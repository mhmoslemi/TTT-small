import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    """Solve for optimal h[p] given current correlation c."""
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    # Unaffected correlations
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max_all = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max_all
    for _ in range(45):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:

            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo = max(x_lo, 0.0)
                xc_hi = min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    # Find optimal x
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0

        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)

                    x_hi = min(x_hi, r1)
                else:

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0
        def _cons_jac(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_init(n, mode, rng):
    """Generate structured initializations."""
    x = np.linspace(0, 2, n)
    if mode == 0:
        # Half-constant
        h = np.where(x < 1.0, 1.0, 0.0)
    elif mode == 1:
        # Periodic: 1 on [0, 0.5), 0 on [0.5, 1.0), 1 on [1.0, 1.5), 0 on [1.5, 2.0)
        h = np.where((x % 1.0) < 0.5, 1.0, 0.0)
    elif mode == 2:
        # Sinusoidal
        h = 0.5 + 0.5 * np.sin(np.pi * x)
    elif mode == 3:
        # Random step function
        h = (rng.rand(n) > 0.5).astype(float)
    elif mode == 4:
        # Triangular
        h = np.where(x < 1.0, 2.0 * x, 2.0 * (2.0 - x))
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def lse_gradient(h, n, alpha):
    """Compute gradient of the LSE-smoothed objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    # Gradient w.r.t. h: for each p, dc_k/dh[p] for weighted sum
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        dc = np.zeros(len(ms))
        if np.any(vf):
            dc[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            dc[vb] -= h[jb[vb]]
        diag = ms == (n - 1)
        dc[diag] = (1.0 - 2.0 * h[p])
        grad[p] = np.sum(weights[ms] * dc) * (2.0 / n)
    return grad * alpha + alpha * (-c.max() * 0)  # simplified

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):

    for alpha in alphas:

        def _cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0
        def _cons_jac(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:

                           bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]

    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))

    neg_mask = a < -1e-15

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    order = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0
        def _cons_jac(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_init(n, mode, rng):
    x = np.linspace(0, 2, n)
    if mode == 0:
        h = np.where(x < 1.0, 1.0, 0.0)
    elif mode == 1:
        h = np.where((x % 1.0) < 0.5, 1.0, 0.0)
    elif mode == 2:
        h = np.clip(0.5 + 0.5 * np.sin(2.0 * np.pi * x), 0, 1)
    elif mode == 3:
        h = (rng.rand(n) > 0.5).astype(float)
    elif mode == 4:
        h = np.where(x < 1.0, 2.0 * x, 2.0 * (2.0 - x))
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)
