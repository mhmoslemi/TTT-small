def upsample(h, n_new):
    n_old = len(h)
    if n_new <= n_old:
        return project_feasible(h[:n_new], n_new)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h)
    n = n_init
    # Phase 1: SLSQP warmup
    h1, c1 = slsqp_run(best_h, n, [1000, 5000], 300, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Exact CD with peak ordering
    remaining = deadline - time.time() - 60
    n_passes = max(20, int(remaining / 5.0))
    h2, c2 = exact_cd(best_h, n, n_passes, deadline - 60, order_type='peak')
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Sequential CD for different trajectory
    if time.time() < deadline - 60:
        h3, c3 = exact_cd(best_h, n, 30, deadline - 60, order_type='sequential')
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-resolution - upsample to 2x
    n2 = min(2 * n, 1400)
    if n2 > n and time.time() < deadline - 120:
        h_up = upsample(best_h, n2)
        h_up, c_up = slsqp_run(h_up, n2, [2000], 200, deadline - 60)
        rem = deadline - time.time() - 30
        np_ = max(10, int(rem / 8.0))
        h_up, c_up = exact_cd(h_up, n2, np_, deadline - 30, order_type='peak')
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n2
    # Phase 5: Try 3x if time permits
    n3 = min(3 * n_init, 2100)
    if n3 > n and time.time() < deadline - 180:
        h_up3 = upsample(best_h, n3)
        h_up3, c_up3 = slsqp_run(h_up3, n3, [3000], 150, deadline - 120)
        rem = deadline - time.time() - 20
        np3 = max(5, int(rem / 10.0))
        h_up3, c_up3 = exact_cd(h_up3, n3, np3, deadline - 20, order_type='peak')
        if c_up3 < best_c:
            best_h, best_c, n = h_up3, c_up3, n3
    # Phase 6: Random restarts on best grid
    n = len(best_h)
    if time.time() < deadline - 60:
        for r in range(8):
            if time.time() > deadline - 30:
                break
            s = 0.02 + 0.02 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = exact_cd(hp, n, 15, deadline - 30, order_type='peak')
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 7: Final polish
    if time.time() < deadline - 10:
        n = len(best_h)
        max_p = max(10, int((deadline - time.time() - 5) / 4.0))
        hf, cf = exact_cd(best_h, n, max_p, deadline, order_type='peak')
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
