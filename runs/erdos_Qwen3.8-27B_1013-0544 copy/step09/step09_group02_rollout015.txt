def subgradient_direction(c_vals, h, n, active_tol=1e-4):
    """Compute the Fleming direction for min max problem."""
    m_max = c_vals.max()
    active = np.where(c_vals >= m_max - active_tol)[0]
    
    # Compute gradients for active lags
    # For lag m, gradient w.r.t. h[p] is:
    # (1 - h[p+m-(n-1)]) - h[p-m+(n-1)] (when valid)
    grads = []
    for m in active:
        g = np.zeros(n)
        for p in range(n):
            jf = p + m - (n-1)
            jb = p - m + (n-1)
            if 0 <= jf < n:
                g[p] += (1.0 - h[jf])
            if 0 <= jb < n:
                g[p] -= h[jb]
        grads.append(g)
    
    grads = np.array(grads)  # shape (len(active), n)
    
    # Find convex combination closest to origin
    # min ||Σ λ_i g_i||² s.t. λ ≥ 0, Σ λ = 1
    # This is a QP: min λ' G G' λ s.t. λ ≥ 0, 1'λ = 1
    # where G = grads.T (n × len(active))
    
    from scipy.optimize import minimize
    k = len(active)
    if k == 1:
        return -grads[0] / (np.linalg.norm(grads[0]) + 1e-15)
    
    # Use SLSQP for the QP
    GtG = grads @ grads.T  # k × k
    def obj(lam):
        return lam @ GtG @ lam
    def obj_grad(lam):
        return 2 * GtG @ lam
    
    bounds = [(0, 1)] * k
    cons = [{'type': 'eq', 'fun': lambda l: l.sum() - 1, 'jac': lambda l: np.ones(k)}]
    lam0 = np.ones(k) / k
    res = minimize(obj, lam0, jac=obj_grad, method='SLSQP', bounds=bounds, constraints=cons)
    
    if res.success:
        direction = -grads.T @ res.x
    else:
        direction = -grads.mean(axis=0)
    
    return direction

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break

            if tot < 1e-15: break
            h += (diff / tot) * h

def exact_coord_delta(h, p, c, n):

    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2*n - 2, max(2*n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)

    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)

    if np.any(vb): vec[vb] -= h[jb[vb]]

    unaff = 0.0

    if mhi < 2*n-2: unaff = max(unaff, float(c[mhi+1:].max()))
    lo, hi = -h0, 1.0 - h0

        mval = float(vals.max())

    m_max = c.max()
    active = np.where(c >= m_max - tol)[0]
    if len(active) <= 1:
        return None
    # Compute gradients
    k = len(active)
    # G[p, i] = gradient of c[active[i]] w.r.t. h[p]
    G = np.zeros((n, k))
    for idx, m in enumerate(active):
        for p in range(n):
            jf = p + m - (n-1)
            jb = p - m + (n-1)
            if 0 <= jf < n: G[p, idx] += (1.0 - h[jf])
            if 0 <= jb < n: G[p, idx] -= h[jb]
    # Find convex combo of columns closest to origin
    # min ||G^T λ||² s.t. λ ≥ 0, 1^T λ = 1
    # = min λ^T G G^T λ s.t. λ ≥ 0, 1^T λ = 1
    GtG = G.T @ G  # k × k
    def obj(lam): return lam @ GtG @ lam
    def obj_grad(lam): return 2.0 * GtG @ lam
    bounds = [(0.0, 1.0)] * k
    cons = [{'type':'eq','fun':lambda l: l.sum()-1.0,'jac':lambda l: np.ones(k)}]
    lam0 = np.ones(k)/k
    res = minimize(obj, lam0, jac=obj_grad, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':200})
    if res.success:
        direction = -G.T @ res.x
    else:
        direction = -G.mean(axis=0)
    return direction

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2*n - 2, max(2*n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf): vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb): vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi+1]
    unaff = 0.0
    if mlo > 0: unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2*n-2: unaff = max(unaff, float(c[mhi+1:].max()))
    lo, hi = -h0, 1.0 - h0

        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())

        elif vmax < -1e-10: lo = mid

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):

                jb = p - ms + (n-1)

def fleming_polish(h, n, max_iter=500, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    for it in range(max_iter):

        c = np.correlate(h, 1.0 - h, mode='full')
        m_max = c.max()
        active = np.where(c >= m_max - 0.01)[0]
        k = len(active)
        if k <= 1:
            # Just use single subgradient
            m_star = int(np.argmax(c))
            g = np.zeros(n)
            for p in range(n):
                jf = p + m_star - (n-1)
                jb = p - m_star + (n-1)
                if 0 <= jf < n: g[p] += 1.0 - h[jf]
                if 0 <= jb < n: g[p] -= h[jb]
        else:
            G = np.zeros((n, k))
            for idx, m in enumerate(active):
                jf = np.arange(n) + m - (n-1)
                jb = np.arange(n) - m + (n-1)
                vf = (jf>=0) & (jf<n)
                vb = (jb>=0) & (jb<n)
                G[vf, idx] += 1.0 - h[jf[vf]]
                G[vb, idx] -= h[jb[vb]]
            GtG = G.T @ G
            def obj(lam): return float(lam @ GtG @ lam)
            def obj_grad(lam): return 2.0 * GtG @ lam
            bounds = [(0.0, 1.0)] * k
            cons = [{'type':'eq','fun':lambda l: l.sum()-1.0,'jac':lambda l: np.ones(k)}]
            lam0 = np.ones(k)/k
            res = minimize(obj, lam0, jac=obj_grad, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':100,'ftol':1e-12})
            g = -(G.T @ res.x) if res.success else -G.mean(axis=0)
        
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-8: break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n): return x.sum() - n/2.0
def sum_con_jac(x, n): return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()

            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except: pass
    return best_h, best_c

                vf = (jf>=0) & (jf<n)
                vb = (jb>=0) & (jb<n)
                delta = np.zeros(len(ms))
                if np.any(vf): delta[vf] += d*(1.0-h[jf[vf]])
                if np.any(vb): delta[vb] -= d*h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p]+d, 0.0, 1.0)
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fleming_grad(h, n, tol):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_max = c.max()
    active = np.where(c >= m_max - tol)[0]
    k = min(len(active), 40)
    if k == 0: return np.zeros(n)
    active = active[:k]
    if k == 1:
        m = int(active[0])
        g = np.zeros(n)
        jf = np.arange(n) + m - (n-1)
        jb = np.arange(n) - m + (n-1)
        vf = (jf>=0) & (jf<n)
        vb = (jb>=0) & (jb<n)
        g[vf] += 1.0 - h[jf[vf]]
        g[vb] -= h[jb[vb]]
        return g
    G = np.zeros((n, k))
    for idx, m in enumerate(active.astype(int)):
        jf = np.arange(n) + m - (n-1)
        jb = np.arange(n) - m + (n-1)
        vf = (jf>=0) & (jf<n)
        vb = (jb>=0) & (jb<n)
        G[vf, idx] += 1.0 - h[jf[vf]]
        G[vb, idx] -= h[jb[vb]]
    GtG = G.T @ G
    def fobj(lam): return float(lam @ GtG @ lam)
    def fgrad(lam): return 2.0 * GtG @ lam
    bounds = [(0.0, 1.0)] * k
    cons = [{'type':'eq','fun':lambda l: l.sum()-1.0,'jac':lambda l: np.ones(k)}]
    lam0 = np.ones(k)/k
    res = minimize(fobj, lam0, jac=fgrad, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':100,'ftol':1e-14})
    if res.success: return -(G.T @ res.x)
    return -G.mean(axis=0)

def fleming_polish(h, n, max_iter=500, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    tol = 2.0
    for it in range(max_iter):
        if deadline and time.time() > deadline: break
        g = fleming_grad(h, n, tol)
        gn = np.linalg.norm(g)
        if gn < 1e-12: break
        h_new = project_feasible(h - lr * g / gn, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new; best_c = c_new
            lr = min(lr * 1.05, 0.15)
        else:
            lr *= 0.7
            if lr < 1e-10: break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n): return x.sum() - n/2.0
def sum_con_jac(x, n): return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline: break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type':'eq','fun':sum_con,'jac':sum_con_jac,'args':(n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons, options={'maxiter':maxiter,'ftol':1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c: best_h, best_c = new_h, new_c
        except: pass
    return best_h, best_c

def fista_smooth(h, n, alpha, max_iter, deadline=None):
    h = h.copy().astype(float)
    y = h.copy()
    t = 1.0
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline and time.time() > deadline: break
        c = np.correlate(y, 1.0 - y, mode='full')
        m = c.max()
        w = np.exp(alpha*(c - m))
        s = w.sum()
        if s < 1e-30: break
        w /= s
        g = np.zeros(n)
        idx = np.where(w > 1e-8)[0]
        for mi in idx:
            wi = w[mi]
            mm = int(mi)
            jf = np.arange(n) + mm - (n-1)
            jb = np.arange(n) - mm + (n-1)
            vf = (jf>=0) & (jf<n)
            vb = (jb>=0) & (jb<n)
            g[vf] += wi*(1.0 - y[jf[vf]])
            g[vb] -= wi*y[jb[vb]]
        lr = 1.0 / (alpha * g.dot(g) + 1e-10)
        lr = min(lr, 1.0)
        x_new = project_feasible(y - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t))/2.0
        y = x_new + ((t-1.0)/t_new)*(x_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = x_new
        t = t_new
        if it % 200 == 0:
            cv = compute_c5(h, n)
            if cv < best_c: best_c = cv
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = np.full(400, 0.5); n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    x_cur = np.linspace(0, 2, n_init)

    # Phase 1: Initial CD
    h1, c1 = coord_cd(best_h, n_init, 100, deadline, rng)
    if c1 < best_c: best_h, best_c = h1, c1

    # Phase 2: Multi-resolution sweep
    sizes = [60, 100, 150, 200, 250, 320, 400, 500, 550, 600, 700, 800, 1000, 1200]
    for n_try in sizes:
        if time.time() > deadline - 100: break
        if n_try == n_init: continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, 60, deadline, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt; x_cur = x_new

    # Phase 3: Fleming + CD alternating
    n_cur = len(best_h)
    for rnd in range(6):
        if time.time() > deadline - 80: break
        hf, cf = fleming_polish(best_h, n_cur, 200, deadline)
        if cf < best_c: best_h, best_c = hf, cf
        hc, cc = coord_cd(best_h, n_cur, 100, deadline, rng)
        if cc < best_c: best_h, best_c = hc, cf

    # Phase 4: FISTA smooth
    n_cur = len(best_h)
    if time.time() < deadline - 60:
        for alpha in [500, 2000]:
            if time.time() > deadline - 30: break
            h_f, c_f = fista_smooth(best_h, n_cur, alpha, 2000, deadline)
            if c_f < best_c: best_h, best_c = h_f, c_f
        hc, cc = coord_cd(best_h, n_cur, 150, deadline, rng)
        if cc < best_c: best_h, best_c = hc, cc

    # Phase 5: SLSQP
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        h5, c5 = slsqp_polish(best_h, n_cur, [5000, 20000, 80000], 400, deadline)
        if c5 < best_c: best_h, best_c = h5, c5

    # Phase 6: Final polish
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        h6, c6 = coord_cd(best_h, n_cur, 300, deadline, rng)
        if c6 < best_c: best_h, best_c = h6, c6
    if time.time() < deadline - 10:
        hf, cf = fleming_polish(best_h, n_cur, 200, deadline)
        if cf < best_c: best_h, best_c = hf, cf
    if time.time() < deadline - 3:
        h7, c7 = coord_cd(best_h, n_cur, 200, deadline, rng)
        if c7 < best_c: best_h, best_c = h7, c7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
