def project_feasible(h, n):

def correlation_grad(h, n, k):
    """Gradient of c[n-1+k] w.r.t. h, for shift k."""
    grad = np.zeros(n)
    if k >= 0:
        # c = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
        # ∂/∂h[p] = (1-h[p+k]) for 0<=p<=n-1-k

            grad[kp:n] += (1.0 - h[0:n-kp])
        if n1 >= 0:
            grad[0:n1+1] -= h[kp:kp+1+n1]

1. Standard CD (as before, for baseline)
2. Pair-coordinate updates (to escape 1D local optima)
3. Active-set gradient steps (for global movement)
4. Multi-resolution (try different n)

Let me be more careful about the pair update. For two coordinates p and q, the correlation c[m] becomes:
c[m] = c[m]_old + (new_p - old_p) * coeff_p[m] + (new_q - old_q) * coeff_q[m] + (new_p - old_p)*(new_q - old_q) * cross_coeff[m]

Wait, it's not quite that simple because c[m] = sum_i h[i]*(1-h[j]) is bilinear. If both p and q appear in the same term (i.e., one is h[p] and the other is h[q] = 1-h[p+k] for some k), then there's a cross term.

Actually, let me think about this differently. For a fixed shift m (with k = m - (n-1)):
c[m] = sum_i h[i] * (1-h[i+k])

If I change h[p] to x and h[q] to y:
- h[p] contributes to c[m] if p is in the sum (as first factor) or if p-k is in the sum (as second factor)
- Similarly for h[q]
- If p = q+k or q = p+k, then h[p] and h[q] appear in the same product term, creating a cross term x*(1-y)

So the pair update is a 2D problem that's piecewise quadratic. We can solve it by:
1. Computing c[m] as a function of (x, y) for all relevant m
2. Finding (x, y) in [0,1]^2 that minimizes max_m c[m]

This is a 2D minimax problem. We can solve it with a grid + refinement approach.

Actually, for efficiency, let me just use a 2D SLSQP or a direct search on a small grid.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def pair_update(h, n, p, q, c):
    h0p, h0q = h[p], h[q]
    if h0p < 1e-12 or h0p > 1.0 - 1e-12 or h0q < 1e-12 or h0q > 1.0 - 1e-12:
        return h0p, h0q
    s_min = max(p - (n-1), q - (n-1))
    s_max = min((n-1) - p, (n-1) - q)
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0p, h0q
    ps, qs = p + s, q + s
    pm, qm = p - s, q - s
    ap = np.zeros(len(s))
    aq = np.zeros(len(s))
    cross = np.zeros(len(s))
    for idx in range(len(s)):
        si = s[idx]
        cp = cq = cx = 0.0
        if 0 <= p + si < n:
            cp += (1.0 - h[p + si])
        if 0 <= p - si < n:
            cp -= h[p - si]
        if 0 <= q + si < n:
            cq += (1.0 - h[q + si])
        if 0 <= q - si < n:
            cq -= h[q - si]
        # Cross term: h[p] appears in (1-h[?]) when q+si==p or q-si==p
        # h[q] appears in (1-h[?]) when p+si==q or p-si==q
        if 0 <= q + si < n and q + si == p:
            cx += -1.0
        if 0 <= q - si < n and q - si == p:
            cx += -1.0
        if 0 <= p + si < n and p + si == q:
            cx += -1.0
        if 0 <= p - si < n and p - si == q:
            cx += -1.0
        # Also h[p] and h[q] in same product: h[i]*(1-h[i+k]) where i=p,i+k=q or i=q,i+k=p
        # This is already captured above
        ap[idx] = cp
        aq[idx] = cq
        cross[idx] = cx
    m = n - 1 + s
    b0 = c[m] - ap * h0p - aq * h0q - cross * h0p * h0q
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base_p = c[n-1] - h0p * (1.0 - h0p)
    c0_base_q = c[n-1] - h0q * (1.0 - h0q)
    
    def obj_2d(xy):
        xp, xq = xy
        vals = b0 + ap * xp + aq * xq + cross * xp * xq
        v = float(vals.max())
        v = max(v, M_unaff)
        # diagonal term for n-1 shift
        c0_new = c[n-1] + (xp - h0p)*(1.0 - h0p) - (xp - h0p)*h0p + (xq - h0q)*(1.0 - h0q) - (xq - h0q)*h0q
        # Actually c[n-1] = sum h[i]*(1-h[i]), changing h[p] and h[q]:
        # delta from p: (xp - h0p)*(1 - h0p) + h0p*(1 - xp) = (xp-h0p) - (xp-h0p)*h0p = (xp-h0p)*(1-h0p)
        # Wait: old contribution = h0p*(1-h0p), new = xp*(1-xp)
        # Actually for the n-1 shift (k=0): c = sum h[i]*(1-h[i])
        # Changing h[p] from h0p to xp: delta = xp*(1-xp) - h0p*(1-h0p)
        # Similarly for h[q]
        c0_new = c[n-1] + xp*(1-xp) - h0p*(1-h0p) + xq*(1-xq) - h0q*(1-h0q)
        v = max(v, c0_new)
        return v
    
    best_xp, best_xq, best_v = h0p, h0q, float('inf')
    # Grid search
    for xp in np.linspace(0, 1, 13):
        for xq in np.linspace(0, 1, 13):
            v = obj_2d((xp, xq))
            if v < best_v:
                best_v, best_xp, best_xq = v, xp, xq
    # Refine
    for _ in range(4):
        step = 1.0 / (13 * (1 << (_ + 1)))
        for dxp in [-step, 0, step]:
            for dxq in [-step, 0, step]:
                xp = np.clip(best_xp + dxp, 0, 1)
                xq = np.clip(best_xq + dxq, 0, 1)
                v = obj_2d((xp, xq))
                if v < best_v:
                    best_v, best_xp, best_xq = v, xp, xq
    return best_xp, best_xq

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-12:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def pair_descent(h, n, max_passes=200, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if len(interior) < 2:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        n_pairs = min(200, len(interior) * (len(interior) - 1) // 2)
        for _ in range(n_pairs):
            p, q = rng.choice(interior, 2, replace=False)
            if p == q:
                continue
            c = np.correlate(h, 1.0 - h, mode='full')
            new_p, new_q = pair_update(h, n, p, q, c)
            if abs(new_p - h[p]) > 1e-10 or abs(new_q - h[q]) > 1e-10:
                h[p], h[q] = new_p, new_q
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
            interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        elif improved == 0:
            break
    return h, best_c

def fourier_smooth(h, n, n_modes):
    """Project h onto its first n_modes Fourier coefficients, then re-clip and project."""
    x = np.linspace(0, 2, n, endpoint=False)
    H = np.fft.rfft(h)
    H[n_modes:] = 0
    h_smooth = np.fft.irfft(H, n=n)
    return project_feasible(h_smooth, n)

def active_grad_step(h, n, step, top_k, temperature):
    """Take one active-set gradient step."""
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    w = np.exp((c[idx] - c[idx].max()) * temperature)
    w /= w.sum()
    grad = np.zeros(n)
    for i, m in enumerate(idx):
        if w[i] < 1e-6:
            continue
        k = m - (n - 1)
        if k >= 0:
            n1 = n - 1 - k
            if n1 >= 0:
                grad[0:n1+1] += w[i] * (1.0 - h[k:n])
            grad[k:n] -= w[i] * h[0:n-k]
        else:
            kp = -k
            n1 = n - 1 - kp
            grad[kp:n] += w[i] * (1.0 - h[0:n-kp])
            grad[0:n1+1] -= w[i] * h[kp:n]
    h_new = h - step * grad
    return project_feasible(h_new, n)
