    n_cur = len(best_h)
    if time.time() < deadline - 40:
        h8, c8 = lbfgs_al(best_h, n_cur, 50000, deadline - 40)
        if c8 < best_c:
            best_h, best_c = h8, c8
    if time.time() < deadline - 10:
        h9, c9 = cd_solve(best_h, best_h.__len__(), 500, deadline - 10, 25)
        h9, c9 = block_cd(h9, len(h9), 50, deadline - 10, 20, 3)
        if c9 < best_c:
            best_h, best_c = h9, c9
    # Final: deep CD + symmetry
    n_f = len(best_h)
    if time.time() < deadline - 2:
        h10, c10 = cd_solve(best_h, n_f, 300, deadline - 2, 30)
        if c10 < best_c:
            best_h, best_c = h10, c10
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
