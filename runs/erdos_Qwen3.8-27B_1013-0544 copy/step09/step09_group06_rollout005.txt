def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    """Solve min_d max(c_peak + d*a, diag term, unaff_max)"""
    T = len(c_peak)
    if T == 0:
        return 0.0, max(unaff_max, 0.0)
    candidates = [lo, hi]

                    candidates.append(d)

            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)

    grad *= (2.0 / n)
    return obj, grad

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    if T == 0:
        return 0.0, unaff_max
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]

                    candidates.append(d)

        c_peak = c[top_idx].astype(float)
        improved = False
        order = np.arange(n)
        if shuffle and rng is not None:
