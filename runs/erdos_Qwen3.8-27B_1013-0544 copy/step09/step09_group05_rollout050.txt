    n_restarts = int(max(4, time_left / 90))
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        h_try = structured_restart(best_h, n, rng, r)
        h_opt, c_opt = coord_descent(h_try, n, 80, deadline, rng, random_order=True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 5: Try nearby n values
    for n_near in [n - 2, n + 2, n + 4, n - 4]:
        if time.time() > deadline - 50:
            break
        if n_near < 50:
            continue
        h_try = resample(best_h, len(best_h), n_near)
        h_opt, c_opt = coord_descent(h_try, n_near, 80, deadline, rng, random_order=False)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        remaining = deadline - time.time() - 3
        max_p = max(20, int(remaining / max(n_f * 0.0008, 0.005)))
        hf, cf = coord_descent(best_h, n_f, max_p, deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
