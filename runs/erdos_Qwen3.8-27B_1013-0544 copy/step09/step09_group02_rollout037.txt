import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def topk_subgrad(h, n, k=5, lr=0.3, max_iter=300, deadline=None):
    h = h.copy()
    best_h = h.copy()
