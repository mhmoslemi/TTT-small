def project_box_sum(x, target):
    x=np.array(x,dtype=float,copy=True)
    lo = float(x.min()) - 1.0
    hi = float(x.max())
    # sum(clip(x-t,0,1)) decreases with t
    for _ in range(50):
        mid=(lo+hi)/2
        y=np.clip(x-mid,0,1)
        s=y.sum()
        if s > target:
            lo=mid
        else:
            hi=mid
    y=np.clip(x-(lo+hi)/2,0,1)
    # adjust residual exactly
    diff=target-y.sum()
    if abs(diff)>1e-15:
        # find interior
        mask=(y>1e-12)&(y<1-1e-12)
        if mask.any():
           y[mask]+=diff/mask.sum()
           y=np.clip(y,0,1)
        else:
           # distribute to all
           y+=diff/x.size
           y=np.clip(y,0,1)
    return y
```

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    if np.all(x >= 0) and np.all(x <= 1):
        s = x.sum()
        if abs(s-target) < 1e-12:
            return x.copy()
        # simple uniform shift if no bounds issue? But shift may violate.
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(40):
       mid=(lo+hi)*0.5
       y=np.clip(x-mid,0.0,1.0)
       if y.sum() > target: lo=mid
       else: hi=mid
    y=np.clip(x-(lo+hi)*0.5,0.0,1.0)
    diff=target-y.sum()
    if diff !=0.0:
       mask=(y>1e-12)&(y<1.0-1e-12)
       if mask.any():
          inc=diff/mask.sum(); y[mask]+=inc; y=np.clip(y,0.0,1.0)
       else:
          # add to all small
          y += diff/x.size
          y=np.clip(y,0.0,1.0)
    # second correction
    diff=target-y.sum()
    if abs(diff)>1e-14:
       mask=(y>1e-12)&(y<1.0-1e-12)
       if mask.any(): y[mask]+=diff/mask.sum()
    return y

import numpy as np
import time

NEG = -1e100

def corr_direct(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(corr_direct(h)) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return x
    s = float(x.sum())
    if abs(s - target) < 1e-13 and float(x.min()) >= 0.0 and float(x.max()) <= 1.0:
        return x.copy()
    if float(x.min()) >= 0.0 and float(x.max()) <= 1.0:
        d = (target - s) / float(n)
        if d + float(x.min()) >= -1e-12 and d + float(x.max()) <= 1.0 + 1e-12:
            y = x + d
            return np.clip(y, 0.0, 1.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x))
    for _ in range(40):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(3):
        diff = target - float(y.sum())
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / float(np.count_nonzero(mask))
            y = np.clip(y, 0.0, 1.0)
        else:
            y += diff / float(n)
            y = np.clip(y, 0.0, 1.0)
    return y

def select_weights(c, temp, kmax=320):
    L = len(c)
    order = np.argsort(c)[::-1]
    K = min(kmax, L)
    idx = order[:K]
    vals = c[idx] - float(c.max())
    if temp <= 1e-12:
        w = np.zeros(K)
        w[0] = 1.0
        return idx, w
    w = np.exp(vals / temp)
    s = float(w.sum())
    if s <= 0.0:
        w[0] = 1.0
    else:
        w /= s
    return idx, w

def active_gradient(h, idx, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for m, w in zip(idx, weights):
        if w <= 1e-15:
            continue
        s = n - 1 - int(m)
        if s == 0:
            g += w * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += w * (1.0 - h[s:])
            g[s:] -= w * h[:n - s]
        else:
            Ls = -s
            g[Ls:] += w * (1.0 - h[:n - Ls])
            g[:n - Ls] -= w * h[Ls:]
    return g

def best_delta(h0, c_sub, v, floor, idx0, iters=24):
    lo = -float(h0)
    hi = 1.0 - float(h0)
    F0 = max(float(c_sub.max()), floor)
    best_d = 0.0
    best_F = F0
    scale = max(1.0, abs(F0))
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * v
        if idx0 >= 0:
            vals[idx0] -= mid * mid
        mval = float(vals.max())
        F = mval if mval > floor else floor
        if F < best_F - 1e-13:
            best_F = F
            best_d = mid
        eps = 1e-10 * scale
        active = vals >= (F - eps)
        if not np.any(active):
            best_d = mid
            break
        min_s = float(v[active].min())
        max_s = float(v[active].max())
        if idx0 >= 0 and vals[idx0] >= F - eps:
            s0 = v[idx0] - 2.0 * mid
            if s0 < min_s:
                min_s = s0
            if s0 > max_s:
                max_s = s0
        if floor >= F - eps:
            if min_s > 0.0:
                min_s = 0.0
            if max_s < 0.0:
                max_s = 0.0
        if min_s > 1e-11:
            hi = mid
        elif max_s < -1e-11:
            lo = mid
        else:
            best_d = mid
            break
    cands = [0.0, best_d, (lo + hi) * 0.5, lo, hi]
    for d in cands:
        vals = c_sub + d * v
        if idx0 >= 0:
            vals[idx0] -= d * d
        mval = float(vals.max())
        F = mval if mval > floor else floor
        if F < best_F - 1e-13:
            best_F = F
            best_d = d
    if best_d < lo:
        best_d = lo
    if best_d > hi:
        best_d = hi
    return best_d

def cd_improve(h, n, deadline, max_passes=100, shuffle=False, rng=None, project_every=2):
    target = n / 2.0
    dx = 2.0 / n
    L = 2 * n - 1
    h = project_box_sum(np.array(h, dtype=float, copy=True), target)
    c = corr_direct(h)
    M = float(c.max())
    best_h = h.copy()
    best_count = M
    best_c = M * dx
    passes = 0
    while passes < max_passes and time.time() < deadline:
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        tick = 0
        for p in order:
            if (tick & 31) == 31 and time.time() > deadline:
                break
            tick += 1

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            if segmax < M - 1e-9:
                continue

            if mlo > 0:

            if mhi < L - 1:
                floor = max(floor, float(c[mhi + 1:].max()))
            if floor >= M - 1e-8 and segmax <= M + 1e-8:

            jf = p + ms - (n - 1)

            vb = (jb >= 0) & (jb < n)
            v = np.zeros(ms.size, dtype=float)

            if np.any(vf):
                v[vf] -= h[jf[vf]]
            idx0 = n - 1 - mlo if mlo <= n - 1 <= mhi else -1

                segmax2 = float(c[ms].max())

                improved = True
        passes += 1
        if not improved:
            break
        if project_every > 0 and passes % project_every == 0:
            hp = project_box_sum(h, target)
            cp = corr_direct(hp)
            cpm = float(cp.max())
            if cpm < best_count:
                best_count = cpm

                c = cp
                M = cpm

                c = corr_direct(h)
                M = float(c.max())
    hf = project_box_sum(h, target)

    cff = float(cf.max())
    if cff < best_count:
        best_h = hf
        best_count = cff

    return best_h, best_c

def upsample_h(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, n)
    return np.interp(x1, x0, h, left=float(h[0]), right=float(h[-1]))

def perturb_h(h, rng, amp):
    n = len(h)
    return project_box_sum(h + amp * rng.standard_normal(n), n / 2.0)

    if time.time() < deadline - 5.0:
        d3 = min(deadline, t0 + 0.88 * budget)
        h3, c3 = cd_improve(best_h, n, deadline=d3, max_passes=cd2, shuffle=True,
                            rng=rng, project_every=proj_every)
        if c3 < best_c:
            best_h, best_c = h3, c3
            best_count = c3 / dx
    if do_sym and time.time() < deadline - 8.0:
        hs = 0.5 * (best_h + best_h[::-1])
        hs = project_box_sum(hs, target)
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_h, best_c = hs, cs
            h4, c4 = cd_improve(hs, n, deadline=deadline,
                                max_passes=int(max(5, int(cd2 * 0.20))),
                                shuffle=True, rng=rng, project_every=proj_every)
            if c4 < best_c:
                best_h, best_c = h4, c4
    if time.time() < deadline - 5.0:
        hf, cf = cd_improve(best_h, n, deadline=deadline,
                            max_passes=int(max(5, int(cd2 * 0.30))),
                            shuffle=False, rng=rng, project_every=proj_every)
        if cf < best_c:
            best_h, best_c = hf, cf
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 18.0)
    hard = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
    except Exception:
        h0 = np.full(512, 0.5)
    best_n = len(h0)
    best_h = project_box_sum(h0, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)

    b1 = min(0.42 * total, hard - time.time())
    h1, c1 = polish_budget(best_h, best_n, b1, rng, do_sym=True, hard_deadline=hard)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, len(h1)

    rem = hard - time.time()
    if rem > 25.0:
        b2 = min(0.09 * total, rem * 0.45)
        hs = 0.5 * (best_h + best_h[::-1])
        hs = project_box_sum(hs, best_n / 2.0)
        h2, c2 = polish_budget(hs, best_n, b2, rng, do_sym=False, hard_deadline=hard)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    n_list = [250, 350, 450, 600, 750, 900]
    n_list = [n for n in n_list if n != best_n]
    reserve = max(70.0, 0.12 * total)
    for idx, nn in enumerate(n_list):
        rem = hard - time.time()
        usable = max(0.0, rem - reserve)
        if usable < 35.0:
            break
        n_left = max(1, len(n_list) - idx)
        b = max(18.0, usable / n_left * 0.88)
        seed_h = upsample_h(best_h, nn)
        amp = 0.018 if nn > 600 else 0.028
        seed_h = perturb_h(seed_h, rng, amp)
        hopt, copt = polish_budget(seed_h, nn, b, rng, do_sym=(nn <= 600),
                                   hard_deadline=hard)
        if copt < best_c:
            best_h, best_c, best_n = hopt, copt, len(hopt)

    rem = hard - time.time()
    if rem > 35.0:
        bf = min(0.55 * rem, rem - 10.0)
        hf, cf = polish_budget(best_h, best_n, bf, rng, do_sym=True, hard_deadline=hard)
        if cf < best_c:
            best_h, best_c, best_n = hf, cf, len(hf)
        rem2 = hard - time.time()
        if rem2 > 35.0:
            nnext = min(1080, int(round(best_n * 1.5 / 10.0) * 10))
            if nnext > best_n:
                sn = upsample_h(best_h, nnext)
                sn = perturb_h(sn, rng, 0.012)
                hn, cn = polish_budget(sn, nnext, rem2 * 0.55, rng,
                                       do_sym=False, hard_deadline=hard)
                if cn < best_c:
                    best_h, best_c, best_n = hn, cn, len(hn)

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
