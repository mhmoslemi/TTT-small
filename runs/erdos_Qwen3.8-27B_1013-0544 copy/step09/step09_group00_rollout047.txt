                x_lo_b, x_hi_b = 0.0, 1.0
                pos = a > 1e-15
                neg = a < -1e-15
                zero = np.abs(a) <= 1e-15
                if np.any(zero) and np.any(c_base[zero] > L):
                    L_lo = L
                    continue
                if np.any(pos):
                    x_hi_b = min(x_hi_b, float(np.min((L - c_base[pos]) / a[pos] + h0)))
                if np.any(neg):
                    x_lo_b = max(x_lo_b, float(np.max((L - c_base[neg]) / a[neg] + h0)))
                if x_lo_b > x_hi_b + 1e-14:
                    L_lo = L
                    continue
                if has_diag:
                    T = L - c_diag_val + h0 * (1.0 - h0)
                    D = 1.0 - 4.0 * T
                    if D >= 0:
                        r1 = (1.0 - np.sqrt(D)) / 2.0
                        r2 = (1.0 + np.sqrt(D)) / 2.0
                        ov1 = (x_hi_b > 0.0) and (x_lo_b < r1)
                        ov2 = (x_hi_b > r2) and (x_lo_b < 1.0)
                        if not (ov1 or ov2):
                            L_lo = L
                            continue
                L_hi = L
            x_lo_b, x_hi_b = 0.0, 1.0
            pos = a > 1e-15
            neg = a < -1e-15
            if np.any(pos):
                x_hi_b = min(x_hi_b, float(np.min((L_hi - c_base[pos]) / a[pos] + h0)))
            if np.any(neg):
                x_lo_b = max(x_lo_b, float(np.max((L_hi - c_base[neg]) / a[neg] + h0)))
            x_lo_b = max(x_lo_b, 0.0)
            x_hi_b = min(x_hi_b, 1.0)
            if has_diag:
                T = L_hi - c_diag_val + h0 * (1.0 - h0)
                D = 1.0 - 4.0 * T
                if D >= 0:
                    r1 = (1.0 - np.sqrt(D)) / 2.0
                    r2 = (1.0 + np.sqrt(D)) / 2.0
                    candidates = [x_lo_b, x_hi_b, (x_lo_b + x_hi_b) / 2.0]
                    if x_lo_b <= r1 <= x_hi_b:
                        candidates.append(r1)
                    if x_lo_b <= r2 <= x_hi_b:
                        candidates.append(r2)
                    best_x, best_L = h0, np.inf
                    for cc in candidates:
                        cc = float(np.clip(cc, 0.0, 1.0))
                        c_new = c_base + (cc - h0) * a
                        c_new = c_new.copy()
                        c_new[diag_pos[0]] = c_diag_val - h0 * (1.0 - h0) + cc * (1.0 - cc)
                        actual = max(unaff_max, float(np.max(c_new)))
                        if actual < best_L:
                            best_L = actual
                            best_x = cc
                    x_star = best_x
                else:
                    x_star = (x_lo_b + x_hi_b) / 2.0
            else:
                x_star = (x_lo_b + x_hi_b) / 2.0
            x_star = float(np.clip(x_star, 0.0, 1.0))
            if abs(x_star - h0) > 1e-14:
                delta = x_star - h0
                c[ms] += delta * a
                if has_diag:
                    c[n - 1] = c_diag_val - h0 * (1.0 - h0) + x_star * (1.0 - x_star)
                h[p] = x_star
                improved = True
        if not improved:
            break
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 1: Initial CD
    c = np.correlate(h_best, 1.0 - h_best, mode='full')
    h_best, c = cd_solve(h_best, n, c, 80, deadline)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 2: Try various n
    n_list = sorted(set([n, 600, 650, 700, 750, 800, 500, 450, 550, 850, 900, 1000]),
                    key=lambda x: abs(x - n))
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_try = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        c_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        h_try, c_try = cd_solve(h_try, n_try, c_try, 60, deadline)
        h_try = project_feasible(h_try, n_try)
        c_val = compute_c5(h_try, n_try)
        if c_val < best_c - 1e-9:
            h_best, best_c, n = h_try, c_val, n_try
    # Phase 3: Multi-perturbation
    n = len(h_best)
    for attempt in range(20):
        if time.time() > deadline - 80:
            break
        scale = 0.015 + 0.015 * (attempt % 6)
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = np.correlate(h_pert, 1.0 - h_pert, mode='full')
        h_pert, c_pert = cd_solve(h_pert, n, c_pert, 100, deadline)
        h_pert = project_feasible(h_pert, n)
        c_val = compute_c5(h_pert, n)
        if c_val < best_c:
            h_best, best_c = h_pert, c_val
    # Phase 4: Bit-flip perturbation
    n = len(h_best)
    for attempt in range(10):
        if time.time() > deadline - 40:
            break
        h_flip = h_best.copy()
        n_flips = max(5, n // 50)
        idx = rng.choice(n, size=n_flips, replace=False)
        h_flip[idx] = 1.0 - h_flip[idx]
        h_flip = project_feasible(h_flip, n)
        c_flip = np.correlate(h_flip, 1.0 - h_flip, mode='full')
        h_flip, c_flip = cd_solve(h_flip, n, c_flip, 80, deadline)
        h_flip = project_feasible(h_flip, n)
        c_val = compute_c5(h_flip, n)
        if c_val < best_c:
            h_best, best_c = h_flip, c_val
    # Phase 5: Final polish
    if time.time() < deadline - 5:
        n = len(h_best)
        c = np.correlate(h_best, 1.0 - h_best, mode='full')
        h_best, c = cd_solve(h_best, n, c, 200, deadline)
        h_best = project_feasible(h_best, n)
        best_c = compute_c5(h_best, n)
    return h_best, best_c, len(h_best)
