def project_feasible(h,n):
   target=n/2
   h=np.clip(h.copy(),0,1)
   for _ in range(100):
      diff=target-h.sum()
      if abs(diff)<1e-13: break
      if diff>0:
         room=1-h
         tot=room.sum()
         if tot<1e-14: break
         step=diff/tot
         if step>1: step=1 # not needed
         h=np.minimum(1, h+step*room)
      else:
         tot=h.sum()
         if tot<1e-14: break
         step=diff/tot
         h=np.maximum(0, h+step*h)
   # fix floating sum by adjusting one interior element
   diff=target-h.sum()
   if abs(diff)>0:
      # find index with room in direction
      if diff>0:
         idx=np.argmax(1-h)
         add=min(diff,1-h[idx])
      else:
         idx=np.argmax(h)
         add=max(diff,-h[idx])
      h[idx]+=add
   return h
```

import time
import numpy as np
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    target = float(n) / 2.0
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    diff = target - h.sum()
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            add = max(diff, -h[idx])
        h[idx] = float(np.clip(h[idx] + add, 0.0, 1.0))
    return h


def build_a(p, h, n):
    L = 2 * n - 1
    base = n - 1
    a = np.zeros(L, dtype=float)
    if p < n - 1:
        a[base + 1:base + n - p] += 1.0 - h[p + 1:]
    if p > 0:
        a[base + 1:base + 1 + p] -= h[p - 1:-1:-1]
        a[base - p:base] += 1.0 - h[:p]
    if p < n - 1:
        a[base - (n - 1 - p):base] -= h[n - 1:p:-1]
    return a


def unique_in_range(xs, lo, hi):
    xs = np.asarray(xs, dtype=float)
    if xs.size == 0:
        return np.array([lo, hi], dtype=float)
    xs = xs[np.isfinite(xs)]
    xs = xs[(xs >= lo - 1e-9) & (xs <= hi + 1e-9)]
    if xs.size == 0:
        return np.array([lo, hi], dtype=float)
    xs = np.clip(xs, lo, hi)
    xs = np.sort(xs)
    out = []
    last = None
    for v in xs:
        v = float(v)
        if last is None or abs(v - last) > 1e-8:
            out.append(v)
            last = v
    if not out:
        return np.array([lo, hi], dtype=float)
    return np.array(out, dtype=float)


def eval_candidates(xs, aa, bb, A, B, D):
    xs = np.asarray(xs, dtype=float)
    q = A + B * xs + D * xs * xs
    if aa.size == 0:
        return q
    if xs.size * aa.size <= 1000000:
        line = np.max(aa[:, None] * xs[None, :] + bb[:, None], axis=0)
    else:
        line = np.empty(xs.size, dtype=float)
        for i in range(xs.size):
            line[i] = float(np.max(aa * xs[i] + bb))
    return np.maximum(line, q)


def solve_1d_small(aa, bb, lo, hi, A, B, D, current):
    xs = [float(lo), float(hi), float(np.clip(current, lo, hi))]
    K = aa.size
    if K > 1:
        for i in range(K - 1):
            da = aa[i] - aa[i + 1:]
            mask = np.abs(da) > 1e-13
            if mask.any():
                roots = (bb[i + 1:][mask] - bb[i]) / da[mask]
                roots = roots[(roots >= lo - 1e-10) & (roots <= hi + 1e-10)]
                if roots.size:
                    xs.extend(roots.tolist())
    if abs(D) > 1e-14:
        for i in range(K):
            c2 = D
            c1 = B - aa[i]
            c0 = A - bb[i]
            disc = c1 * c1 - 4.0 * c2 * c0
            if disc >= -1e-12:
                r = np.sqrt(max(0.0, disc))
                r1 = (-c1 + r) / (2.0 * c2)
                r2 = (-c1 - r) / (2.0 * c2)
                if lo - 1e-10 <= r1 <= hi + 1e-10:
                    xs.append(float(r1))
                if lo - 1e-10 <= r2 <= hi + 1e-10:
                    xs.append(float(r2))
    xs = unique_in_range(xs, lo, hi)
    vals = eval_candidates(xs, aa, bb, A, B, D)
    bv = float(vals.min())
    near = xs[vals <= bv + 1e-10]
    cur = float(np.clip(current, lo, hi))
    return float(near[int(np.argmin(np.abs(near - cur)))])


def affine_info(x, aa, bb, tol=1e-9):
    vals = aa * x + bb
    m = float(np.max(vals))
    mask = vals >= (m - tol)
    if not np.any(mask):
        mask = np.zeros(aa.size, dtype=bool)
        mask[int(np.argmax(vals))] = True
    sl = aa[mask]
    return m, float(np.min(sl)), float(np.max(sl))


def min_affine_bound(aa, bb, lo, hi, iters=35):
    if lo >= hi:
        return float(lo)
    _, mn, mx = affine_info(lo, aa, bb)
    if mn > 1e-10:
        return float(lo)
    _, mn, mx = affine_info(hi, aa, bb)
    if mx < -1e-10:
        return float(hi)
    l, r = lo, hi
    for _ in range(iters):
        m = 0.5 * (l + r)
        _, mn, mx = affine_info(m, aa, bb)
        if mn > 1e-10:
            r = m
        elif mx < -1e-10:
            l = m
        else:
            return float(m)
    return float(0.5 * (l + r))


def g_val(x, aa, bb, A, B, D):
    f = float(np.max(aa * x + bb)) if aa.size else 0.0
    q = A + B * x + D * x * x
    return f - q


def g_info(x, aa, bb, A, B, D):
    f, mn, mx = affine_info(x, aa, bb)
    qd = B + 2.0 * D * x
    q = A + B * x + D * x * x
    return f - q, mn - qd, mx - qd


def min_G_bound(aa, bb, lo, hi, A, B, D, iters=35):
    if lo >= hi:
        return float(lo), g_val(lo, aa, bb, A, B, D)
    g0, gn0, gm0 = g_info(lo, aa, bb, A, B, D)
    if gn0 > 1e-10:
        return float(lo), g0
    g1, gn1, gm1 = g_info(hi, aa, bb, A, B, D)
    if gm1 < -1e-10:
        return float(hi), g1
    l, r = lo, hi
    for _ in range(iters):
        m = 0.5 * (l + r)
        gm, gmn, gmx = g_info(m, aa, bb, A, B, D)
        if gmn > 1e-10:
            r = m
        elif gmx < -1e-10:
            l = m
        else:
            return float(m), gm
    m = 0.5 * (l + r)
    return float(m), g_val(m, aa, bb, A, B, D)


def bisect_G(aa, bb, alo, ahi, A, B, D):
    for _ in range(60):
        am = 0.5 * (alo + ahi)
        gm = g_val(am, aa, bb, A, B, D)
        if gm > 0.0:
            alo = am
        else:
            ahi = am
    return float(0.5 * (alo + ahi))


def solve_1d_large(aa, bb, lo, hi, A, B, D, current):
    if aa.size == 0:
        return solve_1d_small(aa, bb, lo, hi, A, B, D, current)
    xs = [float(lo), float(hi), float(np.clip(current, lo, hi))]
    xf = min_affine_bound(aa, bb, lo, hi)
    xs.append(xf)
    g0 = g_val(lo, aa, bb, A, B, D)
    g1 = g_val(hi, aa, bb, A, B, D)
    xg, gmin = min_G_bound(aa, bb, lo, hi, A, B, D)
    xs.append(xg)
    if gmin < -1e-10:
        if g0 > 1e-10:
            L = bisect_G(aa, bb, lo, xg, A, B, D) if xg > lo + 1e-12 else float(lo)
        else:
            L = float(lo)
        if g1 > 1e-10:
            R = bisect_G(aa, bb, xg, hi, A, B, D) if xg < hi - 1e-12 else float(hi)
        else:
            R = float(hi)
        xs.append(L)
        xs.append(R)
    xs = unique_in_range(xs, lo, hi)
    vals = eval_candidates(xs, aa, bb, A, B, D)
    bv = float(vals.min())
    near = xs[vals <= bv + 1e-10]
    cur = float(np.clip(current, lo, hi))
    return float(near[int(np.argmin(np.abs(near - cur)))])


def solve_1d(aa, bb, lo, hi, A, B, D, current, exact_threshold=70):
    if hi <= lo:
        return float(np.clip(current, lo, hi))
    if aa.size <= exact_threshold:
        return solve_1d_small(aa, bb, lo, hi, A, B, D, current)
    return solve_1d_large(aa, bb, lo, hi, A, B, D, current)


def single_new_x(a, c, y, base, csum, exact_threshold=70):
    maxpos = c.copy()
    pos = a > 0.0
    neg = a < 0.0
    if pos.any():
        maxpos[pos] += a[pos] * (1.0 - y)
    if neg.any():
        maxpos[neg] -= a[neg] * y
    active = maxpos >= (csum - 1.2)
    if base >= 0:
        active[base] = False
    idx = np.where(active)[0]
    A = c[base] - y + y * y
    if idx.size <= exact_threshold:
        aa = a[idx]
        bb = c[idx] - aa * y
        return solve_1d_small(aa, bb, 0.0, 1.0, A, 1.0, -1.0, y)
    aa = a[idx]
    bb = c[idx] - aa * y
    return solve_1d_large(aa, bb, 0.0, 1.0, A, 1.0, -1.0, y)


def coord_single(h, n, deadline, seed, exact_threshold=70):
    h = project_feasible(h, n)
    dx = 2.0 / float(n)
    base = n - 1
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    cur_c = best_c
    rng = np.random.RandomState(seed)
    block_len = max(20, min(90, int(n // 6)))
    stall = 0
    step = 0
    while time.time() < deadline:
        accepted = 0
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline - 1.0:
                break
            y = float(h[p])
            csum = float(np.max(c))
            a = build_a(p, h, n)
            x = single_new_x(a, c, y, base, csum, exact_threshold)
            d = float(x - y)
            if abs(d) > 1e-11:
                old_c0 = float(c[base])
                A = old_c0 - y + y * y
                c += d * a
                c[base] = A + x - x * x
                new_csum = float(np.max(c))
                if new_csum <= csum + 1e-8:
                    h[p] = x
                    accepted += 1
                    step += 1
                    if step % block_len == 0:
                        h = project_feasible(h, n)
                        c = np.correlate(h, 1.0 - h, mode="full").astype(float)
                        cur_c = float(np.max(c)) * dx
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                            stall = 0
                        if time.time() > deadline:
                            break
                else:
                    c -= d * a
                    c[base] = old_c0
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full").astype(float)
        cur_c = float(np.max(c)) * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if accepted == 0 or stall >= 4:
            break
    h = project_feasible(best_h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
    return h, best_c


def interp_h(h, n_new):
    old = np.asarray(h, dtype=float)
    n_old = len(old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(x_new, x_old, old), n_new)


def smooth_fun(x, n, alpha, mu):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    z = alpha * (c - m)
    total = float(np.sum(np.exp(z)))
    lse = np.log(max(total, 1e-300))
    val = dx * (m + lse / alpha)
    if mu:
        val += mu * (x.sum() - n / 2.0) ** 2
    w = np.exp(z) / max(total, 1e-300)
    g = np.zeros(n, dtype=float)
    base = n - 1
    w0 = w[base]
    if w0 > 1e-18:
        g += w0 * (1.0 - 2.0 * x)
    for s in range(1, n):
        wp = w[base + s]
        if wp > 1e-18:
            g[:n - s] += wp * (1.0 - x[s:])
            g[s:] -= wp * x[:-s]
        wm = w[base - s]
        if wm > 1e-18:
            g[s:] += wm * (1.0 - x[:-s])
            g[:n - s] -= wm * x[s:]
    g *= dx
    if mu:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return float(val), g


def smooth_phase(h, n, deadline, seed):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [40.0, 150.0, 500.0, 2000.0]
    mus = [1e-4, 1e-3, 1e-3, 1e-2]
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 30.0:
            break
        mu = mus[i]
        try:
            res = minimize(
                smooth_fun,
                best_h,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxiter": 50, "maxfun": 150, "maxls": 30, "ftol": 1e-12, "gtol": 1e-8},
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass
        end = min(deadline, time.time() + 25.0)
        if end - time.time() > 12.0:
            h3, c3 = coord_single(best_h, n, end, seed + 11 * (i + 1), exact_threshold=60)
            if c3 < best_c:
                best_h, best_c = h3, c3
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 18.0
    if deadline < t0 + 30.0:
        deadline = t0 + max(10.0, float(budget_s) - 3.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
        n0 = int(len(h0))
    except Exception:
        h0 = np.full(600, 0.5, dtype=float)
        n0 = 600
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    main_alloc = min(0.42 * float(budget_s), 460.0)
    main_alloc = max(45.0, main_alloc)
    phase_end = min(deadline, time.time() + main_alloc)
    h2, c2 = coord_single(best_h, best_n, phase_end, seed, exact_threshold=70)
    if c2 < best_c:
        best_h, best_c, best_n = h2, c2, len(h2)

    rem = deadline - time.time()
    if rem > 140.0:
        phase_end = min(deadline, time.time() + min(170.0, rem - 70.0))
        h2, c2 = smooth_phase(best_h, best_n, phase_end, seed + 7)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    for n_try in (900, 700, 520, 450):
        rem = deadline - time.time()
        if rem < 95.0:
            break
        alloc = min(55.0, rem - 70.0)
        if alloc < 30.0:
            break
        phase_end = min(deadline, time.time() + alloc)
        h_try = interp_h(best_h, n_try)
        h2, c2 = coord_single(h_try, n_try, phase_end, seed + 1000 + n_try, exact_threshold=70)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    for r, scale in enumerate((0.01, 0.02, 0.04, 0.08)):
        rem = deadline - time.time()
        if rem < 75.0:
            break
        alloc = min(45.0, rem - 55.0)
        if alloc < 18.0:
            break
        hp = project_feasible(best_h + rng.normal(0.0, scale, best_n), best_n)
        phase_end = min(deadline, time.time() + alloc)
        h2, c2 = coord_single(hp, best_n, phase_end, seed + 500 + 17 * r, exact_threshold=60)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    rem = deadline - time.time()
    if rem > 65.0:
        hb = project_feasible((best_h > 0.5).astype(float), best_n)
        phase_end = min(deadline, time.time() + min(50.0, rem - 45.0))
        h2, c2 = coord_single(hb, best_n, phase_end, seed + 999, exact_threshold=70)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    rem = deadline - time.time()
    if rem > 25.0:
        h2, c2 = coord_single(best_h, best_n, deadline - 4.0, seed + 31415, exact_threshold=80)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
