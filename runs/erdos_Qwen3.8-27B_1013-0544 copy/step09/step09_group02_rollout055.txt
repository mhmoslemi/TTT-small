def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):

2. Compute gradients ∂c_k/∂h_i at h*

4. Update h* = h* + δ
5. Repeat

The gradient of c_k with respect to h_i is:
∂c_k/∂h_i = (1-h[j_forward]) + (-h[j_backward]) (with appropriate domain checks)

This is the same "vec" computed in exact_coord_delta!

So the Frank-Wolfe approach would be:
1. At current h, compute c = correlate(h, 1-h, 'full')
2. Find the active shift k* = argmax c
3. Compute the gradient g_i = ∂c_{k*}/∂h_i (this is the "vec" from the CD)
4. The Frank-Wolfe step is: minimize g · z s.t. z ∈ [0,1]^n, sum(z) = n/2
   The solution is: z_i = 1 if g_i < threshold, z_i = 0 if g_i > threshold, with one fractional coordinate
5. Update: h ← h + α * (z - h)

This is actually a very natural approach! And it's much faster than SLSQP because each iteration only requires:
- One correlation computation: O(n)
- One gradient computation: O(n)
- One FW step: O(n log n) (sorting)

def frank_wolfe(h, n, max_iter=1000, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 2.0 / (max_iter + 2)  # FW step size
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        # Compute gradient of c_k* w.r.t. h
        g = np.zeros(n)
        for i in range(n):
            jf = i + k_star - (n - 1)
            if 0 <= jf < n:
                g[i] += (1.0 - h[jf])
            jb = i - k_star + (n - 1)
            if 0 <= jb < n:
                g[i] -= h[jb]
        # Frank-Wolfe: minimize g · z s.t. z in [0,1], sum(z) = n/2
        # Solution: sort g, set z_i = 1 for smallest g_i, z_i = 0 for largest
        order = np.argsort(g)
        z = np.zeros(n)
        half = n // 2
        z[order[:half]] = 1.0
        # This gives sum(z) = n/2 exactly (n is even)
        # Update
        step = 2.0 / (it + 2)  # standard FW rate
        h_new = h + step * (z - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        # Check convergence
        # FW gap: (h - z) · g
        gap = np.dot(h - z, g)
        if gap < 1e-10:
            break
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]

    if mhi < 2 * n - 2:

    lo, hi = -h0, 1.0 - h0

        elif vmax < -1e-10:

                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def frank_wolfe(h, n, max_iter=500, deadline=None, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Use top-3 active constraints
        top_idx = np.argsort(c)[-3:]
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for i in range(n):
                jf = i + k - (n - 1)
                if 0 <= jf < n:
                    g[i] += (1.0 - h[jf]) / 3.0
                jb = i - k + (n - 1)
                if 0 <= jb < n:
                    g[i] -= h[jb] / 3.0
        # FW step: minimize g·z s.t. z in [0,1], sum(z)=n/2
        order = np.argsort(g)
        z = np.zeros(n)
        z[order[:n//2]] = 1.0
        step = 2.0 / (it + 2)
        h_new = h + step * (z - h)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def diverse_inits(n, rng):

    x = np.linspace(0, 2, n)

        h = np.zeros(n)
