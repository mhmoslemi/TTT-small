    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [2000, 8000], 150)
        h_opt, c_opt = coord_cd(h_opt, n_try, 20, deadline, order='gradient')
        if time.time() < deadline - 30:
            h_opt, c_opt = coord_cd(h_opt, n_try, 15, deadline, order='sequential')
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 6: Perturbation restarts
    for r in range(10):
        if time.time() > deadline - 30:
            break
        s = 0.003 + 0.008 * r
        n = len(best_h)
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        h_o, c_o = coord_cd(hp, n, 12, deadline, order='sequential')
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        max_passes = max(5, int((deadline - time.time() - 3) / 2.5))
        hf, cf = coord_cd(best_h, n, max_passes, deadline, order='gradient')
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
