    if diag_local is not None:

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv = nv.copy()
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(c_aff)
    if k == 0:
        return 0.0
    candidates = [0.0, lo, hi]
    for i in range(k):
        for j in range(i + 1, k):
            di = -1.0 if (diag_local is not None and diag_local == i) else 0.0
            dj = -1.0 if (diag_local is not None and diag_local == j) else 0.0
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = di - dj
            if abs(dd) < 1e-14:
                if abs(da) > 1e-14:
                    d = dc / da
                    if lo <= d <= hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= -1e-14:
                    disc = max(disc, 0.0)
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo <= d <= hi:
                            candidates.append(d)
    for i in range(k):
        di = -1.0 if (diag_local is not None and diag_local == i) else 0.0
        ai = a_coeff[i]
        cm = c_aff[i] - unaff_max
        if abs(di) < 1e-14:
            if abs(ai) > 1e-14:
                d = -cm / ai
                if lo <= d <= hi:
                    candidates.append(d)
        else:
            disc = ai * ai - 4.0 * di * cm
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-ai + sq) / (2.0 * di), (-ai - sq) / (2.0 * di)]:
                    if lo <= d <= hi:
                        candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:

        v = obj_1d(d, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-16:
            best_val = v
            best_d = d
    span = hi - lo
    if span > 1e-12:
        margin = min(span * 0.25, max(abs(best_d) * 0.6, span * 0.1) + 1e-6)

def coord_cd(h, n, max_passes=500, deadline=None, top_k=40, random_order=False, rng=None):

    n_corr = 2 * n - 1
    center = n - 1

            a_coeff = np.zeros(len(ms))
