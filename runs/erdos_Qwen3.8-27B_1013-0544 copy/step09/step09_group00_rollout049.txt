    h = cp.Variable(n)

    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]

    # which means coeff @ h - t <= -const, or (coeff, -1) @ [h, t] <= -const
    
    # For offset k = m - (n-1):
    # coeff[i] = (1 - h0[i+k]) if i+k in [0,n) else 0
    #           + (-h0[i-k]) if i-k in [0,n) else 0

    cols = []
    vals = []

        # h[j+k] gets coefficient (-h0[j])

                    h = h_try

        if abs(prev_c - best_c) < tol:

        prev_c = best_c

    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n

        def _cons_fun(x):
            return x.sum() - n / 2.0

        def _cons_jac(x):
            return np.ones(n)

        def _obj(x):
            cv = np.correlate(x, 1.0 - x, mode='full')
            m = cv.max()
            shifted = alpha * (cv - m)
            mx2 = shifted.max()
            lse = mx2 + np.log(np.sum(np.exp(shifted - mx2)))
            return (m + lse / alpha) * dx

        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(350, 0.5)
        n = 350
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: LP linearization at reduced resolution
    n_lp = min(n, 400)
    x_old = np.linspace(0, 2, len(h_best))
    x_new = np.linspace(0, 2, n_lp)
    h_lp = project_feasible(np.interp(x_new, x_old, h_best), n_lp)
    for it in range(10):
        if time.time() > deadline - 250:
            break
        h_lp, c_lp = lp_iterate(h_lp, n_lp, 3, deadline - 250)
        # Upsample and refine
        x_up = np.linspace(0, 2, n)
        h_up = project_feasible(np.interp(x_up, x_new, h_lp), n)
        h_up, c_up = exact_cd(h_up, n, 80, deadline - 250)
        if c_up < best_c:
            h_best, best_c = h_up, c_up
            n = len(h_best)
            x_old = np.linspace(0, 2, n)

    # Phase 2: Coordinate descent on best
    h_cd, c_cd = exact_cd(h_best, n, 200, deadline - 150)
    if c_cd < best_c:
        h_best, best_c = h_cd, c_cd

    # Phase 3: LP at native resolution
    if time.time() < deadline - 150 and n <= 800:
        h_nat, c_nat = lp_iterate(h_best, n, 4, deadline - 150)
        h_nat, c_nat = exact_cd(h_nat, n, 150, deadline - 100)
        if c_nat < best_c:
            h_best, best_c = h_nat, c_nat

    # Phase 4: SLSQP
    if time.time() < deadline - 100:
        h_s, c_s = slsqp_opt(h_best, n, [5000, 50000], 200, deadline - 50)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 5: Multi-scale
    for n_try in [350, 500, 700, 900]:
        if time.time() > deadline - 80:
            break
        if n_try == n:
            continue
        x1 = np.linspace(0, 2, len(h_best))
        x2 = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(x2, x1, h_best), n_try)
        ho, co = lp_iterate(hi, n_try, 4, deadline - 80)
        ho, co = exact_cd(ho, n_try, 100, deadline - 60)
        if co < best_c:
            h_best, best_c = ho, co
            n = n_try

    # Phase 6: Shake
    n = len(h_best)
    for r in range(5):
        if time.time() > deadline - 20:
            break
        s = 0.02 + 0.015 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = exact_cd(hp, n, 80, deadline - 20)
        if co < best_c:
            h_best, best_c = ho, co

    # Final
    if time.time() < deadline - 3:
        n = len(h_best)
        hf, cf = exact_cd(h_best, n, 150, deadline - 2)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
