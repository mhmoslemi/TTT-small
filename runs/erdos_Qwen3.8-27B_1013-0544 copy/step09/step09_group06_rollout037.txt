    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals = vals.copy()
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    cands = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    cands.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d2 in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d2 <= hi:
                        cands.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                cands.append(d)
    best_d, best_val = lo, float('inf')
    for d in cands:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals = vals.copy()
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val, best_d = v, d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            bd, bv = eval_1d_minimax(c_peak, a_coeff, diag_idx, unaff, lo, hi)
            if bv < cur_max - 1e-15:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                improved = True
                cur_max = bv
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj_grad(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    ex = np.exp(s - sm)
    lse = sm + np.log(np.sum(ex))
    obj = (m + lse / alpha) * (2.0 / n)
    sh = x.sum() - n / 2.0
    obj += pen * sh * sh
    w = ex / np.sum(ex)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh2 = k - (n - 1)
        if sh2 >= 0:
            pe = n - sh2
            grad[:pe] += wk * (1.0 - x[sh2:])
            if sh2 > 0:
                grad[sh2:] += wk * (-x[:pe])
            else:
                grad += wk * (-x)
        else:
            s2 = -sh2
            pe = n - s2
            grad[:pe] += wk * (1.0 - x[s2:])
            grad[s2:] += wk * (-x[:pe])
    grad *= dx
    grad += 2.0 * pen * sh
    return obj, grad

        o, _ = smooth_obj_grad(x, n, alpha, pen)
        return o

        return g

        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        nh = project_feasible(res.x, n)
        return nh, compute_c5(nh, n)
    except Exception:
        return h, compute_c5(h, n)

    cf = compute_c5(hf, n)
    co = compute_c5(h, n)
    return (hf, cf) if cf < co else (h, co)
