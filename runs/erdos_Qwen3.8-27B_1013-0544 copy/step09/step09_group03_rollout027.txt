    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

    c = compute_corr(x)
    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0 / n)

from scipy.optimize import minimize

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:

            h = np.maximum(0.0, h + (diff / tot) * h)

    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0 / n)

def lse_obj_grad(x, n, alpha):
    c = compute_corr(x)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    e = np.exp(shifted - mx)
    w = e / e.sum()
    dx = 2.0 / n
    n2 = len(c)
    n1 = n - 1
    thresh = 1e-10
    active = w > thresh
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for idx in active_idx:
        k = idx - n1
        wk = w[idx]
        if k >= 0:
            if n - k > 0:
                grad[k:] += wk * (1.0 - x[:n-k])
                grad[:n-k] -= wk * x[k:]
        else:
            kp = -k
            if n - kp > 0:
                grad[:n-kp] += wk * (1.0 - x[kp:])
                grad[kp:] -= wk * x[:n-kp]
    return grad * dx

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha),
                          jac=lambda x, n2, a2: lse_obj_grad(x, n2, a2),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(np.clip(res.x, 0, 1), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_nesterov(h, n, steps, deadline, seed=0):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n1 = n - 1
    step0 = 0.05
    mu = 0.9
    for it in range(steps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        cmax = c.max()
        tol = 0.02 / n
        active = c >= (cmax - tol)
        if not active.any():
            active = c >= cmax - 1e-12
        n_act = max(int(active.sum()), 1)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for idx in active_idx:
            k = idx - n1
            if k >= 0:
                if n - k > 0:
                    grad[k:] += (1.0 - h[:n-k])
                    grad[:n-k] -= h[k:]
            else:
                kp = -k
                if n - kp > 0:
                    grad[:n-kp] += (1.0 - h[kp:])
                    grad[kp:] -= h[:n-kp]
        grad *= (dx / n_act)
        step = step0 / (1.0 + 0.05 * it)
        v = mu * v - step * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m_arr = n - 1 + s
    b = c[m_arr] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active_set = vals >= (am - 1e-13)
        if not active_set.any():
            break
        min_sl = float(a[active_set].min())
        max_sl = float(a[active_set].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_corr(h)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with gradient (medium alpha)
    if time.time() < deadline - 450:
        h1, c1 = slsqp_opt(best_h, n, [500, 2000], maxiter_each=500)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Subgradient with Nesterov momentum
    if time.time() < deadline - 350:
        h2, c2 = subgrad_nesterov(best_h, n, 8000, deadline - 340, seed)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: cd after subgrad
    if time.time() < deadline - 250:
        h3, c3 = coord_descent(best_h, n, 2000, deadline - 240, seed + 1)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: SLSQP with high alpha (sharper max)
    if time.time() < deadline - 180:
        h4, c4 = slsqp_opt(best_h, n, [10000, 50000], maxiter_each=600)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_descent(best_h, n, 1000, deadline - 160, seed + 2)
        if c4b < best_c:
            best_h, best_c = h4b, c4b

    # Phase 5: Try different n values
    x_orig = np.linspace(0, 2.0, len(best_h))
    n_trials = [400, 450, 500, 550, 600, 650, 700, 750, 800]
    for n_try in n_trials:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2.0, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 600, deadline - 90, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Fourier perturbation restarts
    n_cur = len(best_h)
    x_grid = np.linspace(0, 2.0, n_cur)
    for r in range(10):
        if time.time() > deadline - 60:
            break
        freq = 1 + r
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = 0.015 * (1.0 + 0.5 * r)
        perturb = amp * np.sin(2.0 * np.pi * freq * x_grid / 2.0 + phase)
        hp = project_feasible(best_h + perturb, n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 400, deadline - 50, seed + 100 + r * 7)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 7: Subgradient restarts with random perturbations
    for r in range(6):
        if time.time() > deadline - 40:
            break
        s = 0.005 + 0.008 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = subgrad_nesterov(hp, n_cur, 4000, deadline - 35, seed + 500 + r * 13)
        h_o2, c_o2 = coord_descent(h_o, n_cur, 500, deadline - 30, seed + 700 + r * 11)
        if c_o2 < best_c:
            best_h, best_c = h_o2, c_o2

    # Phase 8: High-alpha SLSQP final
    if time.time() < deadline - 25:
        h8, c8 = slsqp_opt(best_h, len(best_h), [50000, 100000], maxiter_each=800)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Phase 9: Final CD polish
    if time.time() < deadline - 8:
        n_cur = len(best_h)
        hf, cf = coord_descent(best_h, n_cur, 5000, deadline - 5, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Flip check
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
