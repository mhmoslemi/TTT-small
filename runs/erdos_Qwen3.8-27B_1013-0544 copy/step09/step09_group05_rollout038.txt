from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_max(c, alpha):
    """Smooth approximation of max using log-sum-exp."""
    m = c.max()
    if m < -1e300:
        return m
    shifted = alpha * (c - m)
    mx = shifted.max()
    return (m + mx / alpha) + np.log(np.sum(np.exp(shifted - mx))) / alpha

    x = h.copy()
    y = h.copy()
    t_k = 1.0
    best_x = x.copy()
    best_val = compute_c5(x, n)
    
    c = np.correlate(y, 1.0 - y, mode='full')
    # Compute Lipschitz constant (rough)
    # grad of smooth objective w.r.t. h[i]
    # For simplicity, use a fixed step size
    L = 4.0 * n * alpha  # rough upper bound
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Gradient step
        c_y = np.correlate(y, 1.0 - y, mode='full')
        m = c_y.max()
        shifted = alpha * (c_y - m)
        mx = shifted.max()
        weights = alpha * np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
        
        # gradient of sum_j w_j * c_j where w_j depends on h
        # For simplicity, use the current weights (ignoring the derivative of weights)

            w = weights[j]
            if w < 1e-15:

            p = j - (n-1)
            for i in range(n):
                jf = i + p
                jb = i - p
                if 0 <= jf < n:
                    grad[i] += w * (1.0 - y[jf])
                if 0 <= jb < n:
                    grad[i] -= w * y[jb]
        
        step = 1.0 / L

            if val < best_val:
                best_val = val
                best_x = x.copy()
    
    return best_x, best_val

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-12:

            room = 1.0 - h
            tot = room.sum()

            tot = h.sum()

            h = np.clip(h + diff / tot * h, 0.0, 1.0)

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    T_warm = None
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        T_warm = float(np.max(c))
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            c = np.correlate(h, 1.0 - h, mode='full')
            x_new = solve_coord(c, h, p, n, T_init=T_warm)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    one_mh = 1.0 - h
    for idx in range(2 * n - 1):
        if w[idx] < 1e-16:
            continue
        p = idx - (n - 1)
        wm = w[idx]
        if p >= 0:
            grad[:n-p] += wm * one_mh[p:]
            grad[p:] -= wm * h[:n-p]
        else:
            q = -p
            grad[q:] += wm * one_mh[:n-q]
            grad[:n-q] -= wm * h[q:]
    return grad
