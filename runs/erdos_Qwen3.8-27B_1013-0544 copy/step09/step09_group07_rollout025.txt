def multiplicative_weights_polish(h, n, max_iter=1000, alpha=50, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    p = np.ones(2*n-1) / (2*n-1)  # uniform over all lags
    lr = 1e-3
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        
        # Update weights
        p = np.exp(alpha * c / c.max())
        p /= p.sum()

        grad = np.zeros(n)
        for m in range(2*n-1):
            if p[m] < 1e-6:
                continue
            k = m - (n-1)
            for j in range(n):
                jf = j + k
                if 0 <= jf < n:
                    grad[j] += p[m] * (1.0 - h[jf])
                jb = j - k
                if 0 <= jb < n:
                    grad[j] -= p[m] * h[jb]

        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            h = h_new
            best_c = c_new
            lr = min(lr * 1.02, 5e-3)
        else:
            lr *= 0.7
    
    return h, best_c

def multiplicative_weights_opt(h, n, max_iter=500, alpha_steps=None, deadline=None):
    """Optimize using multiplicative weights on lags + gradient steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    if alpha_steps is None:
        alpha_steps = [10, 50, 100, 200, 500, 1000]
    
    num_lags = 2*n - 1
    p = np.ones(num_lags) / num_lags
    lr = 0.01
    
    for alpha in alpha_steps:
        if deadline is not None and time.time() > deadline:
            break
        for it in range(max_iter):
            if deadline is not None and time.time() > deadline:
                break
            
            c = np.correlate(h, 1.0 - h, mode='full')
            c_max = c.max()
            
            # Update weights (softmax)
            p = np.exp(alpha * (c - c_max))
            p /= p.sum()
            
            # Compute weighted gradient
            grad = np.zeros(n)
            for m in range(num_lags):
                if p[m] < 1e-8:
                    continue
                k = m - (n - 1)
                # ∂R(m)/∂h[j] = (1-h[j-k])·[0≤j-k<n] - h[j+k]·[0≤j+k<n]
                valid_f = np.arange(max(0, -k), min(n, n-k))  # j values where j-k is valid
                # (1-h[j-k]) for j in valid_f
                jk = valid_f - k
                grad[valid_f] += p[m] * (1.0 - h[jk])
                
                valid_b = np.arange(max(0, k), min(n, n+k))  # j values where j+k is valid
                jk2 = valid_b + k
                grad[valid_b] -= p[m] * h[jk2]
            
            # Gradient step with projection
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            
            if c_new < best_c:
                h = h_new
                best_c = c_new
                lr = min(lr * 1.01, 0.05)
            else:
                lr *= 0.7
                if lr < 1e-6:
                    break
    
    return h, best_c
