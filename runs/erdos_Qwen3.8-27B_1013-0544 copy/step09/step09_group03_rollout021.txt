def project_feasible(h, n):

            break
        if diff > 0:

def exact_coord_min(c, p, h, n):

    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)

    if len(s) == 0:

    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0, relax=1.0, patience=8):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if relax < 1.0:
                new_val = h0 + relax * (new_val - h0)
            if abs(new_val - h0) > 1e-14:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            stall = 0
        else:
            stall += 1
        if stall > patience:
            break
    return h, best_c

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    m = c_raw.max()
    shifted = alpha * (c_raw - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    m = c_raw.max()
    shifted = alpha * (c_raw - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        js = idx + s
        mask_p = (js >= 0) & (js < n)
        if mask_p.any():
            grad[mask_p] += w[k] * (1.0 - h[js[mask_p]])
        js2 = idx - s
        mask_m = (js2 >= 0) & (js2 < n)
        if mask_m.any():
            grad[mask_m] -= w[k] * h[js2[mask_m]]
    grad *= dx
    grad = grad - grad.mean()
    return grad

def lbfgs_smoothed(h, n, alpha, maxiter=200, deadline=None):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, n, alpha)
    def grad(x):
        return smooth_grad(x, n, alpha)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 30,
                                'gtol': 1e-14})
        new_h = project_feasible(res.x.copy(), n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Aggressive CD with full precision
    h1, c1 = coord_descent(best_h, n, 5000, deadline - 500, seed, patience=5)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: L-BFGS-B with adaptive smoothing (multiple temperatures)
    for alpha_set in [[300, 1000, 5000, 20000], [500, 2000, 10000, 50000],
                       [1000, 5000, 20000, 100000]]:
        if time.time() > deadline - 300:
            break
        for alpha in alpha_set:
            if time.time() > deadline - 250:
                break
            h2, c2 = lbfgs_smoothed(best_h, n, alpha, 150, deadline - 240)
            if c2 < best_c:
                best_h, best_c = h2, c2
            # Quick CD polish after each smoothing step
            h3, c3 = coord_descent(best_h, n, 300, deadline - 220, seed + 10 + alpha, patience=4)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 3: Relaxed CD to escape potential plateaus
    for relax_val in [0.95, 0.9, 0.8]:
        if time.time() > deadline - 200:
            break
        hr, cr = coord_descent(best_h, n, 800, deadline - 180, seed + 100,
                               relax=relax_val, patience=3)
        if cr < best_c:
            best_h, best_c = hr, cr
        # Follow up with full CD
        h4, c4 = coord_descent(best_h, n, 500, deadline - 160, seed + 200, patience=5)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 4: Multi-resolution exploration
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 800, 1000, 1200]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h, kind='cubic'), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 1500, deadline - 100,
                                      seed + n_try, patience=5)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 5: Binary and structured restarts
    n_cur = len(best_h)
    for r in range(12):
        if time.time() > deadline - 60:
            break
        if r < 4:
            hp = binary_init(n_cur, seed + 500 + r)
        elif r < 8:
            s = 0.003 + 0.008 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        else:
            freq = 1.0 + 0.7 * r
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n_cur)
            hp = project_feasible(0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase), n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 400, deadline - 50,
                                  seed + 600 + r * 13, patience=4)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 6: Final intensive CD
    n_cur = len(best_h)
    hf, cf = coord_descent(best_h, n_cur, 8000, deadline - 5, seed + 999, patience=6)
    if cf < best_c:
        best_h, best_c = hf, cf

    # Flip check
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
