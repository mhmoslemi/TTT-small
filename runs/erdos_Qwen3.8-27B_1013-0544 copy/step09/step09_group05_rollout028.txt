```python
def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(52):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-16:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    xl = 0.0
                    xh = 1.0
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-15:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)
            xh2 = min(x_hi, 1.0)
            if xl2 < r1:
                x_hi = min(x_hi, r1)

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def make_sum_con(n):
    def fun(x):
        return x.sum() - n / 2.0
    def jac(x):
        return np.ones(n)
    return fun, jac

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    fun, jac = make_sum_con(n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': fun, 'jac': jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_sa(h, n, max_iters, deadline, rng):
    """Pair-swap simulated annealing on binary patterns."""
    # Make binary
    h = h.copy()
    # Threshold
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Fix to exactly n/2 ones
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        # Remove excess ones
        idx = np.where(h_bin == 1)[0]
        # Remove ones with lowest "importance" (those most surrounded by ones)
        rng.shuffle(idx)
        for i in idx:
            if n_ones == n_target:
                break
            h_bin[i] = 0
            n_ones -= 1
    elif n_ones < n_target:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        for i in idx:
            if n_ones == n_target:
                break
            h_bin[i] = 1
            n_ones += 1
    
    current_c = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')))
    best_c = current_c
    best_h = h_bin.copy()
    
    T = 0.5
    cool_rate = 0.999
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it > 0 and it % 500 == 0:
            T *= cool_rate
        
        # Pick random i with h=0, j with h=1
        i = rng.randint(n)
        j = rng.randint(n)
        attempts = 0
        while attempts < 20 and h_bin[i] == h_bin[j]:
            i = rng.randint(n)
            j = rng.randint(n)
            attempts += 1
        if h_bin[i] == h_bin[j]:
            continue
        
        # Swap
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        
        # Compute new max
        new_c_vec = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        new_c = float(new_c_vec.max())
        
        delta = new_c - current_c
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_c * (2.0 / n)
```

= sum_n f[n] * g[n + k - 2n + 2]

Hmm, this is getting confusing. Let me just use a direct but vectorized approach:

For each p, grad[p] = sum_m w[m] * [(1-h[p+m])*I(0≤p+m<n) - h[p-m]*I(0≤p-m<n)]

I can compute this as:
term1[p] = sum_m w[m] * (1-h[p+m]) for valid p+m
term2[p] = sum_m w[m] * h[p-m] for valid p-m

term1 is the convolution of w with (1-h), evaluated at the right positions.
term2 is the convolution of w with h.

Specifically:
term1[p] = (w ⊛ (1-h))[p] where ⊛ denotes cross-correlation
= sum_m w[m] * (1-h[p+m])

In numpy: if w has indices from -(n-1) to n-1, and (1-h) has indices 0 to n-1:
np.correlate(1-h, w_padded, 'full') where w_padded is w placed at the right position.

Actually, let me just use FFT-based convolution for speed:

```python
def compute_smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c * (2.0 / n)
    shifted = alpha * (c_scaled - c_scaled.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # w[m] for m in [-(n-1), n-1], where m = k - (n-1)
    # So w is just weights shifted: w[m + (n-1)] = weights[m + (n-1)]
    # Actually weights IS indexed by k from 0 to 2n-2, and m = k-(n-1)
    # So w[m] = weights[m + (n-1)] for m in [-(n-1), n-1]
    
    # grad[p] = (2/n) * sum_m weights[m+(n-1)] * [(1-h[p+m])*I(0≤p+m<n) - h[p-m]*I(0≤p-m<n)]
    
    # First term: sum_m weights[m+(n-1)] * (1-h[p+m]) for 0≤p+m<n
    # = sum_m weights[m+(n-1)] * (1-h[p+m]) for m in [max(-(n-1), -p), min(n-1, n-1-p)]
    # This is like a correlation of weights with (1-h)
    
    # Let's use: np.correlate(1-h, weights, 'full') 
    # = sum_i (1-h)[i] * weights[i + k - (2n-2)]
    # For k = n-1+p: sum_i (1-h)[i] * weights[i + n-1+p - 2n+2] = sum_i (1-h)[i] * weights[i + p - n + 3]
    # That's not quite right either.
    
    # Let me just do it directly with numpy fancy indexing (vectorized over p):
    # For each p in 0..n-1, the valid m values are:
    #   For (1-h[p+m]): m in [0, n-1-p] (since p+m < n means m < n-p, and m >= 0 from the range of valid shifts... wait m can be negative too)
    
    # OK this is getting too complicated. Let me just use the O(n^2) loop but vectorize the inner part.
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * (2.0 / n)
        if w < 1e-15:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            # (1-h[p+m]) for p in [0, n-1-m]
            grad[:n-m] += w * (1.0 - h[m:])
            # -h[p-m] for p in [m, n-1]
            grad[m:] += w * (-h[:n-m])
        else:
            mm = -m
            # (1-h[p+m]) = (1-h[p-mm]) for p in [mm, n-1]
            grad[mm:] += w * (1.0 - h[:n-mm])
            # -h[p-m] = -h[p+mm] for p in [0, n-1-mm]
            grad[:n-mm] += w * (-h[mm:])
    
    return grad
```

This is O(n²) but vectorized (each k iteration is O(n) numpy operations). For n=700, that's 1399 iterations of O(700) numpy ops ≈ 1399 * 700 * ~5ns ≈ 5ms per gradient computation. That's fast enough!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(52):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-16:
            feasible = False
        elif feasible and len(a_nd) > 0:
            xl = 0.0
            xh = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if xl > xh + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    xl = 0.0
                    xh = 1.0
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-15:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = c * (2.0 / n)
    sh = alpha * (cs - cs.max())
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * (2.0 / n)
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] += wk * (-h[:n - m])
        else:
            mm = -m
            grad[mm:] += wk * (1.0 - h[:n - mm])
            grad[:n - mm] += wk * (-h[mm:])
    return grad

def smooth_grad_opt(h, n, alpha, max_steps, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 1.0 / (alpha * 2.0 / n)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h, n, alpha)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c

def binary_sa(h, n, max_iters, deadline, rng):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        for i in idx:
            if n_ones == n_target:
                break
            h_bin[i] = 0
            n_ones -= 1
    elif n_ones < n_target:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        for i in idx:
            if n_ones == n_target:
                break
            h_bin[i] = 1
            n_ones += 1
    current_c = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')))
    best_c = current_c
    best_h = h_bin.copy()
    T = 1.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it > 0 and it % 2000 == 0:
            T *= 0.85
        i = rng.randint(n)
        j = rng.randint(n)
        for _ in range(30):
            if h_bin[i] != h_bin[j]:
                break
            i = rng.randint(n)
            j = rng.randint(n)
        if h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_c = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')))
        delta = new_c - current_c
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c * (2.0 / n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            m = c.max()
            def obj(x):
                cc = np.correlate(x, 1.0 - x, mode='full')
                mm = cc.max()
                sh2 = alpha * (cc - mm)
                mx = sh2.max()
                lse = mx + np.log(np.sum(np.exp(sh2 - mx)))
                return (mm + lse / alpha) * (2.0 / n)
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        h_init = np.full(n, 0.5)
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: CD
    h, c = coord_descent(h, n, 80, deadline, rng, True)
    if c < best_c:
        best_h, best_c = h.copy(), c
    # Phase 2: Smooth gradient
    if time.time() < deadline - 80:
        for alpha in [200, 100, 50]:
            if time.time() > deadline - 60:
                break
            hg, cg = smooth_grad_opt(best_h, n, alpha, 3000, min(deadline - 30, t0 + budget_s * 0.35))
            if cg < best_c:
                best_h, best_c = hg, cg
    # Phase 3: Binary SA
    if time.time() < deadline - 100:
        h_sa, c_sa = binary_sa(best_h, n, 400000, min(deadline - 80, t0 + budget_s * 0.5), rng)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
    # Phase 4: CD from best
    if time.time() < deadline - 60:
        h, c = coord_descent(best_h, n, 80, deadline, rng, True)
        if c < best_c:
            best_h, best_c = h.copy(), c
    # Phase 5: Higher resolution
    for n_try in [1000, 840]:
        if time.time() > deadline - 120:
            break
        if n_try == n:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 50, deadline, rng, True)
        if time.time() < deadline - 30:
            h_opt, c_opt = smooth_grad_opt(h_opt, n_try, 100, 2000, deadline - 10)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 6: Final polish
    n_final = len(best_h)
    if time.time() < deadline - 20:
        h2, c2 = slsqp_run(best_h, n_final, [2000, 10000], 200, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 5:
        h2, c2 = coord_descent(best_h, n_final, 100, deadline, rng, False)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
