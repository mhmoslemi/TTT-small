            n2k = n2 + k  # = i + k
            if 0 <= n2k < n:
                effect[k_idx] -= delta * h[n2k]
    
    return effect

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    idx = np.argsort(-h, kind='stable')
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    # Exact minimization of max(c_aff + d*a_coeff - d^2*diag, unaff_max)
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [max(lo, min(hi, 0.0))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local is not None:
                if diag_local == i:
                    dd -= 1.0
                if diag_local == j:
                    dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    # Also add lo and hi
    candidates.append(lo)
    candidates.append(hi)
    
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    return best_d
