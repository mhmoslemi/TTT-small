def subgradient_step(h, n, lr, n_active=5):
    """Subgradient step considering top n_active lags."""
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:n_active]
    # Average subgradient over active lags
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)  # shift
        # g[p] += (1-h[p+s])*I(0<=p+s<n) - h[p-s]*I(0<=p-s<n)
        ps = np.arange(n) + s
        pm = np.arange(n) - s
        vps = (ps >= 0) & (ps < n)
        vpm = (pm >= 0) & (pm < n)
        g[vps] += (1.0 - h[ps[vps]])
        g[vpm] -= h[pm[vpm]]
    g /= n_active
    h_new = h - lr * g
    return project_feasible(h_new, n)

def block_cd(h, n, block_size=2, max_passes=100, deadline=None, seed=0):
    """Block coordinate descent: optimize small blocks together."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        # Pick a random starting position for the block
        start = rng.randint(0, max(1, n - block_size))
        block = list(range(start, min(start + block_size, n)))
        old_vals = h[block].copy()
        # For each coordinate in block, do exact 1D update (alternating)
        for p in block:
            c = np.correlate(h, 1.0 - h, mode='full')
            if h[p] < 1e-12 or h[p] > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n, 30)
            h[p] = new_val
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif new_c > old_c + 1e-8:
            # Revert if worse (but this is tricky with the max structure)
            pass
        old_c = best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n, iters=35):
    h0 = h[p]
    s_lo = max(-p, p - (n - 1))
    s_hi = min((n - 1) - p, p)
    s = np.arange(s_lo, s_hi + 1)
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    valid_m = (m >= 0) & (m < 2 * n - 1)
    b = np.zeros(len(s))
    b[valid_m] = c[m[valid_m]] - a[valid_m] * h0
    m_all = np.arange(2 * n - 1)
    m_min = n - 1 + s_lo
    m_max = n - 1 + s_hi
    unaff = (m_all < m_min) | (m_all > m_max)
    M_unaff = float(c[unaff].max()) if unaff.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())

        active = vals >= (am - 1e-11)

        min_sl = float(a[active].min())

            hi = mid
        elif max_sl < -1e-10:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:

        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=500, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n, 35)
            if abs(new_val - h0) > 1e-12:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-11:
            best_c = new_c
            stall = 0
        else:
            stall += 1
            if stall > 5 and improved == 0:
                break
    return h, best_c

def subgradient_step(h, n, lr, n_active=5):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:n_active]
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        ps = np.arange(n) + s
        pm = np.arange(n) - s
        vps = (ps >= 0) & (ps < n)
        vpm = (pm >= 0) & (pm < n)
        g[vps] += (1.0 - h[ps[vps]])
        g[vpm] -= h[pm[vpm]]
    g /= n_active
    h_new = h - lr * g
    return project_feasible(h_new, n)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

            new_h = project_feasible(res.x.copy(), n)

            if new_c < best_c:

        except Exception:
            pass
    return best_h, best_c
