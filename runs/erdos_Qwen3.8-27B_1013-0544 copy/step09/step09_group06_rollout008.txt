from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_corr(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        g[-shift:] += (1.0 - h[:n + shift])
        g[:n + shift] -= h[-shift:]
    return g

def subgradient_step(h, n, top_k=10, eta=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        shift = int(k) - (n - 1)
        grad += compute_grad_corr(h, n, shift)
    grad /= len(top_idx)
    grad *= dx
    h_new = h - eta * grad
    return project_feasible(h_new, n)

def subgradient_descent(h, n, max_iters=1000, top_k=10, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.05
    for t in range(max_iters):
        if deadline and time.time() > deadline:
            break
        h_new = subgradient_step(h, n, top_k, eta)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            eta *= 0.95
        if t % 50 == 0:
            eta = max(0.001, eta * 1.1)  # occasional increase
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_corr(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        g[-shift:] += (1.0 - h[:n + shift])
        g[:n + shift] -= h[-shift:]
    return g

def subgradient_h(h, n, top_k=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        shift = int(k) - (n - 1)
        grad += compute_grad_corr(h, n, shift)
    grad /= len(top_idx)
    return grad * dx

def subgradient_descent(h, n, max_iters, top_k, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eta = 0.03
    stall = 0
    for t in range(max_iters):
        if time.time() > deadline:
            break
        g = subgradient_h(h, n, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        h_new = project_feasible(h - eta * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            eta = min(eta * 1.05, 0.1)
        else:
            h = h_new
            stall += 1
            if stall % 20 == 0:
                eta *= 0.8
                h = best_h.copy()
                if eta < 0.001:
                    break
    return best_h, best_c

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def fourier_search(h, n, n_modes, deadline, rng):
    x = np.arange(n) / n * 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for trial in range(8):
        if time.time() > deadline:
            break
        amps = rng.uniform(0.05, 0.4, n_modes)
        phases = rng.uniform(0, 2 * np.pi, n_modes)
        freqs = np.arange(1, n_modes + 1)
        h_f = np.full(n, 0.5)
        for m in range(n_modes):
            h_f += amps[m] * np.sin(2 * np.pi * freqs[m] * x / 2.0 + phases[m])
        h_f = project_feasible(h_f, n)
        h_c, c_c = multi_cd(h_f, n, 80, deadline, 15)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c
    return best_h, best_c

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
