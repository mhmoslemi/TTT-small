    s = target - h.sum() + h[p] + h[q]  # required sum of h[p] + h[q]
    
    if s < 0 or s > 2:
        return h[p], h[q]  # infeasible
    
    # Compute A_m, B_m, C_m for all m
    # c_m = sum_i h[i] * (1 - h[i+m-n+1])
    # Terms involving h[p]: 
    #   When i=p: h[p] * (1-h[p+m-n+1])
    #   When i+m-n+1=p, i.e., i = p-m+n-1: h[p-m+n-1] * (1-h[p])

    # Terms involving both h[p] and h[q]:
    #   When i=p and p+m-n+1=q: h[p]*(1-h[q])
    #   When i=q and q+m-n+1=p: h[q]*(1-h[p])
    
    # Let me compute this more carefully.
    # c_m = sum_i h[i] * (1 - h[i+m-n+1])
    # = sum_i h[i] - sum_i h[i] * h[i+m-n+1]

    # Then c_m = sum_i h[i] - f(m)
    
    # When we change h[p] and h[q]:
    # sum_i h[i] changes by (h[p]_new - h[p]_old) + (h[q]_new - h[q]_old)
    # But h[p]_new + h[q]_new = s (fixed), so sum changes by s - h[p]_old - h[q]_old
    
    # For f(m):
    # f(m) = sum_i h[i] * h[i+m-n+1]
    # Terms involving h[p]:
    #   i=p: h[p] * h[p+m-n+1]
    #   i+m-n+1=p: h[p-m+n-1] * h[p]
    # Terms involving h[q]: similar
    # Cross terms (both p and q):
    #   i=p, p+m-n+1=q: h[p]*h[q]
    #   i=q, q+m-n+1=p: h[q]*h[p]
    
    # This is getting complex. Let me just compute the contribution directly.
    
    # Actually, let me use a different approach. Compute c_m for the current h,
    # then compute how c_m changes when we change h[p] and h[q].
    
    # c_m = sum_i h[i] * (1-h[i+m-n+1])
    # dc_m/dh[p] = (1-h[p+m-n+1]) + h[p-m+n-1] * (-1) [from the -h[i]*h[i+m-n+1] term]
    # Wait, let me be more careful.
    
    # c_m = sum_i h[i] - sum_i h[i]*h[i+m-n+1]
    # dc_m/dh[p] = 1 - h[p+m-n+1] - h[p-m+n-1]
    # (where terms are 0 if index is out of range)
    # But this is the partial derivative holding other h's fixed.
    # Since c_m is bilinear in h, the change when h[p] changes by delta_p is:
    # delta_c_m = delta_p * (1-h[p+m-n+1]) - delta_p * h[p-m+n-1]
    #           = delta_p * (1 - h[p+m-n+1] - h[p-m+n-1])
    # Wait, that's not right either. Let me recompute.
    
    # c_m = sum_i h[i]*(1-h[i+m-n+1])
    # = sum_i h[i] - sum_i h[i]*h[i+m-n+1]

    # 1. sum_i h[i] when i=p: contributes h[p]
    # 2. sum_i h[i]*h[i+m-n+1] when i=p: contributes h[p]*h[p+m-n+1]
    # 3. sum_i h[i]*h[i+m-n+1] when i+m-n+1=p, i.e., i=p-m+n-1: contributes h[p-m+n-1]*h[p]

    # h[p] - h[p]*h[p+m-n+1] - h[p-m+n-1]*h[p]
    # = h[p] * (1 - h[p+m-n+1] - h[p-m+n-1])
    
    # Similarly for h[q].
    
    # But wait, there's a cross term when p+m-n+1 = q (or equivalently q+m-n+1 = p):
    # In that case, h[p]*h[q] appears in the sum, and it involves BOTH h[p] and h[q].
    
    # Let me handle this case by case.
    
    # For simplicity, let me just compute the new c_m directly for candidate values.
    # Since we're solving an LP, let me set it up properly.
    
    # Actually, the simplest approach: for the LP, I need c_m to be linear in (h[p], h[q]).
    # Let me write c_m = a_m * h[p] + b_m * h[q] + d_m * h[p]*h[q] + e_m
    # where d_m is 0 or -1 (from the cross term) and e_m doesn't involve h[p] or h[q].
    
    # The d_m * h[p]*h[q] term makes it non-linear! This is a problem.
    
    # The cross term occurs when p+m-n+1 = q, i.e., m = q-p+n-1
    # or when q+m-n+1 = p, i.e., m = p-q+n-1
    # For a given pair (p,q), there are at most 2 such m values.
    
    # For those m values, c_m has a h[p]*h[q] term, making the LP non-linear.
    
    # Options:
    # 1. Ignore the cross term (it's at most 2 out of 2n-1 terms)
    # 2.Linearize: since h[p]*h[q] ≤ min(h[p], h[q]), we can use an upper bound
    # 3. Use a different approach for those specific m values
    
    # Actually, for the LP to be valid, we need c_m ≤ t for all m.
    # If c_m has a -h[p]*h[q] term (which it does, since it's -sum h[i]*h[i+m-n+1]),
    # then c_m = ... - h[p]*h[q] + ...
    # Since -h[p]*h[q] ≤ 0, we have c_m ≤ (the value without the cross term).
    # So if we solve the LP ignoring the cross term (i.e., setting it to 0),
    # the solution will satisfy c_m ≤ t even with the cross term (since the cross term
    # only decreases c_m).
    
    # Wait, that's not quite right. Let me think again.
    # c_m = a_m*h[p] + b_m*h[q] - h[p]*h[q] + e_m (for the cross term case)
    # We want c_m ≤ t.
    # Since -h[p]*h[q] ≤ 0, we have c_m ≤ a_m*h[p] + b_m*h[q] + e_m.
    # So if a_m*h[p] + b_m*h[q] + e_m ≤ t, then c_m ≤ t.
    # This means ignoring the cross term gives a valid (but possibly conservative) LP.
    
    # Great! So we can safely ignore the cross term in the LP.
    
    # Now let me compute a_m, b_m, e_m for all m.
    
    h0 = h[p]
    h1 = h[q]
    
    # For each m (correlation lag), compute:
    # c_m = sum_i h[i]*(1-h[i+m-n+1])
    # The terms involving h[p] (excluding cross): h[p]*(1-h[p+m-n+1]) - h[p-m+n-1]*h[p]
    #   = h[p] * (1 - h[p+m-n+1] - h[p-m+n-1])
    #   But careful: if p+m-n+1 == q, the term h[p+m-n+1] = h[q], and we've excluded the cross.
    #   Similarly if p-m+n-1 == q.
    
    # Let me just compute a_m, b_m, e_m by:
    # 1. Compute c_m with h[p]=0, h[q]=0 (call it c_m_base)
    # 2. Compute c_m with h[p]=1, h[q]=0 (call it c_m_p)
    # 3. Compute c_m with h[p]=0, h[q]=1 (call it c_m_q)
    # Then c_m = c_m_base + a_m*h[p] + b_m*h[q] (+ cross term)
    # where a_m = c_m_p - c_m_base, b_m = c_m_q - c_m_base
    
    # But computing c_m for each scenario is O(n^2) total, which is fine for n=600.
    
    # Actually, even simpler: I can analytically compute a_m and b_m.
    # a_m = dc_m/dh[p] (ignoring cross) = 1 - h[p+m-n+1] - h[p-m+n-1]
    #      (where terms with out-of-range indices are 0)
    #      But if p+m-n+1 == q, we should not include h[q] in this (since we're treating h[q] separately)
    #      Actually, for the LP, a_m is the coefficient of h[p] when h[q] is held at 0 (or rather, 
    #      when the cross term is ignored).

    # a_m = contribution of h[p] to c_m, treating h[q] as a fixed constant (not part of the variable)
    # = (1 - h_val[p+m-n+1]) - h_val[p-m+n-1]
    # where h_val is the current h with h[p] removed (set to 0 for computation)
    # But this is getting too complicated. Let me use the direct computation approach.
    
    # Direct approach: compute c for h with h[p]=0, h[q]=0, then with h[p]=1,h[q]=0, etc.
    # This is 4 correlation computations, each O(n^2). For n=600, that's 4*360K = 1.4M ops. Fine.
    
    h_base = h.copy()
    h_base[p] = 0
    h_base[q] = 0
    c_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    
    h_p = h.copy()
    h_p[p] = 1.0
    h_p[q] = 0
    c_p = np.correlate(h_p, 1.0 - h_p, mode='full')
    
    h_q = h.copy()
    h_q[p] = 0
    h_q[q] = 1.0
    c_q = np.correlate(h_q, 1.0 - h_q, mode='full')
    
    # c_m ≈ c_base[m] + a_m * h[p] + b_m * h[q]
    a = c_p - c_base
    b = c_q - c_base
    e = c_base
    
    # LP: minimize t
    # subject to: a[m]*x + b[m]*y + e[m] <= t  for all m
    #             x + y = s
    #             0 <= x <= 1
    #             0 <= y <= 1
    # where x = h[p], y = h[q]
    
    # Substituting y = s - x:
    # a[m]*x + b[m]*(s-x) + e[m] <= t
    # (a[m]-b[m])*x + b[m]*s + e[m] <= t
    # 0 <= x <= 1
    # 0 <= s-x <= 1  =>  s-1 <= x <= s
    
    # So: max(s-1, 0) <= x <= min(s, 1)
    # minimize t
    # subject to: (a[m]-b[m])*x + b[m]*s + e[m] <= t  for all m
    
    # This is a 1D LP! Variables: x, t
    # Constraints: (a[m]-b[m])*x - t <= -(b[m]*s + e[m]) for all m
    #              0 <= x <= 1, s-1 <= x <= s
    #              t >= 0 (actually t >= max of the LHS)
    
    # Even simpler: for a given x, t = max_m {(a[m]-b[m])*x + b[m]*s + e[m]}
    # We want to find x that minimizes this.
    # This is the same 1D problem as exact_coord_min!
    
    # So the 2D problem reduces to a 1D problem when we eliminate y = s-x.
    
    # The 1D problem: minimize max_m {d_m * x + f_m} subject to lo_x <= x <= hi_x
    # where d_m = a[m] - b[m], f_m = b[m]*s + e[m]
    # lo_x = max(s-1, 0), hi_x = min(s, 1)
    
    d = a - b
    f = b * s + e
    lo_x = max(s - 1.0, 0.0)
    hi_x = min(s, 1.0)
    
    if lo_x > hi_x:
        return h0, h1  # infeasible
    
    # Binary search / ternary search for optimal x
    # The function g(x) = max_m {d_m * x + f_m} is convex in x (max of linear functions)
    # So we can use ternary search or binary search on the subgradient
    
    best_x = (lo_x + hi_x) / 2
    best_val = float('inf')
    
    # Use gold-section search or just evaluate at a few points
    # Since it's 1D convex, bisection on the subgradient works
    lo, hi = lo_x, hi_x
    for _ in range(80):
        mid = (lo + hi) / 2
        vals = d * mid + f
        am = float(vals.max())
        active = vals >= am - 1e-12
        if not active.any():
            break
        min_sl = float(d[active].min())
        max_sl = float(d[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    
    x_opt = (lo + hi) / 2
    y_opt = s - x_opt
    x_opt = float(np.clip(x_opt, lo_x, hi_x))
    y_opt = float(np.clip(y_opt, 0.0, 1.0))
    
    # Verify
    g = float(np.max(d * x_opt + f))
    g_h0 = float(np.max(d * h0 + f))
    
    if g < g_h0 - 1e-12:
        return x_opt, y_opt
    else:
        return h0, h1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def pair_optimize(h, n, p, q):
    """Optimize h[p] and h[q] jointly."""
    h0, h1 = h[p], h[q]
    target = n / 2.0
    s = target - h.sum() + h0 + h1
    s = np.clip(s, 0.0, 2.0)
    lo_x = max(s - 1.0, 0.0)
    hi_x = min(s, 1.0)
    if lo_x >= hi_x - 1e-15:
        return h0, h1
    h_base = h.copy()
    h_base[p] = 0.0
    h_base[q] = 0.0
    c_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    h_p = h.copy()
    h_p[p] = 1.0
    h_p[q] = 0.0
    c_p = np.correlate(h_p, 1.0 - h_p, mode='full')
    h_q = h.copy()
    h_q[p] = 0.0
    h_q[q] = 1.0
    c_q = np.correlate(h_q, 1.0 - h_q, mode='full')
    a = c_p - c_base
    b = c_q - c_base
    d = a - b
    f = b * s + c_base
    lo, hi = lo_x, hi_x
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = d * mid + f
        am = float(vals.max())
        active = vals >= am - 1e-12
        if not active.any():
            break
        min_sl = float(d[active].min())
        max_sl = float(d[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    x_opt = float(np.clip((lo + hi) * 0.5, lo_x, hi_x))
    y_opt = float(np.clip(s - x_opt, 0.0, 1.0))
    g_new = float(np.max(d * x_opt + f))
    g_old = float(np.max(d * h0 + f))
    if g_new < g_old - 1e-14:
        return x_opt, y_opt
    return h0, h1

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0, use_pairs=True):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        interior = (h > 0.005) & (h < 0.995)
        int_idx = np.where(interior)[0]
        ext_idx = np.where(~interior)[0]
        if len(int_idx) > 0:
            int_idx = int_idx[rng.permutation(len(int_idx))]
        if len(ext_idx) > 0:
            ext_idx = ext_idx[rng.permutation(len(ext_idx))]
        order = np.concatenate([int_idx, ext_idx]) if len(int_idx) > 0 else ext_idx
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            if use_pairs:
                # Try pair with neighbor
                q = (p + 1) % n if p + 1 < n else p - 1
                hq = h[q]
                if 0.005 < hq < 0.995:
                    x_new, y_new = pair_optimize(h, n, p, q)
                    if abs(x_new - h0) > 1e-12 or abs(y_new - hq) > 1e-12:
                        h[p] = x_new
                        h[q] = y_new
                        improved += 1
                        continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-12:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lse_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x.copy(), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    n_cur = len(best_h)

            best_h, best_c = h_o, c_o

        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
