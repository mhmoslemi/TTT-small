# SLSQP with analytical gradient
def lse_objective(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    ac -= ac.max()
    val = (ac.max() + np.log(np.sum(np.exp(ac)))) / alpha * (2.0 / n)
    return val

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()
    dx = 2.0 / n
    # term1[p] = sum_j (1-h[j]) * w[p-j+n-1] for valid indices
    # = np.convolve(1-h, w, 'full')[n-1 : n-1+n]
    term1 = np.convolve(1.0 - h, w, mode='full')[n-1:n-1+n]
    # term2[p] = sum_j h[j] * w[j-p+n-1] for valid indices
    # = np.convolve(h, w[::-1], 'full')[n-1 : n-1+n]
    term2 = np.convolve(h, w[::-1], mode='full')[n-1:n-1+n]
    return dx * (term1 - term2)
