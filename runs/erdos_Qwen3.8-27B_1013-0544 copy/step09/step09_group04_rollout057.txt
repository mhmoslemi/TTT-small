from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def exact_coord_solve(c, h, p, n):
    """Solve for optimal h[p] given all other h fixed, minimizing max(c)."""
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_cd(h, n, max_passes, deadline, order='sequential'):
    h = h.copy()
    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'random':
            idx = n_idx.copy()
            np.random.shuffle(idx)
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = exact_coord_solve(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_swap_search(h_bin, n, deadline, max_sweeps=100):
    """Local search on binary h using swap moves."""
    h = h_bin.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    
    # Precompute S[p, m] = h[p+m-(n-1)] if valid, else 0
    m_idx = np.arange(2 * n - 1)
    # For each p, the valid m range is: 0 <= p+m-(n-1) < n => n-1-p <= m <= 2n-2-p
    S = np.zeros((n, 2 * n - 1))
    for p in range(n):
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, 2 * n - 2 - p)
        S[p, m_lo:m_hi+1] = h[p + m_idx[m_lo:m_hi+1] - (n-1)]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_improvement = 0.0
        best_i, best_j = -1, -1
        best_new_max = cur_max
        
        # Vectorized: for each i in ones, compute the delta for all j in zeros
        for i in ones:
            if time.time() > deadline:
                break
            # Δc[m] ≈ 2*(S[i,m] - S[j,m]) for all j
            # new_max[j] = max_m(c + 2*(S[i] - S[j]))
            base = c + 2.0 * S[i]  # shape (2n-1,)
            diffs = base[None, :] - 2.0 * S[zeros, :]  # shape (len(zeros), 2n-1)
            new_maxs = diffs.max(axis=1)  # shape (len(zeros),)
            
            # Find best j
            min_idx = np.argmin(new_maxs)
            if new_maxs[min_idx] < best_new_max:
                best_new_max = new_maxs[min_idx]
                best_i = i
                best_j = zeros[min_idx]
                best_improvement = cur_max - best_new_max
        
        if best_i < 0 or best_improvement < 1e-15:
            break
        
        # Apply the swap
        h[best_i] = 0
        h[best_j] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Update S
        S[best_i, :] = 0  # h[best_i] is now 0, so S[best_i, m] = h[best_i + m - (n-1)]... 
        # Wait, S[p, m] = h[p+m-(n-1)], not h[p]. So I need to recompute the affected rows.
        # Actually S[p,m] = h[p+m-(n-1)]. Changing h[best_i] affects S[p,m] where p+m-(n-1) = best_i,
        # i.e., m = best_i - p + (n-1). For each p, there's at most one m.
        # Similarly for best_j.
        # This is O(n) to update.
        for p in range(n):
            m_i = best_i - p + (n - 1)
            if 0 <= m_i < 2 * n - 1:
                S[p, m_i] = 0  # was h[best_i] = 1, now 0
            m_j = best_j - p + (n - 1)
            if 0 <= m_j < 2 * n - 1:
                S[p, m_j] = 1  # was h[best_j] = 0, now 1
        
        # Update c
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
    
    return h, cur_max
