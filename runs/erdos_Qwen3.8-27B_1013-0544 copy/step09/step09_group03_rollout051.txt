```python
def exact_coord_min_topk(c, p, h, n, k=10):
    h0 = h[p]
    # Only consider the top-k shifts
    top_idx = np.argsort(c)[-k:][::-1]  # indices of top-k shifts, descending
    s_top = top_idx - (n - 1)  # corresponding shifts
    # For each of these shifts, compute the coefficient
    a = np.zeros(len(s_top))
    for i, s in enumerate(s_top):
        ps = p + s
        pm = p - s
        if 0 <= ps < n:
            a[i] += (1.0 - h[ps])
        if 0 <= pm < n:
            a[i] -= h[pm]
    b = c[top_idx] - a * h0
    # The other shifts (not in top-k) have max value M_unaff
    M_unaff = c[np.setdiff1d(np.arange(2*n-1), top_idx)].max()
    
    # Binary search for optimal h[p]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = vals.max()
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        min_sl = a[active].min()
        max_sl = a[active].max()
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    # Evaluate candidates
    cands = np.array([lo, hi, (lo+hi)/2, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        x = float(np.clip(x, 0, 1))
        v = max((a * x + b).max(), M_unaff)
        if v < best_v:
            best_v = v
            best_x = x
    return best_x
```
