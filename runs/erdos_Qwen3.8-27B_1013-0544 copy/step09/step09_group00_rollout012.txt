    m = c.max()
    excess = np.maximum(c - m, 0.0) + 1e-12
    return m + (np.mean(excess ** p)) ** (1.0 / p)

def slsqp_opt(h, n, p_values, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
             'jac': lambda x, n_=n: np.ones(n_)}]
    for p in p_values:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_power, best_h, args=(n, p), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        if nt == n:

        xo = np.linspace(0, 2, len(h_best))
        xn = np.linspace(0, 2, nt)
        h_int = project_feasible(np.interp(xn, xo, h_best), nt)
        ho, co = slsqp_opt(h_int, nt, [50, 100], 300, deadline)
        ho, co = exact_cd(ho, nt, 500, deadline)
        if co < best_c - 1e-8:
            h_best, best_c, n = ho, co, nt

    for r in range(5):
        if time.time() > deadline - 80:
            break
        s = 0.005 + 0.01 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = exact_cd(hp, n, 300, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    if time.time() < deadline - 5:
        h_best, best_c = exact_cd(h_best, n, 300, deadline)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
