def lse_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m < 1e-14:

    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(600):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, min(k, i + 12)):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = eval_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = eval_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.18, abs(best_d) * 0.4 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(eval_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=80)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=35):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-13:
                new_val = eval_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if m < 1e-15:
        return 0.0
    sh = alpha * (c - m)
    mx = float(sh.max())
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_step(h, n, alpha, maxiter=500, deadline=None):
    def obj(x):
        return lse_obj(x, n, alpha)
    def sf(x):
        return x.sum() - n / 2.0
    def sj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sf, 'jac': sj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def lse_grad_descend(h, n, alpha, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.8 / (alpha * 2.0)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = float(c.max())
        sh = alpha * (c - m)
        mx = float(sh.max())
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for mk in range(2 * n - 1):
            wk = w[mk]
            if wk < 1e-8:
                continue
            for i in range(n):
                j = i + mk - (n - 1)
                if 0 <= j < n:
                    grad[i] += wk * (1.0 - h[j])
                jp = i - mk + (n - 1)
                if 0 <= jp < n:
                    grad[i] -= wk * h[jp]
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            lr = min(lr * 1.03, 2.0)
        else:
            lr *= 0.6
            if lr < 1e-6:
                break
    return h, best_c
