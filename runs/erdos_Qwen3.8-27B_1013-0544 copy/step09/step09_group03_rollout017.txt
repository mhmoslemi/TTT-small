def pair_optimize(h, n, p, q, max_evals=40):
    """Optimize the pair (h[p], h[q]) jointly."""
    target = n / 2.0
    S_other = h.sum() - h[p] - h[q]
    T = target - S_other
    lo = max(0.0, T - 1.0)
    hi = min(1.0, T)
    if lo >= hi:
        return h[p]
    
    # Save original values
    hp, hq = h[p], h[q]
    
    def obj(t):
        h[p] = t
        h[q] = T - t
        c = np.correlate(h, 1.0 - h, mode='full')
        return c.max()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    for _ in range(max_evals):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    # Restore
    t_opt = 0.5 * (lo + hi)  # approximately
    # Actually use the best of c and d
    h[p] = c if fc < fd else d
    h[q] = T - h[p]
    
    return h[p]

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, 0.5 * (lo + hi), 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def golden_pair(h, n, p, q):
    T = n / 2.0 - (h.sum() - h[p] - h[q])
    lo = max(0.0, T - 1.0)
    hi = min(1.0, T)
    if hi - lo < 1e-12:
        return False
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_, b_ = lo, hi
    c_ = b_ - gr * (b_ - a_)
    d_ = a_ + gr * (b_ - a_)
    h[p], h[q] = c_, T - c_
    fc = float(np.correlate(h, 1.0 - h, mode='full').max())
    h[p], h[q] = d_, T - d_
    fd = float(np.correlate(h, 1.0 - h, mode='full').max())
    for _ in range(35):
        if fc < fd:
            b_ = d_
            d_ = c_
            fd = fc
            c_ = b_ - gr * (b_ - a_)
            h[p], h[q] = c_, T - c_
            fc = float(np.correlate(h, 1.0 - h, mode='full').max())
        else:
            a_ = c_
            c_ = d_
            fc = fd
            d_ = a_ + gr * (b_ - a_)
            h[p], h[q] = d_, T - d_
            fd = float(np.correlate(h, 1.0 - h, mode='full').max())
    if fc < fd:
        h[p], h[q] = c_, T - c_
    else:
        h[p], h[q] = d_, T - d_
    return True

def pair_cd(h, n, top_k, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        idx = np.argsort(c)[::-1][:top_k]
        k_vals = np.abs(idx - (n - 1))
        k_vals = k_vals[k_vals > 0]
        k_set = set(k_vals.tolist())
        coord_count = np.zeros(n)
        for m in idx:
            k = m - (n - 1)
            if k >= 0:
                coord_count[:n-k] += 1
                coord_count[k:] += 1
            else:
                coord_count[-k:] += 1
                coord_count[:n+k] += 1
        interior = (h > 0.02) & (h < 0.98)
        coord_count[~interior] = 0
        top_coords = np.argsort(coord_count)[::-1][:100]
        top_coords = top_coords[coord_count[top_coords] > 0]
        if len(top_coords) < 2:
            break
        improved = 0
        n_try = min(60, len(top_coords) * (len(top_coords) - 1) // 2)
        tried = set()
        for _ in range(n_try):
            if time.time() > deadline:
                break
            pi = rng.randint(len(top_coords))
            qi = rng.randint(len(top_coords))
            if pi == qi:
                continue
            p, q = int(top_coords[pi]), int(top_coords[qi])
            key = (min(p, q), max(p, q))
            if key in tried:
                continue
            tried.add(key)
            gap = abs(p - q)
            if gap not in k_set and gap > 0:
                if rng.random() > 0.3:
                    continue
            if golden_pair(h, n, p, q):
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def nesterov_max_autocorr(h, n, max_iter, alpha0, deadline, tol=0.01):
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + 0.3 * v
        y = project_feasible(y, n)
        c = np.correlate(y, y, mode='full')
        c_min = float(c.min())
        active = np.where(c < c_min + tol)[0]
        active = active[abs(active - (n-1)) % n <= n-1]
        if len(active) == 0:
            active = np.array([n-1])
        grad = np.zeros(n)
        for m in active[:30]:
            k = abs(m - (n - 1))
            if k == 0:
                grad += 2.0 * y
            else:
                grad[:n-k] += y[k:]
                grad[k:] += y[:n-k]
        grad /= len(active[:30])
        alpha = alpha0 / (1.0 + 0.05 * it)
        v = 0.3 * v + alpha * grad
        h_new = project_feasible(y + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            h = project_feasible(y - 0.5 * v, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD warmup
    if time.time() < deadline - 500:
        h1, c1 = coord_descent(best_h, n, 400, deadline - 480, seed)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Pair CD (key innovation)
    if time.time() < deadline - 350:
        h2, c2 = pair_cd(best_h, n, top_k=40, max_passes=150, deadline=deadline - 340, seed=seed + 1)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = coord_descent(best_h, n, 300, deadline - 330, seed + 2)
        if c2b < best_c:
            best_h, best_c = h2b, c2b
    # Phase 3: Nesterov on autocorrelation reformulation
    if time.time() < deadline - 250:
        h3, c3 = nesterov_max_autocorr(best_h, n, 500, alpha0=2.0, deadline=deadline - 240)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3b, c3b = coord_descent(best_h, n, 200, deadline - 230, seed + 3)
        if c3b < best_c:
            best_h, best_c = h3b, c3b
    # Phase 4: Alternating pair + single
    for rd in range(4):
        if time.time() > deadline - 180:
            break
        h4, c4 = pair_cd(best_h, n, top_k=25, max_passes=80, deadline=deadline - 175, seed=seed + 10 + rd * 3)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_descent(best_h, n, 200, deadline - 170, seed + 20 + rd * 3)
        if c4b < best_c:
            best_h, best_c = h4b, c4b
    # Phase 5: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 800, 1000]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 150, deadline - 75, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 6: Stochastic restarts
    n_cur = len(best_h)
    for r in range(10):
        if time.time() > deadline - 40:
            break
        s = 0.004 + 0.012 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 150, deadline - 35, seed + 100 + r * 11)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
        if time.time() < deadline - 40 and c_o < best_c + 0.001:
            h_o2, c_o2 = pair_cd(hp, n_cur, top_k=30, max_passes=40, deadline=deadline - 35, seed=seed + 300 + r * 7)
            if c_o2 < best_c:
                best_h, best_c = h_o2, c_o2
    # Phase 7: Final extensive polish
    n_cur = len(best_h)
    if time.time() < deadline - 10:
        hf, cf = coord_descent(best_h, n_cur, 3000, deadline - 5, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, c_f
    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
