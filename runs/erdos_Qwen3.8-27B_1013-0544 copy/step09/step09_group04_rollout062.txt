            hs = gaussian_smooth(best_h, sigma=sg)
            hs = project_feasible(hs, n)
            h_s, c_s = cd_loop(hs, n, deadline - 5)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Phase 6: Final polish
    if time.time() < deadline - 2:
        h_f, c_f = cd_loop(best_h, len(best_h), deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
