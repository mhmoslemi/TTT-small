import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    # Same as before - exact 1D line search for coordinate p
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes, deadline, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

Actually, I realize I should also think about what else could help. Let me think about the structure of the problem more.

The problem is essentially: find a set (or weighted set) A in [0,2] of measure 1 that minimizes the maximum "difference" correlation. In additive combinatorics, this is related to the "difference set" problem.

For the discrete version, we want a binary (or near-binary) sequence h of length n with sum n/2 that minimizes max_k |autocorrelation-like quantity|.

Actually, it's not exactly autocorrelation. It's:
c[k] = sum_i h[i] * (1-h[i+k])

This measures the "non-overlap" between h and its shift. We want to minimize the maximum non-overlap.

Equivalently: c[k] = sum_i h[i] - sum_i h[i]*h[i+k] = (sum h) - autocorr[k]
So minimizing max_k c[k] is the same as minimizing max_k [(sum h) - autocorr[k]] = (sum h) - min_k autocorr[k].

Wait, that's not quite right either because the sum is only over valid indices.

Actually: c[k] = sum_i h[i]*(1-h[i+k]) where the sum is over i such that both i and i+k are in [0,n-1].
= sum_i h[i] (over valid i) - sum_i h[i]*h[i+k] (over valid i)

The first term depends on k (it's the number of valid i times the average of h over those i). For large n and shifts not near the boundary, it's approximately n/2 * (2/n) = 1 (times dx).

So minimizing max_k c[k] is roughly equivalent to maximizing min_k autocorr[k], which is a "flat autocorrelation" problem. This is related to the problem of finding sequences with minimal peak autocorrelation (or maximal flatness).

This suggests that the optimal solution might have a "Golomb ruler" or "Barker sequence" type structure.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes, deadline, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_value_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    log_sum = smax + np.log(np.exp(shifted - smax).sum())
    val = (m + log_sum / alpha) * dx
    w = np.exp(shifted - smax)
    w /= w.sum()
    u = 1.0 - x
    fwd = np.correlate(u, w[::-1], mode='full')
    bwd = np.correlate(x, w, mode='full')
    grad = dx * (fwd[n-1:n-1+n] - bwd[n-1:n-1+n])
    return val, grad

def adam_optimize(h, n, alpha_schedule, iters_per_alpha, lr, deadline):
    m = h.shape[0]
    v = np.zeros(m)
    z = np.zeros(m)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        lr_t = lr
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            t += 1
            val, grad = lse_value_grad(h, n, alpha)
            v = 0.9 * v + 0.1 * grad
            z = 0.999 * z + 0.001 * grad * grad
            v_hat = v / (1 - 0.9 ** t)
            z_hat = z / (1 - 0.999 ** t)
            step = lr_t * v_hat / (np.sqrt(z_hat) + 1e-8)
            h_new = h - step
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    return best_h, best_c
