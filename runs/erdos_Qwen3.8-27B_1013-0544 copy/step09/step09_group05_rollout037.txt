    shifted = shifted - shifted.max()
    exp_s = np.exp(shifted)

    obj += penalty * (x.sum() - n / 2.0) ** 2

    for s in range(-(n-1), n):

        i1hi = min(n, n - s)
        if i1lo < i1hi:

        if i2lo < i2hi:

def coord_descent(h, n, max_passes, deadline, rng, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fista(h, n, alpha_lse, steps, lr, deadline, penalty=100.0):
    """FISTA (accelerated proximal gradient) for the smooth objective."""
    x = h.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_c = compute_c5(x, n)
    
    for k in range(steps):
        if time.time() > deadline:
            break
        _, grad = lse_obj_grad(y, n, alpha_lse, penalty)
        # Gradient step
        x_new = y - lr * grad
        # Proximal step (projection)
        x_new = project_feasible(x_new, n)
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)  # Keep y in bounds for stability
        x = x_new
        t = t_new
        
        # Check improvement
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    
    return best_x, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def lse_fg(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    es = np.exp(sh)
    z = es.sum()
    w = es / z
    obj = (m + (sh.max() + np.log(z)) / alpha) * (2.0 / n)
    obj += penalty * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi < 1e-15:
            continue
        i1lo, i1hi = max(0, -s), min(n, n - s)
        if i1lo < i1hi:
            grad[i1lo:i1hi] += wi * (1.0 - x[i1lo + s:i1hi + s])
        i2lo, i2hi = max(0, s), min(n, n + s)
        if i2lo < i2hi:
            grad[i2lo:i2hi] -= wi * x[i2lo - s:i2hi - s]
    grad *= 2.0 / n
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return obj, grad

def lbfgsb_run(h, n, alphas, maxiter, penalty, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(lse_fg, best_h, args=(n, alpha, penalty),
                           method='L-BFGS-B', jac=True,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-14, 'maxfun': maxiter * 15})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a, hi_a = max(0, n - 1 - p), min(2 * n - 2, n - 1)
    lo_b, hi_b = max(0, p), min(2 * n - 2, p + n - 1)
    m_lo, m_hi = min(lo_a, lo_b), max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf, jb = p + ms - (n - 1), p - ms + (n - 1)
    vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    cb = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_d = len(di) > 0
    cd = 0.0
    if has_d:
        cd = cb[di[0]]
        a[di] = 0.0
    nd = np.ones(len(ms), dtype=bool)
    if has_d:
        nd[di] = False
    a_nd, c_nd = a[nd], cb[nd]
    umax = 0.0
    if m_lo > 0:
        umax = max(umax, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        umax = max(umax, float(np.max(c[m_hi + 1:])))
    L, H = umax, float(np.max(c))
    for _ in range(52):
        if H - L < 1e-15:
            break
        T = (L + H) / 2.0
        ok = True
        if umax > T + 1e-16:
            ok = False
        elif ok and len(a_nd) > 0:
            xl, xh = 0.0, 1.0
            neg, pos = a_nd < -1e-13, a_nd > 1e-13
            if np.any(neg):
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if xl > xh + 1e-15:
                ok = False
        if ok and has_d:
            Cq = T - cd + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1, r2 = (1 - np.sqrt(disc)) / 2, (1 + np.sqrt(disc)) / 2
                if len(a_nd) > 0:
                    neg, pos = a_nd < -1e-13, a_nd > 1e-13
                    xl, xh = 0.0, 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl, xh = max(xl, 0.0), min(xh, 1.0)
                    if xl > xh + 1e-15:
                        ok = False
                    elif xl >= r1 and xh <= r2:
                        ok = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        ok = False
                    elif 0.0 < r1 and 1.0 > r2:
                        ok = False
        if ok:
            H = T
        else:
            L = T
    T = (L + H) / 2.0
    xl, xh = 0.0, 1.0
    if len(a_nd) > 0:
        neg, pos = a_nd < -1e-13, a_nd > 1e-13
        if np.any(neg):
            xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if has_d:
        Cq = T - cd + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1, r2 = (1 - np.sqrt(disc)) / 2, (1 + np.sqrt(disc)) / 2
            if xl < r1:
                xh = min(xh, r1)
            elif xh > r2:
                xl = max(xl, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    xh = min(xh, r1 + 1e-9)
                else:
                    xl = max(xl, r2 - 1e-9)
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if xl > xh + 1e-10:
        return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng, random_order=False):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        imp = False
        order = rng.permutation(n) if random_order else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                h[p] = xn
                imp = True
        if not imp:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_push(h, n, beta):
    hp = 1.0 / (1.0 + np.exp(-beta * (2.0 * h - 1.0)))
    return project_feasible(hp, n)
