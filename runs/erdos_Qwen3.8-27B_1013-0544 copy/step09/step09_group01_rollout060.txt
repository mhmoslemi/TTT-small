from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=60):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        m = c.max()
        # Only update coordinates that could improve the max
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:

            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, c[m_hi + 1:].max())
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval_fast(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval_fast(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval_fast(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval_fast(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval_fast(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = max(unaff_max, c[ms].max())
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

For the breakpoint approach:
- We have m_i(d) = c_aff_i + a_i * d for each affected correlation i
- Plus the self-term: s(d) = c_aff_diag + d*(1-h0-d) [quadratic]
- Plus the constant: K = unaff_max

The max function is M(d) = max(m_1(d), ..., m_k(d), s(d), K)

The minimum of M(d) on [d_lo, d_hi] occurs at:
- A breakpoint where two of the m_i's cross (and both are ≥ all others)
- Or at a breakpoint where s(d) = m_i(d) for some i
- Or at the boundaries d_lo, d_hi

To find all breakpoints is O(k²) which could be expensive for large k. But in practice, only the top few m_i's matter (those near the maximum). So I can:
1. Sort the m_i's by their value at d=0
2. Consider only the top T (say T=20)
3. Find all pairwise crossings among these T
4. Evaluate M(d) at each crossing (check if it's actually on the upper envelope)
5. Also check boundaries

This gives an exact minimum in O(T²) time per coordinate, which is much faster than golden section for large gs_iters.

def cd_exact_min(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_lo, d_hi):
    """Find exact minimum of max(c_aff + d*a, self_term, unaff_max) on [d_lo, d_hi]"""
    n_aff = len(c_aff)
    # Candidate points: boundaries
    candidates = [d_lo, d_hi]
    # Also add d=0 for reference
    # Find top-K affected correlations
    vals_at_0 = c_aff + 0.0 * a  # just c_aff
    # Sort by value descending
    order = np.argsort(-vals_at_0)[:min(30, n_aff)]
    top_c = c_aff[order]
    top_a = a[order]
    # Find pairwise crossings
    T = len(top_c)
    for i in range(T):
        for j in range(i+1, T):
            da = top_a[i] - top_a[j]
            if abs(da) > 1e-15:
                d_cross = (top_c[j] - top_c[i]) / da
                if d_lo <= d_cross <= d_hi:
                    candidates.append(d_cross)
    # Also check where self-term crosses top correlations
    if has_diag:
        # self(d) = c_aff[diag_idx] + d*(1-h0-d)
        sc = c_aff[diag_idx]
        for i in range(T):
            # sc + d*(1-h0-d) = top_c[i] + top_a[i]*d
            # -d² + (1-h0-top_a[i])*d + (sc - top_c[i]) = 0
            A = -1.0
            B = 1.0 - h0 - top_a[i]
            C = sc - top_c[i]
            disc = B*B - 4*A*C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / (2*A)
                d2 = (-B - sq) / (2*A)
                if d_lo <= d1 <= d_hi:
                    candidates.append(d1)
                if d_lo <= d2 <= d_hi:
                    candidates.append(d2)
        # Also where self-term = unaff_max
        # sc + d*(1-h0-d) = unaff_max
        A = -1.0
        B = 1.0 - h0
        C = sc - unaff_max
        disc = B*B - 4*A*C
        if disc >= 0:
            sq = np.sqrt(disc)
            d1 = (-B + sq) / (2*A)
            d2 = (-B - sq) / (2*A)
            if d_lo <= d1 <= d_hi:
                candidates.append(d1)
            if d_lo <= d2 <= d_hi:
                candidates.append(d2)
    # Where each top correlation = unaff_max
    for i in range(T):
        if abs(top_a[i]) > 1e-15:
            d_cross = (unaff_max - top_c[i]) / top_a[i]
            if d_lo <= d_cross <= d_hi:
                candidates.append(d_cross)
    # Evaluate at all candidates
    best_d = d_lo
    best_f = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_lo)
    for d in candidates:
        f = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d)
        if f < best_f:
            best_f = f
            best_d = d
    return best_d, best_f

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def cd(h, n, max_passes, deadline, gs_iters=65, rng=None):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve_count = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = c[:m_lo].max()
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, c[m_hi + 1:].max())
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = max(unaff_max, c[ms].max())
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
        else:
            no_improve_count = 0
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def try_resolution(h_best, n_best, n_try, deadline, rng):
    x_old = np.linspace(0, 2, len(h_best))
    x_new = np.linspace(0, 2, n_try)
    h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
    h_o, c_o = slsqp(h_int, n_try, [2000, 10000, 50000], 250)
    h_o, c_o = cd(h_o, n_try, 50, deadline, gs_iters=65, rng=rng)
    return h_o, c_o
