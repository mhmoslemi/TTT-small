```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m_idx = n - 1 + s
    b = c[m_idx] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

        nv = exact_coord_min(c, p, h, n)
        if abs(nv - h0) > 1e-10:

            improved += 1

    h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1

                A = np.zeros((n_lags, m + 1))

                    i2 = j + offsets

                             - np.where(v2, h[np.clip(i2, 0, n-1)], 0.0)
                A[:, m] = -1.0
                c_obj = np.zeros(m + 1)
                c_obj[m] = 1.0

        h = project_feasible(h, n)

def smooth_slsqp(h, n, alpha, maxiter=500):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        # gradient of max(c) + (1/alpha)*logsumexp(alpha*(c-m))
        # = max indicator + soft indicator
        grad_max = np.zeros(n)
        active = (c >= m - 1e-12)
        grad_max[active] += 1.0 / active.sum()
        # for soft part: d/dx logsumexp(alpha*(c-m)) = alpha * sum_L w_L * dc_L/dx
        # dc_L/dx: for each L, c_L = sum_i x[i]*(1-x[i-off_L])

        # This is O(n^2) which is fine for n=600
        # Actually let's just use numerical gradient or skip jac
        return None  # Let SLSQP use numerical
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

We have:
cor[L] = sum_i h[i] * (1 - h[i - off_L]) where off_L = L - (n-1)

If we change h[j] from h[j] to h[j] + delta (for j in block):
delta_cor[L] = sum_{j in block} delta_j * [(1 - h[j - off_L]) * valid(j - off_L) - h[j + off_L] * valid(j + off_L)]

Wait, I need to be more careful. Let's say we're changing h[j_1], h[j_2], ..., h[j_m] simultaneously.

cor_new[L] = cor[L] + sum_k delta_k * g_k(L)

where g_k(L) is the partial derivative of cor[L] with respect to h[j_k], evaluated at the current h.

g_k(L) = d/dh[j_k] [sum_i h[i] * (1 - h[i - off_L])]
= (1 - h[j_k - off_L]) * valid(j_k - off_L) - h[j_k + off_L] * valid(j_k + off_L)

Wait, that's the partial derivative. But cor[L] is bilinear in h, so the linearization is:
cor_new[L] ≈ cor[L] + sum_k g_k(L) * delta_k

This is exact if only one coordinate changes. If multiple coordinates change, there are cross terms:
cor_new[L] = cor[L] + sum_k g_k(L) * delta_k + sum_{k≠l} h_related * delta_k * delta_l

The cross terms arise when j_k and j_l are related by the offset. Specifically, if j_l = j_k + off_L, then the term h[j_k] * (1 - h[j_l]) has a cross term -delta_k * delta_l.

So the linearization is NOT exact for block size > 1! The LP gives a lower bound on the true max, not the exact value.

This is a problem. For block size 1, the linearization is exact. For block size > 1, it's an approximation.

How bad is the approximation? The cross terms are of order delta_k * delta_l, which is small if the changes are small. So for small steps, the LP solution is approximately correct.

To handle this properly, I have two options:
1. Accept the approximation (good for small steps)
2. Use the exact (bilinear) formulation, which makes the problem non-convex

Option 1 is simpler and should work well in practice, especially if we do many small steps. The LP solution gives a direction, and we can verify the actual improvement after applying it.

s.t. 0 ≤ x_1 ≤ 1, 0 ≤ x_2 ≤ 1

where cross_12(L) = -[j_1+off_L == j_2 valid] - [j_2+off_L == j_1 valid] (i.e., -1 if the two coordinates interact at lag L, 0 otherwise).

This is a min-max of bilinear functions, which is non-convex. Hard to solve exactly.

```python
def block_lp_solve(c, idx, h, n, offsets, top_k=50):
    """Solve block subproblem using cutting-plane LP."""
    m = len(idx)
    # Compute A matrix for all lags
    # A[L, k] = g_k(L) = (1-h[j_k - off])*valid - h[j_k + off]*valid
    A_full = np.zeros((2*n-1, m))
    for k, j in enumerate(idx):
        i1 = j - offsets
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + offsets
        v2 = (i2 >= 0) & (i2 < n)
        A_full[:, k] = np.where(v1, 1.0 - h[np.clip(i1, 0, n-1)], 0.0) \
                     - np.where(v2, h[np.clip(i2, 0, n-1)], 0.0)
    
    # Start with top_k constraints
    order = np.argsort(c)[::-1][:top_k]
    active_set = list(order)
    
    for iteration in range(50):
        # Solve LP with active constraints
        A = A_full[active_set]
```
