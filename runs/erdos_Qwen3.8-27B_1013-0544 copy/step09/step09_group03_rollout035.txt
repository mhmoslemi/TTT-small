def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=500, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:

            new_val = exact_coord_min(c, p, h, n)

    # = -alpha*m + log(sum_k exp(alpha*c_k))
    # By log-sum-exp trick: = -alpha*m + (alpha*m + log(sum_k exp(alpha*(c_k-m))))
    # = log(sum_k exp(alpha*(c_k-m)))  ... circular.
    
    # Let me just compute: LSE = alpha*(c-m), shifted_max = LSE.max(), 
    # log_sum = shifted_max + log(sum(exp(LSE - shifted_max)))
    # F = dx * (m + log_sum/alpha)
    
    LSE = alpha * (c - m)
    shifted = LSE - LSE.max()
    log_sum = LSE.max() + np.log(np.sum(np.exp(shifted)))
    val = (m + log_sum / alpha) * dx
    
    # Penalty
    s = h.sum() - n / 2.0
    val += mu * s * s
    
    # Gradient
    g = grad_smoothed_max(h, n, alpha)
    g += 2 * mu * s * np.ones(n)
    return val, g

    mu = 100.0  # penalty for sum constraint

            return val

            _, g = smooth_obj_and_grad(x, n, alpha, mu)
            return g

                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-8, 'maxcor': 50})
            new_h = project_feasible(res.x.copy(), n)

            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m_idx = n - 1 + s
    b = c[m_idx] - a * h0
    m_min = n - 1 + s.min()
    m_max = n - 1 + s.max()
    all_m = np.arange(2 * n - 1)
    unaff = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff].max()) if unaff.any() else 0.0
    c0 = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:

        active = vals >= (am - 1e-12)
        if not active.any():

        smin = float(a[active].min())
        smax = float(a[active].max())

        elif smax < -1e-11:

        else:

        v = max(v, M_unaff, c0 + x - x * x)
        if v < best_v:

            best_x = float(x)

def coord_descent(h, n, max_passes=500, deadline=None, seed=0):

    h = h.copy()

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    lw = alpha * (c - m)
    lw -= lw.max()
    w = np.exp(lw)
    w /= w.sum()
    ip = np.arange(n)[:, None]
    ii = np.arange(n)[None, :]
    k = n - 1 + ii - ip
    M = np.where((k >= 0) & (k <= 2 * n - 2), w[k], 0.0)
    S1 = M @ (1.0 - h)
    S2 = M.T @ h
    S1 -= w[n - 1] * (1.0 - h)
    S2 -= w[n - 1] * h
    g = (1.0 - 2.0 * h) * w[n - 1] + S1 - S2
    g *= dx / alpha
    return g

def smooth_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    lw = alpha * (c - m)
    lw -= lw.max()
    ls = lw.max() + np.log(np.sum(np.exp(lw)))
    return (m + ls / alpha) * dx

        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n, 1000, deadline - 270, seed + 1)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Proximal gradient
    if time.time() < deadline - 250:
        h3, c3 = prox_grad(best_h, n, 2000, 500, deadline - 240)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 500, deadline - 230, seed + 2)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [400, 450, 500, 550, 650, 700, 750, 800, 900, 1000]:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = coord_descent(hi, n_try, 1200, deadline - 185, seed + n_try)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = x_new

    # Phase 5: L-BFGS on best res
    if time.time() < deadline - 200:
        nc_ = len(best_h)
        h5, c5 = lbfgs_opt(best_h, nc_, [500, 2000, 10000], deadline - 185)
        if c5 < best_c:
            best_h, best_c = h5, c5
        h5, c5 = coord_descent(best_h, nc_, 2000, deadline - 170, seed + 50)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Restarts
    n_cur = len(best_h)
    for r in range(15):
        if time.time() > deadline - 80:
            break
        if r < 5:
            hp = project_feasible(best_h + rng.normal(0, 0.01*(r+1), n_cur), n_cur)
        elif r < 10:
            nb = int(n_cur * 0.04 * (r - 4) / 4)
            idx = rng.choice(n_cur, nb, replace=False)
            hp = best_h.copy()
            hp[idx] = rng.uniform(0, 1, nb)
            hp = project_feasible(hp, n_cur)
        else:
            blk = max(3, n_cur // (r - 9 + 2))
            st = rng.randint(0, n_cur - blk)
            hp = best_h.copy()
            hp[st:st+blk] = 1.0 - hp[st:st+blk]
            hp = project_feasible(hp, n_cur)
        ho, co = coord_descent(hp, n_cur, 500, deadline - 60, seed + 300 + r*19)
        if co < best_c:
            best_h, best_c = ho, c_o if False else co

    # Phase 7: Final CD
    if time.time() < deadline - 10:
        nc_ = len(best_h)
        hf, cf = coord_descent(best_h, nc_, 5000, deadline - 3, seed + 777)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Symmetry check
    nc_ = len(best_h)
    hflip = project_feasible(1.0 - best_h, nc_)
    cfl = compute_c5(hflip, nc_)
    if cfl < best_c:
        best_h, best_c = hflip, cfl

    best_h = project_feasible(best_h, nc_)
    best_c = compute_c5(best_h, nc_)
    return best_h, best_c, nc_
