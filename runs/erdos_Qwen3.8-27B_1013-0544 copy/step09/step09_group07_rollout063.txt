```python
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0

                a_coeff[vb][valid_b] -= h[idx - k_shifts[vb][valid_b]]

            if m_hi < 2*n-2:
```

```python
def golden_min(f, a, b, maxiter=60):
    """Golden section minimization of f on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    for _ in range(maxiter):
        if b - a < 1e-14:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = d, c, fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0, f((a + b) / 2.0)
```

```python
def identify_critical_coords(h, n, num_critical=20):
    """Find coordinates most responsible for the max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    # The coordinates that contribute most to c[k_star] are those
    # where h[i]=1 and h[i+k_star]=0 (for the first factor)
    # or h[i-k_star]=1 and h[i]=0 (for the second factor)
    k = k_star - (n - 1)
    # First factor contribution: h[i]*(1-h[i+k]) for valid i
    scores = np.zeros(n)
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            scores[i] += h[i] * (1 - h[j])
        ip = i - k
        if 0 <= ip < n:
            scores[i] += h[ip] * (1 - h[i])
    return np.argsort(scores)[-num_critical:]
```
