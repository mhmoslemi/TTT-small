import math
import time
import numpy as np


def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(120):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (d / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (d / tot) * h, 0.0, 1.0)

    d = target - h.sum()
    if abs(d) > 1e-13:
        if d > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(d, 1.0 - h[idx])
            h[idx] = min(1.0, h[idx] + add)
        else:
            idx = int(np.argmax(h))
            sub = min(-d, h[idx])
            h[idx] = max(0.0, h[idx] - sub)
    return h


def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(full_corr(h)) * (2.0 / n))


def build_maps(n):
    maps = []
    diag_global = n - 1
    full_len = 2 * n - 1
    for p in range(n):
        lo = min(p, n - 1 - p)
        hi = max(n - 1 + p, 2 * n - 2 - p)
        if lo < 0:
            lo = 0
        if hi >= full_len:
            hi = full_len - 1

        ms = np.arange(lo, hi + 1, dtype=np.int64)
        jf = (p + ms - (n - 1)).astype(np.int64)
        jb = (p - ms + (n - 1)).astype(np.int64)

        dloc = diag_global - lo
        if dloc < 0 or dloc >= len(ms):
            dloc = -1

        m_v = (jf >= 0) & (jf < n)
        m_b = (jb >= 0) & (jb < n)

        if dloc >= 0:
            m_v[dloc] = False
            m_b[dloc] = False

        jfp = np.where(m_v)[0].astype(np.int32)
        jfv = jf[m_v].astype(np.int32)
        jbp = np.where(m_b)[0].astype(np.int32)
        jbv = jb[m_b].astype(np.int32)

        maps.append((lo, hi, jfp, jfv, jbp, jbv, dloc))
    return maps


def coord_update(c, h, p, entry, cur_max, skip_tol=1e-7):
    lo, hi, jfp, jfv, jbp, jbv, dloc = entry
    Llen = hi - lo + 1
    c_aff = c[lo:hi + 1]
    aff_max = float(c_aff.max())
    if skip_tol > 0.0 and aff_max < cur_max - skip_tol:
        return None

    a = np.zeros(Llen, dtype=float)
    if jfp.size:
        a[jfp] += 1.0 - h[jfv]
    if jbp.size:
        a[jbp] -= h[jbv]

    h0 = h[p]
    if dloc >= 0:
        a[dloc] = 0.0
        old = h0 * (1.0 - h0)
        di_base = float(c_aff[dloc] - old)
    else:
        old = 0.0
        di_base = 0.0

    unaff = 0.0
    if lo > 0:
        v = float(c[:lo].max())
        if v > unaff:
            unaff = v
    rest = c.size - hi - 1

        v = float(c[hi + 1:].max())

            unaff = v

    zero_max = float(c_aff[zero].max()) if zero.any() else 0.0
    Llo = unaff

    if Lhi <= Llo + 1e-13:

    pos_a = a[pos]
    pos_b = base[pos]
    neg_a = a[neg]
    neg_b = base[neg]

    for _ in range(32):
        mid = 0.5 * (Llo + Lhi)
        ok = True
        xlo = 0.0

        if pos_a.size:

            elif lb > xlo:

        if ok and xlo > xhi + 1e-12:
            ok = False

        if ok and dloc >= 0:
            Q = mid - di_base
            if Q < 0.0:
                if Q < -1e-11:
                    ok = False
                else:
                    Q = 0.0
            if ok and Q < 0.25 - 1e-12:
                disc = 1.0 - 4.0 * Q
                if disc < 0.0:
                    disc = 0.0
                s = math.sqrt(disc)
                r1 = 0.5 * (1.0 - s)
                r2 = 0.5 * (1.0 + s)
                if r1 < 0.0:
                    r1 = 0.0
                if r2 > 1.0:
                    r2 = 1.0
                seg1 = xlo <= min(xhi, r1) + 1e-12
                seg2 = max(xlo, r2) <= xhi + 1e-12
                if not (seg1 or seg2):
                    ok = False

        if ok:
            Lhi = mid
        else:
            Llo = mid

    Lsol = Lhi
    if Lsol >= Lhi - 1e-10:
        return None

    Lval = Lsol + 1e-10
    xlo = 0.0
    xhi = 1.0
    if pos_a.size:
        ub = float(np.min((Lval - pos_b) / pos_a))
        if ub < 0.0:
            ub = 0.0
        if ub < xhi:
            xhi = ub
    if neg_a.size:
        lb = float(np.max((Lval - neg_b) / neg_a))
        if lb > 1.0:
            lb = 1.0
        if lb > xlo:
            xlo = lb
    if xlo > xhi:
        midx = 0.5 * (xlo + xhi)
        if 0.0 <= midx <= 1.0:
            xlo = xhi = midx
        else:
            return None

    segs = []
    if dloc >= 0:
        Qv = Lval - di_base
        if Qv < 0.0:
            if Qv < -1e-10:
                return None
            Qv = 0.0
        if Qv < 0.25 - 1e-12:
            disc = 1.0 - 4.0 * Qv
            if disc < 0.0:
                disc = 0.0
            s = math.sqrt(disc)
            r1 = 0.5 * (1.0 - s)
            r2 = 0.5 * (1.0 + s)
            if r1 < 0.0:
                r1 = 0.0
            if r2 > 1.0:
                r2 = 1.0
            if xlo <= min(xhi, r1) + 1e-12:

            if max(xlo, r2) <= xhi + 1e-12:
                segs.append((max(xlo, r2), xhi))
        if not segs:
            segs.append((xlo, xhi))
    else:
        segs.append((xlo, xhi))

    best_x = None
    best_pred = float("inf")
    best_dist = float("inf")

    for sl, sh in segs:
        if sh < sl - 1e-12:
            continue
        sl = max(0.0, min(1.0, sl))
        sh = max(0.0, min(1.0, sh))

        for xc in (sl, sh, xc_mid):
            xc = float(np.clip(xc, 0.0, 1.0))
            val = unaff
            if zero_max > val:
                val = zero_max
            if pos_a.size:
                pv = float(np.max(pos_b + pos_a * xc))
                if pv > val:
                    val = pv
            if neg_a.size:
                nv = float(np.max(neg_b + neg_a * xc))
                if nv > val:
                    val = nv
            if dloc >= 0:
                dv = di_base + xc * (1.0 - xc)
                if dv > val:
                    val = dv
            dist = abs(xc - h0)
            if val < best_pred - 1e-14 or (abs(val - best_pred) <= 1e-14 and dist < best_dist):
                best_pred = val
                best_x = xc
                best_dist = dist

    if best_x is None or best_pred >= Lhi - 1e-11:
        return None

    if best_x < 0.0:
        best_x = 0.0
    elif best_x > 1.0:
        best_x = 1.0

    delta = best_x - h0
    if abs(delta) < 1e-14:
        return None

    c[lo:hi + 1] += delta * a
    if dloc >= 0:
        c[lo + dloc] += best_x * (1.0 - best_x) - old
    h[p] = best_x
    return best_pred


def exact_cd(h, n, deadline, order="length", proj_every=8, skip_tol=1e-7, cd_seed=123987):
    h = project_feasible(h, n)
    c = full_corr(h)
    best_c = float(c.max())
    best_h = h.copy()

    maps = build_maps(n)
    lens = np.empty(n, dtype=int)
    for i, ent in enumerate(maps):
        lens[i] = ent[1] - ent[0] + 1
    perm = np.argsort(-lens, kind="mergesort")
    seq_idx = np.arange(n)
    rng = np.random.RandomState(cd_seed + n)

    pass_num = 0
    stall = 0

    while time.time() < deadline:
        if order == "random":
            idx = rng.permutation(n)
        elif order == "seq":
            idx = seq_idx
        else:
            idx = np.roll(perm, pass_num % n)

        improved_any = False
        count = 0
        cur_max = float(c.max())

        for p in idx:
            if time.time() > deadline:
                break
            res = coord_update(c, h, p, maps[p], cur_max, skip_tol)
            if res is not None:
                improved_any = True
                if res > cur_max:
                    cur_max = res
            count += 1
            if count % proj_every == 0:
                h = project_feasible(h, n)
                c = full_corr(h)
                cur_max = float(c.max())
                if cur_max < best_c - 1e-12:
                    best_c = cur_max
                    best_h = h.copy()
                if time.time() > deadline:
                    break

        if time.time() > deadline:
            break

        h = project_feasible(h, n)
        c = full_corr(h)
        cur_max = float(c.max())

        if cur_max < best_c - 1e-10:
            best_c = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if cur_max > best_c + 1e-7:
                h = best_h.copy()
                c = full_corr(h)
                cur_max = best_c
            if stall >= 5:
                break

        pass_num += 1

    return best_h, best_c


def interp_start(src, n):
    m = len(src)
    if m == n:
        return project_feasible(src, n)
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(xn, xo, src), n)

            if cr < best_c:
                best_h, best_c = hr, cr

        start = best_h + rng.normal(0.0, 0.022, base_n)
        start = project_feasible(start, base_n)
        d = min(t0 + 0.51 * B, deadline)
        if time.time() < d:
            hr, cr = exact_cd(start, base_n, d, order="random", proj_every=10, skip_tol=1e-6, cd_seed=9002)
            if cr < best_c:
                best_h, best_c = hr, cr

        cands = []
        for factor in (1.2, 1.4):
            k = int(round(len(best_h) * factor))
            k = max(k, 60)
            k = min(k, 980)
            if k not in cands:
                cands.append(k)

        for nt in cands[:2]:
            rem = deadline - time.time()
            if rem < 70.0:
                break
            d = time.time() + min(max(90.0, 0.18 * rem), 320.0)
            d = min(d, deadline)
            hint = interp_start(best_h, nt)
            ho, co = exact_cd(hint, nt, d, order="length", proj_every=8, skip_tol=1e-7, cd_seed=10000 + nt)
            if co < best_c:
                best_h, best_c = ho, co

        if time.time() < deadline - 5.0:
            hf, cf = exact_cd(best_h, len(best_h), deadline, order="seq", proj_every=6, skip_tol=1e-8, cd_seed=11001)
            if cf < best_c:
                best_h, best_c = hf, cf
    except Exception:
        pass

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
