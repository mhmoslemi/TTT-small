def multi_cd(h, n, max_passes, deadline, top_k):

    """Subgradient method with Nesterov momentum and adaptive learning rate."""
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = 0.7  # momentum
    lr = 0.01  # learning rate
    beta = 50.0  # smoothing parameter
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline: break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c = c.max() * dx
        
        if cur_c < best_c:
            best_h = h.copy()
            best_c = cur_c
        
        # Compute softmax weights
        s = beta * c
        s_max = s.max()
        w = np.exp(s - s_max)
        w /= w.sum()
        
        # Compute gradient: grad[p] = sum_m w[m] * ((1-h[p+m]) - h[p-m])
        # where m = k_idx - (n-1)
        # Vectorize: forward[p] = sum_m w[m] * h[p+m] (h zero outside [0,n))
        #           backward[p] = sum_m w[m] * h[p-m]
        
        # Build extended w for correlation
        # w has length 2n-1, index 0..2n-2, corresponding to m = -(n-1)..(n-1)
        # forward[p] = sum_{m} w[m+n-1] * h[p+m] for 0<=p+m<n
        # This is a cross-correlation
        
        # Use direct loop for clarity (O(n^2) but n is at most 600)
        grad = np.zeros(n)
        for k_idx in range(2*n-1):
            m = k_idx - (n-1)
            wk = w[k_idx]
            if abs(wk) < 1e-15: continue
            # forward term: (1-h[p+m]) for p where 0<=p+m<n
            p_range = np.arange(max(0, -m), min(n, n-m))
            if len(p_range) > 0:
                grad[p_range] += wk * (1.0 - h[p_range + m])
            # backward term: -h[p-m] for p where 0<=p-m<n
            p_range2 = np.arange(max(0, m), min(n, n+m))
            if len(p_range2) > 0:
                grad[p_range2] -= wk * h[p_range2 - m]
        
        # Nesterov momentum update
        v = mu * v - lr * grad * dx
        h_new = project_feasible(h + v, n)
        
        # Adaptive step: if objective increased, halve lr
        c_new = compute_c5(h_new, n)
        if c_new > best_c * 1.01 and it > 10:
            lr *= 0.5
            h_new = h  # revert
        else:
            h = h_new
            # Occasionally increase beta
            if it % 200 == 0 and beta < 5000:
                beta *= 1.5
        
        # Restart momentum periodically
        if it % 500 == 0 and it > 0:
            v *= 0.5
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
