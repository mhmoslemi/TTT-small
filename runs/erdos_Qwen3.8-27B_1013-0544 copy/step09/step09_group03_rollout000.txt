def projected_grad_descent(h, n, deadline, seed=0):

    h = h.copy()

    alpha = 100.0  # smoothness parameter (increase for sharper max approximation)
    step = 0.1
    for it in range(200):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(h, n, alpha)
        # Projected gradient step
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            step *= 0.7
            if step < 1e-6:
                break
    
    return h, best_c
