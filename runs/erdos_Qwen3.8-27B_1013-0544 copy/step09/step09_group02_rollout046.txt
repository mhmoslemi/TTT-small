def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0

        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break

    h = h.copy()

    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False

                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]

                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    obj = (m + mx / alpha + np.log(exp_s.sum()) / alpha) * (2.0 / n)
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        wm = w[m_idx]
        # Forward: p + m_idx - n + 1 in [0, n-1] => p in [n-1-m_idx, m_idx]
        p_lo = max(0, n - 1 - m_idx)
        p_hi = min(n - 1, m_idx)
        if p_lo <= p_hi:
            j_vals = p_lo + m_idx - n + 1  # j for lowest p
            # j ranges from j_vals to j_vals + (p_hi - p_lo)
            j_end = p_hi + m_idx - n + 1
            grad[p_lo:p_hi+1] += wm * (1.0 - h[j_vals:j_end+1])
        
        # Backward: p - m_idx + n - 1 in [0, n-1] => p in [m_idx, m_idx + n - 1]
        p_lo2 = max(0, m_idx)
        p_hi2 = min(n - 1, m_idx + n - 1)
        if p_lo2 <= p_hi2:
            j_vals2 = p_lo2 - m_idx + n - 1
            j_end2 = p_hi2 - m_idx + n - 1
            grad[p_lo2:p_hi2+1] -= wm * h[j_vals2:j_end2+1]
    
    return obj, grad * (2.0 / n)

import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=500, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True

    return h, compute_c5(h, n)

def smooth_obj_and_grad(x, n, alpha, pen_lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    obj = (m + mx / alpha + np.log(exp_s.sum()) / alpha) * (2.0 / n)
    s = x.sum() - n / 2.0
    obj += pen_lam * s * s
    grad = np.zeros(n)
    for mk in range(2 * n - 1):
        wm = w[mk]
        if wm < 1e-14:
            continue
        plo = max(0, n - 1 - mk)
        phi = min(n - 1, mk)
        if plo <= phi:
            j0 = plo + mk - n + 1
            j1 = phi + mk - n + 1
            grad[plo:phi+1] += wm * (1.0 - x[j0:j1+1])
        plo2 = max(0, mk)
        phi2 = min(n - 1, mk + n - 1)
        if plo2 <= phi2:
            j0 = plo2 - mk + n - 1
            j1 = phi2 - mk + n - 1
            grad[plo2:phi2+1] -= wm * x[j0:j1+1]
    grad *= (2.0 / n)
    grad += 2.0 * pen_lam * s
    return obj, grad

def lbfgsb_optimize(h, n, alphas, maxiter_each, deadline, pen_lam=100.0):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_and_grad, best_h, args=(n, alpha, pen_lam),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_polish(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    stall = 0
    for _ in range(max_iter):

        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        jf = np.arange(n) + k_star - (n - 1)
        vb = (jf >= 0) & (jf < n)
        sg[vb] += (1.0 - h[jf[vb]])
        ip = np.arange(n) - k_star + (n - 1)
        va = (ip >= 0) & (ip < n)

        h_new = project_feasible(h - lr * sg, n)

            stall = 0
