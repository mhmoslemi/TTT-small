def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse/alpha) * (2.0/n)

    return x.sum() - n/2.0

    return np.ones(n)

def slsqp_quick(h, n, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [1000, 5000]:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h = np.clip(h + diff/tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h = np.clip(h + diff/tot * h, 0.0, 1.0)
    return h

    jb = p - ms + (n-1)

    vb = (jb >= 0) & (jb < n)
