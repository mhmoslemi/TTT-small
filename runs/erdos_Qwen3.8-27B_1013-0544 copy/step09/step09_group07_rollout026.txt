And c_aff_k + a_k * d (for lags where h[j] appears linearly) is linear in d.

The max of concave functions and linear functions is... not necessarily convex. So the 1D problem might have multiple local minima.

The previous code handles this by considering all pairwise intersections of the quadratics/linear functions, which gives the exact minimum of the max. This is correct but expensive when there are many lags.

2. Use a "target" approach: fix a target T, and try to find h such that all c[k]*dx ≤ T. Use projected gradient or regularized optimization.

**"Spectral" coordinate descent**: Instead of optimizing one variable at a time, optimize a small block of variables (e.g., 3-5 consecutive variables) simultaneously. This allows for better local adjustments.

Or even better:

**"Lag-space" optimization**: Instead of varying h, vary the target for each lag. Start with a uniform target T, then iteratively:
1. For each lag k, compute the "excess" e_k = c[k]*dx - T
2. For lags with e_k > 0, increase T (or decrease the target for that lag)
3. Adjust h to reduce the excess at the worst lags

This is similar to the "dual" approach in optimization.

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_full(c_aff, a_coeff, has_quad, unaff_max, lo, hi):
    """Solve min_d max_k (c_aff[k] + a_coeff[k]*d - has_quad[k]*d^2) subject to lo <= d <= hi, also >= unaff_max"""
    m = len(a_coeff)
    if m == 0:
        return 0.0
    
    def obj(d):
        vals = c_aff + a_coeff * d
        vals[has_quad] -= d * d
        return max(vals.max(), unaff_max)
    
    # Candidate points: intersections of pairs
    candidates = [lo, hi, 0.0]
    for i in range(m):
        for j in range(i+1, m):
            # c_aff[i] + a[i]*d - has_quad[i]*d^2 = c_aff[j] + a[j]*d - has_quad[j]*d^2
            A = a_coeff[i] - a_coeff[j]
            B = c_aff[j] - c_aff[i]
            C = has_quad[i] - has_quad[j]  # coefficient of d^2: -C*d^2
            # C*d^2 + A*d + B = 0
            if abs(C) < 1e-14:
                if abs(A) > 1e-14:
                    d = -B / A
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = A*A - 4*C*B
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-A + sq)/(2*C), (-A - sq)/(2*C)]:
                        if lo < d < hi:
                            candidates.append(d)
    
    # Also add where each function equals unaff_max
    # c_aff[i] + a[i]*d - has_quad[i]*d^2 = unaff_max
    # has_quad[i]*d^2 - a[i]*d + (unaff_max - c_aff[i]) = 0
    for i in range(m):
        if has_quad[i]:
            A2 = 1.0  # has_quad[i]
            B2 = -a_coeff[i]
            C2 = unaff_max - c_aff[i]
            disc = B2*B2 - 4*A2*C2
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B2+sq)/(2*A2), (-B2-sq)/(2*A2)]:
                    if lo < d < hi:
                        candidates.append(d)
        else:
            if abs(a_coeff[i]) > 1e-14:
                d = (unaff_max - c_aff[i]) / a_coeff[i]
                if lo < d < hi:
                    candidates.append(d)
    
    # Evaluate at all candidates
    best_d = lo
    best_val = obj(lo)
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = obj(d)
        if v < best_val:
            best_val = v
            best_d = d
    
    # Golden section refinement
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d - (lo+hi)/2) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj, gl, gh, maxiter=100)
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    
    return best_d

    vals = c_aff + a_coeff * d
    if np.any(has_quad):
        vals = vals.copy()
        vals[has_quad] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d(c_aff, a_coeff, has_quad, unaff_max, lo, hi):
    m = len(a_coeff)
    if m == 0:
        return 0.0
    
    candidates = [lo, hi]
    if lo < 0 < hi:
        candidates.append(0.0)
    
    # Pairwise intersections
    idx = np.where(has_quad)[0]
    lin = np.where(~has_quad)[0]
    
    # Quad-quad
    for i_pos in range(len(idx)):
        for j_pos in range(i_pos+1, len(idx)):
            i, j = idx[i_pos], idx[j_pos]
            A = a_coeff[i] - a_coeff[j]
            B = c_aff[j] - c_aff[i]
            # Both have -d^2, so they cancel: A*d + B = 0
            if abs(A) > 1e-14:
                d = -B / A
                if lo < d < hi:
                    candidates.append(d)
    
    # Quad-linear
    for i_pos in range(len(idx)):
        for j_pos in range(len(lin)):
            i, j = idx[i_pos], lin[j_pos]
            # c_i + a_i*d - d^2 = c_j + a_j*d
            # -d^2 + (a_i-a_j)*d + (c_i-c_j) = 0
            # d^2 - (a_i-a_j)*d + (c_j-c_i) = 0

            A = a_coeff[i] - a_coeff[j]

                d = (c_aff[j] - c_aff[i]) / A
                if lo < d < hi:
                    candidates.append(d)

            #   v*(1-h[idx-s]) + h[idx+s]*(1-v) = v - v*h[idx-s] + h[idx+s] - h[idx+s]*v
            #   = h[idx+s] + v*(1 - h[idx-s] - h[idx+s])
            #   Linear in v.
            
            # For s>0, s > min(idx, n-1-idx) but s <= max(idx, n-1-idx):
            #   Only one term. Either h[idx]*(1-h[idx-s]) [if s <= idx] or h[idx+s]*(1-h[idx]) [if s <= n-1-idx].
            #   Same as above, just one term. Linear in h[idx].
            
            # So indeed, only the diagonal (s=0, full index n-1) has a quadratic term!
            
            # This simplifies things enormously. The 1D subproblem is:
            # min_d max_k (A_k + B_k * d) for k ≠ n-1, and (A_diag + B_diag * d - d²) for k = n-1
            # subject to 0 <= h0 + d <= 1
            
            # This is the max of one concave quadratic and many linear functions.
            # The minimum of this max can be found by checking:
            # 1. Where the quadratic equals each linear (at most 2 solutions per linear)
            # 2. Where the quadratic is at its vertex (if it's below all linears there)
            # 3. The boundaries d = -h0 and d = 1-h0
            
            # Much simpler than I thought!
            
            # Let me compute the linear coefficients for each relevant lag.
            
            # For forward s in [1, min(idx, n-1)]:
            #   c[n-1+s] gets contribution h[idx]*(1-h[idx-s]) if s <= idx
            #   So B = 1 - h[idx-s], A includes the rest
            # For forward s in [1, n-1-idx]:
            #   c[n-1+s] gets contribution h[idx+s]*(1-h[idx]) if s <= n-1-idx
            #   So B = -h[idx+s], A includes the rest
            # For backward s' in [1, idx]:
            #   c[n-1-s'] gets contribution h[idx]*(1-h[idx-s']) if s' <= idx
            #   So B = 1 - h[idx-s'], A includes the rest
            # For backward s' in [1, n-1-idx]:
            #   c[n-1-s'] gets contribution h[idx+s']*(1-h[idx]) if s' <= n-1-idx
            #   So B = -h[idx+s'], A includes the rest
            
            # Note: for s = s' (forward and backward with same magnitude):
            #   c[n-1+s] uses h[i+s](1-h[i])
            #   c[n-1-s] uses h[i](1-h[i-s])
            #   These are different lags!
            
            # Let me just compute the coefficients directly.
            
            # For each full correlation index f (0 to 2n-2), determine if h[idx] appears and with what coefficient.
            # f = n-1+s where s is the shift.
            # If s >= 0: c[f] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            #   h[idx] appears as h[i+s] when i = idx-s (valid if 0<=idx-s and idx-s<=n-1-s, i.e., s<=idx)
            #   h[idx] appears as (1-h[i]) when i = idx (valid if 0<=idx<=n-1-s, i.e., s<=n-1-idx)
            # If s < 0 (s' = -s > 0): c[f] = sum_{i=s'}^{n-1} h[i]*(1-h[i-s'])
            #   h[idx] appears as h[i] when i = idx (valid if s'<=idx)
            #   h[idx] appears as (1-h[i-s']) when i-s'=idx, i=idx+s' (valid if s'<=idx+s'<=n-1, i.e., 0<=idx<=n-1-s')
            
            # For the coefficient of d (where h[idx] = h0 + d):
            # If h[idx] appears as h[i+s] (i.e., in the "first factor"):
            #   Term: (h0+d)*(1-h[other]) => coefficient of d is (1-h[other])
            # If h[idx] appears as (1-h[i]) (i.e., in the "second factor"):
            #   Term: h[other]*(1-(h0+d)) = h[other] - h[other]*(h0+d) => coefficient of d is -h[other]
            # If both (s=0 or s where both conditions hold):
            #   Total coefficient: (1-h[other1]) + (-h[other2])
            
            # For s=0 (f = n-1):
            #   h[idx] appears as h[i] AND (1-h[i]), giving h[idx]*(1-h[idx])
            #   = (h0+d)*(1-h0-d) = h0 - h0² + d - d² - h0*d - h0*d
            #   = (h0 - h0²) + d*(1-2*h0) - d²
            #   So A_diag = c[n-1] - h0*(1-h0) [the rest of the sum]
            #      B_diag = 1 - 2*h0
            #      has_quad = True (coefficient of d² is -1)
            #
            #   Wait, I need to be more careful. c[n-1] = sum_i h[i](1-h[i])
            #   The term for i=idx is h[idx](1-h[idx]) = (h0+d)(1-h0-d)
            #   = h0(1-h0) + d(1-h0) - h0*d - d²
            #   = h0(1-h0) + d(1-2*h0) - d²
            #   So the contribution to c[n-1] from varying h[idx] is:
            #     constant: h0(1-h0) [but this is already in c[n-1] as the current value]
            #     linear: d*(1-2*h0)
            #     quadratic: -d²
            #   So: A_diag = c[n-1] - h0*(1-h0), B_diag = 1-2*h0, has_quad_diag = True
            
            # For s > 0, s <= idx: h[idx] appears as h[i+s] with i = idx-s
            #   Term: h[idx]*(1-h[idx-s]) = (h0+d)*(1-h[idx-s])
            #   Linear: d*(1-h[idx-s]), no quadratic
            #   A = c[n-1+s] - h0*(1-h[idx-s)], B = 1-h[idx-s]
            
            # For s > 0, s <= n-1-idx: h[idx] appears as (1-h[i]) with i = idx
            #   Term: h[idx+s]*(1-h[idx]) = h[idx+s]*(1-h0-d)
            #   Linear: d*(-h[idx+s]), no quadratic
            #   A = c[n-1+s] - h[idx+s]*(1-h0), B = -h[idx+s]
            
            # For s > 0 where both conditions hold (s <= min(idx, n-1-idx)):
            #   Two terms: h[idx]*(1-h[idx-s]) + h[idx+s]*(1-h[idx])
            #   A = c[n-1+s] - h0*(1-h[idx-s]) - h[idx+s]*(1-h0)
            #   B = (1-h[idx-s]) + (-h[idx+s]) = 1 - h[idx-s] - h[idx+s]
            
            # For s < 0 (s' = -s > 0), s' <= idx: h[idx] appears as h[i] with i = idx
            #   Term: h[idx]*(1-h[idx-s']) = (h0+d)*(1-h[idx-s'])
            #   B = 1-h[idx-s'], A = c[n-1-s'] - h0*(1-h[idx-s'])
            
            # For s < 0, s' <= n-1-idx: h[idx] appears as (1-h[i-s']) with i = idx+s'
            #   Term: h[idx+s']*(1-h[idx]) = h[idx+s']*(1-h0-d)
            #   B = -h[idx+s'], A = c[n-1-s'] - h[idx+s']*(1-h0)
            
            # For both (s' <= min(idx, n-1-idx)):
            #   B = (1-h[idx-s']) + (-h[idx+s']) = 1 - h[idx-s'] - h[idx+s']
            #   A = c[n-1-s'] - h0*(1-h[idx-s']) - h[idx+s']*(1-h0)
            
            # Now I realize that for forward s and backward s' with s = s':
            #   Forward: h[idx] appears in c[n-1+s] 
            #   Backward: h[idx] appears in c[n-1-s]
            #   These are DIFFERENT lags (different full indices), so they don't combine.
            #   Unless s = 0, in which case both refer to c[n-1].
            
            # Great, so the algorithm is:
            # 1. For each full index f from 0 to 2n-2, determine if h[idx] appears
            # 2. If yes, compute A_f and B_f
            # 3. Solve min_d max_f (A_f + B_f * d) with the quadratic for f = n-1
            
            # The set of f where h[idx] appears:
            # f = n-1 (always, since s=0 always has h[idx])
            # f = n-1+s for s in [1, idx] (forward, h[idx] as h[i+s])
            # f = n-1+s for s in [1, n-1-idx] (forward, h[idx] as 1-h[i])
            # f = n-1-s for s in [1, idx] (backward, h[idx] as h[i])
            # f = n-1-s for s in [1, n-1-idx] (backward, h[idx] as 1-h[i])
            #
            # So f ranges from n-1-max(idx, n-1-idx) to n-1+max(idx, n-1-idx)
            # Which is from min(idx, n-1-idx) to 2*(n-1)-min(idx, n-1-idx)
            # Wait: n-1-max(idx,n-1-idx) = n-1-(n-1-min(idx,n-1-idx)) = min(idx,n-1-idx)
            # And n-1+max(idx,n-1-idx) = n-1+(n-1-min(idx,n-1-idx)) = 2(n-1)-min(idx,n-1-idx)
            #
            # So the range of f is [min(idx, n-1-idx), 2(n-1)-min(idx, n-1-idx)]
            # The number of lags is 2*max(idx, n-1-idx) + 1
            # For idx near n/2, this is about n lags.
            
            # This is a lot of lags. For the "top_k" approach, I'll select the top_k lags by current value.
            
            # Let me just compute the coefficient for each relevant lag.
            
            s_max = max(idx, n - 1 - idx)
            s_min = min(idx, n - 1 - idx)
            f_lo = s_min  # = n-1-s_max
            f_hi = 2 * (n - 1) - s_min  # = n-1+s_max
            
            f_all = np.arange(f_lo, f_hi + 1)
            
            # Build coefficient arrays
            # For each f in f_all, compute A and B
            s = f_all - (n - 1)  # shift: negative for backward, 0 for diag, positive for forward
            
            A = np.zeros(len(f_all))
            B = np.zeros(len(f_all))
            has_quad = np.zeros(len(f_all), dtype=bool)
            
            # Diagonal: f = n-1, s = 0
            A[n-1 - f_lo] = c[n-1] - h0 * (1 - h0)
            B[n-1 - f_lo] = 1.0 - 2.0 * h0
            has_quad[n-1 - f_lo] = True
            
            # Forward s > 0: f = n-1+s, s from 1 to s_max
            #   h[idx] as h[i+s] (i=idx-s) if s <= idx: B += 1-h[idx-s]
            #   h[idx] as 1-h[i] (i=idx) if s <= n-1-idx: B += -h[idx+s]
            for s_val in range(1, s_max + 1):
                f_idx = n - 1 + s_val - f_lo
                a_val = 0.0
                b_val = 0.0
                if s_val <= idx:
                    a_val += h0 * (1 - h[idx - s_val])
                    b_val += (1 - h[idx - s_val])
                if s_val <= n - 1 - idx:
                    a_val += h[idx + s_val] * (1 - h0)
                    b_val += (-h[idx + s_val])
                A[f_idx] = c[n - 1 + s_val] - a_val
                B[f_idx] = b_val
            
            # Backward s > 0: f = n-1-s, s from 1 to s_max
            #   h[idx] as h[i] (i=idx) if s <= idx: B += 1-h[idx-s]
            #   h[idx] as 1-h[i-s] (i=idx+s) if s <= n-1-idx: B += -h[idx+s]
            for s_val in range(1, s_max + 1):
                f_idx = n - 1 - s_val - f_lo
                a_val = 0.0
                b_val = 0.0
                if s_val <= idx:
                    a_val += h0 * (1 - h[idx - s_val])
                    b_val += (1 - h[idx - s_val])
                if s_val <= n - 1 - idx:
                    a_val += h[idx + s_val]
