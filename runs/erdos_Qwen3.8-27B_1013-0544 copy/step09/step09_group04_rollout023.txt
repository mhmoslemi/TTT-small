def subgradient_run(h, n, deadline, max_iters=200):

        cur_c = float(np.max(c) * (2.0 / n))
        if cur_c < best_c:
            best_h = h.copy()
            best_c = cur_c
            patience = 0
        else:
            patience += 1
            if patience > 20:
                h = best_h.copy()
                lr = max(0.01, lr * 0.5)
                patience = 0
        m_idx = int(np.argmax(c))
        s = m_idx - (n - 1)
        j = np.arange(n)
        g = np.zeros(n)
        valid1 = (j - s >= 0) & (j - s < n)
        valid2 = (j + s >= 0) & (j + s < n)
        if np.any(valid1):
            g[valid1] += (1.0 - h[j[valid1] - s])
        if np.any(valid2):
            g[valid2] += (-h[j[valid2] + s])
        h_new = project_feasible(h - lr * g, n)
        h = h_new
        lr = max(0.001, lr * 0.99)
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
