def lse_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    log_sum = mx + np.log(np.sum(np.exp(shifted - mx)))
    obj = (m + log_sum / alpha) * (2.0 / n)
    
    # Vectorized gradient computation
    # For each p, grad[p] = (2/n) * sum_k weights[k] * [(1-x[p+k-(n-1)]) - x[p-k+(n-1)]]
    # Let's define: for lag k (0-indexed in full correlation), the offset is k - (n-1)
    # dc[k]/dh[p] = (1 - x[p + (k-(n-1))]) * 1[0<=p+(k-(n-1))<n] 
    #              - x[p - (k-(n-1))] * 1[0<=p-(k-(n-1))<n]
    
    # This is: weights @ A @ (1-x) - weights @ B @ x
    # where A[p,k] = 1 if 0<=p+k-(n-1)<n, B[p,k] = 1 if 0<=p-k+(n-1)<n
    # But A and B are (n x (2n-1)) matrices, which is fine for n=600.
    
    # Actually, let's just do it with a loop over k (which is 2n-1 = 1199 iterations):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        # Forward: p + offset in [0, n)
        if offset >= 0:
            p_lo, p_hi = 0, n - offset - 1
        else:
            p_lo, p_hi = -offset, n - 1
        if p_lo <= p_hi:
            jf = np.arange(p_lo, p_hi + 1) + offset
            grad[p_lo:p_hi+1] += w * (1.0 - x[jf])
        # Backward: p - offset in [0, n)
        if offset <= 0:
            p_lo2, p_hi2 = 0, n + offset - 1
        else:
            p_lo2, p_hi2 = offset, n - 1
        if p_lo2 <= p_hi2:
            jb = np.arange(p_lo2, p_hi2 + 1) - offset
            grad[p_lo2:p_hi2+1] -= w * x[jb]
    grad *= (2.0 / n)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = -np.inf
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def run_cd(h, n, max_passes=300, deadline=None, rng=None, shuffle=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_constraint(x, n):
    return x.sum() - n / 2.0

def lse_constraint_jac(x, n):
    return np.ones(n)

def slsqp_lse(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lse_constraint, 'jac': lse_constraint_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def bang_bang_refine(h, n, deadline=None):
    """Try snapping near-0/1 values and see if CD improves."""
    best_h, best_c = h.copy(), compute_c5(h, n)
    for thresh in [0.05, 0.1, 0.15, 0.2]:
        if deadline is not None and time.time() > deadline:
            break
        h2 = h.copy()
        h2[h2 < thresh] = 0.0
        h2[h2 > 1.0 - thresh] = 1.0
        h2 = project_feasible(h2, n)
        h2, c2 = run_cd(h2, n, max_passes=100, deadline=deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    return best_h, best_c

def structured_init(n, rng):
    """Generate a structured initial solution."""
    pattern = rng.randint(0, 4)
    if pattern == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern == 1:
        h = np.zeros(n)
        idx = np.arange(0, n, 2)
        h[idx] = 1.0
        h = np.clip(h * (n * 0.5) / max(h.sum(), 1), 0, 1)
    elif pattern == 2:
        period = max(4, n // 20)
        duty = period // 2
        h = np.zeros(n)
        for i in range(n):
            if i % period < duty:
                h[i] = 1.0
        h = project_feasible(h, n)
    else:
        h = np.full(n, 0.5) + rng.normal(0, 0.1, n)
        h = project_feasible(h, n)
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    # Get initial solution

        if time.time() > deadline - (120 + n_try // 50):
