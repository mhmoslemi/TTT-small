def grad_at_lag(h, n, lag):
    """Gradient of f_lag(h) = sum_i h[i]*(1-h[i+shift])*dx where shift = lag-(n-1)"""
    dx = 2.0 / n
    shift = lag - (n - 1)
    grad = np.zeros(n)
    idx = np.arange(n)
    fwd = idx + shift
    v1 = (fwd >= 0) & (fwd < n)
    grad[v1] += (1.0 - h[fwd[v1]]) * dx
    bwd = idx - shift
    v2 = (bwd >= 0) & (bwd < n)
    grad[v2] -= h[bwd[v2]] * dx
    return grad
