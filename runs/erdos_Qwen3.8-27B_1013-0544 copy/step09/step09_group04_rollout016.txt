import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    # Project to {0 <= h <= 1, sum(h) = n/2}
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = n/2.0 - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.clip(h + diff/tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.clip(h + diff/tot * h, 0.0, 1.0)
    return h

def exact_cd_single(h, n, p, c, cur_max):
    # Exact 1D CD for coordinate p (same as before)
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    
    m_lo = max(0, n - 1 - p)
    m_hi = min(2*n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    
    c_base = c[ms].copy()
    di = np.where(ms == (n-1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    
    unaff = 0.0
    if ms[0] > 0: unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2*n-2: unaff = max(unaff, float(c[ms[-1]+1:].max()))
    
    # Binary search for optimal L
    L_lo = unaff
    L_hi = cur_max
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    
    for _ in range(50):
        if L_hi - L_lo < 1e-16: break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0*(1.0 - h0)
            disc = 1.0 - 4.0*Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0*(1.0-h0)
        disc = 1.0 - 4.0*Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                if (h0 - r1 if x_lo <= r1 else 1e9) < (r2 - h0 if x_hi >= r2 else 1e9):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def pair_cd(h, n, c, cur_max, i, j, n_iter=60):

    hj_old = h[j]
    s = hi_old + hj_old  # must maintain this sum

    d_lo = max(-hi_old, -(1.0 - hj_old))
    d_hi = min(1.0 - hi_old, hj_old)
    if d_hi - d_lo < 1e-14:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def exact_cd_pass(h, n, order='sequential', deadline=None):
    """One pass of exact coordinate descent."""
    n_idx = np.arange(n)
    if order == 'random':
        rng = np.random.RandomState(int(time.time() * 1000) % 10000)
        idx = n_idx.copy()
        rng.shuffle(idx)
    elif order == 'sensitivity':
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        sens = np.zeros(n)
        for p in range(n):
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            sens[p] = cur_max - unaff
        idx = n_idx[np.argsort(-sens)]
    else:
        idx = n_idx
    
    improved = False
    for p in idx:
        if deadline and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d(h, n, p)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
            improved = True
    return h, improved
