def smooth_max_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    nfull = 2 * n - 1
    
    for m_idx in range(nfull):
        if w[m_idx] < 1e-16:
            continue
        wi = w[m_idx]
        k = m_idx - (n - 1)
        # (1-h[j-k]) for j-k in [0,n): j in [k, n+k) ∩ [0,n)
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wi * one_minus_h[j_lo - k : j_hi - k]
        # -h[j+k] for j+k in [0,n): j in [-k, n-k) ∩ [0,n)
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wi * h[j_lo2 + k : j_hi2 + k]
    
    return grad, float(m * (2.0 / n))

import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation values."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    nfull = 2 * n - 1
    
    # Only process lags with significant weight
    threshold = 1e-12
    active = w > threshold
    if not np.any(active):
        # Uniform weights (all lags equal)
        for m_idx in range(nfull):
            k = m_idx - (n - 1)
            j_lo = max(0, k)
            j_hi = min(n, n + k)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += one_minus_h[j_lo - k:j_hi - k] / nfull
            j_lo2 = max(0, -k)
            j_hi2 = min(n, n - k)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= h[j_lo2 + k:j_hi2 + k] / nfull
        return grad, cmax * (2.0 / n)
    
    for m_idx in np.where(active)[0]:
        wi = w[m_idx]
        k = m_idx - (n - 1)
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wi * one_minus_h[j_lo - k:j_hi - k]
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wi * h[j_lo2 + k:j_hi2 + k]
    
    return grad, cmax * (2.0 / n)

    """Adam optimization with annealed smooth max."""
    h = project_feasible(h.copy(), n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stall = 0
    lr_cur = lr
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = t / max(1, max_iter)
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 2)
        
        grad, curr_val = get_gradient(h, n, alpha)
        curr_c = compute_c5(h, n)
        if curr_c < best_c - 1e-14:
            best_c = curr_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 50:
            lr_cur *= 0.5
            if lr_cur < 1e-8:
                break
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
    
    return best_h, best_c
