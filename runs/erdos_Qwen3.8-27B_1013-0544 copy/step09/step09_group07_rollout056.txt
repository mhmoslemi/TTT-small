def linearized_active_set(h, n, t_target, max_iter=100, deadline=None):
    """Solve min max_k c[k]*dx using linearized active set method."""
    dx = 2.0 / n
    h = h.copy()
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max()) * dx
        
        if c_max <= t_target + 1e-10:
            break
        
        # Identify active shifts (top K)
        K = min(50, 2*n-1)
        active_idx = np.argsort(c)[-K:]
        
        # Build LP: minimize t
        # s.t. c[k]*(h) <= t for k in active (linearized)
        #      sum(h) = n/2
        #      0 <= h[i] <= 1
        #
        # c[k](h) ≈ c[k](h₀) + Σ_i grad_k[i] * (h[i] - h₀[i])
        # where grad_k[i] = ∂c[k]/∂h[i] at h₀
        
        import cvxpy as cp
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        
        for k in active_idx:
            # Gradient of c[k] w.r.t. h[i]
            grad = np.zeros(n)
            shift = k - (n-1)
            # c[k] = Σ_i h[i] * (1-h[i+shift]) for valid indices
            # ∂c[k]/∂h[i] = (1-h[i+shift]) - h[i-shift] (for valid indices)
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    grad[i] += (1.0 - h[j])
                j2 = i - shift
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
            
            c_at_k = c[k]
            linearized = c_at_k + np.dot(grad, (h_var - h))
            # But h is a numpy array, so h_var - h gives a cvx expression
            constraints.append(linearized * dx <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
            if h_var.value is not None:
                new_h = project_feasible(h_var.value, n)
                new_c = compute_c5(new_h, n)
                if new_c < c_max - 1e-12:
                    h = new_h
                else:
                    break
        except Exception:
            break
    
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize, golden
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0*dd), (-da - sq) / (2.0*dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d)*0.5 + span*0.05)
        gl, gh = max(lo, best_d - margin), min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=100)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd_fast(h, n, max_passes=500, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2*n - 2, 2*n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2*n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms].copy()
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo, hi_ = -h0, 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi_)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def linearized_as(h, n, max_iter=50, deadline=None, k_active=40, trust_r=0.05):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        
        # Active shifts
        K = min(k_active, 2*n-1)
        active_idx = np.argsort(c)[-K:]
        
        # Build and solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        
        # Trust region (L2)
        constraints.append(cp.norm(h_var - h, 2) <= trust_r)
        
        for k_val in active_idx:
            shift = k_val - (n-1)
            grad = np.zeros(n)
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    grad[i] += (1.0 - h[j])
                j2 = i - shift
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
            
            c_at_k = c[k_val]
            # Linearized: c[k](h_new) ≈ c_at_k + grad @ (h_var - h)
            linearized = c_at_k + grad @ (h_var - h)
            constraints.append(linearized * dx <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
            if h_var.value is not None:
                new_h = np.clip(h_var.value, 0, 1)
                new_h = project_feasible(new_h, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-12:
                    h = new_h
                    best_c = new_c
                else:
                    # Try smaller trust region
                    trust_r *= 0.5
                    if trust_r < 1e-4:
                        break
        except Exception:
            break
    
    return h, best_c

def build_grad_matrix(h, n, active_idx):
    """Build matrix G where G[k_idx, i] = dc[k]/dh[i] for k in active_idx."""
    G = np.zeros((len(active_idx), n))

        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl, gh = max(lo, best_d - margin), min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=100)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd_fast(h, n, max_passes=500, deadline=None, top_k=30):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_lo = min(max(0, n-1-idx), max(0, idx))
            m_hi = max(min(2*n-2, 2*n-2-idx), min(2*n-2, idx+n-1))
            ms_all = np.arange(m_lo, m_hi+1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                unaff_extra = float(c[ms_all[[m not in ms_set for m in ms_all]]].max()) if len(ms_all) > top_k else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms].copy()
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, -h0, 1.0-h0)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def build_grad_matrix(h, n, active_idx):
    G = np.zeros((len(active_idx), n))
    for a, k_val in enumerate(active_idx):
        shift = k_val - (n - 1)
        if shift >= 0:
            G[a, :n-shift] += (1.0 - h[shift:])
            G[a, shift:] -= h[:n-shift]
        else:
            s = -shift
            G[a, s:] += (1.0 - h[:n-s])
            G[a, :n-s] -= h[s:]
    return G

def linearized_as(h, n, max_iter=30, deadline=None, k_active=40, trust_r=0.03):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        K = min(k_active, 2*n-1)
        active_idx = np.argsort(c)[-K:]
        G = build_grad_matrix(h, n, active_idx)
        c_active = c[active_idx]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        constraints.append(cp.norm(h_var - h, 2) <= trust_r)
        # (c_active + G @ (h_var - h)) * dx <= t_var
        constraints.append((c_active + G @ (h_var - h)) * dx <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000, reltol=1e-8, abstol=1e-10)
            if h_var.value is not None:
                new_h = project_feasible(h_var.value, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-12:
                    h = new_h
                    best_c = new_c
                    # Slightly shrink trust region
                    trust_r = max(trust_r * 0.8, 0.005)
                else:
                    trust_r *= 0.5
                    if trust_r < 1e-4:
                        break
        except Exception:
            break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
