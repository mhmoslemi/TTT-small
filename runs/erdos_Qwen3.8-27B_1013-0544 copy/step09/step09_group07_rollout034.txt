def weighted_gradient_fast(h, n, w):
    c1 = np.convolve(w, 1.0 - h, mode='full')[:n]

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    if lo >= hi:
        return 0.0
    k = len(a_coeff)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)

        for j in range(i + 1, k):

            dc = c_aff[j] - c_aff[i]
            dd = 0.0

                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:

                if res.fun < best_val - 1e-15:

                if c5b < best_c:
                    best_h, best_c = h5b, c5b

    # Phase 6: Try different n
    n_cur = len(best_h)
    x_orig = np.linspace(0, 2, n_cur)
    for n_try in [400, 500, 700, 800]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=100, deadline=deadline - 50, top_k=30)
        if cn < best_c:
            best_h, best_c = hn, cn
            n_cur = n_try
            x_orig = x_new

    # Phase 7: Perturbation restarts
    n_fin = len(best_h)
    for r in range(4):
        if time.time() > deadline - 40:
            break
        s = 0.005 + 0.01 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_fin), n_fin)
        ho, co = coord_cd(hp, n_fin, max_passes=80, deadline=deadline - 30, top_k=30)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 8: Final polish
    if time.time() < deadline - 20:
        h8, c8 = coord_cd(best_h, n_fin, max_passes=500, deadline=deadline - 10, top_k=50)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
