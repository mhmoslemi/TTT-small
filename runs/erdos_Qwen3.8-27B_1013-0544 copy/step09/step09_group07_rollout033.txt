    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    log_sum = np.log(np.sum(np.exp(shifted)))

    w = np.exp(shifted - log_sum)  # normalized weights, sum to 1

            m_f_hi = min(2*n-2, 2*n-2-idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2*n-2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff_extra = ms_all, 0.0
            unaff_max = unaff_extra
            if m_lo > 0: unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n-1):
                    diag_local = ii
                    break
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, -h0, 1.0-h0)
            if abs(best_d) > 1e-12:
                nv = c_aff + best_d * a_coeff
                if diag_local is not None:
                    nv[diag_local] -= best_d * best_d
                new_val = max(float(nv.max()), unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick coordinate descent
    if time.time() < deadline - 200:
        h1, c1 = coord_cd_fast(best_h, n, max_passes=150, deadline=deadline-180, top_k=25)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Fourier L-BFGS-B optimization
    for M in [30, 50, 80, 120]:
        if time.time() > deadline - 120:
            break
        for alpha in [200, 500, 1000]:
            if time.time() > deadline - 100:
                break
            for beta in [0.0, 50.0, 200.0]:
                if time.time() > deadline - 90:
                    break
                hf, cf = fourier_lbfhs(best_h, n, M, alpha, beta, deadline, maxiter=150)
                if cf < best_c:
                    best_h, best_c = hf, cf

    # Phase 3: Spectral coordinate descent
    if time.time() < deadline - 100:
        for M in [40, 80]:
            if time.time() > deadline - 80:
                break
            hs, cs = spectral_coord(best_h, n, M, 500, 100.0, deadline, max_passes=40)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 4: More coordinate descent
    if time.time() < deadline - 60:
        h4, c4 = coord_cd_fast(best_h, n, max_passes=400, deadline=deadline-50, top_k=35)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-restart in Fourier space
    if time.time() < deadline - 50:
        for r in range(4):
            if time.time() > deadline - 35:
                break
            M = 50 + 20 * r
            alpha = 300 + 200 * r
            nf, cf = fourier_lbfhs(best_h, n, M, alpha, 100.0, deadline, maxiter=100)
            if cf < best_c:
                best_h, best_c = nf, cf

    # Phase 6: Final polish
    if time.time() < deadline - 20:
        h6, c6 = coord_cd_fast(best_h, n, max_passes=500, deadline=deadline-10, top_k=40)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Try 1-h symmetry
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
