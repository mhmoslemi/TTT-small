    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    """Solve the 1D subproblem exactly."""
    k = len(a_coeff)
    if k == 0:
        return 0.0
    
    def eval_at(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        return max(float(nv.max()), unaff_max)
    
    # Find candidates from pairwise intersections
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: candidates.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2.0*dd), (-da-sq)/(2.0*dd)]:
                        if lo < d < hi: candidates.append(d)
    
    best_d = 0.0
    best_val = eval_at(0.0)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = eval_at(dc)
        if v < best_val:
            best_val = v
            best_d = dc
    
    # Golden section refinement
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(eval_at, gl, gh, maxiter=60)
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except: pass
    return best_d
