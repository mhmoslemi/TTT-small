import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):

            if time.time() > deadline:

def lse_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    exps = np.exp(s - mx)
    lse_val = (m + mx / alpha + np.log(exps.sum()) / alpha) * dx
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-18:
            continue
        jf = np.arange(n) + m_idx - (n - 1)
        jb = np.arange(n) - m_idx + (n - 1)
        af = np.zeros(n)
        vf = (jf >= 0) & (jf < n)
        af[vf] = 1.0 - x[jf[vf]]
        ab = np.zeros(n)
        vb = (jb >= 0) & (jb < n)
        ab[vb] = -x[jb[vb]]
        grad += w * (af + ab)
    grad *= dx
    return lse_val, grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    def obj(x):
        val, _ = lse_obj_and_grad(x, n, alpha)
        return val
    def jac(x):
        _, g = lse_obj_and_grad(x, n, alpha)
        return g
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)
