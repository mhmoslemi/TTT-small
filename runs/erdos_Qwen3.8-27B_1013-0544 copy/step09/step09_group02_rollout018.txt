f(h) = (max(c) + (1/α) * log(Σ exp(α(c_k - max(c))))) * dx

The gradient with respect to h[p]:
∂f/∂h[p] = dx * (1/α) * Σ_k [α * exp(α(c_k - m)) / Σ_j exp(α(c_j - m))] * ∂c_k/∂h[p]
= dx * Σ_k w_k * ∂c_k/∂h[p]

where w_k = exp(α(c_k - m)) / Σ_j exp(α(c_j - m)) and m = max(c).

And ∂c_k/∂h[p] is as computed in the CD code.

So the gradient is: dx * Σ_k w_k * vec_k(p)

where vec_k(p) is the derivative of correlation at lag k with respect to h[p].

    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 20 == 19:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(shifted)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        mlo = max(0, n - 1 - p)
        mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
        ms = np.arange(mlo, mhi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        vec = np.zeros(len(ms))
        if np.any(vf):
            vec[vf] += 1.0 - h[jf[vf]]
        if np.any(vb):
            vec[vb] -= h[jb[vb]]
        grad[p] = np.dot(w[mlo:mhi + 1], vec)
    return grad * dx

def lse_projected(h, n, alpha, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.08
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = lse_grad(h, n, alpha)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h, best_c = h_new, c_new
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 50 or lr < 1e-10:
                break
    return h, best_c

def autocorr_subgrad(h, n, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.04
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        ac = np.correlate(h, h, mode='full')
        k_star = int(np.argmin(ac[n - 1 - (n - 1):n - 1 + (n - 1) + 1])) if n > 1 else 0
        k_star = int(np.argmin(ac))
        n_ac = len(ac)
        shift = n - 1
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + k_star - shift
            j2 = i - k_star + shift
            val = 0.0
            if 0 <= j1 < n:
                val += h[j1]
            if 0 <= j2 < n:
                val += h[j2]
            grad[i] = val
        h_new = project_feasible(h + lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h, best_c = h_new, c_new
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-10:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        try:
            h_init = np.array(initial_h_values, dtype=float).copy()
            n_init = len(h_init)
        except Exception:
            h_init = np.full(400, 0.5)
            n_init = 400
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
        h1, c1 = coord_cd(best_h, n_init, 150, deadline, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
        x_cur = np.linspace(0, 2, len(best_h))
        for n_try in [700, 800, 1000, 1200]:
            if time.time() > deadline - 180:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
            h_opt, c_opt = coord_cd(h_int, n_try, 200, deadline, rng)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                x_cur = x_new
            if time.time() < deadline - 100:
                h_l, c_l = lse_projected(h_opt, n_try, 1500, 200, deadline)
                if c_l < best_c:
                    best_h, best_c = h_l, c_l
                    x_cur = x_new
                h_a, c_a = autocorr_subgrad(h_opt, n_try, 150, deadline)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
                    x_cur = x_new
        n_cur = len(best_h)
        h3, c3 = coord_cd(best_h, n_cur, 300, deadline, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
        if time.time() < deadline - 100:
            h4, c4 = lse_projected(best_h, n_cur, 3000, 300, deadline)
            if c4 < best_c:
                best_h, best_c = h4, c4
            h4b, c4b = coord_cd(best_h, n_cur, 200, deadline, rng)
            if c4b < best_c:
                best_h, best_c = h4b, c4b
            h4c, c4c = autocorr_subgrad(best_h, n_cur, 200, deadline)
            if c4c < best_c:
                best_h, best_c = h4c, c4c
        n_cur = len(best_h)
        for r in range(30):
            if time.time() > deadline - 30:
                break
            s = 0.0005 + 0.0015 * (r % 12)
            mode = r % 4
            if mode == 0:
                pert = rng.normal(0, s, n_cur)
            elif mode == 1:
                pert = s * np.sin(2 * np.pi * rng.randint(1, 5) * np.arange(n_cur) / n_cur)
            elif mode == 2:
                pert = np.zeros(n_cur)
                idx = rng.randint(0, n_cur, size=n_cur // 4)
                pert[idx] = rng.choice([-1, 1], size=len(idx)) * s
            else:
                pert = rng.normal(0, s, n_cur) * np.cos(np.pi * np.arange(n_cur) / n_cur)
            hp = project_feasible(best_h + pert, n_cur)
            maxp = max(40, 120 - 2 * r)
            h_r, c_r = coord_cd(hp, n_cur, maxp, deadline, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
        h6, c6 = coord_cd(best_h, len(best_h), 400, deadline, rng)
        if c6 < best_c:
            best_h, best_c = h6, c6
        if time.time() < deadline - 10:
            h7, c7 = lse_projected(best_h, len(best_h), 5000, 200, deadline)
            if c7 < best_c:
                best_h, best_c = h7, c7
            h7b, c7b = coord_cd(best_h, len(best_h), 200, deadline, rng)
            if c7b < best_c:
                best_h, best_c = h7b, c7b
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
        return best_h, best_c, len(best_h)
    except Exception:
        return best_h, best_c, len(best_h)
