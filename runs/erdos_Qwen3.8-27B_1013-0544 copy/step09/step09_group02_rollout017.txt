  So there IS a cross term: -δ_p*δ_q when q=p+k.
  
  Similarly for the other direction.
  
  This makes pairwise CD a quadratic optimization problem in 2 variables, which can be solved analytically or with a small bisection.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_polish(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.06
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        sg = np.zeros(n)
        for i in range(n):
            j = i + m_star - (n - 1)
            if 0 <= j < n:
                sg[i] += (1.0 - h[j])
            ip = i - m_star + (n - 1)
            if 0 <= ip < n:
                sg[i] -= h[ip]
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 50 or lr < 1e-10:
                break
    return h, best_c

def gen_binary_start(n, rng):
    h = rng.randint(0, 2, size=n).astype(float)
    target = n / 2.0
    n_ones = h.sum()
    if n_ones > target:
        idx_ones = np.where(h == 1.0)[0]
        flip = min(int(n_ones - target), len(idx_ones))
        h[idx_ones[rng.choice(len(idx_ones), flip, replace=False)]] = 0.0
    elif n_ones < target:
        idx_zeros = np.where(h == 0.0)[0]
        flip = min(int(target - n_ones), len(idx_zeros))
        h[idx_zeros[rng.choice(len(idx_zeros), flip, replace=False)]] = 1.0
    return h

def gen_structured_start(n, rng):
    ctype = rng.randint(0, 4)
    if ctype == 0:
        f = rng.uniform(0.5, 5.0)
        h = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi)))
        h = (h > 0.5).astype(float)
    elif ctype == 1:
        blocks = rng.randint(2, 6)
        h = np.zeros(n)
        for b in range(blocks // 2):
            start = int(b * n / (blocks // 2))
            length = int(n / blocks) + rng.randint(-5, 5)
            h[start:start + max(1, length)] = 1.0
    elif ctype == 2:
        center = rng.randint(0, n)
        width = rng.randint(n // 4, n * 3 // 4)
        h = np.zeros(n)
        h[center:center + width] = 1.0
        if center + width > n:
            h[:center + width - n] = 1.0
    else:
        h = np.zeros(n)
        n_ones = n // 2
        idx = rng.choice(n, n_ones, replace=False)
        h[idx] = 1.0
    return h
