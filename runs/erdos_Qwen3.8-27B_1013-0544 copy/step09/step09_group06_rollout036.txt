    """Project y onto {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        tau = (lo + hi) / 2
        h = np.clip(y - tau, 0, 1)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining diff
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff/tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff/tot) * h
    return np.clip(h, 0, 1)

**λ-duality algorithm:**

2. Given λ, solve: min_h Σ_k λ_k c_k(h) subject to 0≤h≤1, Σh=n/2.

   Use projected gradient ascent on the quadratic:
   grad_quad[p] = dx * (Σ_k λ_k * h[j_k(p)] + Σ_k λ_k * h[j'_k(p)])
   where j_k(p) is the "forward" partner and j'_k(p) is the "backward" partner.

   And I do projected gradient ASCENT (since I want to maximize the quadratic, or equivalently minimize -quadratic).
   
   Or better: I want to minimize Σ λ_k c_k = dx*n/2 - dx*h^T M h.
   So I want to MAXIMIZE h^T M h.
   The gradient of h^T M h is (M+M^T)h.

3. Alternate between steps 1 and 2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(y, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        h = np.clip(y - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_weighted_sum(h, n, w):
    dx = 2.0 / n
    A = np.convolve(h, w, 'full')[:n]
    B = np.convolve(h, w[::-1], 'full')[n-1:2*n-1]
    S1 = np.convolve(w, np.ones(n), 'valid')
    return dx * (S1 - A - B)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def nesterov_opt(h, n, alpha, max_iter, deadline, lr_init=0.01):
    h = project_box_sum(h, n)
    x = h.copy()
    y = h.copy()
    mu = 0.9
    lr = lr_init
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for k in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        m = c.max()
        ew = np.exp(alpha * (c - m))
        w = ew / ew.sum()
        g = grad_weighted_sum(y, n, w)
        x_new = project_box_sum(y - lr * g, n)
        if k % 20 == 0:
            if np.sum(np.abs(x_new - y)) < 1e-12:
                break
        y = x_new + mu * (x_new - x)
        x = x_new
        if k % 50 == 0:
            c_val = compute_c5(x, n)
            if c_val < best_c:
                best_c = c_val
                best_h = x.copy()
            if k > 50 and np.max(np.abs(x - best_h)) < 1e-8:
                break
    c_val = compute_c5(x, n)
    if c_val < best_c:
        best_c = c_val
        best_h = x.copy()
    return best_h, best_c

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            Bc = -(a_coeff[diag_idx] - a_coeff[i])
            Cc = -(c_peak[diag_idx] - c_peak[i])
            disc = Bc * Bc - 4.0 * Cc
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-Bc + sq) / 2.0, (-Bc - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, -h0, 1.0 - h0)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lambda_duality(h, n, max_rounds, deadline, alpha_w=50.0):
    h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        ew = np.exp(alpha_w * (c - m))
        w = ew / ew.sum()
        # Inner: minimize sum_k w_k * c_k(h) via projected gradient
        lr = 0.5 / (np.abs(w).max() * 2 * n)
        for inner in range(200):
            if time.time() > deadline:
                break
            g = grad_weighted_sum(h, n, w)
            h_new = project_box_sum(h - lr * g, n)
            if np.max(np.abs(h_new - h)) < 1e-10:
                break
            h = h_new
    return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
