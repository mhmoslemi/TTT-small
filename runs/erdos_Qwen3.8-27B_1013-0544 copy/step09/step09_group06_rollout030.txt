```python
    lse = mx + np.log(np.sum(np.exp(s - mx)))
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-14:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-14:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals = vals.copy()
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val, best_d = v, d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                a_coeff[idx] = ((1.0 - h[jf]) if 0 <= jf < n else 0.0) + ((-h[jb]) if 0 <= jb < n else 0.0)
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, -h0, 1.0 - h0)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

    x_fine = np.linspace(0, 2, n_fine)
    best_h, best_c = h_fine.copy(), compute_c5(h_fine, n_fine)
    for nc in n_coarse_list:
        if time.time() > deadline - 60:
            break
        x_c = np.linspace(0, 2, nc)
        h_c = project_feasible(np.interp(x_c, x_fine, best_h), nc)
        h_c, c_c = multi_cd(h_c, nc, 500, deadline - 60, 20)
        h_back = project_feasible(np.interp(x_fine, x_c, h_c), n_fine)
        h_back, c_back = multi_cd(h_back, n_fine, 300, deadline - 60, 20)
        if c_back < best_c:
            best_h, best_c = h_back, c_back
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick CD warm start
    h1, c1 = multi_cd(best_h, n, 200, deadline - 800, 12)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Coarse-to-fine at multiple scales
    x_cur = np.linspace(0, 2, len(best_h))
    n_cur = len(best_h)
    coarse_sizes = [max(n_cur // 6, 80), max(n_cur // 4, 100), max(n_cur // 3, 120)]
    coarse_sizes = [s for s in coarse_sizes if s < n_cur]
    if coarse_sizes and time.time() < deadline - 400:
        h2, c2 = coarse_to_fine(best_h, n_cur, deadline - 400, coarse_sizes)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n_cur = len(best_h)

    # Phase 3: SLSQP with alpha schedule
    if time.time() < deadline - 350:
        for alpha in [2000, 10000, 50000, 200000, 1000000]:
            if time.time() > deadline - 350:
                break
            hs, cs = slsqp_run(best_h, n_cur, alpha, 300)
            if cs < best_c:
                best_h, best_c = hs, cs
                n_cur = len(best_h)

    # Phase 4: Multiple perturbations + CD
    if time.time() < deadline - 200:
        n_p = len(best_h)
        for r in range(15):
            if time.time() > deadline - 200:
                break
            sigma = 0.001 + 0.003 * r
            hp = project_feasible(best_h + rng.normal(0, sigma, n_p), n_p)
            hp, cp = multi_cd(hp, n_p, 150, deadline - 200, 15)
            hp, cp = try_symmetry(hp, n_p)
            if cp < best_c:
                best_h, best_c = hp, cp
                n_p = len(best_h)

    # Phase 5: Different grid sizes with interpolation
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 800, 900]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        h_int, c_int = multi_cd(h_int, n_try, 300, deadline - 150, 20)
        h_int, c_int = try_symmetry(h_int, n_try)
        if c_int < best_c:
            best_h, best_c = h_int, c_int
            x_best = x_new
            n_cur = n_try

    # Phase 6: Final deep CD
    n_fin = len(best_h)
    if time.time() < deadline - 60:
        h6, c6 = multi_cd(best_h, n_fin, 1000, deadline - 60, 30)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 15:
        hs, cs = slsqp_run(best_h, len(best_h), 2000000, 500)
        if cs < best_c:
            best_h, best_c = hs, cs

    # Phase 8: Ultimate CD
    if time.time() < deadline - 2:
        h8, c8 = multi_cd(best_h, len(best_h), 2000, deadline, 35)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Final symmetry + projection
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
