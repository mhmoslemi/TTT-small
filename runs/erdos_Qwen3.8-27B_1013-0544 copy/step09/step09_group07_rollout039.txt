def project_feasible(h, n):

        if diff > 0:

            if tot < 1e-14:

            h += (diff / tot) * h

def compute_grad(h, q, n):

def binary_snap(h, n):
    h2 = h.copy()
    order = np.argsort(h2)[::-1]

    h2[:] = 0.0
