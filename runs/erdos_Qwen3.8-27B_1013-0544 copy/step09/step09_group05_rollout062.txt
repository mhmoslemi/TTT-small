import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def build_lp_matrix(h_k, n):
    """Build sparse LP coefficient matrix for linearized overlap constraints."""
    from scipy.sparse import CSRMatrix
    n_shifts = 2 * n - 1
    rows = []
    cols = []
    vals = []
    consts = np.zeros(n_shifts)
    
    for m in range(n_shifts):
        offset = m - (n - 1)
        # valid i: 0 <= i < n and 0 <= i + offset < n
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        for i in range(i_lo, i_hi):
            j = i + offset
            # coeff[i] += (1 - h_k[j])
            rows.append(m); cols.append(i); vals.append(1.0 - h_k[j])
            # coeff[j] += -h_k[i]
            rows.append(m); cols.append(j); vals.append(-h_k[i])
            consts[m] += h_k[i] * h_k[j]
    
    A = CSRMatrix((vals, (rows, cols)), shape=(n_shifts, n))
    return A, consts

def solve_lp(h_k, n, solver='ECOS'):
    """Solve the linearized LP."""
    A, consts = build_lp_matrix(h_k, n)
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0
    ]
    
    # For each shift m: A[m,:] @ h - t <= -consts[m]
    # i.e., A @ h - t*ones <= -consts
    ones = np.ones(A.shape[0])
    constraints.append(A @ h - t * ones <= -consts)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if h.value is not None and t.value is not None:
            return np.clip(h.value, 0, 1), t.value
    except Exception:
        pass
    return None, None

def ilp_optimize(h_init, n, max_iters=50, deadline=None, solver='ECOS'):
    """Iterative LP optimization."""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline and time.time() > deadline - 10:
            break
        
        h_lp, t_lp = solve_lp(h, n, solver=solver)
        if h_lp is None:
            break
        
        # Line search: try full step, then bisection
        h_new = project_feasible(h_lp, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
        else:
            # Try smaller steps
            found = False
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_trial = project_feasible(h + alpha * (h_lp - h), n)
                c_trial = compute_c5(h_trial, n)
                if c_trial < best_c:
                    best_h = h_trial.copy()
                    best_c = c_trial
                    h = h_trial
                    found = True
                    break
            if not found:
                break
    
    return best_h, best_c

def ilp_step(h, n):

    # Minimize t
    # s.t. A @ h - t <= -const  (for each shift)
    #      sum(h) = n/2
    #      0 <= h <= 1, t free
    
    # Variables: [h(0), ..., h(n-1), t]
    # c = [0, ..., 0, 1] (minimize t)
    # A_ub: [A, -1] @ [h, t] <= -const
    # A_eq: [ones(1,n), 0] @ [h, t] = n/2

    c_obj[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = -const
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

                  bounds=bounds, method='highs')

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            if co < best_c:
                best_h, best_c = ho, co
    # Phase 6: Final heavy CD polish
    n_final = len(best_h)
    if time.time() < deadline - 5:
        remaining = deadline - time.time() - 3
        max_p = max(30, int(remaining / 0.05))
        hf, cf = coord_descent(best_h, n_final, max_p, deadline, rng, False)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
