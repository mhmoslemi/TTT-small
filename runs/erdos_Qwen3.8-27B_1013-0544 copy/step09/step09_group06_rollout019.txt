def compute_c5(h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:

            tot = room.sum()

            if tot < 1e-14:

            h += (diff / tot) * h

def solve_1d_minimax(c_base, a, lo, hi):
    """Solve min_{d in [lo,hi]} max_i (c_base[i] + a[i]*d)"""
    T = len(c_base)
    if T == 0:
        return 0.0, float(c_base.max()) if len(c_base) > 0 else 0.0
    candidates = [lo, hi]
    # Find all pairwise intersections
    for i in range(T):
        for j in range(i + 1, T):
            da = a[i] - a[j]
            if abs(da) > 1e-15:
                d = (c_base[j] - c_base[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_base + a * d
        v = float(vals.max())
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def get_coefficients(h, n, p):

    return a

def slsqp_refine(h, n, alpha, maxiter=200):
    """SLSQP refinement with smooth max approximation."""
    dx = 2.0/n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse/alpha) * dx
    
    def obj_jac(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        weights = np.exp(s - mx) / np.sum(np.exp(s - mx))
        # dc/dx: derivative of c[m] w.r.t. x[p]
        # c[m] = sum_j x[j]*(1-x[j+m-(n-1)])
        # dc[m]/dx[p] = (1-x[p+m-(n-1)])_valid + (-x[p-m+(n-1)])_valid
        jac = np.zeros(n)
        for p in range(n):
            for m in np.where(weights > 1e-8)[0]:
                jf = p + m - (n-1)
                jb = p - m + (n-1)
                val = 0.0
                if 0 <= jf < n:
                    val += (1.0 - x[jf])
                if 0 <= jb < n:
                    val += (-x[jb])
                jac[p] += weights[m] * val
        return (alpha * jac * dx)  # simplified
    
    # Use SLSQP without explicit gradient (numerical)
    bounds = [(0.0, 1.0)] * n
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}
    ]
    try:

                      constraints=constraints, maxiter=maxiter, ftol=1e-15)
        h_new = project_feasible(res.x, n)
        c_new = float(np.correlate(h_new, 1.0-h_new, mode='full').max()) * dx
        return h_new, c_new
    except:
        return h, float(np.correlate(h, 1.0-h, mode='full').max()) * dx

    # For m where j=p: partner = p+m-(n-1), valid if 0 <= partner < n
    #   i.e., n-1-p <= m <= n-1-p+n-1 = 2n-2-p

One approach: compute the subdifferential of the objective at the current point. The subdifferential of max_k c[k](h) is the convex hull of the gradients of the active correlations. A descent direction is any direction d such that <grad_c[k], d> < 0 for all active k. If no such direction exists, we're at a local minimum.

If a descent direction exists, we can take a step in that direction (projected to maintain feasibility).

The gradient of c[k] = ∑_j h[j]*(1-h[j+k-(n-1)]) w.r.t. h[p] is:
(1-h[p+k-(n-1)])_valid - h[p-k+(n-1)]_valid

This is exactly a[p][k] from before.

    a = np.zeros(2*n-1)
    # Forward: h[p] acts as h[j] with j=p, partner at p+m-(n-1)
    m_lo = max(0, n-1-p)

    """min_{d in [lo,hi]} max_i (c_base[i] + a[i]*d)"""
    k = len(c_base)

        return 0.0, 0.0
    candidates = np.array([lo, hi])

        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-8 or h0 >= 1.0 - 1e-8:
                continue
            a = coef_for_point(h, n, p)
            a_sub = a[top_idx]
            lo = -h0
            hi = 1.0 - h0
            _, best_val = solve_minimax_1d_vec(c_sub, a_sub, lo, hi)
            if best_val < unaff:
                best_val = unaff
            if best_val < current_max - 1e-13:
                best_d, _ = solve_minimax_1d_vec(c_sub, a_sub, lo, hi)
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def swap_cd(h, n, max_passes, deadline, top_k=35, max_pairs=60):
    h = h.copy()
    n_corr = 2 * n - 1
    rng_sw = np.random.RandomState(777)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]
        c_sub = c[top_idx].copy()
        rest_mask = np.ones(n_corr, dtype=bool)
        rest_mask[top_idx] = False
        unaff = float(c[rest_mask].max()) if rest_mask.any() else 0.0
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        for _t in range(max_pairs):
            if time.time() > deadline:
                break
            p = int(rng_sw.choice(interior))
            q = int(rng_sw.choice(interior))
            if p == q:
                continue
            a_p = coef_for_point(h, n, p)
            a_q = coef_for_point(h, n, q)
            a_comb = (a_q - a_p)[top_idx]
            lo = 0.0
            hi = min(h[p], 1.0 - h[q])
            if hi < 1e-8:
                continue
            best_eps, best_val = solve_minimax_1d_vec(c_sub, a_comb, lo, hi)
            if best_val < unaff:
                best_val = unaff
            if best_val < current_max - 1e-13:
                h[p] -= best_eps
                h[q] += best_eps
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb_refine(h, n, rng, deadline, n_trials=20):
    h_best = h.copy()
    c_best = compute_c5(h, n)
    x = np.linspace(0, 2, n)
    for t in range(n_trials):
        if time.time() > deadline:
            break
        trial_type = rng.randint(3)
        if trial_type == 0:
            n_modes = rng.randint(1, 4)
            freqs = rng.choice(np.arange(1, n // 4 + 1), size=n_modes, replace=False)
            phases = rng.uniform(0, 2 * np.pi, size=n_modes)
            amps = rng.uniform(0.003, 0.02, size=n_modes)
            pert = np.zeros(n)
            for f, ph, a in zip(freqs, phases, amps):
                pert += a * np.cos(2 * np.pi * f * x / 2.0 + ph)
            h_p = project_feasible(h_best + pert, n)
        elif trial_type == 1:
            shift = rng.randint(1, 4)
            h_p = project_feasible(np.roll(h_best, shift), n)
            h_p = project_feasible(h_best * (1 - 0.01) + 0.01 * np.roll(h_best, shift), n)
        else:
            s = 0.003 + 0.005 * (t % 5)
            h_p = project_feasible(h_best + rng.normal(0, s, n), n)
        h_r, c_r = cd_optimize(h_p, n, 60, deadline, top_k=25)
        if c_r < c_best - 1e-7:
            h_best, c_best = h_r, c_r
    return h_best, c_best

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
