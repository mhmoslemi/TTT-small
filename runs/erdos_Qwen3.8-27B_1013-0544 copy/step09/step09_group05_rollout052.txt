def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = np.random.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def anneal_step(h, n, c, T, rng):
    """Single simulated annealing step"""
    p = rng.randint(n)
    delta = rng.uniform(-0.05, 0.05)
    h_new = h.copy()
    h_new[p] = np.clip(h[p] + delta, 0, 1)
    h_new = project_feasible(h_new, n)
    c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
    old_max = np.max(c)
    new_max = np.max(c_new)
    if new_max <= old_max or rng.rand() < np.exp(-(new_max - old_max) / max(T, 1e-10)):
        return h_new, c_new
    return h, c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoints
    # Base: 0.5 + small perturbation
    h = np.full(n, 0.5)
    # Add Fourier modes
    for k in range(1, 8):

        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.sin(k * np.pi * x + phase)
    return project_feasible(h, n)

def active_gradient_step(h, n, c, lr, top_k=5):
    """Step to reduce the top-k active correlations"""
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for p in top_idx:
        for i in range(n):
            jf = i + p - (n - 1)
            jb = i - p + (n - 1)
            if 0 <= jf < n:
                grad[i] += (1.0 - h[jf])
            if 0 <= jb < n:
                grad[i] -= h[jb]
    grad -= np.mean(grad)
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def compute_c5(h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))

        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and len(a_nd) > 0:
            xl = 0.0
            xh = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):

            if np.any(pos):

            if xl > xh + 1e-14:
                feasible = False
        if feasible and has_diag:

            disc = 1.0 - 4.0 * Cq
            if disc >= 0:

                if len(a_nd) > 0:

                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):

                else:

                        if 0.0 < r1 or 1.0 > r2:

        if feasible:

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def smart_perturb(h, n, rng, scale=0.01):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[-3:]
    grad = np.zeros(n)
    for k in top_k:
        for i in range(n):
            jf = i + k - (n - 1)
            jb = i - k + (n - 1)
            if 0 <= jf < n:
                grad[i] += (1.0 - h[jf])
            if 0 <= jb < n:
                grad[i] -= h[jb]
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-10:
        pert = rng.normal(0, scale, n)
    else:
        pert = -scale * grad / gnorm + rng.normal(0, scale * 0.5, n)
    return project_feasible(h + pert, n)

def pair_move(h, n, rng, attempts=20, deltas=None):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(attempts):
        p, q = rng.choice(n, 2, replace=False)
        for d in deltas:
            for sp, sq in [(1, 1), (1, -1), (-1, 1), (-1, -1)]:
                h_try = h.copy()
                h_try[p] = np.clip(h_try[p] + sp * d, 0, 1)
                h_try[q] = np.clip(h_try[q] + sq * d, 0, 1)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_h = h_try.copy()
                    best_c = c_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP alpha-annealing
    h1, c1 = slsqp_run(best_h, n, [200, 2000, 20000, 200000], 400, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Heavy coordinate descent
    h2, c2 = coord_descent(best_h, n, 500, deadline, rng, random_order=True)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Smart restarts with CD
    stall = 0
    max_restarts = 8
    for r in range(max_restarts):
        if time.time() > deadline - 80:
            break
        hp = smart_perturb(best_h, n, rng, scale=0.008 + 0.004 * r)
        ho, co = coord_descent(hp, n, 80, deadline, rng, random_order=True)
        if co < best_c - 1e-10:
            best_h, best_c = ho, co
            stall = 0
        else:
            stall += 1
        if stall >= 4:
            break

    # Phase 4: Pair moves to escape
    if time.time() < deadline - 60:
        for _ in range(5):
            if time.time() > deadline - 60:
                break
            hp, cp = pair_move(best_h, n, rng, attempts=30)
            if cp < best_c:
                best_h, cp = hp, cp
            ho, co = coord_descent(hp, n, 50, deadline, rng, random_order=True)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Try different n
    n_cands = [400, 500, 600, 800]
    n_cands = [x for x in n_cands if x != n]
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in n_cands:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = coord_descent(h_int, n_try, 150, deadline, rng, random_order=True)
        if co < best_c:
            best_h, best_c = ho, co
            n = n_try
            x_orig = np.linspace(0, 2, n_try, endpoint=False)

    # Phase 6: Final SLSQP + CD polish
    if time.time() < deadline - 10:
        hi, ci = slsqp_run(best_h, n, [50000, 500000], 500, deadline)
        if ci < best_c:
            best_h, best_c = hi, ci
    if time.time() < deadline - 3:
        remaining = deadline - time.time() - 2
        max_p = max(30, int(remaining / 0.04))
        hf, cf = coord_descent(best_h, n, max_p, deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
