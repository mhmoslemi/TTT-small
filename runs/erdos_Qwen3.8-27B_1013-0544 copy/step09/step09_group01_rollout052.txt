Let me go back to the CD approach but make it smarter.

**Key improvement idea: "Active set" CD**

At each CD step for coordinate p, the max of the affected correlations is achieved by a small number of k values (the "active set"). If I can identify these quickly, I can solve the 1D subproblem more precisely.

For the 1D subproblem: minimize max_{k in S} (c_k + d * a_k) + diagonal term, subject to d ∈ [-h0, 1-h0].

If the active set is {k1, k2, ..., km}, then at the optimum, the active constraints are equal (KKT conditions). So I can solve:
c_ki + d * a_ki = V for all i in active set
This gives d = (V - c_ki) / a_ki for each i, so V = c_ki + a_ki * (V - c_kj) / a_kj
=> V * (1 - a_ki/a_kj) = c_ki - a_ki * c_kj / a_kj
=> V = (c_ki - a_ki * c_kj / a_kj) / (1 - a_ki / a_kj)

This is a small system that can be solved directly. But the active set changes as d changes, so it's iterative.

Actually, for a max of linear functions, the optimum in 1D is at a point where two (or more) of the lines cross, or at a boundary. So the algorithm is:
1. Find the current active set (k values achieving the max at d=0)
2. Find the crossing points of pairs of active lines
3. The optimal d is the one that minimizes the max, which is at a crossing point or boundary

This is essentially what golden section does, but potentially faster if we can identify the active set quickly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def cd_pass(h, n, order, gs_iters, tol):
    """One pass of coordinate descent over given order."""
    c = np.correlate(h, 1.0 - h, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_improved = 0
    for p in order:
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, n - 1 + p)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a = np.zeros(len(ms))
        if np.any(vf):
            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        dp = np.where(ms == (n - 1))[0]
        hd = len(dp) > 0
        di = int(dp[0]) if hd else -1
        c_aff = c[ms]
        um = 0.0
        if m_lo > 0:
            um = max(um, float(c[:m_lo].max()))
        if m_hi < 2 * n - 2:
            um = max(um, float(c[m_hi + 1:].max()))
        
        d_lo = -h0
        d_hi = 1.0 - h0
        d1 = d_hi - gr * (d_hi - d_lo)
        d2 = d_lo + gr * (d_hi - d_lo)
        f1 = _cd_fval(c_aff, a, di, hd, um, h0, d1)
        f2 = _cd_fval(c_aff, a, di, hd, um, h0, d2)
        
        for _ in range(gs_iters):
            if d_hi - d_lo < 1e-14:
                break
            if f1 < f2:
                d_hi = d2; d2 = d1; f2 = f1
                d1 = d_hi - gr * (d_hi - d_lo)
                f1 = _cd_fval(c_aff, a, di, hd, um, h0, d1)
            else:
                d_lo = d1; d1 = d2; f1 = f2
                d2 = d_lo + gr * (d_hi - d_lo)
                f2 = _cd_fval(c_aff, a, di, hd, um, h0, d2)
        
        d_opt = (d_lo + d_hi) / 2.0
        f_opt = _cd_fval(c_aff, a, di, hd, um, h0, d_opt)
        f_curr = _cd_fval(c_aff, a, di, hd, um, h0, 0.0)
        
        if f_opt < f_curr - tol:
            h[p] = h0 + d_opt
            n_improved += 1
            # Update c for affected lags
            # (We'll recompute at the end of the pass for simplicity)
    return h, n_improved

def coord_descent(h, n, max_passes, deadline, gs_iters=55, tol=1e-13, rng=None):
    h = h.copy()
    if rng is None:
        rng = np.random.RandomState(42)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        # Randomize order for better convergence
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        h, n_imp = cd_pass(h, n, order, gs_iters, tol)
        if n_imp == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_refine(h, n, deadline, max_iters=500000):
    """Try to improve by binary swaps."""
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones != n // 2:
        # Fix
        if n_ones > n // 2:
            # Remove excess ones (those closest to 0.5)
            one_indices = np.where(h_bin == 1.0)[0]
            sort_order = np.argsort(h[one_indices])[::-1]  # smallest h values first
            for idx in sort_order[:n_ones - n//2]:
                h_bin[one_indices[idx]] = 0.0
        else:
            zero_indices = np.where(h_bin == 0.0)[0]
            sort_order = np.argsort(h[zero_indices])[::-1]  # largest h values first
            for idx in sort_order[:n//2 - n_ones]:
                h_bin[zero_indices[idx]] = 1.0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    current_c5 = float(c.max()) * (2.0/n)
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        # Swap
        h_bin[i], h_bin[j] = 0.0, 1.0
        c_new = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        new_max = float(c_new.max())
        delta = new_max - float(c.max())
        if delta <= 1e-15:  # accept (equal or better)
            c = c_new
            best_c = min(best_c, new_max * (2.0/n))
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
    
    return best_h, best_c

        h3, c3 = coord_descent(best_h, n, 200, deadline - 30, gs_iters=65, tol=1e-14, seed=7)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Try different n values
    if time.time() < deadline - 120:
        x_orig = np.linspace(0, 2, len(best_h))
        for n_try in [500, 800, 1000]:
            if time.time() > deadline - 40:
                break
            if n_try == len(best_h):
                continue
            x_new = np.linspace(0, 2, n_try)
            h_int = np.interp(x_new, x_orig, best_h)
            h_int = project_feasible(h_int, n_try)
            ho, co = slsqp_optimize(h_int, n_try, [2000, 50000, 500000], 200)
            ho, co = coord_descent(ho, n_try, 150, deadline - 20, gs_iters=55, tol=1e-13, seed=13)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Perturbation restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        for r in range(10):
            if time.time() > deadline - 30:
                break
            if r % 3 == 0:
                s = 0.01 + 0.015 * r
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            elif r % 3 == 1:
                freq = 2.0 + 1.5 * (r // 3)
                hp = project_feasible(best_h + 0.02 * np.sin(np.pi * freq * x), n_cur)
            else:
                freq = 4.0 + 2.0 * (r // 3)
                hp = project_feasible(best_h + 0.015 * np.cos(np.pi * freq * x), n_cur)
            ho, co = coord_descent(hp, n_cur, 100, deadline - 20, gs_iters=55, tol=1e-13, seed=100 + r)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final high-precision polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 200, deadline - 5, gs_iters=80, tol=1e-15, seed=999)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
