def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    if np.any(pos_m):

                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:

            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-12:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, c, n_bisect=30):
    """One pass of coordinate descent. Updates h in place. Returns number of updates."""
    updates = 0
    for p in range(n):
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        x_new = solve_1d(c, h, p, n, n_bisect)
        if abs(x_new - h0) > 1e-13:
            h[p] = x_new
            updates += 1
    return updates

def cd_loops(h, n, max_passes, deadline, n_bisect=25):
    """Multiple passes of coordinate descent."""

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        upd = cd_pass(h, n, c, n_bisect)
        h = project_feasible(h, n)
        if upd == 0:
            break
    return h, compute_c5(h, n)

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
