        # Backward: p + d_shift in [0, n-1] => p in [0, n-1-d_shift]

        grad[:n-s] += (1.0 - h[s:])
        # Backward: p + (-s) = p-s in [0, n-1] => p in [s, n-1]
        grad[s:] -= h[:n-s]
    return grad

            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def soft_grad_descent(h, n, max_iters, deadline, alpha0=100.0, step0=0.02, beta=0.85):
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_val = float(c.max()) * (2.0 / n)
        if cur_val < best_c:
            best_c = cur_val
            best_h = h.copy()
        s = alpha0 * (c - c.max())
        w = np.exp(s)
        w /= w.sum()
        active = np.where(w > 1e-5)[0]
        grad = np.zeros(n)
        for m in active:
            wm = w[int(m)]
            ds = int(m) - (n - 1)
            if ds >= 0:
                grad[ds:] += wm * (1.0 - h[:n - ds])
                grad[:n - ds] -= wm * h[ds:]
            else:
                s2 = -ds
                grad[:n - s2] += wm * (1.0 - h[s2:])
                grad[s2:] -= wm * h[:n - s2]
        step = step0 / (1.0 + 0.1 * it)
        vel = beta * vel + (1.0 - beta) * grad
        h = project_feasible(h - step * vel, n)
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_obj_topk(x, n, alpha, k):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    topk_sum = np.sort(c)[-k:].sum()
    return (topk_sum + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300, use_topk=False, k=10):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if use_topk:
        obj = smooth_obj_topk
        args = (n, alpha, k)
    else:
        obj = smooth_obj
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD convergence
    if time.time() < deadline - 400:
        h2, c2 = multi_cd(best_h, n, 300, deadline - 400, 20)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 2: Soft gradient descent with momentum
    if time.time() < deadline - 350:
        for alpha_try in [50.0, 200.0, 1000.0]:
            if time.time() > deadline - 350:
                break
            h_sg, c_sg = soft_grad_descent(best_h, n, 800, deadline - 350, alpha0=alpha_try, step0=0.015)
            if c_sg < best_c:
                best_h, best_c = h_sg, c_sg
                h2, c2 = multi_cd(best_h, n, 100, deadline - 350, 15)
                if c2 < best_c:
                    best_h, best_c = h2, c2
    # Phase 3: SLSQP with multiple alphas
    if time.time() < deadline - 200:
        for alpha in [3000, 15000, 80000]:
            if time.time() > deadline - 200:
                break
            h_s, c_s = slsqp_run(best_h, n, alpha, 250)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
            h_s, c_s = slsqp_run(best_h, n, alpha, 150, use_topk=True, k=8)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
    # Phase 4: Perturbation restarts
    if time.time() < deadline - 120:
        scales = [0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
        for s in scales:
            if time.time() > deadline - 120:
                break
            for trial in range(2):
                if time.time() > deadline - 120:
                    break
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
                hc, cc = multi_cd(hp, n, 120, deadline - 120, 15)
                if cc < best_c:
                    best_h, best_c = hc, cc
                    h_sg, c_sg = soft_grad_descent(best_h, n, 300, deadline - 120, alpha0=200.0, step0=0.01)
                    if c_sg < best_c:
                        best_h, best_c = h_sg, c_sg
    # Phase 5: Grid size exploration
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 550, 650, 750, 800]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 200, deadline - 80, 15)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new
    # Phase 6: Final deep optimization
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        h_f, c_f = slsqp_run(best_h, n_cur, 300000, 400)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        h_f, c_f = multi_cd(best_h, n_cur, 2000, deadline, 30)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
