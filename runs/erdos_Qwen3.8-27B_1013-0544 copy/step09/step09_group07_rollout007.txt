def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):

        except Exception:

    # Phase 4: Shake and re-optimize (escape local minima)
    n_cur = len(best_h)
    shake_temps = [0.005, 0.01, 0.02, 0.008, 0.015]
    for st in shake_temps:
        if time.time() > deadline - 80:
            break
        hp = project_feasible(best_h + rng.normal(0, st, n_cur), n_cur)
        ho, co = coord_cd(hp, n_cur, max_passes=200, deadline=deadline - 70, top_k=45)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 5: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 800]:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=200, deadline=deadline - 50, top_k=40)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 6: Alternative initializations
    if time.time() < deadline - 50:
        for n_alt in [n_cur, 500]:
            if time.time() > deadline - 35:
                break
            for init_fn in [lambda: make_sinusoidal(n_alt, 1.0, 0.0),
                            lambda: make_sinusoidal(n_alt, 2.0, np.pi/4),
                            lambda: make_block_pair(n_alt)]:
                if time.time() > deadline - 30:
                    break
                h_alt = init_fn()
                ha, ca = coord_cd(h_alt, n_alt, max_passes=150, deadline=deadline - 25, top_k=40)
                if ca < best_c:
                    best_h, best_c = ha, ca

    # Phase 7: Final SLSQP with high precision
    n_fin = len(best_h)
    if time.time() < deadline - 30:
        h7, c7 = slsqp_polish(best_h, n_fin, [5000, 20000, 100000], 500, deadline - 15)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Phase 8: Final coordinate descent
    if time.time() < deadline - 10:
        h8, c8 = coord_cd(best_h, n_fin, max_passes=1000, deadline=deadline - 5, top_k=55)
        if c8 < best_c:
            best_h, best_c = h8, c8

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
