import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def grad_lse(h, n, alpha):
    """Gradient of (1/alpha)*logsumexp(alpha*c)*dx where c = correlate(h,1-h,'full')"""
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    # Compute gradient vectorized over m for each p
    ms = np.arange(2 * n - 1)
    ts = ms - (n - 1)
    grad = np.zeros(n)
    for p in range(n):
        v1 = (0 <= p + ts) & (p + ts < n)
        v2 = (0 <= p - ts) & (p - ts < n)
        g = np.zeros(2 * n - 1)
        if np.any(v1):
            g[v1] += (1.0 - h[p + ts[v1]])
        if np.any(v2):
            g[v2] -= h[p - ts[v2]]
        grad[p] = np.dot(w, g)
    return grad * dx

def nesterov_opt(h, n, steps, alpha, deadline, lr=None):
    """Nesterov accelerated gradient with projection"""
    h = h.copy()
    y = h.copy()
    if lr is None:
        lr = 0.01
    best_h, best_c = h.copy(), compute_c5(h, n)
    for k in range(steps):
        if time.time() > deadline:
            break
        t_k = (k + 1) ** 0.5
        beta = (1 - t_k) / (1 + t_k + 1)  # momentum
        g = grad_lse(y, n, alpha)
        # Projected gradient step
        step = y - lr * g
        step = project_feasible(step, n)
        # Nesterov extrapolation
        t_next = (k + 2) ** 0.5
        beta_next = (1 - t_next) / (1 + t_next + 1)
        h_new = step + beta_next * (step - y)
        h_new = project_feasible(h_new, n)
        y = h_new
        c = compute_c5(y, n)
        if c < best_c:
            best_h, best_c = y.copy(), c
    return best_h, best_c

    w /= w.sum()

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def grad_lse(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    ts = np.arange(2 * n - 1) - (n - 1)
    p_idx = np.arange(n)[:, None]
    t_idx = ts[None, :]
    idx1 = p_idx + t_idx
    valid1 = (idx1 >= 0) & (idx1 < n)
    M1 = np.zeros((n, 2 * n - 1))
    M1[valid1] = 1.0 - h[idx1[valid1]]
    idx2 = p_idx - t_idx
    valid2 = (idx2 >= 0) & (idx2 < n)
    M2 = np.zeros((n, 2 * n - 1))
    M2[valid2] = h[idx2[valid2]]
    return ((M1 - M2) @ w) * dx

def nesterov_opt(h, n, alpha, steps, lr, deadline):
    h = h.copy()
    v = np.zeros(n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    for k in range(steps):
        if time.time() > deadline:
            break
        m = (k + 2) / (k + 5)
        y = h - m * v
        g = grad_lse(y, n, alpha)
        v = m * v + g
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_h, best_c = h.copy(), c
    return best_h, best_c

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    for _ in range(60):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1 and x_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if h.sum() != n / 2.0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    
    # Phase 1: Nesterov optimization
    for alpha in [500, 2000, 10000]:
        if time.time() > deadline - 600:
            break
        for lr in [0.005, 0.01, 0.02]:
            h1, c1 = nesterov_opt(h_best, n, alpha, 200, lr, deadline)
            if c1 < best_c:
                h_best, best_c = h1, c1
    
    # Phase 2: Exact CD (main workhorse)
    h2, c2 = exact_cd(h_best, n, 1000, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2
    
    # Phase 3: Multi-n exploration
    n_list = [350, 400, 500, 600, 800]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        for alpha in [1000, 5000]:
            h_n, c_n = nesterov_opt(h_int, n_try, alpha, 100, 0.01, deadline)
            if c_n < best_c:
                h_int, best_c = h_n, c_n
        h_o, c_o = exact_cd(h_int, n_try, 500, deadline)
        if c_o < best_c - 1e-8:
            h_best, best_c = h_o, c_o
            n = n_try
    
    # Phase 4: Shake and settle
    n = len(h_best)
    for r in range(10):
        if time.time() > deadline - 100:
            break
        s = 0.01 + 0.02 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 300, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
        # Also try Nesterov on perturbed
        h_ns, c_ns = nesterov_opt(hp, n, 5000, 50, 0.01, deadline)
        h_o2, c_o2 = exact_cd(h_ns, n, 200, deadline)
        if c_o2 < best_c:
            h_best, best_c = h_o2, c_o2
    
    # Phase 5: Final CD polish
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 500, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
    
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)

Wait, I'm concerned about the Nesterov method. The momentum formula I used is:
m = (k+2)/(k+5)
y = h - m*v
g = grad(y)
v = m*v + g
h_new = h - lr*v

This is a variant of Nesterov's accelerated gradient. Let me double-check the formula.

Standard Nesterov:
x_{k+1} = y_k - α_k ∇f(y_k)
y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) (x_{k+1} - x_k)
where t_{k+1} = (1 + sqrt(1 + 4t_k²))/2, t_0 = 1

In terms of momentum β_k = (t_k - 1)/t_{k+1}:
x_{k+1} = y_k - α_k ∇f(y_k)
y_{k+1} = x_{k+1} + β_k (x_{k+1} - x_k)

With the "heavy ball" approximation β_k ≈ k/(k+2) (for large k):
y = x - β*v (extrapolation)
v_new = grad(y) + β*(v)  (hmm, this isn't quite right)

Actually, let me use a simpler formulation. The "Nesterov momentum" as implemented in many frameworks:

v = β*v + grad
x = x - lr*v

With β starting at 0 and increasing to ~0.9.

Or equivalently, the "lookahead" method:
y = x + β*(x - x_prev)  (lookahead)
g = grad(y)
x_new = x - lr*g
β_new = (1+sqrt(1+4*β²))/2 ... this gets complicated.

Let me just use a simple momentum SGD approach:
v = β*v + g
x = x - lr*v

with β = 0.9 (or some other value) and lr tuned. This is simpler and often works well.

Actually, for non-convex problems, the momentum can help escape shallow local minima. Let me use:
β = 0.9, lr = 0.01 (or tuned)

But I need to be careful: the projection onto the feasible set (0≤h≤1, sum(h)=n/2) after each step can interfere with the momentum. The "projected" version of momentum SGD is known to work but convergence is not guaranteed.

Let me simplify and just use projected gradient descent with a few different step sizes, plus the exact CD as the main workhorse. The Nesterov is a nice-to-have but the CD is what really matters.
