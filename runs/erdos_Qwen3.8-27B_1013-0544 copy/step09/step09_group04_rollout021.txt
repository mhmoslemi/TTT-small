            else:

            xb = solve_1d(c, h, p+1, n)
            # Try a small grid around (xa, xb)
            best_val = cur_max
            best_a, best_b = h0a, h0b

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def active_order(c, n):
    cur_max = float(c.max())
    tol = 0.05 * cur_max
    active = c > (cur_max - tol)
    score = np.zeros(n)
    for p in range(n):
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, n - 1 + p)
        score[p] = active[m_lo:m_hi + 1].sum()
    return np.argsort(-score)

            rng.shuffle(idx)
        else:
            idx = n_idx

            stall += 1
            if stall > 6:

    return h, compute_c5(h, n)
