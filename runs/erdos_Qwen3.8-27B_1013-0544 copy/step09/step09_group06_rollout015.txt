def legendre_basis(n, M):
    """Build Legendre basis matrix B[i,j] = P_j(x_i - 1) for j=1..M"""
    x = np.linspace(0, 2, n)
    u = x - 1.0  # in [-1, 1]
    B = np.zeros((n, M))
    for j in range(1, M + 1):
        B[:, j-1] = eval_legendre(j, u)
    return B

def h_from_legendre(coeffs, B):
    """h = 0.5 + B @ coeffs"""
    return 0.5 + B @ coeffs

def legendre_objective(coeffs, B, n, alpha, beta):
    """Objective: smooth C5 + barrier for [0,1] constraint."""
    h = h_from_legendre(coeffs, B)
    # Check bounds
    if h.min() <= 1e-10 or h.max() >= 1.0 - 1e-10:
        return 10.0  # penalty for boundary violation
    # Log barrier
    barrier = -beta * np.sum(np.log(h) + np.log(1.0 - h))
    # Smooth C5
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    smooth_c5 = m + np.log(np.sum(np.exp(s - s.max())) + 1e-30) / alpha
    return smooth_c5 + barrier

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    """Exact 1D minimax solution for a single variable update."""
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B_coef = -(a_coeff[diag_idx] - a_coeff[i])
            C_coef = -(c_peak[diag_idx] - c_peak[i])
            disc = B_coef * B_coef - 4.0 * C_coef
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B_coef + sq) / 2.0
                d2 = (-B_coef - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] = vals[diag_idx] - d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    """Multi-peak coordinate descent with exact 1D minimax."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            # Try a few random restarts within the current solution
            r = np.random.RandomState(int(time.time() * 1000) % 2**31)
            for _ in range(5):
                p = r.randint(n)
                h0 = h[p]
                if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                    continue
                # Try a random perturbation
                d = r.uniform(-0.1, 0.1)
                h_new = h.copy()
                h_new[p] = np.clip(h0 + d, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < current_max:
                    h = h_new
                    current_max = c_new
                    improved = True
                    break
            if not improved:
                break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
