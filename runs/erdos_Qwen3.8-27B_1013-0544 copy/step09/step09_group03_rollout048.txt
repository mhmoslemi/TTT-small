    """Gradient descent on smooth max with momentum."""
    x = h.copy()
    v = np.zeros(n)
    best_x = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.1
    
    for alpha in alpha_schedule:

        corr = np.correlate(x, 1.0 - x, mode='full')
        # Softmax weights
        shifted = alpha * (corr - corr.max())

            tot = h.sum()
            if tot < 1e-14:

    return np.clip(h, 0.0, 1.0)
