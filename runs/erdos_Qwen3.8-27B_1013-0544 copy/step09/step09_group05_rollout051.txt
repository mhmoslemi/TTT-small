        j = i + k - (n - 1)
        if 0 <= j < n:
            contribs[i] = h[i] * (1.0 - h[j])
        j2 = i - k + (n - 1)
        if 0 <= j2 < n:
            contribs[i] += h[j2] * (1.0 - h[i])
    top_n = max(10, n // 15)
    top = np.argsort(contribs)[-top_n:]
    h_new = h.copy()
    h_new[top] += rng.uniform(-mag, mag, len(top))
    return project_feasible(h_new, n)

def make_fourier_init(n, seed_val):
    x = np.linspace(0, 2, n)
    g = 0.5 + 0.2 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x) + 0.05 * np.sin(5 * np.pi * x)
    g = project_feasible(g, n)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = make_fourier_init(500, seed)
        n = len(h_init)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Main CD on initial
    h1, c1 = coord_descent(h, n, 150, deadline - 250, rng)
    if c1 < best_c:
        best_h, best_c = h1.copy(), c1

    # Phase 2: Perturbation restarts
    n_restarts = max(5, int((deadline - time.time() - 80) / 40))
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        mag = 0.02 + 0.04 * r / max(1, n_restarts - 1)
        hp = perturbe(best_h, n, rng, mag)
        h2, c2 = coord_descent(hp, n, 80, deadline - 40, rng)
        if c2 < best_c:
            best_h, best_c = h2.copy(), c2

    # Phase 3: Try other resolutions
    n_cands = sorted(set([n, n // 2, int(n * 1.3), 400, 600]))
    n_cands = [x for x in n_cands if x != n and x >= 200]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_cands:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h3, c3 = coord_descent(h_int, n_try, 80, deadline - 30, rng)
        if c3 < best_c:
            best_h, best_c = h3.copy(), c3
            x_orig = np.linspace(0, 2, len(best_h))

    # Phase 4: Fourier init at best n
    n_best = len(best_h)
    if time.time() < deadline - 60:
        for fs in range(3):
            if time.time() > deadline - 40:
                break
            hf = make_fourier_init(n_best, seed + fs * 100)
            hf2, cf2 = coord_descent(hf, n_best, 80, deadline - 20, rng)
            if cf2 < best_c:
                best_h, best_c = hf2.copy(), cf2

    # Phase 5: Final intensive CD polish
    n_fin = len(best_h)
    h_fin, c_fin = coord_descent(best_h, n_fin, 200, deadline - 2, rng)
    if c_fin < best_c:
        best_h, best_c = h_fin.copy(), c_fin

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
