def solve_1d_all(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    # c_aff: array of size K (affine intercepts)
    # a_coeff: array of size K (affine slopes)
    # diag_local: index of diagonal constraint or None
    # unaff_max: constant upper bound from unaffected constraints
    # lo, hi: bounds on d
    
    K = len(a_coeff)
    if K == 0:
        return 0.0
    
    # Evaluate at endpoints
    candidates = [max(lo, min(hi, 0.0)), lo, hi]
    
    # For non-diagonal pairs: linear intersections
    # d* = (c_j - c_i) / (a_i - a_j)
    # Vectorize: for each i, compute d* with all j
    if K > 1:
        # Exclude diagonal from the linear pair computation
        non_diag = [i for i in range(K) if i != diag_local]
        if len(non_diag) > 1:
            cd = np.array(non_diag)
            ci = c_aff[cd][:, None]
            cj = c_aff[cd][None, :]
            ai = a_coeff[cd][:, None]
            aj = a_coeff[cd][None, :]
            da = ai - aj  # a_i - a_j
            dc = cj - ci  # c_j - c_i
            # d* = dc / da where da != 0
            valid = np.abs(da) > 1e-14
            d_star = np.where(valid, dc / np.where(valid, da, 1.0), np.nan)
            in_range = valid & (d_star > lo) & (d_star < hi) & ~np.isnan(d_star)
            if np.any(in_range):
                d_vals = d_star[in_range]
                candidates.extend(d_vals.tolist())
        
        # Diagonal pairs: quadratic
        if diag_local is not None:
            d_idx = diag_local
            for j in range(K):
                if j == d_idx:
                    continue
                # c_d + d*a_d - d² = c_j + d*a_j
                # d² + d*(a_j - a_d) + (c_j - c_d) = 0
                A = 1.0
                B = a_coeff[j] - a_coeff[d_idx]
                C = c_aff[j] - c_aff[d_idx]
                disc = B*B - 4*A*C
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-B+sq)/(2*A), (-B-sq)/(2*A)]:
                        if lo < d < hi:
                            candidates.append(d)
    
    # Also check 0
    candidates = [max(lo, min(hi, c)) for c in candidates]
    
    # Evaluate objective at all candidates
    best_d = candidates[0]
    best_val = obj_at(best_d, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates[1:]:
        v = obj_at(d, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    
    return best_d

import time

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_at(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_golden(c_aff, a_coeff, diag_local, unaff_max, lo, hi, maxiter=80):
    # Convex minimization using golden section search
    if lo >= hi:
        return 0.0
    try:
        res = golden(obj_at, lo, hi, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=maxiter)
        return float(res.x)
    except Exception:
        return 0.0

    """Coordinate descent using ALL affected lags (no top_k truncation)."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        n_pass_improvements = 0
        
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            
            # Determine affected lags
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            
            ms = np.arange(m_lo, m_hi + 1)
            
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            # Compute coefficients
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            
            c_aff = c[ms]
            
            # Find diagonal
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            
            lo = -h0
            hi = 1.0 - h0
            
            best_d = solve_1d_golden(c_aff, a_coeff, diag_local, unaff_max, lo, hi)

                    if new_val < current_max:
                        current_max = new_val
        
        h = project_feasible(h, n)
        if not improved:
            break
    
    return h, compute_c5(h, n)

def golden_search_1d(obj_fn, args, lo, hi, maxiter=100):
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_fn(c, *args)
    fd = obj_fn(d, *args)
    for _ in range(maxiter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_fn(c, *args)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_fn(d, *args)
    return (a + b) / 2
