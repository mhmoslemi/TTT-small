import math, time

def compute_c5(h,n):
    dx=2.0/n
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*dx)

def project_feasible(h,n): ...

def _sum_fun(x,n): return x.sum()-0.5*n
def _sum_jac(x,n): return np.ones(n)

def _slsqp_fun(x,n,alpha):
    dx=2.0/n
    c=np.correlate(x,1.0-x,mode='full')
    m=float(c.max())
    z=alpha*(c-m)
    s=np.sum(np.exp(z))
    return dx*(m + np.log(s)/alpha)

def _slsqp_jac(x,n,alpha):
    dx=2.0/n
    c=np.correlate(x,1.0-x,'full')
    m=float(c.max())
    z=alpha*(c-m)
    ez=np.exp(z)
    s=ez.sum()
    w=ez/s
    base=n-1
    t1=np.convolve(1.0-x, w, mode='full')[base:base+n]
    t2=np.correlate(w, x, mode='full')[2*base:base-1:-1]
    return dx*(t1-t2)

def slsqp_loop(h,n,alphas,maxiter,deadline):
    h=project_feasible(h,n)
    best_c=compute_c5(h,n)
    bounds=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':_sum_fun,'jac':_sum_jac,'args':(n,)}]
    for alpha in alphas:
       if time.time() > deadline - 15: break
       try:
         res=minimize(_slsqp_fun,h,args=(n,alpha),method='SLSQP',jac=_slsqp_jac,
                      bounds=bounds,constraints=cons,
                      options={'maxiter':maxiter,'ftol':1e-14})
         new=project_feasible(res.x,n)
         c=compute_c5(new,n)
         if c<best_c and np.isfinite(c):
             h,best_c=new,c
       except Exception:
         pass
    return h,best_c

import numpy as np
from scipy.optimize import minimize
import math
import time

GOLDEN = (math.sqrt(5.0) - 1.0) / 2.0


def compute_c5(h, n):
    dx = 2.0 / n
    v = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return float(v) if np.isfinite(v) else float("inf")


def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def _sum_fun(x, n):
    return x.sum() - 0.5 * n


def _sum_jac(x, n):
    return np.ones(n)


def _slsqp_fun(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    z = alpha * (c - m)
    s = np.sum(np.exp(z))
    return dx * (m + np.log(s) / alpha)


def _slsqp_jac(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    z = alpha * (c - m)
    ez = np.exp(z)
    w = ez / ez.sum()
    base = n - 1
    t1 = np.convolve(1.0 - x, w, mode="full")[base:base + n]
    t2 = np.correlate(w, x, mode="full")[2 * base:base - 1:-1]
    return dx * (t1 - t2)


def slsqp_loop(h, n, alphas, maxiter, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_fun, "jac": _sum_jac, "args": (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 12.0:
            break
        try:
            res = minimize(
                _slsqp_fun, best_h, args=(n, alpha), method="SLSQP",
                jac=_slsqp_jac, bounds=bounds, constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-14}
            )
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-15:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c


def _add_deriv(a, h, idx, k, sign, n):
    q1 = idx - k
    m1 = (q1 >= 0) & (q1 < n)
    a[m1] += sign * (1.0 - h[q1[m1]])
    q2 = idx + k
    m2 = (q2 >= 0) & (q2 < n)
    a[m2] += sign * (-h[q2[m2]])


def _pair_fval(c_aff, a, unaff, diag, cr1, cr2, d):
    dq = d * d
    vals = c_aff + d * a
    vals[diag] -= 2.0 * dq
    if cr1 >= 0:
        vals[cr1] += dq
    if cr2 >= 0:
        vals[cr2] += dq
    mx = float(vals.max())
    return mx if mx > unaff else unaff


def coord_descent(h, n, max_passes, deadline, gs_iters=22):
    h = project_feasible(h, n)
    base = n - 1
    L = 2 * n - 1
    target = 0.5 * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    skip_tol = 1e-9
    eps = 1e-12

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if abs(h.sum() - target) > 1e-8:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode="full")
        cur_c = float(c.max()) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        interior = np.minimum(h, 1.0 - h)
        if interior.max() < 1e-9:
            break
        r = int(np.argmax(interior))
        improved = False

        for p in range(n):
            if p == r:
                continue
            if time.time() > deadline:
                break
            Dp = max(p, base - p)
            Dr = max(r, base - r)
            D = Dp if Dp > Dr else Dr
            mlo = base - D
            mhi = base + D

            a_max = float(c[mlo:mhi + 1].max())
            u_max = 0.0
            if mlo > 0:
                v = float(c[:mlo].max())
                if v > u_max:
                    u_max = v
            if mhi < L - 1:
                v = float(c[mhi + 1:].max())
                if v > u_max:
                    u_max = v
            if u_max >= a_max - skip_tol:
                continue

            h0p = h[p]
            h0r = h[r]
            dlo = max(-h0p, h0r - 1.0) + eps
            dhi = min(1.0 - h0p, h0r) - eps
            if dhi - dlo < 1e-11:
                continue

            M = 2 * D + 1
            c_aff = c[mlo:mhi + 1]
            k = np.arange(-D, D + 1)
            a = np.zeros(M)
            _add_deriv(a, h, p, k, 1.0, n)
            _add_deriv(a, h, r, k, -1.0, n)

            diag = D
            kcr1 = r - p
            kcr2 = p - r
            cr1 = kcr1 + D if -D <= kcr1 <= D else -1
            cr2 = kcr2 + D if -D <= kcr2 <= D else -1

            f0 = a_max
            best_d = 0.0
            best_f = f0

                ft = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, dt)
                if ft < best_f:
                    best_f = ft
                    best_d = dt

            d_lo, d_hi = dlo, dhi
            d1 = d_hi - GOLDEN * (d_hi - d_lo)
            d2 = d_lo + GOLDEN * (d_hi - d_lo)
            f1 = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, d1)
            f2 = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, d2)
            if f1 < best_f:
                best_f, best_d = f1, d1
            if f2 < best_f:
                best_f, best_d = f2, d2

            for _it in range(gs_iters):
                if d_hi - d_lo < 1e-13:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - GOLDEN * (d_hi - d_lo)
                    f1 = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + GOLDEN * (d_hi - d_lo)
                    f2 = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, d2)
                if f1 < best_f:
                    best_f, best_d = f1, d1
                if f2 < best_f:
                    best_f, best_d = f2, d2

            dm = 0.5 * (d_lo + d_hi)
            fm = _pair_fval(c_aff, a, u_max, diag, cr1, cr2, dm)
            if fm < best_f:
                best_f, best_d = fm, dm

            if best_f < f0 - 1e-12:
                d = best_d
                if abs(d) < 1e-14:
                    continue
                new_vals = c_aff + d * a
                dq = d * d
                new_vals[diag] -= 2.0 * dq
                if cr1 >= 0:
                    new_vals[cr1] += dq
                if cr2 >= 0:
                    new_vals[cr2] += dq
                c[mlo:mhi + 1] = new_vals
                h[p] = h0p + d
                h[r] = h0r - d
                improved = True

        if not improved:
            break

    hfin = project_feasible(h, n)
    cfin = compute_c5(hfin, n)
    if cfin < best_c:
        return hfin, cfin
    return best_h, best_c


def refine(h, n, phase_end, alphas, maxiter, cd_passes, gs_iters):

def resample_h(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
