def resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:

        def obj(x):

            m = c.max()

            mx = shifted.max()

            grad_c = np.zeros((n, len(c)))
            # This is too expensive to compute fully. Use numerical gradient or skip.
            # Actually, let's just use SLSQP without gradient.
            return None
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:

    return best_h, best_c
