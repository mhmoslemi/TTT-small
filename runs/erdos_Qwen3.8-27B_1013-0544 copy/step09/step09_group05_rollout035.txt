    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)
            xh2 = min(x_hi, 1.0)
            if xl2 < r1:
                x_hi = min(x_hi, r1)
            elif xh2 > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def grad_smooth_max(h, n, alpha):
    """Compute subgradient of smooth max of correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        wk = w[k] * dx
        if d == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            if d > 0:
                jf = np.arange(d, n)
                jb = np.arange(0, n - d)
                grad[jf] += wk * (1.0 - h[jf - d])
                grad[jb] -= wk * h[jb + d]
            else:
                d = -d
                jf = np.arange(d, n)
                jb = np.arange(0, n - d)
                grad[jf] += wk * (1.0 - h[jf - d])
                grad[jb] -= wk * h[jb + d]
    grad -= np.mean(grad)
    return grad

def gradient_descent(h, n, deadline, max_iter=500, alpha=100.0, lr_init=0.02):
    h = h.copy()
    lr = lr_init
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        step = lr * np.linalg.norm(g)
        h_new = project_feasible(h - lr * g / (np.linalg.norm(g) + 1e-15), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
            lr = min(lr * 1.05, 0.2)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return h, best_c

def fourier_perturb(h, n, rng, amp_range=(0.02, 0.08), n_modes=4):
    half = n // 2
    free = h[:half].copy()
    x = np.arange(half)
    for k in range(1, min(n_modes + 1, half)):
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        free += amp * np.cos(2 * np.pi * k * x / n + phase)
    free = np.clip(free, 0.005, 0.995)
    h_new = np.empty(n)
    h_new[:half] = free
    h_new[half:] = 1.0 - free
    return project_feasible(h_new, n)

def make_symmetric(h, n):
    half = n // 2
    free = 0.5 * (h[:half] + (1.0 - h[half:2*half]))
    h_sym = np.empty(n)
    h_sym[:half] = free
    h_sym[half:n] = 1.0 - free
    return project_feasible(h_sym, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n_init = 700
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase A: Coordinate descent on initial
    h_a, c_a = coord_descent(best_h, n_init, 200, deadline, rng, True)
    if c_a < best_c:
        best_h, best_c = h_a, c_a
    # Phase B: Gradient + CD alternation with Fourier perturbations
    n = len(best_h)
    for round_i in range(8):
        if time.time() > deadline - 120:
            break
        alpha = 50.0 * (round_i + 1)
        h_g, c_g = gradient_descent(best_h, n, deadline - 200, max_iter=200, alpha=alpha)
        h_c, c_c = coord_descent(h_g, n, 100, deadline - 100, rng, True)
        if c_c < best_c:
            best_h, best_c = h_c, c_c
        # Fourier perturbation to escape
        if time.time() > deadline - 120:
            break
        h_p = fourier_perturb(best_h, n, rng)
        h_c2, c_c2 = coord_descent(h_p, n, 150, deadline - 100, rng, True)
        if c_c2 < best_c:
            best_h, best_c = h_c2, c_c2
    # Phase C: Try symmetric version
    if time.time() < deadline - 150:
        h_sym = make_symmetric(best_h, n)
        h_s, c_s = coord_descent(h_sym, n, 300, deadline - 80, rng, True)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
    # Phase D: Try different n values
    n_cands = [350, 500, 420, 560, 840]
    n_cands = [x for x in n_cands if x != n and x % 2 == 0]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_cands:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 200, deadline - 30, rng, True)
        h_g2, c_g2 = gradient_descent(h_o, n_try, deadline - 20, max_iter=100, alpha=200.0)
        h_o2, c_o2 = coord_descent(h_g2, n_try, 100, deadline - 10, rng, True)
        if c_o2 < best_c:
            best_h, best_c = h_o2, c_o2
            x_orig = np.linspace(0, 2, n_try)
    # Phase E: Final heavy polish
    n = len(best_h)
    if time.time() < deadline - 5:
        remaining = max(10, deadline - time.time() - 2)
        max_p = max(50, int(remaining / 0.03))
        hf, cf = coord_descent(best_h, n, max_p, deadline, rng, False)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
