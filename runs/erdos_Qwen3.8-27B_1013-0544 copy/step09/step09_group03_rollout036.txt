```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_smooth_max(h, alpha):
    """Gradient of soft-max alpha approximation of max(c)."""
    n = len(h)
    c = compute_correlation(h)
    m = c.max()
    # For numerical stability
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        kk = k - (n - 1)
        if kk >= 0:
            grad[:n - kk] -= w[k] * h[kk:]
            grad[kk:] += w[k] * (1.0 - h[:n - kk])
        else:
            kk2 = -kk
            grad[:n - kk2] += w[k] * (1.0 - h[kk2:])
            grad[kk2:] -= w[k] * h[:n - kk2]
    return grad

def smooth_grad_descent(h, n, alphas, steps_per_alpha, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alphas:
        for step in range(steps_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            grad = grad_smooth_max(h, alpha)
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Backtracking line search
            base = h
            obj_base = compute_c5(h, n)
            alpha_step = 0.1
            improved = False
            for _ in range(20):
                h_new = project_feasible(base - alpha_step * grad, n)
                obj_new = compute_c5(h_new, n)
                if obj_new < obj_base - 1e-12:
                    h = h_new
                    improved = True
                    break
                alpha_step *= 0.5
            if not improved:
                break
            if compute_c5(h, n) < best_c:
                best_c = compute_c5(h, n)
                best_h = h.copy()
    return best_h, best_c
```

Wait, the backtracking line search is expensive because each evaluation requires computing the full correlation O(n^2). For n=600, that's 360,000 operations per evaluation, and with 20 backtracking steps, that's 7.2M per gradient step. With 100 gradient steps, that's 720M, which might be too slow.

Let me optimize: I can compute the correlation more efficiently using FFT for the autocorrelation part.

c[k] = sum_n h[n+kk] * (1-h[n]) = sum_n h[n+kk] - sum_n h[n+kk]*h[n]

The first term is just a partial sum of h (easy).
The second term is the autocorrelation of h, which can be computed via FFT in O(n log n).

Let me implement that:

```python
def compute_correlation_fast(h):
    n = len(h)
    # Autocorrelation via FFT
    N = 2 * n
    F = np.fft.rfft(h, N)
    ac = np.fft.irfft(F * np.conj(F), N)[:n]
    # ac[lag] = sum_n h[n]*h[n+lag] for lag >= 0
    
    # c[k] for shift kk = k - (n-1):
    # If kk >= 0: c = sum_{n=0}^{n-1-kk} h[n+kk] - ac[kk]
    #             = (sum(h[kk:]) - sum(h[:kk])) ... 
    # Actually: sum_{n=0}^{n-1-kk} h[n+kk] = sum(h[kk:])
    # And ac[kk] = sum_{n=0}^{n-1-kk} h[n]*h[n+kk]
    # So c[kk] = sum(h[kk:]) - ac[kk]
    
    # If kk < 0 (kk2 = -kk > 0):
    # c[-kk2] = sum_{n=0}^{n-1-kk2} h[n]*(1-h[n+kk2])
    #          = sum(h[:n-kk2]) - sum_{n=0}^{n-1-kk2} h[n]*h[n+kk2]
    #          = sum(h[:n-kk2]) - ac[kk2]
    
    c = np.zeros(2 * n - 1)
    ps = np.concatenate([[0], np.cumsum(h)])  # partial sums
    for kk in range(0, n):
        c[kk + n - 1] = (ps[n] - ps[kk]) - ac[kk]  # sum(h[kk:]) - ac[kk]
        c[n - 1 - kk] = (ps[n - kk] - ps[0]) - ac[kk]  # sum(h[:n-kk]) - ac[kk]
    # Fix: c[n-1] (kk=0) is computed twice, should be sum(h) - ac[0]
    c[n-1] = h.sum() - ac[0]
    return c
```

So my formula is correct. Now for the gradient, can I also speed it up?

The gradient of c[kk] (shift kk≥0) with respect to h[m]:
= I(m≥kk)(1-h[m-kk]) - I(m<n-kk)h[m+kk]

The weighted sum over k:
grad[m] = sum_k w[k] * [I(m≥kk_k)(1-h[m-kk_k]) - I(m<n-kk_k)h[m+kk_k]]

where kk_k is the shift for the k-th correlation value.

This is still O(n²) in general. But for the purposes of this optimization, with n=600 and maybe a few hundred iterations, it should be fine (600² = 360K per gradient, times maybe 200 iterations = 72M, very fast in numpy).

Let me think about what will actually give the best improvement. I think the key insight is:

1. The coordinate descent gets stuck because it optimizes one coordinate at a time, and the landscape has many local minima.
2. A gradient-based method that considers the global structure of the max can potentially escape these local minima.
3. But we need to be careful about the step size and the smoothing parameter.

Let me also consider: what if I use a "simulated annealing" style approach?
- Start with a large temperature (smooth objective)
- Do several gradient steps
- Decrease temperature
- Repeat

This allows large moves early on (when the objective is smooth) and fine adjustments later (when the objective is close to the true max).

Additionally, I can add random perturbations to escape local minima:
- Every K steps, add a small random perturbation to h
- Project back to feasible set
- Continue

Hmm wait, I realize I'm overcomplicating this. Let me look at what the previous solution does and just modify it slightly.

The previous solution's exact_coord_min function computes, for a given coordinate p, the value of h[p] that minimizes the maximum correlation. It does this by:
1. Computing the linear dependence of each c[k] on h[p] (the coefficient a[k] and constant b[k])
2. Finding the value that minimizes max(a[k]*x + b[k]) subject to 0≤x≤1
3. Using binary search

This is already very efficient and exact for the coordinate descent. The issue is that coordinate descent is a local method.

My improvement idea: add a "global" phase that uses the smoothed gradient to potentially escape local optima, then follow up with coordinate descent.

Let me also think about a different angle: what if I try to directly construct a better initial solution?

For the Erdős problem, there's a known connection to Sidon sets and B_h sequences. The optimal set is related to a "rete" (a construction by Lindström and Cantor-like sets).

For a simple construction: take h to be the indicator of a "thickened" Cantor-like set. For example:
- Start with [0, 2]
- Remove the middle third → [0, 2/3] ∪ [4/3, 2]
- Remove the middle third of each → [0, 2/9] ∪ [4/9, 2/3] ∪ [4/3, 10/9] ∪ [14/9, 2]
- etc.

After k iterations, the total length is 2*(2/3)^k. We want this to be 1, so 2*(2/3)^k = 1, giving (2/3)^k = 1/2, k = log(1/2)/log(2/3) ≈ 1.71.

So after about 2 iterations of the Cantor construction, we'd have a set of measure close to 1. The discrete version would be a binary h with a specific pattern.

But this is just a heuristic starting point. The actual optimization should refine it.

Let me focus on writing efficient code that:
1. Uses the parent solution as a starting point
2. Applies smoothed gradient descent with annealing
3. Uses coordinate descent for polishing
4. Tries a few different approaches

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    F = np.fft.rfft(h, N)
    ac = np.fft.irfft(F * np.conj(F), N)[:n]
    ps = np.concatenate([[0.0], np.cumsum(h)])
    c = np.zeros(2 * n - 1)
    for kk in range(n):
        c[n - 1 + kk] = (ps[n] - ps[kk]) - ac[kk]
        if kk > 0:
            c[n - 1 - kk] = ps[n - kk] - ac[kk]
    return c

def grad_smooth_max(h, alpha):
    n = len(h)
    c = compute_correlation(h)
    m = c.max()
    w = np.exp(alpha * (c - m))
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        kk = k - (n - 1)
        if kk >= 0:
            grad[:n - kk] -= wk * h[kk:]
            grad[kk:] += wk * (1.0 - h[:n - kk])
        else:
            kk2 = -kk
            grad[:n - kk2] += wk * (1.0 - h[kk2:])
            grad[kk2:] -= wk * h[:n - kk2]

                return best_h, best_c

    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c[n - 1] - h0 * (1.0 - h0) + x * (1.0 - x))
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_correlation(h)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick CD baseline
    if time.time() < deadline - 500:
        h1, c1 = coord_descent(best_h, n, 2000, deadline - 480, seed)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Smoothed gradient annealing
    if time.time() < deadline - 350:
        h2, c2 = smoothed_grad_descend(best_h, n, deadline - 330,
                                       alphas=[3, 10, 30, 100, 300, 1000, 3000],
                                       steps_per_alpha=150, seed=seed + 100)
        if c2 < best_c:
            best_h, best_c = h2, c2
        if time.time() < deadline - 300:
            h2b, c2b = coord_descent(best_h, n, 800, deadline - 290, seed + 200)
            if c2b < best_c:
                best_h, best_c = h2b, c2b
    # Phase 3: Targeted gradient on active constraints
    if time.time() < deadline - 220:
        h3, c3 = smoothed_grad_descend(best_h, n, deadline - 200,
                                       alphas=[100, 500, 2000, 5000, 20000],
                                       steps_per_alpha=100, seed=seed + 300)
        if c3 < best_c:
            best_h, best_c = h3, c3
        if time.time() < deadline - 180:
            h3b, c3b = coord_descent(best_h, n, 1000, deadline - 170, seed + 400)
            if c3b < best_c:
                best_h, best_c = h3b, c3b
    # Phase 4: Different grid sizes
    x_orig = np.linspace(0, 2, len(best_h))
    n_trials = [400, 500, 700, 800, 1000, 1200]
    for n_try in n_trials:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 600, deadline - 100, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 5: Multi-restart with gradient + CD
    n_cur = len(best_h)
    for r in range(20):
        if time.time() > deadline - 50:
            break
        s = 0.008 + 0.012 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = smoothed_grad_descend(hp, n_cur, deadline - 30,
                                         alphas=[100, 500, 2000],
                                         steps_per_alpha=50, seed=seed + 500 + r * 7)
        h_o, c_o = coord_descent(h_o, n_cur, 300, deadline - 20, seed + 600 + r * 7)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 6: Final CD
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        hf, cf = coord_descent(best_h, n_cur, 5000, deadline - 3, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Check 1-h symmetry
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
```
