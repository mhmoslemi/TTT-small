Actually, the gradient of c[k] with respect to h can be computed more efficiently using the correlation structure.

c[k] = <h, shift_k(1-h)> where shift_k shifts by k-(n-1) positions.

Alternatively, let me use the fact that:
c = correlate(h, 1-h)

The gradient of c[k] with respect to h[i] is:
∂c[k]/∂h[i] = (1-h)[i + k - (n-1)] - h[i - k + (n-1)]

(where out-of-bounds terms are 0)

So the gradient of the smooth objective ∑_k w[k]*c[k] with respect to h[i] is:
∑_k w[k] * ((1-h)[i + k-(n-1)] - h[i - k + (n-1)])

Let me define s = k - (n-1), then k ranges from 0 to 2n-2, and s ranges from -(n-1) to (n-1).

∑_s w[n-1+s] * ((1-h)[i+s] - h[i-s])

= ∑_s w[n-1+s] * (1-h)[i+s] - ∑_s w[n-1+s] * h[i-s]

Let w_s = w[n-1+s] for s in [-(n-1), n-1].

First term: ∑_s w_s * (1-h)[i+s] = ∑_j (1-h)[j] * w_{j-i} = ((1-h) * w)[i] (convolution)
Second term: ∑_s w_s * h[i-s] = ∑_j h[j] * w_{i-j} = (h * w')[i] where w'_s = w_{-s}

So grad[i] = ((1-h) * w)[i] - (h * w')[i] where * denotes convolution.

Let me code this up now. I'll keep it clean and under 300 lines.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, h0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def cd_pass(h, n, update_interval=None, seed=0, direction=1):
    """Single CD pass with optional periodic c updates."""
    c = np.correlate(h, 1.0 - h, mode='full')
    n_updates = 0
    order = np.arange(n) if direction == 1 else np.arange(n-1, -1, -1)
    if seed > 0 and direction == 0:
        rng = np.random.RandomState(seed)
        order = rng.permutation(n)
    count = 0
    for p in order:
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        new_val = exact_coord_min(c, p, h, n)
        if abs(new_val - h0) > 1e-10:
            h[p] = new_val
            n_updates += 1
        count += 1
        if update_interval and count % update_interval == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, n_updates

def coord_descent(h, n, max_passes, deadline, seed=0, update_interval=None):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if time.time() > deadline:
            break
        direction = it % 3  # 0=forward, 1=backward, 2=random
        h, n_updates = cd_pass(h, n, update_interval, seed+rng.randint(1000), direction)
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            stall = 0
        else:
            stall += 1
        if stall > 20 and n_updates < n * 0.005:
            break
    return h, best_c

def smooth_step(h, n, alpha, lr, deadline, max_iter=5000):
    """Nesterov-accelerated gradient step on smooth objective."""
    h = h.copy()
    m = n  # velocity dimension
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

            break

        mx = shifted.max()

        w = exp_s / exp_s.sum()

        f = 1.0 - h
        for s_idx in range(2*n-1):

            wi = w[s_idx]

                grad[s:] -= wi * h[:n-s]
            else:
