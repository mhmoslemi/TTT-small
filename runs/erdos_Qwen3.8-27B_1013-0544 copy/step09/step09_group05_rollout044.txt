        improved = False
        if random_order and rng is not None:

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_direction(h, n, rng):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.where(c == c.max())[0]
    p = idx[rng.randint(len(idx))]
    grad = np.zeros(n)
    for i in range(n):
        jf = i + p - (n - 1)
        jb = i - p + (n - 1)
        if 0 <= jf < n:
            grad[i] += (1.0 - h[jf])
        if 0 <= jb < n:
            grad[i] -= h[jb]
    if p == n - 1:
        for i in range(n):
            grad[i] += (1.0 - 2.0 * h[i])
    grad = grad - np.mean(grad)
    return grad
