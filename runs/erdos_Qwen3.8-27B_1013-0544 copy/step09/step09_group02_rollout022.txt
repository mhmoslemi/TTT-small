import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=500, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if p_idx % 25 == 24:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    weights = exp_s / exp_s.sum()
    dx = 2.0 / n
    n_c = len(c)
    p_vals = np.arange(n_c) - (n - 1)
    grad = np.zeros(n)
    for idx in range(n_c):
        w = weights[idx]
        if w < 1e-16:
            continue
        p = int(p_vals[idx])
        lo_b = max(0, p)
        hi_b = min(n, n + p)
        if hi_b > lo_b:
            grad[lo_b:hi_b] += w * (1.0 - h[lo_b - p:hi_b - p])
        lo_f = max(0, -p)
        hi_f = min(n, n - p)
        if hi_f > lo_f:
            grad[lo_f:hi_f] -= w * h[lo_f + p:hi_f + p]
    return grad * dx, (m + np.log(exp_s.sum()) / alpha) * dx

def pgd_smooth(h, n, max_iter=5000, alpha=10000, deadline=None, lr_init=1.0):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        grad, _ = smooth_max_grad(h, n, alpha)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new
            stall = 0
            lr = min(lr * 1.03, 20.0)
        else:
            lr *= 0.82
            stall += 1
            if stall > 60 or lr < 1e-12:
                break
    return best_h, best_c

def mirror_pgd(h, n, max_iter=3000, top_k=8, deadline=None, lr_init=0.3):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        w = 1.0 / top_k
        p_vals = np.arange(len(c)) - (n - 1)
        grad = np.zeros(n)
        for idx in top_idx:
            p = int(p_vals[idx])
            lo_b = max(0, p)
            hi_b = min(n, n + p)
            if hi_b > lo_b:
                grad[lo_b:hi_b] += w * (1.0 - h[lo_b - p:hi_b - p])
            lo_f = max(0, -p)
            hi_f = min(n, n - p)
            if hi_f > lo_f:
                grad[lo_f:hi_f] -= w * h[lo_f + p:hi_f + p]
        grad *= 2.0 / n
        log_h = np.log(np.maximum(h, 1e-15))
        log_h_new = log_h - lr * grad
        log_h_new -= log_h_new.max()
        h_try = np.exp(log_h_new)
        h_try *= (n / 2.0) / h_try.sum()
        h_new = project_feasible(h_try, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new
            stall = 0
            lr = min(lr * 1.04, 5.0)
        else:
            lr *= 0.85
            stall += 1
            if stall > 70 or lr < 1e-12:
                break
    return best_h, best_c
