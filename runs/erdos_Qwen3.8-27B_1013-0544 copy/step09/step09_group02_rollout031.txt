        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for i in order:
            h0 = h[i]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            mlo = max(0, n - 1 - i)
            mhi = min(n_eff - 1, n - 1 + i)
            ms = np.arange(mlo, mhi + 1)
            shifts = ms - (n - 1)
            jf = i + shifts
            jb = i - shifts
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            vec = np.zeros(len(ms))
            if np.any(vf):
                vec[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                vec[vb] -= h[jb[vb]]
            c_sub = c[mlo:mhi + 1]
            unaff = 0.0
            if mlo > 0:
                unaff = max(unaff, float(c[:mlo].max()))
            if mhi < n_eff - 1:
                unaff = max(unaff, float(c[mhi + 1:].max()))
            lo, hi = -h0, 1.0 - h0
            for _ in range(32):
                mid = (lo + hi) * 0.5
                vals = c_sub + mid * vec
                mval = float(vals.max())
                if mval <= unaff + 1e-13:
                    lo = hi = mid
                    break
                active = vals >= mval - 1e-10
                if not np.any(active):
                    lo = hi = mid
                    break
                vmin = float(vec[active].min())
                vmax = float(vec[active].max())
                if vmin > 1e-10:
                    hi = mid
                elif vmax < -1e-10:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            delta = (lo + hi) * 0.5
            if abs(delta) > 1e-12:
                c[ms] += delta * vec
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p % 15 == 14:
            c = np.correlate(h, 1.0 - h, mode='full')
            h = project_feasible(h, n)
    return h, compute_c5(h, n)

def top_k_gradient(h, n, k):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for ki in top_idx:
        shift = ki - (n - 1)
        if shift >= 0:
            idx = np.arange(n - shift)
            grad[idx] += (1.0 - h[idx + shift])
            grad[idx + shift] -= h[idx]
        elif shift < 0:
            s = -shift
            idx = np.arange(n - s)
            grad[idx + s] += (1.0 - h[idx])
            grad[idx] -= h[idx + s]
    return grad / k

def annealed_topk(h, n, deadline, rng=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    k_list = [80, 40, 25, 15, 10, 5, 3, 1]
    for k in k_list:
        phase_deadline = min(deadline, time.time() + 80)
        lr = 0.15
        stall = 0
        for it in range(150):
            if time.time() > phase_deadline:
                break
            grad = top_k_gradient(h, n, k)
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                stall = 0
            else:
                lr *= 0.8
                stall += 1
                if stall > 35 or lr < 1e-9:
                    break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=400, deadline=None):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        grad = np.zeros(n)
        for ki in range(len(c)):
            if w[ki] < 1e-12:
                continue
            shift = ki - (n - 1)
            if shift >= 0:
                idx = np.arange(n - shift)
                grad[idx] += w[ki] * (1.0 - x[idx + shift])
                grad[idx + shift] -= w[ki] * x[idx]
            elif shift < 0:
                s = -shift
                idx = np.arange(n - s)
                grad[idx + s] += w[ki] * (1.0 - x[idx])
                grad[idx] -= w[ki] * x[idx + s]
        return grad * (2.0 / n)

    def sc(x):
        return x.sum() - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def slsqp_topk(h, n, k, maxiter=400, deadline=None):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        top = np.sort(c)[::-1][:k]
        return top.mean() * (2.0 / n)

    def obj_grad(x):
        g = top_k_gradient(x, n, k)
        return g * (2.0 / n)

    def sc(x):
        return x.sum() - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 600
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Establish baseline with CD
    h1, c1 = coord_cd(best_h, n, max_passes=150, deadline=deadline, rng=rng, shuffle=True)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Multi-resolution exploration
    for n_try in [400, 500, 700, 800, 1000]:
        if time.time() > deadline - 250:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = coord_cd(h_interp, n_try, max_passes=100, deadline=deadline, rng=rng, shuffle=True)
        if c_t < best_c:
            x_back = np.linspace(0, 2, n)
            cand = project_feasible(np.interp(x_back, x_new, h_t), n)
            cand_c = compute_c5(cand, n)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c

    # Phase 3: Top-k annealing (different optimization path)
    if time.time() < deadline - 120:
        h3, c3 = annealed_topk(best_h, n, deadline=time.time() + 150, rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: CD refinement
    h4, c4 = coord_cd(best_h, n, max_passes=300, deadline=deadline, rng=rng, shuffle=True)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: SLSQP with smooth max (multiple alphas)
    if time.time() < deadline - 100:
        for alpha in [3000, 10000, 50000]:
            if time.time() > deadline - 40:
                break
            h5, c5 = slsqp_smooth(best_h, n, alpha, maxiter=500, deadline=deadline)
            if c5 < best_c:
                best_h, best_c = h5, c5
            hb, cb = coord_cd(best_h, n, max_passes=150, deadline=deadline, rng=rng, shuffle=True)
            if cb < best_c:
                best_h, best_c = hb, cb

    # Phase 6: SLSQP with top-k
    if time.time() < deadline - 80:
        for k in [3, 5, 10]:
            if time.time() > deadline - 30:
                break
            h6, c6 = slsqp_topk(best_h, n, k, maxiter=400, deadline=deadline)
            if c6 < best_c:
                best_h, best_c = h6, c6
            hb, cb = coord_cd(best_h, n, max_passes=100, deadline=deadline, rng=rng, shuffle=True)
            if cb < best_c:
                best_h, best_c = hb, cb

    # Phase 7: Structured perturbations for diversity
    for pert in range(12):
        if time.time() > deadline - 50:
            break
        if pert < 4:
            x = np.linspace(0, 2, n)
            freq = 2.0 + 2.0 * pert
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.005 + 0.005 * pert
            pert_vec = amp * np.sin(np.pi * freq * x + phase)
        elif pert < 8:
            pert_vec = np.zeros(n)
            bs = rng.randint(0, n // 2)
            bw = rng.randint(20, n // 5)
            pert_vec[bs:bs + bw] = (0.01 + 0.005 * (pert - 4)) * rng.choice([-1.0, 1.0])
        else:
            scale = 0.003 + 0.003 * (pert - 8)
            pert_vec = scale * rng.randn(n)
        hp = project_feasible(best_h + pert_vec, n)
        hp2, cp2 = coord_cd(hp, n, max_passes=120, deadline=deadline, rng=rng, shuffle=True)
        if cp2 < best_c:
            best_h, best_c = hp2, cp2

    # Phase 8: Final intensive CD
    h8, c8 = coord_cd(best_h, n, max_passes=500, deadline=deadline, rng=rng, shuffle=True)
    if c8 < best_c:
        best_h, best_c = h8, c8

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 5:
        hf, cf = slsqp_smooth(best_h, n, 200000, maxiter=600, deadline=deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n, max_passes=300, deadline=deadline, rng=rng, shuffle=True)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
