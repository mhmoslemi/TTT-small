import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coeff_matrix(h, n, block_arr, top_idx):
    p_vals = block_arr[:, None].astype(int)
    k_vals = top_idx[None, :].astype(int)
    n1 = n - 1
    jf = p_vals + k_vals - n1
    jb = p_vals - k_vals + n1
    jf_c = np.clip(jf, 0, n - 1)
    jb_c = np.clip(jb, 0, n - 1)
    af = np.where((jf >= 0) & (jf < n), 1.0 - h[jf_c], 0.0)
    ab = np.where((jb >= 0) & (jb < n), -h[jb_c], 0.0)
    return af + ab

def block_lp_cd(h, n, deadline, block_size, top_k, max_iter, rng, random_frac=0.3):
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]
        interior = np.where((h > 0.005) & (h < 0.995))[0]
        if len(interior) < block_size:
            break
        if rng.random() < random_frac:
            block = list(rng.choice(interior, block_size, replace=False))
        else:
            coeff = coeff_matrix(h, n, interior, top_idx)
            scores = np.max(np.abs(coeff), axis=1)
            top_i = np.argsort(scores)[::-1][:block_size]
            block = list(interior[top_i])
        block_arr = np.array(block)
        nb = len(block)
        b_val = h[block_arr].copy()
        A_coeff = coeff_matrix(h, n, block_arr, top_idx)
        A_ub = np.zeros((top_k, nb + 1))
        b_ub = np.zeros(top_k)
        A_ub[:, :nb] = A_coeff.T
        A_ub[:, nb] = -1.0
        b_ub = -c[top_idx]
        bounds = [(-b_val[j], 1.0 - b_val[j]) for j in range(nb)]
        bounds.append((0.0, None))
        A_eq = np.zeros((1, nb + 1))
        A_eq[0, :nb] = 1.0
        c_obj = np.zeros(nb + 1)
        c_obj[nb] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        t_new = res.x[nb]
        if t_new >= current_max - 1e-11:
            break
        h[block_arr] = b_val + res.x[:nb]
        h = project_feasible(h, n)
    return h, compute_c5(h, n)[0]

def cd_1d(h, n, deadline, top_k, max_iter):
    h = h.copy()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(top_k)
            diag_idx = None
            for idx in range(top_k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            candidates = [lo, hi, 0.0]
            for i in range(top_k):
                for j in range(i + 1, top_k):
                    if i == diag_idx or j == diag_idx:
                        continue
                    da = a_coeff[i] - a_coeff[j]
                    if abs(da) > 1e-15:
                        d = (c_peak[j] - c_peak[i]) / da
                        if lo <= d <= hi:
                            candidates.append(d)
            if diag_idx is not None:
                for i in range(top_k):
                    if i == diag_idx:
                        continue
                    B = -(a_coeff[diag_idx] - a_coeff[i])
                    C = -(c_peak[diag_idx] - c_peak[i])
                    disc = B * B - 4.0 * C
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        d1 = (-B + sq) / 2.0
                        d2 = (-B - sq) / 2.0
                        if lo <= d1 <= hi:
                            candidates.append(d1)
                        if lo <= d2 <= hi:
                            candidates.append(d2)
            for i in range(top_k):
                if abs(a_coeff[i]) > 1e-15:
                    d = (unaff_max - c_peak[i]) / a_coeff[i]
                    if lo <= d <= hi:
                        candidates.append(d)
            best_d, best_val = 0.0, current_max
            for d in candidates:
                d = float(np.clip(d, lo, hi))
                vals = c_peak + d * a_coeff
                if diag_idx is not None:
                    vals[diag_idx] -= d * d
                v = max(float(vals.max()), unaff_max)
                if v < best_val:
                    best_val = v
                    best_d = d
            if best_val < current_max - 1e-13:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)[0]
