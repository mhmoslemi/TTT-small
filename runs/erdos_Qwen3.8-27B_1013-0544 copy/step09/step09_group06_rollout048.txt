def grad_c_s(h, n, s):
    grad = np.zeros(n)
    # (1-h[p+s]) for valid p+s
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])

    # -h[p-s] for valid p-s
    if s >= 0:
        grad[s:] -= h[:n-s]
    else:
        grad[:n+s] -= h[-s:]
    return grad

        h2, c2 = multi_cd(best_h, n, 400, dl, 50, rng, frontier_only=True)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP (120s)
    if remaining() > 120 + 150:
        dl = time.time() + 120
        for alpha in [100000, 500000]:
            if time.time() > dl:
                break
            h2, c2 = slsqp_run(best_h, n, alpha, 350)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 4: CD after SLSQP (80s)
    if remaining() > 80 + 100:
        dl = time.time() + 80
        h2, c2 = multi_cd(best_h, n, 300, dl, 60, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 5: Try higher resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [800, 1000]:
        if remaining() < 120 + 60:
            break
        dl = time.time() + 100
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h2, c2 = multi_cd(h_int, n_try, 200, dl, 30, rng)
        h2, c2 = try_symmetry(h2, n_try)
        if c2 < best_c:
            best_h, best_c = h2, c2
            x_orig = x_new

    # Phase 6: Final deep CD
    n_cur = len(best_h)
    if remaining() > 50:
        dl = time.time() + min(50, remaining() - 10)
        h2, c2 = multi_cd(best_h, n_cur, 800, dl, 60, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Final symmetry + projection
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
