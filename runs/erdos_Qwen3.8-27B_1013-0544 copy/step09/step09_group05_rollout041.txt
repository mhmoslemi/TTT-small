import numpy as np

def project_feasible(h,n):
    target=0.5*float(n)
    h=np.clip(np.array(h,dtype=float,copy=True),0.0,1.0)
    s=float(h.sum())
    if abs(s-target)<1e-13:
        return h
    lo=float(h.min())-1.0
    hi=float(h.max())+1.0
    for _ in range(70):
        mid=0.5*(lo+hi)
        t=float(np.clip(h-mid,0.0,1.0).sum())
        if t>target:
            lo=mid
        else:
            hi=mid
    h=np.clip(h-0.5*(lo+hi),0.0,1.0)
    r=target-float(h.sum())
    if abs(r)>1e-13:
        m=(h>1e-12)&(h<1.0-1e-12)
        if m.any():
            pos=np.where(m)[0][0]
            h[pos]+=r
        else:
            h+=r/float(n)
            h=np.clip(h,0.0,1.0)
    return h

def compute_c5(h,n):

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(70):
        md = 0.5 * (lo + hi)
        if float(np.clip(h - md, 0.0, 1.0).sum()) > target:
            lo = md
        else:
            hi = md
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(h.sum())
    if abs(r) > 1e-13:
        msk = (h > 1e-12) & (h < 1.0 - 1e-12)
        if msk.any():
            h[np.nonzero(msk)[0][0]] += r
        else:
            h = np.clip(h + r / float(n), 0.0, 1.0)
    return h

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))

def make_plan(n):
    idx = np.arange(n, dtype=np.int32)
    p = idx[:, None]
    src = p + idx[None, :]
    tgt = (n - 1) + idx[None, :] - p
    lo = np.minimum(src[:, 0], tgt[:, 0])
    hi = np.maximum(src[:, -1], tgt[:, -1])
    src_off = src - lo[:, None]
    tgt_off = tgt - lo[:, None]
    cr = (n - 1) - lo
    j_src = (n - 1) - idx
    i_tgt = idx
    return (lo.astype(np.int32), hi.astype(np.int32), src_off.astype(np.int32),
            tgt_off.astype(np.int32), cr.astype(np.int32), j_src.astype(np.int32),
            i_tgt.astype(np.int32))

def eval_sub_vals(csub, a, cr, cc, x0, xs):
    d = xs - x0
    vals = csub[:, None] + a[:, None] * d[None, :]
    if cr >= 0:
        vals[cr, :] = cc + xs * (1.0 - xs) - x0 * (1.0 - x0)
    return np.max(vals, axis=0)

def solve_x_sub(csub, a, cr, cc, x0, off, xs1, K2, K3, tol=1e-8):
    ms = float(np.max(csub))
    if off > ms - tol:
        return x0, (off if off > ms else ms), False
    cur = ms
    bx = x0
    bv = cur
    v = eval_sub_vals(csub, a, cr, cc, x0, xs1)
    if off > -1e90:
        v = np.maximum(v, off)
    i = int(np.argmin(v))
    val = float(v[i])
    if val < bv:
        bv = val
        bx = float(xs1[i])
    step = 1.0 / float(len(xs1) - 1)
    ll = max(0.0, x0 - 2.0 * step)
    hh = min(1.0, x0 + 2.0 * step)
    if hh - ll > 1e-12:
        xl = np.linspace(ll, hh, 17)
        vl = eval_sub_vals(csub, a, cr, cc, x0, xl)
        if off > -1e90:
            vl = np.maximum(vl, off)
        j = int(np.argmin(vl))
        val = float(vl[j])
        if val < bv:
            bv = val
            bx = float(xl[j])
    if bv >= cur - tol:
        return x0, cur, False
    w = 2.0 * step
    lo = max(0.0, bx - w)
    hi = min(1.0, bx + w)
    if hi - lo < 1e-12:
        lo = max(0.0, bx - step)
        hi = min(1.0, bx + step)
    xr = np.linspace(lo, hi, K2)
    vr = eval_sub_vals(csub, a, cr, cc, x0, xr)
    if off > -1e90:
        vr = np.maximum(vr, off)
    j = int(np.argmin(vr))
    val = float(vr[j])
    if val < bv:
        bv = val
        bx = float(xr[j])
    if bv >= cur - tol:
        return x0, cur, False
    w = 2.0 * (hi - lo) / float(max(1, K2 - 1))
    lo2 = max(lo, bx - w)

    if hi2 - lo2 < 1e-12:

    x3 = np.linspace(lo2, hi2, K3)

    val = float(v3[j])

def coord_refine(h, n, deadline, rng=None, random=True, K1=41, K2=33, K3=33):

    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full')
    dx = 2.0 / float(n)
    best_c = float(np.max(c) * dx)
    best_h = h.copy()
    if random and rng is None:
        rng = np.random.RandomState(123)
    lo_p, hi_p, src_off, tgt_off, cr_p, j_src, i_tgt = make_plan(n)
    L = 2 * n - 1

def interp_h(h, n):
    m = len(h)
    if m == n:
        return project_feasible(h.copy(), n)
    xo = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), n)

def make_variants(h, n, rng):
    out = [h[::-1].copy(), 0.5 * (h + 1.0 - h[::-1]), 1.0 - h]
    if n > 1:

        idx = np.argpartition(h, k - 1)[:k]
        b = np.zeros(n, dtype=float)
        b[idx] = 1.0
        out.append(b)
        out.append(1.0 - b)
    return [project_feasible(v, n) for v in out]

    idx = np.argpartition(v, k - 1)[:k]
    b = np.zeros(n, dtype=float)

    return project_feasible(b, n)
