def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    sw = w.sum()
    obj = (m + np.log(sw) / alpha) * (2.0 / n)
    w_norm = w / sw

    N = 1

    U = np.fft.rfft(1.0 - x, N)

    h_rev = x[::-1]
    HR = np.fft.rfft(h_rev, N)
    T2_full = np.fft.irfft(W * HR, N)
    T2 = T2_full[n-1:2*n-1]

    return obj, grad

            dp_aff = dp[affected]
            mask_un = ~affected
            c_unaff_max = c[mask_un].max() if mask_un.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            for _ in range(45):
                mid = (lo + hi) * 0.5
                new_c_aff = c_aff + mid * dp_aff
                mval = new_c_aff.max()
                if mval <= c_unaff_max + 1e-13:
                    lo = hi = mid
                    break
                active = new_c_aff >= mval - 1e-10
                vmin = dp_aff[active].min()
                vmax = dp_aff[active].max()
                if vmin > 1e-10:
                    hi = mid
                elif vmax < -1e-10:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            delta = (lo + hi) * 0.5
            if abs(delta) > 1e-13:
                c += delta * dp
                h[p] = np.clip(h0 + delta, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 8 == 7:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h_old, n_old, n_new):
    x_old = 2.0 * (np.arange(n_old) + 0.5) / n_old
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def structured_perturb(h, n, freq, amp, rng):
    phase = rng.uniform(0, 2 * np.pi)
    pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n * 2.0 + phase)
    return project_feasible(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick CD warmup
    h1, c1 = cd_fast(best_h, n_init, 25, deadline, rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Multi-resolution optimization
    resolutions = [250, 400, 550, 700, 900, 1200, 1600, 2200]
    for n_try in resolutions:
        if time.time() > deadline - 250:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_int = project_feasible(interp_h(best_h, len(best_h), n_try), n_try)
        for alpha in [200, 600, 1500, 4000, 10000]:
            if time.time() > deadline - 100:
                break
            new_h = slsqp_step(h_int, n_try, alpha, 60, deadline)
            new_c = compute_c5(new_h, n_try)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
                h_int = new_h
        h_cd, c_cd = cd_fast(h_int, n_try, 40, deadline, rng)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 3: Extended CD at best resolution
    n_cur = len(best_h)
    h3, c3 = cd_fast(best_h, n_cur, 300, deadline, rng)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: High-alpha SLSQP refinement
    for alpha in [20000, 60000, 150000]:
        if time.time() > deadline - 50:
            break
        new_h = slsqp_step(best_h, n_cur, alpha, 150, deadline)
        new_c = compute_c5(new_h, n_cur)
        if new_c < best_c:
            best_h, best_c = new_h, new_c

    # Phase 5: Structured perturbations (sinusoidal kicks)
    for r in range(12):
        if time.time() > deadline - 40:
            break
        freq = 2 + 3 * r
        amp = 0.005 + 0.003 * r
        hp = structured_perturb(best_h, n_cur, freq, amp, rng)
        hr, cr = cd_fast(hp, n_cur, 50, deadline, rng)
        if cr < best_c:
            best_h, best_c = hr, cr

    # Phase 6: Random perturbations
    for r in range(10):
        if time.time() > deadline - 30:
            break
        s = 0.002 + 0.004 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        hr, cr = cd_fast(hp, n_cur, 60, deadline, rng)
        if cr < best_c:
            best_h, best_c = hr, cr

    # Phase 7: Upsample and final polish
    if time.time() < deadline - 60 and n_cur < 2000:
        n_fin = min(2500, n_cur * 2)
        h_fin = project_feasible(interp_h(best_h, n_cur, n_fin), n_fin)
        h_fin, c_fin = cd_fast(h_fin, n_fin, 100, deadline, rng)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin
            n_cur = n_fin

    # Phase 8: Final CD
    hf, cf = cd_fast(best_h, len(best_h), 300, deadline, rng)
    if cf < best_c:
        best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
