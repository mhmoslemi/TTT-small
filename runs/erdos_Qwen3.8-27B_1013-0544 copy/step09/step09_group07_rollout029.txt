            break

        temp = t0 * (t_end / t0) ** frac

        m_b_lo = max(0, idx)

        m_hi = max(m_f_hi, m_b_hi)

where R_k = sum_i g[i] * g[i + k - (n-1)] for valid i (this is the "autocorrelation" of g at lag k).

maximize t
subject to: R_k ≥ t for all k

            -0.5 ≤ g[i] ≤ 0.5

This is a non-convex problem (R_k is a quadratic form, and we're maximizing the minimum of these quadratics). However, it might be amenable to semidefinite relaxation or other techniques.

For practical purposes, I can use an iterative approach:
1. Start with a feasible g
2. Identify the "bottleneck" lag k* = argmin_k R_k
3. Try to increase R_{k*} while keeping other R_k ≥ t
4. This is a quadratic optimization: maximize g^T A_{k*} g subject to g^T A_k g ≥ t for all k, sum g = 0, bounds on g

This is a QP with quadratic constraints, which is non-convex. But we can solve it approximately using:
- Sequential quadratic programming
- Penalty methods
- Or simply gradient ascent on the bottleneck contrast

**Gradient of R_k with respect to g:**
R_k = sum_i g[i] * g[i+k-(n-1)]
∂R_k/∂g[j] = g[j+k-(n-1)] + g[j-k+(n-1)] (for valid indices, with appropriate boundary handling)

This is actually the "autocorrelation gradient" which has a nice interpretation.

So the gradient of the bottleneck objective (maximize min_k R_k) at the bottleneck lag k* is:
∇ = ∂R_{k*}/∂g

And we can do projected gradient ascent:
g ← project(g + lr * ∇R_{k*})

where project enforces sum(g) = 0 and -0.5 ≤ g ≤ 0.5.

This is essentially the subgradient method applied to the max-min problem!

def grad_ascent_maxmin(h, n, lr=0.01, max_iter=10000, deadline=None):
    h = h.copy()
    g = h - 0.5
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline: break
        
        # Compute autocorrelations of g
        R = np.correlate(g, g, mode='full')
        k_star = int(np.argmin(R))
        
        # Gradient of R_{k_star} w.r.t. g
        grad = np.zeros(n)
        # R[k] = sum_i g[i] * g[i+k-(n-1)] for valid i
        # dR[k_star]/dg[j] = g[j+k_star-(n-1)] (if valid) + g[j-k_star+(n-1)] (if valid)
        m = k_star - (n-1)
        # Forward: g[j+m] for j where 0 <= j < n and 0 <= j+m < n
        # So j ranges from max(0, -m) to min(n-1, n-1-m)
        lo_f = max(0, -m)
        hi_f = min(n-1, n-1-m)
        if lo_f <= hi_f:
            grad[lo_f:hi_f+1] += g[lo_f+m:hi_f+m+1]
        
        # Backward: g[j-m] for j where 0 <= j < n and 0 <= j-m < n
        # So j ranges from max(0, m) to min(n-1, n-1+m)
        lo_b = max(0, m)
        hi_b = min(n-1, n-1+m)
        if lo_b <= hi_b:
            grad[lo_b:hi_b+1] += g[lo_b-m:hi_b-m+1]
        
        # Update
        g_new = g + lr * grad
        # Project: sum = 0, bounds [-0.5, 0.5]
        g_new -= g_new.mean()
        g_new = np.clip(g_new, -0.5, 0.5)
        # Re-project sum to 0 (clip might have broken it)
        for _ in range(50):
            g_new -= g_new.mean()
            g_new = np.clip(g_new, -0.5, 0.5)
        
        h_new = g_new + 0.5
        h_new = project_feasible(h_new, n)  # ensure exact constraint
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            g = h_new - 0.5
            lr = min(lr * 1.05, 0.1)
        else:
            g = g  # keep old g
            lr *= 0.8
    
    return best_h, best_c
