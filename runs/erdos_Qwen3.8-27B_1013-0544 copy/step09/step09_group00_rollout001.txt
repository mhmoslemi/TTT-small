                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP then CD on initial
    h1, c1 = slsqp_opt(h_best, n, [3000, 20000, 100000], 400, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1
    h2, c2 = exact_cd(h_best, n, 300, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 2: Random-order CD with perturbations
    for r in range(20):
        if time.time() > deadline - 120:
            break
        s = 0.008 + 0.006 * (r % 5)
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 200, deadline, rng=rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 3: Multi-n exploration
    n_candidates = [300, 400, 500, 600, 800, 900, 350, 450]
    for n_try in n_candidates:
        if time.time() > deadline - 200:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [5000, 50000], 300, deadline)
        h_o, c_o = exact_cd(h_o, n_try, 150, deadline, rng=rng)
        if c_o < best_c - 1e-7:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 4: Block swap perturbation
    n = len(h_best)
    for r in range(15):
        if time.time() > deadline - 80:
            break
        hc = h_best.copy()
        nswaps = rng.randint(3, 10)
        for _ in range(nswaps):
            i, j = rng.randint(0, n - 1, size=2)
            if i == j:
                continue
            tmp = hc[i]
            hc[i] = hc[j]
            hc[j] = tmp
        hp = project_feasible(hc, n)
        h_o, c_o = exact_cd(hp, n, 250, deadline, rng=rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Fine-tune with SLSQP on best
    if time.time() < deadline - 30:
        n = len(h_best)
        h_o, c_o = slsqp_opt(h_best, n, [100000, 500000], 500, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Final CD polish
    if time.time() < deadline - 3:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 300, deadline, rng=rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
