    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def smooth_gradient(h, n, alpha):
    c = compute_corr(h, n)
    c_shift = alpha * (c - c.max())
    w = np.exp(c_shift)
    w /= w.sum()
    g = np.zeros(n)
    A = 1.0 - h
    for s in range(-(n - 1), n):
        ws = w[n - 1 + s]
        if ws < 1e-12:
            continue
        if s >= 0:
            g[:n - s] += ws * A[s:]
            g[s:] -= ws * h[:n - s]
        else:
            t = -s
            g[t:] += ws * A[:n - t]
            g[:n - t] -= ws * h[t:]
    g /= alpha
    g -= g.mean()
    return g

def gradient_descent(h, n, deadline, alpha_init=100, eta_init=0.05, n_steps=200, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    beta = 0.9
    v = np.zeros(n)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        eta = eta_init / (1.0 + 0.01 * it)
        alpha = alpha_init * (1.0 + 0.005 * it)
        g = smooth_gradient(h, n, alpha)
        v = beta * v - eta * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
        h = h_new
    return h, best_c

def swap_local_search(h, n, deadline, seed, n_trials=200):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_corr(h, n)
    hi_mask = h > 0.6
    lo_mask = h < 0.4
    hi_idx = np.where(hi_mask)[0]
    lo_idx = np.where(lo_mask)[0]
    if len(hi_idx) == 0 or len(lo_idx) == 0:
        return h, best_c
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = hi_idx[rng.randint(len(hi_idx))]
        j = lo_idx[rng.randint(len(lo_idx))]
        if i == j:
            continue
        a, b = h[i], h[j]
        h[i], h[j] = b, a
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-13:
            best_c = new_c
            hi_mask[i], hi_mask[j] = hi_mask[j], hi_mask[i]
            lo_mask[i], lo_mask[j] = lo_mask[j], lo_mask[i]
            if b > 0.6:
                hi_mask[i] = True; hi_mask[j] = False
            if a > 0.6:
                hi_mask[j] = True; hi_mask[i] = False
            if b < 0.4:
                lo_mask[i] = True; lo_mask[j] = False
            if a < 0.4:
                lo_mask[j] = True; lo_mask[i] = False
            hi_idx = np.where(hi_mask)[0]
            lo_idx = np.where(lo_mask)[0]
        else:
            h[i], h[j] = a, b
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial CD convergence
    dl = min(deadline, t0 + 60)
    h1, c1 = coord_descent(best_h, n, 500, dl, seed)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Gradient descent perturbation + CD
    dl2 = min(deadline, t0 + 300)
    for round_i in range(5):
        if time.time() > dl2:
            break
        alpha = 50 + 30 * round_i
        eta = 0.08 / (1.0 + 0.3 * round_i)
        hg, cg = gradient_descent(best_h, n, dl2 - 60, alpha, eta, 100, seed + 10 * round_i)
        hc, cc = coord_descent(hg, n, 400, dl2 - 30, seed + 100 + round_i)
        if cc < best_c:
            best_h, best_c = hc, cc

    # Phase 3: More CD with different seeds (exploit near-convergence)
    dl3 = min(deadline, t0 + 500)
    for r in range(4):
        if time.time() > dl3:
            break
        h2, c2 = coord_descent(best_h, n, 2000, dl3 - 20, seed + 500 + r * 13)
        if c2 < best_c:
            best_h, best_c = h2, cc if False else c2

    # Phase 4: Swap local search
    dl4 = min(deadline, t0 + 650)
    h4, c4 = swap_local_search(best_h, n, dl4, seed + 900, 300)
    if c4 < best_c:
        best_h, best_c = h4, c4
    h4b, c4b = coord_descent(best_h, n, 1000, dl4, seed + 950)
    if c4b < best_c:
        best_h, best_c = h4b, c4b

    # Phase 5: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    n_trials_list = [300, 400, 450, 500, 550, 650, 700, 800]
    for n_try in n_trials_list:
        if time.time() > deadline - 200:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 1500, deadline - 180, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Targeted gradient + CD on best
    if time.time() < deadline - 100:
        hg, cg = gradient_descent(best_h, len(best_h), deadline - 80, alpha_init=200, eta_init=0.03, n_steps=50, seed=777)
        hc, cc = coord_descent(hg, len(best_h), 1000, deadline - 50, seed + 1100)
        if cc < best_c:
            best_h, best_c = hc, cc

    # Phase 7: Final extensive CD
    n_cur = len(best_h)
    df = min(deadline, deadline - 5)
    hf, cf = coord_descent(best_h, n_cur, 5000, df, seed + 9999)
    if cf < best_c:
        best_h, best_c = hf, cf

    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
