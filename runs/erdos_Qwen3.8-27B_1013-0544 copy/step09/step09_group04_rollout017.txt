        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'random':
            idx = n_idx.copy()

def block_cd(h, n, block_size, max_passes, deadline, rng):
    """Update random blocks of coordinates using grid search"""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        # Pick random block
        start = rng.randint(0, n - block_size + 1)
        block = np.arange(start, start + block_size)
        # Grid search over the block
        # For small blocks, use coarse then fine grid
        best_h = h[block].copy()
        best_val = cur_max
        if block_size <= 3:
            grid = np.linspace(0, 1, 8)
            # Enumerate all combinations (8^block_size at most)
            if block_size == 1:
                candidates = grid
            elif block_size == 2:
                candidates = np.array([[a, b] for a in grid for b in grid])
            elif block_size == 3:
                # Too many for full enumeration, use random sampling
                candidates = rng.uniform(0, 1, (200, 3))
            else:
                candidates = rng.uniform(0, 1, (200, block_size))

                h_test = h.copy()

                    best_h = h_test[block].copy()
            h[block] = best_h

        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo, 0) > r1 and min(x_hi, 1) < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
