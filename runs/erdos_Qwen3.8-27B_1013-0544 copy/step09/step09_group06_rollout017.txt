import cvxpy as cp

def linearized_cd(h, n, num_active=20, max_iters=100, time_limit=60):
    t_start = time.time()
    h_cur = h.copy()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(c.max())
        
        # Select active set: peaks within 1% of max
        threshold = c_max - 0.01 * c_max
        active = np.where(c >= threshold)[0]
        if len(active) > num_active:
            active = np.argsort(c)[::-1][:num_active]
        
        # Compute gradients
        # For each active index m, compute dc_m/dh[j] for all j
        grads = np.zeros((len(active), n))
        for a_idx, m in enumerate(active):
            shift = m - (n - 1)
            for j in range(n):
                val = 0.0
                # j appears as first factor (h[i] where i=j)
                j2 = j + shift
                if 0 <= j2 < n:
                    val += (1.0 - h_cur[j2])
                # j appears as second factor (h[i+shift] where i+shift=j, so i=j-shift)
                j1 = j - shift
                if 0 <= j1 < n:
                    val -= h_cur[j1]
                grads[a_idx, j] = val
        
        # LP: minimize t
        # s.t. grads @ delta <= t - c[active]
        #      -h <= delta <= 1-h
        #      sum(delta) = 0
        delta = cp.Variable(n)
        t = cp.Variable()
        constraints = [cp.sum(delta) == 0]
        for a_idx in range(len(active)):
            constraints.append(grads[a_idx] @ delta <= t - c[active[a_idx]])
        constraints.append(delta >= -h_cur)
        constraints.append(delta <= 1.0 - h_cur)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.GLPK, max_time=5.0)
        
        if delta.value is None:
            break
        h_cur = np.clip(h_cur + delta.value, 0, 1)
        h_cur = project_feasible(h_cur, n)
    
    return h_cur, compute_c5(h_cur, n)

def project_feasible(h, n):

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())

        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False

            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue

            diag_idx = None

                jf = p + m - (n - 1)

                ab = (-h[jb]) if 0 <= jb < n else 0.0

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):

            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:

    return best_d, best_val

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):

    if diag_idx is not None:

def lp_direction(h, n, num_active=30, step_search=True):
    """Solve LP to find a descent direction for the maximin problem."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(c.max())
    
    # Select active set
    active = np.argsort(c)[::-1][:num_active]
    
    # Compute gradients for active lags
    # c[m] = sum_{i: valid} h[i] * (1 - h[i + shift]) where shift = m - (n-1)
    # dc[m]/dh[j] = (1-h[j+shift]) [if j+shift valid] + (-h[j-shift]) [if j-shift valid]
    T = 2 * n - 1
    n_active = len(active)
    
    # Build A matrix for LP: A @ delta <= t - c[active]
    # where A[i,j] = dc[m_i]/dh[j]
    A = np.zeros((n_active, n))
    for a_i, m in enumerate(active):
        shift = m - (n - 1)
        for j in range(n):
            val = 0.0
            jp = j + shift
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            jm = j - shift
            if 0 <= jm < n:
                val -= h[jm]
            A[a_i, j] = val
    
    # LP: min t
    # s.t. A @ delta - t <= -c[active]  (i.e., A @ delta <= t - c[active])
    #      -h <= delta <= 1-h
    #      sum(delta) = 0
    
    # Using scipy linprog:
    # Variables: [delta(0..n-1), t]
    # Objective: minimize t -> c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_full @ x <= b_full
    # From A @ delta <= t - c[active]: A @ delta - t <= -c[active]
    A_ineq = np.zeros((n_active, n + 1))
    A_ineq[:, :n] = A
    A_ineq[:, n] = -1.0
    b_ineq = -c[active]
    
    # Bound constraints are handled by bounds parameter
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h <= delta <= 1-h, t free (but bounded by some reasonable value)
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((c_max * 0.5, c_max * 2.0))  # t bounds
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return h, compute_c5(h, n)
    
    delta = res.x[:n]
    
    if not step_search:
        h_new = project_feasible(h + delta, n)
        return h_new, compute_c5(h_new, n)
    
    # Line search
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
        h_try = project_feasible(h + alpha * delta, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved, current_max = True, best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lp_direction(h, n, num_active, max_lags=None):
    """LP-based descent direction for the maximin overlap problem."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(c.max())
    T = 2 * n - 1
    if max_lags is None:
        max_lags = T
    num_active = min(num_active, T)
    active = np.argsort(c)[::-1][:num_active]
    n_act = len(active)
    A = np.zeros((n_act, n))
    for a_i in range(n_act):
        m = int(active[a_i])
        s = m - (n - 1)
        for j in range(n):
            v = 0.0
            jp = j + s
            if 0 <= jp < n:
                v += (1.0 - h[jp])
            jm = j - s
            if 0 <= jm < n:
                v -= h[jm]
            A[a_i, j] = v
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((c_max * 0.5, c_max * 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, compute_c5(h, n)
    if res.status != 0 or res.x is None:
        return h, compute_c5(h, n)
    delta = res.x[:n]
    if np.abs(delta).max() < 1e-12:
        return h, compute_c5(h, n)
    best_h, best_c = h.copy(), c_max
    for alpha in [1.0, 0.6, 0.4, 0.2, 0.1]:
        h_try = project_feasible(h + alpha * delta, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c, best_h = c_try, h_try
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD to local optimum
    for tk in [5, 10, 20, 30]:
        if time.time() > deadline - 500:
            break
        h2, c2 = multi_cd(best_h, n, 200, deadline - 500, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 2: LP-based minimax with CD refinement
    if time.time() < deadline - 300:
        for rnd in range(10):
            if time.time() > deadline - 300:
                break
            h_lp, c_lp = lp_direction(best_h, n, num_active=35)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
            h_cd, c_cd = multi_cd(best_h, n, 80, deadline - 300, 20)
            if c_cd < best_c:
                best_h, best_c = h_cd, c2 if False else c_cd
                best_h, best_c = h_cd, c_cd
    # Phase 3: SLSQP with increasing smoothness
    if time.time() < deadline - 200:
        for alpha in [5000, 30000, 100000, 300000]:
            if time.time() > deadline - 200:
                break
            h_s, c_s = slsqp_run(best_h, n, alpha, 350)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
    # Phase 4: Stochastic restarts with LP+CD
    n_cur = len(best_h)
    if time.time() < deadline - 120:
        for r in range(20):
            if time.time() > deadline - 120:
                break
            s = 0.002 + 0.004 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            hp, cp = lp_direction(hp, n_cur, num_active=25)
            hp, cp = multi_cd(hp, n_cur, 100, deadline - 120, 15)
            if cp < best_c:
                best_h, best_c = hp, cp
    # Phase 5: Different discretization sizes
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [550, 650, 700, 750]:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 150, deadline - 100, 15)
        h_o, c_o = lp_direction(h_o, n_try, num_active=30)
        h_o, c_o = multi_cd(h_o, n_try, 80, deadline - 100, 20)
        h_flip = project_feasible(1.0 - h_o, n_try)
        c_flip = compute_c5(h_flip, n_try)
        if c_flip < c_o:
            h_o, c_o = h_flip, c_flip
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new
    # Phase 6: Final deep refinement
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        h_f, c_f = multi_cd(best_h, n_cur, 1500, deadline - 40, 30)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        h_f, c_f = lp_direction(best_h, n_cur, num_active=50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = multi_cd(best_h, n_cur, 500, deadline, 35)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
