def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:

                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)

                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)

            v *= 0.5
            h_new = h
        h = h_new
        if it % 100 == 99:
            eta = min(eta * 1.1, 200.0)
    if compute_c5(h, n) < best_c:
        best_h, best_c = h, compute_c5(h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Projected gradient with momentum
    t_phase1 = deadline - (budget_s * 0.25)
    h1, c1 = projected_grad_opt(best_h, n, min(t0 + 120, t_phase1), alpha_start=50.0)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Subgradient with top-lag focus
    t_phase2 = min(t0 + 200, t_phase1)
    h2, c2 = subgradient_opt(best_h, n, t_phase2)
    if c2 < best_c:
        best_h, best_c = h2, c1 if c2 < c1 else c2
    best_c = compute_c5(best_h, n)
    # Phase 3: Coordinate descent
    max_cd_passes = max(10, int((deadline - time.time()) / 8.0))
    h3, c3 = coord_descent(best_h, n, min(max_cd_passes, 80), deadline, gs_iters=55)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 600, 800, 1000]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        t_sub = min(time.time() + 80, deadline - 100)
        h_o, c_o = projected_grad_opt(h_int, n_try, t_sub, alpha_start=100.0)
        h_o, c_o = coord_descent(h_o, n_try, 40, deadline, gs_iters=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)
    # Phase 5: Restart with perturbations
    n_cur = len(best_h)
    n_restarts = max(2, int((deadline - time.time()) / 60))
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        s = 0.02 + 0.03 * r / max(1, n_restarts)
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        t_sub = min(time.time() + 40, deadline - 50)
        h_o, c_o = projected_grad_opt(hp, n_cur, t_sub, alpha_start=200.0)
        h_o, c_o = coord_descent(h_o, n_cur, 30, deadline, gs_iters=45)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 6: Final CD polish
    if time.time() < deadline - 5:
        remaining = deadline - time.time() - 3
        max_p = max(5, int(remaining / 3.0))
        h_f, c_f = coord_descent(best_h, n_cur, max_p, deadline, gs_iters=70)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
