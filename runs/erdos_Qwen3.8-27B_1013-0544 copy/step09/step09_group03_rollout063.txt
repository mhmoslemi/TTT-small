import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-13:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-11:
            hi = mid
        elif max_sl < -1e-11:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=2000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def fourier_to_h(coeffs, n, K):
    """Convert Fourier coefficients to h values.
    coeffs = [a1, b1, a2, b2, ..., aK, bK]
    h[i] = 1/2 + sum_{k=1}^{K} [a_k cos(2πk*i/n) + b_k sin(2πk*i/n)]
    """
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = coeffs[2*(k-1)]
        b = coeffs[2*(k-1)+1]
        h += a * np.cos(2*np.pi*k*i/n) + b * np.sin(2*np.pi*k*i/n)
    return h

def fourier_obj(coeffs, n, K, penalty=1000):
    h = fourier_to_h(coeffs, n, K)
    # Penalty for out of [0,1]
    penalty_val = penalty * (np.maximum(0, -h)**2 + np.maximum(0, h-1)**2).sum()
    # Sum constraint penalty
    sum_pen = penalty * (h.sum() - n/2)**2
    # Objective: max correlation
    c = np.correlate(h, 1-h, mode='full')
    obj = c.max() * 2.0/n
    return obj + penalty_val + sum_pen

    s = s_neq0

    pm = p - s

    b = c[m] - a * h0

        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:

            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=2000, deadline=None, seed=0):

            best_c = new_c
        elif improved == 0:
