    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):

            di = rng.uniform(-0.05, 0.05)
            dj = -di
            ni = np.clip(h0i + di, 0.0, 1.0)
            nj = np.clip(h0j + dj, 0.0, 1.0)
            if abs(ni - h0i) < 1e-14 and abs(nj - h0j) < 1e-14:
                continue
            h[i], h[j] = ni, nj
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[i], h[j] = h0i, h0j
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Heavy coordinate descent
    h1, c1 = coord_descent(best_h, n, 60, deadline, gs_iters=100)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Swap + two-point moves to escape local optima
    if time.time() < deadline - 200:
        h2, c2 = swap_optimize(best_h, n, deadline, rng, n_attempts=300)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = two_point_cd(best_h, n, deadline, rng, gs_iters=50, n_pairs=100)
        if c2b < best_c:
            best_h, best_c = h2b, c2b

    # Phase 3: Coordinate descent after mixing
    h3, c3 = coord_descent(best_h, n, 50, deadline, gs_iters=100)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Multi-restart with diverse perturbations
    if time.time() < deadline - 120:
        for r in range(25):
            if time.time() > deadline - 60:
                break
            if r % 5 == 0:
                sigma = 0.01 + 0.04 * (r / 25.0)
                hp = project_feasible(best_h + rng.normal(0, sigma, n), n)
            elif r % 5 == 1:
                hp = best_h.copy()
                bl = rng.randint(5, 30)
                st = rng.randint(0, max(1, n - bl))
                hp[st:st+bl] = 1.0 - hp[st:st+bl]
                hp = project_feasible(hp, n)
            elif r % 5 == 2:
                x = np.linspace(0, 2, n)
                f = rng.uniform(1, 15)
                a = rng.uniform(0.02, 0.07)
                hp = project_feasible(best_h + a * np.sin(2*np.pi*f*x + rng.uniform(0, 6.2832)), n)
            elif r % 5 == 3:
                hp = best_h.copy()
                idx = rng.choice(n, size=rng.randint(10, 50), replace=False)
                hp[idx] = rng.uniform(0, 1, len(idx))
                hp = project_feasible(hp, n)
            else:
                # Tikhonov-smoothed perturbation
                k = np.minimum(n, 20)
                kernel = np.exp(-0.5 * (np.arange(-k//2, k//2+1) / (k//6.0+1))**2)
                kernel /= kernel.sum()
                noise = rng.normal(0, 0.05, n)
                smooth = np.convolve(noise, kernel, mode='same')
                hp = project_feasible(best_h + smooth, n)

            ho, co = coord_descent(hp, n, 30, deadline, gs_iters=70)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Multi-resolution
    if time.time() < deadline - 100:
        for n_try in [400, 500, 600, 800, 900]:
            if time.time() > deadline - 70:
                break
            if n_try == n:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            hi = project_feasible(np.interp(x_new, x_old, best_h, kind='cubic'), n_try)
            ho, co = coord_descent(hi, n_try, 40, deadline, gs_iters=80)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final intensive polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 400, deadline, gs_iters=150)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final SLSQP micro-polish on the most active coordinates
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        c_max = c.max()
        active = np.where(c >= c_max * 0.999)[0]
        # Find which h indices are involved in active correlations
        h_indices = set()
        for k in active:
            shift = int(k) - (n_cur - 1)
            lo = max(0, -shift)
            hi = min(n_cur, n_cur - shift)
            h_indices.update(range(lo, hi))
        h_indices = sorted(h_indices)
        if len(h_indices) > 1 and len(h_indices) < n_cur:
            idx = np.array(h_indices)
            x0 = best_h[idx].copy()
            rest_mask = np.ones(n_cur, dtype=bool)
            rest_mask[idx] = False
            rest_val = best_h[rest_mask].sum()
            target_block = n_cur / 2.0 - rest_val
            bounds = [(0.0, 1.0)] * len(idx)

            def _obj(xb):
                ht = best_h.copy()
                ht[idx] = xb
                return compute_c5(ht, n_cur)

            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_block,
                     'jac': lambda x: np.ones(len(idx))}]
            try:
                res = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-18})
                if res.success:
                    h_new = best_h.copy()
                    h_new[idx] = res.x
                    h_new = project_feasible(h_new, n_cur)
                    c_new = compute_c5(h_new, n_cur)
                    if c_new < best_c:
                        best_h, best_c = h_new, c_new
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
