            if tot < 1e-14:

            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    lo2 = max(0, p - (n - 1))
    hi2 = min(2 * n - 2, p)
    m_lo = min(lo1, lo2)
    m_hi = max(hi1, hi2)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    self_mask = (ms == n - 1)
    has_diag = np.any(self_mask)
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    non_diag = ~self_mask
    for _ in range(35):
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and np.any(non_diag):
            a_nd = a[non_diag]
            c_nd = c[ms[non_diag]]
            xlo = 0.0
            xhi = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                xlo = max(xlo, float(np.max((T - c_nd[neg] + h0 * a_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xhi = min(xhi, float(np.min((T - c_nd[pos] + h0 * a_nd[pos]) / a_nd[pos])))
            if xlo > xhi + 1e-14:
                feasible = False
        if feasible and has_diag:
            Q = T - c[n - 1] + h0 - h0 * h0
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if np.any(non_diag):
                    a_nd = a[non_diag]
                    c_nd = c[ms[non_diag]]
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg] + h0 * a_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos] + h0 * a_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-14:
                        feasible = False
                    elif xl >= r1 - 1e-14:
                        if xh <= r2 + 1e-14:
                            feasible = False
                    elif xh <= r1 + 1e-14:
                        pass
                    elif xl < r1 and xh > r2:
                        feasible = False
                    elif xl < r1:
                        pass
                    else:
                        pass
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(non_diag):
        a_nd = a[non_diag]
        c_nd = c[ms[non_diag]]
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg] + h0 * a_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos] + h0 * a_nd[pos]) / a_nd[pos])))
    if has_diag:
        Q = T - c[n - 1] + h0 - h0 * h0
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo >= r2 - 1e-14:
                pass
            elif x_hi <= r1 + 1e-14:
                pass
            else:
                hi1v = min(x_hi, r1)
                if hi1v >= x_lo - 1e-14:
                    x_hi = hi1v
                else:
                    lo2v = max(x_lo, r2)
                    if x_hi >= lo2v - 1e-14:
                        x_lo = lo2v
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float((x_lo + x_hi) / 2.0)

def cd_loop(h, n, deadline, rng, max_passes=5000, random_order=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                delta = x_new - h0
                h[p] = x_new
                c[n - 1] += x_new * (1.0 - x_new) - h0 * (1.0 - h0)
                lo1 = max(0, n - 1 - p)
                hi1 = min(2 * n - 2, 2 * n - 2 - p)
                lo2 = max(0, p - (n - 1))
                hi2 = min(2 * n - 2, p)
                ms1 = np.arange(lo1, hi1 + 1)
                js1 = p + ms1 - (n - 1)
                v1 = (js1 >= 0) & (js1 < n) & (ms1 != n - 1)
                if np.any(v1):
                    c[ms1[v1]] += delta * (1.0 - h[js1[v1]])
                ms2 = np.arange(lo2, hi2 + 1)
                js2 = p - ms2 + (n - 1)
                v2 = (js2 >= 0) & (js2 < n) & (ms2 != n - 1)
                if np.any(v2):
                    c[ms2[v2]] -= delta * h[js2[v2]]
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, coeffs):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for (freq, amp) in coeffs:
        h += amp * np.sin(np.pi * freq * x)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP warm-up
    h1, c1 = slsqp_run(best_h, n_init, [200, 2000, 20000], 300, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: CD with incremental updates (main workhorse)
    h2, c2 = cd_loop(best_h, n_init, deadline, rng, max_passes=3000, random_order=True)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Try Fourier-based restarts
    n = len(best_h)
    if time.time() < deadline - 120:
        four_sets = [
            [(1, 0.2), (3, 0.1), (5, 0.05)],
            [(1, 0.25), (3, -0.12)],
            [(2, 0.15), (3, 0.1), (4, 0.05)],
            [(1, 0.15), (2, -0.1), (5, 0.08)],
        ]
        for fs in four_sets:
            if time.time() > deadline - 80:
                break
            hf = fourier_init(n, fs)
            hf, cf = cd_loop(hf, n, deadline, rng, max_passes=800, random_order=True)
            if cf < best_c:
                best_h, best_c = hf, cf
    # Phase 4: Random restarts near best
    if time.time() < deadline - 100:
        n_restarts = max(2, int((deadline - time.time() - 60) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            s = 0.02 + 0.03 * r
            hp = best_h + rng.normal(0, s, n)
            hp = project_feasible(hp, n)
            hp, cp = slsqp_run(hp, n, [5000], 200, deadline)
            hp, cp = cd_loop(hp, n, deadline, rng, max_passes=500, random_order=True)
            if cp < best_c:
                best_h, best_c = hp, cp
    # Phase 5: Try different n by interpolation
    if time.time() < deadline - 100:
        x_orig = np.linspace(0, 2, n)
        for n_try in [n + 100, n - 100]:
            if n_try < 100 or time.time() > deadline - 60:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_o, c_o = cd_loop(h_int, n_try, deadline, rng, max_passes=1500, random_order=True)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                n = n_try
                x_orig = np.linspace(0, 2, n_try)
    # Phase 6: Final intensive CD polish
    if time.time() < deadline:
        n = len(best_h)
        remaining = deadline - time.time() - 2
        max_p = max(100, int(remaining / 0.01))
        hf, cf = cd_loop(best_h, n, deadline, rng, max_passes=max_p, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf
        # One more round with random order
        if time.time() < deadline - 5:
            hf, cf = cd_loop(best_h, n, deadline, rng, max_passes=2000, random_order=True)
            if cf < best_c:
                best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
