    if ap.size:

    if an.size:

    if zmx > L + tol: return False,0.0
    if lo > hi + tol: return False,0.0
    if Ah.size:
        D = Bh*Bh + 4.0*Ah*(L - Ch)
        if float(np.min(D)) < -1e-10: return False,0.0
        D = np.maximum(D,0.0)

        r1 = (-Bh - sq)/(2.0*Ah)
        r2 = (-Bh + sq)/(2.0*Ah)
        lo = max(lo, float(np.max(r1)))

def fourier_basis(n,m,rng=None):
    x=np.linspace(0.0,2.0,n)
    cols=[]
    for k in range(1,m+1):
        s=np.sin(np.pi*k*x)
        cc=np.cos(np.pi*k*x)
        cc = cc - cc.mean()

        cols.append(s); cols.append(cc)

    norms=np.linalg.norm(B,axis=0)
    norms[norms<1e-12]=1.0
    B = B / norms
    return B

def pair_cd(h,c,cmax,n,end_time,rng,grad_beta=30.0,beam=6,random_batch=80,no_improve_stop=4000,use_subgrad=True,subgrad_interval=3.0):
    last_grad=time.time(); last_sub=time.time(); noimp=0
    while time.time() < end_time:
        now=time.time()
        if use_subgrad and now-last_sub >= subgrad_interval:
            last_sub=now
            for atol in (0.3,1.0,3.0):
                h2,c2,cm2,imp=subgrad_step(h,c,cmax,n,atol)
                if imp:
                    h[:]=h2; c=c2; cmax=cm2; noimp=0
                    break
        if now-last_grad >= 2.0:
            last_grad=now
            w=softmax_weights(c,n,grad_beta)
            g=grad_weights(h,w,n); g = g - float(np.mean(g))
            order=np.argsort(g)
            m=min(beam, n//2)
            for i in range(m):
                if time.time()>end_time: return h,c,cmax
                p=int(order[i]); q=int(order[n-1-i]);
                if p!=q:

            lows=order[:k]; highs=order[-k:]
            for _ in range(3*beam):
                if time.time()>end_time: return h,c,cmax
                p=int(lows[rng.randint(lows.size)]); q=int(highs[rng.randint(highs.size)])
                if p!=q:

def subgrad_step(h,c,cmax,n,active_tol):

import numpy as np
import time


def project_box_sum(y, target):
    n = y.size
    ymin = float(np.min(y))
    ymax = float(np.max(y))
    lo = -2.0 - ymax
    hi = 2.0 - ymin
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = target - float(y.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = y < 1.0
        else:
            mask = y > 0.0
        cnt = float(mask.sum())
        if cnt <= 0.0:
            break
        y[mask] += diff / cnt
        y = np.clip(y, 0.0, 1.0)
    return y


def c5_from_c(c, n):
    return float(c.max()) * (2.0 / n)


def softmax_weights(c, n, beta):
    q = c * (2.0 / n)
    qmax = float(q.max())
    z = np.exp(beta * (q - qmax))
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(c.shape, 1.0 / c.size)
    return z / s


def grad_weights(h, w, n):
    g = np.zeros(n)
    for r in range(-n + 1, n):
        ww = w[n - 1 + r]
        if ww <= 1e-14:
            continue
        if r >= 0:
            lo = 0
            hi = n - r
        else:
            lo = -r
            hi = n
        g[lo:hi] += ww * (1.0 - h[lo + r:hi + r])
        g[lo + r:hi + r] -= ww * h[lo:hi]
    return g


def pair_feasible(L, smin, smax, unaff, pB, pC, nB, nC, zmx, cA, cB, cC, negA, negB, negC):
    eps = 1e-12
    if L < unaff - eps:
        return False, 0.0
    lo = smin
    hi = smax
    if pB.size:
        hi = min(hi, float(np.min((L - pC) / pB)))
    if nB.size:
        lo = max(lo, float(np.max((L - nC) / nB)))
    if zmx > L + eps:
        return False, 0.0
    if lo > hi + eps:
        return False, 0.0

    if cA.size:
        D = cB * cB + 4.0 * cA * (L - cC)
        if float(np.min(D)) < -1e-9:
            return False, 0.0
        sq = np.sqrt(np.maximum(D, 0.0))
        lo = max(lo, float(np.max((-cB - sq) / (2.0 * cA))))
        hi = min(hi, float(np.min((-cB + sq) / (2.0 * cA))))
        if lo > hi + eps:
            return False, 0.0

    if negA.size:
        for i in range(negA.size):
            a = -negA[i]
            b = float(negB[i])
            cc = float(negC[i])
            D = b * b - 4.0 * a * (L - cc)
            if D < -1e-9:
                continue
            if D < 0.0:
                continue
            sd = float(np.sqrt(D))
            r1 = (b - sd) / (2.0 * a)
            r2 = (b + sd) / (2.0 * a)
            found = False
            if hi <= r1 + eps:
                lof, hif = lo, hi
                found = True
            elif lo >= r2 - eps:
                lof, hif = lo, hi
                found = True
            else:
                if r1 > lo + eps:
                    lof, hif = lo, min(hi, r1)
                    found = True
                elif r2 < hi - eps:
                    lof, hif = max(lo, r2), hi
                    found = True
                elif lo - eps <= r1 <= hi + eps:
                    lof = hif = r1
                    found = True
                elif lo - eps <= r2 <= hi + eps:
                    lof = hif = r2
                    found = True
            if not found:
                return False, 0.0
            lo, hi = lof, hif
            if lo > hi + eps:
                return False, 0.0

    if lo <= hi + eps:
        return True, 0.5 * (lo + hi)
    return False, 0.0


def solve_pair(h, c, cmax, n, p, q):

    hq = float(h[q])
    smin = max(-hp, hq - 1.0)
    smax = min(1.0 - hp, hq)
    if smax - smin < 1e-12:
        return 0.0, cmax

    mn = min(p, q)
    rlo = mn - n + 1
    rhi = n - 1 - mn
    m0 = n - 1 + rlo
    m1 = n - 1 + rhi
    L = m1 - m0 + 1
    cbase = c[m0:m1 + 1]

    unaff = -1e300
    if m0 > 0:
        unaff = max(unaff, float(c[:m0].max()))
    mend = m1 + 1
    if mend < 2 * n - 1:
        unaff = max(unaff, float(c[mend:].max()))

    A = np.zeros(L)
    B = np.zeros(L)
    hloc = h
    ploc = p
    qloc = q
    nloc = n

    for idx in range(L):
        r = rlo + idx

        j = ploc + r
        if 0 <= j < nloc:
            if j == ploc:
                dj = 1.0
            elif j == qloc:
                dj = -1.0
            else:
                dj = 0.0
            A[idx] += -dj
            B[idx] += (1.0 - hloc[j]) - hloc[ploc] * dj
            s1 = ploc
        else:
            s1 = -1

        i = ploc - r
        if 0 <= i < nloc and i != s1:
            da = -1.0 if i == qloc else 0.0
            A[idx] += -da
            B[idx] += da * (1.0 - hloc[ploc]) - hloc[i]
            s2 = i
        else:
            s2 = -1

        j = qloc + r
        if 0 <= j < nloc and qloc != s1 and qloc != s2:
            if j == ploc:
                dj = 1.0
            elif j == qloc:
                dj = -1.0
            else:
                dj = 0.0
            A[idx] += dj
            B[idx] += -(1.0 - hloc[j]) - hloc[qloc] * dj
            s3 = qloc
        else:
            s3 = -1

        i = qloc - r
        if 0 <= i < nloc and i != s1 and i != s2 and i != s3:
            da = 1.0 if i == ploc else 0.0
            A[idx] += da
            B[idx] += da * (1.0 - hloc[qloc]) + hloc[i]

    eps = 1e-13
    aff = np.abs(A) <= eps
    affB = B[aff]
    affC = cbase[aff]
    ap = affB > eps
    an = affB < -eps
    az = ~(ap | an)
    pB = affB[ap]
    pC = affC[ap]
    nB = affB[an]
    nC = affC[an]
    zmx = float(affC[az].max()) if np.any(az) else -1e300

    posm = A > eps
    negm = A < -eps
    convA = A[posm]
    convB = B[posm]
    convC = cbase[posm]
    negA = A[negm]
    negB = B[negm]
    negC = cbase[negm]

    ep1 = cbase + B * smin
    ep2 = cbase + B * smax
    mins = np.minimum(ep1, ep2)
    if posm.any():
        vv = convC - (convB * convB) / (4.0 * convA)
        vtx = -convB / (2.0 * convA)
        mins[posm] = np.where(
            (vtx >= smin - 1e-14) & (vtx <= smax + 1e-14),
            vv,
            np.minimum(ep1[posm], ep2[posm])
        )
    lb = float(np.max(mins)) if L > 0 else 0.0
    if unaff > lb:
        lb = unaff
    if not np.isfinite(lb):
        lb = -1.0
    if lb >= cmax - 1e-11:
        return 0.0, cmax

    Llo = lb - 1e-7
    Lhi = float(cmax)
    if Llo > Lhi:
        Llo = Lhi - 1e-7

    for _ in range(34):
        Lmid = 0.5 * (Llo + Lhi)
        ok, _ = pair_feasible(

            pB, pC, nB, nC, zmx,
            convA, convB, convC,
            negA, negB, negC
        )

        else:
            Llo = Lmid

    ok, spt = pair_feasible(
        Lhi, smin, smax, unaff,
        pB, pC, nB, nC, zmx,
        convA, convB, convC,
        negA, negB, negC
    )
    if not ok:
        spt = 0.0

    best_s = 0.0
    best_val = float(cmax)

        if s < smin - 1e-12 or s > smax + 1e-12:
            continue
        if s < smin:
            s = smin
        elif s > smax:
            s = smax
        vals = cbase + B * s + A * s * s

        val = mv if mv > unaff else unaff

def try_pair(h, c, cmax, n, p, q):
    s, pred = solve_pair(h, c, cmax, n, p, q)
    if abs(s) < 1e-12 or pred >= cmax - 1e-12:
        return c, cmax, False
    hp0 = float(h[p])
    hq0 = float(h[q])
    h[p] = float(np.clip(hp0 + s, 0.0, 1.0))
    h[q] = float(np.clip(hq0 - s, 0.0, 1.0))
    c2 = np.correlate(h, 1.0 - h, mode="full")
    c2max = float(c2.max())
    if c2max < cmax - 1e-12:
        return c2, c2max, True
    h[p] = hp0
    h[q] = hq0
    return c, cmax, False


def subgrad_step(h, c, cmax, n, active_tol):
    active = c >= (cmax - active_tol)
    m = int(active.sum())
    if m == 0:
        return h, c, cmax, False
    w = active.astype(float)
    w /= float(m)
    g = grad_weights(h, w, n)
    norm = float(np.linalg.norm(g))
    if norm < 1e-10:
        return h, c, cmax, False
    d = -g / norm
    target = 0.5 * n
    for step in (2.0, 1.0, 0.5, 0.2, 0.1, 0.05):
        hp = project_box_sum(h + step * d, target)
        cp = np.correlate(hp, 1.0 - hp, mode="full")
        cpmax = float(cp.max())
        if cpmax < cmax - 1e-12:
            return hp, cp, cpmax, True
    return h, c, cmax, False


def smooth_dir(n, rng, m=8):
    x = np.linspace(0.0, 2.0, n)
    d = np.zeros(n)
    for k in range(1, m + 1):
        amp = 1.0 / (k ** 0.6)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        ph2 = rng.uniform(0.0, 2.0 * np.pi)
        d += amp * (np.sin(np.pi * k * x + ph) + 0.7 * np.cos(np.pi * k * x + ph2))
    d -= float(d.mean())
    return d


def block_dir(n, rng, blocks=24):
    d = np.zeros(n)
    st = max(1, n // max(1, blocks))
    for s in range(0, n, st):
        e = min(n, s + st)
        d[s:e] = rng.uniform(-1.0, 1.0)
    d -= float(d.mean())
    return d


def pair_cd(h, c, cmax, n, end_time, rng,
            grad_beta=40.0, beam=6, random_batch=80,
            no_improve_stop=4000, use_subgrad=True, subgrad_interval=3.0):
    last_grad = time.time()
    last_sub = time.time()
    noimp = 0

    while time.time() < end_time:
        now = time.time()

        if use_subgrad and now - last_sub >= subgrad_interval:
            last_sub = now
            for atol in (0.3, 1.0, 3.0):
                h2, c2, cmax2, imp = subgrad_step(h, c, cmax, n, atol)
                if imp:
                    h[:] = h2
                    c = c2
                    cmax = cmax2
                    noimp = 0
                    break

        if now - last_grad >= 2.0:
            last_grad = now
            w = softmax_weights(c, n, grad_beta)
            g = grad_weights(h, w, n)
            g = g - float(np.mean(g))
            order = np.argsort(g)

            m = min(beam, n // 2)
            for i in range(m):
                if time.time() > end_time:
                    return h, c, cmax
                p = int(order[i])
                q = int(order[n - 1 - i])
                if p != q:
                    c, cmax, imp = try_pair(h, c, cmax, n, p, q)
                    noimp = 0 if imp else noimp + 1

            k = max(4, n // 8)
            lows = order[:k]
            highs = order[-k:]
            for _ in range(3 * beam):
                if time.time() > end_time:
                    return h, c, cmax
                p = int(lows[rng.randint(lows.size)])
                q = int(highs[rng.randint(highs.size)])
                if p != q:
                    c, cmax, imp = try_pair(h, c, cmax, n, p, q)
                    noimp = 0 if imp else noimp + 1
                    if noimp > no_improve_stop:
                        return h, c, cmax

        for _ in range(random_batch):
            if time.time() > end_time:
                return h, c, cmax
            p = int(rng.randint(n))
            q = int(rng.randint(n))
            if p == q:
                continue
            c, cmax, imp = try_pair(h, c, cmax, n, p, q)
            noimp = 0 if imp else noimp + 1
            if noimp > no_improve_stop:
                return h, c, cmax

    return h, c, cmax

            best_h = h.copy()
            best_c = c5_from_c(c, n)

    scales = (0.02, 0.05, 0.10, 0.15, 0.25, 0.40)
    for si, scl in enumerate(scales):
        if time.time() > deadline - 130.0:
            break
        for tr in range(2):
            if time.time() > deadline - 130.0:
                break
            if (si + tr) % 2 == 0:
                d = smooth_dir(n, rng, 8)
            else:
                d = block_dir(n, rng, 24)
            dn = float(np.max(np.abs(d)))
            if dn > 0.0:
                d = d / dn
            hp = project_box_sum(best_h + scl * d, target)
            cp = np.correlate(hp, 1.0 - hp, mode="full")
            cpmax = float(cp.max())
            if c5_from_c(cp, n) > best_c + 0.05:
                continue
            endp = min(deadline - 120.0, time.time() + 70.0)
            if endp < time.time() + 5.0:
                continue
            hp, cp, cpmax = pair_cd(
                hp, cp, cpmax, n, endp, rng,
                grad_beta=60.0, beam=5, random_batch=60,
                no_improve_stop=1500, use_subgrad=False, subgrad_interval=0.0
            )
            if c5_from_c(cp, n) < best_c:
                best_h = hp.copy()
                best_c = c5_from_c(cp, n)

    if time.time() < deadline - 20.0:
        hf = best_h.copy()
        cf = np.correlate(hf, 1.0 - hf, mode="full")
        cfmax = float(cf.max())
        hf, cf, cfmax = pair_cd(
            hf, cf, cfmax, n, deadline - 5.0, rng,
            grad_beta=60.0, beam=8, random_batch=110,
            no_improve_stop=8000, use_subgrad=True, subgrad_interval=3.0
        )
        if c5_from_c(cf, n) < best_c:
            best_h = hf.copy()
            best_c = c5_from_c(cf, n)

    best_h = project_box_sum(best_h, target)
    cf = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_c = c5_from_c(cf, n)
    return best_h, best_c, n
