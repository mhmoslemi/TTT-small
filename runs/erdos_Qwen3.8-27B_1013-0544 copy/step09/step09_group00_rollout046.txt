        m_max = 2 * n - 2 - p

Wait, I need to be more careful about the subgradient. Let me re-derive.

c[m] = ∑_i h[i] * (1 - h[i - (m-(n-1))])

For a fixed m, ∂c[m]/∂h[p]:
- h[p] appears as a factor (i=p): contributes (1 - h[p-(m-(n-1))]) if p-(m-(n-1)) is valid
- h[p] appears inside (1-h[·]) (i-(m-(n-1))=p, i.e., i = p+(m-(n-1))): contributes -h[p+(m-(n-1))] if p+(m-(n-1)) is valid
- If both i=p and i-(m-(n-1))=p (i.e., m=n-1), the term is h[p]*(1-h[p]), derivative is 1-2h[p]

So: ∂c[m]/∂h[p] = (1-h[jb]) * δ(vb_valid) - h[jf] * δ(vf_valid)

where jf = p + m - (n-1) and jb = p - m + (n-1)

For m = n-1: ∂c[m]/∂h[p] = (1-h[p]) - h[p] = 1-2h[p] ✓

This matches what's in the solve_1d_bisect function (the `a` array).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.clip(0, 1).copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            scale = diff / tot
            h = np.clip(h + scale * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            scale = diff / tot
            h = np.clip(h + scale * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, tol=1e-11):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        max_change = 0.0
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            d = xn - h0
            if abs(d) > 1e-13:
                max_change = max(max_change, abs(d))
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += d * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= d * d
                h[p] = xn
        if max_change < tol:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_descent(h, n, n_steps, deadline, initial_step=0.01):
    h = h.copy()
    for s in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        thresh = cmax - 0.01 * cmax
        active = c >= thresh
        w = active.astype(float)
        wsum = w.sum()
        if wsum == 0:
            break
        w /= wsum
        step = initial_step / (1.0 + 0.01 * s)
        grad = np.zeros(n)
        for p in range(n):
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                contrib[vb] -= h[jb[vb]]
            mask = active[m_min:m_max + 1]
            if mask.any():
                grad[p] = np.dot(w[m_min:m_max + 1], contrib)
        h = np.clip(h - step * grad, 0.0, 1.0)
        h = project_feasible(h, n)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_polish(h, n, alpha, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return smooth_obj(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(fun, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def directed_perturb(h, n, strength):
    """Perturb h to reduce the worst correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    # Find coordinates that most reduce c[k_star]
    # c[k_star] = sum_i h[i] * (1-h[i-(k_star-(n-1))])
    shift = k_star - (n - 1)
    # To reduce c[k_star], we want to decrease h[i] where (1-h[i-shift]) > 0
    # i.e., where h[i-shift] < 1
    # And increase h[i-shift] where h[i] > 0
    # Simple approach: move h toward (1-h at the shifted position)
    h_new = h.copy()
    valid_fwd = (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n)
    valid_bwd = (np.arange(n) - shift >= 0) & (np.arange(n) - shift < n)
    # Decrease h[i] where h[i-shift] is small (increases 1-h[i-shift] term... wait)
    # c[k] = sum h[i] * (1-h[i-shift])
    # To decrease: decrease h[i] where (1-h[i-shift]) is large, or increase h[i-shift] where h[i] is large
    # Gradient of c[k] w.r.t. h[i] = (1-h[i-shift]) - h[i+shift] (approximately)
    grad_k = np.zeros(n)
    for i in range(n):
        contrib = 0.0
        j_fwd = i - shift
        if 0 <= j_fwd < n:
            contrib += (1.0 - h[j_fwd])
        j_bwd = i + shift
        if 0 <= j_bwd < n:
            contrib -= h[j_bwd]
        grad_k[i] = contrib
    # Move in negative gradient direction (to decrease c[k_star])
