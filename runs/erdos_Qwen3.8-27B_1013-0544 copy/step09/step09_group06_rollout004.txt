def binary_local_search(h, n, max_iters=10000, deadline=None):
    """Local search on binary sequences maintaining sum=n/2"""
    h = np.round(h)
    # Fix sum
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target = n // 2
    while len(ones) > target:
        # Remove a 1 that's "least useful"
        idx = ones[len(ones)-1]
        h[idx] = 0
        ones = ones[:-1]
        zeros = np.append(zeros, idx)
    while len(ones) < target:
        idx = zeros[0]
        h[idx] = 1
        ones = np.append(ones, idx)
        zeros = zeros[1:]
    
    def get_score(h):
        c = np.correlate(h, 1-h, mode='full')
        return c.max()
    
    best_score = get_score(h)
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        # Try a random pair swap: change 1 at i to 0, 0 at j to 1
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_score = get_score(h)
        if new_score > best_score:
            # Revert
            h[i] = 1
            h[j] = 0
        else:
            best_score = new_score
            ones = np.setdiff1d(ones, [i])
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.append(zeros, i)
    return h, best_score * (2.0/n)

                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-8 or h0 > 1.0 - 1e-8:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_local_search(h, n, rng, max_iters, deadline):
    """Local search on binary sequences with incremental correlation tracking."""
    h = np.round(h.copy())
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # Fix count
    while len(ones) > target:
        idx = ones[np.random.RandomState(int(rng[0]*1e6 + len(ones))).randint(len(ones))]
        h[idx] = 0
        ones = np.delete(ones, np.where(ones == idx)[0][0])
        zeros = np.append(zeros, idx)
    while len(ones) < target:
        idx = zeros[0]
        h[idx] = 1
        ones = np.append(ones, idx)
        zeros = np.delete(zeros, 0)
    
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    best = corr.max()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Apply swap: h[i] 1->0, h[j] 0->1
        # Update correlation
        # h[i] was 1, now 0
        # For corr[n-1+s] (s>=0): was contributing (1-h[i+s]), now 0. Change: h[i+s]-1
        # For corr[n-1-s] (s>0): was contributing h[i+s]*(1-1)=0, now h[i+s]*(1-0)=h[i+s]. Change: +h[i+s]
        # h[j] was 0, now 1
        # For corr[n-1+s] (s>=0): was contributing 0, now (1-h[j+s]). Change: 1-h[j+s]
        # For corr[n-1-s] (s>0): was contributing h[j+s]*(1-0)=h[j+s], now h[j+s]*(1-1)=0. Change: -h[j+s]
        
        new_corr = corr.copy()
        ni = int(i)
        nj = int(j)
        
        # i: 1 -> 0
        s_max_i = n - 1 - ni
        for s in range(s_max_i + 1):
            new_corr[n-1+s] += (h[ni+s] - 1)  # remove contribution
        for s in range(1, s_max_i + 1):
            new_corr[n-1-s] += h[ni+s]  # add new contribution
        
        # j: 0 -> 1
        s_max_j = n - 1 - nj
        for s in range(s_max_j + 1):
            new_corr[n-1+s] += (1 - h[nj+s])  # add new contribution
        for s in range(1, s_max_j + 1):
            new_corr[n-1-s] -= h[nj+s]  # remove old contribution
        
        new_max = new_corr.max()
        if new_max <= best:
            corr = new_corr
            best = new_max
            h[ni] = 0
            h[nj] = 1
            ones = np.delete(ones, np.where(ones == ni)[0][0])
            ones = np.append(ones, nj)
            zeros = np.delete(zeros, np.where(zeros == nj)[0][0])
            zeros = np.append(zeros, ni)
        # else: revert (do nothing)
    
    return h, best * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:

def apply_swap(corr, h, i, j, n):
    new_corr = corr.copy()
    ni, nj = int(i), int(j)
    si, sj = n - 1 - ni, n - 1 - nj
    new_corr[n-1 : n-1+si+1] += (h[ni : ni+si+1] - 1)
    new_corr[n-1 : n-1+sj+1] += (1 - h[nj : nj+sj+1])
    if si > 0:
        new_corr[n-1-si : n-1] += h[ni+1 : ni+si+1]
    if sj > 0:
        new_corr[n-1-sj : n-1] -= h[nj+1 : nj+sj+1]
    return new_corr

def binary_ls(h, n, rng, max_iters, deadline):
    h = np.round(h.copy()).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    while len(ones) > target:
        idx = ones.pop(rng.randint(len(ones)))
        h[idx] = 0
        zeros.append(idx)
    while len(ones) < target:
        idx = zeros.pop(rng.randint(len(zeros)))
        h[idx] = 1
        ones.append(idx)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best = float(corr.max())
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        nc = apply_swap(corr, h, i, j, n)
        nm = float(nc.max())
        if nm <= best + 1e-15:
            corr, best = nc, nm
            h[i], h[j] = 0.0, 1.0
            ones.remove(int(i))
            ones.append(int(j))
            zeros.remove(int(j))
            zeros.append(int(i))
    return h, best * (2.0 / n)

def smooth_obj(x, n, alpha, c, m):
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alpha, maxiter=400):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, c, m), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):

    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
