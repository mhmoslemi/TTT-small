            #   h[i] = h[j-k'] when j-k' = i, i.e., j = i+k', valid for 0 <= i <= n-1-k': contributes -h[i+k']

    """Compute grad[m, i] = dc[m]/dh[i]"""
    grad = np.zeros((2*n-1, n))
    # k=0
    grad[n-1, :] = 1.0 - 2.0*h
    # k = 1 to n-1
    for k in range(1, n):
        # For m = n-1+k and m = n-1-k (same gradient)
        g = np.zeros(n)
        # (1-h[i-k]) for i >= k
        g[k:] = 1.0 - h[:n-k]
        # -h[i+k] for i < n-k
        g[:n-k] += -h[k:]
        grad[n-1+k, :] = g
        grad[n-1-k, :] = g
    return grad

    """Optimize pair (i,j). Returns optimal delta and new c, grad if changed."""
    a, b = h[i], h[j]
    s = a + b
    delta_min = max(-a, b-1)
    delta_max = min(1-a, b)
    
    if delta_min >= delta_max:
        return 0.0
    
    c0 = c[n-1]
    d = grad[:, i] - grad[:, j]  # shape (2n-1,)
    d[n-1] = 0  # We handle k=0 separately
    
    # Quadratic: g(delta) = c0 + delta*(b-a) - 2*delta^2
    # Lines (for m != n-1): L_m(delta) = c[m] + delta*d[m]
    
    # Find candidate deltas
    candidates = [delta_min, delta_max, (b-a)/4.0, 0.0]
    
    # Intersections of quadratic with lines
    # g(delta) = c[m] + d[m]*delta
    # c0 + delta*(b-a) - 2delta^2 = c[m] + d[m]*delta
    # -2delta^2 + delta*((b-a) - d[m]) + (c0 - c[m]) = 0
    # 2delta^2 - delta*((b-a) - d[m]) - (c0 - c[m]) = 0
    for m in range(2*n-1):
        if m == n-1:
            continue
        A = 2.0
        B = -((b-a) - d[m])
        C = -(c0 - c[m])
        disc = B*B - 4*A*C
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            delta1 = (-B + sqrt_disc) / (2*A)
            delta2 = (-B - sqrt_disc) / (2*A)
            for delta in [delta1, delta2]:
                if delta_min - 1e-12 <= delta <= delta_max + 1e-12:
                    candidates.append(delta)
    
    # Evaluate objective at each candidate
    best_val = float('inf')
    best_delta = 0.0
    for delta in candidates:
        delta = np.clip(delta, delta_min, delta_max)
        # New c0
        new_c0 = c0 + delta*(b-a) - 2*delta*delta
        # New lines
        new_c = c0 + delta*d  # This is for all m (the linear approximation)
        # But for m=n-1, we need the quadratic
        new_c[n-1] = new_c0
        val = new_c.max()
        if val < best_val:
            best_val = val
            best_delta = delta

With 600 pairs per pass, that's 216M operations per pass. At ~10⁹ ops/s, that's 0.2 seconds per pass. We can do about 5000 passes in 1000 seconds. That's very good!

Actually, I can optimize further. Instead of computing d = grad[:,i] - grad[:,j] from scratch each time, I can note that:
grad[m,i] for m = n-1+k is: (1-h[i-k])[i≥k] - h[i+k][i<n-k]

So d[m] for m = n-1+k is:
(1-h[i-k])[i≥k] - h[i+k][i<n-k] - (1-h[j-k])[j≥k] + h[j+k][j<n-k]

This can be computed in O(1) for each k (just a few comparisons and lookups). So computing the full d vector is O(n), which is what we already have.

def compute_grad_matrix(h, n):
    """Compute grad[m, i] = dc[m]/dh[i] for all m, i."""
    grad = np.zeros((2*n-1, n))
    grad[n-1, :] = 1.0 - 2.0*h
    for k in range(1, n):
        g = np.zeros(n)
        g[k:] = 1.0 - h[:n-k]
        g[:n-k] += -h[k:]
        grad[n-1+k, :] = g
        grad[n-1-k, :] = g
    return grad
