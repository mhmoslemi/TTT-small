                    xl = max(xl, 0.0)

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def active_set_descent(h, n, max_active, max_iter, deadline):
    h = h.copy()
    delta = n - 1
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        active = np.where(c >= c_max * 0.999)[0]
        active = active[:max_active]
        d = np.zeros(n)
        for m in active:
            for i in range(n):
                jb = i - m + delta
                jf = i + m - delta
                if 0 <= jb < n:
                    d[i] += (1.0 - h[jb])
                if 0 <= jf < n:
                    d[i] -= h[jf]
        d = -d
        d -= d.mean()
        dnorm = np.linalg.norm(d, 1)
        if dnorm < 1e-15:
            break
        d = d / dnorm
        improved = False
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = project_feasible(h + alpha * d, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_c = c_try
                h = h_try
                improved = True
                break
        if not improved:
            break
    return h, best_c

def smooth_max_obj(h, n, alpha, lam):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = lam * (h.sum() - n / 2.0) ** 2
    return lse * (2.0 / n) + penalty

def lbfgsb_optimize(h, n, alpha, lam, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h.copy(), args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def project_feasible(h, n):

                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def active_set_descent(h, n, max_active, max_iter, deadline):
    h = h.copy()
    delta = n - 1
    best_c = compute_c5(h, n)
    tol = 1e-4
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        active = np.where(c >= c_max - tol)[0][:max_active]
        d = np.zeros(n)
        for m in active:
            for i in range(n):
                jb = i - m + delta
                jf = i + m - delta
                if 0 <= jb < n:
                    d[i] += (1.0 - h[jb])
                if 0 <= jf < n:
                    d[i] -= h[jf]
        d = -d
        d -= d.mean()
        dn = np.linalg.norm(d, 1)
        if dn < 1e-15:
            break
        d /= dn
        improved = False
        for alpha in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = project_feasible(h + alpha * d, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_c, h, improved = c_try, h_try, True
                break
        if not improved:
            tol *= 0.5
            if tol < 1e-7:
                break
    return h, best_c

def smooth_obj(h, n, alpha, lam):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    return lse * (2.0 / n) + lam * (h.sum() - n / 2.0) ** 2

def lbfgsb_optimize(h, n, alpha, lam, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)
