        top_idx = np.argsort(c)[::-1][:k]
        improved = False
        n_swaps = 0
        for p in range(n):

                break
            h0 = h[p]
            if h0 < 5e-5 or h0 > 1.0 - 5e-5:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):

                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0

                if m == n - 1:
                    diag_idx = idx
            c_peak = c[top_idx].astype(float)
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            candidates = [lo, hi, 0.0]
            for i in range(k):

                    if i == diag_idx or j == diag_idx:
                        continue

                n_swaps += 1
