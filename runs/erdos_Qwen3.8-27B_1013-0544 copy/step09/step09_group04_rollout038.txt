import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def solve_1d(c, h, p, n):
    """Find optimal h[p] given current c and h."""
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi, feas = 0.0, 1.0, True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-15:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo, 0.0) > r1 and min(x_hi, 1.0) < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                if abs(h0 - r1) < abs(r2 - h0):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-13:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order='sequential', skip_boundary=True):
    h = h.copy()
    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order == 'worst_coord':
            # Sort by contribution to worst lag
            k_star = np.argmax(c) - (n - 1)
            contrib = np.zeros(n)
            for p in range(n):
                j1 = p + k_star
                j2 = p - k_star
                if 0 <= j1 < n:
                    contrib[p] = h[p] * (1.0 - h[j1])
                if 0 <= j2 < n:
                    contrib[p] += (1.0 - h[p]) * h[j2]
            idx = n_idx[np.argsort(-contrib)]
        elif order == 'random':
            idx = n_idx.copy()
            np.random.shuffle(idx)
        else:
            idx = n_idx
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if skip_boundary and (h0 < 1e-12 or h0 > 1.0 - 1e-12):
                continue
            x_new = solve_1d(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:

    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0 / n)

def exact_cd(h, n, max_passes, deadline, order='sequential'):

    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order == 'worst':
            k_star = int(np.argmax(c) - (n - 1))
            contrib = np.zeros(n)
            for p in range(n):
                j1 = p + k_star
                j2 = p - k_star
                if 0 <= j1 < n:
                    contrib[p] = h[p] * (1.0 - h[j1])
                if 0 <= j2 < n:
                    contrib[p] += (1.0 - h[p]) * h[j2]
            idx = n_idx[np.argsort(-contrib)]
        elif order == 'random':
            idx = n_idx.copy()
            np.random.shuffle(idx)
        else:
            idx = n_idx
        for p in idx:

                improved = True

            break
        h = project_feasible(h, n)

def weighted_quad_obj(x, n, p_full):
    """Objective: sum_d p[d] * c[n-1+d](x)"""
    # c[m] = sum_i x[i] * (1-x)[i+m-(n-1)]
    # Weighted: sum_m p_full[m] * c[m]
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.dot(p_full, c))

    """Gradient of weighted_quad_obj."""
    # ∂/∂x[m] of sum_i x[i](1-x[i+d]) where d = m_shift
    # = sum_d p[d](1-x[m+d]) - sum_d p[d]x[m-d]
    grad = np.zeros(n)
    center = n - 1
    for m in range(n):
        s1, s2 = 0.0, 0.0
        for d_idx in range(1, n):
            d_pos = d_idx  # shift +d
            d_neg = -d_idx  # shift -d
            # p_full[center + d] is p[d]
            p_pos = p_full[center + d_pos]
            p_neg = p_full[center + d_neg]
            j1 = m + d_pos
            j2 = m + d_neg
            if 0 <= j1 < n:
                s1 += p_pos * (1.0 - x[j1])
            if 0 <= j2 < n:
                s2 += p_neg * x[j2]
        grad[m] = s1 - s2
    return grad

def primal_dual(h, n, deadline, iters=30):
    """Alternating min-max for the Erdős problem."""
    h = h.copy()
    n_full = 2 * n - 1
    p = np.ones(n_full) / n_full  # uniform distribution over lags
    
    for it in range(iters):
        if time.time() > deadline:
            break
        
        # Inner: minimize sum_d p[d] * c[d](h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        
        def obj(x):
            return weighted_quad_obj(x, n, p)
        
        def obj_grad(x):
            return weighted_quad_grad(x, n, p)
        
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h = project_feasible(res.x, n)
        except:
            pass
        
        # Outer: update p based on current c
        c = np.correlate(h, 1.0 - h, mode='full')
        # Use softmax with increasing temperature

        p = np.exp(c / temp)
        p /= p.sum()
        
        # Also do a few CD passes to refine
        h, _ = exact_cd(h, n, 3, deadline, order='sequential')
    
    return h, compute_c5(h, n)

    for freq in [0.7, 1.0, 1.3, 1.6, 2.0, 2.618]:
        if time.time() > deadline - 120:
            break
        x = np.linspace(0, 2, n)
        ph = rng.uniform(0, 2 * np.pi)
        hs = np.clip(0.5 + 0.4 * np.sin(np.pi * freq * x + ph), 0, 1)
        hs = project_feasible(hs, n)
        ho, co = exact_cd(hs, n, 10, deadline, order='sequential')
        if co < best_c:
            best_h, best_c = ho, co
    # Phase 6: Try different n
    for n_try in [600, 800, 1000]:
        if time.time() > deadline - 130:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt = slsqp_step(h_int, n_try, 5000, 150)
        h_opt, c_opt = exact_cd(h_opt, n_try, 12, deadline, order='sequential')
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 7: Final polish
    n_final = len(best_h)
    remaining = deadline - time.time() - 5
    max_passes = max(5, int(remaining / 2.5))
    hf, cf = exact_cd(best_h, n_final, max_passes, deadline, order='sequential')
    if cf < best_c:
        best_h, best_c = hf, cf
    # Phase 8: One more round of tiny restarts if time allows
    if time.time() < deadline - 30:
        for r in range(10):
            if time.time() > deadline - 20:
                break
            hp = project_feasible(best_h + rng.normal(0, 0.001, n_final), n_final)
            ho, co = exact_cd(hp, n_final, 6, deadline, order='sequential')
            if co < best_c:
                best_h, best_c = ho, co
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
