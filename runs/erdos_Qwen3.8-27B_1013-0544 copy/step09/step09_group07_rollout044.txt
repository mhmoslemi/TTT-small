    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

    k = len(a_coeff)
    if k == 0:

    candidates = [lo, hi, 0.0]

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd_fast(h, n, max_passes=200, deadline=None, top_k=80, order='sequential'):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        idx_order = np.arange(n)
        if order == 'random':
            rng.shuffle(idx_order)
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            total_affected = len(ms_all)
            if total_affected > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = ms_all[~np.isin(ms_all, ms)]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def targeted_leveling(h, n, deadline=None, max_iter=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_val = c.max()
        target = top_val - 0.01
        above = np.where(c > target)[0]
        if len(above) == 0:
            break
        for m in above[:5]:
            shift = m - (n - 1)
            sg = np.zeros(n)
            jf = np.arange(n) + shift
            jb = np.arange(n) - shift
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            sg[vf] += (1.0 - h[jf[vf]])
            sg[vb] -= h[jb[vb]]
            h_new = project_feasible(h - 0.005 * sg, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
    return h, best_c

def make_initial(n, seed=42):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.15 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x + 0.5)
    h = h + rng.normal(0, 0.05, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Aggressive CD with high top_k
    phase1_deadline = min(t0 + 250, deadline - 350)
    if time.time() < phase1_deadline:
        h1, c1 = coord_cd_fast(best_h, n, max_passes=300, deadline=phase1_deadline, top_k=80)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP polish
    slq_deadline = min(t0 + 400, deadline - 200)
    if time.time() < slq_deadline:
        h2, c2 = slsqp_polish(best_h, n, [50, 200, 1000, 5000], 200, slq_deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: More CD with even higher top_k
    phase3_deadline = min(t0 + 550, deadline - 150)
    if time.time() < phase3_deadline:
        h3, c3 = coord_cd_fast(best_h, n, max_passes=400, deadline=phase3_deadline, top_k=100)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Targeted leveling
    lev_deadline = min(t0 + 650, deadline - 100)
    if time.time() < lev_deadline:
        h4, c4 = targeted_leveling(best_h, n, deadline=lev_deadline, max_iter=200)
        if c4 < best_c:
            best_h, best_c = h4, c4
        if time.time() < deadline - 80:
            h4b, c4b = coord_cd_fast(best_h, n, max_passes=200, deadline=deadline - 70, top_k=100)
            if c4b < best_c:
                best_h, best_c = h4b, c4b

    # Phase 5: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 700, 400, 800]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd_fast(h_int, n_try, max_passes=150, deadline=deadline - 100, top_k=80)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 6: Multi-restart
    n_cur = len(best_h)
    restart_deadline = deadline - 60
    for r in range(8):
        if time.time() > restart_deadline:
            break
        s = 0.002 + 0.005 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        ho, co = coord_cd_fast(hp, n_cur, max_passes=100, deadline=deadline - 40, top_k=80)
        if co < best_c:
            best_h, best_c = ho, c4 if False else ho, co

    # Phase 7: Final SLSQP + CD
    if time.time() < deadline - 30:
        h7, c7 = slsqp_polish(best_h, n_cur, [5000, 20000, 100000], 400, deadline - 20)
        if c7 < best_c:
            best_h, best_c = h7, c7
        if time.time() < deadline - 10:
            h8, c8 = coord_cd_fast(best_h, n_cur, max_passes=500, deadline=deadline - 5, top_k=120)
            if c8 < best_c:
                best_h, best_c = h8, c8

    # Final: try 1-h symmetry
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    # Final cleanup
    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
