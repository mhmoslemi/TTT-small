    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break

                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:

            m_f_hi = min(2*n-2, 2*n-2-idx)

                unaff_extra = float(c[skipped].max()) if skipped else 0.0

                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == n-1:
                    diag_local = ii

            hi_d = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo_d, hi_d)

def basin_hop(h, n, rng, deadline, base_eps, decay, max_rounds):

    eps = base_eps

            break

        h_pert = h.copy()
        # Random subset of coordinates to perturb

        idx_pert = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx_pert] += rng.normal(0, eps, size=n_pert)
        h_pert = project_feasible(h_pert, n)
        # Re-optimize
        h_opt, c_opt = coord_descent(h_pert, n, max_passes=100, 
                                       deadline=min(deadline, time.time()+30), top_k=25)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
        eps *= decay
    return best_h, best_c

def slsqp_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
