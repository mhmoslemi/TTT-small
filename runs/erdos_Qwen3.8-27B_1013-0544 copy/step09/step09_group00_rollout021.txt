                jb = p - ms + (n - 1)

def smooth_obj_psum(x, n, p):
    """Power-sum smooth approximation of max."""
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if m <= 0:
        return 0.0
    # Use (sum(c^p))^(1/p) as smooth max
    c_pos = np.maximum(c, 0)
    val = (np.sum(c_pos ** p) / len(c)) ** (1.0 / p)
    return val * (2.0 / n)

def lse_obj(x, n, alpha):
    """Log-sum-exp smooth approximation."""
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if m <= 0:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cons(x):
            return x.sum() - n / 2.0
        def _cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, rng, K=10):
    """Generate a Fourier-based initial solution."""
    x = np.linspace(0, 2, n, endpoint=False)
    t = x
    signal = np.zeros(n)

        a = rng.normal(0, 1.0 / np.sqrt(k))
        b = rng.normal(0, 1.0 / np.sqrt(k))
        signal += a * np.cos(2 * np.pi * k * t / 2.0) + b * np.sin(2 * np.pi * k * t / 2.0)

    if sig_max - sig_min > 1e-10:
        h = (signal - sig_min) / (sig_max - sig_min)
    else:
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    return h

                elif z_lo > r2 and z_lo < r2 and z_hi > r2:
                    x_lo = max(x_lo, r2 - h0)
                elif z_lo > r1 and z_lo < r2:
                    x_lo = max(x_lo, r2 - h0)
                elif z_hi > r1 and z_hi < r2:
                    x_hi = min(x_hi, r1 - h0)
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            z_lo, z_hi = x_lo + h0, x_hi + h0
            if z_lo < r1 and z_hi > r2:
                if (r1 - z_lo) <= (z_hi - r2):
                    x_hi = min(x_hi, r1 - h0)
                else:
                    x_lo = max(x_lo, r2 - h0)
            elif z_lo > r1 and z_lo < r2:
                x_lo = max(x_lo, r2 - h0)
            elif z_hi > r1 and z_hi < r2:
                x_hi = min(x_hi, r1 - h0)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy().astype(float)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] += delta * (1.0 - 2.0 * h0) - delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if m <= 0:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x):
            return x.sum() - n / 2.0
        def _cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, rng, K=8):
    t = np.linspace(0, 2, n, endpoint=False)
    sig = np.zeros(n)
    for k in range(1, K + 1):
        amp = 1.0 / np.sqrt(k)
        sig += amp * (rng.normal() * np.cos(np.pi * k * t) + rng.normal() * np.sin(np.pi * k * t))
    smin, smax = sig.min(), sig.max()
    if smax - smin > 1e-10:
        h = (sig - smin) / (smax - smin)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3.0
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD from parent (with fixed diagonal)
    h_o, c_o = exact_cd(h_best, n, 150, deadline)
    if c_o < best_c:
        h_best, best_c = h_o, c_o

    # Phase 2: SLSQP + CD
    if time.time() < deadline - 200:
        h_o, c_o = slsqp_opt(h_best, n, [5000, 20000, 100000], 400, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
        h_o, c_o = exact_cd(h_best, n, 150, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 3: Multi-resolution
    n_list = [300, 400, 500, 600, 350, 450, 550, 650]
    for n_try in n_list:
        if time.time() > deadline - 250:
            break
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 100, deadline)
        if time.time() < deadline - 150:
            h_o, c_o = slsqp_opt(h_o, n_try, [10000, 50000], 300, deadline)
            h_o, c_o = exact_cd(h_o, n_try, 100, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 4: Fourier restarts
    if time.time() < deadline - 200:
        for trial in range(4):
            if time.time() > deadline - 80:
                break
            h_f = fourier_init(n, rng, K=4 + 4 * trial)
            blend = 0.25 + 0.15 * trial
            h_bl = project_feasible(blend * h_f + (1 - blend) * h_best, n)
            h_o, c_o = exact_cd(h_bl, n, 80, deadline)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 5: Gaussian shake
    if time.time() < deadline - 120:
        for r in range(6):
            if time.time() > deadline - 50:
                break
            s = 0.015 + 0.015 * r
            hp = project_feasible(h_best + rng.normal(0, s, n), n)
            h_o, c_o = exact_cd(hp, n, 80, deadline)
            if time.time() < deadline - 40:
                h_o, c_o = slsqp_opt(h_o, n, [50000], 200, deadline)
                h_o, c_o = exact_cd(h_o, n, 80, deadline)
            if c_o < best_c:
                h_best, best_c = h_o, c_o
                n = len(h_best)

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        n = len(h_best)
        h_o, c_o = exact_cd(h_best, n, 300, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
