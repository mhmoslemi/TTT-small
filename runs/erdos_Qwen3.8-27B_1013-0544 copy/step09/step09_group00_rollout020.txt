def exact_cd(h, n, max_passes, deadline):

    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(h, c, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        # Re-project to maintain feasibility
        h_new = project_feasible(h, n)
        if not np.allclose(h_new, h, atol=1e-12):
            c = np.correlate(h_new, 1.0 - h_new, mode='full')
            h = h_new
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 0)
    lse = np.log(np.sum(np.exp(shifted)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted - np.log(np.sum(np.exp(shifted))))
    # grad of lse/alpha with respect to c is w
    # grad of c_m with respect to x[p] is the same 'a' structure
    # This is expensive, so let SLSQP do numerical differentiation
    return None  # Let SLSQP use finite differences

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except:
        return h, compute_c5(h, n)

def fourier_perturb(h, n, rng, n_modes=5, scale=0.05):
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for k in range(n_modes):
        freq = (k + 1) * np.pi
        if rng.random() < 0.5:
            pert += scale * rng.normal() * np.cos(freq * x)
        else:
            pert += scale * rng.normal() * np.sin(freq * x)
    return project_feasible(h + pert, n)
