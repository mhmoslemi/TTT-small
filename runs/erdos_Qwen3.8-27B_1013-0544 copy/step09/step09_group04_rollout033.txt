    #   Valid when 0 <= i+m-(n-1) <= n-1, i.e., n-1-i <= m <= 2n-2-i
    
    # Term involving h[i] as the second factor (l+m-(n-1) = i, so l = i-m+(n-1)):
    #   Contribution: h[i-m+(n-1)] * (1 - h[i])
    #   When h[i] changes from 1 to 0, (1-h[i]) changes from 0 to 1
    #   So delta = h[i-m+(n-1)] * (1 - 0) - h[i-m+(n-1)] * (1 - 1) = h[i-m+(n-1)]
    #   Wait: before: h[l]*(1-h[i]) = h[l]*0 = 0 (when h[i]=1)
    #   After: h[l]*(1-0) = h[l] (when h[i]=0)
    #   So delta = h[l] = h[i-m+(n-1)]
    #   Valid when 0 <= i-m+(n-1) <= n-1, i.e., 0 <= m <= n-1+i and m >= i-(n-1)
    #   Since i < n, i-(n-1) < 1, so m >= 0 (essentially)
    #   And m <= n-1+i
    
    # Similarly for h[j] changing from 0 to 1:
    #   As first factor (l=j): before: 0*(1-h[...]) = 0, after: 1*(1-h[...])
    #   delta = (1-h[j+m-(n-1)])
    #   As second factor (l+m-(n-1)=j): before: h[l]*(1-0) = h[l], after: h[l]*(1-1) = 0
    #   delta = -h[l] = -h[j-m+(n-1)]
    
    # Let me just compute it numerically for the affected m values
    affected = np.zeros(2*n-1)
    affected[0] = 1  # marker
    
    # i as first index
    m1_lo = max(0, n-1-i)
    m1_hi = min(2*n-2, 2*n-2-i)
    for m in range(m1_lo, m1_hi+1):
        k = i + m - (n-1)
        affected[m] += 1
    
    # i as second index
    m2_lo = max(0, max(0, i-(n-1)))
    m2_hi = min(2*n-2, n-1+i)
    for m in range(m2_lo, m2_hi+1):
        affected[m] += 1
    
    # j as first index
    m3_lo = max(0, n-1-j)
    m3_hi = min(2*n-2, 2*n-2-j)
    for m in range(m3_lo, m3_hi+1):
        affected[m] += 1

    m4_lo = max(0, max(0, j-(n-1)))
    m4_hi = min(2*n-2, n-1+j)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_from_continuous(h, n):
    m = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:m]] = 1.0
    return h_bin

def binary_swap_opt(h_bin, n, deadline, rng, max_iters=20000):
    h_bin = h_bin.copy()
    c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    cur_max = float(c.max())
    best_max = cur_max
    patience = 0
    max_patience = 2000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        if cur_max < best_max:
            best_max = cur_max
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break
        found = False
        n_trials = 30
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            new_max = float(c_new.max())
            if new_max < cur_max - 1e-15:
                c = c_new
                cur_max = new_max
                found = True
                break
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        if not found:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h_bin[i] = 0
            h_bin[j] = 1
            c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            new_max = float(c.max())
            temp = max(0.5, 20.0 * (1.0 - it / float(max_iters)))
            if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / temp):
                cur_max = new_max
            else:
                h_bin[i] = 1
                h_bin[j] = 0
    return h_bin, cur_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)

        h_init = np.full(n, 0.5)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warmup
    if time.time() < t0 + budget_s * 0.12:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        for alpha in [300, 1500, 8000]:
            if time.time() > t0 + budget_s * 0.12:
                break
            try:
                res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 150, 'ftol': 1e-16})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass

    # Phase 2: Exact 1D CD on continuous
    cd_deadline = t0 + budget_s * 0.30
    h_cd, c_cd = coord_cd(best_h, n, 80, cd_deadline)
    if c_cd < best_c:
        best_h, best_c = h_cd, c_cd

    # Phase 3: Binary optimization (key new approach)
    bin_deadline = t0 + budget_s * 0.70
    h_bin = binary_from_continuous(best_h, n)
    h_bin_opt, c_bin = binary_swap_opt(h_bin, n, bin_deadline, rng)
    if c_bin < best_c:
        best_h, best_c = h_bin_opt, c_bin

    # Also try random binary starts
    for r in range(3):
        if time.time() > bin_deadline - 10:
            break
        h_rand_bin = np.zeros(n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
