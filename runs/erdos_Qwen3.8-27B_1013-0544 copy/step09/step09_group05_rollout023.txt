For s=0: both give i=p, term is h[p]*(1-h[p]) = x(1-x). Only counted once!
For s≠0: they're different terms (different i values), so both contribute.

So my earlier analysis is correct:
- For s≠0: linear in x, coefficient = (1-h[p-s])*[valid] - h[p+s]*[valid]
- For s=0: x - x^2 (the diagonal term)

Now let me write the final implementation. I'll be careful with the diagonal handling.

Let me also add a key optimization: instead of computing the full correlation from scratch for each coordinate, I can update it incrementally. When h[p] changes from h0 to x:
- For each m, c[m] changes by (x-h0)*a[m] (for m≠diag) or by (x-h0) - (x^2-h0^2) (for m=diag)

So I can maintain c incrementally! This saves the O(n^2) correlation computation per pass.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    s = np.arange(-(n - 1), n)
    js_d = p - s
    js_i = p + s
    vd = (js_d >= 0) & (js_d < n)
    vi = (js_i >= 0) & (js_i < n)
    a = np.zeros(2 * n - 1)
    a[vd] += (1.0 - h[js_d[vd]])
    a[vi] -= h[js_i[vi]]
    diag_m = n - 1
    c_base = c - a * h0
    a[diag_m] = 1.0
    c_base[diag_m] = c[diag_m] - h0 + h0 * h0
    non_diag_mask = np.ones(2 * n - 1, dtype=bool)
    non_diag_mask[diag_m] = False
    pos = non_diag_mask & (a > 1e-13)
    neg = non_diag_mask & (a < -1e-13)
    const_only = non_diag_mask & (np.abs(a) <= 1e-13)
    lo, hi = 0.0, float(np.max(c)) + 0.01
    for _ in range(32):
        T = (lo + hi) / 2.0
        if const_only.any() and c_base[const_only].max() > T + 1e-14:
            lo = T
            continue
        x_lo, x_hi = 0.0, 1.0
        if pos.any():
            x_hi = min(x_hi, float(np.min((T - c_base[pos]) / a[pos])))
        if neg.any():
            x_lo = max(x_lo, float(np.max((T - c_base[neg]) / a[neg])))
        disc = 1.0 - 4.0 * (T - c_base[diag_m])
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo >= r1 and x_hi <= r2:
                lo = T
                continue
        hi = T
    T_opt = hi
    x_lo, x_hi = 0.0, 1.0
    if pos.any():
        x_hi = min(x_hi, float(np.min((T_opt - c_base[pos]) / a[pos])))
    if neg.any():
        x_lo = max(x_lo, float(np.max((T_opt - c_base[neg]) / a[neg])))
    disc = 1.0 - 4.0 * (T_opt - c_base[diag_m])
    if disc >= 0:
        r1 = (1.0 - np.sqrt(disc)) / 2.0
        r2 = (1.0 + np.sqrt(disc)) / 2.0
        if x_lo < r1 < x_hi or (x_lo >= r1 and x_hi <= r2):
            if r1 - max(x_lo, 0) <= min(x_hi, 1) - r2:
                x_hi = min(x_hi, r1 - 1e-9)
            else:
                x_lo = max(x_lo, r2 + 1e-9)
        elif r1 < x_lo < r2:
            x_lo = r2 + 1e-9
        elif r1 < x_hi < r2:
            x_hi = r1 - 1e-9
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng, random_order=True):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def power_mean_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    if cm < 1e-15:
        return 0.0
    val = np.sum(c**p) / (2.0 * n - 1)
    return (val ** (1.0 / p)) * (2.0 / n)

def power_mean_obj2(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    if cm < 1e-15:
        return 0.0
    shifted = (c / cm) ** p
    return cm * (np.mean(shifted) ** (1.0 / p)) * (2.0 / n)

def slsqp_power(h, n, ps, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in ps:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(power_mean_obj2, best_h, args=(n, p), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0}],
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def slsqp_lse(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x, n=n, alpha=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0}],
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def shake_cd(h, n, rng, deadline, n_shakes=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_shakes):
        if time.time() > deadline - 30:
            break
        k = rng.randint(5, max(6, n // 20))
        indices = rng.choice(n, size=k, replace=False)
        pert = best_h.copy()
        for idx in indices:
            pert[idx] += rng.uniform(-0.3, 0.3)
        pert = project_feasible(pert, n)
        hc, cc = coord_descent(pert, n, 30, deadline, rng, random_order=True)
        if cc < best_c:
            best_h, best_c = hc, cc
    return best_h, best_c
