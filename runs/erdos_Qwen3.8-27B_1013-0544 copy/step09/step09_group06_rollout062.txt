import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_autocorr(h, n):
    """Compute d_m for all lags m."""
    # correlate(h, h, 'full') gives sum_i h[i]*h[i+m-(n-1)] for m=0..2n-2
    return np.correlate(h, h, mode='full')

def eval_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = eval_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def iterative_reweighted(h, n, max_iter, deadline, q=4.0, eta=0.5):

    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        d = np.correlate(h, h, mode='full')
        d_min = d.min()
        # Weights emphasizing small autocorrelations
        w = (d + 1e-10) ** (-q)
        w = w / w.sum()  # normalize
        
        # Gradient of Σ w_m * d_m = h^T W h is 2Wh
        # W*h = Σ w_m * (shifted h)_m
        # This is equivalent to: for each lag m, add w_m * circulant_shift(h, m)
        # which is: convolve-like operation
        # Actually, (Wh)_i = Σ_m w_m * Σ_j h[j] * M[m]_{i,j}
        # where M[m] is the shift matrix for lag m
        # This is equivalent to: (Wh)_i = Σ_j h[j] * Σ_m w_m * M[m]_{i,j}
        # = Σ_j h[j] * Wmat_{i,j}
        # where Wmat_{i,j} = Σ_m w_m * M[m]_{i,j}

        for m in range(2 * n - 1):

            if shift >= 0:
                # h[i + shift] for i = 0 to n-1-shift
                Wofh[:n-shift] += w[m] * h[shift:]
            elif shift < 0:
                # h[i + shift] for i = -shift to n-1
                Wofh[-shift:] += w[m] * h[:n+shift]
            else:
                Wofh += w[m] * h

        h_new = project_feasible(h_new, n)

Hmm, the SLSQP without gradient will be slow. Let me implement the analytical gradient.

The objective is: f(h) = dx * (Σ_m c_m^p)^(1/p) where c_m = Σ_i h[i](1-h[i+k_m])

∂f/∂h_j = dx * (1/p) * (Σ_m c_m^p)^(1/p - 1) * Σ_m p * c_m^(p-1) * ∂c_m/∂h_j
= dx * (Σ_m c_m^p)^(1/p - 1) * Σ_m c_m^(p-1) * ∂c_m/∂h_j

And ∂c_m/∂h_j = (1 - h[j+k_m]) + (-h[j-k_m]) = 1 - h[j+k_m] - h[j-k_m] (with boundary=0)

So:
∂f/∂h_j = dx * (Σ_m c_m^p)^(1/p - 1) * Σ_m c_m^(p-1) * (1 - h[j+k_m] - h[j-k_m])

Let me define:
- S = Σ_m c_m^p
- T_j = Σ_m c_m^(p-1) * (1 - h[j+k_m] - h[j-k_m])

Then ∂f/∂h_j = dx * S^(1/p - 1) * T_j

This is O(n * n) to compute (for each j, sum over all m). For n=600, that's 360K operations, which is fine.

Actually, T_j can be computed more efficiently. Notice that:
T_j = Σ_m c_m^(p-1) * (1 - h[j+k_m] - h[j-k_m])
= Σ_m c_m^(p-1) - Σ_m c_m^(p-1) * h[j+k_m] - Σ_m c_m^(p-1) * h[j-k_m]

The first term is constant (independent of j): S1 = Σ_m c_m^(p-1)
The second term: for fixed j, Σ_m c_m^(p-1) * h[j + m - (n-1)]
This is a "weighted correlation" of h with the weight vector c^(p-1).

As I discussed earlier, this can be computed as:
full_corr = np.correlate(c_wp1, h, mode='full')
where c_wp1 = c_m^(p-1) for m = 0..2n-2

And the second term for index j is full_corr[j] (need to verify indexing).

        return dx * float(np.sum(cp ** p) ** (1.0 / p))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    return (h_flip, c_flip) if c_flip < c_orig else (h, c_orig)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Thorough CD
    d1 = min(deadline - 350, t0 + 300)
    if time.time() < d1:
        for tk in [12, 25, 40]:
            if time.time() > d1:
                break
            h2, c2 = multi_cd(best_h, len(best_h), 400, d1, top_k=tk, adaptive=True)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 2: Subgradient
    d2 = deadline - 200
    if time.time() < d2:
        h2, c2 = subgradient_opt(best_h, len(best_h), 8000, d2)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP
    d3 = deadline - 80
    if time.time() < d3:
        for p in [15, 30, 60]:
            if time.time() > d3:
                break
            h2, c2 = slsqp_run(best_h, len(best_h), p, 250)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 4: More CD + random restarts
    n_cur = len(best_h)
    d4 = deadline - 40
    if time.time() < d4:
        h2, c2 = multi_cd(best_h, n_cur, 500, d4, top_k=30, adaptive=True)
        if c2 < best_c:
            best_h, best_c = h2, c2
        for r in range(6):
            if time.time() > d4:
                break
            s = 0.003 + 0.003 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            ho, co = multi_cd(hp, n_cur, 150, d4, top_k=15, adaptive=True)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Different n
    d5 = deadline - 15
    if time.time() < d5:
        x_orig = np.linspace(0, 2, len(best_h))
        for n_try in [550, 620, 580, 650]:
            if time.time() > d5:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_o, c_o = multi_cd(h_int, n_try, 150, d5, top_k=12, adaptive=True)
            h_o, c_o = try_symmetry(h_o, n_try)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_orig = x_new

    # Final
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
