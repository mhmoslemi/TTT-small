        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def soft_max_obj(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

def slsqp_refine(h, n, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in [5000, 20000, 80000]:
        if time.time() > deadline:
            break
        def _obj(x):
            return soft_max_obj(x, n, alpha)
        def _con(x):
            return x.sum() - n / 2.0
        def _con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD on parent
    h1, c1 = exact_cd(h_best, n, 250, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP
    h2, c2 = slsqp_refine(h_best, len(h_best), deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: CD after SLSQP
    h3, c3 = exact_cd(h_best, len(h_best), 200, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    cur_n = len(h_best)

    # Phase 4: Multi-n exploration
    for n_try in [350, 450, 550, 650, 800, 1000, 1200]:
        if time.time() > deadline - 180:
            break
        if n_try == cur_n:
            continue
        x_old = np.linspace(0, 2, cur_n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 120, deadline)
        if time.time() < deadline - 60:
            h_o, c_o = slsqp_refine(h_o, n_try, deadline)
            h_o, c_o = exact_cd(h_o, n_try, 80, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            cur_n = n_try

    # Phase 5: Structured perturbations
    n_cur = len(h_best)
    x_full = np.linspace(0, 2, n_cur)
    freqs = [1, 2, 3, 5, 7]
    for f in freqs:
        if time.time() > deadline - 80:
            break
        for eps in [0.02, 0.05, 0.1]:
            if time.time() > deadline - 40:
                break
            perturb = eps * np.sin(np.pi * f * x_full)
            hp = project_feasible(h_best + perturb, n_cur)
            h_o, c_o = exact_cd(hp, n_cur, 80, deadline)
            if c_o < best_c:
                h_best, best_c = h_o, c_o

    # Phase 6: Random perturbations
    for r in range(6):
        if time.time() > deadline - 40:
            break
        s = 0.003 + 0.005 * r
        hp = project_feasible(h_best + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = exact_cd(hp, n_cur, 100, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
            n_cur = len(h_best)

    # Phase 7: Final aggressive CD
    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 400, deadline)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
