def overlap_for_intervals(intervals, n):
    """Compute C5 for a 0-1 step function defined by intervals.
    intervals: list of (start, end) pairs (in continuous [0,2] coordinates)
    n: number of grid points
    Returns the max correlation value (before multiplying by dx).
    """
    # Create the 0-1 function
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # cell centers
    h = np.zeros(n)
    for a, b in intervals:
        h[(x >= a) & (x < b)] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max())

    """Compute max_t O(t) where O(t) = 1 - A(t) for a 0-1 function.

    Returns the maximum overlap value (continuous, not multiplied by dx).

    k = len(intervals)
    # Find all breakpoints
    bps = [0.0, 2.0]
    for i in range(k):
        for j in range(k):
            a_i, b_i_arr = intervals[i]
            a_j, b_j_arr = intervals[j]
            bps.extend([a_i - a_j, a_i - b_j_arr, b_i_arr - a_j, b_i_arr - b_j_arr])
    # Filter to [0, 2]
    bps = sorted(set(max(0.0, min(2.0, bp)) for bp in bps))
    
    def autocorr(t):

            for j in range(k):
                a_i, b_i_val = intervals[i]
                a_j, b_j_val = intervals[j]

                hi = min(b_i_val, b_j_val - t)
                if hi > lo:

        return total

    for t in bps:
        ov = 1.0 - autocorr(t)
        if ov > max_ov:
            max_ov = ov
    return max_ov

def cd_pass(h, n, order='random', deadline=None):
    # One pass of coordinate descent
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    if order == 'random':
        idx = np.random.permutation(n)
    else:
        idx = np.arange(n)
    for p in idx:
        if deadline and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        x_new = solve_coord(c, h, p, n)
        if abs(x_new - h0) > 1e-12:
            h[p] = x_new
            improved = True
    return improved

def cd_optimize(h, n, max_passes, deadline, order='random'):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = cd_pass(h, n, order, deadline)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:

    h = h.copy()
