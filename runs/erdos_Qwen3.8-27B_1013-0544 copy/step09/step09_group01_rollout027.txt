            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=45):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP init
    h1, c1 = slsqp_optimize(best_h, n, [200, 2000, 20000], 300)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Momentum subgradient
    remaining = deadline - time.time() - 200
    max_iters = max(50, int(remaining / 0.005))
    h2, c2 = momentum_subgrad(best_h, n, max_iters, time.time() + remaining * 0.6)
    if c2 < best_c:
        best_h, best_c = h2, c1 if False else c2
    h2, c2 = coord_descent(best_h, n, 30, deadline, gs_iters=40)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: L-BFGS with penalty (no equality constraint)
    if time.time() < deadline - 150:
        h3, c3 = slsqp_optimize(best_h, n, [500, 5000, 50000], 400, penalty=5000.0)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: CD + subgradient cycle
    if time.time() < deadline - 100:
        h4, c4 = coord_descent(best_h, n, 40, deadline, gs_iters=50)
        if c4 < best_c:
            best_h, best_c = h4, c4
        sub_iters = max(30, int((deadline - time.time() - 60) / 0.004))
        h4, c4 = momentum_subgrad(best_h, n, sub_iters, deadline - 10)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-restart
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        for r in range(5):
            if time.time() > deadline - 70:
                break
            if r % 2 == 0:
                s = 0.005 + 0.02 * (r / 4.0)
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            else:
                freq = 3.0 + 2.0 * r
                amp = 0.02 + 0.02 * (r / 4.0)
                hp = project_feasible(best_h + amp * np.sin(np.pi * freq * x), n_cur)
            h_o, c_o = slsqp_optimize(hp, n_cur, [2000, 20000], 200)
            h_o, c_o = momentum_subgrad(h_o, n_cur, 100, deadline - 60)
            h_o, c_o = coord_descent(h_o, n_cur, 20, deadline, gs_iters=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 30, deadline, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [100000, 1000000], 500)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
