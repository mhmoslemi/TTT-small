import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    # Same as before - find optimal value for h[p] given current c
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        x = float(np.clip(x, 0, 1))
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = x
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def active_subgradient_step(h, n, n_active=20, step_size=0.005):
    """One step of active-set subgradient descent."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    max_c = c_scaled.max()
    thresh = max_c - 0.005
    active_idx = np.where(c_scaled >= thresh)[0]
    if len(active_idx) == 0:

def lp_step(h, n, n_active=15, trust_radius=0.5):
    """Solve LP to reduce max of active correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    max_c = c_scaled.max()
    thresh = max_c - 0.003
    active_idx = np.where(c_scaled >= thresh)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_scaled)])
    active_idx = active_idx[:n_active]
    
    # Build LP: min t
    # s.t. g_i · h ≤ t - c_i for each active i
    #      0 ≤ h ≤ 1
    #      sum(h) = n/2
    #      ||h - h0|| ≤ trust_radius (approximate with box: h0 - r ≤ h ≤ h0 + r)
    
    M = len(active_idx)
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # c vector: [0, ..., 0, 1] (minimize t)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active m: g_m · h - t ≤ -c_m
    A_rows = []
    b_rows = []
    for m in active_idx:
        shift = m - (n - 1)
        g = np.zeros(n)
        if shift >= 0:
            if shift < n:
                g[0:n-shift] += (1.0 - h[shift:n])
                g[shift:n] -= h[0:n-shift]
        else:
            s = -shift
            if s < n:
                g[s:n] += (1.0 - h[0:n-s])
                g[0:n-s] -= h[s:n]
        # g · h - t ≤ -c[m]*dx
        row = np.zeros(n + 1)
        row[:n] = g * dx
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-c[m] * dx)
    
    # Box constraints: h0 - r ≤ h ≤ h0 + r (and 0 ≤ h ≤ 1)
    lo = np.maximum(0.0, h - trust_radius)

def lp_direction(h, n, n_active=20):
    """Find LP direction to reduce active peaks. Returns direction d = h_new - h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    max_c = c_scaled.max()
    thresh = max_c - 0.002
    active_idx = np.where(c_scaled >= thresh)[0][:n_active]
    
    M = len(active_idx)
    # Minimize t s.t. g_m · d ≤ -(c_m - t_target) ... 
    # Actually: minimize t s.t. c_m + g_m · d ≤ t
    # where g_m = ∇c_m(h), d = h_new - h
    # Variables: d (n) + t (1)
    # Objective: minimize t (but we want t < max_c, so minimize t)
    # Constraints: g_m · d - t ≤ -c_m for m in active
    #              -1 ≤ d + h ≤ 1, 0 ≤ d + h ≤ 1 (box on h_new)
    #              sum(d) = 0 (to preserve the constraint)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for m in active_idx:
        shift = m - (n - 1)
        g = np.zeros(n)
        if shift >= 0:
            if shift < n:
                g[0:n-shift] = 1.0 - h[shift:n]
                g[shift:n] -= h[0:n-shift]
        else:
            s = -shift
            if s < n:
                g[s:n] = 1.0 - h[0:n-s]
                g[0:n-s] -= h[s:n]
        row = np.zeros(n+1)
        row[:n] = g * dx
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-c[m] * dx)
    
    # Box on h_new = h + d
    lo = -h  # d ≥ -h (so h+d ≥ 0)
    hi = 1.0 - h  # d ≤ 1-h (so h+d ≤ 1)
    A_box = np.zeros((2*n, n+1))
    b_box = np.zeros(2*n)
    A_box[:n, :n] = -np.eye(n)
    b_box[:n] = -lo  # -d ≤ -lo => d ≥ lo
    A_box[n:, :n] = np.eye(n)
    b_box[n:] = hi   # d ≤ hi
    
    A_ub = np.vstack([A_rows, A_box])
    b_ub = np.concatenate([b_rows, b_box])
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if res.status == 0:
        return res.x[:n]  # direction d
    return None

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        x = float(np.clip(x, 0, 1))
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = x
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def grad_for_shift(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[0:n-shift] = 1.0 - h[shift:n]
            g[shift:n] -= h[0:n-shift]
    else:
        s = -shift
        if s < n:
            g[s:n] = 1.0 - h[0:n-s]
            g[0:n-s] -= h[s:n]
    return g

def lp_reduce(h, n, n_active=25, deadline=None):
    """LP-based simultaneous peak reduction with line search."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    max_c = c_scaled.max()
    thresh = max_c - 0.003
    active_idx = np.where(c_scaled >= thresh)[0][:n_active]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_scaled)])
    M = len(active_idx)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_rows = []
    b_rows = []
    for m in active_idx:
        shift = m - (n - 1)
        g = grad_for_shift(h, n, shift) * dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-c[m] * dx)
    A_box = np.zeros((2 * n, n + 1))
    b_box = np.zeros(2 * n)
    A_box[:n, :n] = -np.eye(n)
    b_box[:n] = -h
    A_box[n:, :n] = np.eye(n)
    b_box[n:] = 1.0 - h
    A_ub = np.vstack([A_rows, A_box])
    b_ub = np.concatenate([b_rows, b_box])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            d = res.x[:n]
            # Line search
            best_h = h.copy()
            best_c = max_c
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.5, 0.3]:
                h_try = project_feasible(h + alpha * d, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
            return best_h, best_c
    except Exception:
        pass
    return h, max_c

def block_cd_2d(h, n, deadline=None, seed=0):
    """2D block coordinate descent on random pairs."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    n_pairs = max(50, n // 3)
    for it in range(n_pairs):
        if deadline is not None and time.time() > deadline:
            break
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        hp, hq = h[p], h[p]
        if hp < 1e-8 or hp > 1 - 1e-8:
            continue
        # Grid search over (h[p], h[q])
        grid = np.linspace(0, 1, 12)
        c_base = np.correlate(h, 1.0 - h, mode='full')
        best_v = float('inf')
        best_xp, best_xq = hp, hq
        for xp in grid:
            h[p] = xp
            for xq in grid:
                h[q] = xq
                # Compute max correlation efficiently
                # Only correlations involving p or q change
                c_new = c_base.copy()
                for s in range(-(n-1), n):
                    affected = False
                    if s != 0:
                        pi = p + s
                        qi = q + s
                        if 0 <= pi < n or 0 <= qi < n:
                            affected = True
                    if affected:
                        val = 0.0
                        for i in range(n):
                            j = i + s
                            if 0 <= j < n:
                                val += h[i] * (1.0 - h[j])
                        c_new[n-1+s] = val
                c_new[n-1] = sum(h[i]*(1-h[i]) for i in range(n))
                c_val = float(c_new.max())
                if c_val < best_v:
                    best_v = c_val
                    best_xp, best_xq = xp, xq
        h[p], h[q] = best_xp, best_xq
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, best_c

def sa_block(h, n, deadline, seed=0, n_blocks=500):
    """Simulated annealing: flip random blocks."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    T_start = 0.005
    T_end = 0.0001
    block_sizes = [3, 5, 8, 12, 20]
    for it in range(n_blocks):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_blocks - 1))
        bs = block_sizes[rng.randint(len(block_sizes))]
        start = rng.randint(max(1, n - bs))
        old_block = h[start:start+bs].copy()
        h[start:start+bs] = rng.uniform(0, 1, bs)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        c_old = best_c
        if c_new < c_old or rng.uniform() < np.exp(-(c_new - c_old) / T):
            best_c = c_new
        else:
            h[start:start+bs] = old_block
            h = project_feasible(h, n)
    return h, best_c
