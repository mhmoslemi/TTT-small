      c = np.correlate(x, 1-x, mode='full')
      m = c.max()
      shifted = alpha*(c - m)
      mx = shifted.max()
      return (m + np.log(np.exp(shifted - mx).sum()).sum() / alpha * (2/n)

  def project_feasible(h, n):
      # Use linear constraint ∑h[i] ≈ 0.5 (to handle the dx weighting)
      target = n / 2.0
      h = np.clip(h.copy(), 0.0, 1.0)
      while True:
          diff = target - h.sum()
          if abs(diff) < 1e-15: break
          if diff > 0:
              room = 1.0 - h
              tot = np.sum(room)
              h += diff / tot * room
          else:
              tot = h.sum()
              h += diff / tot * h
          np.clip(h, 0.0, 1.0)
      return h
