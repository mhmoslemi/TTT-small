import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))

                    delta[vb] -= d * h[jb[vb]]

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 8 == 7:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def sinusoidal_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.48 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

def block_random_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = rng.randint(1, max(2, n // 30) + 1)
        if rng.random() > 0.5:
            h[pos:pos + blen] = rng.uniform(0.6, 1.0)
        else:
            h[pos:pos + blen] = rng.uniform(0.0, 0.4)
        pos += blen
    return project_feasible(h, n)

def multi_start_optimize(n, rng, deadline, best_c, best_h_ref, max_starts=8):
    best_h = best_h_ref.copy() if best_h_ref is not None else random_binary_init(n, rng)
    best_c = compute_c5(best_h, n)
    inits = []
    for i in range(max_starts):
        if i == 0 and best_h_ref is not None:
            inits.append(best_h_ref.copy())
        elif i < 3:
            inits.append(random_binary_init(n, rng))
        elif i < 5:
            inits.append(sinusoidal_init(n, rng.randint(3, 80), rng.uniform(0, 6.28)))
        else:
            inits.append(block_random_init(n, rng))
    for init in inits:
        if time.time() > deadline - 60:
            break
        h_o, c_o = coord_cd(init, n, max_passes=60, deadline=deadline, shuffle=True, rng=rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 600
        h_init = random_binary_init(n_init, rng)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Thorough CD on parent
    h1, c1 = coord_cd(best_h, n_init, max_passes=120, deadline=deadline, shuffle=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP polish on parent
    if time.time() < deadline - 400:
        h2, c2 = slsqp_polish(best_h, n_init, [5000, 20000, 80000, 200000], 350, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = coord_cd(best_h, n_init, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
        if c2b < best_c:
            best_h, best_c = h2b, c2b

    # Phase 3: Multi-resolution with diverse starts
    x_cur = np.linspace(0, 2, len(best_h))
    n_list = [150, 200, 250, 300, 350, 400, 500, 700, 800, 900, 1000, 1300, 1600]
    for n_try in n_list:
        if time.time() > deadline - 350:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = multi_start_optimize(n_try, rng, deadline, best_c, h_int, max_starts=4)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 4: Annealing on current best
    n_cur = len(best_h)
    temps = [0.12, 0.08, 0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    for t in temps:
        if time.time() > deadline - 250:
            break
        for _ in range(4):
            if time.time() > deadline - 250:
                break
            h_pert = project_feasible(best_h + rng.normal(0, t, n_cur), n_cur)
            h_a, c_a = coord_cd(h_pert, n_cur, max_passes=45, deadline=deadline, shuffle=True, rng=rng)
            if c_a < best_c:
                best_h, best_c = h_a, c_a

    # Phase 5: Diverse initializations at current n
    for r in range(10):
        if time.time() > deadline - 180:
            break
        if r < 3:
            init = random_binary_init(n_cur, rng)
        elif r < 6:
            f = rng.randint(5, 100)
            init = sinusoidal_init(n_cur, f, rng.uniform(0, 6.28))
        else:
            init = block_random_init(n_cur, rng)
        h_d, c_d = coord_cd(init, n_cur, max_passes=70, deadline=deadline, shuffle=True, rng=rng)
        if c_d < best_c:
            best_h, best_c = h_d, c_d

    # Phase 6: Coarse-to-fine from very small n
    if time.time() < deadline - 200:
        h_coarse = random_binary_init(80, rng)
        h_coarse, c_coarse = coord_cd(h_coarse, 80, max_passes=100, deadline=deadline, shuffle=True, rng=rng)
        for n_fine in [160, 320, 640, 960]:
            if time.time() > deadline - 180:
                break
            x_f = np.linspace(0, 2, n_fine)
            x_c = np.linspace(0, 2, 80)
            h_f = project_feasible(np.interp(x_f, x_c, h_coarse), n_fine)
            h_f, c_f = coord_cd(h_f, n_fine, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Phase 7: Final polish
    if time.time() < deadline - 100:
        hf, cf = slsqp_polish(best_h, n_cur, [100000, 400000, 800000, 2000000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    hf, cf = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
    if cf < best_c:
        best_h, best_c = hf, cf
    if time.time() < deadline - 20:
        hf, cf = slsqp_polish(best_h, n_cur, [400000, 1500000], 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n_cur, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
