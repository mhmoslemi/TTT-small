import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def correlate_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    """Find optimal value for coordinate p given current correlation c."""
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    
    m = n - 1 + s
    b = c[m] - a * h0
    
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    
    # Binary search for the minimax point
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x
