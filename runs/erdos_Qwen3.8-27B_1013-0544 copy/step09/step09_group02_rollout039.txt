```python
def m_sequence_init(n, m_bits=8):
    """Create an m-sequence based initialization."""
    # Generate an m-sequence of length 2^m - 1
    length = 2**m_bits - 1
    # Use LFSR with a primitive polynomial
    # For m=8, use polynomial x^8 + x^4 + x^3 + x^2 + 1
    reg = 1  # initial state
    seq = []
    for _ in range(length):
        seq.append(reg & 1)
        # Feedback
        feedback = ((reg >> 7) ^ (reg >> 3) ^ (reg >> 2) ^ (reg >> 1) ^ reg) & 1
        reg = (reg >> 1) | (feedback << 7)
    # Tile to get n elements
    h = np.array([seq[i % length] for i in range(n)], dtype=float)
    # Adjust to satisfy sum constraint
    # h has roughly n/2 ones, need exactly n/2
    n_ones = h.sum()
    if n_ones > n/2:
        # Flip some 1s to 0s
        excess = int(n_ones - n/2)
        one_indices = np.where(h == 1)[0]
        rng = np.random.RandomState(42)
        rng.shuffle(one_indices)
        h[one_indices[:excess]] = 0
    elif n_ones < n/2:
        deficit = int(n/2 - n_ones)
        zero_indices = np.where(h == 0)[0]
        rng = np.random.RandomState(42)
        rng.shuffle(zero_indices)
        h[zero_indices[:deficit]] = 1
    return h
```

```python
def subgradient_step(h, n, lr, deadline=None):
    """One subgradient step on the max."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    dx = 2.0 / n
    # Subgradient of max_k c[k] w.r.t. h
    sg = np.zeros(n)
    shift = m_star - (n - 1)
    for i in range(n):
        jf = i + shift
        if 0 <= jf < n:
            sg[i] += (1.0 - h[jf])
        jb = i - shift
        if 0 <= jb < n:
            sg[i] -= h[jb]
    sg *= dx
    h_new = project_feasible(h - lr * sg, n)
    return h_new, compute_c5(h_new, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        if active.sum() == 0:
            active = vals == mval
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

    for p_idx in range(max_passes):

        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 8 == 7:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def soft_max_grad(h, n, beta, top_k=30):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    shifted = beta * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    g = np.zeros(n)
    idx = np.argsort(w)[::-1][:top_k]
    idx = idx[w[idx] > 1e-12]
    for k in idx:
        shift = k - (n - 1)
        wf = w[k]
        i_range = np.arange(n)
        valid_f = (i_range + shift >= 0) & (i_range + shift < n)
        valid_b = (i_range - shift >= 0) & (i_range - shift < n)
        if np.any(valid_f):
            g[valid_f] += wf * (1.0 - h[i_range[valid_f] + shift])
        if np.any(valid_b):
            g[valid_b] -= wf * h[i_range[valid_b] - shift]
    g *= (2.0 / n)
    return g

def lbfgs_softmax(h, n, deadline=None, betas=None):
    if betas is None:
        betas = [20, 50, 100, 200, 400]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for beta in betas:
        if deadline is not None and time.time() > deadline - 20:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = float(c.max())
            sh = beta * (c - m)
            mx = sh.max()
            return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / beta) * (2.0 / n)
        def obj_grad(x):
            g = soft_max_grad(x, n, beta)
            return g
        def sum_con(x):
            return x.sum() - n / 2.0
        def sum_con_jac(x):
            return np.ones(n)
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
        except Exception:
            pass
    return best_h, best_c

def active_subgrad(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Use top-3 active shifts for a smoothed subgradient
        top3 = np.argsort(c)[::-1][:3]
        vals = c[top3]
        w = np.exp(200 * (vals - vals[0]))
        w /= w.sum()
        sg = np.zeros(n)
        for wi, k in zip(w, top3):
            shift = k - (n - 1)
            i_range = np.arange(n)
            valid_f = (i_range + shift >= 0) & (i_range + shift < n)
            valid_b = (i_range - shift >= 0) & (i_range - shift < n)
            if np.any(valid_f):
                sg[valid_f] += wi * (1.0 - h[i_range[valid_f] + shift])
            if np.any(valid_b):
                sg[valid_b] -= wi * h[i_range[valid_b] - shift]
        sg *= (2.0 / n)
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.85
            stall += 1
            if stall > 50 or lr < 1e-6:
                break
    return h, best_c

def m_seq_init(n, seed=42):
    rng = np.random.RandomState(seed)
    m_bits = 8
    length = (1 << m_bits) - 1
    reg = 1
    seq = np.zeros(length, dtype=np.uint8)
    for s in range(length):
        seq[s] = reg & 1
        fb = ((reg >> 7) ^ (reg >> 3) ^ (reg >> 2) ^ (reg >> 1) ^ reg) & 1
        reg = (reg >> 1) | (fb << (m_bits - 1))
    h = np.tile(seq[:n % length if n % length else length].astype(float), n // length + 1)[:n]
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - cur]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: CD on parent
    h1, c1 = coord_cd(best_h, n_init, max_passes=100, deadline=deadline, shuffle=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Multi-resolution CD
    x_parent = np.linspace(0, 2, len(best_h))
    for n_try in [200, 300, 350, 450, 500, 550, 650, 750, 850]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_parent, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=120, deadline=deadline,
                                shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_parent = x_new
    # Phase 3: Active-set subgradient on best
    n_cur = len(best_h)
    h3, c3 = active_subgrad(best_h, n_cur, max_iter=200, deadline=deadline)
    if c3 < best_c:
        best_h, best_c = h3, c3
    h3b, c3b = coord_cd(best_h, n_cur, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
    if c3b < best_c:
        best_h, best_c = h3b, c3b
    # Phase 4: L-BFGS-B with soft-max
    if time.time() < deadline - 60:
        h4, c4 = lbfgs_softmax(best_h, n_cur, deadline=deadline, betas=[30, 80, 150])
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_cd(best_h, n_cur, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
        if c4b < best_c:
            best_h, best_c = h4b, c4b
    # Phase 5: M-sequence restarts
    for n_try in [300, 400, 500, n_init]:
        if time.time() > deadline - 50:
            break
        h_m = m_seq_init(n_try, seed=seed + n_try)
        h_m = project_feasible(h_m, n_try)
        h_opt, c_opt = coord_cd(h_m, n_try, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 6: Structured perturbations + CD
    n_cur = len(best_h)
    for r in range(25):
        if time.time() > deadline - 40:
            break
        freq = 1 + (r % 4)
        phase = rng.uniform(0, 2 * np.pi)
        x_grid = np.linspace(0, 2, n_cur)
        perturb = np.sin(np.pi * freq * x_grid + phase)
        s = 0.003 + 0.002 * r
        hp = project_feasible(best_h + s * perturb, n_cur)
        h_r, c_r = coord_cd(hp, n_cur, max_passes=50, deadline=deadline, shuffle=True, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    # Phase 7: Final L-BFGS + CD
    if time.time() < deadline - 30:
        h7, c7 = lbfgs_softmax(best_h, len(best_h), deadline=deadline, betas=[100, 300])
        if c7 < best_c:
            best_h, best_c = h7, c7
    h_final, c_final = coord_cd(best_h, len(best_h), max_passes=400, deadline=deadline,
                                shuffle=True, rng=rng)
    if c_final < best_c:
        best_h, best_c = h_final, c_final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
