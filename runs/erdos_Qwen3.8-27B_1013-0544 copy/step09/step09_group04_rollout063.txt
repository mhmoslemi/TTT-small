    return (m + lse / alpha) * (2.0 / n)

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        neg_m = a < -1e-15
        pos_m = a > 1e-15
        zero_m = np.abs(a) <= 1e-15
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-15:
            feas = False
        if feas and has_diag:
            R = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * R
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        R = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * R
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:

                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_sequential(h, n, max_passes, deadline):

**Idea: Subgradient / cutting plane approach**

Instead of optimizing the smoothed objective, I can use a cutting plane approach:
1. Start with a set of "active" lags S (initially just the argmax)
2. Solve: minimize max_{k in S} c_k(h) subject to constraints
3. Check if there's a lag k not in S with c_k > max_{k' in S} c_{k'}
4. If yes, add k to S and go to step 2
5. If no, we've found the true max

Step 2 is: minimize t subject to c_k(h) <= t for k in S, sum(h)=n/2, 0<=h<=1.
This is a non-convex QP (since c_k is quadratic and not convex).

But we can solve it using SLSQP with the active constraints. The advantage is that we only need to satisfy a few constraints (|S| is small), making the SLSQP problem easier.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    L_lo = unaff
    L_hi = cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        neg_m = a < -1e-15
        pos_m = a > 1e-15
        zero_m = np.abs(a) <= 1e-15
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-15:
            feas = False
        if feas and has_diag:
            R = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * R
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        R = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * R
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            x_new = solve_1d(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 8000, 30000]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        maxiter = 400 if alpha > 2000 else 600
        try:
            res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                h, best_c = new_h, new_c
        except Exception:
            pass
    return h, best_c

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def optimize_at_n(h, n, deadline, skip_slsqp=False):
    if not skip_slsqp:
        h_s, c_s = slsqp_optimize(h, n, min(deadline, time.time() + 120))
        if c_s < compute_c5(h, n):
            h = h_s
    remaining = deadline - time.time()
    n_passes = max(15, int(remaining / 3.5))
    h, c = cd_pass(h, n, n_passes, deadline, rng=None)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Optimize at original n (SLSQP + CD)
    t1 = deadline - 250
    h1, c1 = optimize_at_n(best_h, n, t1)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Random-ordered CD for escape
    if time.time() < deadline - 150:
        h2, c2 = cd_pass(best_h, len(best_h), 40, deadline - 150, rng=rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Try different resolutions
    for n_try in [800, 900, 600, 500]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        h_int = interp_to_n(best_h, n_try)
        t_end = min(deadline - 30, time.time() + 130)
        h_t, c_t = optimize_at_n(h_int, n_try, t_end)
        if c_t < best_c:
            best_h, best_c = h_t, c_t

    # Phase 4: Perturbation restarts
    for r in range(10):
        if time.time() > deadline - 25:
            break
        n_cur = len(best_h)
        s = 0.015 + 0.012 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        t_end = min(deadline - 10, time.time() + 35)
        h_p, c_p = cd_pass(hp, n_cur, 30, t_end, rng=rng)
        if c_p < best_c:
            best_h, best_c = h_p, c_p

    # Phase 5: Final sequential CD polish
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        max_passes = max(10, int((deadline - time.time() - 2) / 3.0))
        h_f, c_f = cd_pass(best_h, n_cur, max_passes, deadline, rng=None)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
