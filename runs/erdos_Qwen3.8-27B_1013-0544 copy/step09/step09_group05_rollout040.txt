def balance(arr,n,rng):
    arr=np.asarray(arr,np.int8).copy(); tgt=n//2; s=int(arr.sum())
    if s>tgt:
        idx=np.flatnonzero(arr); k=s-tgt

    elif s<tgt:

def thue_seq(n):
    a=np.array([0],np.int8)
    while a.size<n: a=np.concatenate((a,1-a))
    return a[:n]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(80):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            idx = h < 1.0 - 1e-14
            if not idx.any():
                break
            h[idx] += d / int(idx.sum())
            np.clip(h, 0.0, 1.0, out=h)
        else:
            idx = h > 1e-14
            if not idx.any():
                break
            h[idx] += d / int(idx.sum())
            np.clip(h, 0.0, 1.0, out=h)
    d = tgt - h.sum()
    if abs(d) > 1e-15:
        h += d / float(n)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def balance(arr, n, rng):
    arr = np.asarray(arr, dtype=np.int8).copy()
    tgt = n // 2
    s = int(arr.sum())
    if s > tgt:
        idx = np.flatnonzero(arr)
        k = s - tgt
        if idx.size >= k:
            arr[rng.choice(idx, k, replace=False)] = 0
    elif s < tgt:
        idx = np.flatnonzero(arr == 0)
        k = tgt - s
        if idx.size >= k:
            arr[rng.choice(idx, k, replace=False)] = 1
    return arr

def thue_seq(n):
    a = np.array([0], dtype=np.int8)
    while a.size < n:
        a = np.concatenate((a, 1 - a))
    return a[:n]

def arr_to_X(arr):
    X = 0
    for i in np.flatnonzero(arr):
        X |= 1 << int(i)
    return X

def X_to_arr(X, n):
    a = np.zeros(n, dtype=float)
    for i in range(n):
        if (X >> i) & 1:
            a[i] = 1.0
    return a

def bit_max_count(X, n, mask, cutoff=10**9):
    Z = mask ^ X
    best = 0
    bc = int.bit_count
    for d in range(1, n):
        v = bc((Z << d) & X)
        if v > best:
            best = v
            if best > cutoff:
                return best
        v = bc((X << d) & Z)
        if v > best:
            best = v
            if best > cutoff:
                return best
    return best

def X_lists(X, n):
    ones, zeros = [], []
    for i in range(n):
        if (X >> i) & 1:
            ones.append(i)
        else:
            zeros.append(i)
    return ones, zeros

def binary_sa(arr, n, tlim, rng):
    mask = (1 << n) - 1
    X = arr_to_X(arr)
    cur = bit_max_count(X, n, mask)
    bestX, best = X, cur
    ones, zeros = X_lists(X, n)
    dl = time.time() + tlim
    stall = 0
    B = 128
    while time.time() < dl:
        bn, ba, bb, bia, bib = cur, -1, 0, 0, 0
        cutoff = bn - 1
        for j in range(B):
            if j % 32 == 0 and time.time() > dl:
                break
            ia = int(rng.randint(len(ones)))
            ib = int(rng.randint(len(zeros)))
            a, b = ones[ia], zeros[ib]

    if n % 2:
        n -= 1
    n = max(64, min(896, n))

def coord_intervals(T, U, A, b, D):
    if U > T + 1e-11:
        return None
    lo, hi, e = 0.0, 1.0, 1e-14
    z = np.abs(A) <= e
    if z.any() and float(np.max(b[z])) > T + 1e-10:
        return None
    p = A > e
    if p.any():
        hi = min(hi, float(np.min((T - b[p]) / A[p])))
    q = A < -e
    if q.any():
        lo = max(lo, float(np.max((T - b[q]) / A[q])))
    if lo > hi + 1e-11:
        return None
    td = T - D
    if td >= 0.25 - 1e-12:
        out = [(lo, hi)]
    elif td < -1e-11:
        return None
    else:
        s = np.sqrt(max(0.0, 1.0 - 4.0 * td))
        r1, r2 = (1.0 - s) / 2.0, (1.0 + s) / 2.0
        out = []
        if lo <= r1 + 1e-11 and lo <= min(hi, r1) + 1e-11:
            out.append((lo, min(hi, r1)))
        if hi >= r2 - 1e-11 and max(lo, r2) <= hi + 1e-11:
            out.append((max(lo, r2), hi))
    res = []
    for a, bv in out:
        a = max(0.0, min(1.0, a))
        bv = max(0.0, min(1.0, bv))
        if a <= bv + 1e-12:
            res.append((a, bv))
    return res or None

def solve_coord(c, h, p, n):
    idx = np.arange(p, 2 * n - 1 - p, dtype=int)
    ell = idx - (n - 1)
    base = c[idx]
    A = np.zeros(len(idx), dtype=float)
    e1 = p - ell
    m = (e1 >= 0) & (e1 < n)
    A[m] += 1.0 - h[e1[m]]
    e2 = p + ell
    m = (e2 >= 0) & (e2 < n)
    A[m] -= h[e2[m]]
    h0 = float(h[p])
    di = int(np.flatnonzero(ell == 0)[0])
    D = float(base[di]) - h0 * (1.0 - h0)
    Ad = A.copy()
    Ad[di] = 0.0
    bnd = base - A * h0
    bnd[di] = 0.0
    U = 0.0
    if p > 0:
        U = max(U, float(np.max(c[:p])))
    hi = int(idx[-1])
    if hi < 2 * n - 2:
        U = max(U, float(np.max(c[hi + 1:])))
    L, H = 0.0, float(np.max(c))
    for _ in range(24):
        T = 0.5 * (L + H)
        if coord_intervals(T, U, Ad, bnd, D) is not None:
            H = T
        else:
            L = T
    ints = coord_intervals(H, U, Ad, bnd, D)
    if ints is None:
        return h0
    for a, bv in ints:
        if a <= h0 <= bv:
            return h0
    bx, bd = h0, 1e100
    for a, bv in ints:
        if abs(a - h0) < bd:
            bd, bx = abs(a - h0), a
        if abs(bv - h0) < bd:
            bd, bx = abs(bv - h0), bv
    return float(min(1.0, max(0.0, bx)))

def coord_polish(h, n, dl, maxp=12):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best = compute_c5(h, n)
    bh = h.copy()
    hw = h.copy()
    chunk = 12 if n <= 700 else 24
    for _ in range(maxp):
        if time.time() > dl - 2:
            break
        imp = False
        c = np.correlate(hw, 1.0 - hw, mode='full')
        upd = 0
        for p in range(n):
            if time.time() > dl - 2:
                break
            if upd and upd >= chunk:
                c = np.correlate(hw, 1.0 - hw, mode='full')
                upd = 0
            M = float(np.max(c))
            lo, hi = p, 2 * n - 2 - p
            if M - float(np.max(c[lo:hi + 1])) > 1e-11:
                continue
            xn = solve_coord(c, hw, p, n)
            if abs(xn - hw[p]) > 1e-13:
                hw[p] = xn
                imp = True
                upd += 1
        if not imp:
            break
        hw = project_feasible(hw, n)
        nc = compute_c5(hw, n)
        if nc < best - 1e-13:
            best, bh = nc, hw.copy()
        else:
            break
    return bh, best

def subgrad_polish(h, n, dl, rng, K=4):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best = compute_c5(h, n)
    bh = h.copy()
    lr, fail = 0.05, 0
    while time.time() < dl - 2:
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[-K:]
        g = np.zeros(n, dtype=float)
        for kk in top:
            ell = int(int(kk) - (n - 1))
            if ell >= 0:
                L = n - ell
                g[:L] -= h[ell:]
                g[ell:] += 1.0 - h[:L]
            else:
                d = -ell
                L = n - d
                g[d:] -= h[:L]
                g[:L] += 1.0 - h[d:]
        g /= float(len(top))
        h2 = project_feasible(h - lr * g, n)
        c2 = compute_c5(h2, n)
        if c2 < best - 1e-13:
            h, best, bh, fail = h2, c2, h2.copy(), 0
            lr = min(0.2, lr * 1.05)
        else:
            fail += 1
            lr *= 0.8
            if fail > 30:
                h2 = project_feasible(h + rng.normal(0, 0.03, n), n)
                c2 = compute_c5(h2, n)
                if c2 < best:
                    h, best, bh = h2, c2, h2.copy()
                fail = 0
                lr = 0.05
    return bh, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + max(1.0, float(budget_s))
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(160, 0.5)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h, best_c = h0.copy(), compute_c5(h0, n0)
    init_c = best_c
    alt_h, alt_c = None, 1e9

    base_n = n0 if n0 % 2 == 0 else max(64, n0 - 1)
    if base_n < 128:
        base_n = 384
    if base_n > 896:
        base_n = 896
    nlist = [base_n]
    for nn in (640, 512, 768, 896, 384):
        if nn not in nlist:
            nlist.append(nn)
    maxc = 3 if budget_s < 450 else (4 if budget_s < 800 else 5)
    nlist = nlist[:maxc]

    bin_budget = min(650.0, max(90.0, float(budget_s) * 0.55))
    bin_dl = t0 + bin_budget
    for i, nb in enumerate(nlist):
        left = bin_dl - time.time()
        if left < 25:
            break
        sl = min(150.0, left / max(1, len(nlist) - i))
        if sl < 18:
            sl = left * 0.85
        hb, cb = binary_campaign(best_h, nb, sl, rng)
        if cb is None:
            continue
        if cb < best_c:
            best_h, best_c, alt_h, alt_c = hb, cb, None, 1e9
        elif cb < alt_c:
            alt_h, alt_c = hb, cb

    starts = [(best_h, len(best_h))]
    if alt_h is not None and alt_c < best_c + 2e-4 and len(alt_h) != len(best_h):
        starts.append((alt_h, len(alt_h)))
    if len(best_h) != n0 and init_c <= best_c + 2e-4:
        starts.append((h0, n0))

    cont_dl = dl - 5
    for hs, ns in starts:
        if time.time() > cont_dl - 60:
            break
        rem = cont_dl - time.time()
        alloc = min(300.0, rem * 0.8) if len(starts) > 1 else min(600.0, rem)
        t1 = min(time.time() + alloc, cont_dl - 60)
        if t1 <= time.time():
            continue
        h1, c1 = subgrad_polish(hs, ns, t1, rng)
        cs = compute_c5(hs, ns)
        if c1 < cs:
            hc, cc = h1, c1
        else:
            hc, cc = hs, cs
        if cc < best_c:
            best_h, best_c = hc, cc
        if time.time() < cont_dl - 30:
            h2, c2 = coord_polish(hc, ns, cont_dl, 10)
            if c2 < best_c:
                best_h, best_c = h2, c2

    if time.time() < dl - 20:
        h3, c3 = coord_polish(best_h, len(best_h), dl - 2, 6)
        if c3 < best_c:
            best_h, best_c = h3, c3

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), len(best_h)
