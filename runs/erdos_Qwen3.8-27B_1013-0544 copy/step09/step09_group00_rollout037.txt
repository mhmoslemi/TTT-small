    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

    c = np.correlate(x, 1.0 - x, mode='full')
    s = alpha * c
    sm = s.max()
    return (sm + np.log(np.sum(np.exp(s - sm)))) / alpha * (2.0 / n)

def smooth_grad(x, n, alpha):

    s = alpha * c
    sm = s.max()
    w = np.exp(s - sm) / np.sum(np.exp(s - sm))

        if lo_b < hi_b:

    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
             'jac': lambda x, n_=n: np.ones(n_)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha),
                           jac=lambda x, n_=n, a_=alpha: smooth_grad(x, n_, a_),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            if feasible and np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            if x_lo > x_hi + 1e-14:
                feasible = False
            if feasible and has_diag:
                Cq = L - c_diag + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                        feasible = False
            if feasible:
                L_hi = L
            else:
                L_lo = L
        L = L_hi
        x_lo, x_hi = 0.0, 1.0
        if np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        if has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if x_lo >= r1 and x_hi <= r2:
                    if h0 - r1 <= r2 - h0:
                        x_hi = r1
                    else:
                        x_lo = r2
                elif x_lo < r1:
                    x_hi = min(x_hi, r1)
                elif x_hi > r2:
                    x_lo = max(x_lo, r2)
        if x_lo > x_hi:
            continue
        xn = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
        if abs(xn - h0) > 1e-14:
            delta = xn - h0
            c[ms] += delta * a
            if has_diag:
                c[ms[diag_pos[0]]] += delta * (1.0 - 2.0 * h0) - delta * delta
            h[p] = xn
            improved += 1
    return improved

def exact_cd(h, n, max_passes, deadline, random_order=False, seed=0):
    h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n) if random_order else range(n)
        improved = cd_single_pass(h, c, n, order)
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    s = alpha * c
    sm = s.max()
    return (sm + np.log(np.sum(np.exp(s - sm)))) / alpha * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    s = alpha * c
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    grad = np.zeros(n)
    kvals = np.arange(-(n - 1), n)
    for i in range(len(kvals)):
        wv = w[i]
        if wv < 1e-18:
            continue
        k = int(kvals[i])
        lo_f = max(0, -k)
        hi_f = min(n, n - k)
        if lo_f < hi_f:
            idx = np.arange(lo_f, hi_f)
            grad[idx] += wv * (1.0 - x[idx + k])
        lo_b = max(0, k)
        hi_b = min(n, n + k)
        if lo_b < hi_b:
            idx = np.arange(lo_b, hi_b)
            grad[idx] -= wv * x[idx - k]
    return grad * (2.0 / n)

def _obj(x, n, alpha):
    return smooth_obj(x, n, alpha)

def _jac(x, n, alpha):
    return smooth_grad(x, n, alpha)

def _cons_fun(x):
    return x.sum() - x.size / 2.0

def _cons_jac(x):
    return np.ones_like(x)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_obj, best_h, args=(n, alpha), jac=_jac,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Thorough sequential CD
    h1, c1 = exact_cd(h_best, n, 150, deadline, random_order=False, seed=seed)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP with analytic gradient
    h2, c2 = slsqp_opt(h_best, n, [1000, 5000, 20000, 100000], 200, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Random-order CD for escaping
    h3, c3 = exact_cd(h_best, n, 80, deadline, random_order=True, seed=seed + 7)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: More SLSQP after random CD
    h4, c4 = slsqp_opt(h_best, n, [5000, 50000], 150, deadline)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: Multi-n exploration
    n_list = [400, 550, 650, 800, 350, 450, 500]
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [5000, 20000], 100, deadline)
        h_o, c_o = exact_cd(h_o, n_try, 50, deadline, random_order=True, seed=seed + n_try)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Shake and settle
    n = len(h_best)
    for r in range(5):
        if time.time() > deadline - 60:
            break
        s = 0.004 + 0.008 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_opt(hp, n, [5000, 30000], 120, deadline)
        h_o, c_o = exact_cd(h_o, n, 60, deadline, random_order=False, seed=seed + 200 + r)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 7: Final CD polish (both orders)
    if time.time() < deadline - 5:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 100, deadline, random_order=False, seed=seed + 500)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
        h_f2, c_f2 = exact_cd(h_best, n, 50, deadline, random_order=True, seed=seed + 600)
        if c_f2 < best_c:
            h_best, best_c = h_f2, c_f2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
