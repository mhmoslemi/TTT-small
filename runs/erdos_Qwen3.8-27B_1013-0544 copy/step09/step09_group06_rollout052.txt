def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    m = n // p
    h = np.tile(a, m)
    if len(h) < n:

    h = h[:n]

def optimize_periodic(p, n, a0=None, maxiter=200):
    """Optimize a periodic pattern with period p."""
    if a0 is None:
        a = np.full(p, 0.5)
    else:
        a = a0.copy()

    # Use Lagrange multiplier or just project after each step

        a = vars_[:p-1]
        a_p = p/2.0 - a.sum()
        if a_p < -1e-10 or a_p > 1.0 + 1e-10:
            return 1e6
        a_full = np.append(a, a_p)

        c, _ = periodic_c5(a_full, p, n)
        return c

                candidates.append(d)

            vals[diag_idx] -= d * d
        v = max(vals.max(), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    
    new_val = float(np.clip(h0 + best_d, 0.0, 1.0))
    return new_val, best_val

For a flat autocorrelation, we want R[k] ≈ R[0]/(2n-1) for all k (where R[0] = Σ h[i]^2).

This gives us a new optimization strategy:

- This is: minimize Σ_k (R[k] - mean(R))^2
- This is a 4th-order polynomial in h (since R[k] is quadratic in h)
- But it's a convex function! (The objective is Σ_k R[k]^2 - (Σ_k R[k])^2/(2n-1), and Σ_k R[k]^2 is related to the L4 norm of the Fourier transform, which is convex by Parse...)

However, by Parseval's theorem: Σ_k |R[k]|^2 = (1/N) |FT(h)|^4 (in the DFT domain, up to constants). So minimizing Σ_k R[k]^2 is equivalent to minimizing the L4 norm of the DFT of h, which is related to making the spectrum flat.

This is not a convex problem in general, but it has nice structure.

2. Use gradient descent on the variance objective
3. Then switch to the original max-min objective with CD

∂R[k]/∂h[i] = h[i+k-(n-1)] + h[i-k+(n-1)] (with boundary handling)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    k = len(c_peak)
    cands = [lo, hi]
    for i in range(k):
        if i == diag_idx: continue
        for j in range(i+1, k):
            if j == diag_idx: continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi: cands.append(d)
    if diag_idx >= 0:
        for i in range(k):
            if i == diag_idx: continue
            B = a_coeff[diag_idx] - a_coeff[i]
            C = c_peak[diag_idx] - c_peak[i]
            disc = B*B - 4*C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B+sq)/2, (-B-sq)/2]:
                    if lo <= d <= hi: cands.append(d)
        B = a_coeff[diag_idx]
        C = c_peak[diag_idx] - unaff_max
        disc = B*B - 4*C
        if disc >= 0:
            sq = np.sqrt(disc)
            for d in [(-B+sq)/2, (-B-sq)/2]:
                if lo <= d <= hi: cands.append(d)
    for i in range(k):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi: cands.append(d)
    best_d, best_v = 0.0, float('inf')
    for d in cands:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d*a_coeff
        if diag_idx >= 0: vals[diag_idx] -= d*d
        v = max(float(vals.max()), unaff_max)
        if v < best_v: best_v, best_d = v, d
    return best_d, best_v

def cd_pass(h, n, top_k, refresh=20):
    c = np.correlate(h, 1.0-h, mode='full')
    improved = False
    for p in range(n):
        if p % refresh == 0:
            c = np.correlate(h, 1.0-h, mode='full')
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0-1e-10: continue
        ti = np.argsort(c)[::-1][:top_k]
        cp = c[ti].astype(float)
        ac = np.zeros(top_k); di = -1
        for idx in range(top_k):
            m = int(ti[idx])
            jf = p + m - (n-1); jb = p - m + (n-1)
            ac[idx] = ((1-h[jf]) if 0<=jf<n else 0.0) + ((-h[jb]) if 0<=jb<n else 0.0)
            if m == n-1: di = idx
        mask = np.ones(2*n-1, dtype=bool); mask[ti] = False
        um = float(c[mask].max()) if mask.any() else 0.0
        bd, bv = solve_1d(cp, ac, di, um, -h0, 1.0-h0)
        if abs(bd) > 1e-12:
            h[p] = float(np.clip(h0+bd, 0, 1))
            improved = True
    return improved

def multi_cd(h, n, passes, deadline, top_k=10, refresh=20):
    h = h.copy()
    for _ in range(passes):
        if time.time() > deadline: break
        if not cd_pass(h, n, top_k, refresh): break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = alpha*(c-m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s-mx)))
    return (m + lse/alpha)*(2.0/n)

def lbfghs_run(h, n, alpha, lambda_pen, maxiter=200):
    def obj(x):
        base = smooth_max_obj(x, n, alpha)
        pen = lambda_pen * (x.sum() - n/2.0)**2
        return base + pen
    bounds = [(0,1)]*n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    nh = project_feasible(res.x, n)
    return nh, compute_c5(nh, n)

def periodize_and_opt(p, n, deadline, rng):
    best_val = float('inf'); best_h = None
    m = n // p
    for trial in range(5):
        if time.time() > deadline: break
        if trial == 0:
            a0 = np.full(p, 0.5)
        else:
            a0 = rng.uniform(0, 1, p)
        a0 = project_feasible(a0, p)
        
        def pobj(x):
            a = np.append(x, p/2.0 - x.sum())
            if np.any(a < -0.01) or np.any(a > 1.01): return 1e10
            a = np.clip(a, 0, 1)
            hh = np.tile(a, m)
            if len(hh) < n: hh = np.concatenate([hh, a[:n-m*p]])
            return compute_c5(hh[:n], n)
        
        x0 = a0[:p-1]
        res = minimize(pobj, x0, method='Nelder-Mead',
                      options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-12})
        if res.fun < best_val:
            best_val = res.fun
            a = np.append(res.x, p/2.0 - res.x.sum())
            a = np.clip(a, 0, 1)
            hh = np.tile(a, m)
            if len(hh) < n: hh = np.concatenate([hh, a[:n-m*p]])
            best_h = hh[:n].copy()
    return best_val, best_h

def coarse_to_fine(h_best, n_best, n_coarse, deadline, rng):
    x_f = np.linspace(0, 2, n_best)
    x_c = np.linspace(0, 2, n_coarse)
    h_c = project_feasible(np.interp(x_c, x_f, h_best), n_coarse)
    # Add some random perturbation at coarse level
    h_c = project_feasible(h_c + rng.normal(0, 0.05, n_coarse), n_coarse)
    h_c, c_c = multi_cd(h_c, n_coarse, 80, deadline, top_k=8, refresh=5)
    # Upscale
    x_u = np.linspace(0, 2, n_best)
    h_u = project_feasible(np.interp(x_u, x_c, h_c), n_best)
    h_u, c_u = multi_cd(h_u, n_best, 30, deadline, top_k=10)
    return h_u, c_u

def block_perturb(h, n, rng, block_size, num_blocks):
    h_p = h.copy()
    for _ in range(num_blocks):
        start = rng.randint(0, max(1, n - block_size))
        h_p[start:start+block_size] = rng.uniform(0, 1, block_size)
    return project_feasible(h_p, n)
