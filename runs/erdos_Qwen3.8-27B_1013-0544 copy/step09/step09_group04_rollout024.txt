import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def compute_subgradient(h, n, topk=5):
    """Compute subgradient of max correlation, using top-k active set."""
    c = np.correlate(h, 1.0 - h, mode='full')
    L = c.max()
    # Get top-k indices
    k = min(topk, len(c))
    top_idx = np.argsort(-c)[:k]
    # Weight by how close to max
    weights = np.exp(50 * (c[top_idx] - L))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, m in zip(weights, top_idx):
        lag = int(m) - (n - 1)
        # d(c[m])/d(h[p]) = (1-h[p-lag])*I(0<=p-lag<n) - h[p+lag]*I(0<=p+lag<n)
        p = np.arange(n)
        idx_f = p - lag  # p such that p+lag is in range → h[p+lag]
        idx_b = p + lag  # p such that p-lag is in range → h[p-lag]
        mask_f = (idx_f >= 0) & (idx_f < n)
        mask_b = (idx_b >= 0) & (idx_b < n)
        grad += w * (np.where(mask_b, 1.0 - h[idx_b[mask_b]], 0.0) - np.where(mask_f, h[idx_f[mask_f]], 0.0))
    return grad, c, L

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def compute_subgrad(h, n, topk=10):
    c = np.correlate(h, 1.0 - h, mode='full')
    L = c.max()
    k = min(topk, len(c))
    top_idx = np.argsort(-c)[:k]
    weights = np.exp(30.0 * (c[top_idx] - L))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, m in zip(weights, top_idx):
        lag = int(m) - (n - 1)
        p = np.arange(n)
        idx_f = p - lag
        idx_b = p + lag
        mask_b = (idx_b >= 0) & (idx_b < n)
        mask_f = (idx_f >= 0) & (idx_f < n)
        term_b = np.zeros(n)
        term_f = np.zeros(n)
        term_b[mask_b] = 1.0 - h[idx_b[mask_b]]
        term_f[mask_f] = h[idx_f[mask_f]]
        grad += w * (term_b - term_f)
    return grad

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and float(np.max(c_base[zero_m])) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-15:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = abs(h0 - r1)
                d2 = abs(h0 - r2)
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d_exact(c, h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
    h = project_feasible(h, n)
    return h

def cd_loop(h, n, max_passes, deadline, nudge_every=5, nudge_mag=0.003, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    best_h = h.copy()
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        h = cd_pass(h, n, deadline)
        c = compute_c5(h, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall >= nudge_every:
            grad = compute_subgrad(h, n, topk=10)
            gnorm = np.max(np.abs(grad))
            if gnorm > 1e-12:
                eta = nudge_mag / gnorm
                for scale in [1.0, 0.5, 0.2, 0.1, 0.05]:
                    h_try = project_feasible(h - scale * eta * grad, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c:
                        h = h_try
                        best_h = h.copy()
                        best_c = c_try
                        break
            if rng is not None and time.time() < deadline - 10:
                s = 0.005 * (0.5 ** (stall // nudge_every))
                h_try = project_feasible(h + rng.normal(0, s, n), n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    h = h_try
                    best_h = h.copy()
                    best_c = c_try
            stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(700, 0.5)
        n = 700

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP with alpha sweep
    t1_end = t0 + min(90, (deadline - t0) // 5)
    h1, c1 = slsqp_run(h, n, [200, 800, 3000, 15000, 80000], 400)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: CD loop with subgradient nudge
    t2_end = t0 + min(650, (deadline - t0) * 3 // 5)
    n_passes = max(30, int((t2_end - time.time()) / 1.5))
    h2, c2 = cd_loop(best_h, n, n_passes, t2_end, nudge_every=4, nudge_mag=0.004, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Try alternative n values
    t3_end = t0 + min(900, deadline - 80)
    for n_try in [600, 800, 500, 900]:
        if time.time() > t3_end:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [5000, 50000], 300)
        np2 = max(10, int((t3_end - time.time()) / 2.5))
        h_opt, c_opt = cd_loop(h_opt, n_try, np2, t3_end, nudge_every=3, nudge_mag=0.003, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Try structured restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        structured = [
            0.5 + 0.4 * np.sin(np.pi * x),
            0.5 + 0.3 * np.cos(2 * np.pi * x),
            0.5 + 0.2 * np.sin(3 * np.pi * x) + 0.1 * np.sin(5 * np.pi * x),
        ]
        for s_fn in structured:
            if time.time() > deadline - 40:
                break
            h_mix = best_h + 0.3 * (s_fn - best_h)
            h_mix = project_feasible(h_mix, n_cur)
            h_o, c_o = cd_loop(h_mix, n_cur, 30, min(deadline - 30, time.time() + 120),
                                nudge_every=3, nudge_mag=0.003, rng=rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Final polish
    if time.time() < deadline - 10:
        n_fin = len(best_h)
        n_passes = max(5, int((deadline - time.time() - 5) / 1.5))
        h5, c5 = cd_loop(best_h, n_fin, n_passes, deadline, nudge_every=3, nudge_mag=0.001, rng=rng)
        if c5 < best_c:
            best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
