import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord_fast(c, h, p, n, iters=35):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_k = max(0, n - 1 - p)
    hi_k = min(2 * n - 2, n - 1 + p)
    ks = np.arange(lo_k, hi_k + 1)
    jf = p + ks - (n - 1)
    jb = p - ks + (n - 1)
    a = np.zeros(len(ks))
    base = np.zeros(len(ks))
    has_diag = (lo_k <= n - 1 <= hi_k)
    diag_idx = np.where(ks == n - 1)[0]
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a[vf] += (1.0 - h[jf[vf]])
    a[vb] -= h[jb[vb]]
    for i in range(len(ks)):
        if ks[i] == n - 1:
            base[i] = c[n - 1] - h0 + h0 * h0
            a[i] = 1.0
        else:
            base[i] = c[ks[i]] - h0 * a[i]
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[lo_k:hi_k + 1] = False
    fixed_max = float(c[mask].max()) if mask.any() else 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_ = 0.0
    b_ = 1.0

        vals = base + a * t

        m2 = float(np.max(vals))
        return max(fixed_max, m2)
    f1 = eval_f(x1)
    f2 = eval_f(x2)
    for _ in range(iters):

        else:
            a_ = x1
            x1 = x2
            f1 = f2
            x2 = a_ + gr * (b_ - a_)
            f2 = eval_f(x2)
    return float(np.clip((a_ + b_) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_coord_fast(c, h, p, n)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def active_set_lp(h, n, tol, max_iter, deadline):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c = c_raw * dx
        c_max = c.max()
        active = np.where(c >= c_max - tol)[0]
        if len(active) > 500:
            active = np.where(c >= c_max - 2 * tol)[0]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for m in active:
            p = np.arange(n)
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            g = np.zeros(n)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            g[vf] += (1.0 - h[jf[vf]])
            g[vb] -= h[jb[vb]]
            c_linear = c_raw[m] + g @ h_var
            constraints.append(c_linear <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        new_h = project_feasible(np.array(h_var.value).flatten(), n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
        if new_c >= compute_c5(h, n) - 1e-10:
            break
        h = new_h
    return best_h, best_c

def subgrad_momentum(h, n, steps, deadline, rng):

    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5 / np.sqrt(n)
    beta = 0.9
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        near_max = np.where(c >= c_max - 0.002)[0]
        g = np.zeros(n)
        for m in near_max:
            p = np.arange(n)
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            gm = np.zeros(n)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            gm[vf] += (1.0 - h[jf[vf]])
            gm[vb] -= h[jb[vb]]
            g += gm
        g /= len(near_max)
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g /= g_norm
        v = beta * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        h = h_new
        lr *= 0.98
    return best_h, best_c

def smooth_slsqp(h, n, alphas, maxiter_each, deadline):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * dx
        def obj_jac(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            w = np.exp(alpha * (c - m))
            w /= w.sum()
            grad = np.zeros(n)
            for k_i in range(len(w)):
                if w[k_i] < 1e-10:
                    continue
                p = np.arange(n)
                jf = p + k_i - (n - 1)
                jb = p - k_i + (n - 1)
                gm = np.zeros(n)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                gm[vf] += (1.0 - x[jf[vf]])
                gm[vb] -= x[jb[vb]]
                grad += w[k_i] * gm
            return grad * dx
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
