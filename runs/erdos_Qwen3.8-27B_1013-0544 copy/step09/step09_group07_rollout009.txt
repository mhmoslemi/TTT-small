def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * dx * c
    x -= x.max()
    exp_x = np.exp(x)
    weights = exp_x / exp_x.sum()
    
    # Gradient of LSE wrt c is: alpha * dx * weights
    # Gradient of c[m] wrt h:
    # dc[m]/dh[j] = (1-h[m+j-n+1]) - h[n-1-m+j] (with bounds)
    # So grad[j] = alpha*dx*sum_m weights[m]*dc[m]/dh[j]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        w = alpha * dx * weights[m]
        if w < 1e-15:
            continue
        s = m - (n-1)
        # First part: (1-h[j+s]) for j where 0<=j+s<n
        lo = max(0, -s)
        hi = min(n, n-s)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo+s:hi+s])
        # Second part: -h[j-s] for j where 0<=j-s<n
        lo2 = max(0, s)
        hi2 = min(n, n+s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-s:hi2-s]
    return grad

def fista_smooth(h_init, n, alpha, max_iter=500, deadline=None):
    """FISTA (accelerated gradient) for the smooth approximation"""
    h = project_feasible(h_init.copy(), n)
    y = h.copy()
    t = 1.0
    L = 2.0 * (2.0/n) + alpha * (2.0/n)**2 * n  # rough Lipschitz bound
    # Actually, let me estimate L more carefully
    # The Hessian of smooth_obj has eigenvalues bounded by:
    # First part (from Hessian of c[m]): bounded by 2*dx = 4/n
    # Second part (from Hessian of LSE): bounded by alpha*dx^2 * n = alpha*4/n^2 * n = 4*alpha/n
    # So L ≈ 4/n + 4*alpha/n = 4*(1+alpha)/n
    # For n=600, alpha=1000: L ≈ 4*1001/600 ≈ 6.7
    L = 4.0 * (1.0 + alpha) / n
    step = 1.0 / L
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Compute gradient at y
        gy = smooth_obj_grad(y, n, alpha)
        # Gradient step
        y_new = y - step * gy
        # Projection
        y_new = project_feasible(y_new, n)
        # Update t
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
        # Update y for next iteration
        y = y_new + ((t / t_new) - 1.0) * (y_new - h)
        # Update h
        h = y_new
        t = t_new
    
    return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot)
