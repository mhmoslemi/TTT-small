import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):

    h = h.copy()

    stall = 0
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-12:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    return h, best_c
