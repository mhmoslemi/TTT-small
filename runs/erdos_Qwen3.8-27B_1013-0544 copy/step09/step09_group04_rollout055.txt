import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > target:
            lo = tau
        else:
            hi = tau
        if hi - lo < 1e-15:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = c - c.max()
    w = np.exp(alpha * cs)
    ws = w.sum()
    if ws < 1e-300:
        me = np.argmax(c)
        w = np.zeros_like(c)
        w[me] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(len(c)):
        wv = w[k]
        if wv < 1e-15:
            continue
        d = k - n1
        if d >= 0:
            if d < n:
                grad[:n - d] += wv * (1.0 - h[d:])
                grad[d:] -= wv * h[:n - d]
        else:
            dd = -d
            if dd < n:
                grad[dd:] += wv * (1.0 - h[:n - dd])
                grad[:n - dd] -= wv * h[dd:]
    return grad
