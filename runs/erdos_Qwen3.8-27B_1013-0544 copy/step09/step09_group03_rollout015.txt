    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    thresh = m - 0.05  # Include lags within 0.05 of max

    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for lag in active:
        s = lag - (n - 1)
        if s == 0:
            grad += 1.0 - 2.0 * h
        else:
            # dc/dh[i] = (1-h[i+s]) - h[i-s] with bounds
            i = np.arange(n)
            ip = i + s
            im = i - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            tmp = np.zeros(n)
            tmp[vp] = 1.0 - h[ip[vp]]
            tmp[vm] -= h[im[vm]]
            grad += tmp
    grad /= len(active)
    h_new = project_feasible(h - step_size * grad, n)
    return h_new

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def coord_update(c, p, h, n):
