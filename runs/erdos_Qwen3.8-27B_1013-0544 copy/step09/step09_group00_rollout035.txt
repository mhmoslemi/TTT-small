def compute_c5(h, n):

def project_feasible(h, n):

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            lse = m + shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))
            return lse * (2.0 / n)
        
        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            w = np.exp(shifted - shifted.max())
            w /= w.sum()
            grad = np.zeros(n)
            for p in range(n):
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)

                jb = p - ms + (n - 1)

                    dc[vf] += (1.0 - x[jf[vf]])
                if np.any(vb):
                    dc[vb] -= x[jb[vb]]
                grad[p] = np.dot(w[ms], dc)
            return grad * (2.0 / n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]

                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_optimize(h, n, max_passes, deadline, bisect_iter=30):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        cur_max = float(c.max())
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_base = c[ms]
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            # Tighter bounds
            unaff = 0.0
            if m_min > 0:
                unaff = max(unaff, float(np.max(c[:m_min])))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            L_lo, L_hi = unaff, cur_max
            if L_hi <= L_lo:
                L_lo, L_hi = 0.0, cur_max + 1.0
            for _ in range(bisect_iter):
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                feasible = True
                neg_mask = a < -1e-15
                pos_mask = a > 1e-15
                if np.any(neg_mask):
                    x_lo = max(x_lo, h0 + float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if np.any(pos_mask):
                    x_hi = min(x_hi, h0 + float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if has_diag:
                    di = diag_idx[0]
                    disc = a[di] ** 2 - 4.0 * (L - c_base[di])
                    if disc >= 0:
                        r1 = (a[di] - np.sqrt(disc)) / 2.0
                        r2 = (a[di] + np.sqrt(disc)) / 2.0
                        if not (x_hi <= h0 + r1 or x_lo >= h0 + r2):
                            feasible = False
                if x_lo > x_hi + 1e-14:
                    feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            if np.any(neg_mask):
                x_lo = max(x_lo, h0 + float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, h0 + float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            if has_diag:
                di = diag_idx[0]
                disc = a[di] ** 2 - 4.0 * (L - c_base[di])
                if disc >= 0:
                    r1 = (a[di] - np.sqrt(disc)) / 2.0
                    r2 = (a[di] + np.sqrt(disc)) / 2.0
                    if not (x_hi <= h0 + r1 or x_lo >= h0 + r2):
                        if h0 - (h0 + r1) <= (h0 + r2) - h0:
                            x_hi = min(x_hi, h0 + r1)
                        else:
                            x_lo = max(x_lo, h0 + r2)
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            x_new = h0 if x_lo > x_hi else (x_lo + x_hi) / 2.0
            if abs(x_new - h0) > 1e-12:
                delta = x_new - h0
                c[ms] += delta * a
                if has_diag:
                    c[ms[diag_idx[0]]] -= delta * delta
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m - shifted_max(c, alpha))
    w = np.exp(shifted)
    w /= w.sum()
    obj = (m + np.log(w.sum()) / alpha) * (2.0 / n)
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        dc = np.zeros(len(ms))
        if np.any(vf):
            dc[vf] += (1.0 - x[jf[vf]])
        if np.any(vb):
            dc[vb] -= x[jb[vb]]
        grad[p] = np.dot(w[ms], dc)
    return obj, grad * (2.0 / n)

def shifted_max(c, alpha):
    return c.max() - (c.max() - c.max())  # just return 0, handled in caller

def slsqp_opt(h, n, alpha, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = m + mx / alpha + np.log(np.sum(np.exp(shifted - mx))) / alpha
        return lse * dx

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        g = np.zeros(n)
        for p in range(n):
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            dc = np.zeros(len(ms))
            if np.any(vf):
                dc[vf] += (1.0 - x[jf[vf]])
            if np.any(vb):
                dc[vb] -= x[jb[vb]]
            g[p] = np.dot(w[ms], dc)
        return g * dx

    def cons_fun(x):
        return x.sum() - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c
