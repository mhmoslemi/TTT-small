```python
def thue_morse(n):
    """Generate first n bits of Thue-Morse sequence."""
    t = [0]
    for _ in range(n.bit_length()):
        t = t + [1 - x for x in t]
    # Ensure exactly n/2 ones
    # Count ones
    ones = sum(t[:n])
    need = n//2 - ones
    if need > 0:
        # Flip some 0s to 1s (starting from the end)
        for i in range(n-1, -1, -1):
            if t[i] == 0 and need > 0:
                t[i] = 1
                need -= 1
    elif need < 0:
        for i in range(n-1, -1, -1):
            if t[i] == 1 and need < 0:
                t[i] = 0
                need += 1
    return np.array(t[:n], dtype=float)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h = np.minimum(1.0, h + (diff/tot)*room)
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h = np.maximum(0.0, h + (diff/tot)*h)
    return np.clip(h, 0.0, 1.0)

def thue_morse_binary(n):
    """Generate Thue-Morse based binary sequence with exactly n/2 ones."""
    # Build Thue-Morse to length >= n
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1-t])
    t = t[:n].astype(float)
    # Adjust to have exactly n//2 ones
    n_ones = int(t.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(t == 1)[0]
        # Randomly choose which ones to flip
        rng = np.random.RandomState(42)
        to_flip = rng.choice(idx, n_ones - target, replace=False)
        t[to_flip] = 0
    elif n_ones < target:
        idx = np.where(t == 0)[0]
        rng = np.random.RandomState(42)
        to_flip = rng.choice(idx, target - n_ones, replace=False)
        t[to_flip] = 1
    return t

def project_binary(h, n):
    """Project h to a binary sequence with exactly n//2 ones."""
    # Sort by value, set top n//2 to 1
    target = n // 2
    if n % 2 == 1:
        target = (n+1) // 2  # or (n-1)//2, doesn't matter much
    idx = np.argsort(h)[::-1]  # descending
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_sa(h, n, max_iters, deadline, seed):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    ones_mask = h == 1.0
    n_ones = n // 2
    
    T0 = 0.03
    Tf = 0.001
    total_iters = max_iters
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        T = T0 * (Tf/T0) ** (it / total_iters)
        
        # Random swap
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]
        p = ones_idx[rng.randint(len(ones_idx))]
        q = zeros_idx[rng.randint(len(zeros_idx))]
        
        # Apply swap
        h[p] = 0.0
        h[q] = 1.0
        ones_mask[p] = False
        ones_mask[q] = True
        
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            h[p] = 1.0
            h[q] = 0.0
            ones_mask[p] = True
            ones_mask[q] = False
    
    return best_h, best_c

    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)

        worst_m = np.argmax(c)

        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]

        best_p, best_q = None, None

        # (with boundary checks)

        pm = ones_idx - k
        p_term = np.zeros(len(ones_idx))
        valid_pk = (pk >= 0) & (pk < n)
        p_term[valid_pk] = -(1 - h[pk[valid_pk]])
        valid_pm = (pm >= 0) & (pm < n)

        # For now, ignore the special case (p-k=q) and just use h[p-k]

        q_term = np.zeros(len(zeros_idx))

        valid_qm = (qm >= 0) & (qm < n)
        q_term[valid_qm] += -h[qm[valid_qm]]
        
        # Total delta for (p_i, q_j) = p_term[i] + q_term[j]
        # Best: min p_term + min q_term
        bi = np.argmin(p_term)
        bj = np.argmin(q_term)
        best_delta = p_term[bi] + q_term[bj]
        best_p = ones_idx[bi]
        best_q = zeros_idx[bj]
        
        if best_delta >= -1e-10 or best_p is None:
            break
        
        # Apply swap
        h[best_p] = 0.0
        h[best_q] = 1.0
        ones_mask[best_p] = False
        ones_mask[best_q] = True
        
        new_c = compute_c5(h, n)
        if new_c > best_c:
            # Revert
            h[best_p] = 1.0
            h[best_q] = 0.0
            ones_mask[best_p] = True
            ones_mask[best_q] = False
            break
        best_c = new_c
    
    return h, best_c

            # c[m] = A[p] * h[p] + B[p] where A[p] depends on m and B[p] is constant
            
            # For the max, we want to minimize max_m (A_m * h[p] + B_m)

            # c[m] = ... + h[p]*(1-h[p+k]) (if p+k valid, k = m-(n-1))
            #        + ... + h[p-k]*(1-h[p]) (if p-k valid, k = m-(n-1))

            #   This contributes to c[m] as: h[p] * (1-h[p+k])
            #   As function of x = h[p]: x * (1-h[p+k])
            #   So A += (1-h[p+k]), B += 0 (for this term)

            #   This contributes to c[m] as: h[p-k] * (1-x)

            k_range = np.arange(-(n-1), n)  # m = n-1+k

            # Subtract current contribution of h[p] from c:

            # We want to replace h0 with x:
            # new c[m] = c[m] - h0*(1-h[p+k]) - h[p-k]*(1-h0) + x*(1-h[p+k]) + h[p-k]*(1-x)
            #          = c[m] + (x-h0)*(1-h[p+k]) + h[p-k]*(h0-x)
            #          = c[m] + (x-h0)*((1-h[p+k]) - h[p-k])
            
            # So new c[m] = c[m] + (x-h0) * D[m] where D[m] = (1-h[p+k]) - h[p-k]
            
            D = np.zeros(2*n-1)  # D[m] for m = 0 to 2n-2
            for idx, k in enumerate(k_range):
                m = n - 1 + k
                # D[m] = (1-h[p+k]) if 0≤p+k<n else 0, minus h[p-k] if 0≤p-k<n else 0
                val = 0.0
                if 0 <= p+k < n:
                    val += (1 - h[p+k])
                if 0 <= p-k < n:
                    val -= h[p-k]
                D[m] = val
            
            # We want to minimize max_m (c[m] + (x-h0)*D[m]) over x ∈ [0,1]
            # Let y = x - h0, so y ∈ [-h0, 1-h0]
            # minimize max_m (c[m] + y*D[m])
            
            # This is a 1D convex piecewise-linear optimization
            # The optimal y is where the max is minimized
            
            # Binary search on y:
            lo, hi = -h0, 1.0 - h0

                vals = c + mid * D
                amax = vals.max()

                active = (vals >= amax - 1e-12)
                if not active.any():
                    break
                d_min = D[active].min()
                d_max = D[active].max()
                if d_min > 1e-12:
                    hi = mid  # increasing y increases the max
                elif d_max < -1e-12:
                    lo = mid  # decreasing y increases the max
                else:
                    break
            
            x_new = (lo + hi) / 2 + h0
            x_new = np.clip(x_new, 0.0, 1.0)
            
            if abs(x_new - h0) > 1e-10:
                h[p] = x_new
                improved += 1
        
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    
    return h, best_c
```
