def compute_c5(h, n):

import numpy as np

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + diff * room / tot)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + diff * h / tot)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s = np.arange(p - (n - 1), (n - 1) - p + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m_idx = n - 1 + s
    b = c[m_idx] - a * h0
    m_min = n - 1 + s.min()
    m_max = n - 1 + s.max()
    all_m = np.arange(2 * n - 1)
    unaff = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff].max()) if unaff.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-13:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-11:
            hi = mid
        elif max_sl < -1e-11:
            lo = mid

        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = x

def coord_descent(h, n, max_passes=2000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 25:
                break
    return h, best_c

def binary_sa(n, max_iters, deadline=None, seed=0, h_start=None):

            h[to_rm] = 0
        elif cnt < target:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, target - cnt, replace=False)
            h[to_add] = 1
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T0, T1 = 0.05, 0.0003
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        frac = it / max(1, max_iters)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta <= 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c

def hill_climb_binary(h, n, max_iters, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    stall = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        new_c = compute_c5(h, n)
        if new_c <= cur_c:
            cur_c = new_c
            stall = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
            stall += 1
            if stall > 5000:
                break
    return best_h, best_c
