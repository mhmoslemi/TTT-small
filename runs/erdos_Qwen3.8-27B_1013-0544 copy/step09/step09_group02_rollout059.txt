def compute_smooth_gradient(h, n, T):
    M = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c_sh = c / T
    c_sh -= c_sh.max()
    w = np.exp(c_sh)
    w /= w.sum()
    
    dx = 2.0 / n
    g = np.zeros(n)
    
    for m in range(M):
        wm = w[m]
        if wm < 1e-8:
            continue
        s = m - (n - 1)
        # Forward: (1-h[q+s]) for q where 0<=q+s<n
        if s >= 0:
            g[:n-s] += wm * (1.0 - h[s:])
            # Backward: -h[q-s] for q where 0<=q-s<n, i.e., q>=s
            g[s:] -= wm * h[:n-s]
        else:
            sp = -s
            g[sp:] += wm * (1.0 - h[:n-sp])
            g[:n-sp] -= wm * h[sp:]
    
    return g * dx

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    # Project h to [0,1]^n with sum = n/2
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h += frac * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, n, T):
    """Compute gradient of soft-max objective."""
    M = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c_sh = c / T
    mx = c_sh.max()
    exp_c = np.exp(c_sh - mx)
    w = exp_c / exp_c.sum()
    
    dx = 2.0 / n
    g = np.zeros(n)
    
    for m in range(M):
        wm = w[m]
        if wm < 1e-10:
            continue
        s = m - (n - 1)
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            sp = -s
            g[sp:] += wm * (1.0 - h[:n-sp])
            g[:n-sp] -= wm * h[sp:]
    
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h * (target / tot), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if diff > 0:
            adj = diff / mask.sum() if mask.sum() > 0 else 0
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            adj = diff / mask.sum() if mask.sum() > 0 else 0
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def compute_smooth_grad(h, n, T):
    M = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c_sh = c / T
    mx = c_sh.max()
    exp_c = np.exp(c_sh - mx)
    w = exp_c / exp_c.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(M):
        wm = w[m]
        if wm < 1e-10:
            continue
        s = m - (n - 1)
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n - s]
        else:
            sp = -s
            g[sp:] += wm * (1.0 - h[:n - sp])
            g[:n - sp] -= wm * h[sp:]
    return g * dx

def smooth_grad_momentum(h, n, T, deadline, max_iter=8000, lr_init=0.05, beta=0.9):
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_smooth_grad(h, n, T)
        v = beta * v - lr * g
        h_cand = project_feasible(h + v, n)
        c_new = compute_c5(h_cand, n)
        if c_new < best_c - 1e-15:
            h = h_cand
            best_h = h.copy()
            best_c = c_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 150 or lr < 1e-7:
                break
    return best_h, best_c

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, p + n - 1)
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def exact_cd(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, p + n - 1)
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def alternating_opt(h, n, deadline, n_cycles=10, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    T = 1.0
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        T = max(0.1, T * 0.7)
        h_g, c_g = smooth_grad_momentum(h, n, T, min(deadline, time.time() + 20),
                                        max_iter=2000, lr_init=0.05, beta=0.9)
        if c_g < best_c:
            h, best_c = h_g, c_g
        h_c, c_c = exact_cd(h, n, max_passes=50, deadline=deadline, rng=rng)
        if c_c < best_c:
            h, best_c = h_c, c_c
    return h, best_c
