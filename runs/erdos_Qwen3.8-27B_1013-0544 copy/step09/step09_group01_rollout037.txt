    return (m + lse / alpha) * dx

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = c.max()
    sh = alpha * (c - m) * dx
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g1 = np.correlate(w, 1.0 - x, mode='full')
    g2 = np.correlate(w, x, mode='full')
    return dx * (g1[n - 1:n - 1 + n] - g2[n - 1:n - 1 + n])

def smooth_slsqp(h, n, alphas, maxiter_each=250):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    for alpha in alphas:
        def obj_fn(x, a=alpha):
            return _smooth_obj(x, n, a)
        def grad_fn(x, a=alpha):
            return _smooth_grad(x, n, a)
        def con_fun(x):
            return x.sum() - target
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_fun}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with gradient
    if time.time() < deadline - 200:
        h1, c1 = smooth_slsqp(best_h, n, [1000, 5000, 20000, 100000], 200)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Main precise CD
    if time.time() < deadline - 5:
        h2, c2 = precise_cd(best_h, n, deadline - 30)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Try nearby n
    for n_try in [n - 2, n + 2, n - 5, n + 5]:
        if time.time() > deadline - 60:
            break
        if n_try <= 100 or n_try == n:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = smooth_slsqp(h_int, n_try, [5000, 50000], 150)
        if time.time() < deadline - 20:
            h_o, c_o = precise_cd(h_o, n_try, deadline - 5)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 4: Perturbation restarts
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        for r in range(6):
            if time.time() > deadline - 15:
                break
            s = 0.008 + 0.008 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = smooth_slsqp(hp, n_cur, [5000, 50000], 150)
            if time.time() < deadline - 10:
                h_o, c_o = precise_cd(h_o, n_cur, deadline - 5)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final ultra-precise CD
    if time.time() < deadline - 2:
        h_f, c_f = precise_cd(best_h, len(best_h), deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
