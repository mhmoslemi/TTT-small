    s_lo = max(-h[p], -(h[q] - 1.0))  # h[p]+s >= 0 and h[q]-s >= 0

        return 0.0

    # For coordinate p: lags mlo_p to mhi_p

    mhi_p = min(2*n-2, max(2*n-2-p, p+n-1))
    mlo_q = max(0, n - 1 - q)
    mhi_q = min(2*n-2, max(2*n-2-q, q+n-1))
    
    mlo = min(mlo_p, mlo_q)
    mhi = max(mhi_p, mhi_q)
