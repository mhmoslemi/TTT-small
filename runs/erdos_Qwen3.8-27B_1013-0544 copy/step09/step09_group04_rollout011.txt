    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Intensive CD on initial grid
    h1, c1 = coord_cd(best_h, n_init, 50, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Gradient steps to break symmetry
    if time.time() < deadline - 200:
        h2, c2 = gradient_cd(best_h, len(best_h), deadline, max_steps=200)
        if c2 < best_c:
            best_h, best_c = h2, c2
            h2, c2 = coord_cd(best_h, len(best_h), 20, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
    # Phase 3: Try multiple n values
    n_current = len(best_h)
    for n_try in [400, 500, 600, 800, 900, 1000]:
        if time.time() > deadline - 80:
            break
        if n_try == n_current:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, 30, deadline)
        # Add gradient step
        if time.time() < deadline - 40:
            h_opt, c_opt = gradient_cd(h_opt, n_try, deadline, max_steps=100)
            h_opt, c_opt = coord_cd(h_opt, n_try, 15, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 4: Perturbation + CD cycles
    n = len(best_h)
    for r in range(30):
        if time.time() > deadline - 20:
            break
        s = 0.003 + 0.002 * r
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        h4, c4 = coord_cd(hp, n, 10, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
            # Continue improving from new best
            h4b, c4b = coord_cd(best_h, n, 10, deadline)
            if c4b < best_c:
                best_h, best_c = h4b, c4b
    # Phase 5: Final CD polish
    if time.time() < deadline - 3:
        max_passes = max(3, int((deadline - time.time() - 2) / 1.5))
        hf, cf = coord_cd(best_h, len(best_h), max_passes, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
