import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def eval_corr_max(h):
    return float(np.correlate(h, 1.0 - h, mode='full').max())

def directed_binary_search(h, n, deadline, seed, max_iter=80000):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    curr = float(c.max())
    best = h.copy()
    best_val = curr
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        shift = k_star - (n - 1)
        if shift >= 0:
            valid_i = np.arange(n - shift)
        else:
            valid_i = np.arange(-shift, n)
        bad = valid_i[(h[valid_i] == 1)]
        if shift >= 0:
            bad = bad[h[bad + shift] == 0]
        else:
            bad = bad[h[bad - shift] == 0]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_new = curr
        best_move = None
        n_tries = min(40, max(10, len(bad)))
        for t in range(n_tries):
            if len(bad) > 0:
                i = bad[rng.randint(len(bad))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nv = float(np.correlate(h, 1.0 - h, mode='full').max())
            h[j] = 0
            h[i] = 1
            if nv < best_new:
                best_new = nv
                best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            c = np.correlate(h, 1.0 - h, mode='full')
            curr = float(c.max())
            stall = 0
            if curr < best_val:

                best = h.copy()
        else:

            if stall > 200:
                npert = rng.randint(3, 8)
                for _ in range(npert):

                    zeros = np.where(h == 0)[0]
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                c = np.correlate(h, 1.0 - h, mode='full')
                curr = float(c.max())

    return best, best_val * (2.0 / n)

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d = 0.0
            best_val = current_max
            candidates = [lo, hi, 0.0]
            for i in range(k):
                for j in range(i + 1, k):
                    if i == diag_idx or j == diag_idx:
                        continue
                    da = a_coeff[i] - a_coeff[j]
                    if abs(da) > 1e-15:
                        d = (c_peak[j] - c_peak[i]) / da
                        if lo <= d <= hi:
                            candidates.append(d)
            if diag_idx is not None:
                for i in range(k):
                    if i == diag_idx:
                        continue
                    B = -(a_coeff[diag_idx] - a_coeff[i])
                    C = -(c_peak[diag_idx] - c_peak[i])
                    disc = B * B - 4.0 * C
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                            if lo <= d <= hi:
                                candidates.append(d)
            for i in range(k):
                if abs(a_coeff[i]) > 1e-15:
                    d = (unaff_max - c_peak[i]) / a_coeff[i]
                    if lo <= d <= hi:
                        candidates.append(d)
            for d in candidates:
                d = float(np.clip(d, lo, hi))
                vals = c_peak + d * a_coeff
                if diag_idx is not None:
                    vals[diag_idx] -= d * d
                v = max(float(vals.max()), unaff_max)
                if v < best_val:
                    best_val = v
                    best_d = d
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
