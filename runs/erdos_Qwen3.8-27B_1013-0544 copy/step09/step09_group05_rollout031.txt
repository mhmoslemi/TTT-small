def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    improved_total = True
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_mse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        constr = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                  'jac': lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(smooth_mse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=constr,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass

def subgrad_momentum(h, n, steps, deadline, rng):

    v = np.zeros(n)
    lr = 0.01
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(steps):
        if time.time() > deadline:
            break
        lr = 0.01 / (1 + 0.01 * s)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Subgradient of max_k c_k: pick a random argmax
        max_val = c.max()
        active = np.where(c >= max_val - 1e-10)[0]
        p = active[rng.randint(len(active))]
        # Gradient of c_p w.r.t. h
        grad = np.zeros(n)
        for i in range(n):
            jf = i + p - (n - 1)
            jb = i - p + (n - 1)
            if 0 <= jf < n:
                grad[i] += (1.0 - h[jf])
            if 0 <= jb < n:
                grad[i] -= h[jb]
        # Remove mean to maintain sum constraint approximately
        grad = grad - np.mean(grad)
        v = 0.9 * v - lr * grad
        h_new = project_feasible(h + v, n)
        h = h_new
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_h = h.copy()
            best_c = c_val
    return best_h, best_c

            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    if xl > xh + 1e-15 or xl >= r1 - 1e-14 and xh <= r2 + 1e-14:
                        feasible = False
                else:
                    if r1 <= 0.0 and r2 >= 1.0:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)
            xh2 = min(x_hi, 1.0)
            if xl2 < r1:
                x_hi = min(x_hi, r1)
            elif xh2 > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-10)
                else:
                    x_lo = max(x_lo, r2 - 1e-10)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def multistage_opt(start_h, start_n, deadline, rng):
    best_h = project_feasible(start_h, start_n)
    best_c = compute_c5(best_h, start_n)
    n = start_n
    # Coarse to fine: start with smaller n, interpolate up
    n_coarse = max(100, n // 4)
    x_fine = np.linspace(0, 2, n)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = project_feasible(np.interp(x_coarse, x_fine, best_h), n_coarse)
    h_coarse, c_coarse = coord_descent(h_coarse, n_coarse, 50, deadline, rng, True)
    # Interpolate back and optimize
    h_back = project_feasible(np.interp(x_fine, x_coarse, h_coarse), n)
    h_back, c_back = coord_descent(h_back, n, 100, deadline, rng, True)
    if c_back < best_c:
        best_h, best_c = h_back, c_back
    return best_h, best_c

def structured_init(n, rng, style):
    if style == 0:
        h = 0.5 + 0.5 * np.sign(np.sin(2*np.pi*3*np.linspace(0,1,n)) + rng.randn(n)*0.3)
    elif style == 1:
        h = 0.5 + 0.3*np.cos(2*np.pi*5*np.linspace(0,1,n) + rng.rand()*2*np.pi)
    elif style == 2:
        h = np.clip(0.5 + rng.randn(n)*0.2, 0, 1)
    elif style == 3:
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    elif style == 4:
        t = np.linspace(0, 1, n)
        h = 0.5 + 0.4*np.sin(2*np.pi*2*t) * np.cos(2*np.pi*7*t)
    else:
        h = np.full(n, 0.5) + rng.randn(n)*0.1
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: CD polish
    h1, c1 = coord_descent(best_h, n_init, 150, deadline, rng, True)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: SLSQP
    if time.time() < deadline - 100:
        h2, c2 = slsqp_opt(best_h, n_init, [500, 2000, 10000], 400, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = coord_descent(best_h, n_init, 150, deadline, rng, True)
        if c2b < best_c:
            best_h, best_c = h2b, c2b
    # Phase 3: Multi-stage (coarse-to-fine)
    if time.time() < deadline - 100:
        h3, c3 = multistage_opt(best_h, n_init, deadline, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Structured restarts + CD
    if time.time() < deadline - 60:
        n = len(best_h)
        for style in range(5):
            if time.time() > deadline - 40:
                break
            h_r = structured_init(n, rng, style)
            hr, cr = coord_descent(h_r, n, 100, deadline, rng, True)
            if cr < best_c:
                best_h, best_c = hr, cr
    # Phase 5: Perturb best + CD
    if time.time() < deadline - 50:
        n = len(best_h)
        for scale in [0.02, 0.05, 0.1, 0.15]:
            if time.time() > deadline - 30:
                break
            hp = project_feasible(best_h + rng.randn(n)*scale, n)
            hopt, copt = coord_descent(hp, n, 150, deadline, rng, True)
            if copt < best_c:
                best_h, best_c = hopt, copt
    # Phase 6: Try different n
    if time.time() < deadline - 80:
        n = len(best_h)
        n_cands = sorted(set([n-100, n-50, n+50, n+100, n+200]))
        n_cands = [x for x in n_cands if 300 <= x <= 1500 and x != n]
        x_cur = np.linspace(0, 2, len(best_h))
        for nc in n_cands:
            if time.time() > deadline - 50:
                break
            x_new = np.linspace(0, 2, nc)
            hi = project_feasible(np.interp(x_new, x_cur, best_h), nc)
            ho, co = slsqp_opt(hi, nc, [1000, 5000], 200, deadline)
            ho, co = coord_descent(ho, nc, 100, deadline, rng, True)
            if co < best_c:
                best_h, best_c = ho, co
                x_cur = x_new
    # Phase 7: Final polish
    if time.time() < deadline - 10:
        n = len(best_h)
        rem = deadline - time.time() - 5
        mp = max(30, int(rem / 0.5))
        hf, cf = coord_descent(best_h, n, mp, deadline, rng, False)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
