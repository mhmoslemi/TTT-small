def coord_cd_peak(h, n, max_passes, deadline, target_peaks=5):
    """Coordinate descent that targets peak-contributing coordinates first."""
    h = h.copy()
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        
        # Find peak lags
        peak_lags = np.argsort(-c)[:target_peaks]
        
        # For each peak lag k, find which h indices contribute most
        # Lag k (in full mode) corresponds to shift k-(n-1)
        # h[i] contributes to lag m if i + m - (n-1) is in [0, n) (as first factor)
        # or i - m + (n-1) is in [0, n) (as second factor)
        
        # Score each h index by how much it appears in peak correlations
        score = np.zeros(n)
        for k in peak_lags:
            shift = k - (n - 1)
            # h[i] as first factor: i + shift in [0, n)
            i_lo = max(0, -shift)
            i_hi = min(n, n - shift)
            for i in range(i_lo, i_hi):
                partner = i + shift
                # Contribution is h[i] * (1-h[partner])
                # Sensitivity: changing h[i] by dx changes c[k] by dx*(1-h[partner])
                score[i] += abs(1.0 - h[partner]) * c[k]
                score[partner] += abs(h[i]) * c[k]
        
        # Order by score (descending)
        idx = np.argsort(-score)
        
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_cd(h, n, max_iter, deadline, step_scale=0.1):
    """Subgradient method: pick random peak lag, update in that direction."""
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        
        # Pick the maximizing lag (or a random one among top few)
        top_k = np.argsort(-c)[:3]
        k = top_k[np.random.randint(3)]
        shift = k - (n - 1)
        
        # Subgradient of c[k] with respect to h:
        # c[k] = sum_i h[i] * (1-h[i+shift]) where i+shift in [0,n)
        # dc[k]/dh[i] = (1-h[i+shift]) - h[i-shift] (if in range)
        grad = np.zeros(n)
        i_lo = max(0, -shift)
        i_hi = min(n, n - shift)
        for i in range(i_lo, i_hi):
            partner = i + shift
            grad[i] += (1.0 - h[partner])
            grad[partner] -= h[i]
        
        # Projected gradient step
        step = step_scale / max(1, it // 100)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        new_c = compute_c5(h_new, n)
        if new_c < cur_max:
            h = h_new
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    best_h = h.copy()

            break
        t = t0 * (t1 / t0) ** (passed / max(1, max_passes))

        peak_lags = np.argsort(-c)[:3]

def cd_pass(h, n, order='peak', target_peaks=5):
    """One pass of coordinate descent."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    
    if order == 'peak':
        # Score each coordinate by its contribution to peak lags
        top_peaks = np.argsort(-c)[:target_peaks]
        score = np.zeros(n)
        for k in top_peaks:
            shift = k - (n - 1)
            i_lo = max(0, -shift)
            i_hi = min(n, n - shift)
            for i in range(i_lo, i_hi):
                partner = i + shift
                score[i] += c[k] * (1.0 - h[partner])
                if partner != i:
                    score[partner] += c[k] * h[i]
        idx = np.argsort(-score)
    elif order == 'random':
        idx = np.random.permutation(n)
    else:  # sequential
        idx = np.arange(n)
    
    improved = False
    for p in idx:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d_exact(c, h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
            improved = True
    if improved:
        h = project_feasible(h, n)
    return h, improved

def cd_loop(h, n, deadline, order='peak', target_peaks=5, max_passes=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        h, improved = cd_pass(h, n, order=order, target_peaks=target_peaks)
        if not improved:
            # Try a different ordering
            h, improved = cd_pass(h, n, order='random')
            if not improved:
                h, improved = cd_pass(h, n, order='sequential')
                if not improved:
                    break
        c = compute_c5(h, n)
        if c < best_c:
            best_h = h.copy()
            best_c = c
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max

        x_hi = 1.0

            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):

        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)

                if xl > r1 and xh < r2:

        else:

    L = L_hi
    x_lo = 0.0
    x_hi = 1.0

    if np.any(pos_m):

    x_lo = max(x_lo, 0.0)

def get_peak_order(c, h, n, npeaks):
    top = np.argsort(-c)[:npeaks]
    score = np.zeros(n)
    for k in top:
        shift = int(k) - (n - 1)
        i_lo = max(0, -shift)
        i_hi = min(n, n - shift)
        ck = c[k]
        for i in range(i_lo, i_hi):
            j = i + shift
            score[i] += ck * (1.0 - h[j])
            if j != i:
                score[j] += ck * h[i]
    return np.argsort(-score)

def cd_pass(h, n, order_idx):
    c = np.correlate(h, 1.0 - h, mode='full')
    h = h.copy()
    improved = False
    for p in order_idx:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d_exact(c, h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
            improved = True
    if improved:
        h = project_feasible(h, n)
    return h, improved

def cd_loop(h, n, deadline, npeaks=5, max_passes=50000):

    best_c = compute_c5(h, n)
    n_idx = np.arange(n)

            break

            c = np.correlate(h, 1.0 - h, mode='full')
            order_idx = get_peak_order(c, h, n, npeaks)
        else:
            order_idx = n_idx
        h, improved = cd_pass(h, n, order_idx)
        if not improved:

                c = np.correlate(h, 1.0 - h, mode='full')
                order_idx = np.random.permutation(n)
                h, improved = cd_pass(h, n, order_idx)
            if not improved:
                break
        if passed % 100 == 99:
            c_now = compute_c5(h, n)
            if c_now < best_c:

                best_c = c_now
    c_now = compute_c5(h, n)
    if c_now < best_c:
        best_h, best_c = h, c_now
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=250):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def mass_transfer_perturb(h, n, rng, n_pairs=3, scale=0.05):
    h_p = h.copy()
    for _ in range(n_pairs):
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        delta = rng.uniform(-scale, scale)
        h_p[p] += delta
        h_p[q] -= delta
    return project_feasible(h_p, n)
