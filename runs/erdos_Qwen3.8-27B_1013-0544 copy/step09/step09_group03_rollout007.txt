def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    w = np.exp(alpha * (c - cmax))
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        m = k + n - 1
        wk = w[m]
        if wk < 1e-15:
            continue
        # (1-h[p+k]) for valid p: p in [max(0,-k), min(n, n-k))
        # -h[p-k] for valid p: p in [max(0,k), min(n, n+k))
        p_min1 = max(0, -k)
        p_max1 = min(n, n - k)
        p_min2 = max(0, k)
        p_max2 = min(n, n + k)
        grad[p_min1:p_max1] += wk * (1.0 - h[p_min1:p_max1 + k])
        grad[p_min2:p_max2] += wk * (-h[p_min2 - k:p_max2 - k])
    return dx * grad
```

        for lr in [0.01, 0.005, 0.001, 0.0005, 0.0001]:

            c_trial = compute_c5(h_trial, n)
            if c_trial < c_old:
                h = h_trial
                if c_trial < best_c:
                    best_c = c_trial
                break
        else:
            break
    return h, best_c
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + diff / tot * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + diff / tot * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s.min()
    m_max = n - 1 + s.max()
    all_m = np.arange(2 * n - 1)
    unaff = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff].max()) if unaff.any() else 0.0
    c0 = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(52):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        if a[active].min() > 1e-12:
            hi = mid
        elif a[active].max() < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0 + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=500, deadline=None, seed=0, cyclic=False):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = range(n) if cyclic else rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10:
                h[p] = nv
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, best_c

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    w = np.exp(alpha * (c - cmax))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
        ws = 1.0
    w /= ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-14:
            continue
        lo1, hi1 = max(0, -k), min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1:hi1 + k])
        lo2, hi2 = max(0, k), min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - k:hi2 - k]
    return dx * grad

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    return (cm + np.log(np.exp(alpha * (c - cm)).sum()) / alpha) * (2.0 / n)

def proj_grad(h, n, alpha, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        if np.linalg.norm(g) < 1e-12:
            break
        ho = h.copy()
        oo = smooth_obj(h, n, alpha)
        step = lr
        ok = False
        for _ in range(25):
            ht = project_feasible(ho - step * g, n)
            ot = smooth_obj(ht, n, alpha)
            if ot < oo - 1e-12:
                h = ht
                ok = True
                break
            step *= 0.5
        if not ok:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    return h, best_c

def subgrad_descent(h, n, max_iter=150, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        active = c >= c.max() - 1e-10
        if not active.any():
            break
        grad = np.zeros(n)
        nact = int(active.sum())
        for k in np.where(active)[0]:
            sk = int(k) - (n - 1)
            lo1, hi1 = max(0, -sk), min(n, n - sk)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - h[lo1:hi1 + sk])
            lo2, hi2 = max(0, sk), min(n, n + sk)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2 - sk:hi2 - sk]
        grad /= nact
        ho = h.copy()
        for lr in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            ht = project_feasible(ho - lr * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c or ct < c.max() * dx - 1e-6:
                h = ht
                if ct < best_c:
                    best_c = ct
                break
        else:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    c_flip = compute_c5(h_flip, n)
