**Approach: Frank-Wolfe (conditional gradient) on the smooth approximation**

The problem min max_k c_k(h) can be approximated by min sum_k exp(alpha * c_k(h)) / alpha (log-sum-exp).

The Frank-Wolfe method works by:
1. Computing the gradient of the smooth approximation
2. Finding the vertex of the feasible polytope that minimizes the linear approximation (i.e., solving a linear program)
3. Taking a step toward that vertex

The feasible set is {h: 0≤h≤1, sum(h)=n/2}. The vertices of this set are {0,1}-valued vectors with exactly n/2 ones. So the linear oracle is: find the n/2 coordinates with the smallest gradient values, and set those to 1, the rest to 0.

This is actually very efficient! The "LP oracle" is just sorting.

Let me implement this:

```python
def frank_wolfe_step(grad, n):
    # Find the vertex: n/2 coordinates set to 1, rest to 0
    # Minimize grad @ v subject to v being a 0-1 vector with sum = n/2
    # Solution: set v[i] = 1 for the n/2 smallest grad[i]
    k = n // 2
    idx = np.argsort(grad)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v
```

Then the Frank-Wolfe iteration:
```python
def fw_iter(h, n, alpha, step_size):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Gradient of log-sum-exp(alpha * c) * dx
    # d/dh_p [sum_k exp(alpha*c_k)] = alpha * sum_k exp(alpha*c_k) * dc_k/dh_p
    # dc_k/dh_p = (1-h[p+k]) - h[p-k] (with appropriate bounds) + diagonal term
    
    weights = np.exp(alpha * (c - c.max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for p in range(n):
        # compute dc_k/dh_p for all k, weighted sum
        # This is O(n) per p, so O(n²) total
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        dc = np.zeros(len(ms))
        if np.any(vf):
            dc[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            dc[vb] -= h[jb[vb]]
        diag = ms == (n - 1)
        dc[diag] = (1.0 - 2.0 * h[p])  # derivative of x(1-x) is 1-2x
        grad[p] = np.sum(weights[ms] * dc[ms])
    
    v = frank_wolfe_step(grad, n)
    h_new = (1 - step_size) * h + step_size * v
    return h_new, grad
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(h + (diff / tot) * room, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(h + (diff / tot) * h, 0.0)
    return h

def fw_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        dc = np.zeros(len(ms))
        if np.any(vf):
            dc[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            dc[vb] -= h[jb[vb]]
        diag = (ms == (n - 1))
        if np.any(diag):
            dc[diag] = (1.0 - 2.0 * h[p])
        grad[p] = np.dot(w[ms], dc)
    return grad * dx

def fw_oracle(grad, n):
    k = n // 2
    idx = np.argpartition(grad, k - 1)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def frank_wolfe(h, n, alphas, iters_per_alpha, deadline):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    total_it = 0
    for alpha in alphas:
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c
            total_it += 1
            t = 2.0 / (total_it + 2)
            grad = fw_gradient(h, n, alpha)
            v = fw_oracle(grad, n)
            h_new = (1.0 - t) * h + t * v
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
            h = h_new
    return best_h, best_c

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))

                    x_lo = max(x_lo, r2)

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if h.sum() != n / 2.0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
