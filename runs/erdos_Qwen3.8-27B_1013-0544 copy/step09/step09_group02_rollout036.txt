```python
def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()  # should be 0
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = exp_shifted.sum()
    softmax = exp_shifted / sum_exp
    F = m + (max_shifted + np.log(sum_exp)) / alpha
    # Note: max_shifted = alpha*(c.max()-m) = 0, so F = m + log(sum_exp)/alpha
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        # shift_k = k'-(n-1), k' ranges 0 to 2n-2
        # Valid: 0 ≤ j-shift < n AND 0 ≤ j+shift < n
        # j-shift ≥ 0 → shift ≤ j
        # j-shift < n → shift > j-n → shift ≥ j-n+1
        # j+shift ≥ 0 → shift ≥ -j
        # j+shift < n → shift < n-j → shift ≤ n-j-1
        # shift range: max(-(n-1), j-n+1, -j) to min(n-1, j, n-j-1)
        s_lo = max(-(n-1), j-n+1, -j)
        s_hi = min(n-1, j, n-j-1)
        if s_lo > s_hi:
            continue
        shifts = np.arange(s_lo, s_hi+1)
        k_idxs = shifts + (n-1)  # convert to output index
        sw = softmax[k_idxs]
        grad[j] = np.sum(sw * ((1.0 - h[j - shifts]) - h[j + shifts]))
    
    return F, grad
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    softmax = exp_s / sum_exp
    F = m + (max_s + np.log(sum_exp)) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n-1), j-n+1, -j)
        s_hi = min(n-1, j, n-j-1)
        if s_lo > s_hi:
            continue
        shifts = np.arange(s_lo, s_hi+1)
        k_idxs = shifts + (n-1)
        sw = softmax[k_idxs]
        grad[j] = np.sum(sw * ((1.0 - h[j - shifts]) - h[j + shifts]))
    return F, grad

def adam_optimize(h, n, alpha, n_iter=5000, lr=0.01, deadline=None):

    v = np.zeros(n)
    t = 0
    best_F = smooth_max_and_grad(h, n, alpha)[0]
    best_h = h.copy()
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        if it % 200 == 0:
            c_val = compute_c5(h, n)
            if c_val < best_F * 2.0 / n:  # hmm, need to be consistent
                pass
        F, g = smooth_max_and_grad(h, n, alpha)
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, n)
        F_new = smooth_max_and_grad(h, n, alpha)[0]
        if F_new < best_F:
            best_F = F_new
            best_h = h.copy()
    
    return best_h, best_F * 2.0 / n

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(35):  # reduced from 45

        elif vmax < -1e-10:

                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    softmax = exp_s / sum_exp
    F = m + (max_s + np.log(sum_exp)) / alpha
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n - 1), j - n + 1, -j)
        s_hi = min(n - 1, j, n - j - 1)
        if s_lo > s_hi:
            continue
        shifts = np.arange(s_lo, s_hi + 1)
        k_idx = shifts + (n - 1)
        sw = softmax[k_idx]
        idx_f = j - shifts
        idx_b = j + shifts
        grad[j] = np.dot(sw, (1.0 - h[idx_f]) - h[idx_b])
    return F, grad

def adam_optimize(h, n, alpha, n_iter, lr, deadline=None):
    h = h.copy()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    best_F = smooth_max_and_grad(h, n, alpha)[0]
    best_h = h.copy()
    stall = 0
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        F, g = smooth_max_and_grad(h, n, alpha)
        t += 1
        lr_t = lr / (1.0 + 0.001 * t)
        m_buf = 0.9 * m_buf + 0.1 * g
        v_buf = 0.999 * v_buf + 0.001 * g * g
        m_hat = m_buf / (1.0 - 0.9 ** t)
        v_hat = v_buf / (1.0 - 0.999 ** t)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, n)
        F_new = smooth_max_and_grad(h, n, alpha)[0]
        if F_new < best_F - 1e-10:
            best_F = F_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                break
    return best_h, best_F * 2.0 / n

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=300, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(400, 0.5)
    n = len(h_init)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Adam with progressive alpha sharpening
    alphas = [50, 200, 800, 3000, 10000, 30000]
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 40:
            break
        n_iter = int(min(4000, remaining * 40))
        h_a, c_a = adam_optimize(best_h, n, alpha, n_iter, lr=0.008, deadline=deadline)
        if c_a < best_c:
            best_h, best_c = h_a, c_a
        # Quick CD after each Adam phase
        remaining = deadline - time.time()
        if remaining > 60:
            h_cd, c_cd = coord_cd(best_h, n, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
            if c_cd < best_c:
                best_h, best_c = h_cd, c_cd

    # Phase 2: Extended CD
    remaining = deadline - time.time()
    if remaining > 80:
        h2, c2 = coord_cd(best_h, n, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Multi-resolution
    x_cur = np.linspace(0, 2, n)
    for n_try in [400, 500, 700, 850, 1000, 1200]:
        remaining = deadline - time.time()
        if remaining < 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_m, c_m = adam_optimize(h_int, n_try, 5000, n_iter=2000, lr=0.006, deadline=deadline)
        h_m, c_m = coord_cd(h_m, n_try, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
        if c_m < best_c:
            best_h, best_c = h_m, c_m
            n = n_try
            x_cur = x_new

    # Phase 4: Final high-alpha Adam + CD
    remaining = deadline - time.time()
    if remaining > 60:
        h_f, c_f = adam_optimize(best_h, n, 50000, n_iter=4000, lr=0.003, deadline=deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_cd(best_h, n, max_passes=500, deadline=deadline, shuffle=True, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 5: Random perturbation restarts
    n_cur = len(best_h)
    for r in range(10):
        remaining = deadline - time.time()
        if remaining < 50:
            break
        s = 0.001 + 0.003 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        hr, cr = coord_cd(hp, n_cur, max_passes=100, deadline=deadline, shuffle=True, rng=rng)
        if cr < best_c:
            best_h, best_c = hr, cr

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
