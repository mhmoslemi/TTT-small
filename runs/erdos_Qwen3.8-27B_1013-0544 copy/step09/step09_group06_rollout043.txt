    while time.time() < deadline and moves < max_moves:

        U = float(raw[mask].max()) if mask.any() else 0.0

        t = moves % 5

def project_simplex(v, target):
    v = np.asarray(v, dtype=float)
    n=len(v)
    # bisection tau
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    # x(tau)=clip(v - tau,0,1)
    # sum decreasing in tau.
    for _ in range(60):
        tau=(lo+hi)*0.5
        s=np.clip(v - tau, 0.0, 1.0).sum()
        if s > target: # need increase tau to decrease
            lo=tau
        else:
            hi=tau
    tau=(lo+hi)*0.5
    x=np.clip(v-tau,0,1)
    # fix rounding sum
    diff=target-x.sum()
    if abs(diff)>1e-12:
        # distribute to interior
        mask=(x>1e-8)&(x<1-1e-8)
        if mask.any(): x[mask]+=diff/mask.sum()
    return np.clip(x,0,1)

def linear_envelope_min(c, a, lo, hi, work, steps=8):
    if hi <= lo: return lo
    for _ in range(steps):
       mid=0.5*(lo+hi)

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_simplex(v, target):
    v = np.asarray(v, dtype=float)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-10:
        m = (x > 1e-9) & (x < 1.0 - 1e-9)
        k = int(m.sum())
        if k:
            x[m] += d / float(k)
    return np.clip(x, 0.0, 1.0)

def finalize_h(h, target):
    x = np.clip(h, 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-11:
        m = (x > 1e-9) & (x < 1.0 - 1e-9)
        k = int(m.sum())
        if k:
            x[m] += d / float(k)
        else:
            x = project_simplex(x, target)
    return np.clip(x, 0.0, 1.0)

def corr_grad(h, w):
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(n, dtype=float)
    for idx in range(L):
        wt = float(w[idx])
        if wt <= 1e-12:
            continue
        s = idx - n + 1
        if s >= 0:
            r = n - s
            g[:r] += wt - wt * h[s:]
            g[s:] -= wt * h[:r]
        else:
            r = n + s
            g[r:] += wt - wt * h[:r]
            g[:r] -= wt * h[r:]
    return g

def smooth_val_grad(x, alpha, n, pen):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    eu = np.exp(z)
    s = float(eu.sum())
    if s <= 0.0:
        w = np.zeros_like(eu)
        w[np.argmax(c)] = 1.0
    else:
        w = eu / s
    val = dx * (m + np.log(s) / alpha) if alpha > 0.0 else dx * m
    g = dx * corr_grad(h, w)
    if pen > 0.0:
        d = float(h.sum() - 0.5 * n)
        val += 0.5 * pen * d * d
        g = g + pen * d
    return float(val), g

def lbfgs_obj(x, alpha, pen, n):
    return smooth_val_grad(x, alpha, n, pen)

def lbfgs_run(h, n, alpha, pen, maxiter):
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(lbfgs_obj, h, args=(alpha, pen, n), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': int(maxiter), 'ftol': 1e-15,
                                'gtol': 1e-8, 'disp': False})
        if res.x is not None:
            h2 = finalize_h(res.x, 0.5 * n)
            return h2, compute_c5(h2, n)
    except Exception:
        pass
    return h, compute_c5(h, n)

def linear_envelope_min(c, a, lo, hi, work, steps=8):
    if hi <= lo:
        return lo
    for _ in range(steps):
        mid = 0.5 * (lo + hi)
        np.multiply(a, mid, out=work)
        work += c
        m = float(np.max(work))
        tol = 1e-9 * max(1.0, abs(m)) + 1e-12
        active = a[np.abs(work - m) <= tol]
        if active.size:
            amin = float(np.min(active))
            amax = float(np.max(active))
        else:
            ii = int(np.argmax(work))
            amin = amax = float(a[ii])
        if amin > 1e-10:
            hi = mid
        elif amax < -1e-10:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def solve_minimax_quad(c0, a, b, U, lo, hi):
    T = len(c0)
    cands = [lo, hi, 0.0]
    for i in range(T):
        A = b[i]
        if A > 1e-12:
            cands.append(-a[i] / (2.0 * A))
        A = b[i]
        B = a[i]
        C = c0[i] - U
        if abs(A) < 1e-12:
            if abs(B) > 1e-12:
                cands.append(-C / B)
        else:
            disc = B * B - 4.0 * A * C
            if disc >= -1e-10:
                if disc < 0.0:
                    disc = 0.0
                r = np.sqrt(disc)
                cands.append((-B + r) / (2.0 * A))
                cands.append((-B - r) / (2.0 * A))
    for i in range(T):
        bi = b[i]
        ai = a[i]
        ci = c0[i]
        for j in range(i + 1, T):
            A = bi - b[j]
            B = ai - a[j]
            C = ci - c0[j]
            if abs(A) < 1e-12:
                if abs(B) > 1e-12:
                    cands.append(-C / B)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= -1e-10:
                    if disc < 0.0:
                        disc = 0.0
                    r = np.sqrt(disc)
                    cands.append((-B + r) / (2.0 * A))
                    cands.append((-B - r) / (2.0 * A))
    best_d = lo
    best_v = float('inf')
    for d in cands:
        if not np.isfinite(d):
            continue
        if d < lo:
            d = lo
        elif d > hi:
            d = hi
        vals = c0 + d * a + d * d * b
        v = float(np.max(vals))
        if U > v:
            v = U
        if v < best_v:
            best_v = v
            best_d = float(d)
    return best_d, best_v

def eval_pair_candidate(work, raw, alpha, d, i0, i1, i2):
    np.multiply(alpha, d, out=work)
    work += raw
    d2 = d * d
    work[i0] -= 2.0 * d2
    work[i1] += d2
    work[i2] += d2
    return float(np.max(work))

def unique_candidates(cands, lo, hi, tol=1e-11):
    out = []
    for d in cands:
        d = float(d)
        if not np.isfinite(d):
            continue
        if d < lo:
            d = lo
        elif d > hi:
            d = hi
        ok = True
        for x in out:
            if abs(d - x) < tol:
                ok = False
                break
        if ok:
            out.append(d)
    return out

def pair_cd(h_in, n, deadline, top_k=15, max_moves=8000, grid_every=10, seed=1):

    h = finalize_h(np.asarray(h_in, dtype=float).copy(), target)

    inv = 1.0 - h

    work = np.zeros(L, dtype=float)
    rng = np.random.RandomState(seed)
    moves = 0

    limit = 6 * n

        if moves % 128 == 0:

        elif tp == 2:

            q = int(rng.randint(n))
        if q == p:
            q = (q + 1) % n
        idx1 = n - 1 + (q - p)
        idx2 = n - 1 + (p - q)
        hp = float(h[p])
        hq = float(h[q])
        lo = max(-hp, hq - 1.0)
        hi = min(1.0 - hp, hq)
        if hi - lo < 1e-13:
            moves += 1
            stagnant += 1
            continue
        alpha.fill(0.0)
        a = n - 1 - p
        alpha[a:a + n] += inv
        a = n - 1 - q
        alpha[a:a + n] -= inv
        hrev = h[::-1]
        alpha[p:p + n] -= hrev
        alpha[q:q + n] += hrev
        top_list = []
        for vv in top_idx:
            top_list.append(int(vv))
        for f in (idx0, idx1, idx2):
            found = False
            for vv in top_list:
                if vv == f:
                    found = True
                    break
            if not found:
                top_list.append(f)
        top_arr = np.array(top_list, dtype=int)
        c_peak = raw[top_arr]
        a_peak = alpha[top_arr]
        b_peak = np.zeros(top_arr.shape, dtype=float)
        for ii in range(top_arr.size):
            mm = int(top_arr[ii])
            if mm == idx0:
                b_peak[ii] = -2.0
            elif mm == idx1 or mm == idx2:
                b_peak[ii] = 1.0
        mask = np.ones(L, dtype=bool)
        mask[top_arr] = False
        U = float(np.max(raw[mask])) if mask.any() else 0.0
        d_lin = linear_envelope_min(raw, alpha, lo, hi, work, 6)
        d_top, _ = solve_minimax_quad(c_peak, a_peak, b_peak, U, lo, hi)
        cands = [0.0, lo, hi, 0.5 * (lo + hi), d_lin, d_top]
        if moves % grid_every == 0:
            gg = np.linspace(lo, hi, 7)
            for gv in gg:
                cands.append(float(gv))
        cands = unique_candidates(cands, lo, hi)
        best_v = cur
        best_d = 0.0
        for d in cands:
            v = eval_pair_candidate(work, raw, alpha, d, idx0, idx1, idx2)
            if v < best_v:
                best_v = v
                best_d = d
        if best_v < cur - 1e-12 or (moves % 20 == 0):
            span = hi - lo
            offs = [-0.4, -0.15, -0.05, -0.015, 0.015, 0.05, 0.15, 0.4]
            base = best_d
            for off in offs:
                d = base + off * span
                if lo <= d <= hi:
                    v = eval_pair_candidate(work, raw, alpha, d, idx0, idx1, idx2)
                    if v < best_v:
                        best_v = v
                        best_d = d
        tol_i = 1e-10 * max(1.0, cur)
        if best_v < cur - tol_i and abs(best_d) > 1e-14:
            d = best_d
            d2 = d * d
            raw += d * alpha
            raw[idx0] -= 2.0 * d2
            raw[idx1] += d2
            raw[idx2] += d2
            h[p] += d
            inv[p] -= d
            h[q] -= d
            inv[q] += d
            cur = float(np.max(raw))
            stagnant = 0
        else:
            if abs(best_d) <= 1e-14:
                cur = best_v
            stagnant += 1
        moves += 1
    h = finalize_h(h, target)
    return h, compute_c5(h, n)

def resample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_simplex(np.interp(x_new, x_old, h), 0.5 * n_new)

def lp_alternate(h, n, deadline, iters=2):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    L = 2 * n - 1
    for _ in range(iters):
        if time.time() + 15.0 > deadline:
            break
        g = 1.0 - h
        A = np.zeros((L, n + 1), dtype=float)
        for m in range(L):
            s = m - (n - 1)
            if s >= 0:
                A[m, :n - s] = g[s:]
            else:
                r = -s
                A[m, r:] = g[:n - r]
            A[m, n] = -1.0
        cobj = np.zeros(n + 1, dtype=float)
        cobj[n] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        b_eq = np.array([0.5 * n], dtype=float)
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        opts = {'presolve': True, 'time_limit': max(5.0, min(80.0, deadline - time.time()))}
        res = None
        try:
            res = linprog(cobj, A_ub=A, b_ub=np.zeros(L), A_eq=Aeq, b_eq=b_eq,
                          bounds=bounds, method='highs', options=opts)
        except Exception:
            res = None
        if res is not None and res.status == 0 and res.x is not None:
            x = np.clip(res.x[:n], 0.0, 1.0)
            for beta in (1.0, 0.6, 0.3):
                hx = project_simplex((1.0 - beta) * h + beta * x, 0.5 * n)
                hc = compute_c5(hx, n)
                if hc < best_c:
                    best_h = hx
                    best_c = hc
            h = best_h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(400, 0.5)
    n0 = int(len(h0))
    best_h = finalize_h(h0, 0.5 * n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    d1 = min(t0 + 300.0, deadline - 520.0)
    if d1 > time.time() + 20.0:
        hh, cc = pair_cd(best_h, best_n, d1, top_k=15, max_moves=12000,
                         grid_every=12, seed=seed)
        if cc < best_c:
            best_h = hh
            best_c = cc

    for alpha, pen, iters in ((500, 80, 120), (4000, 500, 80)):
        if time.time() + 30.0 > deadline:
            break
        hh, cc = lbfgs_run(best_h, best_n, alpha, pen, iters)
        if cc < best_c:
            best_h = hh
            best_c = cc

    if best_n > 250 and deadline - time.time() > 320.0:
        nc = int(best_n // 2)
        nc = int((nc // 2) * 2)
        if nc >= 120:
            hc = resample_h(best_h, nc)
            rem = deadline - time.time()
            dc = time.time() + min(150.0, rem - 150.0)
            if dc < time.time() + 10.0:
                dc = min(time.time() + 50.0, deadline - 20.0)
            hh, cc = pair_cd(hc, nc, dc, top_k=15, max_moves=20000,
                             grid_every=8, seed=seed + 10)
            if time.time() + 20.0 < deadline:
                hh, cc = lbfgs_run(hh, nc, 2000, 200, 150)
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = nc

    if deadline - time.time() > 120.0 and best_n <= 800:
        ldl = time.time() + min(60.0, deadline - time.time() - 30.0)
        hh, cc = lp_alternate(best_h, best_n, ldl, iters=2)
        if cc < best_c:
            best_h = hh
            best_c = cc

    if deadline - time.time() > 120.0:
        hh, cc = lbfgs_run(best_h, best_n, 8000, 1000, 80)
        if cc < best_c:
            best_h = hh
            best_c = cc

    if deadline - time.time() > 30.0:
        maxm = 20000 if best_n < 400 else 12000
        hh, cc = pair_cd(best_h, best_n, deadline - 2.0, top_k=20,
                         max_moves=maxm, grid_every=6, seed=seed + 100)
        if cc < best_c:
            best_h = hh
            best_c = cc

    best_h = finalize_h(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
