    jb = p - ms + (n - 1)

def gradient_descent(h, n, deadline, lr=0.005, max_iters=5000):
    """Feasible gradient descent with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = corr.max()
        # Subgradient: for each p, the contribution to the active lags
        grad = np.zeros(n)
        active = corr > cmax - 0.05 * cmax
        for p in range(n):
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                contrib[vb] -= h[jb[vb]]
            di = ms == (n - 1)
            contrib[di] = 1.0 - 2.0 * h[p]
            mask = active[ms]
            if mask.any():
                grad[p] = np.sum(contrib[mask]) / mask.sum()
        # Adaptive step size
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
        # Decay learning rate
        lr = max(lr * 0.999, 1e-5)
    return best_h, best_c

def smooth_logsumexp(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:

        except Exception:
            pass
    return best_h, best_c

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
