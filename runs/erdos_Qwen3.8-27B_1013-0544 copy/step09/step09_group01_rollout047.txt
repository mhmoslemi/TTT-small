def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=400):

    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})

def make_restart(best_h, n, rng, r):

        return project_feasible(best_h + rng.normal(0, s, n), n)
    elif mode == 1:
        x = np.linspace(0, 2, n)
        freq = 2.0 + 6.0 * ((r // 5) / max(1, 20))
        amp = 0.02 + 0.05 * (r / max(1, 10))
        return project_feasible(best_h + amp * np.sin(np.pi * freq * x + rng.randn()), n)
    elif mode == 2:
        shift = rng.randint(-n // 4, n // 4 + 1)
        hp = np.roll(best_h, shift)
        return project_feasible(hp, n)
    elif mode == 3:
        hp = best_h.copy()
        n_flip = n // 4
        idx = rng.choice(n, n_flip, replace=False)
        hp[idx] = 1.0 - hp[idx]
        return project_feasible(hp, n)
    else:
        lo = rng.randint(0, n // 2)
        hi = lo + rng.randint(n // 6, n // 3)
        hp = best_h.copy()
        hp[lo:hi] = 1.0 - hp[lo:hi]
        return project_feasible(hp, n)
