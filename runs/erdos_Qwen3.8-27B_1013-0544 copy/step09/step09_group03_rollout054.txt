```python
    y = h.copy()  # Nesterov momentum

    # Learning rate (need to estimate Lipschitz constant)
    # The gradient of c[k] has norm ~ O(sqrt(n)) for each k
    # The smooth max has gradient norm ~ O(1) (since weights sum to 1)
    # Actually, ||grad f_α|| ≤ ||grad c[k*]|| for any k*
    # And ||grad c[k]||_2 = dx * sqrt(sum of (grad components)²)
    # Each grad component is O(dx), and there are O(n) of them, so norm ~ O(sqrt(n)*dx) = O(1)
    L = 4.0  # estimated Lipschitz constant
    lr = 1.0 / L
    
    for it in range(iters):
        if deadline is not None and time.time() > deadline:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        # Nesterov step
        beta = (t_k - 1.0) / t_new
        y_new = x + beta * (y - x) + (1.0 - beta) * (y - x)
        # Actually standard Nesterov: y = x + (t-1)/t_new * (x - y_prev)
        y_new = x + (t_k - 1.0) / t_new * (x - y)
        t_k = t_new
        
        # Gradient at y_new
        c_y = np.correlate(y_new, 1.0-y_new, mode='full') * dx
        c_y_max = c_y.max()
        w = np.exp(alpha * (c_y - c_y_max) / 2.0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w /= w_sum
        
        # Gradient: grad = sum_k w[k] * grad c[k]
        # grad c[k]/dh[p] = (1-h[p-(k-n+1)]) - h[p+(k-n+1)] (with bounds)
        # This is expensive to compute for all k. Let me use FFT or a simpler approach.
        
        # Actually, the gradient of the weighted sum is:
        # ∑_k w[k] * [(1-y_new[p-shift_k]) - y_new[p+shift_k]]
        # where shift_k = k - (n-1)
        # This can be written as:
        # (1-y_new) * W_pos + W_neg * y_new  (approximately)
        # where W_pos[p] = sum_{k: shift_k = p} w[k] ... hmm not quite
        
        # Let me think differently. 
        # grad[p] = sum_k w[k] * (1-y_new[p-shift(k)]) - sum_k w[k] * y_new[p+shift(k)]
        # Let shift(k) = k - (n-1), so k = n-1+shift
        # First term: sum_k w[n-1+shift] * (1-y_new[p-shift]) for valid p-shift
        # = (1-y_new) convolved with w (with appropriate indexing)
        
        # This is a convolution! We can compute it efficiently.
        # But for n=600, direct computation is fine.
        
        grad = np.zeros(n)
        # For efficiency, precompute which k values have significant weight
        top_idx = np.argsort(-w)[:20]  # top 20 weights
        for k in top_idx:
            shift = k - (n - 1)
            v1 = (np.arange(n) - shift >= 0) & (np.arange(n) - shift < n)
            grad[v1] += w[k] * (1.0 - y_new[np.arange(n)[v1] - shift])
            v2 = (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n)
            grad[v2] -= w[k] * y_new[np.arange(n)[v2] + shift]
        
        # Gradient step
        x = y_new - lr * grad
        x = project_feasible(x, n)
        y = y_new  # Update y (this is a simplification; true Nesterov is more complex with projection)
        
        # Check improvement
        c_new = compute_c5(x, n)
        if c_new < best_c:
            best_h = x.copy()
            best_c = c_new
    
    return best_h, best_c
```

Hmm, the Nesterov with projection is tricky. The standard Nesterov method doesn't handle projections well. Let me use a simpler approach:

**Proximal gradient (FISTA) with projection:**
- The projection onto {0≤h≤1, sum(h)=n/2} is our project_feasible
- FISTA: y_{k+1} = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
  x_{k+1} = proj(y_{k+1} - lr * grad f(y_{k+1}))

This is clean and well-defined. Let me use this.

The main cost per iteration is computing the gradient, which involves a "convolution" of weights with h. If I limit to top-K weights, it's O(K*n) per iteration.

For K=20, n=600: O(12000) per iteration. With 100K iterations: 1.2B operations. In numpy, maybe 60-120 seconds. That's reasonable.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s = np.arange(-(n - 1 - p), n - p)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps, pm = p + s, p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += 1.0 - h[ps[vps]]
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    c0 = c[n - 1] - h0 * (1.0 - h0)
    m_lo, m_hi = n - 1 + s.min(), n - 1 + s.max()
    unaff = np.arange(2 * n - 1)
    mask = ((unaff < m_lo) | (unaff > m_hi)) & (unaff != n - 1)
    M_unaff = float(c[mask].max()) if mask.any() else 0.0
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-13:
            break
        act = vals >= am - 1e-12
        if not act.any():
            break
        if float(a[act].min()) > 1e-11:
            hi = mid
        elif float(a[act].max()) < -1e-11:
            lo = mid
        else:
            break
    best_x, best_v = h0, float('inf')
    for x in [lo, hi, (lo + hi) * 0.5]:
        v = max(float((a * x + b).max()), M_unaff, c0 + x - x * x)
        if v < best_v:
            best_x, best_v = float(x), v
    return best_x

def coord_descent(h, n, iters, deadline, seed=0):
    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(iters):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        imp = 0
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1 - 1e-10:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10:
                h[p] = nv
                imp += 1
        if imp == 0:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, best_c

def compute_smooth_grad(y, n, alpha, top_k=30):
    dx = 2.0 / n
    c = np.correlate(y, 1.0 - y, mode='full') * dx
    cm = c.max()
    w = np.exp(alpha * (c - cm))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        ws = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    idx = np.argsort(-w)[:top_k]
    ar = np.arange(n)
    for k in idx:
        shift = k - (n - 1)
        v1 = (ar - shift >= 0) & (ar - shift < n)
        v2 = (ar + shift >= 0) & (ar + shift < n)
        if v1.any():
            grad[v1] += w[k] * (1.0 - y[ar[v1] - shift])
        if v2.any():
            grad[v2] -= w[k] * y[ar[v2] + shift]
    return grad

def fista_smooth(h, n, alpha=300, iters=80000, deadline=None, seed=0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    x_prev = h.copy()
    t_k = 1.0
    L = 2.0 / dx * 1.5
    lr = 1.0 / L
    top_k = 30
    for it in range(iters):
        if deadline is not None and time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = x + beta * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        grad = compute_smooth_grad(y, n, alpha, top_k)
        x_new = y - lr * grad
        x_new = project_feasible(x_new, n)
        x_prev = x
        x = x_new
        t_k = t_new
        if it % 200 == 0:
            nc = compute_c5(x, n)
            if nc < best_c:
                best_c = nc
                best_h = x.copy()
    nc = compute_c5(x, n)
    if nc < best_c:
        best_h = x.copy()
        best_c = nc
    return best_h, best_c

def binary_search(h, n, deadline, seed=0, max_iters=3000):
    rng = np.random.RandomState(seed)
    h_bin = np.zeros(n)
    top = np.argsort(-h)[:n // 2]
    h_bin[top] = 1.0
    best_c = compute_c5(h_bin, n)
    best_bin = h_bin.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        k_max = int(np.argmax(c))
        shift = k_max - (n - 1)
        improved = False
        for trial in range(80):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if h_bin[i] == h_bin[j] or i == j:
                continue
            h_bin[i] = 1.0 - h_bin[i]
            h_bin[j] = 1.0 - h_bin[j]
            nc = compute_c5(h_bin, n)
            if nc < best_c - 1e-14:
                best_c = nc
                best_bin = h_bin.copy()
                improved = True
                break
            else:
                h_bin[i] = 1.0 - h_bin[i]
                h_bin[j] = 1.0 - h_bin[j]
        if not improved:
            break
    return project_feasible(best_bin, n), best_c
```
