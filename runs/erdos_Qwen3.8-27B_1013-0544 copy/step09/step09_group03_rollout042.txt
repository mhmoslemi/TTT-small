import numpy as np
import time
import math

def correlate_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def bound_of(h):
    n = len(h)
    return float(np.max(correlate_raw(h)) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(h.astype(float, copy=True), 0.0, 1.0)
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    th = (target - s) / n
    h1 = np.clip(h + th, 0.0, 1.0)
    if abs(float(h1.sum()) - target) < 1e-10:
        h = h1
    else:
        lo = -1.0; hi = 1.0
        for _ in range(45):
            th = 0.5 * (lo + hi)
            st = float(np.clip(h + th, 0.0, 1.0).sum())
            if st > target:
                hi = th
            else:
                lo = th
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def cd_refine(start_h, n, deadline, seed=0, iters=22, sum_thr=0.7):
    h = project_feasible(np.array(start_h, dtype=float).copy(), n)
    c = correlate_raw(h)
    dx = 2.0 / n
    target = 0.5 * n
    cmax = float(c.max())
    cval = cmax * dx
    best = h.copy(); best_c = cval
    rng = np.random.RandomState(seed)
    stall = 0
    while time.time() < deadline:
        order = rng.permutation(n)
        improved = 0
        count = 0
        sum_h = float(h.sum())
        cmax = float(c.max())
        for p in order:
            if time.time() > deadline - 0.4: break
            if count % 10 == 0:
                cmax = float(c.max())
            res = coord_update(h, c, n, int(p), cmax, iters)
            if res is not None:
                oldx, newx = res
                sum_h += newx - oldx
                improved += 1
            count += 1
            if abs(sum_h - target) > sum_thr:
                h = project_feasible(h, n)
                c = correlate_raw(h)
                cmax = float(c.max())
                sum_h = float(h.sum())
                count = 0
        h = project_feasible(h, n)
        c = correlate_raw(h)
        cmax = float(c.max())
        cval = cmax * dx
        if cval < best_c:
            best_c = cval; best = h.copy()
        if improved == 0:
            stall += 1
            if stall >= 2: break
        else:
            stall = 0
    return best, best_c

def resample_h(h, n):
    m=len(h)
    if m==n: return project_feasible(h,n)
    xo=np.linspace(0.0,2.0,m)
    xn=np.linspace(0.0,2.0,n)
    h2=np.interp(xn,xo,h,left=float(h[0]),right=float(h[-1]))
    return project_feasible(h2,n)

def balanced_binary(h):
    n=len(h); b=np.zeros(n)
    k=int(round(0.5*n))
    idx=np.argsort(h)[::-1]
    b[idx[:k]]=1.0
    return project_feasible(b,n)

def reflect_sym(h):
    return project_feasible(0.5*(h + (1.0 - h[::-1])), len(h))

import numpy as np

def correlate_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def bound_of(h):
    n = len(h)
    return float(np.max(correlate_raw(h)) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(h.astype(float, copy=True), 0.0, 1.0)
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    th = (target - s) / n
    h1 = np.clip(h + th, 0.0, 1.0)
    if abs(float(h1.sum()) - target) < 1e-10:
        h = h1
    else:
        lo, hi = -1.0, 1.0
        for _ in range(45):
            th = 0.5 * (lo + hi)
            if float(np.clip(h + th, 0.0, 1.0).sum()) > target:
                hi = th
            else:
                lo = th
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def coord_update(h, c, n, p, cmax, iters=22):

        v = float(np.max(c[mhi + 1:]))
        if v > M0:
            M0 = v
    if M0 >= cmax - 1e-11:
        return None

    s = np.arange(smin, smax + 1, dtype=int)
    nz = s != 0
    midx = m0 + s
    a = np.zeros(midx.size, dtype=float)

    i = p - s
    mp = (i >= 0) & (i < n) & nz
    if mp.any():
        a[mp] += 1.0 - h[i[mp]]
    j = p + s
    mm = (j >= 0) & (j < n) & nz
    if mm.any():
        a[mm] -= h[j[mm]]

    x0 = float(h[p])
    caff_full = c[midx]
    bfull = caff_full - a * x0
    a = a[nz]
    b = bfull[nz]
    m_nz = midx[nz]
    C0 = float(c[m0] - x0 * (1.0 - x0))

    if const > high:
        high = const
    lo = const if const > C0 else C0
    if lo < 0.0:
        lo = 0.0
    hi = high
    if hi < lo:
        hi = lo + 1e-12

    pos = a > tol
    neg = a < -tol
    hasp = bool(pos.any())
    hasn = bool(neg.any())
    if hasp:
        pinv = 1.0 / a[pos]
        bpinv = b[pos] * pinv
    if hasn:
        ninv = -1.0 / a[neg]
        bninv = b[neg] * ninv

    qmax = C0 + 0.25
    for _ in range(iters):
        if hi - lo < 1e-8:
            break
        T = 0.5 * (lo + hi)
        L, U, ok = 0.0, 1.0, True
        if hasp:
            um = float(np.min(T * pinv - bpinv))
            if um < U:
                U = um
            if U < -1e-10:
                ok = False
        if ok and hasn:
            lm = float(np.max(bninv - T * ninv))
            if lm > L:
                L = lm
            if L > 1.0 + 1e-10:
                ok = False
        if ok and (const > T + 1e-10 or L > U + 1e-10):
            ok = False
        if ok:
            if T < C0 - 1e-10:
                ok = False
            elif T < qmax - 1e-11:
                dt = math.sqrt(max(0.0, qmax - T))
                gs, ge = 0.5 - dt, 0.5 + dt
                if not (L <= min(U, gs) or max(L, ge) <= U):
                    ok = False
        hi = T if ok else lo

    cands = []
    for Ttry in (hi, hi + 1e-9, hi + 1e-7, hi + 1e-5, high):
        L, U, ok = 0.0, 1.0, True
        if hasp:
            um = float(np.min(Ttry * pinv - bpinv))
            if um < U:
                U = um
            if U < -1e-9:
                ok = False
        if ok and hasn:
            lm = float(np.max(bninv - Ttry * ninv))
            if lm > L:
                L = lm
            if L > 1.0 + 1e-9:
                ok = False
        if ok and (const > Ttry + 1e-9 or L > U + 1e-9):
            ok = False
        if not ok:
            continue
        if Ttry >= qmax - 1e-10:
            l, u = max(0.0, L), min(1.0, U)
            if l <= u + 1e-12:
                cands.extend((float(l), float(u), float(np.clip(x0, l, u))))
            break
        if Ttry < C0 - 1e-9:
            continue
        dt = math.sqrt(max(0.0, qmax - Ttry))
        gs, ge = 0.5 - dt, 0.5 + dt
        l = max(0.0, L)
        u = min(1.0, max(0.0, gs))
        if l <= u + 1e-12:
            cands.extend((float(l), float(u), float(np.clip(x0, l, u))))
        l = max(0.0, L)
        u = min(1.0, U)
        if ge < 1.0:
            l2 = max(l, ge)
            if l2 <= u + 1e-12:
                cands.extend((float(l2), float(u), float(np.clip(x0, l2, u))))
        break

    cands.extend((float(x0), 0.0, 1.0))
    best_x, best_val = x0, 1e100
    for xc in cands:
        xc = float(np.clip(xc, 0.0, 1.0))
        val = const
        if a.size:
            mv = float(np.max(a * xc + b))
            if mv > val:
                val = mv
        qv = C0 + xc - xc * xc
        if qv > val:
            val = qv
        if val < best_val:
            best_val, best_x = val, xc

    if best_val < high - 1e-11 and abs(best_x - x0) > 1e-14:
        h[p] = best_x
        if a.size:
            c[m_nz] = a * best_x + b
        c[m0] = C0 + best_x - best_x * best_x
        return (x0, best_x)
    return None

def bundle_step(h, c, n, active_tol=0.25):
    cmax = float(c.max())
    active = np.where(c >= cmax - active_tol)[0]
    if active.size == 0:
        return h, c, False
    grad = np.zeros(n, dtype=float)
    for idx in active:
        l = int(idx) - (n - 1)
        if l > 0:
            grad[l:] += 1.0 - h[:-l]
            grad[:-l] -= h[l:]
        elif l < 0:
            d = -l
            grad[:-d] += 1.0 - h[d:]
            grad[d:] -= h[:-d]
        else:
            grad += 1.0 - 2.0 * h
    d = -grad
    d -= float(d.mean())
    pos = d > 1e-12
    neg = d < -1e-12
    if not (pos.any() or neg.any()):
        return h, c, False
    amax = 1e100
    if pos.any():
        amax = min(amax, float(((1.0 - h[pos]) / d[pos]).min()))
    if neg.any():
        amax = min(amax, float((h[neg] / (-d[neg])).min()))
    if amax <= 1e-10:
        return h, c, False
    for frac in (0.35, 0.175, 0.0875, 0.04375, 0.021875):
        h2 = np.clip(h + frac * amax * d, 0.0, 1.0)
        h2 = project_feasible(h2, n)
        c2 = correlate_raw(h2)
        if float(c2.max()) < cmax - 1e-10:
            return h2, c2, True
    return h, c, False

def cd_refine(start_h, n, deadline, seed=0, iters=22, sum_thr=0.7):
    h = project_feasible(np.array(start_h, dtype=float).copy(), n)
    c = correlate_raw(h)
    dx = 2.0 / n
    target = 0.5 * n
    cmax = float(c.max())
    best_c = cmax * dx
    best = h.copy()
    rng = np.random.RandomState(seed)
    stall, pass_no = 0, 0
    while time.time() < deadline:
        order = rng.permutation(n)
        improved, count, sum_h, cmax = 0, 0, float(h.sum()), float(c.max())
        for p in order:
            if time.time() > deadline - 0.4:
                break
            if count % 10 == 0:
                cmax = float(c.max())
            res = coord_update(h, c, n, int(p), cmax, iters)
            if res is not None:
                sum_h += res[1] - res[0]
                improved += 1
            count += 1
            if abs(sum_h - target) > sum_thr:
                h = project_feasible(h, n)
                c = correlate_raw(h)
                cmax = float(c.max())
                sum_h = float(h.sum())
                count = 0
        h = project_feasible(h, n)
        c = correlate_raw(h)
        cmax = float(c.max())
        if pass_no % 4 == 0 and time.time() < deadline - 1.0:
            h, c, impb = bundle_step(h, c, n, 0.25)
            if impb:
                improved += 1
                cmax = float(c.max())
        cval = cmax * dx
        if cval < best_c:
            best_c, best = cval, h.copy()
        stall = 0 if improved else stall + 1
        pass_no += 1
        if stall >= 2:
            break
    return best, best_c

def resample_h(h, n):
    m = len(h)
    if m == n:
        return project_feasible(h, n)
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    h2 = np.interp(xn, xo, h, left=float(h[0]), right=float(h[-1]))
    return project_feasible(h2, n)

def balanced_binary(h):
    n = len(h)
    b = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1]
    b[idx[:int(round(0.5 * n))]] = 1.0
    return project_feasible(b, n)

def reflect_sym(h):
    return project_feasible(0.5 * (h + (1.0 - h[::-1])), len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + budget_s - 8.0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 600
        h0 = np.full(600, 0.5, dtype=float)

    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = bound_of(best_h)
    iters = 22

    main_limit = min(450.0, max(30.0, 0.42 * budget_s))
    if hard - t0 > 600.0:
        dl1 = min(hard - 380.0, t0 + main_limit)
    else:
        dl1 = hard - 10.0
    if dl1 > time.time():
        h1, c1 = cd_refine(best_h, best_n, dl1, seed, iters, 0.7)
        if c1 < best_c:
            best_h, best_c = h1, c1

    cands = []
    if best_n < 750:
        cands.append((resample_h(best_h, 700), 700, 180.0, seed + 101, 0.7))
    if best_n > 450:
        cands.append((resample_h(best_h, 500), 500, 90.0, seed + 102, 0.7))
    cands.append((balanced_binary(best_h), best_n, 70.0, seed + 103, 0.3))
    cands.append((reflect_sym(best_h), best_n, 60.0, seed + 104, 0.5))
    rngp = np.random.RandomState(seed + 1234)
    for si, sg in enumerate((0.03, 0.10, 0.18)):
        hh = project_feasible(best_h + rngp.normal(0.0, sg, best_n), best_n)
        cands.append((hh, best_n, 50.0, seed + 105 + si, 0.5))

    for hh, nn, lim, se, thr in cands:
        rem = hard - time.time()
        take = min(lim, rem - 100.0)
        if take < 20.0:
            break
        if bound_of(hh) > best_c + 0.04:
            continue
        h2, c2 = cd_refine(hh, nn, time.time() + take, se, iters, thr)
        if c2 < best_c:
            best_h, best_n, best_c = h2, nn, c2

    if hard - time.time() > 20.0:
        h2, c2 = cd_refine(best_h, best_n, hard - 4.0, seed + 987, 24, 0.15)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = bound_of(best_h)
    return best_h, best_c, best_n
