```
def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for outer in range(60):
        diff=target-h.sum()
        if abs(diff) < 1e-12: break
        if diff > 0:
            room=1.0-h
            mask=room > 1e-14
            if not mask.any(): break
            tot=room[mask].sum()
            if tot <= diff:
                h[mask]=1.0
            else:
                h[mask]+=diff/tot*room[mask]
        else:
            mask=h > 1e-14
            if not mask.any(): break
            tot=h[mask].sum()
            if tot <= -diff:
                h[mask]=0.0
            else:
                h[mask]+=diff/tot*h[mask]
    h=np.clip(h,0.0,1.0)
    # final absorb
    for _ in range(30):
        diff=target-h.sum()
        if abs(diff) < 1e-12: break
        if diff > 0:
            idx=int(np.argmax(1.0-h))
            add=min(diff,1.0-h[idx])
            h[idx]+=add
        else:
            idx=int(np.argmax(h))
            sub=min(-diff,h[idx])
            h[idx]-=sub
    return h
```
If tot <= diff, set all room to 1, but if some room smaller, water fill may next iteration distribute to remaining. Good.

import numpy as np

def compute_c_norm(h, n):
    f = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h) * f))
def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = (1.0 - h) > 1e-14
            if not np.any(mask): break
            room = 1.0 - h
            tot = float(room[mask].sum())
            if tot <= diff: h[mask] = 1.0
            else: h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-14
            if not np.any(mask): break
            tot = float(h[mask].sum())
            if tot <= -diff: h[mask] = 0.0
            else: h[mask] += (diff / tot) * h[mask]
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12: break
        if diff > 0:
            idx = int(np.argmax(1.0 - h)); h[idx] += min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h)); h[idx] -= min(-diff, h[idx])
    return h

def compute_c_norm(h, n):

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = (1.0 - h) > 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h
            tot = float(room[mask].sum())
            if tot <= diff:
                h[mask] = 1.0
            else:
                h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            tot = float(h[mask].sum())
            if tot <= -diff:
                h[mask] = 0.0
            else:
                h[mask] += (diff / tot) * h[mask]
    h = np.clip(h, 0.0, 1.0)

def build_pre(n):

        low = (n - 1) - L
        high = (n - 1) + L

def coord_raw_diff(a, b, M, q0, f, t, r):

def downsample_h(h, m):
    n = len(h)
    if n % m == 0:
        blk = n // m
        return np.asarray(h, dtype=float).reshape(m, blk).mean(axis=1)
    xs = np.linspace(0.0, 1.0, n, endpoint=False)
    xn = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(xn, xs, np.asarray(h, dtype=float))

def upsample_h(h, m):
    n = len(h)
    xs = np.linspace(0.0, 1.0, n, endpoint=False)
    xn = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(xn, xs, np.asarray(h, dtype=float))

def lag_gradient(w, h, n):
    g = np.zeros(n, dtype=float)
    ws0 = float(w[n - 1])
    if ws0 > 1e-20:
        g += ws0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        length = n - s
        wp = float(w[n - 1 - s])
        if wp > 1e-20:
            g[:length] += wp * (1.0 - h[s:])
            g[s:] -= wp * h[:length]
        wn = float(w[n - 1 + s])
        if wn > 1e-20:
            g[s:] += wn * (1.0 - h[:length])
            g[:length] -= wn * h[s:]
    return g

def smooth_value(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x)
    m = float(np.max(c))
    z = alpha * (c - m)
    ez = np.exp(z)
    se = float(np.sum(ez))
    val = (2.0 / float(n)) * (m + np.log(se) / alpha)
    err = float(np.sum(x) - 0.5 * n)
    return val + pen * err * err

def smooth_grad(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x)
    m = float(np.max(c))
    z = alpha * (c - m)
    ez = np.exp(z)
    se = float(np.sum(ez))
    w = ez / se
    g = lag_gradient(w, x, n)
    g *= (2.0 / float(n))
    err = float(np.sum(x) - 0.5 * n)
    g += 2.0 * pen * err
    return g

                       bounds=[(0.0, 1.0)] * n,

        if res is not None and getattr(res, 'x', None) is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return x0
