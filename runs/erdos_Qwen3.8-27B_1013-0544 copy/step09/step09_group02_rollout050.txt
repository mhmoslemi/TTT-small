import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)

                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True

            c = np.correlate(h, 1.0 - h, mode='full')

def nesterov_subgrad(h, n, max_iter=5000, deadline=None, lr0=0.05):
    """Nesterov accelerated projected subgradient method."""
    h = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        # Subgradient of c_{k_star} w.r.t. h
        g = np.zeros(n)
        for i in range(n):
            j = i + k_star - (n - 1)
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
            ip = i - k_star + (n - 1)
            if 0 <= ip < n:
                g[i] -= h[ip]
        lr = lr0 / (1.0 + 0.01 * t)
        beta = (1.0 + 0.5 * np.sqrt(t + 1)) / (2.0 + 0.5 * np.sqrt(t + 1))
        v = beta * v + (1 - beta) * g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
    return best_h, best_c

        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb(h, n, strength, rng, method='noise'):
    """Apply a perturbation to h."""
    h_new = h.copy()
    if method == 'noise':
        mask = rng.random(n) < 0.3  # perturb 30% of coordinates
        h_new[mask] += rng.normal(0, strength, mask.sum())
    elif method == 'swap':
        n_swaps = max(1, int(n * 0.05))
        for _ in range(n_swaps):
            i, j = rng.randint(0, n, 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
    elif method == 'block':
        i = rng.randint(0, n // 2)
        length = rng.randint(n // 10, n // 3)
        length = min(length, n - i)
        h_new[i:i+length] = h_new[i:i+length][::-1]
    return project_feasible(h_new, n)
