def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n, prev_val=0.5):
    """Find optimal h[p] given all other coordinates fixed."""
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s

    h = h.copy()
    best_c = compute_c5(h, n)

        order = rng.permutation(n)
        improved = 0
        max_change = 0.0

            h0 = h[p]
            new_val = exact_coord_min(c, p, h, n, h0)

        if new_c < best_c:

        if improved == 0 or max_change < tol:

    return h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
