import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def coord_step(h, c, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo = max(0, n - 1 - p)
    hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(lo, hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    a[vf] += (1.0 - h[jf[vf]])
    a[vb] -= h[jb[vb]]
    c_aff = c[ms]
    diag_mask = (ms == n - 1)
    has_diag = np.any(diag_mask)
    unaff = 0.0
    if lo > 0:
        unaff = max(unaff, float(np.max(c[:lo])))
    if hi < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[hi + 1:])))
    Llo, Lhi = unaff, float(np.max(c))
    for _ in range(48):
        if Lhi - Llo < 1e-15:
            break
        T = (Llo + Lhi) / 2.0
        if unaff > T:
            Llo = T
            continue
        nd = ~diag_mask
        h_lo, h_hi = 0.0, 1.0
        if np.any(nd):
            a_nd, c_nd = a[nd], c_aff[nd]
            neg = a_nd < -1e-14
            pos = a_nd > 1e-14
            if np.any(neg):
                h_lo = max(h_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                h_hi = min(h_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            h_lo, h_hi = max(h_lo, 0.0), min(h_hi, 1.0)
        if has_diag:
            di = np.where(diag_mask)[0][0]
            c_ds = c_aff[di] - h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * (T - c_ds)
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if h_lo < r1:
                    h_hi = min(h_hi, r1)
                elif h_lo <= r2:
                    h_lo = max(h_lo, r2)
        if h_lo <= h_hi:
            Lhi = T
        else:
            Llo = T
    if Lhi >= float(np.max(c)):
        return h0
    T = Lhi
    h_lo, h_hi = 0.0, 1.0
    nd = ~diag_mask
    if np.any(nd):
        a_nd, c_nd = a[nd], c_aff[nd]
        neg = a_nd < -1e-14
        pos = a_nd > 1e-14
        if np.any(neg):
            h_lo = max(h_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            h_hi = min(h_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
        h_lo, h_hi = max(h_lo, 0.0), min(h_hi, 1.0)
    if has_diag:
        di = np.where(diag_mask)[0][0]
        c_ds = c_aff[di] - h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * (T - c_ds)
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h_lo < r1:
                h_hi = min(h_hi, r1)
            elif h_lo <= r2:
                h_lo = max(h_lo, r2)
    h_lo, h_hi = max(h_lo, 0.0), min(h_hi, 1.0)
    if h_lo > h_hi:
        return h0
    return (h_lo + h_hi) / 2.0

def coord_descend(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            new_val = coord_step(h, c, p, n)
            if abs(new_val - h[p]) > 1e-13:
                h[p] = new_val
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def topk_subgrad(h, n, steps, lr0, top_k, rng, deadline):
    h = h.copy()
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        # Get top-k active lags
        k_use = min(top_k, len(c))
        active = np.argsort(c)[-k_use:][::-1]
        grad = np.zeros(n)
        for p in active:
            for i in range(n):
                jf = i + p - (n - 1)
                jb = i - p + (n - 1)
                if 0 <= jf < n:
                    grad[i] += (1.0 - h[jf])
                if 0 <= jb < n:
                    grad[i] -= h[jb]
                if p == n - 1:
                    grad[i] += (1.0 - 2.0 * h[i])
        grad = grad - grad.mean()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h = project_feasible(h - lr0 * grad / grad_norm, n)
    return h

def smooth_slsqp(h, n, deadline):
    from scipy.optimize import minimize
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > deadline - 30:
            break
        def obj(x, a=alpha, n=n):
            cc = np.correlate(x, 1.0 - x, mode='full')
            m = cc.max()
            sh = a * (cc - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / a) * (2.0 / n)
        def con_fun(x, n=n):
            return x.sum() - n / 2.0
        def con_jac(x, n=n):
            return np.ones(n)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c
