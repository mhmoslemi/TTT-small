    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)

    c0_base = c[n - 1] - h0 * (1.0 - h0)

            break

        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)

def cd_with_refresh(h, n, max_passes=1000, refresh_interval=50, deadline=None, seed=0, priority=True):
    """Coordinate descent with periodic correlation refresh and priority ordering."""
    rng = np.random.RandomState(seed)
    h = h.copy()

        updates_since_refresh = 0

            binary_mask = (h < 0.01) | (h > 0.99)
            non_binary = np.where(~binary_mask)[0]
            if len(non_binary) > 0:

                order = rng.permutation(non_binary)
                binary_coords = np.where(binary_mask)[0]
                order = np.concatenate([order, rng.permutation(binary_coords)])
            else:
                order = rng.permutation(n)
        else:
            order = rng.permutation(n)
        
        for p in order:
            if deadline is not None and time.time() > deadline:
                break

                continue

        if new_c < best_c:
            best_c = new_c
        c = np.correlate(h, 1.0 - h, mode='full')
        if improved == 0:

    return h, best_c

    critical = get_critical_coords(h, n)
    if len(critical) == 0:
        return h, compute_c5(h, n)

    scores = np.array([min(h[p], 1-h[p]) for p in critical])
    critical = np.array(critical)
    if len(critical) > block_size:
        top_idx = np.argsort(scores)[-block_size:]
        critical = critical[top_idx]
    
    h_sub = h[critical].copy()
    
    def obj(x):
        h_temp = h.copy()
        h_temp[critical] = x
        return compute_c5(h_temp, n)
    
    bounds = [(0.0, 1.0)] * len(critical)
    cons = [{'type': 'ineq', 'fun': lambda x: (n/2.0 - h.sum() + x.sum()) + 1e-10, 'jac': lambda x: np.ones(len(x))}]
    # Actually use equality constraint
    base_sum = h.sum() - h_sub.sum()
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() + base_sum - n/2.0, 'jac': lambda x: np.ones(len(x))}]

                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = h.copy()
        h_new[critical] = np.clip(res.x, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def smooth_max_loss(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def global_slsqp(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(len(x))}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x.copy(), n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

            ho, co = cd_refresh(hi, n_try, 3000, 25, deadline - 80, seed + n_try, True)
            if co < best_c:
                best_c = co
                best_h = project_feasible(np.interp(x_orig, x_new, ho), n)
                x_orig = x_new

    # Phase 5: Harmonic kicks
    n_cur = len(best_h)
    x = np.linspace(0, 2, n_cur)
    for r in range(14):
        if time.time() > deadline - 60:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        ks = abs(int(np.argmax(c)) - (n_cur - 1)) + 1
        amp = 0.015 + 0.015 * r
        ph = rng.uniform(0, 2 * np.pi)
        hk = project_feasible(best_h + amp * np.sin(2 * np.pi * ks * x / 2.0 + ph), n_cur)
        ho, co = cd_refresh(hk, n_cur, 2000, 25, deadline - 35, seed + 2000 + r * 11, True)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 6: Diverse initializations
    inits = []
    for period in [3, 5, 7, 11]:
        pat = np.zeros(n_cur)
        pat[:n_cur // 2] = 1.0
        rep = int(np.ceil(n_cur / period))
        full = np.tile(pat[:period], rep)[:n_cur]
        inits.append(project_feasible(full, n_cur))
    for r, hi in enumerate(inits):
        if time.time() > deadline - 40:
            break
        ho, co = cd_refresh(hi, n_cur, 3000, 25, deadline - 20, seed + 5000 + r * 7, True)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 7: Small perturbations
    for r in range(6):
        if time.time() > deadline - 25:
            break
        s = 0.005 + 0.004 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        ho, co = cd_refresh(hp, n_cur, 1500, 20, deadline - 10, seed + 3000 + r * 3, False)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 8: Final polish
    if time.time() < deadline - 5:
        hf, cf = cd_refresh(best_h, n_cur, 10000, 15, deadline - 3, seed + 9999, True)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
