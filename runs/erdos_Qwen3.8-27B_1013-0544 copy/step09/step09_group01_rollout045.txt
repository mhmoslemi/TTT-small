    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()

def compute_gradients(h, n):
    """Compute the gradient of max_k c_k at current h.
    Returns subgradient (using max component) or a smoothed version."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Subgradient: contribution of each h_i to c_k for k = argmax
    m = c.argmax()
    # c_m = sum_i h_i * (1-h)_{i-(m-(n-1))}
    # d c_m / d h_p = (1-h_{p-(m-(n-1))}) - h_{p+(m-(n-1))} (with 0 for OOB)
    shift = m - (n - 1)
    grad = np.zeros(n)
    # Contribution from h_p * (1-h)_{p-shift}
    idx_f = np.arange(n)
    jf = idx_f - shift
    valid_f = (jf >= 0) & (jf < n)
    grad[valid_f] += (1.0 - h[jf[valid_f]])
    # Contribution from h_{p+shift} * (1-h)_{p} -> d/dh_p of h_{p+shift}*(1-h_p)
    # = -h_{p+shift}
    jb = idx_f + shift
    valid_b = (jb >= 0) & (jb < n)
    grad[valid_b] -= h[jb[valid_b]]
    return grad, c, m

def subgradient_descent(h, n, n_iters, deadline):
    """Mirror descent / subgradient method for max of bilinear forms."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad, c, _ = compute_gradients(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
        # Decay learning rate
        lr = max(0.001, lr * 0.999)
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d):
    if has_diag:
        c_aff = c_aff.copy()
        c_aff[diag_idx] += -d * d
    vals = c_aff + d * a
    result = float(np.max(vals))
    if unaff > result:
        result = unaff
    return result

def cd_optimize(h, n, deadline, gs_iters=55, seed=42, max_passes=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng = np.random.RandomState(seed)
    staleness = 0

    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if pass_num % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = 0
        for idx in order:
            p = int(idx)
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p - (ms - (n - 1))
            jb = p + (ms - (n - 1))
            a = np.zeros(len(ms))
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_pos = ms - (n - 1)
            has_diag = np.any(diag_pos == 0)
            diag_idx = int(np.where(diag_pos == 0)[0][0]) if has_diag else -1
            c_aff = c[ms].copy()
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            if d_hi - d_lo < 1e-14:
                continue
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
            f2 = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-13:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, d_opt)
            f_curr = _fval(c_aff, a, diag_idx, has_diag, unaff, h0, 0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved += 1

        if improved == 0:
            staleness += 1
            if staleness > 3:
                break
        else:
            staleness = 0
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

def structured_init(n, rng):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    freq = rng.choice([1, 2, 3, 4])
    amp = rng.uniform(0.1, 0.3)
    h += amp * np.sin(np.pi * freq * x) + amp * 0.5 * np.sin(2 * np.pi * freq * x)
    h = project_feasible(h, n)
    for _ in range(5):
        h += rng.normal(0, 0.05, n)
        h = project_feasible(h, n)
    return h

def multi_start_cd(h0, n, deadline, seed=42, num_starts=4):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    for s in range(num_starts):
        if time.time() > deadline:
            break
        if s == 0:
            hp = h0.copy()
        elif s % 3 == 1:
            hp = project_feasible(h0 + rng.normal(0, 0.02, n), n)
        elif s % 3 == 2:
            x = np.linspace(0, 2, n)
            freq = rng.uniform(2, 8)
            amp = rng.uniform(0.02, 0.06)
            hp = project_feasible(h0 + amp * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi)), n)
        else:
            hp = project_feasible(h0 + rng.normal(0, 0.04, n), n)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) / (num_starts - s))
        h_o, c_o = cd_optimize(hp, n, sub_deadline, gs_iters=50, seed=seed + s * 1000)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    return best_h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    # Phase 3: Try alternative grid sizes
    phase3_deadline = t0 + (budget_s - 3) * 0.90
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 600, 800]:
        if time.time() > phase3_deadline:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        sub_dl = min(phase3_deadline, deadline - 30)
        h_o, c_o = cd_optimize(h_int, n_try, sub_dl, gs_iters=50, seed=seed + n_try)
        if c_o < best_c:

            x_orig = x_new

    h_f, c_f = cd_optimize(best_h, n_cur, deadline, gs_iters=70, seed=seed + 9999)
    if c_f < best_c:
        best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
