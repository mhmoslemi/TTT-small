import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_corrs(h, n):
    """Return all correlation values (not multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def subgradient_step(h, n, lr, momentum_buf=None, mu=0.9):
    """One subgradient step with optional momentum."""
    corrs = compute_all_corrs(h, n)
    k_star = np.argmax(corrs)
    # Gradient of R(k_star) w.r.t. h
    # R(k) = sum_i h[i] * (1-h[i+k])
    # In numpy correlate full mode, index k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    # dR/dh[p] = (1-h[p+lag]) - h[p-lag] (with boundary handling)
    # First term: (1-h[p+lag]) for valid p+lag
    if lag >= 0:
        # p+lag < n => p < n-lag
        valid = np.arange(0, n - lag)
        grad[valid] += (1.0 - h[valid + lag])
    else:
        # p+lag >= 0 => p >= -lag
        valid = np.arange(-lag, n)
        grad[valid] += (1.0 - h[valid + lag])
    # Second term: -h[p-lag] for valid p-lag
    if lag >= 0:
        # p-lag >= 0 => p >= lag
        valid = np.arange(lag, n)
        grad[valid] -= h[valid - lag]
    else:
        # p-lag < n => p < n-lag = n+|lag|
        valid = np.arange(0, n + (-lag))
        grad[valid] -= h[valid - lag]
    # Clip gradient to feasible region (projected gradient)
    if momentum_buf is None:
        momentum_buf = np.zeros(n)
    momentum_buf = mu * momentum_buf - lr * grad
    h_new = h + momentum_buf
    h_new = project_feasible(h_new, n)
    return h_new, corrs.max() * (2.0/n), momentum_buf

def subgradient_method(h, n, deadline, max_iters=50000):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum_buf = np.zeros(n)
    lr = 0.01
    mu = 0.7
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new, c_val, momentum_buf = subgradient_step(h, n, lr, momentum_buf, mu)
        if c_val < best_c:
            best_c = c_val
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        h = h_new
        # Adaptive learning rate

            momentum_buf *= 0.5
        elif c_val < best_c * 0.99:

        # Decay momentum
        if it % 1000 == 0:
            mu = 0.7 + 0.2 * (it / max(it, 1))
        if stagnation > 5000:
            break
    return best_h, best_c

def smooth_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
    # Gradient of sum_k weights[k] * R(k) w.r.t. h
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        w = weights[k]
        if lag >= 0:
            valid = np.arange(0, n - lag)
            grad[valid] += w * (1.0 - x[valid + lag])
            valid2 = np.arange(lag, n)
            grad[valid2] -= w * x[valid2 - lag]
        else:
            valid = np.arange(-lag, n)
            grad[valid] += w * (1.0 - x[valid + lag])
            valid2 = np.arange(0, n + (-lag))
            grad[valid2] -= w * x[valid2 - lag]
    return grad * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def l_bfgs_optimize(h, n, alpha, maxiter=300, deadline=None):
    """L-BFGS-B optimization of the smoothed objective."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_objective, h, args=(n, alpha),
                       jac=lambda x: smooth_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

        for p in order:

            m_lo = max(0, n - 1 - p)

            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):

            if np.any(valid_b):

            diag_pos = np.where(ms == (n - 1))[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    s = weights.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights /= s
    grad = np.zeros(n)
    omx = 1.0 - x
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (omx - x)
        elif lag > 0:
            grad[:n - lag] += w * omx[lag:]
            grad[lag:] -= w * x[:n - lag]
        else:
            lag = -lag
            grad[lag:] += w * omx[:n - lag]
            grad[:n - lag] -= w * x[lag:]
    return grad * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def momentum_subgrad(h, n, deadline, max_iters=30000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    buf = np.zeros(n)
    lr = 0.02
    mu = 0.8
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag == 0:
            grad = (1.0 - h) - h
        elif lag > 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            lag2 = -lag
            grad[lag2:] += (1.0 - h[:n - lag2])
            grad[:n - lag2] -= h[lag2:]
        t += 1
        lr_t = 0.02 / (1.0 + 0.01 * t)
        buf = mu * buf - lr_t * grad
        h_new = project_feasible(h + buf, n)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * (2.0 / n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 5000 == 0 and it > 0:
            mu = min(0.95, 0.7 + 0.05 * (it / 5000.0))
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng_local = np.random.RandomState(12345)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng_local.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == n - 1)[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0

            def fval(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
                r = float(np.max(vals))
                return max(r, unaff_max)

            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1, f2 = fval(d1), fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:

                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fval(d2)
            d_opt = (d_lo + d_hi) / 2.0
            if fval(d_opt) < fval(0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
