Let me also think about what the optimal solution looks like. For the Erdős minimum overlap problem, the optimal h is known to be close to a "two-level" function for large n. That is, h takes values close to a and 1-a for some a, with the fraction of each level determined by the integral constraint.

If h takes two values a (with fraction p) and b (with fraction 1-p), then:
- Integral: p*a + (1-p)*b = 1/2 (since integral over [0,2] is 1, so average is 1/2)
- c_0 = p*a*(1-a) + (1-p)*b*(1-b) = p*a - p*a^2 + (1-p)*b - (1-p)*b^2
  = p*a + (1-p)*b - (p*a^2 + (1-p)*b^2)
  = 1/2 - (p*a^2 + (1-p)*b^2)

To minimize c_0, we want to maximize p*a^2 + (1-p)*b^2, which is maximized when {a,b} = {0,1} and p = 1/2.
So c_0 = 1/2 - 1/2 = 0 for the binary case.

For other lags, the two-level structure gives c_k = (number of pairs at distance k where levels differ) * (contribution per pair).

For a two-level function with values a and b at fraction p and 1-p:
c_k = sum_j h[j](1-h[j+d]) = sum_j [p*a*(1-b) + (1-p)*b*(1-a)] * (indicator that j and j+d are at different levels)
    = [p(1-p)*a(1-b) + p(1-p)*b(1-a)] * (fraction of cross-level pairs at distance d)
    = p(1-p)[a+b-2ab] * (fraction of cross-level pairs)

            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha):
    """Compute smooth max of c_k and its gradient."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    shifted = alpha * c
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    val = (mx + np.log(w_sum)) / alpha
    w_norm = w / w_sum
    
    # Gradient computation (vectorized)
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    A_idx = j_idx + (n - 1) - i_idx
    B_idx = (n - 1) + i_idx - j_idx
    A = (w_norm[A_idx] * (1.0 - x)[None, :]).sum(axis=1)
    B = (w_norm[B_idx] * x[None, :]).sum(axis=1)
    grad = dx * (A - B)
    return val, grad

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    """Run SLSQP with smooth max objective for various alpha values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        bounds = [(0.0, 1.0)] * n
        
        def obj(x):
            v, _ = smooth_obj_and_grad(x, n, alpha)
            return v
        
        def obj_grad(x):
            v, g = smooth_obj_and_grad(x, n, alpha)
            return v, g
        
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(obj_grad, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    shifted = alpha * c
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return lse / alpha

def smooth_obj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    shifted = alpha * c
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    val = (mx + np.log(w_sum)) / alpha
    w_norm = w / w_sum
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    A_idx = j_idx + (n - 1) - i_idx
    B_idx = (n - 1) + i_idx - j_idx
    A = (w_norm[A_idx] * (1.0 - x)[None, :]).sum(axis=1)
    B = (w_norm[B_idx] * x[None, :]).sum(axis=1)
    grad = dx * (A - B)
    s = x.sum() - n / 2.0
    val += mu * s * s
    grad += 2.0 * mu * s
    return val, grad

def lbfgs_opt(h, n, alphas, mu, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj_grad, best_h, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds, jac=True,
                          options={'maxiter': maxiter_each, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 80})
            new_h = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def slsqp_opt(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_obj(x, n, alpha)
        def objg(x):
            v, g = smooth_obj_grad(x, n, alpha, 0.0)
            return v, g
        try:
            res = minimize(objg, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=True,
                          options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo = 0
    hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(lo, hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_mask = ms == (n - 1)
    has_diag = diag_mask.any()
    c_diag = c_base[diag_mask][0] if has_diag else 0.0
    if has_diag:
        a[diag_mask] = 0.0

    a_nd = a[nd]
    c_nd = c_base[nd]

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
