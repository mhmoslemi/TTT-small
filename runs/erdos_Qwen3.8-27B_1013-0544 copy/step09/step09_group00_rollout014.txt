   **Alternative: Lagrangian relaxation of the max constraint**
   
   The problem is: min_h max_k c_k(h) subject to constraints.
   
   We can write: min_h { t : c_k(h) ≤ t ∀k, constraints }
   
   The Lagrangian: L(h, λ, t) = t + ∑_k λ_k (c_k(h) - t) subject to ∑λ_k = 1, λ_k ≥ 0
   
   = t(1-∑λ_k) + ∑_k λ_k c_k(h)
   = ∑_k λ_k c_k(h) (when ∑λ_k = 1)
   
   So the dual is: max_{λ ≥ 0, ∑λ=1} min_h ∑_k λ_k c_k(h) subject to h constraints.

    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()

    mx = shifted.max()

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Heavy CD on initial (the main workhorse)
    h1, c1 = exact_cd(h_best, n, 800, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP polish with increasing smoothing
    if time.time() < deadline - 600:
        h2, c2 = slsqp_opt(h_best, n, [20000, 100000, 500000], 400, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2
        # CD after SLSQP
        h2b, c2b = exact_cd(h_best, n, 500, deadline)
        if c2b < best_c:
            h_best, best_c = h2b, c2b

    # Phase 3: Try different n values
    n_list = [400, 500, 600, 350, 450, 550]
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best, period=2), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 400, deadline)
        if c_o < best_c:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 4: Random restarts with CD
    for r in range(12):
        if time.time() > deadline - 80:
            break
        s = 0.005 + 0.01 * r
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        h_o, c_o = exact_cd(hp, len(h_best), 300, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Subgradient + CD cycles
    n_cur = len(h_best)
    for cycle in range(20):
        if time.time() > deadline - 40:
            break
        alpha_vals = [50, 100, 200, 500]
        for alpha in alpha_vals:
            if time.time() > deadline - 30:
                break
            for step in range(3):
                if time.time() > deadline - 25:
                    break
                h_sg = subgradient_step(h_best, n_cur, alpha, deadline,
                                        scale=0.002)
                h_o, c_o = exact_cd(h_sg, n_cur, 200, deadline)
                if c_o < best_c:
                    h_best, best_c = h_o, c_o

    # Phase 6: Final aggressive CD
    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 1000, deadline)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
