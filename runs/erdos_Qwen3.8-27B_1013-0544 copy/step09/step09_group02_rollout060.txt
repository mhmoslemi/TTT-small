            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_shifted = alpha * (c - c.max())
    lse = c.max() + np.log(np.sum(np.exp(c_shifted))) / alpha
    return lse * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    c_shifted = alpha * (c - c_max)
    w = np.exp(c_shifted)
    w /= w.sum()
    dx = 2.0 / n
    p = np.arange(n)
    q = np.arange(n)
    idx1 = p[:, None] - q[None, :] + n - 1
    idx2 = idx1.T  # = q[:, None] - p[None, :] + n - 1
    return dx * ((w[idx1] * (1.0 - x)).sum(axis=1) - (w[idx2] * x).sum(axis=1))

def slsqp_analytic(h, n, alpha, maxiter=2000, deadline=None):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return lse_obj(x, n, alpha)
    def jac(x):
        return lse_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        return new_h, new_c
    except Exception:
        return h, compute_c5(h, n)

def variance_obj(x, n):

    return float(np.var(c))

Wait, I need to double-check the variance gradient. The variance of c is:
var(c) = (1/M) Σ_k (c_k - mean_c)^2 where M = 2n-1

d(var)/d(c_k) = (2/M)(c_k - mean_c) - (2/M) * (1/M) * Σ_j (2(c_j - mean_c)) * (d mean_c / d c_k)
= (2/M)(c_k - mean_c) - (2/M^2) * Σ_j (c_j - mean_c)
= (2/M)(c_k - mean_c) - 0  (since Σ(c_j - mean_c) = 0)
= (2/M)(c_k - mean_c)

So d(var)/d(c_k) = (2/M)(c_k - mean_c) where M = 2n-1.

And the full gradient:

This is the same form as the lse_grad but with w_k = (2/M)(c_k - mean_c) instead of the softmax weights.
