**Better 1D solver**: Instead of bisection, use the fact that we're finding the minimum of a piecewise-linear convex function (in the 1D subproblem). The breakpoints are where the "active" constraint changes. We can find these breakpoints directly.

The 1D subproblem is: minimize max_k (a_k * x + b_k) subject to 0 ≤ x ≤ 1, where a_k and b_k are known.

The function f(x) = max_k (a_k * x + b_k) is convex and piecewise linear. Its minimum is at a point where two or more of the linear pieces are simultaneously at the maximum (a "crossing point"), or at a boundary (x=0 or x=1).

To find the minimum:
1. For each pair of k, l: find the crossing point x = (b_l - b_k) / (a_k - a_l) (if a_k ≠ a_l)
2. Check if x ∈ [0, 1] and if at that x, the max is indeed achieved by k and l
3. Also check x=0 and x=1
4. Return the x that gives the smallest max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def cd_pass(h, c, order, n):
    improved = 0
    for p in order:
        h0 = h[p]
        if h0 < 1e-11:
            continue
        if h0 > 1.0 - 1e-11:
            continue
        new_val = exact_coord_min(c, p, h, n)
        if abs(new_val - h0) > 1e-11:
            h[p] = new_val
            improved += 1
    return h, improved

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    no_improve_count = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        h, improved = cd_pass(h, c, order, n)
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            no_improve_count = 0
        else:
            no_improve_count += 1
            if improved == 0 and no_improve_count >= 3:
                break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-19})
            new_h = project_feasible(res.x.copy(), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_init(n, mode, seed=0):
    x = np.linspace(0, 2, n)
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = (np.sin(np.pi * x) ** 2 * 0.8 + 0.1)
    elif mode == 2:
        h = (0.5 + 0.4 * np.sin(2 * np.pi * x))
    elif mode == 3:
        h = (x < 1.0).astype(float)
    elif mode == 4:
        h = ((x % 0.5) < 0.25).astype(float)
    elif mode == 5:
        h = (0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.1 * np.sin(7 * np.pi * x))
    else:
        h = np.full(n, 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)
