```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def pm_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    c = np.maximum(c, 0.0)
    return np.sum(c ** p) / len(c)

def pm_grad(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = len(c)
    cpm1 = np.power(np.maximum(c, 1e-30), p - 1)
    omx = 1.0 - x
    grad = np.zeros(n)
    for j in range(n):
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j + n - 1)
        ir = np.arange(i_lo, i_hi + 1)
        k1 = ir - j + (n - 1)
        k2 = j - ir + (n - 1)
        grad[j] = (p / m) * (np.dot(cpm1[k1], omx[ir]) - np.dot(cpm1[k2], x[ir]))
    return grad

def pm_con_fun(x, n):
    return x.sum() - n / 2.0

def pm_con_jac(x, n):
    return np.ones(n)

def lbfgs_round(h, n, p, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': pm_con_fun, 'jac': pm_con_jac, 'args': (n,)}]
    try:
        res = minimize(pm_obj, h, args=(n, p), jac=lambda x: pm_grad(x, n, p),
                       method='L-BFGS-B', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    k_lo = max(0, n - 1 - p, p)
    k_hi = min(2 * n - 2, n - 1 + p, p + n - 1)
    ks = np.arange(k_lo, k_hi + 1)
    idx1 = p + ks - (n - 1)
    idx2 = p - ks + (n - 1)
    D = np.zeros(len(ks))
    v1 = (idx1 >= 0) & (idx1 < n)
    D[v1] += (1.0 - h[idx1[v1]])
    v2 = (idx2 >= 0) & (idx2 < n)
    D[v2] -= h[idx2[v2]]
    
    # Find optimal x to minimize max_k (c[k] + (x-h0)*D[k])
    # = minimize max_k (A[k] + x*D[k]) where A[k] = c[k] - h0*D[k]
    A = c[k_lo:k_hi+1] - h0 * D
    
    # Unaffected max
    u_max = 0.0
    if k_lo > 0:
        u_max = max(u_max, float(c[:k_lo].max()))
    if k_hi < 2*n - 2:
        u_max = max(u_max, float(c[k_hi+1:].max()))
    
    cur_max = float(c.max())
    Llo, Lhi = max(u_max, 0.0), cur_max
    for _ in range(50):
        if Lhi - Llo < 1e-14:
            break
        T = (Llo + Lhi) / 2.0
        if u_max > T + 1e-15:
            Llo = T
            continue
        # Need A[k] + x*D[k] <= T for all k
        # D>0: x <= (T-A[k])/D[k]
        # D<0: x >= (T-A[k])/D[k]
        # D=0: need A[k] <= T
        pos = D > 1e-13
        neg = D < -1e-13
        zero = ~pos & ~neg
        if np.any(zero) and A[zero].max() > T + 1e-14:
            Llo = T
            continue
        xl, xh = 0.0, 1.0
        if np.any(neg):
            xl = max(xl, float(np.max((T - A[neg]) / D[neg])))
        if np.any(pos):
            xh = min(xh, float(np.min((T - A[pos]) / D[pos])))
        if xl > xh + 1e-13:
            Llo = T
        else:
            Lhi = T
    
    T = (Llo + Lhi) / 2.0
    pos = D > 1e-13
    neg = D < -1e-13
    xl, xh = 0.0, 1.0
    if np.any(neg):
        xl = max(xl, float(np.max((T - A[neg]) / D[neg])))
    if np.any(pos):
        xh = min(xh, float(np.min((T - A[pos]) / D[pos])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if xl > xh + 1e-10:
        return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if rng is not None else range(n)
        for p in order:
            if time.time() > deadline:
                break
            nv = solve_coord(c, h, p, n)
            if abs(nv - h[p]) > 1e-13:
                h[p] = nv
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h
```

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)

        grad[j] = (2.0 / n) * (np.dot(w[k1], 1.0 - x[ir]) - np.dot(w[k2], x[ir]))
    return grad

def lse_con(x, n):
    return x.sum() - n / 2.0

def lse_con_j(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lse_con, 'jac': lse_con_j, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h, args=(n, alpha), jac=lambda x: lse_grad(x, n, alpha),
                       method='L-BFGS-B', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    k_lo = max(0, n - 1 - p, p)
    k_hi = min(2 * n - 2, n - 1 + p, p + n - 1)
    ks = np.arange(k_lo, k_hi + 1)
    idx1 = p + ks - (n - 1)
    idx2 = p - ks + (n - 1)
    D = np.zeros(len(ks))
    v1 = (idx1 >= 0) & (idx1 < n)
    D[v1] += (1.0 - h[idx1[v1]])
    v2 = (idx2 >= 0) & (idx2 < n)
    D[v2] -= h[idx2[v2]]
    
    cur_at_p = c[p + ks - (n-1)]  # Not needed, let me just use c directly
    # A[k] = c[k] - h0 * D[k] for k in [k_lo, k_hi]
    A = c[k_lo:k_hi + 1].copy()
    # The actual c values at those k are c[k_lo:k_hi+1]
    # When x[p] changes from h0 to x, c[k] changes to c[k] + (x-h0)*D[k]
    # So new c[k] = c[k] + (x-h0)*D[k] = (c[k] - h0*D[k]) + x*D[k] = A[k] + x*D[k]
    
    # Unaffected max
    u_max = 0.0
    if k_lo > 0:
        u_max = max(u_max, float(c[:k_lo].max()))
    if k_hi < 2 * n - 2:
        u_max = max(u_max, float(c[k_hi + 1:].max()))
    
    cur_max = float(c.max())
    Llo, Lhi = max(u_max, 0.0), cur_max
    for _ in range(48):
        if Lhi - Llo < 1e-15:
            break
        T = (Llo + Lhi) / 2.0
        if u_max > T + 1e-15:
            Llo = T
            continue
        pos = D > 1e-13
        neg = D < -1e-13
        zero = ~(pos | neg)
        if np.any(zero) and float(A[zero].max()) > T + 1e-14:
            Llo = T
            continue
        xl, xh = 0.0, 1.0
        if np.any(neg):
            xl = max(xl, float(np.max((T - A[neg]) / D[neg])))
        if np.any(pos):
            xh = min(xh, float(np.min((T - A[pos]) / D[pos])))
        if xl > xh + 1e-13:
            Llo = T
        else:
            Lhi = T

    pos = D > 1e-13
    neg = D < -1e-13

    if np.any(neg):
        xl = max(xl, float(np.max((T - A[neg]) / D[neg])))
    if np.any(pos):
        xh = min(xh, float(np.min((T - A[pos]) / D[pos])))
