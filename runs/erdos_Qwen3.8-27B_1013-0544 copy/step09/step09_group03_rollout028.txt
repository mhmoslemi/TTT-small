                        coords.add(i)
                        coords.add(j)
            order = np.array(list(coords)) if coords else np.arange(n)
            rest = np.setdiff1d(np.arange(n), order)
            order = np.concatenate([order, rest])
            rng.shuffle(order)
            order = np.concatenate([order[np.argsort(-np.abs(np.cumsum(rng.randn(len(order))))[:len(order)])], np.arange(n)][:n])
            order = rng.permutation(n)
        else:
            order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = coord_min_v2(c, p, h, n)
            if abs(new_val - h0) > 1e-12:
                h[p] = new_val
                # Update c incrementally for affected shifts
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                delta = new_val - h0
                for m in range(m_lo, m_hi + 1):
                    j1 = p + m - n + 1
                    j2 = p - m + n - 1
                    s = 0.0
                    if 0 <= j1 < n and j1 != p:
                        s += (1.0 - h[j1])
                    if 0 <= j2 < n and j2 != p:
                        s -= h[j2]
                    if m != n - 1:
                        c[m] += delta * s
                    else:
                        c[m] += delta * (1.0 - 2.0 * new_val)
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0 and it > 5:
            break
    return h, best_c

def pair_update(h, n, deadline=None):
    """Try 2D grid search on adjacent pairs."""
    h = h.copy()
    best_c = compute_c5(h, n)
    grid_pts = np.linspace(0.0, 1.0, 21)
    for it in range(50):
        if deadline is not None and time.time() > deadline:
            break
        any_improved = False
        for p in range(n):
            q = (p + 1) % n
            h0p, h0q = h[p], h[q]
            if (h0p < 0.01 or h0p > 0.99) and (h0q < 0.01 or h0q > 0.99):
                continue
            c = np.correlate(h, 1.0 - h, mode='full')
            best_val = (h0p, h0q)
            best_f = compute_c5(h, n)
            # Local grid around current values
            idx_p = int(np.argmin(np.abs(grid_pts - h0p)))
            idx_q = int(np.argmin(np.abs(grid_pts - h0q)))
            for di in range(-3, 4):
                for dj in range(-3, 4):
                    xp = grid_pts[min(max(idx_p + di, 0), 20)]
                    xq = grid_pts[min(max(idx_q + dj, 0), 20)]
                    h[p], h[q] = xp, xq
                    f = compute_c5(h, n)
                    if f < best_f - 1e-14:
                        best_f = f
                        best_val = (xp, xq)
            h[p], h[q] = best_val
            if best_val != (h0p, h0q):
                any_improved = True
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif not any_improved:
            break
    return h, best_c

def smooth_lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_lse_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(2 * n - 1)
    # dc/dx_k for each shift m
    # c[m] = sum_i x[i]*(1-x[i+m-n+1])
    # dc[m]/dx[p] = (1-x[p+m-n+1]) - x[p-m+n-1] (approx, for valid indices)
    # This is complex; use numerical gradient for simplicity
    eps = 1e-7
    jac = np.zeros(n)
    # For speed, compute analytically
    for p in range(n):
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, n - 1 + p)
        for m in range(m_lo, m_hi + 1):
            j1 = p + m - n + 1
            j2 = p - m + n - 1
            dcmdx = 0.0
            if 0 <= j1 < n:
                dcmdx += (1.0 - x[j1])
            if 0 <= j2 < n and j2 != p:
                dcmdx -= x[j2]
            grad[m] += dcmdx * weights[m]
    jac[p] = grad[p]
    return jac * (2.0 / n)

def lbfgs_phase(h, n, deadline, seed):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_cur = len(best_h)
    alphas = [500, 2000, 10000, 50000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n_cur
        cons = [{'type': 'eq', 'fun': lambda x, n=n_cur: x.sum() - n_cur/2.0}]
        try:
            res = minimize(smooth_lse_obj, best_h.copy(), args=(n_cur, alpha),
                           method='L-BFGS-B', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-10})
            new_h = project_feasible(res.x.copy(), n_cur)
            new_c = compute_c5(new_h, n_cur)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Aggressive coordinate descent
    t1 = deadline - (budget_s * 0.35)
    h1, c1 = coord_descent(best_h, n, 5000, t1, seed, 'random')
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: More CD with shift-aware
    t2 = deadline - (budget_s * 0.25)
    h2, c2 = coord_descent(best_h, n, 3000, t2, seed + 1, 'shift_aware')
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Pair updates
    t3 = deadline - (budget_s * 0.15)
    h3, c3 = pair_update(best_h, n, t3)
    if c3 < best_c:
        best_h, best_c = h3, c2 if c2 < c3 else c3
        if c3 < best_c:
            best_c = c3

    # Phase 4: L-BFGS-B
    t4 = deadline - (budget_s * 0.15)
    h4, c4 = lbfgs_phase(best_h, n, t4, seed)
    if c4 < best_c:
        best_h, best_c = h4, c4
    # CD after L-BFGS
    h4b, c4b = coord_descent(best_h, n, 2000, t4, seed + 100)
    if c4b < best_c:
        best_h, best_c = h4b, c4b

    # Phase 5: Try nearby n values
    x_orig = np.linspace(0, 2, len(best_h))
    n_trials = [len(best_h) - 2, len(best_h) - 1, len(best_h) + 1, len(best_h) + 2,
                len(best_h) - 10, len(best_h) + 10]
    t5_end = deadline - (budget_s * 0.10)
    for n_try in n_trials:
        if time.time() > t5_end or n_try < 100:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 2000, t5_end, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Multi-restart with perturbations
    n_cur = len(best_h)
    t6_end = deadline - 30
    for r in range(20):
        if time.time() > t6_end:
            break
        sigma = 0.002 + 0.003 * r
        hp = project_feasible(best_h + rng.normal(0, sigma, n_cur), n_cur)
        ho, co = coord_descent(hp, n_cur, 800, t6_end, seed + 1000 + r * 13)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 7: Final polish
    hf, cf = coord_descent(best_h, len(best_h), 5000, deadline - 2, seed + 9999)
    if cf < best_c:
        best_h, best_c = hf, cf

    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
