def lse_objective(h, n, alpha):

def solve_1d_minimax_fast(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    di = diag_idx
    for i in range(T):
        for j in range(i + 1, T):
            if i == di or j == di:

            if abs(da) > 1e-15:

            C = -(c_peak[di] - c_peak[i])

        v = max(float(vals.max()), unaff_max)

            best_d = d
    return best_d, best_val

def cd_pass(h, n, top_k, deadline, order):
    """Single CD pass. Returns (h, improved)."""
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_peak = c[top_idx].astype(float)
    improved = False
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:
                diag_idx = idx
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        lo = -h0
        hi = 1.0 - h0
        best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
        if best_val < current_max - 1e-14:
            h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
            improved = True
            current_max = best_val
    return h, improved

def cd_loop(h, n, top_k, max_passes, deadline, seed=0):
    """Run CD for max_passes or until no improvement."""
    h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        h, improved = cd_pass(h, n, top_k, deadline, order)
        if not improved:

        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_grad(h, n, alpha):
    """Gradient of LSE-smoothed max correlation * dx."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx) / np.sum(np.exp(s - mx))
    h_rev = h[::-1]
    first = np.correlate(w, 1.0 - h, mode='full')[:n]
    second = np.correlate(w, h_rev, mode='full')[n-1::-1]
    return (first - second) * dx

def lse_value(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * dx

def grad_descent(h, n, alpha_init, max_iter, deadline):
    """Projected gradient descent with momentum on LSE objective."""
    h = h.copy()
    alpha = alpha_init
    lr = 0.01
    momentum = 0.85

        val = lse_value(h, n, alpha)
        g = lse_grad(h, n, alpha)
        g_tangent = g - np.mean(g)
        v = momentum * v + (1 - momentum) * g_tangent
        h_new = project_feasible(h - lr * v, n)
        val_new = lse_value(h_new, n, alpha)
        if val_new < val * (1 + 1e-12):

            if i % 30 == 0 and i > 0:

                lr = max(lr * 0.9, 0.0005)
        else:
            lr *= 0.5
            if lr < 0.0001:

    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    di = diag_idx
    for i in range(T):
        for j in range(i + 1, T):
            if i == di or j == di:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if di is not None:
        for i in range(T):
            if i == di:
                continue
            B = -(a_coeff[di] - a_coeff[i])
            C = -(c_peak[di] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
        B = -a_coeff[di]
        C = unaff_max - c_peak[di]
        disc = B * B - 4.0 * C
        if disc >= 0:
            sq = np.sqrt(disc)
            d1 = (-B + sq) / 2.0
            d2 = (-B - sq) / 2.0
            if lo <= d1 <= hi:
                candidates.append(d1)
            if lo <= d2 <= hi:
                candidates.append(d2)
    for i in range(T):
        if i == di:
            continue
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if di is not None:
            vals[di] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def cd_pass(h, n, top_k, deadline, order):
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_peak = c[top_idx].astype(float)
    improved = False
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:
                diag_idx = idx
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        lo = -h0
        hi = 1.0 - h0
        best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
        if best_val < current_max - 1e-14:
            h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
            improved = True
            current_max = best_val
    return h, improved

def cd_loop(h, n, top_k, max_passes, deadline, seed=0):
    h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)
        h, improved = cd_pass(h, n, top_k, deadline, order)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_value(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * dx

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx) / np.sum(np.exp(s - mx))
    h_rev = h[::-1]
    first = np.correlate(w, 1.0 - h, mode='full')[:n]
    second = np.correlate(w, h_rev, mode='full')[n - 1::-1]
    return (first - second) * dx

def grad_momentum(h, n, alpha_init, max_iter, deadline):
    h = h.copy()
    alpha = alpha_init
    lr = 0.008
    momentum = 0.88
    v = np.zeros(n)
    best_val = lse_value(h, n, alpha)
    for i in range(max_iter):
        if time.time() > deadline:
            break
        val = lse_value(h, n, alpha)
        g = lse_grad(h, n, alpha)
        g_t = g - np.mean(g)
        v = momentum * v + (1.0 - momentum) * g_t
        h_new = project_feasible(h - lr * v, n)
        val_new = lse_value(h_new, n, alpha)
        if val_new < val * (1.0 + 1e-12):
            h = h_new
            if i % 40 == 0 and i > 0:
                alpha = min(alpha * 1.4, 300000.0)
                lr = max(lr * 0.88, 0.0004)
        else:
            lr *= 0.45
            if lr < 0.00005:
                break
    return h, compute_c5(h, n)
