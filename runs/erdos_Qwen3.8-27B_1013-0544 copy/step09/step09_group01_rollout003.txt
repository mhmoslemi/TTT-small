    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size, momentum_arr, beta=0.7):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    s = m_star - (n - 1)
    grad = np.zeros(n)
    jf = np.arange(n) + s
    for j in range(n):
        g = 0.0
        if 0 <= j + s < n:
            g += (1.0 - h[j + s])
        if 0 <= j - s < n:
            g -= h[j - s]
        if s == 0:
            g = (1.0 - 2.0 * h[j])
        grad[j] = g
    momentum_arr = beta * momentum_arr + (1.0 - beta) * grad
    h_new = h - step_size * momentum_arr
    h_new = project_feasible(h_new, n)
    return h_new, momentum_arr

def subgradient_phase(h, n, deadline, max_iters=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mom = np.zeros(n)
    step = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        step = max(0.001, 0.05 / (1.0 + 0.01 * it))
        h_new, mom = subgradient_step(h, n, step, mom)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        h = h_new
    return best_h, best_c, mom

def two_point_cd(h, n, max_passes, deadline):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            if time.time() > deadline:
                break
            h0a, h0b = h[p], h[p + 1]
            if h0a <= 1e-13 or h0a >= 1 - 1e-13:
                continue
            if h0b <= 1e-13 or h0b >= 1 - 1e-13:
                continue
            best_d = (0.0, 0.0)
            best_f = compute_c5(h, n)
            base = h.copy()
            c = np.correlate(h, 1.0 - h, mode='full')
            m_lo_a = max(0, n - 1 - p)
            m_hi_a = min(2*n-2, n - 1 + p)
            m_lo_b = max(0, n - 2 - p)
            m_hi_b = min(2*n-2, n - 2 + p)
            m_lo = min(m_lo_a, m_lo_b)
            m_hi = max(m_hi_a, m_hi_b)
            ms = np.arange(m_lo, m_hi + 1)
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            aff = np.zeros(len(ms))
            for j, dvar in [(p, 'a'), (p+1, 'b')]:
                jf = j + ms - (n-1)
                jb = j - ms + (n-1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                if np.any(vf):
                    aff[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    aff[vb] -= h[jb[vb]]
            da_lo, da_hi = -h0a, 1.0 - h0a
            db_lo, db_hi = -h0b, 1.0 - h0b
            da1 = da_hi - gr*(da_hi-da_lo)
            da2 = da_lo + gr*(da_hi-da_lo)
            db1 = db_hi - gr*(db_hi-db_lo)
            db2 = db_lo + gr*(db_hi-db_lo)
            def eval2(da, db):
                vals = c[ms] + da*aff + db*aff
                for j, h0j, dj in [(p, h0a, da), (p+1, h0b, db)]:
                    dp = np.where(ms == n-1)[0]
                    if len(dp) > 0:
                        di = int(dp[0])
                        jf_d = j + (n-1) - (n-1)
                        vals[di] = c[ms][di] + da*(1.0-h0a-da) + db*(1.0-h0b-db)
                        break
                r = float(np.max(vals))
                return max(r, unaff_max)
            fa1 = eval2(da1, db1)
            fa2 = eval2(da1, db2)
            fa3 = eval2(da2, db1)
            fa4 = eval2(da2, db2)
            candidates = [(da1,db1,fa1),(da1,db2,fa2),(da2,db1,fa3),(da2,db2,fa4)]
            best = min(candidates, key=lambda x: x[2])
            if best[2] < best_f - 1e-14:
                h[p] = h0a + best[0]
                h[p+1] = h0b + best[1]
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD on original n
    if time.time() < deadline - 200:
        h1, c1 = coord_descent(best_h, n, 80, deadline - 200, gs_iters=55)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Subgradient to escape local optima
    if time.time() < deadline - 180:
        h2, c2, _ = subgradient_phase(best_h, n, deadline - 180, max_iters=1000)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(h2, n, 40, deadline - 180, gs_iters=50)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Two-point CD
    if time.time() < deadline - 120:
        h3, c3 = two_point_cd(best_h, n, 30, deadline - 120)
        if c3 < best_c:
            best_h, best_c = h3, c2 if False else c3
        h3, c3 = coord_descent(best_h, n, 40, deadline - 120, gs_iters=50)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Try coarser and finer n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [350, 500, 1000, 1400]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 60, deadline - 100, gs_iters=55)
        if c_o < best_c:
            h_o, c_o = slsqp_optimize(h_o, n_try, [2000, 10000, 100000], 300)
            h_o, c_o = coord_descent(h_o, n_try, 40, deadline - 100, gs_iters=55)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)
    # Phase 5: Multi-restart
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 80
        n_restarts = max(4, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            scale = 0.01 + 0.05 * (r / max(1, n_restarts - 1))
            if r % 4 == 0:
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif r % 4 == 1:
                x = np.linspace(0, 2, n_cur)
                hp = project_feasible(best_h + scale * 2 * np.sin(np.pi * (3 + r) * x), n_cur)
            elif r % 4 == 2:
                hp = project_feasible(best_h * (1 + 0.02 * rng.randn(n_cur)), n_cur)
            else:
                block = n_cur // 7
                hp = best_h.copy()
                start = (r * block) % max(1, n_cur - block)
                hp[start:start+block] = 1.0 - hp[start:start+block]
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 30, deadline - 80, gs_iters=45)
            if c_o < best_c:
                h_o, c_o = slsqp_optimize(h_o, n_cur, [5000, 50000], 200)
                h_o, c_o = coord_descent(h_o, n_cur, 30, deadline - 80, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 6: Final SLSQP + CD polish
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [10000, 100000, 500000], 500)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 50, deadline - 15, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
