def weighted_gradient(h,n,w):
   g=np.zeros(n)
   L=w.size
   for p in range(L):
      wp=w[p]
      if wp <= 1e-10: continue
      m=p-(n-1)
      if m>=0:
         length=n-m
         if length>0:
            g[:length] += wp*(1.0 - h[m:])
            g[m:] -= wp*h[:length]
      else:
         q=-m
         length=n-q
         if length>0:
            g[q:] += wp*(1.0 - h[:length])
            g[:length] -= wp*h[q:]
   return g

import numpy as np
import time
from bisect import bisect_right


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(x, n):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def build_hull(lines):
    if not lines:
        lines = [(0.0, -1.0e50)]
    lines.sort()
    uniq = []
    for m, b in lines:
        if uniq and abs(m - uniq[-1][0]) < 1e-13:
            if b > uniq[-1][1]:
                uniq[-1] = (float(m), float(b))
        else:
            uniq.append((float(m), float(b)))
    hm, hb, hx = [], [], []
    for m, b in uniq:
        active = True
        while True:
            if not hm:
                x = -np.inf
                break
            if abs(m - hm[-1]) < 1e-13:
                if b > hb[-1]:
                    hm.pop(); hb.pop(); hx.pop(); continue
                active = False
                break
            x = (hb[-1] - b) / (m - hm[-1])
            if x <= hx[-1] + 1e-12:
                hm.pop(); hb.pop(); hx.pop(); continue
            break
        if active:
            hm.append(m); hb.append(b); hx.append(x)
    if not hm:
        hm = [0.0]; hb = [-1.0e50]; hx = [-np.inf]
    return hx, hm, hb


def hull_value(d, hx, hm, hb):
    i = bisect_right(hx, float(d)) - 1
    if i < 0:
        i = 0
    if i >= len(hm):
        i = len(hm) - 1
    return float(hm[i] * d + hb[i])


def solve_1d_exact(cseg, a, diag, U, lo, hi):
    if hi <= lo:
        return 0.0, 1.0e100
    k = int(a.size)
    c0 = None
    a0 = None
    if diag is not None and 0 <= diag < k:
        c0 = float(cseg[diag])
        a0 = float(a[diag])
    lines = []
    for j in range(k):
        if diag is not None and j == diag:
            continue
        lines.append((float(a[j]), float(cseg[j])))
    if U > -1.0e40:
        lines.append((0.0, float(U)))
    hx, hm, hb = build_hull(lines)

    best_d = 0.0 if lo <= 0.0 <= hi else float(lo)
    best_val = 1.0e100
    cand = [float(lo), float(hi)]
    if lo < 0.0 < hi:
        cand.append(0.0)
    for x in hx[1:]:
        if lo <= x <= hi:
            cand.append(float(x))

    if c0 is not None and len(hm) > 0:
        for j in range(len(hm)):
            x1 = hx[j]
            x2 = hx[j + 1] if j + 1 < len(hx) else np.inf
            seglo = max(lo, x1 if np.isfinite(x1) else lo)
            seghi = min(hi, x2 if np.isfinite(x2) else hi)
            if seghi <= seglo:
                continue
            cand.append(float(seglo)); cand.append(float(seghi))
            cand.append(float(0.5 * (seglo + seghi)))
            Aa = a0 - hm[j]
            C0 = c0 - hb[j]
            disc = Aa * Aa + 4.0 * C0
            if disc >= -1e-9:
                if disc < 0.0:
                    disc = 0.0
                rr = np.sqrt(disc)
                r1 = 0.5 * (Aa + rr)
                r2 = 0.5 * (Aa - rr)
                if seglo - 1e-10 <= r1 <= seghi + 1e-10:
                    cand.append(float(np.clip(r1, lo, hi)))
                if seglo - 1e-10 <= r2 <= seghi + 1e-10:
                    cand.append(float(np.clip(r2, lo, hi)))

    for d in cand:
        if d < lo or d > hi:
            continue
        val = hull_value(float(d), hx, hm, hb)
        if c0 is not None:
            q = c0 + a0 * d - d * d
            if q > val:
                val = q
        if val < best_val - 1e-15:
            best_val = float(val)
            best_d = float(d)

    if hi - lo > 1e-7:
        margin = min(1e-5, 0.1 * (hi - lo))
        for d in np.linspace(best_d - margin, best_d + margin, 17):
            d = float(d)
            if d < lo or d > hi:
                continue
            val = hull_value(d, hx, hm, hb)
            if c0 is not None:
                q = c0 + a0 * d - d * d
                if q > val:
                    val = q
            if val < best_val - 1e-15:
                best_val = float(val)
                best_d = d
    return best_d, float(best_val)


def coordinate_descent(h, n, max_step, passes, deadline, rng):
    h = project_feasible(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    no_gain = 0
    L = 2 * n - 1

    for _ in range(passes):
        if time.time() >= deadline:
            break
        h_work = best_h.copy()
        c = np.correlate(h_work, 1.0 - h_work, mode="full")
        work_count = float(c.max())
        improved = False
        order = rng.permutation(n)

        for idx in order:
            if time.time() >= deadline:
                break
            h0 = float(h_work[idx])
            mf_lo = max(0, n - 1 - idx)
            mf_hi = min(L - 1, L - 1 - idx)
            mb_lo = max(0, idx)
            mb_hi = min(L - 1, idx + n - 1)
            m_lo = mf_lo if mf_lo < mb_lo else mb_lo
            m_hi = mf_hi if mf_hi > mb_hi else mb_hi
            total = m_hi - m_lo + 1

            a = np.zeros(total)
            pf_lo = max(m_lo, n - 1 - idx)
            pf_hi = min(m_hi, L - 1 - idx)
            if pf_lo <= pf_hi:
                j = idx + np.arange(pf_lo, pf_hi + 1) - (n - 1)
                a[pf_lo - m_lo:pf_hi - m_lo + 1] += 1.0 - h_work[j]

            pb_lo = max(m_lo, idx)
            pb_hi = min(m_hi, idx + n - 1)
            if pb_lo <= pb_hi:
                j = idx - np.arange(pb_lo, pb_hi + 1) + (n - 1)
                a[pb_lo - m_lo:pb_hi - m_lo + 1] -= h_work[j]

            self_local = (n - 1) - m_lo
            diag = self_local if 0 <= self_local < total else None
            U = -1.0e50
            if m_lo > 0:
                U = max(U, float(c[:m_lo].max()))
            if m_hi < L - 1:
                U = max(U, float(c[m_hi + 1:].max()))

            lo = max(-h0, -float(max_step))
            hi = min(1.0 - h0, float(max_step))
            if hi <= lo:
                continue

            d, val = solve_1d_exact(c[m_lo:m_hi + 1], a, diag, U, lo, hi)
            if d < lo:
                d = lo
            if d > hi:
                d = hi
            if abs(d) <= 1e-14 or val >= work_count - 1e-11:
                continue

            old = c[m_lo:m_hi + 1].copy()
            c[m_lo:m_hi + 1] += d * a
            if m_lo <= n - 1 <= m_hi:
                c[n - 1] -= d * d
            h_work[idx] = float(h0 + d)
            new_count = float(c.max())
            if new_count < work_count - 1e-11:
                work_count = new_count
                improved = True
            else:
                c[m_lo:m_hi + 1] = old
                h_work[idx] = h0

        new_h = project_feasible(h_work, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-14:
            best_h = new_h
            best_c = new_c
            no_gain = 0
        else:
            no_gain += 1
            if not improved or no_gain > 6:
                break

    return best_h, best_c


def weighted_gradient(h, n, w):
    g = np.zeros(n)
    for p in range(w.size):
        wp = float(w[p])
        if wp <= 1e-10:
            continue
        m = p - (n - 1)
        if m >= 0:
            length = n - m
            if length > 0:
                g[:length] += wp * (1.0 - h[m:])
                g[m:] -= wp * h[:length]
        else:
            q = -m
            length = n - q
            if length > 0:
                g[q:] += wp * (1.0 - h[:length])
                g[:length] -= wp * h[q:]
    return g


def projected_gradient_polish(h, n, deadline):
    h = project_feasible(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    betas = [30.0, 10.0, 4.0, 1.5]
    steps = [0.02, 0.05, 0.10, 0.25]

    for beta in betas:
        no_gain = 0
        for _ in range(35):
            if time.time() + 0.25 >= deadline:
                break
            c = np.correlate(best_h, 1.0 - best_h, mode="full")
            maxc = float(c.max())
            u = (c - maxc) / beta
            u -= u.max()
            w = np.exp(u)
            s = float(w.sum())
            if not np.isfinite(s) or s <= 0.0:
                break
            w /= s
            g = weighted_gradient(best_h, n, w)
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                break
            dirn = g / gn
            ok = False
            for alpha in steps:
                y = project_feasible(best_h - alpha * dirn, n)
                cy = compute_c5(y, n)
                if cy < best_c - 1e-13:
                    best_h = y
                    best_c = cy
                    ok = True
                    break
            if ok:
                no_gain = 0
            else:
                no_gain += 1
                if no_gain > 12:
                    break
    return best_h, best_c


def try_n_value(base_h, base_n, ntry, deadline, rng):
    ntry = int(ntry)
    if ntry % 2:
        ntry += 1
    if ntry < 24 or ntry > 1000:
        return None, None
    x_old = 2.0 * np.arange(base_n) / float(base_n)
    x_new = 2.0 * np.arange(ntry) / float(ntry)
    h = np.interp(x_new, x_old, base_h)
    h = project_feasible(h, ntry)
    c = compute_c5(h, ntry)
    h2, c2 = coordinate_descent(h, ntry, 0.30, 1000, max(time.time(), deadline - 8), rng)
    if c2 < c:
        h, c = h2, c2
    if time.time() < deadline - 4:
        h3, c3 = coordinate_descent(h, ntry, 0.05, 500, deadline, rng)
        if c3 < c:
            h, c = h3, c3
    return h, c
