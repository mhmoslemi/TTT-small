def fw_step(h, n, active_k_set):
    """One Frank-Wolfe step."""
    # Compute gradient of average active f_k
    g = np.zeros(n)
    for k_idx in active_k_set:
        k = k_idx - (n-1)  # actual lag
        for j in range(n):
            # ∂f_k/∂h[j] = (1-h[j+k]) - h[j-k]
            term1 = 0.0

            if 0 <= j-k < n:

    # Sort by g, take n/2 smallest
    order = np.argsort(g)
    y = np.zeros(n)
    y[order[:n//2]] = 1.0
    return y

def cd_pass(h, n, order='sequential', deadline=None):
    """One full pass of coordinate descent."""
    h = h.copy()
    n_idx = np.arange(n)
    if order == 'random':
        np.random.shuffle(n_idx)
    elif order == 'worst':
        c = np.correlate(h, 1.0 - h, mode='full')
        # Compute sensitivity: how much each coordinate contributes to max
        # Use std of affected correlations as proxy
        idx = n_idx.copy()
    else:
        idx = n_idx
    c = np.correlate(h, 1.0 - h, mode='full')
    for p in idx:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d_exact(c, h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
    h = project_feasible(h, n)
    return h

def cd_loop(h, n, max_passes, deadline, order='sequential'):
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h = cd_pass(h, n, order=order, deadline=deadline)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Check convergence
        if np.max(np.abs(c - np.correlate(h, 1.0 - h, mode='full'))) < 1e-15:
            break
    return h

def fw_iterate(h, n, max_iters, deadline, tol=1e-10):
    """Frank-Wolfe iterations."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        # Find active set (correlations within tol of max)
        active = np.where(c > cmax - 0.05 * cmax)[0]
        # Compute gradient of average active f_k
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            # ∂f_k/∂h[j] = (1-h[j+k]) - h[j-k]
            for j in range(n):
                t1 = 0.0
                t2 = 0.0
                if 0 <= j + k < n:
                    t1 = 1.0 - h[j + k]
                if 0 <= j - k < n:
                    t2 = h[j - k]
                g[j] += t1 - t2
        g /= len(active)
        # Solve linear subproblem: min g^T y s.t. 0<=y<=1, sum(y)=n/2
        order = np.argsort(g)
        y = np.zeros(n)
        y[order[:n // 2]] = 1.0
        # Line search
        best_alpha = 0.0
        best_val = best_c
        for alpha in np.linspace(0.01, 1.0, 10):
            h_trial = (1 - alpha) * h + alpha * y
            h_trial = np.clip(h_trial, 0, 1)
            ct = compute_c5(h_trial, n)
            if ct < best_val:
                best_val = ct
                best_alpha = alpha
        if best_alpha == 0.0:
            break
        h = (1 - best_alpha) * h + best_alpha * y
        best_c = best_val
    return h, best_c

def fw_step(h, n, beta=50.0):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w /= w.sum()
    # Gradient of weighted sum
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        g[m]  # placeholder
        if k >= 0:
            g[:n-k] += w[m] * (1.0 - h[k:])
            g[k:] -= w[m] * h[:n-k]
        else:
            d = -k
            g[d:] += w[m] * (1.0 - h[:n-d])
            g[:n-d] -= w[m] * h[d:]
    return g, c.max()

For correlation at lag k (corresponding to index m = n-1+k in the full output):
c[m] = ∑_i h[i] * (1-h[i+k]) where i+k must be in [0, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_aff = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_aff[di[0]]
        c_aff[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = float(c[:ms[0]].max())
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    L_lo = unaff
    L_hi = float(c.max())
    if L_lo >= L_hi:
        return h0
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feas = True
        neg_m = a < -1e-15
        pos_m = a > 1e-15
        zero_m = np.abs(a) <= 1e-15
        if np.any(zero_m) and np.max(c_aff[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_aff[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_aff[pos_m]) / a[pos_m])))
        if feas and has_diag:
            Q = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > xh:
                    feas = False
                elif xl > r1 and xh < r2:
                    feas = False
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        if feas and x_lo <= x_hi:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_m = a < -1e-15

def cd_full(h, n, order='sequential'):
    h = h.copy()
    idx = np.arange(n)
    if order == 'random':
        idx = np.random.permutation(n)
    c = np.correlate(h, 1.0 - h, mode='full')

        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d_exact(c, h, p, n)
        if abs(x_new - h0) > 1e-14:

    h = project_feasible(h, n)
    return h

def fw_optimize(h, n, max_iters, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_shifted = c - c.max()
        w = np.exp(50.0 * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        g = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-14:
                continue
            k = m - (n - 1)
            wm = w[m]
            if k >= 0:
                g[:n - k] += wm * (1.0 - h[k:])
                g[k:] -= wm * h[:n - k]
            else:
                d = -k
                g[d:] += wm * (1.0 - h[:n - d])
                g[:n - d] -= wm * h[d:]
        order_g = np.argsort(g)
        y = np.zeros(n)
        y[order_g[:n // 2]] = 1.0
        best_alpha = 0.0
        best_val = best_c
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_t = (1.0 - alpha) * h + alpha * y
            h_t = np.clip(h_t, 0.0, 1.0)
            ct = compute_c5(h_t, n)
            if ct < best_val:
                best_val = ct
                best_alpha = alpha
        if best_alpha == 0.0:
            break
        h = (1.0 - best_alpha) * h + best_alpha * y
        best_c = best_val
    return h, best_c

        if time.time() > deadline - 50:
            break
        if n_try == n or n_try % 2 != 0:
            continue
        x_orig = np.linspace(0, 2, n)
        x_new_arr = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new_arr, x_orig, best_h), n_try)
        for j in range(100):
            if time.time() > deadline - 40:
                break
            h_int = cd_full(h_int, n_try)
            c_int = compute_c5(h_int, n_try)
            if c_int < best_c:
                best_h, best_c, n = h_int, c_int, n_try
                break
        # Also try FW on this n
        if time.time() < deadline - 40:
            h_fw2, c_fw2 = fw_optimize(h_int, n_try, 30, deadline - 40)
            if c_fw2 < best_c:
                best_h, best_c, n = h_fw2, c_fw2, n_try

    # Phase 4: Final CD polish
    n = len(best_h)
    for i in range(500):
        if time.time() > deadline:
            break
        h_new = cd_full(best_h, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        else:
            break

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
