import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _eval_1d(d, c_peak, a_coeff, diag_lo, diag_hi, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_lo >= 0:
        vals[diag_lo:diag_hi+1] += -d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi, 0.0]
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
        d_quad = c_peak[diag_idx] / a_coeff[diag_idx] if abs(a_coeff[diag_idx]) > 1e-15 else 0.0
        if lo <= d_quad <= hi:
            candidates.append(d_quad)
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = 0.0
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        if diag_idx is not None and diag_idx >= 0:
            v = _eval_1d(d, c_peak, a_coeff, diag_idx, diag_idx, unaff_max)
        else:
            v = _eval_1d(d, c_peak, a_coeff, -1, -1, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    v0 = _eval_1d(0.0, c_peak, a_coeff,
                  diag_idx if diag_idx is not None else -1,
                  diag_idx if diag_idx is not None else -1, unaff_max)
    if v0 < best_val:
        best_d, best_val = 0.0, v0
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k, verify=False):
    h = h.copy()
    current_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
        current_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return h, current_max

def swap_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    current_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        improved = False
        for p in range(n - 1):
            if time.time() > deadline:
                break
            h0a, h0b = h[p], h[p + 1]
            lo = max(-h0a, h0b - 1.0)
            hi = min(1.0 - h0a, h0b)
            if hi - lo < 1e-14:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf_a = p + m - (n - 1)
                jb_a = p - m + (n - 1)
                jf_b = (p + 1) + m - (n - 1)
                jb_b = (p + 1) - m + (n - 1)
                aa = 0.0
                if 0 <= jf_a < n:
                    aa += 1.0 - h[jf_a]
                if 0 <= jb_a < n:
                    aa += -h[jb_a]
                ab = 0.0
                if 0 <= jf_b < n:
                    ab += 1.0 - h[jf_b]
                if 0 <= jb_b < n:
                    ab += -h[jb_b]
                a_coeff[idx] = aa - ab
                if m == n - 1:
                    diag_idx = idx
            a_coeff[diag_idx if diag_idx is not None else 0] *= 1.0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0a + best_d, 0.0, 1.0))
                h[p + 1] = float(np.clip(h0b - best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
        current_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return h, current_max

def subgradient_momentum(h, n, max_iters, deadline, alpha_init=0.001):
    h = h.copy()
    y = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    mu = 0.9
    t = 1.0
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1.0
        y = h + (mu * (t - 1.0) / t) * (h - y + v) if it > 0 else h.copy()
        c = np.correlate(y, 1.0 - y, mode='full')
        m = int(np.argmax(c))
        grad = np.zeros(n)
        jf = np.arange(n) + m - (n - 1)
        jb = np.arange(n) - m + (n - 1)
        mask_f = (jf >= 0) & (jf < n)
        mask_b = (jb >= 0) & (jb < n)
        grad[mask_f] += (1.0 - y[jf[mask_f]])
        grad[mask_b] += (-y[jb[mask_b]])
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        alpha_eff = alpha / (it * np.sqrt(2.0 * np.log(it + 1)) + 1.0)
        y_new = y - alpha_eff * grad
        y_new = project_feasible(y_new, n)
        c_new = compute_c5(y_new, n)
        if c_new < best_c:
            h = y_new
            best_c = c_new
            mu = min(mu * 1.001, 0.95)
        else:
            mu = max(mu * 0.99, 0.5)
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

    if time.time() < deadline - 250:
        h2, c2 = subgradient_momentum(best_h, n, 500, deadline - 250, alpha_init=0.005)
        if c2 < best_c:
            best_h, c2 = h2, c2
            h3, c3 = multi_cd(h2, n, 200, deadline - 200, 20)
            if c3 < best_c:
                best_h, best_c = h3, c3
    # Phase 5: SLSQP refinement
    if time.time() < deadline - 200:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 200:
                break
            h_s, c_s = slsqp_run(best_h, n, alpha, 250)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
    # Phase 6: final deep CD
    if time.time() < deadline - 80:
        for tk in [40, 60, 80]:
            if time.time() > deadline - 80:
                break
            h2, c2 = multi_cd(best_h, n, 500, deadline - 80, tk)
            if c2 < best_c:
                best_h, best_c = h2, c2

    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [650, 550, 700, 500, 800]:

            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)

    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
