    m=6
    h = np.linspace(0.11,0.77,m)
    v = 1.0 - h

import numpy as np
import time


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < (1.0 - 1e-14)
            if not mask.any():
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / float(mask.sum()))
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / float(mask.sum()))
    for _ in range(n):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + diff)
    return h


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def calibrate_orientation():
    m = 6
    h = np.linspace(0.11, 0.77, m)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full")
    f = np.zeros(2 * m - 1)
    mid = m - 1
    for i in range(m):
        hi = h[i]
        for j in range(m):
            f[j - i + mid] += hi * v[j]
    if np.allclose(c, f, rtol=1e-11, atol=1e-12):
        return 0
    if np.allclose(c, f[::-1], rtol=1e-11, atol=1e-12):
        return 1
    return 0


def compute_c_internal(h, n, ori):
    c = np.correlate(h, 1.0 - h, mode="full")
    if ori == 1:
        c = c[::-1].copy()
    return c


def coord_candidate_points(xl, xh, b):
    pts = []
    xl = float(np.clip(xl, 0.0, 1.0))
    xh = float(np.clip(xh, 0.0, 1.0))
    if xl > xh + 1e-14:
        return pts
    tol = 1e-12
    if b < 0.0:
        b = 0.0
    if b >= 0.25 - 1e-13:
        pts.extend((xl, xh, 0.5 * (xl + xh)))
    else:
        s = float(np.sqrt(max(0.0, 1.0 - 4.0 * b)))
        r1 = 0.5 * (1.0 - s)
        r2 = 0.5 * (1.0 + s)
        if xl <= r1 + tol:
            bb = min(xh, r1)
            if xl <= bb + tol:
                if bb > xl:
                    pts.extend((xl, bb, 0.5 * (xl + bb)))
                else:
                    pts.extend((xl, bb))
        if xh >= r2 - tol:
            aa = max(xl, r2)
            if aa <= xh + tol:
                if xh > aa:
                    pts.extend((aa, xh, 0.5 * (aa + xh)))
                else:
                    pts.extend((aa, xh))
        if xl - tol <= r1 <= xh + tol:
            pts.append(r1)
        if xl - tol <= r2 <= xh + tol:
            pts.append(r2)
    for vv in (0.0, 1.0):
        if xl - tol <= vv <= xh + tol:
            pts.append(vv)
    out = []
    for x in pts:
        x = float(np.clip(x, 0.0, 1.0))
        ok = True
        for y in out:
            if abs(x - y) < 1e-10:
                ok = False
                break
        if ok:
            out.append(x)
    if not out:
        out.append(float(np.clip(0.5 * (xl + xh), 0.0, 1.0)))
    return out


def eval_coord_point(x, c_base, a, diag, diag_c, h0, unaff):
    d = x - h0
    vals = c_base + a * d
    vals[diag] = diag_c + x - x * x - h0 + h0 * h0
    m = float(vals.max())
    if unaff > m:
        m = unaff
    return m


def coord_step(c, h, p, n, iters=28, accept_tol=1e-10):
    h0 = float(h[p])
    n1 = n - 1

    lo1 = n1 - p
    hi1 = 2 * n1 - p
    lo2 = p
    hi2 = n + p - 1
    lo = lo1 if lo1 < lo2 else lo2
    hi = hi1 if hi1 > hi2 else hi2
    length = hi - lo + 1

    a = np.zeros(length)
    inv = 1.0 - h
    a[lo1 - lo:hi1 - lo + 1] += inv
    a[lo2 - lo:hi2 - lo + 1] -= h[::-1]

    diag = n1 - lo
    c_base = c[lo:hi + 1]
    a[diag] = 0.0
    diag_c = float(c_base[diag])

    unaff = -1.0
    total = 2 * n1
    if lo > 0:
        tmp = float(c[:lo].max())
        if tmp > unaff:
            unaff = tmp
    if hi < total:
        tmp = float(c[hi + 1:].max())
        if tmp > unaff:
            unaff = tmp

    cur = float(c_base.max())
    if unaff > cur:
        cur = unaff

    tol_a = 1e-14
    pos = a > tol_a
    neg = a < -tol_a
    zero = (np.abs(a) <= tol_a)
    zero[diag] = False

    pos_any = bool(pos.any())
    neg_any = bool(neg.any())
    pos_a = a[pos]
    pos_c = c_base[pos]
    neg_a = a[neg]
    neg_c = c_base[neg]
    zero_max = float(c_base[zero].max()) if zero.any() else -1.0

    llo = unaff if unaff > zero_max else zero_max
    lhi = cur
    h0q = h0 - h0 * h0

    for _ in range(int(iters)):
        if lhi - llo < 1e-12:
            break
        l = 0.5 * (llo + lhi)
        feasible = True
        if zero_max > l:
            feasible = False

        dlo = -h0
        dhi = 1.0 - h0
        if feasible and pos_any:
            v = float(np.min((l - pos_c) / pos_a))
            if v < dhi:
                dhi = v
        if feasible and neg_any:
            v = float(np.max((l - neg_c) / neg_a))
            if v > dlo:
                dlo = v
        if feasible and dlo > dhi + 1e-14:
            feasible = False

        if feasible:
            bq = l - diag_c + h0q
            if bq < -1e-10:
                feasible = False
            elif bq < 0.25 - 1e-13:
                if bq < 0.0:
                    bq = 0.0
                s = float(np.sqrt(max(0.0, 1.0 - 4.0 * bq)))
                r1 = 0.5 * (1.0 - s)
                r2 = 0.5 * (1.0 - (-s))
                xl = h0 + dlo
                xh = h0 + dhi
                if xl < 0.0:
                    xl = 0.0
                if xh > 1.0:
                    xh = 1.0
                if xl > r1 + 1e-13 and xh < r2 - 1e-13:
                    feasible = False

        if feasible:
            lhi = l
        else:
            llo = l

    l_val = lhi
    dlo = -h0
    dhi = 1.0 - h0
    if pos_any:
        v = float(np.min((l_val - pos_c) / pos_a))
        if v < dhi:
            dhi = v
    if neg_any:
        v = float(np.max((l_val - neg_c) / neg_a))
        if v > dlo:
            dlo = v

    xl = h0 + dlo
    xh = h0 + dhi
    if xl < 0.0:
        xl = 0.0
    if xh > 1.0:
        xh = 1.0
    if xl > xh + 1e-13:
        return h0, cur, False

    bq = l_val - diag_c + h0q
    if bq < -1e-9:
        return h0, cur, False
    if bq < 0.0:
        bq = 0.0

    cand = coord_candidate_points(xl, xh, bq)
    best_x = h0
    best_val = cur
    for xx in cand:
        vv = eval_coord_point(xx, c_base, a, diag, diag_c, h0, unaff)
        if vv < best_val:
            best_val = vv
            best_x = xx

    if best_val < cur - accept_tol and abs(best_x - h0) > 1e-14:
        old_slice = c[lo:hi + 1].copy()
        d = best_x - h0
        c[lo:hi + 1] += a * d
        c[n1] += best_x - best_x * best_x - h0 + h0 * h0
        h[p] = best_x
        new_cur = float(c.max())
        if new_cur > cur + 1e-7:
            c[lo:hi + 1] = old_slice
            h[p] = h0
            return h0, cur, False
        return best_x, new_cur, True

    return h0, cur, False


def exact_cd(h, n, deadline, ori, max_passes=300, iters=28,
             shuffle=False, seed=1, project_every=60, accept_tol=1e-10):
    h = project_feasible(h, n)
    c = compute_c_internal(h, n, ori)
    cur = float(c.max())
    rng = np.random.RandomState(int(seed)) if shuffle else None

    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle:
            rng.shuffle(order)

        improved = False
        proj_changed = False
        count = 0

        for p in order:
            if time.time() > deadline:
                break
            old_cur = cur
            _, cur, changed = coord_step(c, h, int(p), n, iters, accept_tol)
            if changed and cur <= old_cur + 1e-10:
                improved = True

            count += 1
            if project_every and count % project_every == 0 and time.time() < deadline - 1.0:
                h2 = project_feasible(h, n)
                if float(np.max(np.abs(h2 - h))) > 1e-13:
                    h = h2
                    c = compute_c_internal(h, n, ori)
                    cur = float(c.max())
                    proj_changed = True

        if time.time() > deadline:
            break

        h2 = project_feasible(h, n)
        if float(np.max(np.abs(h2 - h))) > 1e-13:
            h = h2
            c = compute_c_internal(h, n, ori)
            cur = float(c.max())
            proj_changed = True

        if not improved and not proj_changed:
            break

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def interp_h(h, n, ni):
    if ni == n:
        return project_feasible(h, n)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, ni, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), ni)


def perturb_h(base, scale, rng):
    n = len(base)
    return project_feasible(base + rng.normal(0.0, float(scale), n), n)


def smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, 8):
        f = int(k + rng.randint(0, 3))
        amp = 0.28 / float(k + 1)
        ph = 2.0 * np.pi * float(rng.rand())
        h += amp * np.cos(np.pi * f * x + ph)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + total - 4.0
    rng = np.random.RandomState(int(seed))

    try:
        base = np.array(initial_h_values, dtype=float).copy()
        n = int(len(base))
        if n < 30:
            raise ValueError
    except Exception:
        n = 700
        base = smooth_start(n, rng)

    h = project_feasible(base, n)
    ori = calibrate_orientation()

    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    main_dl = t0 + max(60.0, min(total * 0.55, total - 120.0))
    main_dl = min(main_dl, deadline)
    if main_dl > t0 + 30.0:
        h2, c2 = exact_cd(h, n, main_dl, ori,
                          max_passes=800, iters=30,
                          shuffle=False, seed=int(seed), project_every=60)
        if c2 < best_c:
            best_h, best_c = h2, c2

    for scale in (0.05, 0.18, 0.35):
        rem = deadline - time.time()
        if rem < 80.0:
            break
        st = perturb_h(best_h, scale, rng)
        dl = time.time() + min(240.0, max(45.0, rem * 0.32))
        if dl > deadline - 25.0:
            dl = deadline - 25.0
        h2, c2 = exact_cd(st, len(st), dl, ori,
                          max_passes=400, iters=26,
                          shuffle=True, seed=int(seed + 101 * scale), project_every=60)
        if c2 < best_c:
            best_h, best_c = h2, c2

    ncur = len(best_h)
    for factor in (0.6, 1.3, 0.85):
        nf = int(ncur * factor)
        nf = int(max(250, min(950, nf)))
        if nf == ncur:
            continue
        rem = deadline - time.time()
        if rem < 80.0:
            break
        st = interp_h(best_h, ncur, nf)
        dl = time.time() + min(220.0, max(45.0, rem * 0.30))
        if dl > deadline - 25.0:
            dl = deadline - 25.0
        h2, c2 = exact_cd(st, nf, dl, ori,
                          max_passes=400, iters=26,
                          shuffle=(nf < ncur), seed=nf, project_every=60)
        if c2 < best_c:
            best_h, best_c = h2, c2
            ncur = nf

    rem = deadline - time.time()
    if rem > 130.0:
        st = smooth_start(len(best_h), rng)
        dl = min(deadline - 30.0, time.time() + 120.0)
        h2, c2 = exact_cd(st, len(st), dl, ori,
                          max_passes=300, iters=26,
                          shuffle=True, seed=int(seed + 9999), project_every=60)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if deadline - time.time() > 60.0:
        h2, c2 = exact_cd(best_h, len(best_h), deadline - 2.0, ori,
                          max_passes=600, iters=32,
                          shuffle=False, seed=int(seed + 777), project_every=80)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
