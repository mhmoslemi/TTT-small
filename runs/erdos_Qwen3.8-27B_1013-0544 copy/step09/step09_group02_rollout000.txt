def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, h_best), n_try)
        h_opt, c_opt = combined_cd(h_int, n_try, max_passes=90, deadline=deadline, rng=rng)
        if c_opt < compute_c5(h_best, n_best):
            h_best, n_best = h_opt, n_try
            x_cur = x_new
    return h_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Initial CD
    h1, c1 = coord_cd(best_h, n_init, max_passes=80, deadline=deadline, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Multi-resolution
    h2, n2 = multi_resolution(best_h, len(best_h), deadline, rng)
    if compute_c5(h2, n2) < best_c:
        best_h, best_c = h2, compute_c5(h2, n2)

    # Phase 3: Extended combined CD
    n_cur = len(best_h)
    h3, c3 = combined_cd(best_h, n_cur, max_passes=300, deadline=deadline, rng=rng)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: SLSQP with adaptive alphas
    if time.time() < deadline - 100:
        h4, c4 = slsqp_polish(best_h, n_cur, [3000, 10000, 40000], deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_cd(best_h, n_cur, max_passes=150, deadline=deadline, rng=rng)
        if c4b < best_c:
            best_h, best_c = h4b, c4b

    # Phase 5: More pairwise + single CD
    n_cur = len(best_h)
    if time.time() < deadline - 60:
        h5, c5 = combined_cd(best_h, n_cur, max_passes=200, deadline=deadline, rng=rng)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Random restarts with structured perturbations
    n_cur = len(best_h)
    for r in range(20):
        if time.time() > deadline - 40:
            break
        s = 0.003 + 0.002 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = combined_cd(hp, n_cur, max_passes=60, deadline=deadline, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 7: Final polish
    if time.time() < deadline - 20:
        hf, cf = slsqp_polish(best_h, len(best_h), [80000, 300000], deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    hf, cf = coord_cd(best_h, len(best_h), max_passes=300, deadline=deadline, rng=rng)
    if cf < best_c:
        best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
