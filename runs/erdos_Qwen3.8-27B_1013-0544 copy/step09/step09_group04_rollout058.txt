    """
    boundaries: sorted array of M-1 interior boundaries (x_1, ..., x_{M-1})
    heights: array of M heights (a_1, ..., a_M)
    Returns C_5 = max_k I(k)
    """

    # Full boundary array: [0, x_1, ..., x_{M-1}, 2]
    all_bounds = np.concatenate([[0], boundaries, [2]])
    
    # All possible shift break points: |x_i - x_j| for all i,j
    # k ranges over [0, 2] (or we can consider |k| up to 2)
    # Actually, for the integral ∫_0^2 h(x)(1-h(x+k)) dx, k can be any real
    # but we only need k in [-2, 2] (outside this range, the integral is 0)
    
    # Break points in k: values where a step of h(x+k) aligns with a step of h(x)
    # These are k = x_j - x_i for all pairs (i,j)
    k_vals = set()
    for i in range(M+1):
        for j in range(M+1):
            k_vals.add(all_bounds[j] - all_bounds[i])
    k_vals = sorted(set(k for k in k_vals if -2 <= k <= 2))
    
    # For each k, compute I(k) = ∫_0^2 h(x)(1-h(x+k)) dx
    max_I = 0
    for k in k_vals:
        # h(x+k) has the same heights but shifted
        # The combined step function h(x) * (1-h(x+k)) has steps at all_bounds and all_bounds-k
        combined = sorted(set(list(all_bounds) + [b - k for b in all_bounds]))
        # Clip to [0, 2]
        combined = [max(0, min(2, b)) for b in combined]
        combined = sorted(set(combined))
        
        # Integrate over each sub-interval
        I = 0
        for idx in range(len(combined)-1):
            x_mid = (combined[idx] + combined[idx+1]) / 2
            # Find which step x_mid is in for h
            seg_h = np.searchsorted(all_bounds, x_mid, side='right') - 1
            seg_h = max(0, min(M-1, seg_h))
            # Find which step x_mid + k is in for h(x+k) = h at position x_mid+k
            pos = x_mid + k

                h_shifted = heights[seg_sh]
            I += heights[seg_h] * (1 - h_shifted) * (combined[idx+1] - combined[idx])
        max_I = max(max_I, I)
    
    return max_I
