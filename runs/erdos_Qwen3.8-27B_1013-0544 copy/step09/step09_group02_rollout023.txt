import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def correlation_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_at(h, n, m_star):
    sg = np.zeros(n)
    jf = m_star - (n - 1)
    for i in range(n):
        j = i + jf
        if 0 <= j < n:
            sg[i] += (1.0 - h[j])
        ip = i - jf
        if 0 <= ip < n:
            sg[i] -= h[ip]
    return sg

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi+1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2*n-2:
        unaff = max(unaff, float(c[mhi+1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = correlation_full(h, n)
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 8 == 7:
            c = correlation_full(h, n)
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def nesterov_minmax(h, n, max_iter=8000, deadline=None):
    h = h.copy()
    h0 = h.copy()
    y = h.copy()
    c = correlation_full(y, n)
    best_c = float(c.max()) * (2.0 / n)
    best_h = h.copy()
    active_tol_factor = 0.03
    for t in range(1, max_iter + 1):
        if deadline is not None and time.time() > deadline:
            break
        c = correlation_full(y, n)
        m_max = float(c.max())
        tol = active_tol_factor * m_max + 1e-8
        active = np.where(c >= m_max - tol)[0]
        sg = np.zeros(n)
        for m in active:
            sg += subgradient_at(y, n, int(m))
        sg /= len(active)
        step = 2.0 / (n * (t + 1))
        y_new = project_feasible(y - step * sg, n)
        beta = (t - 1) / (t + 1)
        h_new = beta * y_new + (1.0 - beta) * h0
        c_new = correlation_full(h_new, n)
        c_max_new = float(c_new.max()) * (2.0 / n)
        if c_max_new < best_c:
            best_c = c_max_new
            best_h = h_new.copy()
        h = h_new
        y = y_new
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_perturb(h, n, rng):
    """Apply structured (sinusoidal) perturbation to escape local optima."""
    freq = rng.randint(2, n // 4 + 1)
    amp = 0.01 + 0.02 * rng.random()
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h + pert, n)
