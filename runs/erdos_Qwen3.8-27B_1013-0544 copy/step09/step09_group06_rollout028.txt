def project_euclidean(y, n):
    """Euclidean projection onto {h: 0<=h<=1, sum h = n/2}."""
    target = n / 2.0
    lo, hi = y.min() - 1, y.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    return np.clip(y - (lo+hi)/2, 0, 1)

            hc2, cc2 = multi_cd(hp, n, 120, deadline - 200, 12)
            if cc2 < best_c:
                best_h, best_c = hc2, cc2
    # Phase 4: Nesterov refinement on best
    if time.time() < deadline - 120:
        for alpha in [300, 600, 1200]:
            if time.time() > deadline - 120:
                break
            h4, c4 = nesterov_soft_max(best_h, n, deadline - 120, alpha, 3000)
            if c4 < best_c:
                best_h, best_c = h4, c4
    # Phase 5: Final deep CD
    if time.time() < deadline - 5:
        h5, c5v = multi_cd(best_h, n, 1500, deadline - 5, 35)
        if c5v < best_c:
            best_h, best_c = h5, c5v
    best_h = try_symmetry(best_h, n)
    best_h = project_euclidean(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
