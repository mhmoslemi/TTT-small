            if tot < 1e-15:

        return 0.0

    sh = alpha*(c-m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh-mx)))

    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    sh = alpha*(c-m)
    mx = sh.max()
    w = np.exp(sh-mx)
    w /= w.sum()

        g = 0.0
        for mi in range(2*n-1):
            wi = w[mi]
            if wi < 1e-14: continue
            i1 = j + mi - (n-1)
            i2 = j - mi + (n-1)
            if 0 <= i1 < n: g += wi*(1.0-x[i1])
            if 0 <= i2 < n: g -= wi*x[i2]
        grad[j] = g
    return (2.0/n)*grad

            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass

    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for mi in range(2*n-1):
            wi = w[mi]
            if wi < 1e-14: continue
            i1 = j + mi - (n-1)
            i2 = j - mi + (n-1)
            if 0 <= i1 < n: g += wi * (1.0 - x[i1])
            if 0 <= i2 < n: g -= wi * x[i2]
        grad[j] = g
    return (2.0 / n) * grad

def optimize_smooth(h, n, alphas, deadline, maxiter_each=300):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline: break
        try:
            res = minimize(smooth_max_obj, best_h, args=(n, a), method='L-BFGS-B',
                         jac=lambda x, n_=n, a_=a: smooth_max_grad(x, n_, a_),
                         bounds=bounds, options={'maxiter': maxiter_each,
                                                'ftol': 1e-20, 'gtol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except:
            pass
    return best_h, best_c

def fourier_init(n, freqs, phases, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for f, ph in zip(freqs, phases):
        h += np.cos(np.pi * f * x + ph)
    h = 0.5 + 0.3 * h / (np.abs(h).max() + 1e-10)
    return project_feasible(h, n)
