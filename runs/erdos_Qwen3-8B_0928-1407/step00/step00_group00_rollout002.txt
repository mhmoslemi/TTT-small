import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    current_sum = np.sum(h)
    if not np.isclose(current_sum, n / 2):
        h = h * (n / 2) / current_sum
    
    max_iter = 400
    iteration = 0

    best_h = h.copy()

    while (time.time() - start_time) < budget_s and iteration < max_iter:
        # Compute overlap for all shifts
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap_values = overlap * dx
        max_overlap = np.max(overlap_values)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        if np.abs(max_overlap - best_c5) < 1e-6:

        k_index = np.argmax(overlap_values)

        h_new = cp.Variable(n)
