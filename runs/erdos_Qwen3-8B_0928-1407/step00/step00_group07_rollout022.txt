import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(n_points, 0.5)
    desired_sum = n_points / 2
    current_sum = np.sum(h)
    scaling = desired_sum / current_sum if current_sum > 1e-6 else 1.0
    h = h * scaling
    h = np.clip(h, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h):
        h = h.copy()
        idxs = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.05, 0.05)
        sum_perturb = np.sum(h)
        scaling = desired_sum / sum_perturb
        h = np.clip(h * scaling, 0, 1)
        return h

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    pop_size = 100
    global_iter = 200
    for _ in range(global_iter):
        new_pop = [best_h.copy()]
        for _ in range(pop_size - 1):
            new_pop.append(perturb(best_h))
        evals = [(evaluate_c5(cand), cand) for cand in new_pop]
        evals.sort()
        best_c5, best_h = evals[0]

    learning_rate = 0.01
    max_steps = 500
    for _ in range(max_steps):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr)
        grads = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grads[i] = (1 - best_h[j]) * dx
        best_h -= learning_rate * grads
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        scaling = desired_sum / sum_h if sum_h > 1e-6 else 1.0
        best_h = best_h * scaling

    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
