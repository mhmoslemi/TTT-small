    step_size = 0.1
    decay_factor = 0.999

    for iter in range(max_iterations):
        # Compute overlaps using vectorized operations
        shifted_h = np.array([np.roll(h, k) for k in range(n_points)])
        overlaps = (np.sum(h[:, np.newaxis] * (1 - shifted_h), axis=1)) * dx
        current_c5 = np.max(overlaps)
