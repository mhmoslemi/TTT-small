        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h_clipped)
        factor = target_sum / current_sum
        return h_clipped * factor

    def evaluate_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * (2.0 / n_points)

    pop_size = 100
    max_global_iter = 200
    population = [h.copy() for _ in range(pop_size)]
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    for _ in range(max_global_iter):
        scores = []
        for indiv in population:
            c5 = evaluate_c5(indiv)
            scores.append((c5, indiv))
        scores.sort()
        elites = [x[1] for x in scores[:int(len(scores)*0.2)]]
        new_population = elites.copy()
        while len(new_population) < pop_size:
            a, b = np.random.choice(elites, 2, replace=False)
            child = (a + b) / 2.0
            noise = np.random.normal(0, 0.05, size=n_points)
            child += noise
            child = clip_and_rescale(child, target_sum)
            new_population.append(child)
        population = new_population
        for c5, indiv in scores:
            if c5 < best_c5:
                best_c5 = c5
                best_h = indiv.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    max_local_iter = 100
    learning_rate_local = 0.01
    for _ in range(max_local_iter):
        max_k_value = -1
        best_k = 0
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                val = best_h[i] * (1 - best_h[(i + k) % n_points])
                overlap += val
            overlap *= (2.0 / n_points)
            if overlap > max_k_value:
                max_k_value = overlap
                best_k = k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_k) % n_points
            gradient[i] = (1 - best_h[j]) * (2.0 / n_points)
        best_h -= learning_rate_local * gradient
        best_h = clip_and_rescale(best_h, target_sum)
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = clip_and_rescale(best_h, target_sum)
    return (best_h, best_c5, n_points)
