import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = None

    def interp_h(target_n, source_h):
        x_old = np.linspace(0, 2, len(source_h))
        x_new = np.linspace(0, 2, target_n)
        return np.interp(x_new, x_old, source_h)

    def init_h(use_initial=True):
        if use_initial and initial_h is not None:
            h = interp_h(n_points, initial_h)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h

    h = init_h()
    best_h, best_c5 = h.copy(), np.inf

    def compute_c5(h):
        cc = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cc)

    temp = 1.0
    cooling = 0.99
    learning_rate = 0.1
    time_used = 0.0

    while time_used < budget_s:
        c5 = compute_c5(h)

            best_c5 = c5
            best_h = h.copy()

        h -= learning_rate * grad

        if np.sum(h) != 0:

        else:
            h = np.zeros_like(h)
        temp *= cooling
