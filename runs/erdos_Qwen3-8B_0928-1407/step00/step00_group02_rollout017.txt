        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.full_like(h, 0.5)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    return (h.tolist(), final_c5, n_points)
