                      options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6})
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h, compute_overlap(h), n_points)
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Post-processing: ensure sum is correct and clip to [0,1]
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if abs(total - n_points / 2) > 1e-6:
        # Adjust to ensure sum constraint
        diff = (n_points / 2) - total
        best_h[np.argsort(best_h)[0]] += diff
        best_h = np.clip(best_h, 0, 1)
    
    return (best_h, best_c5, n_points)
