        else:
            k = max_k - n_points + 1
        # Map k to valid indices
        valid_k = np.arange(-n_points + 1, n_points)
        valid_k = valid_k[np.abs(valid_k) <= n_points - 1]
        conv_valid = conv_values[np.abs(valid_k) <= n_points - 1]
        k = valid_k[np.argmax(conv_valid)]
        offset = k + n_points - 1
        
        # Compute gradient approximation
        def obj_func(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        
        def grad_func(h):
            epsilon = 1e-4
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_perturbed = h.copy()
                h_perturbed[i] += epsilon
                corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
                conv = corr * dx
                grad[i] = (np.max(conv) - current_c5) / epsilon
            return grad
        
        # Constrained optimization
        constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(obj_func, h, method='L-BFGS-B', jac=grad_func, bounds=bounds, constraints=constraints)
        h = result.x
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    corr_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_best)
    return best_h, final_c5, n_points
