            h = h * (n_points / 2.0) / current_sum
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Reduce step size if progress stalls
        if it % 100 == 0:
            step_size *= 0.9995
    
    return (best_h, best_c5, n_points)
