        alpha = 0.01 / (1 + _ * 0.01)  # Decaying learning rate
        h -= alpha * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2.0) / sum_h
        h *= scale
        # Update best solution
        current_c5 = max(1 - np.dot(h, h[:(n_points - k) % n_points]) * dx for k in range(n_points))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)
