    def evaluate_erdos_solution(h):
        n = len(h)
        dx = 2.0 / n
        padded = np.zeros(2 * n)
        padded[:n] = h
        corr = np.correlate(padded, 1 - padded, mode="full")

    def objective(x):

        c5 = evaluate_erdos_solution(x)
