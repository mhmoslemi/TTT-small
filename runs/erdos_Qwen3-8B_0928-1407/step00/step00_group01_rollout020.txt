        h_new = h - 0.01 * grad
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize sum
        sum_h_new = np.sum(h_new)
        if sum_h_new != 0:
            h_new = h_new / sum_h_new * (n_points / 2)
        h = h_new
        
        # Evaluate true C5
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < min_c5:
            min_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, min_c5, n_points)
