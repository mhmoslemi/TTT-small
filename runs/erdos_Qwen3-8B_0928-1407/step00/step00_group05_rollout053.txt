        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for i in indices:
            h[i] += np.random.normal(0, sigma)
            h[i] = np.clip(h[i], 0.0, 1.0)
        s = np.sum(h)
        scale = T / s
        h = h * scale
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Check against worst in HM
        h_evals = np.array([evaluate(h) for h in HM])
        worst_idx = np.argmax(h_evals)
        if c5 < h_evals[worst_idx]:
            HM[worst_idx] = h.copy()
        sigma *= 0.99

    return (best_h, best_c5, n_points)
