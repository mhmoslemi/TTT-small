import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if 'initial_h_values' in locals():
        h = np.array(initial_h_values)
    else:
        n_points = 80
        h = np.full(n_points, 0.5)
    if np.sum(h) != n_points / 2:
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    n_points = len(h)
    dx = 2.0 / n_points

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale

    def compute_c5(h_val):
        inv_h = 1.0 - h_val
        max_overlap = 0.0
        for k in range(n_points):
            curr = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                curr += h_val[i] * inv_h[j]
            if curr > max_overlap:
                max_overlap = curr
        return max_overlap * dx

    population_size = 100
    global_iter = 200
    local_iter = 200
    learning_rate = 0.01
    time_budget = budget_s

    # Global exploration phase
    population = [h.copy() for _ in range(population_size)]
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    for _ in range(global_iter):
        scores = [compute_c5(p) for p in population]
        top_idx = np.argsort(scores)[:int(population_size * 0.2)]
        top_candidates = [population[i] for i in top_idx]
        new_population = []
        for c in top_candidates:
            perturb = np.random.normal(0, 0.05, n_points)
            mutated = c + perturb
            mutated = clip_rescale(mutated)
            new_population.append(mutated)
        population = new_population
        curr_scores = [compute_c5(p) for p in population]
        best_idx = np.argmin(curr_scores)
        if curr_scores[best_idx] < best_c5:
            best_c5 = curr_scores[best_idx]
            best_h = population[best_idx].copy()
        if time.time() - start_time > time_budget - 500:
            break

    # Local fine-tuning
    current_h = best_h.copy()
    for _ in range(local_iter):
        inv_h = 1.0 - current_h
        corr = np.zeros(n_points)
        for k in range(n_points):
            val = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                val += current_h[i] * inv_h[j]
            corr[k] = val
        best_k = np.argmax(corr)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_k) % n_points
            gradient[i] = (1.0 - current_h[j]) * dx
        updated = current_h - learning_rate * gradient
        updated = clip_rescale(updated)
        new_c5 = compute_c5(updated)
        if new_c5 < best_c5:
            best_c5 = new_c5
            current_h = updated.copy()
        if time.time() - start_time > time_budget - 500:
            break

    final_h = clip_rescale(current_h)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
