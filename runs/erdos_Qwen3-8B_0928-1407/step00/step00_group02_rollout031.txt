        if time.time() - time_start > budget:
            break
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        grad = np.zeros(len(best_h))
        alpha = 1e-4
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += alpha
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus)
            h_minus = best_h.copy()
            h_minus[i] -= alpha
            h_minus = np.clip(h_minus, 0, 1)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * alpha)
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        total_h = np.sum(best_h)
        if abs(total_h - n_points / 2) > 1e-6:
            idx = np.random.randint(n_points)
            best_h[idx] += (n_points / 2 - total_h)
            best_h = np.clip(best_h, 0, 1)
        best_solution = (best_h.copy(), best_c5, n_points)
    
    return best_solution
