import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def obj(h_vec):
        h = np.array(h_vec)
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = obj(current_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        result = minimize(obj, current_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        current_h += noise
        current_h = np.clip(current_h, 0.0, 1.0)

    final_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = obj(final_h)
    return (final_h, final_c5, n_points)
