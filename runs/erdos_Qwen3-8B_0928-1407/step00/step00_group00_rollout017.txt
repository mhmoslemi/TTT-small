import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    cc = np.correlate(h, one_minus_h, mode='full')
    current_c5 = np.max(cc) * dx
    best_h = h.copy()
    best_c5 = current_c5
    iteration_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        iteration_count += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        one_minus_h = 1.0 - h
        cc = np.correlate(h, one_minus_h, mode='full')
        new_c5 = np.max(cc) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        if iteration_count % 1000 == 0:
            ones_indices = np.where(h == 1)[0]
            np.random.shuffle(ones_indices)
            temp_h = np.zeros(n_points)
