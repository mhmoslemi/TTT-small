import numpy as np
import scipy.optimize as optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_minus = 1 - h
    corr = np.correlate(h, h_minus, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 70
        h_initial = np.full(n_points, 0.5)
    n_points = len(h_initial)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    start_time = time.time()
    
    def objective(h):
        return compute_c5(h, n_points)
    
    result = optimize.minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    elapsed = time.time() - start_time
    
    if elapsed >= budget_s:
        return (np.clip(result.x, 0, 1), result.fun, n_points)
    
    # Perturbation
    h_perturb = h_initial + 0.05 * np.random.randn(n_points)
    h_perturb = np.clip(h_perturb, 0.0, 1.0)
    result_perturb = optimize.minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (np.clip(result_perturb.x, 0, 1), result_perturb.fun, n_points)
    
    best_result = result if result.fun < result_perturb.fun else result_perturb
    final_h = np.clip(best_result.x, 0.0, 1.0)
    final_c5 = compute_c5(final_h, n_points)
    
    return (final_h, final_c5, n_points)
