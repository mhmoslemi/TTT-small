        corr = correlate(h_vec, one_minus_h, mode='full')
        c5_candidate = np.max(corr) * dx
        return c5_candidate

    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
    ]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization settings
    max_iter = 200  # Control the total number of iterations
    timeout = budget_s
    
    # Track the best solution
    best_result = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for iter in range(max_iter):
        start_time = time.time()
        result = minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        elapsed_time += time.time() - start_time
        
        # Update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_result = result.x
            if elapsed_time >= timeout:
                break
            x0 = result.x
        else:
            break
    
    # Ensure some improvements are made
    if best_result is None:
        best_result = x0
    
    # Final validation
    final_h = np.clip(best_result, 0.0, 1.0)
    final_c5 = objective(final_h)
    return (final_h.tolist(), final_c5, n_points)
