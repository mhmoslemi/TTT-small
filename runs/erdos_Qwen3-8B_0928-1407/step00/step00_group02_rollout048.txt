        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr.max() * dx
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for m in range(n_points):
            if 0 <= m + lag < n_points:
                grad[m] += (1 - h[m + lag]) * dx
            if 0 <= m - lag < n_points:
                grad[m] -= h[m - lag] * dx

        # Update h
        h -= learning_rate * grad

        # Clip and scale
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            h = h_clipped

            scale = target_sum / current_sum
