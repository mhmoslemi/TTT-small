import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points

    def initialize_h(use_parent=True):
        if use_parent and 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_old = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_old, h)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h

    h = initialize_h()
    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(h):
        cc = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cc)

    for _ in range(5):  # 5 multi-starts
        start_time = time.time()
        h = initialize_h()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        while time.time() - start_time < budget_s * 0.2:
            # Simulated annealing
            T = max(0.1, np.random.rand() * 0.5)
            noise = np.random.rand(n_points) * 2 * T - T
            h += noise
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            new_c5 = compute_c5(h)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                if np.random.rand() < np.exp((current_c5 - new_c5) / T):
                    h = h.copy()
        # Gradient descent on best_h
        for _ in range(100):
            cc = np.correlate(best_h, 1 - best_h, mode='full') * dx
            max_cc = np.max(cc)
            max_indices = np.where(cc == max_cc)[0]
            grad = np.zeros(n_points)
            for k in max_indices:
                shift = k - (n_points - 1)
                grad += np.roll(1 - best_h, shift)
            grad = (grad - np.mean(grad)) / (np.std(grad) + 1e-8)
            best_h -= 0.01 * grad

            best_h *= (n_points / 2) / np.sum(best_h)
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
    return (best_h, best_c5, n_points)
