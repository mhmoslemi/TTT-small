import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float('inf')

    def obj(x):
        h_flat = x.reshape(n)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
    bounds = [(0, 1)] * n

    def perturb(x):
        x = x + np.random.normal(0, 0.02, n)
        x = np.clip(x, 0, 1)
        sm = np.sum(x)
        if sm == 0:
            return np.zeros(n)
        return x * (n / 2) / sm

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    best_h = result.x
    best_c = obj(best_h)

    while time.time() - start_time < budget_s:
        h_pert = perturb(best_h)
        result = minimize(obj, h_pert, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if obj(result.x) < best_c:
            best_c = obj(result.x)
            best_h = result.x
        time.sleep(0.1)

    best_h = np.clip(best_h, 0, 1)
    sm = np.sum(best_h)
    if sm != 0:
        best_h *= (n / 2) / sm
    return tuple(best_h), best_c, n
