                new_pop[i][idx1] = np.clip(new_pop[i][idx1] - delta, 0, 1)
                new_pop[i][idx2] = np.clip(new_pop[i][idx2] + delta, 0, 1)
                sum_c = np.sum(new_pop[i])
                target = n_points / 2
                new_pop[i] *= (target / sum_c) if sum_c != 0 else 1
                new_pop[i] = np.clip(new_pop[i], 0, 1)
        
        population = new_pop
        
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
