import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = np.array(initial_h_values)
    else:
        n_points = 500
        h_initial = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h_initial)
        h_initial *= scale
        h_initial = np.clip(h_initial, 0, 1)
    
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Initialize HM
    hm_size = 30
    hm = [h_initial.copy()]
    for _ in range(hm_size - 1):
        candidate = h_initial.copy()
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.1)
            candidate[i] += delta
        candidate = np.clip(candidate, 0, 1)
        total = np.sum(candidate)
        scale = (n_points / 2) / total
        candidate *= scale
        hm.append(candidate)
    
    # Evaluate function
    def calculate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Sort HM by C5
    hm = sorted(hm, key=calculate_c5)
    best_c5 = calculate_c5(hm[0])
    best_h = hm[0]
    start_time = time.time()
    sigma = 0.1
    sigma_decay = 0.99
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb a candidate from HM
        candidate = hm[np.random.randint(len(hm))]
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            delta = np.random.normal(0, sigma)
            candidate[i] += delta
        candidate = np.clip(candidate, 0, 1)
        total = np.sum(candidate)
        scale = (n_points / 2) / total
        candidate *= scale
        
        # Evaluate
        c5 = calculate_c5(candidate)
        
        # Update HM

            hm[-1] = candidate
            hm.sort(key=calculate_c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
        
        # Adjust sigma

    return (best_h, best_c5, n_points)
