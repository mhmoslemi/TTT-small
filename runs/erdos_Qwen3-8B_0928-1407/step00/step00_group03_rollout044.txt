import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2) / total
    h *= scale
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()
    
    for _ in range(5):
        def objective(x):
            return evaluate_erdos_solution(x, n_points)
        bounds = [(0, 1)] * n_points
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-5, 'gtol': 1e-5})
        h = result.x
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 100:
            break
    
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, final_c5, n_points)
