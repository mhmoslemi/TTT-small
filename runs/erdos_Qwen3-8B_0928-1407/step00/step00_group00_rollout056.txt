        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5 = compute_c5(h)
    best_h = h.copy()
    max_iter = 500
    elapsed = 0

    for iter in range(max_iter):
        start_time = time.time()
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < c5:
            c5 = current_max
            best_h = h.copy()
        k_max = np.argmax(corr)
        shifted_indices = (np.arange(n_points) + k_max) % n_points

        h_new = cp.Variable(n_points)
        term = h_new * (1 - h_new[shifted_indices])
        obj = cp.Minimize(cp.sum(term) * dx)
        constraints = [0 <= h_new, h_new <= 1, cp.sum(h_new) == total_mass]
        try:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.SCS, verbose=False)
        except:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status not in ['OPTIMAL', 'INFEASIBLE']:
            break
        direction = h_new.value - h
        gamma = 0.5
        h = h + gamma * direction
        h = np.clip(h, 0, 1)
        h = (h - np.min(h)) / (np.max(h) - np.min(h)) * total_mass
        h = h / np.sum(h) * total_mass
        elapsed += time.time() - start_time
        if elapsed > budget_s:
            break

    final_c5 = compute_c5(h)
    return (h, final_c5, n_points)
