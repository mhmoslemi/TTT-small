import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    cross = np.correlate(h, 1 - h, mode='full')
    return np.max(cross) * dx

def run(seed=42, budget_s=1100, **kwargs):
    import random
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=float)
    
    best_c5 = compute_c5(h_initial, n_points)
    best_h = h_initial.copy()
    
    start_time = time.time()
    
    # Multi-start optimization with perturbations
    for _ in range(5):  # 5 restarts for robustness
        if time.time() - start_time > budget_s:
            break
        
        # Perturb initial solution
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h_initial + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Define objective function
        def objective(x):
            return compute_c5(x, n_points)
        
        # Define constraints
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        # Update best solution
        if result.success:
            current_c5 = compute_c5(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        else:
            current_c5 = compute_c5(h_perturbed, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()
    
    # Final check to ensure constraints are satisfied
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-5:
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, n_points)
    
    return (best_h, best_c5, n_points)
