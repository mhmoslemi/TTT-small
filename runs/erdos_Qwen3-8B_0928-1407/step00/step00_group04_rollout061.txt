            grad[m] = (corr_p[k_idx] - corr_m[k_idx]) / (2 * epsilon)
        # Solve linear subproblem
        sorted_indices = np.argsort(grad)
        s = np.zeros(n)
        remaining = n / 2
        for idx in sorted_indices:
            if remaining <= 0:
                break
            if grad[idx] < 0:
                s[idx] = 1
                remaining -= 1
        # Update h
        gamma = 0.5
        new_h = best_h + gamma * (s - best_h)
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * (n / 2) / sum_new
        # Evaluate
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    return (best_h, best_c5, n)
