import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        initial_h_values = np.clip(initial_h_values * scale, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, h, mode='full')
        return np.max(1 - corr * dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h0 = initial_h_values.copy()
    
    start_time = time.time()
    
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (initial_h_values, objective(initial_h_values), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
