import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h) if initial_h is not None else 70
    D = n_points / 2.0

    def mutate(h):
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        perturbation = np.random.normal(0, 0.1, len(indices))
        h = h.copy()
        h[indices] += perturbation
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != D:
            scale_factor = D / current_sum
            h = np.clip(h * scale_factor, 0, 1)
        return h

    HM = [np.array(initial_h, copy=True) if initial_h is not None else np.random.rand(n_points)]
    for _ in range(30 - 1):
        HM.append(mutate(HM[0]))

    best_h = HM[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start = time.time()
    while time.time() - start < budget_s:
        idx = np.random.randint(len(HM))
        current_h = HM[idx].copy()
        mutated_h = mutate(current_h)
        c5 = evaluate_erdos_solution(mutated_h)

        worst_index = np.argmin([evaluate_erdos_solution(h) for h in HM])
        if c5 < evaluate_erdos_solution(HM[worst_index]):

        if c5 < best_c5:
            best_h = mutated_h
