    if total > 0:
        h *= (n_points / 2) / total
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        dx = 2.0 / n_points
        shifts = np.roll(h, range(n_points))
        dot_products = np.dot(h, shifts.T)
        return np.max(1.0 - dx * dot_products)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8, 'maxfun': 1000})
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return result.x, result.fun, n_points
    
    h_best = result.x
    c5 = result.fun
    
    return h_best, c5, n_points
