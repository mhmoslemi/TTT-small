                child = alpha * p1 + (1 - alpha) * p2
                child = repair(child)
            else:
                child = p1.copy()
            children.append(child)
        
        # Mutation
        for child in children:
            if random.random() < 0.1:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                delta = np.random.uniform(-0.1, 0.1)
                child[idx1] += delta
                child[idx2] -= delta
                child = np.clip(child, 0, 1)
                child = repair(child)
        
        # Combine and select new population
        combined = population + children
        combined_fitness = [evaluate_erdos_solution(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        new_population = [combined[i] for i in sorted_indices[:population_size]]
        population = new_population
    
    return (best_h, best_c5, n_points)
