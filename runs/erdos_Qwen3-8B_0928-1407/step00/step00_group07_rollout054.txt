        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
        time_left = budget_s - (time.time() - start_time)
    
    for _ in range(100):
        if time_left <= 0:
            break
        max_overlap = -1
        best_k = 0
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += best_h[i] * (1 - best_h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_k) % n_points
            gradient[i] = (1 - best_h[j]) * dx
        best_h -= 0.01 * gradient
        best_h = np.clip(best_h, 0, 1)
        s = np.sum(best_h)
        if s != 0:
            best_h = best_h * (n_points / 2) / s
        time_left = budget_s - (time.time() - start_time)
    
    return (best_h, best_c5, n_points)
