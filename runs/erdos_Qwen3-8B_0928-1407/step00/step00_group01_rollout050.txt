        for i in range(len(h)):
            h[i] += np.random.normal(0, 0.05)
        return ensure_sum(h)

    def crossover(p1, p2):
        alpha = np.random.rand()
        child = alpha * p1 + (1 - alpha) * p2
        child = np.clip(child, 0, 1)
        child = ensure_sum(child)
        return child

    def mutate(h):
        i, j = np.random.choice(n_points, 2, replace=False)
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            return h
        delta = np.random.uniform(0, max_delta)
        h[i] -= delta
        h[j] += delta
        return h

    def select_parents(fitness):
        parents = []
        for _ in range(len(population) // 2):
            tournament = np.random.choice(len(population), 3, replace=False)
            idx = tournament[np.argmax(fitness[tournament])]
            parents.append(population[idx])
        return parents

    def replace_worst(pop, offspring):
        new_pop = np.concatenate((pop, offspring))
        fit = np.array([-compute_c5(h) for h in new_pop])
        indices = np.argsort(fit)
        return new_pop[indices[:len(pop)]]

    population = []
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = ensure_sum(h)
        population.append(h)
        for _ in range(99):
            population.append(perturb(h))
    else:
        population = [generate_random_h() for _ in range(100)]

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fit = np.array([-compute_c5(h) for h in population])
        parents = select_parents(fit)
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            child = crossover(p1, p2)
            offspring.append(child)
        if len(parents) % 2 == 1:
            offspring.append(population[np.random.randint(len(population))])
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i])
        population = replace_worst(population, offspring)
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return best_h, best_c5, n_points
