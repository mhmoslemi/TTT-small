        if not improved and _ % 500 == 0 and _ > 0:
            perturb = np.random.uniform(-0.01, 0.01, size=n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = target_sum / sum_h
            h = h * scale
        if _ > 1000 and np.abs(current_c5 - best_c5) < 1e-6:
            break
    return (h, best_c5, n_points)
