    h = h * scaling

    best_c5 = compute_c5(h, n, dx)
    start_time = time.time()
    
    for _ in range(500):  # Limit iterations to fit within budget
        # Compute overlaps
        overlaps = []
        for k in range(n):
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted) * dx
            overlaps.append(overlap)
        max_overlap = max(overlaps)
        k_idx = overlaps.index(max_overlap)
        
        # Gradient for the maximum shift
        grad = np.zeros(n)
        for i in range(n):
            j = (i + k_idx) % n
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Gradient descent step
        learning_rate = 0.05 * (1 / (1 + _))
        h = h - learning_rate * grad
        
        # Clamp and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n / 2) / h_sum
        
        # Update best solution
        current_c5 = compute_c5(h, n, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n)

def compute_c5(h, n, dx):
    overlaps = []
    for k in range(n):
        shifted = np.roll(h, k)
        overlap = np.dot(h, 1 - shifted) * dx
        overlaps.append(overlap)
    return max(overlaps)
