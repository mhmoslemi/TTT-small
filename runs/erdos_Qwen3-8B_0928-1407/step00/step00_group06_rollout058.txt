import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    T = n_points / 2
    
    # Initialize h with scaling to ensure ∑h * dx = T
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h) * dx
    if abs(sum_h - T) > 1e-8:
        scale = T / sum_h
        h = h * scale
    
    best_c5 = float('inf')
    start_time = time.time()
    max_iter = int(budget_s / 0.1)
    
    for _ in range(max_iter):
        # Compute max C5 and corresponding lag
        max_c5 = 0.0
        best_k = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            integral = np.dot(h, 1 - shifted) * dx
            if integral > max_c5:
                max_c5 = integral
                best_k = k
        
        if max_c5 < best_c5:
            best_c5 = max_c5
        
        # Compute gradient for best_k
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = (j + best_k) % n_points
            j_minus = (j - best_k) % n_points
            grad[j] = (1 - h[j_plus]) - h[j_minus]
        
        # Gradient descent step
        alpha = 0.05
        h_new = h - alpha * grad
        
        # Clip to [0, 1]
        h_clipped = np.clip(h_new, 0, 1)
        
        # Adjust sum to T
        sum_clipped = np.sum(h_clipped) * dx
        delta = (T - sum_clipped) / dx
        
        if delta > 0:
            # Distribute delta to under-filled elements
            for i in range(n_points):
                if delta <= 0:
                    break
                add = min(delta, 1 - h_clipped[i])
                h_clipped[i] += add
                delta -= add

            # Remove excess from over-filled elements

                if delta >= 0:
                    break
                rem = min(-delta, h_clipped[i])
                h_clipped[i] -= rem
                delta += rem
        
        h = h_clipped
        
        # Early exit if time is up
        if time.time() - start_time > budget_s:
            break
    
    return (h, best_c5, n_points)
