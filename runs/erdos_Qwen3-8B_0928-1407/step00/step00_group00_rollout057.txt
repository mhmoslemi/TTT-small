import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)

    step_size = 0.5

    while time.time() - start_time < budget_s and iter < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[:n_points]
        max_idx = np.argmax(overlaps)
        k = max_idx
        current_c5 = overlaps[max_idx] * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1 - h[j]

        delta_h = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(grad * delta_h))
        constraints = [
            cp.sum(delta_h) == 0,
            0 <= h + delta_h,
            h + delta_h <= 1
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve()

        if prob.status == 'optimal':
            h += delta_h.value * step_size
        iter += 1

    return (best_h, best_c5, n_points)
