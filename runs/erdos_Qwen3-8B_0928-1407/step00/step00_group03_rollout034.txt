import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 140
    dx = 2.0 / n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    h_initial = np.interp(x_new, x_initial, initial_h_values)
    
    def compute_c5(h):
        ones_minus_h = 1 - h
        conv = np.correlate(h, ones_minus_h, mode='full')
        max_conv = np.max(conv)
        return max_conv * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun
