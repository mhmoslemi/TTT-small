import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h and n_points
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 500
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    assert np.isclose(np.sum(h), n_points / 2), "Invalid initial sum"
    
    def compute_c5(h, dx):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    iterations = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i, j = np.random.choice(ones), np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        curr_c5 = compute_c5(h, dx)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        if iterations % 1000 == 0:
            ones = np.where(h == 1)[0]
            np.random.shuffle(ones)
            h[ones] = 1.0
            h[~np.isin(np.arange(n_points), ones)] = 0.0
            curr_c5 = compute_c5(h, dx)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
