import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        h *= scale
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    def update_sum(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h) + 0.5
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def adjust_h(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_shift = max_index - n_points + 1
        k_shift %= n_points
        contrib = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_shift) % n_points
            contrib[i] = h[i] * (1 - h[j])
        top_indices = np.argsort(contrib)[::-1][:5]
        h = h.copy()
        for idx in top_indices:
            h[idx] = np.clip(h[idx] + np.random.choice([-0.05, 0.05]), 0, 1)
        h = update_sum(h)
        return h

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    for _ in range(int(budget_s / (n_points * 2))):
        h = adjust_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s:
            break
