            new_corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
            new_max = np.max(new_corr)
            if new_max < max_c5:
                h = new_h.copy()
                max_c5 = new_max
                alpha *= 2
            else:
                alpha /= 2

    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s != 0:
        h *= desired_sum / s
    h = np.clip(h, 0, 1)
    final_corr = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(final_corr)
    return h, final_c5, n_points
