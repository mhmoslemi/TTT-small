import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Check if initial_h_values is available
    if 'initial_h_values' in locals() and initial_h_values is not None:
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 44  # Ensure even number of points for sum/2
        h = np.zeros(n_points)
        h[::2] = 1  # Evenly spaced ones
    # Ensure sum is exactly n_points / 2
    assert np.isclose(np.sum(h), n_points / 2)
    dx = 2.0 / n_points
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = h.copy()
    best_C5 = compute_C5(h)
    start_time = time.time()
    iterations = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Find indices of 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Randomly select a 1 and a 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        original_i = h[i]
        original_j = h[j]
        h[i] = 0
        h[j] = 1
        # Compute new C5
        new_C5 = compute_C5(h)
        # Check if it's better
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
        else:
            # Revert swap
            h[i] = original_i
            h[j] = original_j
        # Periodic shuffle
        iterations += 1
        if iterations % 1000 == 0:
            # Global shuffle
            all_indices = np.arange(n_points)
            np.random.shuffle(all_indices)
            new_ones = all_indices[:n_points // 2]
            h[:] = 0
            h[new_ones] = 1
    # Final check to ensure best_h is valid
    assert np.isclose(np.sum(best_h), n_points / 2)
    return (best_h.tolist(), best_C5, n_points)
