import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy().astype(float)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    perturb_count = 0
    max_perturb = 5

    def objective(h_vals):
        return evaluate_erdos_solution(h_vals)

    constraint = opt.LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = opt.Bounds(0, 1)

    while time.time() - start_time < budget_s:
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success and np.isclose(result.fun, best_c5, atol=1e-5):
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x.copy()
            h = result.x.copy()
        else:
            perturb_count += 1
            if perturb_count < max_perturb:
                noise = np.random.normal(0, 0.01, size=n_points)
                h += noise
                h = np.clip(h, 0, 1)
            else:
                break

    h = np.clip(best_h, 0, 1)
    return (h, best_c5, n_points)
