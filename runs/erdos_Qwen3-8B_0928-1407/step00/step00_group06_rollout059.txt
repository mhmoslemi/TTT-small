    target_sum = n_points / 2
    best_c5 = np.inf
    best_h = h.copy()
    alpha = 0.01
    max_iterations = 10000
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Compute overlaps using FFT-based cyclic cross-correlation
        g = 1.0 - h
        fft_h = fft.fft(h)
        fft_g = fft.fft(g)
        product = fft_h * np.conj(fft_g)
        cc = fft.ifft(product).real
        overlapped = cc * dx
        current_max = np.max(overlapped)
        
        # Track best solution
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
        
        # Find worst lag
        max_index = np.argmax(overlapped)
        lag = max_index
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + lag) % n_points
            i_minus_k = (i - lag) % n_points
            grad[i] = (1.0 - h[i_plus_k]) - h[i_minus_k]
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != target_sum:
            h = h * (target_sum / sum_h)
    
    return (best_h, best_c5, n_points)
