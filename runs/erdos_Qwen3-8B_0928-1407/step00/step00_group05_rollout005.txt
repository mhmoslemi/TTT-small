        h_initial = np.clip(h_initial, 0, 1)
    
    HM_size = 30
    HM = [h_initial]
    for _ in range(HM_size - 1):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed = h_initial + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)
        HM.append(perturbed)
    
    def evaluate_c5(h):
        h_shift = 1.0 - h
        corr = np.correlate(h, h_shift, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    HM_scores = [evaluate_c5(h) for h in HM]
    HM_sorted = [h for _, h in sorted(zip(HM_scores, HM))]
    HM_scores.sort()
    
    best_h = HM_sorted[0]
    best_c5 = HM_scores[0]
    
    sigma = 0.1
    decay_rate = 0.98
    mutation_rate = 0.1
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        idx = np.random.randint(len(HM_sorted))
        candidate = HM_sorted[idx]
        num_perturb = int(mutation_rate * n_points)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        perturbed = candidate.copy()
        perturbed[indices] += np.random.normal(0, sigma, num_perturb)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)
        new_c5 = evaluate_c5(perturbed)
        if new_c5 < HM_scores[-1]:
            HM_sorted.pop()
            HM_scores.pop()
            HM_sorted.append(perturbed)
            HM_scores.append(new_c5)
            HM_sorted.sort(key=lambda x: evaluate_c5(x))
            HM_scores.sort()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = perturbed
        sigma *= decay_rate
    
    return (best_h, best_c5, n_points)
