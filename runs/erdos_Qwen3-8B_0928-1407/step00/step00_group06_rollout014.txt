import numpy as np
import time

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target_sum = n_points / 2
    if np.isclose(s, target_sum, atol=1e-8):
        return h
    if s < target_sum:
        # Increase some elements
        mask = h < 1
        if np.any(mask):
            factor = target_sum / s
            h[mask] *= factor
    else:
        # Decrease some elements
        mask = h > 0
        if np.any(mask):
            factor = target_sum / s
            h[mask] *= factor
    return h
