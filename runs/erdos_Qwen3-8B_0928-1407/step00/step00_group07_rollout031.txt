    start_time = time.time()

    for _ in range(200):
        f = [compute_c5(p) for p in population]
        top_idx = np.argsort(f)[:int(len(population)*0.2)]
        parents = [population[i] for i in top_idx]
        new_pop = []
        for _ in range(len(population)//2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.1:
                child += np.random.uniform(-0.1, 0.1, len(child))
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                if s:
                    child *= (n_points / 2) / s
            new_pop.append(child)
        population = new_pop
        for p in population:
            current = compute_c5(p)
            if current < best_c5:
                best_c5 = current
                best_h = p.copy()
        if time.time() - start_time > budget_s:
            break

    top = population[:20]
    learning_rate = 0.01
    for p in top:
        corr = np.correlate(p, 1 - p, mode='full') * dx
        max_k = np.argmax(corr)
        grad = np.zeros_like(p)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] = (1 - p[j]) * dx
        h_new = p - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        s = np.sum(h_new)
        if s:
            h_new *= (n_points / 2) / s
        current = compute_c5(h_new)
        if current < best_c5:
            best_c5 = current
            best_h = h_new.copy()
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s:
        best_h *= (n_points / 2) / s
    return (best_h, compute_c5(best_h), n_points)
