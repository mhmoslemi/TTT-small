import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    if not np.isclose(np.sum(initial_h), n_points / 2):
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    pop_size = 50
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturbed = initial_h.copy()
        perturbed += np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        pop.append(perturbed)
    
    def compute_c5(h):
        ones_minus_h = 1 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    generation = 0
    start_time = time.time()
    iter_without_improv = 0
    while time.time() - start_time < budget_s:
        pop_sorted = sorted(pop, key=compute_c5)
        parents = pop_sorted[:int(len(pop_sorted)*0.2)]
        offspring = []
        for _ in range(pop_size - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            mutate = np.random.rand(n_points) < 0.1
            child = child.copy()
            for i in range(n_points):
                if mutate[i]:
                    child[i] += np.random.normal(0, 0.1 * (1 - generation / 1000))

            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        new_pop = parents + offspring
        for h in new_pop:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        pop = new_pop
        if best_c5 < 0.3808 and generation > 100:
            break
        if abs(best_c5 - compute_c5(pop_sorted[0])) < 1e-8:
            iter_without_improv += 1
        else:
            iter_without_improv = 0
        if iter_without_improv >= 300:
            break
        generation += 1
    return (best_h, best_c5, n_points)
