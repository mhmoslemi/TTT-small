        if time.time() - time_start > budget_s:
            break
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(cross_corr)
        m = k_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_m = i + m
            if 0 <= i_plus_m < n_points:
