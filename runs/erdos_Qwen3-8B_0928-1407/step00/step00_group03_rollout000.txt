    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    best_h = h.copy()
    best_c5 = objective(best_h)
    while time.time() - start_time < budget_s - 10:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break
    return (best_h, best_c5, n_points)
