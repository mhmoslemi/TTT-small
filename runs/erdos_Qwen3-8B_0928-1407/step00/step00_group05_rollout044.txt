import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', np.random.rand(70) * 2 - 1)
    n_points = len(initial_h_values)
    required_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize HM
    HM = [initial_h_values.copy()]
    c5_values = [np.max(np.correlate(HM[0], 1 - HM[0], mode='full')) * dx]
    
    # Fill HM with 29 more candidates
    for _ in range(29):
        new_h = HM[0].copy()
        num_perturb = int(0.05 * n_points)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h[indices] += delta[indices]
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        scale = required_sum / current_sum
        new_h = np.clip(new_h * scale, 0, 1)
        HM.append(new_h)
        c5_values.append(np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx)

    best_h = HM[np.argmin(c5_values)]
    best_c5 = min(c5_values)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        idx = np.random.randint(len(HM))
        h = HM[idx].copy()
        num_perturb = int(0.05 * n_points)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.1 * (1 - (time.time() - start_time)/budget_s), 0.1 * (1 - (time.time() - start_time)/budget_s), size=n_points)
        h[indices] += delta[indices]
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = required_sum / current_sum
        h = np.clip(h * scale, 0, 1)
        
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

            c5_values[worst_idx] = c5
