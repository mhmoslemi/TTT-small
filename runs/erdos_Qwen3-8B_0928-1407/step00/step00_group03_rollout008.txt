import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    if 'initial_h_values' in globals():

    dx = 2.0 / n_points

        h_flat = x.copy()

        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = Bounds(0, 1)

    result = opt.minimize(
        fun=objective,
        x0=h,

        bounds=bounds,
