import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)
    total = np.sum(h_initial)
    if abs(total - n_points / 2) > 1e-6:
        scale = (n_points / 2) / total
        h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_flat = h_initial.flatten()
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr / dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(objective_func, h_flat, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    elapsed = time.time() - start_time
    
    if elapsed >= budget_s:
        best_h = result.x
        best_c5 = objective_func(best_h)
        return (best_h, best_c5, n_points)
    
    # Further refinement
    h_refine = result.x
    for _ in range(3):
        result = opt.minimize(objective_func, h_refine, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        h_refine = result.x
        elapsed += time.time() - elapsed
        if elapsed >= budget_s:
            break
    
    best_h = h_refine
    best_c5 = objective_func(best_h)
    return (best_h, best_c5, n_points)
