                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb *= (n_points / 2) / np.sum(h_perturb)
                cc_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full') * dx
                delta_C_perturb = np.max(cc_perturb) - current_C
                grad[i] -= delta_C_perturb / eps
            
            # Update with gradient descent
            current_h -= learning_rate * grad
            current_h = np.clip(current_h, 0, 1)
            if np.sum(current_h) != 0:
                current_h *= (n_points / 2) / np.sum(current_h)
            
            # Annealing
            learning_rate *= 0.99
            temp *= cool_rate
    
    return (best_h, best_C5, n_points)
