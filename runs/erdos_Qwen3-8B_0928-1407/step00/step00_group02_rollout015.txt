        new_h[-1] += delta
        new_h = np.clip(new_h, 0, 1)
        return new_h

    best_h = h_initial
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    best_n = len(best_h)
    h_list = [best_h]

    t0 = time.time()

    for _ in range(3):
        new_h_list = []
        for h in h_list:
            new_h = refiner(h, len(h))
            new_h_list.append(new_h)
        c5_list = [compute_c5(h, 2.0 / len(h)) for h in new_h_list]
        top = sorted(zip(c5_list, new_h_list), key=lambda x: x[0])[:50]
        h_list = [h for _, h in top]
        best_c5, best_h = min((c5, h) for c5, h in top)
        best_n = len(best_h)
        if time.time() - t0 > budget_s * 0.8:
            break

    def objective(x):
        h = x.reshape(-1)
        return compute_c5(h, 2.0 / len(h))

    def constraint(x):
        return np.sum(x) - len(x) / 2

    bounds = [(0, 1) for _ in range(len(best_h))]
    cons = {'type': 'eq', 'fun': constraint}

    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
    best_h = result.x
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    best_n = len(best_h)

    return best_h, best_c5, best_n
