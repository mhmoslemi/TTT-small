import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Initialize from provided initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points, dtype=np.float64)
        h[::2] = 1.0  # Initial uniform distribution of 1s
    
    # Ensure sum is correct
    if np.sum(h) != n_points / 2:
        h = np.zeros(n_points, dtype=np.float64)
        h[::2] = 1.0
    
    # Compute C5 function
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    iterations = 0
    early_stopping = 0
    
    while time.time() - start_time < budget_s:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            early_stopping = 0
        else:
            h[i], h[j] = h[j], h[i]
            early_stopping += 1

            np.random.shuffle(ones_indices)

            new_c5 = compute_c5(new_h)

                best_c5 = new_c5
        
        iterations += 1
        if early_stopping > 500:
