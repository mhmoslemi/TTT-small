        dominant_indices = np.where(cc >= current_C * 0.99)
        gradient = np.zeros_like(h)
        for j in dominant_indices[0]:
            k_abs = j - (n_points - 1)
            shift = k_abs
            h_shifted = np.roll(1 - h, shift)
            gradient += h_shifted
        gradient = gradient - np.mean(gradient)
        update = learning_rate * (-gradient) + temperature * np.random.randn(*h.shape)
        h += update
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h *= (n_points / 2) / h_sum
        else:
            h = np.zeros_like(h)
        temperature *= cooling_factor
        if (time.time() - start_time) % (budget_s / 10) == 0:
            learning_rate *= 0.95
        if (time.time() - start_time) % quake_interval == 0:
            h += 0.01 * np.random.randn(*h.shape)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h *= (n_points / 2) / h_sum
            else:
                h = np.zeros_like(h)
    cc_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_C = np.max(cc_best)
    return (best_h, final_C, n_points)
