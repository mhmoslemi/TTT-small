import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    
    # Repair initial h
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    factor = (n_points / 2) / sum_h if sum_h != 0 else 1
    h = h * factor
    h = np.clip(h, 0, 1)
    
    # Initial population
    population = [h]
    for _ in range(49):
        p = h + np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        sum_p = np.sum(p)
        if sum_p != 0:
            p *= (n_points / 2) / sum_p
        population.append(p)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()
    
    N_POP = 50
    TOURNAMENT_SIZE = 2
    MUTATION_RATE = 0.1
    SIGMA_START = 0.1
    SIGMA_DECAY = 0.98
    MAX_GENERATIONS = 1000
    NO_IMPROVEMENT = 300
    
    for gen in range(MAX_GENERATIONS):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        
        # Select parents
        parents = []
        for _ in range(N_POP):
            idxs = np.random.choice(N_POP, TOURNAMENT_SIZE, replace=False)
            winner = idxs[np.argmin(fitness[idxs])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(0, N_POP, 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < N_POP else parents[i]
            c1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
            c2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])
            sum_c1 = np.sum(c1)
            sum_c2 = np.sum(c2)
            if sum_c1 != 0:
                c1 = c1 * (n_points / 2) / sum_c1
            if sum_c2 != 0:
                c2 = c2 * (n_points / 2) / sum_c2
            offspring.extend([c1, c2])
        
        # Mutate
        sigma = SIGMA_START * (SIGMA_DECAY ** gen)
        for i in range(len(offspring)):
            o = offspring[i]
            m = o.copy()
            idxs = np.random.choice(n_points, size=int(MUTATION_RATE * n_points), replace=False)
            for j in idxs:
                m[j] += np.random.normal(0, sigma)
                m[j] = np.clip(m[j], 0, 1)
            sum_m = np.sum(m)
            if sum_m != 0:
                m *= (n_points / 2) / sum_m
            offspring[i] = m
        
        # Replace
        combined = population + offspring
        combined_fitness = fitness + [evaluate(h) for h in offspring]
        sorted_idx = np.argsort(combined_fitness)
        population = [combined[i] for i in sorted_idx[:N_POP]]
        
        # Update best

            current = evaluate(h)
            if current < best_c5:
                best_c5 = current
                best_h = h.copy()
        
        if gen >= 100 and best_c5 < 0.3808:
            break
    
    return (best_h, best_c5, n_points)
