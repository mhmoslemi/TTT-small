    while iteration < max_iterations and time.time() - time_start < budget_s:
        # Compute cross-correlation for all lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_max = np.max(corr) * dx
        
        # Update best C5 if improved
        if curr_max < best_c5:
            best_c5 = curr_max
            best_h = h.copy()
        
        # Find worst lag index and corresponding k
        worst_lag_index = np.argmax(corr)
        k = worst_lag_index - (n_points - 1)
        
        # Compute gradient for this lag
        idx = (np.arange(n_points) + k) % n_points
        grad = 1.0 - h[idx]
        
        # Perform gradient descent step
        h_new = h - alpha * grad
        
        # Clip to [0, 1] and adjust sum
        h_new = np.clip(h_new, 0.0, 1.0)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            scale = required_sum / current_sum
            h_new *= scale
        else:
            h_new = np.ones_like(h) * 0.5
        
        # Accept the update
        h = h_new
        
        # Random perturbation every 100 steps to escape local minima
        if iteration % 100 == 0 and np.random.random() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h)
            if current_sum > 0:
                h *= required_sum / current_sum
        
        # Adaptive learning rate
        if np.abs(curr_max - best_c5) < 1e-6:
            alpha *= 0.5
        else:
            alpha *= 1.01
        
        iteration += 1
    
    return best_h, best_c5, n_points
