import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    best_h = h
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()

    def objective_func(x):
        c5 = evaluate_erdos_solution(x, n_points)
        sum_h = np.sum(x)
        penalty = 1e4 * (sum_h - n_points / 2) ** 2
        return c5 + penalty

    bounds = [(0, 1) for _ in range(n_points)]

    # Global optimization
    if time.time() - start_time < budget_s * 0.6:
        result = differential_evolution(
            objective_func,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        h_candidate = result.x
        c5_candidate = evaluate_erdos_solution(h_candidate, n_points)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Local optimization
    if time.time() - start_time < budget_s * 0.8:
        result_LBFGS = minimize(
            lambda x: evaluate_erdos_solution(x, n_points),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-6
        )
        h_candidate = result_LBFGS.x
        c5_candidate = evaluate_erdos_solution(h_candidate, n_points)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
        # Ensure sum constraint
        sum_h = np.sum(best_h)
        if abs(sum_h - n_points / 2) > 1e-5:
            best_h = best_h / sum_h * (n_points / 2)
            best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, n_points)
