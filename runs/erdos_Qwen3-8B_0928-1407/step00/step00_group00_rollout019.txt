    # Function to compute overlaps and find max
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(100):  # Limited iterations due to time constraint
        if time.time() - start_time > budget_s:
            break
        
        # Compute current max overlap
        current_c5 = compute_c5(h)
        
        # Find the worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift_idx = max_corr_idx - (n_points - 1)
        
        # For this shift, create a new solution to reduce overlap
        # Binary LP: minimize overlap for this shift
        # Approximate by indicating which positions to set to 0 or 1
        # For simplicity, use gradient direction
        # Calculate gradient approximation for overlap
        # This is simplified, actual gradient is linear wrt h as per the overlap function
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + shift_idx) % n_points
            gradient[i] = (1 - h[j])  # Partial derivative of overlap w.r.t h[i]
        
        # Create new h with flipped values where gradient is highest
        # Select positions to switch between 0 and 1 to minimize overlap
        # Sort indices by gradient in descending order
        sort_indices = np.argsort(-gradient)
        for i in range(n_points):
            pos = sort_indices[i]
            val = h[pos]
            if val == 0:
                h[pos] = 1
            else:
                h[pos] = 0
        
        # Project to the feasible region
        # Ensure sum is exactly n_points / 2
        total = np.sum(h)
        if total > n_points / 2:
            # Set some 1's to 0
            diff = total - n_points / 2
            for i in range(n_points):
                if h[i] == 1:
                    h[i] = 0
                    diff -= 1
                    if diff <= 0:
                        break
        elif total < n_points / 2:
            # Set some 0's to 1
            diff = n_points / 2 - total
            for i in range(n_points):
                if h[i] == 0:
                    h[i] = 1
                    diff -= 1
                    if diff <= 0:
                        break
        
        # Update best solution if improved
        current_c5_new = compute_c5(h)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = h.copy()
    
    # Final check
    if best_c5 > compute_c5(best_h):
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
