    start_time = time.time()
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
    except:
        pass

    # Fallback to fine-tuning
    if best_c5 > 0.38:
        def perturb(x):
            x = x.copy()
            idx = np.random.choice(n_points)
            x[idx] = np.clip(x[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
            scale = (target_sum - np.sum(x)) / n_points
            x += scale
            return x

        for _ in range(100):
            h_candidate = perturb(best_h)
            obj = compute_c5(h_candidate)
            if obj < best_c5:
                best_h = h_candidate
                best_c5 = obj

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Final refinement with small steps
        for _ in range(50):
            h_candidate = best_h.copy()
            idx = np.random.choice(n_points)
            delta = np.random.uniform(-0.01, 0.01)
            h_candidate[idx] += delta
            scale = (target_sum - np.sum(h_candidate)) / n_points
            h_candidate += scale
            h_candidate = np.clip(h_candidate, 0, 1)
            obj = compute_c5(h_candidate)
            if obj < best_c5:
                best_h = h_candidate
                best_c5 = obj
    return best_h, best_c5, n_points
