import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():

    else:
        n_points = 44  # even, small n for efficiency
        h = np.zeros(n_points, dtype=float)
        h[::2] = 1.0  # alternate 1s and 0s to balance sum
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()
    iteration = 0
    shuffle_interval = 1000
    
    while time.time() - start_time < budget_s:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break  # no more swaps possible
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        
        # Recompute correlation and check
        corr = np.correlate(h, 1 - h, mode='full') * dx
        new_c5 = np.max(corr)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
        
        iteration += 1
        if iteration % shuffle_interval == 0:
            ones_indices = np.where(h == 1)[0]
            np.random.shuffle(ones_indices)
            h[ones_indices] = 1.0  # Ensure they remain 1s
    
    best_h = np.clip(best_h, 0, 1)
    return best_h.tolist(), best_c5, n_points
