    iteration = 0
    max_iterations = 100000
    
    def compute_c5(h):
        cc = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(cc)
    
    while time.time() - start_time < budget and iteration < max_iterations:
