        # Calculate overlap for this lag
        overlap_lag = overlap[lag]
        
        # Compute the gradient approximation for overlap_lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + lag < n_points:
                grad[i] += 1 - h[i + lag]
            if i - lag >= 0:
                grad[i + lag] -= h[i]
        
        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h if sum_h != 0 else h
        
        # Cool down learning rate
        learning_rate *= cooling
    
    return best_h, best_c5, n_points
