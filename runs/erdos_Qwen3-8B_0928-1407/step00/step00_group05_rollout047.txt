        perturbed = perturbed * (n_points / 2) / (total + 1e-12)
        population.append(perturbed)
    population = np.array(population)
    # Evaluate initial fitness
    fitness = []
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        fitness.append(c5)
    fitness = np.array(fitness)
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    # Parameters
    n_pop = 50
    mutation_rate = 0.1
    sigma = 0.1
    decay = 0.98
    generations = 0
    max_generations = 1000
    early_stop = 300
    improvement_count = 0
    start_time = time.time()
    while generations < max_generations and time.time() - start_time < budget_s:
        # Tournament selection
        selected = []
        for _ in range(n_pop):
            a, b = np.random.choice(n_pop, 2, replace=False)
            if fitness[a] < fitness[b]:
                selected.append(a)
            else:
                selected.append(b)
        # Crossover
        new_population = []
        for i in range(0, n_pop, 2):
            if i + 1 < n_pop:
                parent1 = population[selected[i]]
                parent2 = population[selected[i+1]]
                split = len(parent1) // 2
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                # Renormalize
                total1 = np.sum(child1)
                if total1 == 0:
                    child1 = np.zeros_like(child1)
                else:
                    child1 = child1 * (n_points / 2) / total1
                total2 = np.sum(child2)
                if total2 == 0:
                    child2 = np.zeros_like(child2)
                else:
                    child2 = child2 * (n_points / 2) / total2
                new_population.append(child1)
                new_population.append(child2)
        if len(new_population) < n_pop:
            idx = np.random.choice(n_pop)
            new_population.append(population[selected[idx]])
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i].copy()
            for j in range(len(h)):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.normal(0, sigma)
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if total == 0:
                h = np.zeros_like(h)
            else:
                h = h * (n_points / 2) / total
            new_population[i] = h
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            new_fitness.append(c5)
        new_fitness = np.array(new_fitness)
        # Combine and select
        combined = np.vstack([population, new_population])
        combined_fitness = np.hstack([fitness, new_fitness])
        indices = np.argsort(combined_fitness)
        population = combined[indices[:n_pop]]
        fitness = combined_fitness[indices[:n_pop]]
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            improvement_count = 0
        else:
            improvement_count += 1
        if improvement_count >= early_stop:
            break
        generations += 1
        sigma *= decay
    return (best_h, best_c5, n_points)
