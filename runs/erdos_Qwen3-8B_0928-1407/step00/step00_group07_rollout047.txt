import numpy as np
import numpy.random as npr
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    npr.seed(seed)
    # Check initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 80
        h_initial = np.full(n_points, 0.5)
    n_points = len(h_initial)
    assert np.isclose(np.sum(h_initial), n_points / 2), "Initial sum invalid"

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    time_used = 0

    # Global phase
    pop_size = 50
    generations = 200
    population = [h_initial.copy() for _ in range(pop_size)]
    while time_used < budget_s:
        scores = [compute_c5(ind) for ind in population]
        elite_idx = np.argsort(scores)[:int(0.2 * pop_size)]
        elites = [population[i] for i in elite_idx]
        new_population = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(elites, 2)
            child = (p1 + p2) / 2
            noise = npr.normal(0, 0.05, size=len(child))
            child += noise
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            scale = (n_points / 2) / current_sum if current_sum != 0 else 0.5
            child *= scale
            new_population.append(child)
        population = new_population
        time_used += 100
        if time_used >= budget_s:
            break
        for ind in population:
            c = compute_c5(ind)
            if c < best_c5:
                best_c5 = c
                best_h = ind.copy()

    # Local refinement
    learning_rate = 0.01
    for _ in range(100):
        if time_used >= budget_s:
            break
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_star = np.argmax(corr)
        grad = np.zeros_like(best_h)
        for i in range(len(best_h)):
            j = (i + k_star) % len(best_h)
            grad[i] = (1 - best_h[j]) * (2.0 / len(best_h))
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        current_sum = np.sum(best_h)
        scale = (n_points / 2) / current_sum if current_sum != 0 else 0.5
        best_h *= scale
        time_used += 10

    return (best_h, best_c5, n_points)
