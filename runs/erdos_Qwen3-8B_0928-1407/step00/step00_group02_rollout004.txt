import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.clip(initial_h_values, 0, 1).copy()
    sum_h_initial = sum(h_initial)
    if abs(sum_h_initial - n_points / 2) > 1e-6:
        h_initial = h_initial * (n_points / 2) / sum_h_initial
    h_initial = np.clip(h_initial, 0, 1).copy()

    h = cp.Variable(n_points, bounds=(0, 1))
    Y = cp.Variable((n_points, n_points), symmetric=True)
    C5 = cp.Variable()

    constraints = [
        cp.sum(h) == n_points / 2,
        Y >> 0,
        cp.diag(Y) == h**2
    ]
    for k in range(n_points):
        sum_expr = cp.sum([Y[i, (i + k) % n_points] for i in range(n_points)])
        constraints.append(sum_expr + (n_points / 2) * C5 >= n_points / 2)

    prob = cp.Problem(cp.Minimize(C5), constraints)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            h_values = h.value
            c5_bound = C5.value
        else:
            h_values = h_initial
            c5_bound = evaluate_erdos_solution()
    except:
        h_values = h_initial
        c5_bound = evaluate_erdos_solution()

    h_values = np.clip(h_values, 0, 1)
    h_array = np.array(h_values)
    shift_corr = np.correlate(h_array, 1 - h_array, mode='full')
    max_correlation = np.max(shift_corr) * dx
    c5_bound = np.min([c5_bound, max_correlation])
    return h_values, c5_bound, n_points
