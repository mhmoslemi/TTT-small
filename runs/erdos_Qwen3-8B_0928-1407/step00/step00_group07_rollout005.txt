def run(seed=42, budget_s=1100, **kwargs):

    best_time = 0

        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
