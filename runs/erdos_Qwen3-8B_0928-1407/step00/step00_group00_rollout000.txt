import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Use initial_h_values if provided
    initial_h = kwargs.pop('initial_h_values', None)
    n_points = 45  # default size if initial_h is not provided
    if initial_h is not None:
        h = np.array(initial_h, dtype=float)
        n_points = len(h)
    else:
        h = np.zeros(n_points, dtype=float)
        h[::2] = 1.0  # Start with alternating 1s and 0s
        if n_points % 2 != 0:
            h[-1] = 0.0  # Adjust last element for odd n

    dx = 2.0 / n_points
    assert np.isclose(sum(h) * dx, 1.0), "Initial h does not meet the constraint"

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    max_iter = 100000
    global_shuffle_steps = 1000
    shuffle_counter = 0

    while (time.time() - start_time) < budget_s and shuffle_counter < max_iter:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if not ones.size or not zeros.size:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        shuffle_counter += 1

        # Global shuffle every 1000 iterations
