import scipy.optimize as opt
import time
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 500
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * 2.0 / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    def objective(h):
        return evaluate_erdos_solution(h)
    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()
    result = opt.minimize(objective, h_initial, method='SLSQP', 
                         bounds=bounds, constraints=cons,
                         options={'ftol': 1e-8, 'maxiter': 100})
    best_c5 = result.fun
    best_h = result.x.copy()
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed = result.x + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        result = opt.minimize(objective, h_perturbed, method='SLSQP', 
                             bounds=bounds, constraints=cons,
                             options={'ftol': 1e-8, 'maxiter': 100})
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = result.x + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            result = opt.minimize(objective, h_perturbed, method='SLSQP', 
                                 bounds=bounds, constraints=cons,
                                 options={'ftol': 1e-8, 'maxiter': 100})
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x.copy()
    return (best_h, best_c5, n_points)
