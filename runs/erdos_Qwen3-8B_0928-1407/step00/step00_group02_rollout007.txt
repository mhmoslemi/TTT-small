    np.random.seed(seed)
    from numpy.random import RandomState

    h = initial_h_values
    n_points = len(h)
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    start_time = time.time()

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / current_sum
        return h * scale

    def compute_C5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h):
        h_perturb = h.copy()
        indices = np.random.choice(len(h_perturb), 5, replace=False)
        delta = np.random.uniform(-0.05, 0.05, size=5)
        h_perturb[indices] += delta
        return normalize_h(h_perturb)

    while time.time() - start_time < budget_s - 10:
        candidates = []
        for _ in range(100):
            perturbed = perturb(best_h)
            c5 = compute_C5(perturbed)
            candidates.append((c5, perturbed))
        candidates.sort()
        best_candidate = candidates[0]
        if best_candidate[0] < best_c5:
            best_c5 = best_candidate[0]
            best_h = best_candidate[1].copy()
    return best_h, best_c5, n_points
