def evaluate_c5(h, n_points):
    h_array = np.array(h, dtype=np.float64)
    ones_minus_h = 1.0 - h_array
    corr = np.correlate(h_array, ones_minus_h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx
