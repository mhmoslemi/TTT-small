        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(h):
        cc = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cc)

    n_starts = 5
    time_per_start = budget_s / n_starts
    for _ in range(n_starts):
        current_h = best_h.copy() + 0.05 * np.random.randn(n_points)
        current_h = np.clip(current_h, 0, 1)
        current_h *= (n_points / 2) / np.sum(current_h)
        start_time = time.time()
        while time.time() - start_time < time_per_start:
            cc = np.correlate(current_h, 1 - current_h, mode='full') * dx
            current_c5 = np.max(cc)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            max_indices = np.where(cc == current_c5)[0]
            gradient = np.zeros_like(current_h)
            for k in max_indices:
                shift = k - (n_points - 1)
                gradient += np.roll(1 - current_h, shift)
            gradient = (gradient - np.mean(gradient)) / (np.std(gradient) + 1e-8)
            current_h -= 0.1 * gradient
            current_h = np.clip(current_h, 0, 1)
            current_h *= (n_points / 2) / np.sum(current_h)
    return (best_h, best_c5, n_points)
