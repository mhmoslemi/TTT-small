import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        delta = (n_points / 2 - total) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        return h
    
    initial_h = repair(initial_h)
    
    pop_size = 50
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        pert = np.random.normal(0, 0.1, n_points)
        cand = initial_h + pert
        cand = repair(cand)
        population.append(cand)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return [compute_c5(ind) for ind in pop]
    
    pop_fitness = eval_pop(population)
