def optimize_h(h, dx, budget_s):

    lowest_C = np.inf

    temperature = 1.0
    cooling_factor = 0.99
    quakes_interval = 100

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        cc = np.correlate(h, 1 - h, mode='full') * dx

        # Update best solution
