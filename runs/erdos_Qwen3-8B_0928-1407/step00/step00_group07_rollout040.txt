        contributions = h * (1.0 - h[(np.arange(n_points) + k) % n_points])
        top_indices = np.argsort(contributions)[-5:]
        for idx in top_indices:
            j = (idx + k) % n_points
            h[idx] -= 0.005
            h[j] += 0.005
            h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum < desired_sum - 1e-6:
            delta = (desired_sum - h_sum) / n_points
            h += delta
        elif h_sum > desired_sum + 1e-6:
            delta = (h_sum - desired_sum) / n_points
            h -= delta
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)
