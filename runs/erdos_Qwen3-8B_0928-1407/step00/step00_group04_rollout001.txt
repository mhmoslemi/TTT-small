def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    n_points = len(h)
    learning_rate = 0.1
    num_iterations = 1000
    cooling_factor = 0.995
    quake_interval = 1000
    start_time = time.time()
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        conv = signal.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # Gradient descent
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += 1e-3
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (n_points / 2) / h_perturbed_sum
            conv_perturbed = signal.correlate(h_perturbed, 1 - h_perturbed, mode='full') * dx
            grad[i] = (conv_perturbed[max_idx] - max_val) / 1e-3
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        learning_rate *= cooling_factor
        if np.random.rand() < 0.05:
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
    final_conv = signal.correlate(h, 1 - h, mode='full') * dx
    c5_final = np.max(final_conv)
    return (h.tolist(), c5_final, n_points)
