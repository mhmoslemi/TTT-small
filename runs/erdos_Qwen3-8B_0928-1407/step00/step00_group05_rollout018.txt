    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    current_h = initial_h.copy()
    
    while time.time() - start_time < budget_s:
        result = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 1000})
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        if time.time() - start_time > 100:
            noise = np.random.normal(0, 0.01, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            sum_perturbed = np.sum(perturbed)
            adjustment = (n_points / 2 - sum_perturbed) / n_points
            perturbed += adjustment
            perturbed = np.clip(perturbed, 0.0, 1.0)
            current_h = perturbed.copy()
    
    return (best_h, best_c5, n_points)
