            dx = 2.0 / n_points
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + best_k) % n_points
                grad[i] = (1 - best_h[j]) * dx
            best_h -= 0.01 * grad
            best_h = np.clip(best_h, 0, 1)
            total = np.sum(best_h)
            if total > 0:
                best_h = best_h * (n_points / 2) / total
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
