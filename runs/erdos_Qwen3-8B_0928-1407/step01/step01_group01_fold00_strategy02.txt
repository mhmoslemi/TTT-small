import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # PSO Parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_velocity = 0.2

    def obj(h_vec):
        h = np.array(h_vec)
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    # Initialize swarm
    swarm = np.tile(initial_h, (n_particles, 1)) + \
            np.random.normal(0, 0.05, (n_particles, n_points))
    swarm = np.clip(swarm, 0.0, 1.0)
    for i in range(n_particles):
        s = np.sum(swarm[i])
        if s < 1e-8:
            s = 1e-8
        swarm[i] *= (n_points / 2) / s

    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = swarm.copy()
    personal_obj = np.array([obj(p) for p in personal_best])
    global_best_idx = np.argmin(personal_obj)
    global_best = personal_best[global_best_idx].copy()

    best_swarm_obj = np.min(personal_obj)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Update velocities
        r1, r2 = np.random.rand(2)
        velocities = w * velocities + \
                    c1 * r1 * (personal_best - swarm) + \
                    c2 * r2 * (global_best - swarm)
        velocities = np.clip(velocities, -max_velocity, max_velocity)

        # Update positions
        swarm += velocities
        swarm = np.clip(swarm, 0.0, 1.0)

        # Normalize sums
        for i in range(n_particles):
            s = np.sum(swarm[i])
            if s < 1e-8:
                s = 1e-8
            swarm[i] *= (n_points / 2) / s

        # Evaluate and update bests
        current_obj = np.array([obj(p) for p in swarm])
        improved = current_obj < personal_obj
        personal_best[improved] = swarm[improved]
        personal_obj[improved] = current_obj[improved]

        global_best_idx = np.argmin(personal_obj)
        global_best = personal_best[global_best_idx].copy()

        # Track best
        if personal_obj[global_best_idx] < best_swarm_obj:
            best_swarm_obj = personal_obj[global_best_idx]

    # Return best solution
    return (global_best, best_swarm_obj, n_points)
