import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            continue
        corr = np.dot(h[:n_points - k], 1 - h[k:n_points])
        corrs.append(corr)
    for k in range(2, n_points):  # symmetric side
        if n_points - k <= 0:
            continue
        corr = np.dot(h[k:n_points], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def project_and_normalize(h, n_points):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total == 0:
        return np.full_like(h, 0.5)
    h = h * (n_points / 2) / total
    return h

def initialize_population(n_points, size=50, init_h=None, seed=42):
    np.random.seed(seed)
    population = []
    if init_h is not None:
        population.append(project_and_normalize(init_h.copy(), n_points))
    while len(population) < size:
        h = np.random.rand(n_points)
        h = project_and_normalize(h, n_points)
        population.append(h)
    return np.array(population)

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return project_and_normalize(child, len(parent1))

def mutate(h, sigma):
    h = h.copy()
    noise = np.random.normal(0, sigma, size=h.shape)
    h += noise
    return project_and_normalize(h, len(h))

def evaluate_population(population, n_points):
    return np.array([compute_c5(ind, n_points) for ind in population])
