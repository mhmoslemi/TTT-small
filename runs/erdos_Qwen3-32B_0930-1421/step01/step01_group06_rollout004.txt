import numpy as np
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    def flip_bit(h, idx1, idx2):
        h_new = h.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        return h_new

    def simulated_annealing_budget_search(n_iter=1000):
        # Binary search: reduce error via flipping
        h_b = (initial_h > 0.5).astype(float)
        h_b[~np.isclose(h_b, 1)] = 0
        current_h = h_b.copy()
        current_sum = np.sum(current_h)
        if current_sum < n_points/2:
            adjust_idx = np.where(current_h == 0)[0]
            current_h[adjust_idx[0]] = 1
        elif current_sum > n_points/2:
            adjust_idx = np.where(current_h == 1)[0]
            current_h[adjust_idx[0]] = 0
        best_h = current_h.copy()
        best_c5 = evaluate_erdos_solution(best_h, n_points)

        T0 = 1.0
        temp = T0

        pbar = 0
        while time.time() - start_time < budget_s - 600 and pbar < n_iter:
            idxs = np.random.randint(0, n_points, size=2)
            if idxs[0] == idxs[1]:
                continue
            h_proposal = flip_bit(current_h, *idxs)
            h_sum = np.sum(h_proposal)

            if h_sum < n_points/2 or h_sum > n_points/2:
                continue

            c5_proposal = evaluate_erdos_solution(h_proposal, n_points)
            delta_c5 = c5_proposal - best_c5

            if delta_c5 < 0 or np.exp(-delta_c5 / temp) > np.random.rand():
                current_h[:] = h_proposal
                best_c5 = max(best_c5, c5_proposal)
                best_h[:] = current_h.copy()

            temp *= 0.999
            pbar += 1

        return best_h, best_c5

    binary_h, binary_c5 = simulated_annealing_budget_search()
    print("Binary search complete")

    # Function to scale binary to meet the sum constraint
    def scale_binary_to_sum(h_b, total_sum):
        h_scaled = h_b.copy()
        h_sum = np.sum(h_b)
        if h_sum == total_sum:
            return h_scaled
        idxs_all_1 = np.where(h_b == 1)[0]
        idxs_all_0 = np.where(h_b == 0)[0]
        if h_sum < total_sum:
            idxs_all_1 = np.append(idxs_all_1, idxs_all_0[0])
            h_scaled[idxs_all_0[0]] = 0.5
        else:
            idxs_all_1[-1] = idxs_all_0[0]
            h_scaled[idxs_all_1[-1]] = 0.5
        return h_scaled

    scaled_h = scale_binary_to_sum(binary_h, n_points / 2)

    # Continuous refinement
    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    cons = (
        {"type": "eq", "fun": constraint_sum},
        {"type": "ineq", "fun": constraint_upper},
        {"type": "ineq", "fun": constraint_lower}
    )

    bounds = [(0.0, 1.0)] * n_points

    result = minimize(
        objective,
        scaled_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 500}
    )

    final_h = result.x
    c5_bound = evaluate_erdos_solution(final_h, n_points)

    return final_h, c5_bound, n_points
