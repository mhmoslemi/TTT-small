import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_ref = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_ref = np.zeros(n_points)
        for i in range(n_points // 2):
            h_ref[i + i] = 2.0 / n_points
        h_ref /= h_ref.sum() * 2.0  # Normalize to ∫h(x) dx = 1

    dx = 2.0 / n_points
    h_best = h_ref.copy()
    c5_best = float('inf')

    def compute_overlap(h):
        h_clipped = np.clip(h, 0, 1)
        h_normalized = h_clipped * (n_points / 2 / h_clipped.sum())
        overlaps = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        overlaps *= dx
        return np.max(overlaps)

    def gradient_refine(h0):
        h0 = h0.copy()
        h0_clipped = np.clip(h0, 0, 1)
        h_sum = h0_clipped.sum()
        h0_scaled = h0_clipped / h_sum * (n_points / 2)

        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func_flat(x_flat):
            h = x_flat.reshape(n_points)
            h_clipped = np.clip(h, 0, 1)
            h_sum = h_clipped.sum()
            h_normalized = h_clipped / h_sum * (n_points / 2)
            overlaps = np.correlate(h_normalized, 1 - h_normalized, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            obj_func_flat,
            h0_scaled,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 150}
        )
        return result.x

    tabu_list = []
    delta = 0.1  # Perturbation scale
    patience = 50
    counter = 0
    best_cc = None
    sticking_point = 0

    while (time.time() - start_time) < budget_s * 0.95:
        # Generate perturbed h
        h = h_best + np.random.uniform(-delta, delta, size=n_points)
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        h_normalized = h_clipped / h_sum * (n_points / 2)

        # Avoid cycles
        rep = tuple(np.round(1000 * h_normalized).astype(int).tolist())
        if rep in tabu_list:
            continue

        tabu_list.append(rep)
        if len(tabu_list) > 10:
            tabu_list.pop(0)

        # Gradient refinement
        h_opt = gradient_refine(h_normalized)

        c5 = compute_overlap(h_opt)
        if c5 < c5_best:
            h_best = h_opt
            c5_best = c5
            sticking_point = 0
            # Further refinement
            h_opt2 = gradient_refine(h_opt)
            c52 = compute_overlap(h_opt2)
            if c52 < c5_best:
                h_best = h_opt2
                c5_best = c52
        else:
            sticking_point += 1
            if sticking_point >= patience:
                delta *= 0.8  # Cool down
                sticking_point = 0

    # Final cleanup
    h_best = np.clip(h_best, 0, 1)
    h_best_sum = h_best.sum()
    h_best = h_best / h_best_sum * (n_points / 2)
    c5_final = compute_overlap(h_best)

    return h_best, c5_final, n_points
