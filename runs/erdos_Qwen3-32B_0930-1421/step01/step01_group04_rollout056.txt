import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import scipy.optimize as opt

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_full = np.concatenate([h, h, h])
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    corrs += corrs[::-1]
    return np.max(np.array(corrs) * dx)

def generate_perturbation(h, scale, n_points):
    noise = np.random.normal(0, scale, n_points)
    h_new = enforce_normalization(h + noise, n_points)
    return h_new

def kernel_rbf():
    return C(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2))

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = len(h0)

    dx = 2.0 / n_points

    samples = []

    t0 = time.time()
    max_iter = 1000

    alpha = 0.95

    for sample in [enforce_normalization(h0 + np.random.normal(0, 0.1, n_points), n_points) for _ in range(20)]:
        samples.append(sample)
        obj = compute_c5(sample, n_points)
        targets.append(obj)

    X = np.array(samples)
    y = np.array(targets)

    gp = GaussianProcessRegressor(kernel=kernel_rbf(), n_restarts_optimizer=10, normalize_y=True)
    gp.fit(X, y)

    current_h = h0.copy()
    current_c5 = compute_c5(current_h, n_points)
    best_c5 = current_c5
    T = T0

    while time.time() - t0 < budget_s:
        try:
            scale = 0.05 + max(0.01, T / 100)

            h_candidate = generate_perturbation(current_h, scale, n_points)
            mean, std = gp.predict(h_candidate.reshape(1, -1), return_std=True)

            if np.random.rand() < (T / max(1.0, std[0])):
                pred_value = mean[0]
            else:
                pred_value = float("inf")

            if pred_value < best_c5:
                trial_value = compute_c5(h_candidate, n_points)
                if trial_value < best_c5:
                    current_h = h_candidate.copy()
                    current_c5 = trial_value
                    best_c5 = trial_value
                    X = np.vstack([X, current_h])
                    y = np.append(y, current_c5)
                    gp.fit(X, y)

            T *= alpha
        except IndexError:
            T *= alpha
        if time.time() - t0 >= budget_s - 1:
            break

    res = opt.minimize(
        lambda x: compute_c5(x, n_points),
        current_h,
        bounds=[(0, 1)] * n_points,
        method='L-BFGS-B',
        options={'maxiter': 100}
    )
    h_refined = res.x
    final_c5 = res.fun

    h_refined = enforce_normalization(h_refined, n_points)
    final_c5 = compute_c5(h_refined, n_points)

    return h_refined, final_c5, n_points
