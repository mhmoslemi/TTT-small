import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    dx = 2.0 / n
    return np.max(corr) * dx

def flip_correlated_bits(h, overlap, k):
    """ Flip two bits near the max overlap at shift k. """
    n = len(h)
    h1 = 1 - h
    shifted_h1 = np.roll(h1, k)
    combined = h * shifted_h1
    peak_pos = np.argmax(combined)
    i = peak_pos
    j = np.argmin(1 - h + (np.roll(1 - h, -k) == 1))

    # Flip 1 -> 0 in h near peak and 0 -> 1 in h near corresponding position
    h_new = h.copy()
    for idx in [i, j]:
        if h_new[idx] == 1:
            h_new[idx] = 0
        elif h_new[idx] == 0:
            h_new[idx] = 1
    return h_new

    current_h = h.copy()
    current_c5 = best_c5

    start_time = time.time()
    max_iter = int(1e6)
    for step in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute current overlap and find worst shift
        h1 = 1 - current_h
        corr = correlate(current_h, h1, mode="full")
        max_corr = np.max(corr)
        k = np.argmax(corr)

        # Local bit-flip to reduce max overlap
        h_new = flip_correlated_bits(current_h, max_corr, k)
        c5_new = evaluate_overlap(h_new)

        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    return (best_h, best_c5, n_points)
