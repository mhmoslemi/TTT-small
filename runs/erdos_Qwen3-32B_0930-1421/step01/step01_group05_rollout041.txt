import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.random.rand(n_points)
        h_init /= h_init.sum() * 2.0 / n_points

    n_fireflies = 50
    fireflies = [np.random.normal(loc=h_init, scale=0.05, size=n_points).clip(0, 1) for _ in range(n_fireflies - 1)]
    fireflies.append(h_init.copy())
    fireflies = [f / f.sum() * n_points / 2 for f in fireflies]

    dx = 2.0 / n_points

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    fireflies = [np.clip(firefly, 0, 1) for firefly in fireflies]
    fireflies = [f / f.sum() * n_points / 2 for f in fireflies]

    def repair_h(fly):
        fly = np.clip(fly, 0, 1)
        fly *= n_points / 2 / fly.sum()
        return np.clip(fly, 0, 1)

    best_h = []
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        current_gens = 1

        # Evaluate all fireflies
        fitness = np.array([evaluate(fly) for fly in fireflies])
        idx = np.argsort(fitness)
        fireflies = [fireflies[i] for i in idx]
        fitness = np.sort(fitness)

        for i in range(n_fireflies):
            for j in range(i):
                if fitness[j] < fitness[i]:
                    r = np.linalg.norm(fireflies[i] - fireflies[j])
                    beta0 = 1.0
                    gamma = 0.01 * (1 + current_gens / 2000)
                    beta = beta0 * np.exp(-gamma * r**2)
                    move = beta * (fireflies[j] - fireflies[i])
                    step = np.random.rand(n_points) * np.random.rand(n_points) - 0.5
                    fireflies[i] += move + 0.2 * step
                    fireflies[i] = repair_h(fireflies[i])
                    new_c5 = evaluate(fireflies[i])
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = fireflies[i].copy()
                    fireflies[i] = fireflies[i].copy()
                    break

        current_gens += 1
        if time.time() - start_time >= budget_s:
            break

    # Final repair
    best_h = repair_h(best_h)

    return best_h.tolist(), best_c5, n_points
