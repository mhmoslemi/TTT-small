import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def round_and_project(h):
    """Project into feasible binary set with rounded values"""
    h_binary = np.ceil(h * 0.5 - 1e-6).clip(0, 1)
    target_sum = len(h) / 2
    curr_sum = h_binary.sum()
    if curr_sum < target_sum - 1e-5:
        # Need to turn some 0s to 1s
        diff = target_sum - curr_sum
        indices = np.argwhere(h_binary == 0).flatten()
        for i in np.argsort(h[indices])[::-1][:int(diff + 1)]:
            h_binary[i] = 1.
    elif curr_sum > target_sum + 1e-5:
        # Need to turn some 1s to 0s
        diff = curr_sum - target_sum
        indices = np.argwhere(h_binary == 1).flatten()
        for i in np.argsort(h[indices])[:int(diff + 1)]:
            h_binary[i] = 0.
    h_binary = ensure_feasible(h_binary)
    return h_binary


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.

    h = ensure_feasible(h)

    # Define variables
    h_var = cp.Variable(n_points, nonneg=True)
    lambda_var = cp.Variable(nonz=True)

    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]

    active_shifts = set()  # store k for shifts we have included
    max_shift = n_points - 1

    # Core loop
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    best_time = time.time()

    while time.time() - start_time < budget_s - 10:  # 10s buffer for final steps

        # Add all missing shifts to constraints
        new_shifts = []
        for k in range(1, max_shift):
            if k not in active_shifts:
                # Enforce constraint for overlap at shift k
                idxs = [(i, i+k) for i in range(n_points - k)]
                terms = [
                    h_var[i] * (1 - h_var[j])
                    for i, j in idxs
                ]
                total = sum(terms)
                constraints.append(lambda_var >= total * dx)
                new_shifts.append(k)
        active_shifts.update(new_shifts)
        if not new_shifts:
            break

        # Optimize current problem
        objective = cp.Minimize(lambda_var)
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.CVXOPT, verbose=False, maxitn=100)

        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break

        h_candidate = h_var.value
        lambda_val = lambda_var.value

        # Feasibility repair
        feasible_h = round_and_project(h_candidate)

        # Update best if improved
        current_C5 = evaluate_C5(feasible_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = feasible_h
            best_time = time.time()

        # Early termination check (evolution small)
        if time.time() - best_time > 30:  # 30s w/o improvement
            break

    return best_h, best_C5, n_points
