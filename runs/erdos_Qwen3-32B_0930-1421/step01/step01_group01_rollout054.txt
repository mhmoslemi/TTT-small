import numpy as np
import time
import cvxpy as cp
import scipy.signal


def evaluate_C5(h, dx):
    """Compute the current C5 bound from the h vector."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum == 0:
        return float('inf'), h
    h = h / h_sum * len(h) / 2.0  # normalize to satisfy the integral constraint
    h_centered = np.roll(h, len(h) - 1)  # padding to handle edge cases
    cor = scipy.signal.correlate(h_centered, 1 - h_centered)[len(h) - 1:]
    cor *= dx
    return np.max(cor), h


def rescale_h(h):
    """Rescale h to ensure the integral is 1."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    return (h / h_sum * len(h) / 2) if h_sum > 1e-10 else h * 0.0


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdos C5 problem using sequential convex optimization."""
    np.random.seed(seed)

    # Use initial_h_values if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points, refcheck=False)
    except:
        h = np.zeros(n_points)
        h[: n_points // 3] = 0.1
        h[n_points // 3: -n_points // 3] = 0.3
        h[-n_points // 3:] = 0.2
        h = rescale_h(h)

    best_h = h.copy()
    best_c5, _ = evaluate_C5(best_h, 2.0 / n_points)

    start_time = time.time()
    timeout_per_qp = 10.0  # per QP iteration

    while time.time() - start_time < budget_s:
        dx = 2.0 / n_points
        h = rescale_h(h)
        current_c5, h = evaluate_C5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Compute overlap integrals for all shifts
        h_centered = np.roll(h, len(h) - 1)
        cor = scipy.signal.correlate(h_centered, 1 - h_centered)[len(h) - 1:]
        max_shift = np.argmax(cor)
        max_offset = max_shift - len(h) + 1

        # Solve convex QP to reduce the worst-case shift
        h_var = cp.Variable(n_points, nonneg=True)
        overlaps = sum(h_var[i] * (1 - h_var[i + max_offset]) for i in range(n_points) if i + max_offset in range(n_points))
        overlaps += sum(h_var[i] for i in range(n_points) if i + max_offset not in range(n_points))  # handle out of bounds naturally
        objective = cp.Minimize(overlaps * dx)

        constraints = [sum(h_var) == n_points / 2.0,
                       h_var <= 1.0,
                       h_var >= 0.0]
        prob = cp.Problem(objective, constraints)

        # Solve the convex problem
        try:
            prob.solve(solver='ECOS', verbose=False, max_iters=250, feastol=1e-7, abstol=1e-7, reltol=1e-7, time_limit=timeout_per_qp)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = h_var.value
        except Exception as e:
            pass

        # Perturb occasionally to escape local optima
        if np.random.rand() < 0.02:
            perturbation = (np.random.rand(n_points) - 0.5) * 0.1
            h = np.clip(h + perturbation, 0, 1)

    return best_h, best_c5, n_points
