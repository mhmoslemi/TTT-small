import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100
    n_points = 47 if not 'initial_h_values' in globals() else len(initial_h_values)
    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i * 2] = 2.0 / n_points
        h_init *= (n_points / 2) / h_init.sum()

    best_h = h_init.copy()
    best_c5 = 0.0
    best_h = best_h.clip(0, 1)
    best_h *= (constraint_sum / best_h.sum())

    def objective(h):
        h = h.reshape(n_points)
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    # Optimize with convex subproblem using CVXPY
    def optimize_fixed_support(h, iter_limit=100):
        h = h.copy()
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        active = h > 1e-6
        w = cp.Variable(shape=n_points)
        w.value = h
        support_indicator = cp.Parameter(n_points, value=active)
        constraint_sum_obj = cp.sum(w) == constraint_sum
        obj = cp.Minimize(cp.max(cp.conv(w, 1 - w, mode='full') * dx))
        constraints = [
            w >= 0,
            w <= 1,
            support_indicator @ w == support_indicator @ w,  # fix support
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)
        h_opt = w.value
        h_opt = h_opt.clip(0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    # Perturbation and support variation
    def perturb_support(h, perturb_rate=0.05):
        h = h.copy()
        new_support = h > 1e-6
        n_perturb = max(1, int(np.round(n_points * perturb_rate)))
        idx = np.random.choice(n_points, size=n_perturb, replace=False)
        new_support[idx] = np.random.choice([False, True], size=n_perturb, p=[0.5, 0.5])
        h_new = np.zeros_like(h)
        h_new[new_support] = h[new_support]
        h_new = h_new.clip(0, 1) * (constraint_sum / h_new.sum())
        return h_new

    def refine_and_evaluate(h):
        h_refined = optimize_fixed_support(h)
        c5 = objective(h_refined)
        return h_refined, c5

    best_h, best_c5 = refine_and_evaluate(h_init)

    # Main optimization loop
    for _ in range(50):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_perturbed = perturb_support(best_h, perturb_rate=0.1)
        h_candidate, c5_candidate = refine_and_evaluate(h_perturbed)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    # Final polish
    h_polished = optimize_fixed_support(best_h, iter_limit=30)
    h_polished = h_polished.clip(0, 1)
    h_polished = h_polished * (constraint_sum / h_polished.sum())
    c5_polished = objective(h_polished)
    if c5_polished < best_c5:
        best_h = h_polished
        best_c5 = c5_polished

    best_h = best_h.clip(0, 1) * (constraint_sum / best_h.sum())
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = best_h.clip(0, 1) * (constraint_sum / best_h.sum())
    return best_h, best_c5, n_points
