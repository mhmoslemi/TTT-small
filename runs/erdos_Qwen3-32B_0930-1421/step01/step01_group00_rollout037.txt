import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    def compute_c5(optimized_h):
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
            corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    def top_problematic_shifts(h, num_shifts=5):
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h_scaled = h * target_sum / h_sum
        h = np.clip(h_scaled, 0, 1)
        ones_minus_h = 1.0 - h
        overlap = []
        for k in range(1, n_points):
            left = np.dot(h[:n_points - k], ones_minus_h[k:])
            right = np.dot(h[k:], ones_minus_h[:n_points - k])
            overlap.append((k, max(left, right) * dx))
        overlap.sort(key=lambda x: -x[1])
        return [k for k, _ in overlap[:num_shifts]]

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = time.time()

    window_size = max(1, n_points // 100)
    perturb_step = 0.01
    adjust_radius = max(1, n_points // 100)

    while time.time() - start_time < budget_s - 10:
        h_diff = np.random.normal(0, 1, size=n_points)
        h_perturbed = h + h_diff * (1 if time.time() - best_time < 30 else perturb_step)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = h_perturbed.sum()
        h_perturbed *= target_sum / h_perturbed_sum

        shifts = top_problematic_shifts(h)
        for k in shifts:
            for i in range(n_points - k):
                if h[i] * (1 - h[i + k]) > 0.01:
                    indices = np.concatenate([np.arange(i - adjust_radius, i + adjust_radius)])
                    indices = np.clip(indices, 0, n_points - 1)
                    indices = np.unique(indices)
                    for j in np.random.choice(indices, size=min(5, len(indices)), replace=False):
                        if h[j] < 1.0 and h[j + k] > 0.0:
                            h[j] += 0.005
                            h[j + k] -= 0.005
                            h = np.clip(h, 0, 1)
                            h_sum = h.sum()
                            if abs(h_sum - target_sum) > 1e-7:
                                h *= target_sum / h_sum

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h
            best_time = time.time()
            if best_c5 < 0.380876:
                break
        if time.time() - best_time > 30:
            h = np.clip(h, 0, 1)
            h_frac = h.sum()
            h *= target_sum / h_frac

    return (best_h, best_c5, n_points)
