import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s:
        h = normalize_h(h)
        current_c5 = compute_c5(h)

        c_values = correlate(h, 1 - h, mode='full')
        max_indices = np.where(c_values == np.max(c_values))[0]
        k_star = max_indices[0]

        grad = np.zeros_like(h)
        grad[:-k_star] -= h[k_star:]
        grad[k_star:] += (1 - h[:-k_star]) if k_star > 0 else (1 - h[k_star+1:])
        grad[k_star + h.shape[0]:] += h[:h.shape[0] - k_star] if k_star < h.shape[0] else h[0]
        grad = grad * dx

        learning_rate = 1.0
        while True:
            trial_h = h - learning_rate * grad
            trial_h = normalize_h(trial_h)
            trial_c5 = compute_c5(trial_h)
            if trial_c5 < current_c5:
                h[:] = trial_h
                break
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                break

        if compute_c5(h) < best_c5:
            best_h = h.copy()
            best_c5 = compute_c5(best_h)

        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
