import numpy as np
import cvxpy as cp
import time
import scipy.signal
import scipy.optimize

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Ensure initial satisfies constraints
    h = np.clip(h_init, 0, 1)
    s = h.sum()
    if abs(s - target_sum) > 1e-7:
        h *= (target_sum / s)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()

    # Function to compute the current bound
    def evaluate_c5(h_eval):
        h_eval = np.clip(h_eval, 0, 1)
        s = h_eval.sum()
        if abs(s - target_sum) > 1e-7:
            h_eval *= (target_sum / s)
        h_eval = np.clip(h_eval, 0, 1)

        ones_minus_h = 1.0 - h_eval
        max_c5 = -np.inf
        for k in range(n_points):
            if k >= n_points - 1:
                continue
            # Forward overlap
            if k < n_points:
                corr = np.dot(h_eval[:n_points - k], ones_minus_h[k:])
                max_c5 = max(max_c5, corr * dx)
            # Backward overlap
            if k < n_points:
                corr = np.dot(h_eval[k:], ones_minus_h[:n_points - k])
                max_c5 = max(max_c5, corr * dx)
        return max_c5

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    print(f"Initial C5: {best_c5:.6f}")

    def feasibility_obj(x):
        h_proposal = x.reshape(n_points)
        h_proposal = np.clip(h_proposal, 0, 1)
        s = h_proposal.sum()
        if abs(s - target_sum) > 1e-7:
            h_proposal *= (target_sum / s)
        h_proposal = np.clip(h_proposal, 0, 1)
        c5_val = evaluate_c5(h_proposal)
        return c5_val

    # Step 1: Solve convex relaxation
    h_cp = cp.Variable(n_points)
    constraints = [
        h_cp >= 0,
        h_cp <= 1,
        cp.sum(h_cp) == target_sum
    ]
    obj = cp.Minimize(cp.maximum(
        *[cp.sum([h_cp[i] * (1 - h_cp[i + k]) for i in range(n_points - k)]) * dx for k in range(1, n_points - 1)] +
        *[cp.sum([h_cp[i + k] * (1 - h_cp[i]) for i in range(n_points - k)]) * dx for k in range(1, n_points - 1)]
    ))
    prob = cp.Problem(obj, constraints)
    try:
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        h_cvx = np.clip(np.array(h_cp.value), 0, 1)
        s = h_cvx.sum()
        if abs(s - target_sum) > 1e-7:
            h_cvx *= (target_sum / s)
        h_cvx = np.clip(h_cvx, 0, 1)
        cvx_c5 = evaluate_c5(h_cvx)
        print(f"CVX bound: {cvx_c5:.6f}")
        if cvx_c5 < best_c5:
            best_h = h_cvx
            best_c5 = cvx_c5
    except Exception as e:
        print("Skipping convex relaxation:", e)

    # Step 2: Local search with adaptive step size
    h = best_h.copy()
    t = 0
    step = 0.01
    while time.time() - start_time < budget_s - 10:
        idx = np.random.choice(n_points, size=min(10, n_points), replace=False)
        noise = np.random.uniform(-step, step, size=idx.shape)
        h_proposal = h.copy()
        h_proposal[idx] += noise
        h_proposal = np.clip(h_proposal, 0, 1)
        s = h_proposal.sum()
        if abs(s - target_sum) > 1e-7:
            h_proposal *= (target_sum / s)
        h_proposal = np.clip(h_proposal, 0, 1)
        c5_prop = evaluate_c5(h_proposal)

        if c5_prop < best_c5:
            h = h_proposal
            best_c5 = c5_prop
            best_h = h.copy()
            step = max(0.005, step / 1.5)
            if best_c5 < 0.380876:
                break
        else:
            step = min(0.05, step * 1.1)

        t += 1
        if t % 1000 == 0:
            print(f"T={t}, C5={best_c5:.6f}, Step={step:.5f}")

    return (best_h, best_c5, n_points)
