import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64, copy=True)
    else:
        h = np.zeros(n_points)
        h[::2] = target_sum / (n_points // 2)
        h = h.clip(0, 1)

    def is_feasible(h):
        return np.all((h >= 0) & (h <= 1)) and abs(h.sum() - target_sum) < 1e-6

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / h.sum())
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def refine_with_cvx(h0):
        h0 = h0.copy()
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == target_sum]
        conv_weights = np.outer(1 - np.arange(n_points), 1 - h0)
        overlap_expr = -cp.max(cp.conv(h, 1 - h, mode='full') * dx)
        problem = cp.Problem(cp.Maximize(overlap_expr), constraints)
        problem.solve(solver='ECOS', verbose=False)
        return h.value

    def perturb(h):
        h = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / h.sum())
        return h

    h_best = h.copy()
    c5_best = objective(h_best)
    remaining_steps = 200

    while time.time() - start_time < budget_s and remaining_steps > 0:
        remaining_steps -= 1
        if np.random.rand() < 0.7:
            h_candidate = refine_with_cvx(h_best)
        else:
            h_candidate = perturb(h_best)

        if is_feasible(h_candidate):
            c5_candidate = objective(h_candidate)
            if c5_candidate < c5_best:
                h_best = h_candidate
                c5_best = c5_candidate

    h_best = h_best.clip(0, 1)
    h_best = h_best * (target_sum / h_best.sum())
    return h_best, c5_best, n_points
