import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def generate_wavelet_bases(n_basis=10, alpha=4, beta=0.3, n_points=100):
    x = np.linspace(0, 2, n_points)
    bases = []
    for c in np.linspace(0, 2, n_basis):
        width = beta / (2 - c + c) * n_points  # scale beta to sample space
        center = int(c * n_points)
        basis = np.zeros(n_points)
        dist = np.abs(np.arange(n_points) - center)
        envelope = np.exp(-alpha * (dist / width) ** 2)
        start_idx = max(0, center - int(width))
        end_idx = min(n_points, center + int(width) + 1)
        basis[start_idx:end_idx] = envelope[start_idx:end_idx]
        bases.append(basis)
    bases = np.asarray(bases)
    return np.clip(bases, 0, 1)


def optimize_basis_coefficients(bases, n_points):
    n_basis = len(bases)
    a = cp.Variable(n_basis, nonneg=True)
    dx = 2.0 / n_points
    constraints = [cp.sum(a) == 1.0]

    for k in range(-n_points + 1, n_points):
        dot_prod = 0
        for i in range(n_basis):
            for j in range(n_basis):
                if k < 0:
                    corr_data = bases[i, -k:] * (1 - bases[j, :k])
                else:
                    corr_data = bases[i, :n_points - k] * (1 - bases[j, k:])
                dot_prod += a[i] * a[j] * np.sum(corr_data * dx)
        target_var = cp.Variable()
        constraints += [dot_prod <= target_var]
    objective = cp.Minimize(target_var)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, maxiter=150)
    a_opt = a.value
    return a_opt, problem


def wavelet_h(bases, a):
    return np.dot(bases.T, a)


def perturb_centers(n_basis, perturbation=0.1, n_points=100):
    centers = np.linspace(0, 2, n_basis)
    pert = np.random.uniform(-perturbation, perturbation, n_basis)
    return np.clip(centers + pert, 0, 2)


def run(budget_s=1100, n_points=200, n_basis=10, alpha=4, beta=0.3, max_iter=50):
    start_time = time.time()
    perturbation = 0.1
    best_h = None
    best_C5 = np.inf
    centers_init = None

    if initial_h_values is not None:
        mass_centers = (initial_h_values * np.linspace(0, 2, n_points)).sum() / (initial_h_values.sum() * (2.0 / n_points))
        centers_init = np.linspace(mass_centers, mass_centers, n_basis)

    for i in range(max_iter):
        if initial_h_values is not None and i == 0:
            bases = generate_wavelet_bases(n_basis, alpha, beta, n_points=n_points)
        else:
            centers = np.linspace(0, 2, n_basis)
            if i > 0:
                centers = perturb_centers(n_basis, perturbation=perturbation, n_points=n_points)
            bases = generate_wavelet_bases(n_basis, alpha, beta, n_points=n_points)
        a_opt, problem = optimize_basis_coefficients(bases, n_points)
        if problem.status != "optimal":
            a_opt = np.ones(n_basis) / n_basis

        h = wavelet_h(bases, a_opt)
        h = ensure_feasible(h)

        if len(h) != n_points:
            h = np.clip(h, 0, 1)
            h *= n_points / 2 / h.sum()

        h_downsampled = np.interp(np.linspace(0, 199, n_points), np.linspace(0, 199, 199), h)

        C5 = evaluate_C5(h_downsampled)
        if C5 < best_C5:
            best_h = h_downsampled
            best_C5 = C5
            print(f"Iteration {i + 1}: C₅ = {best_C5:.6f}")

        if time.time() - start_time >= budget_s:
            break

    return best_h, best_C5, n_points
