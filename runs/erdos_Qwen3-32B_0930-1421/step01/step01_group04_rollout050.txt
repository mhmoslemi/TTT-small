import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = h.astype(np.float64)
    corrs = []
    for k in range(1, n_points):
        left = h[:n_points - k]
        right = 1 - h[k:]
        corr = np.dot(left, right)
        corrs.append(corr)
    for k in range(1, n_points):
        left = h[k:]
        right = 1 - h[:n_points - k]
        corr = np.dot(left, right)
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    scale = (n_points / 2.0) / total
    h *= scale
    h = np.clip(h, 0, 1)
    return h

def objective_surrogate(h, n_points):
    h = h.reshape(-1)
    h = enforce_normalization(h, n_points)
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        left = h[:n_points - k]
        right = 1 - h[k:]
        corr = np.dot(left, right)
        corrs.append(corr)
    for k in range(1, n_points):
        left = h[k:]
        right = 1 - h[:n_points - k]
        corr = np.dot(left, right)
        corrs.append(corr)
    corrs = np.array(corrs)
    return np.sum(corrs**2) * dx

def pso_update(particle, velocity, best_particle, global_best, w, c1, c2):
    r1, r2 = np.random.rand(2)
    new_velocity = (
        w * velocity +
        c1 * r1 * (best_particle - particle) +
        c2 * r2 * (global_best - particle)
    )
    new_position = particle + new_velocity
    return new_position, new_velocity

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    init_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(47)
    n_points = len(init_h)
    init_h = enforce_normalization(init_h, n_points)

    # Add classical raised cosine initialization as second base
    h_cos = np.array([0.5 + 0.5*np.cos(np.pi*i / n_points) for i in range(n_points)])
    h_cos = enforce_normalization(h_cos, n_points)

    # PSO setup: 20 particles, hybrid with multiple behaviors
    n_particles = 20
    particles = np.zeros((n_particles, n_points))
    particle_positions = np.tile(init_h, (n_particles, 1))
    # Perturb from both init_h and h_cos
    for i in range(n_points):
        particle_positions[i % n_particles, :] = (h_cos if (i // n_particles) % 2 == 0 else init_h).copy()
    # Apply random variations
    particle_positions += 0.1 * np.random.randn(n_particles, n_points)
    for i in range(n_particles):
        particle_positions[i] = enforce_normalization(particle_positions[i], n_points)

    velocities = 0.01 * np.random.randn(n_particles, n_points)
    particle_best = particle_positions.copy()
    particle_scores = np.array([compute_c5(p, n_points) for p in particle_positions])
    global_best = particle_positions[particle_scores.argmin()].copy()
    global_score = np.min(particle_scores)
    c1, c2 = 2.0, 2.0
    w = 0.7

    time_start = time.time()
    while time.time() - time_start < budget_s:
        for i in range(n_particles):
            pos = particle_positions[i]
            score = compute_c5(pos, n_points)
            # Periodic evaluation of the true C₅ and update best
            if score < particle_scores[i]:
                particle_scores[i] = score
                particle_best[i] = pos.copy()
                if score < global_score:
                    global_best = pos.copy()
                    global_score = score

            # PSO update inline
            pos, velocities[i] = pso_update(
                pos, velocities[i], particle_best[i], global_best, w, c1, c2
            )
            pos = enforce_normalization(pos, n_points)
            particle_positions[i] = pos

        # Decay inertia
        w = max(0.4, w * 0.9995)
        # Trigger L-BFGS-B on the top 2 particles to refine local optima
        top_indices = np.argsort(particle_scores)[:2]
        for idx in top_indices:
            res = opt.minimize(
                lambda x: objective_surrogate(x, n_points),
                x0=particle_positions[idx],
                bounds=[(0, 1)] * n_points,
                method='L-BFGS-B',
                options={'maxiter': 100}
            )
            candidate = res.x
            candidate = enforce_normalization(candidate, n_points)
            candidate_score = compute_c5(candidate, n_points)
            if candidate_score < particle_scores[idx]:
                particle_positions[idx] = candidate
                particle_scores[idx] = candidate_score
                particle_best[idx] = candidate.copy()
                if candidate_score < global_score:
                    global_best = candidate.copy()
                    global_score = candidate_score

        # Enforce computational budget early and break if time is tight
        if time.time() - time_start > budget_s * 0.98:
            final_score = compute_c5(global_best, n_points)
            return global_best, final_score, n_points

    final_score = compute_c5(global_best, n_points)
    return global_best, final_score, n_points
