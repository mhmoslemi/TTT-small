import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        left = h[:n_points - k]
        right = 1 - h[k:]
        corr = np.dot(left, right)
        corrs.append(corr)
    for k in range(1, n_points):
        left = h[k:]
        right = 1 - h[:n_points - k]
        corr = np.dot(left, right)
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def unit_ball_project(v):
    total = np.sum(v)
    scale = (len(v) / 2.0) / total
    return np.clip(v * scale, 0, 1)

def raised_cosine(n_points):
    i = np.arange(n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
    h = np.clip(h, 0, 1)
    return unit_ball_project(h)

def objective_F(h, n_points, alpha=100.0):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    corr_vals = []

    for k in range(1, n_points):
        left = h[:n_points - k]
        right = 1 - h[k:]
        corr_vals.append(np.dot(left, right))
    
    for k in range(1, n_points):
        left = h[k:]
        right = 1 - h[:n_points - k]
        corr_vals.append(np.dot(left, right))

    corr_array = np.array(corr_vals)
    return np.sum(corr_array ** 2) * dx + alpha * (-np.sum(np.log(h) + np.log(1 - h)) - np.log(np.abs(np.sum(h) - n_points / 2)))

def barrier_project(h, n_points, alpha=10.0):
    h = h.reshape(-1)
    h = np.clip(h, 1e-6, 1 - 1e-6)
    h = h / np.sum(h) * (n_points / 2)
    return h

def particle_update(p, v, pbest, gbest, w=0.7, c1=2.0, c2=2.0):
    r1 = np.random.rand(len(p))
    r2 = np.random.rand(len(p))
    v = w * v + c1 * r1 * (pbest - p) + c2 * r2 * (gbest - p)
    p = np.clip(p + v, 0, 1)
    p = barrier_project(p, len(p))
    return p, v

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    n_particles = 20
    max_iter = 600
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    pbest = np.zeros((n_particles, n_points))
    pbest_obj = np.full(n_particles, np.inf)

    positions[0] = initial_h_values
    for i in range(1, n_particles):
        pos = barrier_project(np.random.rand(n_points), n_points)
        positions[i] = pos

    for i in range(n_particles):
        positions[i] = barrier_project(positions[i], n_points)
        obj = objective_F(positions[i], n_points)
        if obj < pbest_obj[i]:
            pbest[i] = positions[i].copy()
            pbest_obj[i] = obj

    gbest_idx = np.argmin(pbest_obj)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = compute_c5(gbest, n_points)
    best_c5 = gbest_c5
    best_h = gbest.copy()

    time_start = time.time()
    for iter in range(max_iter):
        elapsed = time.time() - time_start
        if elapsed > budget_s:
            break

        if iter % 20 == 0:
            # Refine top 5 particles with long-step L-BFGS
            for idx in np.argsort(pbest_obj)[:5]:
                res = opt.minimize(
                    lambda x: objective_F(x, n_points, alpha=10),
                    pbest[idx],
                    bounds=[(0, 1) for _ in range(n_points)],
                    method='L-BFGS-B',
                    options={'maxiter': 100, 'ftol': 1e-6}
                )
                x = barrier_project(res.x, n_points)
                obj = compute_c5(x, n_points)
                if obj < best_c5:
                    best_c5 = obj
                    best_h = x.copy()
                if objective_F(x, n_points) < pbest_obj[idx]:
                    pbest[idx] = x.copy()
                    pbest_obj[idx] = objective_F(x, n_points, alpha=10)

        # Update velocities and positions
        for i in range(n_particles):
            positions[i], velocities[i] = particle_update(
                positions[i],
                velocities[i],
                pbest[i],
                gbest,
                w=0.6 + 0.3 * (1 - iter / max_iter),
                c1=2.0,
                c2=2.0
            )

            obj = objective_F(positions[i], n_points)
            if obj < pbest_obj[i]:
                pbest[i] = positions[i].copy()
                pbest_obj[i] = obj

        # Update global best
        gbest_idx = np.argmin(pbest_obj)
        if pbest_obj[gbest_idx] < objective_F(gbest, n_points):
            gbest = pbest[gbest_idx].copy()
            gbest_c5 = compute_c5(gbest, n_points)
            if gbest_c5 < best_c5:
                best_c5 = gbest_c5
                best_h = gbest.copy()

    final_c5 = compute_c5(best_h, n_points)
    return best_h, final_c5, n_points
