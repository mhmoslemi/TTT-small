import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    max_n_points = 100  # Cap to ensure small sequences

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init = h_init * (n_points / 2) / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_lp(h):
        h_var = cp.Variable(n_points)
        C = cp.Variable(pos=True)
        obj = cp.Minimize(C)

        constraints = [
            cp.sum(h_var) == constraint_sum,
            0 <= h_var, h_var <= 1
        ]

        for k in range(n_points):
            shifted = np.roll(h, k)  # shift to compute k + overlap operation
            overlap_expr = cp.sum(h_var * (1 - shifted))
            constraints.append(overlap_expr * dx <= C)

        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.GLPK)
        return h_var.value

    def perturb_h(h, noise_factor=0.1):
        h_pert = h.copy()
        pert = np.random.normal(0, noise_factor, h.shape)
        h_pert += pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (constraint_sum / h_pert.sum())
        return h_pert

    best_h = np.array(h_init)
    best_c5 = objective(best_h)

    h = optimize_lp(h_init)
    c5 = objective(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    for attempt in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
        h = perturb_h(best_h)
        res = optimize_lp(h)
        if res is not None:
            c5 = objective(res)
            if c5 < best_c5:
                best_h = res
                best_c5 = c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, best_c5, n_points
