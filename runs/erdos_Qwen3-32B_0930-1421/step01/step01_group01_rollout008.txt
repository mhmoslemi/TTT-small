```python
import numpy as np
import time
import math
import optproblems as optp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def generate_perturbation(h, segment_size=5, scale=0.3):
    """Perturb a segment of h to create candidate solutions."""
    n = len(h)
    idx1 = np.random.randint(0, n - segment_size)
    idx2 = idx1 + segment_size
    delta = np.random.uniform(-scale, scale, size=segment_size)
    h_candidate = h.copy()
    h_candidate[idx1:idx2] += delta
    h_candidate = np.clip(h_candidate, 0, 1)
    return rescale_h(h_candidate)


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = rescale_h(h[:n_points] if len(h) > n_points else np.pad(h, (0, n_points - len(h)), 'constant'))
    except:
        h = np.zeros(n_points)
        h = rescale_h(h)

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)
    T = 1.0
    T_min = 1e-4
    cooling_rate = 0.999
    n_steps = 100

    while time.time() - start_time < budget_s:
        if T < T_min:
            break
        for _ in range(n_steps):
            if time.time() - start_time >= budget_s:
                break
            h_candidate = generate_perturbation(h)
            C5_candidate = evaluate_C5(h_candidate, dx)
            C5_current = evaluate_C5(h, dx)

            if C5_candidate < best_C5:
                best_C5 = C5_candidate
                best_h = h_candidate.copy()

            delta = C5_candidate - C5_current
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()

        T *= cooling_rate

    return best_h, best_C5, n_points
```
