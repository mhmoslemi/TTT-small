import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def perturb_h(h, sigma, n_perturb=5):
    h = h.copy()
    n = len(h)
    idxs = np.random.choice(n, n_perturb, replace=False)
    h[idxs] += np.random.normal(0, sigma, size=len(idxs))
    h = np.clip(h, 0.0, 1.0)
    # Rescale to satisfy sum constraint
    s = h.sum()
    h *= (n / 2) / s
    return h

def project_to_simplex(v):
    # Projects to simplex sum(h) = n/2
    h = v.copy()
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - h.size / 2) > 1e-6:
        h = h / s * (h.size / 2)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    h = project_to_simplex(h)
    current_h = h.copy()
    best_h = h.copy()
    current_c5 = evaluate_overlap(h)
    best_c5 = current_c5
    t0 = time.time()
    T = 1.0
    T_min = 1e-5
    alpha = 0.95
    sigma = 0.2
    sigma_min = 0.001
    no_improvement = 0
    max_no_improvement = 500
    reset_count = 0
    max_reset = 5

    while time.time() - t0 < budget_s:
        # Perturbation step
        h_new = perturb_h(current_h, sigma, n_perturb=5)
        c5_new = evaluate_overlap(h_new)

        # Metropolis criterion
        delta = c5_new - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_c5 = c5_new
            current_h = h_new.copy()
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        # Temperature update and sigma decay
        if no_improvement >= max_no_improvement:
            no_improvement = 0
            reset_count += 1
            if reset_count >= max_reset:
                T = max(T_min, T * 1.01)
                alpha = max(0.9, alpha * 1.05)
                reset_count = 0
            T = max(T_min, T * alpha)
            sigma = max(sigma_min, sigma * 0.99)
        else:
            T = max(T_min, T * 0.999)
            sigma = max(sigma_min, sigma * 0.999)

    # Final validation
    assert abs(best_h.sum() - n_points / 2) < 1e-8, "Constraint violation"
    return (best_h, evaluate_overlap(best_h), n_points)
