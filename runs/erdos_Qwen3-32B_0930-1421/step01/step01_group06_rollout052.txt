import numpy as np
import time
from scipy.optimize import minimize
import math
import random

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    h_initial = np.copy(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2
    result_h = np.copy(h_initial)
    best_c5 = evaluate_erdos_solution(result_h, n_points)

    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(h):
        return h.sum() - target_sum

    # Create a binary initialization
    h_bin = np.zeros(n_points)
    bin_integers = int(np.round(target_sum))
    bin_indices = np.random.choice(n_points, bin_integers, replace=False)
    h_bin[bin_indices] = 1.0

    def simulated_annealing(wrapper_h):
        h_bin = np.copy(wrapper_h)
        T0 = 1.0
        T = T0
        iterations = 2000
        no_improvement_count = 0

        for it in range(iterations):
            if time.time() - start_time > budget_s - 20:
                break
            # Flip one bit to maintain a binary solution
            idx_flip_0 = np.where(h_bin == 0)[0]
            idx_flip_1 = np.where(h_bin == 1)[0]
            if len(idx_flip_0) == 0 or len(idx_flip_1) == 0:
                break
            i0 = np.random.choice(idx_flip_0)
            i1 = np.random.choice(idx_flip_1)
            h_new = np.copy(h_bin)
            h_new[i0] = 1.0
            h_new[i1] = 0.0

            c5_current = evaluate_erdos_solution(h_bin, n_points)
            c5_new = evaluate_erdos_solution(h_new, n_points)
            delta = c5_new - c5_current

            if delta < 0 or np.random.random() < np.exp(-delta / T):
                h_bin[:] = h_new
                if delta < 0:
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1

            T = T * 0.99  # Cooling schedule

        return h_bin

    # Step 1: Binary simulated annealing
    h_bin = simulated_annealing(h_bin)

    # Step 2: Binary to continuous scaling
    def bin_to_float(h_bin):
        bin_sum = h_bin.sum()
        if bin_sum == target_sum:
            return h_bin
        elif bin_sum < target_sum:
            idx_0 = np.where(h_bin == 0)[0]
            idx_insert = np.random.choice(idx_0)
            h_float = np.copy(h_bin)
            h_float[idx_insert] = 0.5
            return h_float
        else:
            idx_1 = np.where(h_bin == 1)[0]
            idx_insert = np.random.choice(idx_1)
            h_float = np.copy(h_bin)
            h_float[idx_insert] = 0.5
            return h_float

    h_float = bin_to_float(h_bin)

    # Step 3: Continuous optimization
    def optimize_h(x):
        return evaluate_erdos_solution(x, n_points)

    cons = [{"type": "eq", "fun": constraint_sum}]
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        optimize_h,
        h_float,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 100}
    )

    if result.success:
        new_c5 = evaluate_erdos_solution(result.x, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            result_h = result.x

    return result_h, best_c5, n_points
