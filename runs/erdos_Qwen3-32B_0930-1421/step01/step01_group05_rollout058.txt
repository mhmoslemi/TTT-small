import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Start from initial_h_values or use random if not provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2  # normalize to unit integral

    dx = 2.0 / n_points

    def compute_c5(h):
        hc = correlate(h, 1 - h, mode='full') * dx
        return np.max(hc)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        h = h * (n_points / 2) / total
        return h

    # Initialize firefly population
    population_size = 50
    population = []
    if 'initial_h_values' in globals():
        base = h.copy()
        population = [base.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            population.append(repair(population[-1] + noise))
    else:
        for _ in range(population_size):
            p = np.random.rand(n_points)
            p = repair(p)
            population.append(p)

    # Track best solution
    best_h = None
    best_c5 = float('inf')

    # Firefly algorithm parameters
    beta0 = 1.0  # initial attraction
    gamma_factor = 1.0 / 100  # distance sensitivity
    alpha = 0.2  # randomization factor
    max_iters = 1000  # will adjust dynamically
    current_time = time.time()
    remaining_time = budget_s - (current_time - start_time)
    iterations_per_sec = 0.8 / (remaining_time / max_iters) if remaining_time > 0 else 1000

    # Main algorithm loop
    for t in range(int(iterations_per_sec * remaining_time) if remaining_time > 0 else max_iters):
        if time.time() - start_time >= budget_s:
            break

        # Evaluate current population
        fitness_list = []
        for firefly in population:
            c5 = compute_c5(firefly)
            fitness_list.append((c5, firefly))
        fitness_list.sort()

        # Sort by C₅ value
        ranked_fireflies = [f[1] for f in fitness_list]
        ranked_fitness = [f[0] for f in fitness_list]
        global_best = ranked_fireflies[0]
        global_best_c5 = ranked_fitness[0]

        if global_best_c5 < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_c5

        # Update each firefly
        for i in range(population_size):
            current = population[i]
            current_fitness = ranked_fitness[i]
            for j in range(i):
                target = ranked_fireflies[j]
                target_fitness = ranked_fitness[j]

                if target_fitness < current_fitness:
                    # Compute Euclidean distance
                    distance = np.linalg.norm(current - target)
                    beta = beta0 * np.exp(-gamma_factor * t * distance**2)
                    step = beta * (target - current) + alpha * np.random.rand(n_points)
                    new_firefly = current + step
                    population[i] = repair(new_firefly)

    return best_h.tolist(), best_c5, n_points
