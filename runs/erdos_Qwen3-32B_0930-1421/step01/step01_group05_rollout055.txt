import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Directly use the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    dx = 2.0 / n_points

    def compute_c5(h_vec):
        c = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(c)

    def clip_and_normalize(h_vec):
        clipped = np.clip(h_vec, 0, 1)
        total = np.sum(clipped)
        scale = (n_points / 2) / total
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = clip_and_normalize(h)
    best_c5 = compute_c5(best_h)

    # CVXOPT convex optimization with L1 promotion and line search
    lambda_vals = [0.1 * (0.5)**i for i in range(6)]
    for lam in lambda_vals:
        if time.time() - start_time >= budget_s:
            break
        h_var = cp.Variable(n_points)
        c = np.arange(n_points)
        overlap = [cp.sum([h_var[i] * (1 - h_var[(i + k) % n_points]) for i in range(n_points)]) * dx for k in range(1, n_points)]
        max_overlap = cp.Max(overlap)
        objective = cp.Minimize(max_overlap + lam * cp.sum_abs(h_var))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT)
            h_candidate = np.clip(h_var.value, 0, 1)
            if np.allclose(prob.status, "optimal"):
                current_c5 = compute_c5(h_candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = clip_and_normalize(h_candidate)
        except Exception as e:
            pass

    # Final refinement with adaptive learning rate
    learning_rate = 0.1
    while time.time() - start_time < budget_s:
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += 1e-5
            h_p = clip_and_normalize(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            if h_n[i] > 1e-5:
                h_n[i] -= 1e-5
            h_n = clip_and_normalize(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / 2e-5
        best_h -= learning_rate * grad
        best_h = clip_and_normalize(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= 0.999
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
