import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def feasible_projection(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Scale to satisfy the integral constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-8:
        scale_factor = (n_points / 2) / current_sum
        h = h * scale_factor
        h = np.clip(h, 0.0, 1.0)  # Ensure values don't exceed [0,1] due to scaling
    return h

def perturb_h(h, num_perturb, sigma):
    n = len(h)
    indices = np.random.choice(n, size=min(num_perturb, n), replace=False)
    h_perturbed = h.copy()
    h_perturbed[indices] += np.random.normal(0, sigma, len(indices))
    return feasible_projection(h_perturbed, n)

        if no_improvement >= max_no_improvement:

            sigma = sigma * 0.9  # Reduce step size
            no_improvement = 0
        elif best_c5 == 0:
            break  # Optimal possible solution found

    final_c5 = evaluate_overlap(best_h)
