import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use provided initial solution if available
    n_points = 47
    constraint_sum = n_points / 2
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=np.float64, copy=True)
    else:
        h_best = np.zeros(n_points)
        h_best[::2] = 2.0 / n_points
        h_best *= constraint_sum / h_best.sum()

    dx = 2.0 / n_points

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    # Gradient-based optimization from previous configuration
    def optimize_gradient(h_seed, n_tries=1000):
        h = h_seed.copy()
        bounds = [(0, 1)] * len(h)

        def obj_func(x):
            h_test = x.copy()
            h_test = h_test.clip(0, 1)
            h_sum = h_test.sum()
            h_test = h_test * (constraint_sum / h_sum)
            return evaluate(h_test)

        result = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': n_tries, 'disp': False})
        h_opt = result.x.clip(0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    # Tabu-based exploration for global search
    tabu_list = []
    tabu_size = 15
    max_perturbations = 40

    h_current = h_best.copy()

    c5_best = evaluate(h_current)
    h_best = h_current.copy()

    perturbation_strength = 0.2
    perturbation_decay = 0.25
    perturbation_min = 0.02
    no_improvement_count = 0
    max_no_improvement = 30

    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        h_current = h_best.copy()

        # Generate perturbed candidate
        noise = np.random.normal(0, perturbation_strength, n_points)
        h_candidate = h_current + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (constraint_sum / h_candidate.sum())

        # Local optimization from perturbed candidate
        h_opt = optimize_gradient(h_candidate)

        # Constraint repair
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())

        # Evaluate candidate and check for improvement
        c5_candidate = evaluate(h_opt)
        if c5_candidate < c5_best:
            h_best = h_opt.copy()
            c5_best = c5_candidate
            no_improvement_count = 0
            # Perform round of gradient refinement
            h_best = optimize_gradient(h_best)
            c5_best = evaluate(h_best)
            c5_best = evaluate(h_best)
        else:
            no_improvement_count += 1

        # Adapt perturbation size
        if no_improvement_count >= max_no_improvement // 2:
            perturbation_strength = max(perturbation_min, perturbation_strength * perturbation_decay)

        # Add current solution to tabu list and remove older items
        h_key = tuple(h_opt.astype(np.float32).round(5))
        if h_key not in tabu_list:
            tabu_list.append(h_key)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)

        # Check termination time
        if time.time() - start_time > budget_s * 0.98:
            break

    h_values = h_best.copy()
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (constraint_sum / h_values.sum())
    c5_final = evaluate(h_values)
    return h_values, c5_final, n_points
