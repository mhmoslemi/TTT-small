import numpy as np
import time
import cvxpy as cp
import scipy.signal
import scipy.fft as fft

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    # Load initial values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # constraint: sum(h) == n_points / 2

    # Ensure constraint is exactly met
    h = h / h.sum() * target_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def evaluate_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= (target_sum / optimized_h_sum)

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= n_points:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)

            if k >= n_points:
                continue
            corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    # Fourier-based optimization setup
    def fourier_perturb_optimize(h_vals):
        nonlocal best_h, best_c5

        def fourier_transform(h_in):
            h_fft = fft.fft(h_in)
            return h_fft

        h_fft = fourier_transform(h_vals)
        num_coeffs = 20
        mask = np.zeros_like(h_fft)
        mask[:num_coeffs] = 1.0
        mask[-num_coeffs:] = 1.0

        for _ in range(100):
            # Perturb only the top 20 Fourier coefficients
            freq_perturbation = np.random.normal(scale=0.1, size=h_fft.shape)
            new_fft = h_fft + mask * freq_perturbation

            # Inverse transform
            new_h = np.real(fft.ifft(new_fft))
            # Normalize and clip
            new_h = new_h / new_h.sum() * target_sum
            new_h = np.clip(new_h, 0, 1)
            current_c5 = evaluate_c5(new_h)

            # Update best
            if current_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.380876:
                    break
            h_fft = new_fft

            # Check time budget
            if time.time() - start_time > budget_s - 10:
                break

    # Main optimization loop
    while time.time() - start_time < budget_s - 30:
        # Random search step
        idx = np.random.choice(n_points, min(5, n_points), replace=False)
        h[idx] += np.random.uniform(-0.05, 0.05, len(idx))
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx  # re-normalize to integral = 1

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Fourier-based step
        fourier_perturb_optimize(best_h)

        # Early stopping
        if best_c5 < 0.380876 or time.time() - start_time > budget_s - 30:
            break

    return (best_h, best_c5, n_points)
