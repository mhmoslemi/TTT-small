import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_peak_shifts(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    all_peak_indices = np.where(np.abs(corr) == np.max(corr))[0]
    return all_peak_indices

def flip_peak_contributors(h, peak_indices):
    n = len(h)
    h1 = 1 - h
    best_flips = None
    best_c5 = np.inf

    for k in peak_indices:
        k -= n  # since correlate is in full mode and indices > n

        if 1 <= k <= n - 1:
            left_idx = int(n - k) % n
            right_idx = int(2 * n - k) % n
            candidates = [(left_idx, right_idx), (right_idx, left_idx),
                          ((left_idx + 1) % n, (right_idx - 1) % n),
                          ((left_idx - 1) % n, (right_idx + 1) % n)]
        else:
            candidates = []

        for i, j in candidates:
            if i == j or h[i] == h[j]:
                continue
            h_new = h.copy()
            h_new[i] = 1 - h[i]
            h_new[j] = 1 - h[j]
            c5_new = evaluate_overlap(h_new)
            if c5_new < best_c5:
                best_flips = (i, j)
                best_c5 = c5_new
    if best_flips:
        i, j = best_flips
        h[i], h[j] = 1 - h[i], 1 - h[j]
    return h
