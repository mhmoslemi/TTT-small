import numpy as np
import time
from scipy.signal import correlate
from copy import copy

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    hsum = h.sum()
    if abs(hsum) < 1e-10:
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / hsum
    return h

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_fires = 50
    amplitude = 0.2
    fireflies = [np.array(initial_h_values, dtype=np.float64).copy() for _ in range(n_fires)]

    # Temperature like spread of distribution in attraction
    def distance(f1, f2):
        return np.sqrt(((f1 - f2)**2).mean())

    best_h = None
    best_c5 = float('inf')
    max_iter = 500
    gamma = 1.0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        eval_times = []
        for i in range(n_fires):
            h = fireflies[i]
            h_repair = repair_h(h, n_points)
            c5 = compute_c5(h_repair, 2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repair.copy()
            fireflies[i] = h_repair

        # Sort fireflies by brightness (negative C5)
        fireflies.sort(key=lambda h: compute_c5(h, 2.0 / n_points))

        # Move dimmer to brighter
        for i in range(n_fires):
            if time.time() - start_time > budget_s:
                break
            for j in range(i):
                h = fireflies[i].copy()
                h_star = fireflies[j]

                d = distance(h, h_star)
                beta = np.exp(-gamma * d**2)
                noise = amplitude * (np.random.rand(n_points) * 2 - 1)

                new_h = h + beta * (h_star - h) + noise
                new_h = repair_h(new_h, n_points)
                fireflies[i] = new_h

        # Dynamically decrease gamma
        gamma += 1.0 * (iter / max_iter)

        # If no improvement for a while, expand search.
        # (add novelty to fireflies)
        for i in range(n_fires):
            if np.random.rand() < 0.1:
                fireflies[i] = repair_h(np.random.rand(n_points), n_points)
