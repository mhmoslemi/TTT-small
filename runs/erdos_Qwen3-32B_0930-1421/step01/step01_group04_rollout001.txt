```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        left = n_points - k
        corr = np.dot(h[:left], 1 - h[k:k+left])
        corrs.append(corr)
    for k in range(1, n_points):
        right = n_points - k
        corr = np.dot(h[k:], 1 - h[:right])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def objective_smooth(h_vec, n_points, beta=100):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    corrs = []
    for k in range(1, n_points):
        left = n_points - k
        corr = np.dot(h[:left], 1 - h[k:k+left])
        corrs.append(corr)
    for k in range(1, n_points):
        right = n_points - k
        corr = np.dot(h[k:], 1 - h[:right])
        corrs.append(corr)
    return smooth_max(np.array(corrs) * dx, beta)

def enforce_normalization(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if total == 0:
        h_res = np.full(n_points, 0.5)
    else:
        h_res = h_clipped * (n_points / 2) / total
    return np.clip(h_res, 0, 1)

def raised_cosine_init(n_points):
    i = np.arange(n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
    return enforce_normalization(h, n_points)

def ranrot_pos(h, n, center=None):
    mask = np.random.rand(n) < 0.25
    std = np.clip(h * (1 - h), 0.1, 0.5)
    dx = np.random.randn(n) * std
    dx[mask] = 0.0
    h_p = h + dx
    h_p = enforce_normalization(h_p, n)
    return h_p

def pso_update(inertia, cogn, soc, h, v, pbest, gbest, n_points):
    omega = np.clip(inertia, 0.4, 0.7)
    phi_p = np.random.rand(n_points) * 2.5
    phi_g = np.random.rand(n_points) * 2.5
    new_v = omega * v + phi_p * (pbest - h) + phi_g * (gbest - h)
    new_h = np.clip(h + new_v, 0, 1)
    return enforce_normalization(new_h, n_points), new_v

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    n_points = 47
    if init_h is None:
        print("Initializing from default")
        h_init = raised_cosine_init(n_points)
    else:
        h_init = enforce_normalization(init_h, n_points)

    n_particles = 50
    particles = np.stack([ranrot_pos(h_init, n_points) for _ in range(n_particles)])
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    pbest = particles.copy()
    pbest_vals = np.array([compute_c5(h, n_points) for h in pbest])

    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_vals[gbest_idx]

    time_start = time.time()
    while time.time() - time_start < budget_s * 0.8:
        for i in range(n_particles):
            h = particles[i]
            beta = 50 if pbest_vals[i] < gbest_c5 else 100
            val = objective_smooth(h, n_points, beta)
            if val < pbest_vals[i]:
                pbest[i] = h.copy()
                pbest_vals[i] = compute_c5(pbest[i], n_points)
                if pbest_vals[i] < gbest_c5:
                    gbest_idx = i
                    gbest = pbest[i].copy()
                    gbest_c5 = pbest_vals[i]
        for i in range(n_particles):
            inertia = max(0.7 - (time.time() - time_start) / budget_s * 0.3, 0.4)
            particles[i], velocities[i] = pso_update(inertia, 1.5, 1.5, particles[i], velocities[i], pbest[i], gbest, n_points)
        gbest_c5 = min(pbest_vals)

    time_left = budget_s - (time.time() - time_start)
    if time_left > 0:
        res = opt.minimize(
            lambda x: objective_smooth(x, n_points, beta=100),
            x0=gbest,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': int(time_left / 0.2), 'eps': 1e-4},
            jac=False
        )
        final_h = enforce_normalization(res.x, n_points)
        final_c5 = compute_c5(final_h, n_points)
        if final_c5 < gbest_c5:
            gbest = final_h.copy()
            gbest_c5 = final_c5
    return gbest, gbest_c5, n_points
```
