import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h_population = [np.array(initial_h_values, dtype=np.float64)]
        n_points = len(h_population[0])
    else:
        h_population = [np.random.rand(45) for _ in range(100)]
        h_population = [h / h.sum() * (len(h) / 2) for h in h_population]
        n_points = 45

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def repair_h(h):
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)

    def evaluate_population():
        evaluations = []
        for h in h_population:
            c5 = compute_c5(h)
            evaluations.append((h, c5))
        evaluations.sort(key=lambda x: x[1])  # Lowest is best
        return evaluations

    best_h = None
    best_c5 = float('inf')

    # Firefly parameters
    num_fireflies = 50
    beta0 = 1.0
    gamma_factor = 0.1
    alpha = 0.2
    max_steps = 3000

    # Initialize population
    if 'initial_h_values' in globals():
        h_population = [np.array(initial_h_values, dtype=np.float64)]
        for _ in range(num_fireflies - 1):
            h = h_population[0].copy()
            h += (np.random.rand(n_points) - 0.5) * 0.1
            h = repair_h(h)
            h_population.append(h)
    else:
        h_population = [np.random.rand(n_points) for _ in range(num_fireflies)]
        h_population = [repair_h(h) for h in h_population]

    evaluations = evaluate_population()
    best_h_so_far = evaluations[0][0]
    best_c5_so_far = evaluations[0][1]

    step = 0
    while time.time() - start_time < budget_s and step < max_steps:
        evaluations = evaluate_population()
        sorted_population = [h for h, _ in evaluations]
        sorted_c5 = [c5 for _, c5 in evaluations]
        best_h = sorted_population[0]
        best_c5 = sorted_c5[0]
        if best_c5 < best_c5_so_far:
            best_h_so_far = best_h.copy()
            best_c5_so_far = best_c5

        for i in range(num_fireflies):
            h_i = sorted_population[i]
            c5_i = sorted_c5[i]
            for j in range(i):
                h_j = sorted_population[j]
                c5_j = sorted_c5[j]
                if c5_j < c5_i:
                    dist = np.sqrt(np.sum((h_i - h_j) ** 2))
                    gamma = step / (gamma_factor * max_steps)
                    beta = beta0 * np.exp(-gamma * dist)
                    rn = (np.random.rand(n_points) - 0.5) * alpha
                    h_i += beta * (h_j - h_i) + rn
                    h_i = repair_h(h_i)

        step += 1

    return best_h_so_far.tolist(), best_c5_so_far, n_points
