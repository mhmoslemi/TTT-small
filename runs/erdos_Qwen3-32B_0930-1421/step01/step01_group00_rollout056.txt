import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum

        ones_minus_h = 1.0 - optimized_h
        corr_full = scipy.signal.correlate(optimized_h, ones_minus_h, mode="full")
        max_corr = np.max(corr_full[n_points - 1 : 2 * n_points - 1]) * dx
        return max_corr

    # Fourier initialization
    def fourier_init(optimized_h, K=40):
        H_fou = np.fft.fft(optimized_h)
        H_fou[K:-K] = 0
        return np.fft.ifft(H_fou).real

    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= target_sum / h_sum
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_time = time.time()

    it_start = time.time()
    num_tests = 0
    freq_step = 2

    while time.time() - start_time < budget_s - 5:
        # Frequencies to perturb: top contributors
        if num_tests % 1000 == 0:
            tmp_h = fourier_init(h, K=20)
            tmp_h = np.clip(tmp_h, 0, 1)
            tmp_h_sum = tmp_h.sum()
            if tmp_h_sum != target_sum:
                tmp_h *= target_sum / tmp_h_sum
            new_c5 = compute_c5(tmp_h)
            if new_c5 < best_c5:
                best_h = tmp_h.copy()
                best_c5 = new_c5
                best_time = time.time()
                h[:] = best_h.copy()
                if verbose:
                    print(f"Rewiring: {best_c5:.6f}")

        # Local perturbation using Fourier space
        perturb_h = fourier_init(h, K=freq_step).copy()
        indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        noise = np.random.normal(loc=0.0, scale=0.02, size=len(indices))
        perturb_h[indices] += noise
        perturb_h = np.clip(perturb_h, 0, 1)
        perturb_h_sum = perturb_h.sum()
        if perturb_h_sum != target_sum:
            perturb_h *= target_sum / perturb_h_sum
        perturb_h = np.clip(perturb_h, 0, 1)

        new_c5 = compute_c5(perturb_h)
        if new_c5 < best_c5:
            h[:] = perturb_h.copy()
            best_h = h.copy()
            best_c5 = new_c5
            best_time = time.time()
            if verbose:
                print(f"Improved to {best_c5:.6f} at time {time.time()-start_time:.2f}s")

        num_tests += 1
        if time.time() - best_time > 60:
            freq_step = max(2, freq_step - 1)

        # Progress check
        if verbose and num_tests % 500 == 0:
            print(f"Iteration: {num_tests}, Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}, Best at {best_time - start_time:.2f}s")

    return (best_h, best_c5, n_points)
