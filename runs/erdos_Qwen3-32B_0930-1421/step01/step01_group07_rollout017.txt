import numpy as np
import time
from scipy.signal import correlate

def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_impactful_flips(h, c5):
    n = len(h)
    idx_0 = np.where(h == 0)[0]
    idx_1 = np.where(h == 1)[0]
    best_impact = 0
    best_pair = None

    max_corr_idx = np.argmax(corr)
    corr_center = max_corr_idx - n + 1  # Convert to shift

    for i in idx_1:
        for j in idx_0:
            if abs(i - j) > 10:  # Local perturbation only, skip non-local flips
                continue
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            new_c5 = compute_overlap(h_new)
            impact = c5 - new_c5
            if impact > best_impact:
                best_impact = impact
                best_pair = (i, j)
    return best_pair
