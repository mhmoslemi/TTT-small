```python
import numpy as np
import time
from scipy.signal import correlate
from collections import deque
import heapq


initial_h_values = None  # Will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    corr = correlate(h, ones_minus_h, mode="full")
    return np.max(corr) * dx


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h[0] = 1.0
        total = 1.0
    scale = (len(h) / 2.0) / total
    h *= scale
    return h


def build_candidates(h, depth=3, refine=True):
    """Build minor variations around current h for search space exploration."""
    n_points = len(h)
    candidates = []
    base = np.copy(h)
    values = [0, 0.25, 0.5, 0.75, 1] if refine else [0, 1]
    scale = len(h) / 2.0

    indices = np.random.choice(n_points, size=min(8, n_points), replace=False)
    for i in indices:
        for v in values:
            new_h = np.copy(base)
            new_h[i] = v
            new_h = ensure_feasible(new_h)
            candidates.append(new_h)
    return candidates


def local_branching(h_base, depth=3, refine=True):
    """Explore local branches with controlled perturbations."""
    n_points = len(h_base)
    todo = deque()
    todo.append((h_base, 0, 0.0))
    best = (None, np.inf)
    best_h = h_base
    best_val = evaluate_C5(h_base)

    while todo and todo[0][1] <= depth:
        h, curr_depth, _ = todo.popleft()
        val = evaluate_C5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if curr_depth < depth:
            new_candidates = build_candidates(h, refine=refine)
            for h2 in new_candidates:
                todo.append((h2, curr_depth + 1, evaluate_C5(h2)))
    return best_h, best_val


def bandit_selection(history, temperature=0.2):
    """Use softmax to select the most promising candidate."""
    scores = [-c for c in history.values()]
    probs = np_softmax(scores, temperature)
    return np.random.choice(list(history.keys()), p=probs)


def np_softmax(scores, temperature=1.0):
    logits = (np.array(scores) / temperature)
    exp_logits = np.exp(logits - np.max(logits))
    return exp_logits / np.sum(exp_logits)


def construct_initial_h(n_points):
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[::4] = 1.0
        h = ensure_feasible(h)
    return h


def run(budget_s=1100, n_points=42, start_depth=2, refine=True, branch_limit=50):
    start_time = time.time()
    best_h = construct_initial_h(n_points)
    best_C5 = evaluate_C5(best_h)

    history = {}
    history[tuple(best_h.round(4))] = best_C5

    local_count = 0
    while time.time() - start_time < budget_s:
        try:
            seed_inputs = np.copy(best_h)
            h_candidate, c5 = local_branching(seed_inputs, depth=start_depth, refine=refine)
            if c5 < best_C5:
                best_h = h_candidate
                best_C5 = c5
                local_count = 0
                history[tuple(h_candidate.round(4))] = best_C5
            else:
                local_count += 1

            # Diversify if stuck
            if local_count > branch_limit:
                best_h = construct_initial_h(n_points)
                best_C5 = evaluate_C5(best_h)
                history.clear()
                history[tuple(best_h.round(4))] = best_C5
                local_count = 0

        except KeyboardInterrupt:
            break
        except:
            continue

    return best_h, best_C5, n_points
```
