import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    n_points = len(h)
    dx = 2.0 / n_points

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total_area = np.sum(clipped) * dx
        scale = (1.0 / total_area) * dx
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # CVXOPT convex optimization with L1 regularization
    variables = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.abs(np.correlate(variables, 1 - variables, mode='full') * dx))
                            + 1e-1 * cp.norm1(variables))
    constraints = [variables >= 0,
                   variables <= 1,
                   cp.sum(variables) * dx == 1]
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.CVXOPT, maxiters=100)
        if problem.status == cp.OPTIMAL:
            h_opt = variables.value
            if h_opt is not None:
                h_opt = np.nan_to_num(h_opt, nan=0.0)
                h_opt = normalize_h(h_opt)
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = current_c5
    except:
        pass

    # Gradient descent refinement
    gradient_steps = 500
    learning_rate = 0.05
    eps = 1e-6

    for step in range(gradient_steps):
        if time.time() - start_time >= budget_s:
            break

        grad = np.zeros(n_points)
        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        best_c5 = compute_c5(best_h)

        # Adaptive learning rate
        if step % 10 == 0:
            learning_rate *= 0.95

    return best_h.tolist(), best_c5, n_points
