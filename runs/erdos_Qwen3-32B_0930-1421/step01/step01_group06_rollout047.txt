import numpy as np
import time
from scipy.optimize import Bounds

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(budget_s=1100):
    start_time = time.time()
    np.random.seed(42)

    h_init = np.copy(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0

    # Normalization step to meet the integral constraint
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = h_init * (target_sum / h_init.sum())

    # PSO parameters
    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    reset_fraction = 0.1

    # Initialize particles
    particles = np.random.rand(num_particles, n_points) * 0.2 + h_init[np.newaxis, :]
    velocities = np.random.uniform(-max_velocity, max_velocity, (num_particles, n_points))

    # Project to feasible region
    def project(x):
        x = np.clip(x, 0.0, 1.0)
        x = x * (target_sum / x.sum(keepdims=True))
        return x

    particles = [project(row) for row in particles]
    velocities = [v * 0.2 for v in velocities]

    p_best = np.copy(particles)
    p_best_vals = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
    g_best = particles[np.argmin(p_best_vals)]
    g_best_val = np.min(p_best_vals)

    last_improvement = start_time

    def update_params(t):
        w = w_start - (w_start - w_end) * (t / 200.0)
        return w

    while time.time() - start_time < budget_s - 5:
        t = int((time.time() - start_time) * num_particles * n_points / 1000)  # simple progress

        for i in range(num_particles):
            w = update_params(t)

            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best[i] - particles[i]) +
                c2 * r2 * (g_best - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i])

            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], n_points)
            if c5 < p_best_vals[i]:
                p_best[i] = np.copy(particles[i])
                p_best_vals[i] = c5
                if c5 < g_best_val:
                    g_best = np.copy(particles[i])
                    g_best_val = c5
                    last_improvement = time.time()

        # Reset a fraction of the swarm to maintain diversity
        num_resets = int(reset_fraction * num_particles)
        reset_ids = np.random.choice(num_particles, num_resets, replace=False)

        for idx in reset_ids:
            noise = np.random.uniform(-0.1, 0.1, size=n_points)
            particles[idx] += noise
            particles[idx] = project(particles[idx])
            # Update velocity for reset particles
            velocities[idx] = np.random.uniform(-max_velocity, max_velocity, n_points)

        # Termination condition
        if time.time() - last_improvement > 60:
            break

    return g_best, g_best_val, n_points
