import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    # Rescale to meet mean=0.5 => sum = n_points / 2
    total = np.sum(h)
    target = n_points / 2.0
    if abs(total - target) > 1e-8:
        h = h * (target / total)
    return h

def compute_gradient(h, lag, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + lag <= n_points - 1:
            grad[i] += (1 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    grad = grad / np.linalg.norm(grad)
    return grad

    h = np.clip(h, 0, 1)
    h = project_to_feasible(h, n_points)

    max_iter = 10000
    alpha = 1e-2
    min_alpha = 1e-6
    best_c5 = evaluate_overlap(h)
    best_h = h.copy()
    no_improvement = 0
    max_no_improvement = 100
    iteration = 0

    while time.time() - t0 < budget_s and iteration < max_iter:
        # Compute current overlap max
        corr = correlate(h, 1 - h, mode="full")
        k_shift = np.argmax(corr)
        lag = k_shift - (n_points - 1)

        # Compute gradient for negative update
        grad = compute_gradient(h, lag, n_points)

        # Line search with backtracking
        temp_alpha = alpha
        update_h = h - temp_alpha * grad
        update_h = project_to_feasible(update_h, n_points)
        new_c5 = evaluate_overlap(update_h)

        # Backtracking
        while new_c5 >= best_c5 and temp_alpha > min_alpha:
            temp_alpha *= 0.5
            update_h = h - temp_alpha * grad
            update_h = project_to_feasible(update_h, n_points)
            new_c5 = evaluate_overlap(update_h)

        # Update if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = update_h.copy()
            h = update_h
            no_improvement = 0
            alpha = min(1.1 * alpha, 1e-1)
        else:
            no_improvement += 1
            alpha = max(alpha * 0.9, min_alpha)

        iteration += 1
        if no_improvement >= max_no_improvement:
            # Perturb to escape local minimum
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h = h + perturb
            h = project_to_feasible(h, n_points)
            no_improvement = 0

    assert np.all(h >= 0) and np.all(h <= 1), "Constraint violated: values out of [0,1]"
    assert abs(np.sum(h) - n_points / 2) < 1e-6, "Sum constraint violated"
    return (best_h, best_c5, n_points)
