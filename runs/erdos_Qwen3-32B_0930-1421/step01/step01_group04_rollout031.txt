import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    h = h * (n_points / 2.0) / total
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    max_overlap = 0.0
    for k in range(1, n_points):
        overlap = 0.0
        for i in range(n_points - k):
            overlap += h[i] * (1.0 - h[i + k])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def setup_initial_lp(n_points, initial_h):
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    # Start with a small set of constraints
    initial_k = [1, 2, 4, 8, 16]
    for k in initial_k:
        if k >= n_points:
            continue
        overlap = 0.0
        for i in range(n_points - k):
            overlap += h[i] * (1 - h[i + k])
        constraints.append(overlap * (2.0 / n_points) <= C)
    
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)

    # Set initial value for h
    h_init = initial_h.copy()
    h_init = enforce_normalization(h_init, n_points)
    for k_t, hi_t in zip(h, h_init):
        k_t.value = hi_t

    return prob, h, C

def column_generation(h_initial, n_points, budget_s):
    h_best = h_initial.copy()
    c_best = compute_c5(h_best, n_points)
    time_start = time.time()

    # Setup initial LP
    h_var = cp.Parameter(n_points)
    h_var.value = h_best
    prob, h_lp, C_lp = setup_initial_lp(n_points, h_best)

    while time.time() - time_start < budget_s:
        # Solve LP with current constraints
        h_init = h_best.copy()
        h_var.value = h_init
        h_val = h_var.value
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except:
            prob.solve(solver=cp.SCIP, verbose=False)

        if prob.status != cp.OPTIMAL:
            break

        h_curr = h_lp.value
        h_curr = enforce_normalization(h_curr, n_points)

        # Compute C5 for current solution
        c_curr = compute_c5(h_curr, n_points)
        if c_curr < c_best:
            h_best = h_curr.copy()
            c_best = c_curr

        # Find the worst constraint (most violating one)
        max_overlap = 0.0
        worst_k = 0
        for k in range(1, n_points):
            overlap = 0.0
            for i in range(n_points - k):
                val = h_curr[i] * (1 - h_curr[i + k])
                overlap += val
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k

        # Add new constraint if violated
        if max_overlap > c_curr + 1e-5:
            overlap = 0.0
            for i in range(n_points - worst_k):
                overlap += h_lp[i] * (1 - h_lp[i + worst_k])
            prob.constraints.append(overlap * (2.0 / n_points) <= C_lp)

    # Final polishing
    h_final, c_final = final_refinement(h_best, n_points)
    return h_final, c_final

def final_refinement(h_curr, n_points):
    bounds = [(0, 1)] * n_points
    def objective(h):
        h = h.reshape(-1)
        return compute_c5(h, n_points)

    result = opt.minimize(objective, h_curr, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
    h = result.x
    h = enforce_normalization(h, n_points)
    return h, compute_c5(h, n_points)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Fallback to a rough distribution
        n_points = 47
        comb_period = int(n_points / 4)
        initial_h = np.zeros(n_points)
        indices = np.arange(0, n_points, comb_period)
        initial_h[indices] = 1.0 / comb_period  # Distribute mass evenly
    n_points = len(initial_h)
    initial_h = enforce_normalization(initial_h, n_points)

    result_h, result_c5 = column_generation(initial_h, n_points, budget_s)

    return result_h, result_c5, n_points
