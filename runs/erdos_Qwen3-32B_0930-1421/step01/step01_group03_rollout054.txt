import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # Cap to ensure small sequences

    # Use provided initial construction or default if none
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()  # Normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2
    max_overlap = float('inf')

    def objective_fn(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_fixed_support(initial_h):
        h = initial_h.copy()
        idxs = np.where(h > 0)[0]
        n_active = len(idxs)
        if n_active == 0:
            return h

        hvars = cp.Variable(n_active)
        ones = np.ones(n_active)
        c = constraint_sum / n_active
        objective = cp.Minimize(cp.Max(cp.conv(hvars, ones - hvars) * dx))
        constraints = [
            0 <= hvars,
            hvars <= 1,
            cp.sum(hvars) == constraint_sum
        ]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.ECOS)
        h_opt = np.zeros_like(h)
        h_opt[idxs] = hvars.value
        return np.clip(h_opt, 0, 1)

    def perturb_support(h, noise_factor=0.05):
        h = h.copy()
        idxs = np.arange(n_points)
        np.random.shuffle(idxs)
        for i in range(min(10, n_points)):
            j = idxs[i]
            h[j] += noise_factor * (2 * np.random.rand() - 1)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        return h

    best_h = h_init.copy()
    best_c5 = objective_fn(best_h)

    # Phase 1: Fix support, convex optimize within it
    h = optimize_fixed_support(best_h)
    c5 = objective_fn(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    perturb_count = 0
    while time.time() - start_time < budget_s * 0.99:
        perturb_count += 1
        h = perturb_support(best_h)
        h = optimize_fixed_support(h)
        c5 = objective_fn(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

        # Break if plateauing
        if perturb_count % 50 == 0:
            if np.isclose(best_c5, max_overlap, atol=1e-5):
                break

    # Last check
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    best_h = best_h * (constraint_sum / best_h_sum)
    best_c5 = objective_fn(best_h)

    return best_h, best_c5, n_points
