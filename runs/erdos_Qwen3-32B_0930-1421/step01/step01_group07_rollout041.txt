import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    i = np.random.choice(indices_1)
    j = np.random.choice(indices_0)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

    # Enforce sum constraint with deterministic correction
    def repair_h(h, target_sum):
        while np.sum(h) != target_sum:
            excess = np.sum(h) - target_sum
            flip_from, flip_to = (1, 0) if excess > 0 else (0, 1)
            candidates = np.where(h == flip_from)[0]
            if len(candidates) == 0:
                # If infeasible, break and return as it is
                break
            idx = np.random.choice(candidates)
            h[idx] = flip_to

    target_sum = n_points / 2
    repair_h(h, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    start = time.time()

    MAX_ITERS = 10000
    NO_IMPROVEMENT_LIMIT = 500
    CONFLICT_RESOLUTION_ITERATIONS = 500
    no_improvement_count = 0

    for it in range(MAX_ITERS):
        if time.time() - start > budget_s:
            break

        h_candidate = h.copy()
        corr = correlate(h_candidate, 1 - h_candidate, mode="full")
        max_corr_idx = np.argmax(corr)
        max_corr = np.max(corr)

        # If already at minimum, we can skip exploring further
        if max_corr <= best_c5:
            continue

        n = len(h_candidate)
        dx = 2.0 / n
        corr_overlap = max_corr * dx
        if corr_overlap < best_c5:
            best_c5 = corr_overlap
            best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= NO_IMPROVEMENT_LIMIT:
            # Try multiple bit flips in the most impactful regions
            for _ in range(CONFLICT_RESOLUTION_ITERATIONS):
                h_candidate = flip_two_bits(h_candidate)
                repair_h(h_candidate, target_sum)
                curr_c5 = evaluate_overlap(h_candidate)
                if curr_c5 < best_c5:
                    h = h_candidate.copy()
                    best_c5 = curr_c5
                    no_improvement_count = 0
                    break
            else:
                no_improvement_count = 0

    # Final validation
    repair_h(best_h, target_sum)
    final_c5 = evaluate_overlap(best_h)
    assert abs(np.sum(best_h) - n_points / 2) < 1.e-6, "Sum constraint violated"
    return best_h, final_c5, n_points
