```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math


initial_h_values = None  # Will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h, dx):
    # Enforce [0,1] bounds and adjust to ∑h = n_points / 2
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total < 1e-12:
        h[:] = 0.0
        h[0] = 1.0
    else:
        scale = (n_points * dx) / total
        h *= scale
    return h


def generate_basis(n_basis, n_points, alpha=4.0, beta=0.3):
    # Create a set of Gaussian-like basis functions
    x = np.linspace(0, 2, n_points)
    centers = np.linspace(0, 2, n_basis + 2)[2:-2]  # skip boundary points

    # Create basis functions
    bases = []
    for xi in centers:
        phi = np.exp(-alpha * (x - xi)**2)
        mask = np.abs(x - xi) < beta
        phi *= mask  # zero outside window
        bases.append(phi)

    return np.array(bases)


def setup_optimization(bases, dx):
    n_basis = len(bases)
    n_points = len(bases[0])

    # Optimization variables
    a = cp.Variable(n_basis, nonneg=True)

    # Compute overlap integral constraints
    max_k = 40  # max shift k in [0, 4]
    overlaps = []
    for k in range(5, max_k + 1):  # symmetric shifts
        shift = k
        if shift >= n_points:
            break
        mat = cp.Parameter((n_points, n_basis))
        C = cp.Parameter(n_basis, n_basis)
        phi1 = np.array([cp.Constant(bases[:, i]) for i in range(n_basis)])
        phi2 = np.array([cp.Constant(bases[(i + shift) % n_points] for i in range(n_basis))])
        overlaps.append(cp.sum(cp.multiply(phi1, phi2 * (1 - phi1))) * dx)

    objective = cp.Minimize(cp.sum(overlaps))
    constraints = [
        cp.sum(cp.multiply(a, np.sum(bases, axis=0) * dx)) == 1,
        cp.sum(a) <= n_basis
    ]
    prob = cp.Problem(objective, constraints)
    return prob, a


def build_h(bases, a_coefs):
    h = np.sum(bases.T * a_coefs, axis=1)
    dx = 2.0 / len(h)
    return ensure_feasible(h, dx)


def run(budget_s=1100, n_basis=10, base_alpha=4.0, base_beta=0.3):
    start_time = time.time()
    n_points = 42  # starting resolution
    best_h = None
    best_C5 = np.inf
    current_h = initial_h_values.copy() if initial_h_values is not None else None

    # Strategy 1: Build initial basis model and optimize
    bases = generate_basis(n_basis, n_points, base_alpha, base_beta)
    prob, a = setup_optimization(bases, 2.0 / n_points)

    while time.time() - start_time < budget_s * 0.9:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            a_vals = a.value
            h_candidate = build_h(bases, a_vals)
            C5_candidate = evaluate_C5(h_candidate)
            if C5_candidate < best_C5:
                best_h = h_candidate
                best_C5 = C5_candidate
                print(f"New best: C5 = {best_C5:.6f}")

            # Perturbation: small change to basis centers
            # TODO: future refinement could explore this
            break
        except Exception as e:
            print(f"Optimization failed: {e}")

    return best_h, best_C5, n_points
```
