import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize h to meet constraint ∑h[i] = n_points / 2
    h_sum = np.sum(h)
    h = h * (target_sum / h_sum)
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(current_h):
        ones_minus_h = 1.0 - current_h
        current_h = np.clip(current_h, 0.0, 1.0)
        corr_forward = np.correlate(current_h, ones_minus_h, mode='full')
        corr_backward = np.correlate(ones_minus_h, current_h, mode='full')
        corr_values = np.concatenate((corr_forward, corr_backward))[n_points:-n_points] * dx
        return np.max(corr_values)

    best_c5 = compute_c5(h)

    # Phase 1: convex relaxation via smooth proxy
    x_var = cp.Variable(n_points)
    obj_c5 = 0
    for k in range(1, n_points - 1):
        obj_k = cp.sum([x_var[i] * (1 - x_var[i + k]) for i in range(n_points - k)])
        obj_c5 += obj_k
    obj_c5 *= dx

    constraints = [
        x_var >= 0,
        x_var <= 1,
        cp.sum(x_var) == target_sum
    ]
    problem = cp.Problem(cp.Minimize(obj_c5), constraints)
    problem.solve(solver=cp.OSQP, max_iter=1000, verbose=False)

    new_h = np.clip(np.array(x_var.value).flatten(), 0., 1.)
    new_c5 = compute_c5(new_h)
    if new_c5 < best_c5:
        best_h = new_h
        best_c5 = new_c5

    # Phase 2: gradient-like local search
    elapsed = time.time() - start_time
    while elapsed < budget_s:
        # Perturb small set of indices
        indices = np.random.choice(n_points, size=min(6, n_points), replace=False)
        proposed_h = best_h.copy()
        proposed_h[indices] += np.random.normal(0, 0.02, size=indices.size)
        proposed_h = np.clip(proposed_h, 0.0, 1.0)
        proposed_h_sum = np.sum(proposed_h)
        if abs(proposed_h_sum - target_sum) > 1e-7:
            scale = target_sum / proposed_h_sum
            proposed_h = np.clip(proposed_h * scale, 0.0, 1.0)

        new_c5 = compute_c5(proposed_h)
        if new_c5 < best_c5:
            best_h = proposed_h.copy()
            best_c5 = new_c5

        elapsed = time.time() - start_time

    # Final clean constraint enforcement
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - target_sum) > 1e-7:
        scale = target_sum / best_h_sum
        best_h = np.clip(best_h * scale, 0.0, 1.0)

    return (best_h, best_c5, n_points)
