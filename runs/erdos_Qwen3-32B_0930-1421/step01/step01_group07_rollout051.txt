import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_max_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    max_index = np.argmax(corr)
    max_shift = max_index - n + 1
    return abs(max_shift), corr[max_index] * dx

def optimize_flip(h, k_opt):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    indices_1_at_k = np.where(h[n - k_opt:] == 1)[0] if k_opt > 0 else np.array([])

    if len(indices_1_at_k) == 0:
        return h  # No optimization possible without violating sum constraint

    frontier = np.minimum(indices_1_at_k + k_opt, n - 1)
    one_idx = indices_1[np.argmin(np.abs(indices_1[:, None] - frontier[None, :]).min(axis=0))]

    zero_idx = indices_0[np.argmin(np.abs(indices_0[:, None] - one_idx + k_opt))]

    h_new = h.copy()
    h_new[one_idx] = 0.0
    h_new[zero_idx] = 1.0
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h[h <= 0.5] = 0.0
    n_points = len(h)
    while np.sum(h) != n_points / 2:
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    best_h = h.copy()
    best_c5 = c5 = np.inf
    t0 = time.time()
    iteration = 0

    while time.time() - t0 < budget_s:
        iteration += 1
        k_opt, c5 = find_max_overlap(best_h)
        h_new = optimize_flip(best_h, k_opt)
        sum_h = np.sum(h_new)
        if not (abs(sum_h - n_points / 2) < 1e-8):  # Feasibility check
            continue
        c5_new = evaluate_overlap(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Final validation failed"
    return (best_h, best_c5, n_points)
