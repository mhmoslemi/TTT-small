import numpy as np
import time
import scipy.signal


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h[:5] = 0.1
        h[5:42] = 0.3
        h[42:] = 0.2
        h = h / h.sum() * n_points / 2

    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_C5 = np.inf
    start_time = time.time()

    def rescale_h_input(x):
        x = np.clip(x, 0, 1)
        x_sum = x.sum()
        if x_sum == 0:
            x[0] = n_points / 2
            return x
        return x / x_sum * n_points / 2

    def compute_C5_seq(h):
        h = rescale_h_input(h)
        cor = scipy.signal.correlate(h, 1 - h, mode="full")
        C5 = np.max(cor) * dx
        return C5

    best_C5 = compute_C5_seq(h)
    best_h = h.copy()
    amplitude = 0.1
    perturb_freq = 50
    step_size = 0.001
    iter_count = 0

    while time.time() - start_time < budget_s:
        h_current = best_h.copy()
        # Random segment update
        flip_start = np.random.randint(0, n_points - 5)
        flip_end = min(flip_start + 5, n_points)
        h_current[flip_start:flip_end] = 1 - h_current[flip_start:flip_end]
        h_current = rescale_h_input(h_current)

        # Gaussian perturbation
        perturb_mask = np.random.rand(n_points) < 0.2
        h_candidate = h_current + amplitude * (2 * np.random.rand(n_points) - 1) * perturb_mask
        h_candidate = rescale_h_input(h_candidate)

        # Compute C5
        C5_candidate = compute_C5_seq(h_candidate)

        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h_candidate.copy()
            amplitude *= 0.9
        elif np.random.rand() < 0.1:
            best_h = h_candidate.copy()

        # Adjust amplitude
        iter_count += 1
        if iter_count % 100 == 0:
            amplitude -= 0.01
            amplitude = max(0.01, amplitude)

    h_vals, c5, n_pts = best_h, best_C5, n_points
    return h_vals, c5, n_pts
