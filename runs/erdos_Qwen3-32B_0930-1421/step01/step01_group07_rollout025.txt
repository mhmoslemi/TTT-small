import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def adjust_h(h):
    n = len(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    i = ones[np.random.randint(len(ones))]
    j = zeros[np.random.randint(len(zeros))]
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    h = initial_h_values.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0

    # Normalize the sum
    def fix_sum(h):
        s = np.sum(h)
        target = n_points / 2
        while s != target:
            if s < target:
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 1
            else:
                idx = np.random.choice(np.where(h == 1)[0])
                h[idx] = 0
            s = np.sum(h)
        return h

    h = fix_sum(h)
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    t0 = time.time()
    t = 0

    while time.time() - t0 < budget_s:
        h_proposal = adjust_h(h.copy())
        h_proposal = fix_sum(h_proposal)
        c5_proposal = evaluate_overlap(h_proposal)

        if c5_proposal < best_c5:
            best_h = h_proposal.copy()
            best_c5 = c5_proposal
            h = best_h.copy()
            # Fast reset temperature-like cooling
            t = 0
        t += 1

        if t > 200:  # arbitration limit
            break

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8
    return (best_h, best_c5, n_points)
