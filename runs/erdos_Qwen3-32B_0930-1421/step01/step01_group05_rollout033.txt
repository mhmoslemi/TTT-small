import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as a starting solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        ones = np.ones_like(h)
        neg_h = ones - h
        corr = np.correlate(h, neg_h, mode='full')
        return np.max(corr * dx)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        scale = (n_points / 2.0) / total
        return np.clip(clipped * scale, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    total_phases = 3
    phase_duration = budget_s / total_phases

    # Phase 1: Local optimization with bounded random perturbations
    remaining_time = time.time() - start_time
    while time.time() - start_time < phase_duration:
        # Perturb the current solution with bounded noise
        step_size = 0.01
        h_update = h + np.random.uniform(-step_size, step_size, n_points)
        h_update = normalize_h(h_update)
        c5_val = compute_c5(h_update)
        if c5_val < best_c5:
            h = h_update.copy()
            best_c5 = c5_val
            best_h = h_update.copy()

    # Phase 2: ILP-inspired optimization (loose relaxation with optimization)
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0 and time.time() - start_time < 2 * phase_duration:
        try:
            # Construct a systematic trade-off of h values using optimization
            h_var = cp.Variable(n_points, nonneg=True)
            constraints = [cp.sum(h_var) == n_points / 2, h_var <= 1]
            obj = cp.Minimize(cp.Max([cp.sum(h_var[i] * (1 - h_var[(i + k) % n_points]) 
                           for i in range(n_points)]) for k in range(1, n_points)])
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.SCIP, verbose=False, time_limit=remaining_time)
            if problem.status == cp.OPTIMAL:
                h = np.clip(h_var.value, 0, 1).copy()
                h = normalize_h(h)
                c5_val = compute_c5(h)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
        except Exception as e:
            pass

    # Phase 3: Fine-tuning using coordinate descent and precise convergence
    learning_rate = 0.05
    decay = 0.995
    while time.time() - start_time < budget_s:
        for idx in np.random.permutation(n_points):
            for delta in [0.001, -0.001]:
                h_test = h.copy()
                h_test[idx] += delta
                h_test = normalize_h(h_test)
                c5_new = compute_c5(h_test)
                if c5_new < best_c5:
                    h = h_test.copy()
                    best_c5 = c5_new
                    best_h = h.copy()
        learning_rate *= decay
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
