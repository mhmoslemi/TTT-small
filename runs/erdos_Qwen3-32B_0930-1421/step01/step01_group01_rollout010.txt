import numpy as np
import time
import scipy.signal
import scipy.optimize as opt


def rescale_h(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum < 1e-6:
        h[:] = 0.0
        h[0] = len(h) / 2
    else:
        h = h / h_sum * len(h) / 2
    return h


def evaluate_C5(h, dx):
    h = rescale_h(h)
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def run(seed=42, budget_s=1100, n_points=84):

        h = rescale_h(h)

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)

    T_initial = 2.0
    alpha = 0.99
    temp = T_initial
    T_min = 1e-4
    N_steps = 50
    perturb_amp = 0.15
    accepted = 0
    move_size = 6

    while time.time() - start_time < budget_s and temp > T_min:
        for _ in range(N_steps):
            if time.time() - start_time >= budget_s:
                break

            # Perturb a random segment to avoid direction bias
            a = np.random.randint(0, n_points - move_size)
            b = a + move_size
            perturb = np.random.normal(0, perturb_amp, move_size)
            new_h = h.copy()
            new_h[a:b] = new_h[a:b] + perturb
            new_h = rescale_h(new_h)
            C5_new = evaluate_C5(new_h, dx)

            if C5_new < best_C5:
                h[:] = new_h[:]

                accepted += 1
            else:

                    h[:] = new_h[:]
                    accepted += 1

        # Adjust temperature regime based on acceptance burn-in
        if accepted == 0 and temp > 0.5:

            temp = max(temp * 1.2, T_min)
