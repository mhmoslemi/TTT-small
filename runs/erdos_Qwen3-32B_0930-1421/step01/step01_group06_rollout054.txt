import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure sum(h) = n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    # Binary initialization with simulated annealing
    def binary_anneal():
        nonlocal best_h, best_c5, last_improvement_time
        h_bin = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h_bin[::2] = 0
        h_bin[1::2] = 1
        total_flips = int(n_points * 0.2)
        h_bin[np.random.choice(n_points, total_flips)] = 1 - h_bin[np.random.choice(n_points, total_flips)]
        total_sum = np.sum(h_bin)
        if total_sum < n_points / 2:
            indices = np.where(h_bin == 0)[0]
            for i in range(n_points):
                if h_bin[indices[i]] == 0:
                    h_bin[indices[i]] = 1
                    if np.sum(h_bin) >= n_points / 2:
                        break
        else:
            indices = np.where(h_bin == 1)[0]
            for i in range(n_points):
                if h_bin[indices[i]] == 1:
                    h_bin[indices[i]] = 0
                    if np.sum(h_bin) <= n_points / 2:
                        break

        T0 = 1.0
        T = T0
        cool_factor = 0.999
        max_iter = int(budget_s // 2)
        best_h_bin = np.copy(h_bin)
        best_c5_bin = evaluate_erdos_solution(best_h_bin, n_points)

        for step in range(max_iter):
            if time.time() - start_time >= budget_s - 20:
                break
            # Flip a random bit
            i1 = np.random.randint(n_points)
            i2 = np.random.randint(n_points)
            old_bit1 = h_bin[i1]
            old_bit2 = h_bin[i2]
            while np.sum(h_bin) + (1 - 2 * h_bin[i1]) + (1 - 2 * h_bin[i2]) < 0 or np.sum(h_bin) + (1 - 2 * h_bin[i1]) + (1 - 2 * h_bin[i2]) > n_points:
                i1 = np.random.randint(n_points)
                i2 = np.random.randint(n_points)
                old_bit1 = h_bin[i1]
                old_bit2 = h_bin[i2]
            h_bin[i1] = 1 - h_bin[i1]
            h_bin[i2] = 1 - h_bin[i2]

            # Evaluate new cost
            c5 = evaluate_erdos_solution(h_bin, n_points)
            if c5 < best_c5_bin:
                best_c5_bin = c5
                best_h_bin = np.copy(h_bin)
                last_improvement_time = time.time()
            else:
                prob = np.exp(-(c5 - best_c5_bin) / max(1e-6, T))
                if np.random.rand() > prob:
                    h_bin[i1] = old_bit1
                    h_bin[i2] = old_bit2
            T *= cool_factor

        # Scale to meet exact constraint
        h_imp = np.copy(best_h_bin)
        h_imp = np.minimum(h_imp, 1.0)
        h_imp = np.maximum(h_imp, 0.0)
        total_sum = np.sum(h_imp)
        if total_sum < n_points / 2:
            indices = np.where(h_imp == 0)[0]
            h_imp[indices[0]] = n_points / 2 - (total_sum - 0)
        elif total_sum > n_points / 2:
            indices = np.where(h_imp == 1)[0]
            h_imp[indices[0]] = total_sum - (n_points / 2)
        return h_imp

    # Run simulated annealing
    h_imp = binary_anneal()
    best_c5_imp = evaluate_erdos_solution(h_imp, n_points)
    if best_c5_imp < best_c5:
        best_c5 = best_c5_imp
        best_h = np.copy(h_imp)
        last_improvement_time = time.time()

    # Continuous refine with bounds and constraints
    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    def optimize_with_bounds(optimizer="SLSQP", max_iter=200):
        nonlocal best_h, best_c5, last_improvement_time
        cons = (
            {"type": "eq", "fun": constraint_sum},
            {"type": "ineq", "fun": constraint_upper},
            {"type": "ineq", "fun": constraint_lower}
        )
        result = minimize(
            objective,
            best_h,
            method=optimizer,
            bounds=[(0.0, 1.0)] * n_points,
            constraints=cons,
            options={"maxiter": max_iter}
        )
        if result.success:
            new_c5 = result.fun
            new_h = result.x
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h.copy()
                last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 5:
        optimize_with_bounds(optimizer="SLSQP", max_iter=200)
        if time.time() - last_improvement_time > 300:
            break

    return best_h, best_c5, n_points
