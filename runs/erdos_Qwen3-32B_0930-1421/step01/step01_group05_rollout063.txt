import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial_h_values as provided
    h = initial_h_values if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2)
    else:
        n_points = len(h)

    h = np.array(h, dtype=np.float64)
    dx = 2.0 / n_points

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        scaled = clipped * (n_points / 2) / np.sum(clipped)
        return np.clip(scaled, 0, 1)

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    learning_rate = 0.1
    patience = 0
    max_patience = 100

    while time.time() - start_time < budget_s:
        # Identify most problematic shift k
        h_ext = np.concatenate((np.zeros(n_points - 1), h, np.zeros(n_points - 1)))
        h_1h = 1 - h
        h_1h_ext = np.concatenate((np.zeros(n_points - 1), h_1h, np.zeros(n_points - 1)))

        overlap = np.zeros(n_points)
        for k in range(n_points):
            o = np.sum(h * h_1h_ext[k:k + n_points])
            overlap[k] = o
        k_max = np.argmax(overlap)

        # Compute gradient only for k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] -= h_1h[k_max + i]
            if i - k_max >= 0:
                grad[i] += h[i - k_max]
        grad *= dx

        # Update and project back to feasible set
        h -= learning_rate * grad
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Schedule learning rate decay
        learning_rate *= 0.999

        # Early stopping
        if patience > max_patience:
            break

    # Final check
    h = best_h
    h /= h.sum() * dx  # Ensure mass constraint
    h = h * np.clip(h, 0, 1)

    return best_h.tolist(), best_c5, n_points
