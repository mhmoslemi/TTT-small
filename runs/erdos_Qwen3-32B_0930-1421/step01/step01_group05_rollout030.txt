import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        c = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(c * dx)

    def normalize_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals *= (n_points / 2) / h_vals.sum()
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals

    h = normalize_h(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    learn_rate = 0.05
    t = 0
    decay = 0.9995
    perturb_rate = 0.02
    restart_period = 100
    restarts = 0

    while time.time() - start_time < budget_s:
        t += 1

        # Compute C5 and find the worst k
        overlaps = correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(overlaps)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            l = i
            r = (i + worst_k) % n_points
            grad[i] = (1 - h[r]) - h[i]
        grad *= dx

        # Perform gradient descent
        h = np.clip(h - learn_rate * grad, 0, 1)
        h = normalize_h(h)

        # Perturb randomly to escape local minima
        if np.random.rand() < perturb_rate:
            noise = np.random.normal(0, 0.02, size=n_points)
            h += noise
            h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            learn_rate *= 1.02  # adaptive increase after improvement
            restarts = 0
        else:
            learn_rate = max(0.001, learn_rate * 0.995)
            restarts += 1

        if restarts > restart_period:
            h = best_h.copy()
            h += np.random.normal(0, 0.01, len(h))
            h = normalize_h(h)
            restarts = 0

        # Track progress
        learn_rate *= decay

    # Final verification
    if 'initial_h_values' in globals() and np.allclose(sum(best_h), n_points / 2):
        assert np.all(best_h >= 0) and np.all(best_h <= 1)
    return best_h.tolist(), best_c5, n_points
