import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()  # normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def evaluate(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * constraint_sum / h.sum()  # normalize
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps), h.tolist()

    def obj_and_constraints(h_var, C_var):
        h = h_var.reshape(-1)
        overlaps = np.array([sum(h[i] * (1 - h[(i - k) % n_points]) for i in range(n_points)) * dx for k in range(n_points)])
        return overlaps - C_var

    best_h, best_c5 = None, float('inf')

    while time.time() - start_time < budget_s:
        for trial in range(3):
            if time.time() - start_time > budget_s:
                break

            # LP formulation with linearization
            h = cp.Variable(n_points)
            C = cp.Variable()

            constraints = [
                cp.sum(h) == constraint_sum,
                0 <= h,
                h <= 1
            ]

            overlaps = [sum(h[i] * (1 - h[(i - k) % n_points]) for i in range(n_points)) * dx for k in range(n_points)]
            for overlap in overlaps:
                constraints.append(overlap <= C)

            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.GLPK, maxiters=50)

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_vals = np.clip(h.value, 0, 1)
                h_vals = h_vals * constraint_sum / h_vals.sum()
                c5, _ = evaluate(h_vals)
                if c5 < best_c5:
                    best_h = h_vals
                    best_c5 = c5

            # perturb and try again
            h_init = best_h + np.random.randn(n_points) * 0.05
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * constraint_sum / h_init.sum()

    if best_h is None:
        best_h = np.array(h_init)
    best_c5, _ = evaluate(best_h)

    return best_h, best_c5, n_points
