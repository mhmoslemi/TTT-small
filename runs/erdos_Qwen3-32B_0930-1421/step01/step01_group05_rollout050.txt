import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import pulp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points, dtype=np.float64) / n_points * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ILP-based optimization
    try:
        model = pulp.LpProblem("ErdosOverlap", pulp.LpMinimize)
        h_vars = [pulp.LpVariable(f"h_{i}", lowBound=0.0, upBound=1.0) for i in range(n_points)]

        # Constraint: unit integral
        model += pulp.lpSum(h_vars) == n_points / 2, "UnitIntegral"

        overlap_constraints = []

        for k in range(-n_points + 1, n_points):
            overlap_terms = [(h_vars[i] * (1 - h_vars[(i + k) % n_points])) for i in range(n_points)]
            overlap = pulp.lpSum(overlap_terms)
            overlap_var = pulp.LpVariable(f"c_{k}", lowBound=0)
            overlap_constraints.append(overlap_var >= overlap)
            model += overlap_var

        max_c5 = pulp.LpVariable("max_c5", lowBound=0)
        for k in range(len(overlap_constraints)):
            model += max_c5 >= overlap_constraints[k]

        model.solve(pulp.SCIP_CMD(msg=0, timeLimit=60))
        h_ilp = [p.values()[f"h_{i}"].value() for i in range(n_points)]
        h_ilp = normalize_h(np.array(h_ilp))
        c5_ilp = compute_c5(h_ilp)

        if c5_ilp < best_c5:
            best_c5 = c5_ilp
            best_h = h_ilp
            print(f"Improved via ILP to C5={best_c5:.6f}")
    except:
        pass

    # Remaining Budget - Gradient Descent
    time_left = budget_s - (time.time() - start_time)
    learning_rate = 0.1

    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        eps = 1e-5

        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        h -= learning_rate * grad
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Reduce learning rate over time
        learning_rate *= 0.998

        # Check budget
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
