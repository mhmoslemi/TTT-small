import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def create_basis_wavelets(n_points, N_basis=10, alpha=4.0, beta_rel=0.15):
    dx = 2.0 / n_points
    x_array = (1 + np.arange(n_points)) * dx - dx
    basis_list = []
    for _ in range(N_basis):
        centers = np.random.uniform(0.3, 1.7, N_basis)
    basis_list = []
    if initial_h_values is not None:
        com = np.sum(initial_h_values * x_array) / (2.0 / n_points)
        base_centers = np.linspace(max(0, com - 0.5), min(2, com + 0.5), N_basis)
    else:
        base_centers = np.linspace(0.2, 1.8, N_basis)
    beta = beta_rel * 2  # width in [0,2]
    for xc in base_centers:
        x_centered = np.abs(x_array - xc)
        psi = np.exp(-alpha * x_centered**2)
        mask = (x_centered > beta) | (x_centered < -beta)
        psi[mask] = 0
        basis_list.append(psi)
    return np.array(basis_list).T


def solve_basis_optimization(basis_array):
    A = basis_array
    n_basis = A.shape[1]

    a = cp.Variable(n_basis)
    h = A @ a

    integral_h = (2.0 / A.shape[0]) * cp.sum(A, axis=0) @ a
    constraint_integral = integral_h == 1

    constraints = [a >= 0]
    constraints.append(integral_h == 1)

    # Simplified approximation: minimize the largest pairwise correlation
    overlaps = h * (1 - h)
    norms = np.sum(basis_array**2, axis=0) * (2.0 / A.shape[0])
    norms = norms + 1e-10  # avoid division by zero
    relative_overlap = A.T @ (A * overlaps) / norms

    corr_flat = cp.reshape(relative_overlap, (1, A.shape[1] ** 2)) @ cp.vec(a)
    corr_flat = cp.diag(A.T @ (A * overlaps)) @ a
    corr_trace = cp.reshape(corr_flat, (-1, 1)).T @ a

    obj = cp.Minimize(corr_trace)
    constraints.extend([a <= 1])

    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver=cp.CVXOPT, verbose=False)

    if prob.status != cp.OPTIMAL:
        return None
    return a.value


def normalize_h(h, n_points):
    total = np.sum(h) * (2.0 / n_points)
    if abs(total - 1) > 1e-5:  # Feasibility repair
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total < 1e-6:
            h = np.zeros(n_points)
            h[0] = n_points / 2
        else:
            h = h / total * (n_points / 2)
    return h


def run(budget_s=1100, N_basis=10, alpha=4.0, n_points=42):
    start_time = time.time()

    best_h = None
    best_C5 = float('inf')

    for trial in range(50):
        if time.time() - start_time > budget_s:
            break

        basis_array = create_basis_wavelets(n_points, N_basis, alpha)
        a_vals = solve_basis_optimization(basis_array)
        if a_vals is None:
            continue

        h = normalize_h(basis_array @ a_vals, n_points)

        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

    if best_h is None and initial_h_values is not None:
        best_h = initial_h_values.copy()
        best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
