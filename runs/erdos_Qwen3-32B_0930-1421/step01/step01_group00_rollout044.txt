import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def constrain_h(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        total = optimized_h.sum()
        if abs(total - target_sum) > 1e-8:
            optimized_h = np.clip(optimized_h * (target_sum / total), 0, 1)
        return optimized_h

    def compute_c5(optimized_h):
        optimized_h = constrain_h(optimized_h)
        max_overlap = -np.inf
        optimized_h_ext = np.concatenate([optimized_h, 1 - optimized_h])
        ones_minus_h = 1 - optimized_h
        cors = np.correlate(optimized_h, ones_minus_h, mode='full')
        cors *= dx
        max_overlap = cors.max()
        return max_overlap

    def find_peak_shift(optimized_h):
        optimized_h = constrain_h(optimized_h)
        optimized_h_ext = np.concatenate([optimized_h, 1 - optimized_h])
        ones_minus_h = 1 - optimized_h
        cors = np.correlate(optimized_h, ones_minus_h, mode='same') 
        peak = np.argmax(cors)
        return peak

    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_time = time.time()

    alpha = 0.01
    perturb_interval = 60
    perturb_timer = 0

    while time.time() - start_time < budget_s:
        if time.time() - best_time > 60 and best_c5 < 0.380876:
            break
        if verbose and (time.time() - start_time) % 10 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

        k = find_peak_shift(h)
        grad = np.zeros_like(h)

        for i in range(n_points - k):
            grad[i] += (1 - h[i + k]) * dx
            grad[i + k] -= h[i] * dx

        h_flat = h.copy()
        centered_h = h_flat - h_flat.mean()
        projected_grad = grad - np.dot(grad, centered_h) / np.dot(centered_h, centered_h) * centered_h

        h -= alpha * projected_grad
        h = constrain_h(h)

        if np.random.rand() < (perturb_interval / max(100, time.time() - start_time)) and time.time() - perturb_timer > 30:
            indices = np.random.choice(n_points, size=min(5, n_points), replace=False)
            h[indices] += np.random.uniform(-0.02, 0.02, size=len(indices))
            h = constrain_h(h)
            perturb_timer = time.time()

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            best_time = time.time()

        if alpha > 1e-4 and best_c5 < 0.380876:
            alpha *= 0.95

    return (best_h, best_c5, n_points)
