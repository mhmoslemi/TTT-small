```python
import numpy as np
import time
import cvxpy as cp

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def run(budget_s=1100, n_points=48, use_parent=True):
    start_time = time.time()

    # Set up the grid and initial point
    h = cp.Variable(n_points, nonneg=True)

    # Objective and constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2.0,
    ]

    # Compute all overlaps
    ones_minus_h = 1 - h
    overlaps = []
    for k in range(-n_points + 1, n_points):
        idxs1 = np.arange(max(0, -k), min(n_points, n_points - k))
        idxs2 = idxs1 + k
        valid = (idxs1 < n_points) & (idxs2 < n_points)
        if not valid.any():
            continue
        idxs1, idxs2 = idxs1[valid], idxs2[valid]
        overlap = cp.sum(cp.multiply(h[idxs1], ones_minus_h[idxs2]))
        overlap_scaled = overlap * (2.0 / n_points)
        overlaps.append(overlap_scaled)

    λ = cp.Variable()
    constraints += [o <= λ for o in overlaps]
    objective = cp.Minimize(λ)
    prob = cp.Problem(objective, constraints)

    if use_parent and initial_h_values is not None:
        h.value = initial_h_values
        h.value = h.value / h.value.sum() * (n_points / 2.0)

    best_sol = None
    best_C5 = np.inf

    try:
        while time.time() - start_time < budget_s:
            prob.solve(solver=cp.CVXOPT, verbose=False)

            if h.value is None:
                h.value = np.random.rand(n_points)
                h.value = h.value / h.value.sum() * (n_points / 2.0)

            h_sol = np.clip(h.value, 0, 1)
            h_sol = h_sol / h_sol.sum() * (n_points / 2.0)

            current_C5 = evaluate_C5(h_sol)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_sol = h_sol

            h.value = h_sol + 0.01 * np.random.randn(n_points)
            h.value = np.clip(h.value, 0, 1)
            h.value = h.value / h.value.sum() * (n_points / 2.0)

    except Exception:
        pass  # Handle interrupted computation

    return best_sol, best_C5, n_points
```
