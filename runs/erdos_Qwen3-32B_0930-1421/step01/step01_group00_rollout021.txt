import numpy as np
import time
import copy
import scipy.signal

def compute_c5(h, dx):
    n = len(h)
    ones_minus_h = 1.0 - h
    # Forward shifts
    corr1 = np.correlate(h, ones_minus_h, mode="full")
    # Backward shifts
    corr2 = np.correlate(ones_minus_h, h, mode="full")
    corr = np.concatenate([
        corr1[n-1:-n],
        corr2[n-1:-n]
    ])
    return np.max(corr * dx)

def get_top_shifts(h, dx, top_k=5):
    n = len(h)
    ones_minus_h = 1.0 - h
    forward_corr = np.correlate(h, ones_minus_h, mode="full")
    backward_corr = np.correlate(ones_minus_h, h, mode="full")
    combined_corr = np.concatenate([
        forward_corr[n-1:-n],
        backward_corr[n-1:-n]
    ])
    all_shifts = np.arange(1, n)
    top_indices = np.argsort(combined_corr)[-top_k:]
    return all_shifts[top_indices], combined_corr[top_indices]

def adjust_h(h, top_shifts, dx, percent_top=0.1, step_size=0.02):
    h_adj = h.copy()
    n = len(h)
    delta = step_size * dx
    for k in top_shifts:
        if k >= n:
            continue
        contrib = h[:n-k] * (1.0 - h[k:])
        top_idx = np.argsort(contrib)[-int(percent_top * len(contrib)):]
        h_adj[top_idx + k] += delta
        h_adj[top_idx] -= delta
        h_adj = np.clip(h_adj, 0, 1)
    return h_adj

def fix_sum(h, target_sum):
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-6:
        h = h * (target_sum / h_sum)
    return np.clip(h, 0, 1)

        top_shifts, _ = get_top_shifts(h, dx, top_k=5)
        adjusted_h = adjust_h(h, top_shifts, dx, percent_top=0.1)
        adjusted_h = fix_sum(adjusted_h, target_sum)
