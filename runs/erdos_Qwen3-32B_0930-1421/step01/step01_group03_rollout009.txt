import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    import numpy as np

    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 47
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)
    if h.size == 0 or np.all(h == 0):
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i * 2] = 2.0 / n_points
        h *= n_points / 2 / h.sum()

    dx = 2.0 / n_points
    sum_const = n_points / 2

    def evaluate(h):
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (sum_const / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps), h

    best_h, best_c5 = h, evaluate(h)[0]
    h_list = [h.copy()]

    max_iterations = int(budget_s * 0.8 // 0.1)  # rough estimate
    for it in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
         # Build OP over h, C minimizing C with bilinear linearized constraints
        h_var = cp.Variable(n_points)
        C = cp.Variable()

        constraints = [
            cp.sum(h_var) == sum_const,
            0 <= h_var,
            h_var <= 1
        ]
        for k in range(n_points):  # shift amounts
            overlap_k = 0
            for i in range(n_points):
                j = (i - k) % n_points
                if j < len(best_h):
                    # Linearize overlap term around current best_h
                    overlap_k += h_var[i] * (1 - best_h[j]) + best_h[i] * (1 - h_var[j])
            overlap_k *= dx
            constraints.append(overlap_k <= C)

        objective = cp.Minimize(C)

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GLPK, verbose=False)

        if prob.status != "optimal":
            continue

        h_new = h_var.value
        if h_new is None:
            continue

        h_new = h_new.clip(0, 1)
        h_new = h_new * (sum_const / h_new.sum())
        h_list.append(h_new)
        c5_new, h_new = evaluate(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        # Perturbation if needed
        if it % 5 == 0 and len(h_list) > 1:
            prev_h = h_list[-2]
            diff = best_h - prev_h
            best_h += (0.1 * np.sign(diff) * np.random.rand(*best_h.shape))
            best_h = best_h.clip(0, 1)
            best_h = best_h * (sum_const / best_h.sum())

    best_h = evaluate(best_h)[1]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    return best_h, best_c5, n_points
