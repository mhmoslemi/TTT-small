import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def construct_initial_basis(n_basis=10, alpha=4.0, beta=0.3):
    n_points = len(initial_h_values) if initial_h_values is not None else 42
    x = np.linspace(0, 2, n_points)
    dt = 2.0 / n_points
    basis_list = []
    nonlinear_bases = []

    if initial_h_values is not None:
        # Find local maximums as centers
        h = initial_h_values.copy()
        centers = []
        for i in range(1, n_points - 1):
            if h[i] > h[i - 1] and h[i] > h[i + 1]:
                centers.append(i)
        while len(centers) < n_basis:
            centers.append(int(n_points * np.random.rand()))
    else:
        # Create uniformly spaced centers
        centers = np.linspace(0, 2, n_basis)

    for x_center in centers:
        i_center = int(x_center / dt)
        # Ensure not at edges
        i_center = max(2, min(n_points - 3, i_center))

        # Define basis function support
        start = int(np.floor((x_center - beta) / dt))
        end = int(np.ceil((x_center + beta) / dt))
        span = np.arange(start, end + 1)

        # Construct a time-limited Gaussian basis
        pos = x[start:end + 1] - x_center
        shape = np.exp(-alpha * pos ** 2)
        shape = shape / np.sum(shape)  # Normalize
        basis = np.zeros(n_points)
        basis[span] = shape
        basis_list.append(basis)

    # Build matrix for inner integrals and normalization
    A = np.vstack(basis_list)
    ones = np.ones_like(h)

    return A, ones


def run(budget_s=1100, n_basis=10, alpha=4.0, beta=0.3, pop_size=15):
    start_time = time.time()

    # Use initial construction or set default
    if initial_h_values is not None:
        h_init = np.copy(initial_h_values)
        n_points = len(h_init)
    else:
        h_init = None
        n_points = 42

    # Construct basis functions
    A, _ = construct_initial_basis(n_basis=n_basis, alpha=alpha, beta=beta)
    A_t = A.T

    # Optimize over coefficients for minimal worst-case overlap
    coefficients = cp.Variable(n_basis, nonneg=True)
    obj = cp.Minimize(0)  # Start with dummy
    constraints = []

    # Constraint: Ensure ∫h dx = 1
    inner_integrals = A @ coefficients - 1
    constraints.append(inner_integrals == 0)

    # Constraint: Ensure 0 <= h(x) <= 1 for all x
    h = A_t @ coefficients
    constraints.append(h <= 1)
    constraints.append(h >= 0)

    # Try up to 10 perturbed basis configurations
    best_h = None
    best_C5 = np.inf
    tried = 0

    while tried < 10 and time.time() - start_time < budget_s:
        # Solve optimization problem
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status not in ['infeasible', 'unbounded']:
                current_coeff = coefficients.value
                h_current = A_t @ current_coeff
                h_current = ensure_feasible(h_current)

                # Evaluate C5 for this h
                C5_current = evaluate_C5(h_current)
                
                if C5_current < best_C5:
                    best_h = h_current
                    best_C5 = C5_current
            tried += 1
        except:
            tried += 1

        # Perturb basis
        centers = [i * (2.0 / n_points) for i in range(n_points)]
        new_centers = np.array(centers) + (0.1 - np.random.rand(n_basis) * 0.2)
        new_centers = np.clip(new_centers, 0.0, 2.0)
        A, _ = construct_initial_basis(n_basis=n_basis, alpha=alpha, beta=beta)

    return best_h, best_C5, n_points
