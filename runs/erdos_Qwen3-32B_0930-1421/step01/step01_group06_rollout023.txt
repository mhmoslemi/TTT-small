import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

        T = 1.0

        while time.time() - start_time < budget_sanneal:
            t_step += 1
            i = np.random.randint(n_points)
            new_h = np.copy(h_bin)

            # Flip bit, maintain sum
            if new_h[i]:

                if new_c5 < best_c5:

    def optimize_slsqp(x0):
        def objective(x):
            return evaluate_erdos_solution(x, n_points)

        def constraint_eq(x):
            return np.sum(x) - n_points / 2.0

        def constraint_ub(x):
            return 1.0 - x

        def constraint_lb(x):
            return x

        cons = (

            {"type": "ineq", "fun": constraint_ub},
