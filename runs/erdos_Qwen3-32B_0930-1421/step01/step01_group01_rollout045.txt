import numpy as np
import time
import scipy.signal as signal

def evaluate_C5(h, dx):
    """Evaluate the current C5 bound using cross-correlation."""
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2  # normalize to the integral constraint
    cor = signal.correlate(h, 1 - h, mode='full')
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Project h into the feasible set [0,1] with total sum n/2."""
    h = np.clip(h, 0, 1).copy()
    total = h.sum()
    if total > 1e-10:
        h *= (len(h) / 2) / total
    else:  # handle edge case
        h[0] = len(h) / 2
    return h
