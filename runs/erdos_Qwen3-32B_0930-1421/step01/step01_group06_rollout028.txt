import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.copy(initial_h_values)
    n_points = len(h_init)

    def project(h, n_points):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            h[:] = 0.5
        else:
            scaling = (n_points / 2.0) / total
            h *= scaling
        return h

    h = project(h_init, n_points)
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    pbest_func = np.empty(num_particles)
    pbest_vec = np.zeros((num_particles, n_points))
    gbest = best_h

    # Initialize particles
    num_fallback = 5
    for i in range(num_fallback):
        perturb = np.random.uniform(-1e-2, 1e-2, n_points)
        particles[i] = project(h_init + perturb, n_points)

    for i in range(num_fallback, num_particles):
        perturb = np.random.uniform(-0.5, 0.5, n_points)
        particles[i] = project(h_init + perturb, n_points)

    velocities = np.random.uniform(-max_velocity, max_velocity, (num_particles, n_points))

    for i in range(num_particles):
        particles[i] = project(particles[i], n_points)
        pbest_vec[i] = np.copy(particles[i])
        pbest_func[i] = evaluate_erdos_solution(particles[i], n_points)
        if pbest_func[i] < evaluate_erdos_solution(gbest, n_points):
            gbest = np.copy(particles[i])

    best_time = start_time

    while time.time() - start_time < budget_s - 10:
        # Compute inertia weight
        elapsed = time.time() - start_time
        frac = min(elapsed / budget_s, 1)
        w = w_start - (w_start - w_end) * frac

        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive = c1 * r1 * (pbest_vec[i] - particles[i])
            social = c2 * r2 * (gbest - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] = particles[i] + velocities[i]
            particles[i] = project(particles[i], n_points)

            c5 = evaluate_erdos_solution(particles[i], n_points)
            if c5 < pbest_func[i]:
                pbest_func[i] = c5
                pbest_vec[i] = np.copy(particles[i])

                if c5 < evaluate_erdos_solution(gbest, n_points):
                    gbest = np.copy(particles[i])

                    current_c5 = evaluate_erdos_solution(gbest, n_points)
                    if current_c5 < best_c5:
                        best_h = np.copy(gbest)
                        best_c5 = current_c5
                        best_time = time.time()

        # Fractional reset
        for i in np.random.choice(num_particles, size=int(reset_fraction * num_particles), replace=False):
            perturb = np.random.uniform(-0.5, 0.5, n_points)
            particles[i] = project(h_init + perturb, n_points)
            pbest_func[i] = evaluate_erdos_solution(particles[i], n_points)
            pbest_vec[i] = np.copy(particles[i])

        # Stopping condition
        if time.time() - best_time > 60:
            break

    return best_h, best_c5, n_points
