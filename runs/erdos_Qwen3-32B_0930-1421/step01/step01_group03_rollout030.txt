import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Configuration
    max_n_points = 200  # Allow slight scaling if needed for optimization
    population_size = 15
    n_elite = 3

    def constraint_normalize(h, n_points):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2 / h.sum())
        return h

    # Setup based on initial construction
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values)
    else:
        n_points = 47
        h_base = np.zeros(n_points)
        for i in range(n_points // 2):
            h_base[i << 1] = 2.0 / n_points
        h_base = constraint_normalize(h_base, n_points)

    def objective(h):
        h = h.reshape(n_points)
        h = constraint_normalize(h, n_points)
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_gradient(h, method="L-BFGS-B", maxiter=300):
        h = h.reshape(n_points)
        h = constraint_normalize(h, n_points)

        bounds = [(0, 1)] * n_points
        def obj_func(x):
            return objective(x)

        optimized = minimize(obj_func, x0=h, method=method, bounds=bounds, options={'maxiter': maxiter})
        return optimized.x

    def mutate_base(h, noise_factor=0.05, perturb_positions=False):
        h = h.copy()
        noise = np.random.randn(n_points) * noise_factor
        h = h + noise
        if perturb_positions:
            k = np.random.randint(1, n_points // 5)
            h = np.roll(h, k)
        return constraint_normalize(h, n_points)

    def crossover(h1, h2):
        alpha = np.random.uniform(0.4, 0.6)
        child = alpha * h1 + (1 - alpha) * h2
        return constraint_normalize(child, n_points)

    # Generate diverse initial population
    population = []
    population.append(h_base.copy())
    if 'initial_h_values' in globals():
        population.extend(mutate_base(h_base, noise_factor=0.1) for _ in range(population_size - 1))
    else:
        for _ in range(population_size - 1):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, n_points // 2, replace=False)
            h[indices] = 1
            h = constraint_normalize(h, n_points)
            population.append(h)

    best_h = h_base.copy()
    best_c5 = objective(best_h)

    # Evolutionary loop
    generation = 0
    while time.time() - start_time < budget_s * 0.9:
        # Evaluate fitness
        fitness = [(objective(h), h) for h in population]
        fitness.sort()

        # Keep best individuals
        population = [h for score, h in fitness[:population_size - n_elite]]

        # Crossover and mutation for new candidates
        for _ in range(n_elite):
            if len(population) < 2:
                break
            p1_idx = random.randint(0, len(population)-1)
            p2_idx = random.randint(0, len(population)-1)
            p1 = population[p1_idx]
            p2 = population[p2_idx]
            child = crossover(p1, p2)
            child = optimize_gradient(child)
            population.append(child)

        for _ in range(n_elite):
            if not population:
                continue
            idx = random.randint(0, len(population)-1)
            child = mutate_base(population[idx])
            child = optimize_gradient(child)
            population.append(child)

        # Update best
        for h in fitness:
            current_c5, h_candidate = h
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate

        # Trim surplus
        while len(population) > population_size:
            population.pop()

        generation += 1

    # Final optimization and check
    best_h = optimize_gradient(best_h, method='L-BFGS-B', maxiter=500)
    best_h = constraint_normalize(best_h, n_points)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
