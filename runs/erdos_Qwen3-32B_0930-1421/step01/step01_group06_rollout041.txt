import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to unit mass
    h = np.clip(initial_h / initial_h.sum() * (n_points / 2), 0.0, 1.0)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement_time = start_time

    dx = 2.0 / n_points

    def solve_for_c5(C5):
        h_var = cp.Variable(n_points)
        z_vars = {}
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1,
        ]

        for i in range(n_points - 1):
            for j in [i + k for k in range(1, n_points - i)]:
                if j >= n_points:
                    break
                z = cp.Variable()
                z_vars[(i, j)] = z
                idx = j - i
                constraints += [
                    z <= h_var[i],
                    z <= 1 - h_var[j],
                    z >= h_var[i] + (1 - h_var[j]) - 1,
                    z >= 0,
                ]

        for i in range(n_points - 1):
            for k in range(1, n_points - i):
                j = i + k
                z = z_vars.get((i, j), cp.Constant(0))
                constraints.append(z <= C5 * dx)

        obj = cp.Minimize(C5)
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.GUROBI, verbose=False, TimeLimit=10)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value, C5
        return None, C5

    # Binary search over C5
    C_low = 0.0
    C_high = best_c5
    EPS = 1e-6
    while time.time() - start_time < budget_s - 5:
        if C_high - C_low < EPS:
            break
        C_mid = (C_low + C_high) / 2
        sol, c5_val = solve_for_c5(C_mid)
        if sol is not None:
            new_h = np.clip(sol, 0, 1)
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
                last_improvement_time = time.time()
            C_high = C_mid
        else:
            C_low = C_mid

        if time.time() - last_improvement_time > 30:
            break

    # Perturbation
    for _ in range(10):
        if time.time() - start_time >= budget_s - 5:
            break
        # Random perturbation
        h_perturbed = np.clip(best_h + np.random.uniform(-0.05, 0.05, n_points), 0, 1)
        h_perturbed = h_perturbed / h_perturbed.sum() * (n_points / 2)
        h_perturbed = np.clip(h_perturbed, 0, 1)

        C_low_p = 0.0
        C_high_p = best_c5
        while C_high_p - C_low_p > 1e-6:
            if time.time() - start_time >= budget_s - 5:
                break
            C_mid = (C_low_p + C_high_p) / 2
            sol, c5_val = solve_for_c5(C_mid)
            if sol is not None:
                new_h = np.clip(sol, 0, 1)
                new_c5 = evaluate_erdos_solution(new_h, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
                    last_improvement_time = time.time()
                C_high_p = C_mid
            else:
                C_low_p = C_mid

    return best_h, best_c5, n_points
