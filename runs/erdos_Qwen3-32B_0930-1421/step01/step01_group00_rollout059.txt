import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    verbose = False
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    h = h_init.copy()
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()

    if abs(h_sum - target_sum) > 1e-8:
        for _ in range(100):
            idx = np.random.choice(n_points)
            val = h[idx]
            if val < 0.5 and h_sum < target_sum:
                h[idx] += h[idx] * 0.01
            elif val > 0.5 and h_sum > target_sum:
                h[idx] -= h[idx] * 0.01
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if abs(h_sum - target_sum) < 1e-8:
                break

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()
    best_idx = 0

    def compute_c5(h):
        optimized_h = np.clip(h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            valid_size = n_points - k
            if valid_size <= 0:
                continue
            corr = np.dot(optimized_h[:valid_size], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    def fourier_transform(h):
        hf = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points, d=dx)
        return hf, freqs

    def inverse_fourier_transform(hf):
        return np.fft.ifft(hf).real

    prev_c5 = float('inf')
    while time.time() - start_time < budget_s:
        hf, freqs = fourier_transform(h)

        # Target top Fourier coefficients (top 10% but at least 5)
        top_n = max(5, n_points // 10)
        top_indices = np.argsort(np.abs(hf))[-top_n:]
        for idx in top_indices:
            delta = np.random.normal(0, 0.2, len(hf[idx]))
            hf[idx] += delta

        new_h = inverse_fourier_transform(hf)
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = new_h.sum()

        if abs(new_h_sum - target_sum) > 1e-7:
            new_h *= target_sum / new_h_sum

        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = new_h[:]
            best_time = time.time()

        if verbose and time.time() - start_time > 200 * best_idx:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")
            best_idx += 1

        if best_c5 < 0.380876:
            break

    if best_c5 >= 0.380876:
        best_h = h_init.copy()

    return (best_h, best_c5, n_points)
