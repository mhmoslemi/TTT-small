import numpy as np
import time
import cvxpy as cp
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1).astype(np.float64)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for unit integral
    cor = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def run(seed=42, budget_s=1100, n_points=84):

        h[5:42] = 0.3
        h[42:] = 0.2
        h = h / h.sum() * n_points / 2

    h = cp.Variable(len(h), nonneg=True)
    constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]

    lamb = cp.Parameter(nonneg=True, value=0.1)
    mu = cp.Parameter(nonneg=True, value=0.01)

    shifted = cp.Constant(0.0)
    for k in range(20):
        shifted = cp.roll(h, shift=k)
        shifted = cp.atoms.affine.max.huber_sparsity(1 - h - shifted) + shifted
        constraints += [shifted <= 1]

    objective = lamb * cp.sum_squares(h) + mu * cp.sum(cp.sum_squares((h + cp.roll(h, k) - 1) for k in range(4)))

    energy_prob = cp.Problem(cp.Minimize(objective), constraints)

    best_C5 = np.inf
    best_h = h.value.copy()

    start_time = time.time()
    total_time = 0
    iter_count = 0

    while total_time < budget_s - 5:
        iter_count += 1
        if iter_count > 1:
            perturb = 0.1 * (np.random.rand(n_points) - 0.5)
            h.value = h.value + perturb
            h.value = np.clip(h.value / h.value.sum() * n_points / 2, 0, 1)

        energy_prob.solve(
            solver=cp.CVXOPT,
            max_iters=500,
            verbose=False
        )

        if h.is_dcp() and h.value is not None:
            h_vals = np.clip(h.value, 0, 1)
            h_vals = h_vals / h_vals.sum() * n_points / 2
            c5, _ = evaluate_C5(h_vals, dx)

            if c5 < best_C5:
                best_C5 = c5
                best_h = h_vals
                mu.value *= 0.9
                if best_C5 < 0.381:
                    break  # early stopping for performance

        mu.value -= 0.005
        total_time = time.time() - start_time

    return best_h, best_C5, n_points
