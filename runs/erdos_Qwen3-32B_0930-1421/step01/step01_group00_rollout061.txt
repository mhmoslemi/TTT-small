        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_sum = optimized_h.sum()
        if abs(optimized_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_sum = optimized_h.sum()
        if abs(optimized_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_sum
        ones_minus_h = 1.0 - optimized_h
        max_corr = 0.0
        for k in range(1, n_points):
            range_len = n_points - k
            if range_len <= 1:
                continue
            corr = np.dot(optimized_h[:range_len], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
        for k in range(1, n_points):
            range_len = n_points - k
            if range_len <= 1:
                continue
            corr = np.dot(optimized_h[k:], ones_minus_h[:range_len])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    best_h = h.copy()
    best_time = time.time()
    num_steps = 0
    perturb_interval = 100
    alpha = 0.01  # initial learning rate

            best_time = time.time()

        # Break if no improvement for >2 min and below target
        if time.time() - best_time > 120 and best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
