import numpy as np
import time

from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.clip(0, 1)  # Enforce bounds
    h = h / h.sum() * len(h) / 2  # Ensure ∫h = 1
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def perturb_h(h, segment_min=5, segment_max=15, amp=0.1):
    """Perturb a random segment of h while keeping the sum constraint."""
    h = h.copy()
    n = len(h)
    start = np.random.randint(0, n - segment_max)
    end = start + np.random.randint(segment_min, segment_max)
    indices = slice(start, end)
    delta = np.random.rand(n) * 2 - 1  # random [-1, 1]
    delta[~indices] = 0  # only change a random segment
    delta = delta / (2 * np.max(np.abs(delta))) * amp  # scale
    h = h + delta
    h = h.clip(0, 1)
    h = h / h.sum() * len(h) / 2  # Normalize to ∫h = 1
    return h


def metropolis_accept(delta_obj, temp):
    """Metropolis acceptance rule."""
    if delta_obj < 0:
        return True
    return np.random.rand() < np.exp(-delta_obj / temp)
