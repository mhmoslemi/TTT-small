import numpy as np

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Ensure constraint is exactly satisfied
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-7:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(n_points, best_h)
    best_time = time.time()

    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1

        # Resample indices for perturbation
        n_perturb = min(8, n_points // 2)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        step_size = min(1e-2, 1.0 / (iteration ** 0.5))

        # Perturb h
        proposed_h = h.copy()
        proposed_h[indices] += np.random.uniform(-step_size, step_size, size=n_perturb)

        # Clip and enforce constraint
        proposed_h = np.clip(proposed_h, 0, 1)
        proposed_sum = proposed_h.sum()
        if abs(proposed_sum - target_sum) > 1e-7:
            proposed_h *= (target_sum / proposed_sum)

        # Evaluate C5
        c5 = evaluate_erdos_solution(n_points, proposed_h)

        # Accept better solution
        if c5 < best_c5:
            best_h[:] = proposed_h
            best_c5 = c5
            best_time = time.time()

        # Accept worse solution with small probability for diversity
        elif np.random.rand() < max(0.05, 1 - 0.95 * (time.time() - start_time) / budget_s):
            h[:] = proposed_h
            best_time = time.time()

        # Fall back to best known solution and try convex update if stuck
        if time.time() - best_time > 60:
            h[:] = best_h
            c5 = evaluate_erdos_solution(n_points, h)

            # Try convex optimization-inspired update
            diff = np.random.normal(size=n_points) * 0.1
            scaled_diff = (diff - diff.min()) / (diff.max() - diff.min())
            proposed_h = h + (scaled_diff - 0.5) * 0.2
            proposed_h = np.clip(proposed_h, 0, 1)
            proposed_sum = proposed_h.sum()
            if abs(proposed_sum - target_sum) > 1e-7:
                proposed_h *= (target_sum / proposed_sum)

            c5 = evaluate_erdos_solution(n_points, proposed_h)
            if c5 < best_c5:
                best_h[:] = proposed_h
                best_c5 = c5
                best_time = time.time()

    return (best_h, best_c5, n_points)
