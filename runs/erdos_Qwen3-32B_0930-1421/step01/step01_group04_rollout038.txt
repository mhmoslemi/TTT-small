```python
import numpy as np
import time
import random
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(-n_points + 1, n_points):
        if k < 0:
            corr = np.dot(h[-k:], 1 - h[:k + n_points])
        else:
            corr = np.dot(h[:-k], 1 - h[k:])
        corrs.append(corr * dx)
    return np.max(corrs)

def project_and_normalize(h_vec, n_points):
    h = np.clip(h_vec, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    return h * (n_points / 2) / total

def fitness(h_vec, n_points):
    h = project_and_normalize(h_vec, n_points)
    return compute_c5(h, n_points)

def mutate(h_vec, sigma, n_points):
    h_vec = h_vec + np.random.normal(0, sigma, n_points)
    return project_and_normalize(h_vec, n_points)

def crossover(h1, h2, alpha):
    h_child = alpha * h1 + (1 - alpha) * h2
    return project_and_normalize(h_child, len(h1))

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    random.seed(seed)
    n_points = len(init_h)
    
    # Initialize population
    if init_h is None:
        raise ValueError("Initial h not provided.")
    pop_size = 40
    pop = [project_and_normalize(np.random.rand(n_points), n_points) for _ in range(pop_size)]
    if init_h is not None:
        pop[0] = project_and_normalize(init_h, n_points)
    
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h, n_points)
    
    time_start = time.time()
    generation = 0
    sigma = 0.1
    alpha_cross = 0.5

    while time.time() - time_start < budget_s:
        new_pop = []
        
        # Selection
        scores = [compute_c5(h, n_points) for h in pop]
        sorted_indices = np.argsort(scores)
        top_half = [pop[i] for i in sorted_indices[:len(pop)//2]]
        
        # Crossover and mutation
        for i in range(len(top_half) * 2):
            if i % 2 == 0:
                p1 = random.choice(top_half).copy()
                p2 = random.choice(top_half).copy()
                child = crossover(p1, p2, alpha_cross)
            else:
                parent = random.choice(top_half).copy()
                child = mutate(parent.copy(), sigma, n_points)
            new_pop.append(child)
        
        # Evaluate and refine
        scores_new = [compute_c5(h, n_points) for h in new_pop]
        combined = np.array(top_half + new_pop)
        combined_scores = np.array(scores + scores_new).flatten()
        indices = np.argsort(combined_scores)[:pop_size]
        pop = [combined[i] for i in indices]
        
        # Track best
        current_best_index = np.argmin(combined_scores[top_half + new_pop])
        current_best = combined[current_best_index]
        current_best_c5 = combined_scores[current_best_index]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Anneal mutation and crossover
        sigma *= 0.999
        alpha_cross = 0.5 + 0.5 * np.random.uniform()
        generation += 1

    # Final refinement with L-BFGS-B
    res = opt.minimize(
        lambda x: compute_c5(project_and_normalize(x, n_points), n_points),
        x0=best_h,
        method="L-BFGS-B",
        bounds=[(0, 1)] * n_points,
        options={"maxiter": 100, "ftol": 1e-8, "gtol": 1e-8}
    )
    final_h = res.x
    final_c5 = compute_c5(project_and_normalize(final_h, n_points), n_points)

    return project_and_normalize(final_h, n_points), final_c5, n_points
```
