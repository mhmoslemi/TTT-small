import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)
    # Normalize initial_h to satisfy the constraint
    initial_h *= (n_points / 2.0) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0.0, 1.0)

    # PSO parameters
    num_particles = 50
    max_velocity = 0.2
    c1 = 2.0
    c2 = 2.0
    w_start = 0.9
    w_end = 0.4
    reset_fraction = 0.1
    T = int(budget_s / 10)  # Expected iterations
    velocities = 0.01 * np.random.randn(num_particles, n_points)
    particles = np.tile(initial_h, (num_particles, 1))
    # Randomize 90% of particles
    perturb_mask = np.concatenate([np.ones(n_points // 10, dtype=bool),
                                   np.random.rand(n_points * 9 // 10) < 0.5])
    for i in range(num_particles):
        if np.random.rand() < 0.9:
            particles[i, :] = np.clip(perturb_mask * particles[i, :],
                                      0.0, 1.0) + 0.1 * np.random.randn(n_points)
            particles[i, :] *= (n_points / 2.0) / np.sum(particles[i, :])
            particles[i, :] = np.clip(particles[i, :], 0.0, 1.0)

    # Initialize personal best and global best
    p_best = particles.copy()
    p_best_vals = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = float(p_best_vals[g_best_idx])

    def project(p):
        p = np.clip(p, 0.0, 1.0)
        p *= (n_points / 2.0) / np.sum(p)
        return p

    # Main PSO loop
    while time.time() - start_time < budget_s - 5:
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            w = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
            velocities[i] = (w * velocities[i]
                             + c1 * r1 * (p_best[i] - particles[i])
                             + c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = project(particles[i])
            val = evaluate_erdos_solution(particles[i], n_points)
            if val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = val
                if val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = val

        # Reset a fraction of the swarm
        reset_idx = np.random.choice(num_particles, int(reset_fraction * num_particles), replace=False)
        for idx in reset_idx:
            particles[idx] = np.random.rand(n_points)
            particles[idx] = project(particles[idx])
            p_best[idx] = particles[idx].copy()
            p_best_vals[idx] = evaluate_erdos_solution(particles[idx], n_points)
            if p_best_vals[idx] < g_best_val:
                g_best = p_best[idx].copy()
                g_best_val = p_best_vals[idx]

    return g_best, g_best_val, n_points
