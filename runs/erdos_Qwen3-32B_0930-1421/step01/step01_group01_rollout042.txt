import numpy as np
import time
import random
import math
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_h(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum < 1e-10:
        return float('inf')
    cor = correlate(h, 1 - h, mode='full')
    C5 = np.max(cor) * dx
    return C5


def perturb_h(h, dx, perturbation_scale=0.15, segment_min=5, segment_max=20):
    n = len(h)
    start = random.randint(0, n - 1)
    seg_len = random.randint(segment_min, min(segment_max, n - start))
    segment = h[start:start+seg_len]

    noise = np.random.normal(0, perturbation_scale, seg_len)
    perturbed_segment = segment + noise

    # Clip to [0,1]
    perturbed_segment = np.clip(perturbed_segment, 0, 1)

    # Adjust to keep the total sum constant
    delta = perturbed_segment.sum() - segment.sum()
    if abs(delta) > 1e-10:
        h_complete = h.copy()
        h_complete[start:start+seg_len] = perturbed_segment
        h_complete = h_complete - h_complete.min()  # shift to zero min
        h_complete = h_complete / h_complete.sum() * h.sum()  # normalize
        return np.clip(h_complete, 0, 1)
    return perturbed_segment
