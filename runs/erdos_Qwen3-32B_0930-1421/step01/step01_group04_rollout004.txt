import numpy as np
import time
import scipy.optimize as opt
import inspect

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_ext = np.concatenate((h, h))
    correlations = []
    for k in range(1, n_points):
        overlap = np.dot(h[:n_points - k], 1 - h[k: n_points])
        correlations.append(overlap)
    for k in range(1, n_points):
        overlap = np.dot(h[k:], 1 - h[:n_points - k])
        correlations.append(overlap)
    return max(correlations) * dx

def fix_boundaries(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

def initialize_population(n_points, size=40, init_h=None):
    if init_h is not None and len(init_h) == n_points:
        population = [fix_boundaries(init_h.copy(), n_points)]
    else:
        population = []
    count = len(population)
    while count < size:
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = fix_boundaries(h_rand, n_points)
        population.append(h_rand)
        count += 1
    return np.array(population)

def evaluate_population(population, n_points):
    return np.array([compute_c5(p, n_points) for p in population])

def select_parents(population, scores, num_parents):
    indices = np.argsort(scores)
    return population[indices[-num_parents:]], scores[indices[-num_parents:]]

def crossover(parent1, parent2, alpha):
    child = alpha * parent1 + (1 - alpha) * parent2
    return fix_boundaries(child, len(parent1))

def mutate(individual, sigma):
    noise = np.random.normal(0, sigma, np.shape(individual))
    mutated = fix_boundaries(individual + noise, len(individual))
    return mutated

def genetic_search(population, scores, n_points, budget_s):
    start = time.time()
    while time.time() - start < budget_s:
        scores = evaluate_population(population, n_points)
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]
        best_h = fix_boundaries(population[best_idx].copy(), n_points)

        pop_size = population.shape[0]
        parent_size = max(2, pop_size // 2)
        parents, parent_scores = select_parents(population, scores, parent_size)

        new_population = [best_h]
        for _ in range(pop_size - 1):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2, alpha=0.5)
            if np.random.rand() < 0.2:
                child = mutate(child, sigma=0.1 * (1 - (time.time() - start) / budget_s))
            new_population.append(child)
        population = np.array(new_population)

        # Periodically refine best solution with local optimization
        if (time.time() - start) / budget_s > 0.7 and len(parents) > 0:
            refiner = opt.minimize(lambda x: compute_c5(x, n_points), best_h, 
                                   method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                                   options={'maxiter': 100, 'eps': 1e-6})
            refined = fix_boundaries(refiner.x, n_points)
            best_c5 = compute_c5(refined, n_points)
            best_h = refined

    final_scores = evaluate_population(population, n_points)
    best_idx = np.argmin(final_scores)
    best_h = fix_boundaries(population[best_idx], n_points)
    return best_h, compute_c5(best_h, n_points)

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    n_points = 47
    if init_h is not None and len(init_h) == n_points:
        h0 = fix_boundaries(init_h.copy(), n_points)
    else:
        h0 = fix_boundaries(np.random.rand(n_points), n_points)

    population = initialize_population(n_points, size=40, init_h=h0)
    h_best, val_best = genetic_search(population, None, n_points, budget_s)
    return h_best, val_best, n_points
