import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_to_simplex(v, z=1.0, axis=None, eps=1e-8):
    v = np.asarray(v)
    if axis is None:
        v = v.ravel()
        w = np.sort(v)[::-1]
        n = w.shape[0]
        cplx_sum = np.cumsum(w, axis=axis)
        ind = np.arange(n) + 1
        sub = (cplx_sum - z) / ind
        i = np.where(w > sub)[0][-1]
        bound = sub[i]
        return np.clip(v - bound, 0, 1)
    else:
        raise NotImplementedError("Multi-axis projection not implemented")

def gradient_step(h, corr, lag, dx):
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        j = i + lag
        if j < n:
            grad[i] += (1 - h[j])
        j = i - lag
        if j >= 0:
            grad[i] -= h[j]
    return grad * dx

        accepted = False
