import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr * dx)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr * dx)
    return np.max(corrs)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if not np.isclose(total, n_points / 2, atol=1e-6):
        h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def constraint_index_set(n_points):
    indices = []
    for k in range(1, n_points):
        indices.append(k)
        if k > n_points - k:
            indices.append(n_points - k)
    return list(set(indices))

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        init_h = initial_h_values.copy()
    else:
        init_h = np.random.rand(47)
    init_h = enforce_normalization(init_h, len(init_h))
    h_best = init_h.copy()
    n_points = len(init_h)
    c5_best = compute_c5(h_best, n_points)

    time_start = time.time()
    indices = constraint_index_set(n_points)
    prob = setup_lp(n_points, indices)
    while time.time() - time_start < budget_s * 0.9:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
            if prob.status != cp.OPTIMAL:
                break
            h_val = prob.variables()[0].value.flatten()
            h_best = enforce_normalization(h_val, n_points)
            new_c5 = compute_c5(h_best, n_points)
            if new_c5 < c5_best:
                c5_best = new_c5
        except Exception as e:
            break

        # Identify most violated constraint
        h_val = h_best
        max_overlap = 0
        worst_k = 0
        for k in range(1, n_points):
            if n_points - k < 1:
                continue
            overlap = np.sum(h_val[:n_points-k] * (1 - h_val[k:])) * (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k
        if max_overlap < c5_best + 1e-5:
            break
        indices.append(worst_k)
        indices = list(set(indices))
        if len(indices) > 4 * n_points:
            break
        prob = setup_lp(n_points, indices)

    # Final refinement
    time_remaining = max(budget_s - (time.time() - time_start), 0.5)
    res = opt.minimize(
        lambda x: compute_c5(x, n_points),
        h_best,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 100, 'eps': 1e-4, 'ftol': 1e-9},
        timeout=time_remaining
    )
    h_best = enforce_normalization(res.x, n_points)
    c5_best = compute_c5(h_best, n_points)

    return h_best, c5_best, n_points

def setup_lp(n_points, indices):
    h = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0.0,
        h <= 1.0
    ]

    for k in indices:
        if k >= n_points:
            continue
        expr = 2.0 / n_points * cp.sum([h[i] * (1 - h[i + k]) for i in range(n_points - k)])
        constraints.append(expr <= C)

    return cp.Problem(cp.Minimize(C), constraints)
