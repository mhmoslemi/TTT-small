import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1).copy()
    h_sum = h.sum()
    if h_sum < 1e-10:
        h[:] = 0.0
        h[0] = len(h) / 2
    else:
        h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem with convex relaxation."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if defined
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.full(n_points, 1 / n_points * (n_points / 2))
    h = np.clip(h, 0, 1)
    h /= h.sum() * (2 / n_points) * n_points  # normalize to satisfy sum

    start_time = time.time()
    best_h = h.copy()
    best_c5, bh = evaluate_C5(best_h, dx)
    best_h = bh.copy()

    h = cp.Variable(n_points)
    objective_weights = [0.1, 0.01]

    while time.time() - start_time < budget_s:
        obj_sum = 0
        for k in range(n_points):
            shift = np.zeros(n_points)
            shifted_id = (k % n_points)
            for i in range(n_points - shifted_id):
                shift[i + shifted_id] = 1
            const = h * (1 - cp.multiply(h, shift))
            obj_sum += cp.sum(const)

        obj_sum /= n_points  # average over shifts
        lambda_k = objective_weights[0]
        mu_tikhonov = objective_weights[1]
        objective = cp.Minimize(lambda_k * obj_sum + mu_tikhonov * cp.sum_squares(h))
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_vals = np.clip(h.value, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * n_points / 2
            current_c5, h_vals = evaluate_C5(h_vals, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals

        # Perturb and restart
        h_assign = np.clip(best_h + 0.1 * (np.random.rand(n_points) - 0.5), 0, 1)
        h_assign = h_assign / h_assign.sum() * n_points / 2
        h.value = h_assign

        objective_weights[0] *= 0.9  # decay overlap weight
        objective_weights[1] *= 1.1  # increase regularization

        if (time.time() - start_time) + 40 > budget_s:
            break

    h_vals, c5, n_pts = best_h, best_c5, n_points
    return h_vals, c5, n_pts
