import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Cap for small sequences to stay fast
    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        h_init[::2] = 2.0 / n_points  # even spaced units
        h_init *= n_points / 2 / h_init.sum()  # normalize

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def repair_h(h):
        h = np.clip(h, 0, 1)
        h *= constraint_sum / h.sum()
        return h

    def calc_c5(h):
        h = h.reshape(n_points)
        h = repair_h(h)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def perturb_h(h, noise_scale):
        h = h.copy()
        h += np.random.randn(n_points) * noise_scale
        return repair_h(h)

    def optimize_h(local_h, max_iter=100):
        h = local_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            h = repair_h(h)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        # Use BFGS-type optimizer
        result = minimize(
            obj_func, h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'eps': 1e-4}
        )
        return repair_h(result.x)

    # Initial solution
    best_h = h_init.copy()
    best_c5 = calc_c5(best_h)

    # Constants for tabu search
    tabu_length = 30
    tabu_list = []

    # Cooling protocol
    perturbation_scale = 0.1
    scale_decay = 0.9
    stall_count = 0
    stall_thresh = 25
    max_iter = 3000

    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Perturb current best and refine
        new_h = perturb_h(best_h, perturbation_scale * (0.1 + i / max_iter))
        if np.array_equal(new_h, best_h) or np.any([np.allclose(new_h, t) for t in tabu_list]):
            continue

        new_h = optimize_h(new_h)
        new_c5 = calc_c5(new_h)

        tabu_list.append(new_h)
        if len(tabu_list) > tabu_length:
            tabu_list.pop(0)

        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            stall_count = 0
            perturbation_scale *= scale_decay
        else:
            stall_count += 1

        if stall_count > stall_thresh:
            perturbation_scale *= 2.0  # emergency rescaling
            stall_count = 0

    best_h = repair_h(best_h)
    best_c5 = calc_c5(best_h)

    return best_h, best_c5, n_points
