import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use provided initial step function if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def clip_and_normalize(h_in):
        clipped = np.clip(h_in, 0.0, 1.0)
        total = np.sum(clipped)
        scale = (n_points / 2) / total
        scaled = clipped * scale
        return np.clip(scaled, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lambda_step = 0.05
    lambda_max = 1.5
    lambdas = np.unique(np.logspace(np.log10(lambda_step), np.log10(lambda_max), num=25))

    # Convex optimization with L1 regularization and warm start
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(correlate(h_var, 1 - h_var, mode="full") * dx) + cp.lambda_ * cp.norm1(h_var))
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    problem = cp.Problem(objective, constraints)
    warm_start_h = h.copy()

    remaining_time = budget_s - (time.time() - start_time)

    # Grid of lambda candidates with warm start
    for l in lambdas:
        objective.expr = cp.max(correlate(h_var, 1 - h_var, mode="full") * dx) + l * cp.norm1(h_var)
        h_var.value = warm_start_h
        try:
            problem.solve(solver=cp.CSCS, max_iters=100, abstol=1e-4, reltol=1e-4, verbose=False)
            if problem.status in {"optimal", "optimal_inaccurate"}:
                h_opt = clip_and_normalize(h_var.value)
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                warm_start_h = h_opt.copy()
                if time.time() - start_time >= budget_s:
                    break
        except Exception:
            pass

    # Gradient descent refinement phase
    learning_rate = 0.05
    h = clip_and_normalize(best_h)
    h_best = h.copy()
    c5_best = compute_c5(h)

    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = clip_and_normalize(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = clip_and_normalize(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        h -= learning_rate * grad
        h = clip_and_normalize(h)
        c5 = compute_c5(h)

        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

        learning_rate *= 0.998

    if c5_best < best_c5:
        best_c5 = c5_best
        best_h = h_best

    return best_h.tolist(), best_c5, n_points
