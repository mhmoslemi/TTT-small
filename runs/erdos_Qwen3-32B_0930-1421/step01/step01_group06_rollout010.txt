import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time

    def is_feasible(C):
        h_var = cp.Variable(n_points, nonneg=True)
        z_var = cp.Variable((n_points, n_points))

        constraints = []
        constraints += [h_var[0] == h[0]]
        constraints += [h_var[-1] == h[-1]]
        constraints += [cp.sum(h_var) == n_points / 2.0]
        constraints += [h_var >= 0, h_var <= 1]
        constraints += [
            z_var[i, j] <= h_var[i],
            z_var[i, j] <= 1 - h_var[j],
            z_var[i, j] >= h_var[i] + (1 - h_var[j]) - 1,
            z_var[i, j] >= 0
            for i in range(n_points)
            for j in range(n_points)
            if j != i
        ]

        overlap_terms = np.zeros_like(z_var.value) if z_var.value is not None else np.zeros((n_points, n_points))
        for k in range(1, n_points - 1):
            s = n_points - k
            idxs = [(i, i + k) for i in range(s)]
            overlap_terms_k = cp.sum([z_var[i, j] for i, j in idxs])
            constraints += [overlap_terms_k <= C * dx]

        prob = cp.Problem(cp.Minimize(0), constraints)
        result = prob.solve(solver=cp.GUROBI, verbose=False,otimeLimit=int(budget_s - time.time()))
        return result == 0

    # Binary search over C₅
    C_low = 0.0
    C_high = evaluate_erdos_solution(h, n_points)
    best_C = best_c5

    while time.time() - start_time < budget_s - 5 and (C_high - C_low) > 1e-6:
        mid = (C_low + C_high) / 2
        if is_feasible(mid):
            C_high = mid
            best_C = mid
        else:
            C_low = mid

        if time.time() - last_improvement_time > 60:
            break

    # Perturb and refine
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        # Apply random perturbation
        h_perturbed = h + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        total_p = np.sum(h_perturbed)
        scale = (n_points / 2.0) / total_p
        h_perturbed = np.clip(h_perturbed * scale, 0.0, 1.0)

        C_low = 0.0
        C_high = evaluate_erdos_solution(h_perturbed, n_points)
        while time.time() - start_time < budget_s - 5 and (C_high - C_low) > 1e-6:
            mid = (C_low + C_high) / 2
            if is_feasible(mid):
                C_high = mid
                if mid < best_C:
                    best_C = mid
            else:
                C_low = mid

    return best_h, best_C, n_points
