import numpy as np
import time
import cvxpy as cp
import cvxpy.settings as cs

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure sum(h) = n_points / 2
    h = np.copy(initial_h)
    h *= (n_points / 2 / h.sum())

    # Initial estimate
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)

    dx = 2.0 / n_points

    def build_cvx_model(h_init, c5_val):
        h = cp.Variable(n_points)
        z = cp.Variable((n_points, n_points))

        constraints = [
            cp.sum(h) == n_points / 2.0,
            0 <= h,
            h <= 1,
            0 <= z,
            z <= h[:, None],
            z <= 1 - h[None, :],
            z >= h[:, None] + (1 - h[None, :]) - 1
        ]

        objective_expr = 0
        for k in range(1, n_points):
            for i in range(n_points - k):
                objective_expr += z[i, i + k]
            constraints.append(objective_expr <= c5_val * dx)

        problem = cp.Problem(cp.Minimize(0), constraints)
        return h, problem

    # Binary search for optimal C₅
    def binary_search(h_init, c5_low=0.0, c5_high=best_c5, precision=1e-6):
        best_model = h_init.copy()
        best_c5 = np.inf
        while time.time() - start_time < budget_s - 10 and c5_high - c5_low > precision:
            c5_mid = (c5_low + c5_high) / 2
            h_cvx, prob = build_cvx_model(h_init, c5_mid)
            prob.solve(solver=cs.GUROBI, verbose=False)
            if prob.status == cs.OPTIMAL:
                h_sol = np.clip(h_cvx.value, 0, 1)
                c5_sol = evaluate_erdos_solution(h_sol, n_points)
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_model = h_sol
                c5_high = c5_mid
            else:
                c5_low = c5_mid
        return best_model, best_c5

    # Main loop
    for _ in range(5):  # Try 5 random perturbations
        if time.time() - start_time >= budget_s:
            break
        perturbed_h = np.clip(best_h + np.random.uniform(-0.05, 0.05, n_points), 0, 1)
        perturbed_h *= (n_points / 2.0 / perturbed_h.sum())
        model, c5 = binary_search(perturbed_h)
        if c5 < best_c5:
            best_h = np.copy(model)
            best_c5 = c5

    return best_h, best_c5, n_points
