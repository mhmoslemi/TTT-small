import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, Bounds
from copy import copy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # Cap n_points to ensure small solution

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        h_init[::2] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')  # Avoid division by zero
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=500):
        h_arr = initial_h.copy()
        bounds = Bounds(0, 1, keep_feasible=True)
        def obj_func(h_flat):
            h = h_flat.reshape(-1)
            h_sum = h.sum()
            h = h * (constraint_sum / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            obj_func, h_arr, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        return result.x

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        perturbation = np.random.randn(n_points) * noise_factor
        h += perturbation
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)
    tabu_list = [tuple(h_init.copy().ravel())]

    iter_counter = 0
    no_impv_count = 0
    noise_factor = 0.2
    while time.time() - start_time < budget_s * 0.98:
        iter_counter += 1

        # Perturb
        current_h = perturb_h(best_h, noise_factor)
        current_tuple = tuple(current_h.ravel())

        # Tabu check
        if any(np.allclose(arr, current_h, atol=1e-6) for arr in tabu_list):
            continue

        # Improve current h via optimization
        optimized_h = optimize_h(current_h)
        optimized_c5 = objective(optimized_h)

        tabu_list.append(tuple(optimized_h.ravel()))
        if len(tabu_list) > 10:
            tabu_list.pop(0)

        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
            no_impv_count = 0
            noise_factor = max(0.05, noise_factor * 0.8)
        else:
            no_impv_count += 1

        if no_impv_count >= 50:
            noise_factor *= 1.2
            no_impv_count = 0

        if iter_counter % 20 == 0:
            print(f"Iteration {iter_counter}, Best C5={best_c5:.6f}")
            print(f"Time elapsed: {time.time() - start_time:.2f}/{budget_s} s")

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    if best_h_sum > 0:
        best_h = best_h * (constraint_sum / best_h_sum)

    final_c5 = objective(best_h)

    return best_h, final_c5, n_points
