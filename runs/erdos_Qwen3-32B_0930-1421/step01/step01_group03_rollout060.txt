import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_candidates = [np.array(initial_h_values, copy=True)]
    else:
        n_points = 47
        h_candidates = []

    # Create initial population from mutations and random constructions
    if len(h_candidates) == 0:
        n_points = 47
        for _ in range(10):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / h.sum()
            h_candidates.append(h)

    constraint_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    # Optimization utilities
    def optimize_h(h, max_iter=1000):
        bounds = Bounds(0, 1)
        h = h.copy()
        h /= h.sum() * 2  # Normalize to sum 1

        def f(h_flat):
            h_out = h_flat.reshape(n_points)
            h_out = np.clip(h_out, 0, 1)
            h_out = h_out * (constraint_sum / h_out.sum())
            overlaps = np.correlate(h_out, 1 - h_out, mode='full') * dx
            return np.max(overlaps)

        result = minimize(f, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_opt = np.clip(result.x.reshape(n_points), 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    def mutate_h(h, noise_factor=0.05, max_shift=3):
        h = h.copy()
        h += np.random.randn(n_points) * noise_factor
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())

        # Add positional perturbation
        values = h.copy()
        indices = np.argsort(np.random.randn(n_points))[:max_shift]
        for idx in indices:
            h = np.roll(h, 1)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def crossover(h1, h2):
        mask = np.random.rand(n_points) > 0.5
        h = np.where(mask, h1, h2)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    # Initial optimization pass
    for i in range(len(h_candidates)):
        h = h_candidates[i]
        # Normalize
        h = h / h.sum() * (n_points / 2)
        h_candidates[i] = optimize_h(h)

    best_c5 = min(objective(h) for h in h_candidates)
    best_h = min(h_candidates, key=objective)

    # Population evolution loop
    while time.time() - start_time < budget_s * 0.95:
        next_generation = []

        # Sort current population
        h_ranked = sorted(h_candidates, key=objective)

        # Keep top 20% of population
        next_generation += h_ranked[:max(2, len(h_ranked) // 5)]

        # Generate children through crossover and mutation
        top_half = h_ranked[:5]
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            if len(top_half) >= 2:
                parent1 = top_half[np.random.randint(len(top_half))]
                parent2 = top_half[np.random.randint(len(top_half))]
                child = crossover(parent1, parent2)
                for _ in range(2):
                    child = optimize_h(child)
                next_generation.append(child)

        # Add mutations of best two performers
        top2 = h_ranked[:2]
        for parent in top2:
            if time.time() - start_time > budget_s * 0.95:
                break
            for _ in range(3):
                mutated = mutate_h(parent)
                h = optimize_h(mutated)
                next_generation.append(h)

        # Remove duplicates and trim to population size
        h_candidates = []
        for h in next_generation:
            if len(h_candidates) >= len(next_generation) // 2:
                break
            if all(not np.allclose(h, x, atol=1e-5) for x in h_candidates):
                h_candidates.append(h)

        # Update best solution
        current_c5 = min(objective(h) for h in h_candidates)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = min(h_candidates, key=objective)

    # Final pure gradient-descent refinement
    if best_h is not None and best_h.size > 0:
        best_h = optimize_h(best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())

    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, objective(best_h), n_points
