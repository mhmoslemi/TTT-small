```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def binary_projection(h):
    h_proj = np.copy(h)
    total_needed = len(h) // 2  # sum(h) == n_points / 2
    h_proj = (h_proj >= 0.5).astype(np.float64)
    count = h_proj.sum()
    if abs(count - total_needed) < 1e-6:
        return h_proj
    delta = total_needed - count
    if delta > 0:
        idx = np.argsort(h - h_proj).astype(np.int64)
        h_proj[idx[:int(delta)]] = 1.0
    else:
        idx = np.argsort(h_proj - h).astype(np.int64)
        h_proj[idx[:int(np.abs(delta))]] = 0.0
    return h_proj


def mccormick_relaxation(xy):
    x, y = xy
    return cp.Minimum(x, y) <= xy, cp.Maximum(x + y - 1, 0) <= xy


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_init = np.copy(initial_h_values)
    else:
        h_init = np.zeros(n_points)
        h_init[: n_points // 2] = 1.0
    solution_h = h_init
    best_C5 = evaluate_C5(h_init)
    violated_shifts = []

    for k in range(1, n_points):
        if k < n_points // 2:  # avoid redundant shifts
            violated_shifts.append(k)

    h = cp.Variable(n_points, nonneg=True)
    constraints = [h <= 1.0, cp.sum(h) == n_points / 2.0]

    variables = {}
    for k in violated_shifts:
        overlap = h[: n_points - k] * (1 - h[k:])
        variables[k] = cp.Variable(n_points - k, nonneg=True)
        for i in range(n_points - k):
            constraints += mccormick_relaxation((h[i], 1 - h[k + i]))
            constraints.append(variables[k][i] == h[i] * (1 - h[k + i]))

    lambda_var = cp.Variable()
    constraints.append(cp.sum([cp.sum(variables[k]) for k in variables]) <= lambda_var * n_points)

    objective = cp.Minimize(lambda_var)
    prob = cp.Problem(objective, constraints)

    while time.time() - start_time < budget_s:
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.clip(h.value, 0, 1)
                h_opt = ensure_feasible(h_opt)
                h_bin = binary_projection(h_opt)
                h_bin = ensure_feasible(h_bin)
                C5 = evaluate_C5(h_bin)
                if C5 < best_C5:
                    solution_h = h_bin
                    best_C5 = C5
            break  # stop after one pass over shifts
        except Exception as e:
            pass

    return solution_h, best_C5, n_points
```
