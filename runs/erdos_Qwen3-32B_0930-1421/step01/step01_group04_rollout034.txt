import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = np.zeros(2*n_points-1)
    for k in range(-n_points+1, n_points):
        i = max(0, -k)
        j = min(n_points, n_points - k)
        if i >= j:
            corrs[k + n_points - 1] = 0.0
        else:
            corrs[k + n_points - 1] = np.dot(h[i:j], 1.0 - h[i + k:j + k])
    return np.max(corrs * dx)

def objective(h, n_points):
    return compute_c5(h, n_points)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    expected_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full(n_points, 0.5)
    h = h * expected_sum / current_sum
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_limit(x, alpha=100):
    return np.log(sum(np.exp(alpha * x))) / alpha

def smooth_objective(h, n_points, alpha=100):
    corrs = np.zeros(2*n_points-1)
    for k in range(-n_points+1, n_points):
        i = max(0, -k)
        j = min(n_points, n_points - k)
        if i >= j:
            corrs[k + n_points - 1] = 0.0
        else:
            corrs[k + n_points - 1] = np.dot(h[i:j], 1.0 - h[i + k:j + k])
    return smooth_limit(corrs * (2.0 / n_points), alpha)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    time_start = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        n_points = 47
        h = np.random.rand(n_points)
        h = enforce_normalization(h, n_points)

    n_points = len(h)

    pos = h.copy()
    vel = np.random.uniform(-0.01, 0.01, size=n_points)
    pbest = pos.copy()
    pbest_score = objective(pbest, n_points)
    gbest = pos.copy()
    gbest_score = pbest_score

    w = 0.7
    c1 = 2.0
    c2 = 2.0

    n_swarm = 20
    particles = np.random.uniform(size=(n_swarm, n_points))
    for i in range(n_swarm):
        particles[i] = enforce_normalization(particles[i], n_points)

    p_scores = np.array([objective(p, n_points) for p in particles])
    p_indices = np.argsort(p_scores)
    best_indices = p_indices[:5]

    for it in range(1000):
        if time.time() - time_start > budget_s:
            break

        # Update velocities
        r1, r2 = np.random.rand(2)
        vel = w * vel + c1 * r1 * (pbest - pos) + c2 * r2 * (gbest - pos)

        # Scale velocities
        vel *= 0.25

        pos = pos + vel
        pos = enforce_normalization(pos, n_points)

        current_score = objective(pos, n_points)

        if current_score < pbest_score:
            pbest = pos.copy()
            pbest_score = current_score

        if current_score < gbest_score:
            gbest = pos.copy()
            gbest_score = current_score

        # Local refinement
        if (it % 16) == 0:
            for ind in best_indices:
                x = particles[ind]
                res = opt.minimize(
                    lambda x: smooth_objective(x, n_points, alpha=10),
                    x,
                    bounds=[(0, 1)] * n_points,
                    method='L-BFGS-B',
                    options={'maxiter': 50}
                )
                x_opt = enforce_normalization(res.x, n_points)
                score_opt = objective(x_opt, n_points)
                if score_opt < p_scores[ind]:
                    particles[ind] = x_opt
                    p_scores[ind] = score_opt
                    p_indices = np.argsort(p_scores)
                    best_indices = p_indices[:5]

    final_score = objective(gbest, n_points)
    return gbest, final_score, n_points
