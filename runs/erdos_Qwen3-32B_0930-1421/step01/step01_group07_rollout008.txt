import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_best_flip_indices(h, n_points):
    current_c5 = evaluate_overlap(h)
    best_delta = 0
    best_i = best_j = -1
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    for i in indices_1:
        for j in indices_0:
            h_test = h.copy()
            h_test[i] = 0
            h_test[j] = 1
            new_c5 = evaluate_overlap(h_test)
            delta = new_c5 - current_c5
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
    return best_i, best_j, best_delta
