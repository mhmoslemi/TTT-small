import numpy as np
import optuna
import time

def compute_c5(h, dx):
    n_points = h.size
    max_corr = 0.0
    for k in range(1, n_points):
        # Forward and backward shifts
        if k < n_points:
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            max_corr = max(max_corr, corr)
        if n_points - k < n_points:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            max_corr = max(max_corr, corr)
    return max_corr * dx

def objective(trial, n_points, dx):
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = trial.suggest_float(f'h_{i}', 0, 1)

    # Enforce normalization
    total = np.sum(h)
    if total == 0:
        return float('inf')  # Avoid division by zero
    h = h * (n_points / 2) / total
    h = np.clip(h, 0.0, 1.0)

    # Compute C₅
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    time_start = time.time()

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Optuna genetic algorithm search
    def objective_wrapper(trial):
        return objective(trial, n_points, dx)

    study = optuna.create_study()
    start_time = time.time()
    remaining_budget = budget_s
    while remaining_budget > 5:
        # Dynamically determine iterations based on remaining time
        timeout = min(remaining_budget - 2, 60)
        study.optimize(objective_wrapper, timeout=timeout, n_jobs=1)
        elapsed = time.time() - start_time
        remaining_budget = budget_s - elapsed
        if remaining_budget <= 5:
            break

    # Ensure final normalization and clipping
    h_best = np.clip(study.best_params.values(), 0, 1)
    total = np.sum(h_best)
    if total == 0:
        h_best = np.full(n_points, 0.5)
    else:
        h_best = h_best * (n_points / 2) / total
    c5_bound = study.best_value

    h_best = np.clip(h_best, 0.0, 1.0)
    return h_best, c5_bound, n_points
