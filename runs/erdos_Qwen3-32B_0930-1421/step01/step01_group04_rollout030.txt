import numpy as np
import optlang
import time
import cvxpy as cp
from scipy.optimize import minimize, Bounds

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        s = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(s)
    return np.max(np.array(corrs) * dx)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if not np.isclose(total, n_points / 2, atol=1e-4):
        h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def setup_lp(n_points, h_init):
    h = cp.Variable(n_points)
    C = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]
    for k in [1, 2, 4, 8, 16]:
        if k >= n_points:
            continue
        i_indices = np.arange(n_points - k)
        k_indices = i_indices + k
        overlap = cp.sum(h[i_indices] * (1 - h[k_indices]))
        constraints.append(overlap * (2.0 / n_points) <= C)
    objective = cp.Minimize(C)
    return h, C, constraints, objective

def solve_lp(h, C, constraints, objective):
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    if prob.status != 'optimal':
        return None, None
    return h.value, C.value

def column_selection(h_val, n_points):
    max_overlap = 0.0
    worst_k = 0
    for k in range(1, n_points):
        if k >= n_points:
            continue
        i_indices = np.arange(n_points - k)
        k_indices = i_indices + k
        overlap = np.sum(h_val[i_indices] * (1 - h_val[k_indices]))
        overlap *= (2.0 / n_points)
        if overlap > max_overlap:
            max_overlap = overlap
            worst_k = k
    return worst_k

    # Initialization
    objective = compute_c5(h, n_points)

    time_start = time.time()
    elapsed_time = 0

    # LP setup
    h_var, C_var, constraints, objective_lp = setup_lp(n_points, h)
    while time.time() - time_start < budget_s:
        # Solve the current LP
        h_val, C_val = solve_lp(h_var, C_var, constraints, objective_lp)
        if h_val is None:
            break
        h_val = np.array(h_val).flatten()
        h_val = enforce_normalization(h_val, n_points)
        current_c5 = compute_c5(h_val, n_points)

        # Update best
        if current_c5 < best_c5:
            best_h = h_val.copy()
            best_c5 = current_c5

        # Identify worst constraint
        worst_k = column_selection(h_val, n_points)

        i_indices = np.arange(n_points - worst_k)
        k_indices = i_indices + worst_k
        new_c = cp.sum(h_var[i_indices] * (1 - h_var[k_indices])) * (2.0 / n_points) <= C_var
        constraints.append(new_c)

        elapsed_time = time.time() - time_start
        if elapsed_time > budget_s - 10:
            break

    # Final refinement
    final_h = minimize(
        lambda x: compute_c5(x, n_points),
        best_h,

    final_h = enforce_normalization(final_h, n_points)
    final_c5 = compute_c5(final_h, n_points)
