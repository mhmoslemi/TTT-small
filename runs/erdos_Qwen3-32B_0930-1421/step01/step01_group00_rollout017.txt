import time
import cvxpy as cp

    # Step 2: Constraint-aware scaling
    # ----------------
    def normalize_h(h):
        h = np.clip(h, 0, 1)  # enforce bounds
        h_sum = h.sum()
        # Fix sum constraint if necessary
        if abs(h_sum - target_sum) > 1e-8:
            attempts = 0
            while abs(h_sum - target_sum) > 1e-8 and attempts < 10:
                excess = h_sum - target_sum
                if abs(excess) < 1e-10:
                    break
                # Adjust the smallest entries to meet the constraint
                h -= (h - np.average(h)) * (excess / h_sum)
                h = np.clip(h, 0, 1)
                h_sum = h.sum()
                attempts += 1
        return h

    h = normalize_h(h_init)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Step 3: Optimization with convex relaxation
    # ----------------
    # Try a convex optimization to find a global minimum candidate
    try:
        x = cp.Variable(n_points)
        ones_minus_h = 1 - h
        max_corr = -np.inf

        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == target_sum
        ]

        # Use multiple overlap positions
        n_conv = n_points - 1
        overlap_costs = []

        for k in range(1, n_points - 1):
            if k >= n_points - 1:
                continue
            ref_h = x[:n_points - k]
            ref_ones_minus_h = ones_minus_h[k:]
            overlap_cost = cp.sum(ref_h * ref_ones_minus_h)
            overlap_costs.append(overlap_cost * dx)

        for k in range(1, n_points - 1):
            if k >= n_points - 1:
                continue
            ref_h = x[k:]
            ref_ones_minus_h = ones_minus_h[:n_points - k]
            overlap_cost = cp.sum(ref_h * ref_ones_minus_h)
            overlap_costs.append(overlap_cost * dx)

        # Minimize the worst case
        cost = cp.Max(overlap_costs)
        prob = cp.Problem(cp.Minimize(cost), constraints)

        # Set solver-based optimization
        prob.solve(solver=cp.ECOS, max_iters=500, verbose=False)
        h_opt = np.clip(x.value, 0, 1)

        h = normalize_h(h_opt)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # ----------------
    def local_search_step(h, perturbation_size=0.02, subset_size=20):

        proposed_h = h.copy()
        noise = np.random.uniform(-perturbation_size, perturbation_size, size=proposed_h[indices].shape)

        proposed_h = normalize_h(proposed_h)

        proposed_c5 = compute_c5(proposed_h)

        if proposed_c5 < best_c5:
            h = proposed_h.copy()
            best_c5 = proposed_c5
            best_h = h.copy()
            best_time = time.time()
            perturbation_size *= perturbation_decay
        elif np.random.rand() < 0.05:  # simulated annealing
            h = proposed_h.copy()
            perturbation_size *= 1.05  # grow to escape flat areas

        if time.time() - best_time > patience:

    # ----------------
    # Step 5: Final evaluation
    # ----------------
    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= optimized_h.shape[0]:
                continue
            corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr1 * dx)
            corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr2 * dx)
        return max_corr

    final_best_c5 = compute_c5(best_h)
    return (best_h, final_best_c5, n_points)
