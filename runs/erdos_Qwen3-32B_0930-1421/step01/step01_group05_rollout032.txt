import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Phase 1: Rapid sweep over resolutions (30% of time)
    res_list = [30, 40, 45, 60, 80]
    for res in res_list:
        if time.time() - start_time >= budget_s:
            break
        h_high = np.random.rand(res)
        h_high = normalize_h(h_high)
        h_inter = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, res), h_inter)
        h_inter = normalize_h(h_inter)

        result = minimize(lambda x: compute_c5(normalize_h(x)), h_inter, 
                          bounds=[(0,1)]*n_points, method='L-BFGS-B', 
                          options={'maxiter': 1000, 'ftol': 1e-8})
        h_opt = normalize_h(result.x)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Adaptive local search (35% of time)
    learning_rate = 0.05
    while time.time() - start_time < 0.65 * budget_s:
        grad = np.zeros_like(best_h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = normalize_h(h_plus)
            grad[i] = compute_c5(h_plus) - best_c5

            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = normalize_h(h_minus)
            grad[i] -= compute_c5(h_minus) - best_c5

        grad /= (2 * eps)
        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        learning_rate *= 0.995

    # Phase 3: Mixed-integer linear programming (35% of time)
    try:
        from cvxpy.error import SolverError
        hvar = cp.Variable(n_points)
        c = [cp.Maximize(0)]  # Dummy for placeholder
        for k in range(-n_points + 1, n_points):
            c_k = 0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    c_k += hvar[i] * (1 - hvar[j])
            c.append(cp.Max(v))
        v = cp.Variable()
        constraints = [
            cp.sum(hvar) == n_points / 2,
            0 <= hvar,
            hvar <= 1,
        ]
        problem = cp.Problem(cp.Minimize(v), constraints + [v >= c[k] for k in range(len(c))])
        problem.solve(solver=cp.SCIP, maximum_total_time=budget_s - time.time() + start_time)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.clip(hvar.value, 0, 1)
            h_opt = normalize_h(h_opt)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n_points = n_points
    except SolverError:
        pass

    return best_h.tolist(), best_c5, best_n_points
