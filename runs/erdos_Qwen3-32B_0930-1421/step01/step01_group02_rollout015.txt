import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def build_bump_basis(n_basis, n_points):
    dx = 2.0 / n_points
    basis = np.zeros((n_basis, n_points))
    centers = np.linspace(0, 2, n_basis)
    sigma = 0.3  # empirical width to minimize overlap

    for i in range(n_basis):
        x = np.arange(n_points) * dx
        distance = np.abs(x - centers[i])
        basis[i] = np.exp(-1.5 * distance / sigma)
        # Cut off at zero to avoid crossing unit bounds
        basis[i][basis[i] > 1] = 1.0

    return basis, centers


def optimize_coefficients(basis):
    n_basis = basis.shape[0]
    n_points = basis.shape[1]
    coeff = cp.Variable(n_basis, nonneg=True)
    budget_constraints = []

    # Energy constraint to ensure total mass = 1
    integral = basis.sum(axis=1) * (2.0 / n_points)
    integral = cp.Parameter(n_basis, value=integral)
    mass_constr = cp.sum(coeff * integral) == 1
    constraints = [mass_constr]

    # Construct all correlations: for all k in [-n_points+1, ..., n_points-1]
    # We use a discretized overlap constraint
    max_overlap = cp.Variable()
    overlap_bounds = []
    for k in range(n_points):
        # Construct the overlap term: sum_i sum_j coeff[i] coeff[j] * corr(basis[i], basis[j], shift=k)
        xcorr = np.zeros(n_basis)
        for i in range(n_basis):
            shifted = np.zeros(n_points)
            if k < n_points:
                shifted[k:] = basis[i][:n_points - k]
            else:
                shifted[:n_points - k] = basis[i][k:]
            xcorr[i] = np.dot(basis[i], shifted) * (2.0 / n_points)
        overlap_bound = cp.sum(cp.multiply(coeff, xcorr)) <= max_overlap
        overlap_bounds.append(overlap_bound)

    constraints += overlap_bounds
    constraints += [coeff <= 1.0]

    objective = cp.Minimize(max_overlap)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.CVXOPT, verbose=False)

    if problem.status != "optimal":
        raise RuntimeError("CVXOPT failed to find optimal coefficients.")

    optimized_coeff = coeff.value
    return optimized_coeff


def construct_h(basis, coefficients):
    n_basis = basis.shape[0]
    h = np.zeros(basis.shape[1])
    for i in range(n_basis):
        h += coefficients[i] * basis[i]
    h = ensure_feasible(h)
    return h


def run(budget_s=1100, n_points=42, n_basis=6, n_restart_attempts=5):
    start_time = time.time()
    best_C5 = np.inf
    best_h = None

        h_init = initial_h_values
    else:
        h_init = np.zeros(n_points)
        h_init[n_points // 4] = 1.0

    h_init = h_init / h_init.sum() * n_points / 2
    h_init = ensure_feasible(h_init)

    basis_atom, centers = build_bump_basis(n_basis, n_points)

    for restart in range(n_restart_attempts):
        # Slight perturbation in basis centers
        perturbed_centers = centers + np.random.uniform(-0.1, 0.1, size=n_basis)
        # Ensure centers still in domain [0,2]
        perturbed_centers = np.clip(perturbed_centers, 0, 2)
        # Renew basis with new centers
        tmp_basis, _ = build_bump_basis(n_basis, n_points, centers=perturbed_centers)

        # Optimize over coefficients
        coeffs = optimize_coefficients(tmp_basis)
        h_candidate = construct_h(tmp_basis, coeffs)

        # Ensure numerical feasibility
        h_candidate = ensure_feasible(h_candidate)

        # Evaluate C5
        C5 = evaluate_C5(h_candidate)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_candidate
            print(f"Restart {restart+1}: C₅ = {C5:.7f}")

        # Check time
        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_C5, n_points
