import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_restarts = 100
    max_iter = 20
    n_points = 47  # Use default or initial if available
    if 'initial_h_values' in globals():
        h_raw = np.array(initial_h_values)
        n_points = len(h_raw)
    else:
        h_raw = np.zeros(n_points)
        for i in range(n_points // 2):
            h_raw[i + i] = 2.0 / n_points

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def normalize_h(h):
        h = np.clip(h, 0, 1).copy()
        h_sum = h.sum()
        if h_sum == 0:
            return np.zeros_like(h)  # avoid division by 0
        return h * (constraint_sum / h_sum)

    def objective(h):
        h = normalize_h(h)
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    def convex_relaxation(h_guess):
        h_var = cp.Variable(n_points)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == constraint_sum
        ]
        all_c5 = []
        for k in range(n_points * 2):
            overlap = cp.conv(h_var, 1 - h_var, mode="full")[k]
            all_c5.append(overlap)
        c5 = cp.MaximizeVariable(overlaps=len(all_c5))
        constraints += [overlap <= c5 for overlap in all_c5]
        problem = cp.Problem(cp.Minimize(c5), constraints)
        problem.solve(solver=cp.MOSEK)
        return h_var.value, c5.value

    def smart_perturbation(h, noise_factor=0.1):
        h_pert = h + np.random.normal(0, noise_factor, h.shape)
        h_pert = normalize_h(h_pert)
        return h_pert

    best_h = h_raw
    best_c5 = objective(h_raw)

    for i in range(max_restarts):
        if time.time() - start_time > budget_s * 0.99:
            break

        if i > 0:
            h_current = smart_perturbation(best_h, noise_factor=0.1)
        else:
            h_current = h_raw.copy()

        try:
            h_opt, c5_opt = convex_relaxation(h_current)
            h_opt = normalize_h(h_opt)
            current_c5 = objective(h_opt)

            if current_c5 < best_c5:
                best_h = h_opt
                best_c5 = current_c5
        except Exception:
            continue

    # Post-process to ensure valid solution
    best_h = normalize_h(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
