import numpy as np
import scipy.optimize as opt
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx


def objective_smooth(h_vec, n_points, beta=100):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return (np.log(np.sum(np.exp(beta * np.array(corrs) * dx))) / beta)


def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)


def raised_cosine_init(n_points):
    x = np.linspace(0, 1, n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    h = enforce_normalization(h, n_points)
    return h


def objective_barrier(h_vec, n_points):
    h = h_vec.reshape(-1)
    h = enforce_normalization(h, n_points)
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    max_corr = np.max(np.array(corrs))
    alpha = 100
    barrier = -np.sum(np.log(h + 1e-8) + np.log(1 - h + 1e-8)) - np.log(np.abs(np.sum(h) - n_points / 2 + 1e-8))
    return max_corr + alpha * barrier

    h_cos = raised_cosine_init(n_points)

    # Particle Swarm Optimization setup
    n_particles = 20
    particles = [enforce_normalization(np.random.rand(n_points), n_points) for _ in range(n_particles)]
    particles[0] = h_start  # First particle seeded from initial_h_values and raised cosine
    velocities = [0.01 * np.random.randn(n_points) for _ in range(n_particles)]
    particle_bests = [p.copy() for p in particles]
    particle_best_values = [np.inf for _ in range(n_particles)]
    global_best = h_start
    global_best_value = np.inf
    time_start = time.time()
    max_iter = 1000
    iteration = 0

    while time.time() - time_start < budget_s:
        iteration += 1
        for i in range(n_particles):
            h = particles[i].copy()
            val = compute_c5(h, n_points)
            if val < particle_best_values[i]:
                particle_bests[i] = h.copy()
                particle_best_values[i] = val
            if val < global_best_value:
                global_best = h.copy()
                global_best_value = val

        # Update velocities and positions
        w = max(0.9 - iteration / max_iter * 0.5, 0.4)
        c1 = 2.0
        c2 = 2.0
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (particle_bests[i] - particles[i])
                + c2 * r2 * (global_best - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            particles[i] = particles[i] + velocities[i]
            particles[i] = enforce_normalization(particles[i], n_points)

        # Local opt on top 5 particles
        top_indices = np.argsort(particle_best_values)[:5]
        for i in top_indices:
            bounds = [(0, 1)] * n_points
            res = opt.minimize(
                lambda x: objective_barrier(x, n_points),
                particle_bests[i],
                bounds=bounds,
                method='L-BFGS-B',
                options={'maxiter': 1000, 'ftol': 1e-6}
            )
            h_new = res.x
            h_new = enforce_normalization(h_new, n_points)
            val_new = compute_c5(h_new, n_points)
            if val_new < particle_best_values[i]:
                particle_bests[i] = h_new
                particle_best_values[i] = val_new
                if val_new < global_best_value:
                    global_best = h_new
                    global_best_value = val_new

        if iteration % 50 == 0:
            print(f"Iter {iteration}, Best C5: {global_best_value:.6f}")


    if global_best_value == np.inf:
        return h_init, compute_c5(h_init, n_points), n_points

    return global_best, global_best_value, n_points
