import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, corr, n_points):
    # compute best negative gradient over shifts, i.e. max norm direction among worst shifts
    lag_shape1 = n_points - 1
    lag_shape2 = 2 * n_points - 1
    best_corr_idx = np.argmax(corr)
    lag = best_corr_idx - (n_points - 1)
    ones = np.zeros_like(h)
    grad = np.zeros_like(h)
    for i in range(n_points):
        if i + lag < n_points:
            grad[i] += ones[i + lag]
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    return grad

def project_onto_constraints(h, n_points):
    # Step 1: clip to [0,1]
    h = np.clip(h, 0, 1)
    # Step 2: project to sum constraint
    sum_h = h.sum()
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    h *= scale
    return h
