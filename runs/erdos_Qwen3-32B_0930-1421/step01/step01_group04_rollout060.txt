import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def find_most_violated_constraint(h, n_points):
    worst_overlap = 0.0
    worst_k = 0
    for k in range(1, n_points):
        overlap = np.dot(h[:n_points - k], 1 - h[k:])
        if overlap > worst_overlap:
            worst_overlap = overlap
            worst_k = k
    return worst_k, worst_overlap

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h_init = initial_h_values.copy()
    h_init = enforce_normalization(h_init, n_points)

    # Set up base LP
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]

    # Start with a small set of constraints
    for k in [1, 2, 4, 8, 16]:
        if k >= n_points:
            continue
        constr = 0
        for i in range(n_points - k):
            constr += h_var[i] * (1 - h_var[i + k])
        constraints.append(constr * (2.0 / n_points) <= C)

    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_points)
    time_start = time.time()

    # Column generation loop
    while time.time() - time_start < budget_s - 10:
        # Solve current LP
        prob.solve(solver=cp.GUROBI, verbose=False)
        if prob.status != cp.OPTIMAL:
            break

        # Extract current variable values
        h_current = h_var.value
        h_current = enforce_normalization(h_current, n_points)
        current_c5 = compute_c5(h_current, n_points)

        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5

        # Find most violated constraint
        worst_k, max_overlap = find_most_violated_constraint(best_h, n_points)
        new_c5 = max_overlap * (2.0 / n_points)
        if new_c5 > best_c5 + 1e-5:
            constr = 0
            for i in range(n_points - worst_k):
                constr += h_var[i] * (1 - h_var[i + worst_k])
            constraints.append(constr * (2.0 / n_points) <= C)
            prob = cp.Problem(objective, constraints)

    # Final refinement with gradient-based optimization
    def refine_obj(h_vec):
        return compute_c5(h_vec, n_points)

    if time.time() - time_start < budget_s:
        result = opt.minimize(
            refine_obj,
            best_h,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            options={'maxiter': 100}
        )
        best_h = enforce_normalization(result.x, n_points)
        best_c5 = compute_c5(best_h, n_points)

    return best_h, best_c5, n_points
