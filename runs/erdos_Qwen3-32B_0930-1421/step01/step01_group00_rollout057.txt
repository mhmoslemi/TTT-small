import numpy as np
import time
import scipy.fft as fft

def run(seed=42, budget_s=1100, verbose=False):

    budget_s = min(budget_s, 1100)

    # Step 1: Load and setup
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ∫h dx = 1 => sum(h) = n_points / 2

    # Normalize to exact constraint
    h_sum = h.sum()
    h = np.clip(h, 0, 1)
    if h_sum != target_sum:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Step 2: Optimization using Fourier-based search
    h_fft = fft.fft(h)

    def evolve_definition_fft(h_fft, learning_rate=0.01):
        idxs = np.random.choice(len(h_fft), replace=False, size=min(40, len(h_fft)))
        direction = np.random.uniform(-1, 1, len(h_fft))
        h_fft_pert = h_fft.copy()
        h_fft_pert[idxs] += learning_rate * direction[idxs]
        return h_fft_pert

    def recover_and_clip(real_space_vals):
        real_space_vals = np.real(real_space_vals)
        real_space_vals = np.clip(real_space_vals, 0, 1)
        s = real_space_vals.sum()
        if s != target_sum:
            real_space_vals *= (target_sum / s)
        real_space_vals = np.clip(real_space_vals, 0, 1)
        return real_space_vals

    def compute_c5(eval_h, dx):
        eval_h = np.ravel(eval_h)
        ones_minus_h = 1 - eval_h
        full_correl = scipy.signal.correlate(eval_h, ones_minus_h, mode='full')
        return full_correl.max() * dx

    while time.time() - start_time < budget_s - 10:
        # Perturb h in the Fourier domain
        h_fft = evolve_definition_fft(h_fft)
        new_h = recover_and_clip(fft.ifft(h_fft).real)
        new_c5 = compute_c5(new_h, dx)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            if verbose:
                print(f"Found better C5: {best_c5:.6f} after {time.time() - start_time:.2f}s")

            if new_c5 < 0.379:
                break

        # Heuristically reduce exploration amplitude as we improve
        if best_c5 < 0.3809:
            learning_rate = 0.002
        elif best_c5 < 0.3814:
            learning_rate = 0.004
        else:
            learning_rate = 0.01

        # Early stopping if no improvement in feasibility
        if time.time() - start_time > 600 and best_c5 > 0.382:
            break

    return (best_h, best_c5, n_points)
