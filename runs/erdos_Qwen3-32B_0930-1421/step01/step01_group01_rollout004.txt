import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=120):
    """Optimize step function h to minimize overlap for Erdős C₅."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if available and appropriate size
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points, refcheck=False)
    except:
        h = np.zeros(n_points)
        h[:30] = 0.1
        h[30:90] = 0.3
        h[90:] = 0.25
        h = rescale_h(h, n_points)

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iter = 1000
    iter_step = 0
    perturb_interval = 30  # Perturb every 30 iterations to escape local minima

    def rescale_h(h, n_points):
        h = np.clip(h, 0, 1)
        total = h.sum()
        if total <= 1e-10:
            h[0] = n_points / 2.0
            return h
        return h * (n_points / 2.0) / total

    def evaluate_C5(h, dx):
        cor = correlate(h, 1 - h, mode="full")
        C5 = np.max(cor) * dx
        return C5

    while time.time() - start_time < budget_s:
        iter_step += 1
        h = rescale_h(h, n_points)
        C5 = evaluate_C5(h, dx)

        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()

        if iter_step % perturb_interval == 0:
            # Perturbation step using random Latin hypercube or affine shift
            h_flat = h.copy()
            mask = np.random.rand(n_points) < 0.1
            h_flat[mask] = np.random.rand(np.sum(mask))
            h = rescale_h(h_flat, n_points)
        else:
            # Adaptive convex optimization for dominant overlap shift
            cor = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(cor)
            k = max_idx - n_points + 1  # shift index

            if 0 <= k < n_points:
                h_var = cp.Variable(n_points, nonneg=True, name='h')
                objective = cp.Minimize(correlate(h_var, 1 - h_var, mode="full")[max_idx] * dx)
                constraints = [h_var.sum() == n_points / 2.0, h_var <= 1.0]
                problem = cp.Problem(objective, constraints)

                try:
                    problem.solve(solver=cp.ECOS, verbose=False, max_iters=50, abstol=1e-6)
                    if problem.status == cp.OPTIMAL:
                        h = h_var.value.copy()
                    else:
                        h = rescale_h(h * 0.9 + np.random.rand(n_points) * 0.2, n_points)
                except:
                    h = rescale_h(h * 0.9 + np.random.rand(n_points) * 0.2, n_points)
        if iter_step % max(20, perturb_interval // 2) == 0:
            print(f"C₅: {best_c5:.7f} | t: {(time.time() - start_time):.2f}s")

    return best_h, best_c5, n_points
