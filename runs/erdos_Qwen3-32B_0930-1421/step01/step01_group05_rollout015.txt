import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    def compute_gradient(h):
        curr_c5 = compute_c5(h)
        max_overlap_idx = np.argmax(correlate(h, 1 - h, mode='full'))

        grad = np.zeros_like(h)
        for i in range(n_points):
            h_i = h.copy()
            h_i[i] += 1e-5
            h_i = normalize_h(h_i)
            c_i = np.max(correlate(h_i, 1 - h_i, mode='full') * dx)
            grad[i] = (c_i - curr_c5) / 1e-5
        return grad

    best_c5 = compute_c5(h)
    best_h = h.copy()

    learning_rate = 0.5
    lr_decay = 0.99
    explore_step = 0.05
    beta = 0.5

    while time.time() - start_time < budget_s:
        grad = compute_gradient(h)
        h -= learning_rate * grad
        h = normalize_h(h)

        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            learning_rate *= lr_decay
        else:
            # Exploration around local minimum
            perturbation = np.random.randn(n_points) * explore_step
            h_explore = np.clip(h + perturbation, 0, 1)
            h_explore = normalize_h(h_explore)
            c_explore = compute_c5(h_explore)
            if c_explore < best_c5:
                h = h_explore
                best_c5 = c_explore
                best_h = h.copy()
            else:
                # Revert with probability beta
                if np.random.rand() < beta:
                    h = best_h.copy()

        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
