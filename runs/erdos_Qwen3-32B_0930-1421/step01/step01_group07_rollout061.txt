import numpy as np
import time
from scipy.signal import correlate

def safe_clip_sum(h):
    h_clipped = np.clip(h, 0.0, 1.0)
    mean = h_clipped.sum() / len(h_clipped)
    target_sum = len(h_clipped) / 2
    alpha = (target_sum / h_clipped.sum()) ** 0.5
    h_clipped = np.clip(h_clipped * alpha, 0.0, 1.0)
    return h_clipped

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = safe_clip_sum(initial_h_values)

    t0 = time.time()
    T = 1.0
    alpha = 0.99
    min_temp = 1e-5
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5

    n_perturb = 8       # Number of positions to perturb per step
    sigma = 0.20       # Initial noise magnitude
    sigma_decay = 0.999
    patience = 0
    patience_threshold = 2000

    while time.time() - t0 < budget_s:
        # Perturb
        indices = np.random.randint(0, n_points, n_perturb)
        noise = sigma * np.random.randn(n_perturb)
        delta = current_h[indices].copy()
        delta += noise
        delta = np.clip(current_h[indices] + noise, 0, 1)

        h_new = current_h.copy()
        h_new[indices] = delta
        h_new = safe_clip_sum(h_new)

        # Evaluate
        c5_new = evaluate_overlap(h_new)
        delta_c5 = c5_new - current_c5

        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

        # Adjust temperature
        if patience >= patience_threshold:
            T = max(T * 2, min_temp)
            patience = 0
        else:
            T *= alpha

        if sigma > 1e-3:
            sigma *= sigma_decay

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Constraint violation"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
