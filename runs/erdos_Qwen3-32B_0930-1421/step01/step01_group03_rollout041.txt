import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use starting construction provided if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True)
        n_points = len(h)
    else:
        h = np.zeros(47)
        n_points = len(h)
        for i in range(n_points // 2):
            h[i + i] = 2.0 / n_points
        h *= n_points / 2 / h.sum()  # Normalize to constraint
    dx = 2.0 / n_points

    # Constraint: sum(h) == n_points / 2
    constraint_sum = n_points / 2

    def objective(h_flattened):
        h = h_flattened.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())  # normalize
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def feasible_projection(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h *= constraint_sum / h.sum()
        return h

    def solve_lp(h):
        h_var = cp.Variable(n_points)
        C = cp.Variable(nobound=True)

        constraints = [0 <= h_var, h_var <= 1, h_var.sum() == constraint_sum]

        max_overlap = -float('inf')
        corr_h = feasible_projection(h).copy()

        for k in range(-n_points + 1, n_points):
            # flatten indices for correlation
            shifted = np.roll(corr_h, k) if k != 0 else corr_h
            i0 = max(0, -k)
            i1 = min(n_points, n_points - k)
            overlaps = np.sum(corr_h[i0:i1] * (1 - shifted[i0:i1]))

            # Linearization of bilinear term with Taylor first-order expansion at h
            # h[i]*(1 - h[(i + k) % n_points]) ≈ (1 - h[(i + k) % n_points]) * h_var[i] 
            #                                   - h[i] * h_var[(i + k) % n_points] + h[i] * h[(i + k) % n_points]
            linear_overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                linear_overlap += (1 - corr_h[j]) * h_var[i] - corr_h[i] * h_var[j] + corr_h[i] * corr_h[j]

            constraints.append(linear_overlap * dx <= C)

        objective = cp.Minimize(C)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status != 'optimal':
            return corr_h, 1e9
        return feasible_projection(h_var.value), C.value

    def sample_perturbation(h, noise_factor):
        h = h.reshape(n_points)
        h = h.copy()
        noise = (np.random.rand(n_points) * 2 - 1) * noise_factor
        h += noise
        return feasible_projection(h)

    best_h = feasible_projection(h)
    best_c5 = objective(best_h)

    h_current = best_h
    h_start = h_current.copy()

    # 50 LP solves with limited steps
    for iter_count in range(50):
        if time.time() - start_time > budget_s * 0.99:
            break

        h_opt, c5_val = solve_lp(h_current)
        c5_val = objective(h_opt)
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val

        h_current = h_opt
        # occasional random perturbation to escape local optima
        if iter_count % 5 == 0 and iter_count > 0:
            h_current = sample_perturbation(h_current, noise_factor=0.05)

    # Final projection to constraints
    h_final = feasible_projection(best_h)
    c5_final = objective(h_final)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_final = feasible_projection(h_final)
        c5_final = objective(h_final)

    return h_final, c5_final, n_points
