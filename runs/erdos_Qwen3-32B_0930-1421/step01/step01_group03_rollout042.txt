import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_init = np.array(initial_h_values, copy=True)

    # Enforce constraint: sum(h) == n_points / 2 exactly
    def normalize_h(h):
        h_clipped = np.clip(h, 0, 1)
        return h_clipped * (n_points / 2) / h_clipped.sum()

    def evaluate_c5(h):
        h = normalize_h(h)
        ones_minus_h = 1 - h
        overlaps = np.correlate(h, ones_minus_h, mode='full')
        return np.max(overlaps) * dx

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    h = h_init.copy()

    # Perturbation parameters
    max_perturbations = 6
    perturbation_size = 0.05
    perturb_freq = 7  # Perturb every 7 iterations to boost exploration

    # Build shifts
    shift_indices = [k for k in range(-n_points + 1, 0) if k != 0] + [k for k in range(1, n_points)]

    # Time tracking
    evals_since_perturb = 0
    max_attempts = 100
    attempt = 0

    while time.time() - start_time < budget_s and attempt < max_attempts:
        # Optimization variables
        h_var = cp.Variable(n_points)
        c5 = cp.Variable()

        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            c5 >= 0
        ]

        # Linear approximation of overlap constraints
        overlaps = []
        for k in shift_indices:
            idxs = np.mod(np.arange(n_points) - k, n_points)
            hk = h[np.arange(n_points), idxs]
            overlap_approx = cp.sum(h_var * (1 - hk))
            overlaps.append(overlaps[-1] if len(overlaps) else overlap_approx)
            overlap_diff = (1 - hk) * h_var + (1 - h_var) * hk
            constraint = c5 >= overlap_approx * dx
            constraint += cp.sum(overlap_diff) <= 1e-4  # Enforces approximation quality
            constraints.append(c5 >= overlap_approx * dx)
        c5 >= overlaps[-1] * dx

        # Optimization problem
        objective = cp.Minimize(c5)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status != 'optimal':
            # Perturb if LP fails
            h += 0.1 * np.random.randn(n_points)
            h = normalize_h(h)
        else:
            candidate_h = h_var.value
            if candidate_h is not None:
                h = normalize_h(candidate_h)
                candidate_c5 = evaluate_c5(h)
                if candidate_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = candidate_c5

        # Perturbation every few iterations
        if evals_since_perturb >= perturb_freq:
            h += perturbation_size * np.random.randn(n_points)
            h = normalize_h(h)
            evals_since_perturb = 0
            attempt += 1
        else:
            evals_since_perturb += 1
            attempt += 1

    best_h = normalize_h(best_h)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
