```python
import numpy as np
import time
from collections import deque


initial_h_values = None  # provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode="full")
    return np.max(correlations) * dx


def ensure_feasible_discrete(h, discretization_step=0.25):
    h = np.clip(h, 0, 1)
    h = np.round(h / discretization_step) * discretization_step
    total = np.sum(h)
    n_points = len(h)
    h *= (n_points / 2) / total
    return h


def energy_contribution(h_partial):
    """Compute the contribution of fixed h values to all overlap integrals"""
    n_points = len(h_partial)
    indices = np.where(h_partial != -1)[0]  # which indices are fixed
    h_fixed = h_partial[indices].astype(float)
    C5_contrib = np.zeros(2 * n_points - 1)
    
    for lag in range(-n_points + 1, n_points):
        shift = lag
        fix_pairs = []
        for i in indices:
            j = (i + shift) % n_points
            if h_partial[j] != -1:
                fix_pairs.append((i, j))
                C5_contrib[lag + n_points - 1] += h_partial[i] * h_partial[j]
    
    return C5_contrib


def run(budget_s=1100):
    np.random.seed(42)
    start_time = time.time()
    best_C5 = float('inf')
    best_h = None
    n_points = 42
    
    if initial_h_values is not None:
        h_curr = np.copy(initial_h_values)
    else:
        h_curr = np.zeros(n_points)
        h_curr[::2] = 1.0
        h_curr[0] = 0.5  # manual fix to meet the integral constraint more easily

    # Discretize current h and ensure feasible
    h_curr = ensure_feasible_discrete(h_curr)

    # Feasibility check
    total = np.sum(h_curr)
    n_points = len(h_curr)
    assert np.allclose(total, n_points / 2), f"Initial h_curr is infeasible: sum is {total}, expected {n_points / 2}"

    # Initialize branch-and-bound with a queue
    frontier = deque()
    curr_vals = np.full(n_points, -1)
    visited_sum = 0.0
    known_indices = []

    for i in range(n_points):
        if h_curr[i] in {0.0, 0.25, 0.5, 0.75, 1.0}:
            curr_vals[i] = h_curr[i]
            visited_sum += h_curr[i]
            known_indices.append(i)

    frontier.append((curr_vals.copy(), visited_sum))

    TOL = 1e-8
    max_lazy = 1000
    lazy_count = 0

    while time.time() - start_time < budget_s:
        if not frontier:
            break

        partial_vals, sum_fixed = frontier.pop()  # DFS
        fixed_indices = np.where(partial_vals != -1)[0]
        free_indices = np.where(partial_vals == -1)[0]
        n_free = len(free_indices)

        # Compute remaining sum to be assigned to free variables
        target_total = n_points / 2
        remaining_sum = target_total - sum_fixed

        if abs(remaining_sum) > 1e-9 and n_free == 0:
            continue  # infeasible leaf

        # Enumerate next variable to assign
        if n_free > 0:
            # Heuristic to choose the most impactful variable
            impacts = []
            for idx in free_indices:
                c = 0.0
                for lag in range(-n_points + 1, n_points):
                    if lag >= 0:
                        j = (idx + lag) % n_points
                    else:
                        j = (idx - lag - 2) % n_points
                    if (idx, j) not in [(a, b) for a, b in zip(fixed_indices, fixed_indices)]:
                        c += 1
                impacts.append((idx, c))
            next_idx, _ = max(impacts, key=lambda x: x[1])
        else:
            # All variables fixed, compute exact C5
            h = partial_vals.copy()
            h = ensure_feasible_discrete(h)
            c5 = evaluate_C5(h)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h
            lazy_count += 1
            if lazy_count > max_lazy:
                break
            continue

        # For the selected variable, try all values in {0, 0.25, 0.5, 0.75, 1}
        for val in [0.0, 0.25, 0.5, 0.75, 1.0]:
            if val <= 1.0 and val >= 0.0:
                # Prune infeasible directions
                if (sum_fixed + val) > target_total + TOL:
                    continue  # too much

                if (sum_fixed + val + (n_free - 1)) < target_total - TOL:
                    continue  # too little

                new_partial_vals = partial_vals.copy()
                new_partial_vals[next_idx] = val
                new_sum_fixed = sum_fixed + val

                # Compute a lower bound on C5 for this branch
                contrib = energy_contribution(new_partial_vals)
                free_count = n_points - np.count_nonzero(new_partial_vals != -1)
                max_lag_contrib = np.max(contrib)  # already fixed values' contributions
                bound = max_lag_contrib * (2 / n_points)

                if bound < best_C5:
                    frontier.append((new_partial_vals, new_sum_fixed))
        # Continue DFS

    # Post-check best solution and return
    h_final = ensure_feasible_discrete(best_h)
    total_final = np.sum(h_final)
    assert np.allclose(total_final, n_points / 2), "Returned solution infeasible"

    return h_final, best_C5, n_points
```
