import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def project_to_constraints(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = h.sum()
        if not np.isclose(h_sum, target_sum):
            h *= target_sum / h_sum
        return h

    best_h = project_to_constraints(h)
    best_c5 = np.inf

    def compute_c5(h_instance):
        h_instance = project_to_constraints(h_instance)
        ones_minus_h = 1.0 - h_instance
        corr1 = scipy.signal.correlate(h_instance, 1 - h_instance, mode="full")
        corr2 = scipy.signal.correlate(1 - h_instance, h_instance, mode="full")
        full_corr = np.concatenate((corr1[n_points - 1:], corr2[:n_points]))
        return (np.max(full_corr) * dx)

    def fourier_step(h_instance):
        n_fft = n_points * 2
        H = np.fft.fft(h_instance, n_fft)
        freqs = np.fft.fftfreq(n_fft, d=1 / n_fft)

        # Sort by frequency energy
        magnitude = np.abs(H)
        indices = np.argsort(magnitude)[::-1]
        top_indices = indices[magnitude[indices] > 1e-3]

        # Keep medium-frequency components (enough to let h stay piecewise simple)
        energy_freqs = np.divide(1.0, 1.0 + np.square(freqs * 2 * np.pi * (target_sum / 2)))

        # Perturb with energy-guide weights
        noise = np.fft.ifft(
            H * np.random.normal(0.0, 0.02 * energy_freqs, n_fft)
            + 1j * np.random.normal(0.0, 0.02 * energy_freqs, n_fft),
            n_points
        ).real
        h_instance = h_instance + noise
        return project_to_constraints(h_instance)

    def adaptative_step(h_instance):
        step_range = np.linspace(0.01, 0.05, 20)
        success = False
        for i, _step in enumerate(step_range):
            indices = np.random.choice(n_points, size=min(15, n_points), replace=False)
            noise = np.random.uniform(-_step, _step, size=len(indices))
            proposed_h = h_instance.copy()
            proposed_h[indices] += noise
            proposed_h = project_to_constraints(proposed_h)

            if compute_c5(proposed_h) < compute_c5(h_instance):
                h_instance = proposed_h
                success = True
                break
        return h_instance, success

    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 5:
        # Sample uniform
        h_uniform = np.random.uniform(0, 1, n_points)
        h_uniform = project_to_constraints(h_uniform)
        c5_uniform = compute_c5(h_uniform)
        if c5_uniform < best_c5:
            best_c5 = c5_uniform
            best_h = h_uniform

        # Fourier-based perturbation
        h_fourier = fourier_step(best_h)
        c5_fourier = compute_c5(h_fourier)
        if c5_fourier < best_c5:
            best_c5 = c5_fourier
            best_h = h_fourier

        # Adaptable local search
        best_h, _ = adaptative_step(best_h)

        if time.time() - start_time > budget_s - 10:
            break

        if verbose and (time.time() - start_time) % 30 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, Best C₅: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
