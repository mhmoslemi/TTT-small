```python
import numpy as np
import scipy.optimize as opt
import time
from cvxpy import Variable, Problem, Minimize, cp

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, initial_h_values=None):
    np.random.seed(seed)
    if initial_h_values is not None:
        h_init = initial_h_values.copy()
    else:
        h_init = np.random.rand(47)
    h_init = enforce_normalization(h_init, 47)

    n_points = 47
    h = Variable(n_points)
    C = Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    # Add initial constraints for k in [1, 2, 4, 8, 16]
    for k in [1, 2, 4, 8, 16]:
        if k < n_points:
            overlap = 0.0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i + k])
            constraints.append(overlap * (2.0 / n_points) <= C)

    objective = Minimize(C)
    prob = Problem(objective, constraints)

    time_start = time.time()
    h_best = h_init.copy()
    c5_best = compute_c5(h_best, n_points)

    while time.time() - time_start < budget_s:
        try:
            prob.solve(solver='ECOS', verbose=False, max_iters=50)
        except:
            break
        if prob.status != 'optimal':
            break
        h_val = np.array(h.value).flatten()
        h_val = enforce_normalization(h_val, n_points)
        c5_val = compute_c5(h_val, n_points)
        if c5_val < c5_best:
            h_best = h_val.copy()
            c5_best = c5_val

        max_overlap = 0.0
        worst_k = 0
        for k in range(1, n_points):
            overlap = 0.0
            for i in range(n_points - k):
                overlap += h_val[i] * (1 - h_val[i + k])
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k

        if max_overlap > c5_best + 1e-6 and worst_k < n_points:
            overlap_expr = 0.0
            for i in range(n_points - worst_k):
                overlap_expr += h[i] * (1 - h[i + worst_k])
            constraints.append(overlap_expr * (2.0 / n_points) <= C)
            prob = Problem(objective, constraints)
    return h_best, c5_best, n_points
```
