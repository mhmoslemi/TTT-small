import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1 - optimized_h
        corr = np.correlate(optimized_h, ones_minus_h, mode='full')
        corr_vals = corr[n_points - 1:n_points]
        return np.max(corr_vals) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_time = start_time
    step = np.random.rand(n_points) * 0.07 - 0.035  # Small step for perturbation

    # Enforce integral constraint before optimization
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-7:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    # Fourier setup
    h_fft = np.fft.fft(h)
    selected_indices = np.argsort(np.abs(h_fft))[::-1][:50]  # Select top Fourier components

    while time.time() - start_time < budget_s:
        h_flat = h.copy()

        # Focus on top Fourier components
        new_fft = h_fft.copy()
        perturb_indices = np.random.choice(selected_indices, min(10, len(selected_indices)), replace=False)
        perturbation = (np.random.rand(len(perturb_indices)) - 0.5) * 0.05
        new_fft[perturb_indices] += perturbation

        # Inverse FFT to time domain
        new_h = np.fft.ifft(new_fft).real
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = new_h.sum()
        if abs(new_h_sum - target_sum) > 1e-7:
            new_h *= target_sum / new_h_sum

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h[:] = new_h[:]
            best_c5 = new_c5
            best_time = time.time()
            h_fft[:] = np.fft.fft(new_h)

        if verbose and (time.time() - start_time) % 10 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

        if time.time() - best_time > 90 or best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
