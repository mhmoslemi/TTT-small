from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

        return 1.0 - x

    def constraint_lower(x):

    def is_sum_valid(h_b):
        return int(np.sum(h_b)) in {22, 23}

        new_h[idx] = 1 - new_h[idx]

            if is_sum_valid(h_b):
                return h_b

    h_b = find_valid_binary()
    best_h_b = np.copy(h_b)
    best_c5_binary = evaluate_erdos_solution(h_b, n_points)
