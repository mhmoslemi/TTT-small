import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()  # Normalize to constraint

    dx = 2.0 / n_points

    constraint_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)  # normalize

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=1000):
        h_arr = np.array(initial_h).copy()
        bounds = [(0, 1) for _ in range(n_points)]

        result = minimize(
            objective,
            h_arr,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = h_opt.clip(0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        noise = np.random.randn(len(h)) * noise_factor
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def simulated_annealing(h_init, delta=0.05, cooling_rate=0.95, max_steps=300):
        h_current = h_init.copy()
        h_current /= h_current.sum()
        h_current *= constraint_sum
        best_h = h_current.copy()
        best_obj = objective(best_h)
        temp = 100.0

        for step in range(max_steps):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_proposal = perturb_h(h_current, noise_factor=delta)
            obj_proposal = objective(h_proposal)

            if obj_proposal < best_obj or np.random.rand() < np.exp((best_obj - obj_proposal) / temp):
                h_current = h_proposal
                if obj_proposal < best_obj:
                    best_h = h_current.copy()
                    best_obj = obj_proposal

            temp *= cooling_rate
            if temp < 1e-5:
                temp = 1e-5

        return best_h, best_obj

    best_h = h_init
    best_c5 = objective(h_init)
    best_h = optimize_h(best_h)

    # Refinement with gradient descent
    optimized_best = optimize_h(best_h)
    obj_opt = objective(optimized_best)
    if obj_opt < best_c5:
        best_c5 = obj_opt
        best_h = optimized_best

    # Tabu search with multiple restarts
    tabu_size = 20
    tabu = set()
    restarts = 5

    for r in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break

        h = perturb_h(best_h, noise_factor=0.2)
        h = h.flatten()
        obj = objective(h)

        # Tabu Avoidance
        h_tuple = tuple(h.round(6).tolist())
        if h_tuple in tabu:
            continue

        tabu.add(h_tuple)
        if len(tabu) > tabu_size:
            tabu.pop()

        h = optimize_h(h)
        obj = objective(h)

        if obj < best_c5:
            best_c5 = obj
            best_h = h

    # Final simulated annealing run with gradient refinement
    final_h, final_obj = simulated_annealing(best_h, delta=0.1, max_steps=200)
    if final_obj < best_c5:
        best_h = final_h
        best_c5 = final_obj

    # Final gradient refinement
    best_h = optimize_h(best_h)
    best_c5 = objective(best_h)
    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
