import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure sum(h) == n_points / 2
    scale_factor = n_points / (2.0 * np.sum(initial_h))
    h = np.clip(initial_h * scale_factor, 0.0, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement = start_time

    def constrained_objective(C):
        # h is the vector of optimization variables
        h_var = cp.Variable(n_points)
        dx = 2.0 / n_points
        corr = np.correlate(h_var, 1 - h_var, mode='full')
        max_corr = cp.Max(corr * dx)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1,
            max_corr <= C
        ]

        problem = cp.Problem(cp.Minimize(C), constraints)
        problem.solve(solver=cp.CVXOPT)
        return problem.status, h_var.value

    # Optimization parameters
    C = 0.4  # Initial guess
    C_step = 0.0005
    min_C = best_c5
    stagnation_time = 0.0

    while time.time() - start_time < budget_s - 5:  # allow 5s for final cleanup
        status, h_sol = constrained_objective(C)
        if status == "optimal":
            h_sol = np.clip(h_sol, 0.0, 1.0)
            c5 = evaluate_erdos_solution(h_sol, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h[:] = h_sol[:]
                last_improvement = time.time()
            # Try to reduce C
            C = max(0.0, C - C_step)
        else:
            # If infeasible, increase C
            C = min(1.0, C + C_step)
            stagnation_time = time.time() - last_improvement

        if time.time() - last_improvement > 90:
            # Restart with perturbed solution to escape local minima
            h_perturbed = h.copy()
            noise = 0.02 * np.random.randn(n_points)
            h_perturbed = np.clip(h + noise, 0.0, 1.0)
            h_perturbed /= h_perturbed.sum() / (n_points / 2)
            best_h[:] = h_perturbed
            last_improvement = time.time()

    return best_h, best_c5, n_points
