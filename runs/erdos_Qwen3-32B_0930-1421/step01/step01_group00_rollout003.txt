import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = float('inf')
    best_time = time.time()

    # Normalize initial h
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h

        corr_pos = np.convolve(optimized_h, ones_minus_h, mode="valid")
        corr_neg = np.convolve(optimized_h[::-1], ones_minus_h[::-1], mode="valid")[::-1]
        corr_all = np.concatenate((corr_neg[1:], corr_pos))
        return np.max(corr_all * dx)

    def perturb(current_h, n=10, std=0.1):
        indices = np.random.choice(n_points, size=min(n, n_points), replace=False)
        noise = np.random.normal(0, std, size=len(indices))
        current_h[indices] += noise
        current_h = np.clip(current_h, 0, 1)
        # Restore sum constraint
        current_sum = current_h.sum()
        if current_sum != target_sum and abs(current_sum) > 1e-8:
            surplus = target_sum - current_sum
            current_h += surplus / n_points
            current_h = np.clip(current_h, 0, 1)
        return current_h

    alpha = 0.1  # learning rate
    patience = 0
    early_stop = 60  # 60 seconds without improvement
    batch_size = 20
    perturbation_std = 0.05

    steps = 0
    while time.time() - start_time < budget_s:
        if patience > early_stop:
            break
        steps += 1

        # Perturbation-based search
        temp_h = h.copy()
        for _ in range(batch_size):
            temp_h = perturb(temp_h, min(10, n_points), perturbation_std)
            temp_h_sum = temp_h.sum()
            if temp_h_sum != target_sum and abs(temp_h_sum) > 1e-8:
                surplus = target_sum - temp_h_sum
                adjustment = surplus / n_points
                temp_h += adjustment
                temp_h = np.clip(temp_h, 0, 1)

        # Compute improvement
        c5_current = compute_c5(h)
        c5_new = compute_c5(temp_h)
        if c5_new < c5_current:
            h[:] = temp_h
            best_c5 = c5_new
            best_h[:] = h[:]
            patience = 0
        else:
            patience += 1

        # Update learning parameters
        if steps % 10 == 0:
            perturbation_std = max(0.001, perturbation_std * 0.99)
            alpha *= 0.995

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
