import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[n_points//3 : 2*n_points//3] = 0.5 + 0.1 * np.random.rand(n_points//3)
        h /= h.sum() * 2.0 / n_points
        h *= n_points / 2
    h = np.clip(h, 0, 1)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        return np.clip(clipped * scale, 0, 1)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase: Convex optimization with L1 regularization
    max_opt_time = 0.7 * budget_s
    while time.time() - start_time < max_opt_time:
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.norm(h_var, 1) + 10 * cp.Max([correlate(h_var, 1 - h_var, mode='valid') * dx]))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break
        h_opt = normalize_h(h_var.value)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase: Gradient descent refinement
    max_grad_time = budget_s - 5  # leave 5s buffer
    learning_rates = [0.1, 0.05, 0.02, 0.01, 0.005]
    for lr in learning_rates:
        grad_time_limit = max_grad_time - time.time() + start_time
        if grad_time_limit <= 0:
            break
        count = 0
        while time.time() - start_time < max_grad_time:
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_p = best_h.copy()
                h_p[i] += eps
                h_p = normalize_h(h_p)
                c_p = compute_c5(h_p)

                h_n = best_h.copy()
                h_n[i] -= eps
                h_n = normalize_h(h_n)
                c_n = compute_c5(h_n)

                grad[i] = (c_p - c_n) / (2 * eps)
            best_h -= lr * grad
            best_h = normalize_h(best_h)
            best_c5 = compute_c5(best_h)
            count += 1
            if count >= 250:
                break

    return best_h.tolist(), best_c5, n_points
