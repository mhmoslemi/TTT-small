import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def repair_binary(h):
    # Binary repair with constraint enforcement
    b = np.round(h).clip(0, 1).copy()
    current_sum = b.sum()
    total_samples = len(b)
    shift_needed = (total_samples / 2) - current_sum
    if np.all(b == 0):
        b = np.zeros_like(b)
        idx = int(np.random.rand() * total_samples)
        b[idx] = 1.0
        b = ensure_feasible(b)
        return b
    if shift_needed > 0:
        sort_idx = np.argsort(h[:np.int32(total_samples / 2)])  # Keep highest h
        toggle_idx = sort_idx[:int(shift_needed)]
        b[toggle_idx] = np.clip(b[toggle_idx] + 1, 0, 1)
    elif shift_needed < 0:
        sort_idx = np.argsort(h[int(total_samples / 2):]) + int(total_samples / 2)  # Keep lowest h
        toggle_idx = sort_idx[:int(-shift_needed)]
        b[toggle_idx] = np.clip(b[toggle_idx] - 1, 0, 1)
    return ensure_feasible(b)


def run(budget_s=1100, n_points=42, verbose=False):
    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    h_var = cp.Variable(n_points, nonneg=True)

    # Discretization
    dx = 2.0 / n_points
    num_total_points = 2 * n_points - 1
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
    ]

    # Precompute valid shifts
    valid_shifts = []
    for k in range(n_points, num_total_points + 1):
        if k >= n_points:
            valid_shifts.append(k - n_points)

    # McCormick relaxation global variable
    x_constraints = []

    while time.time() - start_time < budget_s:
        model = cp.Problem(cp.Minimize(cp.Constant(1e99)), [])
        model = cp.Problem(cp.Minimize(cp.Variable(name="lambda")), constraints + x_constraints)
        try:
            model.solve(solver=cp.OSQP, max_iter=20000, verbose=False)
            if time.time() - start_time >= budget_s:
                break
            if model.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                break
            h_feasible = np.clip(h_var.value, 0, 1)
            current_C5 = evaluate_C5(h_feasible)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_feasible.copy()
            
            # Add cutting-plane constraint
            max_k = 0
            max_Ck = -1
            for k in valid_shifts:
                shifted = np.concatenate([np.zeros(k), h_feasible, np.zeros(k)])
                shift_h = 1 - np.roll(shifted, k)  # shifted_h = 1 - h(x + k)
                term = np.sum(h_feasible * shift_h[:len(h_feasible)])
                if term > max_Ck:
                    max_Ck = term
                    max_k = k
            lam = best_C5 / (n_points / 2)
            if max_Ck / (n_points / 2) > best_C5:
                # Correlation constraint for shift k
                shifted = np.roll(h_var, max_k)
                x = (1 - shifted) * h_var
                # McCormick relaxation for x = h * (1 - shifted)
                constraints.extend([
                    x >= 0,
                    x <= 1 - shifted,
                    x <= h_var,
                    x >= h_var + (1 - shifted) - 1,

    return best_h, best_C5, n_points
