import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def perturb_h(h):
    n = len(h)
    idxs = np.random.choice(n, size=np.random.randint(5, 11), replace=False)
    noise = np.random.normal(0, 0.05, size=len(idxs))
    h_perturbed = h.copy()
    h_perturbed[idxs] += noise
    h_perturbed = np.clip(h_perturbed, 0, 1)
    # Normalize to satisfy sum constraint
    s = np.sum(h_perturbed)
    if s != 0:
        h_perturbed = h_perturbed * (n / 2) / s
    return h_perturbed

    while np.sum(h) != n_points / 2:
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    h_current = h.copy()
    c5_current = best_c5

    T = 1.0
    alpha = 0.95
    no_improvement = 0
    max_no_improvement = 1000
    min_temp = 1e-5

    while time.time() - t0 < budget_s:
        h_new = perturb_h(h_current)
        c5_new = evaluate_overlap(h_new)
        delta = c5_new - c5_current

        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            T = max(T * 2, min_temp)
            alpha = max(alpha * 0.9, 0.1)
            no_improvement = 0

        T = max(T * alpha, min_temp)

    # Final constraint check
    s = np.sum(best_h)
    best_h = best_h * (n_points / 2) / s
    best_c5 = evaluate_overlap(best_h)
    return (best_h, best_c5, n_points)
