import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr

def are_bounds_met(h):
    return np.all((0 <= h) & (h <= 1)) and np.isclose(h.sum(), len(h) / 2, atol=1e-6)

def repair_feasibility(h):
    h = np.clip(h, 0.0, 1.0)
    excess = h.sum() - len(h) / 2
    if abs(excess) < 1e-6:
        return h
    step = 1.0 / len(h)
    h[::2] += excess
    h[h > 1.0] = 1.0
    h[h < 0.0] = 0.0
    return h

def run(budget_s=1100):
    start_time = time.time()
    n_points = 42
    if initial_h_values is not None and are_bounds_met(initial_h_values):
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        middle = n_points // 2
        h[middle - 12:middle + 12] = 0.5
        h = repair_feasibility(h)
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h)
    dx = 2.0 / n_points

    h_var = cp.Variable(n_points, nonneg=True)
    lam = cp.Variable(nonneg=True)
    cons = [
        cp.sum(h_var) == n_points / 2.0,
        h_var <= 1,
    ]

    M = [[cp.Variable(n_points, nonneg=True) for _ in range(n_points)] for _ in range(n_points)]
    for i in range(n_points):
        for j in range(n_points):
            for k in range(n_points):
                if abs(i - k) < n_points:
                    corr = cp.hstack([h_var[k + l] for l in range(n_points - abs(i - k))])
                    ones_minus = cp.hstack([1 - h_var[i + l] for l in range(n_points - abs(i - k))])
                    prod = corr * ones_minus
                    cons.append(corr <= 1)
                    cons.append(ones_minus <= 1)
                    cons.append(corr + ones_minus - 1 <= 0)
                    cons.append(corr <= 1 - M[i][j][k - i])
    obj = cp.Minimize(lam)
    problem = cp.Problem(obj, cons)
    while time.time() - start_time < budget_s:
        try:
            result = problem.solve(solver=cp.ECOS, verbose=False)
            if result is not None and np.isfinite(result):
                h_proposal = h_var.value
                h_proposal = repair_feasibility(h_proposal)
                C5_proposal = evaluate_C5(h_proposal)
                if C5_proposal < best_C5:
                    best_h = h_proposal.copy()
                    best_C5 = C5_proposal
            break
        except Exception:
            pass
    return best_h, best_C5, n_points
