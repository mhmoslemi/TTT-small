import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    n_points = len(h)
    if total < 1e-6:
        h = np.zeros(n_points)
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h


def run(budget_s=1100, n_points=60, solver=cp.ECOS):
    start_time = time.time()

        h0 = np.copy(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
        h0 = ensure_feasible(h0)

    h = cp.Variable(n_points, nonneg=True)
    ones = np.ones(n_points)

    # Construct the maximum overlap as an epigraph constraint
    k_vals = np.arange(-n_points + 1, n_points)
    corr_exprs = []

    for k in k_vals:
        if k < 0:
            h1 = h[abs(k):]
            h2 = 1.0 - h[:n_points - abs(k)]
        elif k > 0:
            h1 = h[:n_points - k]
            h2 = 1.0 - h[k:]
        else:
            h1 = h
            h2 = 1.0 - h
        corr = cp.sum(cp.multiply(h1, h2))
        corr_exprs.append(corr)

    lambda_var = cp.Variable()
    objective = cp.Minimize(lambda_var)
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1, h >= 0,
        [corr <= lambda_var for corr in corr_exprs]
    ]

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=solver, verbose=False)

    best_h = ensure_feasible(h.value)
    best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
