import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Ensure h in [0,1] and normalize to sum to target_sum
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-7:
        h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_sum = optimized_h.sum()
        if abs(optimized_sum - target_sum) > 1e-7:
            optimized_h = optimized_h * (target_sum / optimized_sum)

        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            corr = np.dot(optimized_h[:n_points - k], (1.0 - optimized_h[k:]))
            max_corr = max(max_corr, corr * dx)
        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            corr = np.dot(optimized_h[k:], (1.0 - optimized_h[:n_points - k]))
            max_corr = max(max_corr, corr * dx)
        return max_corr

    best_c5 = compute_c5(h)

    step_size = 0.01
    patience = 60  # seconds without improvement to trigger perturbation
    last_improvement = start_time

    while time.time() - start_time < budget_s:
        if time.time() - last_improvement > patience:
            # Perturb randomly to escape local minima
            indices = np.random.choice(n_points, size=min(5, n_points), replace=False)
            noise = np.random.uniform(-0.05, 0.05, size=len(indices))
            h[indices] += noise
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if abs(h_sum - target_sum) > 1e-7:
                h = h * (target_sum / h_sum)
            h = np.clip(h, 0, 1)
            last_improvement = time.time()
            best_c5 = np.inf

        # Find the worst overlap k
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = time.time()

        overlaps = []
        for k in range(1, n_points):
            if k >= h.shape[0] - 1:
                continue
            f = h[:n_points - k]
            g = (1.0 - h[k:])
            overlaps.append((np.dot(f, g), k))

        # Descending overlaps so can process worst first
        overlaps.sort(reverse=True, key=lambda t: t[0])

        # Gradient update based on worst overlap
        grad = np.zeros_like(h)
        worst_k = overlaps[0][1]
        for i in range(n_points - worst_k):
            grad[i] += h[i + worst_k]
            grad[i + worst_k] -= h[i]

        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h = h * (target_sum / h_sum)
        h = np.clip(h, 0, 1)

        # Progress
        if verbose and (time.time() - start_time) % 10 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
