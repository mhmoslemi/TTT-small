import numpy as np
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-8:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-8:
        h += (target_sum - h_sum) / len(h)
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h = optimized_h[:]
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-8:
            optimized_h *= (target_sum / optimized_h_sum)
        optimized_h = np.clip(optimized_h, 0, 1)
        ones_minus_h = 1.0 - optimized_h

        corr_values = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            corr = max(corr1, corr2)
            corr_values.append(corr * dx)
        return max(corr_values)

    while time.time() - start_time < budget_s - 5:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            best_time = time.time()

        # Compute overlap across all k
        overlaps = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            overlap1 = h[:n_points - k] * (1 - h[k:])
            overlap2 = h[k:] * (1 - h[:n_points - k])
            overlaps.append((k, np.sum(np.maximum(overlap1, overlap2)) * dx))
        overlaps.reverse()
        overlaps.sort(key=lambda x: x[1], reverse=True)

        top_k_indices = [k for k, _ in overlaps[:min(len(overlaps), 5)]]
        top_k = top_k_indices

        for k in top_k:
            if k >= n_points - 1:
                continue
            contrib1 = h[:n_points - k] * (1 - h[k:])
            contrib2 = h[k:] * (1 - h[:n_points - k])
            contrib = np.maximum(contrib1, contrib2)
            idx1 = np.argsort(contrib[:n_points - k])[-int(len(contrib[:n_points - k]) * 0.1):]
            idx2 = np.argsort(contrib[:n_points - k])[-int(len(contrib[:n_points - k]) * 0.1):] + k
            focus = np.unique(np.concatenate([idx1, idx2]))
            if focus.size == 0:
                continue

            proposal = h.copy()
            for i in focus:
                if h[i] < 1 - 1e-3:
                    proposal[i] += 0.01
                if k + i < n_points and h[k + i] > 1e-3:
                    proposal[k + i] -= 0.01
            proposal[proposal > 1] = 1.0
            proposal[proposal < 0] = 0.0
            if abs(proposal.sum() - target_sum) > 1e-8:
                proposal *= (target_sum / proposal.sum())
            proposal[proposal > 1] = 1.0
            proposal[proposal < 0] = 0.0

            new_c5 = compute_c5(proposal)
            if new_c5 < current_c5:
                h[:] = proposal[:]

        if current_c5 < 0.380876:
            break

        if time.time() - best_time > 120:
            break

    return (best_h, best_c5, n_points)
