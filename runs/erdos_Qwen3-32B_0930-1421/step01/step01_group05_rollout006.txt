import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    lambda_vals = [0.1, 0.05, 0.025, 0.0125, 0.00625, 0.003125, 0.0015625]

    for lambda_val in lambda_vals:
        if time.time() - start_time > budget_s * 0.95:
            break

        # CVXOPT setup
        n = len(h)
        h_var = cp.Variable(n, nonneg=True)
        corr_kernel = 1.0 - np.eye(n)
        corr_objective = [cp.sum(cp.multiply(h_var[i], corr_kernel[n - i]) if i < n else cp.sum(cp.multiply(h_var[i - n], corr_kernel[n - (i - n)])) for i in range(2*n - 1)]

        overlap = [cp.sum(cp.multiply(h_var[i], corr_kernel[n - i])) for i in range(n)]
        max_overlap = cp.Max(overlap)

        objective = cp.Minimize(max_overlap + lambda_val * cp.norm(h_var, 1))
        constraints = [cp.sum(h_var) == n / 2, h_var <= 1]

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False)

        if problem.status != 'optimal':
            continue

        candidate = np.clip(h_var.value, 0, 1)
        candidate = normalize_h(candidate)
        current_c5 = compute_c5(candidate)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()

    # Gradient descent refinement
    learning_rate = 0.1
    max_desc_steps = 500
    for _ in range(max_desc_steps):
        if time.time() - start_time > budget_s:
            break

        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_inc = best_h.copy()
            h_inc[i] = min(1, h_inc[i] + eps)
            h_inc = normalize_h(h_inc)
            grad_inc = compute_c5(h_inc)

            h_dec = best_h.copy()
            h_dec[i] = max(0, h_dec[i] - eps)
            h_dec = normalize_h(h_dec)
            grad_dec = compute_c5(h_dec)

            grad[i] = (grad_inc - grad_dec) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)

        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= 0.9995

    return best_h.tolist(), best_c5, n_points
