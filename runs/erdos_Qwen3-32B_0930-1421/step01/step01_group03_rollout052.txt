import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def fix_support_and_optimize(h, max_iter=100):
        h = h.reshape(n_points)
        h = h * (constraint_sum / h.sum())
        support_indices = np.where(h != 0)[0]
        active_weights = cp.Variable(len(support_indices), nonneg=True)
        targets = [h[i] for i in support_indices]
        constraints = [cp.sum(active_weights * dx) == 1.0,
                       active_weights <= 1.0,
                       active_weights >= 0.0]

        # Build bump function model
        h_model = np.zeros(n_points)
        for i, idx in enumerate(support_indices):
            h_model[idx] = active_weights[i]

        corr = cp.Conv(h_model, 1 - h_model, "full")
        c5_model = cp.max(corr * dx)
        prob = cp.Problem(cp.Minimize(c5_model), constraints)

        # Solve
        try:
            result = prob.solve(solver="ECOS", max_iters=100)
        except Exception:
            return h

        # Update model values
        if result is None:
            return h
        h_result = np.zeros(n_points)
        for i, idx in enumerate(support_indices):
            h_result[idx] = active_weights.value[i]
        return h_result

    def perturb_support(h, noise_factor=0.2):
        h = h.copy()
        # Randomly perturb h[i] or swap nearby index contributions
        swap_indices = np.random.permutation(n_points)
        i1 = swap_indices[0]
        i2 = swap_indices[1]
        temp = h[i1]
        h[i1] = h[i2]
        h[i2] = temp
        # Add in-place noise
        h += np.random.randn(n_points) * noise_factor
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    max_attempts = 100
    attempt = 0
    best_opt_h = h_init
    best_opt_c5 = best_c5

    while time.time() - start_time < budget_s and attempt < max_attempts:
        attempt += 1
        h = perturb_support(best_opt_h, noise_factor=0.1)
        h = fix_support_and_optimize(h)
        c5 = objective(h)
        if c5 < best_opt_c5:
            best_opt_h = h
            best_opt_c5 = c5

        if best_opt_c5 < best_c5:
            best_h = best_opt_h
            best_c5 = best_opt_c5

        # Reduce perturbation as needed
        if attempt % 10 == 0:
            max_attempts += 10
            budget_s -= 10

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, best_c5, n_points
