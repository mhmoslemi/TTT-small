```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

initial_h_values = None  # provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-9:
        h[0] = 1.0  # minimal feasible distribution
    else:
        h *= (len(h) / 2) / total
    return h


def construct_basis_functions(n_basis, n_points):
    x_vals = np.linspace(0, 2, n_points)
    alpha = 4.0  # Gaussian width
    beta = 0.3   # basis width

    midpoints = np.linspace(0, 2, n_basis)
    psi = []

    for x0 in midpoints:
        local_basis = np.exp(-alpha * (x_vals - x0)**2)
        local_basis[abs(x_vals - x0) > beta] = 0
        local_basis /= local_basis.sum()
        psi.append(local_basis)
    return np.array(psi), midpoints


def optimize_coefficients(psi, n_points):
    a = cp.Variable(psi.shape[0])
    constraint_int = cp.sum(a) * (2.0 / n_points) == 1.0

    # Use prior C5 evaluator implicitly with proxy correlations at symmetric lags
    max_corr_expr = []
    for k in range(-n_points + 1, n_points):
        h_cross = np.roll(psi.T, shift=k, axis=0)
        h_cross = h_cross / h_cross.shape[0]  # normalize for window alignment
        cross_corr = a @ h_cross @ (1 - a @ psi.T)
        max_corr_expr.append(cross_corr)

    objective = cp.Minimize(cp.max(max_corr_expr))
    constraints = [cp.sum(a) == n_points / 2, a >= 0, a <= 1]

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iter=1500)

    coeff = np.clip(a.value, 0, 1)
    h_opt = ensure_feasible(coeff @ psi.T)
    return h_opt


def refine_centers(h, perturbation=0.2, n_iter=5, n_basis=10, n_points=42):
    x_vals = np.linspace(0, 2, n_points)
    best_h = h
    best_C5 = evaluate_C5(h)

    for _ in range(n_iter):
        midpoints = np.linspace(0, 2, n_basis)
        perturbed = midpoints + np.random.uniform(-perturbation, perturbation, size=len(midpoints))
        perturbed = np.clip(perturbed, 0, 2)
        psi, _ = construct_basis_functions(n_basis, n_points)
        h_new = optimize_coefficients(psi, n_points)

        c5_new = evaluate_C5(h_new)
        if c5_new < best_C5:
            best_h = h_new
            best_C5 = c5_new
    return best_h, best_C5


def run(budget_s=1100, n_basis=10, n_points=42, refine_iter=4):
    start_time = time.time()
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = ensure_feasible(h)

    psi, _ = construct_basis_functions(n_basis, n_points)
    h_opt = optimize_coefficients(psi, n_points)
    h_opt, c5_bound = refine_centers(h_opt, perturbation=0.15, n_iter=refine_iter, n_basis=n_basis, n_points=n_points)

    while time.time() - start_time < budget_s and refine_iter < 8:
        h_opt, c5_bound = refine_centers(h_opt, perturbation=0.1, n_iter=1, n_basis=n_basis, n_points=n_points)
        refine_iter += 1

    h_opt = ensure_feasible(h_opt)
    return h_opt, c5_bound, n_points
```
