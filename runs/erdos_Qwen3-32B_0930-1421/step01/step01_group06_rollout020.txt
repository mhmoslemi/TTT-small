import numpy as np
import time
from scipy.optimize import minimize
import random
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Binary simulated annealing phase to find a good discrete solution
    def binary_simulated_annealing():
        # Create binary vector close to sum n/2
        h_bin = np.zeros(n_points, dtype=int)
        indices = np.random.permutation(n_points)
        target_sum = n_points // 2
        for i in range(target_sum):
            h_bin[indices[i]] = 1

        T0 = 1.0
        temp = T0
        best_h_bin = h_bin.copy()
        best_c5 = evaluate_erdos_solution(h_bin, n_points)

        iter_count = 0
        while time.time() - start_time < budget_s * 0.45:
            # Swap two elements to maintain sum constraint
            i0, i1 = np.random.choice(n_points, size=2, replace=False)
            if h_bin[i0] != h_bin[i1]:
                h_bin[[i0, i1]] = h_bin[[i1, i0]]
                new_c5 = evaluate_erdos_solution(h_bin, n_points)
                delta_c5 = new_c5 - best_c5

                if delta_c5 < 0:
                    best_h_bin[:] = h_bin.copy()
                    best_c5 = new_c5
                else:
                    if np.random.rand() < math.exp(-delta_c5 / temp):
                        # Keep bad move with probability
                        pass
                    else:
                        h_bin[:] = best_h_bin.copy()

            temp *= 0.9999
            iter_count += 1

            if iter_count % 50 == 0:
                print(f"Iters: {iter_count}, C5: {best_c5}")

        return best_h_bin, best_c5

    print("Binary simulated annealing phase...")
    best_h_bin, best_c5 = binary_simulated_annealing()
    print(f"Finished binary SA with C5: {best_c5}")

    # Continuous scaling phase
    def scale_binary_to_continuous(h_bin):
        h = np.clip(h_bin.astype(float), 0.0, 1.0)
        binary_sum = np.sum(h_bin)
        continuous_sum = n_points / 2

        if binary_sum < continuous_sum:
            # Add 0.5 to one of the zero positions
            zero_indices = np.where(h_bin == 0)[0]
            if zero_indices.size > 0:
                h[zero_indices[0]] = 0.5
        elif binary_sum > continuous_sum:
            # Change one of the ones to 0.5
            one_indices = np.where(h_bin == 1)[0]
            if one_indices.size > 0:
                h[one_indices[0]] = 0.5
        return h

    h = scale_binary_to_continuous(best_h_bin)
    best_c5 = evaluate_erdos_solution(h, n_points)

    print("Starting continuous optimization phase...")

    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def optimize_with_bounds(optimizer="Nelder-Mead", max_iter=50, callback=None):
        nonlocal h, best_c5

        cons = (
            {"type": "eq", "fun": constraint_sum},
        )
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            h,
            method=optimizer,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter},
            callback=callback
        )

        if result.success:
            new_h = result.x
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h[:] = new_h

    last_improvement_time = time.time()

    # Coarsely refine with a variety of algorithms
    while time.time() - start_time < budget_s - 20:  # leave 20s margin
        print(f"Time remaining: {budget_s - (time.time() - start_time):.1f}s")
        algo = random.choice(["L-BFGS-B", "Nelder-Mead", "SLSQP", "SLSQP"])
        max_iter = max(1, round((budget_s - (time.time() - start_time)) // 10))
        optimize_with_bounds(optimizer=algo, max_iter=max_iter)

        if time.time() - last_improvement_time > 30:
            print("No improvement in 30s, breaking early")
            break

    return h, best_c5, n_points
