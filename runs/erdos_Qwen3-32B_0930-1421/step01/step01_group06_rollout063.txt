import numpy as np
import time
from scipy.optimize import OptimizeResult
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def project_swarm(x, n_points):
    x = np.clip(x, 0, 1)
    total = np.sum(x, axis=1).reshape(-1, 1)
    x = x * (n_points / 2.0) / total
    return x

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h_initial = np.copy(initial_h_values)
    n_points = len(h_initial)
    h_initial = project_swarm(h_initial[None, :], n_points).ravel()

    num_particles = 50
    dim = n_points

    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_velocity = 0.2
    reset_fraction = 0.1
    max_iter = 0

    particles = np.zeros((num_particles, dim))
    velocities = np.zeros((num_particles, dim))
    pbest = np.zeros((num_particles, dim))
    fbest = np.full(num_particles, np.inf)

    for i in range(num_particles):
        if i < num_particles // 10:
            part = np.copy(h_initial)
        else:
            part = np.random.rand(dim)
            part = project_swarm(part[None, :], dim).ravel()
        particles[i] = part
        pbest[i] = part.copy()
        fbest[i] = evaluate_erdos_solution(pbest[i], n_points)

    gbest_idx = np.argmin(fbest)
    gbest = pbest[gbest_idx].copy()
    best_c5 = fbest[gbest_idx]
    last_improvement_time = start_time

    while time.time() - start_time < budget_s - 3:
        for it in range(max_iter if max_iter > 0 else 1):
            iter_time = time.time()
            t_ratio = (time.time() - start_time) / budget_s
            w = w_start - (w_start - w_end) * t_ratio

            r1, r2 = np.random.rand(dim), np.random.rand(dim)
            velocities = w * velocities + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest - particles)
            velocities = np.clip(velocities, -max_velocity, max_velocity)
            particles += velocities
            particles = project_swarm(particles, n_points)

            for i in range(num_particles):
                curr_c5 = evaluate_erdos_solution(particles[i], n_points)
                if curr_c5 < fbest[i]:
                    pbest[i] = particles[i].copy()
                    fbest[i] = curr_c5
                    if curr_c5 < best_c5:
                        best_c5 = curr_c5
                        gbest = particles[i].copy()
                        last_improvement_time = time.time()

            gbest_idx = np.argmin(fbest)
            gbest = pbest[gbest_idx].copy()

            # Reset a portion of the swarm to avoid stagnation
            num_resets = int(num_particles * reset_fraction)
            reset_indices = np.random.choice(num_particles, size=num_resets, replace=False)
            for idx in reset_indices:
                part = np.random.rand(dim)
                part = project_swarm(part[None, :], dim).ravel()
                particles[idx] = part
                pbest[idx] = part.copy()
                fbest[idx] = evaluate_erdos_solution(pbest[idx], n_points)

            if time.time() - last_improvement_time > 60 and it > 10:
                break

            if time.time() - start_time > budget_s - 3:
                break

    return gbest, best_c5, n_points
