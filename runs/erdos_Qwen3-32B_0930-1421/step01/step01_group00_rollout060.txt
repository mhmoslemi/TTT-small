import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    # Clip and project to ensure constraint
    def project_to_constraint(x):
        x = np.clip(x, 0, 1)
        s = x.sum()
        if abs(s - target_sum) > 1e-8:
            x *= target_sum / s
        return x

    h = project_to_constraint(h)

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    # Efficient cross-correlation-based C5 evaluation
    def evaluate_c5(h_eval):
        hw = h_eval.size
        hw_corr = scipy.signal.correlate(h_eval, 1.0 - h_eval, method='direct', mode='full')
        max_k = min(hw, 2*hw - 1)
        c5 = max(hw_corr[i] * dx for i in range(hw, max_k))  # Only shifted overlaps
        return c5

    # Gradient w.r.t h for shift k
    def gradient(h_grad, k):
        grad = np.zeros_like(h_grad)
        grad[:h_grad.size - k] -= (1.0 - h_grad[k:])
        grad[k:] -= h_grad[:h_grad.size - k]
        return grad

    step_size = 0.01
    reanneal_steps = 300  # Perturb every this many steps
    step_idx = 0

    while time.time() - start_time < budget_s:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
            best_time = time.time()

        if best_c5 <= 0.380876:
            break  # Target achieved, no further action

        if time.time() - best_time > 60:
            break  # No improvement for over a minute

        # Identify max overlap shift
        hw = h.size
        corr = scipy.signal.correlate(h, 1.0 - h, method='direct', mode='full')
        max_k = min(hw, 2*hw - 1)
        k_list = np.arange(hw, max_k)
        if len(k_list) == 0:
            break
        max_k_idx = np.argmax(corr[k_list])
        k_current = k_list[max_k_idx]

        # Compute gradient for this k
        grad = gradient(h, k_current)

        # CG-based update
        direction = grad
        # Mirror descent step
        h = h - step_size * direction
        # Project back to constraint
        h = project_to_constraint(h)

        # Perturb every so often to escape local optima
        step_idx += 1
        if step_idx % reanneal_steps == 0:
            noise = max(0.01 * (1 - 0.99 ** (step_idx // reanneal_steps)), 1e-4)
            perturbation = np.random.normal(0, noise, size=h.shape)
            h += perturbation
            h = project_to_constraint(h)

        # Dynamic step size reduction if stuck
        if (step_idx % 300 == 0 and
            best_c5 - evaluate_c5(h) < 1e-6):
            step_size *= 0.8
            h = project_to_constraint(best_h + 0.05 * (h - best_h))

    return (best_h, best_c5, n_points)
