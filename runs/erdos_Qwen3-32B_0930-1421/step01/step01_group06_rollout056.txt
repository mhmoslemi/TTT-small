import numpy as np
import time
import math
import cvxpy as cp
import cvxpy.settings as s

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h * scale_factor, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)

    # Define the overlap constraint binary search
    def is_feasible(C_5):
        nonlocal h, n_points, dx
        H = cp.Variable(n_points)
        Z = cp.Variable((n_points, n_points))

        # Define constraints
        constraints = [
            cp.sum(H) == n_points / 2,
            0 <= H,
            H <= 1,
        ]

        for i in range(n_points):
            for j in range(n_points):
                if (i - j) >= 0 and (i - j) < n_points:
                    x, y = j, i
                    constraints.extend([
                        Z[x, y] >= 0,
                        Z[x, y] <= H[x],
                        Z[x, y] <= 1 - H[y],
                        Z[x, y] >= H[x] + (1 - H[y]) - 1
                    ])

        corr = [Z[i, (i + k) % n_points] for k in range(1, n_points) for i in range(n_points - k)]
        corr = cp.concat(corr)
        corr_bound = cp.sum(corr) <= C_5 * dx * len(corr)

        constraints.append(corr_bound)

        objective = cp.Minimize(0)
        problem = cp.Problem(objective, constraints)

        result = problem.solve(solver=cp.GLPK_MI, verbose=False, maxit=1000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return H.value, 1
        return None, 0

    # Binary search over C_5
    C_low = 0.0
    C_high = best_c5
    best_found = best_h

    while (time.time() - start_time < budget_s - 10) and (C_high - C_low > 1e-6):
        C_mid = (C_low + C_high) / 2
        H_result, feasibility = is_feasible(C_mid)

        if feasibility:
            best_h = H_result
            best_c5 = C_mid
            C_high = C_mid
        else:
            C_low = C_mid

    # Warmstart with 5 random perturbations
    for _ in range(5):
        h_pert = np.clip(
            best_h + np.random.uniform(-0.1, 0.1, size=n_points),
            0, 1
        )
        total_h = np.sum(h_pert)
        h_pert *= (n_points / 2.0) / total_h

        if evaluate_erdos_solution(h_pert, n_points) < best_c5:
            C_low = 0.0
            C_high = best_c5
            while (time.time() - start_time < budget_s - 10) and (C_high - C_low > 1e-6):
                C_mid = (C_low + C_high) / 2
                H_result, feasibility = is_feasible(C_mid)

                if feasibility:
                    best_h = H_result
                    best_c5 = C_mid
                    C_high = C_mid
                else:
                    C_low = C_mid

    # Final validation and clipping
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2.0) / np.sum(best_h)

    return best_h, best_c5, n_points
