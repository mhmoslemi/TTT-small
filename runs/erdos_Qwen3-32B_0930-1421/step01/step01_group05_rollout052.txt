import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        return np.max(correlate(h, 1 - h, mode='full') * dx)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        return clipped * (n_points / 2) / sum_h if sum_h > 0 else np.zeros_like(h_in)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # CVXOPT convex optimization with L1 regularization
    h_var = cp.Variable(n_points)
    overlap = cp.fmax(correlate(h_var, 1 - h_var, mode='full'), -1e9) * dx
    lambda_reg = cp.Parameter(nonneg=True)

    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    objective = cp.Minimize(lambda_reg * cp.norm(h_var, 1) + cp.fmax(overlap, -1e9))

    problem = cp.Problem(objective, constraints)

    lambda_values = [0.1, 0.05, 0.02, 0.01, 0.005]
    for lambda_val in lambda_values:
        if time.time() - start_time >= budget_s:
            break
        lambda_reg.value = lambda_val
        try:
            problem.solve(solver=cp.CVXOPT)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_candidate = np.clip(h_var.value, 0, 1)
                h_candidate = normalize_h(h_candidate)
                current_c5 = compute_c5(h_candidate)
                if current_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = current_c5
        except Exception:
            continue

    # Refine with gradient descent post relaxation
    learning_rate = 0.1
    while time.time() - start_time < budget_s:
        grad = np.zeros(n_points)
        eps = 1e-6

        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
