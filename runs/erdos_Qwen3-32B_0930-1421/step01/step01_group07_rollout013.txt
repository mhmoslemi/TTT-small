import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def normalize_sum(h, n_points):
    total = np.sum(h)
    while abs(total - n_points / 2) > 1e-8:
        desired = n_points / 2 - total
        delta = np.clip(desired / n_points, -1, 1)
        h += delta
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
    return h

def perturb_h(h):
    n = len(h)
    num_points = np.random.randint(5, 11)
    idxs = np.random.choice(n, size=num_points, replace=False)
    noise = np.random.normal(0, 0.2, size=num_points)
    h_new = h.copy()
    h_new[idxs] += noise
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = normalize_sum(h_new, len(h))
    return h_new
