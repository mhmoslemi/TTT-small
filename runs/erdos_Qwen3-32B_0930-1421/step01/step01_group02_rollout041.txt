```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    return h


def get_binary_repair(h):
    n = len(h)
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[-n // 2:]] = 1.0
    return h_bin


def run(budget_s=1100, n_points=42):
    start_time = time.time()

    if initial_h_values is not None:
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    h = ensure_feasible(h)
    best_h = np.copy(h)
    best_C5 = evaluate_C5(h)

    active_shifts = set()
    shifts = [k for k in range(1, n_points)]

    while time.time() - start_time < budget_s and active_shifts:
        h_cp = cp.Variable(n_points, pos=True, neg=True)
        for k in active_shifts:
            slice_left = slice(None, n_points - k)
            slice_right = slice(k, n_points)
            shifted = h_cp[slice_right]
            slack = cp.Variable(n_points - k, pos=True)
            for i in range(n_points - k):
                # McCormick relaxation for 0 <= h_cp[i] <= 1, 0 <= (1 - h_cp[i + k]) <= 1
                cp.constraints.append(h_cp[i] * (1 - h_cp[i + k]) <= slack[i])
                cp.constraints.append(slack[i] >= 0)
                cp.constraints.append(slack[i] <= 1)

        λ = cp.Variable(pos=True)
        objective = cp.Minimize(λ)
        for i in range(n_points - k):
            objective += slack[i]
        constraints = [
            cp.sum(h_cp) == n_points / 2,
            h_cp <= 1,
            h_cp >= 0,
        ]
        for k in active_shifts:
            slice_left = slice(None, n_points - k)
            slice_right = slice(k, n_points)
            shifted = h_cp[slice_right]
            for i in range(n_points - k):
                constraints += [
                    slack[i] >= h_cp[i],
                    slack[i] >= h_cp[i + k],
                    slack[i] <= 1 - h_cp[i + k],
                    slack[i] <= 1 - h_cp[i] + 2 * h_cp[i + k]
                ]
            constraints += [cp.sum(slack) <= λ * n_points / 2]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.OSQP, verbose=False)

        h_candidate = np.array(h_cp.value)
        h_candidate = ensure_feasible(h_candidate)
        current_C5 = evaluate_C5(h_candidate)

        if current_C5 < best_C5:
            best_h = h_candidate
            best_C5 = current_C5

        # Cutting-plane update
        max_overlap_idx = 0
        max_corr = 0
        for shift in shifts:
            shifted = h_candidate[shift:]
            corr = np.sum(h_candidate[:-shift] * (1 - shifted))
            if corr > max_corr:
                max_overlap_idx = shift
                max_corr = corr
        active_shifts.add(max_overlap_idx)
        if max_corr > λ.value * n_points / 2:
            break  # Converged

    # Feasibility repair
    best_h = get_binary_repair(best_h)
    final_C5 = evaluate_C5(best_h)

    return best_h, final_C5, n_points
```
