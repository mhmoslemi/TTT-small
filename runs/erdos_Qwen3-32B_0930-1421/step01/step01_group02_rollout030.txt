import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def create_gaussian_basis_centered_at(x0, x_span, alpha, beta, n_points):
    x_vals = np.linspace(0, 2, n_points)
    basis = np.zeros(n_points)
    mask = np.abs(x_vals - x0) <= beta
    basis[mask] = np.exp(-alpha * (x_vals[mask] - x0) ** 2)
    basis[~mask] = 0
    return basis / np.sum(basis) * x_span


def optimize_coefficients(basis_functions, n_points, initial_flips=10):
    n_basis = len(basis_functions)
    A = np.vstack([
        np.array([np.dot(bf, bf) for bf in basis_functions]),
    ])

    for k in range(1, n_points // 2):
        overlaps = [np.dot(bf, np.roll(1 - bf, k)) for bf in basis_functions]
        mat_k = np.array(overlaps)
        A = np.column_stack((A, mat_k))

    coeff = cp.Variable(n_basis, nonneg=True)
    lambda_var = cp.Variable()

    constraints = [A[0, :] @ coeff == 1]
    for i in range(1, A.shape[0]):
        constraints.append(A[i, :] @ coeff <= lambda_var)

    objective = cp.Minimize(lambda_var)
    prob = cp.Problem(objective, constraints)

    result = prob.solve(solver=cp.SCIPY, maxIter=100)
    if result is None:
        return np.ones(n_basis) / n_basis

    optimized = np.array(coeff.value)
    optimized = np.clip(optimized, 0, 1)
    return optimized


def run(budget_s=1100, n_points=42, n_basis=10, alpha=4, beta=0.3, perturb_radius=0.1, perturb_count=3):
    start_time = time.time()
    x_vals = np.linspace(0, 2, n_points)
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h0 = initial_h_values
        # estimate centers of mass
        x_centroid = np.sum(h0 * x_vals) * 2 / n_points
        x_centers = np.linspace(x_centroid - beta * 0.5, x_centroid + beta * 0.5, n_basis)
    else:
        x_centers = np.linspace(0.1, 1.9, n_basis)

    best_h = None
    best_C5 = np.inf

    while time.time() - start_time < budget_s and len(x_centers) >= 2:
        basis_list = [
            create_gaussian_basis_centered_at(x, 2.0, alpha, beta, n_points)
            for x in x_centers
        ]
        coefficients = optimize_coefficients(basis_list, n_points)

        h = np.zeros(n_points)
        for b, a in zip(basis_list, coefficients):
            h += a * b

        h = ensure_feasible(h)
        C5 = evaluate_C5(h)
        if C5 < best_C5:
            best_h = np.copy(h)
            best_C5 = C5

        # try perturbing centers
        perturbed_centers = x_centers.copy()
        for i in range(min(n_basis, len(x_centers))):
            jitter = np.random.uniform(-perturb_radius, perturb_radius)
            perturbed_centers[i] = np.clip(
                x_centers[i] + jitter, 0.0 + beta / 2, 2.0 - beta / 2
            )

        if np.allclose(perturbed_centers, x_centers):
            break
        x_centers = perturbed_centers

    # fallback: return best found
    if best_h is None:
        # If no basis optimization ran, try a uniform distribution
        h_fallback = np.zeros(n_points)
        h_fallback[-1] = n_points / 2
        h_fallback = h_fallback / np.sum(h_fallback)
        best_h = h_fallback
        best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
