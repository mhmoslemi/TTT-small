import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1

    def compute_c5(h):
        h = np.clip(h, 0, 1).copy()
        h_sum = h.sum()
        h *= (target_sum / h_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[n_points:2*n_points].max() * dx

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def perturb(h, n_perturb=10, scale=0.05):
        proposal = h.copy()
        indices = np.random.choice(n_points, n_perturb, replace=False)
        proposal[indices] += np.random.uniform(-scale, scale, size=n_perturb)
        proposal = np.clip(proposal, 0.0, 1.0)
        h_sum = proposal.sum()
        if h_sum > 0:
            proposal *= target_sum / h_sum
        return proposal

    def optimize_step(h):
        # Use convex relaxation for refinement
        h_var = cp.Variable(n_points, nonneg=True)
        h_val = h.copy()
        h_sum = h_val.sum()
        h_var = h_var * (target_sum / h_sum)
        h_var = cp.clip(h_var, 0, 1)

        ones_minus_h = 1.0 - h_var
        max_k = 1
        obj = 0.0
        for k in range(1, n_points):
            if k >= n_points:
                break
            overlap = cp.sum(h_var[:n_points - k] * ones_minus_h[k:])
            obj += overlap
        obj = obj * dx
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)
        return np.clip(h_var.value, 0, 1)

    while time.time() - start_time < budget_s:
        h = perturb(h, n_perturb=12, scale=0.025)
        h_c5 = compute_c5(h)
        if h_c5 < best_c5:
            best_h = h.copy()
            best_c5 = h_c5
            if verbose:
                print(f"New best: {best_c5:.8f}")

        # If good enough (below target), early exit bracket
        if best_c5 < 0.380876:
            return (best_h, best_c5, n_points)

        # Intermittent convex refinement
        if np.random.rand() < 0.1:
            h = optimize_step(h)
            h_c5 = compute_c5(h)
            if h_c5 < best_c5:
                best_h = h.copy()
                best_c5 = h_c5

    return (best_h, best_c5, n_points)
