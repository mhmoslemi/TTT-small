```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    correl = correlate(h, 1 - h, mode="full")
    max_corr = np.max(correl) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def optimize_h(n_points):
    h = cp.Variable(n_points)
    dx = 2.0 / n_points

    # Initialize from initial_h_values if available
    if initial_h_values is not None:
        h_init = ensure_feasible(initial_h_values)
    else:
        h_init = np.ones(n_points) / 2
    h_init = ensure_feasible(h_init)
    
    constraints = [0 <= h, h <= 1]
    constraints.append(h.sum() == n_points / 2)

    # Objective function: minimize the maximum overlap over all shifts
    max_overlap = cp.Variable()
    x = h
    y = 1 - h
    corr = correlate(x.value, y.value, mode="full") * dx
    idxs = np.arange(len(corr))

    overlap_overall = cp.Max(cp.Conv(x, y) * dx)
    objective = cp.Minimize(overlap_overall)

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK, warm_start=True)
    if problem.status != cp.OPTIMAL:
        problem.solve(solver=cp.CVXOPT, warm_start=True)
        if problem.status != cp.OPTIMAL:
            problem.solve(solver=cp.GLPK, warm_start=True)
    opt_h = h.value
    if opt_h is None:
        return h_init, np.inf
    return ensure_feasible(opt_h), evaluate_C5(opt_h)


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        h_opt, c5 = optimize_h(n_points)
        if c5 < best_C5:
            best_h = h_opt
            best_C5 = c5
        n_points = max(20, min(n_points + 2, int((best_C5 * 100) ** 0.5)))

    return best_h, best_C5, len(best_h)
```
