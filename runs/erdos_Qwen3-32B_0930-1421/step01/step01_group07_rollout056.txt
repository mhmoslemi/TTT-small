import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_best_two_bits(h, max_indices, n_points):
    h1 = 1 - h
    min_contribution = np.inf
    best_i = -1
    best_j = -1

    # Only consider positions contributing to one of the top-active shifts.
    for m in max_indices:
        for i in np.where(h == 1)[0]:
            i_shifted = (i + m) % n_points
            if i_shifted < n_points:
                j = i_shifted
                if h1[j] == 1:
                    contribution = h[i] * h1[j]
                    if contribution < min_contribution:
                        min_contribution = contribution
                        best_i, best_j = i, j

    h_new = h.copy()
    if best_i != -1 and best_j != -1:
        h_new[best_i] = 0
        h_new[best_j] = 1
    # Enforce constraint
    count = np.sum(h_new)
    if count < n_points / 2:
        idx = np.random.choice(np.where(h_new == 0)[0])
        h_new[idx] = 1
    elif count > n_points / 2:
        idx = np.random.choice(np.where(h_new == 1)[0])
        h_new[idx] = 0
    return h_new
