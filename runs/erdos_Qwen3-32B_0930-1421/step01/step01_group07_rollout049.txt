import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_best_flip_pair(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    max_corr_idx = np.argmax(corr)

    # Identify the shift k corresponding to the peak
    k = max_corr_idx - (len(h) - 1)
    # Determine the indices of the contribution to max correlation
    left_start = max(0, -k)
    left_end = min(len(h), len(h) - k)
    right_offset = max(0, k)
    right_start = max(0, right_offset)
    right_end = min(len(h), right_offset + len(h))
    left_indices = np.where(h[left_start:left_end] == 1)[0] + left_start
    right_indices = np.where(h[right_start:right_end] == 0)[0] + right_start - k

    if len(left_indices) == 0 or len(right_indices) == 0:
        # Fallback random flip if no targeted ones exist
        indices_1 = np.where(h == 1)[0]
        indices_0 = np.where(h == 0)[0]
        flip_0 = indices_1[np.random.randint(len(indices_1))]
        flip_1 = indices_0[np.random.randint(len(indices_0))]
        return flip_0, flip_1

    flip_idx0 = np.random.choice(left_indices)
    flip_idx1 = np.random.choice(right_indices)
    return flip_idx0, flip_idx1
