def compute_c5(optimized_h, n_points, dx):
    optimized_h = np.clip(optimized_h, 0, 1)
    optimized_h_sum = optimized_h.sum()
    target_sum = n_points / 2
    if abs(optimized_h_sum - target_sum) > 1e-7:
        optimized_h *= target_sum / optimized_h_sum
    ones_minus_h = 1.0 - optimized_h
    correlations = []
    for k in range(1, n_points):
        if k >= n_points:
            continue
        corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
        corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
        correlations.append(max(corr1, corr2) * dx)
    return max(correlations)
