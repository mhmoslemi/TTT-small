import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h_init = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    # Constraint enforcement via projection
    def project_to_feasible(x):
        x = np.clip(x, 0.0, 1.0)
        s = np.sum(x)
        return x * (n_points / 2.0) / s if s != 0 else x

    # Initialize PSO parameters
    n_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0

    # Initialize particles
    particles = np.empty((n_particles, n_points))
    velocities = np.random.uniform(-0.05, 0.05, size=particles.shape)
    particles[:10] = np.array([project_to_feasible(h_init + np.random.normal(0, 0.05, n_points)) for _ in range(10)])
    particles[10:] = np.array([project_to_feasible(np.random.rand(n_points)) for _ in range(40)])

    p_best = particles.copy()
    p_best_score = np.array([evaluate_erdos_solution(p, n_points) for p in p_best])
    g_best_idx = np.argmin(p_best_score)
    g_best = particles[g_best_idx].copy()
    g_best_score = p_best_score[g_best_idx]

    best_h = g_best.copy()
    best_c5 = g_best_score
    last_improvement_time = start_time

    def evaluate(p): return evaluate_erdos_solution(p, n_points)

    # PSO loop
    while time.time() - start_time < budget_s:
        w_current = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
        w_current = max(w_end, min(w_current, w_start))

        for i in range(n_particles):
            r1, r2 = np.random.rand(2)

            # Velocity and position update
            velocities[i] = w_current * velocities[i] + \
                            c1 * r1 * (p_best[i] - particles[i]) + \
                            c2 * r2 * (g_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i])

        # Update personal and global bests
        current_scores = np.array([evaluate(p) for p in particles])
        idx_improved = current_scores < p_best_score
        p_best[idx_improved] = particles[idx_improved]
        p_best_score[idx_improved] = current_scores[idx_improved]
        new_global = np.argmin(p_best_score)
        if p_best_score[new_global] < g_best_score:
            g_best_score = p_best_score[new_global]
            g_best = particles[new_global].copy()
            if g_best_score < best_c5:
                best_c5 = g_best_score
                best_h[:] = g_best.copy()
                last_improvement_time = time.time()

        # Avoid stagnation by resetting some particles
        if time.time() - last_improvement_time > 30:
            for i in range(int(n_particles * 0.1)):
                particles[i] = project_to_feasible(np.random.rand(n_points))

    return best_h, best_c5, n_points
