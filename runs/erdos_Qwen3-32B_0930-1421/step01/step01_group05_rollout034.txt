import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import warnings

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as a starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    iterations = 0

    while time.time() - start_time < budget_s:
        seed += 1
        np.random.seed(seed)
        # Random exploration phase
        h = np.random.rand(n_points)
        h = normalize_h(h)

        # Improve via gradient descent
        h = improve_with_gradient(h, budget_s - (time.time() - start_time), 0.1)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Ensure we're making progress
        iterations += 1
        if iterations % 50 == 0:
            print(f"Iter: {iterations}, Best C₅: {best_c5:.6f}")

    return best_h.tolist(), best_c5, n_points


def improve_with_gradient(h, remaining_budget_s, learning_rate):
    start_time = time.time()
    n_points = len(h)

    while time.time() - start_time < remaining_budget_s:
        grad = np.zeros(n_points)
        eps = 1e-5

        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_overlap(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_overlap(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        h -= learning_rate * grad
        h = normalize_h(h)

        if time.time() - start_time > remaining_budget_s:
            break

        learning_rate *= 0.999

    return h

def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    max_c = -np.inf

    for shift in range(1, n):
        overlap = np.sum(h[:n - shift] * (1 - h[shift:])) * dx
        if overlap > max_c:
            max_c = overlap

        # Due to circular convolution symmetry, we can compute the reverse shift as well
        overlap = np.sum(h[shift:] * (1 - h[:n - shift])) * dx
        if overlap > max_c:
            max_c = overlap

    return max_c
