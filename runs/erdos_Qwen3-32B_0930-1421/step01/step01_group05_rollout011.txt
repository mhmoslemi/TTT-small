import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Use initial_h_values as known starting point
    if 'initial_h_values' in globals():
        h_pop = [np.array(initial_h_values, dtype=np.float64).copy()]
        n_points = len(h_pop[0])
    else:
        n_points = 45
        h_pop = [np.random.rand(n_points) for _ in range(1)]

    # Scale population to satisfy ∑h = n_points/2
    def normalize_population(h_list):
        for h in h_list:
            h[:] = np.clip(h, 0, 1)
            s = np.sum(h)
            h[:] *= (n_points / 2) / max(s, 1e-12)

    normalize_population(h_pop)

    # Construct remaining fireflies from Gaussian perturbations
    while len(h_pop) < 50:
        perturbation = np.clip(
            h_pop[0] + np.random.normal(0, 0.05, n_points),
            0, 1
        ).copy()
        normalize_population([perturbation])
        h_pop.append(perturbation)

    # Frequency scaling
    dx = 2.0 / n_points

    # Evaluate C5 value for a firefly
    def eval_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    # Measure firefly distances
    def distance(h1, h2):
        return np.linalg.norm(h1 - h2)

    # Fix feasibility of one firefly
    def repair(firefly):
        firefly[:] = np.clip(firefly, 0, 1)
        s = np.sum(firefly)
        if abs(s) > 1e-12:
            firefly[:] *= (n_points / 2) / s

    best_h = h_pop[0].copy()
    best_c5 = eval_c5(best_h)

    # Main firefly optimization loop
    t_total = 10000  # Moderate number of iterations
    beta0 = 1.0
    gamma = 0.01
    alpha = 0.5
    movement_scale = 2.0

    last_check_time = start_time
    progress_check_period = 60  # seconds between progress report

    for t in range(t_total):
        # Budget safety check
        if time.time() - start_time >= budget_s:
            break

        # Sort population by brightness (C5 value)
        h_pop.sort(key=lambda h: eval_c5(h))
        for i in range(len(h_pop)):
            h_i = h_pop[i]
            if np.abs(np.sum(h_i) - n_points / 2) > 1e-12:
                repair(h_i)

            for j in range(i):
                h_j = h_pop[j]
                if eval_c5(h_j) < eval_c5(h_i):
                    r = distance(h_i, h_j)
                    beta = beta0 * np.exp(-gamma * r**2)

                    # Attraction and random move
                    h_i += beta * (h_j - h_i) + alpha * np.random.normal(0, movement_scale, n_points)
                    h_i = np.clip(h_i, 0, 1)

                    # Repair and evaluate
                    repair(h_i)
                    curr_c5 = eval_c5(h_i)

                    # Update best if improved
                    if curr_c5 < best_c5:
                        best_h[:] = h_i.copy()
                        best_c5 = curr_c5

        # Progress tracking
        if time.time() - last_check_time >= progress_check_period:
            print(f"Iteration={t}, Current best C5={best_c5:.6f}, Remaining time={budget_s - (time.time() - start_time):.1f}s")
            last_check_time = time.time()

        # Reduce step size over time
        gamma = t / 100
        movement_scale *= 0.999

    print(f"Optimization complete. Best C5 found: {best_c5}")

    return best_h.tolist(), best_c5, n_points
