import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            corrs.append(corr)
        if n_points - k > 0:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)
    return np.max(np.array(corrs)) * dx if corrs else 0.0

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2.0) / total
    return np.clip(h, 0.0, 1.0)

def mutate(h, sigma=0.05):
    h_new = h + np.random.normal(0, sigma, len(h))
    return enforce_normalization(h_new, len(h))

def crossover(h1, h2):
    alpha = np.random.uniform(0, 1)
    return enforce_normalization(alpha * h1 + (1 - alpha) * h2, len(h1))

def fitness(h, n_points):
    return compute_c5(h, n_points)

def genetic_algorithm(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    n_points = len(init_h)
    population_size = 40
    generations = 0
    mutation_scale = 0.1
    num_elites = 20

    # Initialize population
    if init_h is None:
        pop = [np.random.rand(n_points) for _ in range(population_size)]
        for i in range(population_size):
            pop[i] = enforce_normalization(pop[i], n_points)
    else:
        pop = [init_h.copy()]  # Start from initial_h
        for _ in range(population_size - 1):
            p = np.random.rand(n_points)
            p = enforce_normalization(p, n_points)
            pop.append(p)

    time_start = time.time()
    while time.time() - time_start < budget_s - 0.1:
        # Evaluate fitness for each individual
        fitness_values = np.array([fitness(ind, n_points) for ind in pop])
        best_index = np.argmin(fitness_values)
        best_individual = pop[best_index]
        best_fitness = fitness_values[best_index]

        if generations % 20 == 0:
            print(f"Generation {generations}, C₅ bound: {best_fitness:.6f}")

        # Keep the top num_elites individuals as elites
        idx = np.argsort(fitness_values)
        elites = [pop[i].copy() for i in idx[:num_elites]]

        # Fill population with new individuals
        while len(pop) < population_size:
            # Selection (tournament)
            idx1 = np.random.randint(len(elites))
            idx2 = np.random.randint(len(elites))
            if fitness_values[idx1] < fitness_values[idx2]:
                parent1 = elites[idx1].copy()
            else:
                parent1 = elites[idx2].copy()

            idx1, idx2 = np.random.randint(len(elites), size=2)
            if fitness_values[idx1] < fitness_values[idx2]:
                parent2 = elites[idx1].copy()
            else:
                parent2 = elites[idx2].copy()

            # Crossover
            child = crossover(parent1, parent2)

            # Mutation
            if np.random.rand() < 0.2:
                child = mutate(child, sigma=mutation_scale)

            pop.append(child)
        pop = pop[-population_size:]

        # Decay mutation rate
        mutation_scale *= 0.995
        generations += 1

    # Final local optimization step
    def objective(x):
        h = x.reshape(-1)
        return fitness(h, n_points)

    result = opt.minimize(objective, best_individual, bounds=[(0, 1)] * n_points, method='L-BFGS-B', options={'maxiter': 100})
    final_h = enforce_normalization(result.x, n_points)
    final_c5 = fitness(final_h, n_points)

    return final_h, final_c5, n_points

def run(seed=42, budget_s=1100, init_h=None):
    if init_h is None:
        return None, np.inf, 0
    return genetic_algorithm(seed=seed, budget_s=budget_s, init_h=init_h)
