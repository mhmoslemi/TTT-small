import numpy as np
import time
import scipy.optimize as opt
from scipy.signal import correlate
import cvxpy as cp


def rescale_h(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def compute_C5(h, dx):
    h = h.astype(float)
    h = rescale_h(h)
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def segment_flip(h, segment_size=0.1):
    n = len(h)
    start = np.random.randint(n)
    length = np.random.randint(1, max(2, int(segment_size * n)))
    end = min(start + length, n)
    h[start:end] = 1 - h[start:end]
    return h


def segment_noise(h, noise_level=0.1):
    n = len(h)
    start = np.random.randint(n)
    length = np.random.randint(1, max(2, int(0.1 * n)))
    end = min(start + length, n)
    h[start:end] += np.random.normal(loc=0.0, scale=noise_level, size=end - start)
    return h

    np.random.seed(seed)

        h[70:] = 0.2
        h = rescale_h(h)

    best_h = h.copy()
    best_C5 = compute_C5(best_h, dx)
    best_time = time.time()
    amp = 0.1
    step = 0
    amp_decay = 0.8
    amp_decay_every = 100
    segment_flip_freq = 80

    while time.time() - best_time < budget_s:
        candidate_h = best_h.copy()
        if step % segment_flip_freq == 0:
            candidate_h = segment_flip(candidate_h)
        else:
            candidate_h = segment_noise(candidate_h, noise_level=amp)

        candidate_C5 = compute_C5(candidate_h, dx)

            best_C5 = candidate_C5
            best_h = candidate_h.copy()

        if step % amp_decay_every == 0:
            amp *= amp_decay

        step += 1

    return best_h, best_C5, n_points
