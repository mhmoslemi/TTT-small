import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        def _compute_overlap(h1, h2):
            return correlate(h1, 1 - h2, mode='full') * dx
        return np.max(_compute_overlap(h, h))

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        scaled = clipped * (n_points / 2) / total
        return np.clip(scaled, 0, 1)

    # Initialize firefly population
    n_fireflies = 50
    fireflies = []
    if 'initial_h_values' in globals():
        fireflies.append(np.array(initial_h_values, dtype=np.float64))
        for _ in range(n_fireflies - 1):
            pert = np.random.normal(0, 0.05, n_points)
            proposed = normalize_h(fireflies[0] + pert)
            fireflies.append(proposed)
    else:
        for _ in range(n_fireflies):
            h_tmp = np.random.rand(n_points)
            h_tmp = h_tmp / h_tmp.sum() * n_points / 2
            fireflies.append(h_tmp)

    fireflies = np.array(fireflies)
    best_h = None
    best_c5 = float('inf')

    T = 1000 # max iterations
    max_budget_ms = budget_s * 1000
    gamma_start = 0.001
    gamma_end = 0.1
    alpha = 0.2
    beta_0 = 1.0

    # Simulation loop
    for t in range(T):
        if time.time() - start_time >= budget_s:
            break

        for i in range(n_fireflies):
            current = fireflies[i]
            current_c5 = compute_c5(current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current.copy()

            # Only attract if firefly is not already the brightest
            for j in range(n_fireflies):
                if i == j:
                    continue
                other = fireflies[j]
                other_c5 = compute_c5(other)
                if other_c5 >= current_c5:
                    continue

                # Compute Euclidean distance
                r = np.linalg.norm(current - other)
                if r < 1e-10:
                    continue

                # Time-dependent gamma
                gamma = gamma_start + t * (gamma_end - gamma_start) / T
                beta = beta_0 * np.exp(-gamma * r**2)

                # Move firefly i toward j
                move = beta * (other - current) + alpha * (np.random.rand(n_points) - 0.5)
                proposed = current + move
                proposed = normalize_h(proposed)

                # Update firefly position
                fireflies[i] = proposed

    if best_h is None:
        best_h = fireflies[0]
        best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, n_points
