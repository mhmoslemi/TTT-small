import numpy as np
import time
import cvxpy as cp
import scipy.signal


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h0 = initial_h_values.copy()
        if len(h0) != n_points:
            h0 = np.resize(h0, n_points)
    except NameError:
        h0 = np.zeros(n_points)
        h0[:5] = 0.1
        h0[5:42] = 0.3
        h0[42:] = 0.2
        h0 /= h0.sum() * 2 / n_points

    best_h = h0.copy()
    best_C5 = np.inf
    start_time = time.time()
    max_cvx_time = 80

    idx1 = np.arange(n_points).astype(np.int32)
    idx2 = (np.arange(2 * n_points - 1) - n_points + 1).astype(np.int32)
    overlaps = np.zeros(2 * n_points - 1, dtype=np.float64)

    for k in range(2 * n_points - 1):
        shift = idx2[k]
        valid = (idx1 + shift) >= 0
        valid &= (idx1 + shift) < n_points
        overlaps[k] = np.sum(h0[np.where(valid)] * (1 - h0[np.where(idx1 + shift < n_points)]))

    max_overlap = np.max(overlaps)
    best_C5 = max_overlap * dx

    def eval_C5_cvx(h, dx):
        h = h.reshape(-1)
        for k in range(2 * n_points - 1):
            shift = idx2[k]
            mask = np.zeros_like(h)
            mask += (idx1 + shift) >= 0
            mask &= (idx1 + shift) < n_points
            val = np.sum(h * (1 - np.roll(h, -shift)) * mask)
            overlaps[k] = val
        res = np.max(overlaps)
        return res * dx

    while time.time() - start_time < budget_s and max_cvx_time > 0:
        conv_h = cp.Variable(n_points)
        constraints = [
            cp.sum(conv_h) == n_points / 2,
            conv_h <= 1,
            conv_h >= 0
        ]

        obj = 0
        lambda_ = 0.1
        mu = 0.01
        for k in range(2 * n_points - 1):
            shift = idx2[k]
            valid1 = (idx1 + shift) >= 0
            valid2 = (idx1 + shift) < n_points
            valid = valid1 * valid2
            overlap = cp.sum(conv_h * (1 - cp.roll(conv_h, -shift)) * valid)
            obj += overlap

        obj = lambda_ * obj
        obj += mu * cp.sum_squares(conv_h)

        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.OSQP, max_iter=5000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                hprop = conv_h.value
                hprop = np.clip(hprop, 0, 1)
                hprop = hprop / hprop.sum() * n_points / 2
                C5 = eval_C5_cvx(hprop, dx)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = hprop.copy()
                max_cvx_time -= prob.solver_stats.solve_time
            else:
                max_cvx_time -= 30  # Fallback if not solved in time
        except:
            max_cvx_time -= 10  # Fallback on errors

    if best_C5 >= 0.381:
        return best_h, best_C5, n_points
    return best_h, best_C5, n_points
