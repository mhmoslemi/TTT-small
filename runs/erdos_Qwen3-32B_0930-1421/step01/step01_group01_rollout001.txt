import numpy as np
import time
import cvxpy as cp
import scipy.signal


def get_initial_h(n_points):
    try:
        return initial_h_values.copy()[:n_points] if len(initial_h_values) >= n_points else np.concatenate([initial_h_values, np.zeros(n_points - len(initial_h_values))])
    except:
        return np.zeros(n_points)


def evaluate_C5(h, dx):
    h = np.clip(h.copy(), 0, 1)
    h /= h.sum() * dx * 2  # Ensure integral constraint
    cor = scipy.signal.correlate(h, 1 - h, mode='full')
    return np.max(cor) * dx


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = get_initial_h(n_points)
    except:
        h = np.zeros(n_points)

    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)

    def cvx_optimize(h0, reg=0.01):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints += [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(
            reg * cp.sum_squares(h) +
            cp.max([
                cp.sum([h[i] * (1 - h[j]) for i, j in enumerate((i - k) % n_points for i in range(n_points))])
                for k in range(1, n_points - 1)
            ])
        )
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='ECOS', max_iters=150, verbose=False)
        except Exception:
            return h0
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h0
        return np.clip(np.array(h.value).flatten(), 0, 1)

    start_time = time.time()
    while time.time() - start_time < budget_s * 0.9:
        h = cvx_optimize(h)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h /= h_sum * dx * 2  # Re-normalize
        c5 = evaluate_C5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h

        # Perturb and restart
        h += 0.1 * (np.random.rand(n_points) - 0.5)
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx * 2
        h = cvx_optimize(h)

    return best_h, best_c5, n_points
