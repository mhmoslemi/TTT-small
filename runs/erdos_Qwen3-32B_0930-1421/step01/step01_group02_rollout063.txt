import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None  # Defined at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def generate_basis(x_centers, alpha=4, beta=0.3, n_points=42, dx_scale=None):
    """Construct basis functions centered at x_centers as array indices."""
    if dx_scale is None:
        dx_scale = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    basis = np.zeros((n_points, len(x_centers)))
    for i, xc in enumerate(x_centers):
        i_center = int(np.round(xc / dx_scale))
        left = max(0, i_center - int(np.ceil(beta / dx_scale)))
        right = min(n_points, i_center + int(np.ceil(beta / dx_scale)) + 1)
        x_slice = x[left:right]
        mask = (x_slice >= xc - beta) & (x_slice <= xc + beta)
        x_sub = x_slice[mask]
        basis_slice = np.exp(-alpha * (x_sub - xc)**2)
        basis_slice = (basis_slice - basis_slice.min()) / (basis_slice.max() - basis_slice.min())
        basis[left:right, i][mask] = basis_slice
    return basis


def optimize_coefficients(basis, y_integral=1.0, x_centers=None, dx_scale=None):
    """Optimize basis coefficients to minimize max correlation."""
    n_basis = basis.shape[1]
    a = cp.Variable(n_basis)
    # Normalize ∫h dx = 1
    integral_contribution = cp.sum(a * cp.sum(basis, axis=0)) * dx_scale
    # Logistic-style feasibility bounds
    constraint0 = [0 <= a, a <= 1]
    constraint1 = [integral_contribution == y_integral]
    constraints = constraint0 + constraint1

    # Precompute all cross-correlation matrices for k offsets
    # To reduce computational load, we sample fewer shifts heuristically
    n_points = basis.shape[0]
    dx = 2.0 / n_points
    shifts = np.array([-10, -3, -2, -1, 0, 1, 2, 3, 10])
    # Ensure max offset is ±n_points
    valid_shifts = shifts[(shifts >= -n_points) & (shifts <= n_points)]
    corr_at_shift = []
    for k in valid_shifts:
        shift = int(k)
        shifted_basis = np.roll(basis, shift, axis=0).copy()
        shifted_basis[:shift] = 0 if shift > 0 else shifted_basis[:shift]
        corr = basis * (1 - shifted_basis)
        corr_int = (np.sum(corr, axis=0) * dx)
        corr_at_shift.append(corr_int)

    corr_obj = 0
    for corr_int in corr_at_shift:
        corr_obj += cp.norm(a * corr_int, p=np.inf)

    lambd = cp.Variable()
    constraint_lambd = [a * corr_int <= lambd for corr_int in corr_at_shift]
    constraints += constraint_lambd
    prob = cp.Problem(cp.Minimize(lambd), constraints)
    
    prob.solve(solver=cp.ECOS, max_iters=2000)
    if a.value is None:
        return None
    return a.value


def construct_from_basis(x_centers, n_points=42, alpha=4, beta=0.3, dx_scale=None):
    """Build a feasible h using optimized basis coefficients."""
    if dx_scale is None:
        dx_scale = 2.0 / n_points
    basis = generate_basis(x_centers, alpha, beta, n_points, dx_scale)
    a_vals = optimize_coefficients(basis, y_integral=1.0, x_centers=x_centers, dx_scale=dx_scale)
    if a_vals is None:
        return None
    h = np.sum(basis * a_vals, axis=1)
    h = np.clip(h, 0, 1)
    total = h.sum() * dx_scale
    if total == 0:
        h[0] = 1.0
        h[1:] = 0.0
    else:
        h *= 1.0 / total
    return h


def perturb_centers(x_centers, delta=0.1, n_points=42, dx_scale=None):
    """Perturb center positions slightly to explore nearby solutions."""
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    xp = x_centers + delta * np.random.randn(len(x_centers))
    xp = np.clip(xp, 0, 2)
    return np.unique(np.round(xp / dx) * dx)


def run(budget_s=1100, n_points=42, retries=15, adaptive_centers=True):
    dx = 2.0 / n_points
    total_time = 0.0
    best_h = None
    best_C5 = np.inf
    start_time = time.time()

    # Initialize with parent if available
    initial_used = initial_h_values is not None and np.all(np.isfinite(initial_h_values))

    if initial_used:
        # Compute center of mass from initial distribution
        x = np.linspace(0, 2, n_points)
        cx = np.sum(initial_h_values * x) / np.sum(initial_h_values)
        base_centers = np.round((cx - 0.5 * dx) / dx) * dx
        base_centers = np.concatenate([base_centers - 0.6 * dx, base_centers, base_centers + 0.6 * dx])
        base_centers = np.clip(base_centers, 0, 2)
    else:
        # Otherwise use uniform basis centers
        base_centers = np.arange(0, 2 + dx, dx)[::3][0:6]  # Space out the centers

    for i in range(retries):
        if i > 0 or not initial_used:
            perturbed_centers = perturb_centers(base_centers, delta=0.3 if i == 0 else 0.1, dx_scale=dx)
        else:
            perturbed_centers = base_centers
        h = construct_from_basis(perturbed_centers, n_points=n_points, dx_scale=dx)
        if h is None:
            continue
        end_time = time.time()
        elapsed = end_time - start_time
        total_time = elapsed
        if total_time >= budget_s - 1.0:
            break
        h = np.clip(h, 0, 1)
        ch = np.sum(h) * dx
        if abs(ch - 1.0) > 1e-4:
            h = np.clip(h, 0, 1)
            h *= 1 / np.sum(h) if np.sum(h) > 0 else 1.0 / n_points
        C5 = evaluate_C5(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
            print(f"Improved C5 to {best_C5:.6f} at retry {i}")

    return best_h, best_C5, n_points
