import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if defined
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h = np.linspace(0, 1, n_points) * 0.8
        h[-10:] = 1
        h = h / h.sum() * n_points / 2

    h_var = cp.Variable(n_points, nonneg=True)
    obj energies = []

    best_h = h.copy()
    best_C5 = np.inf
    start_time = time.time()

    def solve():
        nonlocal best_h, best_C5

        offset_all = np.arange(-n_points + 1, n_points)
        offset_masks = []
        for k in offset_all:
            valid = np.arange(n_points)
            if k < 0:
                valid_k = valid[-k:]
                valid = valid[:len(valid_k)]
            elif k >= 0:
                valid_k = valid[:n_points - k]
                valid = valid[:len(valid_k)]
            corr_mask = np.zeros((len(valid), n_points))
            corr_mask[np.arange(len(valid)), valid] = 1
            overlap = (h_var * (1 - h_var[k:] if k > 0 else 1 - h_var[:n_points - k]))[valid]
            obj += cp.sum(overlap)

        energy = 0.01 * cp.sum_squares(h_var)
        constraints = [
            np.all(h_var >= 0),
            np.all(h_var <= 1),
            cp.sum(h_var) == n_points / 2
        ]

        problem = cp.Problem(cp.Minimize(energy), constraints)
        result = problem.solve(solver=cp.SCIPY, verbose=False, max_iter=500)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = np.clip(h_var.value, 0., 1.)
            C5_new = evaluate_C5(h_new, dx)[0]
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
            return C5_new
        return np.inf

    solve_result = solve()

    n_iter = 0
    solution_time = 0.0
    while solution_time < budget_s:
        test_h = best_h + 0.1 * np.random.randn(n_points)
        test_h = np.clip(test_h, 0, 1)
        test_h = rescale_h(test_h)
        x = evaluate_C5(test_h, dx)[0]
        if x < best_C5:
            best_h = test_h
            best_C5 = x

        n_iter += 1
        solution_time = time.time() - start_time

        if solution_time >= budget_s - 10:
            break

    return best_h, best_C5, n_points


def evaluate_C5(h, dx):
    """Evaluate the current C5 bound given a solution vector h."""
    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)
    h /= h.sum()
    h *= len(h) / 2.0
    cor = correlate(h, 1 - h, mode="full")
    c5 = np.max(cor) * dx
    return c5, h


def rescale_h(h):
    h = np.clip(h, 0.0, 1.0)
    h = h / h.sum() * len(h) / 2.0
    return h
