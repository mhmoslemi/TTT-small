import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    max_n_points = 100
    population_size = 10
    crossovers = 2
    gradient_steps = 500
    mutation_factor = 0.15
    patience = 20

    # Use initial_h_values from parent
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        base_h = np.array(initial_h_values)
    else:
        n_points = 47
        base_h = np.zeros(n_points)
        for i in range(n_points // 2):
            base_h[i * 2] = 2.0 / n_points
        base_h *= n_points / 2 / base_h.sum()

    dx = 2.0 / n_points

    # Constraint normalization
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        return h

    def perturb(h, factor=0.1):
        h = h + np.random.randn(n_points) * factor
        h = repair(h)
        return h

    def crossover(h1, h2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, h1, h2)
        return repair(child)

    def optimize(h, steps=500):
        h = h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': steps, 'ftol': 1e-8}
        )
        return repair(res.x)

    # Initialize population with perturbed copies
    population = [base_h]
    if 'initial_h_values' in globals():
        for i in range(1, population_size):
            mut = perturb(base_h, mutation_factor)
            population.append(mut)
    else:
        for i in range(population_size - 1):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, n_points // 4, replace=False)
            h[indices] = 1.0
            h = repair(h)
            h = perturb(h, mutation_factor)
            population.append(h)

    # Evaluate and keep best-so-far
    best_score = np.inf
    best_h = None

    # Main loop
    no_improvement = 0

    while time.time() - start_time < budget_s * 0.95:
        # Evaluate all population
        scores = [objective(ind) for ind in population]
        population = [p for _, p in sorted(zip(scores, population))]
        scores.sort()

        new_population = population[:crossovers]

        # Generate crossovers
        for i in range(2):
            parent1 = population[np.random.randint(1, crossovers)]
            parent2 = population[np.random.randint(crossovers, len(population))]
            child = crossover(parent1, parent2)
            child = optimize(child, gradient_steps)
            scores = [objective(h) for h in [child]]
            if scores[0] < best_score:
                best_score = scores[0]
                best_h = child
                no_improvement = 0
            new_population.append(child)

        # Introduce mutations
        for i in range(2):
            parent = population[np.random.randint(crossovers)]
            child = perturb(parent, mutation_factor)
            child = optimize(child, gradient_steps)
            scores = [objective(h) for h in [child]]
            if scores[0] < best_score:
                best_score = scores[0]
                best_h = child
                no_improvement = 0
            new_population.append(child)

        # Replace old population
        population = new_population
        no_improvement += 1

        if no_improvement > patience:
            break

    # Final optimization on best
    if best_h is None:
        best_h = optimize(population[0], gradient_steps * 2)
    else:
        best_h = optimize(best_h, gradient_steps * 2)

    best_h = repair(best_h)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
