import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits_custom(h, peak_shift, peak_index, n_points):
    h = h.copy()
    # Identify flip positions near the peak overlap
    i = peak_index % n_points
    j = (i - peak_shift + n_points) % n_points
    # Flip a 1 to 0 near the overlap peak
    indices_1 = np.where(h == 1)[0]
    indices_1 = indices_1[np.argsort(np.abs(indices_1 - i))]
    idx_flip1 = indices_1[0]
    # Flip a 0 to 1 near the mirrored position
    indices_0 = np.where(h == 0)[0]
    indices_0 = indices_0[np.argsort(np.abs(indices_0 - j))]
    idx_flip2 = indices_0[0]
    h[idx_flip1] = 0
    h[idx_flip2] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):

    # Enforce sum constraint
    while np.sum(h) != n_points / 2:
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5

    while time.time() - t0 < budget_s:
        corrs = correlate(current_h, 1 - current_h, mode='full')
        max_corr = np.max(corrs)
        # Find first peak and its shift
        peak_index = np.where(corrs == max_corr)[0][0]
        peak_shift = peak_index - (len(current_h) - 1)

        h_new = flip_two_bits_custom(current_h, peak_shift, peak_index, n_points)
        # Check feasibility
        if not np.all((h_new >= 0) & (h_new <= 1)):
            continue
        if np.isclose(np.sum(h_new), n_points / 2):
            new_c5 = evaluate_overlap(h_new)
            if new_c5 < current_c5:
                current_h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
