import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    max_corr = 0
    h_full = np.tile(h, 4)
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        max_corr = max(max_corr, corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        max_corr = max(max_corr, corr)
    return max_corr * dx

def normalize_h(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    return h * (n_points / 2) / total

def mutate(h, sigma):
    h_mut = h + sigma * np.random.randn(len(h))
    return normalize_h(h_mut, len(h))

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return normalize_h(child, len(child))

def evaluate_population(population, n_points):
    return np.array([compute_c5(h, n_points) for h in population])

        fitness = evaluate_population(population, n_points)

            parent = np.random.choice(fittest)
            child = mutate(parent, sigma)

        sigma *= 0.99

    best_h = min(population, key=lambda h: compute_c5(h, n_points))
