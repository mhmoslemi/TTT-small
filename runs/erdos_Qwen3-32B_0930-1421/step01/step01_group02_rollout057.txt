```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h = (h * (len(h) / 2)) / total
    return h


def repair_to_binary(h, n_points):
    h_bin = h.copy()
    h_bin[h_bin > 0.5] = 1.0
    h_bin[h_bin <= 0.5] = 0.0
    total = h_bin.sum()
    if abs(total - n_points / 2) > 1e-6:
        diff = int(n_points / 2) - int(total)
        if diff > 0:
            # Need to activate more points
            idx = np.argsort(h)[::-1]
        else:
            # Need to deactivate some points
            idx = np.argsort(h)
        for i in idx:
            if diff > 0 and h_bin[i] == 0:
                h_bin[i] = 1
                diff -= 1
                if diff == 0:
                    break
            elif diff < 0 and h_bin[i] == 1:
                h_bin[i] = 0
                diff += 1
                if diff == 0:
                    break
    return h_bin


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    max_iter = 1000  # Limit number of cutting-plane iterations

    # Feasible initialization, fallback to random if no initial construction
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
    h = ensure_feasible(h)

    for iter_idx in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s:
            break

        # Evaluate current h
        C5 = evaluate_C5(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = np.copy(h)
            print(f"Iter {iter_idx}: C₅ = {best_C5:.6f}")

        # Create convex variables
        h_vars = cp.Variable(n_points, nonneg=True)
        lambda_vars = cp.Variable(nonneg=True)
        problem = cp.Problem(cp.Minimize(lambda_vars), [])

        # Add bounds on h_vars and sum constraint
        problem.constraints += [
            h_vars <= 1,
            h_vars >= 0,
            cp.sum(h_vars) == n_points / 2
        ]

        # Compute shifts for overlaps (all shifts from -n_points to n_points)
        shifts = np.arange(1 - n_points, n_points)
        for k in shifts:
            # Skip zero shift since h(x)(1 - h(x)) = 0
            if k == 0:
                continue

            # Extract valid indices for this shift
            valid_i = []
            for i in range(n_points):
                i_k = i + k
                if 0 <= i_k < n_points:
                    valid_i.append(i)

            if not valid_i:
                continue

            # Bilinear constraint: h[i] * (1 - h[i + k]) <= lambda_vars
            # McCormick envelopes for bilinear terms
            for i in valid_i:
                i_k = i + k
                expr = h_vars[i] * (1 - h_vars[i_k])
                lb = 0  # lower bound of bilinear term
                ub = h_vars[i] + (1 - h_vars[i_k]) - 1  # upper bound from bilinear
                # Use linear McCormick inequalities: 
                # h[i]*(1-h[i_k]) <= lambda_vars <= h[i] + (1-h[i_k]) - 1
                problem.constraints += [expr <= lambda_vars, lambda_vars <= h_vars[i] + (1 - h_vars[i_k]) - 1]

        # Solve the problem using the primal-dual interior point method
        try:
            problem.solve(solver=cp.OSQP, verbose=False, max_iter=2000)
        except Exception:
            pass

        # If problem is infeasible or unbounded, restore the best_h and continue
        if h_vars.value is None:
            h = ensure_feasible(best_h)
            continue

        # Project new solution to feasible space
        h = ensure_feasible(np.clip(h_vars.value, 0, 1))

    # Final feasibility repair
    h = repair_to_binary(best_h, n_points)
    final_C5 = evaluate_C5(h)

    return h, final_C5, n_points
```
