```python
import numpy as np
import time
import cvxpy as cp


initial_h_values = None  # will be provided at runtime


def evaluate_C5_and_correlations(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode="full") * dx
    max_corr = np.max(correlations)
    return max_corr, correlations


def ensure_feasible(h):
    # Enforce bounds and normalize
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total < 1e-8:
        h.fill(0)
        h[0] = 1
        total = 1
    scaling = (len(h) / 2) / total
    h *= scaling
    return h


def build_optimization_model(n_points, h_guess):
    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    ones_minus_h = 1 - h
    correlations = []
    for k in range(-n_points + 1, n_points):
        idxs = [(i, (i + k) % n_points) for i in range(n_points) if (i + k) % n_points != i]
        corr = cp.sum([h[i] * ones_minus_h[(i + k) % n_points] for i in range(n_points)])
        correlations.append(corr * dx)

    obj = cp.Minimize(cp.Max(correlations))
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

    model = cp.Problem(obj, constraints)
    return model


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    h = ensure_feasible(np.random.rand(n_points)) if initial_h_values is None else ensure_feasible(initial_h_values)
    best_h = np.copy(h)
    best_C5, _ = evaluate_C5_and_correlations(h)

    while time.time() - start_time < budget_s:
        model = build_optimization_model(n_points, best_h)
        model.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s - (time.time() - start_time))
        if model.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.array(model.vars["h"].value).flatten()
            h_solution = ensure_feasible(h_solution)
            current_C5, _ = evaluate_C5_and_correlations(h_solution)
            if current_C5 < best_C5:
                best_h = h_solution
                best_C5 = current_C5

        if time.time() - start_time >= budget_s:
            break

    return best_h, best_C5, n_points
```
