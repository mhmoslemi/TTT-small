import numpy as np
import time

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Normalize to match the integral constraint
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-8:
        h = h * (target_sum / h_sum)
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        h_sum = h_values.sum()
        if abs(h_sum - target_sum) > 1e-8:
            h_values = h_values * (target_sum / h_sum)
        ones_minus_h = 1.0 - h_values
        max_corr = -np.inf

        # Compute forward shifts
        for k in range(1, n_points):
            if n_points - k > 0:
                corr = np.dot(h_values[:n_points - k], ones_minus_h[k:])
                max_corr = max(max_corr, corr * dx)

        # Compute reverse shifts
        for k in range(1, n_points):
            if n_points - k > 0:
                corr = np.dot(h_values[k:], ones_minus_h[:n_points - k])
                max_corr = max(max_corr, corr * dx)

        return max_corr

    # Optimization
    best_c5 = compute_c5(h)
    perturb_size = 10
    step_size = 0.01

    best_time = time.time()
    patience = 0

    while time.time() - start_time < budget_s:
        # Perturbation
        indices = np.random.choice(n_points, size=min(perturb_size, n_points), replace=False)
        proposed_h = h.copy()
        proposed_h[indices] += np.random.uniform(-step_size, step_size, size=indices.shape)
        proposed_h = np.clip(proposed_h, 0.0, 1.0)

        # Normalize and check sum
        h_sum = proposed_h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            proposed_h *= target_sum / h_sum
        proposed_h = np.clip(proposed_h, 0.0, 1.0)

        # Evaluate
        new_c5 = compute_c5(proposed_h)
        time_elapsed = time.time() - best_time

        if new_c5 < best_c5:
            h = proposed_h.copy()
            best_h = h.copy()
            best_c5 = new_c5
            best_time = time.time()
            patience = 0
        else:
            patience += 1

        # Adaptive changes
        if patience > 100:
            step_size *= 1.2
        if patience > 200:
            perturb_size = min(perturb_size + 2, n_points)

        if time_elapsed > 120 and best_c5 < 0.380876:
            break

        # Termination check
        if best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
