import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize h to have exactly the target sum and domain [0,1]
    h = h_init.copy()
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-12:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0.0, 1.0)

    # Final validation
    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-12:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h
        max_corr = 0
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    best_h = h.copy()
    best_c5 = compute_c5(h)
    last_improvement_time = start_time

    while time.time() - start_time < budget_s:
        h = best_h.copy()
        ones_minus_h = 1.0 - h

        # Compute current maximum shift k and its overlap
        corrs = np.zeros(n_points)
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corrs[k] = np.dot(h[:n_points - k], ones_minus_h[k:]) * dx
        kmax = np.argmax(corrs)

        # Construct gradient for this kmax
        grad = np.zeros(n_points)
        for i in range(n_points - kmax):
            grad[i] += 1 - h[i + kmax]
            grad[i + kmax] += -h[i]

        # Gradient descent with step size heuristic
        alpha = 0.01
        h_update = h - alpha * grad
        h_update = np.clip(h_update, 0, 1)
        h_update_sum = h_update.sum()

        # Projection to satisfy the integral constraint
        if abs(h_update_sum - target_sum) > 1e-12:
            h_update *= (target_sum / h_update_sum)

        # Keep only if better
        new_c5 = compute_c5(h_update)
        if new_c5 < best_c5:
            best_h = h_update.copy()
            best_c5 = new_c5
            last_improvement_time = time.time()

        # Perturb every 120 seconds to avoid local traps
        if time.time() - last_improvement_time > 120 and time.time() - start_time < budget_s - 20:
            noise = np.random.uniform(-0.02, 0.02, n_points)
            h_noise = best_h + noise
            h_noise = np.clip(h_noise, 0.0, 1.0)
            h_noise_sum = h_noise.sum()
            if abs(h_noise_sum - target_sum) > 1e-12:
                h_noise *= (target_sum / h_noise_sum)
            noise_c5 = compute_c5(h_noise)
            if noise_c5 < best_c5:
                best_h = h_noise.copy()
                best_c5 = noise_c5
                last_improvement_time = time.time()

        # Early return if bound satisfied
        if best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
