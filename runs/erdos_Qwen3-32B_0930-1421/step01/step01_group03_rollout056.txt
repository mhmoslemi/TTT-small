import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    # Linearize function: returns shifted indices
    def shifted_index(i, shift):
        return (i - shift + n_points) % n_points

    def compute_objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    # Constructing linear constraints by approximating overlaps using linearization
    def build_lp(h):
        h_vars = cp.Variable(n_points, nonneg=True)
        C = cp.Variable()

        constraints = [
            cp.sum(h_vars) == constraint_sum,
            h_vars <= 1,
            h_vars >= 0
        ]

        for k in range(1, n_points):
            # Shift h in correlation with k
            shifted_h = h[shifted_index(np.arange(n_points), k)]
            overlap_approx = cp.sum(h_vars * (1 - shifted_h))

            constraints.append(overlap_approx <= C)

        obj = cp.Minimize(C)
        problem = cp.Problem(obj, constraints)
        return problem, h_vars

    # Run LP with linearized bilinear constraints
    def solve_lp(h):
        problem, h_vars = build_lp(h)
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status != 'optimal':
            return h.copy()
        return np.array(h_vars.value).copy()

    # Feasibility fixer
    def fix_h(h):
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    h_current = h_init
    best_h = h_current
    best_c5 = compute_objective(h_current)

    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        h_current = solve_lp(h_current)
        h_current = fix_h(h_current)
        current_c5 = compute_objective(h_current)

        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5

        # Perturbation every 3 iterations
        if iter_count % 3 == 0:
            noise = np.random.randn(n_points) * 0.02
            h_current += noise
            h_current = fix_h(h_current)

        if time.time() - start_time > budget_s * 0.98:
            break

    best_h = fix_h(best_h)
    return best_h, best_c5, n_points
