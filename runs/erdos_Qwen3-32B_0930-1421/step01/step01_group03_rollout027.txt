import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    n_points = 47 if 'initial_h_values' not in globals() else len(initial_h_values)
    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i + i] = 2.0 / n_points
        h = h * (constraint_sum / h.sum())

    # Normalize and constrain helper
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    h = normalize_h(h)

    def compute_overlap(h):
        h = h.reshape(-1)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    best_h = h.copy()
    best_c5 = compute_overlap(best_h)

    # Define LP formulation with linearization
    def linearize_overlap(h, tolerance=1e-3):
        n = len(h)
        variables = h.copy()
        overlap_values = np.zeros(n)
        for shift in range(1, n):
            idx1 = np.arange(n)
            idx2 = (idx1 - shift) % n
            terms = h[idx1] * (1 - h[idx2])
            overlap = np.sum(terms) * dx
            gradient = []
            dL_dh = np.zeros_like(h)
            for i in range(n):
                dL_dh[i] += (1 - h[(i - shift) % n]) if i in idx1 else 0
                dL_dh[(i + shift) % n] -= h[i] if i in idx1 else 0
            gradient = dL_dh
            overlap_values[shift-1] = overlap
        return overlap_values, gradient

    # Use CVXPY to solve LP
    def solve_lp(h):
        h_var = cp.Variable(h.size, nonneg=True)
        c_var = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h_var) == constraint_sum)
        constraints.append(h_var <= 1)
        constraints.append(h_var >= 0)
        for shift in range(1, n_points):
            idx1 = np.arange(n_points)
            idx2 = (idx1 - shift) % n_points
            overlap_expr = cp.sum(h_var[idx1] * (1 - h[idx2])) * dx
            constraints.append(c_var >= overlap_expr)
        objective = cp.Minimize(c_var)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value, problem.value
        return h.value, float("inf")

    # Iterative improvement loop
    iteration = 0
    tried_perturbation = False
    perturbation_period = 5
    last_improvement = 0

    while start_time + budget_s - time.time() > 0.5:
        h_lp, c5_lp = solve_lp(h)
        if c5_lp < best_c5:
            best_h = h_lp
            best_c5 = c5_lp
            last_improvement = iteration
        iteration += 1
        if iteration % perturbation_period == 0 and iteration > last_improvement + 1:
            perturbation = np.random.normal(0, 0.02, size=n_points)
            h = normalize_h(h + perturbation)

        if time.time() - start_time > budget_s - 1 or iteration > 300:
            break

    # Normalize again and final C5
    best_h = normalize_h(best_h)
    final_c5 = compute_overlap(best_h)

    return best_h, final_c5, n_points
