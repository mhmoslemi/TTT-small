import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h_scaled = h / h_sum * len(h) / 2
    cor = correlate(h_scaled, 1 - h_scaled, mode="full")
    C5 = np.max(cor) * dx
    return C5


def constrain_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum < 1e-10:
        h.fill(0)
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / h_sum
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points
    h = constrain_and_scale(np.copy(initial_h_values), n_points)
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    last_update_time = time.time()
    max_time_per_iter = 10  # time budget per iteration

    while time.time() < last_update_time + budget_s:
        # Rescale and validate
        h = constrain_and_scale(h, n_points)
        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_update_time = time.time()

        # Identify worst shift
        h_scaled = h / h.sum() * n_points / 2
        cor = correlate(h_scaled, 1 - h_scaled, mode="full")
        max_id = np.argmax(cor)
        offset = max_id - n_points + 1

        # Construct problem to reduce the worst overlap
        h_var = cp.Variable(n_points, nonneg=True)
        objective = cp.Minimize(
            cp.sum(cp.multiply(h_var, h_var[offset:])) * dx
        )
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var, h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GUROBI, timeLimit=5, verbose=False)
            h_update = np.array(h_var.value).flatten()
            h = const_percentile_update(h, h_update, cp=0.1)
            h = constrain_and_scale(h, n_points)
        except Exception:
            # If optimization fails, apply random noise
            h = h + 0.05 * (np.random.rand(*h.shape) - 0.5)
            h = constrain_and_scale(h, n_points)

        # Occasionally reset a subset of h values
        if np.random.rand() < 0.02:
            idx = np.random.choice(n_points, size=int(n_points // 2), replace=False)
            h[idx] = np.random.uniform(0, 1, size=int(n_points // 2))
            h = constrain_and_scale(h, n_points)

    return best_h, best_C5, n_points


def const_percentile_update(h, target_h, cp=0.1):
    # Blended between current h and target h toward current
    lambda_ = cp * (h - target_h)
    return np.clip(target_h + lambda_, 0, 1)
