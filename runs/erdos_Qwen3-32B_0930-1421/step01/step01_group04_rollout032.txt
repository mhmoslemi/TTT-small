import numpy as np
import scipy.optimize as opt
import time
import warnings

warnings.filterwarnings("ignore", category=RuntimeWarning)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    maxcorr = 0.0
    h_np = np.asanyarray(h)
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h_np[:n_points - k], 1.0 - h_np[k:k + n_points - k]) * dx
        maxcorr = max(maxcorr, corr)
    for k in range(1, n_points):
        corr = np.dot(h_np[k:], 1.0 - h_np[:n_points - k]) * dx
        maxcorr = max(maxcorr, corr)
    return maxcorr

def objective_surrogate(h_vec, n_points):
    h = h_vec.copy().reshape(-1)
    h = np.clip(h, 1e-6, 1 - 1e-6)
    h = (h / np.sum(h)) * (n_points / 2)
    c5 = compute_c5(h, n_points)
    return c5

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h *= (n_points / 2) / total
    h = np.clip(h, 0.0, 1.0)
    return h

def generate_cosine_basis(n_points):
    return 0.5 + 0.5 * np.cos(np.pi * np.arange(n_points) / n_points)

def pso_run(n_points, budget_s, init_h=None, n_particles=20, iters=1000):
    if init_h is not None:
        x0 = init_h.copy()
    else:
        x0 = generate_cosine_basis(n_points)
    x0 = enforce_normalization(x0, n_points)
    
    # Initialize particles
    pos = np.random.uniform(size=(n_particles, n_points)) * 0.5
    for i in range(n_particles):
        pos[i] = np.clip(pos[i], 0.0, 1.0)
        pos[i] = (pos[i] / np.sum(pos[i])) * (n_points / 2)

    if init_h is not None:
        pos[0] = x0.copy()

    vel = np.random.uniform(-0.1, 0.1, size=pos.shape)
    
    pbest = pos.copy()
    pbest_obj = np.array([objective_surrogate(p, n_points) for p in pos])
    ibest = np.argmin(pbest_obj)
    gbest = pbest[ibest].copy()
    gbest_obj = pbest_obj[ibest]
    
    w = 0.729
    c1, c2 = 1.49445, 1.49445
    t_start = time.time()
    
    for iter in range(iters):
        t_remaining = budget_s - (time.time() - t_start)
        if t_remaining < 0:
            break
            
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel[i] = w * vel[i] + c1 * r1 * (pbest[i] - pos[i]) + c2 * r2 * (gbest - pos[i])
            pos[i] = np.clip(pos[i] + vel[i], 0.0, 1.0)
            total = np.sum(pos[i])
            pos[i] = pos[i] * (n_points / 2) / total
            pos[i] = np.clip(pos[i], 0.0, 1.0)
            obj = objective_surrogate(pos[i], n_points)
            if obj < pbest_obj[i]:
                pbest[i] = pos[i].copy()
                pbest_obj[i] = obj
        ibest = np.argmin(pbest_obj)
        if pbest_obj[ibest] < gbest_obj:
            gbest = pbest[ibest].copy()
            gbest_obj = pbest_obj[ibest]
    
    return gbest, gbest_obj, n_points

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is None:
        raise ValueError("initial_h_values is not provided")
    n_points = len(init_h)

    # Run PSO with surrogate objective
    h_best, c5_best, n_points = pso_run(n_points, budget_s, init_h=init_h)

    # Enforce normalization
    h_best = enforce_normalization(h_best, n_points)

    # Final clean evaluation
    c5_bound = compute_c5(h_best, n_points)

    return h_best, c5_bound, n_points
