import numpy as np
import time
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    cor = correlate(h, 1 - h, mode='full')
    return np.max(cor) * dx


def rescale_h(h):
    """Rescale h to satisfy the integral constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h
