import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = np.tile(h, int(n_points / len(h)) + 1)[:n_points]
            h = h / h.sum() * n_points / 2
    except:
        h = np.zeros(n_points)
        h[:5] = 0.1
        h[5:42] = 0.3
        h[42:] = 0.2
        h = h / h.sum() * n_points / 2

    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.inf
    weights = (0.05, 0.0)  # overlap regularizer λ, Tikhonov μ
    solver = 'ECOS'
    tikhonov_decay = 0.75

    def evaluate_C5(h_vec):
        h = h_vec.copy()
        h[h > 1] = 1
        h[h < 0] = 0
        h = h / h.sum() * n_points / 2
        cor = correlate(h, 1 - h, mode="full")
        return np.max(cor) * dx, h

    def solve_cp(h_init, λ, μ, solver=solver):
        h = cp.Variable(n_points)
        c1 = cp.sum(h) == n_points / 2
        c2 = cp.multiply(h >= 0, h <= 1)
        overlaps = cp.sum([cp.square(correlate(h, 1 - h, mode="full"))])
        E = λ * overlaps + μ * cp.sum_squares(h)
        prob = cp.Problem(cp.Minimize(E), [c1] + list(c2))
        prob.solve(solver=solver, verbose=False)
        if prob.status != 'optimal':
            return h_init
        return h.value

    λ, μ = weights
    iter = 0
    tikhonov_step = 10

    while time.time() - start_time < budget_s:
        iter += 1

        # Solve CP
        h_opt = solve_cp(h, λ, μ, solver)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / h_opt.sum() * n_points / 2
        C5, h_f = evaluate_C5(h_opt)
        if C5 < best_c5:
            best_h = h_f
            best_c5 = C5

        # Perturb and restart
        if iter % tikhonov_step == 0:
            h = best_h + 0.05 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2
            μ *= tikhonov_decay
            λ *= 1.01  # optionally increase this over time

    return best_h, best_c5, n_points
