import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize h to meet the sum constraint
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    if not np.allclose(np.sum(h), n_points / 2, rtol=1e-6, atol=1e-6):
        raise ValueError('Initial h is not feasible')

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    dx = 2.0 / n_points

    def solve_relaxation(C):
        # Formulation using CVXPY
        h_var = cp.Variable(n_points, nonneg=True)
        z_var = cp.Variable(n_points * (n_points - 1) // 2)  # all shifted (i, i + k)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1.0,
            h_var >= 0.0,
        ]

        idx = 0
        for k in range(1, n_points):
            for i in range(n_points - k):
                j = i + k
                z_entry = z_var[idx]
                constraints.append(z_entry >= 0)
                constraints.append(z_entry <= h_var[i])
                constraints.append(z_entry <= 1 - h_var[j])
                constraints.append(z_entry >= h_var[i] + (1 - h_var[j]) - 1)
                idx += 1

        objective = cp.Maximize(0)  # We're checking feasibility
        prob = cp.Problem(objective, constraints)

        # Solve with PDLP for large-scale LPs and low cost
        try:
            result = prob.solve(solver=cp.PDLP, verbose=False, max_iter=100)
        except Exception:
            return None, None

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            feasible_h = h_var.value
            feasible_h = np.clip(feasible_h, 0.0, 1.0)
            # renormalize
            feasible_h *= (n_points / 2) / (float(np.sum(feasible_h)) + 1e-8)
            feasible_h = np.clip(feasible_h, 0.0, 1.0)
            return feasible_h, evaluate_erdos_solution(feasible_h, n_points)
        return None, None

    def binary_search_C(C_low, C_high, max_iter=50):
        nonlocal best_h, best_c5, last_improvement_time
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            C_mid = (C_low + C_high) / 2
            h_feas, c5_feas = solve_relaxation(C_mid)
            if h_feas is not None and c5_feas is not None and c5_feas <= C_high:
                best_h = h_feas
                best_c5 = c5_feas
                last_improvement_time = time.time()
                C_high = C_mid
            else:
                C_low = C_mid
        return best_h, best_c5

    def perturb_h(h, scale=0.1):
        h_pert = h + np.random.uniform(-scale, scale, h.shape)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert *= (n_points / 2) / (float(np.sum(h_pert)) + 1e-8)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        return h_pert

    # Binary search with warm starts
    C_low = 0.0
    C_high = best_c5
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        best_h, best_c5 = binary_search_C(C_low, C_high, max_iter=10)

    # Perturb and search from new starting points
    for i in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        h_pert = perturb_h(best_h)
        C_low = 0.0
        C_high = evaluate_erdos_solution(h_pert, n_points)
        best_h, best_c5 = binary_search_C(C_low, C_high, max_iter=10)

    return best_h, best_c5, n_points
