import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 61  # target small n_points for fast evaluation

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values, copy=True)
    else:
        n_points = max_n_points
        h_base = np.zeros(n_points)
        for i in range(n_points // 2):
            h_base[i * 2 % n_points] = 2.0 / n_points
        # Normalize
        h_base = h_base * (n_points / 2) / h_base.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        return h

    def objective(h):
        h = project_h(h)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def perturb(h, noise_factor=0.1):
        h = h + np.random.randn(h.size) * noise_factor
        return project_h(h)

    def crossover(h1, h2):
        weights = np.random.rand(h1.size)
        return project_h(h1 * weights + h2 * (1 - weights))

    def optimize_h(initial_h, max_iter=500):
        def obj_func(h_flat):
            h = project_h(h_flat.reshape(n_points))
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        bounds = [(0, 1)] * n_points
        result = minimize(obj_func, initial_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_opt = result.x.reshape(n_points)
        return project_h(h_opt)

    population_size = 12
    population = [project_h(np.random.uniform(0.3, 0.7, n_points)) for _ in range(population_size // 2)]

    if 'initial_h_values' in globals():
        # Use parent solution as a base and perturb to level-start the population
        for _ in range(population_size // 2):
            h = project_h(np.random.permutation(perturb(h_base, noise_factor=0.15)))
            if h.sum() > 0:
                population.append(h)

    # Ensure all candidates are feasible
    for i in range(len(population)):
        population[i] = project_h(population[i])

    # Gradient descent and mutation
    best_h = h_base
    best_c5 = objective(best_h)

    iter_count = 0
    patience = 50
    no_improvement = 0

    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1

        # Rank population
        candidates = [(h, objective(h)) for h in population]
        candidates.sort(key=lambda x: x[1])
        top_quarter = [h for h, val in candidates[: max(len(candidates) // 2, 2)]]

        if candidates[0][1] < best_c5:
            best_h = candidates[0][0]
            best_c5 = candidates[0][1]
            no_improvement = 0
        else:
            no_improvement += 1

        # Early break condition
        if no_improvement > patience:
            break

        new_population = []
        for i in range(population_size):
            if np.random.rand() < 0.5:
                parent1 = np.random.choice(top_quarter)
                parent2 = np.random.choice(top_quarter)
                child = crossover(parent1, parent2)
            else:
                parent = np.random.choice(top_quarter)
                child = perturb(parent, noise_factor=0.15)

            if np.random.rand() < 0.2:  # Local improvement attempt
                child = optimize_h(child, max_iter=250)

            new_population.append(child)

        population = new_population

    best_h = project_h(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
