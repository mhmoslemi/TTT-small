import numpy as np
import time
import scipy.optimize as opt

def evaluate_population(population, n_points):
    dx = 2.0 / n_points
    C5 = []
    for h in population:
        values = []
        for k in range(1, n_points):
            if k >= n_points:
                break
            left = h[:n_points - k]
            right = 1.0 - h[k:k + n_points - k]
            values.append(np.dot(left, right))
        for k in range(1, n_points):
            left = h[k:]
            right = 1.0 - h[:n_points - k]
            values.append(np.dot(left, right))
        C5.append(np.max(np.array(values)) * dx)
    return np.array(C5)

def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    sv = np.cumsum(u) - np.arange(1, n + 1) / n
    pos = np.nonzero(sv > 0)[0]
    pos = pos[-1] if len(pos) else -1
    idx = pos + 1
    tau = sv[pos] / idx
    return np.clip(v - tau, 0, 1)

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
    h = project_to_simplex(h)
    return h

def crossover(h1, h2, alpha=0.5):
    h = alpha * h1 + (1 - alpha) * h2
    return enforce_feasibility(h, len(h1))

def mutate(h, sigma):
    return enforce_feasibility(h + sigma * np.random.randn(len(h)), len(h))

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    n_points = len(init_h)
    if n_points < 32:
        n_points = 47  # Minimum to ensure good resolution

    total_population = 50
    mutation_rate = 0.2
    mutation_sigma = 0.1
    sigma_decay = 0.95
    generations = 0
    generation_limit = 100
    time_budget = budget_s
    time_start = time.time()

    # Build population
    population = []
    if init_h is not None:
        population.append(init_h.copy())
    while len(population) < total_population:
        h = np.random.rand(n_points)
        h = enforce_feasibility(h, n_points)
        population.append(h)

    # Main genetic algorithm
    while time.time() - time_start < time_budget:
        generations += 1
        if generations >= generation_limit:
            break

        # Evaluate fitness
        fitness = evaluate_population(population, n_points)
        idx = np.argsort(fitness)
        best_idx = idx[0]
        best_h = population[best_idx].copy()
        best_C5 = fitness[best_idx]

        # Select top 50% as parents
        parent_indices = idx[: total_population // 2]
        parents = [population[i].copy() for i in parent_indices]

        # Crossover and generate new population
        new_population = []
        for i in range(len(parents)):
            for j in range(i + 1, len(parents)):
                child = crossover(parents[i], parents[j], np.random.rand())
                new_population.append(child)

        # Mutation
        for i in range(int(total_population * mutation_rate)):
            child = mutate(best_h, mutation_sigma)
            new_population.append(child)

        # Truncate to match population size
        population = [enforce_feasibility(h, n_points) for h in new_population]
        population = population[:total_population]

        # Decrease mutation intensity
        mutation_sigma *= sigma_decay

    # Final refinement
    final_h = enforce_feasibility(best_h, n_points)
    result = opt.minimize(
        lambda x: evaluate_population([x.reshape(-1)], n_points)[0],
        final_h,
        bounds=[(0, 1)] * n_points,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    final_h = enforce_feasibility(result.x, n_points)
    final_C5 = evaluate_population([final_h], n_points)[0]

    return final_h, final_C5, n_points
