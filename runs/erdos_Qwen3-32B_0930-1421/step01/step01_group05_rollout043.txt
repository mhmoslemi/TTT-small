import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    def compute_c5(h, dx):
        dx = 2.0 / h.shape[0]
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        return clipped * (n_points / 2) / total

    def feasible_perturbation(h, sigma=0.05):
        return normalize_h(h + sigma * (np.random.rand(*h.shape) - 0.5))

    # Firefly algorithm setup
    n_fireflies = 50
    fireflies = [normalize_h(np.random.rand(n_points)) for _ in range(n_fireflies - 1)]
    if 'initial_h_values' in globals():
        fireflies.append(h.copy())
    fireflies = np.array(fireflies)

    best_h = None
    best_c5 = float('inf')
    T = 1000
    gamma = 0.01
    beta0 = 1.0
    alpha = 0.2

    # Main optimization loop
    while time.time() - start_time < budget_s:
        fireflies_c5 = np.array([compute_c5(f, 2.0 / n_points) for f in fireflies])
        indices_sorted = np.argsort(fireflies_c5)
        fireflies = fireflies[indices_sorted]
        fireflies_c5 = fireflies_c5[indices_sorted]

        if fireflies_c5[0] < best_c5:
            best_c5 = fireflies_c5[0]
            best_h = fireflies[0].copy()

        for i in range(n_fireflies):
            for j in range(i):
                if fireflies_c5[j] < fireflies_c5[i]:
                    r = np.linalg.norm(fireflies[i] - fireflies[j])
                    beta = beta0 * np.exp(-gamma * r**2)
                    fireflies[i] = (
                        fireflies[i] + beta * (fireflies[j] - fireflies[i]) +
                        alpha * (np.random.rand(n_points) - 0.5)
                    )
                    fireflies[i] = normalize_h(fireflies[i])

        # Update light absorption coefficient
        t_elapsed = time.time() - start_time
        gamma = t_elapsed / budget_s  # Increase exploration

        # Re-evaluate
        fireflies_c5 = np.array([compute_c5(f, 2.0 / n_points) for f in fireflies])
        if fireflies_c5[0] < best_c5:
            best_c5 = fireflies_c5[0]
            best_h = fireflies[0].copy()

        if best_c5 < 1e-10:
            break

    return best_h.tolist(), best_c5, n_points
