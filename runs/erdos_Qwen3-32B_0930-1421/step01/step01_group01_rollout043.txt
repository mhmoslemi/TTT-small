import numpy as np
import time
import scipy
import scipy.signal


def evaluate_C5(h, n_points):
    """Compute the C5 bound from h, with normalization."""
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    h = h / h_sum * n_points / 2  # normalize for integral constraint
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n_points
    return np.max(cor) * dx


def repair_solution(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = 0.0
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h


def perturb(h, step_size=0.1, mask_fraction=0.3):
    ldims = len(h)
    mask = np.random.uniform(size=ldims) < mask_fraction
    h_perturb = h.copy()
    delta = np.random.uniform(-step_size, step_size, size=ldims) * mask
    h_perturb += delta
    return repair_solution(h_perturb, ldims)

    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, n_points)

    temp_schedule = np.linspace(start=1.2, stop=0.1, num=6000)
    T_idx = 0
    steps_per_temp = 10

    t0 = time.time()
    while time.time() - t0 < budget_s and T_idx < len(temp_schedule):
        T = temp_schedule[T_idx]
        for _ in range(steps_per_temp):
            candidate = perturb(best_h)
            c5_current = best_C5
            c5_candidate = evaluate_C5(candidate, n_points)
            delta = c5_candidate - c5_current
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                if c5_candidate < best_C5:
                    best_h = candidate.copy()
                    best_C5 = c5_candidate

        T_idx += 1

    return best_h, best_C5, n_points
