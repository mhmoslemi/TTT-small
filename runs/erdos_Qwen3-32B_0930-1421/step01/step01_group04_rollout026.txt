import numpy as np
import time
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import ConstantKernel, RBF

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def generate_candidate(h, sigma, n_points):
    h_p = h + np.random.normal(0, sigma, n_points)
    h_p = enforce_normalization(h_p, n_points)
    return h_p

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in globals():
        h_init = initial_h_values
    else:
        h_init = np.random.rand(n_points)
        h_init = enforce_normalization(h_init, n_points)

    X = [h_init.copy()]
    y = [compute_c5(X[0], n_points)]
    time_start = time.time()
    budget = budget_s

    # Sample initial training data
    while len(X) < 10 and time.time() - time_start < 0.8 * budget:
        h_candidate = np.random.rand(n_points)
        h_candidate = enforce_normalization(h_candidate, n_points)
        if not any(np.allclose(h_candidate, x, atol=1e-6) for x in X):
            X.append(h_candidate)
            y.append(compute_c5(h_candidate, n_points))
            if time.time() - time_start >= 0.8 * budget:
                break

    # Build Gaussian process surrogate prior
    X = np.array(X)
    y = np.array(y)

    kernel = ConstantKernel(1.0) * RBF(10.0)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5, normalize_y=True)
    try:
        gp.fit(X, y)
    except:
        pass  # Skip fit if failed, will attempt later

    h_best = X[0].copy()
    val_best = y[0]
    T0 = 1000
    alpha = 0.99
    T = T0
    max_its = 10000
    sigma = 0.2
    retrain_radius = 50
    retrain_cnt = 0
    gp.update = 0

    for it in range(max_its):
        if time.time() - time_start >= 0.95 * budget:
            break
        h = h_best.copy()
        # Sample next with surrogate
        try:
            mu, std = gp.predict([h], return_std=True)
            pert_pct = max(0.05, sigma * (it / max_its)**(1/4))
            sample = h + pert_pct * np.random.randn(n_points)
            sample = sample * (np.sum(sample) == n_points / 2) / np.sum(sample)  # zero if invalid
            sample = enforce_normalization(sample, n_points)
        except:
            sample = generate_candidate(h_best, sigma, n_points)

        val = compute_c5(sample, n_points)
        if val < val_best or np.random.rand() < np.exp((val_best - val) / T):
            h_best = sample.copy()
            val_best = val
            X.append(h_best)
            y.append(val_best)
            retrain_cnt += 1

        if retrain_cnt >= retrain_radius:
            try:
                gp.fit(X, y)
                retrain_cnt = 0
                gp.update += 1
            except:
                pass

        # Annealing schedule
        T *= alpha

    # Final refinement
    h_refined = minimize(lambda x: compute_c5(x, n_points), x0=h_best,
                         bounds=[(0, 1)] * n_points, method='L-BFGS-B', options={'maxiter': 200}).x
    h_refined = enforce_normalization(h_refined, n_points)
    val_final = compute_c5(h_refined, n_points)

    return h_refined, val_final, n_points
