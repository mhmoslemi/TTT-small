import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # ensure small sequences for fast evaluation
    n_points = 47
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i * 2] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h):
        h_arr = initial_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(x):
            h = x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            h = h * (constraint_sum / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            obj_func,
            h_arr,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000}
        )
        h_opt = np.clip(result.x, 0, 1)
        h_opt *= constraint_sum / h_opt.sum()
        return h_opt

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        noise = np.random.normal(0, noise_factor, len(h))
        h = h + noise
        h = np.clip(h, 0, 1)
        h *= constraint_sum / h.sum()
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # Gradient optimization on original
    h = optimize_h(h_init)
    c5 = objective(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Tabu search style restarts with memory of last 15
    tabu_list = []
    tabu_max = 15

    # Cyclic perturbations
    perturbation_magnitude = 0.1
    iter_count = 0
    no_improve_count = 0
    cooling_threshold = 50
    cooling_factor = 0.8

    while time.time() - start_time < budget_s:
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] += np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())

        key = tuple(h)
        if key in tabu_list:
            continue
        tabu_list.append(key)
        if len(tabu_list) > tabu_max:
            tabu_list.pop(0)

        h_opt = optimize_h(h)
        c5 = objective(h_opt)
        iter_count += 1

        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            no_improve_count = 0
            perturbation_magnitude = max(perturbation_magnitude * cooling_factor, 0.005)
        else:
            no_improve_count += 1

        if no_improve_count >= cooling_threshold:
            perturbation_magnitude *= cooling_factor
            no_improve_count = 0

    # Final validation and constraint enforcement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, best_c5, n_points
