import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the initial_h_values provided by parent if exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45  # fallback length if no parent construction
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points
    max_fireflies = 50

    # Repair function to ensure constraints
    def repair_h(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            h_scaled = np.zeros_like(h)
        else:
            h_scaled = h_clipped * (n_points / 2) / sum_h
        return h_scaled

    def evaluate_c5(h):
        c = np.zeros(2 * n_points - 1)
        for k in range(1, c.shape[0]):
            shifted_h = np.roll(h, -(k - n_points + 1))
            shifted_h[:n_points - (k - n_points + 1)] = 0  # truncate shifted part
            c[k - 1] = np.sum(h * (1 - shifted_h[:n_points]))
        c *= dx
        return np.max(c)

    # Initialize fireflies
    fireflies = []
    if 'initial_h_values' in globals():
        # Use provided h as the first firefly
        fireflies.append(h.copy())
        # Perturb initial to generate others
        for _ in range(max_fireflies - 1):
            noise = np.random.normal(0, 0.05, h.shape)
            new_h = h.copy() + noise
            fireflies.append(repair_h(new_h))
    else:
        # Build from scratch
        for _ in range(max_fireflies):
            new_h = np.random.rand(n_points)
            new_h = new_h / new_h.sum() * n_points / 2
            fireflies.append(repair_h(new_h))

    # Firefly parameters
    beta0 = 1.0
    alpha = 0.2
    gamma = 0.01  # Will grow over time
    epsilon = 1e-8
    best_h = []
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        # Update and sort fireflies by brightness (C₅ value)
        c5_values = [evaluate_c5(f) for f in fireflies]
        idx = np.argsort(c5_values)
        fireflies = [fireflies[i] for i in idx]
        c5_values = [c5_values[i] for i in idx]

        # Track best solution if improved
        if c5_values[0] < best_c5:
            best_c5 = c5_values[0]
            best_h = fireflies[0].copy()

        # Update each firefly
        for i in range(max_fireflies):
            for j in range(i):
                # Only move toward brighter fireflies
                if c5_values[j] < c5_values[i]:
                    r = np.linalg.norm(fireflies[i] - fireflies[j])
                    beta = beta0 * np.exp(-gamma * r**2)
                    fireflies[i] += beta * (fireflies[j] - fireflies[i]) + \
                                    alpha * (np.random.rand(n_points) - 0.5) * epsilon
                    fireflies[i] = repair_h(fireflies[i])
        # Adjust gamma over time to increase attraction
        gamma += 0.01

        # Early stop if budget is low
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
