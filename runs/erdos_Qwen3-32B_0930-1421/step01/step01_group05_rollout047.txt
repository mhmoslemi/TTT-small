import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(n_points, 0.5)
        h += np.random.normal(scale=0.05, size=n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h_in):
        overlap = correlate(h_in, 1.0 - h_in, mode='full') * dx
        return np.max(overlap)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        return np.clip(clipped * scale, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    patience_counter = 0
    patience_limit = 50

    while time.time() - start_time < budget_s:
        # Use analytic gradient for the current worst shift
        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        kstar = np.argmax(overlaps)
        if kstar >= n_points:
            kstar -= n_points
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + kstar < n_points:
                grad[i] -= h[i + kstar]
            if 0 <= i - kstar < n_points:
                grad[i] += h[i - kstar]
        grad = grad * dx

        # Line search: adaptive step size with backtracking
        alpha = 0.1
        for _ in range(3):
            h_new = h - alpha * grad
            h_new = normalize_h(h_new)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                break
            alpha *= 0.5

        if new_c5 >= best_c5:
            patience_counter += 1
        else:
            patience_counter = 0

        if patience_counter >= patience_limit:
            # Perturb slightly to escape local minima
            h += np.random.normal(scale=0.01, size=n_points)
            h = normalize_h(h)
            patience_counter = 0

    best_h = normalize_h(best_h)
    best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, n_points
