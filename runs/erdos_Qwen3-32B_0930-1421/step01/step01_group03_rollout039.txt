import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 55  # Higher resolution for initial step
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[2 * i] = 1.0 / (n_points / 2)
        h_init = np.clip(h_init, 0, 1)
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    total_mass = n_points / 2

    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum <= 0:
            return float('inf')
        h = h * (total_mass / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def cvx_relax(h_guess):
        w = cp.Variable(n_points)
        constraints = [
            cp.sum(w) == total_mass,
            0 <= w,
            w <= 1
        ]
        overlaps = np.correlate(w, 1 - w, mode='full') * dx
        obj = overlaps.max()
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver='ECOS', max_iters=250, verbose=False)
        return w.value

    def gradient_descent(h0):
        def inner(x):
            h = np.clip(x, 0, 1)
            h_sum = h.sum()
            if h_sum <= 0:
                return float('inf')
            h = h * (total_mass / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        result = minimize(
            inner,
            h0,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 200}
        )
        return result.x

    best_h = h_init.astype(float).copy()
    best_c5 = objective(best_h)

    # First solve using convex relaxation (ECOS)
    try:
        h_relaxed = cvx_relax(best_h)
        h_relaxed = np.clip(h_relaxed, 0, 1)
        h_relaxed *= total_mass / h_relaxed.sum()
        c5_relaxed = objective(h_relaxed)
        if c5_relaxed < best_c5:
            best_h = h_relaxed
            best_c5 = c5_relaxed
    except:
        pass  # proceed with current best

    # Gradient descent refinement
    h_gradient = gradient_descent(best_h)
    c5_gradient = objective(h_gradient)
    if c5_gradient < best_c5:
        best_h = h_gradient
        best_c5 = c5_gradient

    # Optional: Perturbation + refine scheme
    for attempt in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        noise = np.random.randn(n_points) * 0.05
        h_pert = (best_h + noise).clip(0, 1)
        h_pert *= total_mass / h_pert.sum()
        h_local = gradient_descent(h_pert)
        c5_local = objective(h_local)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local

    # Final validation and projection
    best_h = np.clip(best_h, 0, 1)
    best_h *= (total_mass / best_h.sum())

    return best_h, best_c5, n_points
