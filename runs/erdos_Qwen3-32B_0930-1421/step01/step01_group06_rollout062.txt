def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

    h_init = h_init / h_init.sum() * (n_points / 2.0)
    h_init = np.clip(h_init, 0.0, 1.0)

    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    particles[:10] = h_init  # 10% of particles as the initial solution
    particles[10:] = np.random.uniform(0, 1, (num_particles - 10, n_points))
    for i in range(10, num_particles):
        particles[i] /= particles[i].sum() * (n_points / 2.0)
        particles[i] = np.clip(particles[i], 0.0, 1.0)

    # Objective evaluation
    def evaluate(h):
        return evaluate_erdos_solution(h, n_points)

    # Constraint: sum(h) == n_points / 2
    def project(h):
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * (n_points / 2.0)
        return h

    objectives = np.array([evaluate(particles[i]) for i in range(num_particles)])
    p_best_pos = np.copy(particles)
    p_best_obj = np.copy(objectives)
    g_best_idx = np.argmin(p_best_obj)
    g_best_pos = np.copy(p_best_pos[g_best_idx])
    g_best_obj = p_best_obj[g_best_idx]

    best_h = np.copy(g_best_pos)
    best_c5 = g_best_obj
    last_improvement_time = start_time

    t0 = time.time()
    for iter in range(max_iter):
        # Inertia weight schedule
        w = w_start - (w_start - w_end) * (iter / max_iter)
        r1 = np.random.rand(num_particles, n_points)
        r2 = np.random.rand(num_particles, n_points)

        cognitive = c1 * r1 * (p_best_pos - particles)
        social = c2 * r2 * (np.expand_dims(g_best_pos, axis=0) - particles)
        velocities += w * velocities + cognitive + social
        velocities = np.clip(velocities, -max_velocity, max_velocity)
        particles += velocities

        objectives = np.array([evaluate(p) for p in particles])

            particles[indices] = np.random.uniform(0, 1, (reset_count, n_points))
            for i in indices:

        # Stopping condition
        if time.time() - start_time > budget_s:
            break

        if time.time() - last_improvement_time > 120:
