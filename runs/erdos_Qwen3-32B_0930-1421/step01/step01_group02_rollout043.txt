```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def repair_to_binary(h):
    h = np.array(h)
    h_thresholded = (h > 0.5).astype(float)
    diff = np.sum(h_thresholded) - len(h) / 2
    if diff != 0:
        # Adjust to meet the integer constraint
        idxs = np.argsort(np.abs(h - 0.5))
        for i in idxs:
            if (h_thresholded[i] == 1 and diff > 0):
                h_thresholded[i] = 0
                diff -= 1
                if diff == 0:
                    break
            elif (h_thresholded[i] == 0 and diff < 0):
                h_thresholded[i] = 1
                diff += 1
                if diff == 0:
                    break
    return h_thresholded


def compute_cutting_planes(n_points, h=None):
    dx = 2.0 / n_points
    if h is None:
        h = np.zeros(n_points)
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_idx = np.argmax(correlations)
    shift_k = (max_idx - n_points) if max_idx >= n_points else max_idx
    return shift_k


def run(budget_s=1100, n_points=42, verbose=False):
    start_time = time.time()
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h_initial = np.copy(initial_h_values)
    else:
        # Default uniform initialization
        h_initial = np.full(n_points, 1.0 / n_points * n_points / 2)

    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    alpha = 1.0  # Initial upper bound on C5

    while time.time() - start_time < budget_s:
        # Compute the most violating shift k using the current h values
        h_val = [h[i].value for i in range(n_points)] if h.value is not None else h_initial
        h_val = np.array(h_val)

        # Enforce feasibility during initial phase
        h_val = np.clip(h_val, 0, 1)
        h_val = h_val * (n_points / 2) / np.clip(h_val.sum(), 1e-6, np.inf)

        best_h = np.copy(h_val)
        best_C5 = evaluate_C5(best_h)
        cutoff_k = compute_cutting_planes(n_points, best_h)

        # Add the new constraint for the worst shift
        i_lower = max(0, -cutoff_k)
        i_upper = min(n_points - 1, n_points - cutoff_k)
        if abs(cutoff_k) >= n_points:
            break  # Cannot add constraint for out-of-bounds shift
        obj = cp.Maximize(alpha)
        rhs = objective_value = alpha * (n_points / 2)
        constraints += [
            cp.sum([h[i] * (1 - h[i + cutoff_k]) for i in range(i_lower, i_upper)]) <= rhs
        ]

        # Formulate and solve the problem
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver='ECOS', verbose=False)

        if time.time() - start_time >= budget_s:
            break

        # Repair to binary solution
        repaired_h = repair_to_binary(h.value)
        repaired_C5 = evaluate_C5(repaired_h)

        if repaired_C5 < best_C5:
            best_h = repaired_h
            best_C5 = repaired_C5

    return best_h, best_C5, n_points
```
