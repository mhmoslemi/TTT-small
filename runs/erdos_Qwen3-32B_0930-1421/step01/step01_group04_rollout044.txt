import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        valid_len = n_points - k
        if valid_len <= 0:
            break
        corr = np.dot(h[:valid_len], 1 - h[k:k + valid_len])
        corrs.append(corr)
    for k in range(1, n_points):
        valid_len = n_points - k
        if valid_len <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:valid_len])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if np.isclose(total, 0):
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def run(budget_s=1100):
    np.random.seed(42)
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
    else:
        h_init = np.random.rand(47)
        h_init = enforce_normalization(h_init, 47)

    n_points = len(h_init)
    h_init = enforce_normalization(h_init, n_points)

    h = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    objective = cp.Minimize(C)

    prob = cp.Problem(objective, constraints)
    time_start = time.time()

    while time.time() - time_start < budget_s - 50:
        # Add key shifts: small, medium, and large scales
        key_shifts = [1, 2, 4, 8, 16]
        for k in key_shifts:
            if k >= n_points:
                continue
            expr = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k)) * (2.0 / n_points)
            constraints.append(expr <= C)
        prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=(budget_s - (time.time() - time_start) - 50))

        if prob.status != cp.OPTIMAL:
            break

        h_val = np.clip(h.value.flatten(), 0, 1)
        h_val = enforce_normalization(h_val, n_points)

        # Re-evaluate to find most violated constraint
        c5 = compute_c5(h_val, n_points)
        worst_k = 0
        for k in range(1, n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                break
            corr = np.dot(h_val[:valid_len], 1 - h_val[k:k + valid_len])
            if corr * (2.0 / n_points) > C.value + 1e-5:
                worst_k = k
                break

        if worst_k != 0:
            for i in range(n_points - worst_k):
                expr += h[i] * (1 - h[i + worst_k])
            constraints.append(expr * (2.0 / n_points) <= C)
        else:
            break  # No more violated constraints

    # Final refinement with a local method to polish LP result
    prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=(budget_s - (time.time() - time_start)))
    h_ref = enforce_normalization(np.clip(h.value.ravel(), 0, 1), n_points)
    h_refined = opt.minimize(
        lambda x: compute_c5(x, n_points),
        h_ref,
        method="L-BFGS-B",
        bounds=[(0, 1)] * n_points,
        options={"maxiter": 200, "eps": 1e-4}
    ).x

    h_final = enforce_normalization(h_refined, n_points)
    c5_final = compute_c5(h_final, n_points)

    return h_final, c5_final, n_points
