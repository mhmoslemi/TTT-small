import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
import gurobipy as gp
from gurobipy import GRB

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial_h_values as starting point if available
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = correlate_c5(h, dx)

    # Setup Gurobi model
    model = gp.Model("erdos")
    model.setParam("OutputFlag", 0)
    model.setParam("TimeLimit", budget_s - (time.time() - start_time))

    h_vars = model.addVars(n_points, lb=0.0, ub=1.0, name="h")
    model.addConstr(gp.quicksum(h_vars) == n_points / 2)

    # Precompute overlap indices for each lag k
    overlap_exprs = []
    for k in range(-n_points + 1, n_points):
        x1_vals = [h_vars[i] for i in range(n_points) if i + k < n_points and i + k >= 0]
        x2_vals = [h_vars[i + k] for i in range(n_points) if i + k < n_points and i + k >= 0]
        if len(x1_vals) != len(x2_vals):
            continue
        overlap = gp.quicksum([x1 * (1 - x2) for x1, x2 in zip(x1_vals, x2_vals)]) * dx
        overlap_exprs.append(overlap)

    max_overlap = model.addVar(lb=-GRB.INFINITY, ub=GRB.INFINITY, name="max_overlap")
    for expr in overlap_exprs:
        model.addConstr(max_overlap >= expr)

    model.setObjective(max_overlap, GRB.MINIMIZE)
    model.optimize()

    if model.status == GRB.OPTIMAL:
        h_gurobi = np.array([h_vars[i].X for i in range(n_points)])
        h_gurobi = np.clip(h_gurobi, 0, 1)
        current_c5 = correlate_c5(h_gurobi, dx)
        if current_c5 < best_c5:
            best_h = h_gurobi
            best_c5 = current_c5

    # Phase 2: Gradient descent refinement
    learning_steps = 5000
    learning_rate = 0.1
    for step in range(learning_steps):
        if time.time() - start_time >= budget_s:
            break

        grad = np.zeros_like(best_h)
        eps = 1e-5

        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = correlate_c5(h_p, dx)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = correlate_c5(h_n, dx)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        learning_rate *= 0.999

        current_c5 = correlate_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5

    return best_h.tolist(), best_c5, n_points

def correlate_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def normalize_h(h_in):
    clipped = np.clip(h_in, 0, 1)
    if clipped.sum() == 0:
        return h_in.copy()
    scale = (len(h_in) / 2) / clipped.sum()
    scaled = clipped * scale
    return np.clip(scaled, 0, 1)
