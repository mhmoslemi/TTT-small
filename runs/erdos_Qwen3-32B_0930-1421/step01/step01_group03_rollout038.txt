import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        h_init[::2] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)

    def optimize_convex(h0):
        h = cp.Variable(n_points)
        objective_expr = cp.Minimize(cp.max([cp.sum([(h[i] * (1 - h[j])) for i, j in k_indices]) * dx for k_indices in kernel_indices]))
        constraints = [
            cp.sum(h) == constraint_sum,
            0 <= h <= 1
        ]
        prob = cp.Problem(objective_expr, constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            return h0
        return np.array(h.value).flatten()

    kernel_indices = []
    for shift in range(-n_points + 1, n_points):
        tmp = []
        for i in range(n_points):
            j = (i - shift) % n_points
            tmp.append((i, j))
        kernel_indices.append(tmp)

    def perturb_structure(h, times=5):
        h = h.copy()
        indices = np.argsort(h)[::-1][:times]
        h[indices] = 1 - h[indices]
        h = np.clip(h, 0, 1)
        return h * (constraint_sum / h.sum())

    def refine_structure(h):
        h = h.copy()
        idx = np.where(h > 1e-6)[0]
        weights = cp.Variable(len(idx))
        objective = cp.Minimize(cp.max([cp.sum([weights[i] * (1 - weights[j]) for i, j in k_slice]) for k_slice in kernel_slice]))
        constraints = [
            cp.sum(weights) == constraint_sum,
            0 <= weights <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        refinement = np.zeros(n_points)
        refinement[idx] = np.array(weights.value).flatten()
        return refinement

    kernel_slices = []
    for k_indices in kernel_indices:
        slice = []
        for i, j in k_indices:
            slice.append((i, j))
        kernel_slices.append(slice)

    best_h = h_init
    best_c5 = objective(best_h)
    improvements = 0
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < 1000:
        if improvements >= 10:
            break
        iter_count += 1
        if iter_count % 20 == 0 and iter_count > 100:
            h = perturb_structure(best_h)
        else:
            h = best_h
        h = optimize_convex(h)
        h = refine_structure(h)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            improvements += 1
        else:
            improvements = 0

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
