import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Attempt to use initial_h_values if available
    try:
        h0 = initial_h_values.copy()
        if len(h0) != n_points:
            h0.resize(n_points)
    except NameError:
        h0 = np.zeros(n_points)
        h0[42:] = 0.5
        h0 = np.clip(h0, 0, 1)
        h0 = h0 / h0.sum() * n_points / 2

    best_h = h0.copy()
    best_C5 = float('inf')

    start_time = time.time()
    max_time_per_solve = 0.95 * budget_s
    remaining_time = budget_s

    def compute_energy(obj, x, lam, mu):
        """Helper to build objective function dynamically."""
        k_vals = np.arange(-x.shape[0] + 1, x.shape[0])
        norm = (x.shape[0] ** 2) * dx
        overlap_penalty = cp.sum([cp.sum((x + np.roll(x, -k) - 1) ** 2) for k in k_vals])
        energy = lam * overlap_penalty + mu * cp.sum_squares(x)
        return energy / norm

    while remaining_time > max_time_per_solve and best_C5 > 0.38085:
        remaining_time = budget_s - (time.time() - start_time)

        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]

        lam = cp.Parameter(nonneg=True, value=0.1)
        mu = cp.Parameter(nonneg=True, value=0.01)
        obj = cp.Problem(cp.Minimize(compute_energy(h, n_points, lam, mu)), constraints)

        try:
            obj.solve(solver=cp.OSQP, warm_start=True, verbose=False)
            h_vals = np.array(h.value)
        except:
            h_vals = best_h.copy()

        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / h_vals.sum() * n_points / 2

        # Evaluator
        cor = correlate(h_vals, 1 - h_vals, mode="full")
        C5 = np.max(cor) * dx

        if C5 < best_C5:
            best_C5 = C5
            best_h = h_vals.copy()
            lam.value *= 0.98
            mu.value *= 0.98
        else:
            h_vals = best_h.copy() + 0.1 * np.random.randn(n_points)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / h_vals.sum() * n_points / 2

        if (time.time() - start_time) > max_time_per_solve:
            break

    return best_h, best_C5, n_points
