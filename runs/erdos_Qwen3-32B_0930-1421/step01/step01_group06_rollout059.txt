import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    n_points = len(initial_h_values)

    # Normalize initial_h_values to satisfy the sum constraint and bounds
    h_initial = np.copy(initial_h_values)
    total_h_initial = float(np.sum(h_initial))
    if total_h_initial != 0:
        h_initial *= (n_points / 2.0) / total_h_initial
        h_initial = np.clip(h_initial, 0, 1)

    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement_time = start_time

    def convex_relaxation(C5_candidate, h_init):
        dx = 2.0 / n_points
        h = cp.Variable(n_points, nonneg=True)
        z = cp.Variable((n_points, n_points), nonneg=True)

        constraints = [
            cp.sum(h) == n_points / 2,
            h <= 1,
        ]

        for i in range(n_points):
            for j in range(n_points):
                if i + (j - i) < n_points and (j - i) > 0:
                    k = j - i
                    if 1 <= k < n_points:
                        constraints += [
                            z[i][j] <= h[i],
                            z[i][j] <= 1 - h[j],
                            z[i][j] >= h[i] + (1 - h[j]) - 1,
                            z[i][j] >= 0
                        ]
        # Enforcing max correlation at most C5_candidate * dx
        for k in range(1, n_points):
            fl = n_points - k
            constraints += [cp.sum([z[i][i + k] for i in range(fl)]) <= C5_candidate * dx]

        objective = cp.Minimize(cp.Constant(0))  # Feasibility objective

        problem = cp.Problem(objective, constraints)

        # Use feasible Gurobi if available, else fall back to ECOS
        SOLVERS = ['GUROBI', 'ECOS']
        sols = []
        for s in SOLVERS:
            try:
                problem.solve(solver=s, verbose=False)
                if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    sols.append((s, h.value, problem.status))
                break
            except:
                continue

        if sols:
            for solver_name, h_val, status in sols:
                if h_val is not None:
                    h_val = np.clip(h_val, 0, 1)
                    h_val = h_val / np.sum(h_val) * (n_points / 2.0)
                    new_c5 = evaluate_erdos_solution(h_val, n_points)
                    if new_c5 < best_c5:
                        best_h[:] = h_val.copy()
                        best_c5 = new_c5
                        last_improvement_time = time.time()

    # Binary search over C5 with time budgeting
    C_low = 0.0
    C_high = 1.5
    iteration = 0
    max_bisection_steps = 50
    while best_c5 > 1e-6 and iteration < max_bisection_steps and time.time() - start_time < budget_s - 5:
        C_mid = (C_low + C_high) / 2
        convex_relaxation(C_mid, best_h)
        if best_c5 <= C_mid * 1.001:
            C_high = C_mid
        else:
            C_low = C_mid
        iteration += 1

    # Additional rounds with perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        h_pert = np.clip(best_h + np.random.uniform(-0.1, 0.1, n_points), 0, 1)
        h_pert = h_pert / np.sum(h_pert) * (n_points / 2.0)
        convex_relaxation(best_c5, h_pert)
        if best_c5 != evaluate_erdos_solution(best_h, n_points):
            last_improvement_time = time.time()

    # Final clean pass with ECOS in case GUROBI is missing
    for _ in range(3):
        if time.time() - start_time > budget_s - 2:
            break
        convex_relaxation(best_c5 * 1.0, best_h)

    return best_h, best_c5, n_points
