import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Ensure initial h satisfies constraints
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-7:
        h = h / h_sum * target_sum
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    best_time = start_time

    def compute_correlations(h):
        h = np.clip(h, 0, 1).copy()
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h = h / h_sum * target_sum
            h = np.clip(h, 0, 1)
        ones_minus_h = 1 - h
        xy = []
        for k in range(1, n_points):
            if k < n_points:
                # Forward
                corr = np.dot(h[:n_points - k], ones_minus_h[k:])
                xy.append((k, corr * dx))
            if n_points - k > 1:
                # Reverse
                corr = np.dot(h[k:], ones_minus_h[:n_points - k])
                xy.append((-k, corr * dx))
        return np.array(xy)

    def top_contributors(xy, top_k=5):
        xy = np.array(xy)
        idx = np.argsort(-np.abs(xy[:, 1]))
        return xy[idx[:min(top_k, len(idx))]]

    while time.time() - start_time < budget_s - 0:
        # Compute and analyze overlaps
        xy = compute_correlations(h)
        current_c5 = np.max(np.abs(xy[:, 1]))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            best_time = time.time()

        # Early stopping check
        if best_c5 < 0.380876 or time.time() - best_time > 120:
            break

        top_contributions = top_contributors(xy)
        for (k_sign, c) in top_contributions[:5]:
            k = int(abs(k_sign))
            dir_choice = 1 if k_sign > 0 else -1

            if dir_choice == 1:
                contrib = h[:n_points - k] * (1 - h[k:])
            else:
                contrib = h[k:] * (1 - h[:n_points - k])

            contrib = np.abs(contrib)
            top_indices = np.argsort(-contrib)[:max(1, len(contrib) // 10)]

            for idx in top_indices:
                if dir_choice == 1:
                    i = idx
                    j = k + i
                    if h[i] > 0 and h[j] < 1:
                        delta = min(0.02, h[i], 1 - h[j])
                        h[i] -= delta
                        h[j] += delta
                else:
                    i = idx + k
                    j = idx
                    if h[i] > 0 and h[j] < 1:
                        delta = min(0.02, h[i], 1 - h[j])
                        h[i] -= delta
                        h[j] += delta

            # Constrain and normalize
            h.clip(0, 1, out=h)
            h_sum = h.sum()
            if abs(h_sum - target_sum) > 1e-6:
                h = h / h_sum * target_sum
                h.clip(0, 1, out=h)

    final_c5 = np.max(np.abs(compute_correlations(best_h))[:, 1])
    return (best_h, final_c5, n_points)
