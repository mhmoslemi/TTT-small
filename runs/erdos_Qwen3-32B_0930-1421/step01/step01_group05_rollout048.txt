import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Initialize h with `initial_h_values`
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = correlate_bound(h, dx)
    last_update = time.time()
    best_update_freq = 60.0  # seconds to update best and re-check state

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = clipped.sum()
        return np.clip(clipped * (n_points / 2) / total, 0, 1)

    def correlate_bound(h, dx):
        c = correlate(h, 1.0 - h, mode='full') * dx
        return float(np.max(c))

    # Try solving with ILP for 80% of time budget
    ilp_solver_start = time.time()
    ilp_time_limit = 0.8 * budget_s
    while time.time() - start_time < ilp_time_limit:
        try:
            h_var = cp.Variable(n_points, nonneg=True)
            constraints = [h_var <= 1]
            constraints.append(cp.sum(h_var) == n_points / 2)

            # Minimize the worst-case overlap at any shift k
            overlap_vectors = [
                cp.vec((h_var * cp.roll(1.0 - h_var, shift=k)) * dx)
                for k in range(-n_points + 1, n_points)
            ]
            max_overlap = cp.Max(*[ov[0] for ov in overlap_vectors])  # extract scalar values
            objective = cp.Minimize(max_overlap)

            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.SCIP, verbose=False, time_limit=budget_s - time.time() + start_time)

            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_scip = h_var.value
                h_scip = normalize_h(h_scip)
                c5_scip = correlate_bound(h_scip, dx)
                if c5_scip < best_c5:
                    best_c5 = c5_scip
                    best_h = h_scip.copy()
                    best_update = time.time()
        except Exception:
            pass  # skip errors safely

        # Maintain progress
        if time.time() - last_update > best_update_freq:
            last_update = time.time()
            h_scip = normalize_h(h_scip)
            best_c5 = min(best_c5, correlate_bound(h_scip, dx))
            best_h[:] = h_scip.copy()

        if time.time() - start_time >= ilp_time_limit:
            break

    # Refine with adaptive frequency-weighted descent for the rest
    current_h = best_h.copy()
    rweights = 0.8 + 0.2 * np.sin(np.linspace(0, 4.0 * np.pi, n_points))
    lr_schedule = np.linspace(0.15, 0.001, 1000)
    lrate = 0.1
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        delta = 1e-5
        grad = np.zeros_like(current_h)
        h_copy = current_h.copy()
        for j in range(n_points):
            h_p = h_copy.copy()
            h_p[j] += delta
            h_p = normalize_h(h_p)
            c_p = correlate_bound(h_p, dx)
            h_n = h_copy.copy()
            h_n[j] -= delta
            h_n = normalize_h(h_n)
            c_n = correlate_bound(h_n, dx)
            grad[j] = (c_p - c_n) / (2 * delta)
        current_h -= lrate * grad * rweights[j]
        current_h = normalize_h(current_h)
        c5_new = correlate_bound(current_h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = current_h.copy()
        lrate *= 0.975

    return best_h.tolist(), best_c5, n_points
