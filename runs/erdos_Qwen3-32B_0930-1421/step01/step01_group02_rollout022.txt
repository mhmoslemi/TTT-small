```python
import numpy as np
import time
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def generate_initial_h(n_points):
    if initial_h_values is not None:
        return ensure_feasible(np.copy(initial_h_values[:n_points]))
    h = np.zeros(n_points)
    for i in range(0, n_points, 2):
        h[i] = 0.5
    return ensure_feasible(h)


def discretize(h):
    steps = [0.0, 0.25, 0.5, 0.75, 1.0]
    return np.array([steps[np.argmin(np.abs(h_i - np.array(steps)))] for h_i in h])


def branch_and_bound_search(n_points, budget_s=1100):
    start_time = time.time()
    best_h = generate_initial_h(n_points)
    best_h = discretize(best_h)
    best_h = ensure_feasible(best_h)
    best_C5 = evaluate_C5(best_h)
    queue = [(best_h.copy(), 0.0)]
    visited = set()

    while time.time() - start_time < budget_s:
        if not queue:
            break
        current_h, score = queue.pop()
        current_tuple = tuple(np.round(current_h, 6))
        if current_tuple in visited:
            continue
        visited.add(current_tuple)

        if np.isfinite(score):
            h_score = score
        else:
            h_score = evaluate_C5(current_h)
            if h_score < best_C5:
                best_h = np.copy(current_h)
                best_C5 = h_score

        # Branching step
        for i in range(n_points):
            for delta in [-0.25, 0.25]:
                next_h = np.copy(current_h)
                next_h[i] += delta
                next_h = np.clip(next_h, 0, 1)
                next_h = ensure_feasible(next_h)
                next_tuple = tuple(np.round(next_h, 6))
                if next_tuple in visited:
                    continue
                lower_bound = evaluate_C5(lower_bound_h := np.copy(next_h))
                if lower_bound >= best_C5:
                    continue
                queue.append((next_h, lower_bound))
                if len(queue) >= 1000:
                    break

    return best_h, best_C5, n_points


def run(budget_s=1100, n_points=42):
    return branch_and_bound_search(n_points, budget_s)
```
