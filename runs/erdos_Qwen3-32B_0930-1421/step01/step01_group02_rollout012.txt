```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / h.sum()
    return h


def construct_basis_functions(n_centers=10, alpha=4, beta=0.3, n_points=100):
    x = np.linspace(0, 2, n_points)
    centers = np.linspace(0, 2, n_centers)
    psi = []
    for xc in centers:
        psit = np.zeros(n_points)
        dx = x[1] - x[0]
        for i in range(n_points):
            if abs(x[i] - xc) <= beta:
                psit[i] = np.exp(-alpha * (x[i] - xc) ** 2)
        psit /= psit.sum()
        psi.append(psit)
    return np.array(psi), centers


def optimize_coefficients_basis(psi, n_points, dx, max_k_offset=50):
    n_basis = psi.shape[0]
    a = cp.Variable(n_basis, pos=True)  # coefficients

    # Constraint: total integral of h must be 1
    constraint = [a @ (psi.sum(axis=1) * dx) == 1]

    # Prepare all overlap integrals for different k offsets
    c_overlap = []
    for k in range(1, max_k_offset + 1):
        for i in range(n_basis):
            for j in range(n_basis):
                overlap = np.sum(psi[i, :n_points - k] * (1 - psi[j, k:])) * dx
                c_overlap.append(a[i] * a[j] * overlap)
    max_overlap = cp.Variable()
    cost = max_overlap
    constraints = [constraint, max_overlap >= sum(c_overlap)]

    problem = cp.Problem(cp.Minimize(cost), constraints)
    problem.solve(solver=cp.CVXOPT, verbose=False)
    return a.value, max_overlap.value


def perturb_centers(old_centers, delta=0.1):
    centers = np.copy(old_centers)
    idx = np.random.choice(len(centers), size=int(0.2*len(centers)), replace=False)
    centers[idx] += np.random.uniform(-delta, delta, size=len(idx))
    np.clip(centers, 0, 2, out=centers)
    return np.sort(np.unique(centers))


def run(budget_s=1100, n_points=100, n_tries=20):
    start_time = time.time()
    best_h = None
    best_C5 = np.inf

    # Initialize h with initial_h_values or default bump-like shape
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
        h = ensure_feasible(h)
        c5 = evaluate_C5(h)
        if c5 < best_C5:
            best_h = np.copy(h)
            best_C5 = c5
        centers = np.linspace(0, 2, 10)
    else:
        h = np.zeros(n_points)
        h[n_points//2] = n_points / 2  # one bump in the middle
        h = ensure_feasible(h)
        centers = np.linspace(0, 2, 10)

    # Main optimization loop
    tries = 0
    while time.time() - start_time < budget_s and tries < n_tries:
        print(f"Try {tries + 1} / {n_tries} (Time Used: {time.time() - start_time:.2f}s)")
        tries += 1

        perturbed_centers = perturb_centers(centers)
        psi, _ = construct_basis_functions(n_centers=len(perturbed_centers),
                                          n_points=n_points)
        a, c5_candidate = optimize_coefficients_basis(psi, n_points=n_points)
        if a is None or np.isnan(c5_candidate):
            continue

        h_basis = np.sum(a[:, None] * psi, axis=0)
        h_basis = ensure_feasible(h_basis)
        c5 = evaluate_C5(h_basis)

        if c5 < best_C5:
            best_h = h_basis
            best_C5 = c5
            print(f"New best C₅: {best_C5:.8f}")

    return best_h, best_C5, n_points
```
