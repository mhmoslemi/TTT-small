```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None


def evaluate_C5(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1 - h
    corr = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-9:
        h[:] = 0.0
        if len(h) > 0:
            h[0] = (len(h) / 2)
    else:
        h *= (len(h) / 2) / total
    return h


def run(budget_s=1100, n_points=42, solver=cp.CVXOPT):
    t_start = time.time()

    dx = 2.0 / n_points
    h_var = cp.Variable(n_points, nonneg=True)

    objective = cp.Minimize(cp.Constant(1))
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2.0,
    ]

    def build_objective_and_constraints():
        with cp.Problem(cp.Minimize(0)) as sample_prob:
            vec1 = h_var
            vec2 = 1 - h_var
            patterns = []
            for k in range(-n_points + 1, n_points):
                if k < 0:
                    indices = np.arange(-k, n_points)
                    corr = cp.sum(vec1[-k:] * vec2[:k])
                elif k >= n_points:
                    corr = cp.Constant(0)
                else:
                    indices = np.arange(n_points - k)
                    corr = cp.sum(vec1[:n_points - k] * vec2[k:])
                patterns.append(corr * dx)
            max_corr = cp.Maximize(cp.Max(patterns))
            # Objective is to minimize the max overlap
            sample_prob.objective = max_corr
            sample_prob.solve(solver=solver, verbose=False)
        return sample_prob.value

    h = np.empty(n_points)
    if initial_h_values is not None:
        h[:] = initial_h_values[:]
    else:
        h[:] = np.random.rand(n_points)
        h = ensure_feasible(h)

    best_h = h.copy()
    best_C5 = evaluate_C5(h)

    while time.time() - t_start < budget_s:
        h_var.value = h
        C5 = build_objective_and_constraints()

        if C5 < best_C5:
            best_h = h.copy()
            best_C5 = C5

        # Greedy coordinate descent step
        improved = False
        for i in range(n_points):
            for h_candidate in [0.0, 1.0, best_h[i] - 0.05, best_h[i] + 0.05]:
                if 0 <= h_candidate <= 1:
                    h_proposed = best_h.copy()
                    h_proposed[i] = h_candidate
                    h_proposed = ensure_feasible(h_proposed)
                    proposed_C5 = evaluate_C5(h_proposed)
                    if proposed_C5 < best_C5:
                        best_h = h_proposed.copy()
                        best_C5 = proposed_C5
                        improved = True

        if not improved:
            break

    return best_h, best_C5, n_points
```
