import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs_pos = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        overlap = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs_pos.append(overlap)
    corrs_neg = []
    for k in range(1, n_points):
        overlap = np.dot(h[k:], 1 - h[:n_points - k])
        corrs_neg.append(overlap)
    corrs = np.concatenate([corrs_pos, corrs_neg])
    return np.max(corrs) * dx

def enforce_normalization(h, n_points):
    # Clip to valid range and normalize to target sum
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    h = np.clip(h, 0.0, 1.0)
    return h

def objective(h_vec, n_points):
    h = h_vec.reshape(-1)
    return compute_c5(h, n_points)

def generate_cosine_initial(n_points):
    return 0.5 + 0.5 * np.cos(np.pi * np.arange(n_points) / n_points)

def particle_to_h(particle, n_points):
    h = particle[:n_points].copy()
    return enforce_normalization(h, n_points)

def update_particles(particles, velocities, particle_bests, global_best,
                     inertia=0.7, c1=2.0, c2=2.0, bounds=(0.0, 1.0)):
    for i in range(particles.shape[0]):
        r1 = np.random.rand(particles.shape[1])
        r2 = np.random.rand(particles.shape[1])
        velocities[i] = (
            inertia * velocities[i] +
            c1 * r1 * (particle_bests[i] - particles[i]) +
            c2 * r2 * (global_best - particles[i])
        )
        # Apply velocity clamping
        velocities[i] *= 0.5  # prevent large steps
        particles[i] += velocities[i]
        particles[i] = np.clip(particles[i], *bounds)
        #_fix_sum(particles[i], bounds[1])
        particle_bests[i] = np.clip(particle_bests[i] + 0.001 * velocities[i], *bounds)
    return particles, velocities, particle_bests

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 47
    init_h = initial_h_values if 'initial_h_values' in globals() else generate_cosine_initial(n_points)
    h_start = np.clip(init_h, 0, 1)
    h_start = enforce_normalization(h_start, n_points)
    time_start = time.time()

    # PSO parameters
    n_particles = 20
    n_dim = n_points + 2  # +2 for global_best storage
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_dim))
    velocities = np.zeros_like(particles)
    for i in range(n_particles):
        part = particles[i].copy()
        part[:n_points] = h_start + 0.1 * np.random.randn(n_points)  # perturb
        particles[i] = np.clip(part, 0.0, 1.0)
    particles = np.clip(particles, 0.0, 1.0)
    particle_bests = particles.copy()
    global_best = particles[0, :n_points].copy()

    # Evaluate initial global best
    best_c5 = compute_c5(global_best, n_points)
    best_particle = 0
    for i in range(n_particles):
        particles[i, :n_points] = enforce_normalization(particles[i, :n_points], n_points)
        c5_val = compute_c5(particles[i, :n_points], n_points)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_particle = i
            global_best = particles[i, :n_points].copy()

    # Particle Swarm Optimization Loop
    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - time_start > budget_s - 20:
            break

        if iter % max(int(budget_s / max_iter), 1) == 0:
            elapsed = time.time() - time_start
            remaining = budget_s - elapsed
            break

        particles, velocities, particle_bests = update_particles(
            particles, velocities, particle_bests, global_best, inertia=0.7,
            c1=1.5 * (1 - iter / max_iter), c2=1.5
        )
        for i in range(n_particles):
            x = particles[i, :n_points]
            x = enforce_normalization(x, n_points)
            particles[i, :n_points] = x
            c5_val = compute_c5(x, n_points)
            if c5_val < compute_c5(particle_bests[i, :n_points], n_points):
                particle_bests[i, :n_points] = x.copy()
                if c5_val < compute_c5(global_best, n_points):
                    global_best = x.copy()
                    best_c5 = c5_val

        # Perform local refinements on top 5 particles
        top_indices = np.argsort([compute_c5(particles[i, :n_points], n_points) for i in range(n_particles)])[::-1][0:5]
        for i in top_indices:
            if time.time() - time_start > budget_s - 10:
                break
            bounds_local = [(0.0, 1.0)] * n_points
            res = opt.minimize(
                lambda h: compute_c5(h, n_points),
                particles[i, :n_points], bounds=bounds_local, method='L-BFGS-B',
                options={'maxiter': 100, 'eps': 1e-5}
            )
            x_opt = enforce_normalization(res.x, n_points)
            c5_opt = compute_c5(x_opt, n_points)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                global_best = x_opt.copy()

    final_c5 = compute_c5(global_best, n_points)
    return global_best, final_c5, n_points
