import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    n_points = 47 if not 'initial_h_values' in globals() else len(initial_h_values)
    max_n_points = 100
    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
        h_init = h_init.clip(0, 1)  # Ensure values are in [0, 1]
        h_init = h_init * (constraint_sum / h_init.sum())
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= (constraint_sum / h_init.sum())

    def compute_c5(h):
        h = h.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def normalize(h):
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def optimize_convex(h_start):
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == constraint_sum
        ]

        def conv_target(h_var):
            hk = cp.Constant(h_var)
            conv = cp.Correlation(hk, 1 - hk)
            max_overlap = cp.Max(2 * dx / n_points * cp.RollingSum(conv, n_points))
            return max_overlap

        lam = cp.Parameter(nonneg=True)
        lam.value = 1.0

        obj = cp.Minimize(lam)
        problem = cp.Problem(obj, constraints)

        for k in range(n_points - 1):
            problem.constraints.append(conv_target(h)[k] <= lam)

        problem.solve(solver=cp.ECOS, warm_start=True)
        return np.array(h.value)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    h_history = [best_h.copy()]

    def cosine_perturb(h, scale=0.1):
        t = np.linspace(0, 2, n_points)
        noise = scale * (1 - np.cos(np.pi * t))
        h_noise = h + noise
        return normalize(h_noise)

    remaining_budget = budget_s

    while remaining_budget > 5.0:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        remaining_budget = budget_s - (time.time() - start_time)
        if len(h_history) > 10:
            h = h_history[int(np.random.rand() * min(10, len(h_history)))]
        else:
            h = best_h

        if np.random.rand() < 0.7:
            h_candidate = optimize_convex(h)
        else:
            h_candidate = cosine_perturb(h, scale=0.1)

        h_candidate = normalize(h_candidate)
        c5_candidate = compute_c5(h_candidate)

        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            h_history.append(best_h.copy())

        remaining_budget = budget_s - (time.time() - start_time)

    best_h = normalize(best_h)

    return best_h, best_c5, n_points
