import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as a seed
    h0 = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h0 is None:
        n_points = 45
        h0 = np.random.rand(n_points)
        h0 = h0 / h0.sum() * n_points / 2
    else:
        n_points = len(h0)

    dx = 2.0 / n_points

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        scale = (n_points / 2) / h_sum
        return np.clip(h_clipped * scale, 0, 1)

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def firefly_init():
        if h0 is not None:
            df = np.random.normal(loc=0, scale=0.05, size=(50, n_points))
            candidates = np.clip(h0 + df, 0, 1)
            candidates = np.array([repair_h(h) for h in candidates])
            return [h0] + candidates.tolist()
        else:
            fireflies = []
            for _ in range(50):
                h = np.random.rand(n_points)
                h = h / h.sum() * n_points / 2
                h = repair_h(h)
                fireflies.append(h)
            return fireflies

    # Initialize population
    population = firefly_init()
    population = [repair_h(h) for h in population]
    n_pop = len(population)

    best_h = None
    best_c5 = np.inf
    for h in population:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Firefly algorithm parameters
    beta0 = 1.0
    alpha = 0.2
    gamma_factor = 0.1  # Scales distance dependence

    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(n_pop):
            current = population[i]
            current_c5 = compute_c5(current)

            for j in range(n_pop):
                if compute_c5(population[j]) < current_c5:  # j is brighter
                    r = np.linalg.norm(current - population[j])
                    beta = beta0 * np.exp(- (gamma_factor * time.time()) * r**2)
                    step = beta * (population[j] - current) + alpha * (np.random.rand(n_points) - 0.5)

                    new_firefly = repair_h(current + step)
                    new_population.append(new_firefly)

        population = [repair_h(f) for f in new_population] + population
        population = [repair_h(f) for f in np.unique(population, axis=0)]

        # Update global best
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Prune population to original size to prevent explosion
        if len(population) > n_pop:
            scores = np.array([compute_c5(h) for h in population])
            indices = np.argsort(scores)[:n_pop]
            population = [population[i] for i in indices]

    return best_h.tolist(), best_c5, n_points
