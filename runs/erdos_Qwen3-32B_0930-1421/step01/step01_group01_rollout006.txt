import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def compute_overlap(h, dx):
    cor = correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx


def rescale_h(h):
    h = np.clip(h, 0, 1)
    scale = len(h) / 2 / (h.sum() + 1e-10)
    h *= scale
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

        if len(h) != n_points:

    best_h = h.copy()
    best_C5 = compute_overlap(h, dx)
    start_time = time.time()
    max_iter = 1000000  # high bound; we check time
    iter_idx = 0
    perturb_period = 25
    learning_rate = 0.05
    decay_rate = 0.995

    while time.time() - start_time < budget_s and iter_idx < max_iter:
        iter_idx += 1

        h = rescale_h(h)
        h[0] = max(0, h[0])  # diagnostic clipping
        cor = correlate(h, 1 - h, mode='full')
        max_id = np.argmax(cor)
        max_offset = max_id - len(h) + 1
        grad = np.zeros_like(h)

        for i in range(len(h)):
            k_index = i + max_offset
            if 0 <= k_index < len(h):
                grad[i] = 2 * h[i] - 2 * h[k_index]
            else:
                grad[i] = 2 * h[i] - 1

        # Gradient step
        h -= learning_rate * grad
        h = rescale_h(h)

        # Evaluate new C5
        new_C5 = compute_overlap(h, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()

        # Perturbation if stuck
        if iter_idx % perturb_period == 0:
            if np.random.rand() < 0.3:
                h = rescale_h(h + 0.1 * (np.random.rand(n_points) * 2 - 1))
                h[0] = max(0, h[0])

        # Safeguard learning rate
        learning_rate *= decay_rate

    return best_h, best_C5, n_points
