```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
import math

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h[:] = 0.0
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def run(budget_s=1100, n_points=80, n_cross=8, n_restarts=4):
    start_time = time.time()
    best_C5 = np.inf
    best_h = None

    if initial_h_values is not None:
        best_h = np.copy(initial_h_values)
        best_C5 = evaluate_C5(best_h)
    else:
        h0 = np.zeros(n_points)
        h0[::2] = 1.0
        h0 = ensure_feasible(h0)
        best_h = h0
        best_C5 = evaluate_C5(best_h)

    # Warm start options
    ref_h_list = [best_h]

    for _ in range(n_cross + n_restarts):
        while time.time() - start_time < budget_s:
            # Random initialization
            if len(ref_h_list) == 0:
                h = np.random.rand(n_points)
                h = ensure_feasible(h)
            else:
                ref_idx = np.random.randint(len(ref_h_list))
                h = np.copy(ref_h_list[ref_idx])

            h = cp.Variable(n_points, nonneg=True)
            constraints = [
                h <= 1,
                cp.sum(h) == n_points / 2
            ]

            ones = np.ones(n_points)
            ones_minus_h = 1 - h

            # Define cross correlations of (h, 1-h)
            indices = np.arange(n_points)
            corr_kernels = []
            for k in range(-n_points + 1, n_points):
                if k < 0:
                    corr = cp.sum(ones_minus_h[n_points + k:] * h[:n_points + k])
                elif k > 0:
                    corr = cp.sum(ones_minus_h[:n_points - k] * h[k:])
                else:
                    corr = cp.sum(ones_minus_h * h)
                corr_kernels.append(corr)

            # Instead of exact max, we use convex upper bound,
            # i.e., we minimize the maximum kernel over shifted convolutions
            lam = cp.Variable()
            for corr in corr_kernels:
                constraints.append(corr * (2.0 / n_points) <= lam)

            obj = cp.Minimize(lam)
            problem = cp.Problem(obj, constraints)

            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = np.clip(np.array(h.value), 0, 1)
                if np.abs(np.sum(new_h) - n_points / 2) > 1e-6:
                    new_h = ensure_feasible(new_h)
                new_C5 = evaluate_C5(new_h)

                ref_h_list.append(new_h)
                if new_C5 < best_C5:
                    best_h = new_h
                    best_C5 = new_C5
            else:
                continue

        if time.time() - start_time >= budget_s:
            break

    return best_h, best_C5, n_points
```
