import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        overlap = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(overlap)
    for k in range(1, n_points):
        if n_points - k > 0:
            overlap = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(overlap)
    return (np.max(np.array(corrs)) * dx) if corrs else 0.0

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def raised_cosine(n):
    return np.clip(0.5 + 0.5 * np.cos(np.pi * np.linspace(0, 2, n + 2)[1:-1]), 0, 1)

def pso_run(h0, n_points, budget_s):
    dx = 2.0 / n_points
    particles = 20
    inert_w = 0.7
    c1 = 2.0
    c2 = 2.0
    if h0 is None:
        h0 = raised_cosine(n_points)
    h0 = h0.copy()
    particles_heroes = [enforce_normalization(h0 + np.random.uniform(-0.1, 0.1, n_points), n_points) for _ in range(particles)]
    particles = [p.copy() for p in particles_heroes]
    velocities = [np.random.rand(n_points) * 0.05 for _ in range(particles)]
    best_c5 = float('inf')
    best_particle = None

    start_time = time.time()
    while time.time() - start_time < budget_s * 0.9:
        for i in range(particles):
            h_p = particles[i]
            vel = velocities[i]
            c5 = compute_c5(h_p, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_particle = h_p.copy()
        r1 = np.random.rand()
        r2 = np.random.rand()
        for i in range(particles):
            vel = velocities[i]
            pos = particles[i]
            best = particles_heroes[i]
            gbest = best_particle
            vel = (inert_w * vel +
                   c1 * r1 * (best - pos) +
                   c2 * r2 * (gbest - pos))
            velocities[i] = vel * 0.999
            pos += vel
            pos = enforce_normalization(pos, n_points)
            particles[i] = pos.copy()
    return best_particle, best_c5

def refine_with_bfgs(h_start, n_points, budget_left):
    bounds = [(0, 1)] * n_points

    def objective(x):
        h = x.copy()
        return compute_c5(h, n_points)

    result = opt.minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200, 'eps': 1e-5})
    if result.success:
        refined = result.x
    else:
        refined = h_start
    refined = enforce_normalization(refined, n_points)
    return refined

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is not None:
        n_points = len(init_h)
        h0 = init_h.copy()
    else:
        n_points = 53  # use a different n for more exploration
        h0 = raised_cosine(n_points).copy()
    h0 = enforce_normalization(h0, n_points)

    # Step 1: PSO global exploration
    h_pso, c5_pso = pso_run(h0, n_points, budget_s)

    # Step 2: Local refinement with BFGS
    remaining_time = budget_s - max(0, time.time() - time.time())
    h_bfgs = refine_with_bfgs(h_pso, n_points, remaining_time)

    # Final evaluation
    final_c5 = compute_c5(h_bfgs, n_points)
    return h_bfgs, final_c5, n_points
