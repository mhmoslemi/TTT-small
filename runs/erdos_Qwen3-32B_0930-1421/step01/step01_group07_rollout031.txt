import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def adjust_h(h):
    indices = np.random.choice(len(h), size=np.random.randint(5, 11), replace=False)
    h_new = h.copy()
    h_new[indices] += (np.random.normal(0, 0.2, len(indices)))
    h_new = np.clip(h_new, 0, 1)
    # Project back onto the sum constraint
    total_weight = np.sum(h_new)
    if abs(total_weight) > 1e-8:
        scale = (len(h) / 2) / total_weight
        h_new *= scale
    else:
        h_new /= len(h) / 2
    return h_new

                best_h = current_h.copy()
                best_c5 = c5_new
