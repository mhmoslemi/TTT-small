import numpy as np
import time
import scipy.optimize as opt
import scipy.signal
import math


def perturb_h(h, step_size=0.1, segment_frac=0.25):
    n = len(h)
    idxs = np.random.choice(n, size=int(segment_frac * n), replace=False)
    h_perturbed = h.copy()
    magnitude = np.random.normal(0, step_size, size=h_perturbed.shape)
    h_perturbed[idxs] += magnitude[idxs]
    # Clamp to [0, 1]
    h_clamped = np.clip(h_perturbed, 0, 1)
    # Rescale to preserve total mass
    h_rescaled = h_clamped * (n / 2) / h_clamped.sum()
    return h_rescaled


def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / h.sum()
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5

    best_C5 = evaluate_C5(best_h, dx)
    T_initial = 1.0
    alpha = 0.99
    max_steps = int(budget_s * 0.9)
