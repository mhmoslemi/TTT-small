import numpy as np
import time
from scipy.optimize import minimize
from copy import copy

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / h.sum()
        return h

    # PSO parameters
    num_particles = 50
    max_vel = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_iter = int((budget_s) * 0.9 // (10 * n_points) + 1)
    reset_fraction = 0.1

    # Initialize
    particles = np.array([project(initial_h + np.random.normal(0.0, 0.1, n_points)) for _ in range(int(num_particles * 0.9))])
    for _ in range(int(num_particles * 0.1)):
        particles = np.vstack((particles, project(initial_h)))

    velocities = np.random.uniform(-max_vel, max_vel, (num_particles, n_points))
    particles = np.array([project(p) for p in particles])

    p_best = particles.copy()
    p_best_values = np.array([evaluate_erdos_solution(p, n_points) for p in p_best])
    g_best = p_best[p_best_values.argmin()].copy()
    best_c5 = p_best_values.min()

    last_improvement_time = time.time()

    for i in range(max_iter):
        if time.time() - last_improvement_time > 120 or time.time() - start_time > budget_s - 5:
            break

        r1 = np.random.uniform(0, 1, (num_particles, 1))
        r2 = np.random.uniform(0, 1, (num_particles, 1))
        w = w_start - (w_start - w_end) * (i / max_iter)

        velocities = w * velocities + c1 * r1 * (p_best - particles) + c2 * r2 * (g_best[None, :] - particles)
        np.clip(velocities, -max_vel, max_vel, out=velocities)
        particles += velocities
        particles = np.array([project(p) for p in particles])

        # Evaluate current values
        current_values = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
        for j in range(num_particles):
            if current_values[j] < p_best_values[j]:
                p_best[j] = particles[j].copy()
                p_best_values[j] = current_values[j]
                if current_values[j] < best_c5:
                    g_best = particles[j].copy()
                    best_c5 = current_values[j]
                    last_improvement_time = time.time()

        # Reset a fraction of the worst particles
        worst_indices = current_values.argsort()[-int(num_particles * reset_fraction):]
        for idx in worst_indices:
            particles[idx] = project(initial_h + np.random.normal(0.0, 0.1, n_points))

    return g_best, best_c5, n_points
