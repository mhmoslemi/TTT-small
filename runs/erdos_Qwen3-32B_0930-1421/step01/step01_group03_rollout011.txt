import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def lin_constraints(h, C):
        h = h.reshape(n_points)
        constraints = []

        for k in range(1, n_points):
            shift_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shift_h)) * dx - C
            constraints.append(overlap <= 0)
        return constraints

    def solve_lp(h0, dx, n_points):
        h = cp.Variable(n_points, nonneg=True)
        C = cp.Variable()

        objective = C

        constraints = [cp.sum(h) == constraint_sum]
        for k in range(1, int(n_points / 2) + 2):
            shift_h = np.roll(h0, k % n_points)
            overlap = np.sum(h * (1 - shift_h)) * dx
            constraints.append(overlap <= C)
            constraints.append(h <= 1)
            constraints.append(h >= 0)

        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        h_sol = h.value
        if h_sol is None:
            return h0
        h_sol = np.clip(h_sol, 0, 1)
        h_sol *= constraint_sum / h_sol.sum()
        return h_sol

    def perturb(h, scale_perturbation=0.05):
        h = h.copy()
        h += np.random.randn(n_points) * scale_perturbation
        h = np.clip(h, 0, 1)
        h *= constraint_sum / h.sum()
        return h

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Print only for debugging
    # print("Initial C5:", best_c5)

    max_iterations = 100
    iteration = 0

    while time.time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        iteration += 1

        h_sol = solve_lp(best_h, dx, n_points)
        current_c5 = objective(h_sol)

        if current_c5 < best_c5:
            best_h = h_sol.copy()
            best_c5 = current_c5

        if iteration % 5 == 0:
            h_sol = perturb(best_h)
            h_sol = solve_lp(h_sol, dx, n_points)
            current_c5 = objective(h_sol)
            if current_c5 < best_c5:
                best_h = h_sol.copy()
                best_c5 = current_c5

    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
