import numpy as np
import time
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr1 = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr1)
    for k in range(1, n_points):
        corr2 = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr2)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2.0) / total
    h = np.clip(h, 0.0, 1.0)
    return h

def perturb_h(h, noise_scale, n_points):
    h_new = h + np.random.normal(0, noise_scale, len(h))
    h_new = enforce_normalization(h_new, n_points)
    return h_new

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = enforce_normalization(h, n_points)
    best_h = h.copy()
    best_val = compute_c5(best_h, n_points)

    # Collect initial samples for GP training
    X = []
    y = []
    for _ in range(20):
        h_sample = perturb_h(best_h, 0.2, n_points)
        score = compute_c5(h_sample, n_points)
        X.append(h_sample)
        y.append(score)
    X = np.array(X)
    y = np.array(y)

    # Setup GP
    kernel = ConstantKernel(1.0)**2 * RBF(length_scale=1.0)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-8, n_restarts_optimizer=5)

    time_start = time.time()
    temp = 1000.0
    temp_decay = 0.995

    # Main loop
    while time.time() - time_start < budget_s - 20:
        # Train GP
        gp.fit(X, y)

        # Propose new candidate via GP with exploration strategy (upper confidence bound)
        ucb = lambda x: gp.predict(x)[0] - 1.96 * np.sqrt(gp.predict(x, return_std=True)[1])
        def neg_ucb(x_vec):
            return -ucb(x_vec.reshape(1, -1))
        x_test = h + 0.05 * np.random.randn(n_points)
        x_test = enforce_normalization(x_test, n_points)
        proposed = minimize(neg_ucb, x_test, bounds=[(0, 1)]*n_points).x
        proposed = enforce_normalization(proposed, n_points)

        # Evaluate proposed
        val = compute_c5(proposed, n_points)
        if val < best_val or np.random.rand() < np.exp((best_val - val) / max(temp, 1e-2)):
            best_h = proposed.copy()
            best_val = val
        X = np.vstack((X, proposed))
        y = np.append(y, best_val)

        # Cool temperature
        if len(X) > 150:
            X = X[-100:]  # Limit memory to last 100 points to manage training cost
            y = y[-100:]
        temp *= temp_decay

    return best_h, best_val, n_points
