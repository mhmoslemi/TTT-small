import numpy as np
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h)

    # Binary initialization
    n_target = n_points // 2
    h_binary = np.zeros(n_points, dtype=np.float64)
    indices = np.random.choice(n_points, size=n_target, replace=False)
    h_binary[indices] = 1.0

    # Simulated annealing
    T0 = 1.0
    cooling = 0.99
    max_iter = 5000
    best_h_binary = h_binary.copy()
    best_c5_binary = evaluate_erdos_solution(best_h_binary, n_points)

    current_h = h_binary.copy()
    current_c5 = best_c5_binary
    current_sum = np.sum(current_h)

    def anneal():
        nonlocal current_h, current_c5, best_h_binary, best_c5_binary, T0
        t = 0
        temp = T0
        while t < max_iter and time.time() - start_time < budget_s:
            # Flip a random bit maintaining sum constraint
            idx0 = np.random.choice(np.where(current_h == 1)[0])
            idx1 = np.random.choice(np.where(current_h == 0)[0])
            new_h = current_h.copy()
            new_h[idx0], new_h[idx1] = 0, 1
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            new_c5_delta = new_c5 - current_c5
            if new_c5_delta < 0 or np.random.rand() < np.exp(-new_c5_delta / temp):
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5_binary:
                    best_h_binary = new_h.copy()
                    best_c5_binary = new_c5
            temp = temp * cooling
            t += 1

    anneal()

    # Continuous scaling
    h = best_h_binary.copy()
    sum_h = np.sum(h)
    if sum_h < n_target:
        idx = np.random.choice(np.where(h == 0)[0])
        h[idx] = n_target / sum_h - 1.0
    elif sum_h > n_target:
        idx = np.random.choice(np.where(h == 1)[0])
        h[idx] = 2.0 - n_target / sum_h

    # Ensure exact constraint
    total_h = np.sum(h)
    scale = n_target / total_h
    h = np.minimum(np.maximum(h * scale, 0.0), 1.0)

    # Continuous refinement
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()
    last_improvement_time = time.time()

    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_target

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    cons = (
        {"type": "eq", "fun": constraint_sum},
        {"type": "ineq", "fun": constraint_upper},
        {"type": "ineq", "fun": constraint_lower}
    )

    def callback(x):
        nonlocal last_improvement_time
        current_c5 = evaluate_erdos_solution(x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = x.copy()
            last_improvement_time = time.time()

    while time.time() - start_time < budget_s:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints=cons,
            callback=callback,
            options={'maxiter': 50}
        )
        if not result.success:
            break
        if time.time() - last_improvement_time > 60:
            break

    return best_h, best_c5, n_points
