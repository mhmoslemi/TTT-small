import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64).copy() if 'initial_h_values' in globals() else np.random.rand(45)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h  # Normalize to satisfy sum constraint

    dx = 2.0 / n_points

    def normalize_hvec(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / h.sum()

    def compute_c5(h):
        expanded = np.concatenate([h, np.zeros(n_points - 1)])  # For shifting
        rev = 1 - h
        max_overlap = 0.0
        for k in range(n_points):
            shifted = np.roll(rev, k)
            truncated = shifted[:n_points]
            overlap = np.sum(h * truncated) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    def compute_gradient(h, k):
        rev = 1 - h
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k + i < n_points:
                grad[i] += rev[k + i]
            grad[i] -= rev[i]
        return grad * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    learning_rate = 0.1

    while time.time() - start_time < 0.7 * budget_s:
        # Identify max shift k
        max_overlap = 0.0
        max_k = 0
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            truncated = shifted[:n_points]
            overlap = np.sum(h * truncated) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                max_k = k

        # Compute gradient at max_k
        grad = compute_gradient(h, max_k)

        # Line search
        step_size = learning_rate
        for _ in range(5):
            h_candidate = h - step_size * grad
            h_candidate = normalize_hvec(h_candidate)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = min(best_c5, c5_candidate)
                break
            else:
                step_size *= 0.5

        # Reduce learning rate periodically
        learning_rate *= 0.995
        if time.time() - start_time >= 0.99 * budget_s:
            break

    # Final pass with CVXPY for refinement
    objective = cp.Constant(best_c5)
    H = cp.Variable(n_points)
    objective = cp.Constant(0.0)

    constraints = [0 <= H, H <= 1, cp.sum(H) == n_points / 2]
    for k in range(n_points):
        shifted = np.roll(1 - H, k)
        truncated = shifted[:n_points]
        obj_k = cp.sum(H * truncated) * dx
        objective = cp.maximum(objective, obj_k)

    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCIP, max_iters=100, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = H.value * 1.0
            h = normalize_hvec(h)
            final_c5 = compute_c5(h)
            best_c5 = min(final_c5, best_c5)
        else:
            final_c5 = best_c5
    except Exception:
        pass

    return h.tolist(), best_c5, n_points
