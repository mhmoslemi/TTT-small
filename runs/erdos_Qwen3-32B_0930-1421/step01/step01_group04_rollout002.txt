import numpy as np
import scipy.optimize as opt
import time
import random

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def objective_smooth(h_vec, n_points, beta=100):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    corrs = []
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)
    return np.log(np.sum(np.exp(beta * np.array(corrs) * dx))) / beta

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        h.fill(0.5)
    else:
        h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def init_cosine(n_points):
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * (x**2))
    h = enforce_normalization(h, n_points)
    return h

def pso_update_position(pos, vel, pbest, gbest, w, c1, c2):
    r1 = np.random.rand(len(pos))
    r2 = np.random.rand(len(pos))
    vel = w * vel + c1 * r1 * (pbest - pos) + c2 * r2 * (gbest - pos)
    pos = pos + vel
    return pos, vel

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    random.seed(seed)

    n_points = 47 if init_h is None else len(init_h)
    if init_h is None:
        h0 = init_cosine(n_points)
    else:
        h0 = enforce_normalization(init_h, n_points)

    n_particles = 20
    positions = []
    velocities = []
    pbest = []
    pbest_values = []

    for _ in range(n_particles):
        rand_init = h0 + np.random.normal(0, 0.1, n_points)
        rand_init = enforce_normalization(rand_init, n_points)
        positions.append(rand_init.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
        val = objective_smooth(positions[-1], n_points)
        pbest.append(rand_init.copy())
        pbest_values.append(val)

    gbest = pbest[np.argmin(pbest_values)].copy()
    gbest_value = np.min(pbest_values)

    time_start = time.time()
    remaining_time = budget_s
    w = 0.9
    c1 = 2.0
    c2 = 2.0

    best_h = gbest.copy()
    best_c5 = compute_c5(best_h, n_points)

    settings = dict(beta=100, eps=1e-4, maxiter=1000)

    while time.time() - time_start < budget_s:
        for i in range(n_particles):
            pos = positions[i].copy()
            vel = velocities[i].copy()
            pos, vel = pso_update_position(pos, vel, pbest[i], gbest, w, c1, c2)
            pos = enforce_normalization(pos, n_points)
            positions[i] = pos
            vel = np.clip(vel, -0.2, 0.2)
            velocities[i] = vel
            val = objective_smooth(positions[i], n_points)
            if val < pbest_values[i]:
                pbest[i] = positions[i].copy()
                pbest_values[i] = val

        if pbest_values:
            gbest_index = np.argmin(pbest_values)
            new_gbest_val = pbest_values[gbest_index]
            new_gbest = pbest[gbest_index].copy()

            if new_gbest_val < gbest_value:
                gbest = new_gbest.copy()
                gbest_value = new_gbest_val

        if (time.time() - time_start + 4) < budget_s:
            local_i = np.argmin(pbest_values)
            lbfgs_result = opt.minimize(
                lambda x: objective_smooth(x, n_points),
                pbest[local_i].copy(),
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                options=settings
            )
            h_test = enforce_normalization(lbfgs_result.x, n_points)
            c5_test = compute_c5(h_test, n_points)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test

        w = max(0.4, w * 0.999)
        remaining_time = budget_s - (time.time() - time_start)

    return best_h, best_c5, n_points
