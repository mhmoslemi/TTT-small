```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def construct_initial(n_points):
    if initial_h_values is not None and len(initial_h_values) == n_points:
        return ensure_feasible(initial_h_values)
    h = np.random.rand(n_points)
    return ensure_feasible(h)


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    h = construct_initial(n_points)
    best_h = h
    best_C5 = evaluate_C5(h)
    dx = 2.0 / n_points
    idx = np.arange(n_points)

    while time.time() - start_time < budget_s:

        # Build convex optimization variables
        h_var = cp.Variable(n_points, nonneg=True)
        ones = np.ones(n_points)
        overlap = [correlate(h_var, 1 - h_var, mode="full")[k] * dx for k in range(1, n_points)]

        # Optimization problem formulation
        max_overlap = cp.Variable()
        constraints = [
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        for c in overlap:
            constraints.append(c <= max_overlap)

        objective = cp.Minimize(max_overlap)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CPLEX)

        # Extract solution if it's better
        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
            h_new = ensure_feasible(h_new)
            c5_new = evaluate_C5(h_new)
            if c5_new < best_C5:
                best_h = h_new
                best_C5 = c5_new

    return best_h, best_C5, n_points
```
