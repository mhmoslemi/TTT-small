import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time

    def correlate_objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full') * (2.0 / n_points))

    def optimize_with_cvxpy():
        nonlocal h, best_c5, best_h, last_improvement_time
        x = cp.Variable(n_points)
        dx = 2.0 / n_points
        overlap_obj = np.zeros(n_points*2 - 1)
        for k in range(n_points*2 - 1):
            # Use circular shift to construct indices for convolution-like sum
            corr = 0
            for i in range(n_points):
                j = (i + k - n_points + 1) % n_points  # circular shift
                corr += x[i] * (1 - x[j])
            overlap_obj[k] = corr
        cvar = cp.Variable()
        obj = cp.Minimize(cvar)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1,
        ]
        for val in overlap_obj:
            constraints.append(val <= cvar)

        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, maxiters=1000, kktsolver="robust")

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(x.value, 0, 1).ravel()
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h.copy()
                last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 5:
        optimize_with_cvxpy()
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
