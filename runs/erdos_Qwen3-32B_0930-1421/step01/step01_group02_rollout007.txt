import numpy as np
import time
from scipy.signal import correlate


def ensure_feasible_discrete(h):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if np.abs(total - h.size / 2) > 1e-6:
        h *= (h.size / 2) / np.sum(h)
    return h


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corrs = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(corrs) * dx
    return max_corr


def discrete_perturb(h, levels=[0.0, 0.25, 0.5, 0.75, 1.0], n_perturb=3):
    h_new = h.copy()
    indices = np.random.choice(len(h), n_perturb, replace=False)
    for idx in indices:
        val = np.random.choice(levels)
        h_new[idx] = val
    return ensure_feasible_discrete(h_new)
