# Helper to evaluate the overlap integral C₅
def compute_c5(h, dx):
    h = h.copy()
    h_sum = h.sum()
    h = h.clip(0, 1)
    target_sum = h.size / 2
    if abs(h_sum - target_sum) > 1e-6:
        h *= target_sum / h_sum

    ones_minus_h = 1.0 - h

    max_corr = -np.inf

    for k in range(1, h.size):
        if k >= h.size:
            continue
        a = h[:h.size - k]
        b = ones_minus_h[k:]
        corr = np.dot(a, b)
        max_corr = max(max_corr, corr * dx)

    for k in range(1, h.size):
        if k >= h.size:
            continue
        b = h[k:]
        a = ones_minus_h[:h.size - k]
        corr = np.dot(a, b)
        max_corr = max(max_corr, corr * dx)

    return max_corr
