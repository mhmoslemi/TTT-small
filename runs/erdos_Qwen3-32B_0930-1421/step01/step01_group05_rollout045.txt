import numpy as np
import time
from scipy.signal import correlate
import copy

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_1 = 1 - h
    c = correlate(h, h_1, mode='full') * dx
    return np.max(c)

def normalize_h(h_in, n_points):
    clipped = np.clip(h_in, 0, 1)
    sum_h = np.sum(clipped)
    scale = (n_points / 2) / sum_h
    return np.clip(clipped * scale, 0, 1)

def compute_gradient(h, idx, perturbation=1e-5, n_points=None):
    if n_points is None:
        n_points = len(h)
    dx = 2.0 / n_points
    h_up = h.copy()
    h_down = h.copy()
    
    # Ensure we only perturb within bounds
    if h[idx] < 1 - perturbation:
        h_up[idx] += perturbation
    else:
        h_up[idx] = 1.0
    
    if h[idx] > perturbation:
        h_down[idx] -= perturbation
    else:
        h_down[idx] = 0.0
    
    h_up = normalize_h(h_up, n_points)
    h_down = normalize_h(h_down, n_points)
    
    c_up = evaluate_c5(h_up)
    c_down = evaluate_c5(h_down)
    return (c_up - c_down) / (2 * perturbation)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # If not available, fallback to a random initial guess
        n_points = 45
        h = np.random.rand(n_points)
        h = normalize_h(h, n_points)
    n_points = len(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    def optimize_directions():
        nonlocal best_h, best_c5
        learning_rate = 0.1
        while time.time() - start_time < budget_s:
            grad = np.zeros(n_points)
            for i in range(n_points):
                grad[i] = compute_gradient(h, i, n_points=n_points)
            descent_dir = -np.sign(grad) * np.abs(grad)
            perturbed_h = h + learning_rate * descent_dir
            perturbed_h = normalize_h(perturbed_h, n_points)
            current_c5 = evaluate_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = current_c5
            h = perturbed_h
            # Reduce step size adaptively
            learning_rate *= 0.999

    optimize_directions()

    return best_h.tolist(), best_c5, n_points
