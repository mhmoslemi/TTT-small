import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h
