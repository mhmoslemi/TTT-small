```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum < 1e-9:
        h.flat[0] = 1.0
    else:
        h *= (len(h) / 2) / h_sum
    return h


def project_to_binary_layouts(h):
    n_points = len(h)
    h = np.clip(h, 0, 1)
    # Resample to floor(n_points / 2) ones at the brightest indices
    top_indices = np.argsort(h)[::-1][: int(n_points / 2)]
    h_binary = np.zeros_like(h)
    h_binary[top_indices] = 1.0
    return h_binary


def cut_plane_strategy(h_values, n_points, current_bound):
    h = cp.Variable(n_points)
    lamb = cp.Variable()

    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Max overlap expressions over the relevant shifts
    def get_overlap_expr(shift):
        h1 = h[: n_points - shift]
        h2 = h[shift:]
        return cp.sum(cp.multiply(h1, 1 - h2))

    # Start with a base constraint set
    all_shifts = range(1, n_points // 2 + 1)
    active_shifts = [1, n_points // 4, n_points // 2]  # seed with important shifts

    # Add cuts iteratively to minimize λ while reducing discrepancy
    for k in active_shifts:
        constraints.append(get_overlap_expr(k) <= lamb)

    problem = cp.Problem(cp.Minimize(lamb), constraints)
    problem.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.ECOS, verbose=False)
    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h_values  # fallback
    h_sol = np.clip(np.array(h.value).flatten(), 0, 1)
    h_sol = ensure_feasible(h_sol)

    h_binary = project_to_binary_layouts(h_sol)
    new_bound = evaluate_C5(h_binary)

    if new_bound < current_bound:
        return h_binary, new_bound
    return h_values, current_bound


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[: (n_points // 2)] = 1.0  # uniform initial

    best_C5 = evaluate_C5(h)
    best_h = np.copy(h)

    while time.time() - start_time < budget_s:
        try:
            h, new_C5 = cut_plane_strategy(h, n_points, best_C5)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = np.copy(h)
        except:
            pass  # avoid solver crashes

    return best_h, best_C5, n_points
```
