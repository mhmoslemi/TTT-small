import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize to ensure sum constraint
    total_h = float(np.sum(h))
    scale_factor = target_sum / total_h
    h = np.minimum(h * scale_factor, 1.0)
    h = np.maximum(h, 0.0)

    # Ensure exact normalization
    h = h / np.sum(h) * target_sum

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    def make_convex_relaxation(eps=1e-3):
        x = cp.Variable(n_points)
        correlations = np.correlate(x, 1 - x, mode='full')
        max_correlation = cp.Max(cp.multiply(correlations, dx))
        constraints = [
            cp.sum(x) == target_sum,
            0 <= x,
            x <= 1,
            max_correlation <= best_c5 + eps
        ]
        objective = cp.Minimize(max_correlation)
        prob = cp.Problem(objective, constraints)
        return prob

    scales = [0.5, 0.7, 0.8, 0.9, 1.0]
    epsilons = [1e-4, 1e-3, 5e-3]

    while time.time() - start_time < budget_s - 5:
        for eps in epsilons:
            if time.time() - start_time > budget_s - 5:
                break
            try:
                prob = make_convex_relaxation(eps=eps)
                result = prob.solve(solver=cp.CVXOPT)
                if prob.status == cp.OPTIMAL or prob.status in (cp.OPTIMAL_INACCURATE, cp.OPTIMAL_INCERTAIN, cp.DECOMP_INFEASIBLE):
                    new_h = np.clip(prob.solution[x], 0.0, 1.0).flatten()
                    new_sum = np.sum(new_h)
                    if new_sum > 0:
                        new_h = new_h / new_sum * target_sum
                    new_c5 = evaluate_erdos_solution(new_h, n_points)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h[:] = new_h
                        last_improvement_time = time.time()
            except:
                continue  # Skip solver errors, try next iteration
            # Update solver parameters if improvement stalled
            if time.time() - last_improvement_time > 60:
                last_improvement_time = time.time()

    return best_h, best_c5, n_points
