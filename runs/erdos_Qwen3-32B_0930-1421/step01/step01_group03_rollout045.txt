import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    max_n_points = 100

    # Use parent solution if provided
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        best_h = np.array(initial_h_values)
    else:
        n_points = 47
        best_h = np.zeros(n_points)
        for i in range(n_points // 2):
            best_h[i + i] = 2.0 / n_points  # basic construction
        best_h *= n_points / 2 / best_h.sum()

    dx = 2.0 / n_points

    # Constraint: sum(h) == n_points / 2
    constraint_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        perturbation = np.random.randn(len(h)) * noise_factor
        h = h + perturbation
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def crossover(h1, h2):
        mask = np.random.rand(len(h1)) < 0.5
        child = np.where(mask, h1, h2)
        child = child.clip(0, 1)
        child = child * (constraint_sum / child.sum())
        return child

    def repair(h, noise_factor=0.05):
        h = h.copy()
        h = h.clip(0, 1)
        h = h + np.random.randn(len(h)) * noise_factor
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def optimize_h(h):
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = h_opt.clip(0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    # Initialize population
    population_size = 8
    population = [best_h]
    for _ in range(population_size - 1):
        if 'initial_h_values' in globals():
            candidate = initial_h_values.copy()
        else:
            candidate = np.random.rand(n_points)
        candidate = perturb_h(candidate)
        population.append(candidate)

    best_c5 = objective(best_h)

    # Evolution loop
    while time.time() - start_time < budget_s * 0.9:
        # Evaluate
        scores = [objective(h) for h in population]
        valid_scores = [(i, s) for i, s in enumerate(scores) if not np.isnan(s)]
        if not valid_scores:
            break

        # Sort by score
        valid_scores.sort(key=lambda x: x[1])
        elite_idx = valid_scores[0][0]

        # Keep top 1/3 for elitism
        elite_count = max(1, len(population) // 3)
        elite_indices = [valid_scores[i][0] for i in range(elite_count)]
        elite = [population[i] for i in elite_indices]

        # Crossover and mutate
        new_population = elite[:]
        for _ in range(population_size - elite_count):
            parent1 = population[np.random.choice(elite_indices)]
            parent2 = population[np.random.choice(population_size)]
            child = crossover(parent1, parent2)
            child = perturb_h(child)
            new_population.append(child)

        population = new_population

        # Improve best candidates via gradient descent
        for i in range(min(2, len(population))):
            h_opt = optimize_h(population[i])
            population[i] = h_opt

        # Update best solution
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final refinement
    if time.time() - start_time < budget_s:
        best_h = optimize_h(best_h)
        best_c5 = objective(best_h)

    # Final constraint check
    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    elapsed = time.time() - start_time
    print(f"Final objective C5: {best_c5:.8f} in {elapsed:.2f} seconds.")

    return best_h, best_c5, n_points
