import numpy as np
import time
import opt

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def evaluate_overlap(h, n_points, k):
    return np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    max_overlap = 0.0
    for k in range(1, n_points):
        overlap = evaluate_overlap(h, n_points, k)
        if overlap > max_overlap:
            max_overlap = overlap
    for k in range(1, n_points):
        overlap = evaluate_overlap(h, n_points, k)
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)

    # Use the initial construction if available, or fall back to random
    n_points = len(init_h) if init_h is not None else 47
    if init_h is None:
        comb_period = n_points // 4
        h_init = np.zeros(n_points)
        h_init[::comb_period] = 1.0 / comb_period
        h_init = enforce_normalization(h_init, n_points)
    else:
        h_init = enforce_normalization(init_h, n_points)

    best_h = h_init.copy()
    c5_best = compute_c5(h_init, n_points)

    start_time = time.time()
    refinement = opt.minimize(
        lambda x: compute_c5(x, n_points),
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 200, 'ftol': 1e-7}
    )
    best_h = enforce_normalization(refinement.x, n_points)
    c5_best = compute_c5(best_h, n_points)

    # Perturbation and smart constraint sampling
    attempts = 0
    while (time.time() - start_time) < budget_s - 50:
        k_vals = np.random.choice(np.arange(1, n_points - 1), size=min(50, n_points - 1), replace=False)
        for k in k_vals:
            h_pert = best_h.copy()
            idxs = np.random.choice(n_points - k, size=5, replace=False)
            h_pert[idxs] = 1 - h_pert[idxs]
            h_pert = enforce_normalization(h_pert, n_points)
            c5_pert = compute_c5(h_pert, n_points)
            if c5_pert < c5_best:
                best_h, c5_best = h_pert, c5_pert
                break
        if time.time() - start_time > budget_s:
            break

    # Final optimization
    for _ in range(2):
        refine = opt.minimize(
            lambda x: compute_c5(x, n_points),
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 100}
        )
        h_ref = enforce_normalization(refine.x, n_points)
        c5_ref = compute_c5(h_ref, n_points)
        if c5_ref < c5_best:
            best_h, c5_best = h_ref, c5_ref

    return best_h, c5_best, n_points
