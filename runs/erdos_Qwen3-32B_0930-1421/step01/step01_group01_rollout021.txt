import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2
    cor = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem using convex optimization per worst shift."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if available
    try:

        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.random.rand(n_points) * 2
        h = rescale_h(h)

    h_best, C5_best = np.zeros(n_points), np.inf
    start_time = time.time()
    iter_limit = int((budget_s - 1.0) // 1)
    n_warmup = 10

    for iter in range(iter_limit):
        t_iter = time.time()
        h = rescale_h(h)
        C5_current, h_current = evaluate_C5(h, dx)
        if C5_current < C5_best:
            h_best = h.copy()
            C5_best = C5_current
            print(f"Iter {iter}: C5 improved to {C5_best}")

        cor = np.correlate(h, 1 - h, mode="full")
        k_worst = np.argmax(cor) - n_points + 1
        # Build shifted overlap
        shift_amount = k_worst

        # Build the overlap integrand for this shift
        lhs = h
        rhs_shifted = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i - shift_amount < n_points:
                rhs_shifted[i] = 1 - h[i - shift_amount]
            else:
                rhs_shifted[i] = 1.0

        rhs = rhs_shifted

        # Form QP: minimize overlap at k_worst under constraint ∫h = 1
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(h_var * rhs))
        constraints = [
            cp.sum(h_var) == len(h_var) / 2,
            0 <= h_var, h_var <= 1
        ]
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            h_new = h_var.value
            if h_new is not None:
                h[:] = h_new
        except:
            h = h_best.copy()

        if (iter + 1) % 30 == 0 and n_points > 80:
            h = np.random.rand(n_points) * 2
            h = rescale_h(h)
        # perturbation cooldown
        if (iter + 1) % 100 == 0:
            h = h_best.copy()

        # time control
        if time.time() - start_time > budget_s - 0.25:
            break

    h_vals, c5, n_pts = h_best, C5_best, n_points
    return h_vals, c5, n_pts
