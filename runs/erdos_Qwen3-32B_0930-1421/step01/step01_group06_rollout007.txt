import numpy as np
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    h_best = np.copy(initial_h_values)
    n_points = len(h_best)
    best_c5 = evaluate_erdos_solution(h_best, n_points)

    def simulated_annealing_binary(n_steps=50000):
        nonlocal h_best, best_c5
        h = h_best.copy()
        h[h >= 0.5] = 1.0
        h[h < 0.5] = 0.0

        T0 = 1.0
        alpha = 0.99
        T = T0
        target_sum = n_points / 2.0

        for step in range(n_steps):
            if time.time() - start_time > budget_s - 600:
                return
            # Ensure integer constraint for flip
            one_indices = np.where(h == 1.0)[0]
            zero_indices = np.where(h == 0.0)[0]

            # Temperature schedule
            if step > 0:
                T *= alpha

            # Flip one bit: 1->0 or 0->1, depending on constraint
            if len(one_indices) > target_sum and len(zero_indices) < target_sum:
                idx = np.random.choice(one_indices)
                new_h = h.copy()
                new_h[idx] = 0.0
            elif len(one_indices) < target_sum and len(zero_indices) > target_sum:
                idx = np.random.choice(zero_indices)
                new_h = h.copy()
                new_h[idx] = 1.0
            else:
                # Random flip constrained to stay close to target
                idx_flip1 = np.random.choice(one_indices)
                idx_flip0 = np.random.choice(zero_indices)
                new_h = h.copy()
                new_h[idx_flip1] = 0.0
                new_h[idx_flip0] = 1.0

            c5_old = evaluate_erdos_solution(h, n_points)
            c5_new = evaluate_erdos_solution(new_h, n_points)
            delta = c5_new - c5_old

            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h[:] = new_h[:]
                if c5_new < best_c5:
                    best_c5 = c5_new
                    h_best[:] = h.copy()

    def scale_binary_to_continuous(h_bin):
        total = np.sum(h_bin)
        target = n_points // 2
        if total == target:
            return h_bin
        elif total == target + 1:
            one_indices = np.where(h_bin == 1.0)[0]
            idx = np.random.choice(one_indices)
            h_scaled = h_bin.copy()
            h_scaled[idx] = 0.5
            return h_scaled
        else:
            zero_indices = np.where(h_bin == 0.0)[0]
            idx = np.random.choice(zero_indices)
            h_scaled = h_bin.copy()
            h_scaled[idx] = 0.5
            return h_scaled

    def continuous_refine(h_0):
        nonlocal h_best, best_c5

        def objective(x):
            return evaluate_erdos_solution(x, n_points)

        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0

        cons = (
            {"type": "eq", "fun": constraint_sum},
        )
        bounds = [(0.0, 1.0)] * n_points

        result = minimize(
            objective,
            h_0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 300},
        )

        if result.success:
            new_h = result.x
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                h_best[:] = new_h.copy()
                best_c5 = new_c5

    # Phase 1: Simulated Annealing with binary search
    simulated_annealing_binary(n_steps=50000)

    # Phase 2: Continuous refinement
    h_scaled = scale_binary_to_continuous(h_best)
    continuous_refine(h_scaled)

    # Final check
    final_c5 = evaluate_erdos_solution(h_best, n_points)

    return h_best, final_c5, n_points
