import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def objective_smooth(h_vec, n_points):
    h = np.clip(h_vec.reshape(-1), 0.0, 1.0)
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return smooth_max(np.array(corrs) * (2.0/n_points), beta=100)

def generate_candidate(h, n_points, sigma=0.1):
    h_p = h + np.random.normal(0, sigma, n_points)
    h_p = enforce_normalization(h_p, n_points)
    return h_p

def prior_sampling(n_points, samples=20):
    X = []
    y = []
    for _ in range(samples):
        h = np.random.rand(n_points)
        h = enforce_normalization(h, n_points)
        obj = objective_smooth(h, n_points)
        X.append(h)
        y.append(obj)
    return np.array(X), np.array(y)

    np.random.seed(seed)

    h_current = enforce_normalization(initial_h_values, n_points)
    x_0 = h_current

    X, y = prior_sampling(n_points, samples=20)
    kernel = RBF(length_scale=1.0) + WhiteKernel(noise_level=1e-10)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5)

    best_h = x_0.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        try:
            gp.fit(X, y)
            # Predictive mean and std
            mean_pred, std_pred = gp.predict(X, return_std=True)
            acq_values = mean_pred + 0.2 * std_pred
            idx = np.argmin(acq_values)
            h_candidate = X[idx]
            val_candidate = compute_c5(h_candidate, n_points)

            # Perturb and submit
            h_proposed = generate_candidate(h_candidate, n_points)
            val_proposed = compute_c5(h_proposed, n_points)

            # Accept if better or random
            if val_proposed < best_c5 or np.random.rand() < np.exp((best_c5 - val_proposed) / max(1e-5, best_c5)):
                best_h = h_proposed
                best_c5 = val_proposed
                X = np.vstack((X, best_h))
                new_y = objective_smooth(best_h, n_points)
                y = np.append(y, new_y)
        except:
            # Handle any edge cases: retry with warmed start
            H = generate_candidate(best_h, n_points)
            X = np.vstack((X, H))
            new_y = objective_smooth(H, n_points)
            y = np.append(y, new_y)

        # Check if time is nearly up
        if time.time() - start_time > budget_s - 5:
            break

    # Final refinement with L-BFGS-B
    def wrapper(x):
        h = x.reshape(-1)
        return objective_smooth(h, n_points)
    res = opt.minimize(wrapper, best_h, bounds=[(0, 1)] * n_points, method='L-BFGS-B', options={'maxiter': 100})
    h_refined = res.x
    h_refined = enforce_normalization(h_refined, n_points)
    final_c5 = compute_c5(h_refined, n_points)
    return h_refined, final_c5, n_points
