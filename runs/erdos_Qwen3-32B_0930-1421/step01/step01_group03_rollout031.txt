        h = h * (constraint_sum / h.sum())
        return h

        n = len(parents[0])
        child = np.zeros(n)
        for i in range(n):
            child[i] = np.mean([p[i] for p in parents])
        child = child.clip(0, 1)
        child = child * (constraint_sum / child.sum())
        return child

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def objective(h):
        h = h.reshape(n_points)
        h = normalize_h(h)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

        h = h_flat.reshape(n_points)
        bounds = [(0, 1) for _ in range(n_points)]

            h = normalize_h(h_flat.reshape(n_points))
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
