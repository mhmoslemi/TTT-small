import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def compute_contributors(h, k, n):
    """Identify positions contributing to overlap at shift k"""
    left = (k - n) % n
    right = k % n
    idx1, idx2 = np.where(h[left:] == 1)[0] + left, np.where(h[:right] == 0)[0]
    return idx1, idx2

def greedy_flip_two(h, k, n):
    """Flip two bits to reduce maximum overlap at shift k"""
    h_new = h.copy()
    left = (k - n) % n
    right = k % n
    # Find 1s in h's left segment and 0s in h's right segment
    idx1 = np.where(h[left:] == 1)[0] + left
    idx2 = np.where(h[:right] == 0)[0]
    if len(idx1) > 0 and len(idx2) > 0:
        i, j = idx1[0], idx2[0]
        h_new[i] = 0
        h_new[j] = 1
    return h_new, i, j

def enforce_sum(h, n):
    """Adjust h to maintain exactly n/2 total h[i] == 1"""
    n_half = n // 2
    i1 = np.where(h == 1)[0]
    i0 = np.where(h == 0)[0]
    while np.sum(h) < n_half:
        h[i0[0]] = 1.0
        i0 = i0[1:]  # Remove just fixed index
    while np.sum(h) > n_half:
        h[i1[-1]] = 0.0
        i1 = i1[:-1]
    return h
