import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_peak_indices(h):
    n = len(h)
    h1 = 1 - h
    corr = correlate(h, h1, mode='full')
    maxval = np.max(corr)
    peaks = np.where(corr == maxval)[0]
    mid = n // 2
    best_peak = min(peaks, key=lambda k: abs(k - mid))
    return best_peak

def flip_decisive_bits(h, k, n):
    h_new = h.copy()
    overlap_profile = correlate(h, 1 - h, mode='same')
    max_indices = np.argsort(overlap_profile)[::-1][:2]
    
    # Flip bits in most active regions of original and shifted h
    i = np.where(h == 1)[0][np.argmin(np.abs(np.where(h == 1) - max_indices[0]))]
    j = np.where(h == 0)[0][np.argmin(np.abs(np.where(h == 0) - (k - max_indices[0])))]
    h_new[i] = 0
    h_new[j] = 1
    return h_new

    current_c5 = best_c5

        h_candidate_sum = np.sum(h_candidate)

            h_candidate[np.argmax(h_candidate)] = 0.0

        candidate_c5 = evaluate_overlap(h_candidate)
        if candidate_c5 < current_c5:

                best_h = current_h.copy()
                best_c5 = current_c5

        # Perturbation to escape local optima
        if time.time() - t0 > budget_s - 60:
            break  # Leave 60s for final full evaluation
