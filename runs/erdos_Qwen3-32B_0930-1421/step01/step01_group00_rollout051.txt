import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)

    best_h = h_init.copy()
    best_c5 = np.inf
    current_h = best_h.copy()

    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-8:
            h *= target_sum / h_sum
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h_array):
        h = enforce_constraints(h_array)
        ones_minus_h = 1.0 - h
        corr = scipy.signal.correlate(h, ones_minus_h, mode="full")
        c5 = np.max(corr[1:-1]) * dx
        return c5

    # Use convex optimization for initial improvement
    def convex_initial():
        hw = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(
            [
                cp.sum(cp.multiply(hw, 1 - np.roll(hw, shift=k))) * dx
                for k in range(1, n_points // 2 + 1)
            ],
            p=np.inf
        ))
        constraints = [
            0 <= hw,
            hw <= 1,
            cp.sum(hw) == target_sum,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        return enforce_constraints(hw.value)

    if time.time() - start_time < 60:
        best_h = convex_initial()
        best_c5 = compute_c5(best_h)
        current_h = best_h.copy()

    while time.time() - start_time < budget_s - 30:
        indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        perturbation = np.random.uniform(-0.01, 0.01, size=indices.shape)
        proposed_h = current_h.copy()
        proposed_h[indices] += perturbation
        proposed_h = enforce_constraints(proposed_h)

        new_c5 = compute_c5(proposed_h)
        if new_c5 < best_c5:
            best_h = proposed_h.copy()
            best_c5 = new_c5
            current_h = proposed_h.copy()
        elif np.random.rand() < 0.05:
            current_h = proposed_h.copy()

        # Occasionally apply gradient steps on convex relaxation
        if time.time() - start_time < budget_s - 100 and np.random.rand() < 0.1:
            grad_h = convex_initial()
            grad_h = 0.8 * current_h + 0.2 * grad_h
            grad_h = enforce_constraints(grad_h)
            new_c5 = compute_c5(grad_h)
            if new_c5 < best_c5:
                best_h = grad_h.copy()
                best_c5 = new_c5
                current_h = grad_h.copy()

    return (best_h, best_c5, n_points)
