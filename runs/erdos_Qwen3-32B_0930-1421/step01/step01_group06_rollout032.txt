import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values).astype(np.float64)
    n_points = len(h)

    # Normalize to constraint ∫h = 1
    s = h.sum()
    h = (n_points / 2.0 / s) * h
    h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()
    last_improvement = start_time

    def cvx_optimize(x0, eps=1e-8):
        nonlocal best_c5, best_h, last_improvement

        n = len(x0)
        h_var = cp.Variable(n, nonneg=True)
        corr_k = [cp.sum([x0[i] * (1 - x0[i + k]) for i in range(n - k)]) for k in range(1, n - 1)]
        corr_k += [cp.sum([x0[i + k] * (1 - x0[i]) for i in range(n - k)]) for k in range(1, n - 1)]

        max_corr = cp.Variable()
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1,
            max_corr >= corr for corr in corr_k
        ]

        problem = cp.Problem(cp.Minimize(max_corr), constraints)
        problem.solve(solver=cp.CVXOPT, maxiters=100, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(h_var.value, 0.0, 1.0)
            new_c5 = evaluate_erdos_solution(new_h, n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h
                last_improvement = time.time()
                return True
        return False

    eps_schedule = 10.0
    i_eps = 0
    while time.time() - start_time < budget_s - 5:
        eps = max(1e-6, 10.0 * (0.98 ** i_eps))
        i_eps += 1
        success = cvx_optimize(best_h, eps=eps)
        if not success or time.time() - last_improvement > 120:
            break

    return best_h, best_c5, n_points
