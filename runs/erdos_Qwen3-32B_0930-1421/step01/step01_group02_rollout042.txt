import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr

def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h

def add_bilinear_constraints(problem, h, k_indices, n_points):
    # Create slacks for bilinear terms
    slack = cp.Variable(len(k_indices))
    problem.constraints += [slack >= 0]

    for idx, k in enumerate(k_indices):
        for i in range(n_points - abs(k)):
            j = i + k
            if 0 <= j < n_points:
                problem.constraints += [h[i] * (1 - h[j]) <= slack[idx]]
        # Normalize slack by dx and total mass
        dx = 2.0 / n_points
        problem.constraints += [cp.sum(slack) * dx <= cp.var('λ')]

def optimize_with_cutting_plane(n_points=42, max_iter=100, budget_s=1100):
    h = cp.Variable(n_points, pos=True)
    λ = cp.Variable(pos=True)
    problem = cp.Problem(cp.Minimize(λ))

    # Initial constraints
    problem.constraints += [cp.sum(h) == n_points / 2]
    problem.constraints += [0 <= h]
    problem.constraints += [h <= 1]

    # Cutting-plane loop
    start_time = time.time()
    violated_shifts = set()

    while len(violated_shifts) < n_points and time.time() - start_time < budget_s - 10:
        # Build current problem
        current_problem = cp.Problem(cp.Minimize(λ), problem.constraints)
        current_problem.solve(solver=cp.CLARABEL, verbose=False)

        if current_problem.status == 'infeasible' or current_problem.status == 'unbounded':
            break

        current_h = np.clip(h.value, 0, 1)
        current_h = (current_h / current_h.sum()) * (n_points / 2)

        # Compute overlaps and find violating shifts
        ones_minus_h = 1 - current_h
        correlations = correlate(current_h, ones_minus_h, mode="full")
        dx = 2.0 / n_points
        current_C5 = np.max(correlations * dx)
        max_shift_idx = np.argmax(np.abs(correlations))
        k = max_shift_idx - n_points  # corrected for zero at center

        if k not in violated_shifts and abs(k) < n_points and time.time() - start_time < budget_s - 5:
            violated_shifts.add(k)
            add_bilinear_constraints(problem, h, [k], n_points)

        # Update best solution with current_h and C5
        if current_C5 < 1.0:
            best_h = np.copy(current_h)
            best_C5 = current_C5
        else:
            # Reset to safe value if violated
            best_h = np.zeros(n_points)
            best_h = np.random.rand(n_points)
            best_h = ensure_feasible(best_h)
            best_C5 = evaluate_C5(best_h)

    # Final binary projection
    final_h = np.round(best_h).astype(float)
    final_h = ensure_feasible(final_h)
    final_C5 = evaluate_C5(final_h)

    return final_h, final_C5, n_points

def run(budget_s=1100):
    return optimize_with_cutting_plane(n_points=42, budget_s=budget_s)
