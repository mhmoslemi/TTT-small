import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_population = [np.array(initial_h_values, copy=True) * 0.99]
        h_population.append(h_population[0] + 0.01 * np.random.randn(n_points))
        h_population = [np.clip(h, 0, 1) for h in h_population]
        for _ in range(4):
            h_population.append(np.random.rand(n_points) * 0.5 + 0.25)
        h_population = [h * (n_points / 2 / h.sum()) for h in h_population]
    else:
        n_points = 47
        h_population = [np.zeros(n_points) for _ in range(10)]
        for i, h in enumerate(h_population):
            idx = np.random.choice(n_points, n_points // 4, replace=False)
            h[idx] = 0.5
            h += np.random.rand(n_points) * 0.2
            h = np.clip(h, 0, 1) * (n_points / 2 / h.sum())

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=500):
        h = initial_h.copy()
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1) * (constraint_sum / h.sum())
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1) * (constraint_sum / h_opt.sum())
        return h_opt

    best_h = None
    best_c5 = float('inf')

    generations = 0

    # Population-based search with gradient refinement
    while time.time() - start_time < budget_s * 0.9:
        generation_start = time.time()
        new_population = []

        # Evaluate current population
        population_scores = [(h.copy(), objective(h)) for h in h_population]
        population_scores.sort(key=lambda x: x[1])

        # Keep top 20%
        elite_size = max(2, int(0.2 * len(population_scores)))
        elites = [p[0].copy() for p in population_scores[:elite_size]]

        for i in range(max(5, len(elites) * 2)):
            if time.time() - start_time > budget_s * 0.9:
                break

            # Generate next generation: crossover + mutation
            parents = np.random.choice(elites, size=2)
            child = (parents[0] + parents[1]) / 2
            noise = np.random.randn(n_points) * 0.1
            child += noise
            child = np.clip(child, 0, 1)
            child = child * (constraint_sum / child.sum())

            # Gradient optimization
            child = optimize_h(child, max_iter=200)
            new_population.append(child)

        # Update population with elite and offspring
        h_population = elites + new_population
        if len(h_population) > 20:
            h_population = h_population[:20]

        # Update best
        current_best_vals = [objective(h) for h in h_population]
        for h, val in zip(h_population, current_best_vals):
            if val < best_c5:
                best_c5 = val
                best_h = h.copy()

        generations += 1
        if generations > 50 or time.time() - start_time > budget_s * 0.98:
            break

    # Final refinement of best candidate
    if best_h is not None:
        best_h = optimize_h(best_h, max_iter=200)
        best_h = best_h.clip(0, 1) * (constraint_sum / best_h.sum())
        final_c5 = objective(best_h)

    return best_h, final_c5, n_points
