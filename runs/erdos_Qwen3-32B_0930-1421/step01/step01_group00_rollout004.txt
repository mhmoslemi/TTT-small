import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-8:
        h *= target_sum / h_sum
        h = np.clip(h, 0, 1)

    def compute_c5(h_eval):
        h_eval = np.clip(h_eval, 0, 1)
        h_sum = h_eval.sum()
        if abs(h_sum - target_sum) > 1e-8:
            h_eval *= target_sum / h_sum
        ones_h = 1.0 - h_eval
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr1 = np.dot(h_eval[:n_points - k], ones_h[k:])
            corr2 = np.dot(h_eval[k:], ones_h[:n_points - k])
            max_corr = max(max_corr, corr1 * dx, corr2 * dx)
        return max_corr

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = time.time()
    iteration_time = 0.1

    while time.time() - start_time < budget_s - 5:
        # Compute overlap for all k
        overlaps = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr1 = np.dot(best_h[:n_points - k], 1.0 - best_h[k:])
            corr2 = np.dot(best_h[k:], 1.0 - best_h[:n_points - k])
            overlaps.append((max(corr1, corr2), k))
        overlaps.sort(reverse=True)
        top_k = [k for (__, k) in overlaps[:5]]  # Top 5 problematic shifts

        for k in top_k:
            def evaluate_grad(x, k):
                idxs = np.arange(n_points - k)
                if k >= n_points - 1:
                    continue
                grad = x[k:] - x[:n_points - k]
                grad *= k / n_points * dx
                return grad

            grad1 = evaluate_grad(best_h, k)
            grad2 = -evaluate_grad(best_h, k)
            grad_total = grad1 + grad2
            idxs = np.argsort(np.abs(grad_total))[::-1]
            idxs = idxs[:min(int(n_points * 0.1), n_points)]
            proposed_h = best_h.copy()
            proposed_h[idxs[:len(idxs) // 2]] -= 0.02
            proposed_h[idxs[len(idxs) // 2:]] += 0.02
            proposed_h = np.clip(proposed_h, 0, 1)
            h_sum = proposed_h.sum()
            if abs(h_sum - target_sum) > 1e-8:
                proposed_h *= target_sum / h_sum
                proposed_h = np.clip(proposed_h, 0, 1)

            proposed_c5 = compute_c5(proposed_h)
            if proposed_c5 < best_c5:
                best_h[:] = proposed_h
                best_c5 = proposed_c5
                best_time = time.time()

        if time.time() - best_time > 120:
            break

    return (best_h, best_c5, n_points)
