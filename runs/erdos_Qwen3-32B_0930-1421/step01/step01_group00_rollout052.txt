import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    # Initialize
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Constraint enforcement and clipping
    h_sum = np.sum(h)
    h = np.clip(h, 0, 1)
    if abs(h_sum - target_sum) > 1e-8:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        optimized_h_sum = optimized_h.sum()
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points - 1):
            # Forward correlation
            if len(optimized_h) - k >= 1:
                corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
                max_corr = max(max_corr, corr1 * dx)
                # Backward correlation
                corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
                max_corr = max(max_corr, corr2 * dx)
        return max_corr

    best_c5 = compute_c5(h)
    best_h = h.copy()
    last_improvement_time = start_time

    while time.time() - start_time < budget_s - 5:
        current_c5 = compute_c5(h)
        all_corr = []

        # Precalculate all corr for shift detection
        optimized_h = np.clip(h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        optimized_h_sum = optimized_h.sum()
        ones_minus_h = 1.0 - optimized_h
        for k in range(1, n_points - 1):
            if len(optimized_h) - k < 1:
                continue
            # Forward
            corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            all_corr.append((corr1 * dx, k, 'f'))
            # Backward
            corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            all_corr.append((corr2 * dx, k, 'b'))

        all_corr.sort(reverse=True)
        top_k_shifts = [item[1] for item in all_corr[:5]]

        for k in top_k_shifts:
            if k >= n_points - 1:
                continue
            occupied = h[:n_points - k]
            free = 1.0 - h[k:]
            product = occupied * free
            top_indices = np.argsort(product)[-int(len(product) * 0.1):]

            candidate = h.copy()
            for idx in top_indices:
                if product[idx] > 0.01:
                    # Try reducing h[idx]
                    trial = candidate.copy()
                    trial[idx] -= 0.01
                    trial = np.clip(trial, 0, 1)
                    total = trial.sum()
                    if abs(total - target_sum) > 1e-7:
                        trial *= target_sum / total
                    trial = np.clip(trial, 0, 1)
                    trial_c5 = compute_c5(trial)
                    if trial_c5 < best_c5:
                        best_c5 = trial_c5
                        best_h = trial.copy()
                        last_improvement_time = time.time()
                    # Try increasing h[idx + k]
                    trial = candidate.copy()
                    trial[idx + k] += 0.01
                    trial = np.clip(trial, 0, 1)
                    total = trial.sum()
                    if abs(total - target_sum) > 1e-7:
                        trial *= target_sum / trial.sum()
                    trial = np.clip(trial, 0, 1)
                    trial_c5 = compute_c5(trial)
                    if trial_c5 < best_c5:
                        best_c5 = trial_c5
                        best_h = trial.copy()
                        last_improvement_time = time.time()
                    # Try balanceumat
                    trial = candidate.copy()
                    trial[idx] -= 0.005
                    trial[idx + k] += 0.005
                    trial = np.clip(trial, 0, 1)
                    total = trial.sum()
                    if abs(total - target_sum) > 1e-7:
                        trial *= target_sum / trial.sum()
                    trial = np.clip(trial, 0, 1)
                    trial_c5 = compute_c5(trial)
                    if trial_c5 < best_c5:
                        best_c5 = trial_c5
                        best_h = trial.copy()
                        last_improvement_time = time.time()

        if time.time() - last_improvement_time > 60 and best_c5 < 0.381:
            break  # Early stopping if we're doing well

    h = best_h
    c5_value = compute_c5(h)
    return (h, c5_value, n_points)
