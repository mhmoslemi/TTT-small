import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    return np.max(correlations) * dx


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    n_points = len(h)
    if total < 1e-6:
        h = np.zeros(n_points)
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h

    overlap_all = []
    dx = 2.0 / n_points

        overlap_k = []
        for i in range(n_points):
            if 0 <= i - k < n_points:

        if overlap_k:
            overlap_sum = cp.sum(overlap_k) * dx
            overlap_all.append(overlap_sum)

    max_overlap = cp.Variable(nonneg=True)

    ]

    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, warm_start=True)
