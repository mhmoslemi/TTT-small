import numpy as np
import time
from scipy.signal import correlate


def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx


def perturb_h(h, sigma, n_points):
    h = h.copy()
    n_perturb = np.random.randint(5, 11)
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, sigma)
        h[i] = np.clip(h[i], 0, 1)
    # Normalize to preserve sum constraint
    h_sum = np.sum(h)
    h *= n_points / 2 / h_sum
    return h

    sigma = initial_sigma

    no_improvement = 0
    max_no_improvement = 500
    adaptive_cooling = 1  # increase cooling rate when no improvement

        h_new = perturb_h(current_h, sigma, n_points)
        c5_new = evaluate_overlap(h_new)
        delta = c5_new - current_c5

            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            T *= adaptive_cooling
            no_improvement = 0
        else:
            T *= alpha
