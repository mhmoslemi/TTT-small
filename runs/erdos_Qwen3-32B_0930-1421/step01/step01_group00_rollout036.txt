import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize h to satisfy constraints with tolerances
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    if not np.isclose(h_sum, target_sum):
        h *= target_sum / h_sum
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = start_time
    limit = 1e-7

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if np.abs(optimized_h_sum - target_sum) > limit:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h
        conv = scipy.signal.correlate(optimized_h, ones_minus_h, mode='full')
        correction = np.array([dx for _ in conv])
        return np.max(conv * correction)

    window = int(n_points * 0.05)
    window = max(1, window)

    while time.time() - start_time < budget_s - 10:

        # only attempt top 5 shifts that contribute most to the overlap
        def get_top_k_shifts(h_op):
            h_op = np.clip(h_op, 0.0, 1.0)
            h_op_sum = h_op.sum()
            if np.abs(h_op_sum - target_sum) > 1e-7:
                h_op *= target_sum / h_op_sum
            lcorr = scipy.signal.correlate(h_op, (1 - h_op), mode='full')
            val = lcorr * dx
            k_indices = np.argsort(val)[-5:]
            top_shifts = np.unique(k_indices // n_points)
            return top_shifts[(top_shifts > 0) & (top_shifts < n_points - 1)]

        top_shifts = get_top_k_shifts(h)

        # For each of these top k values, find top percent contribs and perturb
        for k in top_shifts:
            contribs = h[:n_points - k] * (1 - h[k:])
            top_indices = np.argsort(contribs)[-max(1, int(n_points * 0.1)):]
            indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
            proposed_h = h.copy()

            # Adjust the h for top indices accordingly
            for i in top_indices:
                if proposed_h[i] > 0.01:
                    proposed_h[i] -= 0.01
                if proposed_h[i + k] < 0.99 and h[i + k] < h[i]:
                    proposed_h[i + k] += 0.01

            proposed_h = np.clip(proposed_h, 0, 1)
            proposed_sum = proposed_h.sum()
            if np.abs(proposed_sum - target_sum) > limit:
                proposed_h *= target_sum / proposed_sum
                proposed_h = np.clip(proposed_h, 0.0, 1.0)

            new_c5 = compute_c5(proposed_h)
            if new_c5 < best_c5:
                h[:] = proposed_h
                best_c5 = new_c5
                best_h[:] = h[:]
                best_time = time.time()

        if time.time() - best_time > 120 and best_c5 < 0.380876:
            break

    return best_h, best_c5, n_points
