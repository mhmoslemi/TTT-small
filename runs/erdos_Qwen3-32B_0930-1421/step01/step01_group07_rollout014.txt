import numpy as np
import time
from scipy.signal import correlate

def perturb_h(h, sigma, n_pert=5, seed=None):
    n_points = len(h)
    rng = np.random.default_rng(seed)
    indices = rng.choice(n_points, size=min(n_pert, n_points), replace=False)
    delta = rng.normal(scale=sigma, size=len(indices))
    new_h = h.copy()
    new_h[indices] += delta
    new_h = np.clip(new_h, 0.0, 1.0)
    # Normalize to maintain integral constraint
    total = np.sum(new_h)
    scale = (n_points / 2) / total
    new_h *= scale
    return new_h

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx
