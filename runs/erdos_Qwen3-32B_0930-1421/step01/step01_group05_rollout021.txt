import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = normalize_h(h)
    best_c5 = compute_c5(best_h)

    lambdas = [0.1 * (0.5 ** i) for i in range(20)]

    # Try different L1 regularization values
    for lambda_val in lambdas:
        # Break if budget is exhausted
        if time.time() - start_time >= budget_s:
            break

        h_var = cp.Variable(n_points)

        objective = cp.Minimize(cp.max(correlate(h_var, 1 - h_var, mode='full') * dx) + lambda_val * cp.norm1(h_var))

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_candidate = np.array(h_var.value).flatten()
            h_candidate = normalize_h(h_candidate)
            c5_candidate = compute_c5(h_candidate)

            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

    # Post-optimization refinement with linear descent
    learning_rate = 0.1
    n_refinement = 500
    while time.time() - start_time < budget_s and n_refinement > 0:
        grad = np.zeros_like(best_h)
        eps = 1e-5

        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        best_c5 = compute_c5(best_h)

        learning_rate *= 0.99

        n_refinement -= 1

    return best_h.tolist(), best_c5, n_points
