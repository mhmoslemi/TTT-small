import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Helper to clip and scale
    def normalize_h(vec):
        vec = np.clip(vec, 0, 1)
        s = vec.sum()
        if abs(s - target_sum) > 1e-7:
            vec *= (target_sum / s)
        return vec

    # Constraint projection
    def project(h):
        h = normalize_h(h)
        return h

    # Compute C5
    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1).copy()
        s = optimized_h.sum()
        if abs(s - target_sum) > 1e-7:
            optimized_h *= (target_sum / s)

        max_val = -np.inf
        for k in range(1, n_points):
            if k >= n_points:
                continue
            corr = np.dot(optimized_h[:n_points - k], (1 - optimized_h[k:]))
            max_val = max(max_val, corr * dx)

            corr = np.dot(optimized_h[k:], (1 - optimized_h[:n_points - k]))
            max_val = max(max_val, corr * dx)
        return max_val

    h = normalize_h(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_time = start_time
    last_update = start_time

    while time.time() - start_time < budget_s - 5:
        h = best_h.copy()

        # Compute all correlations
        overlap_values = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            val1 = np.dot(h[:n_points - k], (1 - h[k:])) * dx
            val2 = np.dot(h[k:], (1 - h[:n_points - k])) * dx
            overlap = max(val1, val2)
            overlap_values.append((overlap, k))

        overlap_values.sort(reverse=True)
        top_k_shifts = [k for _, k in overlap_values[:5]]

        # Targeted modification for largest overlaps
        for k in top_k_shifts:
            # Consider indices where both h[i] and h[i+k] contribute to overlap
            overlap_idx_l = np.arange(n_points - k)
            overlap_idx_r = overlap_idx_l + k

            # Compute how much each i contributes to the positive part of overlap
            contrib = h[overlap_idx_l] * (1 - h[overlap_idx_r])
            top_idx = overlap_idx_l[np.argsort(contrib)[-10:]]
            contrib2 = h[overlap_idx_r] * (1 - h[overlap_idx_l])
            top_idx2 = overlap_idx_r[np.argsort(contrib2)[-10:]]

            idx_to_alter = np.unique(np.concatenate([top_idx, top_idx2]))

            # For each index in idx_to_alter, decide whether to decrease h[i] or increase h[i+k]
            for i in idx_to_alter:
                j = i + k
                if j >= n_points:
                    continue
                if i < j and j < n_points:
                    # Try decreasing h[i]
                    h[i] -= 0.01
                    h[j] += 0.01
                    h = np.clip(h, 0, 1)
            h = normalize_h(h)

        # Evaluate new c5
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = new_c5
            best_time = time.time()
            last_update = time.time()
        elif new_c5 < 0.380876:
            break

        if time.time() - best_time > 120:
            break

    return (best_h, best_c5, n_points)
