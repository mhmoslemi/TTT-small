import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem using convex relaxation."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h = np.linspace(0, 1, n_points)
        h = h / h.sum() * n_points / 2

    h_var = cp.Variable(n_points)
    obj_vals = []
    best_h, best_c5 = h.copy(), np.inf

    start_time = time.time()
    solver_options = {"max_iter": 500, "tol": 1e-6, "verbose": False}

    # Optimization iterations
    for iter_count in range(1000):
        if time.time() - start_time > budget_s:
            break
        h_val = h if iter_count == 0 else h_var.value

        # Formulate convex relaxation objective
        shifts = [correlate(h_val, np.roll(1 - h_val, k), mode="full") for k in range(5)]
        corr_peak = cp.Max(cp.hstack(shifts))

        def reg_penalty(h):
            return cp.sum((h - 0.5) ** 2)

        problem = cp.Problem(
            cp.Minimize(1.0 * corr_peak + 0.01 * reg_penalty(h_var)),
            [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2],
        )

        try:
            problem.solve(solver=cp.ECOS, **solver_options)
            if h_var.value is None:
                continue
            h = h_var.value
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2  # Enforce integral constraint
            c5 = correlate(h, 1 - h, mode="full").max() * dx
        except cp.SolverError:
            h = h + 0.1 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2
            c5 = correlate(h, 1 - h, mode="full").max() * dx

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    h_vals, c5, n_pts = best_h, best_c5, n_points
    return h_vals, c5, n_pts
