import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, i_flip, j_flip):
    h_new = h.copy()
    h_new[i_flip] = 0
    h_new[j_flip] = 1
    return h_new

def get_problematic_shift(h, dx=1.0):
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    max_corr = np.max(corr)
    shift_idx = np.argmax(corr)
    return shift_idx, max_corr * dx

def find_relevant_indices(h, shift_idx, n_points):
    relevant_indices = []
    for i in range(n_points):
        j = (i - shift_idx) % n_points
        if i != j:
            relevant_indices.append((i, j))
    return relevant_indices

def compute_bit_impact(h, relevant_indices):
    impact = np.zeros(len(h))
    for i, j in relevant_indices:
        impact[i] += h[i] * (1 - h[j])
    return impact

    while np.sum(h) != n_points / 2:
        diff = np.sum(h) - n_points / 2
        if diff < 0:
            idx_flip = np.random.choice(np.where(h == 0)[0])
            h[idx_flip] = 1.0
        else:
            idx_flip = np.random.choice(np.where(h == 1)[0])
            h[idx_flip] = 0.0

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    print("Initial C₅:", best_c5)
    t0 = time.time()
    max_iter = 10000
    no_improvement = 0
    max_no_improvement = 1000

    for iter in range(max_iter):
        if time.time() - t0 >= budget_s:
            print("Budget expired, stopping.")
            break

        shift_idx, c5 = get_problematic_shift(best_h)
        if c5 <= best_c5:
            best_c5 = c5
            no_improvement = 0
            continue

        # Get indices contributing to the problematic shift
        relevant_indices = find_relevant_indices(best_h, shift_idx, n_points)
        impact = compute_bit_impact(best_h, relevant_indices)

        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break  # No room to make helpful bit flips

        i_flip = np.argmin(impact[ones])
        j_flip = np.argmax(impact[zeros])

        try_h = flip_two_bits(best_h, ones[i_flip], zeros[j_flip])
        # Check feasibility
        if np.sum(try_h) == n_points / 2:
            new_c5 = evaluate_overlap(try_h)
            if new_c5 < best_c5:
                best_h = try_h.copy()
                best_c5 = new_c5
                no_improvement = 0
                print(f"Iter {iter}: New C₅={best_c5:.6f}")
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            # Small perturbation to escape local optimum
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            best_h = flip_two_bits(best_h, i, j)
            no_improvement = 0

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
