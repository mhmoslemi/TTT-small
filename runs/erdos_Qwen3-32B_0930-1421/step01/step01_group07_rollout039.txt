import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def clip_and_project(h, total_mass):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h[:] = 0.0
    else:
        h *= total_mass / current_sum
    return h

    best_c5 = evaluate_overlap(h)
    current_c5 = best_c5
    original_alpha = alpha

    for iter_idx in range(max_iter):

        corr = correlate(h, 1 - h, mode="full")
        k_shift = np.argmax(corr)
        lag = k_shift - (n_points - 1)

        grad = np.zeros_like(h)
        for i in range(n_points):

        # Normalize
        grad_len = np.linalg.norm(grad)
        if grad_len > 1e-10:

        step = alpha * grad
        h_candidate = h - step
        h_candidate = clip_and_project(h_candidate, total_mass)

        # Line search
        found_step = False
        backtrack_count = 0
        while backtrack_count < 5:
            candidate_c5 = evaluate_overlap(h_candidate)
            if candidate_c5 < current_c5:
                h[:] = h_candidate

                h_candidate = h - alpha * grad

        # Perturbation to escape local minima
        if iter_idx % 300 == 0:

            h = clip_and_project(h, total_mass)

        # Validation
        assert np.all((h >= -1e-10) & (h <= 1 + 1e-10)), "h out of bounds"
        assert abs(np.sum(h) - total_mass) < 1e-8, "Mass constraint violated"

    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
