import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_full = np.tile(h, 3)  # Occasionally overpadding as a simplification
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr * dx)
    return max(corrs)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if np.isclose(total, 0.0):
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, sigma, n_points):
    h_p = h + np.random.normal(0, sigma, n_points)
    h_p = enforce_normalization(h_p, n_points)
    return h_p

        return h_p
    elif np.random.rand() < np.exp((compute_c5(h, n_points) - mu[0]) / temp):
