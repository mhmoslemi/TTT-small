import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 using convex optimization, quadratic energy and interior point method."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Initialize from accessory global if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.linspace(0.2, 0.4, n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

    best_C5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    # Solver timeout per iteration and restart conditions
    timeout = 60
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1

        # Solve convex relaxation with quadratic energy for overlaps
        h_var = cp.Variable(n_points, nonneg=True)
        C5_energy = cp.sum([cp.sum((h_var + h_var[k:] + h_var[:n_points-k] - 2)**2) for k in range(1, n_points-1)])
        reg_energy = cp.sum_squares(h_var)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]

        # Weight settings for the energy functional
        lambd = 0.1 * (0.5 ** (iter_count // 50))
        mu = 0.01

        problem = cp.Problem(cp.Minimize(lambd * C5_energy + mu * reg_energy), constraints)
        problem.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-6, eps_rel=1e-6, time_limit=timeout)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_var.value
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2  # Re-normalize exactly to constraint
            current_C5 = evaluate_C5(h, dx)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()

        else:
            h = best_h.copy()

        # Perturb to avoid local optima and ensure exploration
        if iter_count % 25 == 0:
            h += 0.1 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2

    return best_h, best_C5, n_points
