import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(n_points):
            if k >= n_points - 1:
                continue
            if n_points - k == 0:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
        for k in range(n_points):
            if k >= n_points - 1:
                continue
            corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_time = time.time()

    # Optimization
    MAX_ITER = 50000
    PROPOSAL_SIZE = 6
    NOISE_SCALE = 0.01
    LOCAL_SEARCH_EVERY = 15
    EPOCHS = 0

    for i in range(MAX_ITER):
        if time.time() - start_time > budget_s - 10:
            break

        # Perturbation
        indices = np.random.choice(n_points, size=min(PROPOSAL_SIZE, n_points), replace=False)
        proposed_h = h.copy()
        noise = np.random.normal(0, NOISE_SCALE, size=indices.shape)
        proposed_h[indices] += noise
        proposed_h = np.clip(proposed_h, 0, 1)
        proposed_h_sum = proposed_h.sum()
        proposed_h *= target_sum / proposed_h_sum
        proposed_h = np.clip(proposed_h, 0, 1)

        c5_new = compute_c5(proposed_h)

        if c5_new < best_c5:
            h[:] = proposed_h[:]
            best_c5 = c5_new
            best_h[:] = h[:]
            best_time = time.time()
            if best_c5 < 0.38088:
                break

        if time.time() - best_time > 90:
            EPOCHS += 1
            if EPOCHS >= 3 and NOISE_SCALE > 1e-3:
                NOISE_SCALE *= 0.8
            if NOISE_SCALE < 1e-4:
                break

        if best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
