import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, i1, i2):
    """Flip bit at i1 and i2"""
    h = h.copy()
    h[i1] = 1 - h[i1]
    h[i2] = 1 - h[i2]
    return h

def perturb_h(h, shift_k, best_sample_h, peak_pos):
    """Recompute and modify h to target the worst overlap peak"""
    i = peak_pos
    # Adjust h in regions around peak and mirrored position
    try:
        flip1 = best_sample_h[np.clip(i, 0, len(best_sample_h)-1)].argmax()
        flip2 = (shift_k - i) + len(best_sample_h) // 2  # Mirrored position
        flip2 = np.clip(flip2, 0, len(best_sample_h) - 1)
    except:
        flip1 = np.random.choice(np.where(h == 1)[0])
        flip2 = np.random.choice(np.where(h == 0)[0])
    return flip_two_bits(h, flip1, flip2)
