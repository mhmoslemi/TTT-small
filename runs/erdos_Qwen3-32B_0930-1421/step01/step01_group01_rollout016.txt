import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem using convex relaxation and interior-point solvers."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Try to use parent construction
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = np.resize(h, n_points)
    except NameError:
        h = np.ones(n_points) / n_points * 2  # Uniform mass distribution

    h = np.clip(h, 0, 1)
    h = h / h.sum() * n_points / 2  # Enforce mass constraint

    # Initialize best solution
    best_h = h.copy()
    best_C5, _ = evaluate_erdos_solution(best_h)
    best_time = time.time()

    lambda_ = 0.1  # Overlap penalty weight
    mu = 0.01      # Tikhonov regularization
    solver_timeout = 100  # s
    perturbation = 0.1
    save_interval = 50

    def energy(h_vec):
        cor = correlate(h_vec, 1 - h_vec, mode="full")
        overlap_term = cp.sum(cp.square(cor + cp.Constant(0)))
        penalty = cp.sum(cp.square(h_vec))
        return (lambda_ / n_points) * overlap_term + mu * penalty

    iter_count = 0
    tic = time.time()

    while time.time() - tic < budget_s and iter_count < 200:
        iter_count += 1
        if time.time() - best_time > 10 or iter_count % save_interval == 0:
            best_h = h.copy()
            best_C5, _ = evaluate_erdos_solution(best_h)
            best_time = time.time()

        # CVXPY convex relaxation
        h_var = cp.Variable(n_points)
        objective = energy(h_var)
        constraints = [
            h_var <= 1,
            h_var >= 0,
            cp.sum(h_var) == n_points / 2
        ]
        problem = cp.Problem(cp.Minimize(objective), constraints)

        try:
            problem.solve(solver=cp.OSQP, verbose=False, max_iter=int(solver_timeout * 10))
            h = h_var.value
            if h is not None:
                h = np.clip(np.nan_to_num(h), 0, 1)
                h = h / h.sum() * n_points / 2
        except Exception as e:
            pass  # fall back to current h

        # Perturbation if stuck
        if iter_count % save_interval == 0 and iter_count > 1:
            h = h * (1 - 0.5 * perturbation * np.random.rand(n_points)) + 0.5 * perturbation * np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2

    # Final verification and update
    h = np.clip(h, 0, 1)
    h = h / h.sum() * n_points / 2
    C5, _ = evaluate_erdos_solution(h)
    if C5 < best_C5:
        best_h = h.copy()
        best_C5 = C5

    return best_h, best_C5, n_points
