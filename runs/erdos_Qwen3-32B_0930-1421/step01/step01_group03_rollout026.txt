import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = max_n_points
        h_init = np.zeros(n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h_init[indices] = 1.0
        h_init = h_init * (n_points / 2) / h_init.sum()

    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    C = cp.Variable()

    objective = cp.Minimize(C)

    constraints = [cp.sum(h) == n_points / 2]
    constraints += [h[i] <= 1 for i in range(n_points)]

    overlap = cp.Constant(0)
    for k in range(1, n_points):
        nh = np.roll(h, shift=k)
        constraints += [C >= dx * cp.sum(h * (1 - nh))]
        
    prob = cp.Problem(objective, constraints)
    best_h = h_init
    best_c5 = float('inf')

    def get_c5(h_val):
        h_val = h_val / h_val.sum() * (n_points / 2)
        h_val = np.clip(h_val, 0, 1)
        overlap_arr = np.correlate(h_val, 1 - h_val, mode='full') * dx
        return np.max(overlap_arr)

    perturbs = 0
    while time.time() - start_time < budget_s:
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
            h_new = np.array(h.value).flatten()
            constraint_corrected = h_new / h_new.sum() * (n_points / 2)
            h_new = np.clip(constraint_corrected, 0, 1)

            c5_new = get_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

            if perturbs % 5 == 0:
                noise = np.random.normal(0, 0.1, n_points)
                h.assign(np.clip(h_new + noise, 0, 1))
            perturbs += 1
        except:
            pass

    # Last clip before return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2)
    best_c5 = get_c5(best_h)
    return best_h, best_c5, n_points
