```python
import numpy as np
import time
import cvxpy as cp

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def construct_optimizer_problem(h_size, dx, h_init):
    h = cp.Variable(h_size, nonneg=True)
    const = h.sum() == h_size / 2  # integral condition
    bounds = [h[i] <= 1 for i in range(h_size)]
    corrs = []
    for k in range(-h_size + 1, h_size):
        overlap = 0
        for i in range(h_size):
            j = i + k
            if 0 <= j < h_size:
                overlap += h[i] * (1 - h[j])
        corrs.append(overlap * dx)
    max_corr = cp.Variable()
    objective = cp.Minimize(max_corr)
    constraints = [const, max_corr >= corr for corr in corrs] + bounds
    problem = cp.Problem(objective, constraints)
    return problem, h


def run(budget_s=1100, n_points=42, solver="ECOS", use_initial=True):
    if use_initial and initial_h_values is not None:
        h_init = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = ensure_feasible(h)
        h_init = h

    dx = 2.0 / n_points

    problem, h_var = construct_optimizer_problem(n_points, dx, h_init)
    problem.solve(solver=solver, verbose=False)

    h_values = np.clip(h_var.value, 0, 1)
    h_values = ensure_feasible(h_values)

    c5_bound = evaluate_C5(h_values)

    return h_values, c5_bound, n_points
```
