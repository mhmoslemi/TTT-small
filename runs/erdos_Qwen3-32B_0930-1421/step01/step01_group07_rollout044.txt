import numpy as np
import time
from scipy.signal import correlate

def perturb_h(h, perturb_count=5, sigma=0.1):
    h_new = h.copy()
    indices = np.random.choice(len(h), size=perturb_count, replace=False)
    delta = np.random.normal(0, sigma, size=perturb_count)
    h_new[indices] += delta
    h_new = np.clip(h_new, 0, 1)
    # Normalize sum to constraint
    delta_sum = np.sum(h_new) - (len(h) / 2)
    if abs(delta_sum) > 1e-8:
        h_new -= delta_sum / len(h)
        h_new = np.clip(h_new, 0, 1)
    return h_new

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx
