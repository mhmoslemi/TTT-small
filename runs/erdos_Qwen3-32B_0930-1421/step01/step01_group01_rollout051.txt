import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    h = h.copy()
    h[h < 0] = 0
    h[h > 1] = 1
    h = h / h.sum() * len(h) / 2  # Enforce normalization
    cor = correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx, h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Initialize h with initial_h_values or fallback
    try:
        h = initial_h_values.copy()
        assert len(h) == n_points, "Mismatch in h length"
    except:
        h = np.linspace(0.1, 0.3, n_points)
        h = h / h.sum() * n_points / 2

    h = cp.Parameter(shape=h.shape, value=h.copy())

    h_opt = cp.Variable(shape=h.shape)
    constraints = [
        0 <= h_opt,
        h_opt <= 1,
        cp.sum(h_opt) == n_points / 2,
    ]

    # Energy function with regularization
    overlap = 0.0
    for k in range(1, n_points):  # Only non-zero lag overlaps are critical
        corr = cp.sum(h_opt * (1 - cp.roll(h_opt, k)))
        overlap += corr

    objective = cp.Minimize(overlap)

    problem = cp.Problem(objective, constraints)
    best_C5 = float('inf')
    best_h = h.value.copy()
    start_time = time.time()

    lambda_ = 0.1
    mu = 0.01
    restart_threshold = 50
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1

        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=200)
            h_f = h_opt.value
            if h_f is None:
                raise ValueError("Infeasible")
            h_f = np.clip(h_f, 0, 1)
            h_f /= h_f.sum() * 2 / n_points * n_points
            C5, _ = evaluate_C5(h_f, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_f.copy()

        except Exception as e:
            pass  # Skip infeasible solves

        if iter_count % restart_threshold == 0:
            # Perturb and restart with a different h
            perturbation = 0.1
            h.value = best_h + perturbation * (np.random.rand(n_points) - 0.5)
            h.value = h.value / h.value.sum() * n_points / 2
            h.value = np.clip(h.value, 0, 1)

        if time.time() - start_time > 0.6 * budget_s:
            break  # Reserve time for final evaluation

    return best_h, best_C5, n_points
