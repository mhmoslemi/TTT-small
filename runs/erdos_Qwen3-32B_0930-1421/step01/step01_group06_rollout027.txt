import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize h to satisfy the sum constraint
    total_h = float(np.sum(h))
    h = np.minimum(h * (n_points / 2.0) / total_h, 1.0)
    h[0] = np.clip(h[0], 0.0, 1.0)
    h[-1] = np.clip(h[-1], 0.0, 1.0)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)

    dx = 2.0 / n_points

    def setup_cvx_model(C5, h_init):
        h_var = cp.Variable(n_points, nonneg=True)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]

        shifts = np.arange(1, n_points)
        for k in shifts:
            idx = np.arange(n_points - k)
            i = idx
            j = idx + k
            z_vars = cp.Variable(len(idx))
            constraints += [
                z_vars >= h_var[i] + (1 - h_var[j]) - 1,
                z_vars <= h_var[i],
                z_vars <= 1 - h_var[j],
                z_vars >= 0,
                cp.sum(z_vars) <= C5 * dx
            ]

        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, max_time=20)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
        return None

    C5_low = 0.0
    C5_high = best_c5
    C5_tolerance = 1e-4
    best_C5 = best_c5

    while time.time() - start_time < budget_s and C5_high - C5_low > C5_tolerance:
        C5_mid = (C5_low + C5_high) / 2
        h_solved = setup_cvx_model(C5_mid, best_h)

        if h_solved is not None:
            h_solved = np.clip(h_solved, 0.0, 1.0)
            current_c5 = evaluate_erdos_solution(h_solved, n_points)
            if current_c5 < best_C5:
                best_C5 = current_c5
                best_h = h_solved
            C5_high = C5_mid
        else:
            C5_low = C5_mid

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)

    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return best_h, final_c5, n_points
