import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp
import cvxpy.optimal_value

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to satisfy the sum constraint
    total_mass = np.sum(h)
    h = np.clip(h * (n_points / 2.0 / total_mass), 0, 1)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement = start_time

    # Define h as a cvxpy variable
    h_cp = cp.Variable(n_points, nonneg=True, max=1.0)

    # Constraint: integral constraint
    constraint_integral = [cp.sum(h_cp) == n_points / 2.0]

    def run_cvx(c5_guess):
        # Construct the problem
        corr = np.correlate(h_cp, 1 - h_cp, mode='full') * (2.0 / n_points)
        obj = cp.Maximize(c5_guess - cp.max(corr))
        constraints = constraint_integral
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.CVXOPT)

        if result > 1e-6:
            h_result = h_cp.value
            h_result = np.clip(h_result, 0.0, 1.0)
            h_result = h_result * (n_points / 2.0 / np.sum(h_result))
            new_c5 = evaluate_erdos_solution(h_result, n_points)
            return h_result, new_c5
        return None, float('nan')

    while time.time() - start_time < budget_s - 10:  # leaving some buffer
        elapsed = time.time() - start_time
        if elapsed - last_improvement > 90:
            break

        # Interleaving with local search for robustness
        # Use a weighted approximation to C₅ for convex relaxation
        for retry in range(3):
            c5_guess = best_c5 * (1.0 - 0.05 * retry)
            h_t, c5_t = run_cvx(c5_guess)
            if h_t is not None and np.isfinite(c5_t):
                if c5_t < best_c5:
                    best_h = np.copy(h_t)
                    best_c5 = c5_t
                    last_improvement = time.time()

    return best_h, best_c5, n_points
