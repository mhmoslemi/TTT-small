```python
import numpy as np
import time

def ensure_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        valid_len = n_points - k
        if valid_len <= 0:
            break
        corr = np.dot(h[:valid_len], 1 - h[k:k + valid_len])
        corrs.append(corr)
    for k in range(1, n_points):
        valid_len = n_points - k
        if valid_len <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:valid_len])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def mutate(h, sigma=0.1):
    h_mut = h + np.random.normal(0, sigma, len(h))
    return ensure_feasible(h_mut, len(h))

def crossover(h1, h2):
    alpha = np.random.uniform(0, 1, len(h1))
    h_child = alpha * h1 + (1 - alpha) * h2
    return ensure_feasible(h_child, len(h_child))

def select_parents(population, fitness, num_parents):
    indices = np.argsort(fitness)
    parents = [population[i] for i in indices[:num_parents]]
    return parents

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)

    if init_h is None:
        raise ValueError("initial_h_values must not be None")
    h_initial = init_h.copy()
    n_points = len(h_initial)

    pop_size = 50
    pop = [ensure_feasible(h_initial + np.random.normal(0, 0.1, n_points), n_points) for _ in range(pop_size - 1)]
    pop.append(h_initial)
    fitness = [compute_c5(h, n_points) for h in pop]
    best_h = pop[np.argmin(fitness)].copy()
    best_c5 = np.min(fitness)
    time_start = time.time()

    while time.time() - time_start < budget_s * 0.9:
        time_now = time.time()
        if time_now - time_start > budget_s:
            break

        # Selection
        num_parents = max(10, int(pop_size * 0.2))
        parents = select_parents(pop, fitness, num_parents)

        # Crossover
        children = []
        while len(children) < pop_size - num_parents:
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            if p1 is p2:
                continue
            child = crossover(p1, p2)
            children.append(child)

        # Mutation
        muts = [mutate(ind, sigma=0.1)
                for ind in children[:int(len(children) * 0.8)]]
        children[:int(len(children) * 0.8)] = muts

        # Build next generation
        pop = parents + children
        fitness = [compute_c5(h, n_points) for h in pop]

        # Update best
        current_idx = np.argmin(fitness)
        if fitness[current_idx] < best_c5:
            best_h = pop[current_idx].copy()
            best_c5 = fitness[current_idx]

        # Logging progress and decay
        pop = pop[:pop_size]
        t_ratio = (time.time() - time_start) / budget_s
        sigma_decay = 0.1 * (1.0 - t_ratio)
        if sigma_decay < 0.01:
            sigma_decay = 0.01

    return best_h, best_c5, n_points
```
