import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None  # Will be injected at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / np.sum(h)
    return h


def basis_function(x, x_i, alpha=4.0, beta=0.3, dx=0.01):
    X = np.arange(-1.0, 1.0 + dx, dx)
    psi = np.zeros_like(X)
    mask = np.abs(X) <= beta
    psi[mask] = np.exp(-alpha * X[mask]**2)
    offset = int((x_i - x[0]) / dx) if len(x) > 0 else 0
    psi = np.roll(psi, offset)
    return np.clip(psi, 0, 1)


def construct_bases(x_grid, N_bases, x0=0.0, xf=2.0, spacing=0.2):
    x_vals = np.linspace(x0, xf, len(x_grid))
    basis_centers = np.linspace(x0 + spacing, xf - spacing, N_bases)
    bases = []
    for xc in basis_centers:
        psi = basis_function(x_vals, xc)
        bases.append(psi)
    bases = np.array(bases)
    return bases


def run(budget_s=1100, N_bases=10, pop_budget=50, max_iter=5, mut_eps=0.1, n_points=42):

    start_time = time.time()
    best_h = None
    best_C5 = np.inf

    x_grid = np.linspace(0, 2, n_points)
    coarse_grid = np.linspace(0, 2, 500)

    if initial_h_values is not None:
        centers = np.argsort(initial_h_values)[-N_bases:][::-1]
        x_init = x_grid[centers]
        spacing = float(np.min(np.diff(x_init)) / 2)
    else:
        x_init = np.linspace(0.1, 1.9, N_bases)

    for iter_round in range(max_iter):
        centers = x_init if iter_round == 0 else x_init + (np.random.rand(N_bases) - 0.5) * mut_eps
        basis_centers = centers.copy()
        bases = construct_bases(coarse_grid, N_bases)
        A = bases / np.sum(bases, axis=0, keepdims=True)  # Normalize over grid

        a = cp.Variable(N_bases, nonneg=True)
        integral = np.sum(A * np.gradient(x_grid, axis=1)).mean(axis=1)  # Integral per basis
        constraint = cp.sum(integral * a) == 1
        C = np.outer(A[0], 1 - A[0])  # Placeholder, will be updated
        overlaps = [cp.convolve(A[i], A[j])[n_points // 2:3 * n_points // 2] for i, j in
                    np.ndindex((N_bases, N_bases)) if i <= j and i != j]
        lambda_var = cp.Variable()

        problem = cp.Problem(cp.Minimize(lambda_var),
                             [constraint,
                              a <= 1,
                              cp.sum(overlaps) <= lambda_var])

        try:
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                a_vals = a.value
                h_c = np.dot(A[:N_bases], a_vals)
                h_resampled = np.interp(x_grid, coarse_grid, h_c)
                h_feasible = ensure_feasible(h_resampled)
                C5 = evaluate_C5(h_feasible)
                if C5 < best_C5:
                    best_h = h_feasible
                    best_C5 = C5
                    x_init = centers + (np.random.rand(N_bases) - 0.5) * mut_eps
        except Exception as e:
            pass

        if time.time() - start_time >= budget_s:
            break

    if best_h is None and initial_h_values is not None:
        best_h = ensure_feasible(np.copy(initial_h_values))
        best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
