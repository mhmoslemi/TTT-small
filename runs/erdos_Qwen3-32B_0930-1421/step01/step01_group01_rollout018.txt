import numpy as np
import time
import cvxpy as cp


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2.0 / n_points
    cor = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h_init = initial_h_values.copy()
        if len(h_init) != n_points:
            h_resize = np.zeros(n_points)
            step = n_points / len(h_init)
            for i in range(n_points):
                h_resize[i] = h_init[int(i // step)]
            h = np.clip(h_resize, 0, 1)
            h /= h.sum() * dx * n_points / 2  # scale to 1
        else:
            h = np.clip(h_init, 0, 1)
            h /= h.sum() * dx * n_points / 2
    except:
        h = np.zeros(n_points)
        h[:-1] = 0.001  # small positive mass
        h[len(h)//3 - 1] = 1.0
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx * n_points / 2

    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)
    start_time = time.time()

    iter_count = 0
    perturbation_amt = 0.05
    lambda_ = 0.1
    mu = 0.001
    iters_per_decay = 50
    solve_timeout = 100  # 100 s per cvxpy call

    while time.time() - start_time < budget_s - 20:  # allow buffer for final evaluation
        iter_count += 1
        h = best_h.copy()
        h = np.clip(h + np.random.rand(*h.shape) * perturbation_amt - perturbation_amt / 2, 0, 1)
        h /= h.sum() * dx * n_points / 2  # renormalize for area = 1

        # Define CVXPY optimization
        h_var = cp.Variable(n_points, nonneg=True)
        obj = lambda_ * cp.sum([cp.square(cp.conv(h_var, 1 - h_var, mode="full")[k]) for k in range(-n_points + 1, n_points)])
        obj += mu * cp.sum_squares(h_var)
        constraints = [cp.sum(h_var) * dx == 1, h_var <= 1, h_var >= 0]
        problem = cp.Problem(cp.Minimize(obj), constraints)

        # Solve with timeout
        start_solver = time.time()
        try:
            problem.solve(solver=cp.OSQP, max_iter=10000, eps_abs=1e-7, eps_rel=1e-7, verbose=False, warm_start=True)
            if (time.time() - start_solver) > solve_timeout:
                raise TimeoutError
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_soln = np.clip(problem.solution.primal_vars[0], 0, 1)
                h_soln /= h_soln.sum() * dx * n_points / 2
                current_C5 = evaluate_C5(h_soln, dx)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_soln
        except:
            pass

        # Adapt regularization
        if iter_count % iters_per_decay == 0:
            lambda_ *= 0.8
            mu *= 0.8

        if iter_count % 4 == 0:
            perturbation_amt = min(0.2, perturbation_amt * 1.05)

    h_vals = best_h.copy()
    c5_bound = best_C5
    n_pts = n_points
    return h_vals, c5_bound, n_pts
