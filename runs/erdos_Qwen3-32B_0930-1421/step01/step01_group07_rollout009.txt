import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, x, y):
    h = h.copy()
    h[x], h[y] = h[y], h[x]
    return h

def find_local_best_flip(h):
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    for i in ones:
        for j in zeros:
            if i != j:
                h_candidate = flip_two_bits(h, i, j)
                c5 = evaluate_overlap(h_candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate
                    if best_c5 == 0:
                        return best_h, best_c5
    return best_h, best_c5

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
