import numpy as np
import time
import scipy.optimize as opt
from scipy.signal import correlate
import scipy.interpolate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def perturb_h(h, amplitude=0.1, segment_size=None, n_points=None):
    """Apply two types of perturbation to h: segment flip and Gaussian noise."""
    h = h.copy()
    if segment_size is None:
        segment_size = max(1, int(0.1 * n_points))
    start_id = np.random.randint(0, n_points - segment_size)
    h[start_id:start_id+segment_size] = 1 - h[start_id:start_id+segment_size]
    h += amplitude * np.random.randn(n_points)
    h = np.clip(h, 0, 1)
    return rescale_h(h)
