import numpy as np
import math
from scipy.signal import correlate
import time

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def localize_contribution_index(h, offset):
    """Find high-contribution positions at a given offset shift"""
    k_middle = offset
    if k_middle >= n:
        mask = np.roll(h, -k_middle)
    else:
        mask = np.roll(h, k_middle - n) if k_middle < n else np.roll(h, -(2 * n - 1 - k_middle))
    overlap = h * (1 - mask)
    flat_overlap = overlap[overlap != 0]
    top_indices = np.argpartition(np.abs(flat_overlap), -2)[-2:]
    sorted_indices = np.flatnonzero(overlap != 0)[top_indices]
    return np.sort(sorted_indices)

def locality_aware_flip(h):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    i = np.random.choice(indices_1 + (indices_1.mean(),)).astype(int)
    i = np.clip(i, 0, n - 1)
    i = indices_1[np.argmin(np.abs(indices_1 - i))]
    j = np.random.choice(indices_0 + (i + n // 4,) % n).astype(int)
    j = np.clip(j, 0, n - 1)
    j = indices_0[np.argmin(np.abs(indices_0 - j))]
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

    t0 = time.time()

            current_h = h_new
            current_c5 = c5_new
            best_h = current_h.copy()
            best_c5 = c5_new
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement > max_no_improvement:
            local_corr = correlate(best_h, 1 - best_h, mode="full")
            top_k = np.argmax(local_corr)
            patch_idx = localize_contribution_index(best_h, top_k)
            if len(patch_idx) >= 2:
                for idx1, idx2 in np.column_stack((patch_idx, np.roll(patch_idx, 1))):
                    if best_h[idx1] == 1 and best_h[idx2] == 0:
