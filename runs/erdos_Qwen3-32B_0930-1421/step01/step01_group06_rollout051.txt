import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h * scale_factor, 1.0)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    def evaluate_objective(h_eval):
        corr = np.correlate(h_eval, 1 - h_eval, mode='full')
        return np.max(corr * (2.0 / n_points))

    while time.time() - start_time < budget_s:

        h_opt = cp.Variable(n_points)
        constraints = [
            0 <= h_opt,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2.0
        ]

        # Surrogate upper bound constraint tightening
        overlap = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        upper_bound = np.max(overlap)
        C5_var = cp.Parameter(value=upper_bound)
        constraints += [cp.max(cp.correlate(h_opt, 1 - h_opt)) <= C5_var * n_points / 2.0]

        objective = cp.Minimize(C5_var)
        problem = cp.Problem(objective, constraints)

        problem.solve(solver=cp.CPLEX, verbose=False)
        if problem.status == 'optimal':
            h_new = np.clip(h_opt.value, 0, 1)
            c5_new = evaluate_objective(h_new)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h[:] = h_new
                last_improvement_time = time.time()

        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
