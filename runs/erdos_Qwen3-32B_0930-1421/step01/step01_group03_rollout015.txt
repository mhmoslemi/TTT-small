import numpy as np

def create_initial_population(n_points):
    base_h = np.zeros(n_points)
    step = n_points // 8
    indices = np.arange(0, n_points, step) % n_points
    base_h[indices[:n_points//4]] = 0.98
    base_h[indices[n_points//4:]] = 0.02

    population = [base_h.copy() for _ in range(5)]
    for idx, h in enumerate(population):
        # Add simple perturbation to similar vectors to form an initial population
        random.seed(idx + 42)
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        population[idx] = np.clip(h + perturbation, 0, 1)
        population[idx] *= (n_points / 2) / population[idx].sum()

    if 'initial_h_values' in globals():
        population.extend([np.array(initial_h_values) * (n_points / 2) / np.array(initial_h_values).sum()])
    
    population = [h for h in population if not np.isnan(h).any()]
    return np.array(population)

def evaluate_population(population, n_points):
    dx = 2.0 / n_points
    results = []
    for h in population:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        score = np.max(overlaps)
        results.append((score, h))
    return sorted(results, key=lambda x: x[0])

def crossover(h1, h2):
    mask = np.random.rand(len(h1)) < 0.5
    child = np.where(mask, h1, h2)
    child = np.clip(child, 0, 1)
    child *= (len(h1) / 2) / child.sum()
    return child

def mutate(h, mutation_strength=0.2):
    h = h.copy()
    perturbation = np.random.normal(loc=0, scale=mutation_strength, size=len(h))
    h += perturbation
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / h.sum()
    return h

def objective(h, n_points):
    h = h.copy()
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / h.sum()
    overlaps = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
    return np.max(overlaps)

def optimize_trajectory(initial_h, max_iter=500):
    h = np.array(initial_h, copy=True)
    dx = 2.0 / len(h)
    constraint_sum = len(h) / 2
    bounds = [(0, 1) for _ in range(len(h))]
    
    def bound_constrained_objective(h_flat):
        h = h_flat.reshape(len(h_flat))
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            h = np.zeros_like(h)
            h[0] = 1
            h_sum = 1
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    result = minimize(
        bound_constrained_objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    h_opt = np.clip(result.x, 0, 1)
    h_sum = h_opt.sum()
    if h_sum == 0:
        h_opt[0] = 1.0
        h_sum = 1.0
    h_opt = h_opt * (constraint_sum / h_sum)
    return h_opt

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 47

    population = create_initial_population(n_points)
    best_c5 = np.inf
    best_h = None
    dx = 2.0 / n_points
    start_time = time.time()

    population = evaluate_population(population, n_points)

    while time.time() - start_time < budget_s * 0.98:
        # Select top 25% of population to breed
        new_population = []
        top_candidates = [ind[1] for ind in population[:len(population) // 4]]

        for _ in range(len(population)):
            parent1 = random.choice(top_candidates)
            parent2 = random.choice(top_candidates)
            child = crossover(parent1, parent2)

            if random.random() < 0.4:
                child = mutate(child)
            
            new_population.append(child)

        # Random perturbation as additional exploration
        for _ in range(len(population) // 2):
            noise_factor = 0.1
            perturbed_h = mutate(top_candidates[-1], mutation_strength=noise_factor)
            new_population.append(perturbed_h)

        # Local optimization of best candidates prior to reselection
        for candidate in top_candidates[:2]:
            if time.time() - start_time > budget_s * 0.98:
                break
            optimized = optimize_trajectory(candidate)
            new_population.append(optimized)

        population = evaluate_population(np.array(new_population), n_points)

        current_c5 = population[0][0]
        current_h = population[0][1]

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        if (time.time() - start_time) > budget_s * 0.99:
            break

    # Final refinement to stabilize result
    if best_h is not None:
        best_h = optimize_trajectory(best_h, max_iter=1000)
        best_h = np.clip(best_h, 0, 1)
        best_h *= (n_points / 2) / best_h.sum()

        dx = 2.0 / n_points
        final_overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = np.max(final_overlaps)
    else:
        best_h = population[0][1]
        best_c5 = population[0][0]

    return best_h, best_c5, n_points
