import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Cap n_points to ensure small sequences
    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        h = h * (constraint_sum / h_sum)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def optimize_with_constraints(h0):
        h = cp.Variable(n_points)
        ones = np.ones(n_points)
        obj = cp.Maximize(-cp.max(cp.conv(h, 1 - h)) * dx)
        constraints = [h >= 0, h <= 1, h @ ones == constraint_sum]
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        return h.value

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        idx = np.random.choice(n_points, size=min(5, n_points), replace=False)
        delta = np.random.uniform(-noise_factor, noise_factor, size=len(idx))
        h[idx] += delta
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # Initial solver try
    h_opt = optimize_with_constraints(best_h)
    if h_opt is not None:
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        new_c5 = objective(h_opt)
        if new_c5 < best_c5:
            best_h = h_opt
            best_c5 = new_c5

    # Perturbation loop
    best_h_record = set()
    tabu_len = 10
    tabu = []
    noise_factor = 0.1
    cooldown_counter = 0
    cooldown_max = 50

    while time.time() - start_time < budget_s * 0.95:
        h_current = perturb_h(best_h, noise_factor)
        h_tuple = tuple(h_current.round(10))
        if h_tuple in best_h_record:
            continue
        best_h_record.add(h_tuple)

        h_current = h_current.copy()
        h_opt = optimize_with_constraints(h_current)
        if h_opt is not None:
            h_opt = np.clip(h_opt, 0, 1)
            cur_sum = h_opt.sum()
            if cur_sum != 0:
                h_opt = h_opt * (constraint_sum / cur_sum)
                obj = objective(h_opt)
                if obj < best_c5:
                    best_h = h_opt
                    best_c5 = obj
                    cooldown_counter = 0  # reset if improvement
                else:
                    cooldown_counter += 1

                if cooldown_counter >= cooldown_max and len(tabu) > 0:
                    tabu.pop(0)
                    noise_factor *= 0.8

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
