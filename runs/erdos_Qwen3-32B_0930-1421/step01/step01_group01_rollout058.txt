import numpy as np
import time
import math
from scipy.signal import correlate


def perturb_h(h, scale=0.2, n_points=None):
    if n_points is None:
        n_points = len(h)
    indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
    delta = np.random.randn(len(indices)) * scale
    h_perturbed = h.copy()
    h_perturbed[indices] += delta
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed /= h_perturbed.sum() * 2.0 / n_points
    return h_perturbed


def evaluate_C5(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * n_points / 2  # normalize to satisfy integral constraint
    cor = correlate(h, 1 - h, mode='full')
    C5 = np.max(cor) * (2.0 / n_points)
    return C5
