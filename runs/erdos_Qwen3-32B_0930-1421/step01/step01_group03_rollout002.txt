import numpy as np
import time
import copy
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_best = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_best = np.zeros(n_points)
        for i in range(n_points // 2):
            h_best[i + i] = 2.0 / n_points
        h_best = h_best / h_best.sum() * (n_points / 2)

    dx = 2.0 / n_points

    def evaluate(h):
        h = h / h.sum() * (n_points / 2)  # Enforce sum constraint
        h = np.clip(h, 0, 1)  # Enforce range constraint
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    best_c5 = evaluate(h_best)

    def tabu_search(max_iter=500, tabu_size=50, delta_start=0.3):
        tabu_list = []

        for it in range(max_iter):
            if time.time() - start_time > budget_s:
                break

            # Perturb h slightly
            h_new = h_best.copy()
            idx = np.random.randint(n_points)
            noise = np.random.randn(n_points) * delta_start
            h_new += noise / 10.0  # scaled small perturbation

            # Feasibility repair
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / h_new.sum() * (n_points / 2)

            # Avoid revisiting known configurations
            if any(np.allclose(h_new, t) for t in tabu_list):
                continue

            # Evaluate new solution
            c5_new = evaluate(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_best[:] = h_new[:]
                delta_start *= 0.9  # Cool perturbation amplitude
            else:
                delta_start *= 1.02  # Hypertune perturbation amplitude

            # Add to tabu
            tabu_list.append(h_new)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)

    # Gradient refinement
    def gradient_refine(max_iter=500):
        bounds = [(0, 1)] * n_points

        def objective_f(h):
            h = h / h.sum() * (n_points / 2)
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlap)

        result = minimize(
            objective_f,
            h_best,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter}
        )
        h_new = np.clip(result.x / result.x.sum() * (n_points / 2), 0, 1)
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            return h_new, c5_new
        return h_best, best_c5

    # Tabu + gradient refinement loop
    rounds = 0
    while time.time() - start_time < budget_s and rounds < 5:
        rounds += 1
        tabu_search(max_iter=500, delta_start=0.3)
        h_best, best_c5 = gradient_refine()

    # Final check of constraints
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best / h_best.sum() * (n_points / 2)

    return h_best, best_c5, n_points
