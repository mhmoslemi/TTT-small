import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel
import scipy.optimize as opt

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_ext = np.tile(h, 4)
    corrs = []
    for k in range(1, n_points):
        val = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(val)
    return np.max(np.array(corrs) * dx)

def proposal_operator(h, sigma, n_points):
    h_p = h + np.random.normal(0, sigma, n_points)
    return enforce_normalization(h_p, n_points)

def score_candidate(gp, h, n_points, temperature):
    h = h.reshape(-1)
    mean, _ = gp.predict(h[None, :], return_std=True)
    delta = 0.5 * temperature * np.random.randn(n_points)
    return h + 0.2 * (delta * np.sign(h - 0.5)).clip(-0.1, 0.1)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h_init = initial_h_values
        n_points = len(h_init)
    else:
        n_points = 47
        h_init = np.random.rand(n_points)
        h_init = enforce_normalization(h_init, n_points)

    h_current = h_init.copy()
    val_current = compute_c5(h_current, n_points)

    X = [h_current.copy()]
    y = [val_current]

    kernel = 1.0 * RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2)) + 0.1 * WhiteKernel()
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-8, normalize_y=True)

    h_best = h_current.copy()
    val_best = val_current
    time_start = time.time()
    T = 1000.0
    alpha = 0.99

    while time.time() - time_start < budget_s * 0.9:
        if len(X) >= max(2, min(n_points, 20)):
            X_train = np.array(X)
            y_train = np.array(y)
            gp.fit(X_train, y_train)

        h_p = proposal_operator(h_current, sigma=0.1, n_points=n_points)
        val_p = compute_c5(h_p, n_points)

        if val_p < val_current or np.random.rand() < np.exp((val_current - val_p) / T):
            h_current = h_p.copy()
            val_current = val_p

        if val_current < val_best:
            h_best = h_current.copy()
            val_best = val_current
            if len(y) < 100:
                X.append(h_best.copy())
                y.append(val_best)

        T *= alpha
        T = max(0.01, T)

    # Final refinement
    bounds = [(0, 1)] * n_points
    res = opt.minimize(lambda x: compute_c5(x, n_points), h_best, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    h_best = enforce_normalization(res.x, n_points)
    val_best = compute_c5(h_best, n_points)

    return h_best, val_best, n_points
