import numpy as np
import scipy.signal
import time

def evaluate_overlap(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h1 = 1 - h
    corr = scipy.signal.correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, indices_1, indices_0):
    i = np.random.choice(indices_1)
    j = np.random.choice(indices_0)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def repair_solution(h, n_points, n_half):
    while np.sum(h) != n_half:
        if np.sum(h) < n_half:
            h[np.argmin(h)] = 1
        else:
            h[np.argmax(h)] = 0
    return h
