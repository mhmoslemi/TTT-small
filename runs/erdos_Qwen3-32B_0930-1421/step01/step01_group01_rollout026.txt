import numpy as np
import time
import math

import scipy.signal


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def evaluate_C5(h, dx):
    """Evaluate C5 = max(correlate(h, 1-h, mode='full')) * dx."""
    h = h.copy()
    h = rescale_h(h)
    h_flat = (1 - h) * h
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def perturb_h(h, amplitude=0.1, segment_size=10):
    """Perturb a segment of h with clipping and normalization."""
    n = len(h)
    idx = np.random.randint(0, n - segment_size + 1)
    h_segment = h[idx:idx + segment_size].copy()
    delta = (np.random.rand(segment_size) - 0.5) * amplitude
    perturbed = h_segment + delta
    perturbed = np.clip(perturbed, 0, 1)
