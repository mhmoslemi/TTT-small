```python
import numpy as np
import time
from scipy.signal import correlate
import heapq

initial_h_values = None  # provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / np.clip(np.sum(h), 1e-8, np.inf)
    return h


def generate_initial_h(n_points):
    if initial_h_values is not None and len(initial_h_values) == n_points:
        return np.copy(initial_h_values)
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    np.random.shuffle(h)
    return ensure_feasible(h)


def discretize_h(h):
    # Map continuous h to discrete values: {0, 0.25, 0.5, 0.75, 1}
    h_disc = np.around(h * 4)
    h_disc = np.clip(h_disc, 0, 4) / 4.0
    return h_disc


def gradient_descent_step(h, step_size=0.01):
    # Compute gradient of C5 w.r.t. h via numerical differentiation
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h

    # Forward difference
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(n_points):
        h_perturbed = np.copy(h)
        h_perturbed[i] += eps
        h_perturbed = ensure_feasible(h_perturbed)
        obj_perturbed = evaluate_C5(h_perturbed)
        obj_default = evaluate_C5(h)
        grad[i] = (obj_perturbed - obj_default) / eps

    h_new = h - step_size * grad
    return ensure_feasible(h_new)


def run(budget_s=1100, n_points=42, pop_size=15, max_steps=500):
    start_time = time.time()
    best_h = generate_initial_h(n_points)
    best_C5 = evaluate_C5(best_h)
    best_h_disc = discretize_h(best_h)

    # Local search with gradient descent and discretization
    t = 0
    while time.time() - start_time < budget_s and t < max_steps:
        t += 1
        # Step 1: Gradient descent in continuous space
        h_i = np.copy(best_h)
        h_i = gradient_descent_step(h_i)
        h_i = discretize_h(h_i)
        h_i = ensure_feasible(h_i)
        C5 = evaluate_C5(h_i)
        if C5 < best_C5:
            best_h_disc = h_i
            best_C5 = C5
            best_h = h_i

        # Step 2: Perturb in discrete space
        idx = np.random.randint(n_points)
        new_val = [0.0, 0.25, 0.5, 0.75, 1.0][np.random.randint(5)]
        h_i = np.copy(best_h_disc)
        h_i[idx] = new_val
        h_i = ensure_feasible(h_i)
        h_i_disc = discretize_h(h_i)
        C5 = evaluate_C5(h_i_disc)
        if C5 < best_C5:
            best_h_disc = h_i_disc
            best_C5 = C5
            best_h = h_i_disc

    return best_h_disc, best_C5, n_points
```
