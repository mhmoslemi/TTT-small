```python
import numpy as np
import time
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total < 1e-6:
        h = np.zeros(n_points)
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h


def unitize_h(h):
    h = np.round(h * 4) / 4  # Snap to 0.25 increments
    return ensure_feasible(h, len(h))


def perturb_h(h, mut_rate=0.1, perturb_amount=0.25):
    h = h.copy()
    n_points = len(h)
    mask = np.random.rand(n_points) < mut_rate
    if not np.any(mask):
        return h
    h[mask] += np.random.choice((-perturb_amount, 0, perturb_amount), size=np.sum(mask))
    h[mask] = np.clip(h[mask], 0, 1)
    return ensure_feasible(h, n_points)


def variable_impact(h, n_points):
    res = np.zeros(n_points)
    for i in range(n_points):
        future = (i + 1) % n_points
        past = (i - 1) % n_points
        res[i] = max(h[future], h[past])
    return res


def tree_search(h, budget_s, n_points=42, res=5, perturb_rate=0.1, max_depth=5):
    from collections import deque
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h)
    start_time = time.time()
    queue = deque()
    queue.append((best_h, best_C5, 0))

    while queue and (time.time() - start_time) < budget_s:
        current, current_C5, depth = queue.popleft()
        if depth >= max_depth:
            if evaluate_C5(current) < current_C5:
                best_h = ensure_feasible(current, n_points)
                best_C5 = evaluate_C5(best_h)
            continue

        # Branching: perturb and queue
        for _ in range(res):
            candidate = perturb_h(current, mut_rate=perturb_rate, perturb_amount=0.25)
            candidate_C5 = evaluate_C5(candidate)
            if candidate_C5 < best_C5:
                queue.append((candidate, candidate_C5, depth + 1))

    return best_h, best_C5


def local_improve(h, local_steps=100, mut_rate=0.05, perturb_amount=0.25):
    current_h = ensure_feasible(h, len(h))
    current_C5 = evaluate_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5

    for _ in range(local_steps):
        candidate = perturb_h(current_h, mut_rate=mut_rate, perturb_amount=perturb_amount)
        candidate = ensure_feasible(candidate, len(candidate))
        c5 = evaluate_C5(candidate)
        if c5 < best_C5:
            best_h = candidate
            best_C5 = c5
            current_h = candidate
            current_C5 = c5

    return best_h, best_C5


def run(budget_s=1100, pop_size=50, n_points=42, res=5):
    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    initial = unitize_h(initial_h_values if initial_h_values is not None else np.ones(n_points) * 0.5)

    while (time.time() - start_time) < budget_s * 0.8:
        # Try tree search
        h, c = tree_search(initial, budget_s * 0.05, n_points, res=res, perturb_rate=0.1, max_depth=3)
        h, c = local_improve(h)
        if c < best_C5:
            best_h = h
            best_C5 = c

    return best_h, best_C5, n_points
```
