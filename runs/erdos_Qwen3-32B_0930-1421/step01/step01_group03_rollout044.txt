import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    pop_size = 10
    max_iter = 50
    patience = 50

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        base_h = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        base_h = np.zeros(n_points)
        step = n_points // 2
        for i in range(step):
            base_h[i + i] = 2.0 / n_points
        base_h *= (n_points / 2 / base_h.sum())

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            return h  # Avoid divide-by-zero edge case
        return h * (target_sum / s)

    def compute_c5(h):
        h.shape = (n_points,)
        h = enforce_constraints(h)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def objective(h):
        return np.clip(compute_c5(h), 0, 1)  # np.clip stabilizes gradient paths

    def perturb(h, noise_strength=0.15):
        h = h.copy()
        per = np.random.rand(n_points)
        noise = per * noise_strength * (np.random.rand(n_points) * 2 - 1)
        h += noise
        h = enforce_constraints(h)
        return h

    def crossover(h1, h2):
        mix = np.random.rand(n_points)
        child = h1 * (1 - mix) + h2 * mix
        return enforce_constraints(child)

    # Initialize population
    population = [perturb(base_h, noise_strength=0.1) for _ in range(pop_size)]
    best_idx = np.argmin([compute_c5(h) for h in population])
    best_h = population[best_idx]
    best_c5 = compute_c5(best_h)

    last_improvement = 0
    for gen in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            # Parent selection by roulette
            p1, p2 = random.choices(population, weights=[1/(1 + h) for h in [v for v in population]], k=2)
            child = crossover(p1, p2)
            child = minimize(objective, child, bounds=[(0, 1)] * n_points, method='L-BFGS-B', options={'maxiter': 10}).x
            child = enforce_constraints(child)
            new_pop.append(child)
        # Add mutations
        for h in population[int(pop_size/2):]:
            new_pop.append(perturb(h))
        # Evaluate
        pop_scores = [objective(h) for h in new_pop]
        population = [h for _, h in sorted(zip(pop_scores, new_pop), key=lambda x: x[0])][:pop_size]
        # Update best
        gen_c5 = min(pop_scores)
        if gen_c5 < best_c5:
            best_c5 = gen_c5
            last_improvement = gen
            best_h = population[0].copy()
        # Early stop if no improvement
        if gen - last_improvement > patience:
            break

    # Final polishing
    final_result = minimize(objective, best_h, bounds=[(0, 1)] * n_points, method='L-BFGS-B', options={'maxiter': 100})
    final_h = enforce_constraints(final_result.x)
    final_c5 = objective(final_h)

    final_h = final_h.clip(0, 1)
    final_h = final_h * (target_sum / final_h.sum())
    elapsed = time.time() - start_time
    return final_h, final_c5, n_points
