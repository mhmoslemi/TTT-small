import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings

warnings.filterwarnings("ignore")

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h_vec, n_points):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    corrs = []
    for k in range(1, n_points):
       蔻 = np.dot(h[:n_points - k], 1 - h[k:]) * dx
        corrs.append(蔻)
    for k in range(1, n_points):
       蔻 = np.dot(h[k:], 1 - h[:n_points - k]) * dx
        corrs.append(蔻)
    return np.max(corrs)

def generate_candidate(h_base, noise_std, n_points):
    h = h_base.copy()
    noise = np.random.normal(0, noise_std, n_points)
    h += noise
    return enforce_normalization(h, n_points)

def eval_surrogate(gp, h_vec):
    return gp.predict(h_vec[None, :])

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    n_init_samples = 20
    h_candidates = []
    
    for _ in range(n_init_samples):
        h = generate_candidate(initial_h_values, 0.1, n_points)
        h_candidates.append(h)
        
    h_array = np.vstack(h_candidates)
    c5_values = [compute_c5(h, n_points) for h in h_array]
    best_idx = np.argmin(c5_values)
    h_best = h_array[best_idx]
    c5_best = c5_values[best_idx]
    
    # Surrogate model setup
    kernel = C(1.0, (1e-2, 1e2)) * RBF(n_points, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6)
    gp.fit(h_array, c5_values)
    
    time_start = time.time()
    T = 1000
    alpha = 0.99
    noise_std = 0.1
    eval_budget = int((budget_s - (time.time() - time_start)) * 0.8)
    
    for t in range(eval_budget):
        if time.time() - time_start >= budget_s:
            break
        
        h = h_best.copy()
        h_candidate = generate_candidate(h, noise_std, n_points)
        c5_pred = eval_surrogate(gp, h_candidate)
        c5_true = compute_c5(h_candidate, n_points)
        
        if c5_true < c5_best or np.random.rand() < np.exp((c5_best - c5_true) / T):
            h_best = h_candidate.copy()
            c5_best = c5_true
        
        if t % 10 == 0:
            gp.fit(h_array, c5_values)
        
        # Update records
        h_array = np.concatenate([h_array, [h_candidate]], axis=0)
        c5_values.append(c5_true)
        
        # Cooling Schedule
        T *= alpha
        noise_std = max(noise_std * alpha, 1e-4)
        
        if t % 50 == 0 and len(c5_values) > n_init_samples:
            n_keep = min(len(c5_values) - 1, 50)
            idx_keep = np.argsort(c5_values)[::-1][n_keep:]
            h_array = h_array[idx_keep, :]
            c5_values = [c5_values[i] for i in idx_keep]
    
    # Final refinement
    def objective(x):
        return compute_c5(x, n_points)
    
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        res = opt.minimize(objective, h_best, bounds=[(0, 1)] * n_points, method='L-BFGS-B')
    
    h_best = res.x
    h_best = enforce_normalization(h_best, n_points)
    c5_best = compute_c5(h_best, n_points)
    
    return h_best, c5_best, n_points
