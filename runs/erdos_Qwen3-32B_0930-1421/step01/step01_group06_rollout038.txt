import numpy as np
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def simulated_annealing_binary(n_points, n_ones, max_iter, T0=1.0, alpha=0.99):
    # Build a binary initial h with n_ones ones
    idx = np.random.choice(n_points, n_ones, replace=False)
    bin_h = np.zeros(n_points)
    bin_h[idx] = 1.0
    current_h = bin_h
    best_h = np.copy(current_h)
    current_cost = evaluate_erdos_solution(current_h, n_points)

    best_cost = current_cost
    T = T0
    for iteration in range(max_iter):
        # Select a random index
        i1 = np.random.randint(n_points)
        while current_h[i1] == 0:
            i1 = np.random.randint(n_points)

        i0 = np.random.randint(n_points)
        while current_h[i0] == 1:
            i0 = np.random.randint(n_points)

        # Flip bits to maintain sum constraint
        new_h = np.copy(current_h)
        new_h[i1] = 0
        new_h[i0] = 1

        # Evaluate new cost
        new_cost = evaluate_erdos_solution(new_h, n_points)
        delta = new_cost - current_cost

        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h = new_h
            current_cost = new_cost
            if current_cost < best_cost:
                best_h = np.copy(current_h)
                best_cost = current_cost

        T *= alpha

    return best_h

def optimize_continuous(h_bin, n_points):
    # Scale binary h to continuous with constraint sum(h) == n_points / 2
    sum_h = np.sum(h_bin)
    h_cont = h_bin.copy().astype(float)

    if not np.isclose(sum_h, n_points / 2.0):
        if sum_h < n_points / 2.0:
            # Need to increase by (n_points / 2 - sum_h)
            idx = np.argmin(h_bin)
            h_cont[idx] = (n_points / 2.0 - np.sum(h_cont)) / 1.0
        else:
            # Need to decrease by (sum_h - n_points / 2)
            idx = np.argmax(h_bin)
            h_cont[idx] = (np.sum(h_cont) - n_points / 2.0) / 1.0
            h_cont[idx] = 1.0 - h_cont[idx]

    def objective(x):
        c5 = evaluate_erdos_solution(x, n_points)
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': constraint_sum})

    result = minimize(
        objective,
        h_cont,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100}
    )

    return result.x if result.success else h_cont
