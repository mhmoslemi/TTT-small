import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.clip(initial_h * scale_factor, 0.0, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time
    dx = 2.0 / n_points

    # Define variables for convex optimization
    h_var = cp.Variable(n_points, nonneg=True)

    # Define constraints: sum to 1, in range [0,1]
    constraints = [
        cp.sum(h_var) == n_points / 2.0,
        h_var <= 1,
        h_var >= 0
    ]

    objective_scale = 1.0 / (2 * dx)

    # Use Cvxpy to minimize the maximum correlation value
    def solve_convex_relaxation(h_init):
        nonlocal best_c5, best_h, last_improvement_time

        # Define slack variable for the constraint
        slack = cp.Variable(obj=value=0.1)

        # Construct the correlation constraints using outer products
        h_values = h_init
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        k_values = [int((len(h_values) - 1) / 2 + k) for k in range(len(corr))]

        conv_objective = 0
        for k in k_values:
            lhs = cp.sum(h_var * (1 - cp.roll(h_var, shift=k)) * dx)
            conv_objective += cp.MaxExpression([lhs - slack])

        # Define objective: minimize the slack
        problem = cp.Problem(cp.Minimize(slack), constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return

        if h_var.value is not None:
            new_h = np.clip(h_var.value, 0, 1)
            new_c5 = evaluate_erdos_solution(new_h, n_points)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h.copy()
                last_improvement_time = time.time()

    # Optimization loop
    while time.time() - start_time < budget_s - 5:
        solve_convex_relaxation(h)
        if time.time() - last_improvement_time > 120:
            break

        # Project solution to unit simplex to maintain feasibility
        h = np.clip(best_h, 0, 1)
        h = h / np.sum(h) * (n_points / 2.0)
        h = np.clip(h, 0, 1)

    return best_h, best_c5, n_points
