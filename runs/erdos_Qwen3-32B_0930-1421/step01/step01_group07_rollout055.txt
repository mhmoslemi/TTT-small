import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def clip_and_rescale(h, total_samples):
    h_clipped = np.clip(h, 0, 1)
    while True:
        h_sum = np.sum(h_clipped)
        if abs(h_sum - total_samples / 2) < 1e-6:
            break
        h_clipped *= (total_samples / 2) / h_sum
        h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

def run(seed=42, budget_s=1100, **kwargs):

    h = np.array(initial_h_values).copy()
    h = np.clip(h, 0, 1)
    h = clip_and_rescale(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    t0 = time.time()
    max_iter = 10000

    for it in range(max_iter):
        if time.time() - t0 >= budget_s:
            break
        corr = correlate(h, 1 - h, mode="full")
        k_shift = np.argmax(corr)
        lag = k_shift - (n_points - 1)
        grad = np.zeros(n_points)

        for i in range(n_points):
            if i + lag < n_points:
                grad[i] += (1 - h[i + lag])
            if i - lag >= 0:
                grad[i] -= h[i - lag]

        grad /= np.linalg.norm(grad)

        alpha = 1.0
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = clip_and_rescale(h_new, n_points)
            c5_new = evaluate_overlap(h_new)
            if c5_new < evaluate_overlap(h) - 1e-6:
                break
            alpha *= 0.5
        else:
            alpha *= 1.1  # Successful step, increase step size

        h = h - alpha * grad
        h = clip_and_rescale(h, n_points)
        c5 = evaluate_overlap(h)

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Escape local minima
        if it % 500 == 0 and np.random.rand() < 0.05:
            h += np.random.normal(0, 0.05, n_points)
            h = clip_and_rescale(h, n_points)

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8
    return (best_h, best_c5, n_points)
