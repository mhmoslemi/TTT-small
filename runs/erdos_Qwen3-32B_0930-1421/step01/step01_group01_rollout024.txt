import numpy as np

from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound for a given h."""
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    """Rescale h to satisfy the constraint ∑h = n / 2."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 0:
        h = h / h_sum * (len(h) / 2)
    else:
        h[:] = 0
        h[0] = len(h) / 2
    return h
