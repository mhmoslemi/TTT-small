import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, shift_peak, n):
    locations = np.arange(n)
    A = h.copy()

    # Find the most overlapping region on original h
    i1 = (shift_peak + np.where(h == 1)[0][:10]) % n
    i0 = (n - shift_peak + np.where(h == 0)[0][:10]) % n
    i1 = np.unique(np.concatenate((i1, i1.reverse().reshape(-1) + 1))) % n
    i0 = np.unique(np.concatenate((i0, i0.reverse().reshape(-1) + 1))) % n

    ones = i1[h[i1].astype(bool)]
    if len(ones) == 0:
        return A
    zeros = i0[~h[i0].astype(bool)]
    if len(zeros) == 0:
        return A

    # Flip the most impactful pair
    A[ones[0]] = 0.0
    A[zeros[0]] = 1.0
    return A
