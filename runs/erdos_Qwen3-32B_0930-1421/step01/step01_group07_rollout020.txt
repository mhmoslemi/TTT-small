import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, corr, k_shift, n_points):
    lag = k_shift - (n_points - 1)
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + lag < n_points:
            grad[i] += (1 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    return grad / np.linalg.norm(grad + 1e-8)

def project_h(h, target_sum):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (target_sum / sum_h)
    return np.clip(h, 0, 1)

    target_sum = n_points / 2

    # Enforce sum constraint
    while not np.isclose(np.sum(h), target_sum, atol=1e-8):
        if np.sum(h) < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5
    t0 = time.time()
    step_size = 1e-2
    backup_h = best_h.copy()
    backup_c5 = best_c5
    iterations = 0
    max_iterations = 10000
    min_step = 1e-8

    while time.time() - t0 < budget_s and iterations < max_iterations:
        corr = correlate(current_h, 1 - current_h, mode="full")
        k_shift = np.argmax(corr)
        grad = compute_gradient(current_h, corr, k_shift, n_points)
        candidate_h = current_h - step_size * grad

        # Projection and constraint enforcement
        h_projected = project_h(candidate_h, target_sum)

        new_c5 = evaluate_overlap(h_projected)
        if not np.isfinite(new_c5) or np.isnan(new_c5):
            current_h = backup_h
            backup_h = current_h.copy()
            current_c5 = backup_c5
            step_size *= 0.5
            step_size = max(step_size, min_step)
            iterations += 1
            continue

        if new_c5 < current_c5 or np.random.rand() < 0.01:
            current_h = h_projected
            current_c5 = new_c5
            backup_h = current_h.copy()
            backup_c5 = current_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
            step_size = min(1.2 * step_size, 1e-1)
        else:
            step_size = max(step_size * 0.8, min_step)
        iterations += 1

        if iterations % 500 == 0 and best_c5 > backup_c5:
            current_h = backup_h
            current_c5 = backup_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
