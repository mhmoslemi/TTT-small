import numpy as np
import time
from scipy.signal import correlate
import math

def ensure_feasibility(h, n_points):
    # Ensure [0,1] bounds
    h = np.clip(h, 0.0, 1.0)
    # Enforce sum constraint
    total = h.sum()
    if abs(total - n_points / 2) > 1e-8:
        h = h / total * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, lag, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + lag < n_points:
            grad[i] += (1 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    grad = grad / np.linalg.norm(grad + 1e-8)
    return grad

        if trial_c5 < best_c5:
            break
        alpha *= 0.5
    h[:] = best_h[:]
    return best_h, best_c5, alpha
