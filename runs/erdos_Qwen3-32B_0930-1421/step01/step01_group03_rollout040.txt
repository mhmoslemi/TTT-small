import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    n_points = 47 if not 'initial_h_values' in globals() else len(initial_h_values)
    dx = 2.0 / n_points

    h0 = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points / 2) / h0.sum()  # normalize to meet constraint

    # Constraint: sum(h) == n_points / 2
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (target_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def feasible_project(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            h = np.zeros_like(h)
        else:
            h = h * (target_sum / h_sum)
        return h

    def build_linearized_constraints(h_ref):
        h_ref = h_ref.reshape(n_points)
        overlaps_ref = np.correlate(h_ref, 1 - h_ref, mode='full') * dx
        maxoverlap_ref = objective(h_ref)

        A = []
        b = []

        for k in range(n_points*2 - 1):
            h_shifted = np.roll(h_ref, k - n_points + 1)
            h_shifted_wrapped = np.roll(h_ref, -k + n_points)
            for i in range(n_points):
                # Linear approximation of h_j * (1 - h_{j + k}) = h_j - h_j * h_{j + k}
                # f(h) = f(h0) + f_h0_j * (h - h0_j) ... partials need to be estimated
                approx = h_ref[i] - 2 * h_ref[i] * h_shifted[i]
                A.append(appprox)
                b.append(overlaps_ref[k])

        return np.array(A), np.array(b)

    best_h = h0
    best_c5 = objective(best_h)

    iter_count = 0
    waveform_users = []

    def perturb(h, noise_factor=0.05):
        h = h.reshape(n_points)
        perturbation = np.random.randn(n_points) * noise_factor
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (target_sum / h_perturbed.sum())
        return h_perturbed

    while time.time() - start_time < budget_s:
        try:
            h = feasible_project(np.random.normal(best_h, 0.05).ravel())
            A, b = build_linearized_constraints(h)

            h_var = cp.Variable(n_points, nonneg=True)
            h_var_opt = cp.Parameter(n_points)
            h_var_opt.value = h
            obj = cp.Minimize(cp.norm(h_var - h_var_opt, p=2))

            # Create constraints from linearized overlaps
            linearized_overlaps = []
            for a_row, b_row in zip(A, b):
                linearized_overlaps.append(a_row.numpy() @ h_var <= b_row.numpy())

            constraints = [
                cp.sum(h_var) == target_sum,
                h_var >= 0,
                h_var <= 1,
                *linearized_overlaps
            ]

            prob = cp.Problem(obj, constraints)
            result = prob.solve(solver="ECOS", verbose=False)
            if h_var.value is None:
                continue

            h_new = feasible_project(h_var.value)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                waveform_users.append((best_h, best_c5))

            iter_count += 1

            if iter_count % 5 == 0:
                h_new = perturb(best_h, noise_factor=0.1)
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    waveform_users.append((best_h, best_c5))

        except Exception as e:
            pass

        # Check for timeout
        if time.time() - start_time > budget_s * 0.98 and waveform_users:
            best_h, best_c5 = min(waveform_users, key=lambda x: x[1])

    return best_h, best_c5, n_points
