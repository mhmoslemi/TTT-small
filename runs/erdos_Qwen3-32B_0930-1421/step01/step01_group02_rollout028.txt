import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h[:] = 0.0
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def build_bases(n_basis, alpha, beta, n_points):
    dx = 2.0 / n_points
    x = np.arange(n_points) * dx
    bases = []
    centers = []  # Store physical x-positions
    for i in range(n_basis):
        x_i = 2.0 * (i + 1) / (n_basis + 1)
        psi = np.zeros(n_points)
        for j in range(n_points):
            if abs((j * dx) - x_i) <= beta:
                psi[j] = np.exp(-alpha * (x[j] - x_i) ** 2)
        bases.append(psi / np.sum(psi))
        centers.append(x_i)
    return np.array(bases), centers


def optimize_coeffs(bases, n_points):
    A = np.hstack([basis.reshape(-1, 1) for basis in bases])
    a = cp.Variable(len(bases))
    overlap = correlate(A @ a, 1.0 - A @ a, mode="full")
    dx = 2.0 / n_points
    overlap *= dx
    constraint = [A @ a >= 0, A @ a <= 1, a >= 0, a <= 1, (A.T @ a).value == (len(bases) / 2)]
    obj = cp.Minimize(cp.max(overlap))
    prob = cp.Problem(obj, constraint)
    result = prob.solve(solver='ECOS', verbose=False)
    return (A @ a).value


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    if initial_h_values is not None:
        prev_h = np.copy(initial_h_values)
    else:
        # default uniform distribution
        prev_h = np.ones(n_points) / n_points * (n_points / 2)
        prev_h = ensure_feasible(prev_h)

    best_h = None
    best_C5 = float('inf')

    n_basis = 10
    alpha = 4
    beta = 0.3

    # Build basis functions
    bases, centers = build_bases(n_basis, alpha, beta, n_points)

    # Main optimization loop
    perturbation_size = 0.1
    perturbed_centers = [c + np.random.uniform(-perturbation_size, perturbation_size) for c in centers]
    while time.time() - start_time < budget_s:
        new_bases = []
        for x in perturbed_centers:
            # Shifted basis around new center
            psi = np.zeros(n_points)
            dx_basis = 2.0 / n_points
            for j in range(n_points):
                if abs((j * dx_basis) - x) <= beta:
                    psi[j] = np.exp(-alpha * ((j * dx_basis) - x) ** 2)
            new_bases.append(psi / np.sum(psi) if np.sum(psi) > 1e-10 else np.zeros(n_points))

        A = np.vstack(new_bases)
        a = cp.Variable(len(new_bases))
        ones = np.ones_like(A[0])
        constraint = [A.T @ a >= 0, A.T @ a <= 1, a >= 0, a <= 1, cp.sum(A.T @ a) == (n_points / 2)]
        obj = cp.Minimize(cp.norm(A.T @ a - prev_h, 'fro'))
        prob = cp.Problem(obj, constraint)
        prob.solve(solver='ECOS', verbose=False)
        h = A.T @ a.value
        h = ensure_feasible(h)
        C5 = evaluate_C5(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
        centers = [np.sum(b[:i+1] * (2.0 / n_points)) for i, b in enumerate(bases)]
        perturbed_centers = [c + np.random.uniform(-perturbation_size, perturbation_size) for c in centers]
        if time.time() - start_time >= budget_s:
            break

    return best_h, best_C5, n_points
