import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    h_sum = float(np.sum(h))
    factor = (n_points / 2.0) / h_sum
    h = np.clip(h * factor, 0.0, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time

    def objective(x):
        c5 = evaluate_erdos_solution(x, n_points)
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    def optimize_with_cvxopt(max_iter=5, callback=None):
        nonlocal h, best_c5, best_h, last_improvement_time

        H = cp.Variable(n_points)
        obj = objective(h)
        C5 = cp.Parameter(nonneg=True, value=obj)

        constraints = [
            cp.sum(H) == n_points / 2.0,
            0 <= H,
            H <= 1
        ]

        overlap_constraint = [H * (1 - np.roll(H, shift=k)) <= C5 * (2 / n_points) for k in range(-n_points + 1, n_points)]
        constraints += overlap_constraint

        problem = cp.Problem(cp.Minimize(C5), constraints)

        for _ in range(max_iter):
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status != "optimal":
                break
            new_h = np.clip(np.array(H.value).ravel(), 0.0, 1.0)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h
                last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 5:
        optimize_with_cvxopt(max_iter=3)
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
