import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h * scale_factor, 1.0)

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time

    def optimize_with_convex_surrogate():
        nonlocal h, best_c5, best_h, last_improvement_time

        h_var = cp.Variable(n_points, nonneg=True)
        c5_var = cp.Parameter(nonneg=True)

        # Construct correlation constraints
        constrs = []
        for k in range(-n_points + 1, n_points):
            valid = np.where(np.arange(len(h)) + k < len(h) + k + 1)[0]
            val_ids = np.array([i for i in range(len(h)) if 0 <= i + k < len(h)])
            h_ids = h[val_ids]
            h_plus_k_ids = h[val_ids + k]
            corr = sum(h_var[val_ids] * (1 - h_var[val_ids + k]) for val_ids in zip(val_ids, val_ids + k))

            constrs.append(corr <= c5_var * dx)

        # Objective function: minimize c5_var
        problem = cp.Problem(
            cp.Minimize(c5_var),
            constrs + [cp.sum(h_var) == n_points / 2],
        )

        # Solve
        problem.solve(solver=cp.CVXOPT)

        if h_var.value is None:
            return  # Skip infeasible

        h_opt = np.clip(h_var.value, 0, 1)
        c5_opt_value = c5_var.value

        # Update the best solution if improved
        c5_current = evaluate_erdos_solution(h_opt, n_points)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h[:] = h_opt.copy()
            last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 5:  # 5s margin
        # Convex optimization with adaptive bias
        optimize_with_convex_surrogate()
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
