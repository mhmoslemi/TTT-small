import numpy as np
import time
from scipy.stats import norm

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points - k:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k]) * dx
        corrs.append(corr)
        corr_rev = np.dot(h[k:], 1 - h[:n_points - k]) * dx
        corrs.append(corr_rev)
    return np.max(corrs)

def surrogate_gp_predict(X, model):
    return model.predict(X)

def train_gp(X, y):
    from sklearn.gaussian_process import GaussianProcessRegressor
    from sklearn.gaussian_process.kernels import RBF, WhiteKernel
    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1000.0)) + WhiteKernel(
        noise_level=1e-5, noise_level_bounds=(1e-10, 1e-2)
    )
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=3, alpha=1e-6)
    gp.fit(X, y)
    return gp

def acquisition_expected_improvement(X, model, current_best):
    mu = surrogate_gp_predict(X, model)
    sigma = np.sqrt(np.diag(model.covariance(X)))
    sigma = np.maximum(sigma, 1e-9)
    imp = current_best - mu
    z = imp / sigma
    ei = imp * norm.cdf(z) + sigma * norm.pdf(z)
    return ei

def perturb_h(h, sigma, n_points):
    h = h.copy()
    h += np.random.normal(0, sigma, n_points)
    return enforce_normalization(h, n_points)

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    start_time = time.time()

    if init_h is None:
        h_current = np.random.rand(47)
    else:
        h_current = init_h.copy()
    h_current = enforce_normalization(h_current, 47)
    n_points = len(h_current)
    dx = 2.0 / n_points
    current_best_c5 = compute_c5(h_current, n_points)

    # Initial random evaluations for surrogate model
    num_init_points = 15
    X = []
    y = []
    for i in range(num_init_points):
        h = np.random.rand(n_points)
        h = enforce_normalization(h, n_points)
        c5 = compute_c5(h, n_points)
        X.append(h)
        y.append(c5)

    X = np.array(X)
    y = np.array(y)
    model = train_gp(X, y)
    best_c5 = np.min(y)
    best_h = X[y.argmin()].copy()

    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.97
    step = 0

    while time.time() - start_time < budget_s - 5:
        # Propose candidate point
        h_candidate = perturb_h(best_h, sigma=0.1 * temp, n_points=n_points)
        c5_candidate = compute_c5(h_candidate, n_points)

        # Compute acceptance based on improvement
        delta = c5_candidate - best_c5
        if c5_candidate < best_c5 or np.exp(-delta / (temp + 1e-6)) > np.random.rand():
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

        # Compute variance of candidate for SE kernel
        _, var = model.predict(np.array([h_candidate]), return_std=True)
        acq = acquisition_expected_improvement(np.array([h_candidate]), model, best_c5)
        X = np.vstack((X, h_candidate))
        y = np.append(y, c5_candidate)

        model = train_gp(X, y)
        temp *= cooling_rate
        step += 1
        if step % 5 == 0:
            # Run local optimization on the best found so far
            opt_result = scipy.optimize.minimize(
                lambda x: compute_c5(x, n_points),
                x0=best_h,
                bounds=[(0, 1)] * n_points,
                method="L-BFGS-B",
                options={"maxiter": 50},
            )
            opt_h = opt_result.x
            opt_h = enforce_normalization(opt_h, n_points)
            opt_c5 = compute_c5(opt_h, n_points)
            if opt_c5 < best_c5:
                best_c5 = opt_c5
                best_h = opt_h.copy()

    return best_h, best_c5, n_points
