import numpy as np
import scipy.stats as stats
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k]) * dx
        corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k]) * dx
        corrs.append(corr)
    return max(corrs)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    return np.clip(h * (n_points / 2) / total, 0.0, 1.0)

def perturb_h(h, noise_scale=0.2):
    h_new = h + noise_scale * np.random.randn(*h.shape)
    return enforce_normalization(h_new, len(h))
