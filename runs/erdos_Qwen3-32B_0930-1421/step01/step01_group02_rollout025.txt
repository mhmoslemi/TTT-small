```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # will be provided by environment


def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_binary_and_feasible(h):
    h = np.copy(h)
    h = np.round(h).clip(0, 1)
    total = h.sum()
    if abs(total - len(h) / 2) > 1e-8:
        # Adjust a single value to satisfy sum constraint
        idx = np.argmin(h)
        if h[idx] < 0.5:
            h[idx] = 1
        else:
            h[idx] = 0
    return h


def add_overlap_constraints(W, n_points, dx, active_shifts):
    overlap_constraints = []
    k_vals = np.arange(-n_points + 1, 1)  # All possible k = i - j
    for k in k_vals:
        if k in active_shifts:
            continue
        # Compute overlap for shift k
        indices = [(i, i + k) for i in range(n_points) if 0 <= i + k < n_points]
        if not indices:
            continue
        # Define bilinear overlaps
        overlap_expr = 0
        for i, j in indices:
            overlap_expr += W[i] * (1 - W[j])
        max_overlap = cp.Parameter(nonneg=True)
        overlap_constraints.append(overlap_expr <= max_overlap)
        active_shifts.append(k)
    return overlap_constraints


def run(budget_s=1100, n_points=42, max_shifts=50):
    start_time = time.time()
    if initial_h_values is not None:
        h0 = initial_h_values
    else:
        h0 = np.zeros(n_points)
        h0[:n_points // 2] = 1 / (n_points // 2)
        h0[n_points // 2:] = 1 / (n_points - n_points // 2)

    h0 = np.clip(h0, 0, 1)

    # Optimization variables: h and lambda (max overlap)
    h = cp.Variable(n_points, nonneg=True)
    lbda = cp.Variable(nonneg=True)

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]

    active_shifts = []
    constraints.extend(
        add_overlap_constraints(h, n_points, 2.0 / n_points, active_shifts)
    )

    # Build and solve optimization
    objective = cp.Minimize(lbda)
    prob = cp.Problem(objective, constraints)

    while time.time() - start_time < budget_s:
        try:
            prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=10)
        except Exception:
            break
        if h.value is None:
            break

        h_sol = np.ascontiguousarray(
            ensure_binary_and_feasible(np.clip(h.value, 0, 1))
        )
        lbda_sol = compute_overlap(h_sol)

        if active_shifts and lbda_sol <= lbda.value:
            # Solution meets all constraints
            break

        # Add new constraints for top-k worst shifts
        overlap = correlate(h.value, (1 - h.value), mode="full") * 2.0 / n_points
        top_k = np.argsort(overlap)[-min(max_shifts, len(overlap)):]  # bottom indices
        for idx in top_k:
            k_val = idx - n_points + 1
            if k_val not in active_shifts:
                active_shifts.append(k_val)
                for i, j in [(i, i + k_val) for i in range(n_points) if 0 <= i + k_val < n_points]:
                    constraints.append(h[i] * (1 - h[j]) <= lbda)
        prob = cp.Problem(objective, constraints)

    return h_sol, lbda_sol, n_points
```
