import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem using convex optimization."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if defined
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.random.uniform(0, 1, n_points)
        h = rescale_h(h)

    # CVXPY problem setup
    h_cvx = cp.Variable(n_points)
    objective = cp.Minimize(cp.norm(
        np.stack(
            [correlate(h_cvx, 1 - h_cvx, mode="full")[i] for i in range(n_points * 2 - 1)]
        ), ord=np.inf
    ) + 0.1 * cp.sum_squares(h_cvx - 0.5)
    )

    constraints = [
        0 <= h_cvx,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]

    prob = cp.Problem(objective, constraints)

    start_time = time.time()
    best_C5 = np.inf
    best_h = h.copy()
    n_opt_calls = 0

    while time.time() - start_time < 0.6 * budget_s and n_opt_calls < 8:
        n_opt_calls += 1

        h_seed = best_h.copy()
        h_seed += 0.1 * (np.random.rand(n_points) - 0.5)
        h_seed = rescale_h(h_seed)

        h_cvx.value = h_seed
        result = prob.solve(max_iter=100, verbose=False, abstol=1e-3, reltol=1e-3, primals_solvers=["ECOS"])

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.clip(h_cvx.value, 0, 1)
            h_sol = rescale_h(h_sol)
            C5, h_sol = evaluate_C5(h_sol, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_sol

        else:
            break  # solver error, aborting

    # Fallback to local gradient search
    h = best_h.copy()
    learning_rate = 0.01
    iter_count = 0
    restart_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        h = rescale_h(h)
        C5_current, _ = evaluate_C5(h, dx)
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()

        if iter_count % 50 == 0:
            learning_rate *= 0.5
            restart_count += 1
            if restart_count % 3 == 0:
                h = best_h.copy()
                h += 0.1 * (np.random.rand(n_points) - 0.5)
                h = rescale_h(h)

        # Gradient step
        cor = correlate(h, 1 - h, mode="full")
        max_id = np.argmax(cor)
        offset = max_id - len(h) + 1
        grad = 2 * h - 1 - 2 * np.roll(h, -offset) + 1
        grad = np.where(np.abs(grad) < 1e-6, 0, grad)
        h = h - learning_rate * grad
        h = rescale_h(h)

    return best_h, best_C5, n_points
