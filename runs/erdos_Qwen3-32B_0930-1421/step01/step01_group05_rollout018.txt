import numpy as np
import time

from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 45 if 'initial_h_values' in globals() else None
    h0 = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None

    dx = 2.0 / (n_points or 45)

    def normalize_h(h_in):
        h_clipped = np.clip(h_in, 0, 1)
        total = h_clipped.sum()
        return h_clipped * (n_points / 2) / total if total > 0 else np.zeros(n_points)

    def compute_c5(h):
        corr = correlate(h, 1 - np.roll(h, shift=np.arange(-n_points + 1, n_points + 1)), mode='full')
        return np.max(corr) * dx

    if h0 is not None:
        h = h0.copy()
    else:
        h = np.zeros(n_points)
        h[np.random.permutation(n_points)[:n_points//2]] = 1.0

    best_h = h.copy()
    best_c5 = float('inf')

    t_outer = budget_s * 0.8
    t_inner = budget_s * 0.2
    outer_steps = 100
    inner_steps = 300
    stepl = 10.0
    decay = 0.90
    eps = 1e-5

    for outer in range(outer_steps):
        if time.time() - start_time > budget_s:
            break

        # Simulated annealing step
        temp = stepl * (decay ** outer)
        for _ in range(inner_steps):
            idx = np.random.randint(n_points)
            stored = h[idx]
            h[idx] = np.clip(h[idx] + np.random.uniform(-0.5, 0.5), 0, 1)
            h = normalize_h(h)

            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            elif np.log(np.random.rand()) < -abs(new_c5 - best_c5) / temp:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                h[idx] = stored
                h = normalize_h(h)

        # Local optimization step
        grad = np.zeros(n_points)
        h_save = h.copy()
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            cp = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            cn = compute_c5(h_n)

            grad[i] = (cp - cn) / (2 * eps)
        h -= 0.1 * grad
        h = normalize_h(h)

        h = h + (best_h - h) * 0.1
        h = normalize_h(h)

    return best_h.tolist(), best_c5, n_points
