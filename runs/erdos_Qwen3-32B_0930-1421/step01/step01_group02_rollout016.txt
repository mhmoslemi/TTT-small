```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h0 = np.copy(initial_h_values)
    else:
        h0 = np.full(n_points, 0.5 / (dx * n_points))
        h0 = ensure_feasible(h0)

    h = cp.Variable(n_points, nonneg=True)
    obj_vars = []

    for k in range(-n_points + 1, n_points):
        if k < 0:
            a = h[-k:]
            b = 1 - h[:k * -1]
        elif k == 0:
            a = h
            b = 1 - h
        else:
            a = h[:-k]
            b = 1 - h[k:]
        obj_var = cp.sum(a * b) * dx
        obj_vars.append(obj_var)

    obj = cp.Minimize(cp.max(obj_vars))
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0,
    ]
    prob = cp.Problem(obj, constraints)

    # Solve
    guess = np.copy(h0)
    h.value = guess
    prob.solve(solver=cp.OSQP, warm_start=True)

    best_h = h.value
    best_C5 = prob.value
    return best_h, best_C5, n_points
```
