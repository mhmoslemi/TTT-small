import numpy as np
import time
import scipy.signal
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2  # normalize for dx = 2 / n_points
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / (h_sum + 1e-12) * len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

        h = initial_h_values.copy()
        h = rescale_h(h)

        idx = np.round(np.linspace(0, n_points - 1, 6)).astype(int)
        h[idx] = 1.0
        h = rescale_h(h)

    best_h = h.copy()
    best_C5, _ = evaluate_C5(best_h, dx)
    start_time = time.time()
    iter_noimprove = 0
    amp = 0.1

    while time.time() - start_time < budget_s:
        h = rescale_h(h)
        C5, _ = evaluate_C5(h, dx)

        # Accept if improvement
        if C5 < best_C5:
            best_h = h.copy()
            best_C5 = C5
            iter_noimprove = 0
        else:
            iter_noimprove += 1

        # Occasionally shake the solution
        if iter_noimprove > 30 or np.isnan(C5):
            # Flip a random segment and add Gaussian perturbation
            seg_len = int(0.1 * n_points)
            seg_start = np.random.randint(0, n_points - seg_len)
            h[seg_start:seg_start + seg_len] = 1 - h[seg_start:seg_start + seg_len]
            h += amp * np.random.randn(n_points)
            h = rescale_h(h)
            iter_noimprove = 0

        # Decrease amplitude over time
        if iter_noimprove > 100:
            amp *= 1.05
        else:
            amp *= 0.998

        # Clamp to boundary
        h[~np.isfinite(h)] = 0.0
        h = np.clip(h, 0, 1)

    return best_h, best_C5, n_points
