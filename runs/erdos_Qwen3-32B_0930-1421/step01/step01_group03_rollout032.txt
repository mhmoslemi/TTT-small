import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100
    n_points = 47
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, copy=True)
    else:
        # Construct a sparse, evenly spaced step function with unit sum
        h_best = np.zeros(n_points)
        for i in range(n_points // 2):
            h_best[i * 2] = 1.0 / (n_points // 2)
        h_best *= n_points / 2 / h_best.sum()

    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (n_points / 2 / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    tabu_set = set()
    best_c5 = objective(h_best)
    delta = 0.15
    iter_no_improve = 0
    max_iter_no_improve = 50
    cooling_factor = 0.8
    iter_limit = 1000
    iter = 0

    def perturb_h(h_vec, delta_perturb=0.1):
        h = h_vec.copy()
        idx_to_perturb = np.random.choice(n_points)
        direction = np.random.choice([-1, 1])
        h[idx_to_perturb] += direction * delta_perturb
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (n_points / 2 / h_sum)
        return h

    while time.time() - start_time < budget_s and iter < iter_limit:
        h_new = perturb_h(h_best, delta)
        h_tuple = tuple(h_new.round(8))
        if h_tuple in tabu_set:
            continue
        tabu_set.add(h_tuple)
        if len(tabu_set) > 10:
            tabu_set.pop()

        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()
            tabu_set.clear()  # escape local minimum
            iter_no_improve = 0
            delta = max(0.01, delta * cooling_factor)
        else:
            iter_no_improve += 1

        if iter_no_improve >= max_iter_no_improve:
            delta = min(0.3, delta / cooling_factor)
            iter_no_improve = 0

        iter += 1

    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(
        objective,
        h_best,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100}
    )
    h_best = result.x.reshape(n_points)
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points / 2 / h_best.sum())
    best_c5 = objective(h_best)

    return h_best, best_c5, n_points
