import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def feasible_project(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale sum constraint: change as small scalar as possible
    target_sum = n_points / 2
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h *= (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    h1 = 1 - h
    corr = correlate(h, h1, mode='full')
    idx_max = np.argmax(corr)
    lag = idx_max - (n_points - 1)

    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + lag
        if 0 <= j < n_points:
            grad[i] += (1 - h[j])
        j = i - lag
        if 0 <= j < n_points:
            grad[i] -= h[j]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum()

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)

    t0 = time.time()
    alpha = 1e-3
    max_iter = int(2000)
    iter_no_improvement_limit = 500
    iter_no_improvement = 0
    min_alpha = 1e-6
    backtrack_threshold = 0.5
    perturb_interval = 300

    for i in range(max_iter):
        if time.time() - t0 >= budget_s:
            break

        grad = compute_gradient(h, n_points)
        norm = np.linalg.norm(grad)
        if norm < 1e-8:
            iter_no_improvement += 1
        else:
            grad = grad / norm
        h_new = h - alpha * grad
        h_new = feasible_project(h_new, n_points)

        c5 = evaluate_overlap(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            iter_no_improvement = 0
            alpha = min(alpha * 1.1, 1e-2)
        else:
            alpha = max(alpha * backtrack_threshold, min_alpha)
            iter_no_improvement += 1

        if iter_no_improvement >= iter_no_improvement_limit:
            alpha = min(alpha * 2, 1e-2)
            noise = np.random.uniform(-0.01, 0.01, n_points)
            h_new = feasible_project(h + noise, n_points)
            iter_no_improvement = 0

        h = h_new.copy()

        if i % perturb_interval == 0 and i > 0:
            perturb = np.random.randn(n_points) * 0.02
            h = feasible_project(h + perturb, n_points)

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
