import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Normalize to sum to n_points / 2
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        scale = (n_points / 2) / total
        h = h * scale
        # Second pass clip
        h = np.clip(h, 0.0, 1.0)
        # Zero-out any tiny negatives due to scaling
        h = np.where(h < 1e-8, 0.0, h)
        h = np.where(h > 1.0 - 1e-8, 1.0, h)
    return h

def local_perturbation(h, n_perturb=5, sigma=0.2):
    h_new = h.copy()
    indices = np.random.choice(len(h), size=n_perturb, replace=False)
    perturbation = np.random.normal(0.0, sigma, size=n_perturb)
    h_new[indices] += perturbation
    h_new = project_h(h_new, len(h))
    return h_new

    while time.time() - t0 < budget_s:

                T *= 0.9

            no_improvement = 0
        else:
