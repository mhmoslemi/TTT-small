import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def surr_objective(h_vec, n_points, beta_sm=100, smooth_beta=60, alpha=10):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    total = np.sum(h)
    h = np.clip(h, 1e-6, 1 - 1e-6)
    norm_penalty = -np.log(np.abs(total - n_points / 2))
    conf_penalty = -np.mean(np.log(h) + np.log(1 - h))
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 1:
            continue
        full_corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(full_corr)
    for k in range(1, n_points):
        if n_points - k <= 1:
            continue
        full_corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(full_corr)
    corrs = np.array(corrs) * dx
    raw_max = np.max(corrs)
    corr_smooth = smooth_max(corrs, beta=smooth_beta)
    return corr_smooth + alpha * (norm_penalty + conf_penalty)

def refine_objective(h_vec, n_points):
    return compute_c5(h_vec, n_points)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

    i = np.arange(n_points)
    return np.clip(0.5 + 0.5 * np.cos(np.pi * i / n_points), 0.0, 1.0)
