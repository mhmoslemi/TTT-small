import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute C5 bound and normalized h."""
    h = h.clip(0.0, 1.0)
    h_sum = h.sum()
    h_normalized = h / h_sum * len(h) / 2  # normalize
    cor = correlate(h_normalized, 1 - h_normalized, mode="full")
    C5 = np.max(cor) * dx
    return C5, h_normalized


def rescale_h(h):
    """Rescale h to satisfy the integral constraint."""
    h = h.clip(0.0, 1.0)
    h_sum = h.sum()
    if h_sum < 1e-10:
        h[:] = 0.0
        h[0] = len(h) / 2
    else:
        h = h / h_sum * len(h) / 2
    return h
