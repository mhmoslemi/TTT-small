import numpy as np
import time
import scipy.signal
import scipy.fft
import cvxpy as cp

def run(seed=42, budget_s=1100):
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def project_sum(h, target_sum, dx):
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0.0, 1.0)
        correction = target_sum - h.sum()
        if abs(correction) > 1e-8:
            indices = np.argsort(h)
            incr = abs(correction) / len(indices)
            if correction > 0:
                h[indices] = np.clip(h[indices] + incr, 0.0, 1.0)
            else:
                h[indices] = np.clip(h[indices] - incr, 0.0, 1.0)
        return h

    def compute_c5(h):
        h = project_sum(h, target_sum, dx)
        ones_minus_h = 1.0 - h
        corr = scipy.signal.correlate(h, ones_minus_h, mode='full')
        corr = corr[n_points : 2 * n_points - 1]
        c5 = np.max(corr * dx)
        return c5

    def fourier_basis(n, m):
        x = np.linspace(0, 2, n, endpoint=False)
        basis = np.empty((2 * m + 1, n))
        for i in range(-m, m + 1):
            if i == 0:
                basis[m] = 1.0 / np.sqrt(n)
            else:
                basis[m + i] = np.exp(1j * np.pi * i * x) / np.sqrt(n)
        return basis

    def residual(h, coeffs, basis):
        return h - np.dot(coeffs, basis)

    # Fourier coef optimization
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    basis = fourier_basis(n_points, 15)  # Reduced basis for faster convergence
    m = basis.shape[0]

    coeffs = np.conj(np.dot(basis, h))
    h_fft = h.copy()
    best_time = time.time()

    while time.time() - start_time < budget_s - 10:  # Leave 10s for final evaluation
        trial_coeffs = coeffs + 0.001 * np.random.randn(m)
        trial_h = np.real(np.dot(trial_coeffs, basis))
        trial_h = project_sum(trial_h, target_sum, dx)

        new_c5 = compute_c5(trial_h)
        if new_c5 < best_c5:
            h_fft = trial_h.copy()
            coeffs = trial_coeffs.copy()
            best_c5 = new_c5
            best_h = h_fft.copy()
            best_time = time.time()

        # Escape local minima
        if time.time() - best_time > 60:
            coeffs = np.conj(np.dot(basis, h_fft))
            trial_coeffs = coeffs + 0.05 * (np.random.rand(m) - 0.5)
            trial_h = np.real(np.dot(trial_coeffs, basis))
            trial_h = project_sum(trial_h, target_sum, dx)
            new_c5 = compute_c5(trial_h)
            if new_c5 < best_c5:
                h_fft = trial_h.copy()
                coeffs = trial_coeffs.copy()
                best_c5 = new_c5
                best_h = h_fft.copy()
                best_time = time.time()

        if best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
