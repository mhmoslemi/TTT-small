import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    h_in = np.copy(initial_h_values)
    n_points = len(h_in)

    # Normalize h_in to match the constraint ∫h = 1
    total_h = float(np.sum(h_in))
    scale_factor = (n_points / 2.0) / total_h
    h_start = np.minimum(h_in * scale_factor, 1.0)

    # Parameters for PSO
    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1
    max_iter = 200

    # Initialize particles and positions
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))

    # Initialization: 10% from initial_h_values, rest from random
    for i in range(num_particles):
        if i < num_particles * 0.1:
            particles[i] = h_start.copy()
        else:
            particles[i] = np.random.rand(n_points)
        # Project to constraint space
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] *= (n_points / 2.0) / particles[i].sum()

    p_best = np.copy(particles)
    p_best_score = np.array([evaluate_erdos_solution(h, n_points) for h in particles])
    g_best = particles[p_best_score.argmin()].copy()
    best_score = p_best_score.min()
    last_improvement_time = start_time

    def evaluate(h):
        return evaluate_erdos_solution(h, n_points)

    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 2:
            break

        t = iteration / max_iter
        w = w_start - t * (w_start - w_end)

        # Update velocities and positions
        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (p_best[i] - particles[i])
                + c2 * r2 * (g_best - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]

        # Project particles to feasible space
        for i in range(num_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2.0) / particles[i].sum()

        # Evaluate and update bests
        scores = np.array([evaluate(p) for p in particles])
        improved = scores < p_best_score
        p_best[improved] = particles[improved].copy()
        p_best_score[improved] = scores[improved]

        if (scores < best_score).any():
            best_score = scores.min()
            g_best = particles[scores.argmin()].copy()
            last_improvement_time = time.time()

        # Apply reset to stagnant particles
        stagnant = (p_best_score == best_score)
        reset_indices = np.random.choice(np.where(stagnant)[0], size=int(num_particles * reset_fraction), replace=False)
        for idx in reset_indices:
            particles[idx] = np.random.rand(n_points)
            particles[idx] = np.clip(particles[idx], 0, 1)
            particles[idx] *= (n_points / 2.0) / particles[idx].sum()

    return g_best, best_score, n_points
