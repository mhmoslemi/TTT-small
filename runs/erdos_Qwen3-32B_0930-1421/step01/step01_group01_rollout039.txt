import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum == 0:
        h[0] = len(h) / 2
    else:
        h = h / h_sum * len(h) / 2
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5
