import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    h_start = np.copy(initial_h_values)
    n_points = len(h_start)

    # Normalize to satisfy ∑h == n / 2
    h_start = np.clip(h_start, 0.0, 1.0)
    h_start = h_start * (n_points / 2.0) / h_start.sum()

    # PSO parameters
    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    reset_fraction = 0.1
    max_velocity_clamp = 0.2

    # Initialize particles and velocities
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    particles[0] = h_start
    for i in range(1, num_particles):
        perturbation = np.random.normal(0.05, 0.1, size=n_points)
        particles[i] = np.clip(h_start + perturbation, 0.0, 1.0)
        particles[i] = particles[i] * (n_points / 2.0) / particles[i].sum()

    # Initialize positions
    best_all = np.inf
    p_best = particles.copy()
    p_best_scores = np.full(num_particles, np.inf)
    g_best = np.copy(h_start)
    g_best_score = best_all

    # Evaluate initial scores
    for i in range(num_particles):
        p_best_scores[i] = evaluate_erdos_solution(particles[i], n_points)
        if p_best_scores[i] < g_best_score:
            g_best[:] = particles[i].copy()
            g_best_score = p_best_scores[i]

    best_c5 = g_best_score
    best_h = g_best.copy()
    last_improvement_time = start_time

    def project_h(x):
        x = np.clip(x, 0.0, 1.0)
        x = x * (n_points / 2.0) / x.sum()
        return x

    # PSO loop
    iteration = 0
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            w = w_start - (w_start - w_end) * (iteration / 200)

            velocities[i] = (w * velocities[i]
                            + c1 * r1 * (p_best[i] - particles[i])
                            + c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity_clamp, max_velocity_clamp)
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])

            score = evaluate_erdos_solution(particles[i], n_points)
            if score < best_c5:
                best_c5 = score
                best_h[:] = particles[i].copy()
                last_improvement_time = time.time()

            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()

            if score < g_best_score:
                g_best[:] = particles[i].copy()
                g_best_score = score

        if time.time() - last_improvement_time > 120:
            break

        # Fractional reset
        num_resets = int(num_particles * reset_fraction)
        for i in np.random.choice(num_particles, num_resets, replace=False):
            perturbation = np.random.normal(0.05, 0.1, size=n_points)
            particles[i] = np.clip(h_start + perturbation, 0.0, 1.0)
            particles[i] = project_h(particles[i])
            p_best_scores[i] = np.inf
            p_best[i] = np.copy(particles[i])

        iteration += 1

    return best_h, best_c5, n_points
