import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def repair_to_binary(h):
    # Project to binary and satisfy constraint
    h_bin = np.round(h)
    h_bin = np.clip(h_bin, 0, 1)
    n_points = len(h)
    total = np.sum(h_bin)
    if total < n_points / 2:
        # Need more 1s
        idx_0 = np.where(h_bin == 0)[0]
        need = int(n_points / 2) - np.sum(h_bin)
        h_bin[np.random.choice(idx_0, size=need, replace=False)] = 1
    elif total > n_points / 2:
        # Need fewer 1s
        idx_1 = np.where(h_bin == 1)[0]
        need = np.sum(h_bin) - int(n_points / 2)
        h_bin[np.random.choice(idx_1, size=need, replace=False)] = 0
    return h_bin

    best_h = None
    best_C5 = np.inf

    # Convex optimization setup
    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    def get_overlap(k):
        idx_i = [i for i in range(n_points - abs(k))]
        idx_j = [i + abs(k) for i in range(n_points - abs(k))]
        sign = 1 if k > 0 else -1
        terms = []
        for i, j in zip(idx_i, idx_j):
            terms.append(h[i] * (1 - h[j]))
        return cp.sum(terms)

    # Start with all immediate neighbors
    k_list = []
    for k in range(1, 3):
        k_list.append(k)
        k_list.append(-k)

    for cut_iter in range(1000):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        # Build objective using current cuts
        overlap_constr = []
        for k in k_list:
            overlap_k = get_overlap(k)
            overlap_constr.append(overlap_k <= 1e9)  # initially inactive

        lambda_var = cp.Variable(nonneg=True)
        objective = cp.Minimize(lambda_var)

        for k in k_list:
            overlap_k = get_overlap(k)
            overlap_constr.append(overlap_k <= lambda_var)

        problem = cp.Problem(
            objective,
            constraints + overlap_constr,
        )

        problem.solve(solver=cp.ECOS, max_iters=1000)

        if h.value is None:
            continue

        h_sol = h.value
        C5 = evaluate_C5(h_sol)

        if C5 < best_C5:
            best_h = h_sol
            best_C5 = C5

        if verbose:
            print(f"Iter {cut_iter}, C5 = {C5:.8f}")

        # Try adding new cutting planes
        for k in range(1, n_points):
            if k in k_list:
                continue
            overlap_k = get_overlap(k)
            model = cp.Problem(cp.Minimize(overlap_k), constraints)
            model.solve(solver=cp.ECOS, verbose=False)
            if model.value > 1e-6:
                k_list.append(k)
                k_list.append(-k)
                break

    # Repair solution to be binary
    if best_h is not None:
        best_h = repair_to_binary(best_h)
        best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
