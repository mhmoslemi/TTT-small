import numpy as np
import time

import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize h to satisfy constraints
    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    while h_sum < target_sum:
        idx = np.argmin(h)
        h[idx] += (target_sum - h_sum) / (n_points - h_sum)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
            corr_rev = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr_rev * dx)
        return max_corr

    best_c5 = compute_c5(h)
    if best_c5 < 0.380876:
        return (h, best_c5, n_points)

    grad_steps = 0
    # Gradient step size parameters
    step_size = 0.01
    step_decay = 0.999

    while time.time() - start_time < budget_s - 10:
        if time.time() - best_time > 300:
            # Perturbation to escape local optima
            perturbation_mask = np.random.randn(n_points) * 0.05
            perturbation_mask -= perturbation_mask.min()
            perturbation_mask /= perturbation_mask.max()
            perturbation_mask *= 0.2 * np.random.rand()  # random scale
            proposed_h = h + perturbation_mask
            proposed_h = np.clip(proposed_h, 0, 1)
            proposed_h_sum = proposed_h.sum()
            if abs(proposed_h_sum - target_sum) > 1e-7:
                proposed_h *= target_sum / proposed_h_sum
            new_c5 = compute_c5(proposed_h)
            if new_c5 < best_c5:
                h[:] = proposed_h
                best_c5 = new_c5
                best_h[:] = h
                best_time = time.time()

        # Compute gradient for the current max overlap and update
        # Identify the max overlap shift k
        max_corr = -np.inf
        kmax = 0
        for k in range(1, n_points):
            if k >= h.shape[0] - 1:
                continue
            corr = np.dot(h[:n_points - k], (1 - h[k:]))
            corr_rev = np.dot(h[k:], (1 - h[:n_points - k]))
            corr_total = max(corr, corr_rev)
            if corr_total > max_corr:
                max_corr = corr_total
                kmax = k
        if kmax == 0:
            break

        # Compute gradient for forward correlation
        grad = np.zeros(n_points)
        for i in range(n_points - kmax):
            grad[i] += (1 - h[i + kmax])  # derivative of h[i]*(1-h[i+k]) w.r.t. h[i]
            grad[i + kmax] -= h[i]  # derivative w.r.t. h[i+k]

        # Compute gradient for reverse correlation
        for i in range(n_points - kmax):
            grad[i] -= h[i + kmax]  # derivative of h[i+k]*(1-h[i]) w.r.t. h[i]
            grad[i + kmax] += (1 - h[i])  # derivative of h[i+k]*(1-h[i]) w.r.t. h[i+k]

        grad *= dx

        # Gradient step
        h -= grad * step_size
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-9:
            h *= target_sum / h_sum

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h
            best_time = time.time()
            step_size = 0.02
        elif time.time() - best_time > 300:
            # Reset when no improvement
            h[:] = best_h
            step_size *= step_decay

    return (best_h, best_c5, n_points)
