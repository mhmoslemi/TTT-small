```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_full = np.tile(h, 4)
    corrs = []
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:n_points])
            corrs.append(corr)
    return np.max(np.array(corrs) * dx)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if np.abs(total - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    n_points = len(init_h)
    h_initial = enforce_normalization(init_h.copy(), n_points)
    time_start = time.time()

    # Initialize variables for LP-based column generation
    h = cp.Variable(n_points)
    C = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    # Initial constraints
    for k in [1, 2, 4, 8, 16]:
        if k >= n_points:
            continue
        idxs = [i for i in range(n_points - k)]
        for i in idxs:
            constraints.append(h[i] * (1 - h[i + k]) <= C)

    # Initial objective
    prob = cp.Problem(cp.Minimize(C), constraints)

    h_best = h_initial.copy()
    C_best = compute_c5(h_initial, n_points)
    time_used = time.time() - time_start

    # Column generation loop
    while time_used < budget_s - 20 and time_used < budget_s * 0.9:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status != cp.OPTIMAL:
            break
        h_val = h.value.flatten()
        h_val = enforce_normalization(h_val, n_points)
        C_val = C.value
        if compute_c5(h_val, n_points) < C_best:
            C_best = compute_c5(h_val, n_points)
            h_best = h_val.copy()
        # Identify worst constraint
        max_overlap = 0.0
        worst_k = 0
        h_agg = np.array([h_val]).squeeze()
        for k in range(1, n_points):
            if n_points - k == 0:
                continue
            corr = np.dot(h_agg[:n_points - k], 1 - h_agg[k:n_points])
            if corr > max_overlap:
                max_overlap = corr
                worst_k = k
        if max_overlap > C_val + 1e-5:
            idxs = [i for i in range(n_points - worst_k)]
            for i in idxs:
                constraints.append(h[i] * (1 - h[i + worst_k]) <= C)
        time_used = time.time() - time_start

    # Final local refinement with optimizing over h
    h_ref = opt.minimize(
        lambda x: compute_c5(x, n_points),
        h_best,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 250, 'ftol': 1e-7}
    ).x
    h_ref = enforce_normalization(h_ref, n_points)
    final_c5 = compute_c5(h_ref, n_points)
    if final_c5 < C_best:
        h_best = h_ref
        C_best = final_c5
    
    return h_best, C_best, n_points
```
