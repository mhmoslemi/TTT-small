import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h):
    # Clip to domain [0, 1]
    h_clipped = np.clip(h, 0.0, 1.0)
    # Enforce sum constraint
    sum_h = np.sum(h_clipped)
    if abs(sum_h - len(h)/2) > 1e-6:
        h_clipped *= (len(h)/2) / sum_h
    return h_clipped

def compute_gradient(h, corr, dx, k_shift):
    n = len(h)
    grad = np.zeros(n)
    lag = k_shift - (n - 1)  # Compute effective lag
    for i in range(n):
        if i + lag < n:
            grad[i] += (1 - h[i + lag]) * dx
        if i - lag >= 0:
            grad[i] -= h[i - lag] * dx
    return grad

        c5_current = evaluate_overlap(current_h)
        if c5_current < best_c5:

        # Perturbation to escape local minima
        if t % iters_per_period == 0 and np.random.rand() < 0.1:
            noise = np.random.randn(n_points) * 0.01
            current_h = project_to_feasible(current_h + noise)
