    dx = 2.0 / n_points

    if abs(h_sum - target_sum) > 1e-9:

        h[idx] += 1e-5
        h_sum = h.sum()
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = float("inf")

    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0.0, 1.0)
        h_sum = h_vals.sum()
        if abs(h_sum - target_sum) > 1e-9:
            h_vals *= target_sum / h_sum
        h_vals = np.clip(h_vals, 0.0, 1.0)
        ones_minus_h = 1.0 - h_vals
        corr_vals = np.zeros(n_points - 1)
        for k in range(1, n_points):
            if k >= len(h_vals):
                break
            if n_points - k >= 1:
                corr_vals[k - 1] = np.dot(h_vals[:n_points - k], ones_minus_h[k:])
        for k in range(1, n_points):
            if k >= len(h_vals):
                break
            if n_points - k >= 1:
                corr_vals[k - 1] = max(corr_vals[k - 1],
                                       np.dot(h_vals[k:], ones_minus_h[:n_points - k]))
        return corr_vals * dx

    best_c5 = np.max(compute_c5(h))

    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)

    def fourier_perturb(h_vals, n_freq=20, scale=0.02):
        H = np.fft.rfft(h_vals, n=2 * n_points)
        freq_indices = np.argsort(np.abs(H))[::-1][:n_freq]
        mask = np.isin(np.fft.rfftfreq(2 * n_points), freq_indices)
        mask[0] = False  # exclude DC
        perturbation = np.fft.fftshift(np.fft.irfft(H * mask))
        perturbation += np.random.normal(scale=scale, size=perturbation.shape)
        return (h_vals + perturbation) / 2.0

    # Optimization loop with Fourier perturbations
    while time.time() - start_time < budget_s - 5:

        proposed_h = fourier_perturb(h, n_freq=20, scale=0.02)

        if np.random.rand() < 0.2:  # 20% chance to do local search instead

            if new_c5 < best_c5:
                h[:] = proposed_h
                best_h[:] = proposed_h
                best_c5 = new_c5

        if verbose and time.time() - start_time > 10 * math.ceil(start_time):
            print(
                f"Time: {time.time()-start_time:.2f}s, C₅: {best_c5:.6f}, Target: 0.380876"
            )

    return (best_h, best_c5, n_points)
