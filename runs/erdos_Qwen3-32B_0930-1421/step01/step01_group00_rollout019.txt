import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = h_init.copy()
    h_sum = h.sum()
    h[:] = np.clip(h, 0, 1)
    if abs(h_sum - target_sum) > 1e-7:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    while h_sum < target_sum:
        idx = np.argmin(h)
        h[idx] += 0.001
        h_sum = h.sum()
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()
    patience = 240
    perturb_size = 6
    eps = 0.01

    def compute_c5(h_array):
        h_array = np.clip(h_array, 0, 1).copy()
        h_sum = h_array.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h_array *= (target_sum / h_sum)
        h_array = np.clip(h_array, 0, 1)
        ones_h = 1.0 - h_array
        max_corr = -np.inf
        for k in range(1, n_points - 1):
            if k > n_points - k:
                continue
            corr = np.dot(h_array[:n_points - k], ones_h[k:])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    # Dynamic convex optimization phase - warm start
    while time.time() - start_time < budget_s * 0.1:
        h_cvx = cp.Variable(n_points)
        overlaps = []
        for k in range(1, n_points - 1):
            if k > n_points - k:
                break
            overlaps.append(cp.sum([h_cvx[i]*(1 - h_cvx[i + k]) for i in range(n_points - k)]))
        overlaps.append(cp.sum([h_cvx[i + k]*(1 - h_cvx[i]) for i in range(n_points - k)]) for k in range(1, n_points - 1))

        objective = cp.Minimize(cp.Max([o * dx for o in overlaps]))
        constraints = [cp.sum(h_cvx) == target_sum, h_cvx >= 0, h_cvx <= 1]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCIP, verbose=False)
        h_candidate = h_cvx.value
        if h_candidate is not None:
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                h[:] = h_candidate
                best_c5 = current_c5
                best_h[:] = h[:]
                best_time = time.time()
                patience = 240
        break  # Comment to allow multiple warm start rounds

    # Local search optimization
    while time.time() - start_time < budget_s - 10:
        indices = np.random.choice(n_points, min(perturb_size, n_points), replace=False)
        proposed_h = h.copy()
        noise = np.random.normal(0, eps, size=len(indices))
        proposed_h[indices] += noise
        proposed_h = np.clip(proposed_h, 0, 1)
        proposed_h_sum = proposed_h.sum()
        if abs(proposed_h_sum - target_sum) > 1e-7:
            proposed_h *= (target_sum / proposed_h_sum)
        proposed_h = np.clip(proposed_h, 0, 1)

        new_c5 = compute_c5(proposed_h)
        if new_c5 < best_c5:
            h[:] = proposed_h
            best_c5 = new_c5
            best_h[:] = h[:]
            best_time = time.time()
            patience = 240
        elif np.random.rand() < 0.1:  # small chance to accept worse to escape local optima
            h[:] = proposed_h

        if time.time() - best_time > patience:
            # If no improvement, try large perturbation
            h = best_h.copy()
            indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
            h[indices] += np.random.normal(0, 0.1, size=len(indices))
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if abs(h_sum - target_sum) > 1e-7:
                h *= (target_sum / h_sum)
            best_time = time.time()
            patience = 240

        if verbose and (time.time() - start_time) % 10 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
