import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def gradient_direction_C5(h, dx):
    """Compute the gradient direction for a single maximum overlap."""
    h = h.astype(np.float64)
    cor = correlate(h, 1 - h, mode="full")
    max_id = np.argmax(cor)
    grad = np.zeros_like(h)
    max_offset = max_id - len(h) + 1
    for i in range(len(h)):
        if 0 <= (i + max_offset) < len(h):
            grad[i] = 2 * h[i] - 1 - 2 * h[i + max_offset] + 1
        else:
            grad[i] = 2 * h[i] - 1
    return grad * dx
