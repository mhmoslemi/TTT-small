import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def localize_max_shift(h):
    """Identify shift k with maximum overlap and localized regions of influence"""
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    k_max = np.argmax(corr)

    # Map shift index to physical offset
    physical_k = k_max - n + 1  # Shift in [-(n-1), (n-2)]
    affected_start = max(0, physical_k)
    affected_end = min(n, n + physical_k)

    return k_max, affected_start, affected_end

def flip_two_bits(h, affected_start, affected_end):
    """Flip 1->0 in original h and 0->1 in affected region"""
    n = len(h)
    indices_1 = np.where((h == 1.0) & (np.arange(n) < affected_start))[0]
    if len(indices_1) == 0:
        indices_1 = np.where(h == 1.0)[0]
    indices_0 = np.where(h[affected_start:affected_end] == 0.0)[0]
    indices_0 = indices_0 + affected_start  # Adjust for offset

    if len(indices_0) == 0 or len(indices_1) == 0:
        return h.copy()  # No valid flip, return original

    i = np.random.choice(indices_1)
    j = np.random.choice(indices_0)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0

    return h_new

def enforce_sum(h):
    """Systematically restore the sum constraint if violated"""
    n = len(h)
    target = n // 2
    diff = int(np.sum(h) - target)

    while np.sum(h) != target:
        if np.sum(h) < target:
            idx = np.random.choice(np.where(h == 0.0)[0])
            h[idx] = 1.0
        elif diff > 0:
            idx = np.random.choice(np.where(h == 1.0)[0])
            h[idx] = 0.0
            diff -= 1
        else:
            idx = np.random.choice(np.where(h == 0.0)[0])
            h[idx] = 1.0
            diff += 1

    return h

    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5
    iter_count = 0
    max_no_improvement = 1000
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        iter_count += 1

        k_max, affected_start, affected_end = localize_max_shift(current_h)
        h_new = flip_two_bits(current_h, affected_start, affected_end)
        h_new = enforce_sum(h_new)

        c5_new = evaluate_overlap(h_new)
        if c5_new < best_c5:

            current_h = h_new.copy()
