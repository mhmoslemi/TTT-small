import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        h_rounded = np.round(h, decimals=10)
        c = correlate(h_rounded, 1 - h_rounded, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        return clipped * (n_points / 2) / sum_h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    learning_rate = 0.1

    while (time.time() - start_time) < budget_s:
        k_values = correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(np.abs(k_values))
        h_grad = np.zeros_like(h)

        if k_idx < n_points:
            # Forward overlap
            for i in range(n_points - k_idx):
                h_grad[i] += 1.0
                h_grad[i + k_idx] -= 1.0
        else:
            # Backward overlap
            for i in range(k_idx - n_points + 1, n_points):
                h_grad[i] += 1.0
                h_grad[i - k_idx + n_points - 1] -= 1.0

        h -= learning_rate * h_grad
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if (time.time() - start_time) > budget_s - 1.0:
            break

        learning_rate *= 0.999  # Decay learning rate

    return best_h.tolist(), best_c5, n_points
