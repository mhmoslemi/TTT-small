import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5_fast(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h *= target_sum / h_sum
        ones_minus_h = 1.0 - h
        corr_all = scipy.signal.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr_all) * dx
        return max_corr

    # Local-refinement optimizer
    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-7:
        h *= target_sum / h_sum

    best_h = h.copy()
    best_c5 = compute_c5_fast(best_h)
    best_time = time.time()

    n_fft = 256
    h_fft = np.fft.fft(h, n_fft)
    effective_indices = np.argsort(np.abs(h_fft))[-20:]
    sweep_coefficient = 0.0005

    while time.time() - start_time < budget_s - 5:
        # Perturb top Fourier modes
        perturb = np.zeros_like(h_fft)
        inds = np.random.choice(effective_indices, min(5, len(effective_indices)), replace=False)
        perturb[inds] = np.random.uniform(-sweep_coefficient, sweep_coefficient, size=len(inds))
        h_fft += perturb

        # Inverse transform, make real, clip and scale
        h_fft_clipped = h_fft.copy()
        h_fft_clipped[0] = np.fft.fft(np.zeros_like(h))[0]  # Keep DC clean

        h_prop = np.fft.ifft(h_fft_clipped).real
        h_prop = np.clip(h_prop, 0, 1)
        h_prop_sum = h_prop.sum()
        if abs(h_prop_sum - target_sum) > 1e-6:
            h_prop = np.clip(h_prop * (target_sum / h_prop_sum), 0, 1)

        # Compute C5
        c5_proposal = compute_c5_fast(h_prop)

        if c5_proposal < best_c5:
            best_c5 = c5_proposal
            best_h = h_prop.copy()
            best_time = time.time()
            if verbose:
                print(f"New best C5: {best_c5:.9f}")
            if best_c5 < 0.380876:
                break
        else:
            # Revert energy back when stuck
            h_fft = np.fft.fft(best_h, n_fft)
            effective_indices = np.argsort(np.abs(h_fft))[-20:]

        # Aggressive restart to escape local minima
        if time.time() - best_time > 60:
            h_prop = np.random.uniform(0, 1, n_points)
            h_prop = np.clip(h_prop, 0, 1)
            h_prop_sum = h_prop.sum()
            h_prop *= (target_sum / h_prop_sum)
            best_c5 = compute_c5_fast(h_prop)
            best_h = h_prop.copy()
            best_time = time.time()
            h_fft = np.fft.fft(best_h, n_fft)
            effective_indices = np.argsort(np.abs(h_fft))[-20:]

        if verbose and (time.time() - start_time) % 60 < 0.1:
            print(f"Time: {time.time() - start_time:.2f} s, Best C5: {best_c5:.9f}")

    return (best_h, best_c5, n_points)
