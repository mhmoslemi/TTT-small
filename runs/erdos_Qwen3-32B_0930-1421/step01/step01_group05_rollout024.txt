```python
import numpy as np
import time
from scipy.signal import correlate

def normalize_h(h, n_points):
    clipped = np.clip(h, 0, 1)
    total = np.sum(clipped)
    factor = (n_points / 2) / total
    return np.clip(clipped * factor, 0, 1)

def compute_c5(h, n_points, dx):
    return np.max(correlate(h, 1 - h, mode='full') * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Extract initial_h_values if available
    global initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_initial)
    else:
        n_points = 45
        h_initial = np.random.rand(n_points)
        h_initial = normalize_h(h_initial, n_points)

    # Parameters
    population_size = 50
    beta0 = 1.0
    alpha = 0.2
    gamma_rate = 1.0
    
    # Initialize population
    population = np.tile(h_initial, (population_size, 1))
    for i in range(1, population_size):
        noise = np.random.normal(0, 0.02, n_points)
        population[i, :] = normalize_h(population[i, :] + noise, n_points)
    
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n_points, dx)

    max_iter = 1000
    current_iter = 0

    while time.time() - start_time < budget_s and current_iter < max_iter:
        # Evaluate C5 for all fireflies
        c5_scores = np.array([compute_c5(f, n_points, dx) for f in population])
        sorted_indices = np.argsort(c5_scores)
        current_iter += 1

        for i in range(population_size):
            for j in range(population_size):
                if c5_scores[j] < c5_scores[i]:
                    r = np.sum((population[i] - population[j]) ** 2)
                    beta = beta0 * np.exp(-gamma_rate * r)
                    population[i] += beta * (population[j] - population[i]) + alpha * (np.random.rand(n_points) - 0.5)
                    population[i] = normalize_h(population[i], n_points)

                    # Update best
                    current_c5 = compute_c5(population[i], n_points, dx)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = population[i].copy()

            if time.time() - start_time >= budget_s:
                break

        gamma_rate += 1.0 / (0.1 * max_iter)

    return best_h.tolist(), best_c5, n_points
```
