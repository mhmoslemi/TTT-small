import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True).clip(0, 1)
    else:
        n_points = 89  # graphic prime for anti-aliasing cross-correlation
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 if i < n_points // 2 else 0.0
        h_init = h_init * (n_points / 2 / h_init.sum())  # normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())  # maintain constraint
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=500):
        h_arr = initial_h.copy()
        bounds = [(1e-10, 1.0) for _ in range(n_points)]

        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 1e-10, 1.0)
            h = h * (constraint_sum / h.sum())
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(obj_func, h_arr.flatten(), method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    def perturb_h(h, noise_factor=0.025):
        h_flat = h.copy().flatten()
        noise = np.random.randn(n_points) * noise_factor
        h_flat += noise
        h_flat = np.clip(h_flat, 0.01, 0.99)
        h_flat = h_flat * (constraint_sum / h_flat.sum())
        return h_flat

    def repair_h(h):
        h = h.reshape(n_points)
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # Gradient refinement
    h = optimize_h(h_init)
    c5 = objective(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Tabu search + cooling schedule
    tabu = []
    tabu_size = 10

    current_delta = 0.025
    cooling_factor = 0.9
    no_improve_count = 0
    max_no_improve = 75

    while time.time() - start_time < budget_s:
        # Copy only to be safe
        perturbed = perturb_h(best_h, noise_factor=current_delta)
        perturbed = optimize_h(perturbed)
        perturbed_flat = perturbed.flatten()
        # Feasibility and tabu check
        is_tabu = any(np.allclose(perturbed_flat, t, rtol=1e-4) for t in tabu)
        if is_tabu:
            continue
        c5_new = objective(perturbed)
        if c5_new < best_c5:
            best_h = perturbed
            best_c5 = c5_new
            no_improve_count = 0
            # Reset upon improvement
            current_delta = min(0.08, current_delta * 1.05)
            tabu = [perturbed_flat]
        else:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                current_delta *= cooling_factor
                no_improve_count = 0
            tabu = [perturbed_flat]

    best_h = repair_h(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
