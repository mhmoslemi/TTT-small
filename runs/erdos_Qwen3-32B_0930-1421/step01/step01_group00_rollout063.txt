import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    # *--------------------------------------------------------------------------*
    #                     Time and basic setup
    # *--------------------------------------------------------------------------*
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    np.random.seed(seed)
    
    # *--------------------------------------------------------------------------*
    #                           Initial setup
    # *--------------------------------------------------------------------------*
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 -> sum(h) = n_points / 2

    # *--------------------------------------------------------------------------*
    #                           Constraint helper
    # *--------------------------------------------------------------------------*
    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if abs(sum_h - target_sum) > 1e-7:
            h *= target_sum / sum_h
            h = np.clip(h, 0, 1)
        return h

    h = project(h, target_sum)
    best_h = h.copy()
    best_c5 = np.inf
    best_time = start_time

    # *--------------------------------------------------------------------------*
    #                       Overlap evaluation
    # *--------------------------------------------------------------------------*
    def compute_c5_fast(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1).copy()
        optimized_h = project(optimized_h, target_sum)
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf

        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            # Left-to-right shift
            corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            # Right-to-left shift
            corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr1 * dx, corr2 * dx)

        return max_corr

    best_c5 = compute_c5_fast(h)
    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)

    # *--------------------------------------------------------------------------*
    #                       Gradient descent with exploration
    # *--------------------------------------------------------------------------*
    alpha = 0.01  # Initial step size
    step_reduction_period = 60  # seconds
    learning_rate_decay_factor = 0.9
    perturb_step = 100
    last_improvement_time = start_time

    while time.time() - start_time < budget_s - 20:
        # Compute current C5 and find worst shift
        h_current = project(h, target_sum)
        ones_minus_h = 1.0 - h_current
        overlap = -np.inf
        worst_k = 0

        for k in range(1, n_points):
            if k >= h_current.shape[0] - 1:
                continue
            corr1 = np.dot(h_current[:n_points - k], (1 - h_current[k:]))
            corr2 = np.dot(h_current[k:], (1 - h_current[:n_points - k]))
            curr_overlap = max(corr1, corr2) * dx
            if curr_overlap > overlap:
                overlap = curr_overlap
                worst_k = k

        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        k = worst_k
        for i in range(n_points - k):
            grad[i] -= (1 - h_current[i + k])
            grad[i + k] += h_current[i]

        grad *= dx  # Apply dx due to integral

        # Gradient descent step
        h -= alpha * grad
        h = project(h, target_sum)
        new_c5 = compute_c5_fast(h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement_time = time.time()

        # Reduce learning rate periodically
        if time.time() - start_time > 20 and (time.time() - start_time) % step_reduction_period < 1e-3:
            alpha *= learning_rate_decay_factor

        # Random perturbation to escape local optima
        if time.time() - start_time > 20 and (time.time() - start_time) % perturb_step < 1e-3:
            noise = np.random.uniform(-0.02, 0.02, size=n_points)
            h += noise
            h = project(h, target_sum)

        # Early stopping
        if time.time() - last_improvement_time > 120:
            h = best_h.copy()
            break

        # Optional verbose reports
        if verbose and (time.time() - start_time) % 10 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.8f}")

    return (best_h, best_c5, n_points)
