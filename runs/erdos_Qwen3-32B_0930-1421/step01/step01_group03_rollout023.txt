import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_n_points = 100

    if 'initial_h_values' in globals():
        h_curr = np.array(initial_h_values, dtype=np.float64)
        n_points = h_curr.size
    else:
        n_points = 47
        h_curr = np.zeros(n_points)
        for i in range(n_points // 2):
            h_curr[2 * i] = 2.0 / n_points
        h_curr *= n_points / 2 / h_curr.sum()

    h_curr = np.clip(h_curr, 0, 1)
    h_curr *= (n_points / 2) / h_curr.sum()
    dx = 2.0 / n_points

    def evaluate(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    best_score = evaluate(h_curr)
    best_h = h_curr.copy()
    t0 = time.time()

    def convex_optimize_with_fixed_support(h0):
        h = cp.Variable(h0.shape)
        obj = cp.Minimize(cp.max(cp.conv(cp.multiply(h, h), cp.Constant(np.ones(n_points)) * dx - cp.conv(h, h, "full"), "full")))
        constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=500)
            if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return None
            h_out = h.value.flatten()
            return h_out
        except Exception:
            return None

    def randomized_support(h, noise_scale=0.1):
        h = h.copy()
        idxs = np.random.permutation(n_points)
        perturb = np.random.normal(0, noise_scale, size=n_points)
        h[idxs[:n_points // 2]] += perturb[idxs[:n_points // 2]]
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()
        return h

    def similar_support(h, smooth_amount=0.1):
        h = h.copy()
        h = h + np.random.normal(0, 0.02, size=h.shape)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    # Initial convex solve
    if time.time() - start_time < budget_s * 0.9:
        h_cvx = convex_optimize_with_fixed_support(h_curr)
        if h_cvx is not None:
            curr = evaluate(h_cvx)
            if curr < best_score:
                best_h, best_score = h_cvx, curr

    ntries = 0
    while time.time() - start_time < budget_s * 0.99:
        ntries += 1
        new_h = randomized_support(best_h) if ntries % 2 == 0 else similar_support(best_h)
        result = convex_optimize_with_fixed_support(new_h)

        if result is not None:
            score = evaluate(result)
            if score < best_score:
                best_h, best_score = result, score

        elif np.random.rand() < 0.5:
            break

        # Safety net
        best_h = np.clip(best_h, 0, 1)
        best_h *= (n_points / 2) / best_h.sum()

    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / best_h.sum()
    return best_h, evaluate(best_h), n_points
