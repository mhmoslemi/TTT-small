import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum < 1e-10:
        h[0] = len(h) / 2
        h /= h.sum() * dx  # rescale back
    else:
        h = h / h_sum * len(h) / 2  # normalize to match integral constraint
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h
