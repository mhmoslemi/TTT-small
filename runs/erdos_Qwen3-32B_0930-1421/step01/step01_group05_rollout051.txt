import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial_h_values if available, else start from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)

    dx = 2.0 / n_points

    def normalize_h(h):
        clipped = np.clip(h, 0, 1)
        total_mass = np.sum(clipped)
        scale = (n_points / 2) / max(total_mass, 1e-10)
        return np.clip(clipped * scale, 0, 1)

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Try multiple structured sampling strategies
    n_points_to_try = [30, 45, 60]
    for n in n_points_to_try:
        if time.time() - start_time >= budget_s:
            break

        # Resample h at new resolution
        res = correlate(h, np.ones(n) / n, mode='same')
        h_new = np.zeros(n)
        for i in range(n):
            start = max(0, int(i * n_points / n - 2))
            end = min(n_points, int(i * n_points / n + 2))
            h_new[i] = np.mean(h[start:end])

        h_new = normalize_h(h_new)
        test_c5 = compute_c5(h_new)

        if test_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = test_c5
            h = h_new.copy()
            n_points = n

    # Optimize best_h with adaptive stepsize
    learning_rates = [0.1, 0.05, 0.025]
    grad_steps = int((budget_s - (time.time() - start_time)) / 2)
    for t in range(grad_steps):
        if time.time() - start_time >= budget_s:
            break
        lr = learning_rates[min(t, len(learning_rates)-1)]

        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        h_flat = h.flatten().copy()
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1
            h1 = normalize_h(h_flat + 0.001 * e)
            h2 = normalize_h(h_flat - 0.001 * e)
            c1 = compute_c5(h1)
            c2 = compute_c5(h2)
            grad[i] = (c1 - c2) / 0.002

        h = normalize_h(h - lr * grad)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if t % 10 == 0:
            print(f"Step {t}, C5={best_c5:.6f}")

    return best_h.tolist(), best_c5, n_points
