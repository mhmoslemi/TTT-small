    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    # Compute C₅ - uses center-symmetric definition
    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()

        if abs(optimized_h_sum - target_sum) > 1e-5:
            optimized_h *= (target_sum / optimized_h_sum)

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf

        for k in range(1, n_points):
            # Left-to-right shift k
            if k <= n_points - 1:
                corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
                max_corr = max(max_corr, corr * dx)

            # Right-to-left shift k
            if k <= n_points - 1:
                corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
                max_corr = max(max_corr, corr * dx)

        return max_corr

    best_c5 = compute_c5(h)
    if verbose:
        print(f"Initial C5: {best_c5:.6f}")

    # Frequency domain definition for h
    def fourier_h(optimized_h):
        # Centered FFT: shift zero component to the center
        return np.fft.fftshift(np.fft.fft(optimized_h))

    def spatial_h(fourier_coeffs):
        return np.real(np.fft.ifft(np.fft.ifftshift(fourier_coeffs)))

    while time.time() - start_time < budget_s:

        # Early stopping if no improvement for 60s

            break
