import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    def normalize_h(h_in):
        h_squared = np.clip(h_in, 0, 1) ** 2
        mass = 2.0 * h_squared.sum() / n_points
        scale = (n_points / 2) / mass
        return np.sqrt(np.clip(h_in * scale, 0, 1))

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Convex optimization with L1 regularization (75% of budget)
    phase1_time_limit = 0.75 * budget_s
    remaining_time = phase1_time_limit - (time.time() - start_time)
    if remaining_time > 0:
        lam = 0.01
        h_opt = cp.Variable(n_points)
        
        constraints = [
            h_opt >= 0, h_opt <= 1,
            cp.sum(h_opt) == n_points / 2
        ]
        
        overlap = correlate(h_opt, 1 - h_opt, mode='full') * dx
        objective = cp.Minimize(cp.max(overlap) + lam * cp.norm1(h_opt))
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.CVXOPT, maxiters=100)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_candidate = np.clip(h_opt.value, 0, 1)
                h_normalized = normalize_h(h_candidate)
                while time.time() - start_time < phase1_time_limit:
                    c5 = compute_c5(h_normalized)
                    if c5 < best_c5:
                        best_h = h_normalized.copy()
                        best_c5 = c5
                    break
        except Exception:
            pass

    # Phase 2: Gradient descent (25% of time)
    gradient_time_limit = budget_s - (time.time() - start_time)
    if gradient_time_limit > 0:
        h = normalize_h(best_h)
        learning_rate = 0.01
        eps = 1e-6

        while time.time() - start_time < budget_s:
            H = h.copy()
            grad = np.zeros_like(h)

            for i in range(n_points):
                h_u = h.copy()
                h_u[i] += eps
                h_u = normalize_h(h_u)
                c_p = compute_c5(h_u)

                h_d = h.copy()
                h_d[i] -= eps
                h_d = normalize_h(h_d)
                c_n = compute_c5(h_d)

                grad[i] = (c_p - c_n) / (2 * eps)

            h -= learning_rate * grad
            h = normalize_h(h)

            # Update best
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

            # Schedule
            learning_rate *= 0.995
            if learning_rate < 1e-4:
                learning_rate = 1e-4

            if time.time() - start_time >= budget_s:
                break

    return best_h.tolist(), best_c5, n_points
