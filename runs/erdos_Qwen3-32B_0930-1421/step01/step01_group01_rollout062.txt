import numpy as np
import time
from scipy.signal import correlate


def evaluate_C5(h, dx):
    h = h.clip(0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    h = h.clip(0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

        if len(h) != n_points:
            h.resize(n_points)
    except:

        h[:10] = 0.1
        h[10:30] = 0.3
        h[30:] = 0.2
        h = rescale_h(h)

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)
    iter_count = 0
    perturb_amp = 0.1

    def perturb(h, amp):
        n_segments = 5
        segment_len = int(n_points / n_segments)
        candidate_h = h.copy()
        for _ in range(2):
            start = np.random.randint(0, n_points - 1)
            end = start + np.random.randint(1, min(n_points - start, segment_len))
            candidate_h[start:end] = candidate_h[start:end] + np.random.randn(end - start) * amp
        return candidate_h

    while time.time() - start_time < budget_s:
        iter_count += 1

        if iter_count % 50 == 0:
            perturb_amp *= 0.95

        candidate_h = perturb(h, perturb_amp)
        candidate_h = rescale_h(candidate_h)
        candidate_C5 = evaluate_C5(candidate_h, dx)

        if candidate_C5 < best_C5:
            best_h = candidate_h.copy()
            best_C5 = candidate_C5
            h = candidate_h.copy()

        elif np.random.rand() < 0.1:
            h = candidate_h.copy()

        if iter_count % 500 == 0:
            perturb_amp = max(0.05, perturb_amp * 0.9)

    return best_h, best_C5, n_points
