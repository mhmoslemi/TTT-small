import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    corrs = []
    for k in range(1, n_points):
        len_k = n_points - k
        if len_k <= 0:
            break
        corr = np.dot(h[:len_k], 1 - h[k:k + len_k])
        corrs.append(corr)
    for k in range(1, n_points):
        len_k = n_points - k
        if len_k <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:len_k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def normalize_h(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total < 1e-10:
        return np.full(n_points, 0.5)
    scale = (n_points / 2) / total
    h = np.clip(h * scale, 0.0, 1.0)
    return h

def raised_cosine_init(n_points):
    i = np.arange(n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
    return normalize_h(h, n_points)

def perturbation_search(h_best, h_gbest, n_points, budget, seed=None):
    np.random.seed(seed)
    N = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 500
    times = []

    # Initialize particles
    if h_best is None:
        h_particles = np.array([raised_cosine_init(n_points) for _ in range(N)])
    else:
        h_particles = np.array([h_best for _ in range(N)])

    # Compute initial velocities
    velocities = 0.1 * np.random.rand(N, n_points)
    pbest = h_particles.copy()
    pbest_vals = [compute_c5(p, n_points) for p in pbest]
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = min(pbest_vals)

    start = time.time()
    for it in range(max_iter):
        if time.time() - start > budget:
            break
        for i in range(N):
            r1, r2 = np.random.rand(2)
            vel = (w * velocities[i] +
                   c1 * r1 * (pbest[i] - h_particles[i]) +
                   c2 * r2 * (gbest - h_particles[i]))
            hnew = h_particles[i] + vel
            hnew = normalize_h(hnew, n_points)
            h_particles[i] = hnew

            val = compute_c5(hnew, n_points)
            # Update pbest
            if val < pbest_vals[i]:
                pbest[i] = hnew.copy()
                pbest_vals[i] = val
            # Update gbest
            if val < gbest_val:
                gbest = hnew.copy()
                gbest_val = val

        w = max(0.4, w * 0.995)
        times.append(time.time() - start)

    return gbest, gbest_val
