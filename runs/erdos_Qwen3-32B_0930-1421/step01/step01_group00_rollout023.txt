import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial normalization
    h = np.clip(h, 0, 1)
    h = (h * target_sum) / h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()

    def compute_c5(h):
        h = np.clip(h, 0, 1).copy()
        h /= h.sum() * 2.0 / n_points  # Normalize to make ∫h dx = 1
        h = (h * target_sum) / h.sum()
        h = np.clip(h, 0, 1)

        ones_minus_h = 1.0 - h
        max_overlap = -np.inf

        for k in range(1, n_points):
            if k >= h.shape[0] - 1:
                continue
            corr_pos = np.dot(h[:n_points - k], ones_minus_h[k:])
            corr_neg = np.dot(h[k:], ones_minus_h[:n_points - k])
            max_overlap = max(max_overlap, corr_pos * dx, corr_neg * dx)

        return max_overlap

    def compute_top_shifts(h, top_k=5):
        overlap = np.zeros_like(h)
        ones_minus_h = 1.0 - h

        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr_pos = np.dot(h[:n_points - k], ones_minus_h[k:])
            corr_neg = np.dot(h[k:], ones_minus_h[:n_points - k])
            max_k = max(corr_pos, corr_neg)
            if np.isfinite(max_k):
                overlap[k] = max_k
        top_k = np.argsort(overlap)[-5:]
        return top_k

    best_c5 = compute_c5(h)

    while time.time() - start_time < budget_s:
        # Find top problematic shifts
        top_shifts = compute_top_shifts(h)

        # Focus adjustments on indices where contribution is large
        for k in top_shifts:
            if k == 0:
                continue
            if k >= n_points - 1:
                continue
            contributions = h[:n_points - k] * (1 - h[k:])
            top_indices = np.argsort(contributions)[-int(len(contributions) * 0.1) - 1:]

            # Decrease h[i], increase h[i+k] if permissible
            for i in top_indices:
                i2 = i + k
                if h[i] > 1e-5 and h[i2] < 1 - 1e-5:
                    # Gradual shift of mass
                    shift = min(0.01, h[i] * 0.1)
                    h[i] -= shift
                    h[i2] += shift

        # Clip and re-normalize
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = (target_sum / h_sum) * h
        h = np.clip(h, 0, 1)

        # Compute new bound
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

        # Early stopping if not improving
        if time.time() - start_time > budget_s - 5:
            break

    return (best_h, best_c5, n_points)
