import numpy as np
import time
from scipy.signal import correlate
import random


def compute_C5(h, dx):
    """Compute the maximum overlap C5 value."""
    cor = correlate(h, 1 - h, mode='full')
    return np.max(cor) * dx


def rescale(h):
    """Normalize h to satisfy the integral constraint."""
    h = np.clip(h, 0, 1)
    norm = h.sum()
    if norm == 0:
        h[0] = len(h)
        return h / h.sum() * len(h) / 2  # ensure average = 1/2
    return h / norm * len(h) / 2


def flip_segment(h, segment_length=10):
    """Randomly flip a segment within h."""
    start = np.random.randint(0, len(h) - segment_length)
    h[start:start + segment_length] = 1 - h[start:start + segment_length]
    return h


def perturb(h, alpha=0.1):
    """Apply a small random perturbation to h."""
    noise = np.random.normal(0.0, alpha, size=len(h))
    h += noise
    return h

    np.random.seed(seed)

    best_h = h.copy()
    best_C5 = compute_C5(best_h, dx)
    iterations = 0
    perturbation_strength = 0.1
    perturbation_decay = 0.995
    restart_freq = 200
    last_improvement = 0

    start_time = time.time()
    iter_limit = int(budget_s * 0.95)  # Leave buffer for final return

    while iterations < iter_limit and time.time() - start_time < budget_s:
        iterations += 1

        # Perturb current h
        h_copy = best_h.copy()
        h_copy = rescale(h_copy)
        h_copy = perturb(h_copy, alpha=perturbation_strength)
        h_candidate = rescale(h_copy)
        h_candidate = flip_segment(h_candidate)

        # Clip to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)

        # Rescale to satisfy constraint
        h_candidate = rescale(h_candidate)

        # Evaluate C5
        candidate_C5 = compute_C5(h_candidate, dx)

        # Store if better
        if candidate_C5 < best_C5:
            best_h = h_candidate.copy()
            best_C5 = candidate_C5
            last_improvement = iterations

        # Occasionally double perturbation strength if no improvement
        if iterations % restart_freq == 0:
            perturbation_strength *= 2
        else:
            # Otherwise decay slowly
            perturbation_strength *= perturbation_decay

        # Decay further if too large
        if perturbation_strength > 0.3:
            perturbation_strength = 0.3

    return best_h, best_C5, n_points
