import numpy as np
import time
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def get_discrete_representation(h, levels=5):
    """Quantize h into levels of [0, 0.25, 0.5, 0.75, 1.0]."""
    return np.round(h * (levels - 1)) / (levels - 1)


def branch_and_bound_search(
    h_init, budget_s, n_points=42, max_branch_depth=5, max_queue_size=1000
):
    dx = 2.0 / n_points
    best_h = np.copy(h_init)
    best_C5 = evaluate_C5(best_h)
    queue = []
    start_time = time.time()

    # Initialize queue with perturbed h
    for seed in range(20):
        np.random.seed(seed)
        perturbations = np.random.randn(n_points)
        h_perturb = h_init + 0.1 * perturbations
        h_quantized = get_discrete_representation(
            ensure_feasible(h_perturb), levels=5
        )
        queue.append((h_quantized, 0))  # (solution, depth)
        if len(queue) > max_queue_size:
            break

    while time.time() - start_time < budget_s and queue:
        # Select node with smallest depth
        h_current, depth = queue.pop(0)
        if depth > max_branch_depth:
            continue

        # Consider all possible branches
        improved = False
        for idx in range(n_points):
            # Skip if already at a discrete level
            if h_current[idx] in {0, 0.25, 0.5, 0.75, 1.0}:
                continue

            for value in {0, 0.25, 0.5, 0.75, 1.0}:
                h_test = np.copy(h_current)
                h_test[idx] = value
                h_test = ensure_feasible(h_test)

                # Compute optimistic bound: if current max_C5 * dx < target_C5, prune
                current_segment = h_test[: idx + 1]
                current_right_segment = h_test[idx:]
                left_overlap = current_segment.dot(1 - current_right_segment)
                lb = left_overlap * dx

                if lb < best_C5:
                    continue

                # If feasible, add to queue
                queue.append((h_test, depth + 1))
                new_C5 = evaluate_C5(h_test)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = np.copy(h_test)
                    improved = True

        # Escape local optimum by random perturbation if no improvement
        if not improved:
            perturb_h = np.copy(best_h)
            idx = np.random.randint(n_points)
            perturb_h[idx] = np.random.choice([0.0, 0.25, 0.5, 0.75, 1.0])
            perturb_h = ensure_feasible(perturb_h)
            queue.append((perturb_h, depth + 1))

    return best_h, best_C5

        h_init = ensure_feasible(h_init)

    h_init_discrete = get_discrete_representation(h_init, levels=5)

    best_h, best_C5 = branch_and_bound_search(
        h_init_discrete, budget_s, n_points
    )
    return best_h, best_C5, n_points
