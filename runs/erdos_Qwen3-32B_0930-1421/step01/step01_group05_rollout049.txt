import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        cor = correlate(h, 1 - h, mode='full') * dx
        return np.max(cor)

    def normalize_h(h_in):
        h_clipped = np.clip(h_in, 0, 1)
        h_sum = h_clipped.sum()
        if abs(h_sum) < 1e-10:
            return np.zeros_like(h_in)
        scale = (n_points / 2) / h_sum
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Try multiple n_points to prevent local optima
    resolutions = [45, 60, 30]
    time_per_resolution = (budget_s / len(resolutions))

    for n_idx, n in enumerate(resolutions):
        if time.time() - start_time >= budget_s:
            break

        rem_time = (n_idx + 1) * time_per_resolution - (time.time() - start_time)
        start_res = time.time()

        # Initialize new h for this resolution
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h_res = np.array(initial_h_values, dtype=np.float64)
        else:
            h_res = np.random.rand(n)
            h_res = h_res / h_res.sum() * n / 2

        def optimize_with_scipy(h_init, time_budget):
            best_h_local = h_init.copy()
            best_c5_local = compute_c5(best_h_local)
            learning_rate = 0.1
            eps = 1e-6

            t_start = time.time()
            while time.time() - t_start < time_budget:
                grad = np.zeros_like(h_init)
                for i in range(n):
                    h_up = h_init.copy()
                    h_up[i] += eps
                    h_up = normalize_h(h_up)
                    c_up = compute_c5(h_up)

                    h_down = h_init.copy()
                    h_down[i] -= eps
                    h_down = normalize_h(h_down)
                    c_down = compute_c5(h_down)

                    grad[i] = (c_up - c_down) / (2 * eps)

                # Apply gradient update
                update = -learning_rate * grad
                h_new = h_init + update
                h_new = normalize_h(h_new)

                c_new = compute_c5(h_new)
                if c_new < best_c5_local:
                    best_h_local = h_new.copy()
                    best_c5_local = c_new

                h_init = h_new
                learning_rate *= 0.99

            return best_h_local, best_c5_local

        if n < len(best_h):
            # Interpolate lower to higher resolution
            h_compat = np.zeros(n)
            scale = n / len(best_h)
            for i in range(n):
                idx = int(i / scale)
                h_compat[i] = best_h[idx]
            h_compat = normalize_h(h_compat)
        else:
            # Interpolate higher to lower resolution
            h_compat = np.zeros(n)
            scale = len(best_h) / n
            for i in range(n):
                j_start = int(i * scale)
                j_end = min(j_start + 1, len(best_h))
                h_compat[i] = best_h[j_start:j_end].sum()

        optimized_h, optimized_c5 = optimize_with_scipy(h_compat, min(rem_time, budget_s))
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h

    return best_h.tolist(), best_c5, n_points
