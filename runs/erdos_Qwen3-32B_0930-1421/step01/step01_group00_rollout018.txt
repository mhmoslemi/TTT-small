import numpy as np
import time
import cvxpy as cp
import scipy.optimize

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def clip_and_rebalance(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        scale = target_sum / h_clipped.sum()
        h_rescaled = h_clipped * scale
        return np.clip(h_rescaled, 0, 1)

    def compute_c5(h):
        h = h.copy()
        h = clip_and_rebalance(h, target_sum)
        ones_minus_h = 1.0 - h
        corr_rev = scipy.signal.correlate(h, ones_minus_h, mode='full')
        corr_rev *= dx
        return np.max(corr_rev)

    h_opt = h.copy()
    best_c5 = compute_c5(h_opt)
    best_h = h_opt.copy()
    best_time = time.time()

    def logl(msg):
        if verbose:
            print(msg)

    # Regularly timed checkpoints to explore efficient landscape navigation
    phases = [
        ("Local Perturb", 0.4 * budget_s),
        ("Refinement", 0.4 * budget_s),
        ("Final Iterations", 0.2 * budget_s),
    ]

    for phase_name, phase_time in phases:
        phase_start = time.time()
        while time.time() - start_time < phase_time:
            # Perturbation
            indices = np.random.choice(n_points, size=min(20, n_points // 2), replace=False)
            noise = np.random.normal(0, 0.05, len(indices))
            h_proposed = h_opt.copy()
            h_proposed[indices] += noise
            h_proposed = clip_and_rebalance(h_proposed, target_sum)

            new_c5 = compute_c5(h_proposed)
            if new_c5 < best_c5:
                h_opt[:] = h_proposed
                best_c5 = new_c5
                best_h[:] = h_opt.copy()
                best_time = time.time()
                logl(f"{phase_name}: Improvement to C5 = {best_c5:.8f}")

        logl(f"Phase {phase_name} ends at time {time.time() - start_time:.2f}s")

    def grad_warm_start(h0, target_sum, dx):
        h = cp.Variable(n_points)
        cons = [
            0 <= h,
            h <= 1,
            cp.sum(h) == target_sum,
        ]

        obj = cp.Minimize(cp.max(
            [
                cp.sum(h[:len(h) - k] * (1 - h[k:]) * dx)
                for k in range(1, n_points)
                if len(h) - k > 0
            ]
        ))
        p = cp.Problem(obj, cons)
        ECOS_RES = {"max_iters": 500, "verbose": False, "abstol": 1e-5}
        try:
            p.solve(solver=cp.ECOS, **ECOS_RES)
        except cp.SolverError:
            pass

        h_sol = np.array(h.value).flatten()
        if h_sol is None:
            h_sol = h0
        return clip_and_rebalance(h_sol, target_sum)

    if time.time() - start_time < budget_s - 10:
        logl("Running Convex Relaxation")
        h_relax = grad_warm_start(h_opt, target_sum, dx)
        new_c5 = compute_c5(h_relax)
        if new_c5 < best_c5:
            h_opt[:] = h_relax
            best_c5 = new_c5
            best_h[:] = h_opt.copy()
            logl(f"Convex Relaxation improves C5 = {best_c5:.8f}")

    logl(f"Final C5Bound: {best_c5:.8f} in {(time.time() - start_time):.2f}s")
    return (best_h, best_c5, n_points)
