import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values)
    total_h = float(np.sum(initial_h_values))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.array(initial_h_values) * scale_factor, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()
    last_improvement_time = start_time

    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    # Binary Simulated Annealing
    n_steps = int((0.4 * budget_s) / (n_points / 10))
    temp_schedule = lambda t: 1.0 / (1 + np.log(1 + t))
    max_energy_swap = 0.1
    target_sum = n_points // 2

    h_binary = (np.random.rand(n_points) < 0.5).astype(float)
    h_binary = h_binary * 2
    h_binary[h_binary > 1] = 0.0
    h_binary = (h_binary + np.random.rand(n_points) * 0.01).clip(0, 1)
    h_binary[h_binary > 0.5] = 1.0
    h_binary[h_binary < 0.5] = 0.0

    if np.sum(h_binary) < target_sum:
        indices = np.where(h_binary == 0.0)[0]
        idx = np.random.choice(indices, size=int(target_sum - np.sum(h_binary)), replace=False)
        h_binary[idx] = 1.0
    else:
        indices = np.where(h_binary == 1.0)[0]
        idx = np.random.choice(indices, size=int(np.sum(h_binary) - target_sum), replace=False)
        h_binary[idx] = 0.0

    for sa_iter in range(n_steps):
        if time.time() - start_time > budget_s:
            break
        t = sa_iter
        T = temp_schedule(t)
        idx1, idx2 = np.random.choice(n_points, size=2, replace=False)
        h_swap = h_binary.copy()
        h_swap[idx1], h_swap[idx2] = h_swap[idx2], h_swap[idx1]
        old_c5 = evaluate_erdos_solution(h_binary, n_points)
        new_c5 = evaluate_erdos_solution(h_swap, n_points)
        delta = new_c5 - old_c5

        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h_binary[:] = h_swap
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = h_binary.copy()
                last_improvement_time = time.time()

        if sa_iter % max(1, n_steps // 100) == 0 and time.time() - last_improvement_time > 120:
            break

    h_scaled = h_binary * scale_factor
    h_scaled = h_scaled.clip(0, 1)

    # Adjust sum(h) to n_points / 2 with 0.5
    h_sum = np.sum(h_scaled)
    delta_sum = h_sum - n_points / 2.0
    if abs(delta_sum) > 1e-10:
        if delta_sum > 0:
            indices = np.where(h_scaled == 1.0)[0]
            idx = np.random.choice(indices, 1, replace=False)[0]
            h_scaled[idx] = 0.5
        else:
            indices = np.where(h_scaled == 0.0)[0]
            idx = np.random.choice(indices, 1, replace=False)[0]
            h_scaled[idx] = 0.5

    # Continuous optimization
    def post_objective(x):
        x = x.copy()
        x[x < 0] = 0.0
        x[x > 1] = 1.0
        return evaluate_erdos_solution(x, n_points)

    cons = (
        {"type": "eq", "fun": constraint_sum},
        {"type": "ineq", "fun": constraint_upper},
        {"type": "ineq", "fun": constraint_lower}
    )

    result = minimize(
        post_objective,
        h_scaled,
        method="SLSQP",
        bounds=[(0.0, 1.0)] * n_points,
        constraints=cons,
        options={"maxiter": 500}
    )

    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h[:] = result.x
        last_improvement_time = time.time()

    return best_h, best_c5, n_points
