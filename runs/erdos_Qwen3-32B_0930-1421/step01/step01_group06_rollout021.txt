import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h_initial = initial_h_values
    n_points = len(h_initial)

    total_h = float(np.sum(h_initial))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h_initial * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    n_target = n_points // 2
    h_binary = np.zeros(n_points, dtype=np.float64)
    indices = np.random.choice(n_points, size=n_target, replace=False)
    h_binary[indices] = 1.0

    def simulated_annealing():
        nonlocal h_binary, best_c5, best_h, last_improvement_time
        h_best_binary = np.copy(h_binary)
        c5_best = evaluate_erdos_solution(h_binary, n_points)

        T0 = 1.0
        T = T0
        cooling_rate = 0.98
        max_iter = 5000

        for i in range(max_iter):
            if time.time() - start_time > budget_s - 200:
                break
            indices = np.random.choice(n_points, size=2, replace=False)
            i1, i2 = indices
            if h_binary[i1] != h_binary[i2]:
                h_binary[i1], h_binary[i2] = h_binary[i2], h_binary[i1]
                new_c5 = evaluate_erdos_solution(h_binary, n_points)
                delta = new_c5 - c5_best
                if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                    c5_best = new_c5
                    h_best_binary = np.copy(h_binary)
                    last_improvement_time = time.time()
                else:
                    h_binary[i1], h_binary[i2] = h_binary[i2], h_binary[i1]
                T *= cooling_rate

        if c5_best < best_c5:
            best_c5 = c5_best
            scale = n_points / 2.0
            total = np.sum(h_best_binary)
            if abs(total - n_target) < 1e-10:
                scale_factor = scale / total
                h_scaled = np.minimum(h_best_binary * scale_factor, 1.0)
                best_h[:] = h_scaled
            else:
                idx = np.where(h_best_binary == 0)[0]
                h_scaled = np.copy(h_best_binary)
                if total < n_target:
                    h_scaled[idx[0]] = 1.0
                else:
                    h_scaled[idx[0]] = 0.0
                h_continuous(h_scaled)

    def h_continuous(h_bin):
        nonlocal best_h, best_c5, last_improvement_time
        h_float = np.copy(h_bin)
        scale = n_points / 2.0
        total = np.sum(h_float)

        if abs(total - scale) < 1e-12:
            pass  # Already valid
        elif total < scale:
            idx = np.where(h_float == 0)[0]
            if idx.size == 0:
                ratio = scale / total
                h_float = h_float * ratio
            else:
                h_float[idx[0]] = 1.0
                h_float = np.minimum(h_float, 1.0)
        else:
            idx = np.where(h_float == 1)[0]
            if idx.size == 0:
                ratio = scale / total
                h_float = h_float * ratio
            else:
                h_float[idx[0]] = 0.0
                h_float = np.maximum(h_float, 0.0)
        h_float = h_float.clip(0, 1)
        h_continuous_result = h_float
        h_continuous_obj = evaluate_erdos_solution(h_continuous_result, n_points)
        if h_continuous_obj < best_c5:
            best_c5 = h_continuous_obj
            best_h[:] = h_continuous_result[:]
            last_improvement_time = time.time()

        def obj_h_continuous(x):
            if time.time() - start_time > budget_s - 5:
                return float('inf')
            return np.max(2.0 / n_points * np.correlate(x, 1 - x, 'full'))

        cons = (
            {"type": "eq", "fun": lambda x: np.sum(x) - n_points / 2},
        )
        bounds = [(0, 1)] * n_points
        result = minimize(obj_h_continuous, h_continuous_result, method='SLSQP',
                          bounds=bounds, constraints=cons, options={'maxiter': 500})
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            c5_opt = evaluate_erdos_solution(h_opt, n_points)
            if c5_opt < best_c5:
                best_h[:] = h_opt[:]
                best_c5 = c5_opt
                last_improvement_time = time.time()

    simulated_annealing()
    h_continuous(h_binary)
    return best_h, best_c5, n_points
