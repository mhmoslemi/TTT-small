import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # Provided at runtime

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    correl = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correl) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total < 1e-8:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def build_basis_functions(n_points, N, x_centers, alpha):
    dx = 2.0 / n_points
    x_grid = np.linspace(0, 2, n_points)
    psis = []
    for xc in x_centers:
        i = int((xc - 0.0) / dx)
        i_start = max(0, i - 5)
        i_end = min(n_points, i + 6)
        xx = x_grid[i_start:i_end]
        psi = np.zeros_like(h)
        mask = np.abs(x_grid - xc) < 0.3
        psi[mask] = np.exp(-alpha * (x_grid[mask] - xc)**2)
        psi = np.clip(psi, 0, 1)  # Normalize to [0, 1]
        psis.append(psi)
    return np.array(psis)


def find_base_centers(h, dx, N=10):
    x_vals = np.linspace(0, 2, len(h))
    weighted_centers = np.sum(h * x_vals) / (np.sum(h) + 1e-8)
    bins = np.linspace(0, 2, N + 1)
    base_centers = (bins[:-1] + bins[1:]) / 2
    return base_centers


def optimize_weights(psis, dx, max_overlap_budget=0.0):
    n_basis = psis.shape[0]
    a = cp.Variable(n_basis, nonneg=True)

    objective = cp.Minimize(cp.norm(a, 1))

    h_expr = cp.sum(cp.multiply(a[..., np.newaxis], psis), axis=0)
    ones_minus_h = 1 - h_expr
    h_lm = h_expr * dx
    constraint = [h_lm == 1.0]

    for k in range(1, len(psis) + 1):
        if k > len(psis):
            break
        corr = cp.sum(cp.multiply(psis, np.roll(ones_minus_h, k, axis=0)), axis=0)
        constraint.append(corr <= max_overlap_budget)

    h_expr = cp.sum(cp.multiply(a[..., np.newaxis], psis), axis=0)
    constraint += [h_expr >= 0, h_expr <= 1]

    prob = cp.Problem(objective, constraint)
    prob.solve(solver=cp.PDLP, verbose=False)
    if a.value is not None:
        a_opt = np.clip(a.value, 0, 1)
        h_opt = np.sum(a_opt[..., np.newaxis] * psis, axis=0)
        h_opt = ensure_feasible(h_opt)
        return h_opt
    else:
        return np.zeros_like(psis[0])


def perturb_base_centers(base_centers, dx, sigma=0.1):
    base_centers = np.array(base_centers)
    perturbation = sigma * dx * np.random.randn(len(base_centers))
    perturbed = base_centers + perturbation
    perturbed = np.clip(perturbed, dx, 2 - dx)
    return np.sort(perturbed)


def run(budget_s=1100, n_points=64, base_centers=None, base_N=10, alpha=4.0):
    start_time = time.time()

    if initial_h_values is not None:
        h = np.copy(initial_h_values)
        base_centers = find_base_centers(h, 2.0 / n_points, N=base_N)
    else:
        if base_centers is None:
            base_centers = np.linspace(0.1, 1.9, base_N)

    best_h = None
    best_C5 = float("inf")
    num_attempts = 0
    dx = 2.0 / n_points

    while time.time() - start_time < budget_s:
        # Build basis functions
        psis = build_basis_functions(n_points, base_N, base_centers, alpha=alpha)
        if psis.shape[0] != base_N or psis.shape[1] != n_points:
            break

        # Solve optimization model
        h_opt = optimize_weights(psis, dx)

        # Evaluate the result
        current_C5 = evaluate_C5(h_opt)
        h_opt = ensure_feasible(h_opt)

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_opt
            print(f"Improved to C₅ = {current_C5:.8f} @ {time.time() - start_time:.2f}s")

        # Perturb base centers and try again
        base_centers = perturb_base_centers(base_centers, dx, sigma=0.1)
        base_centers = np.sort(base_centers)
        num_attempts += 1

    return best_h, best_C5, n_points
