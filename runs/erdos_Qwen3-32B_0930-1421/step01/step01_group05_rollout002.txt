            constraints = [h_vars <= 1]
            constraints += [h_vars.sum() == n_points / 2]

            obj = cp.Minimize(cp.max(overlap))

            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.CBC, maximum_iterations=1000, verbose=False, time_limit=30)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_candidate = h_vars.value
                h_candidate = normalize_h(h_candidate)
                c5 = compute_c5(h_candidate)
                if c5 < best_c5:
                    best_h[:] = h_candidate
                    best_c5 = c5
        except Exception:
            pass

    def optimize_with_local_search():
        nonlocal best_c5, best_h
        learning_rate = 0.2
        patience = 0
        while time.time() - start_time < budget_s:
            for i in range(n_points):
                h_test = best_h.copy()
                h_test[i] = min(max(h_test[i] + (1 - h_test[i]) * np.random.uniform(-1, 1), 0), 1)

                c5_test = compute_c5(h_test)

    remaining_time = budget_s - (time.time() - start_time)
