import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    h = h_init.copy()
    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def get_overlap_constraints(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        return overlaps

    def random_perturb(h_vec, noise=0.05):
        h_vec = h_vec.copy()
        h_vec += np.random.normal(0, noise, h_vec.shape)
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = h_vec.sum()
        h_vec *= (constraint_sum / h_sum)
        return h_vec

    best_h = h.copy()
    best_c5 = objective(best_h)

    # LP with warm starts and linearized constraints
    while time.time() - start_time < budget_s:
        h_current = best_h.copy()
        overlaps = get_overlap_constraints(h_current)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]

        # Linear approximation around current h for bilinear constraint at shift k
        def lin_overlap(h_var, k_shift, h_base):
            h_base = h_base.reshape(n_points)
            overlap = np.correlate(h_base, 1 - h_base, mode='full') * dx
            grads = np.zeros((n_points,))
            for i in range(n_points):
                delta = np.zeros_like(h_var)
                delta[i] = 1
                h_up = (h_base + delta.clip(0,1)) * (constraint_sum / (h_base + delta.clip(0,1)).sum())
                overlap_i = np.correlate(h_up, 1 - h_up, mode='full') * dx
                grads[i] = (overlap_i[k_shift] - overlap[k_shift])
            return grads

        grads = lin_overlap(h_current, max_k, h_current)

        h_var = cp.Variable(n_points)
        C = cp.Variable()

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == constraint_sum,
            grads @ h_var <= overlaps[max_k] - 0.5 * C
        ]
        # Clip k-shifted overlap to set C ≤ 1e-6 away from current
        for k in range(n_points):
            overlap_expr = 0
            for i in range(n_points):
                j = (i - k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j]) * dx
            constraints.append(overlap_expr <= C)

        # Solve LP
        objective_lp = cp.Minimize(C)
        problem = cp.Problem(objective_lp, constraints)
        problem.solve(solver='ECOS', verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_lp = np.clip(h_var.value, 0, 1)
            h_sum = h_lp.sum()
            h_lp = h_lp * (constraint_sum / h_sum)
            c5_lp = objective(h_lp)
            if c5_lp < best_c5:
                best_h = h_lp.copy()
                best_c5 = c5_lp

        # Use perturbation to escape local extrema
        if np.random.rand() < 0.2:
            best_h = random_perturb(best_h, noise=0.1)

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    best_h = best_h * (constraint_sum / best_h_sum)

    return best_h, objective(best_h), n_points
