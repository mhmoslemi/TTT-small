```python
import numpy as np
import time
import scipy.optimize as opt

def clamp_and_normalize(h, n_points):
    h_clamped = np.clip(h, 0.0, 1.0)
    total = h_clamped.sum()
    if total == 0:
        return np.full(n_points, 0.5)
    return h_clamped * (n_points / 2) / total

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def crossover(h1, h2):
    alpha = np.random.uniform(0.4, 0.6)
    h_child = alpha * h1 + (1 - alpha) * h2
    return clamp_and_normalize(h_child, len(h_child))

def mutate(h, sigma):
    h_mut = h + np.random.normal(0, sigma, h.shape)
    return clamp_and_normalize(h_mut, len(h))

def evaluate_population(population, n_points):
    scores = np.array([compute_c5(h, n_points) for h in population])
    return scores

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        init_h = initial_h_values[:]  # Use external initialization
        n_points = len(init_h)
    else:
        n_points = 47  # Default
        init_h = np.random.rand(n_points)
        init_h = clamp_and_normalize(init_h, n_points)

    pop_size = 50
    population = [init_h]
    for _ in range(pop_size - 1):
        random_h = np.random.rand(n_points)
        random_h = clamp_and_normalize(random_h, n_points)
        population.append(random_h)

    population = np.array(population)
    start_time = time.time()

    sigmas = 0.05  # Mutation intensity decay during evolution

    while time.time() - start_time < budget_s - 0.2:  # Reserve time for polish
        scores = evaluate_population(population, n_points)
        top_idx = np.argsort(scores)
        top_half = population[top_idx[:int(pop_size / 2)]]

        new_population = []
        for _ in range(int(pop_size * 0.3)):
            i1, i2 = np.random.choice(len(top_half), 2, replace=False)
            new_h = crossover(top_half[i1], top_half[i2])
            new_population.append(new_h)

        for _ in range(int(pop_size * 0.2)):
            i = np.random.choice(len(top_half))
            new_h = mutate(top_half[i], sigmas)
            new_population.append(new_h)

        for h in top_half:
            new_population.append(h)

        for i in range(len(new_population), pop_size):
            random_h = np.random.rand(n_points)
            random_h = clamp_and_normalize(random_h, n_points)
            new_population.append(random_h)

        population = np.array(new_population)
        sigmas *= 0.992  # Decay mutation

    best_idx = np.argmin(evaluate_population(population, n_points))
    best_h = population[best_idx]

    # Final refinement with L-BFGS-B
    def objective(h_flat):
        h = h_flat.copy()
        h = clamp_and_normalize(h, n_points)
        return compute_c5(h, n_points)

    if time.time() - start_time < budget_s:
        lbfgs_result = opt.minimize(
            objective,
            best_h,
            bounds=[(0, 1)] * n_points,
            method='L-BFGS-B',
            options={'maxiter': 300, 'ftol': 1e-6}
        )
        best_h = lbfgs_result.x
        best_h = clamp_and_normalize(best_h, n_points)

    final_score = compute_c5(best_h, n_points)
    return best_h, final_score, n_points
```
