import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def rescale_h(h):
    """Rescale h to satisfy the ∫h dx = 1 constraint (n_points / 2 sum)."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def evaluate_C5(h, dx):
    h = h.astype(float)
    h = rescale_h(h)
    h = np.clip(h, 0, 1)
    cor = correlate(h, 1 - h, mode='full')
    max_cor = np.max(cor)
    C5 = max_cor * dx
    return C5


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = rescale_h(np.tile(h, n_points // len(h) + 1)[:n_points])
    except:
        h = np.zeros(n_points)
        h[20:30] = 0.5
        h[40:44] = 1
        h[60:80] = 0.3
        h = rescale_h(h)

    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    start_time = time.time()
    segment_size = int(n_points * 0.1)
    perturb_amp = 0.1
    perturb_decay = 0.8
    perturb_freq = 50
    decay_counter = 0

    while time.time() - start_time < budget_s:
        h_current = best_h.copy()

        if np.random.rand() < 0.5:
            # Random segment flip
            start = np.random.randint(0, n_points - segment_size)
            h_current[start:start + segment_size] = 1 - h_current[start:start + segment_size]
        else:
            # Add Gaussian noise
            noise = np.random.normal(0, perturb_amp, n_points)
            h_current = best_h.copy() + noise
        h_current = rescale_h(np.clip(h_current, 0, 1))

        c5_new = evaluate_C5(h_current, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_current.copy()
            decay_counter = 0
        else:
            decay_counter += 1

        # Perturbation amplitude and frequency adjustment
        if decay_counter >= 50 or np.random.rand() < 0.05:
            perturb_amp *= perturb_decay
            perturb_amp = max(perturb_amp, 0.001)

        if time.time() - start_time > budget_s - 30:
            break

    return best_h, best_c5, n_points
