import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    max_corr = 0.0
    for k in range(1, n_points):
        if k < n_points:
            overlap = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        else:
            overlap = 0.0
        max_corr = max(max_corr, overlap)
    for k in range(1, n_points):
        if k < n_points:
            overlap = np.dot(h[k:], 1 - h[:n_points - k])
        else:
            overlap = 0.0
        max_corr = max(max_corr, overlap)
    return max_corr * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def random_candidate(n_points, random_state):
    h = random_state.rand(n_points)
    return enforce_normalization(h, n_points)

def initialize_population(pop_size, n_points):
    state = np.random.Generator(np.random.PCG64(np.random.SeedSequence()))
    return [random_candidate(n_points, state) for _ in range(pop_size)]

def selection(population, fitnesses, select_size):
    idxs = np.argsort(fitnesses)
    return [population[i] for i in idxs[:select_size]], [fitnesses[i] for i in idxs[:select_size]]

def crossover(h1, h2, alpha, state):
    h_child = alpha * h1 + (1 - alpha) * h2
    return enforce_normalization(h_child, len(h1))

def mutate(h, sigma, state):
    noise = state.normal(0, sigma, size=h.shape)
    h_mut = h + noise
    return enforce_normalization(h_mut, len(h))

def genetic_optimize(initial_h, pop_size=40, mut_rate=20, budget_s=1100):
    n_points = len(initial_h)
    start_time = time.time()
    
    # Initialize population
    population = [initial_h]
    pop = initialize_population(pop_size - 1, n_points)
    population.extend(pop)
    fitnesses = [compute_c5(h, n_points) for h in population]
    
    best_index = np.argmin(fitnesses)
    best_h = population[best_index]
    best_c5 = fitnesses[best_index]
    
    # Genetic main loop
    while time.time() - start_time < budget_s:
        # Evaluate current fitnesses
        # Select top parents
        parents, parent_fit = selection(population, fitnesses, len(population)//2)
        next_gen = []

        # Crossover
        for _ in range(mut_rate):
            alpha = np.random.rand()
            p1, p2 = np.random.choice(len(parents), size=2, replace=False)
            child = crossover(parents[p1], parents[p2], alpha, np.random)
            next_gen.append(child)

        # Mutation
        for _ in range(pop_size - mut_rate):
            idx = np.random.randint(len(parents))
            child = mutate(parents[idx], sigma=0.1 * (1 - (time.time() - start_time)/budget_s), state=np.random)
            next_gen.append(child)

        # Evaluate new gen
        new_fitness = [compute_c5(h, n_points) for h in next_gen]
        for h, f in zip(next_gen, new_fitness):
            if f < best_c5:
                best_h = h
                best_c5 = f

        population = next_gen
        fitnesses = new_fitness

    # Final refinement with gradient-based optimization
    def obj_f(h_flat):
        h = h_flat.reshape(-1)
        h = enforce_normalization(h, n_points)
        return compute_c5(h, n_points)

    result = opt.minimize(
        obj_f,
        x0=best_h.flatten(),
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 100}
    )
    final_h = result.x.reshape(-1)
    final_h = enforce_normalization(final_h, n_points)
    final_c5 = compute_c5(final_h, n_points)

    return final_h, final_c5, n_points

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    else:
        raise ValueError("initial_h_values is not defined. Cannot proceed without initial construction.")
    
    return genetic_optimize(h0, budget_s=budget_s)
