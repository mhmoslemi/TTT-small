import numpy as np
import time
import cvxpy as cp

def initialize_h(n_points):
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.copy(initial_h_values)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            h = h * (n_points / 2.0) / h_sum
        h = np.clip(h, 0.0, 1.0)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    return h

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    h = initialize_h(n_points)
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)

    # Define the optimization problem for each binary search step
    def is_feasible(C5, h_init=None):
        if h_init is not None:
            h_init = np.clip(h_init, 0.0, 1.0)
        else:
            h_init = initialize_h(n_points)
        h_var = cp.Variable(n_points, nonneg=True)
        z_var = {}
        # Flatten z to one-dimensional
        z_flat = []
        offsets = []

        # Build McCormick constraints
        for k in range(1, n_points):
            for i in range(n_points - k):
                j = i + k
                var_name = f"z_{i}_{j}"
                z = cp.Variable(n_points - k, nonneg=True)
                z_flat.extend([cp.Constant(0.0)] * i + [z] + [cp.Constant(0.0)] * (n_points - k - i - 1))
                for idx in range(len(z)):
                    # h is the first argument, 1-h is the second
                    z_flat_i = i + idx
                    z_flat_j = j + idx
                    offset = (z_flat_i, z_flat_j)
                    offsets.append(offset)

                    # McCormick constraints
                    constraints += [
                        z_flat[i*len(z):i*len(z) + len(z)] >= 0,
                        z_flat[i*len(z):i*len(z) + len(z)] <= h_var[z_flat_i],
                        z_flat[i*len(z):i*len(z) + len(z)] <= 1 - h_var[z_flat_j],
                        z_flat[i*len(z):i*len(z) + len(z)] >= h_var[z_flat_i] + (1 - h_var[z_flat_j]) - 1
                    ]

        # Total function objective is to minimize C5
        constraints += [
            cp.sum(h_var) == n_points / 2.0,
            h_var <= 1.0,
            h_var >= 0.0
        ]

        corr = np.correlate(h_var.value, 1 - h_var.value, mode="full") * (2.0 / n_points)
        max_corr = cp.Max(corr)
        problem = cp.Problem(cp.Minimize(max_corr), constraints)
        result = problem.solve(solver=cp.GUROBI, verbose=False)
        if result is not None and result < C5 + 1e-6:
            return h_var.value, result
        return None, None

    # Binary search for improved C5
    C_low = 0.0
    C_high = evaluate_erdos_solution(h, n_points)
    best_C5 = C_high
    best_h = np.copy(h)

    while time.time() - start_time < budget_s and C_high - C_low > 1e-6:
        C_mid = (C_low + C_high) / 2
        sol, val = is_feasible(C_mid)
        if sol is not None and val is not None:
            best_C5 = val
            best_h = np.copy(sol)
            C_high = C_mid
        else:
            C_low = C_mid

    # Perturb and repeat for robustness
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(initial_h_values + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        sol, val = is_feasible(0.3808, h_perturbed)
        if sol is not None and val is not None and val < best_C5:
            best_C5 = val
            best_h = sol

    h_validated = best_h
    h_validated[0] = np.minimum(h_validated[0], 1.0)
    h_validated[0] = np.maximum(h_validated[0], 0.0)
    assert np.all(np.isfinite(h_validated))
    h_validated = np.clip(h_validated, 0, 1)
    h_sum = float(np.sum(h_validated))
    h_validated = h_validated * (n_points / 2.0) / h_sum

    final_c5 = evaluate_erdos_solution(h_validated, n_points)
    return h_validated, final_c5, n_points
