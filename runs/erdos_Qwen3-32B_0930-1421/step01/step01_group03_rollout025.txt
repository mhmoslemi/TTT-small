import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    max_n_points = 100  # Cap for small sequences
    dx_tolerance = 1e-6
    iterations_per_lp = 50
    perturb_every = 5  # Perturb every this many LP steps
    max_perturbations = 10

    # Initializer
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h = np.zeros(n_points)
        # Basic uniform construction with two-unit spacing
        for i in range(n_points // 2):
            h[i * 2] = 1.0 / (n_points // 2)
        h *= n_points / 2 / h.sum()  # Normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2
    best_h = h.copy()
    best_c5 = np.inf
    iteration = 0

    def evaluate_c5(h_eval):
        h_eval = h_eval.reshape(n_points)
        h_eval = h_eval.clip(0, 1)
        h_sum = h_eval.sum()
        h_eval = h_eval * (constraint_sum / h_sum)
        overlaps = np.correlate(h_eval, 1 - h_eval, mode='full') * dx
        return np.max(overlaps)

    def lp_solver(h_current):
        C = cp.Variable()
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == constraint_sum,
            0 <= h_opt,
            h_opt <= 1
        ]

        for shift in range(-n_points + 1, n_points):
            shifted_indices = [(i - shift) % n_points for i in range(n_points)]
            overlap = 0.0
            for i in range(n_points):
                overlap += h_opt[i] * (1 - h_opt[shifted_indices[i]])
            overlap = overlap * dx
            constraints.append(overlap <= C)

        problem = cp.Problem(cp.Minimize(C), constraints)
        problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status == 'optimal':
            return h_opt.value, C.value
        else:
            raise RuntimeError("LP problem not solvable via GLPK")

    def perturb_h(h_perturb, perturb_factor=0.1):
        h = h_perturb.copy()
        random_pattern = np.random.randn(n_points)
        # Perturb following symmetry pattern
        for i in range(n_points):
            h[i] += random_pattern[i] * perturb_factor * (i % 2)
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    # Loop over iterations
    while time.time() - start_time < budget_s and iteration < max_perturbations * iterations_per_lp:
        if iteration % perturb_every == 0 and iteration > 0:
            h = perturb_h(h)

        try:
            h_lp, c5_lp = lp_solver(h)
            h = np.clip(h_lp, 0, 1).flatten()
            h = h * (constraint_sum / h.sum())
        except:
            pass

        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        iteration += 1

    # Post-loop projection to ensure constraint
    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, best_c5, n_points
