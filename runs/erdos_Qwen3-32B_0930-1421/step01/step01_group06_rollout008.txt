import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize to ensure sum(h) == n_points / 2
    h = h / h.sum() * n_points / 2
    h[h > 1.0] = 1.0
    h = h / h.sum() * n_points / 2

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    def solve_for_c5(C5):
        h_var = cp.Variable(n_points, nonneg=True, name="h")
        z_var = cp.Variable((n_points, n_points), nonneg=True)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1.0,
        ]

        for i in range(n_points):
            for j in range(n_points):
                constraints += [
                    z_var[i, j] <= h_var[i],
                    z_var[i, j] <= 1 - h_var[j],
                    z_var[i, j] >= h_var[i] + (1 - h_var[j]) - 1,
                ]

        for k in range(1, n_points):  # shifts by k
            valid_start_idx = np.arange(0, n_points - k)
            effective_corr = cp.sum([z_var[i, i + k] for i in valid_start_idx])
            constraints.append(effective_corr <= C5 * dx)

        objective = cp.Minimize(C5)
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver="ECOS", verbose=False)
        return problem.status, h_var.value, z_var.value

    # Binary search for best C5
    C_low = 0.0
    C_high = 1.0
    eps = 1e-6
    feasible_h = None

    timeout = budget_s * 0.9  # 90% of total time
    while time.time() - start_time < timeout and C_high - C_low > eps:
        C_mid = (C_low + C_high) / 2
        status, h_val, _ = solve_for_c5(C_mid)
        if status == "optimal":
            C_high = C_mid
            feasible_h = h_val
        else:
            C_low = C_mid

    if feasible_h is not None:
        feasible_h = np.clip(feasible_h, 0.0, 1.0)
        feasible_h /= feasible_h.sum() * 2.0 / n_points
        c5_val = evaluate_erdos_solution(feasible_h, n_points)
        if c5_val < best_c5:
            best_h = feasible_h * (2.0 / n_points)
            best_c5 = c5_val

    # Return best solution and bound
    best_h = best_h / best_h.sum() * n_points / 2
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, best_c5, n_points)
