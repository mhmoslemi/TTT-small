import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    # Forward lags
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    # Backward lags: handles h(x + k) for k = 1 to n_points-1
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if total == 0:
        return np.full(n_points, 0.5)
    return h_clipped * (n_points / 2) / total

def mutate(h, sigma):
    h_mut = h + np.random.normal(0, sigma, size=h.shape)
    return enforce_normalization(h_mut, len(h))

def crossover(h1, h2):
    alpha = np.random.rand()
    h_child = alpha * h1 + (1 - alpha) * h2
    return enforce_normalization(h_child, len(h_child))

def evaluate_population(population, n_points):
    scores = []
    for h in population:
        score = compute_c5(h, n_points)
        scores.append(score)
    return np.array(scores)
