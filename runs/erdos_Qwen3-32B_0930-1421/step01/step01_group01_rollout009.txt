def perturb_h(h, width, eps=0.2, seed=None):
    n_points = len(h)
    rng = np.random.Generator(np.random.PCG64(seed))
    start = rng.integers(0, n_points - width)
    slice = h[start:start + width].copy()
    delta = rng.uniform(-eps, eps, size=width)
    delta = (delta - delta.mean()) * 2.0
    new_slice = slice + delta
    return h.at[start:start + width].set(np.clip(new_slice, 0.0, 1.0))
