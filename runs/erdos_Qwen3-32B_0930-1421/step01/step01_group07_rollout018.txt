import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, peak_idx, shift_k):
    n = len(h)
    idx = np.random.choice(np.where(h == 1)[0][np.abs(np.where(h == 1)[0] - peak_idx) < n / 4])
    mirrored_idx = (peak_idx - shift_k + n) % n
    mirrored_idx = np.random.choice(np.where(h == 0)[0][np.abs(np.where(h == 0)[0] - mirrored_idx) < n / 4])
    h_new = h.copy()
    h_new[idx] = 0.0
    h_new[mirrored_idx] = 1.0
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)

    while time.time() - t0 < budget_s:
        corr = correlate(h, 1.0 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        shift_k = max_corr_idx - (len(h) - 1)

        # Make sure the shift is in valid range
        if shift_k < 0:
            shift_k = n_points - 1 + shift_k

        peak_x = (max_corr_idx - (len(1 - h) - 1)) % n_points

        # Flip two bits to address the max peak
        h = flip_two_bits(h, peak_x, shift_k)
        while np.sum(h) != n_points / 2:
            if np.sum(h) < n_points / 2:
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 1.0
            else:
                idx = np.random.choice(np.where(h == 1)[0])
                h[idx] = 0.0

        curr_c5 = evaluate_overlap(h)
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
