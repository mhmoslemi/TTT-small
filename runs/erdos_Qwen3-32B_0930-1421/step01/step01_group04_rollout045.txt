import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            corrs.append(corr)
        if k < n_points:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)
    return np.max(np.array(corrs) * dx)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-5:
        h = h * (n_points / 2) / total
        h = np.clip(h, 0.0, 1.0)
    return h

    time_start = time.time()

    # LP-based constraint addition (column generation style)
    def add_critical_constraints(h_candidate, k_list):
        n_points = len(h_candidate)
        critical_k = []
        dx = 2.0 / n_points
        for k in range(1, n_points):
            if n_points - k > 0:
                corr = np.dot(h_candidate[:n_points - k], 1 - h_candidate[k:])
            else:
                corr = 0.0
            overlap = corr * dx
            if overlap > best_c5 - 1e-6:
                critical_k.append(k)
        return critical_k if len(critical_k) > 0 else []

    def solve_lp(h_init, n_points, critical_k_list):
        from cvxpy import Variable, Problem, Minimize, sum, normal, Constant, NonPos

        h = Variable(n_points)
        C = Variable()
        con = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
        ]
        for k in critical_k_list:
            half1 = h[:n_points - k]
            half2 = 1 - h[k:]
            con.append(Constant(2.0 / n_points) * sum(half1 * half2) - C <= 0)
        obj = Minimize(C)
        prob = Problem(obj, con)
        prob.solve(solver='ECOS')
        if prob.status != 'optimal':
            return h_init
        return np.clip(h.value, 0.0, 1.0)

    critical_k_list = [1, 2, 4, 8, 16]
    while time.time() - time_start < budget_s * 0.7:
        h_lp = solve_lp(h, n_points, critical_k_list)

        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k_new = add_critical_constraints(h, critical_k_list)
        critical_k_list += k_new
        critical_k_list = list(set(critical_k_list))
        if not k_new:
            break
        if time.time() - time_start > budget_s * 0.9:
            break

    # Final polishing
    def final_objective(x):
        h_trial = np.reshape(x, (n_points,))
        h_trial = enforce_normalization(h_trial, n_points)
        return compute_c5(h_trial, n_points)

    if time.time() - time_start < budget_s:
        res = opt.minimize(

            c5_p = compute_c5(h_p, n_points)

                best_h = h_p
                best_c5 = c5_p

    final_h = enforce_normalization(best_h, n_points)
    return final_h, best_c5, n_points
