```python
import numpy as np
import time
from scipy.signal import correlate
import bisect


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    n_points = len(h)
    target_sum = n_points / 2
    if abs(total) < 1e-9:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= target_sum / total
    return h


def generate_initial_h(n_points):
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
        h = h[:n_points]
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = ensure_feasible(h)
    return h


def perturb_h(h, n_perturb=1, mut_strength=0.3):
    h = np.copy(h)
    n_points = len(h)
    idxs = np.random.choice(n_points, size=min(n_perturb, n_points), replace=False)
    for i in idxs:
        h[i] += (np.random.rand() * 2 - 1) * mut_strength
        h[i] = np.clip(h[i], 0, 1)
    h = ensure_feasible(h)
    return h


def branch_and_bound_search(h_init, budget_s=1100, n_points=42, n_branches=5000, max_depth=5):
    h = np.copy(h_init)
    best_h = np.copy(h)
    best_val = evaluate_C5(h)
    start_time = time.time()
    n_eval = 0

    while time.time() - start_time < budget_s and n_eval < n_branches:
        h_proposal = perturb_h(h, n_perturb=3, mut_strength=0.3)
        val_proposal = evaluate_C5(h_proposal)

        # Accept based on relative improvement at shallower branches
        if val_proposal < best_val or (np.random.rand() < 0.1 and time.time() - start_time < budget_s * 0.8):
            h = h_proposal
            if val_proposal < best_val:
                best_h = np.copy(h)
                best_val = val_proposal

        n_eval += 1

    return best_h, best_val


def run(budget_s=1100, n_points=42):
    n_points = int(n_points)
    h_init = generate_initial_h(n_points)
    best_h, best_val = branch_and_bound_search(h_init, budget_s=budget_s, n_points=n_points)
    return best_h, best_val, n_points
```
