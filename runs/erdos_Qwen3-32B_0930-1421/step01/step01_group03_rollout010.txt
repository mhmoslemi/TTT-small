import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 47
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points

    constraint_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def cvx_optimize(h_vec):
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        overlap_constraints = []

        for k in range(-n_points + 1, n_points):
            if k < 0:
                indices = [(i - k) % n_points for i in range(n_points)]
            else:
                indices = [i for i in range(n_points)]
            constraint = cp.sum([h_var[i] * (1 - (1.0 * h_var[indices[i]])) for i in range(n_points)]) * dx <= C
            overlap_constraints.append(constraint)

        constraints = [
            cp.sum(h_var) == constraint_sum,
            0 <= h_var,
            h_var <= 1
        ] + overlap_constraints

        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        return h_var.value, prob.value

    def perturb_h(h, perturbation_size=0.05):
        h = h.copy()
        noise = np.random.randn(n_points) * perturbation_size
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Multi-iteration LP with perturbation
    iterations = 0
    perturb_counter = 0
    while time.time() - start_time < budget_s * 0.98:
        iterations += 1
        if iterations % 5 == 0:
            best_h = perturb_h(best_h, perturb_counter / 20.0)
            perturb_counter += 1

        # Run linear program
        h_opt, c5_opt = cvx_optimize(best_h)
        if h_opt is not None and not np.isnan(h_opt).any():
            best_h = h_opt
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt

        if iterations % 25 == 0:
            print(f"Iteration {iterations}, best C5 so far: {best_c5}")
        if time.time() - start_time > budget_s * 0.99:
            break

    # Final projection and constraint validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, objective(best_h), n_points
