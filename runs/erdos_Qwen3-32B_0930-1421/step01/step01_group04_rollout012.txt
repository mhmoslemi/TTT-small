import numpy as np
from cvxpy import Variable, Problem, Minimize, constraints
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    try:
        init_h = initial_h_values.copy()
    except NameError:
        init_h = np.zeros(47)
        # Distribute activity in a pattern
        comb_period = 4
        pattern = np.array([1.0, 0.0, 0.0, 0.0])
        pattern = np.tile(pattern, 12)
        pattern = pattern[:47]
        pattern = pattern * (47 / 2.0) / pattern.sum()
        init_h = np.abs(pattern + 0.2 * np.random.rand(47))
        init_h = enforce_normalization(init_h, 47)
    n_points = len(init_h)

    h = Variable(n_points)
    C = Variable()
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2
    ]

    # Add initial constraints
    initial_ks = [1, 2, 4, 8, 16]
    for k in initial_ks:
        if k < n_points:
            overlap = sum([h[i] * (1 - h[i + k]) for i in range(n_points - k)])
            constraints.append(overlap * (2.0 / n_points) <= C)

    objective = Minimize(C)
    prob = Problem(objective, constraints)

    time_start = time.time()
    h_best = init_h.copy()
    c5_best = compute_c5(h_best, n_points)

    iteration = 0
    while time.time() - time_start < budget_s:
        iteration += 1
        try:
            prob.solve(solver='ECOS', verbose=False)
        except:
            break

        if prob.status != 'optimal':
            break

        h_current = np.clip(h.value, 0.0, 1.0)
        h_current = enforce_normalization(h_current, n_points)
        c5_current = compute_c5(h_current, n_points)
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current

        max_overlap = 0.0
        worst_k = -1
        for k in range(1, n_points):
            if k >= n_points:
                continue
            overlap = sum(h_current[i] * (1 - h_current[i + k]) for i in range(n_points - k))
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k

        if max_overlap > c5_best + 1e-5 and worst_k != -1:
            overlap_expr = sum([h[i] * (1 - h[i + worst_k]) for i in range(n_points - worst_k)])
            overlap_expr *= (2.0 / n_points)
            constraints.append(overlap_expr <= C)
            prob = Problem(objective, constraints)

        if iteration % 50 == 0:
            h_best = h_current.copy()
            c5_best = c5_current

        if time.time() - time_start + 20 > budget_s:
            break

    h_best = enforce_normalization(h_best, n_points)
    c5_best = compute_c5(h_best, n_points)
    return h_best, c5_best, n_points
