```python
import numpy as np
import time
from scipy.signal import correlate
from itertools import product
from bisect import bisect_left

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def discrete_values(h, levels=5):
    """Quantize values to levels at 0, 0.25, 0.5, 0.75, 1"""
    return np.round(h * (levels - 1)) / (levels - 1)


def generate_cellular_neighbor(h, radius=1):
    """Perturb a random segment"""
    n = len(h)
    center = np.random.randint(0, n)
    ops = []
    for i in range(center - radius, center + radius + 1):
        ops.append((i % n, np.random.random()))
    ops.sort()
    h = h.copy()
    idxs, new_vals = zip(*ops)
    h[idxs] = np.array(new_vals)
    return ensure_feasible(h)


def compute_bounds(h, free_mask):
    """Compute lower/upper bounds on C5 for partial assignment"""
    conv_result = correlate(h, 1 - h, mode='full')
    free_vals = np.where(free_mask, h, 0)
    free_conv = correlate(free_vals, 1 - free_vals, mode='full')
    total_conv = conv_result * (2.0 / len(h))

    return (np.min(total_conv), np.max(total_conv))


def get_var_impact(h):
    """How much a variable affects the result"""
    n = len(h)
    max_impact = 0
    best_idx = 0

    for i in range(n):
        # Test all possible values this variable could take
        max_possible = 0
        for v in [0, 0.25, 0.5, 0.75, 1]:
            h_test = h.copy()
            h_test[i] = v
            h_test = ensure_feasible(h_test)
            cand_C5 = evaluate_C5(h_test)
            max_possible = max(max_possible, cand_C5)

        if max_possible > max_impact:
            max_impact = max_possible
            best_idx = i

    return best_idx


def run(budget_s=1100, n_points=42, max_queue=500, branch_depth=3):
    start_time = time.time()
    best_C5 = np.inf
    best_h = None

    # Initialize with discrete versions of the parent if available
    if initial_h_values is not None:
        h = discrete_values(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0

    h = ensure_feasible(h)
    queue = [(h, 0)]  # (solution, depth)

    while time.time() - start_time < budget_s and queue:
        h, depth = queue.pop()
        curr_C5 = evaluate_C5(h)
        if curr_C5 < best_C5:
            best_C5 = curr_C5
            best_h = h

        # Prune by depth
        if depth >= branch_depth:
            continue

        # Compute free variables mask (can be fixed)
        free_mask = np.full(n_points, False)
        fixed_count = int((n_points * 0.1) * (1 - depth/branch_depth))  # fewer fixed as we go deep
        idxs = np.argsort(np.random.rand(n_points))[:fixed_count]
        free_mask[idxs] = True

        # Compute bounds
        lower, upper = compute_bounds(h, free_mask)
        if lower >= best_C5:  # No hope for improvement
            continue

        # Branch on highest-impact variable
        idx = get_var_impact(h)
        base_h = ensure_feasible(h.copy())

        for val in [0, 0.25, 0.5, 0.75, 1]:
            if time.time() - start_time >= budget_s:
                break
            new_h = base_h.copy()
            new_h[idx] = val
            new_h = ensure_feasible(new_h)

            # Add to queue based on potential
            candidate_C5 = evaluate_C5(new_h)
            if candidate_C5 < best_C5:
                queue.insert(0, (new_h, depth + 1))
                if len(queue) > max_queue:
                    queue.pop()

        # Also add a random perturbation with 30% chance
        if np.random.rand() < 0.3:
            perturbed_h = generate_cellular_neighbor(h)
            queue.insert(0, (perturbed_h, depth + 1))
            if len(queue) > max_queue:
                queue.pop()

    return best_h, best_C5, n_points
```
