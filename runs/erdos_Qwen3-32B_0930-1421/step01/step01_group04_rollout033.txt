import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k < 1:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k < 1:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs) * dx)

def objective_raw(h_vec, n_points):
    h = h_vec.reshape(-1)
    return compute_c5(h, n_points)

def enforce_normalization(h, n_points, clipping=True):
    h = np.clip(h, 0.0, 1.0) if clipping else h
    total = np.sum(h)
    if total == 0:
        h = np.full(n_points, 0.5)
    else:
        scale = (n_points / 2.0) / total
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    return h

def raised_cosine_init(n_points):
    i = np.arange(n_points)
    h = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
    h = np.clip(h, 0.0, 1.0)
    return enforce_normalization(h, n_points)

def particle_update(pos, vel, pbest, gbest, w, c1, c2):
    r1 = np.random.rand(*pos.shape)
    r2 = np.random.rand(*pos.shape)
    vel = (w * vel + c1 * r1 * (pbest - pos) + c2 * r2 * (gbest - pos))
    vel = np.clip(vel, -0.2, +0.2)
    pos = pos + vel
    pos = enforce_normalization(pos, len(pos))
    return pos, vel

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # default based on parent structure

    # Get initial_h_values if provided, else use cosine initialization
    if init_h is not None and len(init_h) == n_points:
        h0 = init_h.copy()
    else:
        h0 = raised_cosine_init(n_points)

    h0 = enforce_normalization(h0, n_points)

    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))

    # Initialize particles
    particles[0, :] = h0
    for i in range(1, n_particles):
        if i < 11:
            particles[i] = raised_cosine_init(n_points)
        else:
            particles[i] = np.random.rand(n_points)
        particles[i] = enforce_normalization(particles[i], n_points)
        velocities[i] = 0.01 * (np.random.rand(n_points) - 0.5)

    # Particle and global best tracking
    pbest_vals = np.array([compute_c5(p, n_points) for p in particles])
    pbests = particles.copy()
    gbest_idx = np.argmin(pbest_vals)
    gbest = particles[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    iter_count = 0
    budget_used = 0
    while time.time() - start_time < budget_s * 0.8:
        for i in range(n_particles):
            particles[i], velocities[i] = particle_update(particles[i], velocities[i],
                                                         pbests[i], gbest, w, c1, c2)
            val = compute_c5(particles[i], n_points)
            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbests[i] = particles[i].copy()
        gbest_idx = np.argmin(pbest_vals)
        gbest_val = pbest_vals[gbest_idx]
        gbest = particles[gbest_idx].copy()
        iter_count += 1
        w *= 0.9995

    # Local refinement using lbfgs
    for i in [0, gbest_idx]:
        res = opt.minimize(objective_raw, particles[i], args=(n_points,),
                          method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                          options={'ftol': 1e-5, 'eps': 1e-4, 'maxiter': 200})
        new_val = res.fun
        new_h = enforce_normalization(res.x, n_points)
        if new_val < gbest_val:
            gbest = new_h.copy()
            gbest_val = new_val

    # Final check and return
    gbest = enforce_normalization(gbest, n_points)
    final_val = compute_c5(gbest, n_points)
    return gbest, final_val, n_points
