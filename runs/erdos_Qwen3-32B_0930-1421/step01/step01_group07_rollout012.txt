import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, n_points):
    """Project h to feasible domain: 0 <= h <= 1, sum == n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    if abs(diff) < 1e-8:
        return h
    adjustment = diff / n_points
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def perturb_h(h, n_perturb=5):
    n_points = len(h)
    indices = np.random.choice(n_points, size=min(n_perturb, n_points), replace=False)
    h_perturbed = h.copy()
    sigma = 0.2
    h_perturbed[indices] += np.random.normal(0, sigma, size=len(indices))
    return project_to_feasible(h_perturbed, n_points)

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

    best_c5 = evaluate_overlap(h)
    T = 1.0
    alpha = 0.93
    min_temp = 1e-5
    t0 = time.time()
    max_iterations = 10000
    i = 0
    no_improvement = 0
    max_no_improvement = 500

    while time.time() - t0 < budget_s and i < max_iterations:
        h_new = perturb_h(h)
        c5_new = evaluate_overlap(h_new)
        delta = c5_new - best_c5

        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement = 0
            else:
                h = h_new.copy()
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            T = max(min_temp, T * 1.5)
            no_improvement = 0
        else:
            T = max(min_temp, T * alpha)

        i += 1

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Feasibility constraint violated"
    return (best_h, best_c5, n_points)
