import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Initialize with provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5 / n_points * n_points / 2)

    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)[0]

    def normalize_h(h_in):
        sum_h = np.sum(h_in)
        factor = (n_points / 2) / sum_h
        h_norm = np.clip(h_in * factor, 0, 1)
        return h_norm

    def compute_c5(h):
        fig = correlate(h, 1 - h, mode="full") * dx
        return np.max(fig)

    def convex_optimize_with_regularization(h_init):
        h = cp.Variable(n_points)
        objectives = []
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        obj = 0.0

        for k in range(-n_points + 1, n_points):
            overlap = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    overlap += h[i] * (1 - h[j])
            overlap *= dx
            objectives.append(overlap)

        max_overlap = cp.Max(cp.Constant(np.array(objectives)))
        obj += max_overlap

        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.CVXOPT)
        return h.value

    # Sequence of lambda values for regularization
    lambda_vals = [0.1, 0.05, 0.025, 0.0125, 0.00625]
    for lam in lambda_vals:
        obj = 0.0
        for k in range(-n_points + 1, n_points):
            overlap = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    overlap += h[i] * (1 - h[j])
            overlap *= dx
            obj += overlap

        obj = cp.Max(obj) + lam * cp.norm1(h)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.CVXOPT)

        if cp.math.success:
            candidate = np.clip(h.value, 0, 1)
            candidate = normalize_h(candidate)
            c5_candidate = compute_c5(candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()

        if time.time() - start_time > budget_s - 100:
            break

    # Post-optimization refinement if time permits
    learning_rate = 0.05
    eps = 1e-5
    while time.time() - start_time < budget_s:
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        best_c5 = compute_c5(best_h)

        learning_rate *= 0.995
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
