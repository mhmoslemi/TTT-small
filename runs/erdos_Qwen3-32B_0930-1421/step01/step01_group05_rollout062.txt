import numpy as np
import time
from scipy.signal import correlate

def normalize_h(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    scale = (n_points / 2) / current_sum
    return np.clip(h_clipped * scale, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h
    c = correlate(h, ones_minus_h, mode='full')
    return np.max(c) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)

    h = normalize_h(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    learning_rate = 0.1
    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        eps = 1e-5

        k_max = int(n_points / 2)
        overlaps = np.zeros(k_max * 2 + 1)

        for k in range(-k_max, k_max + 1):
            if k < 0:
                idx = (-k, n_points)
                overlap = np.sum(h[:idx[0]] * (1 - h[idx[1]:]))
            elif k > 0:
                idx = (k, n_points - k)
                overlap = np.sum(h[idx[0]:] * (1 - h[:n_points - idx[0]]))
            else:
                overlap = np.sum(h * (1 - h))
            overlaps[k + k_max] = overlap

        k_star = np.argmax(overlaps)
        k_actual = k_star - k_max

        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p, n_points)

            overlap_p = 0.0
            if k_actual < 0:
                idx = (-k_actual, n_points)
                overlap_p = np.sum(h_p[:idx[0]] * (1 - h_p[idx[1]:]))
            elif k_actual > 0:
                idx = (k_actual, n_points - k_actual)
                overlap_p = np.sum(h_p[idx[0]:] * (1 - h_p[:n_points - idx[0]]))
            else:
                overlap_p = np.sum(h_p * (1 - h_p))

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n, n_points)

            overlap_n = 0.0
            if k_actual < 0:
                idx = (-k_actual, n_points)
                overlap_n = np.sum(h_n[:idx[0]] * (1 - h_n[idx[1]:]))
            elif k_actual > 0:
                idx = (k_actual, n_points - k_actual)
                overlap_n = np.sum(h_n[idx[0]:] * (1 - h_n[:n_points - idx[0]]))
            else:
                overlap_n = np.sum(h_n * (1 - h_n))

            grad[i] = (overlap_p - overlap_n) / (2 * eps)

        h -= learning_rate * grad
        h = normalize_h(h, n_points)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= 0.999
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
