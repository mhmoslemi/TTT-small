```python
import numpy as np
import time
import heapq
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / h.sum()
    return h


def discretize_to_quantized(h, levels=5):
    """Quantizes h to levels (0, 0.25, 0.5, 0.75, 1.0)."""
    return np.round(h * levels) / levels


def generate_initial_h(n_points):
    if initial_h_values is not None:
        h = discretize_to_quantized(np.copy(initial_h_values), n_points=n_points)
        h = ensure_feasible(h)
        return h
    else:
        # Simple alternating pattern
        h = np.tile([1.0, 0.0], n_points // 2 + 1)
        h = ensure_feasible(h)
        h = discretize_to_quantized(h)
        return h


def compute_convolution_bounds(h, n_points):
    dx = 2.0 / n_points
    h = np.tile(h, 3)  # pad to support convolution via FFT

    computed = correlate(h, 1 - h, mode="valid")
    return np.max(computed) * dx


def branch_and_bound(n_points=42, max_depth=5, time_budget=1100):
    start_time = time.time()
    root_h = generate_initial_h(n_points)
    best_h = np.copy(root_h)
    best_bound = evaluate_C5(best_h)

    # Use a priority queue for branch and bound (low bound first)
    heap = []
    heapq.heappush(heap, (best_bound, 0, root_h.copy(), []))

    while not heap.empty() and (time.time() - start_time) < time_budget:
        bound, depth, state, changes = heapq.heappop(heap)
        if depth >= max_depth or bound >= best_bound:
            continue  # Prune for efficiency

        # Try changing variables at current depth
        for idx in [0, 1, 2, 10, 11, 12]:  # Pre_selected critical indices based on heuristics
            for value in np.linspace(0, 1, 5):
                new_h = state.copy()
                new_h[idx] = value
                new_h = ensure_feasible(new_h)

                lb = compute_convolution_bounds(new_h, n_points)
                if lb < best_bound:
                    new_changes = changes + [(idx, value)]
                    heapq.heappush(heap, (lb, depth + 1, new_h, new_changes))

                if lb < best_bound:
                    best_h[:] = new_h
                    best_bound = lb

    return best_h, best_bound, n_points


def run(budget_s=1100, n_points=42):
    h, c5_bound, n_points = branch_and_bound(n_points=n_points, time_budget=budget_s)
    return h, c5_bound, n_points
```
