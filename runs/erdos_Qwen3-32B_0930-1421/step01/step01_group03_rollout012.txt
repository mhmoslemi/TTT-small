import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    max_n_points = 100  # Cap to ensure small sequences
    population_size = 8
    num_generations = 20
    cross_mut_factor = 0.1

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=1000):
        h_arr = np.array(initial_h).copy()
        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            h = h * (constraint_sum / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(obj_func, h_arr, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = h_opt.sum()
        h_opt = h_opt * (constraint_sum / h_opt_sum) if h_opt_sum != 0 else h_opt
        return h_opt

    def perturb_h(h, noise_factor=0.15):
        h = h.copy()
        noise = np.random.randn(n_points) * noise_factor
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def crossover(h1, h2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, h1, h2)
        child = np.clip(child, 0, 1)
        child = child * (constraint_sum / child.sum())
        return child

    population = [h_init]
    for _ in range(population_size - 1):
        rand_h = np.random.uniform(0, 1, n_points)
        sum_rand_h = rand_h.sum()
        population.append(rand_h * (constraint_sum / sum_rand_h))

    best_h = h_init
    best_c5 = objective(best_h)
    population = [optimize_h(h) for h in population]
    population = sorted(population, key=objective)
    best_idx = 0

    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate new candidates
        next_gen = []
        for i in range(len(population)):
            parent1 = population[int(i / 2) % len(population)]
            parent2 = population[int(np.floor(i / 2)) % len(population)]
            child = crossover(parent1, parent2)
            child = optimize_h(child)
            next_gen.append(child)

        # Add mutated copies
        for h in population[:3]:
            child = perturb_h(h, noise_factor=0.15)
            child = optimize_h(child)
            next_gen.append(child)

        population = next_gen
        population = sorted(population, key=objective)
        best_h = population[0]
        best_c5 = objective(best_h)

    # Final gradient descent refinement
    best_h = optimize_h(best_h, max_iter=1000)
    best_h = np.clip(best_h, 0, 1)
    sum_h = best_h.sum()
    best_h = best_h * (constraint_sum / sum_h) if sum_h != 0 else best_h

    if best_c5 > objective(best_h):
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
