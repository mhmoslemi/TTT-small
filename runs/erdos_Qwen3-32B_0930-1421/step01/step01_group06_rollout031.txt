import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n / 2
    total_h = float(np.sum(initial_h))
    if np.isclose(total_h, 0.0):
        raise ValueError("Initial h has zero sum; invalid configuration.")
    h = initial_h * (n_points / 2.0) / total_h
    h = np.clip(h, 0.0, 1.0)

    def project(h):
        # Clip to valid range
        h = np.clip(h, 0.0, 1.0)
        # Normalize to satisfy mass constraint
        s = np.sum(h)
        h *= (n_points / 2.0) / s
        return h

    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    particles = np.zeros((num_particles, n_points))
    velocities = np.random.uniform(-0.01, 0.01, (num_particles, n_points))
    particles[:int(num_particles * 0.1), :] = np.tile(h, (int(num_particles * 0.1), 1))
    for idx in range(int(num_particles * 0.1), num_particles):
        p = h + np.random.normal(0.0, 0.1, n_points)
        p = project(p)
        particles[idx, :] = p

    p_best = np.copy(particles)
    p_best_score = np.array([evaluate_erdos_solution(p, n_points) for p in p_best])
    best_idx = np.argmin(p_best_score)
    g_best = np.copy(p_best[best_idx])
    best_c5 = p_best_score[best_idx]
    best_h = np.copy(g_best)
    last_improvement_time = start_time

    while time.time() - start_time < budget_s - 5:
        t = time.time()
        w = w_start - (w_start - w_end) * (t - start_time) / budget_s
        w = max(w, w_end)

        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)

            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best[i] - particles[i]) +
                c2 * r2 * (g_best - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] = particles[i] + velocities[i]
            particles[i] = project(particles[i])

        scores = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
        improved = scores < p_best_score
        p_best[improved] = particles[improved]
        p_best_score[improved] = scores[improved]
        new_best_idx = np.argmin(p_best_score)
        if p_best_score[new_best_idx] < best_c5:
            best_c5 = p_best_score[new_best_idx]
            best_h = particles[new_best_idx].copy()
            last_improvement_time = time.time()

        # Reset a fraction of particles to random and reproject
        reset_count = int(num_particles * reset_fraction)
        if reset_count > 0:
            reset_ids = np.random.choice(num_particles, reset_count, replace=False)
            for idx in reset_ids:
                p = np.random.uniform(0.0, 1.0, n_points)
                p = project(p)
                particles[idx] = p

        # Early stopping if no improvement recently
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
