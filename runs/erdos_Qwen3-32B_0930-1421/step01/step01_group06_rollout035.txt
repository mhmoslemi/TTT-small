```python
import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Start from the given initial_h_values
    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to constraint: sum(h) = n_points / 2
    h = initial_h * (n_points / 2.0 / np.sum(initial_h))
    h = np.clip(h, 0.0, 1.0)
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement = start_time

    def project_and_normalize(x):
        x = np.clip(x, 0.0, 1.0)
        scale = (n_points / 2.0) / np.clip(np.sum(x), 1e-9, 1e9)
        return x * scale

    def optimize_with_cvxpy():
        nonlocal h, best_c5, best_h, last_improvement
        try:
            H = cp.Variable(n_points)
            dx = 2.0 / n_points
            equal_sum = cp.Sum(H) == n_points / 2
            bounds = 0 <= H, H <= 1

            max_corr = None
            for k_off in range(-n_points + 1, n_points):
                ixs = np.arange(n_points)
                shifted = np.roll(ixs, k_off)
                bool_ixs = (shifted >= 0) & (shifted < n_points)
                ixs = ixs[bool_ixs]
                shifted = shifted[bool_ixs]
                if len(ixs) == 0:
                    continue
                corr_k = cp.sum(H[ixs] * (1 - H[shifted])) * dx
                if max_corr is None:
                    max_corr = corr_k
                else:
                    max_corr = cp.Max(max_corr, corr_k)

            # Formulate optimization with maximization objective of max_corr
            objective = cp.Minimize(max_corr)
            constraints = [equal_sum, bounds]
            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver=cp.ECOS, max_iter=500, feastol=1e-6)
            if result < best_c5:
                h = H.value.copy()
                h = project_and_normalize(h)
                best_c5 = evaluate_erdos_solution(h, n_points)
                best_h = np.copy(h)
                last_improvement = time.time()
        except Exception:
            pass

    def optimize_with_slsqp(max_iter=100):
        nonlocal h, best_c5, best_h, last_improvement

        def objective(x):
            return evaluate_erdos_solution(x, n_points)

        def constrain_sum(x):
            return np.sum(x) - n_points / 2.0

        cons = [
            {"type": "eq", "fun": constrain_sum}
        ]
        bnds = [(0.0, 1.0)] * n_points
        result = minimize(objective, h, bounds=bnds, constraints=cons, method="SLSQP", options={"maxiter": max_iter})
        if result.success:
            new_h = project_and_normalize(result.x)
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                best_h = np.copy(h)
                last_improvement = time.time()

    def perturb_h():
        nonlocal h
        noise = np.random.randn(n_points) * 0.05
        h = project_and_normalize(h + noise)

    while time.time() - start_time < budget_s:
        if time.time() - last_improvement > 120:
            perturb_h()
        if np.random.rand() < 0.5:
            optimize_with_cvxpy()
        else:
            optimize_with_slsqp()
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
```
