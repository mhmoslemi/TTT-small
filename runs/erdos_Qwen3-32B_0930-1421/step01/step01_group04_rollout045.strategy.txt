def column_generation(prob, h, C, n_points, initial_h, budget_s):

        if prob.status != 'optimal':
            break
        
        # Update best solution

            C_best = C_val
            h_best = h_val.copy()
        
        # Identify most violating shift
        max_overlap = 0.0
        worst_k = 0
        for k in range(1, n_points):

    return h_best, C_best
