def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    start_time = time.time()
    T = 1.0  # Initial Temperature
    T_min = 1e-6
    alpha = 0.99  # cooling rate
    steps_per_T = 100
    n_iter = 0

    while T > T_min and time.time() - start_time < budget_s:
        for _ in range(steps_per_T):
            if time.time() - start_time >= budget_s:
                break
            n_iter += 1

            # Generate candidate by perturbing a random segment
            start_seg = np.random.randint(0, n_points)
            end_seg = np.random.randint(start_seg, min(start_seg + 30, n_points))
            shift = np.random.normal(0, 0.1, end_seg - start_seg)
            h_candidate = h.copy()
            h_candidate[start_seg:end_seg] += shift
            h_candidate = rescale_h(h_candidate)

            # Compute C5 for current and candidate
            corr_best = correlate(best_h, 1 - best_h, mode='full')
            corr_new = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_best = np.max(corr_best) * dx
            c5_new = np.max(corr_new) * dx
            delta = c5_new - c5_best

            # Acceptance criterion
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = h_candidate.copy()
                best_c5 = min(best_c5, c5_new)

            # Update h for next step
            h = h_candidate

        # Cool down
        T *= alpha

    return best_h, best_c5, n_points
