import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = time.time()

    # CVXPY optimization
    max_solves = 100
    solve_count = 0
    perturbations = 10

    h_var = cp.Variable(n_points)
    C = cp.Parameter(nonneg=True)

    def feasible_check(h_val, C_val, dx, n_points):
        h_var.value = h_val
        C.value = C_val
        objective = cp.Minimize(0)

        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            0 <= h_var,
            h_var <= 1,
        ]

        # Introduce separate linear constraints for every offset
        for k in range(1, n_points):
            offset_corr = np.correlate(h_var, 1 - h_var, mode='valid')[k-1]
            constraints += [offset_corr * dx <= C]

        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.GUROBI, verbose=False)
        return True if result == 0 and prob.status == 'optimal' else False

    # Binary search
    def binary_search(h_init, c_high):
        c_low = 0.0
        c_high_term = c_high
        best_h_bs = h_init.copy()
        best_c5_bs = c_high_term

        while time.time() - start_time < budget_s - 20 and c_high_term - c_low > 1e-5:
            mid = (c_low + c_high_term) / 2
            feasible = feasible_check(h=h_init, C_val=mid, dx=dx, n_points=n_points)
            if feasible:
                if mid < best_c5_bs:
                    best_c5_bs = mid
                    best_h_bs[:] = h_init
                    nonlocal last_improvement_time
                    last_improvement_time = time.time()
                c_high_term = mid
            else:
                c_low = mid
        return best_h_bs, best_c5_bs

    # First pass
    h = np.copy(best_h)
    h_opt, c_opt = binary_search(h, best_c5)
    if c_opt < best_c5:
        best_h = h_opt
        best_c5 = c_opt

    # Perturbations
    for i in range(perturbations):
        h_perturbed = best_h.copy()
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed *= (n_points / 2.0 / np.sum(h_perturbed))

        h_opt, c_opt = binary_search(h_perturbed, best_c5)
        if c_opt < best_c5:
            best_h = h_opt
            best_c5 = c_opt
            last_improvement_time = time.time()

        if time.time() - start_time >= budget_s:
            break

    # Final clipping and renormalizing
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h *= (n_points / 2.0 / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h, n_points)

    return best_h, final_c5, n_points
