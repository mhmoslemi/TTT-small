import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = np.zeros(2 * n_points)
    h_full = np.tile(h, 4)
    for k in range(1, n_points):

        rhs = 1 - h[k:k + n_points - k]
        corrs[k] = np.dot(lhs, rhs)
    for k in range(1, n_points):
        lhs = h[k:]
        rhs = 1 - h[:n_points - k]
        corrs[n_points + k] = np.dot(lhs, rhs)
    return np.max(corrs) * dx
