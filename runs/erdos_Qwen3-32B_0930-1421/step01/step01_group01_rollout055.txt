import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt

def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h

def is_feasible_h(h, dx, tolerance=1e-6):
    integral = h.sum() * dx
    return abs(integral - 1) < tolerance and np.all((h >= 0) & (h <= 1))

def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points
    start_time = time.time()

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    except:
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    best_C5 = np.inf
    best_h = h.copy()
    perturb_period = 30
    optim_steps = 0
    h = np.clip(h, 0, 1)

    def cvx_optimize_for_k(shift_k, curr_h_vec):
        h_vec = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h_vec) * dx == 1, h_vec <= 1]
        objective = cp.Minimize(
            cp.sum(
                h_vec * (1 - cp.roll(h_vec, shift=shift_k))
            ) * dx
        )
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CPLEX, verbose=False)
        return h_vec.value

    while time.time() - start_time < budget_s:
        h_current = h.copy()
        h = np.clip(h, 0, 1)
        cor = correlate(h, 1 - h, mode="full")
        max_id = np.argmax(cor)
        max_k = max_id - len(h) + 1

        h_candidate = cvx_optimize_for_k(max_k, h)
        h_candidate = np.clip(h_candidate, 0, 1)
        if h_candidate is not None:
            h_candidate = h_candidate / h_candidate.sum() * n_points / 2
            if is_feasible_h(h_candidate, dx):
                h[:] = h_candidate
            else:
                h[:] = h_current

        C5, h = evaluate_C5(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            print(f"New best C5: {best_C5:.6f}")

        if optim_steps % perturb_period == 0:
            h[:] = h * 0.9 + 0.1 * np.random.rand(n_points)
            h = rescale_h(h)

        optim_steps += 1

    return best_h, best_C5, n_points

def rescale_h(h):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * len(h) / 2
    return h
