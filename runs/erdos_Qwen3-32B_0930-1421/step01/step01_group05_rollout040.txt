import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Set up initial configuration
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    n_points = len(h)
    dx = 2.0 / n_points

    # Constraint enforcement utility
    def normalize_h(h_in):
        h_clipped = np.clip(h_in, 0.0, 1.0)
        total = np.sum(h_clipped)
        if abs(total) < 1e-10:
            return np.zeros_like(h_in)
        return h_clipped * (n_points / 2.0) / total

    # Objective function (C₅)
    def compute_c5(h):
        corr = correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Firefly algorithm parameters
    pop_size = 50
    population = [np.random.rand(n_points) for _ in range(pop_size - 1)] + [h.copy()]
    population = [normalize_h(p) for p in population]
    current_c5 = [compute_c5(p) for p in population]

    def firefly_attraction(i, j, gamma):
        distance = np.sqrt(np.sum((population[i] - population[j]) ** 2))
        beta0 = 1.0
        beta = beta0 * np.exp(-gamma * distance**2)
        return beta

    # Main optimization loop
    while time.time() - start_time < budget_s:
        gamma = (time.time() - start_time) / (0.1 * budget_s) + 1e-6
        new_population = []

        for i in range(pop_size):
            new_firefly = population[i].copy()
            alpha = 0.2

            for j in range(pop_size):
                if current_c5[j] < current_c5[i]:  # Move toward brighter fireflies
                    beta = firefly_attraction(i, j, gamma)
                    movement = beta * (population[j] - population[i])
                    new_firefly = new_firefly + movement + alpha * (np.random.rand(n_points) - 0.5)
                    new_firefly = normalize_h(new_firefly)
                    new_c5 = compute_c5(new_firefly)
                    if new_c5 < current_c5[i]:
                        new_population.append(new_firefly)
                        break
            else:
                new_population.append(normalize_h(population[i] + 0.1 * (np.random.rand(n_points) - 0.5)))

            # Always consider clipping and normalization
            new_population[-1] = normalize_h(new_population[-1])

        # Update population if we're not out of time budget
        if time.time() - start_time > budget_s:
            break

        # Evaluate and update
        population = new_population
        current_c5 = [compute_c5(p) for p in population]
        current_idx = np.argmin(current_c5)
        if current_c5[current_idx] < best_c5:
            best_c5 = current_c5[current_idx]
            best_h = population[current_idx].copy()

    return best_h.tolist(), best_c5, n_points
