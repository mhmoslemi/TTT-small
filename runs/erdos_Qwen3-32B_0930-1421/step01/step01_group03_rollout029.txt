        h = h * (n_points / 2 / h.sum())

                h_perturbed = h_perturbed.clip(0, 1)
                h_perturbed = h_perturbed * (n_points / 2 / h_perturbed.sum())
                pop.append(h_perturbed)

                h = create_random_h(n_points)
                pop.append(h)
        return pop

    dx = 2.0 / n_points
    pop = create_initial_population()

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (n_points / 2 / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

        fitness = [(objective(h), h.copy()) for h in pop]

        current_c5, current_h = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # Elitism
        num_elite = int(max_population * elitism_ratio)
        elite = [f[1].copy() for f in fitness[:num_elite]]

            # Crossover
            p1 = random.choice(elite)
            p2 = random.choice(elite)
            child = (p1 * crossover_rate + p2 * (1 - crossover_rate)).copy()
            # Mutation
            mut = np.random.randn(n_points) * mutation_strength
            child += mut
            child = repair(child)
            new_pop.append(child)

        pop = new_pop
        generation += 1

            final_h = repair(final_h)
            final_c5 = objective(final_h)
