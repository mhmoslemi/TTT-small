import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 47
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            h_idx = i + i
            if h_idx < n_points:
                initial_h[h_idx] = 2.0 / n_points
        # Normalize
        initial_h *= (n_points / 2) / initial_h.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def compute_c5(h):
        h = h.clip(0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return np.inf
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)

    def normalize_h(h):
        h = h.clip(0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return h
        return h * (constraint_sum / h_sum)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)

    def perturb(h, noise_scale=0.1, pos_shift=3):
        h = h.copy()
        noise = np.random.randn(n_points) * noise_scale
        h += noise
        h = h.clip(0, 1)
        h = normalize_h(h)

        shift = np.random.randint(-pos_shift, pos_shift + 1)
        h = np.roll(h, shift)
        h = h.copy()
        h = normalize_h(h)
        return h

    def crossover(h1, h2, alpha=0.5):
        h_new = alpha * h1 + (1 - alpha) * h2
        h_new = h_new.clip(0, 1)
        h_new = normalize_h(h_new)
        return h_new

    def gradient_descent(h):
        bounds = [(0, 1) for _ in range(n_points)]
        bound_h = h.clip(0, 1)

        def obj_func(x):
            h_t = x.reshape(n_points)
            return compute_c5(h_t)

        result = minimize(obj_func, bound_h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
        optimized_h = result.x.reshape(n_points)
        optimized_h = optimized_h.clip(0, 1)
        optimized_h = normalize_h(optimized_h)
        return optimized_h

    # Initialize population
    POPULATION_SIZE = 12
    POPULATION = [initial_h]
    for _ in range(POPULATION_SIZE - 1):
        if 'initial_h_values' in globals():
            h = perturb(initial_h, noise_scale=0.2, pos_shift=2)
        else:
            h = np.random.rand(n_points)
        h = normalize_h(h)
        POPULATION.append(h)
    POPULATION = [p.copy() for p in POPULATION]

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    last_improvement_time = 0

    while time.time() - start_time < budget_s * 0.95:
        # Evaluate all population members
        fitness = [(compute_c5(h), h) for h in POPULATION]
        fitness.sort()  # Best is first

        new_population = [h for C5, h in fitness[:POPULATION_SIZE // 2]]
        while len(new_population) < POPULATION_SIZE:
            parent1, parent2 = np.random.choice(fitness[:POPULATION_SIZE // 2], 2, replace=False)
            child = crossover(parent1[1], parent2[1])
            child = gradient_descent(child)
            if np.random.rand() < 0.2:  # Mutation chance
                child = perturb(child)
            new_population.append(child)

        POPULATION = new_population
        current_best = min(fitness, key=lambda x: x[0])

        if current_best[0] < best_c5:
            best_h = current_best[1]
            best_c5 = current_best[0]
            last_improvement_time = time.time()

        # Early stopping if no improvement
        if time.time() - last_improvement_time > budget_s * 0.1 and len(POPULATION) > 1:
            break

    # Final gradient refinement
    if time.time() - start_time < budget_s:
        h = gradient_descent(best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_points
