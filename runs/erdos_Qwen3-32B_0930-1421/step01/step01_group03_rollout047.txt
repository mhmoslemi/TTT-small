import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # Cap to ensure small sequences

    global initial_h_values
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_base = np.zeros(n_points)
        for i in range(n_points // 2):
            h_base[i + i] = 2.0 / n_points
        h_base *= n_points / 2 / h_base.sum()  # Normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = np.clip(h.reshape(n_points), 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        return h * (constraint_sum / h_sum)

    def perturb(h, noise_factor=0.05, repair=True):
        h = h.copy()
        noise = np.random.randn(len(h)) * noise_factor
        h = h + noise
        if repair:
            h = repair(h)
        return h

    def crossover(parent1, parent2, alpha=0.5, repair=True):
        child = alpha * parent1 + (1 - alpha) * parent2
        if repair:
            child = repair(child)
        return child

    def gradient_optimize(h_flat, max_iter=800, repair=True):
        h = h_flat.reshape(n_points)
        h = repair(h) if repair else h.copy()

        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1).copy()
            h_sum = h.sum()
            h = h * (constraint_sum / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_res = result.x.reshape(n_points)
        return repair(h_res)

    # Population-based initialization
    population = []
    for _ in range(5):
        if 'initial_h_values' in globals():
            hp = perturb(h_base, noise_factor=0.3)
        else:
            hp = np.clip(np.random.rand(n_points), 0, 1)
            hp = hp * (constraint_sum / hp.sum())
        population.append(hp)

    best_h = min(population, key=objective)
    best_c5 = objective(best_h)

    gen = 0
    while time.time() - start_time < budget_s * 0.97:
        gen += 1

        scores = [objective(h) for h in population]
        ranked = [h for _, h in sorted(zip(scores, population))]
        survivors = ranked[:2]

        new_candidates = []
        for i in range(len(survivors)):
            for j in range(i+1, len(survivors)):
                child = crossover(survivors[i], survivors[j])
                new_candidates.append(gradient_optimize(child, max_iter=400))

        for _ in range(3):
            h = perturb(survivors[np.random.choice(2)], noise_factor=0.1)
            h = gradient_optimize(h, max_iter=400)
            new_candidates.append(h)

        population = survivors + new_candidates
        for h in population:
            if objective(h) < best_c5:
                best_c5 = objective(h)
                best_h = h

        if gen % 5 == 0:
            h = gradient_optimize(best_h, max_iter=800, repair=False)
            best_h = repair(h)
            best_c5 = objective(best_h)

    best_h = repair(best_h)

    return best_h, best_c5, n_points
