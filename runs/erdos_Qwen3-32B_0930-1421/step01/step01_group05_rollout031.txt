import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = clipped.sum()
        scale = (n_points / 2) / sum_h
        return np.clip(clipped * scale, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Gradient descent with warm start
    learning_rate = 0.1
    patience = 30
    no_improvement = 0
    while time.time() - start_time < budget_s:
        # Compute overlaps for all shifts k
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        c5_value = max_overlap * dx

        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= (i - k_max) < n_points:
                grad[i] += (1 - h[i - k_max])
            if 0 <= (i + k_max) < n_points:
                grad[i] -= h[i + k_max]

        # Update using gradient descent
        h_proposal = h - learning_rate * grad
        h = normalize_h(h_proposal)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Reduce learning rate if no improvement
        if no_improvement > patience:
            learning_rate *= 0.5
            no_improvement = 0

        # Adaptive early stopping
        if current_c5 < 0.382:
            break

        if time.time() - start_time >= budget_s:
            break

    # Final validation
    assert (0 <= best_h).all() and (best_h <= 1).all(), "h out of bounds"
    assert abs(best_h.sum() - n_points / 2) < 1e-8, "Constraint violated"

    return best_h.tolist(), best_c5, n_points
