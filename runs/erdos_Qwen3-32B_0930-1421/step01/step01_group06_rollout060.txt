import numpy as np
import time
from scipy.optimize import OptimizeResult
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def project_to_feasible_space(x, n_points):
    x_clipped = np.clip(x, 0.0, 1.0)
    total = np.sum(x_clipped)
    factor = (n_points / 2.0) / total
    return x_clipped * factor

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize initial h
    h_start = project_to_feasible_space(initial_h, n_points)

    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1
    max_iterations = 500

    # Initialize particles and velocities
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    pbest = np.zeros((num_particles, n_points))
    pbest_c5 = np.full(num_particles, float('inf'))

    # 10% from initial_h, 90% from random perturbations
    particles[0] = h_start
    for i in range(1, num_particles):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        particles[i] = project_to_feasible_space(particles[0] + perturbation, n_points)

    for i in range(num_particles):
        velocities[i] = np.random.uniform(-max_velocity / 10, max_velocity / 10, size=n_points)
        particles[i] = project_to_feasible_space(particles[i], n_points)
        c5 = evaluate_erdos_solution(particles[i], n_points)
        pbest[i] = particles[i].copy()
        pbest_c5[i] = c5

    gbest = pbest[pbest_c5.argmin()].copy()
    gbest_c5 = pbest_c5.min()

    iter_data = {"gbest_c5": [gbest_c5], "time": [time.time() - start_time]}

    while time.time() - start_time < budget_s and len(iter_data["time"]) < max_iterations:
        t = len(iter_data["time"])
        w = w_start - (w_start - w_end) * t / max_iterations

        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (pbest[i] - particles[i])
                + c2 * r2 * (gbest - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            particles[i] += velocities[i]
            particles[i] = project_to_feasible_space(particles[i], n_points)

            c5 = evaluate_erdos_solution(particles[i], n_points)

            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5

            if c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = c5
                iter_data["gbest_c5"].append(gbest_c5)
                iter_data["time"].append(time.time() - start_time)

        # Fractional reset
        reset_indices = np.random.choice(num_particles, size=int(num_particles * reset_fraction), replace=False)
        for idx in reset_indices:
            base = h_start
            particles[idx] = project_to_feasible_space(base + np.random.uniform(-0.1, 0.1, n_points), n_points)
            c5 = evaluate_erdos_solution(particles[idx], n_points)
            pbest[idx] = particles[idx].copy()
            pbest_c5[idx] = c5
            if c5 < gbest_c5:
                gbest = particles[idx].copy()
                gbest_c5 = c5

    return gbest, gbest_c5, n_points
