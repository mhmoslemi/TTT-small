import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def find_worst_shift(h):
    n = len(h)
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    worst_idx = np.argmax(np.abs(corr))
    return worst_idx

def flip_two_bits_with_target(h, worst_idx):
    n = len(h)
    # Construct the shift version of h
    offset = n
    if worst_idx >= n:
        h_shift = np.roll(h, -(worst_idx - n + 1))
        shift_amount = worst_idx - n + 1
    else:
        h_shift = np.roll(h, worst_idx)
        shift_amount = worst_idx

    h1_shifted = 1 - h_shift
    contrib = h * h1_shifted
    indices_selected_1 = np.where((h == 1) & (contrib > np.median(contrib[contrib > 0]))[0]
    indices_selected_0 = np.where((h == 0) & (h_shift == 1))[0]

    if len(indices_selected_1) == 0 or len(indices_selected_0) == 0:
        indices_selected_1 = np.where(h == 1)[0]
        indices_selected_0 = np.where(h == 0)[0]

    i = np.random.choice(indices_selected_1)
    j = np.random.choice(indices_selected_0)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new
