import numpy as np
import time
import scipy.signal as signal


def evaluate_C5(h, dx):
    """Compute C₅ using correlation and return its max overlap value."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    cor = scipy.signal.correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx, h


def rescale_h(h):
    """Ensure h is valid and satisfies ∫h dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h
