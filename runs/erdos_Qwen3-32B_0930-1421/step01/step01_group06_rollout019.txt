import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to the constraint
    total_h = float(np.sum(h))
    if abs(total_h - n_points / 2.0) > 1e-6:
        scale_factor = (n_points / 2.0) / total_h
        h = np.minimum(h * scale_factor, 1.0)
    h = np.clip(h, 0.0, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    def objective(x):
        h = x.reshape(n_points)
        return evaluate_erdos_solution(h, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def optimize_with_cvxpy(epsilon, max_iter=1000, tolerance=1e-5):
        nonlocal h, best_c5, best_h, last_improvement_time

        h_var = cp.Variable(n_points)
        h_ref = best_h
        for i in range(max_iter):
            correlation = np.correlate(h_ref, 1 - h_ref, mode='full')
            max_corr = np.max(correlation)

            # Formulate the problem
            constraints = [
                cp.sum(h_var) == n_points / 2.0,
                0 <= h_var,
                h_var <= 1.0,
                np.correlate(h_var.value, 1 - h_ref, mode='full') <= (1 + epsilon) * max_corr
            ]
            objective = cp.Minimize(cp.sum(cp.abs(h_var - h_ref)))

            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.CDLP, verbose=False)

            if problem.status != cp.OPTIMAL:
                break

            h_candidate = np.array(h_var.value).flatten()
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            candidate_c5 = evaluate_erdos_solution(h_candidate, n_points)

            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h[:] = h_candidate.copy()
                last_improvement_time = time.time()

            h_ref = h_candidate.copy()
            if best_c5 < tolerance:
                break

    try:
        diminishing_epsilon = 0.5  # Initial constraint tolerance
        while time.time() - start_time < budget_s - 10:  # margin
            optimize_with_cvxpy(epsilon=diminishing_epsilon, max_iter=100)
            diminishing_epsilon *= 0.95  # gradual tightening
            if time.time() - last_improvement_time > 60:  # no improvement for a minute
                break
    except Exception:
        pass

    return best_h, best_c5, n_points
