import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points

    target_mean = n_points / 2

    def objective(h):
        h = h.reshape((n_points,))
        # Clamp to [0, 1] and normalize
        h_clamped = np.clip(h, 0, 1)
        h_normalized = (h_clamped * target_mean / h_clamped.sum())
        # Correlation-based overlap
        overlaps = np.correlate(h_normalized, 1 - h_normalized, mode='full') * dx
        return np.max(overlaps)

    bounds = [(0, 1)] * n_points
    method = 'L-BFGS-B'

    tabu_memory = []

    def perturb_h(h_flat, noise_factor=0.15):
        h_new = h_flat.copy()
        idx = np.random.randint(n_points)
        delta = np.random.uniform(-noise_factor, noise_factor)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        return h_new

    def is_unique_in_tabu(h_flat, tabu_memory):
        for h in tabu_memory:
            if np.allclose(h_flat, h, atol=1e-4):
                return False
        return True

    def feasible_repair(h_flat):
        h = h_flat.copy()
        h_clamped = np.clip(h, 0, 1)
        h_repaired = (h_clamped * target_mean / h_clamped.sum())
        return h_repaired

    best_h = feasible_repair(h_init)
    best_c5 = objective(best_h)

    # Gradient descent from initial
    res = minimize(objective, best_h, method=method, bounds=bounds)
    h_opt = feasible_repair(res.x)
    c5 = objective(h_opt)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5

    tabu_length = 10
    perturbation_delta = 0.15
    no_improvement_threshold = 50
    cool_factor = 0.8
    iterations = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s:
        new_h = perturb_h(best_h)
        new_h = feasible_repair(new_h)
        if not is_unique_in_tabu(new_h, tabu_memory):
            continue

        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        tabu_memory.append(new_h)
        if len(tabu_memory) > tabu_length:
            tabu_memory.pop(0)

        iterations += 1
        if no_improvement_count >= no_improvement_threshold:
            perturbation_delta *= cool_factor
            no_improvement_count = 0

        if perturbation_delta < 0.001:
            break

    best_h = feasible_repair(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
