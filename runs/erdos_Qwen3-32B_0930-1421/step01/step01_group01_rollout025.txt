import numpy as np
import time
import math
import scipy.optimize as opt

def evaluate_C5(h, dx):
    cor = np.correlate(h, 1.0 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5

def rescale_h(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total < 1e-10:
        h[:] = 0.0
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h

def perturb_h(h, alpha, perturb_scale=0.1):
    n_points = len(h)
    i = np.random.randint(0, n_points)
    width = np.random.randint(1, min(4, n_points - i))
    segment = h[i:i+width]
    segment += (np.random.rand(width) - 0.5) * perturb_scale
    return rescale_h(h, n_points)

def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = np.resize(h, n_points)
    except:
        h = np.random.rand(n_points)
    h = rescale_h(h, n_points)

    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    tries = 0
    T = 10.0
    alpha = 0.995
    T_min = 1e-6
    t0 = time.time()
    perturb_scale = 0.2
    patience = 0

    while time.time() - t0 < budget_s and T > T_min:
        h_candidate = perturb_h(h.copy(), perturb_scale)
        C5 = evaluate_C5(h_candidate, dx)
        delta = C5 - best_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
            if delta < 0:
                best_h = h.copy()
                best_C5 = C5
                patience = 0
        else:
            patience += 1

        if patience >= 100:
            perturb_scale = min(0.4, perturb_scale * 1.1)
            patience = 0
        else:
            perturb_scale = max(0.05, perturb_scale * 0.95)

        if tries % 200 == 0:
            current_time = time.time() - t0
            remaining = budget_s - current_time
            if remaining < 30:
                alpha = 0.9995
            if remaining < 5:
                break

        T *= alpha
        tries += 1

    return best_h, best_C5, n_points
