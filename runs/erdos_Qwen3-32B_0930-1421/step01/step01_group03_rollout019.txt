import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100
    n_points = 47 if not 'initial_h_values' in globals() else len(initial_h_values)
    h_init = np.array(initial_h_values, copy=True) if 'initial_h_values' in globals() else np.zeros(n_points)

    dx = 2.0 / n_points
    constraint_sum = n_points / 2
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (constraint_sum / h_init.sum())

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=500):
        h = initial_h.copy()
        bounds = [(0, 1)] * n_points

        def obj_func(x):
            x = x.reshape(n_points)
            x = np.clip(x, 0, 1)
            x_sum = x.sum()
            x = x * (constraint_sum / x_sum)
            overlaps = np.correlate(x, 1 - x, mode='full') * dx
            return np.max(overlaps)

        result = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        noise = np.random.randn(n_points) * noise_factor
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def best_neighbor(h, step_size=0.01):
        h = h.copy()
        best_h = h.copy()
        best_val = objective(best_h)
        for i in np.random.permutation(n_points):
            for dh in [-step_size, step_size]:
                h_pert = h.copy()
                h_pert[i] += dh
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (constraint_sum / h_pert.sum())
                val = objective(h_pert)
                if val < best_val:
                    best_h = h_pert
                    best_val = val
        return best_h, best_val

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    # Local optimization
    h_opt = optimize_h(h_init)
    c5 = objective(h_opt)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5

    tabu_list = [tuple(h_opt)]
    max_iter = 25
    tabu_count = 5
    patience = 0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
        h_candidate, c5 = best_neighbor(best_h)
        h_tuple = tuple(h_candidate)
        if h_tuple in tabu_list:
            patience += 1
            continue
        tabu_list.append(h_tuple)
        if len(tabu_list) > tabu_count:
            tabu_list.pop(0)
        if c5 < best_c5:
            best_h = h_candidate
            best_c5 = c5
            patience = 0
        else:
            patience += 1
        if patience > 10:
            break

    # Refinement
    h_refined = optimize_h(best_h)
    c5_refined = objective(h_refined)
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())
    return best_h, best_c5, n_points
