import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize h and project to [0,1]
    h = np.clip(h, 0, 1)
    original_sum = h.sum()
    if not np.isclose(original_sum, target_sum):
        h *= target_sum / original_sum
    h = np.clip(h, 0, 1)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            h *= target_sum / current_sum
        h = np.clip(h, 0, 1)
        h1m = 1 - h
        max_overlap = -np.inf
        for k in range(1, n_points):
            if k >= h.shape[0] - 1:
                continue
            corr = np.dot(h[:n_points - k], h1m[k:])
            max_overlap = max(max_overlap, corr * dx)
            corr = np.dot(h[k:], h1m[:n_points - k])
            max_overlap = max(max_overlap, corr * dx)
        return max_overlap

    def adjust(h, top_shifts, n_steps_adapt=20):
        h = h.copy()
        for _ in range(n_steps_adapt):
            # Normalize
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if not np.isclose(h_sum, target_sum):
                h *= target_sum / h_sum
            h = np.clip(h, 0, 1)

            max_shifts = []
            for k in top_shifts:
                if k >= h.shape[0] - 1:
                    continue
                # Forward overlap
                overlap = np.dot(h[:n_points - k], (1 - h[k:]))
                # Backward overlap
                overlap += np.dot(h[k:], (1 - h[:n_points - k]))
                max_shifts.append((overlap, k))
            if not max_shifts:
                continue
            max_shifts.sort(reverse=True)
            k = max_shifts[0][1]

            idxs = np.argsort(-h[k:])[:int(np.ceil(0.10 * len(h[k:])))]
            idxs = np.unique(idxs + k)
            for i in idxs:
                if h[i] > 0.1:
                    h[i] -= 0.1 * h[i]
                if i >= k and i < n_points:
                    i_rev = i - k
                    if h[i_rev] < 0.9:
                        h[i_rev] += 0.1 * (1 - h[i_rev])
        return h

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    last_improvement = start_time

    top_shifts = []

    while time.time() - start_time < budget_s:
        h_prev = h.copy()

        # Evaluate current position
        current_c5 = evaluate(h)

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = time.time()

        if time.time() - last_improvement > 180:
            break

        if current_c5 < 0.381:
            break

        # Update top_shifts with top 5
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if not np.isclose(h_sum, target_sum):
            h_clipped *= target_sum / h_sum
        h_clipped = np.clip(h_clipped, 0, 1)
        max_shifts = []
        for k in range(1, n_points - 1):
            if k >= h_clipped.shape[0] - 1:
                continue
            # Forward
            val = np.dot(h_clipped[:n_points - k], (1 - h_clipped[k:]))
            # Backward
            val += np.dot(h_clipped[k:], (1 - h_clipped[:n_points - k]))
            max_shifts.append((val, k))
        if max_shifts:
            max_shifts.sort(reverse=True)
            top_shifts = [x[1] for x in max_shifts[:5]]

        if not top_shifts:
            continue

        h = adjust(h, top_shifts, n_steps_adapt=10)

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if not np.isclose(h_sum, target_sum):
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    return (h, evaluate(h), n_points)
