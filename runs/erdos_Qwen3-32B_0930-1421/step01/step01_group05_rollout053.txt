import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        if sum_h <= 1e-10:
            return np.zeros_like(h_in)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        scaled = np.clip(scaled, 0, 1)
        return scaled

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Optimization with L1 regularization
    h_opt = cp.Variable(n_points)
    constraint = [
        cp.sum(h_opt) == n_points / 2,
        h_opt >= 0,
        h_opt <= 1
    ]
    corr = cp.Convolution(h_opt, 1 - h_opt, mode='full')
    max_corr = cp.Max(corr)
    objective = cp.Minimize(max_corr + 0.1 * cp.norm(h_opt, 1))
    problem = cp.Problem(objective, constraint)

    remaining_time = budget_s - (time.time() - start_time)
    warm_start = h.copy()
    line_search = [0.1, 0.05, 0.01, 0.005, 0.001]
    best_lambda = 0.1
    best_h2 = warm_start.copy()
    best_c52 = compute_c5(best_h2)

    for l in line_search:
        if time.time() - start_time >= budget_s:
            break

        objective = cp.Minimize(max_corr + l * cp.norm(h_opt, 1))
        problem = cp.Problem(objective, constraint)
        try:
            problem.solve(solver=cp.CVXOPT, max_hours=remaining_time / 3600.0)
            current_h = np.clip(h_opt.value, 0, 1)
            norm = current_h.sum()
            if norm <= 1e-10:
                continue
            current_h = current_h / norm * n_points / 2
            current_h = np.clip(current_h, 0, 1)
            curr_c5 = compute_c5(current_h)

            if curr_c5 < best_c52:
                best_c52 = curr_c5
                best_h2 = current_h.copy()
                warm_start = current_h.copy()
                best_lambda = l

        except Exception:
            pass

    # Gradient descent refinement
    h = warm_start.copy()
    learning_rate = 0.1
    for step in range(500):
        if time.time() - start_time >= budget_s:
            break

        # Perturbing gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        # Update h
        h -= learning_rate * grad
        h = normalize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c52:
            best_c52 = current_c5
            best_h2 = h.copy()
        learning_rate *= 0.999

    # Combine best from both methods
    final_c5 = min(best_c5, best_c52)
    final_h = best_h if best_c5 < best_c52 else best_h2

    return final_h.tolist(), final_c5, n_points
