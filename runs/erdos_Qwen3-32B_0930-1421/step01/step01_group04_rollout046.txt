import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def setup_lp(n_points, h_init):
    h = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Add initial overlap constraints
    k_values = [1, 2, 4, 8, 16]
    for k in k_values:
        if k >= n_points:
            continue
        overlap = 0.0
        for i in range(n_points - k):
            overlap += h[i] * (1 - h[i + k])
        constraints.append(overlap * (2.0 / n_points) <= C)

    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    return prob, h, C

def identify_worst_k(h_sol, n_points):
    dx = 2.0 / n_points
    max_overlap = 0.0
    worst_k = 0
    for k in range(1, n_points):
        if k >= n_points:
            break
        overlap = np.sum(h_sol[:n_points - k] * (1 - h_sol[k:k + n_points - k]))
        overlap *= dx
        if overlap > max_overlap:
            max_overlap = overlap
            worst_k = k
    return worst_k, max_overlap

def add_worst_k_constraint(prob, h, C, worst_k, n_points):
    overlap_expr = 0.0
    for i in range(n_points - worst_k):
        overlap_expr += h[i] * (1 - h[i + worst_k])
    prob.constraints.append(overlap_expr * (2.0 / n_points) <= C)

def final_refinement(h_best, n_points):
    h_refine = opt.minimize(
        lambda x: compute_c5(x, n_points),
        h_best,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        options={'maxiter': 100, 'eps': 1e-4}
    ).x
    h_refine = enforce_normalization(h_refine, n_points)
    return h_refine, compute_c5(h_refine, n_points)

    h_init = enforce_normalization(h_init, n_points)

    prob, h, C = setup_lp(n_points, h_init)
    h_best = h_init.copy()
    c5_best = compute_c5(h_best, n_points)
    time_start = time.time()
    iter_count = 0

    while time.time() - time_start < budget_s and iter_count < 100:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status != 'optimal':
            break

        h_sol = h.value.flatten()
        h_sol = enforce_normalization(h_sol, n_points)

        k_worst, c5_worst = identify_worst_k(h_sol, n_points)
        if c5_worst > c5_best + 1e-5:
            add_worst_k_constraint(prob, h, C, k_worst, n_points)
            if c5_worst < c5_best:
                h_best = h_sol.copy()
                c5_best = c5_worst
        iter_count += 1

    h_best, c5_best = final_refinement(h_best, n_points)
    return h_best, c5_best, n_points
