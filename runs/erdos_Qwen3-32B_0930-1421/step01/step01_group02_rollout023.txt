import numpy as np
import time
from itertools import product
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    target_total = len(h) / 2
    correction = target_total / (total if total > 1e-6 else 1e6)
    h *= correction
    return h


def discretize_h(h, granularity=4):
    assert granularity >= 1
    return np.round(h * granularity) / granularity


def branch_and_bound(max_budget=1100, initial_granularity=4, n_points=42, use_initial=True):
    base_dx = 2.0 / n_points
    if use_initial and initial_h_values is not None:
        h = np.copy(initial_h_values)
        h = discretize_h(h, granularity=initial_granularity)
    else:
        h = np.zeros(n_points)
        # Initial pattern with balanced sum
        h[::2] = 1.0
    h = ensure_feasible(h)

    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    start_time = time.time()
    queue = [(h, list(range(n_points)))]  # (partial assignment, unassigned indices)

    granularity = initial_granularity
    all_possible_values = np.linspace(0, 1, granularity + 1)

    trial_limit = int(100_000)

    for trial in range(trial_limit):
        if time.time() - start_time > max_budget:
            break
        if not queue:
            break

        # Random sampling from current queue instead of exhaustive branching
        state_h, remaining = queue.pop(np.random.randint(len(queue)))
        if not remaining:
            current_C5 = evaluate_C5(state_h)
            if current_C5 < best_C5:
                best_h = state_h.copy()
                best_C5 = current_C5
            continue

        # Select next variable to assign
        i = remaining[0]
        remaining = remaining[1:]
        for val in all_possible_values:
            next_h = state_h.copy()
            next_h[i] = val
            next_h = ensure_feasible(next_h)
            queue.append((next_h, remaining))

    return best_h, best_C5, n_points


def hyperparameter_sweep():
    best_h = None
    best_C5 = np.inf
    max_budget = 1100
    start_time = time.time()

    # Try different granularities and number of points
    granularities = [4, 5, 10]
    n_points_values = [32, 42, 50]

    for gran in granularities:
        for n in n_points_values:
            if time.time() - start_time > max_budget - 100:
                break
            h, c5, _ = branch_and_bound(
                max_budget=100, initial_granularity=gran, n_points=n, use_initial=True
            )
            if c5 < best_C5:
                best_h = h
                best_C5 = c5
                best_n = n

    return best_h, best_C5, best_n


def run(budget_s=1100):
    # First get a quick baseline
    h, c5, n = branch_and_bound(max_budget=300, use_initial=True)
    best = (h, c5, n)

    # Then do a more thorough search
    try:
        h_full, c5_full, n_full = hyperparameter_sweep()
        best = (h_full, c5_full, n_full)
    except Exception as e:
