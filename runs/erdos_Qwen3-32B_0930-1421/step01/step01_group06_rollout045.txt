import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values and normalize
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h * scale_factor, 1.0)

    def project_to_feasible_space(H):
        H = np.clip(H, 0.0, 1.0)
        H *= (n_points / 2.0) / (np.sum(H) + 1e-8)  # Add small epsilon to avoid division by zero
        return H

    # PSO parameters
    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    # Initialize particles
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best_positions = particles.copy()
    personal_best_values = np.full(num_particles, float('inf'))
    global_best_position = h.copy()
    global_best_value = evaluate_erdos_solution(global_best_position, n_points)

    # Set 10% particles to the normalized provided initial_h_values + noise
    for i in range(int(num_particles * 0.1)):
        particles[i] = h + 0.05 * np.random.randn(n_points)
        particles[i] = project_to_feasible_space(particles[i])

    # Initialize remaining 90% particles with random feasible vectors
    for i in range(int(num_particles * 0.1), num_particles):
        p = np.random.rand(n_points)
        p = project_to_feasible_space(p)
        particles[i] = p

    velocities = 0.01 * np.random.randn(num_particles, n_points)  # Small initial velocities

    last_improvement_time = start_time

    while time.time() - start_time < budget_s - 3:  # Leave some margin
        inertia_weight = w_start - (w_start - w_end) * (time.time() - start_time) / (budget_s - 3)

        # Evaluate all particles
        c5_values = np.zeros(num_particles)
        for i in range(num_particles):
            c5_values[i] = evaluate_erdos_solution(particles[i], n_points)

        # Update personal and global best
        for i in range(num_particles):
            if c5_values[i] < personal_best_values[i]:
                personal_best_values[i] = c5_values[i]
                personal_best_positions[i] = particles[i].copy()
            if c5_values[i] < global_best_value:
                global_best_value = c5_values[i]
                global_best_position[:] = particles[i].copy()
                last_improvement_time = time.time()

        # Reset some particles if we haven't improved in a while
        if time.time() - last_improvement_time > 60:
            num_resets = int(num_particles * reset_fraction)
            indices = np.random.choice(num_particles, num_resets, replace=False)
            for i in indices:
                p = np.random.rand(n_points)
                p = project_to_feasible_space(p)
                particles[i] = p
                velocities[i] = 0.001 * np.random.randn(n_points)

        # Update velocities and positions
        for i in range(num_particles):
            # Generate random vectors for cognitive and social components
            r1, r2 = np.random.rand(), np.random.rand()
            personal_attraction = c1 * r1 * (personal_best_positions[i] - particles[i])
            global_attraction = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = (
                inertia_weight * velocities[i]
                + personal_attraction
                + global_attraction
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            particles[i] += velocities[i]
            particles[i] = project_to_feasible_space(particles[i])

    return global_best_position, global_best_value, n_points
