import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, peak_index, shift_k):
    n = len(h)
    # For symmetric h, assume that both regions around peak_index and shifted index have max contribution
    mapped_index_in_k = (peak_index + shift_k) % n
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    # Flip bits near significant overlap contributors
    i = indices_1[np.argmin(np.abs(indices_1 - peak_index))]
    j = indices_0[np.argmin(np.abs(indices_0 - mapped_index_in_k))]
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = initial_h_values.copy()
    h[h > 0.5] = 1.0

    while not np.isclose(np.sum(h), n_points / 2, atol=1e-8):
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)

    while time.time() - t0 < budget_s:
        h = best_h.copy()
        h_c5 = evaluate_overlap(h)
        full_corr = correlate(h, 1 - h, mode="full")
        full_corr = full_corr[n_points:2 * n_points]  # Only check k in [0, n_points-1]
        max_index = np.argmax(full_corr)
        peak_index = max_index
        shift_k = max_index

        # Flip bits to reduce the worst overlap
        h_new = flip_two_bits(h, peak_index, shift_k)

        # Maintain sum constraint
        while not np.isclose(np.sum(h_new), n_points / 2, atol=1e-8):
            if np.sum(h_new) < n_points / 2:
                idx = np.random.choice(np.where(h_new == 0)[0])
                h_new[idx] = 1.0
            else:
                idx = np.random.choice(np.where(h_new == 1)[0])
                h_new[idx] = 0.0

        new_c5 = evaluate_overlap(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5

    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
