import numpy as np
import time
import scipy.optimize as opt
from scipy.signal import correlate
import math


def rescale_h(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * h.size
    else:
        h[:] = 0.0
        h[0] = h.size
    return h


def evaluate_C5(h, dx):
    h = h.copy()
    h = rescale_h(h)
    h_sum = h.size / 2
    h = h / h.size * h_sum  # normalize to measure 1 over [0,2]
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def run(seed=42, budget_s=1100, n_points=120):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use provided `initial_h_values` if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points, refcheck=False)
    except:
        h = np.zeros(n_points)
        h[n_points // 4 : 3 * n_points // 4] = 0.5
        h = rescale_h(h)

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)[0]

    segment_size = int(n_points * 0.1)
    perturbation_amp = 0.1
    perturbation_freq = 50
    decay_interval = 100
    perturb_count = 0
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1

        # Clone current state
        h_candidate = h.copy()
        # Random segment flip
        start = np.random.randint(0, n_points - segment_size)
        h_candidate[start : start + segment_size] = 1 - h_candidate[start : start + segment_size]
        # Add small Gaussian noise to selected segment
        start = np.random.randint(0, n_points - segment_size)
        noise = (
            np.random.randn(segment_size) * perturbation_amp
            + np.random.rand(segment_size) * 0.2
        )
        h_candidate[start : start + segment_size] += noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = rescale_h(h_candidate)  # Ensure valid

        C5_candidate, _ = evaluate_C5(h_candidate, dx)
        if not np.isnan(C5_candidate) and C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h_candidate.copy()
            h = h_candidate  # adopt improvement as current
        else:
            # If no improvement, swap back
            pass

        if iter_count % decay_interval == 0:
            perturbation_amp *= 0.8
        if iter_count % perturbation_freq == 0:
            perturb_count += 1
            if perturb_count >= 50:
                perturbation_amp = max(0.05, perturbation_amp * 1.2)
                perturb_count = 0

    h_vals, c5, n_pts = best_h, best_C5, n_points
    return h_vals, c5, n_pts
