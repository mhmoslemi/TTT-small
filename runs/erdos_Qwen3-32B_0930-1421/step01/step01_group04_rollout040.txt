import numpy as np
import time
from scipy.optimize import minimize
from scipy.stats import norm

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_full = np.tile(h, 4)
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def sample_feasible(n_points, count):
    h_samples = []
    for _ in range(count):
        h = np.random.rand(n_points)
        h = enforce_normalization(h, n_points)
        h_samples.append(h)
    return np.array(h_samples)

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is None:
        init_h = np.random.rand(47)
    h = enforce_normalization(init_h, 47)
    n_points = len(h)
    t0 = time.time()
    budget_left = budget_s

    # Initial feasibility check
    h = enforce_normalization(h, n_points)

    # Sample feasible points for surrogate
    h_initial_samples = sample_feasible(n_points, 20)
    c5_initial_samples = [compute_c5(h, n_points) for h in h_initial_samples]

    # Approximate Gaussian Mean-only surrogate for direction guidance
    def surrogate(h_vec):
        h_vec = h_vec.reshape(n_points)
        h = enforce_normalization(h_vec, n_points)
        return compute_c5(h, n_points)

    def perturb_h(h, sigma):
        h_new = h + sigma * np.random.normal(size=(n_points,))
        h_new = np.clip(h_new, 0, 1)
        h_new = enforce_normalization(h_new, n_points)
        return h_new

    def simulated_annealing(h, surrogate, budget, init_temp=1000, cooldown=0.99):
        best_h = h.copy()
        best_c5 = compute_c5(best_h, n_points)
        sigma = 0.2
        temperature = init_temp

        t_start = time.time()
        while time.time() - t_start < budget:
            h_proposal = perturb_h(h, sigma)
            c5_proposal = compute_c5(h_proposal, n_points)

            diff = c5_proposal - best_c5
            if diff < 0 or np.random.rand() < np.exp(-diff / temperature):
                h = h_proposal
                if c5_proposal < best_c5:
                    best_h[:] = h_proposal
                    best_c5 = c5_proposal

            temperature *= cooldown
            sigma *= 0.999

        return best_h, best_c5

    # Simulated annealing with Gaussian process-inspired proposal
    h_sag, c5_sag = simulated_annealing(h, surrogate, budget_left - (time.time() - t0), init_temp=200)

    # Final refinement
    res = minimize(lambda x: compute_c5(x, n_points), h_sag, method='L-BFGS-B',
                   bounds=[(0, 1)] * n_points, options={'maxiter': 1000})

    h_best = res.x
    h_best = enforce_normalization(h_best, n_points)
    c5_best = compute_c5(h_best, n_points)

    return h_best, c5_best, n_points
