import numpy as np
import time
import scipy.optimize as opt

np.seterr(all='ignore')

def normalize(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total <= 1e-8:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    h_full = np.concatenate([np.zeros(n_points - 1), h, np.zeros(n_points - 1)])
    for k in range(1, n_points):
        if k >= len(h):
            break
        offset = (len(h) - k)
        full_h_k = h_full[k : k + offset]
        full_1_h = h_full[n_points : n_points + offset]
        corrs.append(np.dot(h[:offset], 1 - full_h_k))
        corrs.append(np.dot(h[k:], 1 - full_1_h[:len(h[k:])]))
    return np.max(np.array(corrs)) * dx

def evaluate_fitness(population, n_points):
    return np.array([compute_c5(h, n_points) for h in population])

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return normalize(child, n_points)

def mutate(h, sigma):
    perturbation = np.random.randn(len(h)) * sigma
    h_p = h + perturbation
    return normalize(h_p, len(h))

def init_population(base_h=None, n_points=47, pop_size=40):
    if base_h is None or len(base_h) != n_points:
        base_h = np.random.rand(n_points)
    base_h = normalize(base_h, n_points)
    pop = [base_h.copy()]
    for _ in range(pop_size - 1):
        h = mutate(pop[0], 0.1)
        pop.append(h)
    return np.array(pop)

def genetic_optimize(budget_s, n_points=47, pop_size=40, base_h=None):
    population = init_population(base_h, n_points, pop_size)
    start_time = time.time()
    sigma = 0.1
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            pop = population.copy()
            parent_idx1 = np.random.choice(pop_size)
            parent_idx2 = np.random.choice(pop_size)
            if parent_idx1 != parent_idx2:
                child = crossover(pop[parent_idx1], pop[parent_idx2])
                if np.random.rand() < 0.2:
                    child = mutate(child, sigma)
                child_fit = compute_c5(child, n_points)
                if child_fit < compute_c5(pop[i], n_points):
                    population[i] = normalize(child, n_points)
        improve_idx = np.argmin(evaluate_fitness(population, n_points))
        sigma *= 0.995
        if time.time() - start_time > budget_s - 5:
            final_refine = opt.minimize(lambda x: compute_c5(x, n_points),
                                        population[improve_idx].copy(),
                                        bounds=[(0,1)]*n_points,
                                        method='L-BFGS-B',
                                        options={'maxiter': 100})
            final_best = normalize(final_refine.x, n_points)
            final_val = compute_c5(final_best, n_points)
            return final_best, final_val
    final_best = normalize(population[np.argmin(evaluate_fitness(population, n_points))], n_points)
    final_val = compute_c5(final_best, n_points)
    return final_best, final_val
