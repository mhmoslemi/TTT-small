import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_worst_slices(h, k_peak):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    left = max(0, k_peak - n // 4)
    right = min(n, k_peak + n // 4)
    contrib_1 = indices_1[(indices_1 >= left) & (indices_1 < right)]
    contrib_0 = indices_0[(indices_0 >= left) & (indices_0 < right)]
    if contrib_1.size == 0 or contrib_0.size == 0:
        return h.copy()
    i = np.random.choice(contrib_1)
    j = np.random.choice(contrib_0)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = initial_h_values.copy()
    h = np.round(h).clip(0, 1)

    while np.sum(h) != n_points / 2:
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    current_c5_history = [best_c5]

    while time.time() - t0 < budget_s:
        c = correlate(current_h, 1 - current_h, mode="full")
        c_peak_idx = np.argmax(c)
        k_peak = c_peak_idx - (len(current_h) - 1)

        # Construct h_new by flipping bits around the peak
        h_new = flip_worst_slices(current_h, k_peak)
        if np.all(h_new == current_h):
            continue  # No flip, skip
        new_c5 = evaluate_overlap(h_new)
        current_c5_history.append(new_c5)
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        elif len(current_c5_history) > 200 and np.std(current_c5_history[-200:]) < 1e-5:
            # Escape local minima by flipping arbitrary 1 and 0
            indices_1 = np.where(current_h == 1)[0]
            indices_0 = np.where(current_h == 0)[0]
            idx1 = np.random.choice(indices_1)
            idx0 = np.random.choice(indices_0)
            h_new = current_h.copy()
            h_new[idx1] = 0
            h_new[idx0] = 1
            if np.sum(h_new) == n_points / 2:
                current_h = h_new
                current_c5 = evaluate_overlap(current_h)
                current_c5_history = [current_c5]
        elif time.time() - t0 > budget_s - 1.0:
            break

    while np.sum(best_h) != n_points / 2:
        if np.sum(best_h) < n_points / 2:
            idx = np.random.choice(np.where(best_h == 0)[0])
            best_h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(best_h == 1)[0])
            best_h[idx] = 0.0

    final_c5 = evaluate_overlap(best_h)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8
    return (best_h, final_c5, n_points)
