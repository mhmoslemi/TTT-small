import numpy as np
import time
from scipy.signal import correlate

def compute_gradient(h, dx, k_shift, n_points):
    grad = np.zeros_like(h)
    lag = k_shift - (n_points - 1)
    for i in range(n_points):
        if i + lag < n_points:
            grad[i] += (1 - h[i + lag]) * dx
        if i - lag >= 0:
            grad[i] -= h[i - lag] * dx
    return grad

def project_to_simplex(v, target_sum):
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-8:
        return v
    return v * (target_sum / current_sum)

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):

    # Discretize if not binary

    # Enforce sum constraint
    h = project_to_simplex(h, target_sum=n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5
    t0 = time.time()
    max_iter = 100000
    iter_count = 0
    alpha = 1e-3
    step_decay = 0.5
    step_increase = 1.1
    min_alpha = 1e-6
    perturb_interval = 500
    perturb_magnitude = 0.01

    while time.time() - t0 < budget_s - 0.1:
        corr = correlate(current_h, 1 - current_h, mode="full")
        k_shift = np.argmax(corr)
        grad = compute_gradient(current_h, 2.0 / n_points, k_shift, n_points)

        # Line search with backtracking
        alpha_test = alpha
        for __ in range(20):
            h_candidate = current_h - alpha_test * grad
            h_proj = project_to_simplex(h_candidate, target_sum=n_points / 2)
            c5_candidate = evaluate_overlap(h_proj)
            if c5_candidate < current_c5:
                current_c5 = c5_candidate
                current_h = h_proj
                alpha = min(alpha * step_increase, 1e-1)
                break
            elif alpha_test > min_alpha:
                alpha_test *= step_decay
            else:
                break
        else:
            # No progress; decay and perturb
            alpha = max(alpha * step_decay, min_alpha)
            if iter_count % perturb_interval == 0:
                current_h = project_to_simplex(current_h + np.random.normal(0, perturb_magnitude, size=n_points), target_sum=n_points / 2)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

        iter_count += 1
        if iter_count >= max_iter:
            break

    # Final assertion and return
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    return (best_h, evaluate_overlap(best_h), n_points)
