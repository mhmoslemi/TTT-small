import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def convexQP_step(h, k, dx):
    """Solve a convex QP to reduce overlap for a specific shift k."""
    n = len(h)
    h_var = cp.Variable(n, nonneg=True, <= 1)
    correlation = np.zeros_like(h)
    for i in range(n):
        if 0 <= i + k < n:
            correlation[i] = -h[(i + k) % n]
    objective = cp.Minimize(h_var @ (2 * correlation))
    constraints = [
        h_var.sum() == (n / 2),
    ]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, feastol_rel=1e-5, max_iter=500)
    if prob.status != cp.OPTIMAL:
        return h
    h_new = h_var.value
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new / h_new.sum() * (n / 2)
    return h_new

    best_h = h.copy()
    best_C5, _ = evaluate_C5(h, dx)
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        h = rescale_h(h)
        h_val, C5_current = evaluate_C5(h, dx)
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()

        # Identify most problematic shift
        h_current = h.copy()
        h_current[h_current < 0] = 0
        h_current[h_current > 1] = 1
        cor = correlate(h_current, 1 - h_current, mode="full")
        max_id = np.argmax(cor)
        k = max_id - len(h) + 1

        # Apply Convex QP
        h_imp = convexQP_step(h_current, k, dx)
        h_imp = rescale_h(h_imp)
        new_C5, _ = evaluate_C5(h_imp, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_imp.copy()
            h = h_imp

        # Random restart every 50 iterations
        if int(100 * (time.time() - start_time) / budget_s) % 50 == 0:

    return best_h, best_C5 * dx, n_points
