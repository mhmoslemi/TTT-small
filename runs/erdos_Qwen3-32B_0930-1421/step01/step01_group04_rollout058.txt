import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
from sklearn.metrics import mean_squared_error
import scipy.optimize as opt

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k < n_points:
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            corrs.append(corr * dx)
    return np.max(np.array(corrs))

def sample_h(n_points):
    h = np.random.rand(n_points)
    return enforce_normalization(h, n_points)

            gp.fit(temp_X, temp_y)

        h_test = h_best + np.random.normal(0, 0.1, n_points)

        # Accept new solution
        h_p = h_test.copy()

        X = np.vstack((X, h_test))
        y = np.append(y, val_p)

        budget_left = max(budget_s - (time.time() - time_start), 0)

    return h_best, val_best, n_points
