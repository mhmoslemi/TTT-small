import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize the initial h to meet the sum constraint
    h_sum = h.sum()
    if not np.isclose(h_sum, target_sum):
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)  # Ensure h stays within [0, 1]

    best_h = h.copy()
    best_c5 = float('inf')
    best_time = time.time()

    def compute_overlaps(h_vec):
        ones_minus_h = 1.0 - h_vec
        overlaps = []

        # Forward overlaps (x from left to right)
        for k in range(1, n_points):
            if k >= n_points:
                break
            valid_length = n_points - k
            overlap = np.dot(h_vec[:valid_length], ones_minus_h[k:])
            overlaps.append(overlap * dx)

        # Reverse overlaps (x from right to left)
        for k in range(1, n_points):
            if k >= n_points:
                break
            valid_length = n_points - k
            overlap = np.dot(h_vec[k:], ones_minus_h[:valid_length])
            overlaps.append(overlap * dx)

        return np.array(overlaps)

    def compute_gradient(h_vec, k):
        ones_minus_h = 1.0 - h_vec
        grad = np.zeros_like(h_vec)

        # Forward: h[i] * (1 - h[i + k])
        for i in range(n_points - k):
            grad[i] += (1 - h_vec[i + k])  # d/dh[i]
            grad[i + k] -= h_vec[i]        # d/dh[i + k]

        # Reverse: h[j] * (1 - h[j - k]) where j > k
        for j in range(k, n_points):
            grad[j] += (1 - h_vec[j - k])
            grad[j - k] -= h_vec[j]

        return grad

    while time.time() - start_time < budget_s:
        overlaps = compute_overlaps(h)
        max_overlap = np.max(overlaps)
        active_shifts = np.where(np.isclose(overlaps, max_overlap))[0]

        # Use a random active shift to compute gradient
        k_idx = np.random.choice(active_shifts)

        max_corr = overlaps[k_idx]
        grad = compute_gradient(h, k_idx)

        # Normalize gradient and apply a step
        step_weight = 0.02
        h_proposed = h - step_weight * grad

        # Normalize the proposed h
        h_proposed = np.clip(h_proposed, 0, 1)
        h_sum_proposed = h_proposed.sum()
        if not np.isclose(h_sum_proposed, target_sum):
            h_proposed *= (target_sum / h_sum_proposed)

        # Evaluate C5 for proposition
        overlaps_proposed = compute_overlaps(h_proposed)
        new_c5 = np.max(overlaps_proposed)

        # Accept if better
        if new_c5 < best_c5:
            h = h_proposed
            best_c5 = new_c5
            best_h = h.copy()
            best_time = time.time()

        # Perturb if no improvement for a while
        if time.time() - best_time > 50:
            perturb_indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
            h[perturb_indices] += np.random.uniform(-0.01, 0.01, size=len(perturb_indices))
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if not np.isclose(h_sum, target_sum):
                h *= (target_sum / h_sum)

        # Logging if verbose
        if verbose and (time.time() - start_time) % 10 < 1e-2:
            print(f"Time: {time.time() - start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
