import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)


def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial h to satisfy the average constraint
    total_h = float(np.sum(initial_h_values))
    if total_h == 0.0:
        raise ValueError("Initial h has zero sum")
    scale = (n_points / 2.0) / total_h
    h_initial = np.minimum(np.maximum(initial_h_values * scale, 0.0), 1.0)
    current_h = h_initial.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement_time = start_time

    def update_best(h, c5):
        nonlocal best_h, best_c5, last_improvement_time
        if c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = c5
            last_improvement_time = time.time()
            return True
        return False

    def solve_feasibility(C_f):
        h = cp.Variable(n_points)
        z = cp.Variable((n_points, n_points))
        objective = 0.0
        constraints = [
            cp.sum(h) == n_points / 2.0,
            0 <= h,
            h <= 1,
        ]

        for i in range(n_points):
            for j in range(n_points):
                if 0 <= i + (j - n_points + 1) < n_points:
                    k = j - n_points + 1
                    if k == 0:
                        continue
                    # Common shift range for index i
                    if i + k < n_points:
                        constraints += [
                            z[i, j] <= h[i],
                            z[i, j] <= 1 - h[i + k],
                            z[i, j] >= h[i] + (1 - h[i + k]) - 1,
                            z[i, j] >= 0,
                        ]
        constraints += [
            cp.sum(z[np.triu_indices(n_points, k=1)]) <= C_f * dx * (n_points - 1)
        ]
        problem = cp.Problem(cp.Minimize(objective), constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)
        return problem.status == cp.OPTIMAL, h.value

    # Binary search for optimal C₅
    C_low = 0.0
    C_high = 2.0 / n_points  # max possible
    C_best = best_c5
    while time.time() - start_time < budget_s:
        if C_high - C_low < 1e-6:
            break
        C_mid = (C_low + C_high) / 2
        feasible, h_sol = solve_feasibility(C_mid)
        if feasible:
            update_best(h_sol, C_mid)
            C_high = C_mid
        else:
            C_low = C_mid

        if time.time() - last_improvement_time > 30:
            perturbed_h = best_h + 0.1 * np.random.rand(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2.0) / np.sum(perturbed_h)
            feasible, h_pert = solve_feasibility(C_mid)
            if feasible:
                update_best(h_pert, C_mid)

        if time.time() - last_improvement_time > 60:
            break

    return best_h, best_c5, n_points
