import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from math import floor, ceil

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Constants and limits
    max_time = budget_s * 0.99
    n_points = 47 if not hasattr(initial_h_values, '__len__') else len(initial_h_values)
    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # Basic initial configuration with uniform selections
        h = np.zeros(n_points)
        h[::2] = 2.0 / n_points
        h = h * constraint_sum / h.sum()

    def relax_convex(h_support, eps=1e-3):
        # Define CVXPY variables over active support
        w = cp.Variable(len(h_support), nonneg=True)
        obj_vals = []

        def overlap_at_k_obj(k):
            # Define (w * (1 - w))^k via convolution
            # Convert to a convolution at index k
            # Construct the weighted sum for the k-th overlap
            weights_1 = [h_support[i] for i in range(len(h_support))]
            weights_2 = [h_support[n_points - (k - i)] if 0 <= n_points - (k - i) < len(h_support) else 0 for i in range(len(h_support))]
            obj = cp.Maximize(cp.sum(w * (1 - w * abs(np.array(weights_1) - np.array(weights_2)))))
            constraints = [
                cp.sum(w * dx) == 1,
                0 <= w,
                w <= 1
            ]
            return obj, constraints

        max_obj = cp.Minimize(0)
        constraints = [cp.sum(w * dx) == 1, 0 <= w, w <= 1]
        for k in range(-n_points + 1, n_points):
            obj, c = overlap_at_k_obj(k)
            max_obj = cp.Minimize(cp.Max(cp.atoms.max(cp.atoms.abs(obj), cp.atoms.abs(max_obj)), eps))
            constraints += c

        prob = cp.Problem(max_obj, constraints)
        prob.solve(solver=cp.CVXOPT)
        return w.value

    def perturb_structure(h, noise_factor=0.2):
        idxs = np.random.choice(n_points, int(n_points * noise_factor), replace=False)
        h_pert = h.copy()
        h_pert[idxs] = np.random.uniform(0, 1, len(idxs))
        h_pert = h_pert.clip(0, 1)
        h_pert = h_pert * (constraint_sum / h_pert.sum())
        return h_pert

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    log_period = 30
    i = 0

    while time.time() - start_time < max_time:
        i += 1
        if i % 10 == 0:
            print(f"Iteration: {i}, Current best C₅: {best_c5:.6f}, Time: {time.time() - start_time:.2f}/{max_time}")

        h = perturb_structure(h, noise_factor=0.1 if i < 20 else 0.05)
        h_relaxed = relax_convex(h)
        idxs = np.where(h > 0)[0]
        h_support = h[idxs]
        new_h = np.zeros(n_points)
        new_h[idxs] = h_relaxed
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (constraint_sum / new_h.sum())

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
