import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel
import scipy.optimize as opt

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        i_vals = h[:n_points - k]
        j_vals = 1 - h[k:]
        corrs.append(np.dot(i_vals, j_vals))
    for k in range(1, n_points):
        i_vals = h[k:]
        j_vals = 1 - h[:n_points - k]
        corrs.append(np.dot(i_vals, j_vals))
    return np.max(np.array(corrs)) * dx

def objective(h, n_points):
    return compute_c5(h, n_points)

    np.random.seed(seed)

    y = [objective(x, n_points) for x in X]
    
    gp_kernel = RBF(length_scale=1.0, length_scale_bounds=(0.01, 10)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=gp_kernel, n_restarts_optimizer=3)

    y_np = np.array(y)

        gp.fit(X_np, y_np)

    alpha = 0.995
    best_h = h_mean
    best_val = objective(best_h, n_points)

    while time.time() - t0 < budget_s:
        h = best_h + 0.1 * (T / 1000) * np.random.normal(size=n_points)
        h = enforce_normalization(h, n_points)
        val = objective(h, n_points)

        if val < best_val or np.random.rand() < np.exp((best_val - val) / T):
            best_h = h
            best_val = val

        X.append(h)
