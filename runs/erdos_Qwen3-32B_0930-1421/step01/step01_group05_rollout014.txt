import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.ones(n_points) / n_points * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = correlate(h, shifted, mode='full')
        corr = corr[n_points - 1: -n_points if n_points != 1 else None] * dx
        return np.max(corr)

    def normalize(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        h_scl = h_clipped * (n_points / 2) / total
        return np.clip(h_scl, 0, 1)

    best_h = normalize(h.copy())
    best_c5 = compute_c5(best_h)

    n_outer = max(10, int(budget_s // 5))
    for outer in range(n_outer):
        if time.time() - start_time >= budget_s:
            break

        # Gradient descent with focus on current worst shift
        h_current = best_h.copy()
        c5_current = compute_c5(h_current)
        corr = correlate(h_current, 1 - h_current, mode='full')
        corr = corr[n_points - 1: -n_points if n_points != 1 else None]
        k_weighted = np.argmax(corr)

        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            # + eps
            h_p = h_current.copy()
            h_p[i] += eps
            h_p = normalize(h_p)
            c_p = compute_c5(h_p)

            # - eps
            h_n = h_current.copy()
            h_n[i] -= eps
            h_n = normalize(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        # Gradient step
        step_size = 0.2 * 0.995**outer
        h_current = h_current - step_size * grad
        h_current = normalize(h_current)

        # Check for improvement
        new_c5 = compute_c5(h_current)
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
            print(f"Iter {outer}: improved C5 = {best_c5}")

        # Additional perturbation to escape local minima
        if np.random.rand() < 0.2:
            h_current = best_h.copy()
            h_current += np.random.normal(0, 0.01, n_points)
            h_current = normalize(h_current)
            new_c5 = compute_c5(h_current)
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5

        # Periodically run a small convex relaxation (first-stage method)
        if outer % 5 == 0 and outer > 0:
            h_relax = cp.Variable(n_points)
            constraints = [cp.sum(h_relax) == n_points / 2, 0 <= h_relax, h_relax <= 1]
            obj = cp.Minimize(compute_c5(h_relax))
            cvx_problem = cp.Problem(obj, constraints)
            try:
                cvx_problem.solve(solver=cp.ECOS, max_iters=150, time_limit=1.0)
                h_relax_val = np.clip(h_relax.value, 0, 1)
                c_relax = compute_c5(h_relax_val)
                if c_relax < best_c5:
                    best_h = h_relax_val
                    best_c5 = c_relax
            except:
                pass

    # Final evaluation with timer
    print(f"Final best bound: C_5 = {best_c5}")
    return best_h.tolist(), best_c5, n_points
