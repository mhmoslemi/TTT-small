import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)

    def simulated_annealing_binary(maxiter=1000):
        h_b = np.zeros(n_points, dtype=bool)
        idx = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h_b[idx] = True
        best_h_b = h_b.copy()
        best_c5 = evaluate_erdos_solution(h_b.astype(float) * 2, n_points)

        T0 = 1.0
        T = T0
        for t in range(maxiter):
            if time.time() - start_time > budget_s:
                return best_h_b
            swap_indices = np.random.choice(n_points, size=2, replace=False)
            if h_b[swap_indices[0]] != h_b[swap_indices[1]]:
                h_b[swap_indices] ^= True  # toggle both to maintain count
                c5 = evaluate_erdos_solution(h_b.astype(float) * 2, n_points)
                if c5 < best_c5 or np.exp((best_c5 - c5) / T) > np.random.rand():
                    best_c5 = c5
                    best_h_b = h_b.copy()
                else:
                    h_b[swap_indices] ^= True  # revert
            T = T0 / np.log(1 + t)
        return best_h_b

    def scale_to_sum(target_sum):
        binary = simulated_annealing_binary()
        binary_sum = binary.sum()
        if binary_sum < target_sum:
            idx = np.where(~binary)[0]
            binary[idx[0]] = True
        elif binary_sum > target_sum:
            idx = np.where(binary)[0]
            binary[idx[0]] = False

        # Now smooth by setting one value to 0.5 if needed
        smooth = binary.astype(float)
        diff = n_points / 2 - binary_sum
        if abs(diff) > 1e-6:
            idx = int(np.round(np.linspace(0, n_points - 1, 1)[0]))
            smooth[idx] = 0.5 * (1 if binary_sum < target_sum else -1) + binary[idx]
        return smooth

    h_binary = scale_to_sum(n_points / 2)

    def objective(x):
        c5 = evaluate_erdos_solution(x, n_points)
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    cons = (
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_upper},
        {'type': 'ineq', 'fun': constraint_lower}
    )
    bounds = [(0.0, 1.0)] * n_points

    result = minimize(
        objective,
        h_binary,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100}
    )

    best_c5 = evaluate_erdos_solution(result.x, n_points)
    return result.x, best_c5, n_points
