import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    i = np.random.choice(indices_1)
    j = np.random.choice(indices_0)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def find_optimal_flip_indices(h, corr, dx, top_k=10):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]

    # Get the top-k worst overlaps (negative values indicate worse overlap)
    worst_overlaps_neg = -corr
    worst_indices = np.argsort(worst_overlaps_neg)[:top_k]
    target_pos = (worst_indices - n + 1).astype(int)

    valid_target_pos = target_pos[(target_pos >= 0) & (target_pos < n)]
    valid_positions = np.unique(np.concatenate((indices_1, valid_target_pos)))
    
    indices_flips = np.random.choice(valid_positions, size=len(valid_positions), replace=False)
    
    for i in indices_flips:
        for j in np.random.choice(indices_0, size=min(len(indices_0), 20), replace=False):
            if i != j:
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1
                if np.isclose(np.sum(h_new), len(h) / 2):
                    return h_new
    return h
