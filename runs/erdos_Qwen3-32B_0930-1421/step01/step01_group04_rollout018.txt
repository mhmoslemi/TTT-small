```python
import numpy as np
import scipy.optimize as opt
import time
import heapq
from itertools import islice

def compute_c5_exact(h, n_points):
    dx = 2.0 / n_points
    h_full = np.tile(h, 4)
    corrs = []
    # Positive lags (k > 0)
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    # Negative lags (k < 0)
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs) * dx), np.array(corrs)

def enforce_constraints(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def initialize_h_cosine(n_points):
    i = np.arange(n_points)
    seed_shape = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
    seed_shape = enforce_constraints(seed_shape, n_points)
    return seed_shape

def initialize_h_random(n_points, seed):
    np.random.seed(seed)
    h = np.random.rand(n_points)
    h = enforce_constraints(h, n_points)
    return h

def objective_surrogate(h_vec, n_points, alpha):
    h = h_vec.reshape(-1)
    _, corrs = compute_c5_exact(h, n_points)
    barrier = -np.sum(np.log(h.clip(1e-6, 1) - 1e-6)) \
              - np.sum(np.log(1.0 - h.clip(0, 1 - 1e-6))) \
              - np.log(np.abs(np.sum(h) - n_points / 2) + 1e-6)
    return np.sum(corrs**2) * (2.0 / n_points)**2 + alpha * barrier

def get_n_points_budget(budget_s, n_start=30):
    # Heuristic: Allocate 20s per n_points across 5 to 100
    max_n_points = int(200 + 30 * budget_s / 1100)
    n_candidates = np.ceil(np.linspace(5, max_n_points, num=5)).astype(int)
    n_candidates = n_candidates[n_candidates >= 20]
    return n_candidates

def particle_update(particle, velocity, best_p, best_g, w, c1, c2):
    r1 = np.random.rand(len(particle))
    r2 = np.random.rand(len(particle))
    velocity = (
        w * velocity +
        c1 * r1 * (best_p - particle) +
        c2 * r2 * (best_g - particle)
    )
    particle += velocity
    velocity = np.clip(velocity, -0.1, 0.1)  # Limit step size
    return particle, velocity

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    time_used = 0.0
    best_h = None
    best_val = float('inf')
    
    if init_h is not None:
        h_init = enforce_constraints(init_h, len(init_h))
    else:
        h_init = initialize_h_cosine(47)

    n_points_list = get_n_points_budget(budget_s)

    for attempt in range(len(n_points_list)):
        n_points = n_points_list[attempt]
        time_per_attempt = budget_s / len(n_points_list)

        if init_h is not None:
            h0 = np.clip(init_h, 0.0, 1.0)
        else:
            h0 = initialize_h_cosine(n_points)

        # Initialize particles
        n_particles = 16
        w = 0.7
        c1, c2 = 1.5, 1.5
        particles = []
        velocities = []
        for _ in range(n_particles):
            # Random particle with normalization
            h = np.random.rand(n_points) * 0.3 + h0 * 0.7
            particles.append(enforce_constraints(h, n_points))
            velocities.append(np.random.rand(n_points) * 0.05)

        # Evaluation and tracking
        best_particles, best_vals = [], []

        # PSO Core
        end_time = time.time() + time_per_attempt
        while time.time() < end_time:
            for i in range(n_particles):
                if best_vals[i] is None:
                    val_exact, _ = compute_c5_exact(particles[i], n_points)
                    best_vals[i] = val_exact
                # Update particle if better
                improved = False
                for _ in range(3):
                    particles[i], velocities[i] = particle_update(
                        particles[i], velocities[i],
                        particles[i], best_particles[i],  # Self as best so far
                        w, c1, c2
                    )
                    particles[i] = enforce_constraints(particles[i], n_points)
                    val_current, _ = compute_c5_exact(particles[i], n_points)
                    if val_current < best_vals[i]:
                        best_vals[i] = val_current
                        best_particles[i] = particles[i].copy()
                        improved = True
                    if not improved:
                        break  # prevent wiggling too much

            # Extract global best
            best_indices = np.argsort(best_vals)
            if best_vals[best_indices[0]] < best_val:
                best_val = best_vals[best_indices[0]]
                best_h = best_particles[best_indices[0]].copy()

            # Reduce inertia for convergence
            w = max(w * 0.999, 0.4)

        # Final refinement
        h = best_particles[best_indices[0]]
        result = opt.minimize(
            lambda x: objective_surrogate(x, n_points, alpha=1000),
            h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        hf = result.x
        hf = enforce_constraints(hf, n_points)
        val_final, _ = compute_c5_exact(hf, n_points)
        if val_final < best_val:
            best_val = val_final
            best_h = hf

        # Update for next attempt
        h_init = best_h.copy()

    return best_h, best_val, len(best_h)
```
