import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement_time = start_time

    # Range for binary sum
    target_sum = n_points // 2
    target_sum_low = max(1, target_sum - 1)
    target_sum_high = min(n_points, target_sum + 1)

    def flip_k_bits(binary_h, k=1):
        idxs = np.random.choice(np.arange(n_points), size=k, replace=False)
        binary_h[idxs] ^= 1
        return binary_h

    def satisfies_sum_constraint(binary_h):
        return target_sum_low <= np.sum(binary_h) <= target_sum_high

    def binary_anneal(num_steps=500, initial_T=1.0, cooling_rate=0.99, flip_per_step=1):
        nonlocal best_c5, best_h, last_improvement_time

        binary_h = np.random.choice([0, 1], size=n_points)
        while not satisfies_sum_constraint(binary_h):
            binary_h = np.random.choice([0, 1], size=n_points)

        best_binary_h = np.copy(binary_h)
        best_binary_c5 = evaluate_erdos_solution(best_binary_h, n_points)

        T = initial_T
        for step in range(num_steps):
            if time.time() - start_time >= budget_s:
                break
            candidate = np.copy(binary_h)
            flip_k_bits(candidate, flip_per_step)
            if satisfies_sum_constraint(candidate):
                candidate_c5 = evaluate_erdos_solution(candidate, n_points)
                delta_c5 = candidate_c5 - best_binary_c5
                if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                    binary_h = candidate
                    best_binary_c5 = candidate_c5
                    if candidate_c5 < best_c5:
                        best_c5 = candidate_c5
                        best_h[:] = np.minimum(binary_h * scale_factor, 1.0)
                        last_improvement_time = time.time()

            T *= cooling_rate
        # Soft scaling of h with fractional position to adjust the sum
        if best_binary_c5 < best_c5:
            s = np.sum(best_binary_h)
            best_h[:] = np.minimum(best_binary_h * (target_sum / s), 1.0)
        return best_binary_h

    binary_opt_h = binary_anneal(num_steps=500, flip_per_step=1)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    # refined continuous optimization from best binary solution
    def optimize_step(x0):
        nonlocal best_c5, best_h, last_improvement_time
        res = minimize(
            evaluate_erdos_solution,
            x0,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints=[
                {"type": "eq", "fun": constraint_sum},
                {"type": "ineq", "fun": constraint_upper},
                {"type": "ineq", "fun": constraint_lower}
            ],
            args=(n_points,)
        )
        if res.success:
            x_new = res.x
            c_new = evaluate_erdos_solution(x_new, n_points)
            if c_new < best_c5:
                best_c5 = c_new
                best_h[:] = x_new
                last_improvement_time = time.time()
    optimize_step(best_h)
    return best_h, best_c5, n_points
