import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)


def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Use the initial_h_values directly from the parent construction
    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to hard constraint: sum(h) == n_points / 2
    current_sum = np.sum(h)
    h = np.clip((n_points / 2.0) / current_sum * h, 0.0, 1.0)
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = time.time()

    def evaluate_obj(h_var):
        return evaluate_erdos_solution(h_var, n_points)

    def optimize_with_cvxpy(max_iter=100):
        nonlocal h, best_c5, best_h, last_improvement_time

        h_var = cp.Variable(n_points)
        sum_h = cp.sum(h_var)
        constraint_sum = [sum_h == n_points / 2.0]
        constraint_bounds = [0.0 <= h_var, h_var <= 1.0]
        constraints = constraint_sum + constraint_bounds

        corr_pos = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr_pos)
        target_C5 = 0.38085  # Target bound
        epsilon = 0.001 * max_corr

        # Relaxation: approximate max correlation as an upper bound on C₅
        obj = cp.Minimize(target_C5 + epsilon)

        constraint_C5 = []
        for lag in range(-n_points + 1, n_points):
            if lag == 0:
                continue  # skip k=0 center term (trivial 0 contribution)
            kernel = np.roll(np.ones(n_points), lag)
            max_corr_k = np.dot(h, (1 - np.roll(h, -lag)))[(lag - 1) % n_points]
            constraint_C5 += [target_C5 >= max_corr_k]
        constraints += constraint_C5
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, maxiters=max_iter)

        h_proposal = np.array(h_var.value)
        if prob.status == cp.OPTIMAL:
            h_proposal = np.clip(h_proposal, 0.0, 1.0)
            h_proposal = (n_points / 2.0 / np.sum(h_proposal)) * h_proposal
            new_c5 = evaluate_erdos_solution(h_proposal, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = h_proposal.copy()
                last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 10:
        optimize_with_cvxpy(max_iter=30)
        if time.time() - last_improvement_time > 50:
            break

    return best_h, best_c5, n_points
