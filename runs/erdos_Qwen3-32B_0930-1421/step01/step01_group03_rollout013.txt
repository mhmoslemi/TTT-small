import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from copy import copy

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        return h * (constraint_sum / h_sum)

    def optimize(h):
        h = h.copy()
        h = repair_h(h)

        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(flat_h):
            return objective(flat_h)

        result = minimize(
            obj_func,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000}
        )
        return repair_h(result.x)

    def perturb(h, scale=0.1):
        h = h.copy()
        h += np.random.randn(n_points) * scale
        return repair_h(h)

    def crossover(h1, h2):
        child = np.zeros(n_points)
        mask = np.random.rand(n_points) > 0.5
        child[mask] = h1[mask]
        child[~mask] = h2[~mask]
        return repair_h(child)

    population_size = min(20, n_points)
    population = [repair_h(h_init + 0.1 * np.random.randn(n_points)) for _ in range(population_size)]
    for p in range(population_size):
        population[p] = optimize(population[p])

    best_h = min(population, key=objective)
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s - 5:
        # New generation
        new_population = []

        # Rank by objective; lower is better
        sorted_indices = np.argsort([objective(p) for p in population])
        top_k = population_size // 3

        for idx in sorted_indices[:top_k]:
            new_population.append(population[idx])
            # Crossover
            if len(new_population) < population_size:
                partner = population[np.random.choice(sorted_indices)]
                new_population.append(crossover(population[idx], partner))
            # Perturbation
            if len(new_population) < population_size:
                new_population.append(perturb(population[idx], scale=0.2))
        population = new_population

        # Optimize the top two
        for idx in sorted_indices[:2]:
            population[idx] = optimize(population[idx])

        # Update best
        current_best = min(population, key=objective)
        c5_current = objective(current_best)
        if c5_current < best_c5:
            best_h = current_best
            best_c5 = c5_current

    # Final check for elapsed time
    best_h = repair_h(best_h)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
