import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    max_n_points = 100
    n_points = min(len(initial_h_values), max_n_points) if 'initial_h_values' in globals() else 47
    constraint_sum = n_points / 2
    dx = 2.0 / n_points

    # Feasibility repair
    def normalize_h(h):
        h = h.clip(0, 1)
        total = h.sum()
        return h * (constraint_sum / total) if total != 0 else h

    # Initialize from available parent, else basic construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[2 * i] = 2.0 / n_points
        h_init = normalize_h(h_init)

    # Compute C5 from h
    def get_c5(h):
        h = h.reshape(n_points)
        h = normalize_h(h)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    # Gradient-based optimization
    def gradient_optimize(h):
        h = h.copy()
        bounds = [(0, 1)] * n_points

        def obj_func(x):
            h_local = np.clip(x, 0, 1)
            total = h_local.sum()
            h_local = h_local * (constraint_sum / total)
            overlaps = np.correlate(h_local, 1 - h_local, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            obj_func,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        h_opt = result.x
        return normalize_h(h_opt)

    # Perturb + coarse combinatorial search
    def perturb_comb(h, delta=0.05):
        h = h.copy()
        idx = np.random.choice(n_points)
        direction = np.random.choice([-1, 1])
        step = direction * delta
        h[idx] = np.clip(h[idx] + step, 0, 1)
        return h

    def comb_search(h, max_rounds=100):
        best = h.copy()
        best_c5 = get_c5(best)
        for _ in range(max_rounds):
            if time.time() - start_time > budget_s * 0.98:
                break
            neighbor = perturb_comb(h)
            neighbor = normalize_h(neighbor)
            c5 = get_c5(neighbor)
            if c5 < best_c5:
                best = neighbor
                best_c5 = c5
                h = best  # guide next neighbor toward local improvement
        return best, best_c5

    best_h = h_init.copy()
    best_c5 = get_c5(best_h)

    # Phase 1: combinatorial search
    best_h, best_c5 = comb_search(best_h)

    # Phase 2: gradient descent
    best_h = gradient_optimize(best_h)
    best_c5 = get_c5(best_h)

    # Phase 3: Restart and refine
    for _ in range(4):
        if time.time() - start_time > budget_s * 0.98:
            break
        variation = np.random.normal(0, 0.1, n_points)
        seed_h = h_init + variation
        seed_h = normalize_h(seed_h)
        refined = gradient_optimize(seed_h)
        c5 = get_c5(refined)
        if c5 < best_c5:
            best_h = refined
            best_c5 = c5

    best_h = normalize_h(best_h)
    final_c5 = get_c5(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s * 1.1:
        best_h = normalize_h(best_h)
        final_c5 = get_c5(best_h)

    return best_h, final_c5, n_points
