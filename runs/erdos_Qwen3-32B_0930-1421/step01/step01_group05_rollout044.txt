import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None

    if h is None:
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        h = h.copy()

    dx = 2.0 / n_points

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        return np.clip(clipped * scale, 0, 1)

    def compute_max_overlap(h):
        return np.max(correlate(h, 1 - h, mode='full') * dx)

    best_h = h.copy()
    best_c5 = compute_max_overlap(best_h)

    max_epochs = budget_s // 10  # Estimate safety margin
    learning_rate_start = 0.5

    for epoch in range(max_epochs):
        # Compute overlaps for all k and identify worst-case k
        overlap_all = correlate(h, 1 - h, mode='full') * dx
        k_worst = np.argmax(overlap_all)
        overlap_worst = overlap_all[k_worst]

        # If no room for improvement, break early
        if overlap_worst == best_c5:
            break
        elif overlap_worst < best_c5:
            best_c5 = overlap_worst
            best_h = h.copy()
            continue

        # Compute gradient for worst-case shift
        grad = np.zeros_like(h)
        shift = k_worst

        def shifted_idx(i):
            return (i + shift) % n_points

        for i in range(n_points):
            grad[i] = (1 - h[shifted_idx(i)]) - (2 * h[i] - 1)

        # Backtracking line search
        step = learning_rate_start
        h_proposal = h - step * grad
        h_proposal = normalize_h(h_proposal)
        new_overlap = compute_max_overlap(h_proposal)

        while new_overlap >= overlap_worst and step > 1e-4:
            step *= 0.5
            h_proposal = h - step * grad
            h_proposal = normalize_h(h_proposal)
            new_overlap = compute_max_overlap(h_proposal)

        # Update only if beneficial
        if new_overlap < overlap_worst:
            h = h_proposal
            if new_overlap < best_c5:
                best_c5 = new_overlap
                best_h = h.copy()
        else:
            break  # No improvement, exit early

        # Adaptive learning rate
        learning_rate_start = step * 0.5

        # Budget management
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
