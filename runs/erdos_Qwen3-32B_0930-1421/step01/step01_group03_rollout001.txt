import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, NonlinearConstraint

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 200
    n_points = 47
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, copy=True)
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points // 2):
            h0[i + i] = 2.0 / n_points
        h0 = h0 * (n_points / 2 / h0.sum())

    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points / 2 / h0.sum())
    best_h = h0.reshape(-1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    def rescale(h):
        h = np.clip(h, 0, 1)
        norm = h.sum()
        return h * (n_points / 2 / norm)

    def objective(h_flat):
        h = rescale(h_flat)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)

    bounds = [(0, 1) for _ in range(n_points)]

    # Objective that handles feasibility within the function
    def feasible_obj(h_flat):
        h = rescale(h_flat)
        overlaps = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlaps * dx)
        return max_overlap

    # Tabu set
    tabu_size = 10
    tabu_hist = []

    # Cooling schedule
    delta = 0.1
    no_improve_count = 0
    patience = 50

    while time.time() - start_time < budget_s:
        # Perturb h with tabu check
        idx = np.random.randint(n_points)
        delta_h = np.random.uniform(-delta, delta)
        h_proposal = best_h.copy()
        h_proposal[idx] += delta_h
        h_proposal = rescale(h_proposal)

        # Skip if in tabu history
        if any(np.allclose(h_proposal, t, atol=1e-6) for t in tabu_hist):
            continue

        overlap = np.max(np.correlate(h_proposal, 1 - h_proposal, mode='full') * dx)
        if overlap < best_c5:
            best_h = h_proposal
            best_c5 = overlap
            no_improve_count = 0
        else:
            no_improve_count += 1

        tabu_hist.append(best_h.copy())
        if len(tabu_hist) > tabu_size:
            tabu_hist.pop(0)

        if no_improve_count >= patience:
            delta *= 0.8
            no_improve_count = 0

    # Final descent from best perturbed result
    try:
        result = minimize(
            feasible_obj, best_h, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        h_local = result.x.reshape(n_points)
        c5_local = np.max(np.correlate(h_local, 1 - h_local, mode='full') * dx)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
    except:
        pass

    best_h = np.clip(best_h, 0, 1)
    best_h = rescale(best_h)

    return best_h, best_c5, n_points
