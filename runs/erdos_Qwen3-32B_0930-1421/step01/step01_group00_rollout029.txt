import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Helper: Enforce constraints
    def project_onto_constraints(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        current_sum = optimized_h.sum()
        if abs(current_sum - target_sum) > 1e-8:
            optimized_h *= (target_sum / current_sum)
            optimized_h = np.clip(optimized_h, 0, 1)
        return optimized_h

    # Evaluate C5
    def compute_c5(optimized_h):
        optimized_h = project_onto_constraints(optimized_h)
        ones_minus_h = 1.0 - optimized_h
        full_corr = scipy.signal.correlate(optimized_h, ones_minus_h, mode="full")
        return full_corr.max() * dx

    # Gradient w.r.t. to h for a fixed shift k
    def get_gradient(h, k):
        grad = np.zeros_like(h)
        if k < 1 or k >= n_points:
            return grad
        grad[:n_points - k] -= h[k:]
        grad[k:] -= (1 - h[:n_points - k])
        return grad

    # Main optimization
    best_c5 = compute_c5(h)
    best_h = h.copy()
    last_improvement = start_time
    learning_rate = 0.01

    while time.time() - start_time < budget_s - 5:
        # Compute current C5 and find k with maximum overlap
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()
            last_improvement = time.time()

        ones_minus_h = 1.0 - h
        corr = scipy.signal.correlate(h, ones_minus_h, mode="full")
        k_max = np.argmax(corr)

        # Compute gradient for worst shift k
        grad = get_gradient(h, k_max)

        # Gradient descent step
        h -= learning_rate * grad
        h = project_onto_constraints(h)

        # Perturbation every 100 iterations or if no improvement for 60s
        if (time.time() - last_improvement > 60) or (np.random.rand() < 0.05):
            idx = np.random.choice(n_points, size=min(10, n_points), replace=False)
            noise = np.random.uniform(-0.05, 0.05, size=len(idx))
            h[idx] += noise
            h = project_onto_constraints(h)

        # Early stopping condition
        if time.time() - last_improvement > 120 and best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
