import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    initial_h = initial_h.flatten()
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(initial_h * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement = start_time
    current_c5 = best_c5

    def objective(x):
        # x: convex variable vector
        return np.max(np.correlate(x, 1 - x, mode='full') * (2.0 / n_points))

    for t in range(3):
        # Tuning epsilon step-down schedule over time
        eps = 1.0 - 0.3 * t

        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        objective_cvx = cp.Minimize(cp.max(cp.multiply(h_var, 1 - cp.roll(h_var, shift=1)) * (2.0 / n_points)) + eps * cp.sum_squares(h_var - h))
        prob = cp.Problem(objective_cvx, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, maxiters=150)

        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
        else:
            h_new = h

        # Normalize again
        total_new = np.sum(h_new)
        if total_new != 0:
            h_new = h_new * (n_points / 2) / np.sum(h_new)

        c5_new = evaluate_erdos_solution(h_new, n_points)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = time.time()

        h = h_new

        if time.time() - last_improvement > 300 or time.time() - start_time > budget_s - 15:
            break

    return best_h, best_c5, n_points
