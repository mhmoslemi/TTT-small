import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use provided initial_h_values or fall back to default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    # Optimization parameters
    n_fireflies = 50
    sigma_init = 0.05
    beta0 = 1.0
    alpha = 0.2
    max_iter = 1000
    tolerance = 1e-6

    # Constraint repair: clip and normalize
    def repair(h_in):
        h_clipped = np.clip(h_in, 0, 1)
        sum_h = h_clipped.sum()
        if abs(sum_h) < 1e-10:
            return np.zeros_like(h_in)
        scale = (n_points / 2) / sum_h
        return h_clipped * scale

    # Compute C5
    def compute_c5(h):
        return max(np.correlate(h, 1-h, mode='full') * (2.0 / n_points))

    # Initialize fireflies
    if 'initial_h_values' in globals():
        fireflies = [repair(h + np.random.randn(n_points) * sigma_init) for _ in range(n_fireflies - 1)]
        fireflies.append(h)  # Add the initial solution
    else:
        fireflies = [repair(np.random.rand(n_points)) for _ in range(n_fireflies)]

    best_firefly = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        # Evaluate all fireflies
        firefly_c5 = [(compute_c5(f), f) for f in fireflies]
        firefly_c5.sort()  # Sort by C5 ascending
        updated_fireflies = []

        for i in range(n_fireflies):
            c5_i, fly_i = firefly_c5[i]
            for j in range(i):  # Only move towards brighter
                c5_j, fly_j = firefly_c5[j]
                distance = np.linalg.norm(fly_i - fly_j)
                gamma = (1 / (0.2 * max_iter)) * (time.time() - start_time) / budget_s
                beta = beta0 * np.exp(-gamma * distance ** 2)

                # Move firefly i toward j with random perturbation
                fly_new = fly_i + beta * (fly_j - fly_i) + alpha * (np.random.rand(n_points) - 0.5) * 1e-4
                fly_new = repair(fly_new)
                updated_fireflies.append(fly_new)

        # Append latest generation and trim
        fireflies = updated_fireflies
        if len(fireflies) < n_fireflies:
            fireflies.append(repair(np.random.rand(n_points)))

        # Evaluate and update best
        best_in_gen = min(firefly_c5, key=lambda x: x[0])
        if best_in_gen[0] < best_c5:
            best_c5 = best_in_gen[0]
            best_firefly = best_in_gen[1]

    return best_firefly.tolist(), best_c5, n_points
