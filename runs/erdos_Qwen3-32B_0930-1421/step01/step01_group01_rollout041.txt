import numpy as np
import time
import math
from scipy.signal import correlate


def project_to_simplex(h, n_points):
    """Normalize h to have total mass 1."""
    h = np.clip(h, 0, 1)
    mass = h.sum()
    if mass < 1e-8:
        h[0] = n_points / 2
        h = np.clip(h, 0, 1)
    else:
        h /= mass
        h *= n_points / 2
        h = np.clip(h, 0, 1)
    return h


def evaluate_objective(h, n_points, dx):
    cor = correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx


def perturbed_h(h, amplitude=0.1, segment_length=None):
    n = len(h)
    if segment_length is None:
        segment_length = int(np.random.uniform(1, n // 2))
    start = np.random.randint(0, n - segment_length)
    indices = np.arange(start, start + segment_length)
    delta = amplitude * (2 * np.random.rand(len(indices)) - 1)
    h_perturbed = h.copy()
    h_perturbed[indices] += delta
    return project_to_simplex(h_perturbed, n)


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h[:5] = 0.1
        h[5:42] = 0.3
        h[42:] = 0.2
        h = project_to_simplex(h, n_points)

    T_initial = 0.5
    T = T_initial
    alpha = 0.95
    n_steps = 100
    best_h = h.copy()
    best_C5 = evaluate_objective(best_h, n_points, dx)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for _ in range(n_steps):
            if time.time() - start_time >= budget_s:
                break
            h_proposal = perturbed_h(h)
            C5_proposal = evaluate_objective(h_proposal, n_points, dx)
            C5_current = evaluate_objective(h, n_points, dx)
            delta_C5 = C5_proposal - C5_current

            if delta_C5 < 0 or np.exp(-delta_C5 / T) > np.random.rand():
                h = h_proposal
                if C5_proposal < best_C5:
                    best_h = h.copy()
                    best_C5 = C5_proposal

        T *= alpha
        n_steps = int(min(n_steps * 1.01, 1000))  # Gently increase steps per temperature

    return best_h, best_C5, n_points
