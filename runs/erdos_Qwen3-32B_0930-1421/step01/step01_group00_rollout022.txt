import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    # Use initial H for consistency
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Constraint handling
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    if not np.isclose(sum_h, target_sum):
        h *= target_sum / sum_h
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    if not np.isclose(sum_h, target_sum):
        while sum_h < target_sum:
            idx = np.argmin(h)
            h[idx] += 1e-4
            h = np.clip(h, 0, 1)
            sum_h = h.sum()

    best_h = h.copy()
    best_c5 = 1.0

    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        ones_minus_h = 1.0 - h_vals
        max_corr = -np.inf
        for k in range(1, n_points):
            if k < n_points - 1:
                corr1 = np.dot(h_vals[:n_points - k], ones_minus_h[k:]) * dx
                corr2 = np.dot(h_vals[k:], ones_minus_h[:n_points - k]) * dx
                corr = max(corr1, corr2)
                max_corr = max(max_corr, corr)
        return max_corr

    def get_top_k_offsets(h_vals, top_n=5):
        h_vals = np.clip(h_vals, 0, 1)
        ones_minus_h = 1.0 - h_vals
        corr_values = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            corr1 = np.dot(h_vals[:n_points - k], ones_minus_h[k:]) * dx
            corr2 = np.dot(h_vals[k:], ones_minus_h[:n_points - k]) * dx
            corr_values.append((k, max(corr1, corr2)))
        return sorted(corr_values, key=lambda x: -x[1])[:top_n]

    while time.time() - start_time < budget_s - 20:
        # Get the top problematic k shifts
        top_k_offsets = get_top_k_offsets(h)
        if not top_k_offsets:
            break

        # Copy current h and propose changes
        candidate_h = h.copy()

        for k, _ in top_k_offsets:
            if k >= n_points or k == 0:
                continue
            indices = []
            for i in range(n_points - k):
                val = h[i] * (1.0 - h[i + k])
                indices.append((i, i + k, val))

            # Sort indices by contribution
            indices.sort(key=lambda x: x[2], reverse=True)
            top_pct = int(0.1 * len(indices))
            top_indices = indices[:top_pct]

            # Try both: decrease h[i], increase h[i + k]
            for i, j, _ in top_indices:
                if h[i] > 0 and candidate_h[j] < 1:
                    # Decrease h[i], increase h[j]
                    shift = min(0.01, h[i])
                    if candidate_h[i] > 1e-7:
                        candidate_h[i] -= shift
                        candidate_h[j] = min(1.0, candidate_h[j] + shift)
                elif h[i] < 1 and candidate_h[j] > 0:
                    # Increase h[i], decrease h[j]
                    shift = min(0.01, 1.0 - h[i])
                    candidate_h[i] = min(1.0, candidate_h[i] + shift)
                    candidate_h[j] = max(0.0, candidate_h[j] - shift)

        # Constraint handling
        candidate_h_sum = candidate_h.sum()
        if not np.isclose(candidate_h_sum, target_sum):
            candidate_h *= target_sum / candidate_h_sum
        candidate_h = np.clip(candidate_h, 0, 1)

        new_c5 = compute_c5(candidate_h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = candidate_h[:]
            if best_c5 < 0.380876:
                break

        if time.time() - start_time > budget_s - 5:
            break

        # Check performance and early stopping
        if best_c5 < 0.380876:
            break

    return (best_h, best_c5, n_points)
