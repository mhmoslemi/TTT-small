import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible_convex(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h[:] = 0.0
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def binary_project(h):
    # Project approximately binary solution to exact binary values while maintaining sum constraint
    h_proj = np.zeros_like(h)
    n_points = len(h)
    mid = n_points / 2
    # Sort indices by increasing distance from 0.5
    indices = np.argsort(np.abs(h - 0.5))
    h_bin = np.round(h.copy())

    def get_total(h_):
        return h_.sum()

    # Initial binary solution
    best_h = h_bin.copy()
    best_diff = abs(get_total(best_h) - mid)
    # Try alternatives by flipping one value at a time
    for i in indices:
        alt_h = h_bin.copy()
        alt_h[i] = 1 - alt_h[i]
        diff = abs(get_total(alt_h) - mid)
        if diff < best_diff:
            best_h = alt_h
            best_diff = diff
            if best_diff == 0:
                break
    h_proj[:] = best_h
    return h_proj


def run(budget_s=1100, n_points=42, init_constraints=5):
    start_time = time.time()

    # 1. Initialize
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h[:] = initial_h_values
    else:
        h = np.random.rand(n_points)
    h = ensure_feasible_convex(h)
    init_C5 = evaluate_C5(h)

    # Constraints: 0 <= h <= 1
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [h_var <= 1.0, h_var >= 0.0]

    # Constraint: sum(h) == n_points / 2
    constraints += [cp.sum(h_var) == n_points / 2]

    # Slack variables for bilinear terms
    slack_vars = dict()

    # 2. Cutting-plane iteration
    cuts = []
    for k in range(0, n_points - 1):
        if len(cuts) >= init_constraints:
            break
        # Manually add an initial cut for shift k
        cuts.append(k)

    h_values = cp.Parameter(n_points, nonneg=True)
    h_values.value = h
    best_h = h.copy()
    best_C5 = init_C5

    while time.time() - start_time < budget_s:
        # Build current optimization problem
        objective = cp.Minimize(1.0)
        constraints_active = constraints.copy()

        for k in cuts:
            # Compute overlap integral for shift k: sum(h_i * (1 - h_{i + k}) * dx)
            # Create linear approximation for slack at current h
            H = np.zeros((n_points, n_points))
            for i in range(n_points):
                if i + k < n_points:
                    H[i, i + k] = 1.0
            overlap = h_var @ (H @ (1 - h_var))
            dx = 2.0 / n_points
            constraints_active.append(overlap <= 1.0 * dx)

        # Solve convex relaxation
        prob = cp.Problem(objective, constraints_active)
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        h_relaxed = h_var.value

        if h_relaxed is None:
            break

        h_relaxed = ensure_feasible_convex(h_relaxed)

        # Compute C5 after fixing bounds and maintaining the constraint
        C5_relaxed = evaluate_C5(h_relaxed)
        if C5_relaxed < best_C5:
            best_h = h_relaxed.copy()
            best_C5 = C5_relaxed
            print(f"New best C5: {best_C5}")

        # Repair solution to feasible binary values
        h_plus = binary_project(h_relaxed.copy())
        C5_plus = evaluate_C5(h_plus)
        if C5_plus < best_C5:
            best_h = h_plus.copy()
            best_C5 = C5_plus
            print(f"Binary repair improves to C5: {best_C5}")

        # Add new cut based on actual constraint violation
        for k in range(n_points - 1):
            if k in cuts:
                continue
            actual_overlap = evaluate_C5(h_relaxed)
            if actual_overlap > best_C5 * 1e-6:
                cuts.append(k)
                break

    return best_h, best_C5, n_points
