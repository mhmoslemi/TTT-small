import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        return np.clip(clipped * scale, 0, 1)

    # Initialize firefly population
    n_fireflies = 50
    population = []
    if 'initial_h_values' in globals():
        population.append(normalize_h(np.array(initial_h_values)))
        for _ in range(n_fireflies - 1):
            h_rand = np.random.rand(n_points) + 0.05 * np.random.randn(n_points)
            population.append(normalize_h(h_rand))
    else:
        for _ in range(n_fireflies):
            h_rand = np.random.rand(n_points)
            population.append(normalize_h(h_rand))

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Firefly algorithm parameters
    beta0 = 1.0
    alpha = 0.2
    gamma_factor = 10.0 / budget_s  # Time-dependent scaling for gamma

    while time.time() - start_time < budget_s:
        # Evaluate each firefly
        evaluations = [(compute_c5(f), f) for f in population]
        evaluations.sort()
        population = [f for _, f in evaluations]

        for i in range(n_fireflies):
            for j in range(i):
                f1 = population[i]
                f2 = population[j]
                c1, c2 = evaluations[i][0], evaluations[j][0]
                if c1 < c2:
                    dist = np.sum((f2 - f1) ** 2)
                    gamma = gamma_factor * (time.time() - start_time)
                    beta = beta0 * np.exp(-gamma * dist)
                    step = beta * (f2 - f1) + alpha * (np.random.rand(n_points) - 0.5) * 1e-3
                    new_f = f1 + step
                    new_f = normalize_h(new_f)
                    if compute_c5(new_f) < c1:
                        population[i] = new_f
                        if compute_c5(new_f) < best_c5:
                            best_h = new_f.copy()
                            best_c5 = compute_c5(new_f)

    # Final best evaluation
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_points
