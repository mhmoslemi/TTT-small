import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def perturb_h(h, sigma, n_perturb, dx):
    h_new = h.copy()
    indices = np.random.choice(len(h), size=min(n_perturb, len(h)), replace=False)
    noise = np.random.normal(0, sigma, size=len(indices))
    h_new[indices] += noise
    h_new = np.clip(h_new, 0, 1)
    # Normalize to maintain the constraint sum(h) = n_points / 2
    mass = np.sum(h_new)
    if abs(mass - len(h) / 2) > 1e-6:
        scale = (len(h) / 2) / mass
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)  # Clip again since scaling could push values out of bounds
    return h_new

            h = h_new

                best_h = h.copy()
                best_c5 = c5_new
                no_improvement = 0
                if sigma > 0.05:
                    sigma *= 0.995

                no_improvement += 1

        if delta < 0:
            T = max(T * alpha, 1e-5)

            T = max(T * 1.001, 1e-5)

        if no_improvement >= max_no_improvement:
            T = max(1.0, T * 1.1)

    return (best_h, final_c5, n_points)
