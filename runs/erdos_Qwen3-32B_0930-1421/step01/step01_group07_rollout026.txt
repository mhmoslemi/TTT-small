import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_bits(h):
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # Prefer local flips
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new
