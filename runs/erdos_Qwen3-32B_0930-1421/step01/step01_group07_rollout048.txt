import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, peak_idx, reverse_peak_idx):
    h_new = h.copy()
    one_indices = np.flatnonzero(h == 1)
    zero_indices = np.flatnonzero(h == 0)
    
    near_peak_indices = np.abs(one_indices - peak_idx)
    flip_from_idx = one_indices[near_peak_indices.argmin()]
    
    near_reverse_peak_indices = np.abs(zero_indices - reverse_peak_idx)
    flip_to_idx = zero_indices[near_reverse_peak_indices.argmin()]
    
    h_new[flip_from_idx] = 0
    h_new[flip_to_idx] = 1
    return h_new
