import numpy as np
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(h * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    def objective(x):
        return evaluate_erdos_solution(x, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    last_improvement = start_time

    T0 = 1.0
    T = 1.0
    alpha = 0.995

    # Binary initialization
    total = int(np.rint(n_points / 2))
    h_bin = np.zeros(n_points)
    chosen = np.random.choice(n_points, total, replace=False)
    h_bin[chosen] = 1.0

    # Simulated Annealing step for binary
    num_steps = 0
    end_anneal = time.time() + budget_s * 0.45  # 45% of total for annealing
    while time.time() < end_anneal and num_steps < 10000:
        num_steps += 1
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_new = h_bin.copy()
        h_new[idx1] ^= 1
        h_new[idx2] ^= 1

        if np.sum(h_new) < 0 or np.sum(h_new) > h_bin.shape[0]:
            continue

        c5_new = evaluate_erdos_solution(h_new, n_points)
        c5_old = evaluate_erdos_solution(h_bin, n_points)

        if c5_new < best_c5 or np.exp(min((c5_old - c5_new) / T, 700)) > np.random.rand():
            h_bin[:] = h_new[:]
            best_c5 = min(best_c5, c5_new)
            last_improvement = time.time()

        T *= alpha

        if time.time() - last_improvement > 180:  # stop if no improvement
            break

    # Apply binary initialization as starting point in continuous optimization
    total_bin = np.sum(h_bin)
    target_total = n_points / 2.0

    if total_bin < target_total:
        diff = int(np.round(target_total - total_bin))
        idxs = np.where(h_bin == 0)[0]
        h_bin[idxs[:diff]] = 1.0
    elif total_bin > target_total:
        diff = int(np.round(total_bin - target_total))
        idxs = np.where(h_bin == 1)[0]
        h_bin[idxs[:diff]] = 0.0

    # Linear relaxation
    h_relax = h_bin.copy()
    delta = target_total - np.sum(h_relax)
    if abs(delta) > 1e-6:
        idxs = np.argwhere(h_relax == 0).flatten() if delta > 0 else np.argwhere(h_relax == 1).flatten()
        idx = idxs[0]
        h_relax[idx] += delta

    # Ensure the vector is valid input
    h_relax = np.clip(h_relax, 0, 1)

    # Continuous optimization
    x0 = h_relax
    cons = (
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_upper},
        {'type': 'ineq', 'fun': constraint_lower}
    )

    def callback(x):
        nonlocal best_c5, best_h, last_improvement
        c5_new = evaluate_erdos_solution(x, n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h[:] = x[:]
            last_improvement = time.time()

    end_optimize = time.time() + budget_s * 0.55  # Avoid exceeding budget
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n_points,
        constraints=cons,
        options={'maxiter': 1000},
        callback=callback
    )

    if result.success:
        final_h = result.x
        final_c5 = evaluate_erdos_solution(final_h, n_points)
        best_c5 = min(best_c5, final_c5)
        best_h = final_h

    return best_h, best_c5, n_points
