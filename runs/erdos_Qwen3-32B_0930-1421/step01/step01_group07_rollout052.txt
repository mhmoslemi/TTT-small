import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project(h, n):
    # Enforce 0 <= h[i] <= 1
    h_clipped = np.clip(h, 0, 1)
    # Ensure sum(h_clipped) == n / 2
    s = np.sum(h_clipped)
    if not np.isclose(s, n / 2):
        h_clipped *= (n / 2) / s
    return h_clipped

def compute_gradient(h, k_shift, n):
    lag = k_shift - (n - 1)
    grad = np.zeros_like(h)
    for i in range(n):
        if i + lag < n:
            grad[i] += (1 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    return grad

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Project to constraints
    h = project(h, n_points)

    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    alpha = 1e-3
    alpha_max = 1e-1
    alpha_min = 1e-6
    t0 = time.time()
    iter_count = 0
    max_iter = 10000
    perturb_interval = 200
    grad_tol = 1e-6

    while time.time() - t0 < budget_s and iter_count < max_iter:
        corr = correlate(current_h, 1 - current_h, mode="full")
        k_shift = np.argmax(corr)
        lag = k_shift - (n_points - 1)
        grad = compute_gradient(current_h, k_shift, n_points)
        norm_grad = np.linalg.norm(grad + 1e-8)  # avoid division by zero
        grad_dir = grad / norm_grad

        # Line search using backtracking
        alpha_test = alpha
        c5_test = evaluate_overlap(current_h - alpha_test * grad_dir)
        c5_current = evaluate_overlap(current_h)
        beta_search = 0.5
        while c5_test > c5_current and alpha_test > alpha_min:
            alpha_test *= beta_search
            c5_test = evaluate_overlap(current_h - alpha_test * grad_dir)

        # Accepting best step
        new_h = current_h - alpha_test * grad_dir
        new_h = project(new_h, n_points)
        c5_new = evaluate_overlap(new_h)

        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            alpha = min(alpha_max, alpha_test * 1.1)
        else:
            alpha = max(alpha_min, alpha_test * 0.5)

        current_h[:] = new_h[:]
        current_c5 = c5_new
        iter_count += 1

        # Perturbation to avoid local minima
        if iter_count % perturb_interval == 0 and np.random.rand() < 0.2:
            eps = 0.05
            perturbation = np.random.uniform(-eps, eps, n_points)
            perturbed_h = project(current_h + perturbation, n_points)
            c5_perturb = evaluate_overlap(perturbed_h)
            if c5_perturb < current_c5:
                current_h[:] = perturbed_h[:]
                current_c5 = c5_perturb

        # Early stopping if too small improvement
        if current_c5 < grad_tol:
            break

    # Final validation
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
    return (best_h, final_c5, n_points)
