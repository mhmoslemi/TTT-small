import numpy as np
import time
from scipy.signal import correlate

def perturb_h(h, noise_level=1e-2):
    h = h.copy()
    noise = np.random.randn(len(h)) * noise_level
    h += noise
    h = np.clip(h, 0, 1)
    # Enforce sum constraint
    weight_sum = h.sum()
    if abs(weight_sum - len(h)/2) > 1e-4:
        h *= (len(h)/2) / weight_sum
    return h

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx
