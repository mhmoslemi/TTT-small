import numpy as np
import time
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    n_points = len(h)
    if total < 1e-8:
        h[:] = 0.0
        h[0::2] = 1.0
        h = ensure_feasible(h)
        return h
    scaling_factor = (n_points / 2) / total
    h *= scaling_factor
    return h


def crossover(parent1, parent2):
    n_points = len(parent1)
    cross_point = np.random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
    return ensure_feasible(child)


def mutate(h, mut_rate=0.2):
    n_points = len(h)
    mutation_mask = np.random.rand(n_points) < mut_rate
    if not np.any(mutation_mask):
        return np.copy(h)
    noisy = np.random.uniform(0, 1, size=n_points)
    h[mutation_mask] = noisy[mutation_mask]
    return ensure_feasible(h)


def tournament_selection(population, scores, size=3):
    participants = np.random.choice(len(population), size)
    indices = np.argsort([scores[i] for i in participants])
    return np.copy(population[participants[indices[0]]])


def generate_population(pop_size, n_points, init_h=None):
    population = []
    for _ in range(pop_size):
        if init_h is not None and np.random.rand() < 0.25:
            h = np.random.permutation(init_h)
        else:
            h = np.random.rand(n_points)
        h = ensure_feasible(h)
        population.append(h)
    return population


def diversify_population(population, n_points, max_attempts=5):
    new_additions = []
    attempts = 0
    while len(population) + len(new_additions) < len(population) + max_attempts:
        h = np.random.rand(n_points)
        h = ensure_feasible(h)
        if not np.any([np.allclose(h, p, atol=1e-4) for p in population + new_additions]):
            new_additions.append(h)
        else:
            attempts += 1
            if attempts > max_attempts:
                break
    return population + new_additions


def run(budget_s=1100, pop_size=40, n_points=42, init_iter=0, **kwargs):
    start_time = time.time()
    if initial_h_values is not None:
        h_start = initial_h_values.copy()
    else:
        h_start = np.ones(n_points) / 2
    h_start = np.copy(ensure_feasible(h_start))
    population = generate_population(pop_size, n_points, init_h=h_start)
    best_h = np.copy(h_start)
    best_c5 = evaluate_C5(best_h)

    cooldown = 100
    while time.time() - start_time < budget_s:
        if init_iter >= 3:
            population = diversify_population(population, n_points)
        scores = [evaluate_C5(ind) for ind in population]
        current_best_idx = np.argmin(scores)
        if scores[current_best_idx] < best_c5:
            best_h = np.copy(population[current_best_idx])
            best_c5 = scores[current_best_idx]

        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population, scores)
            parent2 = tournament_selection(population, scores)
            child = crossover(parent1, parent2)
            child = mutate(child, mut_rate=0.2 + 0.2 * np.exp(-init_iter / 10))
            if np.random.rand() < 0.05:
                child = np.random.rand(n_points)
                child = ensure_feasible(child)
            new_population.append(child)
        population = new_population
        init_iter += 1
        cooldown = max(0, cooldown - 1)
        if cooldown == 0 and init_iter % 16 == 0:
            population = diversify_population(population, n_points)

    return best_h, best_c5, n_points
