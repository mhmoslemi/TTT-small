import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def generate_mccormick_constraints(h, k, dx):
    n = len(h)
    # Bilinear terms h_i * (1 - h_{i+k})
    i_vals = np.arange(n - abs(k))
    j_vals = i_vals + abs(k)
    h = cp.Parameter(n)
    bilinear = h[i_vals] * (1 - h[j_vals])
    w = cp.Variable(len(i_vals))
    for idx in range(len(i_vals)):
        con1 = w[idx] >= 0
        con2 = w[idx] <= 1
        con3 = w[idx] <= h[i_vals[idx]]
        con4 = w[idx] <= 1 - h[j_vals[idx]]
        con5 = w[idx] >= h[i_vals[idx]] + 1 - h[j_vals[idx]] - 1
        yield con1, con2, con3, con4, con5
    return w


def repair_to_binary(h, dx):
    h_repaired = np.zeros_like(h)
    values = np.sort(h)
    cutoff = values[-int(np.floor(len(h) / 2))]
    h_repaired[h > cutoff] = 1.0
    if np.sum(h_repaired) < len(h) / 2:
        idx = np.argsort(h_repaired)
        h_repaired[idx[-1]] = 1.0
    elif np.sum(h_repaired) > len(h) / 2:
        idx = np.argsort(h_repaired)
        h_repaired[idx[0]] = 0.0
    return h_repaired


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    dx = 2.0 / n_points

    # Initialize h and λ
    if initial_h_values is not None:
        h0 = initial_h_values.copy()
    else:
        h0 = np.zeros(n_points)
        h0[:n_points // 2] = 1.0
    h0 = ensure_feasible(h0)

    # Optimization setup
    h = cp.Parameter(n_points)
    lambda_var = cp.Variable()
    objective = cp.Minimize(lambda_var)
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ]
    active_shifts = [1]  # Start with just one shift

    best_h = np.copy(h0)
    best_C5 = evaluate_C5(best_h)
    best_iter = 0
    patience_counter = 0
    patience_limit = 2

    while time.time() - start_time < budget_s and patience_counter < patience_limit:
        # Add McCormick constraints for active shifts
        bilinear_vars = []
        bilinear_indices = []
        bilinear_constraints = []

        for k in active_shifts:
            valid = (k > 0) and (k < n_points)
            if not valid:
                continue
            i_vals = np.arange(n_points - k)
            j_vals = i_vals + k
            for idx, (i, j) in enumerate(zip(i_vals, j_vals)):
                if i >= n_points or j >= n_points:
                    continue
                bilinear_vars.append((i, j))
                bilinear_indices.append(idx)
                bilinear_constraints.extend(
                    generate_mccormick_constraints(h, k, dx))
            # Projected sum of bilinear terms is ≤ λ * (n_points / 2)
            constraints.append(
                cp.sum(sum(bilinear_vars for bilinear_vars in bilinear_vars)) <= lambda_var * (n_points / 2)
            )

        # Solve convex problem
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver='ECOS', verbose=False)
            h_sol = np.array(h.value)
            h_sol = ensure_feasible(h_sol)
            current_C5 = evaluate_C5(h_sol)

            # Update best solution
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_sol.copy()
                patience_counter = 0
            else:
                patience_counter += 1

            # Binary patch for better results
            h_repaired = repair_to_binary(h_sol, dx)
            h_repaired = ensure_feasible(h_repaired)
            current_C5_repair = evaluate_C5(h_repaired)
            if current_C5_repair < best_C5:
                best_C5 = current_C5_repair
                best_h = h_repaired.copy()
                patience_counter = 0

            # Add new constraint to tighten relaxation
            corr = correlate(h_sol, 1 - h_sol, mode="full") * dx
            worst_k = int(np.argmax(corr))
            if worst_k not in active_shifts and 0 < worst_k <= n_points - 1:
                active_shifts.append(worst_k)
                if len(active_shifts) > n_points:
                    active_shifts.pop(0)

        except Exception as e:
            pass  # Skip on solver failure

    return best_h, best_C5, n_points
