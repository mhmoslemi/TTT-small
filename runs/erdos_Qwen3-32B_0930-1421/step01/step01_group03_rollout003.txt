import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (target_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair_h(h_vec, target_sum=target_sum):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return np.zeros_like(h)
        h = h * (target_sum / h_sum)
        return h

    def optimize_h(initial_h, max_iter=300, method='L-BFGS-B'):
        h_arr = initial_h.copy()
        bounds = [(0, 1)] * n_points

        def obj_func(h_flat):
            h = repair_h(h_flat)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            obj_func,
            h_arr,
            method=method,
            bounds=bounds,
            options={'maxiter': max_iter}
        )
        h_opt = repair_h(result.x)
        return h_opt

    def perturb_h(h_vec, noise_factor=0.1):
        h = h_vec.reshape(n_points)
        noise = np.random.randn(n_points) * noise_factor
        h = h + noise
        h = repair_h(h)
        return h

    tabu_configurations = []
    MAX_TABU = 10

    best_h = repair_h(h_init)
    best_score = objective(best_h)

    # Primary loop
    current_h = best_h.copy()
    current_score = best_score
    perturb_step = 0.1
    cool_factor = 0.8
    no_improvement_count = 0
    MAX_NO_IMPROVEMENT = 50

    while time.time() - start_time < budget_s * 0.98:
        # Exploration via perturbations
        for _ in range(5):
            temp_h = perturb_h(current_h, noise_factor=perturb_step)
            temp_h_tuple = tuple(temp_h.tolist())
            if temp_h_tuple not in tabu_configurations:
                opt_h = optimize_h(temp_h)
                opt_score = objective(opt_h)
                if opt_score < best_score:
                    best_h = opt_h.copy()
                    best_score = opt_score
                    tabu_configurations.append(temp_h_tuple)
                    if len(tabu_configurations) > MAX_TABU:
                        tabu_configurations.pop(0)
                    no_improvement_count = 0
                current_h = opt_h.copy()
                current_score = opt_score
            if time.time() - start_time > budget_s * 0.98:
                break

        if no_improvement_count >= MAX_NO_IMPROVEMENT:
            perturb_step *= cool_factor
            no_improvement_count = 0

        no_improvement_count += 1

    best_h = repair_h(best_h)
    final_score = objective(best_h)

    if 'initial_h_values' in globals():
        check_h = np.array(initial_h_values, copy=True)
        assert np.allclose(check_h, best_h) == False, "Failed to escape initial solution."

    elapsed = time.time() - start_time
    residual_time = budget_s - elapsed
    if residual_time > 0:
        final_h = optimize_h(best_h, max_iter=100)
        final_h = repair_h(final_h)
        final_score = objective(final_h)
        return final_h, final_score, n_points
    return best_h, objective(best_h), n_points
