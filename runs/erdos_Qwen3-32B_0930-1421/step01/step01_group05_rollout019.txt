import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial_h_values as fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(45, dtype=np.float64)
        h[0] = 1.0

    n_points = len(h)
    dx = 2.0 / n_points

    def normalize_h(h_in):
        h_clipped = np.clip(h_in, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return h_in * 0.0
        return h_clipped * (n_points / 2) / h_sum

    def compute_c5(h_vec):
        h = normalize_h(h_vec)
        corr = correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: ILP relaxation for assignment using binary variables
    # Converting problem into a convex relaxation over real h_i ∈ [0,1]
    max_iters = 100
    time_threshold = 0.05

    for _ in range(max_iters):
        if time.time() - start_time > budget_s - time_threshold:
            break

        # Define optimization variable
        h_var = cp.Variable(n_points, nonneg=True)

        # Define objective: minimize max correlation over shifts
        # Since the exact non-convex maximum over shifts is hard, we use
        # a convex relaxation approach by restricting to a number of key shifts
        n_shifts = 3  # Limit to key overlap positions
        max_corr = []

        for k in range(n_shifts):
            a = np.roll(h, -k)
            a = np.clip(a, 0, 1)  # Ensure values in [0,1] after shifting
            corr_term = cp.sum(h_var * (1 - a))
            max_corr.append(corr_term)

        # Max over the several terms we care most about
        obj = cp.Minimize(cp.Max(max_corr))

        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1.0,
            h_var >= 0.0
        ]

        # Solve problem
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.SCIP)
        except:
            break

        # Check solver status
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(h_var.value, 0, 1).copy()
            new_h = normalize_h(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
    else:
        # Update time remaining to explore further
        remaining_time = max(budget_s - (time.time() - start_time), 0)
        if remaining_time > 1:
            # Phase 2: Post-solver gradient descent
            learning_rate = 0.05
            while time.time() - start_time < budget_s:
                grad = np.zeros(n_points)
                eps = 1e-5

                for i in range(n_points):
                    h_p = best_h.copy()
                    h_p[i] += eps
                    h_p = normalize_h(h_p)
                    c_p = compute_c5(h_p)
                    h_n = best_h.copy()
                    h_n[i] -= eps
                    h_n = normalize_h(h_n)
                    c_n = compute_c5(h_n)
                    grad[i] = (c_p - c_n) / (2 * eps)

                best_h -= learning_rate * grad
                best_h = normalize_h(best_h)
                best_c5 = compute_c5(best_h)

                learning_rate *= 0.999

                if time.time() - start_time >= budget_s:
                    break

    return best_h.tolist(), best_c5, n_points
