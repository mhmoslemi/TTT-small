import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
            corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)
    return np.max(np.array(corrs) * dx)

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def objective_smooth(h_vec, n_points, beta=100):
    h = h_vec.reshape(-1)
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        if n_points - k > 0:
            corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
            corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k > 0:
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)
    return smooth_max(np.array(corrs) * dx, beta)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def create_raised_cosine(n_points):
    x = np.linspace(0, 2, n_points)
    center = n_points // 2
    width = n_points // 3
    h = np.zeros(n_points)
    for i in range(n_points):
        if abs(i - center) <= width // 2:
            h[i] = 0.5 + 0.5 * np.cos(np.pi * abs(i - center) / (width // 2 + 1e-9))
        if abs(i - center) > width // 2 and abs(i - center) < width // 2 + 4:
            h[i] = 0.0
    return enforce_normalization(h, n_points)

def pso_objective(h_vec, n_points, alpha=100):
    h = h_vec.reshape(-1)
    c5 = compute_c5(h, n_points)
    barrier = -np.sum(np.log(h + 1e-3) + np.log(1.0 - h + 1e-3))
    sum_error = np.sum(h) / n_points * 2 - 1
    return c5 + alpha * (barrier + 1e4 * sum_error**2)

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is None:
        h0 = np.array(create_raised_cosine(47))
    else:
        h0 = init_h.copy()

    n_points = len(h0)
    dx = 2.0 / n_points

    # PSO Setup
    n_particles = 20
    bounds = [(0.0, 1.0)] * n_points
    particles = [enforce_normalization(h0 + np.random.normal(0, 0.1, n_points), n_points) for _ in range(n_particles)]
    velocities = [np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    pbest = [p for p in particles]
    pb_scores = [float('inf')] * n_particles
    gbest = h0.copy()
    gb_score = float('inf')

    time_start = time.time()
    while time.time() - time_start < budget_s:
        for i in range(n_particles):
            p = particles[i]
            v = velocities[i]
            score = compute_c5(p, n_points)

            # Update pbest
            if score < pb_scores[i]:
                pbest[i] = p.copy()
                pb_scores[i] = score

            # Update gbest
            if score < gb_score:
                gbest = p.copy()
                gb_score = score

            # Update velocity and position (PSO update)
            w = 0.7 - 0.2 * ((time.time() - time_start) / budget_s)
            c1, c2 = 2.0, 2.0
            random1 = np.random.rand(n_points)
            random2 = np.random.rand(n_points)
            new_velocity = (
                w * v +
                c1 * random1 * (pbest[i] - p) +
                c2 * random2 * (gbest - p)
            )
            velocities[i] = np.clip(new_velocity, -0.5, 0.5)
            new_position = p + velocities[i]
            new_position = enforce_normalization(new_position, n_points)
            particles[i] = new_position

            if time.time() - time_start > budget_s:
                break

    # Final refinement using L-BFGS-B on the global best
    def c5_func(x):
        h = x.reshape(-1)
        return compute_c5(h, n_points)

    h_best = gbest
    result = opt.minimize(
        c5_func,
        h_best,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50}
    )
    h_best = result.x
    h_best = enforce_normalization(h_best, n_points)

    final_c5 = compute_c5(h_best, n_points)
    return h_best, final_c5, n_points
