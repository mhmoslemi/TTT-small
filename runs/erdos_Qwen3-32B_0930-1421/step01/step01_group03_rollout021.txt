import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def construct_uniform(n):
    h = np.zeros(n)
    h[::2] = 2.0 / n
    h *= n / 2 / h.sum()
    return h

def compute_c5(h, dx):
    h = np.clip(h, 0, 1)
    h /= h.sum() * dx  # Normalize to satisfy integral constraint
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

def objective_cvx(h, n_points):
    h = h.reshape(n_points)
    h = cp.Parameter(n_points, value=h)

    # Convex relaxation for fixed support
    w = cp.Variable(n_points, nonneg=True)
    constraints = [
        h >= w,
        w >= 0,  # Ensure 0 <= w <= h[i]
        sum(w) == h.sum()  # conserve sum_invariant
    ]

    # C5 aware objective
    def correlation_overlap_cost(w_vals):
        w_arr = w_vals.reshape(n_points)
        corr = np.correlate(w_arr, 1 - w_arr, mode='full') * dx
        return np.max(corr)

    prob = cp.Problem(
        cp.Minimize(correlation_overlap_cost(w)),
        constraints
    )
    prob.solve(solver=cp.CPLEX, verbose=False)

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None

    w_opt = w.value
    return w_opt.clip(0, 1)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    time_remaining = budget_s

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_opt = np.array(initial_h_values, dtype=np.float64, copy=True)
    else:
        n_points = 100
        h_opt = construct_uniform(n_points)

    dx = 2.0 / n_points
    best_c5 = compute_c5(h_opt, dx)

    # Fix support regions
    fixed_support = np.isclose(h_opt, 0) | np.isclose(h_opt, 1)
    active_idx = np.flatnonzero(~fixed_support)

    def refine_h(h_current, active_idx):
        # Use convex refinement on active positions
        h_refined = h_current.copy()
        active_h = h_refined[active_idx]
        w_opt = objective_cvx(active_h, len(active_h))
        if w_opt is not None:
            h_refined[active_idx] = w_opt
        return h_refined

    def perturb_support(h, active_idx, noise_factor=0.1):
        h = h.copy()
        active_set = active_idx.copy()
        np.random.shuffle(active_set)
        n_perturb = max(1, len(active_set) // 5)

        # Swap some support positions
        for _ in range(n_perturb):
            i, j = np.random.choice(len(active_set), size=2, replace=False)
            h[active_set[i]], h[active_set[j]] = h[active_set[j]], h[active_set[i]]
        return h

    # Main optimization loop
    best_h = h_opt.copy()
    best_c5 = compute_c5(best_h, dx)
    improvement_steps = 0

    while time.time() - start_time < budget_s * 0.9:
        # Perturb and refine
        h_pert = perturb_support(best_h, active_idx, noise_factor=0.02)
        h_pert = h_pert.clip(0, 1)
        h_pert[active_idx] = refine_h(h_pert, active_idx)
        h_pert = h_pert.clip(0, 1)

        # Satisfy integral constraint
        h_int = h_pert / h_pert.sum() * (n_points / 2)

        c5 = compute_c5(h_int, dx)

        if c5 < best_c5 - 1e-6:
            best_c5 = c5
            best_h = h_int
            improvement_steps += 1
        elif c5 < best_c5 and np.random.rand() < 0.5:
            best_c5 = c5
            best_h = h_int

        if time.time() - start_time > budget_s * 0.98:
            break

    best_h.clip(0, 1, out=best_h)
    best_h /= best_h.sum()
    best_h *= n_points / 2
    best_c5_post = compute_c5(best_h, dx)

    return best_h, best_c5_post, n_points
