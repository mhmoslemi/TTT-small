import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def perturb_h(h, sigma, n_points):
    h_new = h.copy()
    n_flip = np.random.randint(5, 11)
    indices = np.random.choice(n_points, size=n_flip, replace=False)
    h_new[indices] += np.random.normal(0, sigma, size=n_flip)
    h_new = np.clip(h_new, 0, 1)
    # Enforce sum constraint
    delta = np.sum(h_new) - n_points / 2
    h_new -= delta / n_points
    h_new = np.clip(h_new, 0, 1)
    return h_new
