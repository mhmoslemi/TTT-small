import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # Keep n small for faster evaluation

    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        n_points = len(h_base)
    else:
        n_points = 47
        # Basic uniform construction with one unit per 2 bins
        h_base = np.zeros(n_points)
        for i in range(n_points // 2):
            h_base[i * 2] = 2.0 / n_points
        h_base *= n_points / 2 / h_base.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        h = h * (constraint_sum / h_sum)  # Normalize to meet constraint
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair_h(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h.flatten()

    def random_rectified(n, p=5):
        h = np.zeros(n)
        indices = np.sort(np.random.choice(n, size=p, replace=False))
        weights = np.random.rand(p)
        h[indices] = weights
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    def crossover(h1, h2):
        mask = np.random.rand(*h1.shape) <= 0.5
        return np.where(mask, h1, h2)

    def mutate(h, noise_factor=0.15):
        noise = np.random.randn(*h.shape) * noise_factor
        return (h + noise).clip(0, 1)

    def optimize(h0, max_iter=500):
        x0 = h0.copy().flatten()
        bounds = [(0, 1)] * n_points

        def obj_func(x):
            h = repair_h(x).reshape(n_points)
            return objective(h)

        result = minimize(obj_func, x0=x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        return repair_h(result.x)

    # Initialize population
    if 'initial_h_values' in globals():
        population = [h_base]
        for _ in range(9):  # add 9 mutants
            population.append(repair_h(mutate(h_base)))
    else:
        population = [random_rectified(n_points) for _ in range(10)]

    best_h = None
    best_c5 = float('inf')

    generation = 0
    while time.time() - start_time < budget_s * 0.95:
        fitness = [objective(ind) for ind in population]
        sorted_pop = [p for _, p in sorted(zip(fitness, population))]

        survivors = sorted_pop[:max(2, len(population) // 2)]

        if len(survivors) > 0:
            curr_best = survivors[0]
            curr_c5 = fitness[0]
            if curr_c5 < best_c5:
                best_h = curr_best
                best_c5 = curr_c5

        if len(survivors) == 1:
            next_gen = [mutate(survivors[0])] * 10
        else:
            next_gen = []
            for _ in range(10):
                p1, p2 = np.random.choice(survivors, 2, replace=False)
                o = crossover(p1, p2)
                o = repair_h(mutate(o))
                next_gen.append(o)

        for i, indiv in enumerate(next_gen):
            if time.time() - start_time > budget_s * 0.95:
                break
            if i % 2 == 0:
                indiv = optimize(indiv)
            next_gen[i] = indiv

        population = survivors + next_gen
        population = [repair_h(np.array(p)) for p in population]
        generation += 1

    # Final polishing
    if best_h is not None:
        best_h = optimize(best_h, max_iter=1000)
        best_c5 = objective(best_h)
    else:
        best_h = optimize(population[0])

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h.tolist(), objective(best_h), n_points
