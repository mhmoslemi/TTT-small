import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def compute_overlap_corrs(h):
    n = len(h)
    h1 = 1.0 - h
    return correlate(h, h1, mode="full")

def find_worst_shift(corr):
    return np.argmax(corr)

def gradient_flip(h, corr, n_points):
    h = h.copy()
    dx = 2.0 / n_points
    h = h.flatten()

    worst_shift = find_worst_shift(corr)
    worst_corr = correlate(h, 1.0 - h, mode="full")[worst_shift]

    # Construct the shifted array for the worst case
    left = h if worst_shift <= n_points else np.zeros_like(h)
    right = h if worst_shift <= n_points else h[worst_shift - n_points:]

    contribution = np.zeros_like(h)

    for k in range(len(h)):
        i = k
        if i + worst_shift >= len(h):
            continue
        contribution[i] += h1[i + worst_shift]
        contribution[i + worst_shift] -= h[i]

    flat_contrib = -(contribution * h + (1.0 - h) * contribution[worst_shift:]
    flat_contrib = np.abs(flat_contrib)

    top_indices = np.argsort(flat_contrib)[-2:]

    flip_indices = np.zeros(len(h), dtype=bool)
    flip_indices[top_indices] = True
    h[flip_indices] = 1.0 - h[flip_indices]
    return h

def restore_balance(h, n_points, balance=1):
    current_sum = np.sum(h)
    target_sum = balance * n_points / 2.0

    if current_sum < target_sum:
        diff = int(np.round(target_sum - current_sum))
        pos = np.where(h == 0)[0]
        np.random.shuffle(pos)
        h[pos[:diff]] = 1.0

        diff = int(np.round(current_sum - target_sum))
        pos = np.where(h == 1)[0]
        np.random.shuffle(pos)
        h[pos[:diff]] = 0.0
    return h
