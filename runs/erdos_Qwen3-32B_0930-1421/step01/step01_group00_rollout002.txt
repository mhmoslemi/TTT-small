import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def clip_and_enforce_sum(optimized_h):
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
        current_sum = optimized_h.sum()
        if not np.isclose(current_sum, target_sum):
            optimized_h *= target_sum / current_sum
        return optimized_h

    def compute_c5(optimized_h):
        optimized_h = clip_and_enforce_sum(optimized_h.copy())
        ones_minus_h = 1.0 - optimized_h
        first = scipy.signal.correlate(optimized_h, ones_minus_h, mode='full')
        second = scipy.signal.correlate(ones_minus_h, optimized_h, mode='full')
        return dx * np.maximum(np.max(first[n_points:]), np.max(second[n_points:]))

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Scipy optimization wrapper
    def objective(x):
        h_opt = x.reshape(n_points)
        return compute_c5(h_opt)

    from scipy.optimize import minimize

    def optimize(x0):
        nonlocal best_h, best_c5
        res = minimize(objective, x0, bounds=[(0.0, 1.0)] * n_points,
                       constraints={"type": "eq", "fun": lambda x: x.sum() - target_sum},
                       method="SLSQP", tol=1e-6, options={"maxiter": 50})
        if res.fun < best_c5:
            best_h[:] = res.x
            best_c5 = res.fun

    # Main loop
    while ((time.time() - start_time) < budget_s) and (best_c5 > 0.380876):
        idxs = np.random.choice(n_points, size=np.random.randint(1, 10), replace=False)
        h_pert = best_h.copy()
        noise = np.random.uniform(-0.01, 0.01, size=len(idxs))
        h_pert[idxs] += noise
        h_pert = clip_and_enforce_sum(h_pert)
        optimize(h_pert)

    return (best_h, best_c5, n_points)
