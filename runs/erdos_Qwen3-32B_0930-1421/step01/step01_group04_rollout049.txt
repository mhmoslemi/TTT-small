import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        len_k = n_points - k
        dot = np.dot(h[:len_k], 1 - h[k:k+len_k])
        corrs.append(dot)
        dot_rev = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(dot_rev)
    return np.max(np.array(corrs)) * dx

def objective_smooth(h_vec, n_points, beta=100):
    dx = 2.0 / n_points
    h = h_vec
    corrs = []
    for k in range(1, n_points):
        len_k = n_points - k
        dot = np.dot(h[:len_k], 1 - h[k:k+len_k])
        corrs.append(dot)
        dot_rev = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(dot_rev)
    return np.log(np.sum(np.exp(beta * np.array(corrs) * dx))) / beta

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full_like(h, 0.5)
    scale = (n_points / 2) / total
    h = h * scale
    return np.clip(h, 0.0, 1.0)

def create_raised_cosine(n_points):
    h = 0.5 + 0.5 * np.cos(np.pi * np.arange(n_points) / (n_points - 1))
    h = np.clip(h, 0.0, 1.0)
    return enforce_normalization(h, n_points)

def pso_update(
    particles, velocities,
    global_best, particle_best,
    w=0.7, c1=2.0, c2=2.0
):
    r1 = np.random.rand(len(particles), len(particles[0]))
    r2 = np.random.rand(len(particles), len(particles[0]))
    velocities = (
        w * velocities +
        c1 * r1 * (particle_best - particles) +
        c2 * r2 * (global_best - particles)
    )
    particles += velocities
    return velocities

def optimize_pso_harmonic(n_points=47, n_particles=20, budget_s=1100, seed=42):
    np.random.seed(seed)
    h_init = initial_h_values if 'initial_h_values' in globals() else create_raised_cosine(n_points)
    particles = np.array([enforce_normalization(h_init + np.random.uniform(-0.1, 0.1, n_points), n_points) for _ in range(n_particles)])
    velocities = np.random.uniform(-0.5, 0.5, (n_particles, n_points))
    particle_best = particles.copy()
    global_best = particles[0].copy()
    best_c5 = compute_c5(global_best, n_points)

    start_time = time.time()
    for idx in range(800):  # ~1000 loop iterations allowed safely

        # Update positions
        velocities = pso_update(particles, velocities, global_best, particle_best)
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            total = np.sum(particles[i])
            if total > 1e-6:
                particles[i] *= (n_points / 2) / total

        # Evaluate
        for i in range(n_particles):
            c5_i = compute_c5(particles[i], n_points)
            if c5_i < compute_c5(particle_best[i], n_points):
                particle_best[i] = particles[i].copy()
                if c5_i < best_c5:
                    global_best[:] = particles[i].copy()
                    best_c5 = c5_i
        if time.time() - start_time >= budget_s - 10:
            break

        # Time check
        if (idx + 1) % 50 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break

    # Final local refinements
    h_best = global_best.copy()
    h_best = opt.minimize(
        lambda x: objective_smooth(x, n_points, beta=50),
        h_best, bounds=[(0, 1)] * n_points,
        method='L-BFGS-B', options={'maxiter': 500}
    ).x
    h_best = enforce_normalization(h_best, n_points)

    return h_best, compute_c5(h_best, n_points), n_points

def run(seed=42, budget_s=1100, **kwargs):
    return optimize_pso_harmonic(seed=seed, budget_s=budget_s)
