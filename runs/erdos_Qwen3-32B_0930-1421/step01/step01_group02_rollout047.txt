import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # provided by runtime
target_C5 = 0.380876


def compute_basis(x_grid, x_centers, alpha=4.0, beta=0.3):
    n = len(x_grid)
    n_centers = len(x_centers)
    basis_list = []
    for x_i in x_centers:
        distances = np.abs(x_grid - x_i)
        mask = distances <= beta
        psi = np.exp(-alpha * distances[mask] ** 2) if np.any(mask) else np.zeros(n)
        full_psi = np.zeros(n)
        full_psi[mask] = psi
        basis_list.append(full_psi)
    return np.column_stack(basis_list)


def solve_optimization(basis, n_points, max_lag=8):
    A = basis
    n_centers = A.shape[1]

    # Build correlation matrix patterns for each lag k
    k_values = np.arange(-max_lag, max_lag + 1)
    corr_tensors = []
    for k in k_values:
        if abs(k) < n_points:
            if k < 0:
                A1 = A[:, :n_points + k]
                A2 = basis[:, -k:]
            elif k > 0:
                A1 = A[:, k:]
                A2 = basis[:, :n_points - k]
            else:
                A1 = A.copy()
                A2 = basis.copy()

            C = A1.T @ A2
            corr_tensors.append(C)

    # Decision variables
    a = cp.Variable(n_centers, nonneg=True)
    lam = cp.Variable()

    # Constraint: total integral is 1
    integrals = np.sum(basis, axis=0)
    constraint = [a @ integrals == n_points / 2]

    # Objective: minimize max overlap
    correlations = []
    for C in corr_tensors:
        correlations.append(cp.Max(cp.diag(C @ a)))
    obj = cp.Minimize(cp.Max(correlations))

    # Problem
    constraints = constraint
    problem = cp.Problem(obj, constraints)

    try:
        problem.solve(solver=cp.CVXOPT, maxiters=100)
        h_values = basis @ a.value
        h_values = np.clip(h_values, 0, 1)
        h_values /= h_values.sum() * 2.0 / n_points
        return h_values
    except Exception as e:
        return np.zeros(n_points)


def run(budget_s=1100, max_centers=10, alpha=4.0, beta=0.3, dx=2.0 / 42, n_points=42, pop_size=2):
    start_time = time.time()
    x_grid = np.linspace(0, 2, n_points)

    if initial_h_values is not None:
        cm = (initial_h_values * x_grid).sum() / initial_h_values.sum()
        x_centers = np.linspace(0, 2, max_centers)
        x_centers = (x_centers + cm - 0.5) % 2.0
    else:
        x_centers = np.linspace(0.1, 1.9, max_centers)

    best_h = None
        best_C5 = np.inf

    for _ in range(pop_size):
        x_centers = np.sort(np.random.rand(max_centers) * 2.0)
        if initial_h_values is not None:
            x_centers = (x_centers + (initial_h_values * x_grid).sum() / initial_h_values.sum()) % 2.0

        basis = compute_basis(x_grid, x_centers, alpha, beta)
        h = solve_optimization(basis, n_points)
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx  # ensure ∫h = 1

        current_C5 = correlate(h, 1 - h, mode="full")
        current_C5 = np.max(current_C5) * dx

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        if time.time() - start_time > budget_s * 0.9:
            break

    return best_h, best_C5, n_points
