import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h[:] = 0.5
        h_sum = n_points * 0.5
    h = h * (n_points / 2.0) / h_sum
    return h

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    global initial_h_values
    h_init = project_to_feasible(initial_h_values.copy(), len(initial_h_values))
    n_points = len(h_init)

    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    # Initialize particles
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best = np.zeros((num_particles, n_points))
    particle_scores = np.zeros(num_particles)
    best_score = float('inf')

    # Initialize particles (10% from initial_h_values, 90% perturbed)
    for i in range(num_particles):
        if i < num_particles // 10:
            particle = h_init.copy()
        else:
            perturbation = np.random.normal(0, 0.1, n_points)
            particle = np.clip(h_init + perturbation, 0, 1)
        particle = project_to_feasible(particle, n_points)
        particles[i] = particle
        velocities[i] = np.random.uniform(-max_velocity, max_velocity, n_points)
        score = evaluate_erdos_solution(particle, n_points)
        personal_best[i] = particle
        particle_scores[i] = score
        if score < best_score:
            best_score = score
            global_best = particle

    while time.time() - start_time < budget_s - 3:
        cur_time = time.time()
        elapsed = cur_time - start_time

        # Evaluate
        for i in range(num_particles):
            score = evaluate_erdos_solution(particles[i], n_points)
            if score < best_score:
                best_score = score
                global_best = particles[i].copy()
            if score < particle_scores[i]:
                particle_scores[i] = score
                personal_best[i] = particles[i].copy()

        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            w = w_start - (w_start - w_end) * (elapsed / budget_s)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)

        # Apply resets to maintain diversity
        reset_idx = np.random.choice(num_particles, size=int(reset_fraction * num_particles), replace=False)
        for i in reset_idx:
            perturbation = np.random.normal(0, 0.1, n_points)
            particles[i] = np.clip(h_init + perturbation, 0, 1)
            particles[i] = project_to_feasible(particles[i], n_points)
            velocities[i] = np.random.uniform(-max_velocity, max_velocity, n_points)
            score = evaluate_erdos_solution(particles[i], n_points)
            particle_scores[i] = score
            if score < best_score:
                best_score = score
                global_best = particles[i].copy()

    return global_best, best_score, n_points
