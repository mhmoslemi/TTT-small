```python
import numpy as np
import time
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    return np.max(correlations) * dx


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        scale_factor = (len(h) / 2) / total
        h *= scale_factor
    return h


def generate_initial_population(pop_size, n_points):
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = ensure_feasible(h)
        population.append(h)
    if initial_h_values is not None:
        h_init = ensure_feasible(np.copy(initial_h_values))
        population.append(h_init)
    return population


def tournament_selection(population, scores, tournament_size=3):
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = min(idxs, key=lambda i: scores[i])
    return population[best_idx].copy()


def crossover(parent1, parent2):
    n_points = len(parent1)
    cross_idx = np.random.randint(1, n_points)
    child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])
    return ensure_feasible(child)


def mutate(h, mut_rate=0.1, discretization=5):
    levels = np.linspace(0, 1, discretization)
    mask = np.random.rand(len(h)) < mut_rate
    if not mask.any():
        return h.copy()
    h[mask] = np.random.choice(levels, size=np.sum(mask))
    return ensure_feasible(h)


def run(budget_s=1100, pop_size=30, mut_rate=0.1, n_points=42, discretization=5):
    start_time = time.time()

    population = generate_initial_population(pop_size, n_points)
    best_h = np.random.rand(n_points)
    best_h = ensure_feasible(best_h)
    best_C5 = float("inf")

    # Precompute fixed discretization levels based on n_points
    levels = np.linspace(0, 1, discretization)

    while time.time() - start_time < budget_s - 1e-5:
        scores = [evaluate_C5(h) for h in population]
        idx = np.argmin(scores)
        curr_best_C5 = scores[idx]
        if curr_best_C5 < best_C5:
            best_h = population[idx].copy()
            best_C5 = curr_best_C5

        new_population = [best_h.copy()]
        while len(new_population) < pop_size:
            p1 = tournament_selection(population, scores)
            p2 = tournament_selection(population, scores)
            child = crossover(p1, p2)
            child = mutate(child, mut_rate=mut_rate, discretization=discretization)
            new_population.append(ensure_feasible(child))

        population = new_population
        if time.time() - start_time > budget_s - 1e-5:
            break

    return best_h, best_C5, n_points
```
