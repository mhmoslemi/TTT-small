import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    dx = 2.0 / len(h)
    h1 = 1.0 - h
    corr = correlate(h, h1, mode='full')
    return np.max(corr) * dx

def find_best_flip(h, c5, dx):
    n = len(h)
    h1 = 1.0 - h
    corr = correlate(h, h1, mode='full')

    max_k = int(corr.shape[0] // 2)  # Middle corresponds to k=0
    max_idx = np.argmax(corr)

    # Identify the shift with max overlap
    best_k = max_idx - max_k

    # Determine the positions affecting this shift
    def get_indices_for_k(k):
        if k < 0:
            return np.arange(-k, n)
        elif k < n:
            return np.arange(0, n - k)
        else:
            return np.array([], dtype=int)

    idxs_for_k = get_indices_for_k(best_k)
    if len(idxs_for_k) == 0:
        return h, c5  # No indices, cannot improve

    active_i = idxs_for_k + 0
    active_j = idxs_for_k + best_k
    active_i = active_i.compress(active_i < n)
    active_j = active_j.compress(active_j < n)

    best_delta = 0
    best_flip = None

    for i in np.where(h > 0.5)[0]:
        for j in np.where(h < 0.5)[0]:
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0

            delta = evaluate_overlap(h_new) - c5
            if delta < best_delta:
                best_delta = delta
                best_flip = (i, j)

    # Apply best flip if it improves
    if best_flip:
        i, j = best_flip
        h[i], h[j] = 0.0, 1.0
        c5 = evaluate_overlap(h)
    return h, c5
