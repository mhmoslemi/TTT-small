import numpy as np
import time
import math
from scipy.signal import correlate


def logsumexp(a, b):
    """Compute log(exp(a) + exp(b)) in a numerically stable way."""
    return max(a, b) + math.log(1 + math.exp(-abs(a - b)))


def scale_h(h, n_points):
    """Scale h to satisfy mass constraint: ∑h = n_points / 2."""
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    return h


def compute_c5(h, dx):
    """Compute the C5 bound."""
    cor = correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx


def perturb_h(h, pert_amp, seg_width, n_points):
    """Perturb a random segment while preserving conservation law."""
    start = np.random.choice(range(n_points - 1))
    end = min(start + seg_width, n_points)

    h_segment = h[start:end].copy()
    delta = (np.random.rand() * 2 - 1) * pert_amp
    h_segment += delta
    h_segment = np.clip(h_segment, 0, 1)

    bump = h_segment - h[start:end]
    h2 = h.copy()
    h2[start:end] = h_segment

    # Redistribute excess mass to keep total mass stable
    excess = np.sum(h2) - n_points / 2
    if abs(excess) > 1e-10:
        bump_density = np.zeros_like(h2)
        bump_density[start:end] = 1
        s = bump_density.sum()
        scaling = 1.0 - excess / s if s > 0 else 1.0
        h2 = h2 * scaling + bump_density * (1 - scaling)

    return h2

    batch_size = 3  # number of trials per step

    seg_width = 8

    best_h = h.copy()

    while elapsed < budget_s and temp > min_temp:
        h_candidates = []
        c5_list = []

        for _ in range(batch_size):
            try:
                h_candidate = perturb_h(best_h, pert_amp, seg_width, n_points)
                c5 = compute_c5(h_candidate, dx)

                c5_list.append(c5)

        # Find the best among current batch
        idx = np.argmin(c5_list)

        h_new = h_candidates[idx]

        # Acceptance probability for SA
        if c5_new < best_c5:
            best_h[:] = h_new[:]
            best_c5 = c5_new
            # Reduce perturb size only when we acting according to gradient
            pert_amp *= 0.995
        else:
            dE = c5_new - best_c5
            if np.random.rand() < math.exp(-dE / temp):
                best_h[:] = h_new[:]
                best_c5 = c5_new

        temp *= alpha
        elapsed = time.time() - start_time
