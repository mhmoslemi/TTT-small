import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        correlation = correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Solve convex relaxation using LP
    remaining_time = budget_s - (time.time() - start_time)
    max_lp_time = 0.5 * budget_s

    if remaining_time > max_lp_time:
        try:
            h_var = cp.Variable(n_points, nonneg=True)
            constraints = [cp.sum(h_var) == n_points / 2, h_var <= 1]

            # Compute overlaps for all k
            overlaps = []
            for k in range(-n_points + 1, n_points):
                idx1 = np.arange(n_points)
                idx2 = np.arange(n_points) + k
                similar = np.where((idx2 >= 0) & (idx2 < n_points))
                A = np.zeros((n_points, n_points))
                A[idx1[similar], idx2[similar]] = 1
                overlap = h_var @ A @ (1 - h_var)
                overlaps.append(overlap)

            max_overlap = cp.Max(overlaps)
            problem = cp.Problem(cp.Minimize(max_overlap), constraints)
            problem.solve(solver=cp.MOSEK, verbose=False)
            h_lp = h_var.value
            h_lp = normalize_h(np.clip(h_lp, 0, 1))
            current_c5 = compute_c5(h_lp)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_lp.copy()
        except Exception as e:
            pass

    # Phase 2: Local search refinement
    h = best_h.copy()
    max_same_c5_iterations = 50
    step_size = 0.2
    same_c5_count = 0
    eps = 1e-8

    while time.time() - start_time < budget_s:
        old_c5 = best_c5

        # Perturb sample randomly
        idx = np.random.randint(n_points)
        old_val = h[idx]
        h[idx] += np.random.uniform(-step_size, step_size)
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 >= old_c5:
            h = best_h.copy()
            same_c5_count += 1
        else:
            best_c5 = current_c5
            best_h = h.copy()
            same_c5_count = 0

        if same_c5_count > max_same_c5_iterations:
            # Cool down and return to better solution
            h = best_h.copy()
            step_size = max(0.01, step_size * 0.5)
            same_c5_count = 0

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break

    return best_h.tolist(), best_c5, n_points
