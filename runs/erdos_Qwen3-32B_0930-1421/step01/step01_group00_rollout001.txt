import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.inf
    best_time = start_time
    h_sum = h.sum()
    h *= (target_sum / h_sum)
    h = np.clip(h, 0.0, 1.0)

    def evaluate(h):
        h_sum = h.sum()
        if not np.isclose(h_sum, target_sum, atol=1e-6):
            h = h / h_sum * target_sum
        h = np.clip(h, 0.0, 1.0)
        ones_minus_h = 1.0 - h
        corr_full = scipy.signal.correlate(h, ones_minus_h, mode='full')
        corr_relevant = corr_full[n_points - 1:2 * n_points - 2]
        return np.max(corr_relevant * dx)

    try:
        import quadprog
        def solve_quadprog(h0):
            Q = np.eye(len(h0)) * 2.0
            q = np.zeros(len(h0))
            grad = np.zeros(len(h0))
            for k in range(1, len(h0) - 1):
                grad += h0[k:] - h0[:-k]
            grad /= (len(h0) - 1)

            # Quadratic cost approx
            h_var = cp.Variable(h0.shape)
            objective = cp.Minimize(cp.quad_form(h_var - h0, Q))
            constraints = [0 <= h_var,
                           h_var <= 1,
                           cp.sum(h_var) == target_sum]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='OSQP')
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return prob.variables()[0].value
            else:
                return h0

    except ImportError:
        pass

    improved = True
    while time.time() - start_time < budget_s:
        t = time.time() - start_time
        if t >= 0.9 * budget_s:
            break

        idxs = np.random.choice(n_points, min(10, n_points), replace=False)
        prop = h.copy()
        noise = np.random.normal(0.0, 0.02, len(idxs))
        prop[idxs] += noise
        prop = np.clip(prop, 0.0, 1.0)
        prop_sum = prop.sum()
        if not np.isclose(prop_sum, target_sum, atol=1e-6):
            prop = prop * target_sum / prop_sum

        best_c5 = evaluate(h)

        if best_c5 < 0.380876:
            break

        prop_c5 = evaluate(prop)
        if prop_c5 < best_c5:
            h[:] = prop[:]
            best_c5 = prop_c5
            best_h[:] = h[:]
            best_time = time.time()
        elif np.random.rand() < np.exp(-5 * (prop_c5 - best_c5)):
            h[:] = prop[:]
            best_c5 = prop_c5
            best_h[:] = h[:]
            best_time = time.time()

        # Check if stuck
        if t - best_time > 45:
            h = solve_quadprog(h)
            h = np.clip(h, 0.0, 1.0)
            h_sum = h.sum()
            h = h * target_sum / h_sum if not np.isclose(h_sum, target_sum, atol=1e-6) else h
            h = np.clip(h, 0.0, 1.0)
            best_c5 = evaluate(h)
            best_time = time.time()

    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
