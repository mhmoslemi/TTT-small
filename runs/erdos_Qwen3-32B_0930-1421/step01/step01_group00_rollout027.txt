import numpy as np
import time
import scipy.signal
import scipy.fft

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    def compute_c5(h, dx):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h *= target_sum / h_sum
        c5_values = []
        for k in range(-n_points + 1, n_points):
            if abs(k) >= n_points - 1:
                continue
            if k < 0:
                corr = np.dot(h[-k:], 1 - h[:n_points + k])
            else:
                corr = np.dot(h[:n_points - k], 1 - h[k:])
            c5_values.append(corr * dx)
        return max(c5_values)

    def fourier_optimize(h0, max_iter=500, freq_steps=50, verbose=False):
        best_h = h0.copy()
        best_c5 = compute_c5(best_h, dx)
        h_fft = scipy.fft.fft(best_h)
        M = min(freq_steps, n_points // 2)

        for iter in range(max_iter):
            # Perturb top M% frequency components
            perturbation = np.zeros_like(h_fft)
            indices = np.argsort(np.abs(h_fft))[-M:]
            perturbation[indices] = np.random.normal(0, 0.01, size=len(indices))

            h_fft_perturb = h_fft + perturbation
            h_update = np.real(scipy.fft.ifft(h_fft_perturb))

            # Ensure constraints
            h_update = np.clip(h_update, 0, 1)
            h_update_sum = h_update.sum()
            if h_update_sum > 1e-9:  # avoid division by zero
                h_update *= target_sum / h_update_sum

            # Compute new C5
            new_c5 = compute_c5(h_update, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_update.copy()
                if verbose:
                    print(f"Iter {iter}: C5 = {new_c5:.6f}")

            # Time budget check
            if time.time() - start_time > budget_s - 30:
                break
        return best_h, best_c5

    # Run optimization
    h_opt, best_c5 = fourier_optimize(h_init, verbose=False)
    h_opt_sum = h_opt.sum()
    if abs(h_opt_sum - target_sum) > 1e-7:
        h_opt *= target_sum / h_opt_sum
    h_opt = np.clip(h_opt, 0, 1)

    return (h_opt, best_c5, n_points)
