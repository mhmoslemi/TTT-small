import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def normalize_h(h):
    s = np.sum(h)
    scale_factor = (0.5 * len(h)) / s
    h_scaled = h * scale_factor
    h_clipped = np.clip(h_scaled, 0, 1)
    h_scaled = h_clipped * (0.5 * len(h)) / np.sum(h_clipped)
    return h_scaled

def perturb_h(h, step_size=0.2):
    n = len(h)
    perturb_n = np.random.randint(5, 11)
    indices = np.random.choice(n, size=min(perturb_n, n), replace=False)
    delta = np.random.normal(0, step_size, len(indices))
    h_perturbed = h.copy()
    h_perturbed[indices] += delta
    h_perturbed = normalize_h(h_perturbed)
    return h_perturbed
