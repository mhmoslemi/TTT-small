import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(int(seed))
    start_time = time.time()
    MAX_SAMPLES = 100

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        h_init[::2] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    def scale_h(h):
        h = h.astype(float)
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum()
        return h

    def compute_c5(h):
        h = scale_h(h)
        ones = np.ones_like(h)
        kernel = ones - h
        corr = correlate(h, kernel, mode='full') * dx
        return np.max(corr)

    def opt_convex(h0):
        h = cp.Variable(n_points)
        con = [0 <= h, h <= 1]
        con += [cp.sum(h) == n_points / 2]
        obj = cp.Minimize(cp.Max(cp.conv(1 - h, h * dx * 2 / n_points)))
        prob = cp.Problem(obj, con)
        prob.solve(solver=cp.MOSEK, reltol=1e-8, maxtime=10)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.array(h.value).flatten()
        return h0

    best_h = scale_h(h_init.copy())
    best_c5 = compute_c5(best_h)

    # First convex optimization
    h_convex = opt_convex(h_init.copy())
    c5 = compute_c5(h_convex)
    if c5 < best_c5:
        best_h = h_convex
        best_c5 = c5

    # Sequential perturbation with support optimization
    perturb_steps = 50
    noise_schedule = np.linspace(0.4, 0.05, perturb_steps)
    for i in range(perturb_steps):
        if time.time() - start_time > budget_s * 0.95:
            break

        noise = noise_schedule[i]
        h = best_h + np.random.rand(n_points) * noise
        h = scale_h(h)

        # Support random swap
        if np.random.rand() < 0.5:
            ix1, ix2 = np.random.choice(n_points, 2, replace=False)
            h, h[ix1], h[ix2] = h.copy(), h[ix2], h[ix1]

        # Convex refinement
        h_after = opt_convex(h)
        c5 = compute_c5(h_after)

        if c5 < best_c5:
            best_h = h_after
            best_c5 = c5

    # Final verification and output
    best_h = scale_h(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
