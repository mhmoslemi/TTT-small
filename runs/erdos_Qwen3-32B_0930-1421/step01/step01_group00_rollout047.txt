import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Constraint
    
    # Normalize the initial h to satisfy the sum constraint and bounds
    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h[0] += (target_sum - h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != target_sum:
        scale = target_sum / h_sum
        h *= scale
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()
    
    def compute_c5(optimized_h):
        optimized_h = np.copy(optimized_h)
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_sum = optimized_h.sum()
        if abs(optimized_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_sum
        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        
        for k in range(n_points):
            if k >= n_points - 1:
                continue
            # Forward and reverse shifts
            if n_points - k > 1:
                corr1 = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
                corr2 = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
                max_corr = max(max_corr, corr1 * dx, corr2 * dx)
        return max_corr
    
    # Compute gradient for the maximal overlap
    def get_gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] += 1 - h[i + k_max]
            grad[i + k_max] -= h[i]
        return grad
    
    reverse_check_time = 10.0
    prev_time = time.time()
    while time.time() - start_time < budget_s - 5:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_time = time.time()
        
        # Determine current maximal overlap shift
        overlap = []
        for k in range(1, n_points):
            if k >= n_points - 1:
                continue
            if n_points - k > 1:
                corr1 = np.dot(h[:n_points - k], (1 - h[k:]))
                corr2 = np.dot(h[k:], (1 - h[:n_points - k]))
                overlap.append((corr1 * dx, corr1, k, None))
                overlap.append((corr2 * dx, corr2, k, None))
        if not overlap:
            break
        max_corr, _, k_max, _ = max(overlap, key=lambda x: x[0])
        
        # Compute gradient and update
        grad = get_gradient(h, k_max)
        alpha = 0.015
        while True:
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = new_h.sum()
            if abs(new_sum - target_sum) > 1e-7:
                new_h *= target_sum / new_sum
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h_sum = new_h.sum()
            if abs(new_h_sum - target_sum) < 1e-7:
                break
            alpha *= 0.95

        h[:] = new_h

        # Perturbation to escape local minima
        if time.time() - best_time > 60 or time.time() - prev_time > reverse_check_time:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_sum = perturbed_h.sum()
            perturbed_h *= target_sum / perturbed_sum

            if compute_c5(perturbed_h) < best_c5:
                h[:] = perturbed_h
                best_c5 = compute_c5(h)
                best_h[:] = h
                best_time = time.time()
            prev_time = time.time()

        # Short progress output
        if verbose and (time.time() - start_time) % 30 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")
    
    return (best_h, best_c5, n_points)
