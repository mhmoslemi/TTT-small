import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Gradient-based optimization with analytical gradient
    learning_rate = 0.1
    decay_factor = 0.999

    while time.time() - start_time < budget_s:
        # Evaluate all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)

        grad = np.zeros_like(h)
        left = h.copy()
        right = 1 - h.copy()
        for i in range(n_points):
            for di in [0, 1]:
                if 0 <= i + k_max + di < n_points:
                    grad[i] -= (1 - h[i + k_max]) if di == 0 else 0
                if 0 <= i - k_max + di < n_points:
                    grad[i] += h[i - k_max] if di == 0 else 0

        grad *= dx
        h -= learning_rate * grad
        h = normalize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_factor

    # Final validation
    if not np.all((best_h >= 0) & (best_h <= 1)) or not np.isclose(np.sum(best_h), n_points / 2):
        best_h = h.copy()
        best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, n_points
