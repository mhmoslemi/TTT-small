import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    max_n_points = 1000
    n_points = 47
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_cvx(h, noise_factor=0.05):
        h = h.copy()
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        w = cp.Variable(n_points)
        constraints = [
            0 <= w,
            w <= 1,
            cp.sum(w) == constraint_sum
        ]
        obj = cvxpy_nobounds_objective(w, h, dx)
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.CBC, verbose=False)
        if result is None:
            return h
        return np.clip(w.value, 0, 1)

    def cvxpy_nobounds_objective(w, h, dx):
        norm_h = w / cp.sum(w)
        shifted_h = cp.reshape(norm_h, (n_points, 1)) - cp.reshape(w, (n_points, 1))
        cross_term = cp.sum(cp.multiply(norm_h, shifted_h), axis=0) * dx
        return cp.MaxExpression([cross_term[k] for k in range(cross_term.shape[0])])

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        pert = np.random.randn(n_points) * noise_factor
        h = h + pert
        h = h.clip(0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # First run with CVXPY convex inner loop
    h = optimize_cvx(h_init)
    c5 = objective(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Perturb and optimize
    for _ in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
        noisy_h = perturb_h(h, noise_factor=0.15)
        improved_h = optimize_cvx(noisy_h)
        c5 = objective(improved_h)
        if c5 < best_c5:
            best_h = improved_h
            best_c5 = c5

    # Post-processing normalization
    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = best_h.clip(0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
