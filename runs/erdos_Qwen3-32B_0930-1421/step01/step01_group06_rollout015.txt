import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    total_h = np.sum(initial_h_values)
    scale_factor = (n_points / 2.0) / total_h
    h_start = np.clip(initial_h_values * scale_factor, 0.0, 1.0)

    # PSO constants
    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    # Initialize particles and velocities
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))

    # Start with 10% particles as perturbed initial values
    for i in range(num_particles):
        if i < num_particles // 10:
            particles[i] = h_start + 0.1 * np.random.randn(n_points)
        else:
            particles[i] = np.random.rand(n_points)

    # Normalize the particles to meet the constraint sum(h) == n_points / 2
    def project(p):
        p = np.clip(p, 0.0, 1.0)
        return p * (n_points / 2.0) / np.sum(p)

    particles = np.array([project(p) for p in particles])
    velocities = 0.01 * np.random.randn(num_particles, n_points)

    # Initialize bests
    p_best = np.copy(particles)
    p_best_vals = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
    g_best_index = np.argmin(p_best_vals)
    g_best = np.copy(p_best[g_best_index])
    best_c5 = p_best_vals[g_best_index]
    best_h = np.copy(g_best)
    last_improvement_time = start_time

    # Main PSO loop
    while time.time() - start_time < budget_s - 10:  # keep 10s grace period
        t = time.time() - start_time
        num_iterations = int((t * 2) ** 0.5)  # Determine per-time iteration count
        for _ in range(min(3, num_iterations)):  # Limit iterations per loop
            # Update inertia
            inertia = w_start - (w_start - w_end) * (time.time() - start_time) / min(budget_s, 600)

            # Random coefficients for cognitive and social updates
            r1 = np.random.rand(num_particles, n_points)
            r2 = np.random.rand(num_particles, n_points)

            # Velocity and position updates
            velocities = (inertia * velocities +
                          c1 * r1 * (p_best - particles) +
                          c2 * r2 * (np.expand_dims(g_best, 0) - particles))
            velocities = np.clip(velocities, -max_velocity, max_velocity)
            particles += velocities

            # Project to feasible region
            for i in range(num_particles):
                particles[i] = project(particles[i])

            # Evaluate and update bests
            for i in range(num_particles):
                c5 = evaluate_erdos_solution(particles[i], n_points)
                if c5 < p_best_vals[i]:
                    p_best[i] = particles[i].copy()
                    p_best_vals[i] = c5
            g_best_index = np.argmin(p_best_vals)
            new_g_best = p_best[g_best_index]
            new_c5 = p_best_vals[g_best_index]

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_g_best
                last_improvement_time = time.time()

            if time.time() - last_improvement_time > 120:
                break

            # Apply fractional reset
            reset_indices = np.random.choice(num_particles, int(reset_fraction * num_particles), replace=False)
            for idx in reset_indices:
                particles[idx] = np.random.rand(n_points)
                particles[idx] = project(particles[idx])

        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
