import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        return np.max(c)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Define optimization parameters
    lambda_vals = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    current_lambda = lambda_vals[0]
    solver_options = {'maxiters': 500}

    # Phase 1: Convex optimization with L1 regularization
    phase1_budget = 0.7 * budget_s
    h_cvx = cp.Variable(n_points)
    h = h.astype(np.float64)  # Ensure h is of type float
    obj = cp.Minimize(cp.max(cp.correlate(h_cvx, 1 - h_cvx, mode='full') * (2.0 / n_points)) + current_lambda * cp.norm1(h_cvx))

    for current_lambda in lambda_vals:
        if time.time() - start_time >= budget_s:
            break
        constraints = [cp.sum(h_cvx) == n_points / 2, h_cvx >= 0, h_cvx <= 1]
        problem = cp.Problem(obj, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, **solver_options)
            h_new = h_cvx.value
            if h_new is not None:
                h_new = normalize_h(h_new)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass  # Ignore and move to next lambda

    # Phase 2: Gradient descent refinement
    learning_rate = 0.1
    phase2_budget = 0.3 * budget_s
    while time.time() - start_time < budget_s:
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= 0.99

    return best_h.tolist(), best_c5, n_points
