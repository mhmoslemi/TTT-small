import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)

    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    if abs(h_sum - target_sum) > 1e-8:
        h *= target_sum / h_sum
    h = np.clip(h, 0.0, 1.0)

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
        optimized_sum = optimized_h.sum()
        if abs(optimized_sum - target_sum) > 1e-8:
            optimized_h *= target_sum / optimized_sum
        optimized_h = np.clip(optimized_h, 0.0, 1.0)

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= n_points:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)
        return max_corr

    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    last_improved = time.time()

    # Hyperparameters
    step_size = 0.01
    perturb_interval = 50
    perturb_step = 0.002
    improvement_patience = 120

    while time.time() - start_time < budget_s - improvement_patience:
        if time.time() - last_improved > improvement_patience:
            break
        if (time.time() - start_time) % perturb_interval < 1e-4:
            indices = np.random.choice(n_points, size=min(20, n_points), replace=False)
            h[indices] += np.random.uniform(-perturb_step, perturb_step, size=len(indices))
            h = np.clip(h, 0.0, 1.0)
            h_sum = h.sum()
            if abs(h_sum - target_sum) > 1e-8:
                h *= target_sum / h_sum
            h = np.clip(h, 0.0, 1.0)

        # Compute gradient for the max-lag overlap
        h_grad = np.zeros(n_points)
        ones_minus_h = 1.0 - h
        max_corr = -np.inf
        k_max = -1

        for k in range(1, n_points):
            if k >= n_points:
                continue
            corr = np.dot(h[:n_points - k], ones_minus_h[k:])
            if corr > max_corr:
                max_corr = corr
                k_max = k

        # Gradient for the worst lag k_max
        for i in range(n_points - k_max):
            h_grad[i] += ones_minus_h[k_max + i]  # derivative w.r. to h[i]
        for i in range(n_points - k_max):
            h_grad[k_max + i] -= h[i]  # derivative w.r. to h[k + i]

        # Apply gradient step
        grad_dir = h_grad
        grad_dir = (grad_dir - np.mean(grad_dir)) / (np.std(grad_dir) + 1e-6)  # Normalize

        h_unscaled = h - step_size * grad_dir

        # Project back onto constraints
        h = h_unscaled.copy()
        h = np.clip(h, 0.0, 1.0)
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-8:
            h *= target_sum / h_sum
        h = np.clip(h, 0.0, 1.0)

        # Re-evaluate
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improved = time.time()
    return (best_h, best_c5, n_points)
