import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_current = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_current = np.zeros(n_points)
        h_current[::2] = 2.0 / n_points
        h_current *= n_points / 2 / h_current.sum()

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.copy().flatten()
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    h_best = h_current
    c5_best = objective(h_current)

    def linear_model(h):
        h = h.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return overlaps

    def minimize_overlap_lp(h0):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            h <= 1,
            h >= 0,
            sum(h) == constraint_sum,
            h[0, :] >= 0
        ]

        for k in range(1, n_points):
            shifted = np.roll(h, -k)
            overlap = cp.sum(h * (1 - shifted)) * dx
            constraints.append(overlap <= C)
        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.GLPK_MI, verbose=False,
                   time_limit=budget_s * 0.85 - (time.time() - start_time))
        return h.value, C.value

    # Perturb functions
    def perturb_h(h, noise_factor=0.05):
        h_pert = h + noise_factor * np.random.randn(n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (constraint_sum / h_pert.sum())
        return h_pert

    # Main optimize loop
    restart = 0
    while time.time() - start_time < budget_s:
        # Solve LP from current point
        best_h_lp, best_c5_lp = minimize_overlap_lp(h_current)
        h_current = best_h_lp

        # Evaluate and update best solution
        if h_current is not None:
            h_current = h_current.clip(0, 1)
            h_sum = h_current.sum()
            h_current = h_current * (constraint_sum / h_sum)
            c5 = objective(h_current)
            if c5 < c5_best:
                h_best = h_current.copy()
                c5_best = c5

        # Randomly perturb every few iterations to escape local minima
        restart += 1
        if restart % 5 == 0:
            h_current = perturb_h(h_current)
            c5 = objective(h_current)
            if c5 < c5_best:
                h_best = h_current.copy()
                c5_best = c5

    # Final validation and normalization
    h_best = h_best.clip(0, 1)
    h_best = h_best * (constraint_sum / h_best.sum())
    c5_best = objective(h_best)

    return h_best, c5_best, n_points
