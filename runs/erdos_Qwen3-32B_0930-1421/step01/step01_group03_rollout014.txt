import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_base = np.zeros(n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h_base[indices] = 2.0 / n_points
        h_base *= n_points / 2 / h_base.sum()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * target_sum / h_sum
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def repair(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / h.sum()

    def mutate(h, noise_factor=0.05):
        h_mut = h + np.random.randn(n_points) * noise_factor
        h_mut = repair(h_mut)
        return h_mut

    def cross(h1, h2):
        mask = np.random.rand(n_points) < 0.5
        h_cross = np.where(mask, h1, h2)
        h_cross = repair(h_cross)
        return h_cross

    def optimize(h0, max_iter=500):
        best_h = h0.copy()
        best_val = evaluate(best_h)
        for _ in range(max_iter):
            h = h0 + np.random.randn(n_points) * 0.02
            h = repair(h)
            val = evaluate(h)
            if val < best_val:
                best_h = h
                best_val = val
        return best_h, best_val

    # Initialize population
    population = [mutate(h_base) for _ in range(5)]
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.8:
            break
        # Evaluate round
        evals = [(h, evaluate(h)) for h in population]
        evals.sort(key=lambda x: x[1])
        top_half = [e[0] for e in evals[:5]]

        # Generate new offspring
        for i in range(5):
            for j in range(i + 1, 5):
                if time.time() - start_time > budget_s * 0.99:
                    break
                child = cross(top_half[i], top_half[j])
                child = optimize(child, max_iter=200)[0]
                population.append(child)

        population = top_half + random.sample(population[5:], 5)

    # Final optimization step
    best_h = min(population, key=evaluate)
    best_h, best_val = optimize(best_h, max_iter=1000)
    best_h = repair(best_h)

    # Final clean sweep
    best_val = evaluate(best_h)

    return best_h, best_val, n_points
