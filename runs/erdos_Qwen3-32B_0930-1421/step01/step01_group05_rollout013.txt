import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use predefined initial_h_values to start
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        ones = np.ones_like(h_vals)
        overlap = correlate(h_vals, ones - h_vals, mode='full')
        return np.max(overlap) * dx

    def normalize_h(h_vals):
        h_clipped = np.clip(h_vals, 0, 1)
        scale = (n_points / 2) / np.clip(np.sum(h_clipped), 1e-10, None)
        return np.clip(h_clipped * scale, 0, 1)

    h_best = h.copy()
    c5_best = compute_c5(h_best)

    learning_rate = 0.1
    while time.time() - start_time < budget_s:
        # Identify the max-overlap shift k for the current h
        ones = np.ones_like(h)
        overlap = correlate(h, ones - h, mode='full')
        k_best = np.argmax(overlap)
        overlap_best = overlap[k_best]

        # Gradient for current best k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k_best + i < n_points:
                grad[i] += -h[i]
                grad[k_best + i] += h[i]
            else:
                grad[i] += h[i]
                grad[k_best + i - n_points] += h[i]
        grad *= dx

        # Gradient descent step with line search
        for step in range(10):  # Max line search steps
            trial_h = normalize_h(h - learning_rate * grad)
            trial_c5 = compute_c5(trial_h)
            if trial_c5 < c5_best:
                h = trial_h.copy()
                c5_best = trial_c5
                h_best = h.copy()
                break
            else:
                learning_rate *= 0.5

        # Track time and reduce learning rate if progress stalls
        learning_rate *= 0.999

    return h_best.tolist(), c5_best, n_points
