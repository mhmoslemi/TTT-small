import time

    dx = 2.0 / n_points

    def enforce_constraints(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        scaling = target_sum / current_sum if current_sum > 0 else 1.0
        h *= scaling
        h = np.clip(h, 0.0, 1.0)

    best_c5 = np.inf

    # FFT-based overlap calculator
    def compute_c5_fft(h):
        h = h.copy()
        h = enforce_constraints(h)

        # Compute (1 - h)
        ones_minus_h = 1.0 - h

        # Compute full cross-correlation via FFT
        cross_corr = scipy.signal.correlate(h, ones_minus_h, mode='full')
        cross_corr = cross_corr * dx

        # Slice from k=1 (shift 1) to k = n_points -2 (overlap only at start)
        max_k_forward = np.max(cross_corr[n_points:])
        max_k_backward = np.max(cross_corr[:n_points - 1])

        return max(max_k_forward, max_k_backward)

    # Fourier domain optimization
    h_f = fft.rfft(h_init)
    N = h_f.size
    M = min(20, N - 1)  # Use top M+1 coefficients (real and symmetric)

    while time.time() - start_time < budget_s - 5:
        # Select top M frequency components
        indices = np.argsort(np.abs(h_f))[::-1][:M]
        proposed_coeffs = h_f.copy()

        # Add small perturbation to the top coefficients in Fourier domain
        perturbation = np.random.uniform(-0.01, 0.01, size=proposed_coeffs[indices].shape)
        proposed_coeffs[indices] += perturbation

        # Reconstruct h in spatial domain
        h_proposed = fft.irfft(proposed_coeffs, n=n_points)

        # Enforce constraints
        h_proposed = enforce_constraints(h_proposed)

        # Evaluate new C5
        new_c5 = compute_c5_fft(h_proposed)

        # If improved, save the solution
        if new_c5 < best_c5:
            best_h = h_proposed.copy()
            best_c5 = new_c5
            best_time = time.time()

        # Check early stopping if we are really close to the goal and no improvement in 120s
        if best_c5 < 0.380876 and time.time() - best_time > 120:
