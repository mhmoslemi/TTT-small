import numpy as np
import time
import cvxpy as cp
import scipy.signal


def evaluate_C5(h, dx):
    """Evaluate the C5 bound from h using fast convolution."""
    h = np.clip(h, 0, 1).astype(np.float64)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    ones_h = 1 - h
    max_overlap = 0.0
    cor = scipy.signal.correlate(h, ones_h, mode="full")
    max_overlap = np.max(cor) * dx
    return max_overlap, h
