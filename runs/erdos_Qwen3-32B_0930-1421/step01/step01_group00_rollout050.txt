import numpy as np
import time
import cvxpy as cp
import scipy.signal


def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Rescale to meet constraint exactly
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float("inf")
    best_time = time.time()

    # Warm up with convex relaxation of key overlaps
    def convex_relax_step(h):
        # Formulate a semidefinite relaxation to minimize max_k overlap
        h_var = cp.Variable(n_points)
        ones_minus_h_var = 1 - h_var
        corr_vars = [
            cp.sum(cp.multiply(h_var[:n_points - k], ones_minus_h_var[k:]))
            for k in range(1, n_points - 1)
        ]
        max_corr = cp.Variable()
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ] + [corr <= max_corr for corr in corr_vars]
        objective = cp.Minimize(max_corr)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.OSQP, max_iter=5000, eps_abs=1e-6)

        if prob.status == "optimal":
            h_candidate = np.clip(h_var.value, 0, 1)
            if np.isclose(h_candidate.sum(), target_sum):
                return h_candidate
        return h

    # Run convex relaxation
    if time.time() - start_time < 60:
        h = convex_relax_step(h)

    # Local search with gradient approximation and adaptive perturbation
    while time.time() - start_time < budget_s - 5:
        t_since_best = time.time() - best_time

        # Perturb small subset
        n_perturb = min(10, n_points)
        idx = np.random.choice(n_points, size=n_perturb, replace=False)
        delta = np.clip(
            np.random.randn(n_perturb) * (0.05 + 0.03 * (t_since_best > 120)), -0.1, 0.1
        )
        h_proposal = h.copy()
        h_proposal[idx] = np.clip(h_proposal[idx] + delta, 0, 1)

        # Normalize to volume constraint
        h_proposal_sum = h_proposal.sum()
        if not np.isclose(h_proposal_sum, target_sum):
            if h_proposal_sum < target_sum:
                max_idx = np.argmax(1 - h_proposal)
                h_proposal[max_idx] = min(1, h_proposal[max_idx] + (target_sum - h_proposal_sum))
            elif h_proposal_sum > target_sum:
                min_idx = np.argmin(h_proposal)
                h_proposal[min_idx] = max(0, h_proposal[min_idx] - (h_proposal_sum - target_sum))

        # Evaluate proposal
        def compute_c5(h_eval):
            max_corr = -np.inf
            h_clip = np.clip(h_eval, 0, 1)
            h_clip_sum = h_clip.sum()
            if not np.isclose(h_clip_sum, target_sum, atol=1e-6):
                h_clip *= target_sum / h_clip_sum
            ones_h = 1 - h_clip
            for k in range(1, n_points - 1):
                if k >= len(h_clip):
                    continue
                lhs = h_clip[: len(h_clip) - k]
                rhs = ones_h[k:]
                corr = np.dot(lhs, rhs) * dx
                max_corr = max(max_corr, corr)

                lhs = h_clip[k:]
                rhs = ones_h[: len(h_clip) - k]
                corr = np.dot(lhs, rhs) * dx
                max_corr = max(max_corr, corr)
            return max_corr

        c5_proposal = compute_c5(h_proposal)
        c5_current = compute_c5(h)

        # Accept better or similar proposals probabilistically
        if c5_proposal < best_c5 or np.random.rand() < 0.1:
            h[:] = h_proposal[:]
            c5_current = c5_proposal
            best_time = time.time()

        # Update best solution
        if c5_current < best_c5:
            best_h[:] = h[:]
            best_c5 = c5_current

        # Exit early or after timeout
        if best_c5 <= 0.380876:
            print(f"Bound C5 = {best_c5} <= 0.380876 achieved.")
            h = best_h.copy()
            break

        # Periodic logging
        if verbose and time.time() - start_time % 30 < 1:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

        # Progress check
        if time.time() - best_time > 120 and verbose:
            print(f"No improvement for 120s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
