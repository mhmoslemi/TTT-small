```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / h.sum()
    return h


def mcco_rlx(n_points):
    h = cp.Variable(n_points, pos=True)
    lam = cp.Variable(pos=True)

    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]

    overlaps = []
    for k in range(-n_points + 1, n_points):
        if -n_points < k < 0:
            indices = [(i, i - k) for i in range(-k, n_points)]
        elif k == 0:
            continue  # skip zero overlap
        else:
            indices = [(i, i + k) for i in range(n_points - k)]

        ov = sum(h[i] * (1 - h[j]) for i, j in indices)
        overlaps.append((ov, k))

    return h, lam, overlaps


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    h_init = np.copy(initial_h_values) if initial_h_values is not None else ensure_feasible(np.random.rand(n_points))
    best_h, best_C5 = h_init, evaluate_C5(h_init)
    dx = 2.0 / n_points

    h_var, lam, all_overlaps = mcco_rlx(n_points)
    constraints = []

    # Add bilinear McCormick constraints at each cutting-plane step
    while time.time() - start_time < budget_s:
        constraints.extend([o[0] <= lam * (n_points / 2) for o in all_overlaps])
        problem = cp.Problem(cp.Minimize(lam), constraints)

        problem.solve(solver=cp.OSQP, verbose=False)

        if problem.status != "optimal":
            break

        curr_h = np.clip(np.array(h_var.value).flatten(), 0, 1)
        curr_h = ensure_feasible(curr_h)
        curr_C5 = evaluate_C5(curr_h)

        if curr_C5 < best_C5:
            best_h, best_C5 = curr_h, curr_C5

        # Add the worst-case overlap constraint
        all_ov_corr = [corr * dx for corr in correlate(best_h, (1 - best_h), mode="full")]
        worst_k = int(np.argmax(all_ov_corr))
        if worst_k in [o[1] for o in all_overlaps]:
            continue
        else:
            if 0 <= worst_k < n_points:
                indices = [(i, i + worst_k) for i in range(n_points - worst_k)]
            else:
                indices = [(i, i - worst_k) for i in range(-worst_k, n_points)]
            new_ov = sum(h_var[i] * (1 - h_var[j]) for i, j in indices)
            constraints.append(new_ov <= lam * (n_points / 2))

    return best_h, best_C5, n_points
```
