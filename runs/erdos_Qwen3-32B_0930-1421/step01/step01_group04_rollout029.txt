import numpy as np
import time
import cvxpy as cp
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

    ]

    for k in [1, 2, 4, 8, 16]:

            constraints.append(overlap <= C)
    
    prob = cp.Problem(cp.Minimize(C), constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    
    h_value = h.value.copy()
    h_value = enforce_normalization(h_value, n_points)
    best_c5 = compute_c5(h_value, n_points)
    best_h = h_value.copy()
    time_start = time.time()
    
    # Column generation loop
    while time.time() - time_start < budget_s * 0.9:
        # Evaluate worst violated constraint
        corrs = []
        for k in range(1, n_points - 1):
            corr = np.dot(h_value[:n_points - k], 1 - h_value[k:k + n_points - k])
            corrs.append(corr * (2.0 / n_points))
        worst_index = np.argmax(corrs)
        worst_k = worst_index + 1
        
        # Check if constraint is violated
        viol = corrs[worst_index] - best_c5
        if viol > 1e-5:
            overlap = cp.sum(h[:n_points - worst_k] * (1 - h[worst_k:n_points]) * (2.0 / n_points))
            constraints.append(overlap <= C)
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            if prob.status == "optimal":
                h_value = h.value.copy()
                h_value = enforce_normalization(h_value, n_points)
                c5 = compute_c5(h_value, n_points)
                if c5 < best_c5:
                    best_h = h_value.copy()
                    best_c5 = c5
        else:
            break

    # L-BFGS refinement
    def objective(x):
        h_noisy = x.reshape(-1)
        return compute_c5(h_noisy, n_points)

    bounds = [(0, 1)] * n_points
    result = opt.minimize(
        objective,
        best_h,

        method='L-BFGS-B',

    c5_final = compute_c5(h_return, n_points)
