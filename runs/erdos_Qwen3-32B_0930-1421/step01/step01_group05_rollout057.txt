import numpy as np
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_best)
    else:
        n_points = 45
        h_best = np.random.rand(n_points)
        h_best = h_best / h_best.sum() * (n_points / 2)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def repair_h(h):
        # Clip to [0, 1]
        clipped = np.clip(h, 0, 1)
        # Normalize to ensure sum(h) = n_points / 2
        sum_h = np.sum(clipped)
        if sum_h == 0:
            return np.zeros_like(h)
        scale_factor = (n_points / 2) / sum_h
        scaled = clipped * scale_factor
        # Final clip in case of numerical error
        return np.clip(scaled, 0, 1)

    # Firefly initialization
    n_fireflies = 50
    h_population = [h_best.copy()]  # Start from initial_h_values
    for _ in range(n_fireflies - 1):
        h = h_best.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = repair_h(h)
        h_population.append(h)

    # Parameters
    beta0 = 1.0
    alpha = 0.2
    gamma = 0.02

    best_c5 = compute_c5(h_best)
    H_pop = [repair_h(h) for h in h_population]
    H_pop = np.array(H_pop)

    t_max = int(budget_s // 0.02)  # max iterations per budget
    keep_running = True

    while time.time() - start_time < budget_s and keep_running:
        keep_running = False
        new_pop = H_pop.copy()
        current_c5 = compute_c5(H_pop)

        for i in range(n_fireflies):
            idx_sort = np.argsort(current_c5)
            for j in idx_sort:
                if current_c5[j] < current_c5[i]:  # brighter firefly
                    r = np.linalg.norm(H_pop[j] - H_pop[i])
                    beta = beta0 * np.exp(-gamma * r**2)
                    move = beta * (H_pop[j] - H_pop[i]) + alpha * (np.random.rand(n_points) - 0.5)
                    new_pos = H_pop[i] + move
                    new_pos = repair_h(new_pos)
                    c5_new = compute_c5(new_pos)

                    if c5_new < current_c5[i]:
                        new_pop[i] = new_pos
                        current_c5[i] = c5_new
                        keep_running = True

            new_pop[i] = repair_h(new_pop[i])

        H_pop = new_pop
        idx_min = np.argmin(current_c5)
        h_best = H_pop[idx_min]
        best_c5 = current_c5[idx_min]

    return h_best.tolist(), best_c5, n_points
