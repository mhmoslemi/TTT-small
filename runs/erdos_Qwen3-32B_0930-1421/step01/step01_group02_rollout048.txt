import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    h *= (len(h) / 2) / np.sum(h)
    return h


def construct_convex_model(h_init, dx, tol=1e-6):
    n_points = len(h_init)
    h = cp.Variable(n_points, pos=True)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]
    objective = cp.Minimize(cp.Parameter(n_points * 2 - 1).max())
    problem = cp.Problem(objective, constraints)
    return h, problem

    best_h = ensure_feasible(initial_h)

    h = best_h

    # Solve convex relaxation to optimize C5 directly
    n_points = len(h)
    h_var = cp.Variable(n_points, pos=True)
    ones_minus_h_var = 1 - h_var
    overlaps = []
    for k in range(-(n_points - 1), n_points):
        shifted = np.roll(ones_minus_h_var, shift=k)
        overlap = cp.sum(h_var * shifted.T)
        overlaps.append(overlap * dx)

    objective = cp.Minimize(cp.max(overlaps))
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.SCS, max_iters=1500, verbose=False)
    except Exception as e:
        pass

    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_solution = np.clip(np.array(h_var.value).flatten(), 0, 1)
        h_solution = ensure_feasible(h_solution)
    else:
        h_solution = best_h

    final_C5 = evaluate_C5(h_solution)
    return h_solution, final_C5, n_points
