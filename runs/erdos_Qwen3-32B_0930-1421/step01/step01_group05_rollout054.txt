import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from math import log

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the initial_h_values provided or fallback to a default
    if 'initial_h_values' in globals():
        h_global = np.array(initial_h_values, dtype=np.float64)
    else:
        h_global = np.ones(45, dtype=np.float64) / 45 * 45 / 2

    n_points = len(h_global)
    dx = 2.0 / n_points

    def compute_c5(h):
        k_min = -n_points + 1
        k_max = n_points - 2
        c_ks = []
        for k in range(k_min, k_max + 1):
            if 0 <= k < n_points:
                a = h
                b = np.roll(1 - h, -k)
                b[-k:] = 0  # Zero-padding
            else:
                a = h[:n_points - abs(k)] if k > 0 else h[abs(k):]
                b = h[-(n_points - abs(k)):] if k > 0 else h[:-abs(k)]
                b = (1 - b) if k > 0 else (1 - b)
            c_k = np.sum(a * b) * dx
            c_ks.append(c_k)
        return max(c_ks)

    def clip_and_normalize(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        return clipped * (n_points / 2) / total

    h = h_global.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Sequence of λ values to use in line search
    lambdas = [1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]

    # Phase 1: CVXOPT with L1-regularization (70% of time)
    cvx_time_limit = 0.7 * budget_s
    for l in lambdas:
        now = time.time()
        if now - start_time > cvx_time_limit:
            break

        # Define problem variables
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(
            [cp.sum((h_var[i:] if k >= 0 else h_var[:n_points - k]) * (1 - h_var[np.roll(np.arange(n_points), -k)])[:len(h_var[i:])]) * dx for k in range(-n_points + 1, n_points - 1)]
        ) + l * cp.norm1(h_var))

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1,
            h_var >= 0
        ]

        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.CVXOPT)
        except Exception:
            continue

        new_h = h_var.value
        if new_h is None:
            continue
        new_h = clip_and_normalize(new_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

    # Phase 2: Refined gradient descent with adaptive learning
    grad_time_limit = budget_s - (time.time() - start_time)
    learning_rate = 0.01
    grad_steps = 0
    grad_max = 5000  # Conservative upper bound on steps
    while time.time() - start_time < budget_s and grad_steps < grad_max:
        eps = 1e-3
        grad = np.zeros(n_points)

        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = clip_and_normalize(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = clip_and_normalize(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        # Trust region bound on grad step size
        norm_grad = np.sum(grad ** 2) ** 0.5
        step_size = max(0.0001, min(1.0, (1 / (1 + norm_grad)) * learning_rate))
        h -= step_size * grad
        h = clip_and_normalize(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= 0.995
        grad_steps += 1

    return best_h.tolist(), best_c5, n_points
