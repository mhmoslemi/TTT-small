import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to desired constraint sum(h) == n_points / 2.0
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)
    h = np.maximum(h, 0.0)

    dx = 2.0 / n_points
    h_cvx = cp.Parameter(n_points, nonneg=True)
    h_var = cp.Variable(n_points, nonneg=True)
    C5_var = cp.Variable(nonneg=True)

    # Rewrite evaluate function into CVXPY constraints
    constraints = [
        cp.sum(h_var) == n_points / 2.0,
        h_var <= 1.0,
        C5_var >= 0.0,
    ]

    for k in range(-n_points + 1, n_points):
        if k >= 0:
            i_l = 0
            i_r = n_points - k
        else:
            i_l = -k
            i_r = n_points
        lhs = h_var[i_l:i_r] * (1 - h_var[i_l + k:i_r + k])
        constraints.append(C5_var >= cp.sum(lhs) * dx)

    objective = cp.Minimize(C5_var)
    problem = cp.Problem(objective, constraints)

    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement_time = start_time

    def cvx_run(h_start):
        nonlocal best_c5, best_h, h_var, last_improvement_time
        h_var.value = h_start
        problem.solve(solver=cp.MOSEK, verbose=False, max_time=30)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(h_var.value, 0, 1).copy()
            c5 = float(C5_var.value)
            if c5 < best_c5:
                best_c5 = c5
                best_h[...] = new_h
                last_improvement_time = time.time()
                return c5
        return best_c5

    cvx_run(h)

    while time.time() - start_time < budget_s - 5:
        idx = np.random.choice(n_points)
        h_perturb = h + 0.1 * (np.random.rand(n_points) * 2 - 1)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * n_points / 2.0
        h_perturb = np.clip(h_perturb, 0, 1)
        c5 = cvx_run(h_perturb)
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
