import numpy as np
import time
from scipy.signal import correlate
import heapq


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-9:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def initialize_branching_space(h, n_levels=5):
    step = 1.0 / (n_levels - 1)
    return np.arange(0, 1.0 + 1e-9, step)


def discrete_feasibility(h, n_points):
    if h.shape != (n_points,):
        return False
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    return np.isclose(h.sum(), n_points / 2, atol=1e-6)


def perturb_h(h, perturbation_budget=3):
    h = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, size=perturbation_budget, replace=False)
    discrete_levels = initialize_branching_space(h)
    for i in indices:
        h[i] = discrete_levels[np.random.randint(len(discrete_levels))]
    return ensure_feasible(h)


def compute_upper_bound(h_partial, n_points):
    h = h_partial.copy()
    unknown = h == -1
    h[unknown] = 0.0
    total_fixed = h.sum()
    remaining = (n_points / 2) - total_fixed
    h[unknown] = 0.0 if remaining <= 0 else 1.0

    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    return np.max(correlations) * (2.0 / n_points)


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    n_levels = 5  # Discretization: 0, 0.25, 0.5, 0.75, 1

    if initial_h_values is not None:
        h_init = initial_h_values.copy()
    else:
        h_init = np.zeros(n_points)
        h_init[::2] = 1.0
        h_init = ensure_feasible(h_init)

    h_discrete = perturb_h(h_init)
    best_h = h_discrete.copy()
    best_C5 = evaluate_C5(best_h)
    queue = []

    # Branching state: partial assignment (with -1 as unassigned), indices sorted
    # by impact. We use 0 for minimal entropy.
    discrete_levels = initialize_branching_space(h_discrete, n_levels=n_levels)

    def push_to_queue(state, current_bound):
        state_t = state.tobytes(), current_bound
        # Use heapq to manage most promising states
        heapq.heappush(queue, (-current_bound, state_t, state.shape))

    # Start state
    h_test = h_discrete.copy()
    push_to_queue(h_test, best_C5)

    while time.time() - start_time < budget_s and queue:
        _, (state_bytes, _, shape) = heapq.heappop(queue)
        state = np.frombuffer(state_bytes).reshape(shape)
        if not discrete_feasibility(state, n_points):
            continue

        # Try to complete
        success = True
        for i in range(n_points):
            if state[i] < 0:
                state[i] = np.random.choice(discrete_levels)
        state = ensure_feasible(state)

        current_val = evaluate_C5(state)

        if current_val < best_C5:
            best_C5 = current_val
            best_h = state.copy()

        # Prune if lower bound is too bad
        lb = compute_upper_bound(state, n_points)
        if lb >= best_C5:
            continue

        # Branch on next unassigned variable
        unknown = np.where(state < 0)[0]
        if len(unknown) > 0:
            idx = unknown[0]
            for level in discrete_levels:
                new = state.copy()
                new[idx] = level
                if discrete_feasibility(new, n_points):
                    new = ensure_feasible(new)
                    new_bound = evaluate_C5(new)
                    push_to_queue(new, new_bound)

    return best_h, best_C5, n_points
