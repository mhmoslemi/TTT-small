import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
import itertools

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Grid of nearby n_points for exploring different resolutions
    n_candidates = [30, 45, 60]

    # Optimize over all candidate resolutions
    for n_candidate in n_candidates:
        if time.time() - start_time >= budget_s:
            break

        # Adjust h to fit new grid
        h_new = np.interp(np.linspace(0, 2, n_candidate), np.linspace(0, 2, n_points), h)
        h_new = normalize_h(h_new)
        n_points = n_candidate
        h = h_new
        dx = 2.0 / n_points

        # Random perturbation + optimization
        for _ in range(5):
            perturbation = np.random.randn(n_points) * h * (1 - h) * 0.1
            perturbed_h = h + perturbation
            perturbed_h = normalize_h(perturbed_h)

            # Gradient descent
            for step in range(50):
                if time.time() - start_time >= budget_s:
                    break
                grad = np.zeros_like(h)
                eps = 1e-5
                for i in range(n_points):
                    h_p = h.copy()
                    h_p[i] += eps
                    h_p = normalize_h(h_p)
                    c_p = compute_c5(h_p)
                    h_n = h.copy()
                    h_n[i] -= eps
                    h_n = normalize_h(h_n)
                    c_n = compute_c5(h_n)
                    grad[i] = (c_p - c_n) / (2 * eps)

                h -= 0.1 * grad / np.max(np.abs(grad))
                h = normalize_h(h)

            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                print(f"Improved C5: {best_c5} at n={n_candidate}")

    return best_h.tolist(), best_c5, n_points
