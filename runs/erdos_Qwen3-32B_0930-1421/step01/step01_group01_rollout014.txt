import numpy as np
import time
import scipy.signal


def run(seed=42, budget_s=1100, n_points=100):
    """Optimize the Erdős C5 problem with perturbation-based greedy search."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if defined
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h[:n_points//4] = 0.3
        h[n_points//4:3*n_points//4] = 0.2
        h[3*n_points//4:] = 0.25
        h = h / h.sum() * n_points / 2

    best_h = h.copy()
    best_C5 = float('inf')
    best_time = time.time()

    def evaluate_C5(h_eval):
        cor = scipy.signal.correlate(h_eval, 1.0 - h_eval, mode='full')
        C5 = np.max(cor) * dx
        return C5

    def rescale_h(h_eval):
        h_eval = np.clip(h_eval, 0.0, 1.0)
        h_sum = h_eval.sum()
        if h_sum > 1e-10:
            h_eval = h_eval / h_sum * n_points / 2
        return h_eval

    start_time = time.time()
    learning_rate = 0.02
    perturbation_amp = 0.1
    segment_len = int(0.15 * n_points)
    perturb_step = 0

    while time.time() - start_time < budget_s:
        perturb_step += 1

        if perturb_step % 100 == 0:
            perturbation_amp = max(0.02, perturbation_amp * 0.9)
        candidate_h = h.copy()
        if perturb_step % 50 == 0:
            # Apply segment flip
            start = np.random.randint(0, candidate_h.shape[0] - segment_len)
            candidate_h[start:start+segment_len] = 1.0 - candidate_h[start:start+segment_len]
        else:
            # Apply Gaussian noise to a segment
            start = np.random.randint(0, candidate_h.shape[0] - segment_len)
            noise = perturbation_amp * np.random.randn(segment_len)
            candidate_h[start:start+segment_len] += noise
        candidate_h = rescale_h(candidate_h)

        # Evaluate candidate
        C5_candidate = evaluate_C5(candidate_h)
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = candidate_h.copy()
            best_time = time.time()

        # Update h if improved
        if C5_candidate < evaluate_C5(h):
            h = candidate_h
        else:
            # Small regularization step
            h = rescale_h(h)

    return best_h, best_C5, n_points
