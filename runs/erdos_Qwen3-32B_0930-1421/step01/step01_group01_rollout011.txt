import numpy as np
import time
import math
from scipy.signal import correlate


def evaluate_C5(h, dx):
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def perturb_h(h, dx, n_points, amp=0.1):
    h = h.copy()
    idxs = np.random.choice(n_points, size=min(7, n_points), replace=False)
    change = np.random.normal(
        scale=np.sqrt(amp / len(idxs)), size=len(idxs))  # Gaussian perturb
    h[idxs] += change
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h *= (n_points / 2) / h_sum
    return h


def run(seed=42, budget_s=1100, n_points=84):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n_points

        h_sum = h.sum()
        h *= (n_points / 2) / h_sum

    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)

    T = 1.0
    T_min = 1e-8
    alpha = 0.99
    iter_per_T = 100
    t0 = time.time()

    while time.time() - t0 < budget_s:
        for _ in range(iter_per_T):
            if time.time() - t0 > budget_s:
                break
            h_candidate = perturb_h(h, dx, n_points, amp=0.2)
            C5_candidate = evaluate_C5(h_candidate, dx)
            C5_current = evaluate_C5(h, dx)
            delta = C5_candidate - C5_current

                if C5_candidate < best_C5:
                    best_C5 = C5_candidate
                    best_h = h.copy()
                    if best_C5 < 0.381:
                        print(f"Improved to C5={best_C5:.6f}")  # Debug message
        T *= alpha

    return best_h, best_C5, n_points
