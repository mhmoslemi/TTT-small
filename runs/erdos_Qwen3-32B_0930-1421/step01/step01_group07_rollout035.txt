import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h, pos1, pos2):
    h_new = h.copy()
    h_new[pos1] = 1 - h[pos1]
    h_new[pos2] = 1 - h[pos2]
    return h_new

def flip_pair_and_validate(h, pos1, pos2):
    h_new = flip_two_bits(h, pos1, pos2)
    if (abs(np.sum(h_new) - len(h) / 2) < 1e-8) and (np.all((h_new >= 0) & (h_new <= 1))):
        return h_new
    return None
