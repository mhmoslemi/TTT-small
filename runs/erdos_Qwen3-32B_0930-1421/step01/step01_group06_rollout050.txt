import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure sum(h) is n_points / 2
    total_h = float(np.sum(initial_h))
    h = np.clip(initial_h * (n_points / 2 / total_h), 0.0, 1.0)
    h = h.astype(np.float64)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    best_h_values = np.copy(h)
    last_improvement_time = start_time

    dx = 2.0 / n_points

    def objective(x):
        h = x  # h is unrolled across the array
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    # Use convex optimization to enforce constraints directly
    h_var = cp.Variable(n_points)
    c5_var = cp.Variable()

    constraints = [
        cp.sum(h_var) == n_points / 2.0,
        0 <= h_var,
        h_var <= 1,
        cp.max(cp.conv(h_var, 1 - h_var, mode='full') * dx) <= c5_var,
    ]

    epsilons = np.logspace(-2, -5, 20)  # decreasing epsilon for tightening
    current_c5 = evaluate_erdos_solution(best_h, n_points)

    while time.time() - start_time < budget_s:
        for epsilon in epsilons:
            if time.time() - last_improvement_time > 120:
                break

            obj = c5_var + epsilon * cp.sum_squares(h_var - best_h)

            prob = cp.Problem(cp.Minimize(obj), constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    optimized_h = h_var.value
                    optimized_c5 = c5_var.value
                    if optimized_h is not None and optimized_c5 < current_c5:
                        best_h = np.clip(optimized_h, 0, 1)
                        best_h = best_h * (n_points / 2 / np.sum(best_h))
                        best_h = np.clip(best_h, 0, 1)
                        current_c5 = max(evaluate_erdos_solution(best_h, n_points), optimized_c5)
                        last_improvement_time = time.time()
                if time.time() - start_time >= budget_s - 1:
                    break
            except Exception as e:
                pass

    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return best_h, final_c5, n_points
