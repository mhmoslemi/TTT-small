import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

    h_init = enforce_normalization(h_init, n_points)

    # Initialize problem with a few key constraints
    h = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Base constraints
    base_shifts = [1, 2, 4, 8, 16]
    for k in base_shifts:
        if k >= n_points:
            continue
        lhs = cp.sum(h[:n_points - k] * (1 - h[k:k + n_points - k]))
        constraints.append(lhs * (2.0 / n_points) <= C)

    # Build and solve initial problem
    prob = cp.Problem(cp.Minimize(C), constraints)
    prob.solve(solver=cp.GUROBI, verbose=False)
    if prob.status != cp.OPTIMAL:
        return h_init, compute_c5(h_init, n_points), n_points

    h_best = h.value.flatten()
    h_best = enforce_normalization(h_best, n_points)
    C_best = compute_c5(h_best, n_points)

    time_start = time.time()
    early_stop = 100
    stagnation_count = 0

    while time.time() - time_start < budget_s:
        # Find the best violated constraint
        h_current = h.value.flatten()
        h_current = enforce_normalization(h_current, n_points)
        max_overlap = 0.0
        worst_k = 0
        for k in range(1, n_points):
            if n_points - k <= 0:
                continue
            lhs = np.dot(h_current[:n_points - k], 1 - h_current[k:k + n_points - k])
            lhs *= (2.0 / n_points)
            if lhs > max_overlap:
                max_overlap = lhs
                worst_k = k

        # Check if current C is higher than maximum overlap and add constraint
        if abs(max_overlap - C.value) > 1e-5:
            # Add constraint for worst_k
            lhs = cp.sum(h[:n_points - worst_k] * (1 - h[worst_k:worst_k + n_points - worst_k]))
            constraints.append(lhs * (2.0 / n_points) <= C)
            prob = cp.Problem(cp.Minimize(C), constraints)
            result = prob.solve(solver=cp.GUROBI, verbose=False)
            if prob.status != cp.OPTIMAL:
                break

            h_current = h.value.flatten()
            h_current = enforce_normalization(h_current, n_points)
            current_c5 = compute_c5(h_current, n_points)

            if current_c5 < C_best:
                C_best = current_c5
                h_best = h_current.copy()
                stagnation_count = 0

    h_refine = opt.minimize(

        method='L-BFGS-B',

    final_c5 = compute_c5(h_refine, n_points)

    return h_refine, final_c5, n_points
