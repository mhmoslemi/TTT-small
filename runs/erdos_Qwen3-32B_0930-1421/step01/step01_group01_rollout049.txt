import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points, refcheck=False)
    except:
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.9
        h[n_points // 2:] = 0.1
        h = h / h.sum() * n_points / 2

    h_init = h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    total_start = time.time()

    steps = 0
    solver_options = {"verbose": False, "max_iter": 200}

    while time.time() - total_start < budget_s:
        h_var = cp.Variable(n_points, pos=True)
        ones = np.ones(n_points)

        # Constraint: mass constraint
        mass_constraint = [cp.sum(h_var) <= n_points / 2 + 1e-3,
                           cp.sum(h_var) >= n_points / 2 - 1e-3]
        # Range constraint
        val_constraint = [h_var <= 1.0, h_var >= 0.0]

        constraints = mass_constraint + val_constraint
        h_correlated = correlate(h_var, 1.0 - h_var, mode="full")
        max_overlap = cp.Max(h_correlated * dx)

        objective = cp.Minimize(max_overlap)

        problem = cp.Problem(objective, constraints)
        try:
            lmbda = 1.5 if steps % 2 == 0 else 0.1

            # Introduce regularization
            reg = lmbda * cp.sum_squares(h_var - h)
            problem = cp.Problem(objective + reg, constraints)

            problem.solve(solver=cp.CVXOPT, **solver_options)
            h_sol = h_var.value
            if problem.status == "optimal":
                h_sol = np.clip(h_sol, 0, 1)
                h = rescale_h(h_sol)
                C5_val = evaluate_erdos_solution(h, dx)[0]
                if C5_val < best_c5:
                    best_c5 = C5_val
                    best_h = h.copy()

                # Perturbation to avoid local optima
                h = h + 0.03 * (np.random.rand(n_points) - 0.5)
                h = rescale_h(h)

            steps += 1
        except Exception as e:
            continue

    return best_h, best_c5, n_points


def rescale_h(h):
    h = np.clip(h, 0, 1)
    if h.sum() < 1e-10:
        h[0] = 1.0
    h = h / h.sum() * len(h) / 2
    return h
