import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1

    np.random.seed(seed)

    # Normalize h to meet integral constraints
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != target_sum:
        h *= target_sum / h_sum
    h = np.clip(h, 0, 1)
    h_sum = h.sum()

    # Enforce the constraint exactly
    while abs(h_sum - target_sum) > 1e-10:
        idx = np.argmin(h)
        h[idx] += 0.001
        h = np.clip(h, 0, 1)
        h_sum = h.sum()

    best_h = h.copy()
    best_c5 = np.inf

    def evaluate_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
        ones_minus_h = 1.0 - optimized_h

        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= n_points:
                continue
            lcorr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            rcorr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, lcorr, rcorr)
        return max_corr * dx

    best_c5 = evaluate_c5(h)

    start_iter = 0
    step_size = 0.01
    patience = 30
    last_improvement = 0

    while time.time() - start_time < budget_s:
        # Compute shifted overlaps to determine critical segments
        ones_minus_h = 1.0 - h
        max_shift = 0
        max_corr = -np.inf

        for k in range(1, n_points):
            if k >= n_points:
                continue
            lcorr = np.dot(h[:n_points - k], (1 - h[k:]))
            rcorr = np.dot(h[k:], (1 - h[:n_points - k]))
            if lcorr > max_corr or rcorr > max_corr:
                max_corr = max(lcorr, rcorr)
                max_shift = k

        # Targeted gradient descent on overlapping segment for max_shift
        grad = np.zeros_like(h)

        if max_shift < n_points:
            for i in range(n_points - max_shift):
                grad[i] += (1 - h[i + max_shift])
                grad[i + max_shift] -= h[i]

        # Normalize gradient to avoid large steps
        grad = grad / np.abs(grad).max()

        # Update h
        proposed_h = h - step_size * grad
        proposed_h = np.clip(proposed_h, 0, 1)
        proposed_sum = proposed_h.sum()

        if proposed_sum != target_sum:
            proposed_h *= target_sum / proposed_sum
        proposed_h = np.clip(proposed_h, 0, 1)

        new_c5 = evaluate_c5(proposed_h)

        if new_c5 < best_c5:
            h[:] = proposed_h
            best_c5 = new_c5
            best_h[:] = h
            last_improvement = time.time()
        else:
            # Try a random perturbation to escape local optima
            if np.random.rand() < 0.05:
                indices = np.random.choice(n_points, size=5, replace=False)
                pert = np.random.uniform(-0.02, 0.02, size=5)
                h[indices] += pert
                h = np.clip(h, 0, 1)
                h_sum = h.sum()
                if abs(h_sum - target_sum) > 1e-7:
                    h *= target_sum / h_sum

        if time.time() - last_improvement > 60 and start_iter > 10:
            step_size *= 0.5
            if step_size < 1e-4:
                break

        start_iter += 1

    return (best_h, best_c5, n_points)
