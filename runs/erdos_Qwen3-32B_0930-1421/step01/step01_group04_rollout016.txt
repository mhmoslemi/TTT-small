import numpy as np
import scipy.optimize as opt
import time
import random

def compute_c5(h, n_points):
    h = np.asarray(h)
    h_full = np.tile(h, 3)
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        sub_len = n_points - k
        corr = np.dot(h[:sub_len], (1 - h[k:k + sub_len]))
        corrs.append(corr)
    for k in range(1, n_points):
        sub_len = n_points - k
        corr = np.dot(h[k:], (1 - h[:sub_len]))
        corrs.append(corr)
    return np.max(np.array(corrs)) * (2.0 / n_points)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def cosine_init(n_points):
    x = np.linspace(0, np.pi, n_points, endpoint=False)
    h = 0.5 + 0.5 * np.cos(x)
    return enforce_normalization(h, n_points)

def objective(h_vec, n_points, exact=True):
    h = h_vec.reshape(-1)
    return compute_c5(h, n_points)

            return p_clipped

def weighted_proposal(n_points, h1, h2, alpha=0.5):

    return enforce_normalization(h, n_points)
