```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(2, n_points + 1):  # k=0 is allowed again in full overlap mode
        if k >= n_points:
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 47
    if init_h is not None:
        h = init_h.copy()
    else:
        h = np.random.rand(n_points)
    h = enforce_normalization(h, n_points)

    # Preparations
    def sample_random_h(num):
        h_list = []
        for i in range(num):
            hp = np.random.rand(n_points)
            hp = enforce_normalization(hp, n_points)
            h_list.append(hp)
        return np.array(h_list)

    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-3, 1e3))
    gpr = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=2, normalize_y=True)

    X = []
    y = []

    # Initial sampling for the GP
    X_init = sample_random_h(20)
    for h_init in X_init:
        y_init = compute_c5(h_init, n_points)
        X.append(h_init)
        y.append(y_init)

    X = np.array(X)
    y = np.array(y)

    gpr.fit(X, y)

    best_c5 = min(y)
    best_h = X[y.argmin()]

    T = 1000.0
    alpha = 0.99
    perturbation_sigma = 0.1
    evaluations = 0
    evaluation_time = []

    while time.time() - start_time < budget_s:
        # Propose new candidate from current best
        h_curr = best_h.copy()
        h_new = h_curr + np.random.normal(0, perturbation_sigma, n_points)
        h_new = enforce_normalization(h_new, n_points)

        # Predict using GP
        y_pred, y_std = gpr.predict([h_new], return_std=True)

        # Accept with Metropolis criterion
        y_curr = compute_c5(best_h, n_points)
        delta = y_pred[0] - y_curr

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
            y_new = compute_c5(h_new, n_points)
            best_c5 = min(best_c5, y_new)
            best_h = h_new
            X = np.vstack((X, h_new))
            y = np.append(y, y_new)
            gpr.fit(X, y)  # Refit the GP
        T *= alpha

        if time.time() - start_time > budget_s - 1:
            break

    # Final refinement with L-BFGS-B
    def objective(x):
        h = enforce_normalization(x, n_points)
        return compute_c5(h, n_points)

    result = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, options={'maxiter': 100, 'ftol': 1e-5})

    h_final = result.x
    h_final = enforce_normalization(h_final, n_points)
    final_c5 = compute_c5(h_final, n_points)

    return h_final, final_c5, n_points
```
