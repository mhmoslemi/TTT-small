import numpy as np
import time
import scipy.fft

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    # Step 1: Setup
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    # Precompute FFT axes
    h_fft = scipy.fft.fft(h_init, n_points)

    # Step 2: Optimization parameters
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()
    best_update = time.time()

    # Enforce constraints once at start
    def enforce_constraints(h, target_sum):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        h *= scale
        return np.clip(h, 0, 1)

    h[:] = enforce_constraints(h, target_sum)

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h.copy(), 0, 1)
        optimized_h_sum = optimized_h.sum()
        if np.abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum

        ones_minus_h = 1.0 - optimized_h

        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
            max_corr = max(max_corr, corr * dx)

        for k in range(1, n_points):
            if k >= optimized_h.shape[0] - 1:
                continue
            corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr * dx)

        return max_corr

    best_c5 = compute_c5(h)

    while time.time() - start_time < budget_s - 5:
        # Perturb high-frequency Fourier components
        freq_indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        h_fft_perturbed = h_fft.copy()
        noise = np.random.normal(0, 0.1, freq_indices.shape)
        noise += 1j * np.random.normal(0, 0.1, freq_indices.shape)
        h_fft_perturbed[freq_indices] += noise

        h_proposed = np.real(scipy.fft.ifft(h_fft_perturbed, n_points))

        h_proposed = enforce_constraints(h_proposed, target_sum)

        new_c5 = compute_c5(h_proposed)

        if new_c5 < best_c5:
            h[:] = h_proposed
            best_c5 = new_c5
            best_h[:] = h[:]
            best_update = time.time()

        if time.time() - best_update > 90 and best_c5 < 0.380876:
            break

        # Progress logging
        if verbose and (time.time() - start_time) % 10 < 2:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
