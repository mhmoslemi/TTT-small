import numpy as np
import time
import scipy.signal as signal


def evaluate_c5(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if abs(h_sum) < 1e-10:
        h[:] = 0.0
        h[0] = h.shape[0] / 2
    else:
        h = h / h_sum * h.shape[0] / 2
    cor = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(cor) * dx
    return c5, h

        h[42:] = 0.2

    h = h / h.sum() * n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)[0]
    start_time = time.time()

    tau = 0.1  # Perturbation scale
    decay_rate = 0.95  # per-epoch decay
    epochs = 0
    patience = 0
    max_patience = 200
    move_freq = 30

    while time.time() - start_time < budget_s:
        epochs += 1

        # Random segment flip or perturbation
        if epochs % move_freq == 0:
            # Select two random indices for segment
            l, r = np.sort(np.random.choice(n_points, 2, replace=False))
            # Sometimes flip, sometimes add noise
            if np.random.rand() < 0.5:
                h[l:r] = 1.0 - h[l:r]
            else:
                h[l:r] += np.random.randn(r - l) * tau
            h = np.clip(h, 0, 1)

        # Normalize h to meet the constraint
        h = h / h.sum() * n_points / 2

        # Compute C5 with this h
        c5, h = evaluate_c5(h, dx)

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
            move_freq = max(15, move_freq * 0.99)
        else:
            patience += 1

        # Reduce perturbation if no improvement
        if patience > max_patience:
            h = best_h.copy()
            patience = 0
            tau = max(0.01, tau * decay_rate)

        # Decay scale
        tau = max(0.01, tau * decay_rate)

    return best_h, best_c5, n_points
