import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def optimize_local_C5_k(h, k, dx, n_points, verbose=False):
    """Solve a local QP to reduce the overlap at a specific shift k."""
    h = cp.Parameter(n_points, nonneg=True)
    h_val = h.value if h.value is not None else np.random.rand(n_points)

    # Variables
    x = cp.Variable(n_points)
    # Constraints
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n_points / 2,
    ]
    # Objective: minimize h * (1 - h.shifted)
    overlap = 0
    for i in range(n_points):
        j = (i + k) % n_points
        overlap += x[i] * (1 - x[j])
    objective = cp.Minimize(overlap * dx)

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    x_sol = x.value
    return x_sol, prob.value * dx
