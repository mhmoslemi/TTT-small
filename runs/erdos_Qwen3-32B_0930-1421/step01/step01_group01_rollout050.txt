import numpy as np
import time
import cvxpy as cp


def evaluate_C5(h, dx):
    h = h.copy()
    h[h < 0] = 0
    h[h > 1] = 1
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2
    cor = np.correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx, h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h = np.linspace(0.0, 1.0, n_points)
        h = h / h.sum() * n_points / 2

    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    # Define symbolic variable
    h_variable = cp.Variable(n_points)
    obj_core = 0.0
    for k in range(len(h)):
        for i in range(len(h) - k):
            obj_core += (h_variable[i] + h_variable[i + k] - 1)**2

    objective = cp.Minimize(cp.sum(obj_core) + 0.01 * cp.sum_squares(h_variable))
    constraints = [cp.sum(h_variable) == n_points / 2, 0 <= h_variable, h_variable <= 1]
    problem = cp.Problem(objective, constraints)

    perturbations = [0.01, 0.05, 0.10, 0.20]
    for t in range(4):
        if time.time() - start_time > budget_s * 0.95:
            break

        h_seed = np.random.rand(n_points) * 0.5 + 0.25
        h_seed = h_seed / h_seed.sum() * n_points / 2
        h_variable.value = h_seed

        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iter=1000, warm_start=True)
            h_new = h_variable.value
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.abs(np.sum(h_new)) * n_points / 2
            c5, _ = evaluate_C5(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception as e:
            pass

        # Perturb and restart
        for p in perturbations:
            if time.time() - start_time > budget_s * 0.95:
                break

            h_next = (1 - p) * best_h + p * np.random.rand(n_points)
            h_next = h_next / h_next.sum() * n_points / 2
            h_variable.value = h_next

            try:
                problem.solve(solver=cp.ECOS, verbose=False, max_iter=1000, warm_start=False)
                h_new = h_variable.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new / np.abs(np.sum(h_new)) * n_points / 2
                c5, _ = evaluate_C5(h_new, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5, n_points
