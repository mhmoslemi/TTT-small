import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def mutation(h, sigma, n_points):
    h_mut = h + np.random.normal(0, sigma, n_points)
    return enforce_normalization(h_mut, n_points)

def crossover(p1, p2, alpha):
    h_c = alpha * p1 + (1 - alpha) * p2
    return enforce_normalization(h_c, len(p1))

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is None:
        raise ValueError("initial_h_values is required")
    n_points = len(init_h)
    population_size = 40
    mut_rate = 0.2
    elite_size = population_size // 2
    sigma_max = 0.1
    time_start = time.time()

    # Initialize population
    population = [init_h]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        population.append(enforce_normalization(h, n_points))

    # Evaluate initial population
    fitness = [compute_c5(h, n_points) for h in population]
    best_idx = np.argmin(fitness)
    best = population[best_idx].copy()
    best_val = fitness[best_idx]

    while time.time() - time_start < budget_s:
        # Selection
        indices = np.argsort(fitness)
        elites = [population[i] for i in indices[:elite_size]]
        new_population = elites.copy()

        # Crossover
        while len(new_population) < population_size:
            i, j = np.random.choice(elite_size, 2, replace=False)
            alpha = np.random.rand()
            child = crossover(population[i], population[j], alpha)
            new_population.append(child)

        # Mutation
        for idx in range(len(new_population)):
            if np.random.rand() < mut_rate:
                sigma = sigma_max * (1 - (time.time() - time_start) / budget_s)
                new_population[idx] = mutation(new_population[idx], sigma, n_points)

        # Evaluate new population
        population = new_population
        fitness = [compute_c5(h, n_points) for h in population]
        current_best = np.min(fitness)
        if current_best < best_val:
            best_idx = np.argmin(fitness)
            best = population[best_idx].copy()
            best_val = current_best

        # Early convergence check (reduce mutation intensity)
        if max(fitness) - min(fitness) < 1e-5:
            mut_rate *= 0.9

        # Final local refinement every 50 generations
        if (len(new_population)) % 40 == 0 and len(new_population) > 0:
            def local_obj(x):
                return compute_c5(x.reshape(-1), n_points)
            lb = np.zeros(n_points)
            ub = np.ones(n_points)
            res = np.optimize.minimize(lambda x: local_obj(x), best, bounds=list(zip(lb, ub)),
                                       method="L-BFGS-B", options={"maxiter": 50})
            if res.success:
                refined = enforce_normalization(res.x, n_points)
                refined_val = compute_c5(refined, n_points)
                if refined_val < best_val:
                    best = refined
                    best_val = refined_val

    return best, best_val, n_points
