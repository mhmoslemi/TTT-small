import numpy as np
import time
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    n_points = len(h)
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = n_points / 2
    else:
        h *= (n_points / 2) / total
    return h


def discretize_values(h, levels=[0.0, 0.25, 0.5, 0.75, 1.0]):
    n_points = len(h)
    discretized = np.take_closest(np.array(levels), h)
    return ensure_feasible(discretized)


def random_neighbor(h, mut_steps=3):
    h = np.copy(h)
    idxs = np.random.permutation(len(h))[:mut_steps]
    for i in idxs:
        h[i] = np.random.rand()
    return ensure_feasible(h)
