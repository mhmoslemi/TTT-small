import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # ensure ∫h dx = 1
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if h.shape[0] != n_points:
            h = np.resize(h, n_points)
    except:
        h = np.zeros(n_points)
        h = rescale_h(h)

    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)[0]
    start_time = time.time()

    iters = 0
    max_iter = 3000
    perturb_amp = 0.1
    perturb_fraction = 0.1
    segment_size = int(n_points * 0.1)
    prev_update_time = time.time()

    while time.time() - start_time < budget_s and iters < max_iter:
        iters += 1

        # Perturbation 1: random flip in a segment
        segment_start = np.random.randint(0, n_points - segment_size)
        h_flipped = best_h.copy()
        h_flipped[segment_start:segment_start + segment_size] = 1 - h_flipped[segment_start:segment_start + segment_size]
        h_flipped = rescale_h(h_flipped)

        # Perturbation 2: Gaussian noise in a segment
        h_noised = best_h.copy()
        noise_segment_len = min(n_points, max(int(n_points * perturb_fraction), 1))
        noise_segment_start = np.random.randint(0, n_points - noise_segment_len)
        noise_segment_end = noise_segment_start + noise_segment_len
        noise = np.random.normal(0, perturb_amp, size=noise_segment_len)
        h_noised[noise_segment_start:noise_segment_end] += noise
        h_noised = np.clip(h_noised, 0, 1)
        h_noised = rescale_h(h_noised)

        candidates = [h_flipped, h_noised]
        C5_list = [evaluate_C5(c, dx)[0] for c in candidates]

        for candidate, C5 in zip(candidates, C5_list):
            if C5 < best_C5:
                best_h = candidate.copy()
                best_C5 = C5

        # Dynamic perturbation tuning
        if iters % 100 == 0:
            perturb_amp *= 0.8

        # Safeguard logging
        if time.time() - prev_update_time > 60:
            prev_update_time = time.time()

    return best_h, best_C5, n_points
