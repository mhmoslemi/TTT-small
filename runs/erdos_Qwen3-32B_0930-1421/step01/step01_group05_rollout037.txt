import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        h = 0.5 * np.ones(n_points)  # Centralized start on 0-1 scale

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        scale = (n_points / 2) / sum_h
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_model_time = time.time() - start_time

    # Phase 1: Convex optimization with sparsity regularization (80% of time)
    lambda_vals = [0.05, 0.1, 0.2, 0.5, 1.0]
    phase1_time_limit = 0.8 * budget_s

    for lam in lambda_vals:
        if time.time() - start_time > phase1_time_limit:
            break

        h_opt = cp.Variable(n_points)
        objective = cp.Minimize(lam * cp.norm1(h_opt) + h_opt.T @ correlate(1 - h_opt, h_opt, mode='full') * dx)
        constraints = [
            h_opt >= 0,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, max_iters=200, verbose=False)
            h_proposed = np.clip(h_opt.value, 0, 1)

            # Check feasibility (unit integral & bounded)
            if not np.isclose(np.sum(h_proposed), n_points / 2):
                continue

            h_proposed = normalize_h(h_proposed)
            c5_proposed = compute_c5(h_proposed)

            if c5_proposed < best_c5:
                best_c5 = c5_proposed
                best_h = h_proposed.copy()
                print(f"Improved to {best_c5:.6f} with λ={lam}")
        except Exception as e:
            continue  # Skip invalid optimizations

    # Phase 2: Perturbation refinement (local search) (remaining time)
    learning_rate = 0.05
    patience_counter = 0
    max_patience = 200
    perturbation_scale = 0.1

    while time.time() - start_time < budget_s and patience_counter < max_patience:
        h_new = h.copy()
        idx = np.random.randint(n_points)
        h_new[idx] += perturbation_scale * (2 * np.random.rand() - 1)
        h_new = normalize_h(h_new)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1

    return best_h.tolist(), best_c5, n_points
