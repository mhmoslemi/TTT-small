import numpy as np
import time
import math
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.clip(h * scale_factor, 0.0, 1.0)

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time
    last_norm_time = start_time
    last_projection_time = start_time
    eps_decay = 1e-4  # Reduces constraint tolerance during successive iterations

    def objective(x):
        # x is the vector of h values
        c5 = evaluate_erdos_solution(x, n_points)
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    def optimize_step(optimizer="SLSQP", max_iter=100):
        nonlocal h, best_c5, best_h, last_improvement_time

        cons = (
            {"type": "eq", "fun": constraint_sum},
            {"type": "ineq", "fun": constraint_upper},
            {"type": "ineq", "fun": constraint_lower}
        )

        result = minimize(
            objective,
            h,
            method=optimizer,
            bounds=[(0.0, 1.0)] * n_points,
            constraints=cons,
            options={"maxiter": max_iter}
        )

        if result.success:
            new_h = result.x
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h.copy()
                last_improvement_time = time.time()

    def optimize_cvxpy():
        nonlocal h, best_c5, best_h, last_improvement_time
        h_var = cp.Variable(n_points)
        corr_all = np.correlate(h_var, 1 - h_var, mode='full')
        max_corr = cp.max(corr_all) * (2.0 / n_points)
        objective = cp.Minimize(max_corr)
        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        h_proposed = np.clip(h_var.value, 0.0, 1.0)

        delta_regularization = 1e-7
        h_proposed = h_proposed + delta_regularization * (np.random.random(n_points) - 0.5)  # perturbation to escape degeneracy
        h_proposed = np.clip(h_proposed, 0.0, 1.0)

        h_proposed_sum = np.sum(h_proposed)
        if abs(h_proposed_sum - n_points / 2.0) > 1e-5:
            h_proposed = h_proposed / h_proposed_sum * (n_points / 2.0)

        c5_val = evaluate_erdos_solution(h_proposed, n_points)
        if c5_val < best_c5:
            best_h[:] = h_proposed.copy()
            best_c5 = c5_val
            last_improvement_time = time.time()

    def perturb_best_h(energy=0.1):
        nonlocal h
        direction = np.random.rand(n_points)
        direction -= np.mean(direction)
        direction /= np.max(np.abs(direction))
        h += energy * np.random.uniform(-1, 1) * direction
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h / h_sum * (n_points / 2.0)

    def evaluate_kfold():
        nonlocal best_h, h, best_c5
        h_proj = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_proj)
        if h_sum != 0:
            h_proj = h_proj / h_sum * (n_points / 2.0)
        c5_val = evaluate_erdos_solution(h_proj, n_points)
        if c5_val < best_c5:
            best_h[:] = h_proj.copy()
            best_c5 = c5_val
            last_improvement_time = time.time()

    step_counter = 0
    try:
        while time.time() - start_time < budget_s - 1:
            t_remain = budget_s - (time.time() - start_time)

            if t_remain < 20:
                break

            step_counter += 1
            if step_counter % 3 == 0:
                optimize_cvxpy()
            else:
                optimize_step(optimizer="COBYLA", max_iter=int(min(50, 0.8 * t_remain)))

            # Perturbation step
            if step_counter % 5 == 0:
                perturb_best_h()

            # Ensure feasibility and constraint met
            if step_counter % 7 == 0:
                evaluate_kfold()

            if time.time() - last_improvement_time > 150:
                break

    except Exception as e:
        pass

    return best_h, best_c5, n_points
