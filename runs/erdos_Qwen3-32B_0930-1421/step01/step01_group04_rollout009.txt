import numpy as np
import time
import scipy.optimize as opt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def compute_c5(h_vec, n_points):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    corrs = []
    for k in range(1, n_points):
        corr1 = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr1)
    for k in range(1, n_points):
        corr2 = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr2)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def generate_candidate(h, noise_scale):
    h_p = h + np.random.normal(0, noise_scale, h.shape)
    return enforce_normalization(h_p, len(h))
