import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        if np.abs(sum_h) < 1e-12:
            return np.zeros_like(h_in)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    # Phase 1: Convex optimization with L1 regularization
    max_lamdas = 25
    best_c5_penalty = float('inf')
    lam = 0.01
    while time.time() - start_time < 0.7 * budget_s and lam < 100:
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(correlate(h_var, 1 - h_var, mode='full') * dx) + lam * cp.norm(h_var, 1))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.clip(h_var.value, 0, 1)
            h = normalize_h(h_sol)
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                best_c5_penalty = c5_val + lam * np.linalg.norm(h, 1)
        lam *= 1.25  # geometric step in lambda search

    # Phase 2: Local gradient descent after convex optimization solution
    learning_rate = 0.01
    for phase in range(2):
        while time.time() - start_time < budget_s:
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_p = h.copy()
                h_p[i] += eps
                h_p = normalize_h(h_p)
                cp = compute_c5(h_p)

                h_n = h.copy()
                h_n[i] -= eps
                h_n = normalize_h(h_n)
                cn = compute_c5(h_n)

                grad[i] = (cp - cn) / (2 * eps)
            h -= learning_rate * grad
            h = normalize_h(h)
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                learning_rate *= 1.0025  # encourage early exploration
            else:
                learning_rate *= 0.975
            if time.time() - start_time >= budget_s:
                break

    return best_h.tolist(), best_c5, n_points
