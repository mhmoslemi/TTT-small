```python
import numpy as np
import time
from scipy.optimize import minimize as optmin
from scipy.stats import norm
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return max(corrs) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, sigma, budget, n_points):
    h = enforce_normalization(h, n_points)
    pert = np.clip(np.random.randn(n_points) * sigma, -0.5, 0.5)
    h_p = enforce_normalization(h + pert, n_points)
    return h_p

def acquisition(x, gp, xi=0.01):
    mean, std = gp.predict([x], return_std=True)
    return - (mean - xi * std)  # Minimizing negative = maximizing acquisition

def build_gp(X, y):
    kernel = C(1.0, (1e-3, 1e1)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, normalize_y=False, n_restarts_optimizer=5)
    gp.fit(X, y)
    return gp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(47, 0.5)  # 1/2 distribution is feasible and centered
        h[0:2] = 1.0  # bias initial guess to be non-flat

    n_points = len(h)
    h = enforce_normalization(h, n_points)
    X = [h.ravel()]
    y = [compute_c5(h, n_points)]
    best_h = h.copy()
    best_val = y[0]
    t_start = time.time()

    # Collect more initial data for GP initialization
    for _ in range(5):
        h_init = perturb_h(h, 0.2, budget_s, n_points)
        val = compute_c5(h_init, n_points)
        X.append(h_init.ravel())
        y.append(val)

    x = np.array(X)
    y = np.array(y)

    gp = build_gp(x, y)
    iter_count = 0
    while time.time() - t_start < budget_s * 0.85:
        iter_count += 1
        if iter_count % 10 == 0:
            gp = build_gp(x, y)
        h_candidate = perturb_h(best_h, 0.1, budget_s, n_points)
        acq_val = acquisition(h_candidate.ravel(), gp)
        if acq_val < 0:
            h_candidate_val = compute_c5(h_candidate, n_points)
            x = x.tolist()
            y = y.tolist()
            x.append(h_candidate.ravel())
            y.append(h_candidate_val)
            x = np.array(x)
            y = np.array(y)
            if h_candidate_val < best_val:
                best_h = h_candidate.copy()
                best_val = h_candidate_val
        if iter_count % 10 == 0:
            xi = np.sqrt(2 * np.log(iter_count + 1))
        if iter_count % 100 == 0:
            print(f"Iter {iter_count}, best C5: {best_val:.6f}, time used: {time.time()-t_start:.1f}s")

    # Final refinement
    h_refined = optmin(fun=lambda x: compute_c5(x.reshape(n_points), n_points),
                       x0=best_h,
                       bounds=[(0.0, 1.0)] * n_points,
                       method='L-BFGS-B',
                       options={'maxiter': 100}).x
    h_refined = enforce_normalization(h_refined, n_points)
    final_val = compute_c5(h_refined, n_points)

    return h_refined, final_val, n_points
```
