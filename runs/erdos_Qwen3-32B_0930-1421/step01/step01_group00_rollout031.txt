import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    def compute_c5_fast(h):
        h = np.clip(h, 0, 1).copy()
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-7:
            h *= target_sum / h_sum

        ones_minus_h = 1.0 - h
        max_corr = -np.inf
        full = scipy.signal.correlate(h, ones_minus_h, mode='full')
        for i in range(-n_points + 2, n_points - 1):
            if i < 0:
                overlap_vals = full[n_points + i]
            else:
                overlap_vals = full[n_points - 1 + i]
            max_corr = max(max_corr, overlap_vals * dx)
        return max_corr

    def gradient_descend(h, k, dx):
        grad = np.zeros_like(h)
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k]) * dx
            grad[i + k] -= h[i] * dx
        return grad

    def project_h(h, target_sum):
        with np.errstate(invalid='ignore'):
            h = np.clip(h, 0, 1).copy()
        h_sum = h.sum()
        if abs(h_sum - target_sum) > 1e-6:
            h *= target_sum / h_sum
        return h

    h = h_init.copy()
    h = project_h(h, target_sum)

    best_c5 = compute_c5_fast(h)
    best_h = h.copy()
    best_time = start_time

    step_size = 0.01
    perturb_period = 60  # seconds between perturbations

    while time.time() - start_time < budget_s:
        c5 = compute_c5_fast(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_time = time.time()

        # Identify shift that gives max overlap
        _, corr = scipy.signal.correlate2d(h.reshape(1, -1), (1 - h).reshape(1, -1), mode='full', boundary='fill', fillvalue=0)
        max_overlaps = [corr[0, k] * dx for k in range(1, n_points)]
        max_overlaps += [corr[0, -k - 1] * dx for k in range(1, n_points - 1)]
        k_max = np.argmax(max_overlaps)
        sign = -1 if k_max >= len(max_overlaps) // 2 else 1
        k_overlap = abs(1 - k_max) % (n_points - 1)

        # Gradient descent on max overlap
        grad = gradient_descend(h, k_overlap, dx)
        h -= step_size * grad

        # Project back to feasible space
        h = project_h(h, target_sum)

        # Perturb every few seconds
        if time.time() - best_time > perturb_period:
            indices = np.random.choice(range(n_points), size=min(10, n_points), replace=False)
            h[indices] += np.random.uniform(-0.02, 0.02, size=len(indices))
            h = project_h(h, target_sum)
            best_time = time.time()

        # Early stop if no better result in 180s
        if time.time() - best_time > 180:
            break

    return (best_h, best_c5, n_points)
