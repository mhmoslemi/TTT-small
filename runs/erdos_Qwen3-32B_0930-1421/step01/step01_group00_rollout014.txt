import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce constraints
    h = np.clip(h, 0, 1)
    h *= target_sum / h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        s = optimized_h.sum()
        if abs(s - target_sum) > 1e-6:
            optimized_h *= target_sum / s
        ones_minus_h = 1 - optimized_h
        c5 = -np.inf
        for k in range(1, optimized_h.size):
            if k >= optimized_h.size - 1:
                continue
            corr1 = optimized_h[:optimized_h.size - k] * ones_minus_h[k:]
            corr1 = np.sum(corr1) * dx
            corr2 = optimized_h[k:] * ones_minus_h[:optimized_h.size - k]
            corr2 = np.sum(corr2) * dx
            c5 = max(c5, corr1, corr2)
        return c5

    def directional_derivative(h, k, step=0.01):
        grad = np.zeros_like(h)
        for i in range(h.size - k):
            grad[i] += -(1 - h[i + k]) * step
            grad[i + k] += h[i] * step
        for i in range(h.size - k):
            grad[h.size - k + i] += -(1 - h[i]) * step
            grad[i] += h[h.size - k + i] * step
        return grad

    last_improvement = start_time
    h = best_h.copy()

    while time.time() - start_time < budget_s:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            last_improvement = time.time()

        if best_c5 <= 0.380876:
            return best_h, best_c5, n_points

        correlations = np.zeros(n_points - 1)
        for k in range(1, n_points):
            if k >= h.size - 1:
                continue
            corr1 = h[:n_points - k] @ (1 - h[k:]) * dx
            corr2 = h[k:] @ (1 - h[:n_points - k]) * dx
            correlations[k - 1] = max(corr1, corr2)

        max_k = np.argmax(correlations) + 1
        grad = directional_derivative(h, max_k)
        alpha = 0.01

        while True:
            h_candidate = h - alpha * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            s = h_candidate.sum()
            if s == 0:
                break
            h_candidate *= target_sum / s
            h_candidate = np.clip(h_candidate, 0, 1)
            if (h_candidate == h).all():
                alpha *= 0.5
                if alpha < 1e-6:
                    break
                continue
            h[:] = h_candidate
            current_c5 = compute_c5(h)
            if current_c5 >= best_c5:
                alpha *= 0.5
            else:
                break

        # Perturbation to escape local minima
        if time.time() - last_improvement > budget_s * 0.1:
            h = best_h.copy()
            indices = np.random.choice(n_points, min(10, n_points), replace=False)
            noise = np.random.uniform(-0.05, 0.05, size=indices.shape)
            h[indices] += noise
            h = np.clip(h, 0, 1)
            s = h.sum()
            if s != 0:
                h *= target_sum / s
            h = np.clip(h, 0, 1)

            last_improvement = time.time()

        if verbose and (time.time() - start_time) % 10 < 0.1:
            print(f"Elapsed: {time.time()-start_time:.1f}s, Best C5: {best_c5:.8f}")

    return best_h, best_c5, n_points
