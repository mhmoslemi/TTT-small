import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def optimistic_perturb(h, scale=0.1):
    n_points = len(h)
    h_perturbed = h + np.random.uniform(-scale, scale, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    # Enforce sum constraint
    current_sum = np.sum(h_perturbed)
    if current_sum > 0:
        h_perturbed = h_perturbed / current_sum * (n_points / 2.0)
    else:
        h_perturbed = np.ones_like(h_perturbed) / n_points * (n_points / 2.0)
    return h_perturbed

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Normalize to enforce sum constraint
    current_sum = np.sum(initial_h)
    if np.isclose(current_sum, 0):
        h = np.ones(n_points) * (n_points / 2.0) / n_points
    else:
        h = initial_h * (n_points / 2.0 / current_sum)
    h = np.clip(h, 0, 1)
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)

    # Convert to CVXPY variables
    h_var = cp.Variable(n_points, nonneg=True)
    z_var = {}
    for k in range(1, n_points):
        for i in range(n_points - k):
            key = (i, i+k)
            z_var[key] = cp.Variable(nonneg=True)

    def build_convex_restriction(C5):
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1,
            h_var >= 0,
        ]
        for key in z_var:
            i, j = key
            z_ij = z_var[key]
            h_i = h_var[i]
            h_j = h_var[j]
            constraints += [
                z_ij <= h_i,
                z_ij <= 1 - h_j,
                z_ij >= h_i - h_j
            ]
        for k_shift in range(1, n_points):
            corr_sum = sum(z_var.get((i, i + k_shift), 0) for i in range(n_points - k_shift))
            constraints += [corr_sum <= C5 * dx]
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        return prob

    # Warm-start the search with initial bound
    lower = 0.0
    upper = best_c5
    best_c5_candidate = best_c5
    best_h_candidate = np.copy(h)

    def solve_and_update(C5_to_try):
        nonlocal best_c5_candidate, best_h_candidate
        try:
            prob = build_convex_restriction(C5_to_try)
            prob.solve(solver="GUROBI", verbose=False)
            if prob.status in [cp.OPTIMAL, 'optimal']:
                best_h_candidate = h_var.value.copy()
                best_h_candidate = np.clip(best_h_candidate, 0, 1)
                return True
            else:
                return False
        except:
            return False

    while time.time() - start_time < budget_s - 20:
        mid = (lower + upper) / 2
        if solve_and_update(mid):
            best_c5_candidate = mid
            upper = mid
        else:
            lower = mid
        if time.time() - start_time < budget_s - 20:
            # Perturb the solution and try again
            h_perturbed = optimistic_perturb(best_h_candidate)
            best_h_candidate = h_perturbed
            best_c5_candidate = evaluate_erdos_solution(h_perturbed, n_points)
            lower = 0
            upper = best_c5_candidate

        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = best_h_candidate.copy()

    # Final improvement pass with basic local optimization
    def objective(x):
        return evaluate_erdos_solution(x, n_points)
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    def constraint_upper(x):
        return 1.0 - x
    def constraint_lower(x):
        return x
    cons = (
        {"type": "eq", "fun": constraint_sum},
        {"type": "ineq", "fun": constraint_upper},
        {"type": "ineq", "fun": constraint_lower}
    )
    res = minimize(
        objective,
        best_h,
        method="L-BFGS-B",
        bounds=[(0, 1)]*n_points,
        constraints=cons,
        options={"maxiter": 100}
    )
    if res.success:
        final_h = res.x
        best_c5 = evaluate_erdos_solution(final_h, n_points)
        best_h = final_h

    return best_h, best_c5, n_points
