import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(budget_s=1100, n_points=42, initial_cutting_plane_k=None):
    start_time = time.time()

    if initial_h_values is not None:
        h_init = np.copy(initial_h_values)
        sum_h = h_init.sum()
        if not np.isclose(sum_h, n_points / 2):
            h_init = h_init * (n_points / 2) / sum_h
    else:
        h_init = np.zeros(n_points)
        h_init[: n_points // 2] = 1.0
    h = cp.Variable(n_points, nonneg=True)
    zero_one_constraints = [h <= 1.0]
    meanConstraint = [cp.sum(h) == n_points / 2]

    all_ks = [k for k in range(1, n_points)]
    if initial_cutting_plane_k is not None:
        active_k = [initial_cutting_plane_k]
    else:
        active_k = []

    dx = 2.0 / n_points
    num_constraints_added = 0
    best_C5 = np.inf
    best_h = h_init

    while time.time() - start_time < budget_s:
        objective = cp.Minimize(cp.Constant(0))
        constraints = zero_one_constraints + meanConstraint

        # Build C5 minimization problem
        for k in active_k:
            offset = k
            for i in range(n_points - offset):
                constraints.append(h[i] + h[i + offset] - h[i] * h[i + offset] <= 1)
                constraints.append(h[i] + h[i + offset] - h[i] * h[i + offset] >= 0)
            for i in range(n_points - offset):
                constraints.append(cp.Constant(1) * h[i] - h[i] * h[i + offset] >= 0)

        # Objective is the maximum overlap
        obj_c5 = cp.Variable()
        for i in range(n_points - offset):
            obj_c5 += 1 * h[i] - h[i] * h[i + offset]
        objective = cp.Minimize(obj_c5 * dx)

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False)
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break

        feasible_h = h.value
        feasible_h[feasible_h < 0.5] = 0
        feasible_h[feasible_h >= 0.5] = 1

        feasible_h = feasible_h * (n_points / 2) / feasible_h.sum()

        candidates = []
        for k in range(1, n_points):
            overlap = correlate(feasible_h, 1 - feasible_h, mode='full')
            c5 = np.max(overlap[k:]) * dx
            candidates.append((c5, k))

        new_c5, new_k = min(candidates)
        if new_c5 < best_C5:
            best_C5 = new_c5
            best_h = feasible_h

        if num_constraints_added >= len(all_ks):
            break

        next_k = all_ks.pop(0)
        active_k.append(next_k)
        num_constraints_added += 1

    return best_h, best_C5, n_points
