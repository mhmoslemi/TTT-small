import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def run(budget_s=1100):
    start_time = time.time()

    n_points = 42
    h_init = np.array(ensure_feasible(initial_h_values)) if initial_h_values is not None else np.ones(n_points) / n_points * n_points / 2

    h = cp.Variable(n_points, nonneg=True)
    x = h
    corr_k = []

    # Circular convolution indices (shifts k = -n + 1, ..., n - 1)
    for k in range(n_points):
        shifted_u = np.zeros(n_points)
        shifted_d = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            shifted_u[i] = x[i]
            shifted_d[i] = (1 - x[j])
        corr_k.append(cp.sum(cp.multiply(shifted_u, shifted_d)))

    lambda_max = cp.Variable()
    dx = 2.0 / n_points

    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ]

    overlap_constraints = [
        corr * dx <= lambda_max for corr in corr_k
    ]
    constraints += overlap_constraints

    objective = cp.Minimize(lambda_max)

    prob = cp.Problem(objective, constraints)

    prob.solve(solver=cp.ECOS, max_iters=10000)
    h_sol = np.array(h.value).flatten()

    C5 = evaluate_C5(h_sol)
    best_h = ensure_feasible(h_sol)

    def refine_grid(h_fine):
        h_fine = np.zeros(len(h_fine))
        for i in range(len(h_fine)):
            if i < n_points:
                h_fine[i] = h_sol[i]
            else:
                h_fine[i] = h_sol[i - n_points]
        return h_fine

    n_refined = 126
    h_refined = refine_grid(np.zeros(n_refined))
    prob_refined = cp.Problem(objective,
        [
            cp.sum(h_refined) == n_refined / 2,
            h_refined <= 1,
            h_refined >= 0
        ] + [corr * dx <= lambda_max for corr in corr_k * (n_refined // n_points)]
    )
    prob_refined.solve(solver=cp.ECOS)

    if prob_refined.value < C5:
        best_h = np.array(h_refined.value)
        best_h = ensure_feasible(best_h)
        C5 = evaluate_C5(best_h)

    return best_h, C5, n_points
