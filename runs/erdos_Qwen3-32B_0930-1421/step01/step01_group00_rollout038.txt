import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Constraint handling
    h = np.clip(h, 0, 1)
    scale = target_sum / h.sum()
    h *= scale
    h = np.clip(h, 0, 1)

    # Ensure the sum constraint is met with floating-point tweak
    while abs(h.sum() - target_sum) > 1e-7:
        error = h.sum() - target_sum
        idx_low = h.argmin()
        idx_high = h.argmax()
        step = 2 * abs(error) / n_points
        if error < 0:
            h[idx_high] = max(0.0, h[idx_high] - step)
        else:
            h[idx_low] = min(1.0, h[idx_low] + step)

    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float("inf")

    def calc_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-7:
            optimized_h *= target_sum / optimized_h_sum
            optimized_h = np.clip(optimized_h, 0, 1)

        ones_minus_h = 1.0 - optimized_h
        max_corr = -np.inf
        for k in range(1, n_points):
            if k >= optimized_h.shape[0]:
                continue
            corr_fw = np.dot(optimized_h[:-k], ones_minus_h[k:])
            corr_bw = np.dot(optimized_h[k:], ones_minus_h[:-k])
            max_corr = max(max_corr, corr_fw * dx, corr_bw * dx)
        return max_corr

    best_c5 = calc_c5(h)
    best_h = h.copy()
    last_update = time.time()

    while time.time() - start_time < budget_s:
        # Identify worst shifts
        corr_vals = []
        for k in range(1, n_points):
            if k >= h.shape[0]:
                continue
            corr_fw = np.dot(h[:-k], (1.0 - h[k:]))
            corr_bw = np.dot(h[k:], (1.0 - h[:-k]))
            corr_vals.append((k, max(corr_fw * dx, corr_bw * dx)))
        worst_shifts = sorted(corr_vals, key=lambda x: x[1], reverse=True)[:5]

        # Apply focused perturbations
        modified = False
        for k, _ in worst_shifts:
            # Find indices that contribute
            raw_corr_fw = h[:-k] * (1.0 - h[k:])
            raw_corr_bw = h[k:] * (1.0 - h[:-k])
            top_indices_fw = np.argsort(raw_corr_fw)[-max(1, int(0.1 * len(raw_corr_fw))) :]
            top_indices_bw = np.argsort(raw_corr_bw)[-max(1, int(0.1 * len(raw_corr_bw))) :]

            # Apply smoothing shifts
            for i in top_indices_fw:
                h[i] = max(0.0, h[i] - 0.01)
                h[i + k] = min(1.0, h[i + k] + 0.01)
            for i in top_indices_bw:
                h[i + k] = max(0.0, h[i + k] - 0.01)
                h[i] = min(1.0, h[i] + 0.01)

            # Recompute constraint
            h = np.clip(h, 0, 1)
            scale = target_sum / h.sum()
            h *= scale
            h = np.clip(h, 0, 1)

            # Update best if improved
            new_c5 = calc_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_update = time.time()
                modified = True

        if not modified:
            # Degenerate case: shake to escape local minima
            h += np.random.normal(0, 0.005, size=h.shape)
            h = np.clip(h, 0, 1)
            scale = target_sum / h.sum()
            h *= scale
            h = np.clip(h, 0, 1)

        # Early stopping check
        if time.time() - last_update > max(60, 0.05 * budget_s):
            break

    # Final constraint check
    if abs(h.sum() - target_sum) > 1e-7:
        h = best_h.copy()

    return (best_h, best_c5, n_points)
