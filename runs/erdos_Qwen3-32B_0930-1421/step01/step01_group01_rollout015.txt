import numpy as np
import time
import scipy.signal


def run(seed=42, budget_s=1100, n_points=None):
    """Optimize the Erdős C5 constant using systematic perturbation and rescaling."""
    np.random.seed(seed)
    # Set n_points from parent if available, else use default
    n_points = len(initial_h_values) if "initial_h_values" in globals() and n_points is None else 84
    dx = 2.0 / n_points

    # Initialize h
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.zeros(n_points)
        h[:20] = 0.5
        h[20:40] = 1.0
        h[40:60] = 0.3
        h[60:] = 0.1
        h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)  # rescale to integral constraint

    def evaluate_C5_fast(h, dx):
        h = np.clip(h, 0, 1).astype(np.float64)
        normalize_factor = h.sum() / (n_points / 2)
        h = h / normalize_factor
        cor = scipy.signal.correlate(h, (1 - h), mode="full")
        C5 = np.max(cor) * dx
        return C5, h

    best_h = h.copy()
    best_C5 = evaluate_C5_fast(best_h, dx)[0]
    start_time = time.time()
    amplitude = 0.1
    freq_adjust = 50
    perturb_fraction = 0.1  # segment size as fraction of n_points
    segment_length = int(n_points * perturb_fraction)
    max_iter = int(2e3)

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        h = best_h.copy()

        # Perturbation 1: segment flip
        start_idx = np.random.randint(0, n_points - segment_length)
        h[start_idx : start_idx + segment_length] = 1 - h[start_idx : start_idx + segment_length]

        # Perturbation 2: small Gaussian noise
        noise = amplitude * (2 * np.random.rand(n_points) - 1)
        h += noise

        # Feasibility clipping and rescaling
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)

        # Compute new C5
        new_C5, new_h = evaluate_C5_fast(h, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()

        # Adapt algorithm parameters
        if i % freq_adjust == 0:
            amplitude *= 0.8  # decay perturbation amplitude over time
            freq_adjust = min(freq_adjust * 2, 1000)

    return best_h, best_C5, n_points
