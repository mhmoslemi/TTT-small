import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c)

def normalize_h(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = h_clipped.sum()
    scale = (n_points / 2) / total
    return np.clip(h_clipped * scale, 0, 1)

    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # CVXOPT convex optimization with L1 regularization
    h_var = cp.Variable(n_points)
    overlap = cp.Maximize(compute_c5(h_var, dx))
    objective = cp.Minimize(overlap + 0.1 * cp.norm1(h_var))

    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
    ]

    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.CBC, verbose=False)
        h_cvx = np.clip(h_var.value, 0, 1)
        h_cvx = normalize_h(h_cvx, n_points)
        c5_cvx = compute_c5(h_cvx, dx)
        if c5_cvx < best_c5:
            best_h = h_cvx
            best_c5 = c5_cvx
    except:
        pass

    # Gradient descent refinement
    learning_rate = 0.01
    max_steps = 10000
    step = 0
    while time.time() - start_time < budget_s and step < max_steps:
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_p = best_h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p, n_points)
            c_p = compute_c5(h_p, dx)

            h_n = best_h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n, n_points)
            c_n = compute_c5(h_n, dx)

            grad[i] = (c_p - c_n) / (2 * eps)

        best_h -= learning_rate * grad
        best_h = normalize_h(best_h, n_points)
        current_c5 = compute_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= 0.999
        step += 1

    return best_h.tolist(), best_c5, n_points
