import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)

    # Normalize
    total_h = float(np.sum(h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.clip(h * scale_factor, 0.0, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    last_improvement_time = start_time

    dx = 2.0 / n_points
    n = n_points

    # Convex problem parameters
    h_var = cp.Variable(n)
    z_vars = {}
    for k in range(1, n):
        indices = [(i, i + k) for i in range(n - k)]
        for i, j in indices:
            z_vars[(i, j)] = cp.Variable(nonneg=True)

    # Constraints
    constraints = [
        cp.sum(h_var) == n / 2,
        h_var >= 0,
        h_var <= 1
    ]

    # McCormick inequalities for z_{i,j} = h_i * (1 - h_j)
    for (i, j), z in z_vars.items():
        hj = h_var[j]
        hi = h_var[i]
        constraints += [
            z >= 0,
            z <= hi,
            z <= 1 - hj,
            z >= hi + (1 - hj) - 1
        ]

    # Correlation constraints
    for k in range(1, n - 1):
        corr_sum = 0
        for i in range(n - k):
            corr_sum += z_vars.get((i, i + k), cp.Constant(0))
        constraints.append(corr_sum <= best_c5 * dx)

    # Objective
    c5 = cp.Variable(nonneg=True)
    obj = cp.Minimize(c5)

    prob = cp.Problem(obj)

    num_outer_loops = 0
    while time.time() - start_time < budget_s - 10 and num_outer_loops < 10:
        C_low = 0
        C_high = best_c5
        for _ in range(30):
            if C_high <= C_low:
                break
            C_mid = (C_low + C_high) / 2
            for k in range(1, len(constraints) - (n - 2)):  # remove old correlation constraints
                if "corr" in str(constraints[k]):
                    constraints.pop(k)

            for k in range(1, n - 1):
                corr_sum = 0
                for i in range(n - k):
                    corr_sum += z_vars.get((i, i + k), cp.Constant(0))
                constraints.append(corr_sum <= C_mid * dx)

            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.GUROBI, verbose=False)

            if prob.status == cp.OPTIMAL:
                best_c5 = C_mid
                best_h = np.array(h_var.value).flatten()
                last_improvement_time = time.time()
                C_high = C_mid
            else:
                C_low = C_mid

        # Perturb for next round
        perturbation = np.random.uniform(-0.1, 0.1, n)
        h = np.clip(best_h + perturbation, 0.0, 1.0)
        total_h = np.sum(h)
        h = np.clip(h * (n / 2 / total_h), 0.0, 1.0)
        best_h = np.copy(h)
        best_c5 = evaluate_erdos_solution(best_h, n)
        num_outer_loops += 1

    # Final post-check and normalization
    total_h = np.sum(best_h)
    best_h = np.clip(best_h * (n / 2 / total_h), 0.0, 1.0)
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    return best_h, best_c5, n_points
