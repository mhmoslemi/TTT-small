import numpy as np

import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Retrieve initial_h_values if available
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Simple adaptive ILP-inspired greedy optimization with constraints
    for t in range(1000):
        if time.time() - start_time >= budget_s:
            break

        # Compress to 0-1 by thresholding around 0.5
        threshold = 0.5
        h_bin = (h >= threshold).astype(int)
        h_float = h.copy()
        # Only consider changing the middle 10%: most uncertain
        i = np.random.randint(0, n_points)
        h_float[i] = 1.0 - h_float[i]

        # Renormalize
        h_float = normalize_h(h_float)
        c5 = compute_c5(h_float)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_float.copy()
            h = h_float.copy()
        elif np.random.rand() < 0.1:
            # Occasionally allow a random perturbation for escape
            i = np.random.randint(0, n_points)
            h[i] = np.random.rand()
            h = normalize_h(h)

    # Final gradient descent with decreasing learning rate
    learning_rate = 0.1
    eps = 1e-5
    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        h -= learning_rate * grad
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= 0.995
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
