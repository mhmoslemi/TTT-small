import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    # Load initial solution and setup
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce constraints strictly
    def constrain_h(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if not np.isclose(h_sum, target_sum):
            h = h * target_sum / h_sum
        h = np.clip(h, 0.0, 1.0)
        return h

    h = constrain_h(h)
    best_h = h.copy()
    best_c5 = np.inf
    best_time = start_time

    # Use convex optimization as a baseline solver
    def convex_optimization_step(initial_h):
        x = cp.Variable(n_points)
        obj = cp.Minimize(cp.max(
            [cp.sum((initial_h[i] * (1 - initial_h[i + k])) * dx for i in range(n_points - k))
             for k in range(1, n_points)]
        ))
        constraints = [0 <= x, x <= 1, cp.sum(x) == target_sum]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False, max_iter=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return np.clip(x.value, 0.0, 1.0)
        except:
            pass
        return initial_h

    if time.time() - start_time < 120:
        h_co = convex_optimization_step(h)
        if h_co is not None:
            h = constrain_h(h_co)

    # Build a correlation cache to reduce redundant computation
    def compute_c5_fast(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = h.sum()
        if not np.isclose(h_sum, target_sum):
            h = h * target_sum / h_sum
        h = np.clip(h, 0.0, 1.0)

        ones_minus_h = 1.0 - h
        corr_values = []

        for k in range(1, n_points):
            if k >= n_points:
                corr = 0.0
            else:
                corr = np.sum(h[:n_points - k] * ones_minus_h[k:])
                corr_values.append(corr * dx)

        for k in range(1, n_points):
            if k >= n_points:
                corr = 0.0
            else:
                corr = np.sum(h[k:] * (1.0 - h[:n_points - k]))
                corr_values.append(corr * dx)

        max_corr = np.max(corr_values)
        return max_corr

    best_c5 = compute_c5_fast(h)

    # Perturbation with adaptive step size
    step_size = 0.02
    perturb_size = min(10, n_points)

    while time.time() - start_time < budget_s:
        if np.random.rand() < 0.2:
            step_size = max(0.005, step_size * 0.98)

        indices = np.random.choice(n_points, size=perturb_size, replace=False)
        proposed_h = h.copy()
        noise = np.random.uniform(-step_size, step_size, size=perturb_size)
        proposed_h[indices] += noise
        proposed_h = constrain_h(proposed_h)

        # Fast C5 evaluation on proposed_h
        new_c5 = compute_c5_fast(proposed_h)

        if new_c5 < best_c5:
            h[:] = proposed_h
            best_h[:] = h.copy()
            best_c5 = new_c5
            best_time = time.time()
            step_size = max(0.005, step_size * 1.05)
        else:
            step_size = max(0.005, step_size * 0.95)

        if time.time() - best_time > 60 and best_c5 < 0.380876:
            break

        if verbose and (time.time() - start_time) % 10 < 1:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    final_c5 = compute_c5_fast(best_h)

    return (best_h, final_c5, n_points)
