import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h0 = np.copy(initial_h_values)
    n_points = len(h0)
    total_h0 = float(np.sum(h0))
    scale_factor = (n_points / 2.0) / total_h0
    h0 = np.minimum(h0 * scale_factor, 1.0)

    num_particles = 50
    max_velocity = 0.2
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    reset_fraction = 0.1
    max_iter = 500

    # Initialize particles
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best = particles.copy()
    personal_best_values = np.full(num_particles, np.inf)
    global_best = h0.copy()
    best_c5 = evaluate_erdos_solution(global_best, n_points)

    def project(h):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            h[:] = 0.5  # fallback for numerical safety
        else:
            h *= (n_points / 2.0) / s
        return h

    def init_particle(i):
        if i < num_particles // 10:
            particle = project(np.copy(h0))
        else:
            noise = np.random.uniform(-0.01, 0.01, n_points)
            particle = project(h0 + noise * scale_factor)
        return particle

    for i in range(num_particles):
        particles[i] = init_particle(i)
        particles[i] = project(particles[i])
        c5 = evaluate_erdos_solution(particles[i], n_points)
        if c5 < personal_best_values[i]:
            personal_best[i] = particles[i].copy()
            personal_best_values[i] = c5
        if c5 < best_c5:
            global_best[:] = particles[i].copy()
            best_c5 = c5

    # Optimization loop
    iters = 0
    last_improvement_time = start_time

    while time.time() - start_time < budget_s:
        if iters >= max_iter:
            break
        iters += 1
        inertia = w_start - (w_start - w_end) * (iters / max_iter)

        r1 = np.random.rand(num_particles, 1)
        r2 = np.random.rand(num_particles, 1)

        velocities = (inertia * velocities +
                      c1 * r1 * (personal_best - particles) +
                      c2 * r2 * (global_best - particles))

        np.clip(velocities, -max_velocity, max_velocity, out=velocities)
        particles += velocities

        for i in range(num_particles):
            particles[i] = project(particles[i])
            c5 = evaluate_erdos_solution(particles[i], n_points)
            if c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = c5
            if c5 < best_c5:
                global_best[:] = particles[i].copy()
                best_c5 = c5
                last_improvement_time = time.time()

        # Reset stalled particles
        stalled = (personal_best_values > best_c5 + 1e-3)
        reset_count = int(reset_fraction * num_particles)
        for i in np.random.permutation(num_particles)[stalled][:reset_count]:
            particles[i] = project(np.random.uniform(0.0, 1.0, n_points))

        if time.time() - last_improvement_time > 60:
            # Restart with small perturbations if no progress
            particles = np.clip(particles + np.random.uniform(-0.01, 0.01, particles.shape), 0, 1)
            for i in range(num_particles):
                particles[i] = project(particles[i])

    best_h, final_c5 = global_best, best_c5

    return best_h, final_c5, n_points
