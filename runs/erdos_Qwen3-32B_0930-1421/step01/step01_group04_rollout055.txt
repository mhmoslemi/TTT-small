import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = np.bitwise_and(np.dot(h, 1 - h) < 1, np.abs(np.arange(n_points)[:, None] - np.arange(n_points)[None, :]) < n_points * 0.8)
    return np.max(np.diag(np.dot(np.outer(h, 1 - h), corrs.toarray()[:, None])).max() * dx)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    return (h * (n_points / 2) / total).clip(0, 1)

def initial_population(n_points, size=50, seed=42):
    np.random.seed(seed)
    pop = []
    for i in range(size):
        rand_h = np.random.rand(n_points)
        pop.append(enforce_normalization(rand_h, n_points))
    if 'initial_h_values' in globals():
        pop = [initial_h_values.copy()] + pop
    return pop

def mutation(h, std_dev=0.05):
    noise = np.random.normal(loc=0, scale=std_dev, size=len(h))
    return enforce_normalization(h + noise, len(h))

def crossover(h1, h2, alpha=None):
    alpha = alpha if alpha is not None else np.random.rand()
    child = alpha * h1 + (1 - alpha) * h2
    return enforce_normalization(child, len(h1))

def evaluate_population(pop, n_points):
    return [compute_c5(ind, n_points) for ind in pop]

def select_parents(pop, scores, num_parents):
    ranked_indices = np.argsort(scores)
    return [pop[i] for i in ranked_indices[:num_parents]]

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    time_start = time.time()
    
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 47
    
    # Initialize population
    population = initial_population(n_points, size=max(50, n_points // 5 + 1), seed=seed)
    scores = evaluate_population(population, n_points)
    best_index = np.argmin(scores)
    best_h = population[best_index].copy()
    best_c5 = scores[best_index]

    # Genetic search parameters
    gen = 0
    eval_budget = 0.8 * budget_s
    local_refine_intervals = 5

    while time.time() - time_start < budget_s:
        # Select top parent candidates
        parents = select_parents(population, scores, len(population) // 2)
        
        # Generate offspring via crossover and mutation
        offspring = []
        for _ in range(len(parents) // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            offspring.append(child)
        
        for p in parents:
            child = mutation(p, std_dev=0.1 * (1 - (time.time() - time_start) / budget_s))
            offspring.append(child)
        
        # Evaluate offspring
        offspring_scores = evaluate_population(offspring, n_points)
        offspring_with_scores = list(zip(offspring, offspring_scores))
        offspring_with_scores.sort(key=lambda x: x[1])
        
        # Merge and select new population
        population = parents + [ind for ind, _ in offspring_with_scores[:]]  # preserve parents
        scores = [compute_c5(ind, n_points) for ind in population]
        
        # Maintain minimum size
        while len(population) > max(50, n_points // 3):
            worst_index = np.argmax(scores)
            population.pop(worst_index)
            scores.pop(worst_index)

        # Update best
        current_best_index = np.argmin(scores)
        if scores[current_best_index] < best_c5:
            best_h = population[current_best_index].copy()
            best_c5 = scores[current_best_index]
        
        # Occasionally refine with L-BFGS-B
        if gen % local_refine_intervals == 0 and time.time() - time_start + 30 < budget_s:
            lbfgs_bounds = [(0, 1)] * n_points
            refined_model = opt.minimize(lambda x: compute_c5(x.reshape(-1), n_points), 
                                        x0=best_h, bounds=lbfgs_bounds,
                                        method="L-BFGS-B", options={"maxfun": 1000})
            if refined_model.success:
                refined_score = compute_c5(refined_model.x.reshape(-1), n_points)
                if refined_score < best_c5:
                    best_h[:] = refined_model.x.reshape(-1)

        gen += 1
        if time.time() - time_start > eval_budget:
            break  # Final refinement phase

    # Extract best solution
    return best_h, best_c5, n_points
