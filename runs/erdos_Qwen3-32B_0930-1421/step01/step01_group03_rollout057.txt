import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 47
    dx = 2.0 / max_n_points
    constraint_sum = max_n_points / 2

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[2 * i] = 2.0 / n_points
        h_init = h_init.clip(0, 1)
        h_init = h_init * (constraint_sum / h_init.sum())

    def objective_C(h):
        h = h.reshape(n_points)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps), h

    def lp_inner(h):
        h = h.reshape(n_points)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)

        # variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()

        # constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == constraint_sum]
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                mask = np.roll(h, -k)
            else:
                mask = np.roll(h, -k)
            overlap_expr = cp.multiply(h_var, np.roll(1 - h, -k))
            overlap_sum = cp.sum(overlap_expr) * dx
            constraints.append(overlap_sum <= C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)

        if prob.status != 'optimal' and prob.status != 'hiatus':
            return h, objective_C(h)[0]
        return h_var.value, prob.value

    best_h = h_init.copy()
    best_c5 = objective_C(best_h)[0]
    cur_h = h_init.copy()
    iters = 0
    perturb_freq = 5
    max_perms = 30
    while time.time() - start_time < budget_s and iters < max_perms:
        iters += 1
        cur_h, c5 = lp_inner(cur_h)
        cur_h = cur_h.clip(0, 1)
        cur_h = cur_h * (constraint_sum / cur_h.sum())
        c5_new = objective_C(cur_h)[0]
        if c5_new < best_c5:
            best_h = cur_h.copy()
            best_c5 = c5_new
        if iters % perturb_freq == 0:
            noise = np.random.randn(n_points) * 0.05
            cur_h = (cur_h + noise).clip(0, 1)
            cur_h = cur_h * (constraint_sum / cur_h.sum())

    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    return best_h, best_c5, n_points
