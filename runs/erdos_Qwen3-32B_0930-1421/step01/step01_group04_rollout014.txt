import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

class ColumnGenerationSolver:
    def __init__(self, initial_h, n_points):
        self.n_points = n_points
        self.h = initial_h.copy()
        self.C = cp.Variable()
        self.h_var = cp.Variable(n_points)
        self.locally_better_h = initial_h.copy()
        self.best_C = compute_c5(self.locally_better_h, n_points)
        self.constraints = []

    def setup_initial_constraints(self):
        self.constraints.extend([
            self.h_var >= 0,
            self.h_var <= 1,
            cp.sum(self.h_var) == self.n_points / 2
        ])
        # Start with a few key lags
        for k in [1, 2, 4, 6, 10]:
            if k >= self.n_points:
                continue
            sm = 0.0
            for i in range(self.n_points - k):
                sm += self.h_var[i] * (1 - self.h_var[i + k])
            self.constraints.append(sm * (2.0 / self.n_points) <= self.C)

    def solve_lp(self):
        obj = cp.Minimize(self.C)
        prob = cp.Problem(obj, self.constraints)
        prob.solve(solver='CLARABEL', verbose=False)
        if prob.status != 'optimal':
            return False
        self.h[self.h_var.value >= 0] = np.clip(self.h_var.value, 0.0, 1.0)
        return True

    def find_violated_constraint(self):
        for k in range(1, self.n_points):
            if self.n_points - k <= 0:
                continue
            corr = np.sum(self.h[:n_points - k] * (1 - self.h[k:]))
            if corr > self.best_C * self.n_points * 0.5 + 1e-5:
                return k
        return None

    def add_violated_constraint(self, k):
        sm = 0.0
        for i in range(self.n_points - k):
            sm += self.h_var[i] * (1 - self.h_var[i + k])
        self.constraints.append(sm * (2.0 / self.n_points) <= self.C)

    def finalize(self):
        res = opt.minimize(
            lambda x: compute_c5(x, self.n_points),
            self.h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * self.n_points,
            options={'maxiter': 200, 'eps': 1e-4}
        )
        h_refined = enforce_normalization(res.x, self.n_points)
        c5_val = compute_c5(h_refined, self.n_points)
        if c5_val < self.best_C:
            self.locally_better_h = h_refined
            self.best_C = c5_val
        return self.locally_better_h, self.best_C

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if initial_h_values is None:

        raise ValueError("initial_h_values is required")
    n_points = len(initial_h_values)
    h_best = initial_h_values.copy()
    C_best = compute_c5(h_best, n_points)

    solver = ColumnGenerationSolver(h_best, n_points)
    solver.setup_initial_constraints()
    time_start = time.time()
    budget = budget_s

    solver.solve_lp()
    elapsed = time.time() - time_start
    remaining = budget - elapsed

    while remaining > 0:
        k = solver.find_violated_constraint()
        if k is not None:
            solver.add_violated_constraint(k)
            if not solver.solve_lp():
                break
        else:
            break
        elapsed = time.time() - time_start
        remaining = budget - elapsed

    h_best, C_best = solver.finalize()
    h_best = enforce_normalization(h_best, n_points)
    return h_best, C_best, n_points
