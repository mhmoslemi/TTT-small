import numpy as np
import math
from scipy.signal import correlate
import time

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_targeting_peak(h, peak_idx):
    n = len(h)
    left = peak_idx % n
    right = (peak_idx - n) % n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for i in np.abs(ones - left).argsort():
        if h[ones[i]] == 1:
            for j in np.abs(zeros - right).argsort():
                if h[zeros[j]] == 0:
                    h_new = h.copy()
                    h_new[ones[i]] = 0
                    h_new[zeros[j]] = 1
                    return h_new
    return h
