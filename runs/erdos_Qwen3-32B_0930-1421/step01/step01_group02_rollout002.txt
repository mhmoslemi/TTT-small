```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def create_convolution_constraints(h_vec, shift_range):
    n = len(h_vec)
    dx = 2.0 / n
    h = cp.Parameter(n, nonneg=True)
    constraints = [
        0 <= h[i] for i in range(n)
    ] + [
        h[i] <= 1 for i in range(n)
    ] + [
        cp.sum(h) == n / 2
    ]

    conv_obj = []
    for k in shift_range:
        # Compute circular convolution values at shift k
        idx1 = np.arange(n)
        idx2 = np.mod(idx1 - k, n)
        obj_val = cp.sum([h[i] * (1 - h[j]) for i, j in zip(idx1, idx2)])
        conv_obj.append(obj_val * dx)

    return h, constraints, conv_obj


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    # Use the initial_h_values if available
    if initial_h_values is not None:
        h0 = np.copy(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
        h0 = ensure_feasible(h0)

    h_vec = cp.Variable(n_points)
    constraints = [
        0 <= h_vec[i] for i in range(n_points)
    ] + [
        h_vec[i] <= 1 for i in range(n_points)
    ] + [
        cp.sum(h_vec) == n_points / 2
    ]

    dx = 2.0 / n_points
    conv_values = correlate(h0, 1 - h0, mode="full")
    overlap_positions = np.where((np.arange(len(conv_values)) >= n_points)
                                 & (np.arange(len(conv_values)) < 2 * n_points))
    max_corr = np.max(conv_values[overlap_positions]) * dx

    # Optimization problem setup
    max_overlap = cp.Variable()
    obj = cp.Minimize(max_overlap)
    conv_vals = []

    for k in range(-n_points + 1, n_points):
        if k == 0:
            continue
        idx1 = np.arange(n_points)
        idx2 = np.mod(-k + idx1, n_points)
        val = cp.sum([h_vec[i] * (1 - h_vec[j]) for i, j in zip(idx1, idx2)])
        val = val * dx
        conv_vals.append(val)

    constraints += [cp.Max(conv_vals) <= max_overlap]
    problem = cp.Problem(obj, constraints)

    best_h = None
    best_C5 = np.inf

    # Try to use warm start and different initializations
    for attempt in range(2):
        if initial_h_values is None:
            h0 = np.random.rand(n_points)
            h0 = ensure_feasible(h0)
        else:
            h0 = np.copy(initial_h_values)

        h_vec.value = h0

        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            pass

        if problem.status not in ['infeasible', 'unbounded', 'optimal']:
            candidate_h = h0
        else:
            candidate_h = h_vec.value
            if candidate_h is not None:
                candidate_h = np.clip(candidate_h, 0, 1)
                candidate_h /= np.sum(candidate_h)
                candidate_h *= n_points / 2
        candidate_h = np.clip(candidate_h, 0, 1)

        while np.sum(candidate_h) < n_points / 2 - 1e-4:
            idx = np.argmin(candidate_h)
            candidate_h[idx] = min(1, candidate_h[idx] + 1e-2)

        while np.sum(candidate_h) > n_points / 2 + 1e-4:
            idx = np.argmax(candidate_h)
            candidate_h[idx] = max(0, candidate_h[idx] - 1e-2)

        current_C5 = evaluate_C5(candidate_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = candidate_h

        if time.time() - start_time >= budget_s:
            break

    return best_h, best_C5, n_points
```
