import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if n_points - k <= 0:
            continue
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        if n_points - k <= 0:
            continue
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    h = initial_h_values.copy()
    h = enforce_normalization(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    # Column generation loop
    time_start = time.time()
    while time.time() - time_start < budget_s * 0.9:
        h_cp = cp.Variable(n_points)
        C = cp.Variable()

        constraints = [
            h_cp >= 0,
            h_cp <= 1,
            cp.sum(h_cp) == n_points / 2,
        ]

        # Initially use 10 explicit k values
        initial_k = [1, 2, 4, 8, 16, 23, 31, 37, 42, 45]
        for k in initial_k:
            if k >= n_points:
                continue
            overlap = cp.sum([h_cp[i] * (1 - h_cp[i + k]) for i in range(n_points - k)])
            constraints.append(overlap * (2.0 / n_points) <= C)

        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GLPK_MI, abstol=1e-8, reltol=1e-6, maxit=1000, verbose=False)

        if prob.status != cp.OPTIMAL:
            break

        h_val = np.clip(h_cp.value.flatten(), 0.0, 1.0)
        h_val = enforce_normalization(h_val, n_points)
        val = compute_c5(h_val, n_points)

        if val < best_c5:
            best_h = h_val.copy()
            best_c5 = val

        max_overlap = 0.0
        worst_k = 0
        for k in range(1, n_points):
            if n_points - k <= 0:
                continue
            corr = np.dot(h_val[:n_points - k], 1 - h_val[k:k + n_points - k])
            overlap = corr * (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k

        # Add only if the worst constraint is violated
        if max_overlap > best_c5 + 1e-7:
            overlap_expr = sum([h_cp[i] * (1 - h_cp[i + worst_k]) for i in range(n_points - worst_k)])
            constraints.append(overlap_expr * (2.0 / n_points) <= C)

    # Local refinement after column generation
    def refine_objective(x):
        return compute_c5(x, n_points)

    result = opt.minimize(
        refine_objective,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-4}
    )
    h_refine = result.x
    h_refine = enforce_normalization(h_refine, n_points)
    c5_refine = compute_c5(h_refine, n_points)

    final_h, final_c5 = (h_refine, c5_refine) if c5_refine < best_c5 else (best_h, best_c5)
    return final_h, final_c5, n_points
