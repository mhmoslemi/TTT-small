import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    if h.sum() == 0:
        h = np.zeros_like(h)
        h[0] = len(h) / 2
    else:
        h = h / h.sum() * len(h) / 2
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h

def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if defined

            h[5:42] = 0.3
            h[42:] = 0.2
            h = h / h.sum() * len(h) / 2
    except:
        h = np.zeros(n_points)
        h[:5] = 0.1
        h[5:42] = 0.3
        h[42:] = 0.2
        h = h / h.sum() * len(h) / 2

    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)[0]
    start_time = time.time()
    max_k = None
    iter = 0

    while time.time() - start_time < budget_s:
        # Evaluate current solution
        h = np.clip(h, 0, 1)
        h = h / h.sum() * len(h) / 2
        C5, h_current = evaluate_C5(h, dx)
        if C5 < best_C5:
            best_C5, best_h = C5, h.copy()
            max_k = None

        # Find the problematic k
        cor = correlate(h, 1 - h, mode="full")
        max_id = np.argmax(cor)
        if max_k is None or cor[max_id] > cor[np.argmax(cor[(max_id-5):(max_id+5)])]:
            max_k = max_id - (len(h) - 1)

        if max_k is not None:
            # Build support vectors for shift k
            L = len(h)
            idxs = (np.arange(L) + max_k) % L
            h_k = h[idxs]
            h_k_mask = np.arange(L)[(max_k + np.arange(L)) % L < L]

            # Build quadratic program
            h_var = cp.Variable(L, nonneg=True)
            objective = cp.Maximize(h @ (1 - h_k))
            constraints = [
                h_var >= 0,
                h_var <= 1,
                h_var.sum() == h.sum()
            ]

            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver=cp.ECOS, max_iters=500, verbose=False)

            if result is not None:
                h = h_var.value
                h = np.clip(h, 0, 1)
                h = h / h.sum() * len(h) / 2

        elif max_k is None:
            pert = np.random.rand(n_points) * 0.3 - 0.15
            h = np.clip(h + pert, 0, 1)
            h = h / h.sum() * len(h) / 2

        if iter % 50 == 0:
            # Restart with random perturbation
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * len(h) / 2

        iter += 1

        if iter % 100 == 0:
            C5_check, h_check = evaluate_C5(h, dx)
            if C5_check < best_C5:
                best_C5, best_h = C5_check, h_check.copy()

    return best_h, best_C5, n_points
