import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize to ensure the integral constraint
    cor = np.correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx, h


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem using sequentially focused QP."""
    np.random.seed(seed)

        if len(h) != n_points:

    except:

        h[:] = 0.0
        h[0] = n_points / 2
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

    best_h, best_C5 = h.copy(), np.inf
    start_time = time.time()
    iter_count = 0
    perturb_count = 0
    noise_frac = 0.1

    while time.time() - start_time < budget_s:
        iter_count += 1
        h = np.clip(h.copy(), 0, 1)
        h_sum = h.sum()
        h = h / h_sum * n_points / 2
        C5_curr, h_curr = evaluate_C5(h, dx)

        if C5_curr < best_C5:
            best_C5 = C5_curr
            best_h = h_curr.copy()

        cor = correlate(h_curr, 1 - h_curr, mode="full")
        max_id = np.argmax(cor)
        worst_k = max_id - len(h_curr) + 1

        # Construct h_left and h_right for the shift k
        h_left = h_curr.copy()
        h_right = h_curr.copy()
        h_right = np.roll(h_right, -worst_k)

        h_left = cp.Parameter(n_points, nonneg=True, name="h_left")
        h_right = cp.Parameter(n_points, nonneg=True, name="h_right")
        h_var = cp.Variable(n_points, nonneg=True)

        delta = cp.multiply(h_var, 1 - h_right)

        # QP Formulation
        objective = cp.Minimize(h_var.T @ delta)
        constraints = [
            cp.sum(h_var) * dx == 1.0,
            h_var <= 1,
            h_var >= 0
        ]

        prob = cp.Problem(objective, constraints)

        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
            if result is not None:
                h = h_var.value
                h = np.clip(h, 0, 1)
                h_sum = h.sum()
                h = h / h_sum * n_points / 2
        except:
            perturb_count += 1
            if perturb_count % 5 == 0:
                h = np.clip(h + (np.random.rand(n_points) - 0.5) * noise_frac, 0, 1)
                h = h / h.sum() * n_points / 2
            continue

        if iter_count % 50 == 0:
            h = np.clip(h, 0, 1)
            perturb_guess = h + (np.random.rand(n_points) - 0.5) * 0.2
            perturb_guess = np.clip(perturb_guess, 0, 1)
            perturb_guess = perturb_guess / perturb_guess.sum() * n_points / 2
            C5_perturb, _ = evaluate_C5(perturb_guess, dx)
            if C5_perturb < best_C5:
                h = perturb_guess
                best_C5 = C5_perturb
                best_h = h.copy()

    return best_h, best_C5, n_points
