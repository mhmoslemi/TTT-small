import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Enforce constraints
    h = np.clip(h, 0.0, 1.0)
    h *= target_sum / h.sum()
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = np.inf
    h_history = [h.copy()]

    # Utility function to evaluate C5 efficiently
    def compute_c5(h_eval):
        h_eval = np.clip(h_eval, 0.0, 1.0)
        h_sum = h_eval.sum()
        if abs(h_sum - target_sum) > 1e-6:
            h_eval *= target_sum / h_sum

        ones_minus_h = 1.0 - h_eval
        max_corr = -np.inf
        full_corr = scipy.signal.correlate(h_eval, ones_minus_h, mode="full")
        corr_k = full_corr[n_points - 1 : n_points + n_points - 1]
        max_corr = np.max(corr_k[n_points - 1 : 2 * n_points - 1]) * dx
        return max_corr

    best_c5 = compute_c5(h)

    # Gradient update for max shift with projection
    def grad_update(h_update, best_shift):
        grad = np.zeros_like(h_update)
        for i in range(n_points - best_shift):
            grad[i] += -(1 - h_update[i + best_shift]) * dx
            grad[i + best_shift] += h_update[i] * dx
        h_update -= 0.01 * grad
        h_update = np.clip(h_update, 0.0, 1.0)
        h_sum = h_update.sum()
        if abs(h_sum - target_sum) > 1e-6:
            h_update *= target_sum / h_sum
        return h_update

    def add_random_perturbation(h_perturbed):
        idx = np.random.choice(n_points, size=5)
        h_perturbed[idx] += np.random.uniform(-0.02, 0.02, size=5)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_sum = h_perturbed.sum()
        if abs(h_sum - target_sum) > 1e-6:
            h_perturbed *= target_sum / h_sum
        return h_perturbed

    # Optimization loop
    last_best_time = start_time
    while time.time() - start_time < budget_s:
        c5 = compute_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
            last_best_time = time.time()
        if time.time() - last_best_time > 60:
            break

        # Find max shift and gradient
        ones_minus_h = 1.0 - h
        full_corr = scipy.signal.correlate(h, ones_minus_h, mode="full")
        corr_k = full_corr[n_points - 1 : 2 * n_points - 1]
        best_shift = np.argmax(corr_k) + 1

        # Gradient step on worst shift
        h = grad_update(h, best_shift)

        # Perturbation every so often to avoid local trapping
        if np.random.rand() < 0.01:
            h = add_random_perturbation(h)

    # One-time refinement
    h = best_h.copy()
    for i in range(5):
        ones_minus_h = 1.0 - h
        full_corr = scipy.signal.correlate(h, ones_minus_h, mode="full")
        corr_k = full_corr[n_points - 1 : 2 * n_points - 1]
        worst_shift = np.argmax(corr_k) + 1
        h = grad_update(h, worst_shift)

    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h[:] = h.copy()

    return (best_h, best_c5, n_points)
