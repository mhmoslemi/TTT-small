import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:n_points - k + k])
        corrs.append(corr)
    for k in range(1, n_points):
        if k >= n_points:
            break
        corr = np.dot(h[k:n_points - k + k], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h):
    h = np.clip(h, 0, 1.0)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / total
    return h

def mutate(parent, sigma=0.05):
    child = parent + np.random.normal(0, sigma, len(parent))
    return enforce_normalization(child)

def crossover(parent1, parent2):
    alpha = np.random.uniform(0, 1)
    child = alpha * parent1 + (1 - alpha) * parent2
    return enforce_normalization(child)

def evaluate_population(population, n_points):
    return [compute_c5(p, n_points) for p in population]

def run(seed=42, budget_s=1100):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    pop_size = 40
    mutation_rate = 0.2
    sigma_max = 0.1
    init_pop = [initial_h_values + np.random.uniform(-0.1, 0.1, n_points) for _ in range(9)]
    pop = [enforce_normalization(h) for h in init_pop]
    pop.extend([initial_h_values] if initial_h_values is not None else [])
    while len(pop) < pop_size:
        pop.append(enforce_normalization(np.random.rand(n_points)))

    time_start = time.time()
    while time.time() - time_start < budget_s * 0.8:
        fitness = evaluate_population(pop, n_points)
        pop = [h for _, h in sorted(zip(fitness, pop))]
        parents = pop[:int(pop_size * 0.5)]
        new_pop = parents.copy()
        for _ in range(int(pop_size * 0.8)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < mutation_rate:
                child = mutate(child, sigma_max * (1 - (time.time() - time_start) / budget_s))
            new_pop.append(child)
        pop = new_pop[:pop_size]

    final_h = pop[0]
    final_c5 = compute_c5(final_h, n_points)
    # Final refinement with L-BFGS-B
    def refined_obj(x):
        x = x.reshape(-1)
        return compute_c5(enforce_normalization(x), n_points)
    result = opt.minimize(lambda x: refined_obj(x), final_h, method='L-BFGS-B',
                          bounds=[(0,1)] * n_points,
                          options={'maxiter': 200})
    final_h = enforce_normalization(result.x)
    final_c5 = compute_c5(final_h, n_points)
    return final_h, final_c5, n_points
