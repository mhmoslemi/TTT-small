import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h_neg = 1.0 - h
    corr = correlate(h, h_neg, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    # Clip values to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    # Rescale to satisfy the average
    target_sum = n_points / 2
    current_sum = np.sum(h)
    while not np.allclose(current_sum, target_sum, atol=1e-6):
        correction = (target_sum - current_sum) / (current_sum * n_points)
        h = np.clip(h + correction * h, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
    return h

        for i in range(n_points):

                grad[i] += (1 - h[i + lag])
            if i - lag >= 0:
                grad[i] -= h[i - lag]

        grad = grad / (np.linalg.norm(grad) + 1e-10)

        # Line search
        learning_rate = min(min_learning_rate * (1.1 ** iter), 1.0)
        alpha = learning_rate
        for _ in range(10):

            # Perturb to escape local minima
            perturbation = 0.01 * np.random.randn(n_points)
