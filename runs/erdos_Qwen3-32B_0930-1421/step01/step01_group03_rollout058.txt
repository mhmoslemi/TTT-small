import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    max_n_points = 100  # Limit size to small sequences
    n_points = 47 if not hasattr(initial_h_values, '__len__') or len(initial_h_values) <= max_n_points else min(len(initial_h_values), max_n_points)
    dx = 2.0 / n_points

    # Load or initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
        h_init = h_init[:n_points]
    else:
        h_init = np.zeros(n_points)
        h_init[::2] = 2.0 / n_points
        h_init = h_init * (n_points / 2 / h_init.sum())  # Normalize

    # Constraint: sum(h) == n_points / 2
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_with_lp(h_guess, dx, constraint_sum):
        h = cp.Variable(n_points)
        C = cp.Variable(nonneg=True)

        constraints = [0 <= h, h <= 1, cp.sum(h) == constraint_sum]

        overlap_shifts = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = overlap_shifts.max()
        constraints += [overlap_shifts[k] <= C for k in range(len(overlap_shifts))]

        objective = C
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver='CVXOPT', maxiters=200, verbose=False)
        except:
            return h_guess
        h_opt = np.clip(h.value, 0, 1)
        h_opt = h_opt * (constraint_sum / h_opt.sum())
        return h_opt

    def perturb_structure(h, noise_factor=0.1):
        h = h.copy()
        noise = np.zeros_like(h)
        freqs = np.linspace(0.1, 2.0, len(h))
        for i in range(3):
            noise += np.sin(2 * np.pi * freqs * i) * np.random.randn()
        h += noise * noise_factor
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # Perturbations and LP refinements
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        h_current = perturb_structure(h_init if iteration == 1 else best_h, noise_factor=0.1)
        h_lp = optimize_with_lp(h_current, dx, constraint_sum)
        new_c5 = objective(h_lp)
        if new_c5 < best_c5:
            best_h = h_lp
            best_c5 = new_c5
        if iteration % 10 == 0:
            perturb_scramble = np.random.rand() * 0.2
            h_current = np.random.rand(n_points)
            h_current = h_current.clip(0, 1)
            h_current = h_current * (constraint_sum / h_current.sum())
            h_lp = optimize_with_lp(h_current, dx, constraint_sum)
            new_c5 = objective(h_lp)
            if new_c5 < best_c5:
                best_h = h_lp
                best_c5 = new_c5

    # Final constraint enforces valid solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
