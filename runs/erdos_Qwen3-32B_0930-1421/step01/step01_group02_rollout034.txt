```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def run(budget_s=1100, n_points=42):
    start_time = time.time()
    h = cp.Variable(n_points, nonneg=True, name="h")
    h_values = h.copy()
    constraint_list = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]

    max_C5 = np.inf
    best_h = None

    # Setup grid direction shifts
    dx = 2.0 / n_points
    indices = np.arange(-n_points + 1, n_points)
    grid_shift_ops = []

    for shift in indices:
        left = max(0, -shift)
        right = min(n_points, n_points - shift)
        vals = np.roll(np.eye(n_points), shift=shift, axis=1)[:, left:right]

        # Operators to compute h(x)*(1-h(x+k)) for shift k
        op_h = h[left:right]
        op_1h = 1 - h[(n_points + shift) % n_points : (n_points + shift) % n_points + n_points]
        op = cp.sum(cp.multiply(op_h, op_1h))

        # Add to constraints and track maximum
        max_C5 = cp.max([max_C5] + [op * dx])

    # Construct problem
    objective = cp.Minimize(max_C5)
    problem = cp.Problem(objective, constraint_list)

    if initial_h_values is not None:
        constraint_list.append(h == initial_h_values)

    problem.solve(solver=cp.ECOS, verbose=False)

    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        best_h = h.value
        best_C5 = max_C5.value
    else:
        best_h = ensure_feasible(np.random.rand(n_points))
        best_C5 = evaluate_C5(best_h)

    return best_h, best_C5, n_points
```
