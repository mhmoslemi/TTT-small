import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∫h dx = 1 => sum(h) = n_points / 2

    # Ensure initial h satisfies constraints
    h = h_init.copy()
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    h = h / h_sum * target_sum
    h = np.clip(h, 0.0, 1.0)
    h_sum = h.sum()
    while h_sum < target_sum:
        idx = np.argmin(h)
        h[idx] += min(1.0 - h[idx], target_sum - h_sum)
        h_sum = h.sum()

    h_best = h.copy()
    c5_best = np.inf

    # Helper to compute overlap max for optimization feedback
    def compute_overlap(h_vec):
        h_vec = np.clip(h_vec.copy(), 0.0, 1.0)
        h_sum = h_vec.sum()
        if abs(h_sum - target_sum) > 1e-8:
            h_vec *= target_sum / h_sum
        ones_minus_h = 1.0 - h_vec
        max_corr = -1e18
        for k in range(1, n_points - 1):
            if k >= h_vec.shape[0] - 1:
                continue
            corr1 = np.dot(h_vec[:n_points - k], ones_minus_h[k:])
            corr2 = np.dot(h_vec[k:], ones_minus_h[:n_points - k])
            max_corr = max(max_corr, corr1, corr2)
        return max_corr * dx

    # Evaluate initial point
    c5_current = compute_overlap(h)
    c5_best = min(c5_best, c5_current)
    h_best[:] = h.copy()
    best_time = time.time()
    epsilons = [0.01, 0.005, 0.001]
    eps_index = 0

    # Main optimization
    while time.time() - start_time < budget_s - 5:
        c5_current = compute_overlap(h)

        if c5_current < c5_best:
            c5_best = c5_current
            h_best[:] = h.copy()
            best_time = time.time()

        # Identify the most problematic shifts
        k_vals = []
        overlap_vals = []
        for k in range(1, n_points - 1):
            k_ = k
            if k_ >= h.shape[0] - 1:
                continue
            corr1 = np.dot(h[:n_points - k_], (1.0 - h)[k_:])
            corr2 = np.dot(h[k_:], (1.0 - h)[:n_points - k_])
            overlap_vals.append(max(corr1, corr2))
            k_vals.append(k_)

        top_k = np.array(k_vals)[np.argsort(overlap_vals)[-5:]]
        top_k = np.unique(top_k)

        # For each problematic shift, adjust subset of samples
        for k in top_k:
            contrib = []
            for i in range(n_points - k):
                contrib.append(h[i] * (1.0 - h[i + k]))
            contrib = np.array(contrib)
            max_indices = np.argsort(-contrib)[:max(10, int(len(contrib) * 0.1))]
            for i in max_indices:
                dir = np.random.choice([-1, 1])
                prop = h.copy()
                prop[i] += dir * epsilons[eps_index]
                prop = np.clip(prop, 0.0, 1.0)
                prop_sum = prop.sum()
                if abs(prop_sum - target_sum) > 1e-10:
                    prop = prop / prop_sum * target_sum
                    prop = np.clip(prop, 0.0, 1.0)
                c5_proposal = compute_overlap(prop)
                if c5_proposal < c5_current:
                    h[:] = prop[:]
                    c5_current = c5_proposal
                    break  # break early per shift improvement

        # Adaptive change to epsilon based on improvement
        if (time.time() - best_time) > 30 and eps_index < len(epsilons) - 1:
            eps_index += 1  # fine-tune policy
            best_time = time.time()

    c5_final = compute_overlap(h_best)
    return (h_best, c5_final, n_points)
