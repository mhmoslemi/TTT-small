import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    n_points = 47
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        # Construct basic uniform distribution
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * target_sum / h_sum
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def optimize_with_random_perturbations(x0, max_restarts=100):
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            h = h * target_sum / h_sum
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        bounds = [(0, 1) for _ in range(n_points)]
        x = x0.copy()
        x = x.clip(0, 1)
        x = x * target_sum / x.sum()
        result = minimize(
            objective,
            x,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 200}
        )
        return result.x

    def perturb_support(h, noise_factor=0.3):
        h = h.copy()
        perturbation = np.random.normal(0, 1, n_points) * noise_factor
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / perturbed_h.sum()
        return perturbed_h

    # Main perturbation + opt loop
    h = best_h
    for restart in range(60):
        if time.time() - start_time > budget_s * 0.95:
            break
        h = perturb_support(h, noise_factor=0.1 * (0.95**restart))
        h = optimize_with_random_perturbations(h)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5

    # Feasibility repair
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / best_h.sum()
    final_c5 = evaluate_c5(best_h)

    return best_h, final_c5, n_points
