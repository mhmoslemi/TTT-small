import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize to satisfy integral constraint
    cor = correlate(h, 1 - h, mode="full")
    return np.max(cor) * dx, h


def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)
    dx = 2.0 / n_points

    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            h.resize(n_points)
    except:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * (n_points / 2)

    best_h = h.copy()
    best_C5, _ = evaluate_C5(best_h, dx)

    eval_time = time.time()
    iter_count = 0
    max_iter = 1000

    while time.time() - eval_time < budget_s and iter_count < max_iter:
        iter_count += 1
        h_var = cp.Variable(n_points)
        objective = (
            cp.sum_squares(h_var)
            + cp.Parameter(nonneg=True) * cp.sum(
                [cp.sum(
                    (
                        h_var[x] + h_var[(x + k) % n_points]
                        - 1
                    ) ** 2
                    for x in range(n_points)
                ) for k in range(1, n_points)]
            )
        )
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, max_iter=500, verbose=False)

        if h_var.value is not None:
            h = np.clip(h_var.value, 0, 1)
            h = h / h.sum() * (n_points / 2)
            C5, _ = evaluate_C5(h, dx)
            if C5 < best_C5:
                best_C5, best_h = C5, h

        if iter_count % 10 == 0:
            h = np.clip(h + 0.01 * (2 * np.random.rand(n_points) - 1) + h * 0.1, 0, 1)
            h = h / h.sum() * (n_points / 2)
            C5, _ = evaluate_C5(h, dx)
            if C5 < best_C5:
                best_C5, best_h = C5, h

    return best_h, best_C5, n_points
