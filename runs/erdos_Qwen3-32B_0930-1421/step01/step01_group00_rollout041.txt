import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = h_init.shape[0]
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Ensure initial h satisfies constraints
    h = h_init.copy()
    h_sum = np.clip(h, 0, 1).sum()
    if abs(h_sum - target_sum) > 1e-10:
        h = np.clip(h, 0, 1)
        h *= target_sum / h_sum
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(optimized_h):
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h_sum = optimized_h.sum()
        if abs(optimized_h_sum - target_sum) > 1e-10:
            optimized_h *= target_sum / optimized_h_sum
        optimized_h = np.clip(optimized_h, 0, 1)

        corrs = scipy.signal.correlate(optimized_h, 1.0 - optimized_h, mode='full')
        n_upper = corrs.size // 2 - 1
        n_lower = -corrs.size // 2
        upper_corrs = corrs[n_upper:2 * n_points - 1 + n_upper]
        lower_corrs = corrs[n_lower:2 * n_lower + n_points]
        total_corrs = np.concatenate((upper_corrs, lower_corrs))
        return (np.max(total_corrs) * dx)

    def local_search(h):
        nonlocal best_c5, best_h
        h = h.copy()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
        for _ in range(100):
            indices = np.random.choice(n_points, min(5, n_points))
            noise = np.random.normal(0, 0.01, size=len(indices))
            trial = h.copy()
            trial[indices] += noise
            trial = np.clip(trial, 0, 1)
            trial_sum = trial.sum()
            if abs(trial_sum - target_sum) > 1e-7:
                trial *= (target_sum / trial_sum)
            trial = np.clip(trial, 0, 1)
            trial_c5 = compute_c5(trial)
            if trial_c5 < current_c5:
                h[:] = trial
                current_c5 = trial_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h[:] = h[:]
            if time.time() - start_time > budget_s - 1.0:
                break

    try:
        while time.time() - start_time < budget_s - 2.0:
            local_search(best_h)
            if verbose:
                elapsed = time.time() - start_time
                print(f"{elapsed:.1f}s, best C₅: {best_c5:.6f}")
    except:
        pass

    return best_h, best_c5, n_points
