import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100  # Cap to ensure small sequences
    n_points = 47 if 'initial_h_values' not in globals() else len(initial_h_values)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, copy=True)
    else:
        # Construct initial h as approx evenly spaced intervals with unit value
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + (i // 1)] = 2.0 / n_points
        h_init = h_init.clip(0, 1)
        h_init = h_init * (n_points / 2 / h_init.sum())  # Normalize to constraint

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(h0, max_iter=200):
        bounds = [(0, 1)] * n_points
        best_h = h0.copy()
        best_obj = objective(best_h)

        def minimize_func(x):
            x = x.reshape(n_points)
            x = x.clip(0, 1)
            x_sum = x.sum()
            x = x * (constraint_sum / x_sum)
            overlaps = np.correlate(x, 1 - x, mode='full') * dx
            return np.max(overlaps)

        result = minimize(
            minimize_func,
            best_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-6, 'gtol': 1e-5},
        )
        best_h = result.x.reshape(n_points)
        best_h = best_h.clip(0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        return best_h

    def perturb(h, noise_factor=0.2, explore=True):
        h = h.copy()
        perturbation = np.random.normal(0, noise_factor, n_points)
        if explore:
            idx = np.random.choice(n_points, size=int(n_points * 0.5), replace=False)
            perturbation = np.zeros(n_points)
            perturbation[idx] = np.random.uniform(-5e-2, 5e-2, size=len(idx))
        h += perturbation
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h_vals = h_init
    best_c5 = objective(best_h_vals)

    current_h = best_h_vals
    tabu_list = []
    max_tabu_size = 10
    perturb_scale = 0.1
    cooling_rate = 0.8
    noimpr_count = 0
    patience_limit = 10

    while time.time() - start_time < budget_s:
        current_h = perturb(current_h, noise_factor=perturb_scale)
        current_h = optimize_h(current_h)
        current_c5 = objective(current_h)

        if current_c5 < best_c5 and not any(np.allclose(current_h, t, atol=1e-5) for t in tabu_list):
            best_h_vals = current_h.copy()
            best_c5 = current_c5
            noimpr_count = 0
        else:
            noimpr_count += 1

        if noimpr_count >= patience_limit:
            perturb_scale *= cooling_rate
            noimpr_count = 0

        current_h = optimize_h(current_h)
        tabu_list.append(current_h.copy())
        if len(tabu_list) > max_tabu_size:
            tabu_list.pop(0)

        if time.time() - start_time > budget_s * 0.9:
            break

    best_h_vals = best_h_vals.clip(0, 1)
    best_h_vals = best_h_vals * (constraint_sum / best_h_vals.sum())
    final_c5 = objective(best_h_vals)

    return best_h_vals, final_c5, n_points
