import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(budget_s=1100, **kwargs):
    start_time = time.time()

    # Load initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def clip_and_normalize(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    def compute_c5(h):
        hh = 1 - h
        corr = correlate(h, hh, mode='full')
        c5 = np.max(corr * dx)
        return c5

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def solve_cvxopt(h0, lam, max_iter=5):
        h = cp.Variable(n_points)
        overlap = 0
        for lag in range(-n_points + 1, n_points):
            idx1 = np.arange(n_points - np.abs(lag))
            if lag >= 0:
                idx0 = idx1 + lag
            else:
                idx0 = idx1 - lag
            overlap += cp.sum(h[idx0] * (1 - h[idx1]))
        objective = cp.Minimize(cp.norm(h, 1) * lam + cp.Max(overlap * dx))
        constraints = [
            cp.sum(h) == n_points / 2,
            h <= 1,
            h >= 0
        ]
        prob = cp.Problem(objective, constraints)
        val = np.inf
        for i in range(max_iter):
            if time.time() - start_time >= budget_s - 10:
                break
            try:
                prob.solve(solver=cp.CVXOPT)
                if prob.status == 'optimal':
                    h_val = h.value
                    h_sol = clip_and_normalize(h_val)
                    c5 = compute_c5(h_sol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
                    val = best_c5
            except Exception as e:
                pass
        return val, best_h.copy(), best_c5

    # Grid search over lambda
    lam_list = [0.01, 0.03, 0.1, 0.3, 1.0]
    for lam in lam_list:
        t_start = time.time()
        val, h_sol, c5 = solve_cvxopt(best_h, lam, max_iter=3)
        print(f"lam={lam}: C5={c5:.8f}")

    # Gradient descent refinement
    learning_rate = 0.01
    h = clip_and_normalize(best_h)
    n = len(h)
    grad_smooth = np.zeros_like(h)
    while time.time() - start_time < budget_s:
        eps = 1e-4
        for i in range(n):
            h_p = h.copy()
            h_p[i] += eps
            h_p = clip_and_normalize(h_p)
            c5_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = clip_and_normalize(h_n)
            c5_n = compute_c5(h_n)

            grad_smooth[i] = (c5_p - c5_n) / (2 * eps)
        h -= learning_rate * grad_smooth
        h = clip_and_normalize(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        learning_rate *= 0.9999
        if time.time() - start_time >= budget_s - 1:
            break

    return best_h.tolist(), best_c5, n_points
