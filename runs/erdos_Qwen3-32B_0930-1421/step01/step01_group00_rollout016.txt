import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    start_time = time.time()
    budget_s = min(budget_s, 1100)
    np.random.seed(seed)

    # Step 1: Setup
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h * (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    diff = target_sum - h_sum
    if diff != 0:
        idx = np.where(h > 1e-6)[0]
        np.random.shuffle(idx)
        h[idx[:int(np.ceil(abs(diff)))]] += np.sign(diff) * 1e-6 * abs(diff)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = 1.0
    t_best = time.time()
    early_stop_timer = time.time()

    def evaluate(h_eval):
        h_eval = np.clip(h_eval, 0, 1).copy()
        h_sum = h_eval.sum()
        h_eval = h_eval * (target_sum / h_sum)
        h_eval = np.clip(h_eval, 0, 1)
        ones_minus_h = 1.0 - h_eval
        ones_minus_h = np.concatenate((ones_minus_h[1:], [0, 0]))

        corr = scipy.signal.correlate(h_eval, ones_minus_h, mode='full')
        c5 = np.max(corr[1:-1] * dx)
        return c5

    def perturb(h_eval):
        idxs = np.random.choice(n_points, size=int(n_points**0.5), replace=False)
        scale = 0.1 if evaluate(h_eval) > 1e-10 else 0.01
        noise = scale * (np.random.rand(len(idxs)) - 0.5)
        h_eval[idxs] += noise
        h_eval = np.clip(h_eval, 0, 1)
        return h_eval

    while time.time() - start_time < budget_s - 5:
        h_proposed = perturb(h)
        c5_proposed = evaluate(h_proposed)
        if c5_proposed < best_c5:
            best_c5 = c5_proposed
            best_h[:] = h_proposed.copy()
            t_best = time.time()
            early_stop_timer = time.time()
        elif np.random.rand() < 0.001:  # small chance to accept worse
            h[:] = h_proposed.copy()
        h[:] = best_h.copy() if best_c5 < evaluate(h) else h_proposed.copy()
        if best_c5 < 0.38085:
            return (best_h, best_c5, n_points)
        if time.time() - early_stop_timer > 60 and best_c5 < 0.5:
            break
    return (best_h, best_c5, n_points)
