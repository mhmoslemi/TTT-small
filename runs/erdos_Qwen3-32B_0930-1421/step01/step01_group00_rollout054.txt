def run(seed=42, budget_s=1100, verbose=False):

        h_old = h.copy()
        c5, max_shifts = compute_c5_with_shifts(h)

            best_c5 = c5

        top_shifts = max_shifts[:5]

            ones_minus_h = 1.0 - h
            if k < len(h):
                corr = h[:n_points - k] * (1 - h[k:])
            else:
                corr = h[k:] * (1 - h[:n_points - k])

            # Find top 10% indices where overlap is largest
            top_idx = np.argsort(corr)[::-1][:int(np.ceil(len(corr) * 0.1))]

            for i in top_idx:
                if i >= len(h):
                    continue
                j = i + k
                if j >= len(h):
                    continue

                # Decide whether to lower h[i] or raise h[j]
                lower_i = h[i] > 0.1 and (1 - h[j]) > 0.3
                raise_j = h[j] < 0.9 and (1 - h[j]) > 0.3

                if lower_i:
                    h[i] = max(0.0, h[i] - 0.02)
                elif raise_j:
                    h[j] = min(1.0, h[j] + 0.02)

        h.clip(0.0, 1.0, out=h)
        current_sum = h.sum()
        if not np.allclose(current_sum, target_sum, atol=1e-6):

        if verbose and (budget_remain < budget_s - 10):
            print(f"Time left: {budget_remain:.2f}s, Best C5: {best_c5:.10f}")

    return (best_h, best_c5, n_points)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h

    max_overlap = 0.0
    for k in range(1, n - 1):
        if k >= n:
            continue
        if k <= n:
            overlap = np.dot(h[:n - k], (1.0 - h[k:])) * dx
        else:
            overlap = np.dot(h[k:], (1.0 - h[:n - k])) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap


def compute_c5_with_shifts(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h

    overlaps = []
    for k in range(1, n - 1):
        if k >= n:
            continue
        if k <= n:
            overlap = np.dot(h[:n - k], (1.0 - h[k:])) * dx
        else:
            overlap = np.dot(h[k:], (1.0 - h[:n - k])) * dx
        overlaps.append((overlap, k))
    overlaps.sort(reverse=True)
    top_c5 = overlaps[0][0]
    top_shifts = [k for (v, k) in overlaps[:5] if v > 1e-6]
    return top_c5, top_shifts
