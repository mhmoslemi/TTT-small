import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def create_basis_functions(n_points, cen, alpha=4.0, beta=0.3):
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    psi = np.zeros((len(cen), n_points))
    for i, xi in enumerate(cen):
        left = max(0, int(np.ceil((xi - beta) / dx)))
        right = min(n_points, int(np.ceil((xi + beta) / dx)) + 1)
        x_sub = x[left:right]
        psi[i, left:right] = np.exp(-alpha * (x_sub - xi) ** 2)
    return psi


def optimize_coefficients(psi):
    n_basis = psi.shape[0]
    n_points = psi.shape[1]
    a = cp.Variable(n_basis, nonneg=True)
    dx = 2.0 / n_points
    # Normalization constraint
    constraint_int = []
    for i in range(n_basis):
        constraint_int.append(psi[i].sum() * dx)
    constraint_total = [a @ constraint_int == 1.0]

    # Overlap constraints
    C_overlap = np.zeros((10, n_basis, n_basis))  # shift from -9 to 9
    for k in range(-9, 10):
        k = int(k)
        for i in range(n_basis):
            for j in range(n_basis):
                shifted = np.roll(psi[j], k)
                C_overlap[k + 9, i, j] = (psi[i] * shifted).sum() * dx
    lambda_ = cp.Variable()
    constraints = [constraint_total]
    for k in range(10):  # forward shifts only, reverse symmetric
        for i in range(n_basis):
            for j in range(n_basis):
                constraints += [a[i] * a[j] * C_overlap[k, i, j] <= lambda_]
    # Objective: minimize maximum overlap
    objective = cp.Minimize(lambda_)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.CVXOPT, maxiters=100)
    return a.value


def run(budget_s=1100, n_points=42):
    start_time = time.time()

    # Step 1: Initial basis config
    if initial_h_values is not None:
        mass_grid = initial_h_values
    else:
        mass_grid = np.zeros(n_points)
        mass_grid[n_points // 2] = 1.0

    # Step 2: Basis centers using center of mass
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    com = np.sum(x * mass_grid)
    # Place 10 basis centers around COM with spacing
    centers = com - 0.9 + np.linspace(0, 1.8, 10)
    centers = np.clip(centers, 0.01, 1.99)

    # Step 3: Basis function optimization
    best_c5 = np.inf
    best_h = None
    while time.time() - start_time < budget_s * 0.8:
        psi = create_basis_functions(n_points, centers)
        a = optimize_coefficients(psi)
        if a is None:
            break

        h_basis = np.zeros(n_points)
        for i in range(psi.shape[0]):
            h_basis += a[i] * psi[i]
        h = ensure_feasible(h_basis)
        c5 = evaluate_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Perturb basis centers
        centers_pert = centers + np.random.uniform(-0.1, 0.1, len(centers))
        centers_pert = np.clip(centers_pert, 0.1, 1.9)

    if best_h is None:
        # fallback to uniform
        best_h = np.zeros(n_points)
        best_h += 1.0 / n_points * n_points / 2.0
        best_h = ensure_feasible(best_h)
        best_c5 = evaluate_C5(best_h)

    return best_h, best_c5, n_points
