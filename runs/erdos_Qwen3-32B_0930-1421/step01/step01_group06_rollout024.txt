import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Normalize initial_h_values
    total_h = float(np.sum(initial_h_values))
    h = np.minimum(np.copy(initial_h_values) * (n_points / 2.0 / total_h), 1.0)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)

    # Build feasible regions and product constraints
    def solve_feasibility(C5):
        h_opt = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt <= 1.0
        ]
        for k in range(1, n_points):
            offset = n_points
            corr = np.correlate(h_opt, 1 - h_opt, mode='valid')[k-1]
            constraints.append(corr * dx <= C5)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            result = prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=5)
            return h_opt.value if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] else None
        except:
            return None

    # Binary search for minimal C5
    C_low, C_high = 0.0, best_c5
    best_h = None
    while time.time() - start_time < budget_s - 10:
        if C_high - C_low < 1e-6:
            break
        C_mid = (C_low + C_high) / 2
        h_sol = solve_feasibility(C_mid)
        if h_sol is not None:
            h = np.clip(h_sol, 0.0, 1.0)
            total_h = np.sum(h)
            h = h * (n_points / 2.0) / total_h if total_h > 1e-9 else h
            c5_val = evaluate_erdos_solution(h, n_points)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            C_high = C_mid
        else:
            C_low = C_mid

    # Perturbation with warming
    if best_h is not None:
        perturbation_magnitude = 0.05
        for _ in range(3):
            h_pert = best_h + np.random.randn(n_points) * perturbation_magnitude
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert *= (n_points / 2.0) / np.sum(h_pert)
            c5_val = evaluate_erdos_solution(h_pert, n_points)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_pert.copy()

    return best_h, best_c5, n_points
