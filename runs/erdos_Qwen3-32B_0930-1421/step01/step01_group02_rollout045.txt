```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


initial_h_values = None  # Will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def build_cvx_model(centers, widths, support_length, n_points):
    dx = 2.0 / n_points
    grid = np.linspace(0.0, 2.0 - dx, n_points)
    N = len(centers)

    # Build matrix of basis functions
    W = np.zeros((n_points, N))
    for i in range(N):
        Φ = 2.0 * widths[i]  # Control width w.r.t. Gaussian's parametrization
        idxs_centroid = (grid - centers[i]) ** 2
        envelope = np.exp(-Φ * idxs_centroid)
        envelope /= np.sum(envelope)
        envelope_enforced = np.clip(envelope, 0, 1)
        W[:, i] = envelope_enforced

    a = cp.Variable(N)
    lam = cp.Variable()

    # Constraint: ∫h dx = 1
    integral_h = W.T @ a * dx
    obj_integral = cp.Constraint(integral_h == 1)

    # Objective: minimize worst overlap correlation
    K = np.zeros((N, N))
    for i in range(N):
        for j in range(N):
            shifted = np.roll(W[:, j], 1)
            K[i, j] = W[:, i] @ (1 - shifted) * dx

    # Envelope-wise maximum correlation: minimize λ such that a^T * K_i <= λ for all i
    obj_ows = [cp.Constraint(W[:, i] @ (1 - np.roll(W[:, i], k)) * dx <= lam for k in range(-n_points + 1, n_points)) for i in range(N)]

    # Overall constraints and objective
    constraints = [a >= 0, obj_integral]
    for o in obj_ows:
        constraints += o
    objective = cp.Minimize(lam)

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS)
    a_opt = a.value
    h_opt = W @ a_opt
    h_opt = ensure_feasible(h_opt)
    return h_opt


def run(budget_s=1100, n_points=42):
    start_time = time.time()

    if initial_h_values is not None:
        initial_h = initial_h_values
    else:
        # Define some reasonable initial configuration
        initial_centers = np.linspace(0.1, 1.9, 5)
        h = np.zeros(n_points)
        for x in initial_centers:
            idx = int(x * n_points / 2)
            w = 0.2
            left, right = max(0, idx - w * n_points), min(n_points, idx + w * n_points)
            h[left:right] = 1.0
        h = ensure_feasible(h)

    N = 10  # Number of basis functions to test
    α_range = np.linspace(2.0, 8.0, 6)
    β_range = np.linspace(0.25, 0.75, 6)
    centers_range = [np.linspace(0.1 + dx, 1.9 - dx, N)
                     for dx in np.linspace(0.0, 0.1, 4)]

    best_h = initial_h.copy()
    best_C5 = evaluate_C5(best_h)
    elapsed = time.time() - start_time

    for α in α_range:
        for β in β_range:
            if elapsed >= budget_s * 0.9:
                break
            centers = np.sort(np.random.uniform(0.1, 1.9, N))
            widths = np.full_like(centers, β)
            try:
                h_candidate = build_cvx_model(centers, widths, β, n_points)
                curr_C5 = evaluate_C5(h_candidate)
                if curr_C5 < best_C5:
                    best_h = h_candidate.copy()
                    best_C5 = curr_C5
            except Exception:
                pass
            elapsed = time.time() - start_time

    # Perturbation refinement
    for _ in range(4):
        if elapsed >= budget_s:
            break
        center_pert = best_h.copy()
        indices = np.argsort(center_pert)[-6:]
        for i in np.random.choice(len(center_pert), size=3, replace=False):
            center_pert[i] += np.random.uniform(-0.1, 0.1)
        centers = np.linspace(0.1, 1.9, N) + np.random.normal(0, 0.1, N)
        widths = np.full_like(centers, β)
        try:
            h_candidate = build_cvx_model(centers, widths, β, n_points)
            curr_C5 = evaluate_C5(h_candidate)
            if curr_C5 < best_C5:
                best_h = h_candidate.copy()
                best_C5 = curr_C5
        except:
            pass
        elapsed = time.time() - start_time

    return best_h, best_C5, n_points
```
