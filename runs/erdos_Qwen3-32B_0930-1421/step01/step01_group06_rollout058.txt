import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    last_improvement_time = start_time

    dx = 2.0 / n_points
    idxs = np.arange(n_points)

    def solve_convex_relaxation(C5, h_start):
        h = cp.Parameter(n_points, value=h_start)
        z = cp.Parameter((n_points, n_points), value=0.0)
        for k in range(1, n_points):  # Shifting distances
            for i in range(n_points - k):
                j = i + k
                z[i, j] = h[i] * (1 - h[j])

        corr = np.correlate(h, 1 - h, mode='full')[n_points - 1:2 * n_points - 1]
        corr = corr * dx
        constraint = cp.sum(cp.multiply(corr, [1.0 for _ in corr])) <= C5

        constraints = [
            constraint,
            cp.sum(h) == n_points / 2.0,
            0 <= h,
            h <= 1,
        ]

        for i in range(n_points):
            for j in range(n_points):
                if j - i >= 1 and j < n_points:
                    # McCormick envelopes for h[i]*(1 - h[j])
                    l = 0.0; u = 1.0
                    v = 0.0; w = 0.0
                    z[i, j] = cp.Parameter(n_points, 1.0)
                    constraint_z1 = z[i, j] >= l*1 + h[i]*(1 - w) + (1 - h[i])*((1 - l) - w) - l*w
                    constraint_z2 = z[i, j] <= 1 - h[j]
                    constraint_z3 = z[i, j] <= h[i]
                    constraint_z4 = z[i, j] >= h[i] + (1 - h[j]) - 1
                    constraints.append(constraint_z1)
                    constraints.append(constraint_z2)
                    constraints.append(constraint_z3)
                    constraints.append(constraint_z4)

        objective = cp.Minimize(C5)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI, verbose=False, time_limit=5)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return problem.value, h.value
        return None, h.value

    # Binary search on C5
    C_low, C_high = 0.0, best_c5
    while time.time() - start_time < budget_s - 10:
        if C_high - C_low < 1e-6:
            break
        C_mid = (C_low + C_high) / 2
        c5_scale, sol_h = solve_convex_relaxation(C_mid, best_h)
        if c5_scale is not None:
            C_high = C_mid
            if c5_scale < best_c5:
                best_c5 = c5_scale
                best_h = sol_h.copy()
                last_improvement_time = time.time()
        else:
            C_low = C_mid

        # Perturbation warm_starts
        if (time.time() - last_improvement_time) > 60 and np.random.rand() < 0.2:
            # Perturb and normalize
            perturbed_h = best_h + np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h *= (n_points / 2) / perturbed_h.sum()
            # Run one warm_start iteration
            c5_scale, sol_h = solve_convex_relaxation(C_high, perturbed_h)
            if c5_scale is not None and c5_scale < best_c5:
                best_c5 = c5_scale
                best_h = sol_h.copy()
                last_improvement_time = time.time()

    return best_h, best_c5, n_points
