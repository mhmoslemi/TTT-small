import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Simple uniform initialization
    else:
        n_points = len(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        kh = 1 - h
        c = correlate(h, kh, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        sum_h = np.sum(h_in)
        scale = (n_points / 2) / sum_h
        return np.clip(h_in * scale, 0.0, 1.0)

    def gradient_step(h, learning_rate):
        kh = 1 - h
        c = correlate(h, kh, mode='full') * dx
        k_star = np.argmax(c)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points - k_star):
                if i + k_star < n_points:
                    grad[i] -= h[i + k_star]
            grad[i] -= kh[np.abs(i - k_star)] if 0 <= np.abs(i - k_star) < n_points else 0.0
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h = normalize_h(h)

    while time.time() - start_time < budget_s - 10:
        g = gradient_step(h, 0.0)
        learning_rate = 0.1
        alpha = 0.5
        for _ in range(5):
            h_test = normalize_h(h - learning_rate * g)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                h = h_test
                best_c5 = c5_test
                best_h = h.copy()
                break
            else:
                learning_rate *= alpha

    return best_h.tolist(), best_c5, n_points
