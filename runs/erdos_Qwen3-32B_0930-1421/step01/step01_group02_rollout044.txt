```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
        h = ensure_feasible(h)
    else:
        h *= (len(h) / 2) / total
    return h


def construct_bases(n_bases=10, alpha=4, beta=0.3, n_points=42):
    x = np.linspace(0, 2, n_points)
    bases = []
    for i in range(n_bases):
        x_center = 0.5 + 1.0 * i / n_bases
        g = np.zeros_like(x)
        idxs = np.abs(x - x_center) <= beta
        g[idxs] = np.exp(-alpha * (x[idxs] - x_center) ** 2)
        bases.append(g)
    return np.array(bases)


def perturb_bases(bases, scale=0.1):
    new_bases = []
    for b in bases:
        x_centers = b.argmax()
        x_centers = x_centers + np.random.uniform(-scale, scale)
        x_centers = np.clip(x_centers, 0, 1)
        new_bases.append(x_centers)
    return np.array(new_bases)


def run(budget_s=1100):
    start_time = time.time()
    n_points = 42
    if initial_h_values is not None:
        h_init = initial_h_values
    else:
        h_init = np.zeros(n_points)
        h_init[n_points // 2] = 1.0
        h_init = ensure_feasible(h_init)

    dx = 2.0 / n_points
    best_h = np.copy(h_init)
    best_C5 = evaluate_C5(best_h)
    n_bases = 10

    while time.time() - start_time < budget_s:
        bases = construct_bases(n_bases, n_points=n_points)
        A = bases.T

        a = cp.Variable(n_bases)
        ones_vec = np.ones(n_points)
        constraint = cp.sum([a[i] * cp.sum(bases[i]) for i in range(n_bases)]) == 1.0

        corr_exprs = []

        for shift in range(-n_points + 1, n_points):
            corr = 0.0
            for i in range(n_bases):
                for j in range(n_bases):
                    displaced = np.roll(bases[j], shift)
                    displaced = np.clip(displaced, 0, n_points - 1)
                    corr += a[i] * (1 - a[j]) * cp.sum(bases[i] * displaced)
            corr_exprs.append(corr)

        λ = cp.Variable()
        objective = cp.Minimize(λ)
        constraints = [constraint] + [cp.sum(corr_exprs[k]) <= λ for k in range(len(corr_exprs))]
        constraints += [0 <= a, a <= 1.0]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT)

        if prob.status != 'optimal':
            continue

        a_vals = a.value
        h = np.zeros(n_points)
        for i in range(n_bases):
            h += a_vals[i] * bases[i]

        h = ensure_feasible(h)
        C5 = evaluate_C5(h)
        
        if C5 < best_C5:
            best_h = np.copy(h)
            best_C5 = C5

        perturbed_bases = construct_bases(n_bases, n_points=n_points)
        bases = np.array([np.roll(b, shift= int(np.random.uniform(-2,2))) for b in perturbed_bases])

    return best_h, best_C5, n_points
```
