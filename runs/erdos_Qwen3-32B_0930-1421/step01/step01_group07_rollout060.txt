import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def smooth_perturb(h, num_perturb):
    n = len(h)
    indices = np.random.choice(n, num_perturb, replace=False)
    delta = np.random.normal(0, 0.1, size=num_perturb)
    h_new = h.copy()
    h_new[indices] += delta
    h_new = np.clip(h_new, 0.0, 1.0)
    # Normalize to satisfy sum constraint
    current_sum = h_new.sum()
    if abs(current_sum - n / 2) > 1e-5:
        h_new = np.clip(h_new * (n / 2) / current_sum, 0.0, 1.0)
    return h_new

    current_h = h.copy()
    current_c5 = best_c5
    no_improvement = 0
