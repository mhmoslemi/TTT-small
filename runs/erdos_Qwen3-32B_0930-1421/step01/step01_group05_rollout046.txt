import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Load initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: Random initialization with 45 points
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = float('inf')

    def compute_c5(h):
        # Evaluation of the max overlap integral
        h1_h = 1 - h
        c = correlate(h, h1_h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        # Clip and scale to ensure constraint
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        scaled = clipped * (n_points / 2) / total
        return np.clip(scaled, 0, 1)

    # Gradient descent with line search
    def gradient(h, epsilon=1e-5):
        n = len(h)
        grad = np.zeros(n)
        max_k = -1
        max_overlap = -1e10
        h1_h = 1 - h

        # Find the k (shift) that produces the max overlap
        for k in range(-n + 1, n):
            i = max(0, k)
            j = min(n, n + k)
            overlap = np.sum(h[i:j] * h1_h[:j - i])
            if overlap > max_overlap:
                max_overlap = overlap
                max_k = k

        # Gradient for the max-k overlap
        for i in range(n):
            # Forward finite difference
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = normalize_h(h_plus)
            h1_h_plus = 1 - h_plus
            overlap_plus = 0.0
            for k in range(-n + 1, n):
                if k == max_k:
                    i_k = max(0, k)
                    j_k = min(n, n + k)
                    overlap_plus = np.sum(h_plus[i_k:j_k] * h1_h_plus[:j_k - i_k])
                    break

            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = normalize_h(h_minus)
            h1_h_minus = 1 - h_minus
            overlap_minus = 0.0
            for k in range(-n + 1, n):
                if k == max_k:
                    i_k = max(0, k)
                    j_k = min(n, n + k)
                    overlap_minus = np.sum(h_minus[i_k:j_k] * h1_h_minus[:j_k - i_k])
                    break

            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        return grad

    learning_rate = 0.08
    momentum = 0.7
    velocity = np.zeros_like(h)
    
    while time.time() - start_time < budget_s:
        grad = gradient(h)
        velocity = momentum * velocity + learning_rate * grad
        h_proposed = h - velocity
        h_proposed = normalize_h(h_proposed)

        c5_proposed = compute_c5(h_proposed)
        c5_current = compute_c5(h)

        if c5_proposed < c5_current:
            h = h_proposed.copy()
            if c5_proposed < best_c5:
                best_c5 = c5_proposed
                best_h = h.copy()

        # Line search adaptation
        if c5_proposed < best_c5:
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9

        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
