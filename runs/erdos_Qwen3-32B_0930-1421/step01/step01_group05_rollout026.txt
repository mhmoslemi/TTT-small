import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_pop = [np.array(initial_h_values, dtype=np.float64)]
        n_points = len(h_pop[0])
    else:
        n_points = 45
        h_pop = [np.random.rand(n_points) for _ in range(50)]

    # Normalize the population
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        if abs(total) > 1e-10:
            h *= (n_points / 2) / total
        else:
            h[:] = 0.5
        return h

    for h in h_pop:
        normalize_h(h)

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Firefly algorithm parameters
    n_pop = len(h_pop)
    beta0 = 1.0
    gamma = 0.01
    alpha = 0.2
    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        # Compute brightness (C₅) for all fireflies
        c5_values = [compute_c5(h) for h in h_pop]
        idx_sorted = np.argsort(c5_values)
        h_pop = [h_pop[i] for i in idx_sorted]
        c5_values = [c5_values[i] for i in idx_sorted]
        best_idx = 0
        if c5_values[0] < best_c5:
            best_c5 = c5_values[0]
            best_h = h_pop[0].copy()

        # Update firefly positions
        for i in range(1, n_pop):
            for j in range(i):
                if c5_values[j] < c5_values[i]:
                    r = np.linalg.norm(h_pop[i] - h_pop[j])
                    alpha_t = alpha * (1 - i / n_pop)  # decaying randomization
                    beta = beta0 * np.exp(-gamma * r**2)
                    delta = beta * (h_pop[j] - h_pop[i]) + alpha_t * (np.random.rand(n_points) - 0.5)
                    h_pop[i] += delta
                    normalize_h(h_pop[i])

        # Reduce step sizes over time
        gamma *= 1.01

        # Check budget
        remaining = budget_s - (time.time() - start_time)
        if remaining < 1:
            break

    return best_h.tolist(), best_c5, n_points
