import numpy as np
import time
from scipy.signal import correlate

# This version implements a projected gradient descent with backtracking line search
def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h_proj = np.clip(h, 0, 1)
    # Normalize to ensure sum(h) == n_points / 2
    sum_h = np.sum(h_proj)
    if sum_h > 0:
        h_proj *= (n_points / 2) / sum_h
    else:
        # If all zeros (due to clipping), assign half 1s and half 0s
        idxs = np.random.permutation(n_points)
        h_proj[:] = 0
        h_proj[idxs[:n_points//2]] = 1.0
    # Final clip to handle small rounding errors
    return np.clip(h_proj, 0, 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + lag < n_points:

            if i - lag >= 0:
                grad[i] -= h[i - lag]

        # Normalize gradient
        norm_grad = np.linalg.norm(grad)
        if norm_grad > 0:

        # Backtracking line search
        alpha = alpha_start
        success = False
        while not success and time.time() - t0 < budget_s:
            h_proposed = h - alpha * grad
            h_proposed = project_to_feasible(h_proposed, n_points)

                h = h_proposed
                if c5_proposed < best_c5:
                    best_c5 = c5_proposed
                    best_h = h.copy()
                alpha *= 1.1  # Success - increase step size
                success = True
            else:
                # Backtrack
                alpha *= gamma
                if alpha < 1e-5:

        if it % 500 == 0:

    final_c5 = evaluate_overlap(best_h)
