import numpy as np
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import scipy.optimize as opt

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corrs = []
    for k in range(1, n_points):
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, sigma, n_points):
    h_new = h + sigma * np.random.randn(n_points)
    return enforce_normalization(h_new, n_points)

    n_init = 10

        c5_proposal = compute_c5(h_proposal, n_points)
        delta = c5_proposal - best_c5

        # Metropolis criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_best = h_proposal.copy()
            best_c5 = c5_proposal
            last_improve = time.time()
            X = np.vstack([X, h_proposal.reshape(1, -1)])
            y = np.append(y, best_c5)
            gp.fit(X, y)

        # Adaptive cooling
        T *= alpha
        sigma *= 0.999
        T = max(T, 0.01)

        # End early if needed
        if time.time() - last_improve > 100 and T < 10.0:
            if sigma > 0.001:
                sigma *= 0.5
            else:
                alpha **= 0.9

    # Final refinement
    def objective(x):
        return compute_c5(x.reshape(-1), n_points)

    res = opt.minimize(
        objective,
        h_best,
        bounds=[(0.0, 1.0)] * n_points,
        method="L-BFGS-B",
        options={'maxiter': 100, 'eps': 1e-4}
    )
    final_h = enforce_normalization(res.x, n_points)
    final_c5 = compute_c5(final_h, n_points)
