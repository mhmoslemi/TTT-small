import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).astype(np.float64)
    else:
        h = np.ones(n_points) / n_points * (n_points / 2)
    h = np.clip(h, 0, 1.0)

    dx = 2.0 / n_points

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        total = np.sum(clipped)
        scale = (n_points / 2) / total
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_c5 = float('inf')
    best_h = h.copy().tolist()

    def phase_gradient_descent(h, time_end, learning_rate=0.1):
        while time.time() < time_end:
            grad = np.zeros_like(h)
            eps = 1e-5

            for i in range(n_points):
                h_p = h.copy()
                h_p[i] += eps
                h_p = normalize_h(h_p)
                c_p = compute_c5(h_p)

                h_n = h.copy()
                h_n[i] -= eps
                h_n = normalize_h(h_n)
                c_n = compute_c5(h_n)

                grad[i] = (c_p - c_n) / (2 * eps)

            h -= learning_rate * grad
            h = normalize_h(h)

            current_c5 = compute_c5(h)
            nonlocal best_c5, best_h
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy().tolist()

            learning_rate *= 0.999

    # Phase 1: Gradient descent
    phase1_end = start_time + 0.9 * budget_s
    phase_gradient_descent(h.copy(), phase1_end, learning_rate=0.01)

    return best_h, best_c5, n_points
