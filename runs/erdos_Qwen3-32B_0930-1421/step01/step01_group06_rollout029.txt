import numpy as np
import time
from scipy.optimize import OptimizeResult

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values)
    h0 = initial_h_values * (n_points / 2.0) / initial_h_values.sum()
    h0 = np.clip(h0, 0, 1)

    num_particles = 75
    max_velocity = 0.15
    w_start = 0.9
    w_end = 0.3
    c1 = 2.0
    c2 = 2.0
    reset_fraction = 0.1

    # Initialize particles
    particles = np.tile(h0, (num_particles, 1))
    velocities = 0.01 * np.random.randn(num_particles, n_points)
    particles[:int(num_particles * 0.25)] = np.random.rand(int(num_particles * 0.25), n_points)
    particles = np.clip(particles, 0, 1)
    for i in range(num_particles):
        particles[i] *= (n_points / 2.0) / np.sum(particles[i])

    # Initialize personal and global best
    p_best = np.copy(particles)
    p_best_scores = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
    best_idx = np.argmin(p_best_scores)
    g_best = np.copy(p_best[best_idx])
    best_c5 = p_best_scores[best_idx]
    last_improvement_time = start_time

    def evaluate_all():
        return np.array([evaluate_erdos_solution(p, n_points) for p in particles])

    def update_positions_and_velocities(scores, velocities, particles, p_best, g_best, w):
        r1 = np.random.rand(num_particles, 1)
        r2 = np.random.rand(num_particles, 1)
        velocities = w * velocities + \
                     c1 * r1 * (p_best - particles) + \
                     c2 * r2 * (g_best - particles)
        velocities = np.clip(velocities, -max_velocity, max_velocity)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(num_particles):
            s = np.sum(particles[i])
            if s > 0:
                particles[i] *= (n_points / 2.0) / s
            else:
                particles[i] = h0
        return velocities, particles

    while time.time() - start_time < budget_s:

        # Evaluate all particles
        scores = evaluate_all()
        improved = (scores < p_best_scores)
        p_best[improved] = particles[improved]
        p_best_scores[improved] = scores[improved]

        if np.any(scores < best_c5):
            best_idx = np.argmin(scores)
            g_best = np.copy(particles[best_idx])
            best_c5 = scores[best_idx]
            last_improvement_time = time.time()

        # Stopping criterion
        if time.time() - last_improvement_time > 90:
            break

        # Update velocities and positions
        w = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
        w = max(w, w_end)
        velocities, particles = update_positions_and_velocities(scores, velocities, particles, p_best, g_best, w)

        # Reset some particles to avoid stagnation
        num_resets = max(1, int(num_particles * reset_fraction))
        reset_mask = np.random.choice(num_particles, size=num_resets, replace=False)
        for i in reset_mask:
            particles[i] = initial_h_values * (n_points / 2.0) / initial_h_values.sum()
            particles[i] = np.clip(particles[i], 0, 1)

    result = OptimizeResult()
    result["fun"] = best_c5
    result["x"] = g_best
    result["success"] = True

    print(f"Final C5 bound: {best_c5:.6f}")
    return g_best, best_c5, n_points
