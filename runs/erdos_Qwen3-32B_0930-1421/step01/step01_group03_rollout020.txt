import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 100
    n_points = max_n_points

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values, copy=True)
    else:
        # Construct balanced initialization
        h_base = np.zeros(n_points)
        splits = int(np.floor(n_points / 4))
        step = int(n_points / (2 * splits + 1))
        h_base[::step] = (2.0 / n_points)
        h_base = (h_base / h_base.sum()) * (n_points / 2)

    dx = 2.0 / n_points
    constraint_sum = n_points / 2

    def compute_c5(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    best_h = h_base.copy()
    best_c5 = compute_c5(best_h)

    # Try different swarm origins
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.9:
            break
        h_start = np.random.uniform(0, 1, n_points)
        h_start = (h_start / h_start.sum()) * (n_points / 2)

        # Smooth weight refinement
        bounds_sm = [(0, 1) for _ in range(n_points)]
        result_sm = minimize(
            fun=compute_c5,
            x0=h_start,
            method='L-BFGS-B',
            bounds=bounds_sm,
            options={'maxiter': 100}
        )
        if result_sm.success:
            h_sm = result_sm.x
            h_sm = h_sm.clip(0, 1)
            h_sm_sum = h_sm.sum()
            h_sm = h_sm * (constraint_sum / h_sm_sum)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5:
                best_h = h_sm
                best_c5 = c5_sm

    # Perturb active support with structured search
    def perturb_structure(h, perturbation_rate=0.05):
        h = h.copy()
        perturb_indices = np.random.choice(n_points, replace=False, size=int(n_points * perturbation_rate))
        h[perturb_indices] = np.random.uniform(0, 1, len(perturb_indices))
        h = h.clip(0, 1)
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)
        return h

    for _ in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        h_pert = perturb_structure(best_h)
        h_ref = h_pert.copy()

        # Structural refinement
        bounds_ref = [(0, 1) for _ in range(n_points)]
        result_ref = minimize(
            fun=compute_c5,
            x0=h_ref,
            method='L-BFGS-B',
            bounds=bounds_ref,
            options={'maxiter': 100}
        )
        if result_ref.success:
            h_refined = result_ref.x
            h_refined = h_refined.clip(0, 1)
            h_sum = h_refined.sum()
            h_refined = h_refined * (constraint_sum / h_sum)
            c5_refined = compute_c5(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

    # Final constraint projection
    best_h = best_h.clip(0, 1)
    best_h_sum = best_h.sum()
    best_h = best_h * (constraint_sum / best_h_sum)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = best_h.clip(0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
