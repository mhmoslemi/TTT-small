```python
import numpy as np
import time
from scipy.signal import correlate
import heapq
from itertools import product

initial_h_values = None  # will be provided at runtime


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    corr = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr


def is_feasible(h):
    return np.all((0 <= h) & (h <= 1)) and np.isclose(h.sum(), np.ceil(len(h) / 2))


def discretize(h, levels=5):
    """Map h to nearest discrete values (0, 0.25, 0.5, 0.75, 1)"""
    step = 1 / (levels - 1)
    return step * np.round(h / step)


def make_feasible(h):
    """Ensure sum constraint and bounds"""
    h = np.clip(discretize(h), 0, 1)
    total = h.sum()
    if total == 0:
        h[0] = 1.0
        return make_feasible(h)
    scale = (len(h) / 2) / total
    h = np.clip(h * scale, 0, 1)
    return h


def ensure_feasible_sum(h):
    """Fix rounding errors"""
    current_sum = h.sum()
    goal = np.ceil(len(h) / 2)
    diff = goal - current_sum
    if diff == 0:
        return h
    h = np.where(h >= 1 - 1e-6, 1.0, np.where(h <= 1e-6, 0.0, h))
    to_adjust = min(abs(diff), np.isclose(h, 0).sum() + np.isclose(h, 1).sum())
    if diff > 0 and to_adjust > 0:
        h[np.isclose(h, 0)] = 1.0
        h = h[:to_adjust]
        h = np.concatenate([np.ones_like(h), h[to_adjust:]])
    elif diff < 0 and to_adjust > 0:
        h[np.isclose(h, 1)] = 0.0
        h = h[:to_adjust]
        h = np.concatenate([np.zeros_like(h), h[to_adjust:]])
    h = make_feasible(h)
    return h


def get_neighbors(h, n_points):
    h = ensure_feasible_sum(h)
    indices = list(range(n_points))
    neighbors = []
    for idx in indices:
        for val in [0, 0.25, 0.5, 0.75, 1]:
            if abs(h[idx] - val) < 1e-12:
                continue
            h_new = h.copy()
            h_new[idx] = val
            h_new = ensure_feasible_sum(h_new)
            if not np.array_equal(h_new, h):
                neighbors.append(h_new)
        if len(neighbors) >= 100:
            break
    return neighbors


def local_search(h, max_iter=500, n_points=42):
    h = ensure_feasible_sum(h)
    best = h
    best_c5 = evaluate_C5(best)
    for _ in range(max_iter):
        neighbors = get_neighbors(best, n_points)
        if not neighbors:
            break
        scores = [evaluate_C5(n) for n in neighbors]
        next_idx = np.argmin(scores)
        next_h = neighbors[next_idx]
        if scores[next_idx] < best_c5:
            best = next_h
            best_c5 = scores[next_idx]
        else:
            break
    return best, best_c5


def branch_and_bound(h, depth=0, max_depth=5, best_c5=np.inf):
    h = ensure_feasible_sum(h)
    current_c5 = evaluate_C5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
    if depth >= max_depth:
        return h, best_c5

    candidates = get_neighbors(h, len(h))
    candidates = [c for c in candidates if np.any(c != h)]

    for c in candidates:
        c, c5 = branch_and_bound(c, depth + 1, max_depth, best_c5)
        if c5 < best_c5:
            best_c5 = c5
            h = c
    return h, best_c5


def perturb_h(h, n_perturb=3):
    h = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, n_perturb, replace=False)
    for i in indices:
        h[i] = np.random.choice([0, 0.25, 0.5, 0.75, 1])
    return ensure_feasible_sum(h)


def run(budget_s=1100, n_points=42, init_perturb=5, max_depth=5):
    start_time = time.time()
    if initial_h_values is not None:
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)
    h = ensure_feasible_sum(discretize(h))
    best_c5 = evaluate_C5(h)
    best_h = h.copy()

    trial_best_c5 = []
    elapsed = 0

    while elapsed < budget_s:
        h_perturbs = [perturb_h(best_h, n_perturb=init_perturb) for _ in range(2)]
        for h_p in h_perturbs:
            h_bb, c5 = branch_and_bound(h_p, max_depth=max_depth, best_c5=best_c5)
            if c5 < best_c5:
                best_h = h_bb
                best_c5 = c5
        # Local refinement
        h_ls, c5_ls = local_search(best_h, max_iter=50, n_points=n_points)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls
        # Log progress
        trial_best_c5.append(best_c5)
        if len(trial_best_c5) > 10:
            trial_best_c5.pop(0)
        # Time check
        elapsed = time.time() - start_time
        # Downgrade memoization to avoid memory bloat
        if len(trial_best_c5) > 3 and all(np.isclose(best_c5, trial_best_c5[-4:-1])):
            break
    return best_h, best_c5, n_points
```
