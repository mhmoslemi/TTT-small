import numpy as np
import time
import scipy.signal

def compute_c5(h, dx):
    n_points = len(h)
    h = np.clip(h, 0, 1)
    ones_minus_h = 1.0 - h.copy()
    max_corr = -np.inf
    for k in range(1, n_points):
        if k >= n_points:
            continue
        corr1 = np.dot(h[:n_points - k], ones_minus_h[k:])
        corr2 = np.dot(h[k:], ones_minus_h[:n_points - k])
        max_corr = max(max_corr, corr1 * dx, corr2 * dx)
    return max_corr
