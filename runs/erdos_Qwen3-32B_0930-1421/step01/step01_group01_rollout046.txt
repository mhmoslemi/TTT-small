import numpy as np
import time
import scipy.optimize as opt
from scipy.signal import correlate
from numpy.lib.stride_tricks import as_strided


def rolling_window(a, window, padding="valid"):
    """Create a rolling window view of the array."""
    pad_width = [(0, window - 1)] + [(0, 0)] * (a.ndim - 1)
    a_padded = np.pad(a, pad_width=pad_width, mode="constant", constant_values=0)
    shape = a_padded.shape[:-1] + (a_padded.shape[-1] - window + 1, window)
    strides = a_padded.strides + (a_padded.strides[-1],)
    return as_strided(a_padded, shape=shape, strides=strides)


def evaluate_C5(h, dx):
    h1 = h.copy()
    h1[h1 < 0] = 0
    h1[h1 > 1] = 1
    h_sum = h1.sum()
    h1 = h1 / h_sum * len(h1) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h1, 1 - h1, mode="full")
    C5 = np.max(cor) * dx
    return C5


def rescale_h(h):
    """Rescale h to satisfy the constraint sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def random_segment_flip(h, segment_len=10, p_flip=0.5):
    """Randomly flip a segment with probability p_flip."""
    n = len(h)
    start = np.random.randint(0, n - segment_len + 1)
    slice_ = slice(start, start + segment_len)
    if np.random.rand() < p_flip:
        h[slice_] = 1 - h[slice_]
    return h


def perturb_h(h, amplitude=0.1, noise_type="gauss"):
    """Perturb h with small amounts of noise, ensure [0, 1]."""
    if noise_type == "uni":
        pert = amplitude * (np.random.rand(len(h)) - 0.5)
    elif noise_type == "gauss":
        pert = amplitude * np.random.randn(len(h))
    else:
        pert = 0.0
    h += pert
    return np.clip(h, 0, 1)
