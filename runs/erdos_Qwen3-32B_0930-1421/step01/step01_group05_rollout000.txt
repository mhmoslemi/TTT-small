import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Default to initial_h_values when available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    patience_idx = 0
    patience_limit = 30  # Number of iterations without improvement before stopping

    # ILP formulation using binary vectors with relaxation
    while time.time() - start_time < budget_s:
        # ILP Step
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            h_var <= 1.0,
            cp.sum(h_var) == n_points / 2
        ]
        overlap = 0.0
        for shift in range(1, n_points):  # Avoid self-overlap at 0
            overlap += cp.sum(h_var * (1 - cp.roll(h_var, shift)))

        # Objective: Minimize the worst-case overlap across shifts
        objective = cp.Minimize(cp.norm(overlap, 1))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.SCIP)
        ilp_h = np.clip(h_var.value, 0, 1)

        # Update best solution if improvement
        normalized_ilp_h = normalize_h(ilp_h)
        c5_val = compute_c5(normalized_ilp_h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = normalized_ilp_h.copy()
            patience_idx = 0
        else:
            patience_idx += 1

        # Exit early if patience is exhausted
        if patience_idx >= patience_limit:
            break

    return best_h.tolist(), best_c5, n_points
