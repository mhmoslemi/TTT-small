import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    # Rescale to zero mean
    mean = 0.5  # Expected mean for uniform [0,1]
    h_centered = h - mean
    # Clipping
    h_clipped = np.clip(h_centered, -mean, mean)
    # Shift back
    h_fixed = h_clipped + mean
    # Project to feasible set
    h_proj = np.clip(h_fixed, 0.0, 1.0)
    # Rescale to meet ∑h = n_points / 2
    scale = (n_points / 2.0) / np.sum(h_proj)
    h_proj *= scale
    return h_proj

def run(**params):
    np.random.seed(42)
    global initial_h_values
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    # Ensure h is in feasible region
    h_prev = h.copy()
    h = project_h(h, n_points)
    assert abs(np.sum(h) - n_points / 2) < 1e-8, "Initial h violates sum constraint"
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)

    t0 = time.time()
    budget_s = 1100

    alpha = 1e-2  # Initial step size
    min_alpha = 1e-6
    beta = 0.9  # Momentum damping
    noise_every = 100

    momentum = np.zeros_like(h)

    while time.time() - t0 < budget_s:

        h1 = 1.0 - h
        corr = correlate(h, h1, mode="full")
        k_shift = np.argmax(corr)
        n_half = n_points - 1
        lag = k_shift - (n_half)

        # Build gradient of h[k] * (1 - h[k + lag]) for + lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + lag < n_points:
                grad[i] += (1 - h[i + lag])
            if i - lag >= 0:
                grad[i] -= h[i - lag]

        grad = grad / (np.linalg.norm(grad) + 1e-10)

        # New search gradient with momentum
        grad = beta * momentum + (1 - beta) * grad

        new_h = h - alpha * grad

        # Feasibility projection
        new_h = project_h(new_h.copy(), n_points)

        new_c5 = evaluate_overlap(new_h)

        # Check if better, or accept via line search
        if new_c5 < best_c5:
            h = new_h.copy()
            momentum = grad
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= 1.1  # Increase acceptance rate success
            alpha = min(alpha, 1e-2)
        else:
            # Backtracking line search
            alpha *= 0.5
            alpha = max(alpha, min_alpha)

        # Perturbation to escape local minima
        if abs(np.sum(h)) > n_points:
            h = project_h(h, n_points)

        if np.random.rand() < 0.01 or time.time() - t0 > budget_s - 60:  # Add noise once in a while
            pert = (np.random.normal(0, 0.02, size=n_points) * (1 - 2 * h))
            h = project_h(h + pert, n_points)

    # Final validation
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Constraint violated"
    return (best_h, best_c5, n_points)
