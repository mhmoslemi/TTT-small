import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = min(budget_s, 1100)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def clip_and_scale(optimizer_h):
        optimizer_h = np.clip(optimizer_h, 0, 1)
        current_sum = optimizer_h.sum()
        if abs(current_sum - target_sum) > 1e-8:
            optimizer_h *= target_sum / current_sum
        return optimizer_h

    def compute_c5(optimizer_h):
        optimizer_h = clip_and_scale(optimizer_h)
        ones_minus_h = 1 - optimizer_h
        correlations = scipy.signal.correlate(optimizer_h, ones_minus_h, mode='full')
        correlations[0:n_points-1] = -np.inf  # filter out k=0
        return correlations[n_points:].max() * dx

    h = clip_and_scale(h)
    best_h, best_c5 = h.copy(), compute_c5(h)

    last_improvement = start_time
    target_improvement = 0.380876

    while time.time() - start_time < budget_s - 5:
        current_corr = scipy.signal.correlate(h, 1-h, mode='full')
        over_threshold = current_corr[n_points:] > (best_c5 * dx / 1.2)
        problematic_lags = np.where(over_threshold)[0]

        if len(problematic_lags) > 0:
            k = int(np.random.choice(problematic_lags))
            # Indices j where h[j]*(1 - h[j+k]) is large
            j_above = np.where(h[:n_points-k] * (1 - h[k:]) > 0.02)[0]
            if len(j_above) > 0:
                num_adjust = min(5, len(j_above))
                j_adjust = np.random.choice(j_above, size=num_adjust, replace=False)
                h[j_adjust] *= 0.95  # Lower h[j] when it is large and h[j+k] is small
                h[j_adjust + k] *= 1.05  # Increase h[j+k] complementarily

        h = clip_and_scale(h)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = new_c5
            last_improvement = time.time()
            if new_c5 < target_improvement:
                break  # Success condition met early

        if time.time() - last_improvement > 120 and best_c5 > target_improvement:
            # Stagnation or improvement not sufficient
            break

        if verbose and (time.time() - start_time) % 5 < 1e-3:
            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
