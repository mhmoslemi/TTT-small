import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, lag, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + lag < n_points:
            grad[i] += (1 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    grad *= (2 / n_points)  # Scale by dx
    return grad

def project_l1_l2(h, sum_target):
    h_clipped = np.clip(h, 0, 1)
    while not np.allclose(np.sum(h_clipped), sum_target, atol=1e-6):
        scale = sum_target / np.sum(h_clipped)
        h_clipped *= scale
        h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

    best_c5 = evaluate_overlap(h)
    t0 = time.time()

    alpha_incr = 1.01
    alpha_decr = 0.5
    patience = 0

        corr = correlate(h, 1 - h, mode="full")

        # Compute gradient
        grad = compute_gradient(h, lag, n_points)

        current_c5 = new_c5
