import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    max_n_points = 93  # Max sample size to stay small yet expressive
    n_points = 93 if 'initial_h_values' not in globals() else len(initial_h_values)
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True).astype(np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        unit_per_dp = n_points // 2
        h[np.linspace(0, n_points-1, num=unit_per_dp, dtype=int)] = 1 / dx  # 1 unit over [0,2]
        h = h * (n_points / 2 / h.sum())  # project to constraint

    def objective(h):
        h = h.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        h_normalize = h_clipped * (n_points / 2 / s)
        overlaps = np.correlate(h_normalize, 1 - h_normalize, mode='full') * dx
        return np.max(overlaps)

    # Convex support set: Fix indices where h > 0.01 to reduce combinatorics
    def convex_fine_tune(h):
        active = h > 0.01  # Active indices
        h_active = h[active]
        len_active = len(h_active)

        w = cp.Variable(len_active)
        constraint = [w >= 0, w <= 1, w.sum() == n_points / 2]
        overlap = 0

        # Precompute (1 - h_normalize) for active indices shifted by k
        h_active_arr = h_active.reshape((len_active, 1))
        shifted = np.hstack([h_active_arr] * n_points)

        for k in range(-n_points + 1, n_points):
            if k <= 0:
                a = h_active_arr[:, -k:]
                b = shifted[:, :k + n_points]
            else:
                a = h_active_arr[:, :n_points - k]
                b = shifted[:, k:]
            prod = a * b
            overlap += cp.max(1 - prod.sum(axis=0))

        prob = cp.Problem(cp.Minimize(cp.Max(overlap * dx)))
        prob.solve(solver=cp.MOSEK, verbose=False)
        w_opt = w.value

        h_new = np.zeros_like(h)
        h_new[active] = w_opt

        # Normalize total mass to constraint
        h_new = h_new.clip(0, 1)
        h_new = h_new * (n_points / 2 / h_new.sum())

        return h_new

    # Structure perturbation
    def perturb_structure(h, sparsity=0.1):
        h = h.copy()
        random_indices = np.random.rand(n_points) < sparsity
        h[random_indices] = 0  # Zero out some positions
        h = h.clip(0, 1) * (n_points / 2) / h.clip(0, 1).sum()
        return h

    best_h = h.copy()
    best_c5 = objective(best_h)

    # Try convex fine-tune from initial
    if time.time() - start_time < budget_s * 0.3 and 'initial_h_values' in globals():
        try:
            h_tuned = convex_fine_tune(h)
            c5_tuned = objective(h_tuned)
            if c5_tuned < best_c5:
                best_h = h_tuned
                best_c5 = c5_tuned
        except:
            pass  # Skip if MOSEK unavailable or issue

    # Run gradient-based refinement
    def gradient_based(h):
        h_opt = minimize(
            lambda x: objective(x),
            h,
            bounds=[(0, 1)] * n_points,
            method='L-BFGS-B',
            options={'maxiter': 100},
        ).x
        return h_opt

    if time.time() - start_time < budget_s * 0.6:
        h_gb = gradient_based(best_h)
        c5_gb = objective(h_gb)
        if c5_gb < best_c5:
            best_h = h_gb
            best_c5 = c5_gb

    # Perturb and retry
    for _ in range(15):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_pert = perturb_structure(best_h)
        h_gb = gradient_based(h_pert)
        c5_gb = objective(h_gb)
        if c5_gb < best_c5:
            best_h = h_gb
            best_c5 = c5_gb

    # Normalize and clip
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2 / best_h.sum())

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = best_h * (n_points / 2 / best_h.sum())
        best_h = best_h.clip(0, 1)

    return best_h, best_c5, n_points
