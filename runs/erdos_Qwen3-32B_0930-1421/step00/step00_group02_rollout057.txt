import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 128  # Optimized number of points
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * 0.5 / h.mean()

    def normalize(h):
        s = np.sum(h)
        return h * (n_points / 2) / s

    h = normalize(h)
    h[h > 1] = 1.0
    h[h < 0] = 0.0

    best_c5 = float("inf")
    best_h = h.copy()

    # Gradient descent with adaptive learning rate
    def evaluate_c5(h):
        c = correlate(h, 1 - h, mode='full') * dx
        return np.max(c[n_points-1:2 * n_points - 1])

    def grad(w, h):
        corr = correlate(h, 1 - h, mode='full')
        stat_k = np.argwhere(corr[n_points-1:2*n_points-1] == best_c5)[0,0] + n_points - 1
        grad = correlate(2 * h - 1, h, mode='full')
        grad = grad * 0
        grad[stat_k - n_points + 1:stat_k + 1] = - (1 - 2 * h)
        grad = grad[n_points-1:2 * n_points - 1] * dx
        return grad

    def min_stochastic_prox(w, lr):
        h_t = w * 0.5 + 0.5
        grad = grad(w, h)

        def prox(u):
            return np.clip(u, 0, 1)

        h_upt = prox(h_t - lr * grad)
        chg = h_upt - h_t
        cdiff = np.sum(chg)
        if abs(cdiff) > 1e-4:
            scale = (n_points / 2 - np.sum(h_upt)) / (n_points / 2 - np.sum(h_t))
            if abs(scale) > 1e-2:
                h_upt += (h_t - (1 - scale) * (h_t - h_upt) / lr) * lr
                h_upt = prox(h_upt)

        w_upt = (h_upt - 0.5) * 2
        grad_upt = grad(w_upt, h_upt)
        return w_upt, grad_upt

    while time.time() - start_time < budget_s:
        h = normalize(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
        w = (best_h - 0.5) * 2
        for _ in range(5 if c5 > 0.8 else 3):
            w, grad_val = min_stochastic_prox(w, lr=0.1)
        h = (w * 0.5 + 0.5)
        h = np.clip(h, 0, 1)
        h = normalize(h)

    return (best_h, best_c5, n_points)
