def run(seed=42, budget_s=1100, **kwargs):

    # Optimization parameters
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    target_sum_dx = 1.0

    # Validate initial h
    if not (np.all((h0 >= 0) & (h0 <= 1))):
        raise ValueError("Initial h0 is out of bounds")
    if abs(np.sum(h0) - target_sum) > 1e-8:
        raise ValueError("Initial h0 has incorrect sum")

    h_best = h0.copy()
    c5_best = np.inf
    start_time = time.time()

    # Gradient descent loop
    while time.time() - start_time < budget_s:
        # Evaluate current C5
        corr = np.correlate(h_best, 1 - h_best, mode='full')
        corr = corr[n_points - 1: 2 * n_points - 1]
        c5_current = np.max(corr) * dx
        if c5_current < c5_best:
            c5_best = c5_current
            h_best = h_best.copy()

        # Compute lag with max overlap
        k_star = np.argwhere(corr == np.max(corr))[0, 0]

        # Construct gradient for this lag
        grad = np.zeros_like(h_best)
        for i in range(n_points):
            if 0 <= i - k_star < n_points:
                grad[i] += (1 - h_best[i - k_star])  # gradient w.r.t h[i], for h[i - k]
            if 0 <= i + k_star < n_points:
                grad[i] += -h_best[i + k_star]        # gradient w.r.t h[i], for h[i + k]

        # Normalize gradient
        grad = grad / max(np.linalg.norm(grad), 1e-8)

        # Apply update with adaptive step size
        step_size = 1.0
        h_update = h_best - step_size * grad

        # Projection onto [0, 1]^n
        h_update = np.clip(h_update, 0, 1)

        # Project to constraint: sum(h) = n_points / 2
        s = np.sum(h_update)
        scaling_factor = (target_sum / s)
        h_update = np.clip(h_update * scaling_factor, 0, 1)

        # Replace h_best only if it improved
        c5_update = np.max(np.correlate(h_update, 1 - h_update, mode='full')[n_points - 1: 2 * n_points - 1]) * dx
        if c5_update < c5_best:
            h_best = h_update.copy()

        # Reduce step size over time
        step_size *= 0.995

        # Early termination if no progress
        if time.time() - start_time > 0.9 * budget_s:
            break

    # Final eval of best solution
    corr_best = np.correlate(h_best, 1 - h_best, mode='full')
    c5_final = np.max(corr_best[n_points - 1: 2 * n_points - 1]) * dx

    c5_bound = c5_final

    return (h_best.tolist(), c5_bound, n_points)
