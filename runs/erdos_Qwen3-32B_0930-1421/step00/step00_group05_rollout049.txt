import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_c5(h, dx):
    h_projected = np.clip(h, 0.0, 1.0)
    ones_minus_h = 1.0 - h_projected
    corr = correlate(h_projected, ones_minus_h, mode='full')
    corr *= dx
    c5 = np.max(corr)
    return c5

def project_to_feasible(h, dx, target_sum=None):
    h_projected = np.clip(h, 0.0, 1.0)
    if target_sum is None:
        target_sum = (len(h) / 2.0)
    total = h_projected.sum()
    if abs(total) < 1e-10:
        return h_projected
    h_projected *= (target_sum / total)
    return h_projected

    best_h = h.copy()
    learning_rate = 0.01
    beta = 10
    max_iterations = 20000
    iteration = 0

    while time.time() - time_start < budget_s - 0.5:
        iteration += 1
        overlaps = np.zeros(n_points + 1)
        for k in range(1, n_points):
            l = k
            a = h[:n_points - l]
            b = 1 - h[l:]
            overlaps[k] = np.sum(a * b) * dx
        for k in range(n_points + 1):
            if k < n_points:
                l = n_points - k
                a = h[k:]
                b = 1 - h[:l]
                overlaps[k + n_points] = np.sum(a * b) * dx
        values = np.concatenate([overlaps, overlaps[n_points - 1:0:-1]])
        exp_vals = np.exp(beta * (values - values.max()))
        grad_approx = beta * exp_vals / np.sum(exp_vals)
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            l = k
            grad = grad_approx[k] * (-1 * h[l:] + (1 - h[:n_points - l]) * dx)
            gradient[l:] -= grad
            gradient[:n_points - l] += grad
        for k in range(1, n_points):
            l = n_points - k
            grad = grad_approx[k + n_points - 1] * (-1 * h[:l] + (1 - h[k:]) * dx)
            gradient[:l] -= grad
            gradient[k:] += grad
        gradient[n_points - 1:] -= grad_approx[0] * (1 - h[1:]) * dx
        gradient[:n_points - 1] += grad_approx[0] * h[1:] * dx
        gradient[0] -= grad_approx[0] * (1 - h[0]) * dx
        h -= learning_rate * gradient
        h = project_to_feasible(h, dx, target_sum)
        current_c5 = evaluate_erdos_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iteration % 50 == 0:
            learning_rate *= 0.965
        if iteration % 100 == 0:
            beta *= 1.02
        if learning_rate < 1e-6:
            break

    return (best_h.tolist(), best_c5, n_points)
