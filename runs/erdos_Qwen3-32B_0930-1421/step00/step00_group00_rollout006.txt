import numpy as np
import time
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Functions
    def evaluate_c5(h):
        h_norm = h / (np.sum(h) * dx) * target_sum * dx  # rescale in case of numerical drift
        full_corr = np.correlate(h_norm, 1 - h_norm, mode="full")
        full_corr = full_corr[int(n_points / 2):3 * int(n_points / 2)]
        return np.max(full_corr) * dx

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h_clipped * scale, 0, 1)

    def backtrack_step(h, grad, alpha, beta=0.5, max_backtracks=10):
        for i in range(max_backtracks):
            h_new = h - alpha * grad
            h_new = project(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                return h_new, c5_new
            alpha *= beta
        return h, evaluate_c5(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    c5_best_history = [best_c5]
    iter_time = time.time()

    while time.time() - t_start < budget_s:
        fh = h / (np.sum(h) * dx) * target_sum  # re-normalize if needed
        corr = np.correlate(fh, 1 - fh, mode="full")
        max_idx = np.nanargmax(corr[int(n_points / 2):3 * int(n_points / 2)]) + int(n_points / 2)
        k = max_idx - n_points + 1
        if not (0 <= k < n_points):
            break
        grad = (1 - np.roll(h, -k)) * dx
        h, c5 = backtrack_step(h, grad, alpha=1.0)

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        if time.time() - iter_time > 10:
            break

    # Ensure final constraint compliance
    best_h = project(best_h)
    final_c5 = evaluate_c5(best_h)
    return best_h.tolist(), min(final_c5, *c5_best_history), n_points
