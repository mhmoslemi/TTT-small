import numpy as np
import cvxpy as cp
from time import time


def run(seed=42, budget_s=1100):
    # Initial setup
    t0 = time()
    n_points = 45  # Start with initial construction
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) / n_points - 0.5) < 1e-6  # verify ∫h = 1

    def project_round(x):
        x_clipped = np.clip(x, 0, 1)
        x_scaled = x_clipped / np.sum(x_clipped) * (n_points // 2)
        return x_scaled

    def compute_c5(h):
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    eps = 1e-4
    remaining_time = budget_s

    while remaining_time > 0.1:
        # Compute current overlap
        conv = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(conv)
        overlap_k = conv[k_max] * dx

        # Formulate optimization
        h_var = cp.Variable(n_points)
        theta = cp.Variable(n_points)

        obj = h_var[k_max] * (1 - h_var[(k_max - n_points) % n_points]) * dx

        overlaps = [h_var[i] * (1 - h_var[(i - k_max) % n_points]) * dx
                    for i in range(n_points)]
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            theta >= obj,
            theta >= cp.sum(h_var) * dx / n_points  # minimum possible mean
        ]

        prob = cp.Problem(cp.Minimize(theta), constraints)
        prob.solve(solver=cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.SCIP)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_update = h_var.value
            h_update = project_round(h_update)
            c5_update = compute_c5(h_update)
            if c5_update < best_c5 - 1e-6:
                best_h = h_update.copy()
                best_c5 = c5_update
                h = project_round(h_update + np.random.normal(0, 0.01, n_points))
            else:
                h = h_update.copy()

        # Check time
        elapsed = time() - t0
        remaining_time = max(0, budget_s - elapsed)

    return best_h, best_c5, n_points
