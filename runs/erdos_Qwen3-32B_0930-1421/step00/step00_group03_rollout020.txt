import numpy as np
import time
from numpy import correlate
from scipy.optimize import minimize


def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    full_corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(full_corr * dx)
    return c5


def project_to_constraints(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    original_sum = h.sum()
    if abs(original_sum - target_sum) < 1e-10:
        return h
    diff = original_sum - target_sum
    if diff > 0:
        indices = np.argsort(-h)
    else:
        indices = np.argsort(h)
    idx = 0
    while abs(diff) > 1e-10 and idx < n_points:
        h[indices[idx]] = h[indices[idx]] - diff / (n_points - idx)
        diff = h.sum() - target_sum
        idx += 1
    h = np.clip(h, 0, 1)
    return h


def sa_run(h, n_points, max_steps=1000, init_temp=1.0, cooling_factor=0.99, step_size=0.1):
    t = init_temp
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, n_points)

    for step in range(max_steps):
        try_h = h.copy() + step_size * np.random.randn(n_points)
        try_h = project_to_constraints(try_h, n_points)
        trial_c5 = evaluate_c5(try_h, n_points)

        if trial_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - trial_c5) / t):
            h[:] = try_h
            best_c5 = min(best_c5, trial_c5)

        if np.isclose(best_c5, 0):
            break

        t *= cooling_factor

    return h, best_c5


def gd_run(h, n_points, max_steps=500, learning_rate=0.01, tol=1e-6):
    h = project_to_constraints(h, n_points).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, n_points)

    for step in range(max_steps):
        h_grad = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            idx = (k + h.shape[0]) % h.shape[0]
            c = correlate(h, 1 - h, mode='full')
            max_k = np.argmax(c)
            for i in range(h.shape[0]):
                if i + max_k < h.shape[0]:
                    h_grad[i] += (1 - h[(i + max_k) % h.shape[0]]) - h[i]
        h_new = h[:] - learning_rate * h_grad
        h_new = project_to_constraints(h_new, n_points)

        new_c5 = evaluate_c5(h_new, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h_new
            h[:] = h_new
        else:
            # reduce learning rate on plateau
            learning_rate *= 0.9

    return best_h, best_c5
