import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def normalize(h, n_points):
        desired_sum = n_points / 2
        current_sum = h.sum()
        h = h * (desired_sum / current_sum)
        h = np.clip(h, 0, 1)
        return h

    h = normalize(h, n_points)

    best_h = h.copy()
    best_c5 = np.inf

    # Configuration parameters
    lrate = 0.1
    niters = 0
    max_niters = 100000
    patience = 0
    max_patience = 50

    while True:
        niters += 1
        if time.time() - start_time >= budget_s:
            break
        if niters >= max_niters:
            break

        # Compute overlaps using convolution/correlation
        cor = correlate(h, 1 - h, mode='full')
        cor = cor * dx
        k_range = np.arange(len(cor))
        mask = (k_range >= n_points - 1) & (k_range <= 2 * n_points - 2)
        cor_k = cor[mask]
        # Use only valid indices
        idx = np.arange(n_points + 1)
        cor_k = cor[idx + n_points - 1]

        c5 = cor_k.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break

        # Identify shift to improve
        k_max = np.argmax(cor_k)
        lag = k_max - (n_points - 1)

        if lag < 0:
            overlap_mask = np.arange(n_points + lag, n_points)
            other_mask = np.arange(-lag, 0)
        else:
            overlap_mask = np.arange(0, n_points - lag)
            other_mask = np.arange(lag, n_points)

        # Gradient for maximizing contribution to the max C_k
        grad = np.zeros(n_points)
        grad[overlap_mask] = 1 - 2 * h[overlap_mask]
        grad[other_mask] = -1 + 2 * h[other_mask]
        grad = -grad # For minimizing the max overlap

        # Update step
        h = h + lrate * grad
        h = normalize(h, n_points)

    return best_h, best_c5, n_points
