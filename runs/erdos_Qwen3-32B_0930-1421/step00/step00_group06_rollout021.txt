import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    # Use the provided initial_h_values, already of size n=95
    h_init = np.array(initial_h_values)
    n_points = h_init.size
    dx = 2.0 / n_points

    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1.0,
        h >= 0.0,
    ]
    overlaps = [cp.sum(h * (1 - cp.roll(h, -k))) for k in range(1, 5)]
    overlaps += [cp.sum(h * (1 - cp.roll(h, k))) for k in range(1, 5)]
    C5 = cp.Max(overlaps)
    objective = cp.Minimize(C5 * dx)

    problem = cp.Problem(objective, constraints)

    best_c5 = None
    best_h = h_init.copy()

    def optimize_and_update():
        nonlocal best_c5, best_h
        try:
            problem.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.SCS,
                          verbose=False,
                          max_iters=1000,
                          eps=1e-5)
            h_sol = h.value
            if h_sol is not None and np.all(np.isfinite(h_sol)):
                h_sol = np.clip(h_sol, 0, 1)
                h_sol /= np.sum(h_sol) / (n_points / 2) + 1e-12
                sol_c5 = evaluate_erdos_solution(h_sol, n_points)
                if best_c5 is None or sol_c5 < best_c5:
                    best_c5 = sol_c5
                    best_h = h_sol.copy()
        except Exception:
            pass

    # First optimization pass with solver
    optimize_and_update()

    # Second pass with warm start
    h_init_value = cp.Parameter(n_points)
    h_init_value.value = best_h
    h = cp.Variable(n_points, nonneg=True)
    problem = cp.Problem(objective, [
        h == h_init_value - cp.Variable(n_points),
        cp.sum(h) == n_points / 2,
        h <= 1.0,
        h >= 0.0
    ])
    optimize_and_update()

    # Run until time budget is exhausted
    while time.time() - start_time < budget_s - 20:
        try:
            perm = np.random.permutation(n_points)
            v = np.zeros(n_points)
            for i in range(n_points):
                c = 1 - 2 * best_h[perm[i]]
                v[perm[i]] = (0.5 if c < 0 else best_h[perm[i]] + (0.1 if c > 0 else -0.1))
            v = np.clip(v, 0, 1)
            v /= np.sum(v) / (n_points / 2) + 1e-12
            c5 = evaluate_erdos_solution(v, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h[:] = v[:]
        except Exception:
            pass

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h) / (n_points / 2) + 1e-12
    return best_h.tolist(), best_c5, n_points
