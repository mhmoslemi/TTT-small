```python
import numpy as np
import time
from math import exp
import cvxpy as cp
from numpy.random import default_rng

def evaluate_overlap(h, dx):
    return max(np.correlate(h, 1 - h, mode="full") * dx)

def run(seed=42, budget_s=1100):
    global initial_h_values
    rng = default_rng(seed)

    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 80

    # Convert to binary, normalized
    if 'initial_h_values' in globals():
        h = (initial_h_values > 0.5).astype(int)
        errors = sum(h) - n_points / 2
        if errors < 0:
            idx = rng.choice(np.where(h == 0)[0], abs(int(errors)), replace=False)
            h[idx] = 1
        elif errors > 0:
            idx = rng.choice(np.where(h == 1)[0], abs(int(errors)), replace=False)
            h[idx] = 0
    else:
        n_on = n_points // 2
        idx = rng.choice(n_points, n_on, replace=False)
        h = np.zeros(n_points, dtype=int)
        h[idx] = 1

    h = h.astype(int)
    dx = 2.0 / n_points
    T = 1.0  # Initial temperature
    alpha = 0.99  # Cooling rate
    best_h = h.copy()
    current_score = evaluate_overlap(h, dx)
    best_score = current_score
    start_time = time.time()
    no_improvement = 0
    patience = 1000  # Consecutive steps with no improvement
    min_temp = 1e-5

    while time.time() - start_time < budget_s:
        if no_improvement > patience:
            T = max(T * 0.99, min_temp)
            alpha *= 0.995

        # Proposal: flip two bits to maintain area constraint
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        h_prime = h.copy()
        h_prime[ones[i]] = 0
        h_prime[zeros[j]] = 1
        h_prime = h_prime.copy()

        # Evaluate
        score_prime = evaluate_overlap(h_prime, dx)
        delta = score_prime - current_score

        if delta < 0:
            h = h_prime
            current_score = score_prime
            if score_prime < best_score:
                best_h = h_prime
                best_score = score_prime
                no_improvement = 0
                T = max(T * alpha, min_temp)  # Cool only for real improvement
            else:
                no_improvement += 1
        else:
            if rng.random() < exp(-delta / T):
                h = h_prime
                current_score = score_prime
            no_improvement += 1
        T *= alpha

    best_h = best_h.astype(float)
    return best_h, best_score, n_points
```
