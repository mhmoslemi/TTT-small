import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    initial_alpha = 0.5
    decay_factor = 0.98
    max_iterations = 1000
    improvement_threshold = 1e-4
    recompute_interval = 50

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    total_mass = n_points / 2

    h = np.copy(initial_h_values)
    h_prev = np.copy(h)
    h_shifted = np.copy(h)
    alpha = initial_alpha

    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        h_shifted = alpha * h + (1 - alpha) * h_prev
        overlap = np.correlate(h_shifted, 1 - h, mode='full') * dx
        k_max = np.argmax(overlap)

        # CVXPY convex optimization
        var_h = cp.Variable(n_points)
        objective = cp.Minimize(h_shifted[k_max] * var_h[-k_max] - h_shifted[k_max] * var_h[-k_max])
        objective += cp.Maximize(-overlap[k_max])

        constraints = [
            0 <= var_h,
            var_h <= 1,
            cp.sum(var_h) == total_mass,
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.OSQP, reltol=1e-6)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_prev = np.copy(h)
            alpha *= decay_factor
            continue

        h_new = var_h.value
        h_new = np.clip(h_new, 0, 1)
        h = h_new
        h_prev = np.copy(h)

        current_c5 = compute_c5(h, dx)
        if best_c5 - current_c5 > improvement_threshold:
            best_c5 = current_c5
            alpha = max(alpha * decay_factor, 0.1)
        if iter % recompute_interval == 0:
            h_shifted = np.copy(h)

    # Validate the final solution
    h = np.clip(h, 0, 1)
    h = h / np.mean(h) * (n_points / 2)
    c5_bound = compute_c5(h, dx)
    return (h, c5_bound, n_points)


def compute_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlap)
