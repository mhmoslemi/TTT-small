import numpy as np
import cvxpy as cp
import math
import time
from evaluate_erdos_solution import evaluate_erdos_solution
from math import log

def run(seed=42, budget_s=1100):
    # Time tracking
    start_time = time.time()

    # Parameters
    n_points = 95  # Use initial_h_values size
    alpha = 0.5  # Initial convex combination parameter of frozen state
    max_iterations = 2000
    improvement_threshold = 1e-4
    reanneal_interval = 50
    h_prev = np.copy(initial_h_values)  # Start with given initial h

    dx = 2.0 / n_points

    # Precompute sum constraint: sum(h) must be n_points / 2
    target_sum = n_points / 2

    # Utility function to compute overlap integral
    def compute_c5(h):
        h = np.array(h).flatten()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize best solution
    best_c5 = float('inf')
    best_h = np.copy(h_prev)

    # Optimization loop
    for iteration in range(max_iterations):
        # Time check
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute current h_shifted (convex combination)
        h_shifted = alpha * h_prev + (1 - alpha) * h_prev

        # Compute overlap at all lags
        overlap = np.correlate(h_shifted, 1 - h_prev, mode='full') * dx
        k_max = np.argmax(overlap)

        # Construct overlap operator for lag k_max
        def overlap_for_h(h):
            overlap_k = (np.sum(
                h * (1 - np.roll(h, k_max)) if k_max < n_points else h * (1 - np.concatenate([h[k_max:], h[:k_max - 1]]))
            )) * dx
            return overlap_k

        # Define variables and problem for this k
        h_var = cp.Variable(n_points)
        problem = cp.Problem(
            cp.Minimize(overlap_for_h(h_var)),
            [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]
        )

        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except:
            problem.solve(solver=cp.GLPK, verbose=False)

        # Update h_prev
        h_prev = np.clip(h_var.value, 0, 1)

        # Evaluate new c5
        new_c5 = compute_c5(h_prev)

        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = np.copy(h_prev)
            alpha = max(alpha * 0.9, 0.1)  # Small improvement, slowly contract the surrogate
        else:
            alpha = max(alpha * 1.05, 0.5)  # No improvement, expand exploration

        # Reanneal every reanneal_interval
        if iteration % reanneal_interval == 0:
            h_prev = np.random.rand(n_points) * 0.5 + 0.25
            h_prev = np.clip(h_prev, 0, 1)
            h_prev = h_prev / np.mean(h_prev) * (n_points / 2)

    # Final checks and return
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values / np.mean(h_values) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)

    final_c5 = compute_c5(h_values)
    return (h_values, final_c5, n_points)
