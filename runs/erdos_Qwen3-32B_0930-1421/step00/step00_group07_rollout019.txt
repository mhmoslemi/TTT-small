import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    corrs = correlate(h, 1 - h, mode="full") * dx
    return np.max(corrs)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = initial_h_values.copy()

    # Normalize initial h to have sum equal to n_points / 2
    h /= np.sum(h)
    h *= n_points / 2
    h = np.clip(h, 0, 1)

    alpha = 0.01
    max_iter = 10000
    patience = 200
    best_c5 = np.inf
    best_h = h.copy()
    best_iter = 0

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            print(f"Time budget expired after {iteration} iterations.")
            break

        # Compute cross-correlation values and find maximum index
        corrs = correlate(h, 1 - h, mode="full")
        k = int(np.argmax(corrs))

        # Construct the gradient vector based on the maximum overlapping offset
        grad_h = np.zeros(n_points)
        if k <= n_points - 1:
            # handle left padding in full correlation
            overlap = h * (1 - h[k:])
            for i in range(len(overlap)):
                grad_h[i] += overlap[i]
        if k >= n_points:
            # handle right padding in full correlation
            overlap = h[k - n_points:] * (1 - h[:n_points - (k - n_points)])
            for i in range(len(overlap)):
                grad_h[i + (k - n_points)] += overlap[i]

        # Update h
        h_proposed = h - alpha * grad_h
        # Project onto [0,1]^n
        h_proposed = np.clip(h_proposed, 0, 1)
        # Project onto the integral constraint: sum(h) == n_points/2
        scaling_factor = (n_points / 2) / np.sum(h_proposed)
        h_proposed *= scaling_factor
        h = h_proposed

        # Track best h
        c5 = evaluate_C5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_iter = iteration
            alpha = 0.01  # Reset learning rate on improvement
        elif iteration - best_iter >= patience:
            alpha *= 0.5  # Reduce learning rate if no improvement
            if alpha < 1e-6:
                break

    # Final validation of constraints
    h = best_h
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2):
        h *= (n_points / 2) / h_sum
    c5_final = evaluate_C5(h, n_points)

    return h, c5_final, n_points
