import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
    except NameError:
        n_points = 95
        dx = 2.0 / n_points
        h = np.full(n_points, n_points / 2.0 / n_points)
        # Perturb slightly to break symmetry
        h += np.random.rand(n_points) * 0.05 - 0.025
        h = np.clip(h, 0, 1)

    # Normalize to ensure ∫h(x)dx = 1
    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0, 1)

    h_prev = h.copy()
    alpha = 0.5
    max_iter = 1000
    iter_gap = 50
    last_improvement = 0.0
    best_c5 = 1.0
    best_h = h.copy()

    for it in range(max_iter):
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Frozen surrogate approximation
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Compute overlap for all k
        corr = np.correlate(h_shifted, 1 - h, mode="full") * dx
        overlap_all = corr[n_points:2 * n_points]  # Only consider k in 1 <= k < n_points

        k_max = np.argmax(overlap_all)
        c5 = overlap_all[k_max]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = elapsed

        # Formulate the convex problem at k_max
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var <= 1, h_var.sum() == n_points / 2]
        overlap_k = np.dot(h_shifted[n_points - k_max:], h_var[:k_max])
        overlap_k += np.dot(h_shifted[:n_points - k_max], h_var[k_max:])
        obj = cp.Minimize(overlap_k)
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.GLPK, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_var.value
            h = np.clip(h, 0, 1)

        h_prev = h.copy()

        # Decay alpha if no improvement
        if it % iter_gap == 0 and (elapsed - last_improvement) > 2 * iter_gap:
            alpha = max(0.1, alpha * 0.9)

    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    overlap_all = corr[n_points:2 * n_points]
    c5 = overlap_all[np.argmax(overlap_all)]

    return h.tolist(), c5, n_points
