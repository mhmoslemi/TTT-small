import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate


def evaluate_overlap(h, dx):
    one_minus_h = 1.0 - h
    correlations = correlate(h, one_minus_h, mode='full')
    max_overlap = np.max(correlations * dx)
    return max_overlap


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = h_init.size
    dx = 2.0 / n_points

    best_h = h_init.copy()
    best_c5 = evaluate_overlap(best_h, dx)

    lower = 0.38
    upper = 0.4
    tol = 1e-5
    total_time = 0
    max_iter = 50

    for step in range(max_iter):
        t0 = time.time()
        mid = (lower + upper) * 0.5
        h = cp.Variable(n_points, nonneg=True)

        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1,
        ]

        integrals = []
        for k in range(-n_points + 1, n_points):
            if k < 0:
                offset = -k
                integrals.append(cp.sum(cp.multiply(h[offset:], one_minus_h[:n_points - offset])) * dx - mid <= 0)
            elif k >= 0:
                offset = k
                integrals.append(cp.sum(cp.multiply(h[:n_points - offset], one_minus_h[offset:])) * dx - mid <= 0)
        constraints += integrals

        # Objective is feasibility
        objective = cp.Minimize(cp.Constant(0))
        prob = cp.Problem(objective, constraints)

        try:
            result = prob.solve(solver="ECOS", max_iters=150, verbose=False)
            new_h = np.clip(np.round(h.value, decimals=8), 0, 1)
            new_h = new_h / np.sum(new_h) * n_points / 2
            new_c5 = evaluate_overlap(new_h, dx)
            if prob.status in ("optimal", "optimal_inaccurate"):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    upper = mid
                else:
                    lower = mid
        except Exception:
            lower = mid

        total_time += time.time() - t0
        if total_time > budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
