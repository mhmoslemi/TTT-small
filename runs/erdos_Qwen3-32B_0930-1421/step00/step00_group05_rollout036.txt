import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    Tstart = time.time()
    
    # Extract initial solution and determine n_points
    h0 = cp.Parameter(shape=initial_h_values.shape, value=initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Define step function as optimization variable
    h = cp.Variable(n_points, nonneg=True)
    
    # Define constraints
    constraints = [
        h <= 1,
        cp.sum(h) == n_points / 2,  # ∫ h(x) dx = 1
    ]
    
    # Define correlation-like (overlap) cost for all k
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = cp.Max(corr)

    # Define objective
    objective = cp.Minimize(max_corr * dx)
    
    # Create problem and solve
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s - 20)  # Reserve time

    # Extract best solution
    h_values = h.value if h.value is not None else h0.value

    # Final validation
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode="full")) * dx
    assert np.allclose(np.sum(h_values), n_points / 2), "Integrability constraint violated"
    assert np.all((h_values >= 0) & (h_values <= 1)), "Range constraint violated"
    
    return h_values, c5_bound, n_points
