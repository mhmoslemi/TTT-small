import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    h = h_initial.copy().astype(np.float64)
    best_h = h.copy()
    alpha = 0.1
    beta = 0.7

    best_c5 = float('inf')
    start_time = time.time()

    def compute_c5(_h):
        full_corr = np.correlate(_h, 1 - _h, mode='full')
        c5 = np.max(full_corr) * dx
        return c5

    def project_h(_h):
        # Clip to [0, 1]
        _h = np.clip(_h, 0.0, 1.0)
        # Maintain tight total mass by distributing surplus/deficit
        total_mass = _h.sum()
        target_mass = n_points / 2.0
        delta_mass = target_mass - total_mass
        # Adjust by adding/subtracting to non-saturated regions
        if abs(delta_mass) > 1e-8:
            if delta_mass > 0:
                eligible = _h < 1.0
                if np.sum(eligible) > 0:
                    _h[eligible] += delta_mass / np.sum(eligible)
                else:
                    _h += delta_mass / n_points
            else:
                eligible = _h > 0.0
                if np.sum(eligible) > 0:
                    _h[eligible] += delta_mass / np.sum(eligible)
                else:
                    _h += delta_mass / n_points
        return _h

    # Initial value
    best_c5 = compute_c5(h)

    it = 0
    MAX_ITERS = 200

    while time.time() - start_time < budget_s and it < MAX_ITERS:
        it += 1

        # Step 1: Compute full cross-correlation and find worst shift
        full_corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(full_corr)

        # Step 2: Compute gradient
        if k_max < n_points:
            gradient = 1 - h[n_points - k_max - 1::]
            gradient = np.concatenate((gradient, np.zeros(k_max)))
        else:
            gradient = 1 - h[:n_points - (k_max - n_points):]
            gradient = np.concatenate((np.zeros(k_max - n_points), gradient))

        gradient = gradient * dx

        # Step 3: Apply gradient update and project back
        h_proposed = h - alpha * gradient
        h_proposed = project_h(h_proposed)

        # Step 4: Check if this improved us
        c5_proposed = compute_c5(h_proposed)
        if c5_proposed < best_c5:
            h = h_proposed
            best_c5 = c5_proposed
            best_h = h_proposed.copy()
            alpha = 0.1  # Reset step size when progress is made
        else:
            alpha *= beta  # Reduce step size

    return (best_h.tolist(), best_c5, n_points)
