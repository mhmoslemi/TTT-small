import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Pre-imported global from the harness
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values).copy().astype(np.float64)

    # Optimization settings
    time_budget = budget_s
    iter_budget = 100  # heuristic for progress check

    # Variables
    h = cp.Variable(n_points, nonneg=True)

    # Define objective and constraints
    def compute_obj(h_vals):
        overlaps = []
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(1 - h_vals, k)
            o = np.sum(h_vals * shifted * dx)
            overlaps.append(o)
        return np.max(overlaps)

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Objective
    objective = cp.Minimize(cp.Max([
        cp.sum([h[i] * (1 - h[cp.mod((i + k), n_points)]) * dx for i in range(n_points)])
        for k in range(-n_points + 1, n_points)
    ]))

    # Build and solve the problem
    problem = cp.Problem(objective, constraints)
    
    # Warm start
    h.value = h0

    # Solve within time limit
    try:
        problem.solve(solver=cp.GUROBI, verbose=False, timeLimit=budget_s)
    except:
        pass

    h_values = np.clip(h.value, 0, 1)
    h_values /= np.sum(h_values)  # normalize in case constraint violation
    h_values *= n_points / 2

    # Final evaluation
    overlaps = []
    ref = h_values.astype(np.float64)
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(1 - ref, k)
        o = np.sum(ref * shifted * dx)
        overlaps.append(o)
    c5_bound = np.max(overlaps)
    return (h_values, c5_bound, n_points)
