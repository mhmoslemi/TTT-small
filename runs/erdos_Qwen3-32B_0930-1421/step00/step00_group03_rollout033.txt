import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70  # Keep low for efficiency
    dx = 2.0 / n_points
    np.random.seed(seed)

    # Initialize h based on provided starting point or fallback to uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 1  # Half ones, half zeros as fallback

    # Normalize h to ensure constraint sum(h) = n_points / 2
    h /= h.sum() * 2

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    t0 = time.time()
    iter_count = 0
    max_iter = 1000
    lr = 0.05  # Starting learning rate
    gamma = 0.99  # Learning rate decay
    no_improvement_count = 0
    max_no_improvement = 50

    while time.time() - t0 < budget_s and iter_count < max_iter:
        # Compute gradient of the max overlap
        dh = np.zeros_like(h)
        c5, max_k = compute_c5_and_max_k(h, n_points, dx)
        for i in range(n_points):
            # Save original value for backtracking
            h_i = h[i]
            h[i] = min(max(h[i] + 1e-5, 0.0), 1.0)
            c5_perturbed, _ = compute_c5_and_max_k(h, n_points, dx)
            dh[i] = (c5_perturbed - c5) / 1e-5
            h[i] = h_i  # Restore

        # Normalize gradient and apply learning rate
        dh = dh / (1e-8 + np.max(np.abs(dh)))
        h -= lr * dh

        # Clip to bounds [0,1] and rescale to maintain sum constraint
        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2) / h.sum()

        # Compute new C₅
        new_c5 = compute_c5(h, n_points, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Decay learning rate and enforce termination
        lr *= gamma
        iter_count += 1
        if no_improvement_count >= max_no_improvement:
            break

    return best_h.tolist(), best_c5, n_points

def compute_c5_and_max_k(h, n_points, dx):
    # Compute all overlaps with correlate
    corr = correlate(h, 1 - h, mode='full')
    all_c5 = corr[n_points - 1:] * dx  # Only shifts k >= 0 are valid
    c5 = all_c5.max()
    max_k_idx = np.argmax(all_c5)
    return c5, max_k_idx

def compute_c5(h, n_points, dx):
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr[n_points - 1:]) * dx
    return c5
