import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def evaluate_candidate(h_values, dx):
    h = np.array(h_values, dtype=np.float64)
    result = evaluate_erdos_solution(h)
    if len(result) != 2:
        return np.inf
    return result[0]

def mutate(h, dx, pm=0.1, adjustment_rate=0.5):
    n = len(h)
    h = np.copy(h)
    indices = np.random.choice(n, size=int(pm * n), replace=False)

    delta = np.zeros(n)
    for i in indices:
        h[i] = 1 - h[i]
        delta[i] = (-1 if h[i] == 0 else 1)

    over = np.argmax(np.cumsum(np.maximum(delta, 0)))
    under = np.argmin(np.cumsum(np.minimum(delta, 0)))
    while not np.isclose(np.sum(h), n / 2):
        excess = np.sum(h) - n / 2
        if excess > 0:
            change = np.clip(-adjustment_rate * np.abs(excess), -1, 1)
            for i in indices:
                if excess > 0 and h[i] == 1:
                    h[i] = max(h[i] + change, 0)
                    break
        else:
            change = np.clip(-adjustment_rate * np.abs(excess), -1, 1)
            for i in indices:
                if excess < 0 and h[i] == 0:
                    h[i] = min(h[i] - change, 1)
                    break
    return h

def tournament_selection(population, fitnesses, k=2, t_size=3):
    indices = np.random.choice(len(population), size=t_size, replace=False)
    best_idx = indices[np.argmin(np.take(fitnesses, indices))]
    return population[best_idx], fitnesses[best_idx]

def crossover(parent1, parent2):
    n = len(parent1)
    idx1, idx2 = sorted(np.random.choice(n, 2, replace=False))
    child = np.copy(parent1)
    child[idx1:idx2] = parent2[idx1:idx2]
    child = np.clip(child, 0.0, 1.0)
    child = child.astype(np.float64)
    child *= (len(child) / 2) / np.sum(child)
    return child

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    h_init /= 2.0 / n_points  # Normalize to [0,1]
    dx = 2.0 / n_points

    pop_size = 40
    generations = 2000
    t_size = 3
    pm = 0.1
    pop = [h_init.copy()]

    def generate_random():
        h = np.random.rand(n_points)
        h = h.astype(np.float64)
        h *= (n_points / 2) / np.sum(h)
        return h

    # Initialize population
    for _ in range(pop_size - 1):
        pop.append(generate_random())

    fitnesses = [evaluate_candidate(h, dx) for h in pop]
    best_h = pop[np.argmin(fitnesses)]
    best_c5 = min(fitnesses)
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s - 19:
            break
        next_pop = [best_h.copy()]

        for _ in range(pop_size - 1):
            p1, f1 = tournament_selection(pop, fitnesses, t_size=t_size)
            p2, f2 = tournament_selection(pop, fitnesses, t_size=t_size)
            child = crossover(p1, p2)
            child = mutate(child, dx, pm=pm)
            child = np.clip(child, 0.0, 1.0)
            child = child.astype(np.float64)
            child *= (n_points / 2) / np.sum(child)
            next_pop.append(child)

        pop = next_pop
        fitnesses = [evaluate_candidate(h, dx) for h in pop]
        current_best_idx = np.argmin(fitnesses)
        current_best = pop[current_best_idx]
        current_c5 = fitnesses[current_best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best

    best_h *= 2.0 / n_points
    return (best_h.tolist(), best_c5, n_points)
