import numpy as np
import time
import random
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    n_points = 80  # initial h_values is for 80 points

    # Use the provided initial_h_values and convert to binary
    h = initial_h_values.astype(float)
    target_sum = n_points / 2

    # Threshold to binary and normalize to ensure the sum constraint
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h_sum = h.sum()
    while h_sum != target_sum:
        if h_sum < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0
        h_sum = h.sum()

    h_best = h.copy()
    c5_best = evaluate_erdos_solution.evaluate_erdos_solution(h_best)

    # Simulated annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    patience = 1000
    iteration = 0
    last_improvement = 0

    def evaluate_overlap(h_vec):
        co = np.correlate(h_vec, 1 - h_vec, mode="full")
        dx = 2.0 / len(h_vec)
        return max(co) * dx

    while time.time() - start_time < budget_s - 1.0:
        # Propose new candidate (flip a 1 and a 0)
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i1 = random.choice(ones_idx)
        i0 = random.choice(zeros_idx)
        h_candidate = h.copy()
        h_candidate[i1] = 0.0
        h_candidate[i0] = 1.0

        c5_candidate = evaluate_overlap(h_candidate)
        c5_current = evaluate_overlap(h)

        # Acceptance condition with simulated annealing
        if c5_candidate < c5_best:
            h_best = h_candidate.copy()
            c5_best = c5_candidate
            last_improvement = iteration
        elif c5_candidate < c5_current:
            h = h_candidate.copy()
        else:
            delta = c5_candidate - c5_current
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()

        T = max(T * alpha, min_T)
        iteration += 1

        # Safety reset if no improvement for 1000 iterations
        if iteration - last_improvement > patience:
            T = max(T * 10, min_T)
            alpha = max(alpha * 0.9, 0.95)
            last_improvement = iteration

    # Final validation
    assert h_best.sum() == n_points / 2, "Sum constraint violated!"
    c5_final = evaluate_erdos_solution.evaluate_erdos_solution(h_best)

    return (h_best.tolist(), c5_final, n_points)
