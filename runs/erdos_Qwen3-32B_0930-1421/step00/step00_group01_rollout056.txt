import numpy as np
import scipy.optimize as opt
import time
import evaluate_erdos_solution

def objective(h, n_points, dx):
    h = h.reshape((n_points,))
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2

def bounds_and_box(h, n_points):
    return [(0, 1)] * n_points

def opt_func(h_flat, n_points, dx):
    return objective(h_flat, n_points, dx)

def setup_optimization(h0, n_points, dx):
    constr = [{
        'type': 'eq',
        'fun': lambda h: constraint_sum(h, n_points)
    }]
    bounds = bounds_and_box(h0, n_points)
    return constr, bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h0 = initial_h_values.copy()
    n_points = h0.size
    dx = 2.0 / n_points

    # Set up optimization
    constr, bounds = setup_optimization(h0, n_points, dx)
    result = None

    def callback(xk, state):
        nonlocal result
        result = np.max(np.correlate(xk.reshape(-1), 1 - xk.reshape(-1), mode='full') * dx)
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            raise opt.OptimizeResult('Budget exhausted', result)

    try:
        res = opt.minimize(
            lambda h: objective(h, n_points, dx),
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constr,
            callback=callback,
            options={'ftol': 1e-8, 'maxiter': 10000}
        )
        h_opt = res.x.reshape(-1)
    except:
        h_opt = np.clip(h0, 0, 1)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)

    # Ensure h_opt is within bounds and satisfies constraints
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)

    # Final evaluation
    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        print('Budget exhausted before completion.')

    return h_opt, c5_bound, n_points
