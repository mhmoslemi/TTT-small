import numpy as np
import scipy.optimize as opt
import time

# Global variables for tracking and timing
start_time = time.time()

def evaluate_c5(h, dx):
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n - 1)])
    one_minus_h = 1 - h_extended
    corr = np.correlate(h_extended, one_minus_h, mode="full")
    c5 = np.max(np.abs(corr[n - 1:2 * n - 1 + 1]) * dx)
    return c5

def objective(x, n_points, dx):
    return evaluate_c5(x.reshape(n_points), dx)

def constraint_sum(x, n_points):
    return np.sum(x) - (n_points / 2)

def constraint_upper(x):
    return 1 - x

def constraint_lower(x):
    return x

def optimize_h(n, dx, x0):
    cons = (
        {'type': 'eq', 'fun': lambda x: constraint_sum(x, n)},
        {'type': 'ineq', 'fun': constraint_upper},
        {'type': 'ineq', 'fun': constraint_lower}
    )
    result = opt.minimize(
        lambda x: objective(x, n, dx),
        x0, method='SLSQP',
        constraints=cons,
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': 1500}
    )
    return result.x, result.fun
