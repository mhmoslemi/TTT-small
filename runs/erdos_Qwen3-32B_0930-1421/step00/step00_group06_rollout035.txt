import numpy as np
import cvxpy as cp
import math
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()

    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    prev_best_c5 = best_c5 + 2 * abs(best_c5)

    learning_rate = 0.01
    max_iterations = 1_000
    lr_decay = 0.95

    for i in range(max_iterations):
        t = time.time()
        if t - start_time > budget_s:
            break

        # Step 1: Evaluate overlap for all k and find best k
        h1h = 1 - h
        overlap = np.correlate(h, h1h, mode='full')
        overlap *= dx
        k = (np.argmax(overlap) - n_points + 1) % n_points

        # Step 2: Compute gradient for current k
        idxs = np.arange(n_points)
        lag_k = k
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = h1h[(i + lag_k) % n_points] - h[(i - lag_k) % n_points]

        # Step 3: Update h with gradient descent and project to feasible region
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

        # Step 4: Normalize to maintain constraint ∫h dx = 1
        h_sum = h.sum()
        h *= n_points / (2.0 * h_sum)

        # Step 5: Compute C₅ for current h
        c5 = evaluate_erdos_solution(h)

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            prev_best_c5 = best_c5 + 1e-6  # Reset threshold
        else:
            # Decay learning rate if not improving enough
            if abs(prev_best_c5 - best_c5) < 1e-6:
                learning_rate *= lr_decay
            prev_best_c5 = best_c5

    return (best_h.tolist(), best_c5, n_points)
