import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    n_points = 70  # Discretization level for h
    dx = 2.0 / n_points

    # Basis function construction
    def make_basis(M_basis, n_points):
        x_vals = np.linspace(0, 2, n_points)
        centers = np.linspace(0, 2, M_basis)
        width = 2 / (M_basis / 2.0)
        basis_functions = np.zeros((M_basis, n_points))
        for j in range(M_basis):
            basis_functions[j] = np.maximum(0, 1 - np.abs(x_vals - centers[j]) / width)
        return basis_functions / basis_functions.max(axis=1, keepdims=True)

    # Initial guess conversion
    def h_to_basis_weights(h_samples, basis_functions):
        h_avg = np.mean(h_samples)
        W = basis_functions
        _, _, V = np.linalg.svd(W.T @ W)
        return V[0, :] * (4 * h_avg)

    def compute_c5(h_vals, dx):
        h = h_vals
        one_minus_h = 1 - h
        max_c5 = 0.0
        for k in range(1, 6):
            shifted_h = np.roll(h, -k)
            shifted_h[-k:] = 0
            shifted_one_minus_h = 1 - shifted_h
            tmp = np.sum(h * shifted_one_minus_h)
            max_c5 = max(max_c5, tmp * dx)
        return max_c5

    def construct_h_from_basis(W_basis, w_opt):
        return (w_opt @ W_basis).clip(0, 1)

    M_basis_values = [80, 100]  # Evaluate for different basis sizes
    best_solution = None
    best_c5 = float("inf")

    for M_basis in M_basis_values:
        print(f"Processing M_basis = {M_basis} (elapsed: {time.time() - start_time:.1f}s)")
        if time.time() - start_time > budget_s - 60:
            break

        W_basis = make_basis(M_basis, n_points)
        n_basis = W_basis.shape[0]

        # Initialize with the current solution's h
        h_init = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
        h_init[h_init < 0] = 0
        h_init[h_init > 1] = 1
        w0 = h_to_basis_weights(h_init, W_basis)

        w = cp.Variable(n_basis, nonneg=True)
        t = cp.Variable()

        constraints = [
            cp.sum(w) == 1,
            w >= 0,
        ]

        temp_c5_max = 0.0
        for k in range(1, 6):
            # Construct non-convex term: w^T A_k w
            A_k = np.outer(W_basis, 1 - np.roll(W_basis, -k))
            A_k = np.triu(A_k)
            temp_c5 = cp.trace(A_k @ cp.diag(w) @ cp.diag(w))
            constraints.append(temp_c5 <= t)
            temp_c5_max += 1e-3  # heuristic to aid convergence

        objective = cp.Minimize(t)
        problem = cp.Problem(objective, constraints)

        if "MOSEK" in cp.installed_solvers():
            solver = cp.MOSEK
        elif "GUROBI" in cp.installed_solvers():
            solver = cp.GUROBI
        else:
            solver = cp.ECOS_BB

        try:
            problem.solve(solver=solver, verbose=False, abstol=1e-4)

            if problem.status in ["optimal", "optimal_inaccurate"]:
                w_opt = w.value
                if w_opt is None:
                    continue
                h_opt = construct_h_from_basis(W_basis, w_opt)
                # Correct for numerical inaccuracies
                h_opt[h_opt < 0] = 0
                h_opt[h_opt > 1] = 1
                c5_opt = compute_c5(h_opt, dx)

                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_solution = h_opt
        except:
            continue

    if best_solution is None:
        best_solution = initial_h_values if 'initial_h_values' in globals() else np.ones(n_points) / n_points * n_points / 2
        best_c5 = compute_c5(best_solution, dx)

    return (best_solution, best_c5, n_points)
