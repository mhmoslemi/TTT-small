import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values

    np.random.seed(seed)
    start_time = time.time()

    n_points = 45
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Structured initialization if no parent exists
        h = np.zeros(n_points)
        h[n_points//2 - n_points//4:n_points//2 + n_points//4] = 1
        # Ensure constraint sum(h) = n_points / 2
        h = np.clip(h, 0, 1)
        if h.sum() == 0:
            h.fill(0.5)
        h *= n_points / h.sum()

    dx = 2.0 / n_points
    step_size = 0.05
    best_h = h.copy()
    best_c5 = None
    patience = 0
    max_patience = 20

    while time.time() - start_time < budget_s:
        # Compute correlations and find C5
        padded_h = np.concatenate([np.zeros(len(h) - 1), h, np.zeros(len(h) - 1)])
        ones_minus_h = 1.0 - h
        cors = np.correlate(h, ones_minus_h, mode="full")
        cors *= dx  # scale by dx
        c5 = cors.max()

        # Track best solution
        if best_c5 is None or c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            print(f"Improved C5: {best_c5:.6f}")
        else:
            patience += 1

        if patience >= max_patience:
            # Reinitialize if no improvement
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h *= n_points / h.sum()
            patience = 0

        # Steepest descent direction
        grad_copy = np.zeros_like(h)
        for k in range(1, n_points):
            valid = np.arange(n_points - k)
            grad_copy[valid] += (1 - h[valid + k])
        # Subtract self-correlation
        for k in range(n_points):
            if 0 <= k < len(cors) and 0 <= k + n_points < len(cors):
                grad_copy -= (1 - h) * cors[k + n_points]
        grad_copy *= dx
        grad_copy /= max(1e-7, np.abs(grad_copy).max())

        # Update and project
        h -= step_size * grad_copy
        h = np.clip(h, 0, 1)
        h *= n_points / h.sum()

        if (time.time() - start_time) % 50 < 1:
            print(f"Current C5: {c5:.6f} after {time.time() - start_time:.2f}s")

    return (best_h.tolist(), best_c5, n_points)
