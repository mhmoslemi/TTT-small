import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Fix n_points
    n_points = 95 if "initial_h_values" in globals() else 95
    if "initial_h_values" in globals():
        h = initial_h_values.copy()
    else:
        # Feasible initialization: uniform distribution
        h = np.ones(n_points) / n_points * (n_points / 2)
    
    assert np.all(0 <= h) and np.all(h <= 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-8
    
    dx = 2.0 / n_points

    alpha = 0.5
    h_prev = h.copy()
    best_h = h.copy()

    max_iterations = 1000
    tic_total = time.time()
    iter_no_improve = 0
    best_c5 = float("inf")

    for iter in range(max_iterations):
        tic = time.time()

        # Compute moving convexification variable
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Correlate with 1 - h
        h_flip = 1 - h
        overlap = correlate(h_shifted, h_flip, mode="full") * dx
        k_candidates = np.arange(1, n_points)

        # Only consider k where overlap is positive and not out of bounds
        overlap_idx_k = overlap[n_points:2 * n_points]
        valid_k = k_candidates[overlap_idx_k > 0]
        if len(valid_k) == 0:
            break
        k_max = valid_k[np.argmax(overlap_idx_k[valid_k - 1])]

        # Formulate convex problem for h with fixed h_shifted
        opt_h = cp.Variable(shape=h.shape, nonneg=True)
        obj = cp.Minimize(
            cp.sum(
                opt_h * (1 - cp.roll(opt_h, shift=-k_max)) 
            ) * dx
        )
        constraints = [
            cp.sum(opt_h) == n_points / 2,
            opt_h <= 1
        ]

        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status != cp.OPTIMAL:
            h_prev = h.copy()
            iter_no_improve += 1
            if iter_no_improve >= 10: 
                break
            if time.time() - tic_total > budget_s * 0.98:
                break
            continue

        h = opt_h.value
        h = np.clip(h, 0, 1)

        # Update best
        h_flip = 1 - h
        overlap = correlate(h, h_flip, mode="full") * dx
        overlap_idx_k = overlap[n_points:2 * n_points]
        c5 = np.max(overlap_idx_k)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_no_improve = 0
            if alpha > 0.1:
                alpha *= 0.95
        else:
            iter_no_improve += 1

        # Budget check
        if time.time() - tic_total > budget_s * 0.98:
            break

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)

    h_flip = 1 - best_h
    overlap = correlate(best_h, h_flip, mode="full") * dx
    overlap_idx_k = overlap[n_points:2 * n_points]
    c5 = np.max(overlap_idx_k)

    return best_h, c5, n_points
