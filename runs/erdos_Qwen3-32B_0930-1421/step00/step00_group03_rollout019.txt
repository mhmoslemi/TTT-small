import time

    patience = 60  # seconds of no improvement before quitting
    learning_rate = 0.01
    max_iterations = 5000
    iteration = 0

    def compute_c5(h, dx):
        h = h * dx
        linear = np.cumsum(h) * dx
        conv = np.correlate(linear, 1-linear, mode='full')
        return np.max(conv)

    def compute_c5_index(h, dx):
        h = h * dx
        linear = np.cumsum(h) * dx
        conv = np.correlate(linear, 1-linear, mode='full')
        c5_val = np.max(conv)
        max_index = np.argmax(conv)
        return c5_val, max_index

    # Optimization loop
    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        # Compute C5 and identify the most critical shift
        c5_val, max_index = compute_c5_index(h, dx)

        # Check for improvement
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            last_improvement_time = time.time()

        # Early stopping
        if time.time() - last_improvement_time > patience:
            break

        # Perform an update step
        h_shifted = np.roll(h, -max_index - 1)
        gradient = (1 - h_shifted) * dx
        grad = np.zeros_like(h)
        grad[(max_index + 1):] = gradient[:(- (max_index + 1))]

        h += -learning_rate * grad

        # Ensure [0,1] bounds
        h = np.clip(h, 0, 1)

        # Normalize to satisfy the integration condition
        scale_factor = (n_points / 2.0) / h.sum()
        h *= scale_factor

    h = h * (n_points / 2.0 / h.sum())

    c5_bound = compute_c5(h, dx)

    return h.tolist(), c5_bound, n_points
