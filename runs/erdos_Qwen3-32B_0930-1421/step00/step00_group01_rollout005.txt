import numpy as np
import time

def evaluate_candidate(h):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    h = h.copy()
    s = float(np.sum(h))
    if not np.isclose(s, n / 2):
        h *= (n / 2) / s
    h_flipped = 1 - h
    corr = np.correlate(h, h_flipped, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 47  # align with initial_h_values
    global initial_h_values
    h_best = np.array(initial_h_values, copy=True)
    c5_best = evaluate_candidate(h_best)

    pop_size = 50
    mut_rate = 0.02
    pop = [h_best + np.random.normal(0, 0.1, n_points) for _ in range(pop_size - 1)]
    pop.append(h_best)
    for h in pop:
        s = float(np.sum(h))
        if not np.isclose(s, n_points / 2):
            h *= (n_points / 2) / s

    max_generations = 1000
    generation = 0

    while time.time() - start_time < budget_s and generation < max_generations:
        generation += 1

        # Evaluate the population
        fitnesses = []
        for h in pop:
            f = evaluate_candidate(h)
            fitnesses.append((h.copy(), f))

        # Sort by fitness
        fitnesses.sort(key=lambda x: x[1])
        pop = [f for h, f in fitnesses]
        best_curr = fitnesses[0][1]
        if best_curr < c5_best:
            h_best = fitnesses[0][0]
            c5_best = best_curr

        # Select top candidates
        top_k = int(0.2 * len(pop))
        selected = pop[:top_k]

        # Crossover and mutation
        new_pop = []
        while len(new_pop) < len(pop) and time.time() - start_time < budget_s:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            s = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:s], p2[s:]])
            # mutation
            if np.random.rand() < mut_rate:
                mask = np.random.rand(n_points) < 0.05
                child[mask] = np.random.rand(np.sum(mask))
                s = float(np.sum(child))
                if not np.isclose(s, n_points / 2):
                    child *= (n_points / 2) / s
            new_pop.append(child)

        pop = selected + new_pop

    return h_best, c5_best, n_points
