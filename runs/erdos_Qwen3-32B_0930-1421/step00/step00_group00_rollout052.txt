def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction
    h = np.copy(initial_h_values["h_values"])
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = evaluate_solution(h)

    # Optimization parameters
    max_iter = 1000
    alpha = 1.0  # Initial step size
    beta = 0.5   # Line search backtracking factor
    grad_norm_threshold = 1e-6
    n_evaluations = 0

    for iter in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break

        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        corr *= dx
        # Find the worst overlap shift k*
        k_star = int(round((n_points - 1) / 2))
        worst_index = np.argmax(corr)
        k_star = worst_index - (n_points - 1)

        # Compute gradient based on k*
        grad = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = (i + k_star) % n_points
            if shifted_i >= n_points:
                continue
            grad[i] = (1 - h[shifted_i]) * dx

        # Compute gradient norm for stopping condition
        grad_norm = np.linalg.norm(grad)
        if grad_norm < grad_norm_threshold:
            break

        # Line search to find optimal step size
        t = alpha
        for _ in range(5):
            # Trial step
            new_h = h - t * grad
            # Project back to constraint domain
            new_h = np.clip(new_h, 0, 1)
            # Ensure integral constraint
            new_h += (target_sum - np.sum(new_h)) / n_points

            # Evaluate
            new_c5 = evaluate_solution(new_h)
            n_evaluations += 1
            if new_c5 < best_c5:
                h = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = np.copy(h)
                break
            t *= beta  # Reduce step size

        # Over time if no improvement in last 100 iterations
        if (best_c5 == evaluate_solution(h)) and (iter % 100 == 0):
            break

    # Final projection and print
    best_h = np.clip(best_h, 0, 1)
    best_h += (target_sum - np.sum(best_h)) / n_points
    final_c5 = evaluate_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_solution(h):
    corr = correlate(h, 1 - h, mode='full')
    corr *= 2.0 / len(h)
    return np.max(corr)
