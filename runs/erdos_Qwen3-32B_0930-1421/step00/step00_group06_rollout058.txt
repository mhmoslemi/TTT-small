import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Configuration
    n_points = 95
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h_values = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.mean() * (n_points / 2)
        h = np.clip(h, 0, 1)
        h_values_init = h

    h = cp.Parameter(n_points, value=h_values)
    h_values = cp.Variable(n_points)
    constraints = [
        0 <= h_values,
        h_values <= 1,
        cp.sum(h_values) == n_points / 2
    ]

    # Start timer
    start_time = time.time()
    max_time = budget_s
    max_iter = 500

    h_opt = h
    C5_opt = 0.0

    for iter in range(max_iter):
        if time.time() - start_time > max_time:
            break

        h_current = h_opt.value
        if isinstance(h_current, (int, float, type(None))):
            h_current = np.random.rand(n_points)
            h_current = h_current / h_current.mean() * (n_points / 2)
            h_current = np.clip(h_current, 0, 1)

        overlap = np.correlate(h_current, 1.0 - h_current, mode='full')
        relevance_mask = np.zeros_like(overlap, dtype=bool)
        lags = np.arange(-n_points + 1, n_points)
        for lag in lags:
            idx = lag + n_points - 1
            if 0 <= lag < n_points and overlap[idx] > 0:
                relevance_mask[idx] = True

        overlap_truncated = overlap[relevance_mask]
        nu_k = np.copy(overlap_truncated)

        proxy_objective = cp.Minimize(cp.sum(cp.log(nu_k + 1e-6)))

        problem = cp.Problem(proxy_objective, constraints)
        problem.solve(solver=cp.OSQP if "OSQP" in cp.installed_solvers() else cp.SCS)

        if h_values.value is None:
            break

        h_opt = h_values
        h_current_new = np.clip(np.copy(h_values.value), 0, 1)
        h_current_new = h_current_new / h_current_new.mean() * n_points / 2
        h_current_new = np.clip(h_current_new, 0, 1)
        h_current_new = np.copy(h_current_new)

        # Recompute C5_match with corrected np.correlate
        C5_match = np.max(np.correlate(h_current_new, 1.0 - h_current_new, mode='full') * dx)

        if iter == 0 or C5_match < C5_opt:
            C5_opt = C5_match
            h_values = np.copy(h_current_new)

    return (h_values, C5_opt, n_points)
