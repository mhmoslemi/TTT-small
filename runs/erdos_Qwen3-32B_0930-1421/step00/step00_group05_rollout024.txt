import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    while True:
        # Compute the overlap for all shifts
        overlaps = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        
        # Find the shift with the most overlap (excluding too small/large shifts)
        valid_k = np.arange(max(1, n_points - 40), min(2*n_points - 1, n_points + 40))
        overlaps = overlaps[valid_k]
        idx = np.argmax(overlaps)
        k_max = valid_k[idx]
        
        # Project k to equivalent shift within [0, n_points)
        k_centered = k_max - n_points + 1
        
        # Create a gradient focused on reducing the overlap at k_centered
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= (i + k_centered) < n_points:
                gradient[i] = 1.0 - 2.0 * h[i] - (1.0 - 2.0 * h[i + k_centered])
            else:
                gradient[i] = 1.0 - 2.0 * h[i]
        
        # Gradient descent with backtracking line search
        alpha = 1.0
        max_iter_line_search = 5
        
        for _ in range(max_iter_line_search):
            # Compute new h with this step size
            h_new = h + alpha * gradient
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to maintain the integral constraint (must sum to n_points / 2)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute overlap after this step
            overlaps_new = scipy.signal.correlate(h_new, 1 - h_new, mode='full') * dx
            val_new = np.max(overlaps_new)
            val_old = np.max(scipy.signal.correlate(h, 1 - h, mode='full') * dx)
            
            # If move reduces the maximum overlap, accept it
            if val_new < val_old * 1.001:  # Slight tolerance to avoid zero progress
                h = h_new.copy()
                break
            # Otherwise, reduce step size
            alpha *= 0.5
        else:
            # If we couldn't make progress, break
            break
        
        # Check time budget
        elapsed = time.time() - budget_s_start
        if elapsed > budget_s:
            break

    # Final cleanup to ensure exact constraint satisfaction
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate final C5 bound
    overlaps = scipy.signal.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(overlaps)
    
    return h, c5_bound, n_points
