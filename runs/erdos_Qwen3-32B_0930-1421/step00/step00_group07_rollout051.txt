        return h_normalized

    alpha = 0.01

    counter = 0

    for it in range(max_iter):
        # Compute overlap for all lags
        h_1_h = 1.0 - h
        corr = np.correlate(h, h_1_h, mode='full') * dx
        lag_max = np.argmax(np.abs(corr))
        max_overlap = corr[lag_max]

        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            counter = 0
        else:
            counter += 1

        # Early stopping
        if counter >= patience:
            alpha *= 0.5
            counter = 0
            if alpha < 1e-4:
                break

        # Time constraint
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute gradient manually for max lags
        def get_gradient(h, lag_max):
            n = len(h)
            grad = np.zeros(n)
            for i in range(n):
                idx1 = np.arange(n)
                idx2 = (idx1 + lag_max) % n
                grad += h_1_h[idx1] * h[idx2] - h[idx1] * h[idx2]
            grad *= dx
            return grad

        # Update
        h = h - alpha * g
        h = clip_and_normalize(h)

    # Final validation
    h = clip_and_normalize(h)
