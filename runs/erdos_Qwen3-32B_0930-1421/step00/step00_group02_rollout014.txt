import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        h = np.ones(50) / 25  # basic initialization
        n_points = len(h)

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = float('inf')

    def compute_overlap(h):
        h = h * dx if h.ndim == 1 else h
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    def evaluate(h):
        return compute_overlap(h)

    def project_simplex(v, s=1.0):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(1, len(u) + 1) * u >= cssv
        _, = np.where(ind)
        idx = max(np.nonzero(ind)[0])
        theta = cssv[idx] / (idx + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

    # Normalize h to have sum equal to target_sum
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            h[:] = target_sum / h.shape[0]
        else:
            h[:] *= (target_sum / h_sum)
        h = np.clip(h, 0, 1)
        return h

    h = normalize_h(h)

    best_h = h.copy()
    best_c5 = compute_overlap(best_h)

    iter_count = 0

    while time.time() - start < budget_s:
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlap)
        S = overlap[k_max]
        
        # Compute gradient at max overlap
        grad = np.zeros_like(h)
        grad[:n_points - k_max] -= h[k_max:]
        grad[k_max:] += h[:n_points - k_max]
        grad *= dx

        # Define optimization variable
        h_var = cp.Variable(n_points)

        # Define objective: stay close to gradient descent step
        step_size = 0.1
        obj = cp.Minimize(cp.sum_squares(h_var - (h - step_size * grad)))

        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ]

        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=200, abstol=1e-4, reltol=1e-4)
            h_new = h_var.value
        except:
            h_new = h - step_size * grad

        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new)

        c5 = compute_overlap(h_new)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        h = h_new

        iter_count += 1
        if iter_count > 2000:
            break

    return (best_h, best_c5, n_points)
