import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the provided initial_h_values as the starting point
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else None
    if h is None:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    else:
        n_points = len(h)

    dx = 2.0 / n_points
    tolerance = 1e-10

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def normalize_h(h_in):
        clipped = np.clip(h_in, 0, 1)
        sum_h = np.sum(clipped)
        scale = (n_points / 2) / sum_h
        scaled = clipped * scale
        return np.clip(scaled, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Frequency domain optimization (60% of time)
    phase1_time_limit = 0.6 * budget_s
    while time.time() - start_time < phase1_time_limit:
        try:
            H = np.fft.fft(h)
            n = h.shape[0]

            # Create CVXPY variable in frequency domain
            H_new = cp.Variable(H.shape, complex=True)

            # Constraints
            constraints = [H_new == H]  # Start with existing Fourier coeff
            constraints += [cp.abs(H_new[n // 2]) == 0]  # Zero-padding in frequency domain
            constraints += [cp.maximum(cp.abs(H_new)) == cp.Minimize(cp.max(cp.abs(H_new)))]

            # Form and solve the problem
            problem = cp.Problem(cp.Minimize(cp.max(cp.abs(H_new))), constraints)
            problem.solve(solver=cp.CVXOPT)

            # Update the time domain signal
            h = np.fft.ifft(H_new.value).real
            h = normalize_h(h)

            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        except Exception as e:
            # Skip failed optimization
            pass

    # Phase 2: Time domain gradient descent optimization (40% of time)
    phase2_time_limit = 0.4 * budget_s
    learning_rate = 0.1
    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        eps = 1e-5

        # Compute gradient with finite differences
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = normalize_h(h_p)
            c_p = compute_c5(h_p)

            h_n = h.copy()
            h_n[i] -= eps
            h_n = normalize_h(h_n)
            c_n = compute_c5(h_n)

            grad[i] = (c_p - c_n) / (2 * eps)

        # Update h with gradient descent
        h -= learning_rate * grad
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Reduce learning rate over time
        learning_rate *= 0.999

        # Check budget
        if time.time() - start_time >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
