import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h / h.size, 0, 1)  # normalize for dx and [0,1] range
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return c5

def run(seed=42, budget_s=1100, n_points=45):
    start_time = time.time()
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        np.random.seed(seed)
        h = np.zeros(n_points)
        h[:int(n_points/2)] = 1.0
        h = np.random.permutation(h)

    def objective(h):
        h_sum = np.sum(h)
        if h_sum < n_points / 2 - 1e-6 or h_sum > n_points / 2 + 1e-6:
            return float('inf')
        corr = correlate(h / n_points, (1 - h / n_points), mode="full") * 2.0
        c5 = np.max(corr)
        return c5

    t = 0.5
    alpha = 0.1
    alpha_decay = 0.999
    best_h = h.copy()
    best_c5 = objective(h)
    step = 1

    while time.time() - start_time < budget_s:
        # Compute C5 and find maximum lag
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)

        # Find the most influential lag k
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Compute gradient at relevant indices
        indices = np.arange(n_points).astype(int)
        valid_i = np.where((indices + k) < n_points)
        grad = np.zeros_like(h)
        grad = 1 - h[(indices + k) % n_points]

        # Gradient update
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)

        # Projection onto the feasible set
        h = h / np.sum(h) * (n_points / 2)

        # Check and update best
        c5_current = objective(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            alpha = min(alpha * alpha_decay, 1e-3)  # decay step size
        else:
            # Restart with new step size
            h = best_h.copy()
            alpha = max(alpha * 2, 1e-3)

    h_values = best_h
    c5_bound = best_c5
    h_values = np.clip(h_values, 0, 1)
    assert np.all((0 <= h_values) & (h_values <= 1))
    assert np.isclose(np.sum(h_values), n_points / 2.0)

    return h_values, c5_bound, n_points
