import numpy as np
import math
import time
import random
import cvxpy as cp
from collections import deque

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = len(initial_h_values)
    n_particles = 50
    max_iterations = 200
    inertia = 0.7
    c1, c2 = 1.5, 1.5
    H = initial_h_values
    ones = np.ones_like(H)
    dx = 2.0 / n_points
    
    def objective(h):
        return max(np.correlate(h, 1 - h, mode="full") * dx)
    
    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - n_points / 2) > 1e-5:
            delta = (n_points / 2 - s) / h.size
            h += delta
            h = np.clip(h, 0, 1)
            h = h / h.sum() * (n_points / 2)
        return h
    
    particles = [H + np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    particles = [enforce_constraints(p) for p in particles]
    velocities = [np.zeros_like(H).astype(float) for _ in range(n_particles)]
    
    pbest_positions = [p.copy() for p in particles]
    pbest_scores = [objective(p) for p in particles]
    gbest_index = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_index]
    gbest_score = pbest_scores[gbest_index]
    
    iteration_scores = deque(maxlen=20)
    
    for i in range(max_iterations):
        time_taken = time.time() - start_time
        if time_taken > budget_s - 1:
            break
        if i > 0 and i % 20 == 0:
            iteration_scores.append(gbest_score)
            if len(set(iteration_scores)) == 1 and i >= 100:
                break
        
        for idx in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[idx] = inertia * velocities[idx] + c1 * r1 * (pbest_positions[idx] - particles[idx]) + \
                              c2 * r2 * (gbest_position - particles[idx])
            old = particles[idx].copy()
            particles[idx] = particles[idx] + velocities[idx]
            particles[idx] = enforce_constraints(particles[idx])
            
            new_score = objective(particles[idx])
            if new_score < pbest_scores[idx]:
                pbest_scores[idx] = new_score
                pbest_positions[idx] = particles[idx].copy()
                if new_score < gbest_score:
                    gbest_score = new_score
                    gbest_position = particles[idx].copy()
    
    return (gbest_position.tolist(), gbest_score, n_points)
