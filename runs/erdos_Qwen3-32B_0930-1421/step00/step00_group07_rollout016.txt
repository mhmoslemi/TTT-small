import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t0 = time.time()
    initial_h = initial_h_values
    h = initial_h.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    # Normalize initial h to have correct average
    current_sum = h.sum()
    h = h * (target_sum / current_sum)
    
    alpha = 0.01
    max_iter = 5000
    patience = 200
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_iter = 0
    early_stop_counter = 0

    for i in range(max_iter):
        if time.time() - t0 > budget_s:
            break
            
        # Compute 1 - h as a separate array
        one_minus_h = 1.0 - h
        
        # Compute cross-correlation (full mode for all possible lags)
        corr = correlate(h, one_minus_h, mode='full')
        corr = corr * dx
        C5 = np.max(corr)

        # Calculate gradient based on the maximum overlap lag
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        overlap = corr[max_index]
        dh = np.zeros_like(h)
        if lag >= 0:
            # h is on the left, 1 - h on the right
            dh[:n_points - lag] = h[lag:]
            dh[n_points - lag:] = one_minus_h[:lag]
        else:
            # h is on the right, 1 - h on the left
            dh[-lag:] = h[:lag]
            dh[:-lag] = one_minus_h[lag:]            
        
        # Gradient descent update
        h = h - alpha * dh
        
        # Apply box constraint and normalize
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        scale = target_sum / current_sum
        h = h * scale
        
        # Check progress
        c5_val = evaluate_erdos_solution(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            best_iter = i
            early_stop_counter = 0
        else:
            early_stop_counter += 1
            if early_stop_counter >= patience:
                alpha *= 0.5
                early_stop_counter = 0

    return (best_h, best_c5, n_points)
