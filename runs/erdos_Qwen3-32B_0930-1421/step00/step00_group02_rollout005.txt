import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        dx = 2.0 / n_points
        assert np.all(h >= 0) and np.all(h <= 1)
        assert np.isclose(np.sum(h) * dx, 1.0)
    else:
        # simple initialization: sparse step function
        n_points = 45
        h = np.zeros(n_points, dtype=np.float64)
        h[::2] = 1.0
        h *= (n_points / 2) / np.sum(h)

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Track time usage and progress
    last_update = time.time()
    patience_counter = 0
    patience_limit = 1000
    step_size = 0.001
    adaptive_alpha = 0.995

    while time.time() - start_time < budget_s:
        dx = 2.0 / n_points
        lag_product = correlate(h, 1 - h, mode='full')  # raw overlap at each lag
        c5 = np.max(lag_product * dx)

        # Identify the most problematic overlap lag
        max_idx = np.argmax(lag_product * dx)
        k = max_idx - n_points + 1  # lag corresponding to the max

        # Compute gradient approx w.r.t h[i]
        # (1 - h[i + k]) for forward shift; (1 - h[i - k]) for reverse shift
        grad = np.zeros_like(h)
        if k > 0:
            # forward overlap
            grad[k:] += (1 - h[:n_points - k])
        else:
            # backward overlap
            grad[:n_points + k] += (1 - h[-k:])

        # Normalize gradient by dx and scale
        grad *= step_size
        h -= grad
        h = np.clip(h, 0, 1)

        # Maintain constraint
        h *= (n_points / 2) / np.sum(h)

        # evaluate improvement
        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_update = time.time()
            patience_counter = 0
        else:
            patience_counter += 1

        # adaptive step size
        if patience_counter > 100:
            step_size *= adaptive_alpha
            patience_counter = 0

        if current_c5 <= 0.38086:  # early stopping target
            break

    return best_h.tolist(), best_c5, n_points
