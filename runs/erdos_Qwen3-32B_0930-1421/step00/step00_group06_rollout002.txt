import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        ones = np.ones_like(h)
        corr = np.correlate(h, ones - h, mode="full")
        return np.max(corr * dx)

    def project_h(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped)
        return h_clipped * (n_points / 2) / total

    t0 = time.time()

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    imp_threshold = 1e-6
    lr = 0.01
    max_iter = 10000
    decay = 0.99
    iters = 0

    while iters < max_iter and time.time() - t0 < budget_s * 0.95:
        iters += 1

        ones = np.ones_like(h)
        ones_minus_h = ones - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        c5 = np.max(corr * dx)
        
        if iters % 100 == 0:
            print(f"Iteration {iters}, C5: {c5}, Time: {time.time() - t0}")
        
        if abs(c5 - best_c5) < imp_threshold:
            break
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find the worst k
        shifted = h[n_points:]
        target = corr[n_points - 1:]
        k = np.argmax(target[n_points - 1:])

        k += n_points - 1  # Align index
        k %= n_points

        # Gradient update
        # See mathematical gradient in strategy description
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = (1 - h[(i + k) % n_points]) - ones_minus_h[(i - k) % n_points]
        
        # Gradient descent update
        h += lr * grad
        h = project_h(h)

        # Learning rate decay
        lr *= decay

    return (best_h.tolist(), best_c5, n_points)
