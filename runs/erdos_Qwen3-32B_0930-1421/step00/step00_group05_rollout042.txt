import numpy as np
import time
from math import isclose

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = h.size
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_str = False

    print("Starting c5:", best_c5)

    learning_rate = 0.1
    max_iter = 100000
    patience = 0
    best_iter = -1

    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break

        # Compute overlaps
        h_flip = 1 - h
        corr = np.correlate(h, h_flip, mode="full")
        corr = corr * dx

        # Find max overlap index k
        k_max = np.argmax(corr)
        k_center = n_points

        # Only care about current k = k_center to k_center + 1
        # since the ones beyond are small if we use mirrored padding
        if k_max < k_center - 3:
            h = h[::-1].copy()  # Mirror the function
            h_flip = 1 - h  # Adjust accordingly
            corr = np.correlate(h, h_flip, mode="full")
            corr = corr * dx
            k_max = np.argmax(corr)

        # Compute local gradient for current k_max
        pad_h = np.concatenate([h, np.zeros(n_points + n_points)], dtype=np.float64)
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx = k_max - i
            if idx >= 0 and idx < len(grad):
                # dC_k / dh[i] = -1 + 2h[i] - 2h[i]h[i+k]
                grad[i] = -1 + 2 * pad_h[i] - 2 * pad_h[i] * pad_h[i + k_max]

        # Gradient descent step
        step = -learning_rate * grad
        new_h = h + step

        # Projection onto the box constraint [0, 1]
        new_h_proj = np.clip(new_h, 0, 1)

        # Projection onto the integral constraint
        new_sum = np.sum(new_h_proj)
        if abs(new_sum - n_points / 2) > 1e-5:
            scale = (n_points / 2) / (new_sum + 1e-8)
            new_h_proj *= scale
            new_h_proj = np.clip(new_h_proj, 0, 1)

        # Check if better
        new_c5 = compute_c5(new_h_proj, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h_proj.copy()
            best_str = True
            patience = 0
        else:
            patience += 1

        # Early stopping
        if patience > 100:
            learning_rate *= 0.8
            patience = 0

        # Quick fallback after X iterations if we exceed time
        if time.time() - start_time > budget_s * 0.99 or it % 10000 == 0:
            if not best_str and it < 100:
                # If we haven't improved within 100 iterations, fallback to reevaluating
                h = best_h.copy()

    # Ensure we sanitize final h
    h = best_h
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-5:
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / (np.sum(h) + 1e-8)
        h *= scale
    h = np.clip(h, 0, 1)

    final_c5 = compute_c5(h, dx)
    print("Final c5:", final_c5)
    print("Time elapsed:", time.time() - start_time)

    return h.tolist(), final_c5, n_points

def compute_c5(h, dx):
    h_flip = 1 - h
    corr = np.correlate(h, h_flip, mode="full")
    return np.max(corr * dx)
