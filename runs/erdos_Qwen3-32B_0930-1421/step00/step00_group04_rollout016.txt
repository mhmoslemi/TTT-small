import numpy as np
import time

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    h1 = h - h ** 2
    gov = np.correlate(h, h1, mode='full') * dx
    return gov.max()

def run(seed=42, budget_s=1100, n_points=42, smoothing=0.001, learning_rate=0.01, tolerance=1e-6):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            n_points = len(h)
    else:
        h = np.ones(n_points, np.float64)
        h[:n_points//2] = 0.0

    target_sum = n_points / 2
    best_h = h.copy()
    best_score = evaluate_erdos_solution(h)

    while time.time() - t0 < budget_s:
        # Compute gradient
        h1 = h - h ** 2
        gov = np.correlate(h, h1, mode='full')
        max_g = gov.argmax()
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= max_g - i < n_points:
                grad[i] += (h[i] - h[max_g - i])
        grad = grad * (2.0 / n_points)

        # Gradient update
        h = h - learning_rate * grad
        # Add tiny smoothing to avoid degeneracy
        h = np.clip(h, smoothing, 1.0 - smoothing)

        # Normalize to maintain volume constraint
        h *= target_sum / h.sum()

        # Check improvement and update best solution
        score = evaluate_erdos_solution(h)
        if score < best_score - tolerance:
            best_score = score
            best_h = h.copy()
        elif abs(score - best_score) < tolerance:
            break  # Stagnation threshold

    h_values = best_h
    c5_bound = best_score
    return h_values, c5_bound, n_points
