import numpy as np
import math
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        h_flipped = 1 - h
        corr = np.correlate(h, h_flipped, mode='full')
        return np.max(corr * dx)

    def project_to_feasible(h):
        h_proj = np.clip(h, 0, 1)
        total = np.sum(h_proj)
        target = n_points / 2
        if abs(total - target) < 1e-8:
            return h_proj
        diff = target - total
        if diff > 0:
            # Need to increase values at the smallest
            indices = np.argsort(h_proj)
            h_proj[indices[:int(abs(diff) + 1)]] = 1.0
        else:
            # Need to decrease values at the largest
            indices = np.argsort(-h_proj)
            h_proj[indices[:int(abs(diff) + 1)]] = 0.0
        return h_proj

    def sa_step(h, temp, step_size):
        h_proposal = np.clip(h + step_size * np.random.randn(n_points), 0, 1)
        h_proposal = project_to_feasible(h_proposal)
        c5_proposal = compute_c5(h_proposal)
        c5_current = compute_c5(h)
        if c5_proposal < c5_current or np.random.rand() < np.exp((c5_current - c5_proposal) / max(temp, 1e-5)):
            return h_proposal, c5_proposal
        return h, c5_current

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 1.0
    alpha = 0.99
    step_size = 0.1
    sa_iter = 0
    sa_max = 1000
    timeout_padding = 10

    while time.time() - start_time < budget_s - timeout_padding and sa_iter < sa_max:
        h, c5 = sa_step(best_h, T, step_size)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        T *= alpha
        step_size *= math.sqrt(1 - sa_iter / sa_max)
        sa_iter += 1

    # Gradient descent
    def gradient(h):
        h_clipped = np.clip(h, 0, 1)
        h_flipped = 1 - h_clipped
        full_corr = np.correlate(h_clipped, h_flipped, mode='full')
        max_idx = np.argmax(full_corr)
        k = max_idx - (n_points - 1)

        if k < 0:
            shift = -k
            result = -h_clipped[:n_points - shift] + h_flipped[shift:]
        else:
            shift = k
            result = h_clipped[shift:] - h_flipped[:n_points - shift]
        return result / n_points

    def objective(h_flat):
        h = h_flat.copy()
        h = project_to_feasible(h)
        return compute_c5(h)

    result = minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-6}
    )

    h_opt = result.x
    h_opt = project_to_feasible(h_opt)
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    return best_h, best_c5, n_points
