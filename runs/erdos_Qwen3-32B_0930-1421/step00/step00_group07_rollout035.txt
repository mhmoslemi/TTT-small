import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    dx = 0.025  # 80 point discretization
    n_points = int(2 / dx)
    assert initial_h_values is not None, "initial_h_values should be provided"
    h = np.copy(initial_h_values)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum  # normalize mass

    def compute_c5(h):
        dx_ = 2.0 / len(h)
        hh = np.copy(h)
        corr = correlate(hh, 1 - hh, mode="full")
        c5 = np.max(corr) * dx_
        return c5

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.025
    patience = 0
    max_iter = 5000
    val_freq = 300
    eps = 1e-6
    curr_iter = 0

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        # Compute the cross-correlation values of h with 1 - h
        corr = correlate(h, 1 - h, mode="full")
        dx_ = 2.0 / n_points
        corr *= dx_
        idx_max = np.argmax(corr)
        lag = idx_max - n_points + 1
        mask_len = n_points + np.abs(lag)

        # Compute gradient (simplified for first-order approx)
        # Gradient towards reducing maximum over lap
        grad = np.zeros(n_points)
        if lag >= 0:
            grad[:n_points - lag] -= (1 - h[lag:])
            grad[lag:] -= h[:n_points - lag]
        else:
            grad[-lag:] -= (1 - h[:n_points + lag])
            grad[:n_points + lag] -= h[-lag:]

        grad = grad / (np.linalg.norm(grad, ord=2) + eps)

        # Update
        h -= alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = (h * (n_points / 2) / h.sum())

        # Validation / Update best solution
        curr_iter += 1
        if i % val_freq == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            if patience >= 3:
                alpha *= 0.5
            if alpha < 1e-6:
                break

    return best_h, best_c5, n_points
