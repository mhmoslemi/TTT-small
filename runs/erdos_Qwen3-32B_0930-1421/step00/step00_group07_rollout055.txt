import numpy as np

from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_limit = start_time + budget_s

    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    n_pixels = 1000  # number of sample points for overlap integral estimation

    initial_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(initial_h, n_points=n_points)[0]
    best_h = initial_h.copy()

    intervals = []
    for i in range(0, n_points - 1, 2):
        if initial_h[i] == initial_h[i + 1] == 1:
            a = i * dx
            b = min((i + 2) * dx, 2)
            intervals.append((a, b))
        elif initial_h[i] == 1:
            a = i * dx
            b = (i + 1) * dx
            intervals.append((a, b))
        elif initial_h[i + 1] == 1:
            a = (i + 1) * dx
            b = min((i + 2) * dx, 2)
            intervals.append((a, b))

    if not intervals:
        # fallback to simple centered interval if no initialization
        intervals = [(0.5, 1.5)]

    def intervals_to_h(intervals):
        h = np.zeros(n_points)
        for a, b in intervals:
            start = int(n_points * a // 2)
            end = int(n_points * b // 2) + 1
            h[start:end] = 1
            if b - a > 1.0 or b > 2 or a < 0:
                pass
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(intervals):
        h = intervals_to_h(intervals)
        total = sum(b - a for a, b in intervals)
        if abs(total - 1) > 1e-4:
            return float('inf')  # infeasible
        if np.sum(h) < 0.5 or np.sum(h) > 1.5:
            return float('inf')
        result = evaluate_erdos_solution(h, n_points=n_points)[0]
        return result

    # Perturbation parameters
    movement_scale = 0.08
    temperature = 1.0
    temp_decay = 0.995
    step_size = 0.02
    initial_intervals = intervals.copy()
    best_intervals = intervals.copy()

    for iteration in range(2000):
        if time.time() > budget_limit:
            break
        if iteration % 100 == 0:
            print(f"Iteration {iteration}: C₅ = {compute_c5(best_intervals):.6f}")

        # Copy current intervals
        new_intervals = best_intervals.copy()

        # Perturb a random interval
        if not new_intervals:
            a, b = 0.0, min(2.0, 1.0)
            new_intervals = [(a, b)]
        else:
            idx = np.random.choice(len(new_intervals))
            a, b = new_intervals[idx]

            # Choose random movement direction
            move_type = np.random.randint(0, 3)

            if move_type == 0:
                # Change start point
                delta = np.random.uniform(-movement_scale, movement_scale)
                new_a = max(0.0, a + delta)
                if idx > 0:
                    new_a = max(new_a, new_intervals[idx - 1][1] + 1e-4)
                if new_a < b:
                    new_intervals[idx] = (new_a, b)
            elif move_type == 1:
                # Change end point
                delta = np.random.uniform(-movement_scale, movement_scale)
                new_b = min(2.0, b + delta)
                if idx < len(new_intervals) - 1:
                    new_b = min(new_b, new_intervals[idx + 1][0] - 1e-4)
                if new_b > a:
                    new_intervals[idx] = (a, new_b)
            elif move_type == 2:
                # Shrink/expand the entire interval
                total_length = b - a
                delta = np.random.uniform(-0.5 * movement_scale, 0.5 * movement_scale)
                new_a = max(0.0, a + delta * 0.5)
                new_b = min(2.0, b + delta * 0.5)
                if new_b > new_a:
                    new_intervals[idx] = (new_a, new_b)

        # Compute Objective
        c5_new = compute_c5(new_intervals)
        c5_current = compute_c5(best_intervals)

        # Simple Metropolis acceptance criteria
        if c5_new < c5_current or \
           np.random.rand() < np.exp((c5_current - c5_new) / temperature):
            best_intervals = new_intervals.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = intervals_to_h(best_intervals)

        # Temperature decay and scale reduction
        temperature *= temp_decay
        if movement_scale > 1e-3 and iteration % 100 == 1:
            movement_scale *= 0.95

    return best_h, best_c5, n_points
