        # Compute the cross correlation h * (1 - h) using convolution
        correlate_result = np.correlate(h, ones_h, mode='full')
        # Multiply by dx^2 to match integration over [0,2] × [0,2]
        c5_values = (correlate_result[31:-32]) * dx
        return c5_values

            h = h + (target_sum - current_sum) / n_points

        h = np.clip(h, 0, 1)
        return h

                break

            h_var = cp.Variable(n_points)

            # Compute exact correlation for stability

                    new_c5 = np.max(compute_c5(new_h, n_points))
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5

                    best_c5 = np.max(compute_c5(best_h, n_points))
            except Exception as e:

                best_c5 = np.max(compute_c5(best_h, n_points))

            if np.random.rand() < 0.01 or best_c5 >= float('inf'):
