import numpy as np
import time
import random
import math
import cvxpy as cp
from numpy import correlate
from numpy.random import uniform
from numpy.polynomial.polynomial import Polynomial


def project_feasible(h, n_points):
    # Box constraint
    h = np.clip(h, 0.0, 1.0)
    # Integral constraint
    sum_h = np.sum(h)
    target_sum = n_points / 2
    if sum_h == target_sum:
        return h
    delta = (target_sum - sum_h) / n_points
    if delta > 0:
        h += delta
        h = np.clip(h, 0.0, 1.0)
    else:
        idx = np.argsort(h)
        h[idx] -= -delta
        h = np.clip(h, 0.0, 1.0)
    return h


def evaluate_particle(h):
    n_points = len(h)
    dx = 2.0 / n_points
    try:
        c_full = correlate(h, 1.0 - h, mode="full")
    except:
        c_full = np.ma.correlate(h, 1.0 - h, mode="full").data
    return np.max(c_full * dx), h


def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)

    if 'initial_h_values' in globals():

    else:
        h_best = uniform(0, 1, 70)
        h_best = project_feasible(h_best, len(h_best))

    h_best = project_feasible(h_best, len(h_best))
    c5_best, _ = evaluate_particle(h_best)
    n_points = len(h_best)

    swarm_size = 50
    inertia = 0.7
    c1 = 1.49
    c2 = 1.49

    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h = initial_h_values + 0.1 * np.random.randn(n_points)
        else:
            h = uniform(0, 1, n_points)
        h = project_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h)
        val, _ = evaluate_particle(h)
        personal_best_vals.append(val)

    global_best = personal_best[np.argmin(personal_best_vals)]

    start_time = time.time()
    max_iter = 1000
    max_no_improve = 100
    no_improve_count = 0
    best_c5 = np.inf

    for iter in range(max_iter):
        iter_start = time.time()
        now = time.time()
        if now - start_time >= budget_s - 10:
            break
        for p in range(swarm_size):
            r1 = random.random()
            r2 = random.random()
            v = inertia * velocities[p] + c1 * r1 * (personal_best[p] - particles[p]) + c2 * r2 * (global_best - particles[p])
            new_h = particles[p] + v
            new_h = project_feasible(new_h, n_points)
            new_val, _ = evaluate_particle(new_h)

            # Update personal best
            if new_val < personal_best_vals[p]:
                personal_best[p] = new_h
                personal_best_vals[p] = new_val
                if new_val < best_c5:
                    best_c5 = new_val
                    h_best = new_h
                    no_improve_count = 0
            else:
                no_improve_count += 1

            particles[p] = new_h
            velocities[p] = v

        # Update global best
        idx = np.argmin(personal_best_vals)
        if personal_best_vals[idx] < evaluate_particle(global_best)[0]:
            global_best = personal_best[idx]

        iter_time = time.time() - iter_start
        budget_left = budget_s - (time.time() - start_time)
        if budget_left < iter_time * 2:
            break

        if no_improve_count >= max_no_improve:
            break
    # Final validation
    h_best = project_feasible(h_best, n_points)
    c5, h = evaluate_particle(h_best)
    return h_best, c5, n_points
