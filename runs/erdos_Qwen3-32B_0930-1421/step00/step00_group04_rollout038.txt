import time

    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2
        return h

    population = [h_best]
    while len(population) < pop_size:
        population.append(create_individual())

    start_time = time.time()

    for generation in range(generations):
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)

        # Update best so far
        idx = np.argmin(fitness)
        if fitness[idx] < c5_best:
            h_best = population[idx].copy()
            c5_best = fitness[idx]

        # Time check
        if time.time() - start_time > budget_s:
            break

        next_gen = [h_best.copy()]

        # Genetic operations
        for _ in range((pop_size - 1) // 2):
            # Select parents
            parents = np.random.choice(population, size=2, p=(1.0 / np.array(fitness)) / np.sum(1.0 / np.array(fitness)))
            parent1, parent2 = parents

            # Crossover
            mask = np.random.rand(n_points) < 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)

            # Normalize to maintain ∑h[i] = n_points / 2
            for child in [child1, child2]:
                child_sum = np.sum(child)
                if child_sum != 0:
                    child = child / child_sum * (n_points / 2)
                else:
                    child = np.zeros_like(child)

                # Mutation
                if np.random.rand() < mutation_rate:
                    idxs = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
                    vals = np.random.rand(len(idxs)) * 0.5
                    child[idxs] = vals

                    # Re-normalize
                    child_sum = np.sum(child)
                    if child_sum != 0:
                        child = child / child_sum * (n_points / 2)

                next_gen.append(child)

        # Replace with new population
        population = next_gen

    return h_best.tolist(), c5_best, n_points
