import numpy as np
import cvxpy as cp
import math
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Use the pre-imported initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize to meet the mean constraint
    h_sum = np.sum(h)
    scaling = (n_points / 2) / h_sum
    h *= scaling
    h = np.clip(h, 0, 1)

    # Optimization parameters
    max_iterations = 1000
    alpha = 0.5
    improvement_threshold = 1e-4
    last_best_c5 = float('inf')

    h_prev = h.copy()

    for iter in range(max_iterations):
        if time() - start_time >= budget_s:
            break
        # Blend h_prev and current h to create h_shifted
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Compute current overlap
        # (compute using only one k for now; adjust in advanced versions)
        conv = np.correlate(h_shifted, 1 - h, mode="full") * dx
        k_max = np.argmax(conv)
        current_c5 = conv[k_max]

        # If improvement is below threshold, decay alpha
        if abs(last_best_c5 - current_c5) < improvement_threshold:
            alpha = max(alpha * 0.9, 0.1)

        # Re-define optimization variable
        h_opt = cp.Variable(n_points, nonneg=True)

        # Objective: minimize the overlap (for k_max only)
        # overlap = sum(h_opt * (1 - h_shifted[(i + k_max) % n_points]) * dx for i in range(n_points))
        h_shifted_kmax = np.roll(h_shifted, -k_max)
        h_shifted_kmax = h_shifted_kmax[:n_points]
        obj = cp.Minimize(cp.sum(h_opt * dx * (1 - h_shifted_kmax)))

        # Constraints
        constraints = [
            0 <= h_opt,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2
        ]

        # Solve problem
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.SCIP)

        if prob.status != cp.OPTIMAL:
            # If solver fails, copy previous h and continue
            h = h_prev.copy()
            continue

        h = h_opt.value
        h_prev = h.copy()

        # Update best_c5
        conv = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(conv)
        last_best_c5 = min(last_best_c5, current_c5)

        # Project back to feasible region if needed
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            scaling = (n_points / 2) / h_sum
            h *= scaling
            h = np.clip(h, 0, 1)

    conv = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(conv)
    return h, c5_bound, n_points
