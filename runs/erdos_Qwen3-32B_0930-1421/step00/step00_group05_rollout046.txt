import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Configuration
    n_points = 70  # As per the hint from the current best solution
    dx = 2.0 / n_points

    # Truncate time budget
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.zeros(n_points)
        h_initial[n_points // 2] = n_points / 2  # Point mass at center

    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    max_shift = 19  # Shifts greater than 50% of the domain are symmetric
    overlaps = []

    # Precompute for shifts k
    for k in range(1, max_shift + 1):
        shifted = np.roll(h, k)
        overlap = cp.sum(h * (1 - shifted) * dx)
        overlaps.append(overlap)

    # Use LP approx relaxation of max with equal bounds
    C5 = cp.Variable()
    constraints += [C5 >= o for o in overlaps]

    # Objective is to minimize the max overlap
    objective = cp.Minimize(C5)

    # Problem
    prob = cp.Problem(objective, constraints)

    # Solve using ECOS (supporting time limit)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, verbose=True, time_limit=time_left())
    except cp.SolverError:
        # Fallback to local solution if solver error
        h_result = h_initial
    else:
        h_result = np.clip(h.value, 0, 1)

    # Final verification and evaluation
    h_values = h_result
    c5_bound = evaluate_erdos_solution(h_values)

    return h_values, c5_bound, n_points
