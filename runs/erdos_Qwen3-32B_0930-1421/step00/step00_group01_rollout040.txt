import numpy as np
import scipy.optimize as opt
import time
from typing import Tuple

initial_h_values = None  # Will be injected at runtime

def evaluate_objective(h: np.ndarray, dx: float) -> float:
    h = h.clip(0.0, 1.0)
    h1 = 1 - h
    n = h.size
    max_overlap = 0
    for k in range(-n + 1, n):
        if k <= 0:
            i = np.arange(-k, n)
            j = i + k
        else:
            i = np.arange(0, n - k)
            j = i + k
        overlap = np.dot(h[i], h1[j]) * dx
        max_overlap = max(max_overlap, abs(overlap))
    return max_overlap

def normalize_h(h: np.ndarray) -> np.ndarray:
    total = h.sum()
    if total == 0:
        return h.copy()
    target_sum = h.size / 2
    h = h / total * target_sum
    return h.clip(0.0, 1.0)

def objective(packed_h: np.ndarray) -> float:
    n_points = len(packed_h)
    h = packed_h.copy()
    h = h.reshape(n_points)
    h = normalize_h(h)
    dx = 2.0 / n_points
    return evaluate_objective(h, dx)

def gradient(packed_h: np.ndarray) -> np.ndarray:
    n_points = len(packed_h)
    h = packed_h.copy().reshape(n_points)
    h = normalize_h(h)
    dx = 2.0 / n_points
    h1 = 1.0 - h
    n = n_points
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        if k <= 0:
            i = np.arange(-k, n)
            j = i + k
        else:
            i = np.arange(0, n - k)
            j = i + k
        cent = np.dot(h1[j], h[i]) + np.dot(h[j], h1[i])  # ~dCdh = -2(1-h)(1'(x)) -
        grad[i] += (1 - h[i]) * (1 - h1[j]) * dx
        grad[j] += (h[j]) * (1 - h1[i]) * dx
    dh_over_dhpacked = np.identity(n)
    return grad.flatten().dot(dh_over_dhpacked).flatten()

def trust_region_update(h_flat: np.ndarray, grad_flat: np.ndarray, trust_radius: float, **kwargs):
    def f(packed: np.ndarray):
        return objective(packed)

    result = opt.minimize(
        fun=f,
        x0=h_flat,
        method='L-BFGS-B',
        jac=grad_flat,
        bounds=[(0, 1)] * len(h_flat)
    )
    h_new = result.x.clip(0, 1)
    return h_new

def run(seed=42, budget_s=1100, n_points=47, trust_radius=0.05):
    global initial_h_values
    if initial_h_values is not None:
        h0 = initial_h_values.copy()
    else:
        h0 = np.zeros(n_points)
        h0[: h0.size // 2] = 1.0  # Half the domain set to 1

    h0 = h0.reshape(-1)
    h0 = normalize_h(h0)
    best_h = h0.copy()
    best_c5 = evaluate_objective(h0, 2.0 / n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        try:
            h_flat = best_h.copy()
            grad_flat = gradient(h_flat).flatten()
            new_h_flat = trust_region_update(h_flat, grad_flat, trust_radius)
            h = new_h_flat.reshape(n_points)
            h = normalize_h(h)
            dx = 2.0 / n_points
            current_c5 = evaluate_objective(h, dx)
            if current_c5 < best_c5:
                best_h = new_h_flat
                best_c5 = current_c5
        except Exception as e:
            break

    h_values = best_h.reshape(n_points)
    return (h_values.tolist(), best_c5, n_points)
