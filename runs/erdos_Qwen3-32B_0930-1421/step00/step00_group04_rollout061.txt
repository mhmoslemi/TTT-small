import numpy as np
import time
import scipy.signal
from math import isclose


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    t_max = t_start + budget_s

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        raise ValueError("initial_h_values missing.")

    h = np.clip(h, 0, 1)

    def rescale_integral(h):
        dx = 2.0 / len(h)
        current = np.sum(h) * dx
        return h * (1.0 / current)

    def evaluate_c5(h):
        dx = 2.0 / len(h)
        ones_h = 1 - h
        corr = scipy.signal.correlate(h, ones_h, mode='full')
        return np.max(corr) * dx

    def step_func(h, grad, alpha):
        h_new = h - alpha * grad  # Gradient step
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain integral 1.0
        return rescale_integral(h_new)

    # Learn how to adjust simplexes in search space
    def simplex_operation(h_list, f_values, c_idx):
        n = len(h_list)
        imax = np.argmax(f_values)
        x_bar = np.mean(np.delete(h_list, imax, axis=0), axis=0)
        x_refl = x_bar + (x_bar - h_list[imax]) * 1.0
        x_refl = np.clip(x_refl, 0, 1)
        x_refl = rescale_integral(x_refl)
        return x_refl

    # Gradient update (vector)
    def compute_grad(h):
        n = len(h)
        dx = 2.0 / n

        # Compute correlations
        one_minus_h = 1 - h
        corr = scipy.signal.correlate(h, one_minus_h, mode='full')
        int_corr = corr * dx
        k_max = np.unravel_index(np.argmax(int_corr, axis=None), int_corr.shape)[0]

        # Build Phi
        def reshape_and_expand(v):
            return np.concatenate([v, v])  # Extend for full convolution
        h_exp = reshape_and_expand(h)
        one_h_exp = reshape_and_expand(one_minus_h)

        Phi = scipy.signal.correlate(h_exp, one_h_exp, mode='full', method='direct')
        Phi = Phi[k_max - n + 1: k_max + 1]
        dPhi = 2 * h - 1  # Derivative of h(1 - h)
        grad = 2 * np.sum(np.reshape(Phi, (n, n)) * dPhi, axis=1) * dx / n

        # Normalize by number of terms
        grad /= (2 * n)
        return grad - 2 * np.sum(grad) * dx / n  # Residual drift fix

    # Optimization loop with simplex restarts
    num_iterations = 0
    best_c5 = np.inf
    current_best = h.copy()
    current_iter = 0
    learn_rate = 0.01

    # Main loop
    while time.time() < t_max:
        # Compute gradient and update h
        grad = compute_grad(h)
        h = step_func(h, grad, learn_rate)
        c5 = evaluate_c5(h)
        num_iterations += 1

        if c5 < best_c5:
            current_best = h.copy()
            best_c5 = c5
            current_iter = 0
        else:
            current_iter += 1

        if num_iterations % 50 == 0:
            # Restart with simplex update if not improving
            h_list = [current_best]
            for _ in range(2):
                h2 = current_best + (np.random.rand(len(current_best)) - 0.5) * 0.1
                h2 = np.clip(h2, 0, 1)
                h2 = rescale_integral(h2)
                h_list.append(h2)

            f_values = [evaluate_c5(hx) for hx in h_list]
            x_refl = simplex_operation(h_list, f_values, 1)

            h = x_refl.copy()
            learn_rate *= 0.5

        if time.time() > t_max:
            break

    return current_best, best_c5, n_points
