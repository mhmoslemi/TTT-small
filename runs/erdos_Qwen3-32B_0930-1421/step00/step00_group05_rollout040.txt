import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float).copy()
    h /= (h.sum() * dx)  # ensure normalization
    h *= n_points / 2  # Projects back to integral constraint

    # Ensure the integral is correct
    h_sum = h.sum()
    if h_sum == 0:
        h[:] = 0.5
    else:
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

    def compute_C5(h):
        return np.max(correlate(h, 1 - h, mode="full") * dx)

    best_h = h.copy()
    best_c5 = compute_C5(h)
    start_time = time.time()

    max_iter = 100000
    iter_step = 0
    learning_rate = 0.1
    tolerance = 1e-4
    momentum = 0.9
    velocity = np.zeros_like(h)
    grad_history = []

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iter_step >= max_iter:
            break

        iter_step += 1

        if iter_step % 100 == 0:
            current_c5 = compute_C5(h)
            if current_c5 < best_c5 - 1e-5:
                best_c5 = current_c5
                best_h = h.copy()
                learning_rate = min(learning_rate * 1.05, 0.2)
            elif current_c5 >= best_c5 + 1e-5:
                h = best_h.copy()
                learning_rate = max(learning_rate * 0.5, 0.01)
                velocity.fill(0)

        corr = correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        # Compute gradient ∇C at k = k_max
        # Gradient is 2 * (h - 2h^2 + h^3 - 1) at shifted indices
        dh = np.zeros_like(h)
        if k_max >= 0 and k_max < n_points:
            dh[k_max:] -= h[:-k_max]
            dh = dh + dh[::-1]
        else:
            # If k_max is out of bounds, reflect
            pass

        # Compute gradient of overlap at k_max
        grad = - (2 * h - 4 * h**2 + 2 * h**3 - 1)
        if k_max >= 0 and (k_max + n_points) <= len(grad):
            grad_shifted = grad[k_max : k_max + n_points]
        else:
            pad_width = n_points
            grad_padded = np.pad(grad, (pad_width, pad_width), mode='reflect')
            low = k_max + pad_width
            high = low + n_points
            grad_shifted = grad_padded[low : high]

        velocity = momentum * velocity + (1 - momentum) * grad_shifted
        h_update = h - learning_rate * velocity
        h_update = np.clip(h_update, 0, 1)
        h_update_sum = h_update.sum()
        h = h_update * (n_points / 2) / h_update_sum

        current_c5 = compute_C5(h)
        if current_c5 < best_c5 - 1e-7:
            best_c5 = current_c5
            best_h = h.copy()

    final_c5 = compute_C5(best_h)

    return (best_h, final_c5, n_points)
