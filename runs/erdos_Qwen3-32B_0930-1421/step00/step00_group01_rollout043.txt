import numpy as np
from scipy.optimize import minimize, Bounds
import evaluate_erdos_solution as ev
from functools import partial

def run(seed=42, budget_s=1100):
    start_time = np.datetime64("now")
    np.random.seed(seed)

    # Load the initial construction
    h_initial = np.array(initial_h_values)

    n_points = h_initial.size
    dx = 2.0 / n_points
    sum_h = n_points / 2.0

    def objective(h, return_grads=False):
        h = h.reshape(-1)
        c5_array = []
        for k in range(2 * n_points):
            sl0 = np.s_[:]
            sl1 = np.s_[:]
            if k < n_points:
                sl1 = np.s_[k:]
                sl0 = np.s_[:-k]
            elif k >= n_points and k < 2 * n_points:
                sl0 = np.s_[:-k + n_points]
                sl1 = np.s_[k - n_points:]
            overlap = np.sum(h[sl0] * (1 - h[sl1])) * dx
            c5_array.append(overlap)
        c5_current = np.max(c5_array)
        if return_grads:
            # Gradient computation via perturbation (finite differences)
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(len(h)):
                h_p = h.copy()
                h_m = h.copy()
                h_p[i] = np.clip(h[i] + eps, 0, 1)
                h_m[i] = np.clip(h[i] - eps, 0, 1)
                grad[i] = (objective(h_p) - objective(h_m)) / (2 * eps)
            return c5_current, grad
        return c5_current

    def constraint_sum(h):
        return np.sum(h) - sum_h

    def clip_and_renormalize(x):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        if s == 0:
            return h_initial
        x_clipped *= sum_h / s
        return x_clipped

    # Bounds and constraints for scipy.optimize
    bounds = Bounds(0, 1)
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    constraints = [cons_sum]

    # Set up a budget-aware optimization loop
    result = None
    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    for iter in range(1000):
        time_remaining = np.datetime64("now") - start_time < np.timedelta64(budget_s, "s")
        if not time_remaining:
            break

        # Objective and callback for minimize
        def objective_wrap(x):
            val = objective(x)
            global best_c5, best_h
            if val < best_c5:
                best_c5 = val
                best_h = x.copy()
            return val

        result = minimize(
            objective_wrap,
            x0=h_initial,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-6, 'maxiter': 1}
        )

        h_initial = clip_and_renormalize(result.x)

    c5_bound = objective(best_h)
    h_values = best_h.tolist()
    return h_values, c5_bound, n_points
