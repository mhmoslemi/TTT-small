import numpy as np
import time
import random
from math import exp

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)
    
    t0 = time.time()
    max_time = budget_s
    
    # Use initial_h_values if available, or define a simple starting point
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        n_points = 80
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    n_points = len(h)
    dx = 2.0 / n_points
    h = (h > 0.5).astype(np.int64)  # force binary
    norm = np.sum(h)
    if not (abs(norm - n_points / 2) < 1e-10):
        # Ensure exact normalization
        diff = n_points // 2 - norm
        if diff > 0:
            # Need to add ones
            ones_idx = np.where(h == 0)[0]
            sample = np.random.choice(ones_idx, diff, replace=False)
            h[sample] = 1
        else:
            # Need to remove ones
            ones_idx = np.where(h == 1)[0]
            sample = np.random.choice(ones_idx, -diff, replace=False)
            h[sample] = 0

    temperature = 1.0
    T_min = 1e-5
    decay = 0.999
    no_improve = 0
    best_h = h.copy()
    best_c5 = float('inf')

    def compute_overlap(h):
        dx = 2.0 / len(h)
        h_vals = (h > 0.5).astype(np.float64)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr * dx)

    while time.time() - t0 < max_time:
        # Flip two bits - one 1 -> 0, one 0 -> 1
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i0 = random.choice(ones_idx)
        i1 = random.choice(zeros_idx)
        h[i0], h[i1] = h[i1], h[i0]

        # Check normalization is maintained
        if not (abs(np.sum(h) - n_points / 2) < 1e-10):
            # Revert
            h[i0], h[i1] = h[i1], h[i0]
            continue

        # Compute c5
        c5 = compute_overlap(h)

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            # Annealing acceptance
            if random.random() >= exp((best_c5 - c5) / temperature):
                # Reject - revert
                h[i0], h[i1] = h[i1], h[i0]
            no_improve += 1

        # Cool down
        temperature *= decay
        if temperature < T_min:
            break

        # Mechanical reset to escape plateaus
        if no_improve > 1000:
            temperature = max(temperature * 1.5, T_min)
            no_improve = 0

    h = best_h
    c5 = compute_overlap(h)

    # Final check
    h = (h > 0.5).astype(np.float64)  # Ensure in [0, 1]
    assert abs(np.sum(h) - n_points / 2) < 1e-10
    return h.tolist(), c5, n_points
