import numpy as np
import time
import random
from evaluate_erdos_solution import evaluate_erdos_solution

def normalize_h(n_points, h):
    total = 0.5 * n_points
    current_sum = h.sum()
    scale_factor = total / current_sum
    h = np.clip(h * scale_factor, 0, 1)
    correction = total - h.sum()
    if abs(correction) > 1e-8:
        for i in np.random.permutation(n_points):
            if h[i] > 1e-8 and h[i] < 1 - 1e-8:
                adjusted = h[i] + correction
                if 0 <= adjusted <= 1:
                    h[i] = adjusted
                    break
    return h

def evaluate_c5(n_points, h):
    dx = 2.0 / n_points
    h = h.astype(np.float64)
    ones_minus_h = 1.0 - h
    c5 = np.max(np.correlate(h, ones_minus_h, mode="full") * dx)
    return c5

def mutate(h, mutation_rate=0.03):
    h = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, int(n_points * mutation_rate))
    for i in indices:
        h[i] = np.random.uniform(0, 1)
    return normalize_h(n_points, h)

def crossover(h1, h2, split_frac=(0.25, 0.75)):
    h1 = h1.copy()
    h2 = h2.copy()
    n_points = len(h1)
    split = int(random.uniform(*split_frac) * n_points)
    child = np.concatenate((h1[:split], h2[split:]))
    return normalize_h(n_points, child)

def initialize_population(n_points, initial_h, population_size=50):
    population = [normalize_h(n_points, initial_h.copy())]
    for _ in range(population_size - 1):
        h = normalize_h(n_points, initial_h + np.random.normal(0, 0.1, n_points))
        population.append(h)
    return population

def run(seed=42, budget_s=1100, n_points=47, population_size=50, max_generations=1000, mutation_rate=0.03):
    np.random.seed(seed)
    random.seed(seed)

    for gen in range(max_generations):
        population = sorted(population, key=lambda h: evaluate_c5(n_points, h))
        if time.time() - start_time > budget_s - 10:
            break
        current_best = population[0]
        current_c5 = evaluate_c5(n_points, current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5

        top_idx = int(0.2 * population_size)
        top_individuals = population[:top_idx]
        new_population = top_individuals.copy()

        # Breeding
        for _ in range(top_idx, population_size):
            parent1 = random.choice(top_individuals)
            parent2 = random.choice(top_individuals)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)

        population = new_population

    return best_h, best_c5, n_points
