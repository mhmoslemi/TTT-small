        c5 = evaluate_erdos_solution(h, n_points=n_points)

            best_c5 = c5

        # Compute gradient
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx

        max_idx = np.argmax(corr)

        # Construct a simple gradient (preferred shift minus unpreferred shift)
        grad = np.zeros_like(h)
        shift = max_idx - n_points + 1
        if shift < 0:
            idx = np.arange(-shift, n_points)
            grad[idx] += (h[idx + shift] - h[idx])
        else:
            idx = np.arange(n_points - shift)
            grad[idx] += (h[idx] - h[idx + shift])

        # Gradient step
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Project to constraint set

        # Normalize to satisfy the integral constraint
        h_sum = h.sum()
        if abs(h_sum) > 1e-10:
            h *= (n_points / 2) / h_sum

        # Early stop if no improvement
        if iter % patience == 0:
            current_c5 = evaluate_erdos_solution(h, n_points=n_points)
            if current_c5 >= best_c5:
                break

    return (best_h.tolist(), best_c5, n_points)
