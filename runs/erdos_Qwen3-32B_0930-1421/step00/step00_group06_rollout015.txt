import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        assert h.ndim == 1 and h.size > 0
    else:
        n_points = 95
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = np.clip(h, 0, 1)
        h = h / np.mean(h) * (n_points / 2)

    h = np.clip(h, 0, 1)
    h /= np.sum(h) * (2 / n_points)
    h *= (n_points / 2)
    h = np.clip(h, 0, 1)

    # Convex parameters
    alpha = 0.5
    max_iterations = 500
    improvement_threshold = 1e-4
    last_c5 = evaluate_erdos_solution(h)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = last_c5

    dx = 2.0 / n_points

    for it in range(max_iterations):
        t_iter = time.time()
        h_shifted = alpha * h + (1 - alpha) * h_prev
        h_shifted = np.clip(h_shifted, 0, 1)

        overlap = np.correlate(h_shifted, 1 - h, mode='full') * dx
        k_max = np.argmax(overlap)

        # Formulate and solve convex program
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var <= 1, cp.sum(h_var) == n_points / 2]
        objective = cp.Minimize(
            cp.sum(
                h_shifted[k_max:] * (1 - h_var[:n_points - k_max]) +
                h_shifted[:n_points - k_max] * (1 - h_var[k_max:])
            ) * dx
        )
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.clip(np.array(h_var.value).flatten(), 0, 1)
                h_new /= np.sum(h_new) * (2 / n_points)
                h_new *= (n_points / 2)
                h_new = np.clip(h_new, 0, 1)
                curr_c5 = evaluate_erdos_solution(h_new)
                if curr_c5 < last_c5 - improvement_threshold:
                    last_c5 = curr_c5
                    best_c5 = curr_c5
                    best_h = h_new.copy()
                    alpha *= 0.9  # reduce exploration
                h_prev = h.copy()
                h = h_new.copy()
        except:
            pass

        # Early stopping
        if time.time() - t_start > budget_s - 2:
            break

    return (best_h, best_c5, len(best_h))
