import numpy as np
import cvxpy as cp
import time
import math

from evaluate_erdos_solution import evaluate_erdos_solution
from numpy import correlate


def evaluate(h, dx):
    # Compute overlap integrals via cross-correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # We only care about k in 1 to len(h) - 1
    corr = corr[len(h) - 1 : 2 * len(h) - 1]
    return np.max(corr) * dx


def project(h, n_points):
    # Clip values
    h = np.clip(h, 0, 1)
    # Normalize to constraint: sum(h) == n_points / 2
    total = np.sum(h)
    if total == 0:
        h = np.zeros_like(h)
    else:
        h *= n_points / 2 / total
    h = np.clip(h, 0, 1)
    return h


def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    start = time.time()

    # Use initial_h_values from parent
    h = np.array(initial_h_values)
    n_points = h.shape[0]

    dx = 2.0 / n_points
    best_c5 = evaluate(h, dx)
    best_h = h.copy()

    # Optimization parameter
    max_iter = 1000
    iter_start = 0
    for iter in range(max_iter):
        if time.time() - start >= budget_s:
            break

        # Compute overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[len(h) - 1 : 2 * len(h) - 1]
        k_max = np.argmax(corr)
        worst_overlap = corr[k_max] * dx

        if worst_overlap >= best_c5 - 1e-6:
            # No improvement
            iter_start = iter + 1
            continue

        if worst_overlap < best_c5:
            best_c5 = worst_overlap
            best_h = h.copy()
            print(f"New best C₅: {best_c5:.8f} at iteration {iter}")
        else:
            continue

        # Compute gradient for the worst k
        k = k_max
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k])  # derivative of h[i]*(1-h[i+k])
            grad[i + k] -= h[i]  # derivative of h[i]*(1-h[i+k])
        grad *= dx

        # Optimize h: min ||h' - (h - alpha * grad)||^2
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var - (h - grad ** 2 / (best_c5 + 1e-6))))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            h = h_var.value
            h = project(h, n_points)
        except Exception as e:
            print(f"CVXPY solve failed: {e}")
            h = project(h - 0.1 * grad, n_points)

    # Return best solution found
    final_c5 = evaluate(best_h, dx)
    return (best_h, final_c5, n_points)
