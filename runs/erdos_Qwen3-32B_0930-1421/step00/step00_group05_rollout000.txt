    def integrate(h):
        # Compute overlaps for k = 0, ..., n_points * 2 - 1
        valid_shifts = range(-(n_points//2), n_points//2 + 1)
        c5 = []
        for k in valid_shifts:
            shifted = np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted)) * dx
            c5.append(overlap)
        return np.array(c5)

    def project_h(h):
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize
        mass = np.sum(h)
        h *= n_points / 2.0 / mass
        return h

    # Optimize with gradient descent
    best_h = h.copy()
    best_c5 = np.max(integrate(h))
    
    start_time = time.time()
    beta = 10  # Log-sum-exp parameter
    lr = 0.01  # Learning rate
    iter_no_improve = 0
    
    while time.time() - start_time < budget_s:
        # Compute overlaps
        c5 = integrate(h)
        max_idx = np.argmax(c5)
        max_c5 = c5[max_idx]
        
        # Compute gradient for step function h
        grad = np.zeros_like(h)

            if k == 0:
                grad += np.roll(h, -k) - 2*h + 1
            else:
                term = np.roll(h, -k) - h
                grad += term if k < 0 else -term
        
        # Normalize gradient
        grad = np.clip(grad / beta, -1, 1)
        
        # Gradient step
        h = h - lr * grad
        
        # Project to feasible set
        h = project_h(h)

            if time.time() - start_time > budget_s - 0.1:

        # Adjust log-sum-exp parameter every 100 iterations
        if iter_no_improve % 100 == 0 and beta < 1000:
            beta *= 1.2
