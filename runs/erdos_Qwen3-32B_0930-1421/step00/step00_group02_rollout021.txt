import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

# Global access to the initial solution if available
try:
    from initial_h_values import initial_h_values as _h0
except ImportError:
    _h0 = None

def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values, dtype=np.float64)
    dx = 2.0 / n_points
    max_c5 = 0.0
    for k in range(1, n_points):
        w = np.roll(h, -k)
        overlap = h * (1 - w)
        c5 = np.sum(overlap) * dx
        max_c5 = max(max_c5, c5)
        w = np.roll(h, k)
        overlap = h * (1 - w)
        c5 = np.sum(overlap) * dx
        max_c5 = max(max_c5, c5)
    return max_c5

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use provided construction or fallback to standard initialization
    if _h0 is not None:
        n_points = len(_h0)
        h = np.copy(_h0).astype(np.float64)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    # Constraint: sum(h) == n_points / 2
    def normalize_h(h):
        total = np.sum(h)
        correct_sum = n_points / 2
        if np.isclose(total, correct_sum):
            return h
        scale_factor = (correct_sum) / (total or 1.0)
        return h * scale_factor

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_time = 0.0
    dx = 2.0 / n_points

    # Optimization loop
    while time.time() - start_time < budget_s:

        current_c5 = evaluate_erdos_solution(h, n_points)

        # Store best result
        if current_c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = current_c5
            best_time = time.time() - start_time

        # Compute overlaps for all lags
        corr = correlate(h, 1.0 - h, mode='full')
        valid_corr = corr[n_points - 1 : 2 * n_points - 1]
        c5_values = valid_corr * dx

        max_c5 = np.max(c5_values)
        worst_lag_indices = np.where(np.abs(c5_values - max_c5) < 1e-8)[0]
        worst_lag = worst_lag_indices[0]

        # Update: nudge h values to reduce overlap at worst lag
        step_size = 0.01 / np.sqrt(best_time + 1e-6)  # Adaptive step size
        delta_h = np.zeros_like(h)
        k = worst_lag
        left_shift = (1.0 - h[k:]) * h
        right_shift = (1.0 - np.roll(h, -k))[:(n_points - k)] * h[:(n_points - k)]
        delta_h[k:] += left_shift
        delta_h[:(n_points - k)] -= right_shift

        h = h + step_size * delta_h
        h = np.clip(h, 0.0, 1.0)
        h = normalize_h(h)

        # Early termination
        if time.time() - best_time > budget_s / 2 and current_c5 - best_c5 < 1e-6:
            break

    return best_h, best_c5, n_points
