import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Get initial points
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    assert abs(np.sum(h) - target_sum) < 1e-8

    # Parameters
    lr = 0.1  # Initial learning rate
    max_iter = 50000
    iter_update_print = 1000
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')

    for iter in range(max_iter):
        # Step 1: Compute overlap integrals
        c = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(c)
        
        # Early stopping if time expired
        if time.time() - start_time > budget_s:
            break

        # Step 2: Identify the worst overlap
        k_max = np.argmax(c)
        diff = h.shape[0]
        if k_max < diff:
            shift = k_max
            direction = 1
        else:
            shift = h.shape[0] + k_max - diff
            direction = -1

        # Step 3: Gradient for C_k_max
        gradient = np.zeros_like(h)
        if direction == 1:
            gradient[:h.shape[0] - shift] = 1 - 2 * h[:h.shape[0] - shift]
            gradient[shift:] = -h[shift:]
        else:
            gradient[shift:] = 1 - 2 * h[shift:]
            gradient[:h.shape[0] - shift] = -h[:h.shape[0] - shift]

        # Step 4: Line search with backtracking (in-place)
        alpha = lr
        while True:
            temp_h = h - alpha * gradient
            temp_h = np.clip(temp_h, 0, 1)
            temp_sum = np.sum(temp_h)
            if temp_sum > 1e-8:
                temp_h *= (target_sum / temp_sum)
                break
            alpha /= 2

        h = temp_h

        # Track the best
        c_new = np.correlate(h, 1 - h, mode="full") * dx
        c5_new = np.max(c_new)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new

        # Print progress
        if iter % iter_update_print == 0 or iter == max_iter - 1:
            print(f'Iter: {iter}, C5: {best_c5:.8f}, t: {time.time() - start_time:.2f}s')

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if abs(best_sum - target_sum) > 1e-8:
        best_h *= (target_sum / best_sum)

    return best_h, best_c5, n_points
