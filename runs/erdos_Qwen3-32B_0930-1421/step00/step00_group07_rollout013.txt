import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_overlap(h, dx):
    h_bin = (h > 0.5).astype(float)
    corr = correlate(h_bin, 1 - h_bin, mode='full')
    return float(np.max(corr * dx))

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Initialize from provided parent construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_mass = n_points / 2

    # Convert h to binary and adjust mass
    h_bin = (h > 0.5).astype(float)
    mass_diff = int(np.round(target_mass - np.sum(h_bin)))
    if mass_diff != 0:
        # Find indices to flip
        indices = np.where((h_bin == 0) if mass_diff > 0 else (h_bin == 1))[0]
        np.random.shuffle(indices)
        for i in indices[:abs(mass_diff)]:
            h_bin[i] = 1 - h_bin[i]

    best_h = h_bin.copy()
    best_c5 = evaluate_overlap(best_h, dx)

    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    patience = 1000
    no_improve_counter = 0
    initial_temp_decay = 1.0

    # Configuration
    while time.time() - start_time < budget_s:
        mass_diff = int(np.round(target_mass - np.sum(h_bin)))
        if mass_diff != 0:
            # Find indices to flip if h is invalid (should not happen)
            indices = np.where((h_bin == 0) if mass_diff > 0 else (h_bin == 1))[0]
            np.random.shuffle(indices)
            for i in indices[:abs(mass_diff)]:
                h_bin[i] = 1 - h_bin[i]

        # Generate a candidate by flipping two bits
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break  # Infeasible, cannot flip further
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_candidate = h_bin.copy()
        h_candidate[i] ^= 1
        h_candidate[j] ^= 1

        # Compute new C5 bound
        c5_candidate = evaluate_overlap(h_candidate, dx)
        delta_c5 = c5_candidate - best_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_bin = h_candidate
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        T *= alpha

        # Reduce patience and slow decay if no improvement
        if no_improve_counter >= patience:
            no_improve_counter = 0
            T = max(T * 10, min_T)
            alpha *= 0.9

    # Final validation
    best_h = (best_h > 0.5).astype(float)
    assert np.abs(np.sum(best_h) - target_mass) <= 1e-8, "Feasibility constraint violated"
    final_c5 = evaluate_overlap(best_h, dx)

    return (best_h, final_c5, n_points)
