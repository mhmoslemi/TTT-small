import numpy as np
import cvxpy as cp
import time
import math


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use parent construction
    h0 = initial_h_values.copy()
    n = len(h0)
    dx = 2.0 / n

    # Initial checks and set-up
    h = cp.Parameter(n, nonneg=True, value=h0)
    assert abs(np.sum(h.value) - n / 2) < 1e-8, "Initial h violates sum constraint"

    start_time = time.time()

    # CVXPY variables
    x = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(x) == n / 2, 0 <= x, x <= 1]

    best_h = h.value
    best_c5 = float('inf')
    patience = 0
    MAX_PATIENCE = 20

    while time.time() - start_time < budget_s:
        objective = cp.Maximize(cp.max([cp.sum([x[i] * (1 - x[(i + k) % n])
                                                for i in range(n) if (i + k) < n])
                                        * dx for k in range(1, n // 2 + 1)]))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status != 'optimal':
            break

        current_h = x.value
        current_c5 = evaluate_erdos_solution(n, current_h)['c5_value']

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            patience = 0
        else:
            patience += 1

        if patience >= MAX_PATIENCE:
            break

    return (best_h, best_c5, n)
