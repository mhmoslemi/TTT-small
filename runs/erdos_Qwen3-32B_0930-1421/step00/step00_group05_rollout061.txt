        widths = 0.02 * n_points

            return np.maximum(0, 1 - np.abs(x - c) / widths)
        return np.stack([triangular_basis(np.arange(n_points)*dx + dx/2, c) for c in centers], axis=1)

    C5 = cp.max(ones_h[ :n_points] @ Phi[:,:n_points] * dx)

    objective = cp.Minimize(C5)
    problem = cp.Problem(objective, constraints)
    
    t_elapsed = time.time() - t_start
    if t_elapsed < budget_s:
        problem.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - t_elapsed)
