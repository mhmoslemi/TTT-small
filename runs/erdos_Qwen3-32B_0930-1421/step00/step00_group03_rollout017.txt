import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / n_points
    h = np.array(initial_h_values)

    def clamp_and_project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        h_vals = np.clip(h, 0, 1)
        one_minus_h = 1.0 - h_vals
        full_corr = correlate(h_vals, one_minus_h, mode="full")
        c5 = np.max(np.abs(full_corr[full_corr.size // 2:])) * dx
        return c5

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    patience = 0
    max_patience = 100
    lr = 0.01

    while time.time() - start_time < budget_s:
        gradient = np.zeros_like(h)
        h_vals = np.clip(h, 0, 1)
        one_minus_h = 1.0 - h_vals

        # Manually perform gradient for dominant C₅ term via brute force estimate
        full_corr = correlate(h_vals, one_minus_h, mode="full")
        c5_idx = np.argmax(np.abs(full_corr[full_corr.size // 2:])) + full_corr.size // 2
        shift = c5_idx - h_vals.size + 1
        for i in range(max(0, shift), min(h_vals.size, h_vals.size - shift)):
            gradient[i] -= 2 * (h_vals[i] - h_vals[i - shift]) * dx

        h += lr * gradient
        h = clamp_and_project(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            break

    return best_h.tolist(), best_c5, n_points
