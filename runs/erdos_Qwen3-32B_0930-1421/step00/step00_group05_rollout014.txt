import numpy as np
import cvxpy as cp
import time

def construct_basis_functions(n_points, n_basis=100):
    basis_x = np.linspace(0, 2, n_points)
    w_basis = 2.0 / n_basis
    basis_widths = np.full(n_basis, w_basis)
    centers = np.linspace(0, 2 - w_basis, n_basis)
    Phi = np.zeros((n_points, n_basis))
    for j in range(n_basis):
        center = centers[j]
        for i in range(n_points):
            dx = abs(basis_x[i] - center)
            phi = max(0.0, 1.0 - dx / w_basis)
            Phi[i, j] = phi
    return Phi

def evaluate_C5(h, dx):
    h_array = np.concatenate([h, [0, 0, 0, 0]])  # pad for full convolution
    h1_h = h_array[:-4]
    h2_h = h_array[1:-3]
    h3_h = h_array[2:-2]
    h4_h = h_array[3:-1]
    h5_h = h_array[4:]
    
    h_vals = np.stack([h1_h, h2_h, h3_h, h4_h, h5_h])
    overlap = np.sum(h_vals * (1 - h_vals), axis=1) * dx
    c5 = np.max(overlap)
    return c5
