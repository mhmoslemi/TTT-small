import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate


def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode='full') * dx
    return np.max(overlap)


def run(seed=42, budget_s=1100, verbose=False, gamma=100, **kwargs):
    np.random.seed(seed)

        overlap = correlate(h, 1 - h, mode='full')
        overlap[:n_points] = -np.inf
        k_star = np.argmax(overlap)
        k = k_star

        # Compute gradients at position shift k
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] -= h[i + k]  # derivative w.r.t h[i]
            grad[i + k] += h[i]  # derivative w.r.t h[i + k]

        # Form objective and constraints for convex projection
        new_h = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h - new_h - grad / gamma))
        constraints = [
            new_h >= 0,
            new_h <= 1,
            cp.sum(new_h) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)

        if prob.status != 'optimal':
            if verbose:
                print("Projection failed, reusing previous h.")
            break

        h = new_h.value.copy()

        # Ensure h is within bounds and satisfies the constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum

        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if verbose and (time.time() - t0) % 20 < 1:
            print(f"Iteration: {int((time.time()-t0)/1)}s, best C5: {best_c5:.5f}")

    return best_h, best_c5, n_points
