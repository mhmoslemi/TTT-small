def evaluate_erdos_solution(h, dx):

    return C5

        grad = np.zeros_like(h)
        mask_start = max(0, -k)
        mask_end = min(n_points, n_points - k)
        grad[mask_start:mask_end] -= rhs
        grad[max(0, -k):] += h[:n_points - max(0, k)]
        grad *= np.sign(corr[max_index])

        h_flat = h.flatten()
        grad_flat = grad.flatten()
        h_flat -= alpha * grad_flat

        # Projection back into [0, 1]
        np.clip(h_flat, 0, 1, out=h_flat)
        total = np.sum(h_flat)
        if abs(total - n_points / 2) > 1e-6:
            h_flat = h_flat / total * (n_points / 2)

        C5 = evaluate_erdos_solution(h, dx)

        if C5 < best_C5:

            best_h = h.copy()

        if alpha > 1e-5:
