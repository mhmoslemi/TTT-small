import numpy as np
import time
from scipy.fft import fft, ifft, fftshift
from scipy.optimize import minimize


def evaluate_overlap(h, dx):
    # Compute 1 - h with cyclic padding to match full convolution output
    h1m = 1 - h
    h_fft = fft(h)
    h1m_fft = fft(h1m)
    corr_fft = h_fft * np.conj(h1m_fft)
    corr = ifft(corr_fft).real * dx
    # We only care about overlaps in range where it is meaningful (0 to 2 - shift)
    c5 = np.max(np.abs(corr[n_points // 2 : 3 * n_points // 2]))
    return np.clip(c5, 0, 1)


def proj_simplex(v):
    # Projects a vector onto the simplex ∑x=1 and x ∈ [0,1]
    def max_pool():
        np.clip(v, 0, 1, out=v)
        total = v.sum()
        v *= (1 + (1 - total) / total)
        v[np.isnan(v)] = 0
        return v

    return max_pool()


def cost(h_flat):
    h = h_flat.reshape((n_points,))
    h = proj_simplex(h)
    c5 = evaluate_overlap(h, dx)
    return c5


def run(seed=42, budget_s=1100, verbose=False):
    global n_points, dx

    np.random.seed(seed)

    # Use parent h if provided
    if "initial_h_values" in globals():
        h0 = initial_h_values
    else:
        # Create a uniform distribution
        h0 = np.ones(n_points) / n_points
        h0 *= n_points / 2  # Scale to sum to n_points / 2

    # Determine n_points adaptively but keep under 800 for portability
    n_points = min(800, 2 * len(h0))
    dx = 2.0 / n_points

    h0 = h0[:n_points]
    h0 = h0 / h0.sum() * (n_points / 2)  # Normalize to desired integral

    start = time.time()
    best_cost = np.inf
    best_h = h0.copy()

    def cb(xk):
        nonlocal best_cost, best_h
        current_cost = cost(xk)
        nonlocal start
        elapsed = time.time() - start
        if current_cost < best_cost:
            best_cost = current_cost
            best_h[:] = xk.copy()
            print(f"Iteration {i} - c5={best_cost:.6f}, elapsed={elapsed:.1f}s")
        return elapsed < budget_s

    # Run optimization
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h0.copy()
    result = minimize(
        cost,
        x0,
        bounds=bounds,
        method="L-BFGS-B",
        callback=cb,
        options={"ftol": 1e-16, "maxfun": 1000},
    )

    h_opt = proj_simplex(best_h)
    c5_bound = evaluate_overlap(h_opt, dx)
    return (h_opt.tolist(), c5_bound, n_points)
