import numpy as np
import time
from scipy.signal import correlate
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    h = initial_h_values.copy()
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5

    dx = 2.0 / n_points
    idxs = np.arange(n_points)
    shifts = np.arange(-n_points + 1, n_points)

    # Feasibility constraints
    def project_to_feasible(h):
        np.clip(h, 0, 1, out=h)
        h *= n_points / 2 / h.sum()

    learning_rate = 0.01
    patience = 50
    no_improvement_count = 0

    max_iter = 10000
    iter_ = 0

    while time.time() - start_time < budget_s and iter_ < max_iter:
        iter_ += 1

        # Compute gradients via finite differences (numerical)
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(h.size):
            h[i] += eps
            hp = h.copy()
            project_to_feasible(hp)
            c5p = evaluate_erdos_solution(hp)

            h[i] -= 2 * eps
            hm = h.copy()
            project_to_feasible(hm)
            c5m = evaluate_erdos_solution(hm)

            grad[i] = (c5p - c5m) / (2 * eps)
            h[i] += eps  # Restore

        # Projected gradient descent
        h -= learning_rate * grad
        project_to_feasible(h)

        current_c5 = evaluate_erdos_solution(h)

        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Learning rate decay
        if no_improvement_count > patience:
            learning_rate *= 0.9
            if learning_rate < 1e-5:
                break
            no_improvement_count = 0

    # Final validation
    project_to_feasible(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
