import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = h
    h2 = 1 - h
    corr = np.correlate(h1, h2, mode='full')
    return np.max(corr * dx)

def objective(h, n_points):
    return compute_c5(h, n_points)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2

def bounds_and_constraints(n_points):
    bounds = [(0, 1)] * n_points
    cons = [
        {'type': 'eq', 'fun': lambda h: constraint_sum(h, n_points)}
    ]
    return bounds, cons

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # Use the initial construction's sample count

    def objective_wrapper(x):

    bounds, cons = bounds_and_constraints(n_points)

    # Initial point has already satisfied the constraints: ensure it does
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)

    # Optimization
    result = minimize(
        objective_wrapper,
        h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-6}
    )

    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)  # Enforce sum constraint once more

    c5_bound = objective(h_opt, n_points)

    best_time = time.time() - start_time

    return h_opt.tolist(), c5_bound, n_points
