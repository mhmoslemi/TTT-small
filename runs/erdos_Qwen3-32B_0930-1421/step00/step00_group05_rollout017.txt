import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)

    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64).copy()
    h /= h.sum() * dx
    h *= n_points / 2.0
    h = np.clip(h, 0.0, 1.0)

    def projected_grad_desc(h, eps=0.01, max_iter=5000, lr=0.01, beta=10.0):
        best_h = h.copy()
        best_c5 = evaluate(h)
        alpha = lr
        iter_since_improve = 0
        for t in range(max_iter):
            if time.time() - t_start > budget_s:
                break

            C_k = np.zeros(n_points + 1)
            for k in range(n_points + 1):
                if k < len(h):
                    aligned_right = np.roll(h, -k)
                else:
                    aligned_right = np.zeros_like(h)
                C_k[k] = (h * (1.0 - aligned_right)).sum() * dx

            C_k_valid = C_k[1:]  # Skip zero shift
            c5 = np.max(C_k_valid)

            if t % 100 == 0:
                print(f"Iter {t}: c5={c5}")
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    iter_since_improve = 0
                else:
                    iter_since_improve += 1

            if iter_since_improve > 50:
                alpha *= 0.5
                iter_since_improve = 0

            grads = np.zeros_like(h)
            for k in range(1, len(C_k)):
                if abs(C_k[k] - best_c5) < 1e-6:
                    delta = C_k[k] - c5
                else:
                    delta = -h * (1 - np.roll(h, -k))
                grads -= delta

            h -= alpha * grads
            h = np.clip(h, 0.0, 1.0)
            h = h / h.sum() * n_points / 2.0
            if beta < 100:
                beta = min(beta * 1.01, 100.0)

        return best_h, best_c5

    def evaluate(hv):
        C_k = np.zeros(n_points + 1)
        for k in range(1, n_points + 1):
            aligned_right = np.roll(hv, -k) if k <= n_points else np.zeros_like(hv)
            C_k[k] = (hv * (1.0 - aligned_right)).sum() * (2.0 / n_points)
        return np.max(C_k[1:])

    h, c5 = projected_grad_desc(h, lr=0.01, beta=10.0)

    return h.tolist(), c5, n_points
