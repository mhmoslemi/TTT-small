import numpy as np
import scipy.optimize
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time()
    
    # Extract the initial solution
    n_points = len(initial_h_values)
    h0 = np.array(initial_h_values, copy=True)
    
    dx = 2.0 / n_points

    def evaluate(h):
        total_integral = np.sum(h)
        if not np.isclose(total_integral, n_points / 2, atol=1e-6):
            return np.inf
        
        # Compute the shifted overlaps
        corr_values = []
        for k in range(-n_points + 1, n_points):
            if -1 < k < n_points:
                if k > 0:
                    overlap = np.sum(h[:n_points - k] * (1 - h[k:]))
                elif k < 0:
                    k = abs(k)
                    overlap = np.sum(h[k:] * (1 - h[:-k]))
                else:
                    overlap = 0.0
                corr = overlap * dx
                corr_values.append(corr)
        return np.max(np.abs(corr_values))

    # Define constraints and bounds
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # COBYLA optimization
    lb = np.zeros(n_points)
    ub = np.ones(n_points)
    bounds = list(zip(lb, ub))
    constraint = {'type': 'eq', 'fun': constraint_sum}

    def objective(h_flat):
        h = np.clip(h_flat, 0, 1)
        return evaluate(h)

    result = scipy.optimize.minimize(
        objective,
        h0,
        method='COBYLA',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 10000, 'rhobeg': 0.1, 'tol': 1e-5}
    )
    
    h_opt = np.clip(result.x, 0, 1)
    
    # Post-process to ensure integral constraint
    h_opt /= np.sum(h_opt) * 2.0 * dx
    h_opt = np.clip(h_opt, 0, 1)
    
    c5_bound = evaluate(h_opt)
    elapsed = time() - t_start
    
    if elapsed < budget_s and result.fun < c5_bound:
        h_opt = result.x
        c5_bound = evaluate(h_opt)
    
    return h_opt, c5_bound, n_points
