```python
import numpy as np
import cvxpy as cp
from time import time
import scipy.signal

dx_rate = 0.016  # Rough estimate for 125 samples
n_points = int(2 / dx_rate)  # => 125 samples
dx = 2.0 / n_points

def evaluate_c5(h):
    """Compute the Erdos constant C5 for a step function h"""
    h = np.array(h).flatten()
    ones_min_h = 1.0 - h
    corr = scipy.signal.correlate(h, ones_min_h, mode='full')
    result = np.max(corr * dx)
    return result

def evaluate_and_project(h):
    """Evaluate C5 and project back to feasible set"""
    h = np.clip(h, 0, 1)  # clip to [0,1]
    # Enforce the integral constraint (area = 1)
    current_sum = np.sum(h)
    required_sum = n_points / 2  # since 2 * h(x) dx = 1 => sum(h) = n_points / 2
    if not np.allclose(current_sum, required_sum):
        scaling = required_sum / current_sum
        h = np.clip(h * scaling, 0, 1)
    return h

def run(seed=42, budget_s=1100):
    start_time = time()
    best_h = None
    best_c5 = np.inf

    # Heuristic starting point: a discretized version of best known construction
    if 'initial_h_values' in globals():
        global initial_h_values
        # Interpolate initial_h_values to match the new n_points
        old_n = len(initial_h_values)
        if old_n != n_points:
            x_old = np.linspace(0, 2, old_n)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, initial_h_values)
            # Normalize to satisfy the constraint
            h = evaluate_and_project(h)
        else:
            h = np.array(initial_h_values).copy()
    else:
        # Create a reasonable starting vector: random with a reasonable mean
        np.random.seed(seed)
        h = np.random.uniform(0.3, 0.7, size=n_points)
        h = evaluate_and_project(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    improvement_count = 0
    improvement_max = 50
    improvement_timeout = 10.0  # seconds before forcing perturbation if no improvement

    while time() - start_time < budget_s - 5:  # leave buffer time
        # Create a CVXPY variable for the optimization
        h_var = cp.Variable(n_points)

        # Define the objective based on top-k overlap integrals
        ones = np.ones(n_points)
        for k in [0, 1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 20, -20]:
            if 0 <= k < n_points:
                overlap = cp.sum(cp.multiply(h_var, 1 - h_var[k:]))
                cp.MaxExpression([overlap])
            if -n_points < k < 0:
                overlap = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:n_points + k]))
                cp.MaxExpression([overlap])

        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        # Form and solve problem
        problem = cp.Problem(cp.Minimize(cp.Max(ones)), constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(h_var.value, 0, 1)
            new_h = evaluate_and_project(new_h)
            new_c5 = evaluate_c5(new_h)

            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                improvement_count = 0
                print(f"Found better solution: C5 = {best_c5:.7f}")
            else:
                improvement_count += 1
                if improvement_count >= improvement_max or time() - improvement_timeout > start_time:
                    # Perturb the current best solution
                    perturbation = 0.1 * (2 * np.random.rand(n_points) - 1)
                    best_h = evaluate_and_project(best_h + perturbation)
                    print(f"No improvement for {improvement_count} iterations - perturbing with C5={best_c5:.7f}")
                    improvement_count = 0
        else:
            # If solver fails, perturb current solution
            perturbation = 0.1 * (2 * np.random.rand(n_points) - 1)
            best_h = evaluate_and_project(best_h + perturbation)
            improvement_count += 1
        
        if time() - start_time > budget_s - 2:
            break

    # Final projection to feasible set
    best_h = evaluate_and_project(best_h)
    final_c5 = evaluate_c5(best_h)

    return best_h, final_c5, n_points
```
