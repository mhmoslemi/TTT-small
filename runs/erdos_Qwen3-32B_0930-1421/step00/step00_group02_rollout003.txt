import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=400):
    np.random.seed(seed)
    t_start = time.time()

    # Step 0: Initial setup
    dx = 2.0 / n_points
    h_current = np.random.rand(n_points)
    h_current = h_current / h_current.sum() * n_points / 2  # normalize to meet constraint
    h_best = h_current.copy()
    c5_best = evaluate_erdos_solution(h_best, dx)[0]

    h_init = initial_h_values
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    if n_points != n_init:
        h_current = np.interp(np.arange(n_points) * dx, np.arange(n_init) * dx_init, h_init)
        h_current = h_current / h_current.sum() * n_points / 2  # normalize again
    h_best = h_current.copy()
    c5_best = evaluate_erdos_solution(h_best, dx)[0]

    # Step 1: Optimization loop
    iter = 0
    while time.time() - t_start < budget_s - 5:
        iter += 1
        h = cp.Variable(n_points)
        obj = []
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                lhs = h[0 : n_points - k]
                rhs = 1 - h[k : n_points]
            else:
                lhs = 1 - h[-k : n_points]
                rhs = h[0 : n_points + k]
            obj.append(cp.sum(lhs * rhs) * dx)
        max_obj = cp.Variable()
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2, max_obj >= cp.Max(obj)]
        problem = cp.Problem(cp.Minimize(max_obj), constraints)
        problem.solve(max_iter=100, solver=cp.ECOS, verbose=False)
        h_proposed = h.value
        if h_proposed is None:
            continue

        c5_proposed = max(np.correlate(h_proposed, 1 - h_proposed, mode="full") * dx)
        if c5_proposed < c5_best:
            c5_best = c5_proposed
            h_best = h_proposed.copy()

        # Augment h with a mutation if no improvement is observed
        if iter % 20 == 0:
            # Small random perturbation
            mut = np.random.uniform(-0.01, 0.01, n_points)
            h_current = np.clip(h_best + mut, 0, 1)
            h_current = h_current / h_current.sum() * n_points / 2

        if time.time() - t_start + 20 >= budget_s:
            break

    return h_best, c5_best, n_points
