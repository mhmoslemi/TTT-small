import numpy as np
import time
import scipy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h_initial = np.copy(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h_current = np.copy(h_initial)
    best_h = np.copy(h_current)
    best_c5 = np.inf

    alpha = 0.1  # Initial step size
    beta = 0.7   # Step reduction factor
    max_iter = 200  # Max iterations allowed
    start_time = time.time()

    def evaluate(h):
        # Compute all cross-correlations
        h_ext = np.concatenate([h, h])
        correl = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correl * dx)
        return c5

    def project(h):
        # Clip to bounds
        h = np.clip(h, 0.0, 1.0)
        # Compute sum difference from constraint
        sum_h = np.sum(h)
        diff = target_sum - sum_h
        if abs(diff) < 1e-8:
            return h
        if diff > 0:
            # Need to increase values where possible
            mask = h < 1.0
            num = np.sum(mask)
            if num > 0:
                h[mask] += diff / num
        else:
            # Need to decrease values
            mask = h > 0.0
            num = np.sum(mask)
            if num > 0:
                h[mask] += diff / num
        h = np.clip(h, 0.0, 1.0)  # Final clip after correction
        return h

    iter = 0
    while iter < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        c5 = evaluate(h_current)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h_current)

        # Find the shift k with maximum overlap
        h_ext = np.concatenate([h_current, h_current])
        correl = np.correlate(h_current, 1 - h_current, mode="full")
        k_max = np.argmax(correl)

        # Compute gradient based on k_max
        k_offset = k_max
        grad = np.zeros_like(h_current)
        if 0 <= k_offset < n_points:
            grad = (1 - np.roll(h_current, k_offset)) * dx
        elif k_offset >= n_points and k_offset < 2 * n_points:
            k_offset = k_offset - n_points
            grad = (1 - np.roll(h_current, -k_offset)) * dx
        else:
            continue

        # Gradient descent update
        h_proposed = h_current - alpha * grad
        h_proposed = project(h_proposed)

        c5_proposed = evaluate(h_proposed)
        if not np.isfinite(c5_proposed):
            alpha *= beta
            continue
        if c5_proposed < best_c5:
            h_current = h_proposed
            iter += 1
        else:
            alpha *= beta  # Reduce step size
            if alpha < 1e-8:
                alpha = 0.1  # Reset if too small

    return (best_h.tolist(), best_c5, n_points)
