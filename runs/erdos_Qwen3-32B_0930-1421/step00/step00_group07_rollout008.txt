import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_C5 = float('inf')

    # Gradient descent parameters
    alpha = 0.02  # Initial learning rate
    patience = 0
    max_patience = 200  # Early stopping
    max_iter = 1000
    iter_count = 0

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()
        return np.clip(h, 0, 1)

    def compute_C5(h):
        corr = scipy.signal.correlate(h, 1 - h, mode="full")
        C5 = np.max(corr * dx)
        return C5

    while time.time() - start_time < budget_s and iter_count < max_iter and patience < max_patience:
        iter_count += 1

        corr = scipy.signal.correlate(h, 1 - h, mode="full")
        C5 = corr * dx
        k = np.argmax(C5)

        # Compute gradient only for the worst case shift k
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            if 0 <= i_plus_k < n_points:
                grad[i] += (1 - h[i_plus_k])
            if 0 <= i_minus_k < n_points:
                grad[i] -= h[i_minus_k]

        h -= alpha * grad
        h = normalize_h(h)

        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            alpha = min(alpha * 1.1, 0.1)  # Increase on progress
        else:
            patience += 1
            alpha = max(alpha * 0.9, 1e-5)  # Decrease on no progress

    return best_h, best_C5, n_points
