import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial configuration from parent
    h = np.array(initial_h_values).copy()
    n_original = len(h)
    n_points = n_original
    dx = 2.0 / n_points
    max_overlap = evaluate_erdos_solution(h, dx)

    def intervals_from_h(h, dx):
        positions = []
        accum_length = 0
        for i, val in enumerate(h):
            if val > 0.5:
                if not positions or positions and positions[-1][1] != i:
                    positions.append([i * dx, ])
                else:
                    positions[-1][1] = i * dx
            elif positions:
                positions[-1][1] = i * dx
        return np.array(positions), accum_length

    def h_from_intervals(intervals, n_points):
        h = np.zeros(n_points)
        for a, b in intervals:
            a_idx = int(np.floor(a / dx))
            b_idx = int(np.ceil(b / dx))
            h[a_idx:b_idx] = 1
        # Ensure the total integral is exactly 1.0
        h[h == 0] = 1.0 - (np.sum(h * dx) - 1.0) / np.sum(h == 0)
        return h

    def evaluate_intervals(intervals, dx):
        h = h_from_intervals(intervals, len(intervals) * 4)
        overlaps = evaluate_erdos_solution(h, dx)
        return overlaps

    # Initial interval extraction
    intervals, _ = intervals_from_h(h, dx)
    best_intervals = intervals.copy()
    best_c5 = max_overlap

    T = 1.0
    total_iterations = 0
    no_improvement = 0
    INTVAR = min(2, len(intervals))  # Max simultaneously adjusted intervals
    max_iterations = 100000
    while time.time() - start_time < budget_s and total_iterations < max_iterations and no_improvement < 1000:
        total_iterations += 1

        # Randomly select which interval to tweak
        idx = np.random.randint(len(intervals))
        a, b = intervals[idx]
        a2 = intervals[idx][0]
        b2 = intervals[idx][1]

        # Suggest a new length
        if np.random.rand() < 0.5:
            # Shift start
            delta = np.random.uniform(-0.02, 0.02)
            a_new = a - delta
            a_new = max(0.0, a_new)
            if idx > 0:
                a_new = max(a_new, intervals[idx - 1][1] + 1e-5)
            a_new = min(a_new, b2 - 1e-5)
            intervals[idx] = [a_new, b2]
        else:
            # Shift end
            delta = np.random.uniform(-0.02, 0.02)
            b_new = b + delta
            b_new = min(2.0, b_new)
            if idx < len(intervals) - 1:
                b_new = min(b_new, intervals[idx + 1][0] - 1e-5)
            b_new = max(b_new, a2 + 1e-5)
            intervals[idx] = [a2, b_new]

        # Normalize intervals
        intervals.sort(key=lambda x: x[0])  # Sort by start
        merged = []
        for i in range(len(intervals)):
            if not merged:
                merged.append(intervals[i])
            else:
                last = merged[-1]
                if intervals[i][0] <= last[1] + 1e-5:
                    # Merge overlapping or adjacent intervals
                    new_a = last[0]
                    new_b = max(last[1], intervals[i][1])
                    merged[-1] = [new_a, new_b]
                else:
                    merged.append(intervals[i])
        intervals = np.array(merged)

        # Check for invalid or no change
        if intervals.shape[0] == 0 or intervals.shape[0] > n_points:
            continue

        # Compute new h and evaluate
        new_h = h_from_intervals(intervals, n_points)
        overlaps = evaluate_erdos_solution(new_h, dx)
        new_c5 = max(overlaps)

        # Acceptance via simulated annealing
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
            best_intervals = intervals.copy()
            best_c5 = new_c5
            no_improvement = 0
        else:
            intervals = best_intervals.copy()
            no_improvement += 1

        # Animate temperature
        T *= 0.999
        if T < 1e-6:
            T = 1e-6

    # Final conversion and validation
    h = h_from_intervals(best_intervals, n_points)
    h[h > 1.0] = 1.0
    h[h < 0.0] = 0.0
    sum_h = np.sum(h) * dx
    h -= ((sum_h - 1.0) / np.sum(h > 1e-3)) * (h > 1e-3).astype(float)
    h = np.clip(h, 0.0, 1.0)
    final_overlaps = evaluate_erdos_solution(h, dx)

    print(f"Final C5: {np.max(final_overlaps)}", flush=True)

    return h.tolist(), np.max(final_overlaps), n_points
