import numpy as np
import itertools
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def _hcorr(h, n_points):
    dx = 2.0 / n_points
    h = h.astype(np.float64)
    corr = np.correlate(h, 1 - h, mode='full')
    valid = corr[n_points - 1 : 2 * n_points - 1]
    C = np.max(valid * dx)
    return C

def discretize_intervals(intervals, n_points):
    grid = np.linspace(0, 2, n_points)
    h = np.zeros(n_points, dtype=np.float64)
    for a, b in intervals:
        idx_a = np.searchsorted(grid, a, side='right')
        idx_b = np.searchsorted(grid, b, side='left')
        h[idx_a:idx_b] = 1.0
    return h

def compute_overlap(intervals, k, dx):
    total = 0.0
    shift = k * dx
    for a, b in intervals:
        a_shifted = a + shift
        b_shifted = b + shift
        for a2, b2 in intervals:
            if b_shifted <= a2 or b2 <= a:
                continue  # no overlap
            total += (max(a, a2) - min(b, b2)) if max(a, a2) < min(b, b2) else 0
    return total * dx if total > 0 else 0.0

def interval_overlap_intervals(intervals, k, dx):
    overlaps = []
    for k in range(-n_points + 1, n_points):
        overlaps.append(compute_overlap(intervals, k, dx))
    C5 = np.max(overlaps)
    return C5

def generate_initial_intervals(h_vals):
    grid = np.linspace(0, 2, len(h_vals))
    intervals = []
    i = 0
    while i < len(h_vals):
        if h_vals[i] > 0.5:
            start = grid[i]
            j = i
            while j < len(h_vals) and h_vals[j] > 0.5:
                j += 1
            end = grid[j]
            intervals.append((start, end))
            i = j
        else:
            i += 1
    return intervals

def adjust_interval(intervals, i, delta, dx):
    a, b = intervals[i]
    a_new = a + delta
    b_new = b + delta
    if a_new < 0 or b_new > 2:
        return None  # out of bounds
    # Ensure no overlap
    intervals_new = [interval for j, interval in enumerate(intervals) if j != i]
    intervals_new.append((a_new, b_new))
    intervals_new.sort()
    for j in range(len(intervals_new) - 1):
        if intervals_new[j][1] > intervals_new[j + 1][0]:
            return None  # overlapping
    return intervals_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    best_h = h_init.copy()
    C_best = _hcorr(best_h, n_points)

    intervals = generate_initial_intervals(initial_h_values)
    measure = sum(b - a for a, b in intervals)
    dx = 2.0 / n_points
    if abs(measure - 1.0) > 1e-6:
        for i in range(len(intervals)):
            a, b = intervals[i]
            while measure > 1.0:
                intervals[i] = (a, b - 1e-5)
                measure -= 1e-5
            while measure < 1.0:
                intervals[i] = (a, b + 1e-5)
                measure += 1e-5

    best_intervals = intervals[:]
    best_score = C_best
    T = 1.0
    alpha = 0.999
    delta = 0.02

    while time.time() - start_time < budget_s - 1:
        # Propose interval adjustment
        idx = np.random.randint(len(intervals))
        d = np.random.uniform(-delta, delta)
        candidate = adjust_interval(intervals, idx, d, dx)
        if candidate is None:
            continue
        # Compute correlation
        h_candidate = discretize_intervals(candidate, n_points)
        C_candidate = _hcorr(h_candidate, n_points)

        if C_candidate < best_score or np.random.rand() < np.exp((best_score - C_candidate) / T):
            intervals = candidate[:]
            best_score = C_candidate
            T *= alpha
        if time.time() - start_time > budget_s - 1:
            break

    best_h = discretize_intervals(intervals, n_points)
    h_clean = np.clip(best_h, 0, 1)
    final_C5 = _hcorr(h_clean, n_points)
    return list(h_clean), final_C5, n_points
