import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_pad = np.concatenate([h, np.zeros(n_points - 1)])
    overlaps = []
    for k in range(1, n_points):
        if h.size + k <= h_pad.size:
            overlap = np.sum(h * (1 - h_pad[k: k + n_points]))
            overlaps.append(overlap * dx)
    return max(overlaps)

def objective(h_vec, n_points):
    h = h_vec.reshape(n_points)
    return compute_c5(h, n_points)

def constraint_sum(h_vec, n_points):
    return np.sum(h_vec) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: constraint_sum(h, n_points)}
    return bounds, cons
