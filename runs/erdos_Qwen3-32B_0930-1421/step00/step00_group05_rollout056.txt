import numpy as np
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Enforce sum constraint
    h *= n_points / 2.0 / h.sum()

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    h_best = h.copy()
    lowest_c5 = best_c5

    max_iter = 1000
    iters_since_improvement = 0

    for t in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break

        # Compute overlaps
        k_offsets = np.arange(-n_points + 1, n_points)
        overlap = np.full_like(k_offsets, np.nan)
        for k in range(0, n_points):
            # Left pad with 0s for negative k
            # Right pad with 0s for positive k
            if k < n_points // 2:
                pad_wid = n_points - k
                hp = np.pad(h, ((0, pad_wid),), "constant", constant_values=(0,))
                hpl = hp[:2 * n_points - k]
                hpr = 1 - np.roll(hpl, k)
                c = correlate(hp[:n_points], hpr, mode="full")[-n_points:]
                val = (c * dx).max()
            else:
                pad_wid = k
                hp = np.pad(h, ((pad_wid, 0),), "constant", constant_values=(0,))
                hpl = hp[k:]
                hpr = 1 - np.roll(hpl, -k)
                c = correlate(hpl, hpr, mode="full")
                val = (c * dx).max()
            overlap[k] = val

        c5 = overlap[y].max() if (y := ~np.isnan(overlap)).sum() else 0

        if c5 < lowest_c5:
            lowest_c5 = c5
            h_best = h.copy()
            iters_since_improvement = 0
        else:
            iters_since_improvement += 1

        # Stop if no improvement after a while
        if iters_since_improvement > 100:
            break

        # Find the worst overlap and compute the direction to go
        valid_idx = np.where(overlap < np.inf)[0]
        if valid_idx.size == 0:
            continue
        k_max = valid_idx[overlap[valid_idx].argmin()]
        val_k = overlap[k_max]

        # Compute gradient descent update
        if k_max >= 0:
            # Positive k: h(x) * (1 - h(x+k))
            padded_ext = n_points + k_max + 1
            hpl = np.zeros(padded_ext)
            hpr = np.zeros(padded_ext)
            hpl[:n_points] = h
            hpr[k_max:] = 1 - h[: padded_ext - k_max]
            conv = correlate(hpl, hpr, mode="full")
            grad = hpr * conv[n_points:]

        else:
            # Negative k: h(x) * (1 - h(x + k)), k is negative shift
            k_neg = -k_max
            hpl = np.zeros(n_points + k_neg)
            hpr = np.zeros(n_points + k_neg)
            hpl[k_neg:] = h[:n_points]
            hpr[:n_points] = 1 - h[k_neg:]
            conv = correlate(hpl, hpr, mode="full")
            grad = hpr * conv[n_points + k_neg :]

        # Normalize gradient and update
        update_direction = -grad / (np.linalg.norm(grad) + 1e-8)
        learning_rate = 0.2 / (t + 1)
        h += learning_rate * update_direction

        # Project back into constraints
        h = np.clip(h, 0.0, 1.0)

        # Normalize sum constraint
        h *= n_points / 2.0 / h.sum()

    final_c5 = evaluate_erdos_solution(h_best)
    return h_best, final_c5, n_points
