import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Ensure global initial_h_values is used directly
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.copy(initial_h_values).astype(np.float64)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.inf
    it = 0

    while time.time() - start_time < budget_s:
        it += 1

        # Compute all overlaps
        conv = np.correlate(h, 1 - h, mode='full')
        C = conv * dx
        max_index = np.argmax(np.abs(C))
        k_max = max_index - n_points + 1
        k_min = k_max
        k_max = k_max + 2 * (max_index - n_points + 1 + 1) if k_max < 0 else k_max

        if k_max < 0 or k_max >= len(C):
            continue

        # Compute gradient with respect to current C[k_max]
        # Only one shift index for now
        shifts = [k_max]

        grad = np.zeros_like(h)
        for k in shifts:
            i_vals = np.clip(np.arange(n_points) + k, 0, n_points - 1)
            j_vals = np.arange(n_points)
            overlap_pos = np.isin(j_vals, i_vals)
            mask = h[overlap_pos]
            grad[overlap_pos] += (1 - 2 * mask) * dx

        # Gradient step
        learning_rate = 0.1
        prev_h = h.copy()
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Normalize to sum constraint
        h /= h.sum() * dx
        h *= n_points / 2

        # Check improvement
        conv = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(np.abs(conv * dx))

        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

        # Backtracking line search if not improving
        if new_c5 >= best_c5:
            h = prev_h.copy()
            learning_rate *= 0.5

        # Frequency of normalization
        if it % 50 == 0:
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2

    conv = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(np.abs(conv * dx))
    return (best_h, c5_bound, n_points)
