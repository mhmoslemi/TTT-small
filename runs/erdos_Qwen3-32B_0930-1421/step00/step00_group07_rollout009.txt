import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h_start = initial_h_values[:]
    n_points = len(h_start)
    assert n_points * 2 % 2 == 0, "n_points must be divisible by 2"

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Copy initial h
    h = h_start.copy()

    best_h = h.copy()
    best_c5_bound = evaluate_erdos_solution(h)
    best_time = 0.0
    start_time = time.time()

    alpha = 0.02
    patience_counter = 0
    max_iter = 1000
    patience = 200

    def reweight_h(h, target_sum):
        # Project back to constraint: sum(h) = target_sum
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        # Rescale to maintain mean of 0.5
        return h_clipped / current_sum * target_sum

    for iter_idx in range(max_iter):
        t_start = time.time()
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = corr * dx
        max_c5_idx = np.argmax(c5)
        k = max_c5_idx - n_points + 1

        # Compute gradient for the worst shift k
        grad = np.zeros_like(h)
        usable_k = min(k, n_points - 1)
        for i in range(n_points):
            j_plus = (i + usable_k) % n_points
            j_minus = (i - usable_k) % n_points
            grad[i] += (1 - h[j_plus]) - h[j_minus]

        # Update h
        h_proposal = h.copy()
        h_proposal = h_proposal - alpha * grad
        h_proposal = reweight_h(h_proposal, target_sum)

        # Accept or reject to minimize C₅
        if evaluate_erdos_solution(h_proposal) < best_c5_bound:
            h = h_proposal.copy()
            best_h = h.copy()
            best_c5_bound = evaluate_erdos_solution(best_h)
            patience_counter = 0
        else:
            patience_counter += 1

        if iter_idx % 50 == 0 or patience_counter >= patience:
            print(f"Iter {iter_idx}, alpha={alpha:.6f}, C₅={best_c5_bound:.6f}")

        # Update learning rate every 500 iterations
        if iter_idx % 500 == 0 and alpha > 1e-5:
            alpha *= 0.5

        # Safety check for time
        if time.time() - start_time > budget_s - 10:
            print("Time budget almost expired. Terminating.")
            break

    return (best_h.tolist(), best_c5_bound, n_points)
