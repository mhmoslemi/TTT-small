import numpy as np
import cvxpy as cp
import time
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = default_rng(seed)

    start_time = time.time()

    # Configuration
    max_n_points = 1000
    initial_n_points = len(initial_h_values["h"])
    n_points = initial_n_points
    initial_h = initial_h_values["h"].copy()
    initial_C5 = initial_h_values["c5_bound"]

    dx = 2.0 / n_points
    total_mass = n_points / 2.0

    def objective(h):
        corrs = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corrs * dx)

    def solve_fixed_n_points(n, h0, budget_s_left):
        dx = 2.0 / n
        h = cp.Variable(n, nonneg=True)
        ones = np.ones(n)

        # Constraints
        constraints = [
            h <= 1.0,
            h >= 0.0,
            cp.sum(h) == total_mass
        ]

        def create_upper_bound_constraints(C5):
            def compute_overlap_cost_CVXPY(k):
                shifted = np.roll(h, shift=k)
                overlap = cp.sum(h * (1.0 - shifted))
                return overlap / n * 2.0

            upper_bounds = []
            for k in range(-(n-1), n):  # All valid lags
                if k == 0:
                    continue
                upper_bounds.append(compute_overlap_cost_CVXPY(k) <= C5)
            return upper_bounds

        def binary_search_with_solver(search_min, search_max, tolerance=1e-5, max_iter=50):
            best_C5 = initial_C5
            best_h = h0
            iter = 0
            while time.time() - start_time < budget_s and iter < max_iter:
                mid = (search_min + search_max) / 2.0
                print(f"Iter {iter}: C5 = {mid:.7f}, budget left: {budget_s - (time.time() - start_time):.1f}s")
                current_constraints = constraints + create_upper_bound_constraints(mid)
                problem = cp.Problem(cp.Minimize(0), current_constraints)
                problem.solve(solver=cp.MOSEK, verbose=False)
                if h.value is not None:
                    candidate_h = np.clip(h.value, 0, 1)
                    cur_C5 = objective(candidate_h)
                    if cur_C5 < best_C5:
                        best_C5 = cur_C5
                        best_h = candidate_h
                    if cur_C5 <= mid + 1e-6:
                        search_max = mid
                    else:
                        search_min = mid
                else:
                    search_min = mid
                iter += 1
            return best_h, best_C5

        h_best_fixed_n, C5_best_fixed_n = binary_search_with_solver(0.38085, 0.4, budget_s_left=600)
        return h_best_fixed_n, C5_best_fixed_n

    n_points_to_try = [70, 90, 100, 112]

    best_C5 = initial_C5
    best_h = initial_h
    best_n = len(initial_h)

    for n in n_points_to_try:
        if time.time() - start_time > budget_s - 30:
            break
        print(f"Trying n_points = {n}")
        if n == len(initial_h):
            h0 = initial_h
        else:
            h0 = np.linspace(0, 2, n)
            h0 = np.interp(h0, initial_h_values["x"], initial_h_values["h"])
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (total_mass) / np.sum(h0)

        h1, c5 = solve_fixed_n_points(n, h0, budget_s=500)
        if c5 < best_C5:
            best_C5 = c5
            best_h = h1
            best_n = n

    print(f"Best C5: {best_C5:.8f}")
    print("Constraints:")
    print("  All h in [0,1]:", np.all((best_h >= 0) & (best_h <= 1)))
    print("  Integral = 1    :", np.abs(np.sum(best_h) - total_mass) < 1e-6)
    return best_h, best_C5, best_n
