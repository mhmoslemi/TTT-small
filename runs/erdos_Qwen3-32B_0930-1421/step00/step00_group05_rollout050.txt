import numpy as np
import time

def evaluate_erdos_solution(h_values, n_points):
    h = h_values
    dx = 2.0 / n_points
    h_extended = np.concatenate([h, h])
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    C = np.max(overlaps[int(n_points/2):int(3*n_points/2)])
    return C

def project_constraints(h, n_points):
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to ensure sum(h) = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / current_sum
    # Clip again after normalization
    return np.clip(h, 0, 1)

def compute_overlap(h, n_points, beta=10):
    h_extended = np.tile(h, 3)  # for circular padding and overlap window
    L = n_points
    overlaps = np.zeros(L - 1)
    for k in range(1, L):  # shifts from 1 to n_points - 1
        if k >= h_extended.shape[0] - L:
            break
        overlap = np.sum(h_extended[:L] * (1 - h_extended[k:k+L])) / n_points
        overlaps[k - 1] = overlap
    max_overlap_idx = np.argmax(overlaps)
    # Log-sum-exp approximation of max
    exps = np.exp(beta * (overlaps - overlaps[max_overlap_idx]))
    approx_max = overlaps[max_overlap_idx] - np.log(np.sum(exps)) / beta
    return approx_max, overlaps

def compute_gradients(h, k, n_points):

    # Derivative with respect to h[i]

            - 2 * h[i] * (1 - h_extended[i + k]) / n_points
            - np.sum(h_extended[:n_points] * (1 - h_extended[k:k + n_points])) / (n_points)
        )
        grad[i] = dC_dh
    return grad
