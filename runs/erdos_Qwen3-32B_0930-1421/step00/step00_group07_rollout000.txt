import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = 80
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h /= h.sum() * dx * 2  # Renormalize to ensure sum == n_points / 2

    alpha = 0.01
    patience = 200
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    iteration = 0

    def compute_c5_and_gradient(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        corr_values = corr * dx
        max_index = np.argmax(np.abs(corr_values))
        max_lag = max_index - len(h_vec) + 1
        max_corr = corr_values[max_index]

        def compute_gradient(h_vec, lag):
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                if 0 <= i + lag < len(h_vec):
                    grad[i] += (1 - h_vec[i + lag])
                if 0 <= i - lag < len(h_vec):
                    grad[i] -= h_vec[i - lag]
                grad[i] *= dx
            return grad

        grad = compute_gradient(h_vec, max_lag)
        return max_corr, grad

    def project(h_vec):
        with np.errstate(divide='ignore'):
            h_proj = np.clip(h_vec, 0, 1)
        s = h_proj.sum()
        scale = (n_points / 2) / s
        h_proj *= scale
        return h_proj

    while time.time() - start_time < budget_s - 0.1:
        c5_current, grad = compute_c5_and_gradient(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                alpha *= 0.5
                patience_counter = 0

        h += -alpha * grad
        h = project(h)

        iteration += 1

    return best_h.tolist(), best_c5, n_points
