import numpy as np
import time
from scipy import optimize

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h_circ = np.concatenate([h, h])
    cvec = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cvec)

def optimize_h_admm(initial_h, max_iter=1000, rho=1.0, eps_rel=1e-4, eps_abs=1e-4, verbose=False):
    n_points = len(initial_h)
    h = initial_h.copy()
    z = np.zeros(n_points - 1)
    u = np.zeros(n_points - 1)
    dx = 2.0 / n_points
    ones = np.ones(n_points)

    # Normalize initial h to satisfy the mean constraint
    h /= h.sum()
    h *= n_points / 2

    for iter in range(max_iter):
        old_h = h.copy()

        # Compute overlaps for each lag
        h_circ = np.concatenate([h, h])
        overlaps = np.array([
            np.dot(h_circ[i:], 1 - h_circ[:len(h_circ) - i]) * dx for i in range(n_points - 1)
        ])

        # Update z
        for k in range(n_points - 1):
            z[k] = max(overlaps[k] + u[k], 0)
        
        # Compute augmented objective to optimize h
        def cost(hw):
            h_new = np.nan_to_num(np.reshape(hw, (n_points,)))
            h_circ = np.concatenate([h_new, h_new])
            overlaps = np.array([
                np.dot(h_circ[k:], 1 - h_circ[:len(h_circ) - k]) for k in range(n_points - 1)
            ])
            obj = np.sum(z - overlaps - u)  # Owing to ADMM augmented Lagrangian form
            return obj

        # Bounds for variables and linear constraint
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        A = np.array([np.ones(n_points)])
        b = np.array([n_points / 2.0])

        args = ()
        cons = [{'type': 'eq', 'fun': lambda h, A=A, b=b: A @ h - b}]
        result = optimize.minimize(
            cost,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
        )

        h = np.clip(result.x, 0.0, 1.0)

        # Update dual variables
        h_circ = np.concatenate([h, h])
        overlaps_new = np.array([
            np.dot(h_circ[k:], 1 - h_circ[:len(h_circ) - k]) for k in range(n_points - 1)
        ])
        s = z - overlaps_new - u
        for k in range(n_points - 1):
            u[k] += s[k]

        # Check convergence
        r_norm = np.sqrt(np.sum((z - overlaps - u) ** 2))
        s_norm = np.sqrt(np.sum((z - old_z) ** 2) if iter > 0 else 0.0)
        eps_pri = eps_abs + eps_rel * max(np.sqrt(np.sum(overlaps ** 2)), np.sqrt(np.sum(z ** 2)))
        eps_dual = eps_abs + eps_rel * np.sqrt(np.sum(u ** 2))
        if r_norm < eps_pri and s_norm < eps_dual:
            if verbose:
                print(f"ADMM converged at iteration {iter}")
            break

        old_z = z.copy()

    C5 = evaluate_c5(h, n_points)
    return h, C5

def run(seed=42, budget_s=1100, **kwargs):
    if "initial_h_values" not in globals():
        raise ValueError("Missing required input 'initial_h_values'")
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    
    start_time = time.time()
    h_opt, best_c5 = optimize_h_admm(initial_h)
    elapsed_time = time.time() - start_time
    
    if elapsed_time < budget_s:
        # Try further random perturbation to escape local minima
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed_h = np.clip(h_opt + noise, 0.0, 1.0)
            perturbed_h = perturbed_h / perturbed_h.sum() * n_points / 2
            candidate_h, candidate_c5 = optimize_h_admm(perturbed_h)
            if candidate_c5 < best_c5:
                h_opt, best_c5 = candidate_h, candidate_c5

    return h_opt, best_c5, n_points
