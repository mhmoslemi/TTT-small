import numpy as np
import scipy.optimize
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.copy(initial_h_values)
    c5_initial, _ = evaluate_erdos_solution(h_initial)

    pop_size = 50
    max_gen = 200
    mutation_rate = 0.1
    crossover_rate = 0.5
    parent_frac = 0.2
    target_sum = n_points / 2.0

    # Initialize population around the parent solution
    def perturb(h):
        h = np.copy(h)
        noise = np.random.normal(0, mutation_rate, h.shape)
        h += noise
        h = np.clip(h, 0, 1)
        return h

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()
        return h

    def fitness(h):
        h_projected = project_to_feasible(np.copy(h))
        c5, _ = evaluate_erdos_solution(h_projected)
        penalty = 1e4 * (abs(h_projected.sum() - target_sum) + np.max(h_projected - 1) + np.max(-h_projected))
        return c5 + penalty

    # Initial population
    population = [project_to_feasible(h_initial)]
    for _ in range(pop_size - 1):
        sample = perturb(h_initial)
        sample = project_to_feasible(sample)
        population.append(sample)

    best_h = np.copy(population[0])
    best_c5 = fitness(best_h)
    best_h = project_to_feasible(best_h)

    # Genetic algorithm loop
    for gen in range(max_gen):
        t_start = time.time()
        if t_start - start_time > budget_s - 5:
            break

        # Evaluate all candidates
        fits = [fitness(h) for h in population]
        order = np.argsort(fits)
        population = [population[i] for i in order]

        # Select top candidates
        parents = population[:int(pop_size * parent_frac)]

        # Generate new offspring
        new_candidates = []
        for h1 in parents:
            for h2 in parents:
                if len(new_candidates) >= pop_size:
                    break
                # Crossover: blend
                h_new = crossover_rate * h1 + (1 - crossover_rate) * h2
                # Mutate
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, mutation_rate, h_new.shape)
                    h_new += noise
                h_new = project_to_feasible(h_new)
                new_candidates.append(h_new)

        population = new_candidates[:pop_size]

        # Track best
        for h in population:
            c = fitness(h)
            if c < best_c5:
                best_c5 = c
                best_h = h

    # Final projection and evaluation
    best_h = project_to_feasible(best_h)
    c5_final, _ = evaluate_erdos_solution(best_h)

    if abs(best_h.sum() - target_sum) > 1e-6:
        # Last force normalization
        best_h *= target_sum / best_h.sum()
    return (best_h, c5_final, n_points)
