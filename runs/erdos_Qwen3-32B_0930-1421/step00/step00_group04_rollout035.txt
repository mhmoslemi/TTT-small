import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_objective(h, dx):
    h_padded = np.concatenate([h, h])
    k_vals = []
    for k in range(h.size):
        shifted = np.roll(h_padded, k)
        shifted = shifted[:h.size]
        corr = correlate(h, 1 - shifted, mode="full")
        c5 = np.max(corr * dx)
        k_vals.append(c5)
    return min(k_vals)

def project_h(h, dx):
    h_proj = np.clip(h, 0, 1)
    scaling = (h.size / 2) / np.sum(h_proj)
    h_proj *= scaling
    return h_proj

def run(seed=42, budget_s=1100, n_points=42):
    np.random.seed(seed)
    dx = 2.0 / n_points

    if "initial_h_values" in globals():
        h_start = initial_h_values
    else:
        h_start = np.random.rand(n_points)
        h_start = project_h(h_start, dx)

    h = h_start.copy()
    best_h = h.copy()
    best_bound = evaluate_erdos_objective(h, dx)
    best_decay = -np.inf
    iter_count = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        iter_count += 1

        # Compute overlaps
        h_padded = np.concatenate([h, h])
        overlap_vals = []
        for k in range(h.size):
            shifted = np.roll(h_padded, k)
            shifted = shifted[:h.size]
            corr = correlate(h, 1 - shifted, mode="full")[h.size - 1:]
            max_val = np.max(corr * dx)
            overlap_vals.append(max_val)

        current_bound = min(overlap_vals)

        # Check for improvement
        if current_bound < best_bound:
            best_bound = current_bound
            best_h = h.copy()
            print(f"New best bound: {best_bound:.9f} at iter {iter_count}")

        # Gradient update
        grad = np.zeros_like(h)
        for k in range(h.size):
            shifted = np.roll(h_padded, k)
            shifted = shifted[:h.size]
            corr = correlate(h, 1 - shifted, mode="full")[h.size - 1:]
            max_idx = np.argmax(corr * dx)
            grad_idx = h.size - 1 + max_idx
            grad[grad_idx] -= dx * h[max_idx]

        alpha = 0.01
        h = h - alpha * grad
        h = project_h(h, dx)

    return best_h.tolist(), best_bound, n_points
