import numpy as np
import time
from scipy.signal import correlate

def parse_intervals(h_values):
    n = len(h_values)
    dx = 2.0 / n
    intervals = []
    a = 0
    while a < n:
        if h_values[a] > 0.5:
            b = a + 1
            while b < n and h_values[b] > 0.5:
                b += 1
            intervals.append((a * dx, (b - 1) * dx))
            a = b
        else:
            a += 1
    return intervals

def apply_intervals(intervals, n_points):
    h = np.zeros(n_points)
    for a, b in intervals:
        a_idx = int(np.floor(a * n_points / 2))
        b_idx = int(np.ceil(b * n_points / 2))
        h[a_idx:b_idx] = 1
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = h
    h2 = 1 - h
    corr = correlate(h1, h2, mode='full')
    c5 = np.max(corr * dx)
    return c5

def adjust_interval(intervals, i, delta, eps=1e-6):
    new_intervals = list(intervals)
    a, b = new_intervals[i]
    new_a = a + delta
    new_b = b + delta
    if new_a < new_b - eps and 0 <= new_a <= 2 - eps and b <= 2 - eps:
        new_intervals[i] = (new_a, new_b)
    return new_intervals

def mutate_intervals(intervals, scale=0.02, n_try=1):
    for _ in range(n_try):
        i = np.random.randint(len(intervals))
        delta = np.random.normal(0, scale)
        new_intervals = adjust_interval(intervals, i, delta)
        if new_intervals != intervals:
            return new_intervals
    return intervals
