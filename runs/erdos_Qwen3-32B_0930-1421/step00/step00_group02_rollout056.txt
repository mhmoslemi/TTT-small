import numpy as np
import cvxpy as cp
from time import time
import math


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Parameters
    n_points_initial = len(initial_h_values)
    targets = [n_points_initial, n_points_initial]
    best_h = None
    best_c5 = np.inf
    times = []
    valid_solutions = []

    max_time = budget_s
    start_time = time()
    best_time = time()

    # Helper to compute C₅
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = float(np.max(overlaps) * dx)
        return c5

    # Phase 1: Frequency domain objective
    def optimize_phase1(n_points):
        dx = 2.0 / n_points
        h_start = (
            np.interp(np.linspace(0, 2, n_points), [0, 2], [0, 1]) if 'initial_h_values' not in globals()
            else np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(initial_h_values)), 
                           initial_h_values)
        )
        h = cp.Variable(shape=n_points, nonneg=True)
        objective = cp.Minimize(cp.max(cp.abs(cp.signal.fft.fft(h * (1 - h)))))

        constraints = [
            h <= 1,
            h >= 0,
            cp.sum(h) == n_points / 2.0
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != 'optimal':
            return h_start
        h_val = h.value
        return h_val.clip(0, 1)

    # Phase 2: Time domain gradient descent
    def optimize_phase2(h_init, alpha_start=0.5, alpha_decay=0.95, max_iter=200, eps=1e-5):
        h = np.array(h_init).copy()
        n = len(h)
        alpha = alpha_start
        dx = 2.0 / n
        max_iter = math.floor(max_iter)

        for _ in range(max_iter):
            # Compute cross correlation
            diff = np.correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(diff)
            # Precompute shifted (1-h) version in [-k, ...] where k = max_idx
            k = max_idx - (n - 1)
            if k < n:
                cor = 1 - h[[((i - k) % n) for i in range(n)]]
            else:
                k = int(n - n + max_idx)
                cor = h[[((i - k) % n) for i in range(n)]]
                cor = 1 - cor

            # Gradient step: direction is h*(h-1) * target overlap push toward smoother h
            grad = cor - h
            h += alpha * grad
            h = h.clip(0, 1)

            # Enforce sum constraint
            scaling_factor = (n / 2.0) / h.sum()
            h = h * scaling_factor
            h = h.clip(0, 1)

            # Recompute objective and check for improvement
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            alpha *= alpha_decay

            if time() - best_time > 9.0 * max_time / 10.0:
                break

    for n_points in targets:
        dx = 2.0 / n_points
        h_init = (
            np.interp(np.linspace(0, 2, n_points), [0, 2], [0, 1])
            if 'initial_h_values' not in globals()
            else np.interp(np.linspace(0, 2, n_points),
                           np.linspace(0, 2, len(initial_h_values)),
                           initial_h_values)
        )
        h_freq = optimize_phase1(n_points)
        h_freq /= h_freq.sum() * dx  # normalize to constraint
        h_freq *= n_points / 2.0
        h_freq = h_freq.clip(0, 1)
        optimize_phase2(h_freq)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        if time() - start_time > max_time:
            break

    # Postprocessing
    h_values = best_h.tolist()
    return (h_values, best_c5, len(h_values))
