import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = np.inf
    t_start = time.time()
    patience_counter = 0
    alpha = 0.02
    max_iter = 1000
    patience = 200

    def compute_overlap(h):
        ones = np.ones(n_points)
        corr = correlate(h, ones - h, mode='full')
        return np.max(corr * dx)

    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        # Shift index k corresponds to the overlap between h and (1-h) shifted by k
        for i in range(n_points):
            if 0 <= i - k < n_points:
                grad[i] -= 1 - h[i - k]
            if 0 <= i + k < n_points:
                grad[i] += h[i + k]
        return grad

    while time.time() - t_start < budget_s:
        for iter in range(max_iter):
            if time.time() - t_start > budget_s:
                break
            corr = correlate(h, 1 - h, mode='full')
            overlaps = corr * dx
            k = np.argmax(overlaps) - (n_points - 1)
            if k < 0:
                k += n_points
            # Current overlap
            c5 = overlaps.max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                patience_counter = 0
                print(f"Iter {iter}: new best c5={best_c5:.6f}")
            else:
                patience_counter += 1
                if patience_counter >= patience:
                    alpha *= 0.5
                    patience_counter = 0
                    if alpha < 1e-7:
                        break
            grad = compute_gradient(h, k)
            grad = grad / np.abs(grad).max()
            h = h - alpha * grad
            # Clip to [0, 1]
            h = np.clip(h, 0, 1)
            # Enforce sum constraint
            h = h / h.sum() * target_sum

    return (best_h.tolist(), best_c5, n_points)
