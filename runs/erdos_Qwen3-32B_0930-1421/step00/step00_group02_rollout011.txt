import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # 1. Initialization
    initial = initial_h_values
    n_points = len(initial)
    dx = 2.0 / n_points
    total_mass = n_points / 2
    h = initial * 1.0  # starts with initial construction

    h = np.clip(h, 0, 1)
    h *= total_mass / np.sum(h)  # normalize to satisfy mass constraint

    best_h = h.copy()
    best_c5 = float("inf")
    patience = 0
    max_patience = 20

    def calc_c5(h):
        h_mass = h.sum()
        if abs(h_mass - total_mass) > 1e-6:
            warning("Mass constraint violation: correcting")
            h *= total_mass / h_mass
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def line_search(h, dh, lr=0.1):
        h_proposal = np.clip(h + dh * lr, 0, 1)
        mass = h_proposal.sum()
        h_proposal *= total_mass / mass
        return h_proposal

    best_c5 = calc_c5(h)
    best_h = h.copy()
    print(f"Initial C5: {best_c5:.8f}")

    # 2. Frequency domain optimization
    phase1_time = 0
    freq_iter = 0
    max_freq_iter = 10

    while freq_iter < max_freq_iter and time() - start_time + phase1_time < 0.6 * budget_s:
        freq_iter += 1
        h_fft = np.fft.rfft(h)

        # Frequency domain problem
        H = cp.Variable(h_fft.shape)
        obj = cp.Minimize(cp.norm(H, p=float('inf')))
        constraint = [0 <= H] if h_fft.dtype == np.complex128 else [0 <= H[1:].real]
        # Require 1-h * h to be minimized in frequency domain
        combined = H * (1 - H)
        prob = cp.Problem(obj, [cp.norm(H, 'inf') <= 1] + constraint)
        prob.solve(solver=cp.ECOS, verbose=False)

        h_opt_fft = prob.value
        h_new = np.fft.irfft(h_opt_fft)

        # Project onto constraint space
        h_new = line_search(h, h_new - h)
        h = np.clip(h_new, 0, 1)
        h *= total_mass / h.sum()

        c5 = calc_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            break

    # 3. Gradient-Based Refinement
    gradient_iter = 0
    max_gradient_iter = 500

    while gradient_iter < max_gradient_iter and time() - start_time < budget_s - 60:
        gradient_iter += 1

        k_best = np.argmax(np.correlate(h, 1 - h, mode="full"))
        # Use directional derivative approx
        grad_plus = h.copy()
        grad_plus[k_best-1] = np.clip(h[k_best-1] + 0.05, 0, 1)
        grad_plus = line_search(h, grad_plus - h, 0.01)

        c5_plus = calc_c5(grad_plus)

        if c5_plus < best_c5:
            h = grad_plus
            if c5_plus < best_c5:
                best_c5 = c5_plus
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

        if patience >= max_patience:
            break

    # 4. Final Output
    best_c5 = calc_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
