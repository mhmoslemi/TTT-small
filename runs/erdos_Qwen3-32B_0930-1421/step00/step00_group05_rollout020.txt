import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Discretization setup
    try:
        h_start = np.array(initial_h_values)
        n_start = len(h_start)
        n_points = n_start  # Start from existing size
        dx = 2.0 / n_points
    except:
        # Fallback to default if no initial_h_values is defined
        n_points = 70
        h_start = np.ones(n_points) / n_points * (n_points / 2)

    # Normalize to meet the constraint sum(h) == n_points / 2
    h_start = h_start / h_start.sum() * (n_points / 2)
    h_start = np.clip(h_start, 0, 1)

    # Ensure h is feasible (0 ≤ h ≤ 1, sum is correct)
    def is_feasible(h):
        return np.all((0 <= h) & (h <= 1)) and np.isclose(h.sum(), n_points / 2, atol=1e-6)

    # Compute C5 from h using correlation
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        one_minus_h = 1.0 - h
        corr = correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    # Initial feasible h and c5
    h_best = np.copy(h_start)
    c5_best = compute_c5(h_best)
    best_time = time.time()
    budget_used = best_time - start_time

    # Define context for variables: decision vector h
    h = cp.Variable(n_points, nonneg=True)
    constraints = [h <= 1, h.sum() == n_points / 2]

    # Slack for max optimization
    t = cp.Variable()
    corr_k = []
    for k in range(n_points-1, n_points+1):
        # Compute correlation for each k (shifts from -1 to +1 in this range)
        offset_h = np.roll(h, k).copy()
        t_k = cp.sum(h * (1 - offset_h))
        corr_k.append(t_k)
    corr_max = cp.Max(corr_k)
    objective = cp.Minimize(corr_max * dx)

    prob = cp.Problem(objective, constraints)
    prob.solve()

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_candidate = np.clip(np.array(h.value), 0, 1)
        h_candidate = (h_candidate / h_candidate.sum()) * (n_points / 2)
        h_candidate = np.clip(h_candidate, 0, 1)

        if is_feasible(h_candidate):
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5_best:
                h_best = h_candidate
                c5_best = c5_candidate

    # Iterative optimization with smaller perturbations
    while budget_used < budget_s - 5:
        # Random perturbation
        h_perturbed = np.copy(h_best) + 0.1 * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = (h_perturbed / h_perturbed.sum()) * (n_points / 2)

        if is_feasible(h_perturbed):
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < c5_best:
                h_best = h_perturbed
                c5_best = c5_perturbed

        budget_used = time.time() - start_time
        if (budget_used - best_time) > 5:
            best_time = budget_used

    return (h_best.tolist(), c5_best, n_points)
