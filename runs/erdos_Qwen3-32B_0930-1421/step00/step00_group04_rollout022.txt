import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = 42
    dx = 2.0 / n_points
    h_sum = n_points // 2  # Constraint: sum(h) == n_points / 2

    # Start with an initial solution from parent construction
    if 'initial_h_values' in globals():
        current_best_h = np.array(initial_h_values)
    else:
        # Construct an initial feasible solution: spread 1 with uniform distribution
        current_best_h = np.zeros(n_points)
        current_best_h[::2] = 1
        current_best_h = current_best_h / np.sum(current_best_h) * h_sum

    best_c5 = evaluate_erdos_solution(current_best_h)

    def evaluate_fitness(h):
        h = h.clip(0, 1)
        h = h / h.sum() * h_sum  # Enforce ∑h[i] = n_points / 2
        return evaluate_erdos_solution(h)

    def tournament_selection(population, fitnesses, size=3, tourn_size=2):
        indices = np.random.choice(len(population), (tourn_size, size))
        return np.array([np.argmin([fitnesses[i] for i in idx]) for idx in indices])

    def crossover(parent1, parent2):
        # Discrete crossover
        alpha = np.random.rand(n_points)  # Use real-valued crossover
        child = alpha * parent1 + (1 - alpha) * parent2
        child = child.clip(0, 1)
        child = child / child.sum() * h_sum
        return child

    def mutate(child, mutation_rate=0.1):
        # Small perturbation mutation
        noise = np.random.normal(0, 0.1, n_points)
        mask = np.random.rand(n_points) < mutation_rate
        child = np.where(mask, child + noise, child)
        child = child.clip(0, 1)
        child = child / child.sum() * h_sum
        return child

    # GA parameters
    pop_size = 40
    generations = 300
    mutation_rate = 0.1

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(initial_h_values)
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h / h.sum() * h_sum
        population.append(h)
    
    population = np.array(population)
    fitnesses = np.array([evaluate_fitness(h) for h in population])

    # Track start time
    start_time = time.time()
    current_best_h = population[np.argmin(fitnesses)]
    best_c5 = np.min(fitnesses)

    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break

        # Tournament selection
        selected_indices = tournament_selection(population, fitnesses, pop_size, 3)
        selected = population[selected_indices]

        # Crossover & Mutation
        new_population = []
        for i in range(0, pop_size, 2):
            if i + 1 >= pop_size:
                break
            child1 = crossover(selected[i], selected[i+1])
            child2 = mutate(child1, mutation_rate)
            new_population.extend([child1, child2])

        # Evaluate new individuals
        new_population = np.array(new_population)
        new_fitnesses = np.array([evaluate_fitness(h) for h in new_population])

        # Combine with old population and select the best
        all_pop = np.vstack([population, new_population])
        all_fitness = np.concatenate([fitnesses, new_fitnesses])
        # Get index of best pop_size individuals
        idxs = np.argsort(all_fitness)[:pop_size]
        population = all_pop[idxs]
        fitnesses = all_fitness[idxs]

        # Update best
        current_idx = np.argmin(fitnesses)
        current_c5 = fitnesses[current_idx]
        if current_c5 < best_c5:
            current_best_h = population[current_idx].copy()
            best_c5 = current_c5

    return current_best_h, best_c5, n_points
