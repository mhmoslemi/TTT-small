import numpy as np
import cvxpy as cp
import time
from math import floor


def run(seed=42, budget_s=1100, **kwargs):
    timer = time.time

    np.random.seed(seed)

    # Use initial_h_values if provided
    if 'initial_h_values' not in kwargs or kwargs['initial_h_values'] is None:
        n_points = 80
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        initial_h = np.clip(initial_h, 0, 1)
    else:
        n_points = len(kwargs['initial_h_values'])
        initial_h = kwargs['initial_h_values'].copy()

    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h_time = 0

    t0 = timer()
    max_iter = 10000
    iter_ = 0
    patience = 200
    no_improve_count = 0
    alpha = 0.005
    dx = 2.0 / n_points

    def compute_c5(h):
        h_fft = np.fft.fft(h)
        h_inv = 1 - h
        h_inv_fft = np.fft.fft(h_inv)
        conv_fft = h_fft * h_inv_fft
        conv = np.fft.ifft(conv_fft).real
        c5 = np.max(conv) * dx
        return c5

    while timer() - t0 < budget_s - 1:
        h_fft = np.fft.fft(h)
        h_inv_fft = np.fft.fft(1 - h)
        conv_fft = h_fft * h_inv_fft
        conv = np.fft.ifft(conv_fft).real

        idx_all = np.arange(len(conv))
        center = len(h) // 2
        mask = (idx_all > center) & (idx_all < (len(conv) - center))
        idx_choice = np.argwhere(mask)
        lag = idx_choice[np.argmin(conv[mask])]

        # Direction to update h
        corr_h = np.roll(1 - h, shift=int(lag))
        grad = -corr_h * dx
        h_new = h + alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # Normalize to constraint

        h = h_new.copy()
        current_c5 = evaluate_erdos_solution(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_h_time = timer() - t0
            no_improve_count = 0
            alpha *= 1.1  # Could even decrease learning rate
        else:
            no_improve_count += 1
            alpha *= 0.99  # Small decay on no improvement

        if no_improve_count > patience:
            alpha = max(1e-5, alpha * 0.8)
            no_improve_count = 0

        iter_ += 1
        if iter_ % 200 == 0:
            print(f"Iter: {iter_} | C5: {best_c5:.6f} | alpha: {alpha:.4f} | time: {timer()-t0:.2f}s")

    return best_h, best_c5, n_points
