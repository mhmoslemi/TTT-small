```python
import numpy as np
import time
import scipy.signal
import cvxpy as cp
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use initial_h_values if provided
    initial_h_values_available = "initial_h_values" in globals()
    if initial_h_values_available:
        n_points = len(initial_h_values)
        h = np.copy(initial_h_values)
    else:
        n_points = 84  # Start with 84 points for resolution
        h = np.ones(n_points) / n_points * n_points / 2

    dx = 2.0 / n_points

    def evaluate_c5(h_vec):
        h_rev = 1 - h_vec
        corr = scipy.signal.correlate(h_vec, h_rev, mode='full')
        corr *= dx
        return float(np.max(corr))
    
    # Normalize constraint: sum(h) == n_points / 2
    def enforce_constraint(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec /= h_vec.sum() * dx
        h_vec *= n_points / 2
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    # Gradient descent with restart and simplex
    t0 = time.time()
    c5_best = evaluate_c5(h)
    h_best = np.copy(h)
    learning_rate = 0.01
    restart_count = 0
    simplex_size = 0.1
    simplex_steps = 0

    while time.time() - t0 < budget_s:
        h_vec = np.copy(h_best)

        for _ in range(50):  # 50 parameter steps between check and restart
            if time.time() - t0 > budget_s:
                break

            h_rev = 1 - h_vec
            corr = scipy.signal.correlate(h_vec, h_rev, mode='full')
            corr *= dx

            shift_indices = np.where(corr == corr.max())[0]
            if len(shift_indices) > 0:
                target_shift = shift_indices[0]
                overlap = scipy.signal.correlate(h_vec, h_rev[target_shift:target_shift + n_points], mode='valid') * dx

                # Compute gradient update: reduce overlap
                grad = -overlap
                h_vec += learning_rate * grad
                h_vec = enforce_constraint(h_vec)

                current_c5 = evaluate_c5(h_vec)
                if current_c5 < c5_best:
                    c5_best = current_c5
                    h_best = np.copy(h_vec)
                    learning_rate *= 0.95  # Adaptive learning rate
                else:
                    learning_rate = max(0.001, learning_rate * 0.98)

        if time.time() - t0 > budget_s:
            break

        restart_count += 1
        print(f"Restart {restart_count}, Current C₅: {c5_best}, Time: {time.time()-t0:.2f}s")

        if np.random.rand() < 0.1:
            # Stochastic simplex exploration
            simplex = []
            for _ in range(5):
                perturbation = simplex_size * (2 * np.random.rand(n_points) - 1)
                perturbed = np.copy(h_best) + perturbation
                perturbed = enforce_constraint(perturbed)
                simplex.append(perturbed)
            simplex.append(h_best)
            simplex_c5 = np.array([evaluate_c5(p) for p in simplex])
            best_idx = simplex_c5.argmin()
            h_best = simplex[best_idx]
            if simplex_c5[best_idx] < c5_best:
                c5_best = simplex_c5[best_idx]

    return (h_best, c5_best, n_points)
```
