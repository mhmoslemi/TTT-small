import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    initial_h = initial_h_values.copy()
    n_points = initial_h.size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h * (target_sum / initial_h.sum())

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    max_iterations = 5000
    alpha = 0.01
    patience = 200
    wait = 0
    best_iter = 0

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        # Normalize to ensure sum constraint
        h = h / h.sum() * target_sum
        return h

    for i in range(max_iterations):
        t = time.time()
        if t - start_time >= budget_s - 1:
            break

        corr = correlate(h, 1 - h, mode="full")
        k = np.argmax(np.abs(corr))
        dh = correlate(np.eye(1, n_points, k - (n_points - 1)), 1 - h, mode="same") - 2 * h * (1 - h[:, None])
        dh = correlate(np.eye(1, n_points, k - (n_points - 1)), h, mode="same")
        grad = dh.flatten()

        # Update h
        h_new = h - alpha * grad
        h_new = project(h_new)

        c5_new = compute_c5(h_new)

        # Early stopping
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h[:] = h_new.copy()
            best_iter = i
            wait = 0
        else:
            wait += 1
            if wait >= patience:
                alpha *= 0.5
                wait = 0

        h[:] = h_new.copy()

    # Validation: confirm constraints and C5
    best_h = project(best_h)
    return tuple(best_h), compute_c5(best_h), n_points
