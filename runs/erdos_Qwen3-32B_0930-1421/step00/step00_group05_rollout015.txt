import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def build_triangle_bases(n_points, n_bases):
    x = np.linspace(0, 2, n_points)
    width = 4.0 / n_bases
    bases = []
    for i in range(n_bases):
        center = 2 * (i / n_bases)
        base = np.zeros(n_points)
        for j in range(n_points):
            d = abs(x[j] - center)
            if d <= width:
                base[j] = max(0, 1 - d / width)
        bases.append(base)
    return np.array(bases)

def objective(h_values):
    dx = 2.0 / len(h_values)
    h_values = np.clip(h_values, 0.0, 1.0)
    h = h_values.astype(np.float64)
    h_f = np.r_[h, 0.0, 0.0]
    one_minus_h = 1 - h_f
    conv = correlate(h_f, one_minus_h, mode="full")
    result = np.zeros(conv.shape[0])
    for k in range(1, 5):
        idx = k if k <= len(h_values) else k - 1
        if idx < conv.shape[0]:
            result[k - 1] = conv[idx]
   ecal_objective = np.max(result) * dx
    total = np.sum(h_values)
    assert np.allclose(total, len(h_values) / 2.0, atol=1e-6), f"Integral constraint violated: {total} != {len(h_values)/2.0}"
    return ecal_objective

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use 100 basis functions over 140 points for a good complexity balance
    n_points_built = 140
    n_bases = 100
    bases = build_triangle_bases(n_points_built, n_bases)

    # Build superposition model
    w = cp.Variable(n_bases, nonneg=True)
    constraints = [sum(w) == 1.0]
    h = bases @ w

    # Build the overlap expressions
    dx = 2.0 / n_points_built
    e = []
    for k in range(1, 5):
        shifted_h = np.roll(h, k)
        shifted_h = shifted_h[:(n_points_built - k)]  # Trim overlapping values
        overlap = sum(h[:(n_points_built - k)] * (1 - shifted_h))
        e.append(overlap * dx)
    t = cp.Variable()
    constraints.extend([t >= expr for expr in e])

    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK, verbose=False, mosek_params={"MSK_DPAR_OPTIMIZER_MAX_TIME": 60})

    h_values_built = h.value

    # Resize to 70 samples
    h_values = h_values_built[::2]  # downsample
    h_values[-1] += (h_values_built[-1] if len(h_values_built) > len(h_values) else 0)

    # Normalize to meet the integral constraint
    total = np.sum(h_values)
    if not np.allclose(total, 70 / 2.0, atol=1e-6):
        h_values = h_values * (35.0 / total)

    best_c5 = objective.value

    # Final check over runtime and post-process to 70 points
    while time.time() - start_time < (0.99 * budget_s):
        # Try random perturbation
        try:
            idx = np.random.choice(len(h_values))
            h_values_old = h_values[idx].copy()
            h_values[idx] = np.random.uniform(0, 1)
            h_values = np.clip(h_values, 0, 1)
            new_total = np.sum(h_values)
            if new_total < 35:
                scaling = 35 / new_total
                h_values = h_values * scaling
            new_c5 = objective(h_values)
            if new_c5 < best_c5:
                best_c5 = new_c5
                print(f"Improved C5 to {best_c5:.8f}")
            h_values = h_values_old  # Revert
        except:
            pass

    return (h_values.tolist(), best_c5, 70)
