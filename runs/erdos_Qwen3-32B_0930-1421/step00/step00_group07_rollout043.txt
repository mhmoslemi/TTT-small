```python
import numpy as np
import time
import scipy.signal

initial_h_values = getattr(globals(), 'initial_h_values', None)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    use_initial = initial_h_values is not None
    n_points = 80 if use_initial else 80

    if use_initial:
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = h / h.sum() * (n_points / 2)  # normalize to satisfy the constraint
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    alpha = 0.02
    patience = 200
    no_improve = 0
    max_iter = 1000

    def compute_c5(h):
        dx = 2 / n_points
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    while time.time() - start_time < budget_s and no_improve < patience:
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = corr * (2 / n_points)
        k = np.argmax(c5)
        shift = k - (n_points - 1)

        # Construct gradient for the worst shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            left = (i - shift) % n_points
            right = (i + shift) % n_points
            grad[i] = (1 - h[right]) - h[left]

        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 500 == 0:
            alpha /= 2  # halve learning rate

        if no_improve >= patience or alpha < 1e-6:
            break

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
```
