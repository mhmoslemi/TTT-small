import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use given initial solution
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = np.copy(h)
    
    # Gradient descent optimization
    learning_rate = 0.01
    max_iterations = 3000
    update_interval = 100
    for iteration in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

        # Compute correlation for all valid shifts
        h_corr = correlate(h, (1 - h), mode='full')
        valid_shifts = np.arange(min(len(h), len(h)))#.astype(int)
        C = h_corr[n_points-1:] * dx
        c5 = np.max(C)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)

        if iteration % update_interval == 0:
            print(f"Iteration {iteration}, C5 = {c5}")

        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(1, n_points):
            mask = (np.arange(n_points - k)) < (n_points - k)
            for i in range(len(h) - k):
                grad[i] += h[i + k]
                grad[i + k] -= h[i]

        grad *= dx

        # Update h using gradient descent
        h -= learning_rate * grad

        # Clip to [0,1]
        h = np.clip(h, 0, 1)

        # Normalize to satisfy the constraint sum(h) = target_sum
        h *= target_sum / h.sum()

    # Final cleanup
    h = best_h
    h = np.clip(h, 0, 1)  # Final enforcement of [0,1]
    h *= target_sum / h.sum()  # Final normalization
    
    # Re-calculate final C5
    h_corr = correlate(h, (1 - h), mode='full')
    valid_shifts = np.arange(n_points)
    C = h_corr[n_points-1:] * dx
    final_c5 = np.max(C)

    return (h.tolist(), float(final_c5), n_points)
