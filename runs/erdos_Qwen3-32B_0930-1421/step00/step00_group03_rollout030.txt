import numpy as np
import time
from scipy.signal import correlate

def clip_and_scale(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    target_sum = n_points / 2.0
    scale = target_sum / current_sum if current_sum != 0 else 1.0
    h_scaled = h_clipped * scale
    return h_scaled

def compute_c5(h, n_points):
    dx = 2.0 / n_points

    result = correlate(h, one_minus_h, mode='full') * dx
    return np.max(result)
