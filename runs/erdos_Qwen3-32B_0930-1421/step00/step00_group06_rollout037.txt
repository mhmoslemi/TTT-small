import numpy as np
import scipy.signal
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Start from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        assert np.allclose(np.sum(h) * dx, 1.0)
    else:
        n_points = 95
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = h / np.sum(h) * n_points / 2  # normalize to have sum = n_points / 2
        dx = 2.0 / n_points

    h = cp.Parameter(n_points, value=h, nonneg=True)
    z = cp.Variable(n_points)

    # Define the maximum overlap constrained by z
    correl = scipy.signal.correlate(h.value, 1 - h.value, mode='full')
    max_lag = n_points  # Only consider lags up to n_points

    overlap = np.zeros(n_points)

    for k in range(n_points):
        # Compute the sum of h[i] * (1 - h[(i + k) % n_points]) * dx
        overlap[k] = np.sum(h.value * (1 - np.roll(h.value, -k))) * dx

    # Initial z is the overlap
    best_z = np.max(overlap)
    best_h = h.value.copy()

    # Optimize via L-BFGS-B
    def objective(h_vec):
        h_flat = h_vec.reshape(-1)
        if not (np.all(h_flat >= 0) and np.all(h_flat <= 1)):
            return np.inf
        if not np.isclose(np.sum(h_flat), n_points / 2):
            return np.inf
        correl = scipy.signal.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(correl[:2 * n_points - 1]) * dx  # only evaluate [0, 2] interval
        return c5

    bounds = [(0, 1)] * n_points
    from scipy.optimize import minimize
    result = minimize(
        objective,
        x0=h.value,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-6}
    )

    if result.success and result.fun < best_z:
        best_h = result.x
        best_z = result.fun

    # Return best solution found
    c5_bound = best_z
    h_values = best_h

    return h_values, c5_bound, n_points
