import numpy as np
import scipy.optimize as sopt
from time import time

def objective(h, n_points):
    dx = 2.0 / n_points
    h.resize((1, n_points))
    corr = np.correlate(h[0], 1 - h[0], mode='full')
    return np.max(corr * dx)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2.0

def bounds(n_points):
    return [(0.0, 1.0)] * n_points

def optimize_from_h0(h0, n_points):
    dx = 2.0 / n_points

    def opt_func(h):
        h.resize((1, n_points))
        corr = np.correlate(h[0], 1 - h[0], mode='full')
        return np.max(corr * dx)

    def jacobian(h):
        h_resize = h.reshape((1, n_points))
        dhdp = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                j = (i - k) % n_points  # negative wraps around
                dcorr = h_resize[0][i] - h_resize[0][i] * h_resize[0][j]
                dhdp[i] += dcorr * dx / n_points
        return dhdp

    cons = (
        {'type': 'eq', 'fun': lambda h: constraint_sum(h, n_points)}
    )
    bounds_obj = bounds(n_points)
    res = sopt.minimize(opt_func, x0=h0, method='SLSQP',
                        jac=jacobian, bounds=bounds_obj, constraints=cons,
                        options={'ftol': 1e-6, 'maxiter': 1000})
    if not res.success:
        return h0
    h_opt = res.x
    # Project on to the simplex for safety
    h_opt = np.clip(h_opt, 0, 1)
    h_opt += (n_points / 2.0 - np.sum(h_opt)) / n_points
    h_opt = np.clip(h_opt, 0, 1)
    return h_opt

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = initial_h_values.shape[0]
    h0 = initial_h_values.copy()
    best_h = h0
    best_c5 = objective(h0, n_points)
    start_time = time()

    for _ in range(3):
        # Use different starting points to avoid local optima
        h0 = np.random.default_rng(seed + _).uniform(0, 1, n_points)
        h0 += (n_points / 2.0 - h0.sum()) / n_points
        h0 = np.clip(h0, 0, 1)
        # Local search
        h_opt = optimize_from_h0(h0, n_points)
        c5_opt = objective(h_opt, n_points)
        if c5_opt < best_c5:
            best_h[:] = h_opt[:]
            best_c5 = c5_opt
        elapsed = time() - start_time
        if elapsed > budget_s * 0.5:
            break

    # Final polishing with initial_h_values
    h_opt = optimize_from_h0(h0, n_points)
    c5_opt = objective(h_opt, n_points)
    if c5_opt < best_c5:
        best_h[:] = h_opt[:]
        best_c5 = c5_opt
    h_opt = best_h
    c5_opt = best_c5
    dx = 2.0 / n_points
    c5_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
    return (h_opt.tolist(), c5_opt, n_points)
