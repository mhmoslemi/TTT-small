import time
from scipy.optimize import minimize


def evaluate_solution(h, n_points):
    dx = 2.0 / n_points
    h_vals = h.reshape(-1)

    # Compute all shifted overlaps: h(x) * (1 - h(x + k)) for all k
    max_overlap = -np.inf
    for k in range(1, n_points):
        if k >= n_points:
            break
        # Compute overlap for shift k
        overlap = np.sum(h_vals[: n_points - k] * (1 - h_vals[k:])) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap


def objective(h_flat, n_points):
    h = h_flat.reshape(-1)
    return evaluate_solution(h, n_points)
