import numpy as np
import time
from scipy.signal import correlate
import random
import cvxpy as cp


def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    values = correlate(h, 1 - h, mode="full") * dx
    return np.max(values)


def project_to_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    h *= n_points / 2.0 / np.sum(h)
    return h


def run(seed=42, budget_s=1100, n_points=70, n_particles=50, max_iter=500):
    np.random.seed(seed)
    random.seed(seed)
    init_h = initial_h_values
    best_h = init_h.copy()
    best_c5 = evaluate_c5(best_h, n_points)
    particle_positions = [init_h + 0.1 * np.random.randn(n_points) for _ in range(n_particles)]
    for p in particle_positions:
        p[:] = project_to_feasible(p, n_points)
    particle_velocities = [np.zeros_like(p) for p in particle_positions]
    personal_bests = [p.copy() for p in particle_positions]
    personal_best_scores = [evaluate_c5(p, n_points) for p in personal_bests]
    global_best = personal_bests[np.argmin(personal_best_scores)]
    global_best_score = np.min(personal_best_scores)
    start_time = time.time()
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            particle_velocities[i] = (
                omega * particle_velocities[i]
                + c1 * r1 * (personal_bests[i] - particle_positions[i])
                + c2 * r2 * (global_best - particle_positions[i])
            )
            particle_positions[i] += particle_velocities[i]
            particle_positions[i] = project_to_feasible(particle_positions[i], n_points)
            c5 = evaluate_c5(particle_positions[i], n_points)
            if c5 < personal_best_scores[i]:
                personal_bests[i] = particle_positions[i].copy()
                personal_best_scores[i] = c5
                if c5 < global_best_score:
                    global_best = personal_bests[i].copy()
                    global_best_score = c5
                    best_h = global_best.copy()
                    best_c5 = global_best_score
            if c5 < best_c5:
                best_h[:] = particle_positions[i].copy()
                best_c5 = c5
        omega *= 0.995  # inertia decay
    return best_h.tolist(), best_c5, n_points
