import numpy as np
import time
from scipy.signal import correlate

def project(h, target_sum):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Scale to ensure sum constraint
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling_factor = target_sum / h_sum
    return np.clip(h_clipped * scaling_factor, 0, 1)

def eval_c5(h, dx):
    ones_minus_h = 1 - h
    corr = correlate(h, ones_minus_h, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_c5 = eval_c5(best_h, dx)
    
    max_iter = 1000
    iter_start = 0

    alpha = 1.0  # Initial step size
    beta = 0.5   # Step size reduction factor

    for iter in range(iter_start, max_iter):
        # Compute current c5 and the shift index k that gives the maximum overlap
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full')
        k_star = np.argmax(corr)
        
        # Compute gradient vector (h(x) derivative is not taken, but we know which shift is max)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_star <= i < k_star + n_points:
                grad[i - k_star] = ones_minus_h[i - k_star]
        
        # Normalize gradient
        grad *= 1.0  # No scaling because already in dx units
        
        # Line search with backtracking
        tau = alpha
        for backtrack in range(10):
            h_candidate = h - tau * grad
            h_candidate = project(h_candidate, target_sum)
            c5_candidate = eval_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                h = np.copy(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = np.copy(h)
                break
            else:
                tau *= beta
        if time.time() - start_time >= budget_s:
            break
    
    return best_h.tolist(), best_c5, n_points
