import numpy as np
import cvxpy as cp
import math
import time

from evaluate_erdos_solution import evaluate_erdos_solution
import initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points

    h = h / np.sum(h) * (n_points / 2)  # ensure sum constraint

    h_prev = h.copy()
    alpha = 0.5
    best_c5 = np.inf
    best_h = h.copy()
    max_iter = 1000
    iter_without_improvement = 0
    improvement_threshold = 1e-4
    alpha_decay_factor = 0.95

    def compute_c5(h_in):
        correlations = np.correlate(h_in, 1 - h_in, mode='full')
        return np.max(correlations * dx)

    best_c5 = compute_c5(h)
    best_h[:] = h

    for iter in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        h_shifted = alpha * h + (1 - alpha) * h_prev
        h_shifted = np.clip(h_shifted, 0, 1)
        h_prev[:] = h  # Save old h before CVX

        # Compute all cross-correlations
        correlations = np.correlate(h_shifted, 1 - h, mode='full') * dx
        k_max = np.argmax(correlations)

        # CVXPy optimization
        h_var = cp.Variable(n_points)
        obj = correlations[k_max]
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]

        # Re-define problem only in terms of minimizing correlations[k_max]
        # Since correlations[k] = sum(h_var[i] * (1 - h_shifted[(i + k) % n_points]))
        # For fixed k_max, define it as linear obj
        corr_coeff = np.array([(1 - h_shifted[(i + k_max) % n_points]) for i in range(n_points)])
        obj = -cp.sum(h_var * corr_coeff)

        problem = cp.Problem(cp.Minimize(obj), constraints)
        try:
            problem.solve(solver=cp.LINEAR, verbose=False)
        except:
            problem.solve(solver=cp.GLOP, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h[:] = h_var.value
            h = np.clip(h, 0, 1)  # Safety bound

            current_c5 = compute_c5(h)
            if current_c5 < best_c5 - 1e-6:
                best_c5 = current_c5
                best_h[:] = h
                iter_without_improvement = 0
                alpha = max(alpha * 1.0, 0.1)
            else:
                iter_without_improvement += 1
                if iter_without_improvement >= 50:
                    alpha *= alpha_decay_factor
                    iter_without_improvement = 0
        elif problem.status in [cp.INFEASIBLE, cp.UNBOUNDED]:
            h[:] = best_h
            alpha *= alpha_decay_factor

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
