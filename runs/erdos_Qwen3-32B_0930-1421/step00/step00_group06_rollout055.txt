import numpy as np
import time
import cvxpy as cp

def compute_c5(h, dx):
    n = len(h)
    h_cors = np.correlate(h, 1 - h, mode="full")
    return np.max(h_cors * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Use initial_h_values from harness
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points

    max_iter = 300
    rho = 1.0
    eps_abs = 1e-3
    eps_rel = 1e-3
    max_rho = 1e4
    n_lags = n_points - 1  # All correlated lags

    # Initialize auxiliary variables: z_k ≈ overlap(k)
    z = np.zeros(n_lags)
    u = np.zeros(n_lags)

    for iter_count in range(max_iter):
        t_iter = time.time()
        if t_iter - t0 > budget_s:
            break

        # Compute overlaps for all lags
        h_cors = np.correlate(h, 1 - h, mode='full')
        overlaps = h_cors[1:n_points]
        obj_vals = overlaps * dx

        # Primal residual
        r = obj_vals - z
        primal_norm = np.linalg.norm(r)

        # Update z
        z_new = np.maximum(obj_vals, z + u)

        # Dual update
        u += obj_vals - z_new

        # Project h to feasible set: [0,1], normalize to integral constraint
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / np.sum(h)

        # Check convergence
        dual_norm = np.linalg.norm(z_new - z)
        z = z_new

        abs_eps = eps_abs * (1 + np.linalg.norm(obj_vals))
        rel_eps = eps_rel * (np.linalg.norm(z) + np.linalg.norm(u))
        if primal_norm < abs_eps and primal_norm < rel_eps and dual_norm < abs_eps and dual_norm < rel_eps:
            break

        # Update rho if necessary
        if primal_norm > 10 * dual_norm and rho < max_rho:
            rho *= 2

    c5_bound = compute_c5(h, dx)
    return (h.tolist(), c5_bound, n_points)
