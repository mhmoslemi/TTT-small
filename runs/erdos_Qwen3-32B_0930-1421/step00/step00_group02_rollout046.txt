import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 45
        dy = 2.0 / n_points
        h = np.full(n_points, 0.5 / dy)
        h[: int(n_points / 4)] = 1.0 / dy
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

    dx = 2.0 / n_points
    h *= dx
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)

    budget_remaining = budget_s
    best_h = np.copy(h)
    best_c5 = np.inf
    max_iter = 500
    start_time = time.time()

    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1:]  # Only positive lags
        max_pos = np.argmax(corr)
        S_kest = corr[max_pos]
        grad = np.zeros_like(h)

        for i in range(n_points):
            for j in range(n_points):
                if j - i == max_pos:
                    grad[i] += 1 - h[j]
                    grad[j] -= h[i]

        H_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(H_var - (h - 0.05 * grad)))
        constraints = [
            0 <= H_var,
            H_var <= 1,
            cp.sum(H_var) == n_points / 2,
        ]
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS)
            h = H_var.value
        except:
            h = best_h
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        corr_new = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr_new[n_points - 1:]) * dx
        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

        if time.time() - start_time > budget_s:
            break

    return (best_h * dx).tolist(), best_c5, n_points
