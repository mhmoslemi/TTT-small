import numpy as np
import time
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h0 = initial_h_values.copy()
    n_points = len(h0)

    dx = 2.0 / n_points

    def compute_c5(h):
        outer = np.zeros_like(h)
        for k in range(1, n_points):
            h_shifted = np.roll(h, -k)
            val = np.sum(h * (1 - h_shifted))
            outer[k] = val
        vals = outer[1:] * dx
        return np.max(vals)

    def clip_and_project(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        target_sum = n_points / 2
        if abs(s - target_sum) < 1e-10:
            return h_clipped
        diff = target_sum - s
        if diff > 0:
            # increase some values
            mask = (h_clipped >= np.median(h_clipped))
            to_change = h_clipped[mask]
            num_change = np.sum(mask)
            step = diff / num_change
            h_clipped[mask] += step
        else:
            # decrease some values
            mask = (h_clipped <= np.median(h_clipped))
            to_change = h_clipped[mask]
            num_change = np.sum(mask)
            step = -diff / num_change
            h_clipped[mask] -= step
        return np.clip(h_clipped, 0, 1)

    def sa_step(h, T, sigma):
        h_perturbed = h + sigma * np.random.randn(n_points)
        h_feasible = clip_and_project(h_perturbed)
        c5 = compute_c5(h_feasible)
        return h_feasible, c5

    start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    T = 1.0
    alpha = 0.99
    sigma = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        h_perturbed, c5 = sa_step(best_h, T, sigma)
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / T):
            best_h = h_perturbed.copy()
            best_c5 = c5
        T *= alpha
        sigma *= np.sqrt(1 - time.time() / (start_time + budget_s))

    def grad(h):
        h = h.copy()
        outer = np.zeros_like(h)
        for k in range(1, n_points):
            h_shifted = np.roll(h, -k)
            val = np.sum(h * (1 - h_shifted))
            outer[k] = val
        max_idx = np.argmax(outer[1:]) + 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            dx_shifted = np.where(i >= max_idx, 0, -(1 - h[(i + max_idx) % n_points]))
            grad[i] = (1 - h[(i - max_idx) % n_points]) + dx_shifted
        grad *= dx
        return -grad

    def obj(h):
        return compute_c5(h)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    def callback(h, val=None):
        nonlocal best_h, best_c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Run GD
    opt = minimize(obj, x0=best_h, method='L-BFGS-B', bounds=bounds,
                   constraints=cons, jac=grad, callback=callback)
    if opt.success:
        best_h = opt.x.copy()
        best_c5 = compute_c5(best_h)
    best_h = clip_and_project(best_h)
    best_c5 = compute_c5(best_h)

    # Return best solution found within time
    return best_h.tolist(), best_c5, n_points
