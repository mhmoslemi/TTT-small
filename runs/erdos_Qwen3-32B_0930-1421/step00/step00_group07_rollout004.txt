import numpy as np
import time
import math
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup from parent construction
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    # Convert h into intervals for effective manipulation
    def h_to_intervals(h, dx):
        intervals = []
        a = None
        for i in range(n_points):
            x = i * dx
            if h[i] > 0.5:
                if a is None:
                    a = x
            else:
                if a is not None:
                    b = x
                    intervals.append((a, b))
                    a = None
        if a is not None:
            intervals.append((a, 2.0))
        return intervals

    def intervals_to_h(intervals, n_points):
        h = np.zeros(n_points, dtype=np.float64)
        for a, b in intervals:
            start_idx = int(math.floor(a / dx))
            end_idx = int(math.ceil(b / dx))
            start_idx = max(start_idx, 0)
            end_idx = min(end_idx, n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1
        # Normalize area to 1
        if np.sum(h) > 1e-9:
            h *= n_points / 2 / np.sum(h)
        return h

    def evaluate_C5(h, dx):
        corr = correlate(h, 1.0 - h, mode="full")
        C5 = np.max(corr * dx)
        return C5

    # Initial state
    current_intervals = h_to_intervals(h, dx)
    current_h = h.copy()
    best_C5 = evaluate_C5(current_h, dx)
    best_h = current_h.copy()
    current_C5 = best_C5

    # Anneal parameters
    T = 1.0
    alpha = 0.995
    num_tries = 0
    max_attempt = 3000
    step_size = 0.02
    adaptive_clip = 0.02
    stalling = 0
    max_stall = 300

    # Optimizer loop
    while time.time() - start_time < budget_s and num_tries < max_attempt and stalling < max_stall:
        # Choose an interval to adjust
        if len(current_intervals) == 0:
            break
        i = np.random.choice(len(current_intervals))
        a, b = current_intervals[i]
        delta = np.random.normal(0.0, step_size)

        if np.random.rand() < 0.5:
            # Shift start of interval
            a_new = max(a + delta, 0.0)
            if i > 0:
                a_new = max(a_new, current_intervals[i-1][1] + 1e-8)
            if a_new < b:
                new_intervals = current_intervals.copy()
                new_intervals[i] = (a_new, b)
                if i < len(new_intervals) - 1:
                    next_a, next_b = new_intervals[i+1]
                    if a_new > next_a - 1e-8:
                        new_intervals[i] = (a_new, next_a - 1e-8)
                current_intervals[i] = (a_new, b)
        else:
            # Shift end of interval
            b_new = min(b + delta, 2.0)
            if i < len(current_intervals) - 1:
                b_new = min(b_new, current_intervals[i+1][0] - 1e-8)
            if b_new > a:
                new_intervals = current_intervals.copy()
                new_intervals[i] = (a, b_new)
                if i > 0:
                    prev_a, prev_b = new_intervals[i-1]
                    if b_new < prev_b + 1e-8:
                        new_intervals[i] = (prev_b + 1e-8, b_new)
                current_intervals[i] = (a, b_new)

        # Build new h
        h_new = intervals_to_h(current_intervals, n_points)

        # Evaluate candidate
        C5_new = evaluate_C5(h_new, dx)

        # Accept if improves or via metropolis criterion
        if C5_new < current_C5 or np.random.rand() < np.exp( (current_C5 - C5_new) / max(T, 1e-5) ):
            current_h = h_new.copy()
            current_C5 = C5_new
            if C5_new < best_C5:
                best_h = current_h.copy()
                best_C5 = C5_new
                stalling = 0
            else:
                stalling += 1
        else:
            stalling += 1

        num_tries += 1
        T *= alpha
        if stalling >= 150:
            step_size = max(0.005, step_size * 0.8)

    return best_h, best_C5, n_points
