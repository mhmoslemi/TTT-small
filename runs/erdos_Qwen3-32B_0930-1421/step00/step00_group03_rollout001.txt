import numpy as np
import time

def objective(h, dx):
    # Compute the maximum overlap integral
    n_points = h.size
    h1 = 1 - h
    corr = np.correlate(h, h1, mode="full")
    corr_center = corr[n_points:2 * n_points]
    return np.max(corr_center * dx)

def optimize_h(initial_h, n_points, dx, budget_s, tolerance=1e-10, max_epochs=1000):
    h = np.array(initial_h, copy=True, dtype=np.float64)
    start_time = time.time()
    min_c5 = float("inf")
    best_h = np.copy(h)

    for epoch in range(max_epochs):
        # Enforce constraints
        h = np.clip(h, 0.0, 1.0)
        while np.sum(h) != n_points / 2.:
            scaling = (n_points / 2.) / np.sum(h)
            h = h * scaling
            h = np.clip(h, 0.0, 1.0)

        # Compute objective and gradients numerically
        c5 = objective(h, dx)

        if c5 < min_c5:
            min_c5 = c5
            best_h = np.copy(h)

        if time.time() - start_time > budget_s:
            break

        # Gradient update (simple numerical gradient)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h[i] += 1e-5
            c_plus = objective(h, dx)
            h[i] -= 2e-5
            c_minus = objective(h, dx)
            h[i] += 1e-5
            grad[i] = (c_plus - c_minus) / (2e-5)

        learning_rate = 0.01 * (0.95 ** epoch)
        h -= learning_rate * grad

    h = best_h
    h = np.clip(h, 0.0, 1.0)
    scaling = (n_points / 2.) / np.sum(h)
    h = h * scaling
    h = np.clip(h, 0.0, 1.0)
    c5_bound = objective(h, dx)
    return h, c5_bound
