import numpy as np
import math
import time
from functools import partial

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    c5 = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return c5

def normalize_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    target_sum = n_points / 2.0
    scale = target_sum / h_sum
    h = np.clip(h * scale, 0, 1)
    return h

def mutate_h(h, n_points, mutation_rate=0.03):
    n_mutations = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, n_mutations, replace=False)
    h[indices] = np.random.rand(n_mutations)
    h = np.clip(h, 0, 1)
    return h

def crossover_h(parent1, parent2, n_points):
    s = np.random.randint(n_points // 4, 3 * n_points // 4)
    child = np.concatenate([parent1[:s], parent2[s:]])
    return child

def evaluate_and_normalize(h, n_points):
    h = normalize_h(h, n_points)
    c5 = evaluate_c5(h, n_points)
    return h, c5
