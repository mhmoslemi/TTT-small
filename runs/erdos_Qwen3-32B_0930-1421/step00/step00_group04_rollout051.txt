import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_candidate(h_values: np.ndarray):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = h_values.copy()
    h[h < 0] = 0.0
    h[h > 1.0] = 1.0
    h_sum = h.sum()
    assert h_sum > 0, "Zero vector or empty vector not allowed"
    h = h / h_sum * (n_points / 2.0)
    ones_h = np.ones_like(h) - h
    corrs = correlate(h, ones_h, mode="full")
    c5 = np.max(corrs) * dx
    return h, c5

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    initial_n = 42
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True)
        h[np.abs(h) > 1.0] = 0.0
        h_sum = np.clip(h.sum(), 1e-6, None)
        h = h / h_sum * (initial_n / 2.0)
        n_points = initial_n
    else:
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    learning_rate = 0.01
    best_h = h.copy()
    best_c5 = np.inf
    step_count = 0
    max_steps = 10_000
    proj_freq = 10
    early_stop = 100

    while (time.time() - start_time) < budget_s and step_count < max_steps:
        h_eval, c5 = evaluate_erdos_candidate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_eval.copy()
            counter = 0
            print(f"New best C5: {best_c5:.6f}")
        else:
            counter += 1
            if counter > early_stop:
                print(f"Early stopping at step {step_count}")
                break

        # Gradient step
        ones_h = np.ones_like(h) - h
        corrs = correlate(h, ones_h, mode="full")
        grad = np.gradient(corrs).copy()
        grad_shifted = np.zeros_like(h)
        for i in range(1, len(h)+1):
            idx = len(h)-1 + i -1
            grad_shifted = grad_shifted + grad[i]*i / (2.0 / len(h))
        h = h - learning_rate * grad_shifted

        # Projection
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h / h_sum * (n_points / 2.0)

        step_count += 1

    return best_h, best_c5, n_points
