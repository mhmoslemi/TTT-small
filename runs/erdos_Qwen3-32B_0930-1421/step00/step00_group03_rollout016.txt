import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Create a uniform distribution starting point
        h = np.zeros(n_points)
        h[:35] = 1.0 / 35 * n_points / 2
        h[35:] = 0.0
        h /= dx  # Normalize to satisfy integral constraint

    # Feasibility projection
    def project_h(h):
        h = np.clip(h, 0, 1)
        h -= (np.sum(h) - n_points / 2) / np.sum(h != 0.5)  # Adjust minimal change to satisfy sum constraint
        return np.clip(h, 0, 1)

    h = project_h(h)

    # Compute overlap integrals for all shifts
    def compute_c5(h):
        correlate = np.correlate(h, 1 - h, mode='full')
        valid_correlate = correlate[n_points - 1: 2 * n_points - 1]
        c5 = np.max(valid_correlate * dx)
        return c5

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    learning_rate = 0.01
    num_iter = 0
    max_iterations = int(budget_s * 100 / n_points)

    while time.time() - start_time < budget_s and num_iter < max_iterations:
        num_iter += 1

        # Compute numerical gradient of C5
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - best_c5) / eps

        # Normalize gradient to unit vector
        grad = np.nan_to_num(grad / (np.linalg.norm(grad) + 1e-8))
        
        # Take gradient descent step and project back
        h_new = h - learning_rate * grad
        h_new = project_h(h_new)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            learning_rate *= 1.1  # Increase learning rate if we're improving
        else:
            learning_rate *= 0.9  # Decrease learning rate if we're not improving

        h = h_new

    return (best_h, best_c5, n_points)
