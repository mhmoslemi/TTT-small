import numpy as np
import time
from itertools import islice
from operator import itemgetter

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Load initial solution and set defaults
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Set GA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.05
    elitism_ratio = 0.2
    tournament_size = 10

    def normalize(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        correction = target_sum / current_sum if current_sum != 0 else 1.0
        h_corrected = h_clipped * correction
        return np.clip(h_corrected, 0, 1)

    def objective(h):
        h_norm = normalize(h)
        sum_h = np.sum(h_norm)
        assert np.isclose(sum_h, target_sum), f"Sum is {sum_h}, should be {target_sum}"
        h_flip = 1 - h_norm
        corrs = np.correlate(h_norm, h_flip, mode="full") * dx
        return np.max(corrs)

    def create_individual():
        h = initial_h_values.copy() if np.random.rand() < 0.5 else np.random.rand(n_points)
        return normalize(h)

    def crossover(parent1, parent2):
        s = np.random.randint(1, n_points - 2)
        child = np.concatenate((parent1[:s], parent2[s:]))
        return normalize(child)

    def mutate(individual):
        n_mut = np.random.randint(1, max(2, int(n_points * mutation_rate)))
        idxs = np.random.choice(n_points, size=n_mut, replace=False)
        individual[idxs] = np.random.rand(n_mut)
        return normalize(individual)

    def tournament_selection(pop, fit, size):
        contestant_idx = np.random.choice(len(pop), size=size, replace=False)
        return pop[contestant_idx[fit[contestant_idx].argmin()]]

    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    best_c5 = objective(population[0])
    best_h = population[0].copy()

    for gen in range(generations):
        current_time = time.time()
        if current_time - start_time > budget_s - 30:  # reserve time for finalization
            break

        # Evaluate all individuals
        fitness = [objective(ind) for ind in population]
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()

        # Select elites
        elite_count = max(1, int(pop_size * elitism_ratio))
        elite_indices = np.argsort(fitness)[:elite_count]
        elites = [population[i] for i in elite_indices]

        # Generate new population
        new_population = elites.copy()
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)

        population = new_population

    return best_h, best_c5, n_points
