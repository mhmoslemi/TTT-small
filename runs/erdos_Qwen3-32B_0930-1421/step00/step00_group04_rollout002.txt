import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 42
    h = initial_h_values.copy()
    n_points = len(h)

    # Check constraint
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-6

    start_time = time.time()
    max_time = budget_s
    alpha = 0.1  # Learning rate
    best_c5 = float('inf')
    best_h = h.copy()

    def evaluate(h):
        k_vals = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(k_vals) * dx
        return c5

    # Optimization loop
    while time.time() - start_time < max_time:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient (simplified approximation)
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n_points - 1)
        if -n_points + 1 < shift < 0:
            # Left overlap
            deriv = dx * (h[:(n_points + shift)] - (1 - h[-shift:]))
            h += alpha * deriv
        elif 0 < shift < n_points - 1:
            # Right overlap
            deriv = dx * (h[shift:] - (1 - h[:(n_points - shift)]))
            h += alpha * deriv
        else:
            # Full or no overlap
            h += alpha * dx * (h - h)
            continue

        # Projection step: clip and normalize
        h[h > 1] = 1
        h[h < 0] = 0
        h *= (n_points / 2) / np.sum(h)

    c5_bound = evaluate(best_h)
    return (best_h.tolist(), c5_bound, n_points)
