import numpy as np
import time
from scipy.optimize import minimize, Bounds
from math import exp

def run(seed=42, budget_s=1100, n_points=47):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n = n_points  # shorthand

    def clip_and_normalize(h):
        h_clipped = np.clip(h, 0, 1)
        return h_clipped * (target_sum / np.sum(h_clipped))

    def objective_smooth(h, overlaps, weights=None, smoothness=1.0):
        corrs = np.zeros(len(overlaps))
        for i in range(len(overlaps)):
            if overlaps[i] is not None:
                corrs[i] = np.sum(h * overlaps[i])
        corrs *= dx
        if weights is None:
            return np.max(corrs)
       lse = np.log(np.sum(weights * np.exp(smoothness * corrs))) / smoothness
        return lse

    def objective_true(h, overlaps):
        corrs = np.zeros(len(overlaps))
        for i in range(len(overlaps)):
            if overlaps[i] is not None:
                corrs[i] = np.sum(h * overlaps[i]) * dx
        return np.max(corrs)

    def make_overlaps(h):
        max_len = 2 * n - 1
        overlaps = [None] * max_len
        h_ext = np.concatenate([h, np.zeros_like(h)])  # pad for full convolution
        for k in range(1, n):
            overlap_k = 1 - np.roll(h, -k)
            overlap_k = overlap_k[:-k]
            overlaps[k] = overlap_k
            overlaps[-k-1] = overlap_k[::-1]  # symmetric for negative shifts
        overlaps[0] = np.zeros_like(h)
        return overlaps

    def initial_solution():
        if 'initial_h_values' in globals():
            return clip_and_normalize(np.array(initial_h_values, dtype=float))
        else:
            h = np.random.rand(n)
            return clip_and_normalize(h)

    def surrogate_evaluate(h, overlaps, smoothness=10.0):
        return objective_smooth(h, overlaps, smoothness=smoothness)

    def simulated_annealing(h0, overlaps, T0=100.0, cooling=0.995, max_iter=500):
        h = h0.copy()
        best_h = h.copy()
        T = T0
        best_c5 = objective_true(h, overlaps)
        for _ in range(max_iter):
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.2
            new_h = h.copy()
            new_h[idx] = np.clip(new_h[idx] + delta, 0, 1)
            new_h = clip_and_normalize(new_h)
            current_c5 = objective_true(new_h, overlaps)
            if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / T):
                h = new_h
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
            T *= cooling
        return best_h

    overlaps = make_overlaps(np.zeros(n))
    best_h = initial_solution()
    best_c5 = objective_true(best_h, overlaps)
    start_time = time.time()
    elapsed = 0

    # Hybrid optimization
    while elapsed < budget_s:
        h0 = best_h.copy()
        idx = np.random.choice(n, int(n * 0.1), replace=False)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(idx))
        h0[idx] = np.clip(h0[idx] + perturbation, 0, 1)
        h0 = clip_and_normalize(h0)

        bounds = Bounds(0, 1)
        result = minimize(
            lambda h: surrogate_evaluate(h, overlaps, smoothness=20.0),
            h0,
            method='L-BFGS-B',
            bounds=bounds,
        )
        post_opt = result.x if not result.success else h0
        post_opt = clip_and_normalize(post_opt)

        post_c5 = objective_true(post_opt, overlaps)
        if post_c5 < best_c5:
            best_h = post_opt
            best_c5 = post_c5

        sa_h = simulated_annealing(best_h, overlaps)
        sa_c5 = objective_true(sa_h, overlaps)
        if sa_c5 < best_c5:
            best_h = sa_h
            best_c5 = sa_c5

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
