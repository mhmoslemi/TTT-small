import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h to minimize the Erdős constant C₅."""
    start_time = time.time()
    dx_tiny = 1e-8
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    np.random.seed(seed)

    def evaluate_c5(h):
        h_padded = np.concatenate([h, 1 - h])
        corr_full = np.correlate(h, 1 - h, mode='full')
        corr_full[n_points:] *= 1.5  # adjust the tail for partial overlaps
        return np.max(corr_full * dx)

    best_h = np.copy(h)
    best_c5 = evaluate_c5(h)
    t_inner = time.time()

    # ADMM parameters
    rho = 1.0
    eps_abs = 1e-4
    eps_rel = 1e-3
    max_iter = 1000
    dual_weight = 1.0
    iter = 0

    n_lags = 2 * n_points
    z = np.copy(best_c5)
    u = np.full_like(z, 0.0)

    while time.time() - start_time < budget_s:
        iter += 1
        h_flattened = h.flatten()
        h_vals = h_flattened.reshape((1, -1))
        hh = np.concatenate((h_vals, 1 - h_vals), axis=0)
        overlaps = correlate(h_vals, 1 - h_vals, mode='full')
        overlaps[n_points:] *= 1.5
        overlaps *= dx

        # Update z
        z_update = (overlaps * (1.0 - u) + 2 * z) / (2 + rho)
        z = np.maximum(np.minimum(z_update, 1.0), 0.0)

        # Build objective function
        obj_vals = z - u
        grad = np.zeros_like(h)
        for k in range(n_points):
            i0 = max(0, n_points - k - 1)
            i1 = min(n_points, n_points + k)
            J_nz = slice(k, n_points + k)
            grad_term = (1 - 2 * h[J_nz]) * 2  # derivative w.r. to h at lag k
            grad += grad_term

        # Project h to feasible values
        h_step = h - grad.flatten() / (2 * rho) + u.flatten() / rho
        h_projected = np.clip(h_step, 0.0, 1.0)

        # Normalize the integral
        norm = np.sum(h_projected) * dx
        if abs(norm - 1.0) > 1e-5:
            scaling_factor = np.sqrt((1.0 / (norm)) / (1.0 / norm))
            h_projected = h_projected * scaling_factor

        h = h_projected

        # Primal residual
        r = h - z + u
        # Dual residual
        s = rho * (np.mean((h + z) / 2))

        # Update Lagrange multiplier u
        u += r

        # Check convergence
        if np.max(np.abs(r)) < eps_abs and np.max(np.abs(s)) < eps_abs:
            print("Converged")
            break

        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

        if iter >= max_iter:
            break

    # Final check at the end of optimization
    h_best = np.copy(best_h)
    norm = np.sum(h_best) * dx
    if not np.isclose(norm, 1.0, atol=1e-8):
        scaling_factor = 1.0 / norm
        h_best *= scaling_factor
    h_best = np.clip(h_best, 0.0, 1.0)

    return h_best, best_c5, n_points
