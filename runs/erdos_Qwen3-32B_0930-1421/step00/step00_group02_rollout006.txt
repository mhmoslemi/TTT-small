import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    # Normalize to ensure the constraint sum(h) == n_points / 2
    total_h = float(np.sum(initial_h))
    scale_factor = (n_points / 2.0) / total_h
    h = np.minimum(np.copy(initial_h) * scale_factor, 1.0)

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    last_improvement_time = start_time

    def objective(x):
        # x is the vector of h values
        c5 = evaluate_erdos_solution(x, n_points)
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    def constraint_upper(x):
        return 1.0 - x

    def constraint_lower(x):
        return x

    def optimize_with_bounds(optimizer="SLSQP", max_iter=100, callback=None):
        nonlocal h, best_c5, best_h, last_improvement_time

        cons = (
            {"type": "eq", "fun": constraint_sum},
            {"type": "ineq", "fun": constraint_upper},
            {"type": "ineq", "fun": constraint_lower}
        )

        result = minimize(
            objective,
            h,
            method=optimizer,
            bounds=[(0.0, 1.0)] * n_points,
            constraints=cons,
            options={"maxiter": max_iter},
            callback=callback
        )

        if result.success:
            new_h = result.x
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h.copy()
                last_improvement_time = time.time()

    while time.time() - start_time < budget_s - 5:  # leave 5s margin
        optimize_with_bounds(optimizer="SLSQP", max_iter=100)
        if time.time() - last_improvement_time > 120:
            break

    return best_h, best_c5, n_points
