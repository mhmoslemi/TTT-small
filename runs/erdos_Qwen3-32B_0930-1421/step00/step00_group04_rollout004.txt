import numpy as np
import time
import heapq
import random
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    n_points = 42
    dx = 2.0 / n_points
    total_mass = n_points / 2

    def is_valid(h):
        return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - total_mass) < 1e-8

    def compute_c5(h, dx):
        h = np.asarray(h)
        corr = correlate(h, 1 - h[::-1], mode='full')
        return np.max(corr * dx)

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = total_mass / s
        h = h * scale
        h = np.clip(h, 0, 1)
        error = total_mass - np.sum(h)
        while error > 1e-5:
            idx = np.argmin(h)
            if h[idx] + error <= 1:
                h[idx] += error
                break
            else:
                addable = 1 - h[idx]
                h[idx] += addable
                error -= addable
        return h

    def random_individual():
        h = np.random.rand(n_points)
        h = repair(h)
        return h

    def evaluate(individual):
        return compute_c5(individual, dx)

    def crossover(a, b):
        alpha = np.random.rand(n_points)
        c = alpha * a + (1 - alpha) * b
        c = repair(c)
        return c

    def mutate(individual, rate=0.1):
        mask = np.random.rand(n_points) < rate
        noise = np.random.rand(n_points) * 0.2
        c = individual.copy()
        c[mask] += noise[mask]
        c[~mask] -= noise[~mask] / (n_points - np.sum(mask))
        c = repair(c)
        return c

    def tournament_selection(population, fitnesses, n_tournament=3):
        indices = np.random.choice(len(population), n_tournament)
        return population[indices[np.argmin(fitnesses[indices])]]

            history_best_individual = best_individual.copy()

        generations += 1

    best_c5 = compute_c5(history_best_individual, dx)
