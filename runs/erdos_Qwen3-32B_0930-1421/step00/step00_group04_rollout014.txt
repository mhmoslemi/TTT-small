import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1 - h
    cors = correlate(h, one_minus_h, mode="full")
    return np.max(cors) * dx

def rescale(h, n_points):
    required_sum = n_points / 2.0
    total = np.sum(h)
    return h * (required_sum / total)

    start_time = time.time()

    for iter in range(max_steps):
        if time.time() - start_time > budget_s:
            break

        # Gradient step (approximate using finite differences)
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            hp = h.copy()
            hm = h.copy()
            hp[i] = np.clip(hp[i] + eps, 0, 1)
            hm[i] = np.clip(hm[i] - eps, 0, 1)

            cm = evaluate_erdos_solution(hm, n_points)
            grad[i] = (cp - cm) / (2 * eps)

        h = h - 0.01 * grad
        h = np.clip(h, 0, 1)
        h = rescale(h, n_points)

        # Periodic simplex restart
        if iter % step_update_rate == 0:
            # Define a simplex around current h
            simplex = [h.copy()]
            for j in range(3):
                simplex.append(h + 0.1 * np.random.randn(n_points))
                simplex[j + 1] = np.clip(simplex[j + 1], 0, 1)
                simplex[j + 1] = rescale(simplex[j + 1], n_points)

            # Evaluate C₅ for each simplex point
            values = [evaluate_erdos_solution(simplex[k], n_points) for k in range(4)]
            best_simplex_idx = np.argmin(values)
            h = simplex[best_simplex_idx]

        # Update best solution
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
