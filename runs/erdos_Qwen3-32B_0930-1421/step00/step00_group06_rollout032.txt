import numpy as np
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = h.size
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start_time = time.time()

    # Hyperparameters
    alpha = 0.01
    decay_rate = 0.95
    max_iter = 1000
    tolerance = 1e-6
    n_warn_proj = 0

    def compute_c5(h, dx):
        h_1m = 1 - h
        corr = np.correlate(h, h_1m, mode='full')
        corr *= dx
        c5 = np.max(corr[n_points:(2*n_points)])
        return c5

    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        h_1m = 1.0 - h
        corr = np.correlate(h, h_1m, mode='full')
        lag_corr = corr[n_points:(2*n_points)] * dx
        k = np.argmax(lag_corr)

        # Compute gradient
        idxs = np.arange(n_points)
        idxs_k_pos = (idxs + k) % n_points
        idxs_k_neg = (idxs - k) % n_points
        grad = (1.0 - h[idxs_k_pos]) - h[idxs_k_neg]

        # Update step
        h_update = h - alpha * grad

        # Projection onto [0,1]
        h = np.clip(h_update, 0.0, 1.0)

        # Normalize to maintain the integral constraint
        h *= n_points / 2.0 / np.sum(h)

        # Check constraint
        if abs(np.sum(h) - n_points / 2.0) > 1e-6 and n_warn_proj < 3:
            h = best_h.copy()
            n_warn_proj += 1

        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Adjust learning rate
        alpha *= decay_rate

        # Convergence check
        if i > 0 and abs((prev_c5 - best_c5) / prev_c5) < tolerance:
            break
        prev_c5 = best_c5

    return best_h, best_c5, n_points
