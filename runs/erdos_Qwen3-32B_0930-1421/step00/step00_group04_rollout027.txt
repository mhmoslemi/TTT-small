        fft_h = fft(h)
        one_minus_h = 1 - h
        fft_one_minus_h = fft(one_minus_h)

        cross_corr_fft = fft_h * fft_one_minus_h.conjugate()
        cross_corr = ifft(cross_corr_fft).real * dx

    patience = 100

    for _iter in range(max_iter):
        elapsed = time.time() - start
        if elapsed >= budget_s - 1:
            break

        # Compute gradient using approx derivative
        grad = np.zeros_like(h)
        for shift in [1, -1]:
            h_shifted = np.roll(h, shift)
            for i in range(n_points):
                idx = (i + shift) % n_points
                grad[i] += -(h[i] - h_shifted[i])
        grad = grad / n_points

        h_candidate = h - alpha * grad
        h_candidate = np.clip(h_candidate, 0, 1)

        # Project to feasible set
        current_sum = h_candidate.sum()
        scale_factor = (n_points / 2) / current_sum

        new_c5 = evaluate()

            best_h = h_candidate.copy()

            no_improvement = 0
