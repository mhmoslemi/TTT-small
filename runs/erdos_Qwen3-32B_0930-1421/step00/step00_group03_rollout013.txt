import numpy as np
import time
import random
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Extract information
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    # Constants for PSO
    n_particles = 50
    max_iter = 500
    w = 0.7  # Inertia
    c1 = 1.5  # Personal coefficient
    c2 = 1.5  # Social coefficient

    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros_like(particles)
    personal_best = np.zeros_like(particles)
    global_best = np.zeros(n_points)

    # Use initial_h_values as a starting point for each particle
    for i in range(n_particles):
        particles[i] = h_init * (1 + 0.1 * np.random.randn(n_points))
        particles[i] = np.clip(particles[i], 0, 1)
        # Enforce integral constraint
        scale = n_points / 2.0
        particles[i] = particles[i] / np.sum(particles[i]) * scale
        particles[i] = np.clip(particles[i], 0, 1)
        personal_best[i] = particles[i].copy()

    # Evaluate and find global best
    gbest_idx = 0
    gbest_score = float("inf")
    for i in range(n_particles):
        score = evaluate_particle(particles[i], dx)
        if score < gbest_score:
            gbest_score = score
            global_best[:] = particles[i]

    # Track time and results
    start_time = time.time()
    best_h = global_best.copy()
    best_c5 = gbest_score
    iter_no_improvement = 0
    prev_best_c5 = best_c5

    # PSO main loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Update velocities
        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)
        velocities = w * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (
            np.tile(global_best, (n_particles, 1)) - particles
        )

        # Update positions
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            # Enforce integral constraint
            scale = n_points / 2.0
            particles[i] = particles[i] / np.sum(particles[i]) * scale
            particles[i] = np.clip(particles[i], 0, 1)

        # Evaluate and update personal and global best
        for i in range(n_particles):
            score = evaluate_particle(particles[i], dx)
            if score < gbest_score:
                gbest_score = score
                global_best[:] = particles[i]
                best_h[:] = particles[i]
                best_c5 = gbest_score
                iter_no_improvement = 0
                prev_best_c5 = best_c5
            if score < evaluate_particle(personal_best[i], dx):
                personal_best[i, :] = particles[i, :]

        # Early stopping if no improvement
        if abs(best_c5 - prev_best_c5) < 1e-5:
            iter_no_improvement += 1
            if iter_no_improvement >= 50:
                break

    return (best_h, best_c5, n_points)


def evaluate_particle(h, dx):
    # Compute 1 - h and cross-correlate
    h_one_minus = 1 - h
    # Use correlate with "full" mode and take the middle n_points - 1 + n_points = 2*n_points - 1 values
    correlations = correlate(h, h_one_minus, mode="full")
    # Rescale by dx
    correlations = correlations * dx
    return np.max(correlations)
