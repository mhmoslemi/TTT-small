import numpy as np

def run(seed=42, budget_s=1100):
    start_time = time.time()

    # Initial parameters
    initial_n_points = len(initial_h_values)
    h = cp.Variable(initial_n_points, nonneg=True)
    dx = 2.0 / initial_n_points

    # Objective and constraints
    constraint = [h >= 0, h <= 1, cp.sum(h) == initial_n_points / 2]
    overlaps = []
    for k in range(1, initial_n_points):
        # Construct (1 - h) shifted by k
        shifted = cp.reshape(h[k:], (1, initial_n_points - k))
        left = cp.reshape(h[:initial_n_points - k], (1, initial_n_points - k))
        overlap = cp.sum(left * (1 - shifted))
        overlaps.append(overlap)

    obj = cp.Minimize(cp.max(cp.hstack(overlaps) * dx))

    prob = cp.Problem(obj, constraint)

    # Use Gurobi with progress monitoring
    prob.solve(solver=cp.GUROBI, verbose=False)

    # While no convergence and time allows, repeat with max_iter increase
    timeout = budget_s - 20  # reserve 20s for fallback
    valuation_dt = 5 # seconds
    best_c5 = 1.0
    best_h = initial_h_values
    final_dx = dx

    while time.time() - start_time < timeout and np.isfinite(prob.value) and prob.value < best_c5:

        elapsed = time.time() - start_time
        if elapsed % 60 == 0:
            print(f"Elapsed time {elapsed:.0f} / {budget_s - 20} s, C₅: {prob.value * dx:.8f}")

        best_h = h.value
        overlap_values = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
        if overlap_values < best_c5:
