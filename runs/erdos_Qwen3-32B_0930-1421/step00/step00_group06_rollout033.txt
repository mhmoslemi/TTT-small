import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = compute_c5(h, n_points, dx)
    MAX_ITER = 1000
    learning_rate = 0.02
    decay_rate = 0.97

    for iteration in range(MAX_ITER):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute the overlap integrals for each lag
        overlap = np.zeros(n_points * 2 - 1)
        for k in range(n_points * 2 - 1):
            k_center = k - n_points + 1
            idx_valid = np.where((np.arange(n_points) + k_center) % n_points)
            overlap_vals = h[idx_valid] * (1.0 - h[(idx_valid + k_center) % n_points])
            overlap[k] = np.sum(overlap_vals) * dx

        # Identify the maximum overlap and its corresponding lag
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx]
        k_opt = max_overlap_idx - n_points + 1

        # Compute gradient for C₅
        grad = np.zeros(n_points)
        if k_opt != 0:
            grad = (1.0 - np.roll(h, k_opt)) - np.roll(h, -k_opt)

        # Update h using gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * n_points / 2  # Maintains integral constraint

        h_new_c5 = compute_c5(h_new, n_points, dx)
        if h_new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = h_new_c5

        learning_rate *= decay_rate
        if iteration > 10 and abs(best_c5 - compute_c5(best_h, n_points, dx)) < 1e-6:
            break

    return best_h, best_c5, n_points

def compute_c5(h, n_points, dx):
    kernel = 1 - h
    corr = np.correlate(h, kernel, mode='full')
    c5 = np.max(corr) * dx
    return c5
