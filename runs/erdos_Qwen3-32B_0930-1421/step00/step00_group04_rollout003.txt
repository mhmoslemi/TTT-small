```python
import numpy as np
import time
from scipy.ndimage import convolve1d
from evaluate_erdos_solution import evaluate_erdos_solution

def project_simplex(v):
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    scaling_factor = v_sum / (len(v) / 2)
    return v / scaling_factor

def compute_c5(h, dx):
    h_rev = 1 - h
    conv_result = convolve1d(h, h_rev[::-1], mode='full')
    return np.max(conv_result * dx)

def run(seed=42, budget_s=1100, n_points=120):
    np.random.seed(seed)
    start_time = time.time()
    budget_limit = start_time + budget_s
    alpha = 0.01
    best_c5 = np.inf
    best_h = None

    # Use existing initial_h_values if available, else create a uniform initialization
    if 'initial_h_values' in globals():
        h = project_simplex(initial_h_values.copy())
    else:
        h = np.zeros(n_points)
        idx = int(n_points * 0.5)
        h[idx - 1:idx + 2] = 0.2
        h = project_simplex(h)

    h = h.copy()
    last_improvement = start_time
    patience = 0
    max_patience = 100
    iter = 0

    while time.time() < budget_limit:
        dx = 2.0 / n_points

        # Compute C5 and gradients
        rev_h = 1 - h
        conv_result = convolve1d(h, rev_h[::-1], mode='full')
        max_idx = np.argmax(conv_result)
        shift_k = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        left_side = np.maximum(0, n_points + shift_k - 1)
        right_side = np.minimum(n_points, n_points + shift_k)
        grad[shift_k:right_side] -= h[left_side:shift_k]
        grad[left_side:n_points] += h[shift_k:right_side]

        # Update with projected gradient
        h = project_simplex(h - alpha * grad)

        # Compute and check if it's better
        c5 = compute_c5(h, dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            last_improvement = time.time()
            patience = 0
        else:
            patience += 1

        if time.time() > last_improvement + 20:
            alpha *= 0.95  # reduce learning rate if no improvement

        iter += 1
        if patience >= max_patience:
            break

    return best_h, best_c5, n_points
```
