```python
import numpy as np
import time
import random
import math

def evaluate_overlap(h, dx):
    h = np.array(h, dtype=float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def run(seed=42, budget_s=1100):
    random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        # Convert initial_h_values to binary, ensuring exactly n_points/2 1s
        h = np.array([1.0 if x > 0.5 else 0.0 for x in initial_h_values])
        one_count = sum(h)
        target = n_points / 2.0
        if one_count < target:
            h[random.choice(np.where(h == 0.0)[0])] = 1.0
        elif one_count > target:
            h[random.choice(np.where(h == 1.0)[0])] = 0.0
        h = h.tolist()
    else:
        h = [0.0] * n_points
        for i in random.sample(range(n_points), int(n_points / 2)):
            h[i] = 1.0

    # Annealing parameters
    T = 1.0
    alpha = 0.99
    t0 = time.time()
    patience = 1000
    no_improvement = 0
    best_h = h[:]
    best_c5 = evaluate_overlap(h, dx)

    while time.time() - t0 < budget_s:
        # Flip two bits (1->0 and 0->1) to preserve constraint
        idx1 = random.choice([i for i in range(n_points) if h[i] == 1])
        idx2 = random.choice([i for i in range(n_points) if h[i] == 0])
        new_h = h[:]
        new_h[idx1], new_h[idx2] = 0.0, 1.0

        # Evaluation
        current_c5 = evaluate_overlap(h, dx)
        candidate_c5 = evaluate_overlap(new_h, dx)

        # Acceptance criterion
        if candidate_c5 < current_c5 or random.random() < math.exp(-(candidate_c5 - current_c5) / T):
            h = new_h[:]
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h[:]
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        # Temperature adjustment and reset
        T *= alpha
        if T < 1e-5:
            T = max(T, 1e-5)
            alpha = max(alpha * 0.99, 0.95)
        if no_improvement > patience and random.random() < 0.05:
            T = max(T * 2, 1e-4)
            alpha = 0.99
            no_improvement = 0

    # Final validation
    h = best_h
    assert sum(h) == n_points / 2, (sum(h), n_points / 2)
    c5 = evaluate_overlap(h, dx)
    assert c5 <= 1.0 and c5 >= 0.0, c5

    return (h, c5, n_points)
```
