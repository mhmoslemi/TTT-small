```python
import numpy as np
import time
import random
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 42  # From the given example
    dx = 2.0 / n_points
    total_mass = n_points / 2
    target_c5 = 0.380876  # Our target

    # If initial_h_values provided in the global context, use it
    global initial_h_values
    try:
        initial_population = [initial_h_values.copy()]
    except:
        # Default initialization if no parent provided
        initial_h_values = np.zeros(n_points)
        initial_h_values[:int(n_points / 2)] = 1.0
        initial_population = [initial_h_values.copy()]

    # Genetic algorithm parameters
    pop_size = 20
    num_generations = 300
    mutation_rate = 0.1
    tournament_size = 5
    top_fraction = 0.2  # number of top solutions to retain
    time_start = time.time()
    
    # Build initial population with random individuals
    def build_random_h():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[indices] = 1.0
        return h
    
    for _ in range(pop_size - len(initial_population)):
        initial_population.append(build_random_h())

    # Compute fitness of a candidate
    def calculate_c5(h):
        h_array = np.array(h)
        one_minus_h = 1.0 - h_array
        corr_values = correlate(h_array, one_minus_h, mode="full")
        corr_values = corr_values * dx
        return np.max(corr_values)

    # Evaluate initial population
    population = initial_population
    fitness = [calculate_c5(h) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_c5 = fitness[best_index]

    # Track progress and time
    for gen in range(num_generations):
        # Stop if time is nearly up (keeping 1 second buffer)
        if time.time() - time_start > budget_s - 1:
            break

        # Selection using tournament
        parents = []
        for _ in range(pop_size):
            tournament = random.sample(range(pop_size), tournament_size)
            winner = min(tournament, key=lambda idx: fitness[idx])
            parents.append(population[winner])

        # Crossover operation
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 < len(parents):
                parent_a = parents[i]
                parent_b = parents[i + 1]
                child1 = parent_a.copy()
                child2 = parent_b.copy()
                # One-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1[crossover_point:], child2[crossover_point:] = child2[crossover_point:], child1[crossover_point:].copy()
                # Ensure the sum constraint is satisfied
                for child in [child1, child2]:
                    sum_child = np.sum(child)
                    delta = sum_child - total_mass
                    if abs(delta) > 1e-8:
                        indices = np.where(child > 0)[0]
                        random_index = np.random.choice(indices) if delta < 0 else np.random.choice(np.where(child < 1)[0])
                        child[random_index] += -delta / n_points  # normalize over n_points for diff
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])

        # Mutate offspring
        for child in offspring:
            if random.random() < mutation_rate:
                # Select multiple random positions to flip
                num_flips = np.random.randint(1, int(n_points * 0.15))
                mutation_idx = np.random.choice(n_points, size=num_flips, replace=False)
                for idx in mutation_idx:
                    old_val = child[idx]
                    new_val = old_val + np.random.uniform(-0.5, 0.5)
                    new_val = max(0.0, min(1.0, new_val))
                    child[idx] = new_val

                # Adjust to satisfy sum constraint
                sum_child = np.sum(child)
                delta = sum_child - total_mass
                if abs(delta) > 1e-8:
                    indices = np.where(child > 0)[0]
                    random_index = np.random.choice(indices) if delta < 0 else np.random.choice(np.where(child < 1)[0])
                    child[random_index] += -delta / n_points  # small correction over n_points

        # Evaluate offspring
        for o in offspring:
            c5 = calculate_c5(o)
            # Prune high C5 offspring to save time
            if c5 < best_c5 + 0.1:
                population.append(o)
                fitness.append(c5)

        # Elitist replacement
        population.sort(key=lambda h: calculate_c5(h))
        population = population[:pop_size]
        fitness = [calculate_c5(h) for h in population]

        # Update best solution
        gen_best_index = np.argmin(fitness)
        gen_best_c5 = fitness[gen_best_index]
        if gen_best_c5 < best_c5:
            best_c5 = gen_best_c5
            best_h = population[gen_best_index].copy()
            if best_c5 < target_c5:
                # If already below target, we can return early
                print(f"Early stopping: C5 {best_c5} < target {target_c5}")
                break

    return best_h, best_c5, n_points
```
