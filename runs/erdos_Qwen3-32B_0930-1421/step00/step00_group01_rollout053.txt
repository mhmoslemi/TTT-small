import numpy as np
import time
import random

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)

    MAX_GEN = 1000

    TOP_FRACTION = 0.2
    MUTATION_RATE = 0.05
    INIT_SAMPLES = len(initial_h_values)  # Preserve parent construction
    initial_sum = INIT_SAMPLES / 2.0
    dx = 2.0 / INIT_SAMPLES

    # Fitness function: C5 = max np.correlate(h, 1-h, mode='full') * dx
    def compute_c5(h):
        k_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(k_vals)

    def normalize_h(h):
        # Ensure sum == init_sum and values in [0, 1]
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if abs(current_sum - initial_sum) > 1e-5:
            delta = (initial_sum - current_sum) / INIT_SAMPLES
            h = np.clip(h + delta, 0, 1)
        return h

    # Initialize population: initial_h_values + randomized variations
    population = [np.copy(initial_h_values)]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(INIT_SAMPLES)
        h = normalize_h(h)
        population.append(h)

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    for gen in range(MAX_GEN):
        # Evaluate all candidates
        candidates = []
        for h in population:
            c5 = compute_c5(h)
            candidates.append((c5, h))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.copy(h)

        # Stop if time is up
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Sort candidates by fitness (lowest C5 is best)
        candidates.sort()
        top_idx = int(TOP_FRACTION * len(candidates))
        top_h = [h for c5, h in candidates[:top_idx]]

        # Generate new population via crossover and mutation
        new_pop = []
        while len(new_pop) < POP_SIZE:
            # Tournament selection
            if len(top_h) < 2:
                break
            i1, i2 = random.sample(range(len(top_h)), 2)
            parent1, parent2 = top_h[i1], top_h[i2]
            s = np.random.randint(int(0.2 * INIT_SAMPLES), int(0.8 * INIT_SAMPLES))
            child = np.concatenate([parent1[:s], parent2[s:]])
            # Mutation
            if random.random() < MUTATION_RATE:
                flip_n = int(MUTATION_RATE * INIT_SAMPLES)
                indices = random.sample(range(INIT_SAMPLES), flip_n)
                child[indices] = np.random.rand(flip_n)
            # Normalize
            child = normalize_h(child)
            new_pop.append(child)

        # Refresh population
        population = new_pop

    # Final output
    return best_h.tolist(), best_c5, INIT_SAMPLES
