import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)

    dx = 2.0 / n_points

    # Ensure initial_h_values is valid and feasible
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h)  # Normalize to uniform
        h *= n_points / 2  # Scale to satisfy integral constraint

    # Projection to feasibility region
    def project_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h / total * (n_points / 2)
        return h

    h = project_h(h)

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    # Precompute convolution pattern
    def compute_c5(h, dx):
        full_conv = np.convolve(h, 1.0 - h, mode="full")
        return np.max(full_conv * dx)

    best_c5 = compute_c5(h, dx)

    # Iterate over the improvements
    iteration = 0
    max_iterations = int(budget_s / 1.0)  # Assume ~1s per iteration

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        # Evaluate the overlaps for all possible lags
        # Avoid recomputation by Naive convolution, compute it only once
        conv = np.zeros(2 * n_points - 1)
        for i in range(2 * n_points - 1):
            k = i - n_points + 1
            if -n_points + 1 <= k <= n_points - 1:
                valid = (np.arange(n_points) + k) % n_points
                conv[i] = np.sum(h * (1.0 - h[valid.astype(int)])) * dx

        k_idx = np.argmax(conv)

        # Formulate and solve the convex subproblem for k=k_max
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        overlap = 0.0
        k = k_idx - n_points + 1
        valid = (np.arange(n_points) + k) % n_points
        overlap = cp.sum(h_var * (1 - h_var[valid.astype(int)])) * dx
        objective = cp.Minimize(overlap)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCIPY)

        if prob.status != cp.OPTIMAL:
            h = project_h(h + 0.01 * np.random.randn(n_points))
            continue

        h_candidate = np.array(h_var.value.flat, dtype=np.float32)
        h = project_h(h_candidate)

        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
