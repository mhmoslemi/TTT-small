import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=45):
    # Ensure reasonable n_points if no initial_h_values provided
    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values must be defined in parent scope.")

    np.random.seed(seed)
    start_time = time.time()

    # Discretization
    h = np.array(initial_h_values, dtype=np.float64)
    h = h / h.sum() * n_points / 2
    dx = 2.0 / n_points

    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Precompute shifted indices
    shifts = np.arange(-n_points + 1, n_points, 1)
    idxs = np.arange(n_points)

    max_iter = 1000
    current_iter = 0

    while time.time() - start_time < budget_s and current_iter < max_iter:
        current_iter += 1

        # Compute overlaps
        overlap = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            shift = k - n_points + 1
            idx_offset = (n_points - 1 - (idxs - shift)) % n_points
            overlap[k] = np.sum(h * (1 - h[idx_offset])) * dx

        k_worst = np.argmax(overlap)

        # Formulate and solve convex relaxation for worst shift
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(overlap[k_worst])
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)

        # Use ECOS as default solver (free and efficient for convex subproblems)
        try:
            prob.solve(solver='ECOS')
        except:
            prob.solve(solver='SCIPY', ee=False)

        # Project solution
        h_new = np.clip(h_var.value, 0, 1)
        h_new /= h_new.sum()
        h_new *= n_points / 2
        h = h_new

        # Update best solution
        c5_candidate = evaluate_erdos_solution(h)
        if c5_candidate < best_c5:
            best_h = h.copy()
            best_c5 = c5_candidate

    return best_h.tolist(), best_c5, n_points
