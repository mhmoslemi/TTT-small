import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    t_budget = budget_s

    # Start optimization
    while time.monotonic() - start_time < t_budget:
        # Compute overlap integrals via cross-correlation
        c_vals = np.correlate(h, 1 - h, mode='full') * dx
        c5_val = np.max(c_vals)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val

        # Get the index of the max C_k
        k_max = int(np.argmax(np.abs(c_vals)))

        # Compute the gradient (numerical approximation): -h[x] * (1 - h[x - k_max])
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = (i - k_max) % n_points
            grad[i] = -h[i] + 2 * h[i] * h[idx]

        # Gradient descent step
        step_size = 0.1
        h_candidate = h - step_size * grad
        h_candidate = np.clip(h_candidate, 0, 1)

        # Normalize to maintain sum constraint
        h_candidate_sum = np.sum(h_candidate)
        scale = n_points / (2 * h_candidate_sum)
        h_candidate *= scale
        h_candidate = np.clip(h_candidate, 0, 1)

        # Check if this improves our best solution
        h = h_candidate

    return (best_h, best_c5, n_points)
