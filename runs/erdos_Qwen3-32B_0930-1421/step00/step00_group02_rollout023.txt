import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100, adapt=False, learning_rate=None, n_points=45):
    start_time = time.time()

    np.random.seed(seed)

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    if abs(np.sum(h) - n_points / 2) > 1e-6:
        h *= (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    print(f"Initial C5: {best_c5}")

    patience = 20
    no_improvement_counter = 0
    lr = 0.05 if learning_rate is None else learning_rate
    h_best = h.copy()

    while time.time() - start_time < budget_s and no_improvement_counter < patience:
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx

        lags = np.arange(len(overlap))
        lag_idx = np.argmax(overlap)
        best_lag = lags[lag_idx]

        # Box constraint projection and sum constraint
        grad = np.zeros_like(h)
        if best_lag < n_points:
            grad[best_lag:] = 1 - h[:-best_lag]
        else:
            grad[:n_points - best_lag] = -h[best_lag:]

        # Step descent + projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)

        c5_new = evaluate_erdos_solution(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        h = h_new.copy()

        # Dynamic learning rate decrease
        if c5_new > best_c5:
            lr *= 0.95

    print(f"Best C5: {best_c5}")
    return (best_h.tolist(), best_c5, n_points)
