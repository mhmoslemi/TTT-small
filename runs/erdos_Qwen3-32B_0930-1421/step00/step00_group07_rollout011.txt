import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    assert np.all((h >= 0) & (h <= 1)), "Initial h violates box constraints"
    assert np.isclose(h.sum(), target_sum), "Initial h violates integral constraint"
    
    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.02
    max_iter = 1000
    patience = 200
    no_improve = 0
    start_time = time.time()
    
    def compute_c5(h):
        corr = scipy.signal.fftconvolve(h, 1 - h[::-1], mode='full')
        overlaps = corr * dx
        return np.max(overlaps)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute overlap values
        corr = scipy.signal.fftconvolve(h, 1 - h[::-1], mode='full')
        overlaps = corr * dx
        c5 = np.max(overlaps)
        k = np.argmax(overlaps)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            i1 = j - k + n_points  # Account for wrap-around at left boundary
            i2 = j + k
            if 0 <= i1 < n_points:
                grad[j] -= h[i1]
            if 0 <= i2 < n_points:
                grad[j] += (1 - h[i2])
        
        # Gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain integral constraint
        h_new *= target_sum / np.clip(h_new.sum(), 1e-6, 2 * target_sum)
        h = h_new
        
        # Check for improvement
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
            alpha = min(0.04, alpha * 1.05)  # Increase learning rate with improvement
        else:
            no_improve += 1
            if no_improve >= patience:
                alpha = max(5e-4, alpha * 0.5)  # Halve learning rate
        
    return (best_h, best_c5, n_points)
