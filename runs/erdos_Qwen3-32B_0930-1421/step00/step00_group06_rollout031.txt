import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    # Use the initial_h_values provided in the environment
    h = np.copy(initial_h_values)
    n_points = h.size
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = float('inf')

    alpha = 0.5
    max_iterations = 1000
    improvement_threshold = 1e-4
    check_interval = 50
    h_prev = np.copy(h)

    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Budget time exceeded, stopping iteration.")
            break

        # Compute h_shifted as convex combination
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Compute overlap integrals at all k
        h1_h = h_shifted
        h2_1m = 1 - h
        corr = np.correlate(h1_h, h2_1m, mode='full') * dx
        # The positions k = 1 to n_points-1 are valid shifts on periodic interval
        valid_corr = corr[n_points:2*n_points]

        # Find the maximum overlap position (shift k)
        k_max = np.argmax(valid_corr)  # 0-based in [0, n_points-1]
        max_overlap = valid_corr[k_max]

        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = np.copy(h)
            print(f"Iteration {iter}: Improved C₅ = {best_c5:.6f}") if iter % check_interval == 0 else None

        # Formulate convex problem for current k_max
        h_var = cp.Parameter(shape=n_points, value=h, nonneg=True)
        h_opt = cp.Variable(n_points)

        # Ensure feasible h_opt
        constraints = [
            0 <= h_opt,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2,
        ]

        # Objective function for minimizing the overlap at k_max
        def get_overlap(h_opt, k):
            shifted = np.roll(h_opt, -k)
            overlap = cp.sum(h_opt * (1 - shifted))
            return overlap

        objective = cp.Minimize(get_overlap(h_opt, k_max))

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CBC, verbose=False)

        if problem.status != 'optimal':
            h_prev = np.copy(h)
            print("Warning: Non-optimal solution, skipping.")
            continue

        h = h_opt.value
        h_prev = alpha * h + (1 - alpha) * h_prev

        if iter % check_interval == 0:
            # Recompute full C₅ bound
            h1_h = h
            h2_1m = 1 - h
            corr = np.correlate(h1_h, h2_1m, mode='full') * dx
            overlaps = corr[n_points:2*n_points]
            current_c5 = np.max(overlaps)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.copy(h)
                print(f"Current C₅ after epoch {iter} = {best_c5:.6f}")

    # Final bound
    h1_h = best_h
    h2_1m = 1 - best_h
    corr = np.correlate(h1_h, h2_1m, mode='full') * dx
    overlaps = corr[n_points:2*n_points]
    final_c5 = np.max(overlaps)

    # Validate constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Constraint violation"
    assert np.isclose(np.sum(best_h), n_points / 2), "Measure constraint violated"

    return (best_h.tolist(), final_c5, n_points)
