import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp
try:
    from optimize_erdos import evaluate_erdos_solution
    from optimize_erdos import initial_h_values
    has_initial_h = True
except ImportError:
    has_initial_h = False

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Parameters
    n_points = len(initial_h_values) if has_initial_h else 45
    pop_size = 50
    max_gen = 100
    mutation_std = 0.1
    crossover_rate = 0.5
    starting_h = initial_h_values.copy() if has_initial_h else None

    start_time = time.time()

    def compute_overlap(h, dx):
        h_full = np.zeros(2 * n_points)
        h_full[:n_points] = h
        ones_minus_h = 1 - h_full
        corr = np.correlate(h_full, ones_minus_h, mode='full')
        corr[:n_points] = -np.inf  # Discard negative lags
        return np.max(corr[n_points:]) * dx

    def eval_fitness(h):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        dx = 2.0 / n_points
        return compute_overlap(h, dx)

    if starting_h is None:
        starting_h = np.zeros(n_points)
        half_indices = np.random.choice(n_points, int(n_points / 2), replace=False)
        starting_h[half_indices] = 1.0
        starting_h = starting_h.copy()

    population = []
    for _ in range(pop_size):
        perturbed = starting_h + mutation_std * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed /= perturbed.sum() * (2.0 / n_points)
        population.append(perturbed)

    best_h = starting_h.copy()
    best_c5 = eval_fitness(best_h)

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break

        fitness = [eval_fitness(ind) for ind in population]
        fitness_idx = np.argsort(fitness)

        # Track best solution from population
        current_best_idx = fitness_idx[0]
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5

        # Select parents as top 20%
        parent_idx = fitness_idx[:int(pop_size * 0.2)]
        parents = [population[i] for i in parent_idx]

        # Crossover and mutation to create new candidates
        new_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            crossover = np.random.rand()
            if crossover < crossover_rate:
                child = (parent1 + parent2) / 2  # Arithmetic blending
            else:
                child = parent1.copy()

            # Add mutation
            noisy_child = child + mutation_std * np.random.randn(n_points)
            noisy_child = np.clip(noisy_child, 0, 1)
            noisy_child /= noisy_child.sum() * (2.0 / n_points)

            new_population.append(noisy_child)

        population = parents + new_population

    # Post-processing with local gradient descent
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return compute_overlap(h, 2.0 / n_points)

    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, x0=best_h, bounds=bounds, method='L-BFGS-B')
    final_h = result.x.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / final_h.sum() * (n_points / 2)
    final_c5 = eval_fitness(final_h)

    # Final evaluation
    dx = 2.0 / n_points
    h_full = np.zeros(2 * n_points)
    h_full[:n_points] = best_h
    ones_minus_h = 1 - h_full
    corr = np.correlate(h_full, ones_minus_h, mode='full')
    corr[:n_points] = -np.inf
    max_corr = np.max(corr[n_points:])
    final_c5_bound = max_corr * dx

    return final_h, final_c5_bound, n_points
