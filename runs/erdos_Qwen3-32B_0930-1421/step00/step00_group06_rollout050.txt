import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = h0.shape[0]
    dx = 2.0 / n_points

    h_val = h0.copy()
    best_h = h_val.copy()
    best_c5 = float('inf')

    # Objective function
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    # Constraints
    def normalize_h(h):
        h_sum = np.sum(h)
        h_norm = h * (n_points / 2.0 / h_sum)
        return np.clip(h_norm, 0, 1)

    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    alpha_decay = 0.95
    tolerance = 1e-6
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute current C₅ and best lag k
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        k = np.argmax(corr)

        # Compute gradient
        grad = (1 - np.roll(h_val, k)) - np.roll(1 - h_val, -k)
        grad = np.roll(grad, -k)

        # Update step
        h_proposal = h_val - learning_rate * grad
        h_proposal = normalize_h(h_proposal)

        # Update best solution
        c5_current = compute_c5(h_val)
        c5_proposal = compute_c5(h_proposal)

        if c5_proposal < best_c5:
            best_c5 = c5_proposal
            best_h = h_proposal.copy()

        h_val = h_proposal.copy()
        learning_rate *= alpha_decay

        # Check convergence
        if iter > 0 and abs(c5_current - best_c5) < tolerance:
            break

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, n_points
