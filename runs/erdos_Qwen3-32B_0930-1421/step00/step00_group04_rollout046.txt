import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize, Bounds
import cvxpy as cp

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h = h * dx
    c = correlate(h, 1 - h, mode="full")
    c5 = np.max(c) * n_points**2
    return c5

def constrain_h(h, n_points):
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / np.sum(h)
    return h

def objective(x, n_points):
    h = x.reshape(-1)
    return evaluate_c5(h, n_points)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 42 * 2  # start with 84 samples, balances speed and accuracy

    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy() if n_points == initial_h_values.size else np.tile(initial_h_values, n_points // initial_h_values.size + 1)[:n_points]
    else:
        h0 = np.zeros(n_points)
        mid = n_points // 2
        h0[max(0, mid - 10):min(n_points, mid + 10)] = 1
        h0 = constrain_h(h0, n_points)

    def optimized_f(x):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            raise RuntimeError("Budget exceeded")
        h = x.reshape(-1)
        return evaluate_c5(h, n_points)

    h_flat = h0.flatten()

    # First pass with a simple optimization
    res = minimize(optimized_f, h_flat, bounds=Bounds(0, 1))
    h_opt = constrain_h(res.x, n_points)

    # Second pass with a restarted gradient descent with adaptive rate
    max_iterations = 300
    best_c5 = evaluate_c5(h_opt, n_points)
    best_h = h_opt
    step_size = 0.01
    restart_every = 50
    i = 0

    while time.time() - start_time < budget_s:
        # Compute the gradient numerically
        eps = 1e-5
        grad = np.zeros(n_points)
        for idx in range(n_points):
            h_up = h_opt.copy()
            h_up[idx] += eps
            h_up = constrain_h(h_up, n_points)
            loss_up = evaluate_c5(h_up, n_points)

            h_down = h_opt.copy()
            h_down[idx] -= eps
            h_down = constrain_h(h_down, n_points)
            loss_down = evaluate_c5(h_down, n_points)

            grad[idx] = (loss_up - loss_down) / (2 * eps)

        h_opt -= step_size * grad
        h_opt = constrain_h(h_opt, n_points)

        c5_current = evaluate_c5(h_opt, n_points)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_opt

        if i % restart_every == 0:
            # Generate a simplex using random perturbations
            simplex = [best_h]
            for _ in range(3):
                hpert = best_h.copy()
                perturbation = 0.1 * (2 * np.random.rand(n_points) - 1)
                hpert += perturbation
                hpert = constrain_h(hpert, n_points)
                simplex.append(hpert)

            # Evaluate the simplex and select the best
            simplex_c5 = [evaluate_c5(h, n_points) for h in simplex]
            best_idx = np.argmin(simplex_c5)
            best_h = simplex[best_idx]
            h_opt = best_h

        i += 1

    elapsed_time = time.time() - start_time
    print(f"Time used: {elapsed_time:.2f}, final C5: {best_c5:.6f}, n_points: {n_points}")
    return (best_h, best_c5, n_points)
