import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    n_pop = 50
    n_gens = 1000
    mut_rate = 0.02
    elitism = 0.2
    n_elite = max(5, int(n_pop * elitism))
    n_pairs = n_pop - n_elite
    best_h = None
    best_c5 = float('inf')

    population = [h0.copy() for _ in range(int(n_pop / 2))]  # Start with one h0
    # Add random variations and perturbed h0
    for _ in range(n_pop - len(population)):
        h_copy = h0.copy()
        h_copy += np.random.normal(0, 0.05, size=n_points)
        h_copy = np.clip(h_copy, 0, 1)
        h_copy *= target_sum / np.sum(h_copy)
        population.append(h_copy)

    def compute_c5(h):
        ones = 1 - h
        corr = np.correlate(h, ones, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(n_gens):
        if time.time() - start_time > budget_s:
            break

        # Evaluate candidates
        fitness = []
        for i, h in enumerate(population):
            # Repair constraints
            h_mask = np.isfinite(h)
            h = h * h_mask
            h = np.clip(h, 0, 1)
            cur_sum = np.sum(h)
            if abs(cur_sum - target_sum) > 1e-6:
                h *= (target_sum / cur_sum)
            c = compute_c5(h)
            fitness.append((c, h))

        # Sort: lower C₅ is better
        fitness.sort()
        current_c5, current_h = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # Select top candidates
        top = [h for _, h in fitness[:n_elite]]
        next_gen = top

        # Mate top pairs
        selected_idx = np.random.choice(range(n_elite), size=(n_pairs, 2), replace=False)
        for i, j in selected_idx:
            h1, h2 = top[i], top[j]
            s = np.random.randint(0, n_points)
            child = np.concatenate([h1[:s], h2[s:]]).copy()
            # Add mutation
            m = np.random.binomial(1, mut_rate, size=n_points).astype(bool)
            child[m] = np.random.uniform(0, 1, size=np.sum(m))
            # Repair constraints
            child = np.clip(child, 0, 1)
            child *= (target_sum / np.sum(child))
            next_gen.append(child)
        population = next_gen

    # Re-evaluate best solution
    best_h = np.clip(best_h, 0, 1)
    best_h *= (target_sum / np.sum(best_h))
    return (best_h.tolist(), float(best_c5), n_points)
