import numpy as np
import time
import cvxpy as cp
from collections import deque

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n_points = 400
        h0 = np.zeros(n_points)
        h0[n_points // 2 - 50 : n_points // 2 + 50] = 0.5

    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    h = h0.copy()

    window = 11
    history = deque(maxlen=20)

    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        area = np.sum(h_clipped) * dx
        if np.isclose(area, 1.0):
            return h_clipped
        scaling_factor = n_points / 2 / area
        h_scaled = h_clipped * scaling_factor
        return project_h(h_scaled)  # clip again in case of edge scaling issues

    while time.time() - start_time < budget_s - 5:
        # Adjust window if convergence is not adequate
        if len(history) == history.maxlen and all(np.abs(x - history[0]) <= 1e-6 for x in history):
            window = max(3, window // 2)
        elif len(history) == history.maxlen:
            pass  # maintain window

        # Construct problem with focus on non-smooth target max
        H = cp.Variable(n_points)
        constraints = [
            0 <= H,
            H <= 1,
            cp.sum(H) == n_points / 2,
            H >= h * 0.9,
            H <= h * 1.1
        ]

        overlap_func = []
        bases = [1]
        for m in range(-window, window + 1):
            if m == 0:
                continue
            shifted = cp.roll(h, -m * 1)
            overlap = H * (1 - shifted[:n_points])
            integral = cp.sum(overlap)
            bases.append(1)
            overlap_func.append(integral)

        objective = cp.Minimize(cp.Max(overlap_func))
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.MOSEK, verbose=False)
            if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                continue
            h = project_h(H.value)
        except Exception:
            continue

        h = project_h(h)
        c5 = evaluate_erdos_solution(h)

        history.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        if len(history) > 0 and history[-1] < best_c5 + 1e-4:
            window += 2

    return best_h, best_c5, n_points
