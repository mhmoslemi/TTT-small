import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate


def optimize_h(n_points, initial_h=None, random_seed=42, max_time=1100):
    np.random.seed(random_seed)
    dx = 2.0 / n_points

    if initial_h is not None:
        h = initial_h.copy()
    else:
        h = np.random.rand(n_points)
        h /= h.sum() * dx
        h *= 1.0

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    def compute_c5(h):
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full')
        corr *= dx
        return np.max(corr)

    while time.time() - start_time < max_time:
        idxs = np.random.choice(n_points, size=2, replace=False)
        old_val = h[idxs].copy()

        # Try perturbation
        noise = np.random.normal(0, 0.1, size=2)
        new_vals = np.clip(h[idxs] + noise, 0, 1)
        h[idxs] = new_vals

        # Enforce the average constraint: sum == n_points / 2
        sum_h = h.sum()
        delta = np.sum(new_vals - old_val)
        if sum_h + delta > n_points / 2 or sum_h + delta < n_points / 2:
            correction = (n_points / 2) - sum_h
            idx = idxs[0] if abs(correction) < abs(h[idxs[0]] - new_vals[0]) else idxs[1]
            h[idx] = max(0, min(1, h[idx] + correction))

        current_c5 = compute_c5(h)
        if time.time() - start_time > max_time:
            break
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            print(f"Improved C5: {best_c5:.6f}")

        if np.random.rand() > 0.8:  # sometimes reset back to better starting point
            h = best_h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, n_points=350):
    t0 = time.time()

    try_initial = True
    if try_initial and 'initial_h_values' in globals():
        initial = initial_h_values
        n_points_initial = len(initial)
        if n_points_initial == n_points:
            h_init = initial
        else:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial)
            h_init /= h_init.sum() * (2.0 / n_points)
            h_init *= 1.0
    else:
        h_init = None

    optimized_h, best_c5 = optimize_h(n_points, initial_h=h_init, random_seed=seed, max_time=budget_s)

    optimized_h = np.clip(optimized_h, 0, 1)
    optimized_h *= (1 / (2 * np.sum(optimized_h))) * n_points

    final_c5 = np.max(correlate(optimized_h, 1 - optimized_h, mode='full') * (2.0 / n_points))

    return (optimized_h.tolist(), final_c5, n_points)
