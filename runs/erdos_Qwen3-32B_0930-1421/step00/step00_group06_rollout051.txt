import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure constraint: sum(h) == n_points / 2
    h = h.astype(np.float64)
    h_sum = h.sum()
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h / h_sum * (n_points / 2)

    start_time = time.time()
    max_iter = 1000
    learning_rate = 0.01
    lr_decay = 0.95
    best_h = h.copy()
    best_c5 = np.inf
    prev_c5 = np.inf

    for i in range(max_iter):
        corrs = np.correlate(h, 1 - h, mode="full")
        corrs = corrs[n_points - 1:2 * n_points - 1]  # Lags from 0 to n_points - 1
        c5 = corrs.max() * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Early stopping
        if abs(prev_c5 - c5) < 1e-6:
            break
        prev_c5 = c5

        k = np.argmax(corrs)

        # Compute gradient manually
        grad = np.zeros_like(h)
        for i in range(n_points):
            wrap_i = (i - k) % n_points
            grad[i] = (1 - h[(i + k) % n_points]) - h[wrap_i]

        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * (n_points / 2)

        learning_rate *= lr_decay

        # Enforce time budget
        if time.time() - start_time > budget_s - 5:
            break

    return (best_h, best_c5, n_points)
