def evaluate_erdos_solution(h):
    h = np.clip(h, 0.0, 1.0)
    n_points = len(h)
    dx = 2.0 / n_points
    ones = np.ones_like(h)
    c5_values = correlate(h, ones - h, mode="full") * dx
    return np.max(c5_values)

def project_to_cone(h, goal_sum):
    """Project h onto the cone [0,1]^n ∩ {x : sum(x) = goal_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - goal_sum) < 1e-12:
        return h
    elif current_sum > goal_sum:
        # Decrease largest values
        h_sorted = np.sort(h)
        threshold_idx = np.searchsorted(np.cumsum(h_sorted), goal_sum, side="right")
        h_clipped = np.zeros_like(h_sorted)
        h_clipped[threshold_idx:] = h_sorted[threshold_idx:]
        # Scale clipped values
        if threshold_idx < len(h_sorted):
            scale = (goal_sum - np.sum(h_clipped)) / (len(h_sorted) - threshold_idx)
            h_clipped[threshold_idx:] *= scale
        return np.sort(h_clipped)[::-1]  # restore original order
    else:
        # Increase smallest values
        h_sorted = np.sort(h)
        threshold_idx = np.searchsorted(np.cumsum(h_sorted), goal_sum, side="right")
        h_clipped = np.zeros_like(h_sorted)
        h_clipped[:threshold_idx] = h_sorted[:threshold_idx]
        # Scale clipped values
        if threshold_idx > 0:
            scale = (goal_sum - np.sum(h_clipped)) / threshold_idx
            h_clipped[:threshold_idx] += scale
        return np.sort(h_clipped)[::-1]  # restore original order
