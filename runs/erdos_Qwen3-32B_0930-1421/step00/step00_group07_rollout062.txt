import numpy as np
import math
import time

def evaluate_overlap(h, dx):
    # Compute (1 - h) once
    one_minus_h = 1 - h
    # Compute full cross-correlation using np.correlate
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    np.random.seed(seed)

    # Binary conversion and normalization of the initial solution
    h = (initial_h_values > 0.5).astype(np.float64)

    # Make sure the sum of h is exactly n / 2
    sum_h = np.sum(h)
    diff = int(n / 2) - sum_h
    if diff != 0:
        if diff > 0:
            indices = np.where(h == 0)[0]
            np.random.shuffle(indices)
            for i in indices[:int(diff)]:
                h[i] = 1
        else:
            indices = np.where(h == 1)[0]
            np.random.shuffle(indices)
            for i in indices[:int(-diff)]:
                h[i] = 0

    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated Annealing parameters
    T = 1.0
    t_decay = 0.99
    min_T = 1e-5
    patience = 1000
    no_improvement = 0

    while time.time() - start_time < budget_s:
        # Generate a new candidate by flipping two bits
        idx0 = np.random.choice(np.where(current_h == 1)[0])
        idx1 = np.random.choice(np.where(current_h == 0)[0])
        new_h = current_h.copy()
        new_h[idx0] = 0
        new_h[idx1] = 1
        new_c5 = evaluate_overlap(new_h, dx)

        if new_c5 < best_c5:
            best_h[:] = new_h
            best_c5 = new_c5
            no_improvement = 0
        elif np.log(np.random.rand()) < -(new_c5 - current_c5) / T:
            # Accept worse solution with probability exp(-(C_new - C_current)/T)
            current_h[:] = new_h
            current_c5 = new_c5
            no_improvement += 1
        else:
            no_improvement += 1

        if T > min_T:
            T *= t_decay
        else:
            T = max(min_T, T * t_decay)

        if no_improvement > patience:
            T = max(min_T, T * 0.9)
            no_improvement = 0

    return best_h, best_c5, n
