import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = budget_s

    # Copy initial_h_values if provided, else use uniform initialization
    if "initial_h_values" in globals():
        h = np.copy(initial_h_values)
        n_points = len(h)
    else:
        # Default initialization: constant 0.5
        n_points = 80
        h = np.full(n_points, 0.5)
    
    # Normalize to ensure exact integral = 1
    dx = 2.0 / n_points
    h *= n_points / (2 * np.sum(h))

    best_h = np.copy(h)
    best_c5 = float('inf')

    alpha = 0.02  # initial learning rate
    patience_counter = 0
    max_patience = 200
    max_iter = 1000

    for iter_idx in range(max_iter):
        if time.time() - start_time > max_time:
            break

        corr = correlate(h, 1 - h, mode="full")
        overlap_values = corr * dx
        c5 = np.max(overlap_values)
        k = np.argmax(overlap_values)

        # If improvement
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
            patience_counter = 0
        else:
            patience_counter += 1

        if patience_counter >= max_patience:
            alpha *= 0.5
            patience_counter = 0
            if alpha < 1e-6:
                break

        # Compute gradient on worst-case shift k
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + k
            i_minus = i - k
            if i_plus < n_points:
                grad[i] += (1 - h[i_plus])
            if i_minus >= 0:
                grad[i] -= h[i_minus]

        # Take gradient step
        h_new = h - alpha * grad

        # Project back into feasible region [0, 1]
        h_new = np.clip(h_new, 0, 1)

        # Re-normalize to maintain ∑h = n / 2
        if np.any(h_new):  # avoid divide by zero
            h_new *= (n_points / 2) / np.sum(h_new)
        else:
            h_new[:] = 0.5 * n_points / 2  # fallback to uniform

        h[:] = h_new

    # Evaluate to return best solution and clean up
    c5_bound, best_h = evaluate_erdos_solution(best_h)

    return (best_h, c5_bound, len(best_h))
