    start_time = time.time()

    h[0] += correction

    # Helper: compute C5
    def compute_c5(h):
        dh = 1 - h
        corr = correlate(h, dh, mode="full") * dx
        return np.max(np.abs(corr[n_points-1:2*n_points-1]))

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Optimization Loop
    while time.time() - start_time < budget_s - 10:  # Reserve 10s for final steps
        h = best_h.copy()

        # Phase 1: Frequency-domain approximate optimization
        H = np.fft.fft(h)
        k_shift = np.fft.fftfreq(n_points) * (2 * np.pi / dx)
        w = -(0.5 * (1 + np.cos(k_shift / 2)))  # Weight function to suppress high-frequency errors
        w[0] = 0  # Don't penalize DC component

        # Surrogate frequency-bound objective
        h_var = cp.Variable(n_points)
        x_var = h_var * (1 - h_var)

        objective = cp.Minimize(cp.max(cp.abs(cp.fft.fft(x_var)) * w))
        constraints = [
            0 <= h_var, h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        # Solve
        prob = cp.Problem(objective, constraints)
        ECOS_BACKEND_ID = min([b for b in cp.installed_solvers() if 'ECOS' in b and cp.installed_solvers().index(b)])
        prob.solve(solver=cp.ECOS)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            candidate_h = np.array(h_var.value).flatten()
            candidate_h = np.clip(candidate_h, 0, 1)

            # Project to feasible set (satisfy sum constraint)
            avg = np.sum(candidate_h) / n_points
            adjustment = (n_points / 2 - np.sum(candidate_h)) / n_points
            candidate_h += adjustment
            candidate_h = np.clip(candidate_h, 0, 1)

            c5 = compute_c5(candidate_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate_h
                print(f"New best C5: {best_c5}")

        # Phase 2: Time-domain stochastic gradient descent
        alpha = 0.01
        for _ in range(50):
            dh = 1 - best_h
            corr = correlate(best_h, dh, mode="full") * dx
            max_k = np.argmax(np.abs(corr[n_points-1:]))
            k = max_k
            shift_dh = np.roll(dh, k) * dx

            grad = -shift_dh[n_points-1:]  # Sign of negative gradient
            grad += 0.0001  # Small regularization
            step = alpha * grad
            h_trial = best_h + step
            h_trial = np.clip(h_trial, 0, 1)

            # Enforce sum constraint
            h_trial = h_trial / np.sum(h_trial) * n_points / 2

            c5 = compute_c5(h_trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_trial
                print(f"Improved C5: {best_c5}")

        # Stop if time is nearly up
        if time.time() - start_time > budget_s - 10:
            break

    return best_h.tolist(), best_c5, n_points
