def evaluate_overlap(h, dx):
    """Compute max overlap C₅ via cross-correlation"""
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode='full') * dx
    return np.max(corr)


def compute_gradient(h, dx):
    """Compute gradient via convolution in real and frequency domain"""
    hcorr = np.correlate(h, 1 - h, mode="full")
    valid_corr = hcorr[len(h) - 1 : 2 * len(h) - 1]
    total_corr = hcorr[:3 * len(h) - 1]
    max_index = np.argmax(total_corr)
    sign = np.sign(hcorr[max_index])
    return sign * (h >> (max_index - len(h) + 1) - h << (len(h) - 1 - max_index)), dx
