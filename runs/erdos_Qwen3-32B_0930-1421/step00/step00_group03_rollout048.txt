import time
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import math

def evaluate_erdos_solution(h):
    h = np.ascontiguousarray(h)
    dh = h - h**2
    corr = correlate(h, dh, mode='full')
    n_points = h.shape[0]
    dx = 2.0 / n_points
    result = np.max(corr[n_points:n_points*2]) * dx
    return result


def project(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    scaling = (n_points / 2.0) / current_sum
    h *= scaling
    return h

    np.random.seed(seed)

    while time.time() - start < budget_s:
        h = project(h, n_points)
        overlap = h * (1 - h)
        corr = correlate(h, overlap, mode='full')
        idx = np.arange(n_points, n_points * 2)

        grad = correlate(overlap, h, mode='full') - correlate(overlap * 2, h, mode='full')

    return best_h.tolist(), best_c5, n_points
