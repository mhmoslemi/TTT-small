```python
import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
import time


def basis_transform(x_coords, centers, widths):
    return np.array([np.clip(1 - np.abs(x_coords - c) / w, 0, 1) for c, w in zip(centers, widths)]).T


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    initial_h_values = np.array(evaluate_erdos_solution.initial_h_values())

    # Use 70-sample version of the function as discretization
    n_points = 70
    x = np.linspace(0, 2, n_points, endpoint=False)
    dx = 2.0 / n_points

    h = initial_h_values.copy()

    # Define MICP formulation with basis function approach
    M = 50  # Number of triangular basis functions
    centers = np.linspace(dx, 2 - dx, M)
    widths = np.full(M, 0.1)  # Fixed width for triangular basis functions

    B = basis_transform(x, centers, widths)

    w = cp.Variable(B.shape[1], nonneg=True)
    constraints = [
        cp.sum(w) == 1,
        cp.sum(B @ w) * dx == 1,  # Integrate to 1
        B @ w <= 1,
        B @ w >= 0
    ]

    h_cvxpy = B @ w
    overlaps = [h_cvxpy @ (1 - h_cvxpy[k:]) * dx for k in range(n_points)]
    t = cp.Variable()
    for k in range(n_points):
        constraints += [t >= overlaps[k]]

    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    last_save = 0

    while time.time() - start_time < budget_s * 0.9:  # Leave 10% for final evaluation
        try:
            problem.solve(solver=cp.GUROBI, TimeLimit=budget_s * 0.3, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = (B @ w.value).clip(0, 1)
                h_new = h_new / h_new.sum() * n_points / 2
                h_new = h_new.clip(0, 1)
                c5 = evaluate_erdos_solution(h_new)
                
                if c5 < best_c5:
                    if abs(sum(h_new) * dx - 1) < 1e-6 and np.all((h_new >= 0) * (h_new <= 1)):
                        best_h = h_new.copy()
                        best_c5 = c5
                        last_save = time.time()
        except Exception:
            # Fallback to standard subgradient descent if MICP fails
            break

    # Final verification
    h = best_h.copy()
    if time.time() - start_time < budget_s:
        # Refine with gradient-based methods
        from scipy.optimize import minimize
        def objective_func(coeffs):
            h = (B @ coeffs).clip(0, 1)
            h = h / h.sum() * n_points / 2
            c5 = evaluate_erdos_solution(h)
            return c5
        
        result = minimize(
            objective_func,
            x0=np.random.rand(B.shape[1]) * 0.1 + 0.1 / n_points,
            method='L-BFGS-B',
            bounds=[(0, 1)] * B.shape[1],
            options={'maxiter': 100}
        )
        if result.success:
            h_new = (B @ result.x).clip(0, 1)
            h_new = h_new / h_new.sum() * n_points / 2
            c5 = evaluate_erdos_solution(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    return best_h, best_c5, n_points
```
