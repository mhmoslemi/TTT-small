import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, initial_h_values=None):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70  # Start with minimal initial construction support
    dx = 2.0 / n_points

    # Construct basis functions (triangular) over the domain
    def construct_basis_functions(n_basis, x_values):
        basis = np.zeros((n_points, n_basis))
        centers = np.linspace(0, 2, n_basis)
        widths = 2 / (2 * n_basis)
        for j in range(n_basis):
            center = centers[j]
            idxs = np.where((x_values >= center - widths) & (x_values <= center + widths))[0]
            for i in idxs:
                x = x_values[i]
                basis[i, j] = max(0, 1.0 - abs(x - center) / widths)
        # Normalize basis functions to uniform sup-norm 1
        return basis / np.max(basis, axis=0)

    x_values = np.linspace(0, 2 - dx, n_points)

    # Initial basis: a reasonable number of functions to balance expressivity and tractability
    n_basis = 30
    basis = construct_basis_functions(n_basis, x_values)

    # Variables
    w = cp.Variable(n_basis, nonneg=True)
    h = basis @ w  # Construct h(x)

    # Soft constraints to ensure h ∈ [0, 1]
    constraint_h_le1 = cp.sum(cp.pos(basis @ w - 1)) <= 1e-5
    constraint_h_ge0 = cp.sum(cp.neg(basis @ w)) <= 1e-5

    # Integral constraint
    constraint_int = cp.sum(w @ basis[:, ::-1].flatten()) * dx == 1.0  # np.sum(h) * dx = 1

    # One-step correlation computation via convolutions mapped to basis combinations
    # Compute discrete shifted (1 - h) convolved with h
    ones = np.ones_like(h)
    shifts = [2 ** k for k in range(6)]  # check k = 1,...,5
    max_corr = np.zeros_like(h)

    expressions = []
    for k in shifts:
        if k >= n_points:
            continue
        expr = ones[k:] * h[:-k] - h[k:] * h[:-k]
        expressions.append(cp.sum(expr) * dx)

    obj = cp.Minimize(cp.Max(expressions))

    constraints = [cp.sum(w) == 1.0, constraint_int, constraint_h_le1, constraint_h_ge0]

    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCIP, verbose=False, time_limit=60)

    h_opt = basis @ w.value
    h_opt = np.clip(h_opt, 0, 1)  # Postprocessing safety
    h_opt = h_opt / (np.sum(h_opt) * dx) * 1.0  # Re-normalize in case of numeric drift

    h_values = h_opt
    if initial_h_values is not None and np.allclose(h_opt, initial_h_values):
        pass  # Same as known solution

    def compute_c5(h):
        h = np.array(h)
        ones = np.ones_like(h)
        corrs = []
        for k in range(1, len(h) + 1):
            if k >= len(h):
                continue
            corr = np.sum(h * (ones[k:] - h[k:])) * dx
            corrs.append(corr)
        return max(corrs)

    c5_bound = compute_c5(h_values)
    return h_values.tolist(), c5_bound, n_points
