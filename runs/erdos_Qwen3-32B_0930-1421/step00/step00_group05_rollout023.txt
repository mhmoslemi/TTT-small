import numpy as np
import cvxpy as cp
from time import perf_counter
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    start_time = perf_counter()

    # Use the initial_h_values from the parent construction
    n = len(initial_h_values)
    dx = 2.0 / n
    dx_reciprocal = 1 / dx  # Reciprocal for scaling convolution quickly

    # Initial copy from the parent construction
    h = initial_h_values.copy()
    h = h.reshape(-1)
    best_h = h.copy()
    best_c5 = np.inf

    # Precompute 1 - h for evaluation speed
    one_minus_h = 1.0 - h
    window = n  # full convolution range is 2n - 1: from -n+1 to +n-1
    overlap_integrals = (np.correlate(h, one_minus_h, mode="full") * dx)

    # Find which k gave the maximum overlap
    best_c5 = overlap_integrals[n:-n].max()
    best_h = h.copy()

    for time_used in range(1, 2):  # allows up to 1 iteration if time allows
        # Check time budget
        elapsed = perf_counter() - start_time
        if elapsed >= budget_s:
            # Return best solution so far
            best_result = evaluate_erdos_solution(best_h)
            return (best_h.tolist(), best_result, n)

        # Reformulate the optimization: CVXVAR
        h_var = cpVariable(n, nonneg=True)
        max_overlap = cp.Parameter(noneg=True)

        constraints = [
            cp.sum(h_var) == n / 2,
            h_var <= 1,
            h_var >= 0,
        ]

        # Simulate max overlap manually with binary search feasibility
        def optimize_for_overlap_bound(target_c5):
            overlap_integrals_var = cp.conv(self=h_var, other=1 - h_var, mode="full")
            shifted_integrals = overlap_integrals_var[n:2*n]
            obj = cp.Maximize(cp.Constant(0))  # Feasibility check
            prob = cp.Problem(obj, constraints + [shifted_integrals <= target_c5])
            return prob.solve(solver="ECOS", verbose=False)

        # Binary search on max overlap: fast feasibility check
        low_bound = 0.0
        high_bound = 0.4  # can increase if needed
        while high_bound - low_bound > 1e-5:
            mid = (low_bound + high_bound) / 2
            result = optimize_for_overlap_bound(mid)
            if result is not None and result <= mid + 1e-6:
                high_bound = mid
            else:
                low_bound = mid

        best_c5 = high_bound
        best_h = h_var.value.copy()

        # Final evaluation
        best_result = evaluate_erdos_solution(best_h)

        elapsed = perf_counter() - start_time
        if elapsed >= budget_s:
            return (best_h.tolist(), best_result, n)

    return (best_h.tolist(), best_result, n)
