import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    h = h_values
    n_points = len(h)
    dx = 2.0 / n_points
    h_array = h.reshape(-1)
    corr = correlate(h_array, 1.0 - h_array, mode="full")
    c5_bound = (np.max(corr) * dx)
    return c5_bound

        # Constraint penalties
        sum_diff = abs(np.sum(h) - n_points / 2)
        clamping = max(0, np.max(h) - 1) + max(0, 0 - np.min(h))
        penalty = 1000.0 * (sum_diff + clamping)
        return val + penalty  # Higher val + penalty are worse

    def mutate(h, sigma=0.1):
        h_new = h.copy()
        perturbation = np.random.randn(*h_new.shape) * sigma
        h_new += perturbation
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # Project back to feasible space
        h_total = np.sum(h_new)
        factor = (n_points / 2) / h_total
        return h_new * factor

    def crossover(p1, p2):
        alpha = np.random.rand(*p1.shape)
        child = alpha * p1 + (1 - alpha) * p2
        return child

    def project_simplex(h):
        h = h.clip(0, 1)
        h_total = h.sum()
        return h * (n_points / 2) / h_total

    # Simulation parameters
    pop_size = 50
    max_gen = 200
    mutation_sigma = 0.1
    best_h = project_simplex(initial_h)
    best_val = evaluate(best_h)

    population = [project_simplex(initial_h + np.random.randn(n_points) * 1e-3) for _ in range(pop_size)]

    while time.time() - t0 < budget_s and max_gen > 0:

        evaluations = [evaluate(ind) for ind in population]

        sorted_indices = np.argsort(evaluations)
        parents = [population[i] for i in sorted_indices[:5]]

        offspring = []
        for _ in range(pop_size):
            # Crossover
            p1, p2 = np.random.choice(parents, 2)
            child = crossover(p1, p2)
            # Mutation
            child = mutate(child, mutation_sigma)
            offspring.append(project_simplex(child))
