import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Constants
    dx = 2.0 / 45  # Initial resolution
    n_points = 45
    target_improvement = 1e-5  # Target improvement from best known constant

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Simple initialization: uniform grid with fixed support
        h = np.zeros(n_points)
        middle_idx = n_points // 2
        h[middle_idx - 2:middle_idx + 2] = 0.25
        h = h * 2.0 / np.sum(h)  # Normalize to satisfy the sum constraint

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iter = 500

    def evaluate_c5(h, dx):
        return evaluate_erdos_solution.evaluate_erdos_solution(h, dx)

    def project_to_constraints(h):
        h = np.clip(h, 0, 1)
        scale = 1.0
        if np.sum(h) != 0:
            scale = n_points / 2 / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        return h

    def optimize_subproblem(h, k, dx):
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(h_var[k] * dx)

        constraints = [
            h_var >= 0,
            h_var <= 1,
        ]
        for i in range(n_points):
            # Use indices correctly for fixed k
            if k <= i < n_points:
                constraints.append(h_var[i] <= h[i - k])
            if k > i:
                constraints.append(h_var[i] <= 1 - h[i + k])

        # Total sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)

        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        return h_var.value

    # Evaluate the current solution
    c5 = evaluate_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    print(f"Initial C5: {best_c5}")

    # Iterative optimization
    for it in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        # Project current h to satisfy constraints
        h = project_to_constraints(h)
        
        # Evaluate C5 at current h
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"Iteration {it}: C5 improved to {best_c5}")

        # Identify the worst-case shift k
        overlaps = []
        for k in range(n_points):
            idx = np.arange(n_points)
            val = np.sum(h[idx] * (1 - h[(idx - k) % n_points]))
            val *= dx
            overlaps.append(val)

        k_max = np.argmax(overlaps)
        val_kmax = overlaps[k_max]
        print(f"Worst-case shift k = {k_max} with value {val_kmax}")

        # Solve convex subproblem around worst shift k_max
        h = optimize_subproblem(h, k_max, dx)

        # Project back to constraints
        h = project_to_constraints(h)

    # Final evaluation
    best_h = project_to_constraints(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    print(f"Final C5 bound: {best_c5}")

    return best_h, best_c5, n_points
