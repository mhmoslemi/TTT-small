import numpy as np
import scipy.fft as fft
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 84
    n_points = 84
    h = initial_h_values.copy()
    total_mass = n_points / 2.0
    h = h / h.sum() * total_mass
    start_time = time.time()

    # Optimization routine
    max_iter = 1000
    no_improvement_count = 0
    best_c5 = float("inf")
    best_h = h.copy()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute FFT(h) and FFT(1 - h)
        H = fft.fft(h)
        G = fft.fft(1 - h)

        # Cross-correlation in frequency domain
        prod = H * np.conj(G)
        corr_real = fft.ifft(prod).real

        # Only consider indices where k in range(0, n_points-1)
        c5 = np.max(corr_real[n_points : 2 * n_points - 1] * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count > 50:
            break  # Exit early if no progress

        # Gradient: approximate by using the convolution result
        grad = fft.ifft(fft.fft(h * (1 - h))).real  # Symmetric gradient
        h -= 0.005 * grad  # Small step by line search

        # Projection to maintain constraints
        h = np.clip(h, 0, 1)
        h = h / h.sum() * total_mass  # Normalize

    return best_h, best_c5, n_points
