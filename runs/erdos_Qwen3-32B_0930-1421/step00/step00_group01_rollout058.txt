import numpy as np
import scipy.optimize
import time

def compute_c5(h, dx):
    n = len(h)
    c5 = 0.0
    for k in range(1, n):
        overlap = np.sum(h * (1 - np.roll(h, -k))) * dx
        c5 = max(c5, overlap)
    return c5

def objective(h_vec, n_points):
    dx = 2.0 / n_points
    h = h_vec.reshape((n_points,))
    return compute_c5(h, dx)

def constraint_sum(h_vec, n_points):
    return np.sum(h_vec) - n_points / 2

def bounds_func(h_vec, n_points):
    return np.clip(h_vec, 0, 1)
