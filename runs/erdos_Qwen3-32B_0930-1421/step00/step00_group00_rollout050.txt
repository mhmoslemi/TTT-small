import numpy as np
import time
from scipy.signal import correlate
import math
import cvxpy as cp
import scipy.optimize as opt

def evaluate_erdos_solution(h):
    h = np.array(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1 - h
    corr = correlate(h, one_minus_h, mode='full')
    normalized_corr = corr * dx
    c5 = np.max(normalized_corr)
    return c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
        total_mass = sum(h0) * (2.0 / n)
        assert abs(total_mass - 1.0) < 1e-6, "Initial h must sum to 1"
    else:
        n = 45
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    # Parameters for GA
    pop_size = 50
    max_gen = 200
    mutation_rate = 0.1
    mutation_strength = 0.1

    # Initialize
    population = [h0.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        mutation = np.random.normal(0, 1, n)
        population[i] += mutation_strength * mutation
        population[i] = np.clip(population[i], 0, 1)
        scale = (n / 2) / np.sum(population[i])
        population[i] *= scale

    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Optimization loop
    for gen in range(max_gen):
        current_time = time.time()
        if current_time - start_time > budget_s - 0.5:
            break

        # Evaluate fitness
        fits = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fits.append(c5)

        idxs = np.argsort(fits)
        top_idx = idxs[:int(pop_size * 0.2)]
        top_pop = [population[i] for i in top_idx]
        for i in top_idx:
            if fits[i] < best_c5:
                best_c5 = fits[i]
                best_h[:] = population[i].copy()

        # Crossover and mutation
        new_pop = []
        for _ in range(pop_size - len(top_pop)):
            p1 = np.random.choice(top_idx)
            p2 = np.random.choice(top_idx)
            child = 0.5 * (population[p1] + population[p2])
            if np.random.rand() < mutation_rate:
                mu = np.random.normal(0, 1, n)
                child += mutation_strength * mu
            child = np.clip(child, 0, 1)
            scale = (n / 2) / np.sum(child)
            child *= scale
            new_pop.append(child)

        population = top_pop + new_pop

    # Local refinement with projection and optimization
    def objective(h, dx):
        h = h.reshape(-1)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def projection(u):
        u = np.clip(u, 0, 1)
        s = np.sum(u)
        if s > 0:
            u *= (n / 2) / s
        return u

    def cb(xk):
        nonlocal best_h, best_c5
        h = projection(xk)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h
        return c5 > 1e-10

    # Run gradient-based optimization to refine
    h_opt = projection(best_h)
    res = opt.minimize(
        lambda x: objective(x, 2.0 / n),
        h_opt,
        method='L-BFGS-B',
        jac=None,
        bounds=[(0.0, 1.0)] * n,
        constraints=(),
        callback=cb,
        options={'maxiter': 20}
    )
    h_refined = projection(res.x)
    c5_refined = evaluate_erdos_solution(h_refined)

    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined

    return (best_h, best_c5, n)
