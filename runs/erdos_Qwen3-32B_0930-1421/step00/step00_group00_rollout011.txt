import time
import numpy as np
from evaluate_erdos_solution import evaluate_erdos_solution
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    h_initial = initial_h_values
    h_current = np.copy(h_initial)
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()
    best_c5 = np.inf
    best_h = np.copy(h_current)
    max_iter = 2000
    alpha = 0.1  # Initial step size
    alpha_min = 1e-6
    beta = 0.7   # Backtracking step size reduction factor

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute C₅ and identify k_max
        # Correlate h with (1-h)
        h1 = np.concatenate([h_current, h_current])
        h2 = 1 - h_current
        corr = correlate(h1, h2, mode='full') * dx
        corr = corr[n_points - 1 : 3 * n_points - 1]
        max_idx = np.argmax(corr)
        c5 = corr[max_idx]

        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h_current)

        # Identify shift k_max
        k_max = max_idx

        # Compute gradient based on k_max
        h1_pad = np.concatenate([h_current, h_current])
        grad = (1 - h1_pad[k_max : k_max + n_points]) * dx

        # Compute proposed h
        h_proposed = h_current - alpha * grad

        # Project back to feasible region
        # First: clip to [0, 1]
        h_proposed = np.clip(h_proposed, 0, 1)
        current_sum = np.sum(h_proposed)
        sum_diff = target_sum - current_sum

        # Adjust sum constraint
        if abs(sum_diff) < 1e-8:
            h_current = np.copy(h_proposed)
        else:
            # Count positions where we can adjust
            if sum_diff > 0:  # Need to increase value
                mask = (h_proposed < 1.0)
                adjust_num = np.sum(mask)
                if adjust_num == 0:
                    h_current = h_current  # No valid positions to increase
                else:
                    adjust_amount = sum_diff / adjust_num
                    h_proposed[mask] += adjust_amount
                    h_proposed = np.clip(h_proposed, 0, 1)
                    h_current = np.copy(h_proposed)
            else:  # Need to decrease value
                mask = (h_proposed > 0.0)
                adjust_num = np.sum(mask)
                if adjust_num == 0:
                    h_current = h_current  # No valid positions to decrease
                else:
                    adjust_amount = -sum_diff / adjust_num
                    h_proposed[mask] -= adjust_amount
                    h_proposed = np.clip(h_proposed, 0, 1)
                    h_current = np.copy(h_proposed)

        if i % 10 == 0:
            # Re-evaluate C₅
            h1 = np.concatenate([h_current, h_current])
            h2 = 1 - h_current
            corr = correlate(h1, h2, mode='full') * dx
            corr = corr[n_points - 1 : 3 * n_points - 1]
            potential_c5 = np.max(corr)

            if np.isnan(potential_c5) or potential_c5 > best_c5:
                h_current = np.copy(best_h)
                alpha *= beta
                continue

    # Final evaluation
    h1 = np.concatenate([best_h, best_h])
    h2 = 1 - best_h
    corr = correlate(h1, h2, mode='full') * dx
    corr = corr[n_points - 1 : 3 * n_points - 1]
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), float(final_c5), n_points)
