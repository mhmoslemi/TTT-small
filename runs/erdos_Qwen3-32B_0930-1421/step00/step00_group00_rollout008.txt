import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = cp.Parameter(n_points)
    h_curr = initial_h.copy()
    assert np.all(h_curr >= 0) and np.all(h_curr <= 1)
    assert abs(np.sum(h_curr) - target_sum) < 1e-7

    def compute_c5(h_vec):
        if np.any(h_vec < -1e-6) or np.any(h_vec > 1 + 1e-6):
            return np.inf
        h_norm = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full') * dx
        return np.max(corr)

    full_time = time.time()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    alpha = 0.1
    beta = 0.75
    max_iter = 2000
    tol = 1e-8
    it = 0

    while time.time() - full_time < budget_s - 0.1:
        it += 1
        h1 = h_curr
        corr = np.correlate(h1, 1 - h1, mode='full') * dx
        k_max = np.argmax(corr)
        lag = (h1.shape[0] - 1) - k_max
        grad = np.zeros_like(h1)
        grad[:lag] -= 1 - h1[lag:]  # decrease where h[x] affects 1 - h[x+k]
        grad[lag:] += 1 - h1[:lag]  # increase where 1 - h[x+k] affects h[x]

        step = alpha * grad
        prop = h1 - step
        prop = np.clip(prop, 0, 1)
        s = np.sum(prop)
        diff = target_sum - s
        if diff > 0:
            indices = prop < 1 - tol
            prop[indices] += diff / np.count_nonzero(indices)
        elif diff < 0:
            indices = prop > tol
            prop[indices] += diff / np.count_nonzero(indices)

        prop_c5 = compute_c5(prop)
        if prop_c5 < best_c5 - 1e-8:
            best_c5 = prop_c5
            best_h = prop.copy()
            h_curr = prop.copy()
            alpha *= 1.15  # success -> increase step size if feasible
        else:
            alpha *= beta  # backtrack if not improving
            if alpha < tol:
                alpha = 0.1

        if it % 100 == 0:
            print(f"Iter {it}, c5={best_c5:.6f}, time={time.time()-full_time:.1f}s")

        if time.time() - full_time > budget_s - 0.2:
            break

    return best_h.tolist(), best_c5, n_points
