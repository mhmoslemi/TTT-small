import numpy as np
import time
import random
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points // 2

    def initialize_population(pop_size):
        pop = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, copy=True).astype(float)
            pop.append(h)
        else:
            h = np.zeros(n_points)
            h[:target_sum] = 1.0
            np.random.shuffle(h)
            pop.append(h)

        for _ in range(pop_size - 1):
            h = np.sort(np.random.rand(n_points) / np.random.rand(n_points) * n_points)  # Dispersed random
            h = np.clip(h, 0.0, 1.0)
            h = h / h.sum() * target_sum
            h = np.round(h).astype(int)
            h = np.clip(h, 0, 1)
            pop.append(h)
        return np.array(pop)

    def fitness(h):
        h = np.clip(h, 0, 1).astype(float)
        h_sum = h.sum()
        if h_sum == 0:
            return float('inf')
        scale = target_sum / h_sum
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        c5 = evaluate_erdos_solution(h_scaled)
        return c5

    def crossover(h1, h2):
        s = random.randint(n_points // 4, 3 * n_points // 4)
        child = np.hstack((h1[:s], h2[s:]))
        child = child / child.sum() * target_sum
        child = np.round(child).astype(int)
        child = np.clip(child, 0, 1)
        return child

    def mutate(h):
        mutate_indices = random.sample(range(n_points), random.randint(1, 5))
        h_mut = h.copy()
        for i in mutate_indices:
            h_mut[i] = random.choice([0, 1])
        h_mut = h_mut / h_mut.sum() * target_sum
        h_mut = np.round(h_mut).astype(int)
        h_mut = np.clip(h_mut, 0, 1)
        return h_mut

    population_size = 50
    generations = 1000
    elitism = int(0.1 * population_size)

    pop = initialize_population(population_size)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break

        fitness_values = []
        for h in pop:
            c5 = fitness(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
            fitness_values.append((h, c5))

        fitness_values.sort(key=lambda x: x[1])
        elites = [x[0] for x in fitness_values[:elitism]]

        new_population = elites[:]
        while len(new_population) < population_size:
            h1, h2 = random.choices(elites, k=2)
            child = crossover(h1, h2)
            child = mutate(child)
            new_population.append(child)

        pop = np.array(new_population)

    h_opt = best_h / best_h.sum() * target_sum
    h_opt = np.round(h_opt).astype(int)
    h_opt = np.clip(h_opt, 0, 1)
    final_c5 = evaluate_erdos_solution(h_opt)
    return (h_opt, final_c5, n_points)
