import numpy as np
import scipy.optimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    lag_values = corr[len(h) - 1 : len(h) + len(h) - 1]
    c5 = np.max(lag_values) * dx
    return c5


def project_to_feasible_space(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to maintain integral = 1
    total_mass = np.sum(h)
    h *= n_points / (2.0 * total_mass)
    return h


def compute_gradient(h, n_points, best_lag):
    grad = np.zeros(n_points)
    for i in range(n_points):
        j_plus = (i + best_lag) % n_points
        j_minus = (i - best_lag) % n_points
        grad[i] = (1 - h[j_plus]) - h[j_minus]
    return grad
