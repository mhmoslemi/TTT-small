import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # Use initial_h_values provided by the harness
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    # Feasibility check: ensure ∫h dx = 1
    assert np.isclose(np.sum(h_init) * dx, 1.0, atol=1e-8)

    h = cp.Variable(n_points, nonneg=True)
    constraints = [h <= 1, h @ dx == 1]

    k_all = np.arange(-n_points + 1, n_points)
    # Compute all correlations: overlay of h(x) over 1-h(x + k)
    # Use cvxpy's piece-wise max over all k to minimize max_k C5
    overlaps = []
    for k in k_all:
        if -n_points < k < n_points:  # valid for full mode
            # Handle shifting via circular indices
            idx1 = np.arange(n_points)
            idx2 = (idx1 + k) % n_points
            overlap = cp.sum(
                h[idx1] * (1 - h[idx2]) * dx
            )
            overlaps.append(overlap)

    overlap_max = cp.Max(*overlaps)
    objective = cp.Minimize(overlap_max)

    prob = cp.Problem(objective, constraints)
    prob.solve(solver='GUROBI', verbose=False, TimeLimit=0.9 * budget_s, warm_start=True)
    if h.value is None:
        raise RuntimeError("Solver failed to find a feasible solution in time")

    h_values = np.clip(h.value, 0, 1)
    c5_bound = prob.value

    elapsed = time.time() - start_time
    if elapsed < 0.99 * budget_s:
        print(f"Attempt refinement with additional iterations for {budget_s - elapsed:.1f}s remaining")

        # Optional: refine by fixing some positions and height values
        # Extract peaks and attempt finer control over segment boundaries
        from scipy.ndimage import gaussian_filter1d
        h_density = gaussian_filter1d(h_values, sigma=n_points / 10)
        for attempt in range(3):
            # Add small zero-mean noise to smooth h_values slightly
            h_rand = np.clip(h_values + 0.05 * np.random.randn(n_points), 0, 1)
            h_rand_sum = h_rand.sum()
            if h_rand_sum > 0:
                h_rand *= (n_points / 2) / h_rand_sum  # enforce constraint

            h.value = h_rand
            overlap_max = cp.Max(*[cp.sum(h[idx1] * (1 - h[idx2]) * dx) for idx1, idx2 in [(np.arange(n_points), (np.arange(n_points) + k) % n_points) for k in k_all[:]]])
            prob = cp.Problem(cp.Minimize(overlap_max), constraints)
            prob.solve(solver='GUROBI', verbose=False, TimeLimit=0.2 * budget_s)
            if h.value is None:
                break
            h_values = np.clip(h.value, 0, 1)
            c5_bound = prob.value
            elapsed = time.time() - start_time
            if elapsed >= 0.99 * budget_s:
                break

    return h_values, c5_bound, n_points
