```python
import numpy as np
import math
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.where(globals()['initial_h_values'] > 0.5, 1.0, 0.0)
        target_sum = n_points / 2
        current_sum = np.sum(h)
        diff = int(target_sum - current_sum)
        while diff != 0:
            if diff > 0:
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 1
                diff -= 1
            else:
                idx = np.random.choice(np.where(h == 1)[0])
                h[idx] = 0
                diff += 1
    else:
        h = np.zeros(n_points, dtype=float)
        ones_idx = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[ones_idx] = 1.0

    h = h.copy()

    # Custom overlap evaluation
    def evaluate_overlap(h):
        def wrap_index(i):
            i %= n_points
            return i

        max_overlap = 0
        for k in range(-n_points + 1, n_points):
            overlap = 0
            for i in range(n_points):
                if h[i] > 0:
                    overlap += h[i] * (1 - h[wrap_index(i + k)])
            overlap *= dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    c5_current = best_c5

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    T = 1.0
    alpha = 0.99
    min_T = 1e-5

    patience_counter = 0
    max_patience = 1000

    while time.time() - start_time < max_time:
        idx1 = np.random.choice(np.where(h == 1)[0])
        idx0 = np.random.choice(np.where(h == 0)[0])

        h_new = h.copy()
        h_new[idx1] = 0
        h_new[idx0] = 1

        c5_candidate = evaluate_overlap(h_new)

        if c5_candidate < c5_current or random.random() < math.exp(-(c5_candidate - c5_current) / T):
            h[:] = h_new
            c5_current = c5_candidate
            if c5_candidate < best_c5:
                best_h[:] = h_new
                best_c5 = c5_candidate
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1

        if patience_counter >= max_patience:
            T = max(T * alpha * 10, min_T)
            patience_counter = 0
        else:
            T = max(T * alpha, min_T)

    return (best_h.tolist(), best_c5, n_points)
```
