import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_h(h):
    h = np.clip(h, 0, 1)
    n = len(h)
    total = np.sum(h)
    delta = total - n / 2
    if abs(delta) > 1e-8:
        if delta > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        scale = 1 + delta / np.sum(h[idx[:5]])
        h = h.copy()
        h[idx[:5]] *= scale
        h = np.clip(h, 0, 1)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr * dx), h

def run(seed=42, budget_s=1100, n_particles=50, n_iterations=500, n_points=70):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h0 = np.copy(initial_h_values)
    else:
        h0 = np.ones(n_points) / 2
    h0, _ = evaluate_h(h0)

    particles = np.random.normal(0.5, 0.1, (n_particles, n_points))
    velocities = np.zeros_like(particles)
    pbests = np.array([np.copy(p) for p in particles])
    pbest_scores = np.array([np.inf] * n_particles)
    gbest = np.copy(h0)
    gbest_score = np.inf

    t0 = time.time()
    for iter in range(n_iterations):
        if time.time() - t0 > budget_s - 1.0:
            break
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            velocities[i] = np.clip(velocities[i], -1, 1)
            score, corrected_h = evaluate_h(particles[i])
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbests[i] = np.copy(corrected_h)
            if score < gbest_score:
                gbest_score = score
                gbest = np.copy(corrected_h)
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbests[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
        if (iter + 1) % 10 == 0:
            print(f"iter={iter + 1}, gbest_score={gbest_score:.6f}", flush=True)
            if gbest_score < 0.380876:
                break

    best_h = gbest
    final_c5, _ = evaluate_h(best_h)

    return best_h, final_c5, n_points
