```python
import numpy as np
import time
from scipy.optimize import minimize
import evaluate_erdos_solution as evaluator


def smooth_c5(h, eta=1.0, dx=1.0):
    h = h.reshape(-1)
    n = len(h)
    correl = np.correlate(h, 1 - h, mode="full")
    correl = correl[n - 1:2 * n - 1]
    correl *= dx
    return (1.0 / eta) * np.log(np.sum(np.exp(eta * correl)))

def objective(h, smooth_eta, dx):
    return smooth_c5(h, smooth_eta, dx)

def constraint_sum(h, n):
    return np.sum(h) - n / 2

def constraints(h, n):
    con1 = constraint_sum(h, n)
    con2 = [{'type': 'ineq', 'fun': lambda x: 1 - x[i]} for i in range(len(h))]
    con3 = [{'type': 'ineq', 'fun': lambda x: x[i]} for i in range(len(h))]
    return [con1] + con2 + con3

def get_c5(h, n):
    h = np.array(h).clip(0, 1).reshape(-1)
    h = h / np.sum(h) * (n / 2)
    h = np.nan_to_num(h, copy=True)
    d = evaluator.evaluate_erdos_solution(h)
    return d['c5_bound'], h

def apply_budget(start_time, *args):
    return time.time() - start_time < budget_s

def generate_starting_pts(h0, num_pts=10, perturbation=0.2):
    base_h = np.copy(h0)
    base_h = base_h.reshape(-1)
    x0 = base_h.copy()
    n = len(x0)
    pts = []
    for i in range(num_pts):
        perturbation = (np.random.rand(n) - 0.5) * 2. * perturbation
        perturbed = x0 + perturbation
        perturbed /= np.sum(perturbed)
        perturbed *= n / 2
        perturbed = np.clip(perturbed, 0, 1)
        pts.append(perturbed)
    return pts

def run(seed=42, budget_s=1100, initial_smooth_eta=1.0, final_smooth_eta=100.0, n_sim_anneal=500):
    global initial_h_values
    assert 'evaluate_erdos_solution' in globals(), "Evaluator not available"
    assert 'initial_h_values' in globals(), "Initial h_values not available"

    start_time = time.time()
    h0 = np.array(initial_h_values).reshape(-1).copy()
    n = len(h0)

    smooth_eta = initial_smooth_eta
    best_c5 = float("inf")
    best_h = h0.copy()

    dx = 2.0 / n
    bound_constraints = ([0, 1] for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]

    starting_points = generate_starting_pts(h0, num_pts=5)

    for x0 in starting_points:
        x = x0.copy()
        t0 = time.time()
        while time.time() - t0 < 10 * 60 and time.time() - start_time < budget_s:
            if not apply_budget(start_time): break

            res = minimize(
                lambda h: objective(h, smooth_eta, dx),
                x0=x.copy(),
                bounds=bound_constraints,
                constraints=cons,
                method='SLSQP'
            )

            if res.success:
                x = res.x / np.sum(res.x) * (n / 2)
                x = x.clip(0, 1)
                c5, x_clipped = get_c5(x, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x_clipped
                smooth_eta = min(2 * smooth_eta, final_smooth_eta)
                break
        if not apply_budget(start_time): break

    if best_c5 == float("inf"):
        best_h, best_c5, n = evaluator.evaluate_erdos_solution(initial_h_values)

    return (list(best_h), best_c5, n)
```
