import numpy as np
import math
import time
import scipy
from scipy.fft import fft, ifft, fftshift
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute (1 - h) shifted and unshifted
    h1 = 1 - h
    # Compute correlation via FFT convolution (mode full)
    h_fft = fft(h, len(h) * 2)
    h1_fft = np.conj(fft(h1, len(h) * 2))
    # Convolution in frequency domain
    conv_fft = h_fft * h1_fft
    conv = np.real(ifft(conv_fft))
    # Extract [2*(n-1) ... 2] = [n ... 3n-2] indices for ``mode="full"``
    n = len(h)
    conv = conv[:2*n - 1]
    c5 = np.max(conv * dx)
    return c5

def proj_simplex(v, normalize=True):
    # Project onto simplex ∑x = total, x ∈ [0, 1]
    n = len(v)
    v = np.asarray(v)
    if normalize:
        total = n_points / 2
    else:
        return v.clip(0, 1)
    # Project to unit interval
    u = v.clip(0, 1)
    p = np.sort(u, axis=0)[::-1]
    ind = np.cumsum(p, axis=0) - np.linspace(total, total, n)
    idx = np.sum(ind > 0, axis=0)
    if idx == 0:
        return u
    thres = (np.cumsum(p, axis=0)[idx - 1:idx] - total) / idx
    p[p - thres > 0] -= thres
    p[p < 0] = 0
    return p

def run(seed=42, budget_s=1100, n_points=1024):
    global initial_h_values
    t0 = time.time()
    maxiter = 1000
    n_points_adj = min(n_points, 1024)
    dx = 2.0 / n_points_adj
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h[:n_points_adj]  # Truncate or pad
        h = np.pad(h, (0, max(0, n_points_adj - len(h))), mode='constant', constant_values=0)
    else:
        h = np.zeros(n_points_adj, dtype=np.float64)
        h[: int(0.4 * n_points_adj)] = 1.0
    h = h / np.sum(h) * (n_points_adj / 2)
    h = np.clip(h, 0, 1)

    alpha = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement = 0
    iteration = 0
    while time.time() - t0 < budget_s and iteration < maxiter:
        # FFT-based correlation
        h_fft = fft(h, len(h) * 2)
        h1_fft = np.conj(fft(1 - h, len(h) * 2))
        conv_fft = h_fft * h1_fft
        conv = np.real(ifft(conv_fft))
        conv = conv[:2 * n_points_adj - 1]
        c5 = np.max(conv * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Compute gradient via (h - shifted h) * dx
        # The convolution result aligns with h shifted and unshifted
        # Here used for simplistic heuristics to guide updates
        conv /= dx
        max_index = np.argmax(conv)
        k = max_index - n_points_adj + 1  # Shift in the full convolution
        if k != n_points_adj - 1:  # No wraparound
            h_shifted = np.zeros_like(h)
            if k > 0:
                h_shifted[k:] = h[:-k]
            else:
                h_shifted[-k:] = h[:-len(h_shifted) + k]
            gradient = (h - h_shifted) * dx
        else:
            gradient = np.zeros_like(h)

        # Update with projection
        h -= alpha * gradient
        h = proj_simplex(h, normalize=True)
        h = np.clip(h, 0, 1)
        iteration += 1
        if no_improvement > 100:
            alpha *= 0.9  # Decay learning rate on plateau
            no_improvement = 0

    return best_h, best_c5, n_points_adj
