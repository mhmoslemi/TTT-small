import numpy as np
import time
import cvxpy as cp
import scipy

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # initialize with provided construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure constraint: sum(h) == n_points / 2
    h_sum = np.sum(h)
    if not np.abs(h_sum - n_points / 2) < 1e-6:
        h *= (n_points / 2) / h_sum

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)

    # Optimization settings
    max_iter = 3000
    iter_count = 0
    stop_condition = max_iter

    while time.time() - start_time < budget_s and iter_count < stop_condition:
        # Compute overlap values
        conv = np.correlate(h, 1 - h, mode='full')
        conv = conv[(n_points): 2 * n_points - 1]
        max_idx = np.argmax(conv)

        # If maximum is at an endpoint, mirror it from the other side
        if max_idx == 0:
            max_idx = 1
        if max_idx == conv.size - 1:
            max_idx = conv.size - 2

        # Compute gradient at max point
        k = max_idx
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            idx1 = i
            idx2 = i + k
            grad[idx1] += (1 - h[idx2])
            grad[idx2] += -h[idx1]

        grad *= dx  # Scale by dx

        # Solve QP to update h in the direction of -grad
        h_new = cp.Variable(n_points)
        constraints = [h_new >= 0, h_new <= 1, cp.sum(h_new) == n_points / 2]
        objective = cp.Minimize(cp.sum_squares(h_new - (h - grad * 0.1)))
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            h = np.array(h_new.value).flatten()
        except:
            h = h - 0.01 * grad
            # Project back onto feasible set
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if not np.abs(h_sum - n_points / 2) < 1e-6:
                h *= (n_points / 2) / h_sum

        # Update best solution
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        iter_count += 1

    return (best_h, best_c5, n_points)
