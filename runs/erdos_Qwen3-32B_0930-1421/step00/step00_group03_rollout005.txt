import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    h0 = initial_h_values.copy()
    n = len(h0)
    dx = 2.0 / n

    def energy(h):
        h = h.astype(np.float64)
        h_normalized = h / dx
        overlaps = scipy.signal.correlate(h, 1.0 - h, mode='full')
        overlaps = overlaps * dx
        max_overlap = np.max(overlaps.astype(np.float64))
        return max_overlap

    start_time = time.time()

    energy_hat = energy(h0)
    best_h = h0.copy()
    best_energy = energy_hat
    temp = 1.0
    alpha = 0.99
    sigma = 0.1
    iter_energy = 0

    while time.time() - start_time < budget_s / 2:
        iter_energy += 1
        h2 = np.clip(h0 + sigma * np.random.randn(n), 0.0, 1.0)

        # Project h2 onto feasible set
        volume = np.sum(h2)
        if not np.isclose(volume, n / 2):
            diff = n / 2 - volume
            idx = np.argpartition(h2, -int(np.abs(diff)))[:int(np.abs(diff))]
            if diff < 0:
                h2[idx] -= np.where(h2[idx] > 0, diff / len(idx), 0)
            else:
                h2[idx] += np.where(h2[idx] < 1, diff / len(idx), 0)

        energy2 = energy(h2)
        if energy2 < best_energy or np.random.random() < np.exp((best_energy - energy2) / temp):
            h0 = h2
            h0 = np.clip(h0, 0, 1)
            energy_hat = energy2
            best_energy = energy_hat
            best_h = h0.copy()

        temp *= alpha
        if iter_energy % 10 == 0:
            h0 = best_h.copy()
            temp = max(temp, 0.1)

        if iter_energy % 10 == 0 and time.time() - start_time > budget_s / 3:
            # Restart if no improvement, taking care to not violate hard constraints
            h0 = np.clip(h0 + 0.3 * np.random.randn(n) * (1 - h0) * h0, 0, 1)
            temp = 0.8 if h0.std() > 0.1 else 1.0
            h0 = best_h * 0.9 + h0 * 0.1
            h0 = np.clip(h0, 0, 1)

    def grad(h):
        h = h.astype(np.float64)
        correl = scipy.signal.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(correl)
        factor = dx

        k = max_idx - (n - 1)
        correl = correl * dx
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                dh = (1.0 - h[j]) - h[i]
                deriv = dh * factor
                yield deriv
            else:
                yield 0.0

    def grad_energy(h):
        derivs = np.array(list(grad(h)))
        weights = np.abs(derivs)
        weights /= weights.sum() + 1e-12
        p = scipy.optimize.NonlinearConstraint(
            lambda x: np.array([x.sum() - n / 2]), lb=0.0, ub=0.0)
        
        for _ in range(10):
            def constraint(x):
                return np.sum(x) - n / 2
            def f(x):
                return derivs @ x

            result = scipy.optimize.minimize(
                f, x0=h, bounds=[(0.0, 1.0)] * n, constraints=[p], method='SLSQP')
            if result.success:
                h = result.x
                return h

        return h

    # Use Gradient Ascent Algorithm
    optimize_time = 0.0
    t_start = time.time()
    while optimize_time < budget_s / 2:
        g = np.array(grad(best_h))
        g = g - best_h * 0.1
        z = np.zeros(n)
        p = scipy.optimize.NonlinearConstraint(
            lambda x: np.sum(x) - n / 2, -1e-8, 1e-8)

            return -np.max(scipy.signal.correlate(x, 1.0 - x, mode='full') * dx)
        result = scipy.optimize.minimize(
            optimize_f, best_h, bounds=[(0, 1)] * n, method='SLSQP', constraints=[p])
        optimize_time = time.time() - t_start
        if not result.success:
            break
        if -result.fun < best_energy:
            best_energy = -result.fun
            best_h = result.x
        best_h = np.clip(best_h, 0, 1)

    h_seq = best_h
    C5_bound = energy(h_seq)

    return (best_h, C5_bound, n)
