import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_erdos_solution(h_values, n_points=None):
    if n_points is None:
        n_points = len(h_values)
    dx = 2.0 / n_points
    corr = correlate(h_values, 1 - h_values, mode='full')
    return np.max(corr) * dx

def project_to_feasible_set(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-8:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        while abs(diff) > 1e-8:
            h[idxs] += diff / len(idxs)
            h = np.clip(h, 0.0, 1.0)
            current_sum = h.sum()
            diff = target_sum - current_sum
            idxs = np.argsort(h)
    else:
        idxs = np.argsort(-h)
        while abs(diff) < -1e-8:
            h[idxs] -= (-diff) / len(idxs)
            h = np.clip(h, 0.0, 1.0)
            current_sum = h.sum()
            diff = target_sum - current_sum
            idxs = np.argsort(-h)
    return h

        h_array = np.array(h).ravel()
        result = evaluate_erdos_solution(h_array, n_points)
        obj_calls += 1
        return result

    n_particles = 50
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    global_best = None
    global_best_c5 = float('inf')

    if initial_h_values is not None:
        base = initial_h_values.copy()
    else:
        # basic construction
        base = np.zeros(n_points)
        base[n_points // 2] = n_points / 2
        base /= n_points / 2

    for _ in range(n_particles):
        h = base.copy()
        h += 0.1 * np.random.randn(n_points)
        h = project_to_feasible_set(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        c5 = compute_obj(h)
        personal_best.append(h.copy())
        personal_best_c5.append(c5)
        if c5 < global_best_c5:
            global_best = h.copy()
            global_best_c5 = c5

    start_time = time.perf_counter()
    iteration = 0
    no_improvement = 0
    max_no_improvement = 100
    omega = 0.7
    c1 = 1.5
    c2 = 1.5

    while time.perf_counter() - start_time < budget_s and no_improvement < max_no_improvement:
        for idx in range(n_particles):
            h = particles[idx]
            v = velocities[idx]
            p_best = personal_best[idx]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)

            cognitive = c1 * r1 * (p_best - h)
            social = c2 * r2 * (global_best - h)
            v = omega * v + cognitive + social
            v_clip = np.clip(v, -0.5, 0.5)
            h_new = h + v_clip
            h_new = project_to_feasible_set(h_new, n_points)

            c5_new = compute_obj(h_new)
            if c5_new < personal_best_c5[idx]:
                personal_best[idx] = h_new.copy()
                personal_best_c5[idx] = c5_new
                if c5_new < global_best_c5:
                    global_best = h_new.copy()
                    global_best_c5 = c5_new
                    no_improvement = 0
                else:
                    no_improvement += 1
            particles[idx] = h_new
            velocities[idx] = v_clip

        iteration += 1

    return global_best, global_best_c5, n_points
