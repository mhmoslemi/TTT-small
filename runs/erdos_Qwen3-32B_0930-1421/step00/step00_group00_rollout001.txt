import numpy as np
import time
from scipy.optimize import BFGS, minimize
from dataclasses import dataclass
from typing import Tuple, Optional, List, Dict

@dataclass
class Candidate:
    h: np.ndarray
    c5: float
    feasible: bool

def project_onto_constraints(h: np.ndarray, n_points: int) -> np.ndarray:
    """Project h onto the feasible region [0,1]^N with sum(h) = N/2"""
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    return h

def fitness(h: np.ndarray, n_points: int) -> Tuple[float, float, bool]:
    """Compute negative fitness from overlap integral"""
    dx = 2.0 / n_points
    if not (all(0 <= x <= 1 for x in h) and abs(np.sum(h) - n_points/2) < 1e-6):
        return float('-inf'), float('inf'), False

    h_func = np.trim_zeros(h, 'f')
    c = np.correlate(h_func, 1 - h_func, mode='full')
    c5 = dx * np.max(c)
    return -c5, c5, True

def mutate(h: np.ndarray, mutation_strength: float = 0.1) -> np.ndarray:
    """Add Gaussian noise and project to feasible set"""
    return project_onto_constraints(
        h + np.random.normal(0, mutation_strength, h.shape),
        len(h)
    )

def crossover(h1: np.ndarray, h2: np.ndarray) -> np.ndarray:
    """Arithmetic crossover"""
    alpha = np.random.uniform(0.4, 0.6)
    return project_onto_constraints(
        alpha * h1 + (1 - alpha) * h2,
        len(h1)
    )

def sample_population(parent: np.ndarray, pop_size: int, noise_scale: float = 0.1) -> List[Candidate]:
    """Generate initial population around parent solution"""
    candidates = []
    for i in range(pop_size):
        h = project_onto_constraints(
            parent + np.random.normal(0, noise_scale, parent.shape),
            len(parent)
        )
        fit, c5, feasible = fitness(h, len(parent))
        candidates.append(Candidate(h, c5, feasible))
    return candidates

def optimize_with_bounds(population: List[Candidate], n_points: int) -> Tuple[float, float]:
    """Optimize fitness while maintaining sum(h) = n_points/2 and 0 <= h <= 1"""
    best = min(population, key=lambda x: x.c5 if x.feasible else float('inf'))

    # BFGS optimization
    def objective(h_flat):
        h = h_flat.copy().reshape(n_points)
        h = project_onto_constraints(h, n_points)
        return fitness(h, n_points)[0]

    result = minimize(objective, best.h, method='BFGS', bounds=[(0, 1)] * n_points)
    h = project_onto_constraints(result.x.reshape(n_points), n_points)
    fit, c5, feasible = fitness(h, n_points)

    return (h, c5)
