import numpy as np
import time
import scipy
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Start from initial_h_values
    h = initial_h_values
    n_points = h.shape[0]
    dx = 2.0 / n_points

    # Ensure the constraint sum(h) == n_points / 2
    if np.sum(h) != n_points / 2:
        h = h / h.sum() * (n_points / 2)

    beta = 5.0
    step_size = 0.01
    max_iter = 100_000
    improvement_threshold = 0.001
    comp_start_time = time.time()
    best_c5 = None
    best_h = h.copy()

    for it in range(max_iter):
        elapsed = time.time() - comp_start_time
        if elapsed > budget_s - 1e-3:
            break
        if it % 100 == 0:
            beta *= 1.001  # slowly increase beta to focus more on max C_k

        # Compute current overlaps
        h1h = 1.0 - h
        overlaps = correlate(h, h1h, mode="full") * dx

        # Exclude the zero-shift and symmetric repeats
        k_start = n_points // 2
        k_end = 2 * n_points - k_start
        Ck = overlaps[k_start:k_end]

        exponential_term = np.exp(beta * (Ck - Ck.max()))
        weight = exponential_term / exponential_term.sum()

        # Compute gradient using the weighted average of individual overlaps
        norm_h1h = h1h.copy()
        for k in range(1, n_points):
            norm_h1h[k:(k + n_points)] += h * h1h[n_points - k]

        grad = (1.0 / beta) * norm_h1h * weight[:(n_points - 1)]

        # Gradient descent step
        h = h - step_size * grad.sum(axis=1)

        # Projection step: clip and rescale
        h_clipped = np.clip(h, 0.0, 1.0)
        h_scaled = h_clipped / h_clipped.sum() * (n_points / 2)
        h = h_scaled

        # Evaluate current C5
        overlaps = correlate(h, 1.0 - h, mode="full") * dx
        C5 = overlaps[n_points:n_points + n_points].max()

        if best_c5 is None or C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()

        #annealing
        if it > 0 and it % 500 == 0:
            if best_c5 - C5 < improvement_threshold:
                step_size *= 0.5
                initial_h_values[:] = h.copy()
                h = initial_h_values.copy()

        if it % 1000 == 0:
            print(f"Iteration {it}, C5: {best_c5:.6f}, elapsed: {elapsed:.1f}s")

    h_values = best_h
    c5_bound = max(np.correlate(h_values, 1 - h_values, mode="full") * dx)

    return h_values, c5_bound, n_points
