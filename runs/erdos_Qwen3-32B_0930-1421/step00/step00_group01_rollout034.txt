import numpy as np
import scipy.optimize as opt
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    max_c5 = 0.0
    for k in range(-n_points + 1, n_points):
        if k < 0:
            idx = slice(-k, None)
            other_idx = slice(0, n_points + k)
        elif k > 0:
            idx = slice(0, n_points - k)
            other_idx = slice(k, None)
        else:
            continue
        integral = np.sum(h[idx] * (1.0 - h[other_idx])) * dx
        max_c5 = max(max_c5, integral)
    return max_c5

def objective(h, n_points):
    return evaluate_c5(h, n_points)

def bounds_and_constraints(n_points):
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    A_eq = np.ones((1, n_points))
    b_eq = [n_points / 2.0]
    return bounds, A_eq, b_eq

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, n_points)

    def callback(x):
        nonlocal best_h, best_c5
        val = objective(x, n_points)
        if val < best_c5:
            best_h = x.copy()
            best_c5 = val
        return

    result = opt.minimize(
        lambda x: objective(x, n_points),
        h0=h0,
        method='SLSQP',
        bounds=bounds_and_constraints(n_points)[0],
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        callback=callback,
        options={'maxiter': 1000}
    )

    while time.time() - start_time < budget_s - 10:
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        candidate_h = np.clip(h0 + perturbation, 0, 1)
        candidate_h *= (n_points / 2.0) / np.sum(candidate_h)

        candidate_c5 = evaluate_c5(candidate_h, n_points)
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            print(f"Found better C₅ = {best_c5:.8f}")

    if result.fun < best_c5:
        best_h = result.x.copy()
        best_c5 = result.fun

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h *= (n_points / 2.0) / np.sum(best_h)
    return best_h, best_c5, n_points
