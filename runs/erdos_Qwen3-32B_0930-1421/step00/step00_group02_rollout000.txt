import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = 45  # Start from initial_h_values (45 points)
    initial_h = np.array(initial_h_values).copy() if 'initial_h_values' in globals() else None

    if initial_h is None:
        # Re-initialize: uniform values adjusted to meet the sum constraint
        h = np.full(n_points, 0.5)
        dx = 2.0 / n_points
        h *= (0.5 / (dx * h.sum()))  # scale to ensure sum constraint is met
        h = np.clip(h, 0, 1)
    else:
        h = initial_h.copy()

    best_h = h.copy()
    best_c5 = evaluate_h(C5_by_h)
    lowest_c5 = best_c5
    dx = 2.0 / n_points

    max_iter = 2000
    progress_threshold = 1.0e-4
    it = 0
    start_time = time.time()

    while time.time() < start_time + budget_s and it < max_iter:
        it += 1
        h_var = cp.Variable(n_points)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        overlaps = []
        for k in range(-n_points + 1, n_points):
            if (-k * dx < 0) or (k * dx > 2):
                continue  # indices out of bounds
            truncated = max(0, min(n_points - k, n_points - abs(k)))
            if truncated <= 0:
                continue
            offset = np.bitwise_and(abs(k) < n_points, k < n_points)
            local_sum = 0
            if k >= 0:
                local_sum += cp.sum((h_var[:n_points - k] * (1 - h_var[k:])))
            else:
                local_sum += cp.sum((h_var[abs(k):] * (1 - h_var[:n_points - abs(k)])))
            overlaps.append(local_sum * dx)

        obj = cp.Minimax(overlaps)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        result = prob.solve(solver='ECOS', verbose=False)

        if h_var.value is None:
            continue  # skip infeasible solutions

        h_new = np.clip(h_var.value, 0, 1)
        if np.isclose(np.sum(h_new), n_points / 2, atol=1e-4).all():
            c5_new = C5_by_h(h_new)
            if c5_new < lowest_c5:
                lowest_c5 = c5_new
                best_h = h_new
                print("Average C5 improved to", lowest_c5)

        # Perturb h and restart solver if no progress
        h = h_new.copy() if h_var.value is not None else h
        h += 5e-3 * np.random.normal(size=n_points)
        h = np.clip(h, 0, 1)

    return (best_h.tolist(), lowest_c5, n_points)


def evaluate_h(func):
    h = func(initial_h_values)
    dx = 2.0 / len(h)
    h_values = np.array(h)
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    conv *= dx
    return conv.max()

def C5_by_h(h):
    return evaluate_erdos_solution(h)[0]
