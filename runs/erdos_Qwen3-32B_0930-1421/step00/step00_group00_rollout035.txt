import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
        dx = 2.0 / n_points
    else:
        # Default construction: uniform distribution
        n_points = 45
        h_init = np.full(n_points, 0.5)

    # Define step function evaluation
    def evaluate(h):
        corr = correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    # Enforce constraints
    def normalize_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h)
        h_scaled = h_clipped * (n_points / 2) / total
        return h_scaled
    
    # Fitness function for GA
    def fitness(h):
        h_feasible = normalize_h(h)
        c5 = evaluate(h_feasible)
        return -c5  # Minimize C5

    # Initial population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        h = h_init.copy() + 0.1 * np.random.randn(n_points)
        h = normalize_h(h)
        population.append(h)

    # Genetic algorithm main loop
    max_gen = 200
    generation = 0

    while generation < max_gen and time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_values = [fitness(h) for h in population]
        indices = np.argsort(fitness_values)
        best_idx = indices[0]
        best_h = population[best_idx]
        best_c5 = -fitness_values[best_idx]
        
        # Selection (top 20%)
        elite_count = max(1, int(pop_size * 0.2))
        elites = [population[idx] for idx in indices[:elite_count]]
        
        # Crossover
        offspring = []
        for _ in range(pop_size - elite_count):
            parent1 = population[np.random.choice(indices[:pop_size//2])]
            parent2 = population[np.random.choice(indices[:pop_size//2])]
            child = 0.5 * (parent1 + parent2)
            # Mutation
            child += 0.1 * np.random.randn(n_points)
            child = normalize_h(child)
            offspring.append(child)
        
        # Replace population with elites + offspring
        population = elites + offspring
        generation += 1

    # Post-processing with gradient descent
    def obj_func(h_vec):
        h = h_vec.reshape(n_points)
        h = normalize_h(h)
        return evaluate(h)
    
    def grad_func(h_vec):
        h = h_vec.reshape(n_points)
        h = normalize_h(h)
        # Compute gradient numerically
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_mod = h.copy()
            h_mod[i] += eps
            h_mod = normalize_h(h_mod)
            f_plus = evaluate(h_mod)
            h_mod[i] -= 2 * eps
            h_mod = normalize_h(h_mod)
            f_minus = evaluate(h_mod)
            grad[i] = (f_plus - f_minus) / (2 * eps)
        return grad

    h_flat = best_h.copy().ravel()
    res = minimize(obj_func, h_flat, method='BFGS', jac=grad_func, options={'maxiter': 20, 'disp': False})
    h_refined = res.x.reshape(n_points)

    # Final evaluation
    h_final = normalize_h(h_refined)
    c5_bound = evaluate(h_final)

    return h_final.tolist(), c5_bound, n_points
