import numpy as np
import cvxpy as cp
from time import time
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    n_points = 45 if 'initial_h_values' in globals() else 101
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        assert np.all(h >= 0.0) and np.all(h <= 1.0)
        assert abs(np.sum(h) - n_points / 2.0) < 1e-6
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0  # basic initial guess
        h = h / h.sum() * n_points / 2.0  # normalize
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Define ranges for shifting
    k_vals = np.arange(1, n_points)

    for iter in range(10000):
        current_time = time()
        if current_time - start_time > budget_s * 0.99:  # allow testing with some margin
            break

        # Step 1: compute overlaps for all known k
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        worst_k_index = np.argmax(corr[1:-1])
        worst_k = worst_k_index

        # Step 2: Formulate a new convex problem for the worst-k subproblem
        class_wrapper = np.array(h)
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2.0]
        shifted_window = np.zeros(n_points)

        # Flatten and place 1s in shifted region
        for i in range(n_points):
            if i - worst_k >= 0 and i - worst_k < n_points:
                shifted_window[i] = 1.0 - class_wrapper[i - worst_k]

        overlap = cp.sum(cp.multiply(h_var, shifted_window)) * dx
        objective = cp.Minimize(overlap)
        problem = cp.Problem(objective, constraints)

        problem.solve(solver=cp.GUROBI if hasattr(cp, "GUROBI") else cp.SCIP)  # fallback solvers
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            continue

        new_h = np.clip(h_var.value, 0.0, 1.0)
        new_c5 = evaluate_erdos_solution(new_h)

        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

        # Safety limits to prevent instability
        if iter % 200 == 0:
            # Gradient descent-style perturbation
            perturbation = np.random.normal(0, 0.001, n_points)
            best_h = best_h + perturbation
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = best_h / best_h.sum() * n_points / 2.0  # re-normalize
            best_c5 = evaluate_erdos_solution(best_h)

        if iter % 100 == 0 and best_c5 < 0.381:
            print(f"Iter {iter}, C5 = {best_c5:.6f}")

    return (best_h.tolist(), best_c5, n_points)
