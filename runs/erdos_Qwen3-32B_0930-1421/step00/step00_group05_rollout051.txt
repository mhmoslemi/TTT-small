import numpy as np

def evaluate_step_function(h, dx):
    # Compute the overlap for all k (k in 0 to 2n)
    # For k in 0 to n-1: shifts to the right
    # For k in n to 2n: shifts to the left (wrapping around)
    h_full = np.concatenate([h, h])
    overlaps = np.zeros(2 * h.shape[0])
    for k in range(h.shape[0] * 2):
        h_shifted = np.roll(h_full, -k)[h.shape[0]:]
        overlap = np.sum(h * (1 - h_shifted)) * dx
        overlaps[k] = overlap
    return np.max(overlaps)

def project_h(h, n_points):
    # Clip to feasible region [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Normalize to satisfy the integral constraint
    h_clipped /= np.sum(h_clipped)
    h_clipped *= n_points / 2.0
    return h_clipped

def gradient_step(h, dx, beta=10.0, n_points=70):
    h_full = np.concatenate([h, h])
    h_correction = np.zeros(h.shape)

    overlaps = np.zeros(2 * n_points)

        h_shifted = np.roll(h_full, -k)[n_points:]
        overlap = np.sum(h * (1 - h_shifted)) * dx
        overlaps[k] = overlap

    max_index = int(np.argmax(overlaps))
    w = np.exp(beta * (overlaps - overlaps[max_index]))
    w /= np.sum(w)

        h_shifted = np.roll(h_full, -k)[n_points:]
        weight = w[k]
        h_corr = (1 - h_shifted) * weight
        h_correction += h_corr
    
    grad_flat = n_points * (h_correction - np.mean(h_correction)) * dx
    h_new = h - 0.01 * grad_flat
    return project_h(h_new, n_points)
