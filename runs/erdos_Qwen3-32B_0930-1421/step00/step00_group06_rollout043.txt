import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)

    h = cp.Variable(n_points, nonneg=True)
    dx = 2.0 / n_points
    obj_sum = n_points / 2
    constants = [1 - h[i] for i in range(n_points)]
    shifts = [int(2 * dx * k) for k in range(1, 5)]

    # Formulate bilinear max with convex surrogation
    nu_k = cp.Parameter(n_points, pos=True)
    constraint = [
        h <= 1,
        h >= 0,
        cp.sum(h) == obj_sum
    ]
    for k in shifts:
        shifted = [h[(i + k) % n_points] for i in range(n_points)]
        overlap = cp.sum([h[i] * constants[(i + k) % n_points] for i in range(n_points)]) * dx
        constraint.append(nu_k[k] >= overlap)

    epsilon = 1e-6
    objective_surrogate = cp.Minimize(cp.sum(cp.log(nu_k + epsilon)))

    start_time = time.time()
    h_opt = initial_h.copy()
    nu = np.ones(n_points) * dx

    while time.time() - start_time < budget_s - 5:
        nu_k.value = nu
        prob = cp.Problem(objective_surrogate, constraint)
        prob.solve(solver=cp.OSQP, verbose=False)
        if h.value is None:
            h_opt = initial_h.copy()
            break
        h_opt = np.clip(h.value, 0, 1)
        h_opt = h_opt / np.mean(h_opt) * (n_points / 2)

        # Compute actual overlaps
        shifted_h = np.roll(h_opt, shift=-shifts, axis=0)
        overlaps = [np.sum(h_opt[i] * (1 - shifted_h[i, k]) for i in range(n_points)) * dx
                    for k in range(len(shifts))]

        nu = np.maximum(nu, overlaps)

        # Early stopping
        nu_converg = np.std(nu)
        if nu_converg < 1e-4:
            break

    C5_bound = max(np.correlate(h_opt, 1 - h_opt, mode="full")[1:] * dx)
    return h_opt, C5_bound, n_points
