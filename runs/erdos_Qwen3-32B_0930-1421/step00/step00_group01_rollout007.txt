import numpy as np
import time
import random
from copy import deepcopy

def evaluate_c5(n_points, h_values):
    h = np.array(h_values, dtype=np.float64)
    dx = 2.0 / n_points
    total = h.sum()
    if not np.isclose(total, n_points / 2.0):
        # Normalization to meet constraint: sum(h) = n_points / 2
        h = h / total * (n_points / 2.0)

    clip_h = np.clip(h, 0.0, 1.0)
    ones_minus_h = 1.0 - clip_h

    corr = np.correlate(clip_h, ones_minus_h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def mutate(solution, n_points, noise_level=0.02):
    n_mutate = random.randint(1, max(2, n_points // 100))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    values = np.random.uniform(0, 1, size=n_mutate)
    new_h = solution.copy()
    new_h[indices] = values
    new_h = new_h / new_h.sum() * (n_points / 2.0) # equalize sum
    return np.clip(new_h, 0.0, 1.0)

def crossover(parent1, parent2, n_points):
    split_idx = random.randint(n_points // 4, 3 * n_points // 4)
    child = np.concatenate((parent1[:split_idx], parent2[split_idx:]))
    child = child / child.sum() * (n_points / 2.0)
    return np.clip(child, 0.0, 1.0)

def run():
    start_time = time.time()
    budget_s = 1100
    n_points = 300
    pop_size = 50
    generations = 1000
    elite_pct = 0.2
    mut_pct = 0.02
    early_stop_threshold = 50

    # Initialize population
    population = []

    # Use initial solution if available
    if 'initial_h_values' in globals():
        eig = np.array(initial_h_values, copy=True)
        # Resize to target n_points
        eig = np.interp(np.linspace(0, len(eig)-1, n_points), range(len(eig)), eig)
        population.append(eig.tolist())

    # Fill with random solutions that meet the constraints
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2.0)
        h = np.clip(h, 0.0, 1.0)
        population.append(h.tolist())

    best_solution = None
    best_c5 = float('inf')
    no_improvement = 0

    # Main evolutionary loop
    gen = 0
    while time.time() - start_time < budget_s and gen < generations:
        # Evaluate population
        pop_with_c5 = []
        for individual in population:
            c5 = evaluate_c5(n_points, individual)
            pop_with_c5.append((individual, c5))

        # Sort by fitness (C₅)
        pop_with_c5.sort(key=lambda x: x[1])

        current_best, current_c5 = pop_with_c5[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = current_best
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= early_stop_threshold:
            break  # No progress for a while, exit early

        # Generate next generation
        new_population = [p[0] for p in pop_with_c5[:int(pop_size * elite_pct)]]

        # Select parents for breeding
        selected = pop_with_c5[:int(pop_size * 0.4)]
        np.random.shuffle(selected)
        for i in range(0, min(len(selected), pop_size - len(new_population)), 2):
            if i+1 < len(selected):
                parent1 = selected[i][0]
                parent2 = selected[i+1][0]
                child = crossover(parent1, parent2, n_points)
                new_population.append(child)

        # Add mutations
        while len(new_population) < pop_size:
            candidate = random.choice(population)
            mutated = mutate(candidate, n_points, mut_pct)
            new_population.append(mutated)

        population = new_population
        gen += 1

    # Final evaluation
    final_c5 = evaluate_c5(n_points, best_solution)
    return best_solution, final_c5, n_points
