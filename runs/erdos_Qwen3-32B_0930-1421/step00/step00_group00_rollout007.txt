import numpy as np

from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    assert np.abs(np.sum(h) - target_sum) < 1e-6

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h)

    # Optimization parameters
    alpha = 1.0  # Initial step size
    beta = 0.5   # Backtracking decrease factor
    gamma = 0.9  # Acceptance ratio
    max_iter = 10000
    print_iter = 100

    grad_cache = np.zeros_like(h)

    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 5:
            break

        # Compute overlap for all shifts
        # Avoid mode="full" to prevent unnecessary numpy optimization
        h1 = np.concatenate([h, np.zeros(n_points - 1)])
        h2 = np.concatenate([1 - np.roll(h, shift=1), np.zeros(1)])
        corr = np.correlate(h1, h2, mode="valid") * dx

        k_star = np.argmax(np.abs(corr))
        max_value = corr[k_star]

        # Gradient for optimal shift
        k_start = max(0, k_star - 1)
        k_end = min(n_points, k_star + 1)
        grad_cache[:] = 0.0
        for k in range(k_start, k_end):
            shifted_indices = np.clip(np.arange(n_points) + k, 0, n_points - 1)
            grad_cache += (1 - h[shifted_indices]) * dx

        grad_norm = np.max(np.abs(grad_cache))
        if grad_norm < 1e-6:
            break  # Converged

        grad_cache /= grad_norm  # Normalize

        # Backtracking line search
        current_alpha = alpha
        new_h = np.copy(h) - current_alpha * grad_cache
        new_h = np.clip(new_h, 0, 1)

        # Projection to sum constraint
        def project_sum_constraint(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if abs(sum_x - target_sum) > 1e-6:
                x = (x / sum_x) * target_sum
                x = np.clip(x, 0, 1)
            return x

        new_h = project_sum_constraint(new_h)

        # Make sure projected sum is enforced with higher precision
        sum_diff = np.sum(new_h) - target_sum
        if abs(sum_diff) > 1e-8:
            adjustment = np.round(sum_diff)
            adjustment_idx = np.argmin(new_h)
            new_h[adjustment_idx] += adjustment

        # Objective evaluation
        new_c5 = evaluate_erdos_solution(new_h)

        if new_c5 < best_c5:
            h = new_h
            best_h = np.copy(h)
            best_c5 = new_c5

        if iter % print_iter == 0 and iter > 0:
            print(f"Iter {iter}: best C5 = {best_c5}")

    # Ensure constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = project_sum_constraint(best_h)
    final_c5 = evaluate_erdos_solution(best_h)

    return (best_h, final_c5, n_points)
