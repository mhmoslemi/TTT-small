import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import scipy.optimize as opt


def evaluate_C5(h, dx):
    """Compute the C5 bound from the current h."""
    h = h.copy()
    h[h < 0] = 0
    h[h > 1] = 1
    h_sum = h.sum()
    h = h / h_sum * len(h) / 2  # normalize for dx = 2 / n_points
    cor = correlate(h, 1 - h, mode="full")
    C5 = np.max(cor) * dx
    return C5, h


def rescale_h(h):
    """Rescale h to satisfy the integral = 1 constraint."""
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum > 1e-10:
        h = h / h_sum * len(h) / 2
    else:
        h[:] = 0.0
        h[0] = len(h) / 2
    return h


def gradient_direction_C5(h, dx):
    """Compute the gradient of C5 w.r.t h for a single maximum overlap point."""
    h = h.astype(np.float64)
    cor = np.correlate(h, 1 - h, mode="full")
    max_id = np.argmax(cor)
    grad = np.zeros_like(h)
    max_offset = max_id - len(h) + 1
    for i in range(len(h)):
        if 0 <= (i + max_offset) < len(h):
            grad[i] = 2 * h[i] - 1 - 2 * h[i + max_offset] + 1
        else:
            grad[i] = 2 * h[i] - 1
    return grad * dx


def run(seed=42, budget_s=1100, n_points=84):
    """Optimize the Erdős C5 problem."""
    np.random.seed(seed)
    dx = 2.0 / n_points

        h[5:42] = 0.3
        h[42:] = 0.2
        h = rescale_h(h)

    start_time = time.time()
    best_h, best_C5 = h.copy(), np.inf
    learning_rate = 0.01
    simplex_size = 0.1
    iter_count = 0
    simp_proj = 0.9  # contraction parameter for simplex

    while time.time() - start_time < budget_s:
        iter_count += 1
        h = rescale_h(h)
        h_current = h.copy()
        C5, h = evaluate_C5(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

        if iter_count % 50 == 0:
            # Restart with a simplex exploration
            h_vals = np.array([rescale_h(h_current + simplex_size * (np.random.rand(n_points) - 0.5)) for _ in range(3)])
            h_vals = np.vstack([h_current] + h_vals.tolist())
            C5_vals = [evaluate_C5(h, dx)[0] for h in h_vals]
            best_simplex_id = np.argmin(C5_vals)
            h = h_vals[best_simplex_id].copy()
            learning_rate *= 0.5

        else:
            # Gradient step
            grad = gradient_direction_C5(h, dx)
            h = h - learning_rate * grad
            h = rescale_h(h)

        # Recalculate C5 again
        C5_post = evaluate_C5(h, dx)[0]
        if C5_post < best_C5:
            best_C5 = C5_post
            best_h = h.copy()

    h_vals, c5, n_pts = best_h, best_C5, n_points
    return h_vals, c5, n_pts
