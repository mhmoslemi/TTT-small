import time
import numpy as np
import cvxpy as cp
import scipy.signal as signal
from evaluate_erdos_solution import evaluate_erdos_solution
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    n_points = 256  # Sample count - high enough for accuracy, low enough for speed
    max_iter = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h /= np.sum(h) * dx  # Normalize to maintain constraint
    else:
        h = 0.5 * np.ones(n_points)
        h = h / np.sum(h) * n_points / 2  # Normalize

    ttracker = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        H = h - h**2
        result = np.correlate(h, H, mode='full') * dx
        return np.max(result)

    def feasible(h):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    # Phase 1: Frequency domain optimization
    def frequency_optimize(h, dx):
        H = np.fft.fft(h)
        freq = np.fft.fftfreq(n_points) * n_points
        mask = np.abs(freq) < n_points / 2
        z = cp.Variable(n_points)
        obj = cp.Minimize(cp.max(cp.abs(z)))
        constraints = [
            z == cp.fft(h - h ** 2),
            h >= 0, h <= 1,
            cp.sum(h) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT)
        except Exception as e:
            return h
        return h

    # Phase 2: Gradient descent on energy landscape
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.70:
            break

        h = frequency_optimize(h, dx)
        h = feasible(h)

        corre = np.correlate(h, h - h**2, mode='full') * dx
        k_max = np.argmax(np.abs(corre))
        k = int(k_max)

        gradient = np.zeros_like(h)
        gradient[n_points - k:] += 1.0 - 2.0 * h[n_points - k:]
        gradient[:len(gradient) - k] -= 1.0 - 2.0 * h[:len(gradient) - k]
        gradient *= (1.0 - h) * h
        h -= 0.05 * gradient
        h = feasible(h)

        current_c5 = compute_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if time.time() - start_time > budget_s:
            break

    # Final refinement using Nelder-Mead
    def wrapper(x):
        x = x.reshape(h.shape)
        x = feasible(x)
        return compute_c5(x, n_points)

    from scipy.optimize import minimize
    res = minimize(wrapper, x0=best_h, bounds=[(0, 1)] * n_points, method='Nelder-Mead')
    h_nelder = feasible(res.x)
    final_c5 = compute_c5(h_nelder, n_points)

    if final_c5 < best_c5:
        return h_nelder, final_c5, n_points
    return best_h, best_c5, n_points
