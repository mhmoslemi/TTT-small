import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h = initial_h_values
    h = np.copy(initial_h)
    n_points = len(h)
    dx = 2.0 / n_points

    max_iter = 1000
    lr = 0.01
    tolerance = 1e-6
    best_h = np.copy(h)
    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    for iter_ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute overlap for all lags
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx_plus = (i + k) % n_points
            idx_minus = (i - k) % n_points
            grad[i] = (1 - h[idx_plus]) - h[idx_minus]
        grad *= dx

        # Update h and project to [0, 1]
        h -= lr * grad
        h = np.clip(h, 0, 1)

        # Enforce integral constraint
        h *= n_points / 2 / np.sum(h)

        # Track best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

        # Early stopping
        if np.abs(c5 - best_c5) < tolerance:
            break

        # Reduce learning rate
        lr *= 0.95

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    n_points = len(h)
    corr = correlate(h, 1 - h, mode="full")
    corr *= dx  # Weight for dx
    c5 = np.max(corr)
    return c5
