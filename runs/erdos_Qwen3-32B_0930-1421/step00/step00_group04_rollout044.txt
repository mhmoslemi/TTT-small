import numpy as np
import time
import scipy.optimize
from math import isclose

def evaluate(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if not isclose(h_sum, 1.0, abs_tol=1e-6):
        h *= (1.0 / h_sum) / dx
        h = np.clip(h, 0, 1)

    ones_h = 1 - h
    c = np.correlate(h, ones_h, mode="full") * dx
    return np.max(c)

def objective(x, n_points):
    h = x.reshape(-1)
    dx = 2.0 / n_points
    return evaluate(h, dx)

def constraint_sum(x, n_points):
    dx = 2.0 / n_points
    return np.sum(x) * dx - 1.0

def get_initial_h(n_points, h_base=None):
    if h_base is not None:
        h = np.resize(h_base, n_points)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid - 1:mid + 1] = (1.0 / (2 * (2.0 / n_points)))
    return h

def optimize_h(n_points, h_initial=None, budget_s=1100):
    start_time = time.time()
    dx = 2.0 / n_points
    best_h = get_initial_h(n_points, h_initial)
    best_obj = objective(best_h, n_points)

    # Simplex setup
    simplex_size = 0.05
    simplex = [best_h + np.random.uniform(-simplex_size, simplex_size, n_points) for _ in range(4)]
    simplex = [np.clip(s, 0, 1) for s in simplex]
    for s in simplex:
        s_sum = np.sum(s)
        s *= (1.0 / s_sum) / dx
        s = np.clip(s, 0, 1)

    while time.time() - start_time < budget_s - 5:
        # Gradient descent step
        result = scipy.optimize.minimize(
            objective,
            x0=best_h.copy(),
            args=(n_points,),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum, 'args': (n_points,)},
            options={'maxiter': 10}
        )
        if result.success:
            h_test = result.x
            new_obj = objective(h_test, n_points)
            if new_obj < best_obj:
                best_h = h_test
                best_obj = new_obj

        # Simplex expansion
        all_h = [best_h] + simplex
        all_obj = [objective(h, n_points) for h in all_h]
        best_simplex_idx = np.argmin(all_obj)
        best_simplex_h = all_h[best_simplex_idx]
        if all_obj[best_simplex_idx] < best_obj:
            best_h = best_simplex_h
            best_obj = all_obj[best_simplex_idx]
            simplex_size *= 1.1
        else:
            simplex_size *= 0.9

        # Renew simplex
        simplex = []
        for _ in range(3):
            delta = np.random.uniform(-simplex_size, simplex_size, n_points)
            new_h = best_h + delta
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h) * dx
            new_h /= new_h_sum
            new_h = np.clip(new_h, 0, 1)
            simplex.append(new_h)

    return best_h, best_obj
