```python
import numpy as np
import time
import cvxpy as cp
import scipy.signal


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    initial_h_values = globals().get('initial_h_values', None)
    n_points = 45 if initial_h_values is not None else 45
    dx = 2.0 / n_points

    h = np.array(initial_h_values) if initial_h_values is not None else np.zeros(n_points)
    current_h = h * 0.0 + 0.5  # initialize to 0.5

    def clip_and_scale(values):
        values = np.clip(values, 0, 1)
        h_sum = np.sum(values)
        values *= n_points / 2 / h_sum
        values = np.clip(values, 0, 1)
        return values

    def compute_c5(h_values):
        h_flat = np.array(h_values)
        corr = scipy.signal.correlate(h_flat, 1 - h_flat, mode='full')
        corr_flat = corr[n_points - 1: n_points * 2 + 1]
        return np.max(np.abs(corr_flat * dx))

    # Optimization phase 1: Frequency domain (CVXOpt)
    def freq_optimize(h0):
        h = cp.Variable(shape=(n_points), nonneg=True)
        real = h * (1 - h)
        # Initialize with previous h
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2,
        ]
        obj = cp.Minimize(cp.norm_fft(real, 1))
        problem = cp.Problem(obj, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, verbose=False, maxitn=1000, feastol=1e-6)
        except Exception:
            problem.solve(solver=cp.ECOS, verbose=False, maxiters=1000)

        return np.clip(h.value, 0, 1)

    current_c5 = compute_c5(current_h)
    best_c5 = current_c5
    best_h = np.copy(current_h)

    best_h_time = time.time()
    print(f"Initial C5: {best_c5:.6f}")

    dx = 2.0 / n_points

    while time.time() - start_time < max_time:
        trial_h = freq_optimize(current_h)
        trial_h = clip_and_scale(trial_h)
        new_c5 = compute_c5(trial_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = trial_h
            print(f"New best C5: {best_c5:.6f} at {time.time() - start_time:.2f}s")
            best_h_time = time.time()
        current_h = trial_h.copy()

        if time.time() - best_h_time > max_time * 0.4:
            break

    return best_h.tolist(), best_c5, n_points
```
