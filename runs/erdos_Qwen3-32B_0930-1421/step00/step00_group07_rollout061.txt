import numpy as np
import math
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, initial_h_values=None):
    np.random.seed(seed)
    
    # Use provided initial_h_values or create balanced random binary h
    if initial_h_values is not None:
        n_points = len(initial_h_values)
        h = (initial_h_values > 0.5).astype(int)
        # Enforce balance if needed
        while np.sum(h) != n_points // 2:
            diff = n_points // 2 - np.sum(h)
            if diff > 0:
                idxs_0 = np.where(h == 0)[0]
                replace = np.random.choice(idxs_0, size=abs(diff), replace=False)
                h[replace] = 1
            else:
                idxs_1 = np.where(h == 1)[0]
                replace = np.random.choice(idxs_1, size=abs(diff), replace=False)
                h[replace] = 0
    else:
        n_points = 80  # Reasonable size that balances quality and speed
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    patience_counter = 0
    patience_limit = 1000
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate move (flip two bits: one 1 to 0 and one 0 to 1)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        # Evaluate candidate
        current_c5 = evaluate_erdos_solution(h)
        new_c5 = evaluate_erdos_solution(h_new)

        # Accept new solution
        if new_c5 < best_c5 or np.random.rand() < math.exp((current_c5 - new_c5) / T):
            h[:] = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                patience_counter = 0
            else:
                patience_counter += 1
            if patience_counter >= patience_limit:
                T = max(T * 1.5, min_T)
                patience_counter = 0
        if T > min_T:
            T *= alpha

    return (best_h, best_c5, n_points)
