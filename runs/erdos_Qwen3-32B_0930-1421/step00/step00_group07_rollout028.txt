import numpy as np
import random
import time
import math

def evaluate_overlap(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = np.correlate(h, ones_minus_h, mode='full')
    return correlations * dx


def energy(h):
    overlaps = evaluate_overlap(h, len(h))
    return np.max(overlaps)


def generate_neighbor(h, n_points):
    h = h.copy()
    indices_0 = np.where(h == 0)[0]
    indices_1 = np.where(h == 1)[0]
    i = random.choice(indices_0)
    j = random.choice(indices_1)
    h[i], h[j] = h[j], h[i]
    return h
