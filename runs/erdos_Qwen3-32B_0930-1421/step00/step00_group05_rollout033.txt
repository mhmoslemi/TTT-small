import numpy as np
import cvxpy as cp
import time
from numpy import correlate
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial = initial_h_values.copy()
    n_points = len(initial)
    dx = 2.0 / n_points
    target_sum = int(n_points / 2)

    h = cp.Parameter(n_points, nonneg=True)
    h.value = initial.copy()

    # Ensure sum(h) == target_sum by soft constraints using L2 penalty
    def loss_func(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals /= h_vals.sum() * dx
        h_vals += (target_sum - h_vals.sum()) / (n_points*dx)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals_sum = h_vals.sum()
        scale = target_sum / h_vals_sum
        h_vals = h_vals * scale
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals

    def compute_c5(h_vals):
        h_vals = np.ascontiguousarray(h_vals)
        ones = np.ones_like(h_vals)
        corr = correlate(h_vals, ones - h_vals, mode="full")
        corr = corr[n_points - 1 : 2 * n_points - 1]
        corr *= dx
        return np.max(corr)

    start_time = time.time()
    best_h = h.value.copy()
    best_c5 = np.inf
    alpha = 0.001
    decay_rate = 0.5
    patience = 30
    no_improvement = 0

    while time.time() - start_time < budget_s:
        h_vals = h.value.copy()
        grad = np.zeros_like(h_vals)

        for k in range(1, n_points):
            shift = k
            window = n_points - shift
            if window < 1:
                continue
            h_k = h_vals[shift:]
            one_minus_h_k = 1 - h_vals[:window]
            overlap = h_k * one_minus_h_k
            overlap = np.mean(overlap)
            delta = np.zeros_like(h_vals)
            delta[shift:] += one_minus_h_k
            delta[:window] += -h_k
            grad += 1 * delta

        h_vals -= alpha * grad
        h_vals = loss_func(h_vals)
        h.value = h_vals

        current_c5 = compute_c5(h.value)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.value.copy()
            no_improvement = 0
            alpha *= 1.01
        else:
            no_improvement += 1
            alpha *= decay_rate
            if no_improvement >= patience:
                break

    return best_h, best_c5, n_points
