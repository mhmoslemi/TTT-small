import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    dx = 2.0 / 80
    n_points = 80

    # Get initial configuration
    h = np.copy(initial_h_values)

    # Convert to binary
    h = (h > 0.5).astype(float)
    count = h.sum()

    # Normalize to have half ones
    n_half = n_points / 2
    diff = int(count - n_half)
    if diff > 0:
        for _ in range(diff):
            idx = np.random.choice(np.where(h > 0.5)[0])
            h[idx] = 0.0
    elif diff < 0:
        for _ in range(-diff):
            idx = np.random.choice(np.where(h < 0.5)[0])
            h[idx] = 1.0

    # Main optimization loop
    best_h = np.copy(h)
    best_c5 = evaluate_c5(h)

    T = 1.0
    min_T = 1e-5
    alpha = 0.99
    patience = 1000
    no_improvement = 0

    while time.time() - start_time < budget_s:
        # Flip two random bits to maintain balance
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        a = np.random.choice(ones_idx)
        b = np.random.choice(zeros_idx)
        h_candidate = np.copy(h)
        h_candidate[a] = 0.0
        h_candidate[b] = 1.0

        c5_current = evaluate_c5(h)
        c5_candidate = evaluate_c5(h_candidate)

        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            no_improvement = 0
            T = max(T * 1.05, min_T)  # boost temperature on improvement
        else:
            if np.random.rand() < np.exp(-(c5_candidate - c5_current) / T):
                h = h_candidate
            else:
                no_improvement += 1

        T = max(T * alpha, min_T)

        # Reset patience occasionally
        if no_improvement > patience:
            T = max(T * 1.5, min_T)
            no_improvement = 0

    # Final validation
    assert abs(best_h.sum() - n_half) < 1e-10, "Constraint violation"
    c5_bound = evaluate_c5(best_h)

    return (best_h.tolist(), c5_bound, n_points)

def evaluate_c5(h):
    h = np.array(h)
    dx = 2.0 / len(h)
    cross_corr = correlate(h, 1 - h, mode="full")
    return np.max(cross_corr * dx)
