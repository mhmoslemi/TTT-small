import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    h = np.asarray(h)
    dx = 2.0 / len(h)
    overlaps = np.array([np.sum(np.roll(1 - h, -k) * h) * dx for k in range(len(h))])
    return max(overlaps), h[np.isfinite(h)].mean()

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 45
    dx = 2.0 / n_points
    total = n_points / 2

    # Safeguard: fallback in case initial_h_values is invalid
    if "initial_h_values" in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        middle = n_points // 2
        h[middle - 1 : middle + 1] = dx / 2  # Try narrow base
        h = np.clip(h, 0.0, 1.0)
        h /= np.sum(h) * dx
        h *= total

    best_h, best_c5 = h.copy(), np.inf

    while time.time() - start_time < budget_s:
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == total,
        ]

        # Approximate max overlap convolution
        # For each shift k, compute overlap = h * (1 - shift(h, k)) * dx
        # Here use linearization via a convex envelop of the max term
        # Use randomized 0.05 step heuristic to stimulate transmitter
        h_val = h.value if ('h.value' in locals()) else best_h
        shifts = range(1, 8)
        overlaps = [np.sum(np.roll(1 - h_val, -k) * h_val) * dx for k in shifts]
        w = np.exp(-np.abs(shifts - 2)) / np.sum(np.exp(-np.abs(shifts - 2)))  # Weight forward
        objective = cp.Maximize(-cp.sum(overlaps * w))  # minimize approx max overlap

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.PDLP, max_rounds=100, verbose=False)

        if prob.status == 'optimal':
            h_new = h.value
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = h_new / (np.sum(h_new) * dx) * total
            c_new, _ = evaluate_erdos_solution(h_new)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new

        # Early stopping if no improvement
        if (time.time() - start_time) > (budget_s - 30) and best_c5 >= 0.380876:
            break

    return best_h.tolist(), best_c5, n_points
