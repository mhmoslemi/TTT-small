import numpy as np
import time
from scipy import optimize
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    # Constants
    global initial_h_values
    try:
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # If no initial_h_values is available, use a uniform distribution
        n_points = 95
        h = np.full(n_points, 1.0 / n_points, dtype=np.float64)

    dx = 2.0 / n_points
    np.random.seed(seed)
    t0 = time.time()

    # Helper: compute overlap integrals
    def compute_overlap(h_vals):
        h_tiled = np.concatenate([h_vals, h_vals])
        overlap = np.zeros(2 * n_points)
        for k in range(1, n_points + 1):
            h_k = np.roll(h_tiled, k)
            overlap[k - 1] = np.sum(h_vals * (1 - h_k[:n_points])) * dx
        for k in range(1, n_points + 1):
            h_k = np.roll(h_tiled, -k)
            overlap[k + n_points - 1] = np.sum(h_vals * (1 - h_k[:n_points])) * dx
        return overlap[n_points - 1:n_points + n_points]

    # Optimization objective
    def objective(x):
        h_vals = x.reshape((n_points,))
        overlap = compute_overlap(h_vals)
        return np.max(overlap)

    # Constraints: integral = 1 and bounds [0,1]
    def constraint_int(x):
        return np.sum(x) - (n_points / 2)

    constraints = {'type': 'eq', 'fun': constraint_int}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Starting point refinement
    h = np.clip(h, 0.0, 1.0)
    h *= (n_points / 2) / np.sum(h)

    # Main optimization loop with time budget
    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 300

    # Record start time
    t_start = time.time()

    for i in range(max_iter):
        if time.time() - t_start > budget_s * 0.98:
            break
        res = optimize.minimize(objective, x0=h, method='SLSQP', bounds=bounds,
                                constraints=constraints, options={'maxiter': 10, 'ftol': 1e-6})
        h = res.x
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)

        # Compute and track best solution
        overlap = compute_overlap(h)
        current_c5 = np.max(overlap)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final validation and edge correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    overlap = compute_overlap(best_h)
    c5_final = np.max(overlap)

    return best_h.tolist(), max(c5_final, 0.0), n_points
