import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    
    # Copy and extract initial values
    h = np.array(initial_h_values, copy=True)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initial validation
    assert np.all((h >= 0) & (h <= 1))
    assert np.isclose(np.sum(h), n_points / 2)
    
    best_h = h.copy()
    target_sum = n_points / 2
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    alpha = 0.1
    beta = 0.7
    max_iter = 200
    for iter in range(max_iter):
        if time.time() - t_start > budget_s - 10:
            break
        
        # Compute overlaps and find worst shift
        overlap = correlate(h, 1 - h, mode="full") * dx

        # Get active indices (with non-zero correlation)
        active_mask = (overlap > 1e-6)
        active_overlaps = overlap[active_mask]
        if len(active_overlaps) == 0:
            break
        k_max = np.argmax(overlap[active_mask])
        
        # Get the actual index corresponding to the maximum
        active_indices = np.where(active_mask)[0]
        k_max_actual = active_indices[k_max]
        
        # Compute gradient direction based on this worst shift
        max_shift = k_max_actual
        indices = np.arange(n_points)
        if max_shift >= n_points:
            g = np.zeros(n_points)
            for i in range(n_points):
                j = i - max_shift
                if j < 0:
                    j += n_points
                g[i] = (1 - h[j]) * dx
        else:
            g = (1 - np.roll(h, shift=-max_shift)) * dx
        
        # Create proposed direction
        h_proposed = h - alpha * g
        
        # Project onto feasible region
        h_proposed = np.clip(h_proposed, 0, 1)
        
        sum_diff = target_sum - np.sum(h_proposed)
        if abs(sum_diff) > 1e-6:
            # Find eligible indices
            if sum_diff > 0:
                # Need to add more
                eligible_mask = h_proposed < 1
            else:
                # Need to reduce
                eligible_mask = h_proposed > 0
            eligible_idx = np.where(eligible_mask)[0]
            if len(eligible_idx) > 0:
                delta = sum_diff / len(eligible_idx)
                h_proposed[eligible_idx] += delta
        
        h_proposed = np.clip(h_proposed, 0, 1)
        
        # Check and update best solution
        c5_proposed = evaluate_erdos_solution(h_proposed)
        if np.isfinite(c5_proposed) and c5_proposed < best_c5:
            best_h[:] = h_proposed[:]
            best_c5 = c5_proposed
            h = h_proposed.copy()
        else:
            alpha *= beta
    
    # Final validation
    assert np.all((best_h >= 0) & (best_h <= 1))
    assert np.isclose(np.sum(best_h), target_sum)
    
    return (best_h, best_c5, n_points)
