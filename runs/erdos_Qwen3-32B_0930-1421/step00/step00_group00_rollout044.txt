import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    np.random.seed(seed)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h_arr = h.reshape(-1)
        max_overlap = 0.0
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(h_arr, shift=k)
            overlap = np.sum(h_arr * (1 - shifted))
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    def project_h(h, n_points):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = n_points / 2 / total if total != 0 else 1.0
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    def optimize_inner(h, k, n_points, dx):
        h = cp.Parameter(n_points)
        h = h.value
        overlap = 0.0
        shifted = np.roll(h, shift=k)
        for i in range(n_points):
            overlap += h[i] * (1 - shifted[i])
        overlap *= dx
        return overlap

    # Use initial construction if provided
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    else:
        # Initialize with uniform height: 1 / 2
        n_points = 45
        h = np.full(n_points, 0.5)

    # Compute initial C5
    c5 = compute_c5(h, n_points)
    best_h = h.copy()
    best_c5 = c5
    dx = 2.0 / n_points

    max_iter = 1000
    print_interval = 100

    for iter in range(max_iter):
        # Enforce constraints and project
        h = project_h(h, n_points)

        # Evaluate all possible overlaps
        L = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(h, shift=k)
            L[k + n_points - 1] = np.sum(h * (1 - shifted)) * dx

        k_max = np.argmax(L)

        # Form convex subproblem to minimize C5 at this k
        h_var = cp.Variable(n_points)

        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]

        objective = cp.Minimize(L[k_max + n_points - 1])
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iters=1000)

        h = h_var.value

        # Project h to feasible region
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = n_points / 2 / total
        h *= scale
        h = np.clip(h, 0, 1)

        # Evaluate new C5
        c5 = compute_c5(h, n_points)

        # Update best if improved
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Check budget
        if time.time() - start_time > budget_s - 5:
            break

        # Progress info
        if iter % print_interval == 0:
            print(f"Iter {iter}: Best C5 = {best_c5:.6f}")

    # Final check of constraints
    best_h = project_h(best_h, n_points)
    c5 = compute_c5(best_h, n_points)

    return (best_h, c5, n_points)
