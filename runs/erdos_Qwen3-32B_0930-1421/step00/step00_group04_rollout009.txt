import numpy as np
import time
from scipy.fft import fft, ifft, fftfreq
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Discretization parameters
    n_points_opt = 84  # Use the parent construction
    dx = 2.0 / n_points_opt
    
    # Initial guess
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.uniform(0, 1, n_points_opt)
    h = h / h.sum() * (n_points_opt / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.1
    lr_decay = 0.995
    
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Fourier approach
        H = fft(h)
        G = fft(1 - h)
        C = ifft(H * G.conj()).real * dx
        
        k_max = np.argmax(np.abs(C))
        grad = np.zeros_like(h)
        grad = np.roll(h - h[k_max:], k_max)  # Based on overlap displacement
        
        # Gradient descent step
        h_candidate = h - lr * grad
        
        # Projection onto feasible set
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / h_candidate.sum() * (n_points_opt / 2.0)
        
        # Evaluate new candidate
        c5 = evaluate_erdos_solution(h_candidate)
        
        # Update solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            lr *= lr_decay
        else:
            h = h.copy()
        
    # Return the best solution found
    return best_h, best_c5, n_points_opt
