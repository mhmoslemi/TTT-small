import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Preload initial construction
    n_points = len(initial_h_values)
    h_best = initial_h_values.copy()
    c5_best = evaluate_erdos_solution(h_best)

    dx = 2.0 / n_points

    # Define symbolic variables
    h = cp.Variable(n_points)
    t = cp.Variable()

    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2  # Ensure ∫h dx = 1
    ]

    # Compute all C_k from convolution: h * (1 - h) evaluated in full mode
    # Let's define it symbolically
    Ck = []
    for k in range(1, n_points - 1):  # k=1 to k=n-2 avoid edge issues with 'full'
        shift = np.roll(h, -k)
        overlap = cp.sum(h * (1 - shift))
        Ck.append(overlap * dx <= t)  # scale by dx

    obj = cp.Minimize(t)
    constraints += Ck
    problem = cp.Problem(obj, constraints)

    # Solve for h that also obey constraints
    problem.solve(solver=cp.GUROBI, TimeLimit=budget_s * 0.8)
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        c5_opt = t.value
        # Report only if objective improved to respect record significantly
        if c5_opt < c5_best:
            h_best = h_opt

    # Final cleanup and return
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.95:
        # If time budget allows, rerun second pass with tighter tolerances
        # or apply local search around current h_best

        # Here, we might refine using numerical method like basin-hopping
        from scipy.optimize import minimize
        def objective_func(x):
            h1 = x.reshape((-1,))
            corr = np.correlate(h1, 1 - h1, mode='full') * dx
            max_corr = np.max(corr)
            return max_corr

        bounds = [(0, 1)] * n_points
        constraint = {'type': 'eq', 'fun': lambda h1: np.sum(h1) - n_points / 2}
        res = minimize(objective_func, h_best, bounds=bounds, constraints=constraint, method="SLSQP")
        if res.success:
            h_best = res.x
            c5_best = res.fun

    # Final validation of the best solution
    c5_bound = evaluate_erdos_solution(h_best)

    return h_best, c5_bound, n_points
