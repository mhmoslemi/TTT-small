import numpy as np
import time
import math
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

from evaluate_erdos_solution import evaluate_erdos_solution

# Ensure the initial_h_values is available from imports
n_points = len(initial_h_values)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-8:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        indices = np.argsort(h)
        to_increase = indices[:int(abs(diff) + 1)]
        h[to_increase] += np.repeat(diff / len(to_increase), len(to_increase)).clip(0.0, 1.0 - h[to_increase])
    else:
        indices = np.argsort(-h)
        to_decrease = indices[:int(abs(diff) + 1)]
        h[to_decrease] -= np.repeat(abs(diff) / len(to_decrease), len(to_decrease)).clip(0.0, h[to_decrease])
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    dx = 2.0 / len(h)
    c = correlate(h, 1 - h, mode='full')
    c = c * dx
    return np.max(c)

def sa_step(h, T, sigma):
    n_points = len(h)
    h_proposal = h + sigma * np.random.randn(n_points)
    h_proposal = project_to_feasible(h_proposal, n_points)
    c_new = compute_c5(h_proposal)
    c_curr = compute_c5(h)
    if c_new < c_curr or np.random.rand() < np.exp((c_curr - c_new) / T):
        return h_proposal, c_new
    return h, c_curr

def run(seed=42, budget_s=1100, *args, **kwargs):
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    h = project_to_feasible(h, len(h))

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    sa_iterations = 0
    max_sa_iterations = 500
    no_improvement = 0
    max_no_progress = 100

    while time.time() - start_time < budget_s and sa_iterations < max_sa_iterations:
        h, current_c5 = sa_step(h, T, sigma)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= max_no_progress:
            T = max(T * 2.0, 0.1)
            alpha = 0.995
            no_improvement = 0
        else:
            T *= alpha
        sa_iterations += 1

    return best_h, best_c5, len(h)
