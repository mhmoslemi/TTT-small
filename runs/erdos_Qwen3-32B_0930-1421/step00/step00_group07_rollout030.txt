import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    return np.max(corr) * dx

def flip_two_bits(h):
    n = len(h)
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    i = np.random.choice(indices_1)
    j = np.random.choice(indices_0)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

    h = initial_h_values.copy()
    h[h > 0.5] = 1.0

    # Enforce sum constraint
    while np.sum(h) != n_points / 2:
        if np.sum(h) < n_points / 2:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0.0

    t0 = time.time()
    T = 1.0
    alpha = 0.99
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    current_h = h.copy()
    current_c5 = best_c5
    no_improvement = 0
    max_no_improvement = 1000
    min_temp = 1e-5

    while time.time() - t0 < budget_s:
        h_new = flip_two_bits(current_h)
        c5_new = evaluate_overlap(h_new)
        delta = c5_new - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            T = max(min_temp, T * 2)  # Reset temperature and halve decay
            alpha = max(alpha * 0.9, min(alpha, 0.1))
            no_improvement = 0

        T = max(T * alpha, min_temp)

    # Final validation
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    final_c5 = evaluate_overlap(best_h)
