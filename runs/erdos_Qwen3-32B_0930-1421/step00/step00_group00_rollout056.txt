import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    full_corr = correlate(h, 1 - h, mode='full') * dx
    c5 = np.max(full_corr)
    return c5, full_corr

def project_feasible(h, target_sum):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    diff = target_sum - current_sum

    # Adjust sum constraint
    if abs(diff) > 1e-8:
        indices = np.where((h_clipped > 0) | (h_clipped < 1))[0]
        if diff > 0:
            # Need to add
            h_clipped[indices] += diff / len(indices)
            h_clipped = np.clip(h_clipped, 0, 1)
        else:
            # Need to subtract
            h_clipped[indices] += diff / len(indices)
    return h_clipped

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    start_time = time.time()
    best_h = np.copy(h)
    best_c5, _ = evaluate_erdos_solution(h)

    alpha = 0.1
    beta = 0.7
    max_iter = 200
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        h_current = np.copy(best_h)

        # Compute full cross correlation
        corr = correlate(h_current, 1 - h_current, mode='full') * dx
        k_max = np.argmax(corr)
        k_max = int(k_max)

        # Compute gradient: index adjustment for circular shift
        grad = np.zeros_like(h_current)
        if k_max < len(h_current):
            # Regular shift
            grad = (1 - h_current[k_max:]) * dx
            grad = np.concatenate([np.zeros(k_max), grad])
        else:
            # Wrap-around shift
            wrap = k_max - n_points
            grad = (1 - h_current[wrap:]) * dx
            grad = np.concatenate([np.zeros(wrap), grad, (1 - h_current[:wrap]) * dx])

        # Propose update
        h_proposed = h_current - alpha * grad
        h_proposed = project_feasible(h_proposed, target_sum)

        # Evaluate proposal
        c5_proposal, _ = evaluate_erdos_solution(h_proposed)
        if c5_proposal < best_c5 and not np.isnan(c5_proposal):
            best_h = np.copy(h_proposed)
            best_c5 = c5_proposal
            alpha *= 1.05  # Slight increase
        else:
            alpha *= beta  # Reduce step size

    return best_h, best_c5, n_points
