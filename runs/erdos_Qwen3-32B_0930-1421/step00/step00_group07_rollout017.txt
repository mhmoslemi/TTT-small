import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)

    # Rescale h to satisfy sum(h) == n_points / 2
    h = initial_h * (n_points / 2.0 / np.sum(initial_h))

    # Hyperparameters
    max_iter = 5000
    alpha = 0.01
    patience = 200
    decay_factor = 0.5
    dx = 2.0 / n_points

    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    no_improvement = 0
    current_alpha = alpha

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute cross-correlation
        corr = correlate(h, 1.0 - h, mode="full")
        C5 = np.max(corr) * dx

        # Early stopping check
        if C5 < best_C5 - 1e-6:
            best_C5 = C5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                current_alpha *= decay_factor
                no_improvement = 0

        # Gradient estimation
        grad = np.zeros_like(h)
        # Compute derivative contribution from all overlapping positions
        for k in range(-n_points + 1, n_points):
            left = max(0, -k)
            right = min(n_points, n_points - k)
            if left >= right:
                continue
            overlap = (1 - h[k + left:right]) if k >= 0 else (1 - h[left:right - k])
            grad[left:right] += (-overlap if k >= 0 else -overlap)

        # Gradient descent
        h -= current_alpha * grad
        h = np.clip(h, 0, 1)
        h *= (n_points / 2.0) / np.sum(h)

        # Periodic reinitialization if no improvement
        if np.random.rand() < 1e-3:
            h = initial_h * (n_points / 2.0 / np.sum(initial_h))

    # Final validation step
    final_C5 = evaluate_C5(best_h, dx)

    return (best_h, final_C5, n_points)


def evaluate_C5(h, dx):
    corr = correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
