import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_vec = initial_h_values
    n_points = len(h_vec)

    dx = 2.0 / n_points

    max_iter = 500
    eps = 1e-6
    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Initial values for duals
    nu = []

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        h_np = np.array(np.clip(h.value if h.value is not None else h_vec, 0, 1), dtype=np.float64)
        # Compute overlaps
        one_minus_h = 1 - h_np
        corrs = np.correlate(h_np, one_minus_h, mode="full")
        for k in range(1, n_points):
            corr_k = corrs[k] * dx
            if len(nu) <= k:
                nu.append(corr_k)
            else:
                nu[k] = max(nu[k], corr_k)

        # Update surrogate objective
        surrogate = 0
        for k_val in nu:
            surrogate += cp.log(k_val + eps)

        objective = cp.Minimize(surrogate)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iters=50)

        if h.value is None:
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec / h_vec.mean() * (n_points / 2)
        else:
            h_val = np.array(h.value, dtype=np.float64)
            h_vec = np.clip(h_val, 0, 1)
            if h_vec.sum() == 0:
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec / h_vec.mean() * (n_points / 2)
            else:
                h_vec = h_vec / h_vec.mean() * (n_points / 2)

    # Final evaluation of the function
    one_minus_h = 1 - h_vec
    corrs = np.correlate(h_vec, one_minus_h, mode="full")
    c5_bound = np.max(corrs[1: n_points] * dx)

    h_values = h_vec
    return h_values, c5_bound, n_points
