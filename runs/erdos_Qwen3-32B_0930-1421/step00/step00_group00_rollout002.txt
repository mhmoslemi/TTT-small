import numpy as np
import time
import scipy.optimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        h1 = np.array(h)
        h1[h1 < 0] = 0
        h1[h1 > 1] = 1
        corr = correlate(h1, 1 - h1, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h1)
        if abs(sum_h - target_sum) > 1e-4:
            c5 += 10000  # huge penalty for sum constraint violation
        return c5

    def project_to_simplex(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - target_sum) > 1e-8:
            h *= target_sum / total
        return h

    def crossover(a, b):
        alpha = np.random.rand(*a.shape)
        return project_to_simplex(alpha * a + (1 - alpha) * b)

    def mutate(h, noise_strength=0.1):
        h_noise = h + noise_strength * (2 * np.random.rand(*h.shape) - 1)
        return project_to_simplex(h_noise)

    def select_parents(population, fitness):
        ranks = np.argsort(fitness)
        return population[ranks[:int(0.2 * len(population))]]

    def initialize_population(parent, noise_strength=0.1, pop_size=50):
        pop = [parent]
        for _ in range(pop_size - 1):
            h = mutate(parent, noise_strength)
            pop.append(h)
        return np.array(pop)

    population = initialize_population(initial_h_values, noise_strength=0.1, pop_size=50)
    best_h = initial_h_values
    best_c5 = compute_c5(initial_h_values)
    best_h = project_to_simplex(best_h)

    generation = 0
    max_gen = 200

    while time.time() - start_time < budget_s and generation < max_gen:
        # Evaluate fitness
        fitness = np.array([compute_c5(h) for h in population])
        # Select parents
        parents = select_parents(population, fitness)
        # Create offspring
        offspring = []
        for _ in range(len(population)):
            a = parents[np.random.randint(len(parents))]
            b = parents[np.random.randint(len(parents))]
            child = crossover(a, b)
            if np.random.rand() < 0.3:  # Additional mutation with limited strength
                child = mutate(child, noise_strength=0.05)
            offspring.append(child)
        offspring = np.array(offspring)
        # Replace worst half of population
        new_population = np.where(fitness.reshape(-1, 1) < np.median(fitness),
                                  population, offspring)
        population = new_population
        # Update best
        local_best_c5 = min(compute_c5(h) for h in population)
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = project_to_simplex(population[np.argmin(fitness)])
        generation += 1

    # Post-processing: refine best solution
    def objective(x):
        h = project_to_simplex(x.reshape(-1))
        return compute_c5(h)

    result = scipy.optimize.minimize(objective, x0=best_h,
                                       method='BFGS',
                                       options={'maxiter': 20,
                                                'disp': False})
    final_h = project_to_simplex(result.x)
    final_c5 = compute_c5(final_h)

    elapsed = time.time() - start_time
    print(f"Final C5: {final_c5:.7f} found in {elapsed:.1f}s")

    return (final_h, final_c5, n_points)
