import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    h_padded = np.concatenate((h, np.zeros_like(h)))
    corr = correlate(h, 1 - h_padded, mode='valid')
    return np.max(corr * dx)

def smooth_max_correlate(h, dx, eta=1.0):
    h_padded = np.concatenate((h, np.zeros_like(h)))
    corr = correlate(h, 1 - h_padded, mode='valid')
    return eta**-1 * np.log(np.sum(np.exp(eta * corr * dx)))

def normalize_h(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if np.isclose(s, 0):
        h[:] = 1.0 / n_points
        return
    h *= (n_points / 2) / s

def perturb_h(h, noise_strength=0.1):
    h = h + np.random.uniform(-noise_strength, noise_strength, h.shape)
    normalize_h(h, h.shape[0])
    return h
