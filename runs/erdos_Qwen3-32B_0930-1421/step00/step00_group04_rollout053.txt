import numpy as np
import time

def evaluate_c5_single(n_points, h_values):
    dx = 2.0 / n_points
    h_arr = np.array(h_values).astype(float)
    one_minus_h = 1.0 - h_arr
    corr = np.correlate(h_arr, one_minus_h, mode='full') * dx
    c5 = np.max(np.abs(corr))
    return c5

def mutate(h, mutation_rate=0.05):
    h = np.array(h)
    n = len(h)
    indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in indices:
        delta = np.random.uniform(-0.2, 0.2)
        old_val = h[i]
        new_val = np.clip(old_val + delta, 0.0, 1.0)
        h[i] = new_val
    # Ensure ∑h = n_points / 2
    h_sum = np.sum(h)
    h += (n_points / 2 - h_sum) / n
    h = np.clip(h, 0.0, 1.0)
    return list(h)

def crossover(parent1, parent2):
    n = len(parent1)
    idx = np.random.randint(1, n-1)
    child = parent1[:idx] + parent2[idx:]
    # Adjust to meet area constraint
    child = np.array(child)
    child += (n / 2 - np.sum(child)) / n
    child = np.clip(child, 0.0, 1.0)
    return list(child)

def tournament_selection(population, fitnesses, k=3):
    idxs = np.random.choice(len(population), k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx], fitnesses[best_idx]

def clean_population(population):
    n = len(population[0])
    # Remove NaNs
    population = [p if np.all(np.isfinite(p)) else np.random.rand(n) for p in population]

    # Normalize to constraint 0 ≤ h[i] ≤ 1
    population = [np.clip(p, 0.0, 1.0) for p in population]

    # Correct to meet ∑h = n / 2
    for i in range(len(population)):
        total = np.sum(population[i])
        difference = n / 2 - total
        population[i] += difference / n
        population[i] = np.clip(population[i], 0.0, 1.0)
    return population

    total = h0.sum()

    # Generate initial population
    population = [h0]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        total = h.sum()
        h += (n_points / 2 - total) / n_points
        h = np.clip(h, 0.0, 1.0).tolist()
        population.append(h)

    population = clean_population(population)
    
    best_c5 = float('inf')
    best_h = h0
    start_time = time.time()
    elapsed_time = 0

    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 10:

        # Evaluate fitness for each individual
        fitness = []
        for ind in population:
            c5 = evaluate_c5_single(n_points, ind)
            fitness.append(c5)

        # Update best solution

        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]

        # Create new population
        new_population = []
