import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h_values, n_points):
    dx = 2.0 / n_points
    h = np.clip(h_values, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2):
        scale_factor = (n_points / 2) / h_sum
        h = np.clip(h * scale_factor, 0, 1)
    ones_minus_h = 1 - h
    corr = correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5, h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        n_points = 70
        h_init = np.zeros(n_points)
        h_init[:n_points // 2] = 1.0

    n_points = len(h_init)
    n_particles = 60
    max_iter = 1000
    w = 0.7
    c1 = 1.5
    c2 = 1.5

    # Initialize PSO swarms
    particles = [np.random.rand(n_points) * 0.2 + h_init for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest_pos = particles.copy()
    pbest_values = [evaluate_c5(p, n_points)[0] for p in particles]
    gbest_pos = particles[np.argmin(pbest_values)].copy()
    gbest_value = min(pbest_values)

    start_time = time.time()
    iter_without_improvement = 0
    max_no_improve = 100

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (pbest_pos[i] - particles[i])
                + c2 * r2 * (gbest_pos - particles[i])
            )
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)  # Normalize to satisfy integral constraint

        # Evaluate and update pbest and gbest
        for i in range(n_particles):
            c5_current, h_current = evaluate_c5(particles[i], n_points)
            if c5_current < pbest_values[i]:
                pbest_values[i] = c5_current
                pbest_pos[i] = h_current
            if c5_current < gbest_value:
                gbest_value = c5_current
                gbest_pos = h_current
                iter_without_improvement = 0
            elif time.time() - start_time > budget_s - 1.0:
                break
        iter_without_improvement += 1
        if iter_without_improvement > max_no_improve:
            break

    # Final evaluation
    c5_final, h_final = evaluate_c5(gbest_pos, n_points)
    return h_final, c5_final, n_points
