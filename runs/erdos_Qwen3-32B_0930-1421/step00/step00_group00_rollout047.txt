import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use n_points = 45 to start from the initial construction
    initial_h = np.array(initial_h_values, copy=True)
    n_points = len(initial_h)

    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    objective = cp.Minimize(1)  # Dummy objective to make setup correct
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ]
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.CBC, verbose=False)

    best_h = h.value  # Start with initial_h_values via projection
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = np.inf

    def compute_c5(h_vals):
        h_vals = np.array(h_vals)
        L = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(L) * dx

    # Cleaning up initial h if needed
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    best_c5 = compute_c5(h)

    max_iterations = 500
    it = 0
    while time.time() - start_time < budget_s - 0.1 and it < max_iterations:
        it += 1

        correlations = []
        for shift in range(2 * n_points - 1):
            shifted_h = np.roll(h, shift - n_points + 1)
            overlap = h * (1 - shifted_h)
            correlations.append(np.sum(overlap) * dx)

        k_max = np.argmax(correlations)
        k_max_overlap = correlations[k_max]

        # Form subproblem to minimize overlap at k_max
        overlap_vector = h * (1.0 - np.roll(h, k_max - n_points + 1))
        objective = cp.Minimize(k_max_overlap)
        problem = cp.Problem(objective, [cp.sum(h) == n_points / 2, h >= 0, h <= 1])
        try:
            problem.solve(solver=cp.CBC, verbose=False)
            new_h = h.value
        except:
            problem.solve(solver=cp.CVXOPT, verbose=False)
            new_h = h.value

        if new_h is None or any(np.isnan(new_h)):
            continue

        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / np.sum(new_h) * n_points / 2
        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

        h = best_h  # Update h for next iteration

    return (best_h, best_c5, n_points)
