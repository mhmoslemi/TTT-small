import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h_array, n_points):
    dx = 2.0 / n_points
    try:
        corrs = signal.correlate(h_array, 1 - h_array, mode='full')
        c5 = np.max(np.abs(corrs * dx))
    except Exception:
        c5 = float('inf')
    return c5

def run():
    start_time = time.time()
    budget_s = 1100

    # Use the provided initial construction

    n_points = len(h)
    target_sum = n_points / 2  # Constraint that ∫ h(x) dx = 1
    dx = 2.0 / n_points

    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    print(f"Initial C5 = {best_c5:.8f}")

    # Hyperparameters for gradient descent
    alpha = 0.1  # Initial step size
    beta = 0.7   # Step size reduction factor
    max_iterations = 1000  # Upper limit to avoid over/underflow
    history = []

    def project_h(h):
        # Clip between [0, 1]
        np.clip(h, 0, 1, out=h)
        # Fix the sum constraint
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-8:
            diff = target_sum - current_sum
            if abs(diff) > 0:
                n_candidates = (h < 1).sum() if diff > 0 else (h > 0).sum()
                idxs = np.where(h < 1 if diff > 0 else h > 0)[0]
                step = diff / n_candidates
                h[idxs] += step
                np.clip(h, 0, 1, out=h)
        return h

    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget reached.")
            break

        # Evaluate C5 and get max shift for gradient
        corrs = signal.correlate(h, 1 - h, mode='full')
        corrs *= dx
        max_corr_index = np.argmax(np.abs(corrs))
        k_max = max_corr_index  # Index corresponds to shift

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            # The index (i - k_max) is taken modulo n_points
            idx = (i - k_max) % n_points
            grad[i] = (1 - h[idx]) * dx  # Gradient step to reduce overlap

        # Update h
        h_proposed = h - alpha * grad
        project_h(h_proposed)

        # Evaluate new point
        c5_proposed = evaluate_erdos_solution(h_proposed, n_points)

        # Accept new point if better
        if c5_proposed < best_c5:
            best_h[:] = h_proposed[:]
            best_c5 = c5_proposed
            alpha *= 1.1  # Step size expansion on improvement
            print(f"Improvement: C5 = {best_c5:.8f} at iteration {iteration}")
        else:
            alpha *= beta  # No improvement: reduce step size
            h[:] = best_h.copy()

        # Prevent step size collapse
        if alpha < 1e-8:
            print("Alpha is too small, restarting from best solution.")
            h[:] = best_h.copy()
            alpha = 0.1
            beta = 0.7

        # Track time and break if necessary
        if len(history) >= 20 and all(abs(h - prev["h"]) < 1e-3 for prev in history[-20:]):
            print("Stall detected, terminating early.")
            break
        history.append({"h": h.copy(), "c5": c5_proposed})

    print(f"Best found C5 = {best_c5:.8f}")
    return (best_h.tolist(), best_c5, n_points)
