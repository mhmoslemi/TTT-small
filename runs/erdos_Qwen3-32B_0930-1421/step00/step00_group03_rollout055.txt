import numpy as np
import time
from math import exp

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c5 = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c5)

def project_sum_constraint(h, target_sum):
    while True:
        curr_sum = float(np.sum(h))
        if abs(curr_sum - target_sum) < 1e-4:
            break
        diff = target_sum - curr_sum
        idx = np.argsort(h)
        if diff > 0:
            n = int(min(100, len(h) // 10))
            h[idx[-n:]] += diff / n
        else:
            n = int(min(100, len(h) // 10))
            h[idx[:n]] -= diff / n
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100):
    start = time.time()
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    h = project_sum_constraint(h, target_sum)

    best_h = np.copy(h)
    best_c5 = compute_c5(h, n_points)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    max_sa_iter = 2000
    sa_iter = 0
    no_improve = 0
    patience = 200

    # GD parameters
    learning_rate = 0.01
    max_gd_iter = 500

    while time.time() - start < budget_s and sa_iter < max_sa_iter and no_improve < patience:
        # Perturb current solution
        h_perturbed = np.copy(best_h) + sigma * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = project_sum_constraint(h_perturbed, target_sum)

        # Evaluate C₅
        current_c5 = best_c5
        new_c5 = compute_c5(h_perturbed, n_points)

        # Accept or reject
        if new_c5 < current_c5 or exp((current_c5 - new_c5) / T) > np.random.rand():
            best_h[:] = h_perturbed
            best_c5 = new_c5
            no_improve = 0

        # Update temperature and step size
        T *= alpha
        sigma = sigma / (1 + sa_iter ** 0.5)  # adaptive sigma decay
        sa_iter += 1
        if sa_iter % 50 == 0:
            print(f"SA Iter {sa_iter}, Best C5: {best_c5:.7f}, Time: {time.time()-start:.1f}s")
        no_improve += 1

    # Run gradient descent if time allows
    for _ in range(max_gd_iter):
        if time.time() - start >= budget_s:
            break
        grad = np.zeros(n_points)
        for shift in range(n_points):
            h_trunc = np.roll(h=best_h, shift=shift)
            h_trunc[:shift] = 0.0
            grad += h_trunc - best_h
        grad[best_h < 1e-4] = max(grad[best_h < 1e-4]) / 100
        grad[best_h > 1 - 1e-4] = -max(grad[best_h > 1 - 1e-4]) / 100
        best_h = np.clip(best_h - learning_rate * grad, 0, 1)
        best_h = project_sum_constraint(best_h, target_sum)
        best_c5 = compute_c5(best_h, n_points)
        learning_rate *= 0.995  # decay learning rate

    print(f"Final C5 bound: {best_c5:.7f}, in {time.time()-start:.1f}s")
    return (best_h.tolist(), best_c5, n_points)
