import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    h = initial_h_values.copy()
    n_points = h.size
    dx = 2.0 / n_points

    # Constrained optimization setup
    h = cp.Parameter(n_points)
    alpha = 0.7
    h_prev = h.value.copy() if h.value is not None else np.random.rand(n_points)
    h_prev = h_prev * (n_points / 2) / np.sum(h_prev)
    h_prev = np.clip(h_prev, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    improved = True
    last_improve_time = time.time()

    while time.time() - start_time < budget_s:
        h_shifted = alpha * h.value + (1 - alpha) * h_prev
        h_cvx = cp.Variable(n_points)
        overlap = np.correlate(h_shifted, 1 - h.value, mode='full') * dx
        k_max = np.argmax(overlap)

        # Set up shifted indices
        shifted_index = (np.arange(n_points) + k_max) % n_points
        obj = cp.Minimize(cp.sum(h_cvx * (1 - cp.Parameter(n_points, value=h_shifted[shifted_index]))))
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            0 <= h_cvx,
            h_cvx <= 1,
        ]
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False)

        if result is not None and np.isfinite(result):
            h.value = h_cvx.value
            h.value = np.clip(h.value, 0, 1)
            h.value = h.value * (n_points / 2) / np.sum(h.value)

            current_overlap = np.correlate(h.value, 1 - h.value, mode='full') * dx
            current_c5 = np.max(current_overlap)

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.value.copy()
                alpha = max(alpha * 0.99, 0.1)
                last_improve_time = time.time()

            h_prev = h.value.copy()

        else:
            h.value = np.clip(h.value + 2 * np.random.rand(n_points) - 1, 0, 1)
            h.value = h.value * (n_points / 2) / np.sum(h.value)

        # Projection step to restore constraint
        h.value = h.value * (n_points / 2) / np.sum(h.value)
        h.value = np.clip(h.value, 0, 1)

        # Moderately penalize time stagnation
        if time.time() - last_improve_time > 60:
            h.value = np.random.rand(n_points) * 0.5 + 0.25
            h.value = h.value * (n_points / 2) / np.sum(h.value)
            h.value = np.clip(h.value, 0, 1)
            last_improve_time = time.time()

    # Final validation and clean up
    best_h[:] = np.clip(best_h, 0, 1)
    best_h[:] = best_h * (n_points / 2) / best_h.sum()
    validate = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(validate)

    return best_h, c5_bound, n_points
