import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h_norm = h / h.sum() * n_points / 2  # Normalize to satisfy constraint
    h = h_norm

    dx = 2.0 / n_points

    # Initialize ADMM variables
    max_lag = 2 * n_points
    z = 0.38085 * np.ones(max_lag)
    u = np.zeros_like(z)
    rho = 1.0
    max_iter = 1000
    eps_abs = 1e-4
    eps_rel = 1e-2

    start = time.time()
    for iter in range(max_iter):
        # Compute overlaps
        h_padded = np.concatenate([h, h])
        overlaps = np.zeros(max_lag)
        for k in range(max_lag):
            j_indices = np.arange(n_points)
            i_indices = (j_indices + k) % n_points
            overlaps[k] = np.dot(h, (1.0 - h_padded[i_indices + n_points // 2])) * dx

        # Update z
        for k in range(max_lag):
            residual = overlaps[k] + u[k]
            z[k] = np.clip(residual + rho * u[k], overlaps[k], float('inf'))

        # Update h
        for i in range(n_points):
            sum_z = 0
            for k in range(n_points):
                if (i + k) < n_points:
                    sum_z += (1 - h[(i + k) % n_points])
            gradient = (sum_z - 0.5) * dx / rho  # Using gradient and projection
            h[i] = np.clip(h[i] - gradient, 0, 1)

        # Project h to satisfy sum constraint
        h = h / h.sum() * n_points / 2

        # Update u
        for k in range(max_lag):
            u[k] += (overlaps[k] - z[k]) / rho

        # Check convergence
        r = overlaps - z + u
        s = rho * (z - u)
        primal_res = np.max(np.abs(r))
        dual_res = np.max(np.abs(s))
        if primal_res < eps_abs and dual_res < eps_abs:
            break
        if time.time() - start > budget_s - 0.1:
            break
        # Adjust rho
        if primal_res > 10 * dual_res:
            rho *= 2
        elif dual_res > 10 * primal_res:
            rho /= 2

    overlaps = np.zeros(n_points * 2)
    for k in range(n_points * 2):
        if k >= n_points:
            continue
        j_indices = np.arange(n_points)
        i_indices = (j_indices + k) % n_points
        overlaps[k] = np.dot(h, 1.0 - h[i_indices]) * dx

    c5_bound = np.max(overlaps)
    return (h, c5_bound, n_points)
