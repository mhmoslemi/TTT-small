import numpy as np
import cvxpy as cp
import time
from functools import partial
import scipy.signal

def compute_c5(h):
    dx = 2.0 / len(h)
    overlaps = scipy.signal.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

    h_init = np.copy(initial_guess)
    h_init /= np.sum(h_init) * 2.0 / len(h_init)
    h_init *= len(h_init) / 2.0
    return h_init
