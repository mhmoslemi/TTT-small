import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1.0 - h_values, mode="full")
    corr = corr[n_points - 1:2 * n_points - 1]  # Only symmetric lags where k ∈ [0, n_points - 1]
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    MAX_OPT_ITER = 500
    epsilon = 1e-6

    if 'initial_h_values' in globals():
        h = np.array(globals()['initial_h_values'], dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 95
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    dx = 2.0 / n_points

    for iter_idx in range(MAX_OPT_ITER):
        if time.time() - start_time > budget_s - 5:
            break

        h_var = cp.Variable(n_points)
        obj = 0.0
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]

        overlaps = np.correlate(h, 1 - h, mode="full")[n_points - 1: 2 * n_points - 1] * dx
        nu_k = np.max(overlaps)
        proxy = cp.Minimize(cp.sum(cp.log(overlaps + epsilon) * (h_var - h) ** 2))
        prob = cp.Problem(proxy, constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * (n_points / 2)
                    h_new = np.clip(h_new, 0, 1)
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                    h = h_new
            else:
                break
        except:
            break

    return (best_h.tolist(), best_c5, n_points)
