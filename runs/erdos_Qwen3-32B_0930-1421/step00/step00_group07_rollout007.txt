import numpy as np

import math
import time
import random


def grid_to_intervals(grid, n_points):
    dx = 2.0 / n_points
    intervals = []
    in_interval = False
    start = 0

    for i, val in enumerate(grid):
        x = i * dx
        if val >= 0.5:
            if not in_interval:
                start = x
                in_interval = True
        else:
            if in_interval:
                end = x
                intervals.append((start, end))
                in_interval = False
    if in_interval:
        intervals.append((start, 2.0))

    # Ensure intervals are in order and non-overlapping
    intervals.sort()
    cleaned = []
    for a, b in intervals:
        if not cleaned:
            cleaned.append([a, b])
        else:
            last_a, last_b = cleaned[-1]
            if a < last_b:
                # Overlap, merge intervals
                cleaned[-1][1] = max(last_b, b)
            else:
                cleaned.append([a, b])
    return np.array(cleaned)


def intervals_to_grid(intervals, n_points):
    grid = np.zeros(n_points)
    dx = 2.0 / n_points
    for a, b in intervals:
        start_idx = int(np.floor(a / dx))
        end_idx = int(np.ceil(b / dx))
        grid[start_idx:end_idx] = 1
    # Adjust to meet integral 1 constraint
    scale = (np.sum(grid) * 2.0 / n_points) / 1.0
    grid *= scale
    grid = np.clip(grid, 0, 1)
    return grid


def compute_c5(h):
    h_array = h.flatten()
    n_points = len(h_array)
    dx = 2.0 / n_points
    # Compute cross-correlation
    corr = np.correlate(h_array, 1 - h_array, mode="full")
    return(np.max(corr * dx))


def acceptance_probability(old_cost, new_cost, temp):
    if new_cost < old_cost:
        return 1.0
    return math.exp((old_cost - new_cost) / temp)
