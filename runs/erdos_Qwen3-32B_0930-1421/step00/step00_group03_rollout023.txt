import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

def feasibility_projection(h, total_mass):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, total_mass):
        return h
    # Adjust by scaling the largest elements if over, smallest if under
    idx = np.argsort(h)
    diff = current_sum - total_mass
    if diff > 0:  # Over the target, decrease the largest values
        adjust = idx[::-1]
        scale = diff / np.sum(h[adjust[:len(adjust)//2]])
        h[adjust[:len(adjust)//2]] *= (1 - scale)
    else:  # Under the target, increase the smallest values
        adjust = idx[:len(idx)//2]
        scale = -diff / np.sum(1 - h[adjust])
        h[adjust] += scale
    h = np.clip(h, 0, 1)
    return h

def gradient_descent(h, dx, total_mass, max_iter=500, lr=0.01, tol=1e-5):
    for _ in range(max_iter):
        grad = np.zeros_like(h)
        for k in range(-len(h)+1, len(h)-1):
            i_vals = np.arange(0, len(h))
            overlap = ((h[i_vals]) * (1 - h[i_vals - k])) * dx
            overlap_index = (len(h) - 1 + k)
            overlap = np.sum(overlap)
            grad += ((1 - h[i_vals - k]) if k > 0 else -h[i_vals - k + len(h)]) * dx
        grad = np.clip(grad, -1e5, 1e5)
        h = h - lr * grad
        h = feasibility_projection(h, total_mass)
        prev_c5 = compute_c5(h, dx)
        if _ > 0 and abs(prev_c5 - compute_c5(h, dx)) < tol:
            break
    return h

def simulated_annealing(h, dx, total_mass, max_iter=1000, initial_temp=1.0, cooling_rate=0.99, noise_scale=0.1):
    n_points = len(h)
    T = initial_temp
    h_best = h.copy()
    c5_best = compute_c5(h_best, dx)

    for i in range(max_iter):
        h_prop = h_best + noise_scale * np.random.randn(n_points)
        h_prop = feasibility_projection(h_prop, total_mass)
        c5_prop = compute_c5(h_prop, dx)

        if c5_prop < c5_best or np.random.rand() < np.exp((c5_best - c5_prop) / T):
            h_best = h_prop
            if c5_prop < c5_best:
                c5_best = c5_prop

        if i % 100 == 0:
            print(f"SA Step {i}, T={T:.3f}, best C5={c5_best:.6f}")

        T *= cooling_rate
        if T < 1e-4:
            break
    return h_best

def run(seed=42, budget_s=1100, n_points=70, initial_temp=1.0, cooling_rate=0.99, noise_scale=0.1, sa_max_iter=1000, gd_max_iter=500, lr=0.01):
    np.random.seed(seed)
    dx = 2.0 / n_points
    total_mass = n_points / 2

    # Initialize from known solution
    try:
        h = initial_h_values.copy()
    except NameError:
        # No parent, construct from scratch
        h = np.clip(np.random.rand(n_points), 0, 1)
        h = feasibility_projection(h, total_mass)

    start_time = time.time()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    while time.time() - start_time < budget_s - 20:
        # Simulated Annealing Phase
        h = simulated_annealing(h, dx, total_mass, int(sa_max_iter), initial_temp, cooling_rate, noise_scale)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Gradient Descent Phase
        h = gradient_descent(h, dx, total_mass, gd_max_iter, lr)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        if time.time() - start_time > budget_s - 5:
            break

    print(f"Final C5: {best_c5:.6f}")
    print(f"Solution sum: {np.sum(best_h)} (should be {total_mass})")
    return (best_h.tolist(), best_c5, n_points)
