import numpy as np
import cvxpy as cp
import time
import math

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h, dtype=np.float64)
    ones_minus_h = 1.0 - h
    c5 = []
    for k in range(-n_points + 1, n_points):
        corr = np.sum(h * np.roll(ones_minus_h, k))
        c5.append(corr * dx)
    return np.max(c5)

def smooth_c5(h, n_points, eta=100.0):
    h = h.reshape(-1)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    val = []
    for k in range(-n_points + 1, n_points):
        corr = np.sum(h * np.roll(ones_minus_h, k))
        val.append(corr * dx)
    log_sum_exp = math.log(np.sum(np.exp(eta * np.array(val)))) / eta
    return log_sum_exp

def perturb_h(h, noise_level=0.1):
    h = h.copy()
    h += np.random.normal(0.0, noise_level, size=len(h))
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    h = h * (h.shape[0] / 2) / h_sum
    return h

def normalize_h(h):
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    if h_sum == 0:
        h[0] = 1.0
    h = h * (h.shape[0] / 2) / h_sum
    return h
