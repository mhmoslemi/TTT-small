import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    ones_h = 1.0 - h
    corr = correlate(h, ones_h, mode="full")
    return np.max(corr) * dx

def project_onto_constraints(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to ensure sum(h) == n_points / 2
    total = h.sum()
    if total == 0:
        return np.zeros(n_points)
    scale = (n_points / 2) / total
    h *= scale
    return h

    best_h = h.copy()
    best_c5 = evaluate_C5(best_h, n_points)
    no_improvement = 0
    start_time = time.time()

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        dx = 2.0 / n_points
        ones_h = 1.0 - h
        corr = correlate(h, ones_h, mode="full")
        max_idx = np.argmax(corr)
        lag = (max_idx - n_points + 1)

        # Compute gradient: centered at the overlapping region
        grad = np.zeros(n_points)
        if lag >= 0:
            # Lag k >= 0: overlapping tail of h against head of 1-h
            grad[:n_points - lag] += 1 - h[lag:]
        else:
            # Lag k < 0: overlapping head of h against tail of 1-h
            grad[-lag:] += 1 - h[:n_points + lag]

        grad = grad * h * dx  # approximate gradient of max overlap
        h -= alpha * grad
        h = project_onto_constraints(h, n_points)

        c5 = evaluate_C5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= patience:
            alpha *= 0.5
            no_improvement = 0

    return (best_h.tolist(), best_c5, n_points)
