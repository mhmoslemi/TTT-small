import numpy as np
import cvxpy as cp
from time import time


def project_to_simplex(h, dx, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h) * dx
    if np.abs(s - 1) > 1e-6:
        h *= (1 / s)
    return h


def compute_overlap_integral(h, dx):
    correlate_result = np.correlate(h, (1 - h), mode='full')
    return np.max(correlate_result * dx)


def warm_start(h, dx):
    return project_to_simplex(h, dx, 1.0)


def run(budget_s=1100, **kwargs):
    start_time = time()
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_vals = initial_h_values
    else:
        h_vals = np.ones(n_points) / n_points
        h_vals = h_vals / np.sum(h_vals) * n_points / 2

    h_vals = warm_start(h_vals, dx)
    h = cp.Parameter(shape=n_points, value=h_vals)

    h_opt = cp.Variable(n_points)
    obj_vals = []

    # Initial solution
    c5_initial = compute_overlap_integral(h_vals, dx)
    best_h = h_vals.copy()
    best_c5 = c5_initial
    obj_vals.append(c5_initial)

    while time() - start_time < budget_s - 0.1:
        k = int(np.round(n_points / 2))
        overlap = []
        for shift in range(-n_points + 1, n_points):
            if shift < 0:
                a = h[abs(shift):]
                b = 1 - h[: len(h) - abs(shift)]
            elif shift > 0:
                a = h[: len(h) - shift]
                b = 1 - h[shift:]
            else:
                a = h
                b = 1 - h
            overlap.append(cp.matmul(a, b))

        theta = cp.Variable()
        objective = cp.Minimize(theta)
        constraints = [
            theta >= ov * dx for ov in overlap
        ] + [
            h_opt == project_to_simplex(h_opt, dx, 1.0),
            0 <= h_opt,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2,
        ]

        cp.Problem(objective, constraints).solve(solver=cp.CVXOPT)
        h_vals = h_opt.value
        new_c5 = compute_overlap_integral(h_vals, dx)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_vals.copy()

        obj_vals.append(new_c5)

    return best_h, best_c5, n_points
