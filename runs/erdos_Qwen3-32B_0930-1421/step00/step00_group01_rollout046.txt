import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_objective(h_vec, n_points):
    dx = 2.0 / n_points
    h = h_vec.reshape(n_points)
    ones_h = 1.0 - h
    corr = correlate(h, ones_h, mode='full')
    corr = corr * dx  # Scale the correlation
    return np.max(corr)

def smooth_max(corr, eta=10.0):
    # Smooth the max with softmax-like log-sum-exp
    max_val = np.max(corr)
    corr_centered = corr - max_val
    return max_val + (1 / eta) * np.log(np.sum(np.exp(eta * corr_centered)))

def smooth_objective(h_vec, n_points, eta=10.0):
    dx = 2.0 / n_points
    h = h_vec.reshape(n_points)
    ones_h = 1.0 - h
    corr = correlate(h, ones_h, mode='full') * dx  # Apply dx immediately
    return smooth_max(corr, eta)

def objective_with_constraints(h_vec, n_points):
    return evaluate_objective(h_vec, n_points)

def generate_new_h(h_seed, n_points):
    h = h_seed.copy()
    idxs = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
    h[idxs] += (np.random.rand(len(idxs)) * 0.2 - 0.1)  # perturb up to 0.1
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2) * 2.0 / n_points  # renormalize
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # fixed to n=47 for compatibility with `initial_h_values`

    if 'initial_h_values' not in globals():
        # fallback to random initialization
        h_vec = np.random.rand(n_points)
        h_vec = h_vec / h_vec.sum() * (n_points / 2) * 2.0 / n_points
    else:
        h_vec = initial_h_values.reshape(n_points).copy()

    best_h = h_vec.copy()
    best_c5 = evaluate_objective(best_h, n_points)
    current_h = h_vec.copy()
    current_c5 = best_c5

    # Initial population with various perturbations
    candidates = [current_h]
    for _ in range(5):
        perturbed = generate_new_h(best_h, n_points)
        candidates.append(perturbed)

    for cand in candidates:
        h0 = cand.flatten()
        h0 = h0 / h0.sum() * (n_points / 2) * 2.0 / n_points

        # Perturbed constraints
        def constraint_func(x):
            return np.sum(x) - n_points / 2 * 2.0 / n_points

        constraints = [{'type': 'eq', 'fun': constraint_func}]
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            # Gradient-based optimization (SLSQP local)
            result = minimize(
                smooth_objective,
                x0=h0,
                args=(n_points, 10.0),
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 100}
            )

            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h = new_h / new_h.sum() * (n_points / 2) * 2.0 / n_points
                new_c5 = evaluate_objective(new_h, n_points)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except:
            pass  # silence optimization failure

        if time.time() - start_time > budget_s - 30:
            break  # leave 30 seconds for final verification

    # Final evaluation and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2) * 2.0 / n_points
    best_c5 = evaluate_objective(best_h, n_points)

    return (best_h, best_c5, n_points)
