import numpy as np
import scipy.signal
import time
import cvxpy as cp
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    def evaluate(h, n_points):
        dx = 2.0 / n_points
        h = h.reshape(-1)
        diff = 1 - h
        conv = scipy.signal.correlate(h, diff, mode="full")
        return np.max(conv) * dx

    def project(h, total_mass):
        h_clipped = np.clip(h, 0, 1)
        mass = np.sum(h_clipped)
        if abs(mass) < 1e-8:
            return h_clipped
        scale_factor = total_mass / mass
        return np.clip(h_clipped * scale_factor, 0, 1)

    def optimize_gd(h_init, n_points, total_mass=1.0, iterations=200, alpha=0.01, restart_period=40):
        h_best = h_init.copy()
        c5_best = evaluate(h_best, n_points)

        h = h_init.copy()
        best_on_path = h_best.copy()
        time_start = time.time()
        for iter_count in range(iterations):
            conv = scipy.signal.correlate(h, 1 - h, mode="full")
            indices = np.argmax(conv)
            delay = indices - n_points + 1
            if delay <= n_points:
                delayed_h = np.roll(h, shift=delay)
                delayed_h[:delay] = 0
            else:
                delayed_h = np.roll(h, shift=delay)
                delayed_h[abs(delay):] = 0

            grad = (1 - h - delayed_h) * 2 * h * (1 - h)

            h_flat = h.flatten()
            grad_flat = grad.flatten()

            h_flat -= alpha * grad_flat
            h = project(h_flat, total_mass)
            c5 = evaluate(h, n_points)

            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
                best_on_path = h_best.copy()

            if iter_count % restart_period == 0:
                h = best_on_path.copy()
                alpha *= 0.5
                if alpha < 1e-6:
                    break

            if time.time() - time_start > budget_s * 0.9:
                break

        return h_best, c5_best

    def simplex_search(h_center, n_points, total_mass=1.0, simplex_size=0.1, steps=20):
        h_center = h_center.copy()
        simplex = [h_center + simplex_size * np.random.randn(n_points),
                   h_center - simplex_size * np.random.randn(n_points),
                   h_center.copy()]
        simplex = [project(s, total_mass) for s in simplex]
        scores = [evaluate(s, n_points) for s in simplex]
        best_idx = np.argmin(scores)

        for step in range(steps):
            # Reflect
            centroid = np.mean([simplex[i] for i in range(len(simplex)) if i != best_idx], axis=0)
            reflected = centroid + (centroid - simplex[best_idx])
            reflected = project(reflected, total_mass)
            score_reflect = evaluate(reflected, n_points)
            if score_reflect < scores[best_idx]:
                simplex[best_idx] = reflected
                scores[best_idx] = score_reflect
                best_idx = np.argmin(scores)
            else:
                # Contract if reflection fails
                contracted = centroid + 0.5 * (simplex[best_idx] - centroid)
                contracted = project(contracted, total_mass)
                score_contract = evaluate(contracted, n_points)
                if score_contract < scores[best_idx]:
                    simplex[best_idx] = contracted
                    scores[best_idx] = score_contract
                    best_idx = np.argmin(scores)
                else:
                    # Shrink if contraction fails
                    for i in range(len(simplex)):
                        simplex[i] = 0.5 * (simplex[i] + simplex[best_idx])
                        simplex[i] = project(simplex[i], total_mass)
                        scores[i] = evaluate(simplex[i], n_points)
                    best_idx = np.argmin(scores)

        return simplex[best_idx], min(scores)

    def main():
        best_h = None
        best_c5 = np.inf
        n_points = 84

        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            h[:n_points // 2] = 1
            h = project(h, total_mass=1.0)

        time_start = time.time()
        while time.time() - time_start < budget_s:

            h_gd, c5_gd = optimize_gd(h.copy(), n_points=n_points, iterations=200, alpha=0.1)
            h_simplex, c5_simplex = simplex_search(h_gd.copy(), n_points=n_points)

            c5_total = min(c5_gd, c5_simplex)
            if c5_total < best_c5:
                best_c5 = c5_total
                if c5_gd < c5_simplex:
                    best_h = h_gd.copy()
                else:
                    best_h = h_simplex.copy()

            if time.time() - time_start > budget_s * 0.995:
                break

            # Next iteration with small random move
            h_candidate = best_h.copy()
            h_candidate += 0.1 * np.random.randn(n_points)
            h = project(h_candidate, total_mass=1.0)

        return best_h, best_c5, n_points

    best_h, best_c5, n_points = main()
    return (best_h, best_c5, n_points)
