import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    best_h = np.copy(h)
    best_c5 = float('inf')

    def evaluate(h, dx):
        h = np.clip(h, 0, 1)
        h_sum = h.sum() * dx
        scale = 1.0 / h_sum
        h = np.clip(h * scale, 0, 1)
        L = np.correlate(h, 1 - h, mode='full') * dx
        return L

    # Initial evaluation
    L = evaluate(h, dx)
    best_c5 = L[max(L)]
    best_c5_idx = np.argmax(L)

    max_iter = 1000
    for iter_idx in range(max_iter):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 0:
            break

        # Warm start
        k_max = best_c5_idx

        # Convexification trick with fixed k
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(h_var @ (1 - np.roll(h, k_max)) * dx)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        prob = cp.Problem(obj, constraints)
        solver = cp.OSQP() if 'GUROBI' in cp.installed_solvers() else cp.OSQP
        prob.solve(solver=solver, verbose=False)
        if prob.status == 'optimal':
            h_new = np.clip(prob.value, 0, 1)
            h_sum = h_new.sum() * dx
            h_new = np.clip(h_new * (1.0 / h_sum), 0, 1)
            h = h_new
            L = evaluate(h, dx)
            current_c5 = L[max(L)]
            if current_c5 < best_c5 - 1e-4:
                best_h[:] = h
                best_c5 = current_c5
                best_c5_idx = np.argmax(L)
        else:
            h[:] = best_h

    h_sum = best_h.sum()
    scale = n_points / (2 * h_sum)
    best_h *= scale
    best_h = np.clip(best_h, 0, 1)
    h_sum = best_h.sum()
    assert abs(h_sum - n_points / 2) < 2e-4, f"Feasibility failed: sum(h) = {h_sum:.6f}"

    L = evaluate(best_h, dx)
    c5_bound = L[max(L)]
    return best_h.tolist(), c5_bound, n_points
