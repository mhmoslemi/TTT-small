import numpy as np
import time
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)

    # Normalize h to exactly satisfy the integral constraint
    dx = 2.0 / n_points
    expected_sum = n_points / 2.0
    h = h / h.sum() * expected_sum

    # Optimization parameters
    alpha = 0.01
    max_iter = 5000
    patience = 200
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    def evaluate_c5(_h):
        corr = correlate(_h, 1 - _h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(_h):
        corr = correlate(_h, 1 - _h, mode='full')
        max_index = np.argmax(corr)
        pad_h = np.concatenate([np.zeros_like(_h), _h, np.zeros_like(_h)])
        shifted_1h = 1 - np.roll(_h, shift=1)  # Avoid shifting 1-h directly

        grad = np.zeros_like(_h)
        for i in range(n_points):
            for k in range(2 * n_points + 1):
                if k + i < n_points:
                    if k < n_points:
                        grad[i] += (1 - _h[k]) * (1 if (k + i) == max_index else 0)
                    else:
                        grad[i] += _h[k - n_points] * (1 if (k + i) == max_index else 0)

        return grad * dx * 2  # Heuristic scaling to strengthen updates

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        grad = compute_gradient(h)
        h -= alpha * grad

        # Projection: Clip and normalize
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * expected_sum

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif iter % patience == 0:
            alpha *= 0.5  # Reduce learning rate

    return best_h.astype(np.float64).tolist(), best_c5, n_points
