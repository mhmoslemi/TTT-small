import numpy as np
import time
import math
from itertools import combinations
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def get_distance_distribution(x, k, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(x, 1 - x, mode='full')
    corr *= dx
    return corr[k]

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum_constraint(h):
    s = np.sum(h)
    n = len(h)
    total = n / 2.0
    if s == total:
        return h
    diff = total - s
    while abs(diff) > 1e-8:
        if diff > 0:
            idx = np.argmin(h)
            setattr(repair_sum_constraint, 'idx_count', (getattr(repair_sum_constraint, 'idx_count', 0) + 1) % n)
            if idx == h.shape[0]:
                idx = 0
            max_add = min(1.0 - h[idx], diff)
            h[idx] += max_add
            diff -= max_add
        else:
            idx = np.argmax(h)
            max_subtract = min(h[idx], -diff)
            h[idx] -= max_subtract
            diff += max_subtract
    return h

def get_initial_population(n_points, pop_size, initial_h):
    pop = []
    for _ in range(pop_size):
        h = np.copy(initial_h)
        noise = np.random.randn(n_points)
        noise = (noise - np.min(noise)) / (np.max(noise) - np.min(noise)) * 2 * 1e-5
        h = np.clip(h + noise, 0, 1)
        h = repair_sum_constraint(h)
        pop.append(h)
    return np.array(pop)

def tournament_selection(pop, fitness, tournament_size=3):
    indices = np.random.choice(len(pop), tournament_size)
    fitnesses = [fitness[i] for i in indices]
    return pop[indices[np.argmin(fitnesses)]]

def crossover(parent1, parent2):
    n_points = len(parent1)
    mask = np.random.rand(n_points) < 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    child1 = repair_sum_constraint(child1)
    child2 = repair_sum_constraint(child2)
    return child1, child2

def mutate(child, mutation_rate=0.05):
    n_points = len(child)
    mask = np.random.rand(n_points) < mutation_rate
    perturb = np.random.rand(n_points) * 0.2 - 0.1
    child_new = np.clip(child + mask * perturb, 0, 1)
    child_new = repair_sum_constraint(child_new)
    return child_new

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    pop_size = 30
    max_generations = 2000
    population = get_initial_population(n_points, pop_size, initial_h)
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = evaluate_c5(best_h, n_points)
    
    for gen in range(max_generations):
        current_time = time.time()
        if current_time - start_time > budget_s - 10:  # Allow 10 sec early exit
            break
            
        # Evaluate
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, n_points)
            fitness.append(c5)
        
        # Track best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]
        
        # Selection and reproduction
        new_population = [best_h]
        for _ in range((pop_size - 1) // 2):
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            c1, c2 = crossover(p1, p2)
            new_population.extend([mutate(c1), mutate(c2)])
        
        new_population = new_population[:pop_size]
        population = np.array(new_population)
    
    return best_h, best_c5, n_points
