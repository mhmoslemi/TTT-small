import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed
    np.random.seed(seed)

    # Load initial values
    global initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
    assert np.abs(np.sum(h) - target_sum) < 1e-6, "Initial h must sum to n_points/2"

    # Avoid divergence by tracking best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Optimization parameters
    start_time = time.time()
    max_iter = 1000
    alpha = 1.0  # Step size multiplier
    beta = 0.9   # Backtracking step decay
    min_alpha = 1e-8

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute overlap across shifts
        h1 = h
        h2 = 1.0 - h
        corr = scipy.signal.correlate(h1, h2, mode="full") * dx

        # Find the shift that gives the maximum overlap
        k = np.argmax(abs(corr))

        # Compute indices involved
        n = len(h)
        overlap_start = max(0, k - n + 1)
        overlap_end = min(n, k + 1)

        # Build the gradient only for indices with overlap
        gradient = np.zeros(n)
        if k <= n - 1:
            gradient[:n - k] = (1.0 - h[k:n])
        if k > 0:
            gradient[k - 1:] = h[:-k + 1]

        # Compute the step size via backtracking line search
        # Start with full step size
        current_alpha = alpha
        current_h = h.copy()

        # Try step size until improvement is found
        while current_alpha > min_alpha:
            new_h = current_h - current_alpha * gradient
            new_h = np.clip(new_h, 0, 1)  # Ensure bounds

            # Maintain sum constraint
            delta_sum = np.sum(new_h) - target_sum
            if np.abs(delta_sum) > 1e-9:
                new_h -= delta_sum / n

            # Evaluate new solution
            new_corr = scipy.signal.correlate(new_h, 1.0 - new_h, mode="full") * dx
            new_c5 = np.max(np.abs(new_corr))

            # Accept improvement
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                h = best_h
                break

            # Try smaller step
            current_alpha *= beta

        # For next iteration, use current h
        if np.allclose(h, best_h, atol=1e-8):
            break  # No more improvement, exit early

    # Final check of constraints
    best_h = np.clip(best_h, 0, 1)
    delta_sum = np.sum(best_h) - target_sum
    if abs(delta_sum) > 1e-9:
        best_h -= delta_sum / n

    # Return the best solution
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, n_points)
