import time
import numpy as np
from scipy import optimize
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        return evaluate_erdos_solution(h)

    def fitness(h_vec):
        h = h_vec.reshape(n_points)
        sum_h = h.sum()
        penalty = 1000 * abs(sum_h - n_points / 2)
        if np.any(h > 1) or np.any(h < 0):
            return 1e6
        c5 = compute_c5(h)
        return c5 + penalty

    def clip_and_project(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - n_points / 2) > 1e-8:
            factor = (n_points / 2) / s
            h = h * factor
        return h

    # Perturb initial solution to form initial population
    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    for i in range(1, pop_size):
        pop[i] += 0.1 * np.random.randn(n_points)
        pop[i] = clip_and_project(pop[i])

    best_idx = 0
    best_c5 = fitness(pop[best_idx])
    best_h = pop[best_idx].copy()

    max_gen = 200
    for gen in range(max_gen):
        if time.time() - t0 > budget_s * 0.9:
            break
        # Evaluate fitness
        fitness_values = [fitness(p) for p in pop]
        # Select top 20% as parents
        sorted_indices = np.argsort(fitness_values)
        top_pct = int(pop_size * 0.2)
        parents = [pop[i] for i in sorted_indices[:top_pct]]
        # Generate offspring
        new_pop = parents.copy()
        while len(new_pop) < pop_size:
            a, b = np.random.choice(len(parents), 2, replace=False)
            pa, pb = parents[a], parents[b]
            alpha = (np.random.rand() + 0.5)  # Ensure diversity
            child = alpha * pa + (1 - alpha) * pb
            # Mutate
            child += 0.1 * np.random.randn(n_points)
            child = clip_and_project(child)
            new_pop.append(child)
        pop = new_pop
        # Update best
        current_best_idx = np.argmin(fitness_values)
        current_best_c5 = fitness_values[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = pop[current_best_idx].copy()

    # Final local refinement
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        return compute_c5(h)
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{"type": "eq", "fun": lambda x: x.sum() - n_points / 2}]
    def proj(x):
        x = x.copy().reshape(n_points)
        return clip_and_project(x).ravel()
    result = optimize.minimize(
        objective,
        best_h.ravel(),
        method="L-BFGS-B",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 20, "ftol": 1e-8},
        jac=None
    )
    final_h = clip_and_project(result.x.reshape(n_points))

    final_c5 = evaluate_erdos_solution(final_h)

    return final_h, final_c5, n_points
