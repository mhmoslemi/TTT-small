import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 45
    h = np.array(initial_h_values)
    n_points = len(h)
    ds = int(0.9 * budget_s)

    def evaluate(h, dx):
        # Evaluate the C5 bound using cross-correlation
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    best_val = evaluate(h, dx)
    best_h = h.copy()

    start = time.time()

    # Phase 1: Convex relaxation in frequency domain
    H = cp.Variable(n_points, nonneg=True, nonpos=1.0)  # Refine this as needed
    objective = cp.Minimize(cp.norm(H, np.inf))  # Surrogate frequency domain objective
    constraints = [
        cp.sum(H) * dx == 1,  # Volume constraint
        H >= 0, H <= 1
    ]
    prob = cp.Problem(objective, constraints)

    # Using ECOS as a lightweight convex solver
    for _ in range(50):
        if time.time() - start > ds:
            break
        try:
            H.value = best_h.copy()
            prob.solve(solver=cp.ECOS, verbose=False)
            h_update = H.value
            if np.isfinite(h_update).all():
                # Normalize back to vol constraint
                h_update *= (0.5 * n_points) / np.sum(h_update)
                h_update = np.clip(h_update, 0, 1)
                val = evaluate(h_update, dx)
                if val < best_val:
                    best_val = val
                    best_h[:] = h_update
        except:
            pass

    # Phase 2: Gradient-based refinement
    h = best_h.copy()
    learning_rate = 0.01
    for _ in range(200):
        if time.time() - start > budget_s - 50:
            break
        dh = 0.0
        corr = correlate(h, 1 - h, mode="full") * dx
        max_pos = np.argmax(corr)
        max_val = corr[max_pos]
        if abs(max_val) < 1e-10:
            continue
        # "Push down" where corr is positive (overlap is too much)
        width = 5  # influence window
        for offset in range(-width, width + 1):
            idx = (max_pos + offset) % n_points
            dh += h[idx]
        dh = h - dh / (2 * width + 1)
        h = np.clip(h - learning_rate * dh, 0, 1)
        h *= (0.5 * n_points) / np.sum(h)
        val = evaluate(h, dx)
        if val < best_val:
            best_val = val
            best_h[:] = h

    # Return the best feasible solution
    return best_h.tolist(), float(best_val), n_points
