import numpy as np
import cvxpy as cp
import time

import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Extract initial_h_values with n=70 samples
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_c5(h):
        # Normalize to [0, 2]
        h_interp = np.interp(np.linspace(0, 2, 4 * n_points, endpoint=False), np.linspace(0, 2, n_points), h)
        h_flip = 1 - h_interp
        conv = np.correlate(h_interp, h_flip, mode="same")
        c5 = np.max(conv * dx)
        return c5

    # Binary search for optimal C₅
    low, high = 0.38085, 0.4
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    timespent = 0

    while timespent < budget_s:
        # Midpoint of current binary search
        mid = (low + high) / 2

        # CVXPy optimization problem
        h = cp.Variable(n_points)
        c5 = cp.Variable()

        objective = cp.Minimize(c5)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2,
            c5 >= 0
        ]

        for k in range(1, n_points):
            # Compute correlation at lag k
            h_pad = np.zeros(n_points * 2)
            h_pad[:n_points] = h
            h_flip = 1 - cp.vstack([cp.roll(h_pad, shift=i) for i in range(k, k + n_points - 1)])

            integral = cp.sum([h[i] * (1 - h_pad[k + i]) for i in range(n_points)])
            constraints.append(integral <= mid)

        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(solver=cp.MOSEK, verbose=False)
            if problem.status == "optimal":
                h_opt = np.clip(h.value, 0, 1)
                c5_opt = compute_c5(h_opt)

                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    print(f"New best C₅ = {best_c5:.6f}")
                    low = mid
                else:
                    high = mid
            else:
                high = mid
        except:
            high = mid

        timespent = time.time() - start_time
        if high - low < 1e-6:
            break

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
