import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run():
    start_time = time.time()
    budget_s = 1100

    # Initial setup
    h0 = initial_h_values.astype(float)
    n_points = len(h0)
    dx = 2.0 / n_points

    def evaluate(h):
        """Evaluate overlap integral and return C₅ (max overlap)."""
        h = np.clip(h, 0.0, 1.0)
        h_expanded = np.concatenate([h, [0]] * n_points)
        ones_minus_h = 1 - h

        corrs = np.full(2 * n_points, np.nan)
        for k in range(n_points, 2 * n_points):
            corrs[k] = np.dot(h_expanded[:k], ones_minus_h[(2 * n_points - k):])
        return np.nanmax(corrs), {0}

    def penalty(h):
        """Penalize infeasible solutions."""
        total = np.sum(h)
        area = total * dx
        integral_penalty = abs(area - 1.0)
        bound = max(max(h), 1.0 - min(h))
        penalization = integral_penalty + 0.1 * max(bound - 1.0, 0.0)
        return penalization * 100.0

    def fitness(h):
        """Fitness is negative of objective with border penalties."""
        c5, _ = evaluate(h)
        return -(c5 + penalty(h))

    def project_to_feasible(h):
        """Project back to feasible region (sum = n_points / 2)."""
        h_clipped = np.clip(h, 0.0, 1.0)
        rescale = h_clipped * n_points / (2 * np.sum(h_clipped))
        return np.clip(rescale, 0.0, 1.0)

    def mutate(parent, sigma=0.1):
        """Generate a perturbed offspring by adding Gaussian noise."""
        offspring = parent + np.random.normal(0, sigma, n_points)
        return project_to_feasible(offspring)

    def crossover(parent1, parent2, alpha=0.5):
        """Blend two parents into one offspring."""
        child = alpha * parent1 + (1 - alpha) * parent2
        return project_to_feasible(child)

    def evaluate_population(population):
        """Return list of (fitness, c5, idx) for individuals in population."""
        results = []
        for i, h in enumerate(population):
            c5, _ = evaluate(h)
            results.append((c5, i))
        return results

    def ga_optimization(population, generations=200):
        """Run genetic algorithm with crossover and mutation."""
        best_c5 = float('inf')
        best_h = population[0].copy()

        for gen in range(generations):
            # Evaluate current population
            eval_results = evaluate_population(population)
            c5_list = [r[0] for r in eval_results]
            best_idx = np.argmin(c5_list)
            best_c5_gen = c5_list[best_idx]

            if best_c5_gen < best_c5:
                best_c5 = best_c5_gen
                best_h = population[best_idx].copy()

            # Sort by fitness (lowest C₅)
            eval_results.sort()
            _, best_indices = zip(*eval_results)

            # Top 20% elitism
            elite_size = int(0.2 * len(population))
            elites = [population[i] for i in best_indices[:elite_size]]

            # Generate offspring
            offspring = []
            for _ in range(len(population) - elite_size):
                # Select parents and breed
                parent1 = elites[np.random.randint(len(elites))]
                parent2 = elites[np.random.randint(len(elites))]
                if np.random.rand() < 0.5:
                    child = mutate(parent1)
                else:
                    child = crossover(parent1, parent2)
                offspring.append(child)

            # Recombine
            population = elites + offspring

            # Timeout check
            if time.time() - start_time > budget_s * 0.9:
                break

        return best_h, best_c5

    # Initialize population: start from parent and add mutants
    population_size = 50
    population = [project_to_feasible(initial_h_values + 0.1 * np.random.rand(n_points)) for _ in range(population_size)]

    # Run optimization
    best_h, best_c5 = ga_optimization(population)

    # Post-processing: gradient descent
    def obj(h_vec):
        h = h_vec.reshape(-1)
        c5, _ = evaluate(h)
        return c5

    def bounds_func(h_vec):
        h = h_vec.reshape(-1)
        return np.array([0] * len(h) + [1] * len(h) + [np.sum(h) - n_points / 2.0])

    # Naive constraints: all in [0, 1], sum is n_points / 2
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8)

    optimized_h = project_to_feasible(result.x)
    final_c5, _ = evaluate(optimized_h)

    elapsed = time.time() - start_time
    if time.time() - start_time > budget_s:
        print("Warning: small time buffer left")
    elif final_c5 > 0.38085:
        print("Solution did not meet the bound")

    return optimized_h, final_c5, n_points
