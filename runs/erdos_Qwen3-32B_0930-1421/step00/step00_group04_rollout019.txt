import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        h = initial_h_values.copy()
    else:
        n_points = 42
        h = np.ones(n_points) / n_points * (n_points / 2)

    # Normalize to satisfy integral constraint
    if not np.isclose(np.sum(h), n_points / 2):
        h = h / np.sum(h) * (n_points / 2)

    # Global parameters
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')

    # Projection to feasible region [0, 1] and volume constraint
    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_clipped)
        return h_clipped * scale

    def evaluate_c5(h):
        h = h.reshape(-1)  # Ensure flat
        one_m_h = 1 - h
        corr = np.correlate(h, one_m_h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Optimization loop
    max_steps = 100000
    alpha = 0.005
    improvement_window = 100
    no_improvement_count = 0
    step = 0

    while time.time() - start_time < budget_s:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            alpha *= 1.05  # encourage exploration
        else:
            no_improvement_count += 1
            alpha *= 0.97  # reduce step if stuck

        if no_improvement_count >= improvement_window:
            # Heuristic: perturb to escape local minima
            noise = np.random.randn(n_points) * 0.05
            h += noise / np.linalg.norm(noise, ord=np.inf) * (n_points / 10)
            h = project_h(h)
            no_improvement_count = 0

        # Gradient step
        h_diff = np.correlate(h, -h, mode='full')  # Improved gradient approximation
        grad = np.gradient(c5, h) + dx * h_diff
        alpha_clamped = np.clip(alpha, 1e-5, 1e-1)
        h -= alpha_clamped * grad
        h = project_h(h)

        step += 1

    # Final projection and validation
    h = project_h(h)
    h_counts = np.bincount(h.astype(np.intc), minlength=n_points)
    final_c5 = evaluate_c5(h)

    return h, final_c5, n_points
