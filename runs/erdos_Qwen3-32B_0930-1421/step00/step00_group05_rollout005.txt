import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution


def run():
    import time

    n_points = 70  # Start from existing parent solution

    h0 = initial_h_values
    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    h_sum = cp.sum(h)
    c5_expr = cp.Max(
        [cp.sum(cp.multiply(h, 1 - np.roll(h, k))) * dx for k in range(1, n_points)]
    )

    constraints = [
        h <= 1.0,
        h_sum == n_points / 2.0,  # Constraint from uniform measure
    ]

    # Use warm start
    prob = cp.Problem(cp.Minimize(c5_expr), constraints)
    result = prob.solve(warm_start=True, solver="GUROBI", verbose=False, TimeLimit=1050)
    h_values = h.value

    if h_values is None:
        return h0, 1.0, n_points

    # Ensure solution is valid and masked to [0, 1]
    h_values = np.round(np.clip(h_values, 0.0, 1.0), 10)

    start_time = time.time()
    elapsed_time = 0
    budget_s = 1100

    # Local refinement: try small amplitude perturbations in wavelet-like fashion
    for _ in range(10):
        if elapsed_time >= budget_s - 5:
            break

        idx = np.random.choice(n_points, size=5)
        h_values[idx] += np.random.uniform(-0.05, 0.05, len(idx))
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = np.round(h_values, 10)

        # Enforce integral-1 constraint
        S = np.sum(h_values)
        h_values *= n_points / 2.0 / S

        elapsed_time = time.time() - start_time

    # Evaluate final result
    best_h = h_values
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    return best_h, best_c5, n_points
