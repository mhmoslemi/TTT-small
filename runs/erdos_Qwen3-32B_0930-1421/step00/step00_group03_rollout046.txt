import numpy as np
import time
from numpy.random import uniform, rand, choice

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    max_c5 = 0.0
    for k in range(1, n_points):
        cross_correlation = np.correlate(h_values, 1 - h_values, mode="full")
        current_c5 = cross_correlation[k] * dx
        if current_c5 > max_c5:
            max_c5 = current_c5
    return max_c5

def ensure_feasible(h, n_points):
    # Ensure bounds and integral constraint
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-8:
        return h
    diff = target_sum - current_sum
    indices = np.argsort(h)
    if diff > 0:
        # Increase the smallest values to meet the constraint
        n = len(h)
        for i in indices[:n]:
            change = max(0.0, 1.0 - h[i])
            if diff <= change:
                h[i] += diff
                break
            h[i] = 1.0
            diff -= change
    else:
        # Decrease the largest values to meet the constraint
        n = len(h)
        indices = indices[::-1]
        for i in indices[:n]:
            change = max(0.0, h[i])
            if abs(diff) <= change:
                h[i] += diff
                break
            h[i] = 0.0
            diff += change
    return h
