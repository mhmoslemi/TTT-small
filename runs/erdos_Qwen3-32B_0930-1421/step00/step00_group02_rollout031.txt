import numpy as np
import cvxpy as cp
import time
import scipy.signal


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    np.random.seed(seed)

    # Use the initial_h_values from the harness if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        current_c5 = evaluate_erdos_solution(h)
    else:
        # Fallback initialization
        n_points = 45
        h = np.random.rand(n_points)
        h /= h.sum() * 2  # Scale so sum(h) dx == 1

    # Compute dx
    dx = 2.0 / n_points

    # Projection function to maintain constraints
    def project(h):
        h[h < 0] = 0.0
        h[h > 1] = 1.0
        total_mass = h.sum()
        scaling = n_points / 2 / total_mass
        h *= scaling
        return h

    # Convolution for computing overlaps
    def compute_c5(h):
        conv = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5

    best_h = np.copy(h)
    best_c5 = compute_c5(h)

    while time.time() - start_time < budget_s:
        all_overlaps = scipy.signal.correlate(h, 1 - h, mode='full')

        # Determine k where overlap is maximal
        k_star = int(np.argmax(all_overlaps))
        # Compute overlap at k_star
        cor_kstar = all_overlaps[k_star]
        # Indices overlapping for k_star
        ii = np.arange(max(0, k_star - 128), min(n_points - k_star, 128 + 1))
        i_vals = ii
        j_vals = (i_vals + k_star) % n_points  # periodic shift (for function wrapping)

        # Gradient at h for k_star
        grad_h = np.zeros_like(h)
        grad_h[i_vals] += (1 - h[j_vals])  # derivative of h[i]*(1-h[j]) w.r.t h[i]
        grad_h[j_vals] -= h[i_vals]  # derivative of h[i]*(1-h[j]) w.r.t h[j]

        # Normalize gradient so it doesn't overstep the box constraints
        grad_h -= grad_h.min()  # shift gradient to non-negative

        # Constrained optimization step
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_var - (h - 0.02 * grad_h)))
        constraints = [
            h_var >= 0, h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS)
            h = h_var.value
        except:
            pass

        if h is not None:
            h = np.clip(h, 0, 1)
            h = project(h)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = np.copy(h)

    return best_h, best_c5, n_points
