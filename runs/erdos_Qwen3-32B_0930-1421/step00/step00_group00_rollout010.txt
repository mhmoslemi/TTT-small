import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_c5(h):
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def project_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    target_sum = len(h) / 2.0
    if current_sum < target_sum:
        excess = target_sum - current_sum
        indices = np.where(h < 1.0)[0]
        h[indices] += excess / len(indices)
    elif current_sum > target_sum:
        excess = current_sum - target_sum
        indices = np.where(h > 0.0)[0]
        h[indices] -= excess / len(indices)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        h = np.array(kwargs['initial_h_values'], dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 90
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / h.sum() * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_erdos_c5(best_h)
    start_time = time.time()
    max_iter = 200
    alpha = 0.1
    beta = 0.7

    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        overlaps = np.correlate(h, 1 - h, mode="full")
        max_idx = np argmax(overlaps)
        k_max = max_idx - (len(h) - 1)  # Adjust shift index

        if k_max <= -len(h) + 1 or k_max >= len(h) - 1:
            continue  # Skip invalid shift

        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i - k_max) % len(h)
            gradient[i] = (1 - h[j])
        gradient *= (2.0 / len(h))

        # Projected descent
        h_proposed = h - alpha * gradient
        h_proposed = project_feasible(h_proposed)
        
        c5_proposed = evaluate_erdos_c5(h_proposed)
        if c5_proposed < best_c5:
            best_h = h_proposed.copy()
            best_c5 = c5_proposed
            h = h_proposed.copy()
            alpha = max(alpha * 1.1, 0.01)
        else:
            alpha *= beta

    # Return h_values, best_c5, n_points
    return (best_h.tolist(), best_c5, n_points)
