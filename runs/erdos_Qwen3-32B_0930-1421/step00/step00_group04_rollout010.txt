import numpy as np
import scipy.fft as fft
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(42, 1.0 / 42) * 42
    n_points = len(initial_h)

    dx = 2.0 / n_points
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)

    start_time = time.time()
    max_iter = 1000
    alpha = 0.1
    threshold = 0.38085
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    def evaluate_c5(h):
        h_1_minus_h = np.convolve(h, 1 - h, mode="full")
        return np.max(h_1_minus_h * dx)

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        if i % 10 == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            print(f"Iter {i}, c5 = {current_c5:.7f}")

        H = fft.fft(h)
        G = fft.fft(1 - h)
        overlap_freq = H * np.conj(G)
        overlap = fft.ifft(overlap_freq).real
        k_max = np.argmax(overlap)

        # Gradient update
        shift = int((h.size / 2 + k_max) % h.size)
        h_new = np.roll(h, -shift)
        delta = h - h_new
        h = h - alpha * delta

        # Projection to [0, 1]
        h = np.clip(h, 0, 1)

        # Rescale to match ∑h = n_points / 2
        h = h / h.sum() * (n_points / 2)

    best_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
