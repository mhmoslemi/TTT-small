import numpy as np
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    eval_mode = "full"

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        target = n_points / 2
        if abs(s - target) > 1e-8:
            diff = (target - s) / n_points
            h += diff
            h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        h = h.flatten()
        corr = np.correlate(h, 1 - h, mode=eval_mode)
        return np.max(corr) * dx

    best_h = project_to_feasible(h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    h = best_h.copy()
    c5 = best_c5
    T = 1.0
    alpha = 0.99
    max_sa_steps = 1000
    max_gd_steps = 500
    sigma = 0.1
    eta = 0.01
    convergence_threshold = 1e-5

    while time.time() - start_time < budget_s:
        # Simulated Annealing Phase
        for sa_step in range(max_sa_steps):
            if time.time() - start_time > budget_s:
                break

            h_proposal = h + sigma * np.random.randn(n_points)
            h_proposal = project_to_feasible(h_proposal)
            c5_proposal = compute_c5(h_proposal)

            if c5_proposal < best_c5:
                best_h[:] = h_proposal
                best_c5 = c5_proposal

            delta = c5 - c5_proposal
            if delta >= 0 or np.random.rand() < np.exp(delta / T):
                h[:] = h_proposal
                c5 = c5_proposal

            T *= alpha
            if sa_step % 100 == 0 and time.time() - start_time > budget_s:
                break

        # Gradient Descent Phase
        for gd_step in range(max_gd_steps):
            if time.time() - start_time > budget_s:
                break

            h_flat = h.flat
            grad = np.zeros(n_points)

            corr = np.correlate(h, 1 - h, mode="full")
            idx = np.argmax(corr)
            k = idx - n_points  # k is the shift causing the max
            if k < 0:
                conv1 = np.correlate(1 - h, 1 - h, mode="valid")
                conv2 = np.correlate(h, 1 - h, mode="full")
                k = np.argmax(conv2) - n_points

            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += 1 - 2 * h[i]
            grad /= n_points
            h -= eta * grad
            h = project_to_feasible(h)

            new_c5 = compute_c5(h)
            if np.isclose(new_c5, c5, atol=convergence_threshold):
                break
            c5 = new_c5
            if c5 < best_c5:
                best_h[:] = h
                best_c5 = c5

    return (best_h, best_c5, n_points)
