import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = h_values.reshape(-1)
    max_overlap = 0.0
    for k in range(1, n_points):
        if k < n_points:
            overlap = np.sum(h[:n_points - k] * (1 - h[k:])) * dx
            max_overlap = max(max_overlap, overlap)
    return max_overlap

    dx = 2.0 / n_points
