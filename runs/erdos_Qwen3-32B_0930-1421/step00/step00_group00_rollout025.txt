import numpy as np
import time
import scipy.signal

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == target_sum:
        return h
    diff = target_sum - s
    idx = np.flatnonzero(h < 1 if diff > 0 else h > 0)
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx) * (1 if diff > 0 else -1)
    h = np.clip(h, 0, 1)
    return h

def calculate_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = scipy.signal.correlate(h, 1 - h, mode="full")
    overlaps = overlaps * dx
    return np.max(overlaps)

    n_points = len(h)
    target_sum = n_points / 2

    best_c5 = calculate_c5(best_h)
    dx = 2.0 / n_points

    # Gradient descent parameters
    alpha = 0.1
    beta = 0.7
    max_iter = 200
    iter_idx = 0

    while time.time() - start_time < budget_s - 0.1:
        h_current = project_to_constraints(best_h.copy(), target_sum)
        overlaps = scipy.signal.correlate(h_current, 1 - h_current, mode="full")
        overlaps = overlaps * dx
        max_idx = np.argmax(overlaps)
        k_max = max_idx - n_points + 1  # shift index
        if k_max < 0:
            k_max += n_points  # wrap-around indexing

        grad = np.zeros_like(h_current)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = (1 - h_current[j]) * dx
        h_proposed = h_current - alpha * grad
        h_proposed = project_to_constraints(h_proposed, target_sum)

        c5_proposed = calculate_c5(h_proposed)
        if c5_proposed < best_c5:
            best_h[:] = h_proposed
            best_c5 = c5_proposed
            alpha = 0.1
        else:
            alpha *= beta
            
        iter_idx += 1
        if iter_idx >= max_iter:
            break

    h_feasible = project_to_constraints(best_h, target_sum)
    c5_bound = calculate_c5(h_feasible)
    return (h_feasible.tolist(), c5_bound, n_points)
