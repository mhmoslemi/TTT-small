def run(seed=42, budget_s=1100, n_points=70, n_particles=50, n_iterations=400):
    np.random.seed(seed)

    # Global integral constraint
    total = n_points / 2.0
    dx = 2.0 / n_points

    # Load initial construction
    h = initial_h_values
    assert np.all(h >= 0) and np.all(h <= 1), "Initial construction is invalid"
    assert abs(np.sum(h) - total) < 1e-8, "Initial construction violates the integral constraint"

    # Precompute indices for all valid lags
    lags = np.arange(-n_points + 1, n_points)

    # Define PSO parameters
    omega = 0.65
    c1 = 1.5
    c2 = 1.5
    positions = np.array([h + 0.1 * np.random.randn(n_points) for _ in range(n_particles)])
    positions = np.clip(positions, 0, 1)
    # Enforce integral constraint by proportionally scaling
    for i in range(n_particles):
        positions[i] = _enforce_integral(positions[i], total)

    velocities = 0.01 * np.random.randn(n_particles, n_points)
    personal_best = positions.copy()
    global_best = personal_best[np.argmin([_compute_c5(p, lags, dx) for p in personal_best])].copy()
    best_c5 = _compute_c5(global_best, lags, dx)

    start_time = time.time()

    for iteration in range(n_iterations):
        time_passed = time.time() - start_time
        if time_passed >= budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] +
                            c1 * r1 * (personal_best[i] - positions[i]) +
                            c2 * r2 * (global_best - positions[i]))
            positions[i] = np.clip(positions[i] + velocities[i], 0, 1)
            positions[i] = _enforce_integral(positions[i], total)
            new_c5 = _compute_c5(positions[i], lags, dx)
            if new_c5 < _compute_c5(personal_best[i], lags, dx):
                personal_best[i] = positions[i]
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    global_best = positions[i].copy()
        # Perturbation decay heuristic to escape local optima
        if iteration % 50 == 0:
            velocities *= 0.8

    # Final verification
    assert np.all(global_best >= 0) and np.all(global_best <= 1), "Solution out of bounds"
    assert abs(np.sum(global_best) - total) < 1e-8, "Integral constraint violated"
    c5_bound = _compute_c5(global_best, lags, dx)
    return (global_best.tolist(), c5_bound, n_points)

def _enforce_integral(h, total):
    current_sum = np.sum(h)
    if abs(current_sum - total) < 1e-8:
        return h
    h = np.clip(h, 0, 1)
    if current_sum > total:
        # Too much mass: reduce non-zero elements
        mask = h > 1e-6
        scaling_factor = (np.sum(h[mask]) - total) / np.sum(h[mask])
        h[mask] *= (1 - scaling_factor)
    elif current_sum < total:
        # Too little mass: increase lowest non-zero elements
        mask = h < 1 - 1e-6
        scaling_factor = (total - np.sum(h[mask])) / np.sum(h[mask])
        h[mask] *= (1 + scaling_factor)
    return h

def _compute_c5(h, lags, dx):
    for_valid_lags = lags[np.abs(lags) < len(h) - 1]
    result = correlate(h, 1 - h, mode='full')
    result[for_valid_lags < 0] = 0  # Invalid lags have zero contribution
    max_c5 = np.max(result[np.abs(lags) < len(h) - 1])
    return max_c5 * dx
