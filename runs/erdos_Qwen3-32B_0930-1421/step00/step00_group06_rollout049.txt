import numpy as np
import numpy.random as npr
import time

def compute_c5(h, dx):
    n = len(h)
    ones_h = 1.0 - h
    corr = np.correlate(h, ones_h, mode='full')
    corr[0:n] = -np.inf  # Exclude zero-lag
    return np.max(corr * dx)

def project_h(h, dx, n_points):
    h_proj = np.clip(h, 0.0, 1.0)
    scale = (n_points / 2.0) / (np.sum(h_proj) + 1e-10)
    return h_proj * scale

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    learning_rate = 0.01
    decay = 0.95
    max_iter = 1000
    tolerance = 1e-6
    prev_c5 = float('inf')

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute correlation and find the key lag that maximizes the overlap
        corr = np.correlate(h, 1 - h, mode='full')
        corr[0:n_points] = -np.inf  # exclude zero-lag to avoid 0
        max_idx = np.argmax(corr)
        k = max_idx if max_idx < n_points else max_idx - n_points

        # Gradient: for each i, the derivative is (1 - h[(i + k) % n]) - h[(i - k) % n]
        grad = np.zeros_like(h)
        for i in range(n_points):
            left = (i - k) % n_points
            right = (i + k) % n_points
            grad[i] = (1 - h[right]) - h[left]

        # Update step with learning rate
        h_update = h - learning_rate * grad
        h_update = project_h(h_update, dx, n_points)

        # Compute new C₅
        c5_update = compute_c5(h_update, dx)

        # Update best solution
        if c5_update < best_c5:
            best_h = h_update.copy()
            best_c5 = c5_update
        elif abs(c5_update - best_c5) < tolerance:
            # Convergence
            break

        h = h_update
        learning_rate *= decay
        prev_c5 = c5_update

    return (best_h, best_c5, n_points)
