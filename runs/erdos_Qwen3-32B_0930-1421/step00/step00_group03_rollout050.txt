import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_objective(h, dx):
    h = np.array(h, dtype=float)
    # Compute 1 - h
    one_minus_h = 1 - h
    # Compute the cross-correlation
    corr = correlate(h, one_minus_h, mode='full')
    # Only keep the positive shifts (k > 0)
    max_overlap = np.max(corr[len(h)-1:]) * dx
    return max_overlap

def clip_and_rescale(h, n_points):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    target_sum = n_points / 2
    scale_factor = target_sum / sum_h
    h *= scale_factor
    h = np.clip(h, 0, 1)  # Post-scale clipping
    # Second pass to enforce sum constraint again due to saturation
    sum_h = np.sum(h)
    if sum_h < 0.9999 * target_sum or sum_h > 1.0001 * target_sum:
        h = (h * target_sum / sum_h).clip(0, 1)
    return h

def gradient_descent_search(h0, dx, max_iter=1000, lr=0.01, kill_after_no_improvement=50):
    h = np.copy(h0)
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_objective(best_h, dx)
    last_improvement = 0
    t0 = time.time()
    n_points = len(h)
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 1:
            break
        # Compute gradient numerically
        eps = 1e-5
        grad = np.zeros_like(h)
        current_c5 = evaluate_erdos_objective(h, dx)
        for j in range(n_points):
            h_p = h + eps * np.eye(1, n_points, j)[0]
            h_p = clip_and_rescale(h_p, n_points)
            c_p = evaluate_erdos_objective(h_p, dx)
            grad[j] = (c_p - current_c5) / eps
        # Normalize the gradient
        grad /= (1e-8 + np.linalg.norm(grad))
        # Optimize: move h in negative gradient direction
        h -= lr * grad
        h = clip_and_rescale(h, n_points)
        new_c5 = evaluate_erdos_objective(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = i
        elif i - last_improvement > kill_after_no_improvement:
            break
    return best_h, best_c5
