import numpy as np
import time

from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Initialization
    h_start = initial_h_values

    n_points = len(h_start)
    dx = 2.0 / n_points
    h = h_start.copy()
    h_flatness = 0.995  # For projection
    best_h = h.copy()
    best_c5 = float("inf")

    start_time = time.time()

    def compute_c5(h_vec):
        h_vec[np.abs(h_vec) < 1e-6] = 0.0
        h_one_minus = 1.0 - h_vec
        corr = correlate(h_vec, h_one_minus, mode="full")
        max_c5 = np.max(corr) * dx
        return max_c5

    def project_h(h_vec):
        h_proj = np.clip(h_vec, 0.0, 1.0)
        sum_diff = np.sum(h_proj) - n_points / 2
        if abs(sum_diff) < 1e-6:
            return h_proj
        h_proj -= sum_diff / n_points  # project onto feasible sum
        h_proj = np.clip(h_proj, 0.0, 1.0)
        return h_proj

    def adaptive_grad_descent_step(h_vec, alpha, grad):
        h_new = h_vec - alpha * grad
        h_new = project_h(h_new)
        return h_new

    while time.time() - start_time < budget_s:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if best_c5 < 0.38080:
            break  # Early stop if below target
        if abs(c5 - best_c5) < 1e-8:
            break  # No improvement, exit loop
        corr = correlate(h, 1 - h, mode="full")
        lag_idx = np.argmax(corr)
        k = lag_idx - len(h) + 1  # Correct the lag index

        # Compute gradient for the maximum lag
        i_valid = np.where(np.arange(len(h)) + k < len(h))
        grad = - (1 - h[i_valid + k])  # Minimize the overlap
        grad = np.zeros_like(h)
        grad[i_valid] = - (1 - h[i_valid + k])
        grad = grad * dx  # Scale for integrals

        # Learning rate schedule
        learning_rate = 0.025
        if c5 < best_c5 + 0.001:
            learning_rate *= 0.95  # Decay rate on stabilization

        h = adaptive_grad_descent_step(h, learning_rate, grad)

    return best_h, best_c5, n_points
