import time

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    budget_start_time = start_time
    n_points = 70  # number of samples

    sum_h = np.sum(h)
    h *= (n_points / 2) / sum_h  # ensure sum constraint
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    
    # Optimization parameters
    lr = 0.01
    num_epochs = 1000
    patience = 50
    improve_epochs = 0

    def compute_c5(h, n_points, dx):
        # Compute overlap integral using cross-correlation
        h_inv = 1.0 - h
        correl = np.correlate(h, h_inv, mode="full")
        c5 = np.max(correl[:(n_points - 1)] * dx)
        return c5

    def project_h(h, n_points):
        # Rescale h to maintain integral constraint
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h *= (n_points / 2) / sum_h
        return h

    def get_gradient(h, n_points, dx, use_max=True):
        # Numerical gradient of the objective (C5)
        h_flattened = h.flatten()
        grad = np.zeros_like(h)
        eps = 1e-4
        baseline_c5 = compute_c5(h, n_points, dx)

        for i in range(n_points):
            h_up = h_flattened.copy()
            h_down = h_flattened.copy()
            h_up[i] += eps
            h_down[i] -= eps
            h_up = project_h(h_up, n_points)
            h_down = project_h(h_down, n_points)
            obj_up = compute_c5(h_up, n_points, dx)
            obj_down = compute_c5(h_down, n_points, dx)
            grad[i] = (obj_up - obj_down) / (2 * eps)
        return grad

    for epoch in range(num_epochs):
        if time.time() - budget_start_time > budget_s:
            break
        grad = get_gradient(h, n_points, dx)
        # Update h
        h -= lr * grad
        h = project_h(h, n_points)
        
        # Compute new C5 value
        new_c5 = compute_c5(h, n_points, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            improve_epochs = 0
        else:
            improve_epochs += 1
        
        # Learning rate decay
        lr *= np.clip(0.99, 0.5, 1.0)
        
        if improve_epochs > patience:
            break

    return best_h, best_c5, n_points
