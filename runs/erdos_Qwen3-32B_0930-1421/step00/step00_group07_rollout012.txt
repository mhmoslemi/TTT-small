```python
import numpy as np
import time
import math
from numpy import correlate

def evaluate_overlap(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    return max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = (globals()['initial_h_values'] > 0.5).astype(float)
        n_points = len(h)
    else:
        n_points = 80
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        np.random.shuffle(h)

    sum_h = int(round(np.sum(h)))
    required_sum = n_points // 2

    if sum_h != required_sum:
        diff = required_sum - sum_h
        indices = np.where(h > 0.5)[0]
        np.random.shuffle(indices)
        h[indices[:abs(diff)]] = 0.0 if diff < 0 else 1.0

    h_bin = (h > 0.5).astype(int)

    best_h = h_bin.copy()
    best_c5 = evaluate_overlap(best_h)
    T = 1.0
    alpha = 0.999
    min_T = 1e-5
    patience = 1000

    last_improvement = 0
    iteration = 0

    while time.time() - start_time < budget_s:
        new_h = best_h.copy()

        idx0 = np.random.choice(np.where(new_h == 1)[0])
        idx1 = np.random.choice(np.where(new_h == 0)[0])
        new_h[idx0] = 0
        new_h[idx1] = 1

        current_c5 = evaluate_overlap(best_h)
        candidate_c5 = evaluate_overlap(new_h)

        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                last_improvement = 0
            else:
                last_improvement += 1

        T = max(min_T, T * alpha)

        iteration += 1
        if last_improvement > patience:
            T = max(min_T, T * 0.5)
            alpha = max(alpha * 0.5, min_T)
            last_improvement = 0

        if iteration % 1000 == 0:
            print(f"Iteration: {iteration}, Best C5: {best_c5}, Temperature: {T}")

    # Final verification
    assert np.isfinite(best_c5), "Non-finite result!"
    assert np.sum(best_h) == required_sum, "Constraint violation!"
    assert all(x in (0, 1) for x in best_h), "Solution invalid!"

    return (best_h.tolist(), best_c5, n_points)
```
