import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        h1_minus_h = 1 - h_vec
        corr = correlate(h_vec, h1_minus_h, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    learning_rate = 0.01
    max_iter = 1000
    tolerance = 1e-6
    decay_factor = 0.95

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break

        # Compute the overlap integral for all k
        h1_minus_h = 1 - h
        corr = correlate(h, h1_minus_h, mode='full') * dx
        k_max = int(np.argmax(corr))

        # Compute gradient based on k that gives max overlap
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = h1_minus_h[j] - h[(i + k_max) % n_points]

        # Gradient update
        h_unclipped = h + learning_rate * grad

        # Projection to [0,1]
        h_proj = np.clip(h_unclipped, 0, 1)

        # Normalize to preserve constraint ∫h = 1
        vol = np.sum(h_proj) * dx
        if vol > 1e-8:
            h_proj *= 1.0 / vol

        # Evaluate improvement
        new_c5 = compute_c5(h_proj)
        if new_c5 < best_c5 - 1e-6:
            best_c5 = new_c5
            best_h[:] = h_proj[:]
            learning_rate = min(learning_rate * 1.05, 0.1)
        else:
            h[:] = h_proj[:]
            learning_rate *= decay_factor

    return best_h, best_c5, n_points
