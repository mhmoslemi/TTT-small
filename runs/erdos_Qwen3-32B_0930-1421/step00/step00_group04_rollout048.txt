import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=42, adapt_n=False):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_initial = h.size
        if n_points != n_initial and adapt_n:
            # Adapt initial h to new n_points with interpolation
            x = np.linspace(0, 2, n_initial)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x, h)
            h = np.clip(h, 0, 1)
            scale = n_points / 2 / np.sum(h)
            h *= scale
        else:
            h = h[:n_points].copy()

    else:
        # Simple initialization: concentration toward the center
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - (1 - x) / 1.0, 0, 1)
        h = h / h.sum() * (n_points / 2)

    h = h.astype(float)
    dx = 2.0 / n_points
    grad_alpha = 0.01  # Learning rate
    proj_eps = 1e-5  # Projection tolerance

    while time.time() - start_time < budget_s - 0.2:
        conv = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(conv)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"Improved: C5 = {c5:.6f}")

        # Compute gradient: h - h_shifted for indices overlapping in the convolution
        grad = np.zeros_like(h)
        for k in range(n_points):
            indices = np.arange(np.max([0, -k]), np.min([n_points, n_points - k]))
            overlap = len(indices)
            shifted_h = np.roll(h, -k).copy()
            grad[indices] += -dx * (h[indices] - shifted_h[indices])

        # Projective gradient descent step
        h = h - grad_alpha * grad

        # Projection: clipping and scaling to sum constraint
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        scale = (n_points / 2) / s
        h = h * scale

        # Termination
        if k % 50 == 0 and np.abs(c5 - best_c5) < 1e-8:
            break

    if best_h is None:
        best_h = h.copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    return best_h, best_c5, n_points
