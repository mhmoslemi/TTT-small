import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    full_corr = correlate(h, 1.0 - h, mode="full")
    return np.max(full_corr * dx)

def repair_h(h, n_points):
    total = np.sum(h)
    h = np.clip(h, 0, 1)
    if np.abs(total - n_points / 2) > 1e-5:  # scale only when necessary
        h_copy = h * (n_points / 2) / total
        h_copy = np.round(h_copy, 6)
        return np.clip(h_copy, 0, 1)
    return h

def random_population(pop_size, n_points):
    sum_needed = n_points / 2
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(low=0, high=1, size=n_points)
        h = repair_h(h, n_points)
        population.append(h)
    return population

def crossover(parent1, parent2, n_points):
    s = np.random.randint(n_points // 4, 3 * n_points // 4)
    child = np.concatenate([parent1[:s], parent2[s:]])
    return repair_h(child, n_points)

def mutate(h, n_points, mutation_rate=0.02):
    mutated = h.copy()
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    mutated[indices] = np.random.uniform(0, 1, size=len(indices))
    return repair_h(mutated, n_points)

def evolve_to_minimize_c5(population, n_points, budget_s, initial_h_values=None):
    start = time.time()
    max_gen = 1000
    pop_size = 50
    best_c5 = float('inf')
    best_h = None

    if initial_h_values is not None:
        population.insert(0, initial_h_values)

    generation = 0
    while time.time() - start < budget_s and generation < max_gen:
        generation += 1
        # Evaluate
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, n_points)
            fitness.append((c5, h))
        fitness.sort()
        current_c5, current_h = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Select top 20%
        cutoff = int(len(population) * 0.2)
        parents = [h for _, h in fitness[:cutoff]]
        # Reproduce
        children = []
        for _ in range(pop_size - len(parents)):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(parent1, parent2, n_points)
            child = mutate(child, n_points)
            children.append(child)
        population = parents + children
    return best_h, best_c5

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except NameError:
        print("No initial_h_values found. Using default n_points=47.")
        n_points = 47
    population = random_population(50, n_points)
    if 'initial_h_values' in globals():
        h_best, c5_best = evolve_to_minimize_c5(population, n_points, budget_s, initial_h_values)
    else:
        h_best, c5_best = evolve_to_minimize_c5(population, n_points, budget_s)
    return list(h_best), c5_best, n_points
