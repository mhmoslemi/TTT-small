import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h.copy()

    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(h, n_points)
    best_c5_value = best_c5
    alpha = 0.1
    beta = 0.7

    def compute_c5(h):
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    iterations = 0
    max_iter = 1000

    while time.time() - start_time < budget_s - 10 and iterations < max_iter:
        # Compute C5 and find the problematic shift
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)

        # Gradient (we take -direction to minimize overlap at shift k_max)
        if k_max < 0:
            g = (1 - h[-k_max:]) * dx
            g = np.concatenate((g, np.zeros(n_points + k_max)))
        elif k_max >= n_points:
            g = np.zeros(n_points + k_max - n_points)
            g = np.concatenate((g, (1 - h[:n_points + k_max - n_points]) * dx))
        else:
            g = (1 - np.roll(h, k_max)) * dx

        # Descent step
        h_proposed = h - alpha * g

        # Projection: clip to [0, 1]
        h_proposed = np.clip(h_proposed, 0, 1)

        # Project to sum constraint
        sum_h = np.sum(h_proposed)
        delta = target_sum - sum_h
        if abs(delta) < 1e-6:
            pass
        elif delta > 0:
            idx = np.where(h_proposed < 1)[0]
            if len(idx) > 0:
                h_proposed[idx] += delta / len(idx)
        else:
            idx = np.where(h_proposed > 0)[0]
            if len(idx) > 0:
                h_proposed[idx] -= abs(delta) / len(idx)

        # Check feasibility
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed = h_proposed / h_proposed.sum() * target_sum

        # Evaluate new value
        new_c5 = compute_c5(h_proposed)

        # Accept if better
        if not np.isfinite(new_c5) or new_c5 >= best_c5_value:
            alpha *= beta
            iterations += 1
            continue
        else:
            best_c5_value = new_c5
            best_h = h_proposed.copy()
            h = h_proposed.copy()
            iterations += 1
            alpha = 0.1

    # Ensure final result is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * target_sum
    c5_bound, _ = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
