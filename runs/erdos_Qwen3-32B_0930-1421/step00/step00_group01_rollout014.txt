import numpy as np
import time
from scipy.optimize import minimize
from copy import deepcopy

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    h_array = np.asarray(h, dtype=np.float64)
    ones_minus_h = 1 - h_array
    corr = np.correlate(h_array, ones_minus_h, mode="full")
    return float(np.max(corr * dx))

def smooth_max(values, eta):
    return (1 / eta) * np.log(np.sum(np.exp(eta * np.array(values))))

def project_to_feasible_space(h):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    n = len(h)
    target = n / 2
    if abs(total - target) < 1e-10:
        return h
    h = h * (target / total)
    h = np.clip(h, 0, 1)
    h = h * (target / np.sum(h))
    return h

def objective_smooth(h, eta, n_points):
    h = h.reshape(n_points)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    return smooth_max(corr * dx, eta)

def objective_true(h, n_points):
    h = h.reshape(n_points)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    return np.max(corr * dx)
