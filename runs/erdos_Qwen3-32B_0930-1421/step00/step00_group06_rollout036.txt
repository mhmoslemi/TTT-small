import numpy as np
import math
import time

def evaluate_erdos_overlap(h, dx):
    # Compute max correlation of h and 1-h over lags
    h1 = np.concatenate([h, h])
    corrs = np.correlate(h, 1 - h, mode='full')
    return np.max(corrs * dx)

def project_to_feasible(h, dx):
    # Enforce 0 <= h <= 1 and integral constraint
    h_clipped = np.clip(h, 0, 1)
    avg = np.mean(h_clipped)
    return (h_clipped / avg) * 0.5 / dx

    import cvxpy as cp

    n_points = len(h0)
    dx = 2.0 / n_points

    # Define max lag to consider (overlap only relevant within [0,2])
    max_lag = int(2 / dx)
    npoints = n_points

    h = cp.Variable(npoints)
    z = cp.Variable(max_lag + 1)
    obj = cp.Minimize(cp.max(z))

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    lag = cp.Parameter(npoints, nonneg=True, value=0.0)
    for k in range(max_lag + 1):
        kvec = []
        for i in range(npoints):
            if i + k < npoints:
                kvec.append(h[i] * (1 - h[i + k]))
            else:
                j = (i + k) % npoints
                kvec.append(h[i] * (1 - h[j]))
        overlap = cp.sum(cp.multiply(kvec, dx))
        constraints += [z[k] >= overlap + k / npoints * lag[k]]

    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.ECOS_BB, verbose=False, max_iters=500)

    best_h = h.value
    h = project_to_feasible(best_h, dx)
    best_c5 = evaluate_erdos_overlap(h, dx)

    t0 = time.time()
    budget_s = 1100
    best_h = h0.copy()
    best_c5 = evaluate_erdos_overlap(best_h, dx)

    while time.time() - t0 < budget_s:
        # ADMM step: update h, z
        corr = np.correlate(h, 1 - h, mode='full')
        z = np.maximum(corr, z - lag)
        residual = corr - z
        h_adj = h - 0.5 / npoints * lag / dx
        h = project_to_feasible(h_adj, dx)
        lag = lag + residual

        current_c5 = evaluate_erdos_overlap(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    result = (
        best_h.tolist(),
        best_c5,
        n_points
    )
    return result
