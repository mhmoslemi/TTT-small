import numpy as np

def evaluate_erdos_solution(h_values):
    h = np.array(h_values, dtype=np.float64)
    n_points = h.size
    dx = 2.0 / n_points
    # Compute all correlations of h and 1 - h
    corr = np.correlate(h, 1 - h, mode='full')
    sel_corr = corr[n_points - 1: n_points + 1]  # Relevant overlaps
    c5 = np.max(sel_corr) * dx  # Scale by dx for integral
    return c5

def project_h(h, n_points):
    h = np.clip(h, 0, 1)  # Project to valid range
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h *= (n_points / 2) / total
    return h
