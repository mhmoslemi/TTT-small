import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_current = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_current)
    dx = 2.0 / n_points

    def compute_c5(h):
        h_normalized = h / n_points * n_points  # No normalization; h is already consistent
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    alpha = 0.1
    beta = 0.7
    max_iter = 2000
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break

        corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
        k_max = np.argmax(corr)
        # Compute gradient for shift k_max
        k_indices = (np.arange(n_points) - k_max) % n_points
        gradient = (1 - h_current[k_indices.astype(int)]) * dx

        h_proposed = h_current - alpha * gradient
        # Projection step: clip
        h_proposed = np.clip(h_proposed, 0.0, 1.0)
        # Enforce sum constraint
        total = np.sum(h_proposed)
        target = n_points / 2
        excess = total - target
        if abs(excess) < 1e-8:
            pass  # Already at target
        elif excess > 0:
            # Need to decrease
            can_reduce = h_proposed > 1e-8
            total_reducible = np.sum(h_proposed[can_reduce]) - (can_reduce.sum() * 1e-8)
            if total_reducible > 0:
                delta = -excess / can_reduce.sum()
                h_proposed[can_reduce] += delta
        else:
            # Need to increase
            can_increase = h_proposed < 1 - 1e-8
            total_increaseable = (1 - h_proposed[can_increase]) - 1e-8
            if total_increaseable > 0:
                delta = -excess / can_increase.sum()
                h_proposed[can_increase] += delta

        h_proposed = np.clip(h_proposed, 0.0, 1.0)
        c5_proposed = compute_c5(h_proposed)

        if c5_proposed < best_c5:
            h_current = h_proposed.copy()
            best_h = h_proposed.copy()
            best_c5 = c5_proposed
            alpha *= 1.1  # Increase step size after improvement
        else:
            alpha *= beta  # Reduce step size on failure

        # Prevent zero step
        if alpha < 1e-5:
            break

    final_score, _ = evaluate_erdos_solution(best_h)
    return best_h.tolist(), final_score, n_points
