import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    # Parameters
    max_iter = 1000
    n_points = 84  # Starting with the parent construction size

    if "initial_h_values" in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points)
        h = h / h.mean() * 0.5

    dx = 2.0 / n_points
    h = h * (n_points / 2) / h.sum()

    def objective(h):
        # Compute (1-h) shifted and h(x)(1 - h(x + k)) via FFT
        f1 = fft(h)
        f2 = fft(1 - h)
        overlaps = ifft(f1 * np.conj(f2)).real * dx
        return np.max(overlaps)

    # Optimization loop
    alpha = 0.05
    patience = 100
    best_c5 = np.inf
    best_h = h.copy()
    niters_since_update = 0

    for i in range(max_iter):
        t0 = time.time()

        # Compute overlaps
        f1 = fft(h)
        f2 = fft(1 - h)
        overlaps = ifft(f1 * np.conj(f2)).real * dx
        c5 = np.max(overlaps)
        niters_since_update += 1

        # Track progress
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            niters_since_update = 0
            print(f"Iter {i} improved to {best_c5:.8f} (cost: {time.time()-t0:.2f}s)")

        # Stopping condition
        if time.time() - start > budget_s:
            print("Time budget exceeded, quitting...")
            break
        if niters_since_update > 0:
            alpha *= 0.99
        if niters_since_update > patience:
            print("No progress for patience steps, early stopping.")
            break

        # Compute gradient via numerical approximation
        eps = 1e-5
        grad_est = np.zeros_like(h)
        for j in range(n_points):
            # Perturb a single point
            h_finite = h.copy()
            h_finite[j] += 2 * eps
            h_finite = np.clip(h_finite, 0, 1)
            h_finite = h_finite * (n_points / 2) / h_finite.sum()  # Projection
            obj_p = objective(h_finite)

            h_finite = h.copy()
            h_finite[j] -= 2 * eps
            h_finite = np.clip(h_finite, 0, 1)
            h_finite = h_finite * (n_points / 2) / h_finite.sum()
            obj_m = objective(h_finite)

            grad_est[j] = (obj_p - obj_m) / (4 * eps)

        # Gradient descent with projection
        h = h - alpha * grad_est
        h = np.clip(h, 0, 1)  # Enforce box constraints
        h = h * (n_points / 2) / h.sum()  # Scale to meet integral constraint

    return (best_h, best_c5, n_points)
