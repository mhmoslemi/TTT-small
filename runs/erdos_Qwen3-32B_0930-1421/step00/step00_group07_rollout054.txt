import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initial h as intervals
    def get_intervals(h):
        h_bounded = np.clip(h, 0.0, 1.0)
        idx = np.where(np.abs(h_bounded - 1.0) < 1e-4)[0]
        intervals = []
        if len(idx) == 0:
            return intervals
        curr_start = idx[0]
        for j in range(1, len(idx)):
            if idx[j] != idx[j-1]+1:
                intervals.append((curr_start, idx[j-1] + 1))
                curr_start = idx[j]
        intervals.append((curr_start, len(h_bounded) - 1))
        return [(i[0] * dx, i[1] * dx) for i in intervals]
    
    def flatten_intervals(intervals):
        h = np.zeros(n_points)
        for a, b in intervals:
            a_idx = int(np.floor(a / dx))
            b_idx = int(np.ceil(b / dx))
            h[a_idx : b_idx] = 1.0
        h *= (np.sum(h) == n_points / 2.0)
        return h
    
    def compute_c5(h):
        ones = np.zeros(n_points)
        zeros = np.zeros(n_points)
        h_h_bounded = np.clip(h, 0.0, 1.0)
        ones[(h_h_bounded > 0.5).astype(int)] = 1
        zeros[(h_h_bounded < 0.5).astype(int)] = 1
        corr = correlate(ones - zeros, ones, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    intervals = get_intervals(initial_h_values)
    best_intervals = intervals
    h_best = flatten_intervals(intervals)
    c5_best = compute_c5(h_best)
    print(f'Initialized with C5: {c5_best}')
    
    def tweak_intervals(intervals, step_size=0.01):
        intervals = [list(interval) for interval in intervals]
        total_length = sum(b - a for a, b in intervals)
        i = np.random.choice(len(intervals))
        a, b = intervals[i]
        jitter = np.random.normal(0.0, step_size, 2)
        new_a = max(0.0, a + jitter[0])
        new_b = max(new_a, min(2.0, b + jitter[1]))
        if new_b - new_a + total_length - (b - a) < 1e-8:
            return intervals
        intervals[i] = [new_a, new_b]
        for j in [i-1, i+1]:
            if 0 <= j < len(intervals) and abs(j - i) < 2:
                a_adj, b_adj = intervals[j]
                if a_adj >= intervals[i][1]:
                    intervals[j][0] = np.min([a_adj, intervals[i][1]])
                    intervals[j][1] = np.max([b_adj, intervals[i][1]])
        return intervals
    
    for iter in range(5000):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.98:
            break
        if iter % 200 == 0:
            print(f'Iteration {iter}, C5: {c5_best}')
        intervals_proposed = tweak_intervals(best_intervals, step_size=0.1)
        proposed_h = flatten_intervals(intervals_proposed)
        proposed_c5 = compute_c5(proposed_h)
        if proposed_c5 < c5_best:
            best_intervals = intervals_proposed
            c5_best = proposed_c5
            h_best = proposed_h
        elif iter % 20 == 0 and np.random.rand() < 0.03:
            best_intervals = intervals_proposed
            c5_best = proposed_c5
            h_best = proposed_h
    h_best = np.clip(h_best, 0.0, 1.0)
    assert abs(np.sum(h_best) - n_points / 2.0) < 1e-6, "Constraint violated!"
    final_c5 = compute_c5(h_best)
    print(f'Final C5: {final_c5}')
    return h_best, final_c5, n_points
