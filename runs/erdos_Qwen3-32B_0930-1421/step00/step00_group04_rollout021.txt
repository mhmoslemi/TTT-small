import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, pop_size=20, generations=100, mutation_rate=0.05, n_points=42):
    np.random.seed(seed)

    # Load initial solution
    h0 = np.array(initial_h_values)
    assert np.all(h0 >= 0) and np.all(h0 <= 1) and np.isclose(np.sum(h0), n_points / 2)

    # Discretization step
    dx = 2.0 / n_points

    def compute_c5(h):
        correlations = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlations) * dx
        return c5

    def tournament_selection(population, fitness, size=3):
        indices = np.random.choice(len(population), size)
        return population[indices[np.argmin(fitness[indices])]]

    def crossover(parent1, parent2):
        alpha = np.random.rand(len(parent1))
        child = alpha * parent1 + (1 - alpha) * parent2
        child *= n_points / 2 / child.sum()
        return np.clip(child, 0, 1)

    def mutate(individual):
        mask = np.random.rand(len(individual)) < mutation_rate
        individual[mask] = np.random.rand(np.sum(mask))
        individual *= n_points / 2 / individual.sum()
        return np.clip(individual, 0, 1)

    start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Initialize population
    population = [h0.copy()]
    while len(population) < pop_size:
        h_random = np.random.rand(n_points)
        h_random *= n_points / 2 / h_random.sum()
        h_random = np.clip(h_random, 0, 1)
        population.append(h_random)

    population = np.array(population)

    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break

        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population, [compute_c5(h) for h in population])
            parent2 = tournament_selection(population, [compute_c5(h) for h in population])
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)

        population = np.array(new_population)
        current_c5 = [compute_c5(h) for h in population]
        current_best_idx = np.argmin(current_c5)
        current_best_c5 = current_c5[current_best_idx]
        current_best_h = population[current_best_idx].copy()

        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5

    return best_h, best_c5, n_points
