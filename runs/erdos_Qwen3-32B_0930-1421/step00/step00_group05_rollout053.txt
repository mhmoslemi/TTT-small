import numpy as np
import cvxpy as cp
import scipy as sp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Set discretization size
    n_points = 70  # Start with the parent's discretization
    dx = 2.0 / n_points

    # Load the initial construction
    h0 = np.copy(initial_h_values)
    if h0.shape[0] != n_points:
        h0 = sp.interpolate.interp1d(np.linspace(0, 2, h0.shape[0]), h0, kind="linear", fill_value="extrapolate")(
            np.linspace(0, 2, n_points)
        )
        h0 = np.round(h0, decimals=5)
        h0 = np.clip(h0, 0.0, 1.0)  # Ensure box constraint
        # Adjust for mean 1/2
        total_mass = np.sum(h0) * dx
        adjustment = (1 - total_mass) / (n_points)
        h0 += adjustment
        h0 = np.clip(h0, 0.0, 1.0)

    h = cp.Variable(n_points, nonneg=True)
    objective = []
    objective_sum = 0
    for k in range(0, n_points):
        left = h * (1 - cp.roll(h, -k))
        objective_sum += cp.sum(left) * dx
        objective.append(objective_sum)
    C5 = cp.Max(objective)

    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1.0,
        h >= 0.0,
    ]

    problem = cp.Problem(cp.Minimize(C5), constraints)
    problem.solve(solver=cp.GUROBI, warm_start=True, timeLimit=1090)
    print(f"{problem.value=}")

    h_values = np.array(h.value).ravel()

    # Ensure the solution is valid
    h_values = np.clip(h_values, 0.0, 1.0)
    total_mass = np.sum(h_values) * dx
    if abs(total_mass - 1.0) > 1e-6:  # mass mismatch
        adjustment = (1.0 - total_mass) / (n_points * dx)
        h_values += adjustment
        h_values = np.clip(h_values, 0.0, 1.0)

    elapsed = time.time() - start_time
    print(f"Elapsed time: {elapsed:.2f}s")

    return (
        h_values,
        evaluate_erdos_solution(h_values)[0],
        h_values.size,
    )
