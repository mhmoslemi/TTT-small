import numpy as np
import cvxpy as cp
import time
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use initial_h_values if it exists (globally defined in the harness)
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        n_points = 45
        h = np.zeros(n_points)
        h[n_points//2] = 1.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    max_iter = 1000
    alpha = 0.01  # learning rate / step size for smooth approx
    curr_iter = 0

    while time.time() - start_time < budget_s:
        h = best_h.copy()
        h_var = cp.Variable(n_points)

        # Objective approx: smooth approx to max_k overlap
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            overlaps.append(cp.sum(cp.multiply(h_var, shifted)) * dx)

        # Huber approximation for max
        avg_overlap = sum(overlaps) / n_points
        obj = avg_overlap + alpha * cp.norm((avg_overlap - cp.bmat(overlaps)), p='inf')

        # Constraints
        constraints = [
            h_var <= 1.0,
            h_var >= 0.0,
            cp.sum(h_var) == n_points / 2  # ∫h dx = 1
        ]

        # Solve
        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.SCIP, verbose=False)

        if problem.status != 'optimal':
            break

        candidate = h_var.value
        if candidate is None:
            break

        # Project onto valid domain
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        candidate = candidate * (n_points / 2) / candidate_sum

        # Evaluate
        c5 = evaluate_erdos_solution(candidate)

        # Update best
        if c5 < best_c5 - 1e-8:  # positive improvement threshold
            best_c5 = c5
            best_h = candidate
            print(f"Improved C5 to {best_c5:.7f} at iteration {curr_iter}")

        curr_iter += 1

    return (best_h, best_c5, n_points)
