import numpy as np
import time
from scipy.optimize import minimize
import functools

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    valid_k = (n_points, 2 * n_points - 1)  # k = n_points corresponds to offset of 0
    max_val = 0.0
    for k in range(n_points, 2 * n_points):
        overlap = np.dot(h, 1 - np.roll(h, -k) % 1) * dx
        max_val = max(max_val, overlap)
    return max_val

def project_to_feasible_space(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if abs(current_sum - n_points / 2) > 1e-6:
        scale_factor = (n_points / 2) / current_sum
        h *= scale_factor
        h = np.clip(h, 0, 1)  # clip again after scaling
    return h

def evolve_population(population, n_points, mutation_strength=0.1):
    # Crossover with arithmetic blending
    parent_idx = np.argsort([compute_c5(ind, n_points) for ind in population])[:2]
    parent1, parent2 = population[parent_idx[0]], population[parent_idx[1]]
    child = 0.5 * (parent1 + parent2)

    # Mutation
    noise = np.random.normal(0, mutation_strength, n_points)
    child += noise
    child = project_to_feasible_space(child, n_points)

    return child

def evaluate_population(population, n_points):
    return np.array([compute_c5(ind, n_points) for ind in population])
