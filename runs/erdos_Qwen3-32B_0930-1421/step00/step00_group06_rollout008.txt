import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    correlate = np.correlate(h, 1 - h, mode="full")
    return np.max(correlate * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the parent construction if available
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_vals)
    else:
        # Fallback initialization with uniform distribution
        n_points = 95  # Start with n=95 as we're optimizing the known initial solution
        h_vals = np.ones(n_points) / n_points * (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)

    best_c5 = float('inf')
    best_h = h_vals.copy()
    terminated = False

    while time.time() < start_time + budget_s and not terminated:
        try:
            # Optimization step - convex relaxation
            h = cp.Variable(n_points, nonneg=True)
            nu = cp.Variable()

            # Compute overlaps for lags that would involve in [0, n_points]
            lags = []
            for k in range(1, n_points):
                if k <= n_points - k:  # exploit symmetry
                    lags.append(k)
                if k > n_points - k:
                    break

            obj_vars = []
            for k in lags:
                idxs = np.arange(n_points)
                shifted = np.roll(h_vals, k)
                overlap = h @ (1 - shifted) * (2.0 / n_points)
                obj_vars.append(overlap)
            C = cp.MaxExpression(obj_vars)

            objective = cp.Minimize(C)
            constraints = [
                cp.sum(h) == n_points / 2,
                h <= 1.0,
                h >= 0.0
            ]

            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver="ECOS", verbose=False)
            if np.isfinite(result):
                h_vals = np.ravel(h.value).astype(np.float64)
                h_vals = np.clip(h_vals, 0, 1)
                h_vals = h_vals * (n_points / 2) / np.sum(h_vals)  # normalize to constraint

                # Evaluate improved C5
                c5_current = compute_c5(h_vals, n_points)

                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_vals.copy()

        except Exception as e:
            # Continue optimization with best so far in case of error
            pass

        # Termination based on time or acceptable C5
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s or best_c5 <= 0.38085:
            terminated = True

    return (best_h.tolist(), best_c5, n_points)
