import numpy as np
import time
from scipy.optimize import minimize, BFGS
import cvxpy as cp
from collections import deque


def evaluate_population(population, n_points, dx):
    results = []
    for h in population:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr * dx)
        results.append((c5_val, h))
    return sorted(results, key=lambda x: x[0])


def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    total = h.sum()
    target_sum = n_points / 2.0
    if abs(total - target_sum) > 1e-6:
        h = h * (target_sum / total)
    return h


def crossover(parent1, parent2):
    alpha = np.random.uniform(0, 1)
    return (alpha * parent1 + (1 - alpha) * parent2)


def mutate(h, sigma=0.1):
    mutation = np.random.normal(loc=0, scale=sigma, size=len(h))
    h_mutant = h + mutation
    return project_to_feasible(h_mutant, len(h))


def refine_local(h_best, n_points, dx):
    def objective(h_vec):
        h = h_vec.reshape(-1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)
    
    def constraint_eq(h_vec):
        return h_vec.sum() - n_points / 2.0

    def constraint_lower(h_vec):
        return h_vec

    def constraint_upper(h_vec):
        return 1.0 - h_vec

    # Box constraints and equality constraints
    cons = (
        {'type': 'eq', 'fun': constraint_eq},
        {'type': 'ineq', 'fun': constraint_lower},
        {'type': 'ineq', 'fun': constraint_upper}
    )

    # Initial point must be feasible
    h_init = h_best.copy()
    h_init = project_to_feasible(h_init, n_points)
    
    result = minimize(
        objective,
        x0=h_init,
        method='SLSQP',
        jac=None,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 20},
    )
    if result.success:
        return result.x.reshape(-1)
    return h_best


def run(seed=42, budget_s=1100, n_points=45, pop_size=50, max_gen=50):
    start_time = time.time()
    np.random.seed(seed)
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_parent = initial_h_values
    else:
        h_parent = np.zeros(n_points, dtype=np.float32)
        h_parent[15:30] = 1.0  # Fill middle with 15 samples of 1
        h_parent = project_to_feasible(h_parent, n_points)

    population = [h_parent]
    for _ in range(pop_size - 1):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_sample = h_parent + perturbation
        h_sample = project_to_feasible(h_sample, n_points)
        population.append(h_sample)

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 10:
            break

        population = evaluate_population(population, n_points, dx)
        top_fraction = int(0.2 * pop_size)
        elites = [p[1] for p in population[:top_fraction]]

        while len(population) > pop_size:
            population.pop()

        for _ in range(len(population) - pop_size // 2):
            parent1, parent2 = np.random.choice(elites, 2)
            offspring = crossover(parent1, parent2)
            offspring = mutate(offspring)
            population.append((0, offspring))

    population = evaluate_population(population, n_points, dx)
    best_h = population[0][1]
    best_h = refine_local(best_h, n_points, dx)
    best_h = project_to_feasible(best_h, n_points)

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr * dx)

    return (best_h.tolist(), c5_bound, n_points)
