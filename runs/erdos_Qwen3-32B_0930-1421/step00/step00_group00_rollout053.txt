import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(h, n_points=n_points)
    max_iter = 1000
    alpha = 1.0  # Initial step size
    beta = 0.5  # Backtracking reduction factor
    grad_scale = 1.0

    for iter in range(max_iter):
        # Compute the cross-correlation using fast evaluation
        c5_new, overlaps = evaluate_erdos_solution(h, n_points=n_points)
        # Find the shift with maximum overlap
        overlap_flat = overlaps.flatten()
        k_star = np.argmax(np.abs(overlap_flat)) % n_points
        
        if k_star > n_points:
            gradient = np.zeros_like(h)
        else:
            # Compute gradient using only the critical shift k_star
            # Only the central summands can contribute meaningfully
            gradient = np.zeros_like(h)
            i = np.arange(n_points)
            k_idx = (i - k_star + n_points) % n_points
            gradient[i] = - (1 - h[k_idx]) * dx

        # Gradient descent step
        h_new_unclipped = h - alpha * grad_scale * gradient
        # Clipping onto [0,1] followed by sum-projection
        h_new_clipped = np.clip(h_new_unclipped, 0.0, 1.0)
        current_sum = np.sum(h_new_clipped)
        if current_sum != 0:
            h_new = h_new_clipped * (target_sum / current_sum)
        else:
            break

        c5_new, _ = evaluate_erdos_solution(h_new, n_points=n_points)

        # Line search and backtracking
        backtrack_count = 0
        while c5_new > best_c5 and backtrack_count < 10:
            alpha *= beta
            h_new_unclipped = h - alpha * grad_scale * gradient
            h_new_clipped = np.clip(h_new_unclipped, 0.0, 1.0)
            current_sum = np.sum(h_new_clipped)
            if current_sum != 0:
                h_new = h_new_clipped * (target_sum / current_sum)
            else:
                break
            c5_new, _ = evaluate_erdos_solution(h_new, n_points=n_points)
            backtrack_count += 1

        if c5_new < best_c5 or np.isclose(c5_new, best_c5):
            best_h = h_new.copy()
            best_c5 = c5_new
            h = best_h.copy()
        else:
            # No improvement, revert
            pass

        if time.time() - start_time > budget_s - 5:
            print(f"Budget almost exhausted, returning best solution with C₅ = {best_c5:.6f}")
            break

    return (best_h, best_c5, n_points)
