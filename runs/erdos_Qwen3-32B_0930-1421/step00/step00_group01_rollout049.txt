import numpy as np
from scipy.optimize import minimize

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h_array = h.reshape(-1)
    max_overlap = 0.0
    for k in range(1, n_points):
        overlap = np.sum(h_array[:n_points - k] * (1 - h_array[k:])) * dx
        max_overlap = max(max_overlap, overlap)
        overlap_flip = np.sum(h_array[k:] * (1 - h_array[:n_points - k])) * dx
        max_overlap = max(max_overlap, overlap_flip)
    return max_overlap

    return evaluate_c5(h, n_points)
