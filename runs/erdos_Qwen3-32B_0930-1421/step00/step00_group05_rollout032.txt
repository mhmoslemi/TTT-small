import numpy as np

    start_time = time.time()
    max_time = budget_s

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.zeros(n_points)
        n_true = n_points // 2
        h[:n_true] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2) / h.sum()

    dx = 2.0 / n_points
    h *= n_points / 2
    h /= h.sum() * dx

    best_h = h.copy()
    best_c5 = float('inf')
    last_improvement_time = time.time()

    # Optimization parameters
    learning_rate = 0.01
    beta = 10.0
    beta_step = 100
    beta_increase = 1.5
    patience = 300
    grad_clip = 1.0
    n_hist = int(n_points * 2)
    
    while time.time() - start_time < max_time:
        # Compute overlap integrals using convolution
        h_dot = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(h_dot)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement_time = time.time()
        elif time.time() - last_improvement_time > max_time * 0.05:
            # Fresh start if no improvement
            h = np.zeros(n_points)
            n_true = n_points // 2
            h[:n_true] = 1.0
            np.random.shuffle(h)
            h = h * (n_points / 2) / h.sum()
            h = h.clip(0, 1)
            last_improvement_time = time.time()

        # Use log-sum-exp approximation of max
        log_sum = np.log(np.sum(np.exp(beta * (h_dot - c5)), where=np.isfinite(h_dot)))
        objective = (np.log(log_sum) / beta) + c5

        # Compute weights for gradients
        grad_weights = 1e-8 + ((beta * np.exp(beta * (h_dot - c5)))
                               / np.sum(beta * np.exp(beta * (h_dot - c5)) * (h_dot - c5)))

        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(n_hist):
            if k < n_points:
                grad += grad_weights[k] * ((1 - h[k]) * np.ones_like(h))
            if k < n_points:
                grad -= grad_weights[k] * (h * np.ones_like(h[k]))

        # Projected gradient descent
        h -= learning_rate * (grad / np.max(np.abs(grad) + 1e-8))
        h = np.clip(h, 0, 1)
        
        # Normalize to exactly meet the constraint sum(h) == n_points / 2
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum

        # Adjust parameters
        if (time.time() - start_time) - (beta_step * learning_rate) > beta_increase:
            beta *= 1.2
            learning_rate = max(learning_rate * 0.98, 1e-5)

    return best_h, best_c5, n_points
