import numpy as np
import cvxpy as cp
import time
import random

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(corr * dx)
    return C5

def project_to_constraints(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    scale_factor = n_points / 2 / total
    h_scaled = h_clipped * scale_factor

    # Enforce [0, 1] again to be safe
    return np.clip(h_scaled, 0, 1)

    np.random.seed(seed)
    random.seed(seed)

    n_points = len(initial_h)

    n_particles = 40
    max_iter = 1000

    c2 = 1.5
    omega = 0.7
    omega_decay = 0.99

    particles = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    velocities = [0.0 * np.random.rand(n_points) for _ in range(n_particles)]

    # Enforce constraints after perturbation
    particles = [project_to_constraints(p, n_points) for p in particles]
    velocities = [np.random.rand(n_points) for _ in range(n_particles)]

    personal_best = particles.copy()
    personal_best_scores = [evaluate_C5(p) for p in particles]

    global_best = particles[np.argmin(personal_best_scores)]
    global_best_score = min(personal_best_scores)

    start_time = time.time()
    iter_no_improve = 0
    best_score = global_best_score

    for t in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
        if iter_no_improve > 200:
            break
        if t > 0 and abs(best_score - global_best_score) < 1e-8:
            iter_no_improve += 1
        best_score = global_best_score

        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()

            velocities[i] = (
                omega * velocities[i]
                + c1 * r1 * (personal_best[i] - particles[i])
                + c2 * r2 * (global_best - particles[i])
            )
            particles[i] += velocities[i]
            particles[i] = project_to_constraints(particles[i], n_points)

            current_score = evaluate_C5(particles[i])
            if current_score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = current_score
                if current_score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = current_score
                    iter_no_improve = 0

        omega *= omega_decay

    return (global_best.tolist(), global_best_score, n_points)
