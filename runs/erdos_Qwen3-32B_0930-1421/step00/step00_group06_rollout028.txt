import numpy as np
import cvxpy as cp
import time

def evaluate_overlap(h, dx):
    h_r = h.copy()
    h_rc = np.concatenate([h_r, h_r])  # For all shifts
    correlations = np.correlate(h_r, 1 - h_rc, mode='full')[: len(h_r) + len(h_r) - 1]
    return np.max(correlations * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 95  # default as per task description
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / np.sum(h)

    dx = 2.0 / n_points
    h_prev = h.copy()
    h_shifted = h.copy()
    alpha = 0.5
    best_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    iteration = 0
    max_iterations = 1000

    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Update h_shifted as a convex combination
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Compute overlap
        h_shifted_r = h_shifted
        h_shifted_rc = np.concatenate([h_shifted_r, h_shifted_r])
        corr = np.correlate(h_shifted_r, 1 - h_shifted_rc, mode='full')[: len(h_shifted_r) + len(h_shifted_r) - 1]
        k_max = np.argmax(corr)
        overlap_kmax = corr[k_max]

        # Formulate CVX problem - minimize the worst overlap at k_max
        x = cp.Variable(n_points, nonneg=True)
        obj = cp.Minimize(overlap_kmax)
        const = [
            cp.sum(x) == n_points / 2,
            x <= 1,
            x >= 0
        ]

        prob = cp.Problem(obj, const)
        prob.solve(solver=cp.ECOS, verbose=False)
        h_candidate = np.array(x.value).flatten()

        # Compute the actual overlap on this candidate h
        h_candidate_dx = evaluate_overlap(h_candidate, dx)
        if h_candidate_dx < best_c5:
            best_c5 = h_candidate_dx
            best_h = h_candidate.copy()
            h_prev = h.copy()
            h = h_candidate.copy()
            alpha = 0.5
        else:
            alpha = max(alpha * 0.95, 0.1)

        iteration += 1

    # Validation
    best_h[best_h > 1.0] = 1.0
    best_h[best_h < 0.0] = 0.0
    best_h = best_h / np.sum(best_h) * n_points / 2

    return best_h, best_c5, n_points
