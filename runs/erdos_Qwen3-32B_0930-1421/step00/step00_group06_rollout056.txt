import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 95  # start with the initial size
    dx = 2.0 / n_points

    # Initialize h from the provided initial construction
    h = initial_h_values()

    # Featurize and ensure constraints
    h = np.clip(h, 0, 1)
    h = h / h.mean() * (n_points / 2 / n_points) * n_points

    max_iter = 500
    eps = 1e-6
    nu = None

    for iter in range(max_iter):
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx

        # Trim to relevant lags (from -n_points+1 to n_points-1, shift to 0:+ve)
        overlaps = overlaps[n_points - 1 : 2 * n_points - 1]

        if iter == 0:
            nu = overlaps.copy()
        else:
            nu = np.maximum(nu, overlaps)

        # Formulate convex subproblem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.log(nu[n_points - 1 :] + eps)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.OSQP, verbose=False)

        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # on failure, fall back
            h = h / h.mean() * (n_points / 2 / n_points) * n_points
            continue

        h = np.clip(h_var.value, 0, 1)
        # Ensure average constraint is satisfied
        h = h / np.mean(h) * (n_points / 2 / n_points) * n_points

        # Check budget exit
        if time.time() - start_time > budget_s:
            break

    # Final evaluation
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    overlaps = overlaps[n_points - 1 : 2 * n_points - 1]
    c5_bound = overlaps.max()

    # Validation
    assert np.all(h >= 0) and np.all(h <= 1), "Solution out of [0,1]"
    assert abs(sum(h) - n_points / 2) < 1e-6, "Sum constraint violated"

    return h, c5_bound, n_points
