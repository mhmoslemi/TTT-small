import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = np.copy(h)
    best_c5, _ = evaluate_erdos_solution(h)
    
    # Line search parameters
    alpha = 1.0
    beta = 0.5
    grad_tol = 1e-6
    max_line_search = 20
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        overlap = correlate(h, 1 - h, mode="full")
        overlap *= dx
        k_star = np.argmax(overlap) - h.size + 1  # Adjust for mode="full"
        
        # Compute gradient (reverse indices due to cross-correlation)
        grad = np.zeros_like(h)
        if -n_points < k_star < n_points:
            k_idx = int(k_star)
            if k_idx > 0:
                grad[:-k_idx] = (1 - h[k_idx:])
            elif k_idx < 0:
                grad[-k_idx:] = (1 - h[:-k_idx])
        
        # Skip if zero gradient to avoid numerical issues
        if np.abs(grad).max() < grad_tol:
            break
        
        # Gradient direction
        grad = grad / np.linalg.norm(grad + 1e-8)
        
        # Backtracking line search
        t = alpha
        new_h = None
        for _ in range(max_line_search):
            candidate = np.clip(h - t * grad, 0, 1)  # Box constraint

            # Maintain sum constraint: project onto sum(S) = target_sum
            # Projection via scaling (after clipping)
            cur_sum = candidate.sum()
            if cur_sum != 0 and np.abs(cur_sum) > 1e-4:
                scale_factor = target_sum / cur_sum
                candidate = candidate * scale_factor
                candidate = np.clip(candidate, 0, 1)

            new_c5, _ = evaluate_erdos_solution(candidate)

            if new_c5 < best_c5:
                new_h = candidate
                h = new_h
                best_c5 = new_c5
                best_h = np.copy(h)
                break
            else:
                t *= beta
        else:
            break  # Failed to improve: break early

        if (time.time() - start_time) >= budget_s:
            break

    return (best_h, best_c5, n_points)
