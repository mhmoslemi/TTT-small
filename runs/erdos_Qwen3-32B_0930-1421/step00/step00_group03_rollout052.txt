import numpy as np
import random
import time
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corr = correlate(h, ones_minus_h, mode='full')
    C5 = (corr * dx).max()
    return C5

def project_to_sum(h, n_points):
    total = h.sum()
    if np.isclose(total, n_points / 2):
        return h
    s = n_points / 2
    diff = s - total
    idx = np.argsort(h)
    if diff > 1e-8:
        idx = idx[:int(np.ceil(len(idx)*0.1))]
        while diff > 1e-8 and len(idx) > 0:
            count = len(idx)
            step = diff / count
            if step > 1.0 - h[idx].min():
                step = min(1.0 - h[idx], np.full_like(idx, step))
                diff -= (step * (2 / n_points)).sum()
            else:
                h[idx] += step
                diff = s - h.sum()
            idx = idx[np.argsort(h[idx])]  # maintain sorted indices
    elif diff < -1e-8:
        idx = idx[::-1][:int(np.ceil(len(idx)*0.1))]
        while diff < -1e-8 and len(idx) > 0:
            count = len(idx)
            step = -diff / count
            if step > h[idx].min():
                step = np.clip(h[idx] - 0, -diff / count, 0)
                diff += (step * (2 / n_points)).sum()
            else:
                h[idx] -= step
                diff = h.sum() - s
            idx = idx[np.argsort(h[idx])[::-1]]
    return h.clip(0, 1)
