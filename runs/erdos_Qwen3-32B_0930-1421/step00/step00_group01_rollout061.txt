from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    best_h = None

    clip = lambda h: np.clip(h, 0, 1)
    norm = lambda h: h / h.sum() * n_points / 2

    def objective(h):
        h = h.reshape(-1)
        h = clip(norm(h))
        dx = 2.0 / n_points
        correlations = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlations)

    def callback(x, status=None):
        nonlocal best_c5, best_h
        current_c5 = objective(x)
        if current_c5 < best_c5:
            best_h = x.copy()
            best_c5 = current_c5

    def generate_starting_points(num=20):
        h_initial = np.copy(initial_h_values)
        perturbation = np.random.uniform(-0.1, 0.1, h_initial.shape)
        starting_points = [clip(norm(h_initial + p * h_initial.std()))
                           for p in perturbation]
        for i in range(num - len(starting_points)):
            starting_points.append(clip(norm(np.random.rand(n_points))))
        return starting_points

    starting_points = generate_starting_points()
    
    for x0 in starting_points:
        if time.time() - start_time >= budget_s - 5:
            break
        bounds = [(0, 1)] * n_points
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', callback=callback, options={'maxiter': 500})
        if result.fun < best_c5:
            best_h = result.x.copy()
