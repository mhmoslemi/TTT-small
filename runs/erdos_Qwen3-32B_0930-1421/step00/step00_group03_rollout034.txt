import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_bounds
from scipy.optimize import Bounds

def evaluate_c5(h_values, n_points, dx):
    h = h_values.reshape(-1)
    h_array = np.zeros(n_points << 1)
    h_array[:n_points] = h
    c = correlate(h_array, 1 - h_array, mode='full')
    c = c * dx
    max_c = np.max(c[n_points:2 * n_points])
    return max_c

def project_onto_simplex(v, n_points):
    v = v.clip(0, 1)
    v_sum = v.sum()
    if v_sum == n_points / 2:
        return v
    alpha = (v_sum - n_points / 2) / n_points
    return v - alpha
