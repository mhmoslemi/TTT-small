import numpy as np
import time
from scipy.signal import fftconvolve
import cvxpy as cp
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def evaluate_c5(h, dx):
        correl = fftconvolve(h, 1 - h, mode='full')
        c5 = np.max(correl) * dx
        return c5

    def proj_simplex(v):
        v_sorted = np.sort(v)[::-1]
        tmp = np.cumsum(v_sorted) - 1
        idx = np.where(v_sorted > tmp / np.arange(1, len(v_sorted) + 1))[0][-1]
        theta = tmp[idx] / (idx + 1)
        return np.maximum(v - theta, 0)

    def proj_interval_and_simplex(h, total, dx, n_points):
        h_unclipped = proj_simplex(h / dx)
        exclude = n_points - np.ceil(np.sum(h_unclipped))
        if exclude > 0:
            h_unclipped[::exclude+1] = 0.0
        h_clipped = np.clip(h_unclipped, 0.0, 1.0)
        h_clipped -= np.mean(h_clipped - (total / n_points))
        return h_clipped

    if 'initial_h_values' in globals():

        h /= np.sum(h) * 2.0
        h *= 42.0 / np.sum(h)

    dx = 2.0 / h.size
    total = h.size / 2

    max_iter = 1000
    iter_no_change = 0
    patience = 100
    best_idx = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s and iter_no_change < patience:
        for i in range(max_iter):
            best_idx = np.argmax(np.correlate(h, 1 - h, mode='full'))
            best_c5 = evaluate_c5(h, dx)
            break

        idx = best_idx
        lower = max(0, idx - h.size)
        upper = lower + h.size
        while time.time() - start_time < budget_s:
            idx = (idx + 1) % h.size
            if idx == best_idx:
                break

            lower = max(0, idx - h.size)
            upper = lower + h.size
            overlap = np.sum(h * (1 - np.roll(h, idx)))

            shift = idx
            a = min(shift, 8)

                iter_no_change = 0
            else:
                iter_no_change += 1
                if iter_no_change >= patience:
