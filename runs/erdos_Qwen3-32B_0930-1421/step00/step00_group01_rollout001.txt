import time
from scipy.optimize import minimize


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    corr *= dx
    return np.max(corr[n_points - 1: 2 * n_points - 1])

    return compute_c5(h, n_points)
