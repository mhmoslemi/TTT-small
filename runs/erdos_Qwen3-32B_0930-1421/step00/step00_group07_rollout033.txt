import numpy as np
import evaluate_erdos_solution
import time
import scipy.signal
import cvxpy as cp
from math import sqrt

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Setup timer and tracking
    t0 = time.time()
    max_time = budget_s
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_mass = n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution.evaluate_C5(h)

    # Gradient descent parameters
    learning_rate = 0.1
    patience = 200
    no_improvement = 0
    max_iter = 5_000
    iter_ = 0
    batch_size = 3  # Small batch size to avoid getting stuck

    def clip_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scaling = target_mass / max(1e-10, total)
        h = h * scaling
        return h

    def compute_update(h):
        h1h = 1 - h
        corr = scipy.signal.correlate(h, h1h, mode="full")
        dx_corr = corr * dx
        peak_idx = np.argmax(np.abs(dx_corr))
        lag = peak_idx - (len(h) - 1)
        # Get the overlapping indices
        def get_overlap_indices(lag):
            if lag >= 0:
                start = 0
                stop = len(h) - lag
                indices = (np.arange(start, stop), np.arange(start + lag, len(h)))
            else:
                start = -lag
                stop = len(h)
                indices = (np.arange(start, len(h)), np.arange(0, stop))
            return indices

        ixs = get_overlap_indices(lag)
        grad = np.zeros(len(h))
        grad[ixs[0]] -= 2 * (1 - h[ixs[1]])
        grad[ixs[1]] += 2 * h[ixs[0]]
        return grad

    while time.time() - t0 < max_time and iter_ < max_iter:
        # Stochastic Objective Selection
        if np.random.rand() < 0.2:
            # Occasionally shift to a random lag to diversify search
            lag = np.random.randint(-len(h) + 1, len(h))
            def get_overlap_indices(lag):
                if lag >= 0:
                    start = 0
                    stop = len(h) - lag
                    indices = (np.arange(start, stop), np.arange(start + lag, len(h)))
                else:
                    start = -lag
                    stop = len(h)
                    indices = (np.arange(start, len(h)), np.arange(0, stop))
                return indices

            ixs = get_overlap_indices(lag)
            if len(ixs[0]) == 0:
                continue  # Skip if no overlap
            grad = np.zeros(len(h))
            grad[ixs[0]] -= 2 * (1 - h[ixs[1]])
            grad[ixs[1]] += 2 * h[ixs[0]]
        else:
            grad = compute_update(h)

        # Projected update
        h_update = h - learning_rate * grad
        h_update = clip_h(h_update)

        # Test if better and track
        new_c5 = evaluate_erdos_solution.evaluate_C5(h_update)
        if new_c5 < best_c5:
            best_h = h_update.copy()
            best_c5 = new_c5
            no_improvement = 0
            learning_rate = min(learning_rate * 1.05, 1.0)
        else:
            no_improvement += 1

        # Learning rate scheduling
        if no_improvement > patience:
            learning_rate *= 0.5
            no_improvement = 0
            if learning_rate < 1e-7:
                break

        h = h_update
        iter_ += 1
        if iter_ % 100 == 0:
            print(f"Iteration {iter_}: C5 = {best_c5:.6f}, learning rate: {learning_rate:.4f}, time: {time.time() - t0:.1f}s")

    return best_h, best_c5, n_points
