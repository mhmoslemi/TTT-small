import numpy as np
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Use the initial h if available
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, copy=True)
    dx = 2.0 / n_points

    # Enforce sum constraint
    total_sum = np.sum(h)
    scale_factor = (n_points / 2.0) / total_sum
    h *= scale_factor

    h = np.clip(h, 0, 1)

    # Precompute 1 - h
    one_minus_h = 1.0 - h

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    c5_history = [best_c5]
    patience = 20
    wait = 0

    while time.time() - t0 < budget_s:
        cs = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(cs)
        cs *= dx

        # Compute which lags to adjust
        top_lag_indices = np.argsort(cs)[::-1][: min(5, len(cs) - 1)]

        for lag in top_lag_indices:
            if lag < len(h):
                grad = 1 - h[lag:]
            else:
                grad = 1 - h[:(len(h) - lag)]
            # Add zero-padding to align gradient with h
            grad = np.concatenate([np.zeros(lag), grad])
            grad = grad[:len(h)]
            grad = (grad - np.mean(grad)) / (np.std(grad) + 1e-8)

            # Try step size
            alpha = 0.03 * (1.0 - wait / 100.0)
            h_candidate = h + alpha * grad

            # Enforce constraints
            h_candidate = np.clip(h_candidate, 0, 1)
            total_sum = np.sum(h_candidate)
            if total_sum > 1e-6:
                h_candidate *= (n_points / 2.0) / total_sum

            c5_candidate = evaluate_erdos_solution(h_candidate)

            if c5_candidate < best_c5:
                wait = 0
                h = h_candidate
                best_c5 = c5_candidate
                best_h = h.copy()
                c5_history.append(best_c5)
            else:
                wait += 1

            if wait >= patience:
                break

        if wait >= patience:
            break

    if best_c5 >= 0.380876:
        h_opt = np.zeros_like(h)
        h_opt[n_points // 3 : 2 * n_points // 3] = 1.0
        evaluate_erdos_solution(h_opt)
        best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
