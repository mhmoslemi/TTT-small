```python
import numpy as np
import math
import time
from scipy.signal import correlate

def evaluate_C5(h, dx):
    n_points = len(h)
    # Compute max overlap in convolution mode="full"
    what = h
    ones = np.ones_like(h)
    ones_h = ones - h
    overlap = correlate(what, ones_h, mode="full", precision="double")
    overlap = overlap * dx
    return np.max(overlap)

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) > 1e-8:
        h_clipped = h_clipped * (target_sum / current_sum)
    return h_clipped

def compute_gradient(h, dx):
    n_points = len(h)
    what = h
    ones = np.ones_like(h)
    ones_h = ones - h
    overlap = correlate(what, ones_h, mode="full", precision="double")
    max_idx = np.argmax(overlap)

    # Determine positions involved in the max overlap
    k = max_idx - (n_points - 1)
    grad = np.zeros_like(h)
    if k <= 0:
        grad[:n_points + k] = h[k:] - 1
        grad[k:] = h[:n_points + k]
    else:
        grad[k:] = h[:-k] - 1
        grad[:-k] = h[k:]

    grad *= dx
    return grad

def run(seed=42, budget_s=1100, n_points=512, learning_rate=0.01, max_steps=10000, patience=100):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial = initial_h_values
        h = np.full(n_points, np.nan)
        start_idx = max(0, n_points // 2 - len(initial) // 2)
        end_idx = start_idx + len(initial)
        h[start_idx:end_idx] = initial
        h[np.isnan(h)] = 0.5
        h = project(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = project(h, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    step = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        step += 1
        if step > max_steps:
            break

        grad = compute_gradient(h, dx)
        h = h - learning_rate * grad
        h = project(h, target_sum)

        current_c5 = evaluate_C5(h, dx)
        if current_c5 < best_c5 - 1e-5:
            best_c5 = current_c5
            best_h = h.copy()
            if np.isnan(best_c5).any():
                break
            patience = 100
        else:
            learning_rate *= 0.9999  # adaptive step size
            patience -= 1
            if patience <= 0:
                break

    return best_h, best_c5, n_points
```
