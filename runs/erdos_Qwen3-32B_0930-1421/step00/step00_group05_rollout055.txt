import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    n_points = len(initial_h_values)
    h_opt = initial_h_values.copy()

    # Number of points to try during search
    try:
        hf = h_opt.astype(float)  # Convert to float for optimization
        hf = hf / hf.sum() * n_points / 2  # Normalize to 1-dx sum

        # Solver setup: MIP
        variables = cp.Variable(n_points)
        constraints = [
            variables >= 0,
            variables <= 1,
            cp.sum(variables) == n_points / 2.0,
        ]

        # Precompute all shifted versions of (1 - h)
        all_shifts = []
        for k in range(n_points):
            shifted = np.roll(1 - hf, k)
            shifted[:k] = 0.0  # Only consider shifted overlaps within [0, 2]
            all_shifts.append(shifted)

        all_shifts = np.array(all_shifts)
        obj = cp.max(cp.multiply(np.array(range(n_points)) / float(n_points), cp.sum(cp.multiply(hf, all_shifts), axis=1)))

        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.GUROBI, verbose=True, TimeLimit=1100 - (time.time() - start_time))

        h_opt = variables.value
        h_opt = np.clip(h_opt, 0.0, 1.0)

        def evaluate_c5(h):
            dx = 2.0 / len(h)
            overlaps = np.array([correlate(h, (1 - h[i:]), mode='valid') * dx for i in range(n_points)])
            return overlaps.max()

        c5_bound = evaluate_c5(h_opt)

    except Exception as e:
        print(e)
        c5_bound = 0.380876

    elapsed = time.time() - start_time
    print(f"Elapsed time: {elapsed:.2f}s")
    if elapsed < 1100:
        print(f"Time remaining: {budget_s - elapsed:.2f}s")

    return (h_opt, c5_bound, n_points)
