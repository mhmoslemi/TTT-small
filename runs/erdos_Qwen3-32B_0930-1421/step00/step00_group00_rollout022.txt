import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use the provided initial_h_values
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Project function to constraint: sum(h) = target_sum and values in [0,1]
    def project(h, target_sum, n_points):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        h_adjusted = h_clipped * scale
        # Final clipping to ensure numerical stability
        h_final = np.clip(h_adjusted, 0, 1)
        return h_final
    
    # Evaluate current C_5
    def evaluate_c5(h, dx):
        h_array = np.array(h)
        overlap = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = overlap * dx
        return np.max(c5)
    
    # Record best solution
    best_h = np.copy(h)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Optimization parameters
    alpha = 1.0  # Step size
    beta = 0.5   # Backtracking line search parameter
    max_line_search = 10
    max_iter = 1000
    iter = 0
    
    while time.time() - start_time < budget_s - 5 and iter < max_iter:
        iter += 1
        
        # Compute correlation and find max overlap shift
        h_array = np.array(h)
        overlap = np.correlate(h_array, 1 - h_array, mode="full")
        k_star = np.argmax(overlap)
        
        # Compute gradient for k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            offset_k = (i + k_star) % n_points  # Cyclic shift assumption
            if offset_k < n_points:
                grad[i] = - (1 - h_array[offset_k])
        
        # Gradient descent with backtracking line search
        for line_search in range(max_line_search):
            current_alpha = alpha * (beta ** line_search)
            new_h = h - current_alpha * grad
            new_h = project(new_h, target_sum, n_points)
            
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                break  # Successful step
        
        # Stop if no improvement in last 100 iterations
        if iter % 20 == 0 and abs(best_c5 - evaluate_c5(h, dx)) < 1e-6:
            break
    
    # Final projection
    best_h = project(best_h, target_sum, n_points)
    
    return best_h.tolist(), best_c5, n_points
