import numpy as np
import time
from scipy.signal import correlate


def compute_c5(h, dx):
    # Compute all correlation lags (h * (1 - h)), and return max value
    n = len(h)
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    corr = corr[int(n/2):int(3*n/2)] * dx
    return np.max(corr)


def project_l1_sum(h, n_points):
    # Project h to meet constraint: sum(h) = n_points / 2
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    scale = target_sum / current_sum
    return scale * h


def project_bounds(h):
    return np.clip(h, 0.0, 1.0)

        best_h = project_bounds(best_h)

    best_c5 = compute_c5(best_h, 2.0 / n_points)
    best_time = start

    learning_rate = 0.05
    patience = 60.0  # seconds with no improvement to allow for a small update

    while time.time() - start < budget_s:
        h = best_h.copy()
        h_flat = h.flatten()

        # Perform gradient descent on the lags that contribute to C5
        dx = 2.0 / n_points
        corr_result = correlate(h, 1 - h, mode='full')
        lag_values = corr_result[int(n_points / 2):int(3 * n_points / 2 - 1)]

        max_idx = np.argmax(lag_values)
        max_lag = max_idx

        # Get positions contributing to max overlap
        h1 = 1 - np.roll(h, -max_lag)
        grad = h1.copy()

        # Apply gradient update
        h_flat -= learning_rate * grad
        h = project_bounds(project_l1_sum(h_flat, n_points))
        current_c5 = compute_c5(h, dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
            best_time = time.time()

        elif time.time() - best_time > patience:
            # rebuild from lower C5 if stalled
            value_below = best_c5 - 0.01
            mask = lag_values <= value_below
            selected_idx = np.where(mask)[0]
            if len(selected_idx) > 0:
                selected_lag = np.random.choice(selected_idx)
                h1 = 1 - np.roll(h, -selected_lag)
                h_flat -= learning_rate * h1
                h = project_bounds(project_l1_sum(h_flat, n_points))

                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h
                    best_time = time.time()
