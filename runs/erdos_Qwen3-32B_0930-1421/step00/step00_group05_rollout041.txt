import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h = np.array(initial_h_values, dtype=float)
    n_points = h.size
    dx = 2.0 / n_points
    
    # Ensure h satisfies sum constraint
    target_sum = n_points / 2
    h /= h.sum() * dx
    h *= target_sum * dx

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    patience = 0

    while time.time() - start_time < budget_s:
        # Compute overlap integrals
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx

        # Identify the worst shift
        max_idx = np.argmax(overlap)
        k_max = max_idx - n_points  # shift value

        # Flip-range k to get valid x indices
        k_range = np.arange(n_points)
        shifted_k = max_idx - k_range
        valid_shifted_k = shifted_k[(shifted_k >= 0) & (shifted_k < n_points)]

        # Adjust h only at overlapping points
        grad = np.zeros_like(h)
        grad[valid_shifted_k] += (1 - 2 * h[valid_shifted_k])
        h -= learning_rate * grad

        # Project for constraints
        h = np.clip(h, 0, 1)

        # Normalize to satisfy sum constraint
        current_sum = h.sum()
        scaling_factor = target_sum / current_sum
        h *= scaling_factor

        # Evaluate C5 on original h
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        c5 = np.max(overlap)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Adjust learning rate
        if patience > 50:
            learning_rate *= 0.5
            patience = 0

    # Final evaluation
    overlap = np.correlate(best_h, 1 - best_h, mode="full")
    overlap = overlap * dx
    final_c5 = np.max(overlap)

    return (best_h.tolist(), final_c5, n_points)
