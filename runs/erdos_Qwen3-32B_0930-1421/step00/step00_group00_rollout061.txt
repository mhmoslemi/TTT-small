import numpy as np
import cvxpy as cp
import time

def project_h(h, n_points):
    # Enforce bounds
    h = np.clip(h, 0, 1)
    # Enforce sum constraint
    total = np.sum(h)
    h = h * (n_points / 2.0) / total
    return h

def compute_overlap(h, dx):
    # Compute the overlap for all shifts k
    return np.correlate(h, (1 - h), mode="full") * dx

    t0 = time.time()
    np.random.seed(seed)

    best_c5 = np.max(compute_overlap(h, dx))
    best_h = h.copy()

    best_h_best_c5 = (best_h.copy(), best_c5)

        L = compute_overlap(h, dx)

        h_opt = cp.Variable(n_points)

        constraints = [

        # Minimize the worst overlap for current candidate shift
        problem = cp.Problem(cp.Minimize(theta), constraints)
        problem.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.CVXOPT, verbose=False)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
