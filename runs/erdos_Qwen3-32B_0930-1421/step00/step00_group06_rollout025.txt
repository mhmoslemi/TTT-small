```python
import numpy as np
import cvxpy as cp
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' not in kwargs:
        raise ValueError("initial_h_values must be provided in keyword arguments.")
    h0 = kwargs['initial_h_values']
    n_points = len(h0)

    dx = 2.0 / n_points
    max_k = n_points  # maximum lag in the correlation

    # Use initial h and clip to guarantee feasibility
    h = np.clip(h0.copy(), 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)

    # Initialize dual variables
    nu = np.zeros(max_k)

    # Convex optimization setup
    h_var = cp.Variable(n_points)
    obj_proxy = cp.Minimize(np.sum(cp.log(nu + 1e-6)))
    constraints_proxy = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    prob = cp.Problem(obj_proxy, constraints_proxy)

    start_time = time.time()
    max_iter = 500
    best_c5 = 0.0
    best_h = h.copy()

    for it in range(max_iter):
        # Compute overlap for current test
        h_np = np.array(h)
        ones_minus_h = 1.0 - h_np
        correlate_result = np.correlate(h_np, ones_minus_h, mode="full")

        c5 = np.max(correlate_result[:n_points]) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Update dual variables
        for k in range(1, n_points):
            Ik = correlate_result[n_points - k]
            nu[k] = max(nu[k], Ik)

        # Solve convex problem
        h_init = h.copy()
        for i in range(n_points):
            constraints_proxy[0] = 0 <= h_var[i]
            constraints_proxy[1] = h_var[i] <= 1
        prob.solve(warm_start=True)  # reuse previous solution if possible
        h[:] = np.array(h_var.value).flatten()

        # Project to feasible space
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == 0.0:
            h[:] = h0.copy()
        else:
            h = h / np.sum(h) * (n_points / 2)

        # Check elapsed time and break early if needed
        if time.time() - start_time > budget_s:
            break

    # Final evaluation of best solution
    final_h = best_h
    final_c5 = 0.0
    h_correlate = final_h / dx
    ones_minus_h = 1.0 - h_correlate
    corr = np.correlate(h_correlate, ones_minus_h, mode="full")
    final_c5 = np.max(corr[:n_points]) * dx

    return final_h.tolist(), final_c5, n_points
```
