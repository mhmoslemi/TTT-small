import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    ones = np.ones_like(h)
    c = correlate(h, ones - h, mode='full') * dx
    return np.max(c)


def enforce_constraints(h, n_points):
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    return h


def run(seed=42, budget_s=1100, n_points=None):
    np.random.seed(seed)

    # Default to initial_h_values if available, adapt if n_points changes
    if 'initial_h_values' in globals():
        if n_points is None:
            h = np.array(initial_h_values)
            n_points = len(h)
        else:
            old_n = len(initial_h_values)
            if old_n == n_points:
                h = np.array(initial_h_values)
            else:
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, old_n), initial_h_values)
                h = enforce_constraints(h, n_points)
    else:
        # Construct basic even rectangle
        h = np.zeros(n_points)
        h[n_points // 4:3 * n_points // 4] = 2.0 / n_points * (n_points / 2)

    start_time = time.time()

    # Gradient descent parameters
    learning_rate = 0.01
    restart_period = 50
    grad_steps = 200
    simplex_steps = 10
    best_c5 = np.inf
    best_h = h

    dx = 2.0 / n_points
    ref = np.arange(-n_points + 1, n_points)

    while time.time() - start_time < budget_s:
        for step in range(grad_steps):
            ones = np.ones_like(h)
            corr = correlate(h, ones - h, mode='full')
            dx_corr = corr * dx

            # Select k with max value
            k_max = np.argmax(dx_corr)
            ones_h1 = (ones - h[ref[k_max]::1]).copy()
            ones_h1[ref[k_max]::1] = 1 - h[ref[k_max]::1]

            # Gradient of inner product h * (1 - h_{shifted})
            dL_dh = 1 - 2 * h[ref[k_max]::1]
            grad = np.zeros_like(h)
            grad[ref[k_max]::1] = dL_dh

            h += learning_rate * grad
            h = enforce_constraints(h, n_points)

            current_c5 = evaluate_c5(h, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        if step % restart_period == 0:
            # Simplex exploration
            simplex = np.array([
                h.copy(),
                h.copy() + 0.1,
                h.copy() - 0.1
            ])
            for i in range(3):
                simplex[i] = enforce_constraints(simplex[i], n_points)

            scores = [evaluate_c5(simplex[i], n_points) for i in range(3)]
            best_idx = np.argmin(scores)
            h = simplex[best_idx].copy()
            learning_rate *= 0.5

        if learning_rate < 1e-4:
            break

    final_c5 = evaluate_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
