import scipy.optimize
import time

def evaluate_objective(h, dx):
    # Convert h into a full array with zero-padding to handle full convolution
    n = len(h)
    h_full = np.concatenate([h, np.zeros(n - 1)])
    overlap = np.correlate(h_full, 1 - h_full, mode='valid')
    return np.max(overlap * dx)
