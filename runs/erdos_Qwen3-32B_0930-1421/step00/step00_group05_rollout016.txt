import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial_h_values as starting point
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    # Ensure sum constraint and bounds
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum()

    learning_rate = 0.01
    beta = 10.0
    iters = 0
    best_h = h.copy()
    best_c5 = float("inf")

    while time.time() - start_time < budget_s:
        iters += 1

        # Compute all overlapping integrals for each shift
        cc = np.correlate(h, 1 - h, mode='full') * dx

        # Evaluate current C5
        c5_current = np.max(cc)

        # Improve if necessary
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h[:] = h.copy()

        # Compute gradient using log-sum-exp formulation
        scores = cc + 1e-8  # tiny shift to avoid log(0)
        grad = np.zeros_like(h)
        for k in range(n_points):
            grad[:] += np.correlate(
                h, -2 * (1 - 2 * h) * ((1 - h) * dx), mode='full'
            )[k] * scores[k] * (1 - h[(n_points - k) % n_points]) if k not in [0, n_points // 2] else 0
        grad /= n_points

        # Normalize gradient magnitude
        grad = grad / (np.linalg.norm(grad) + 1e-8)

        # Update h
        h = h - learning_rate * grad

        # Projection step
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

        # Adjust learning rate and beta
        if iters % 50 == 0:
            learning_rate *= 0.9
        if iters % 100 == 0:
            beta = min(beta * 1.1, 1e4)

        # Early exit if time is exhausted
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final verification and cleanup
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / h.sum()
    c5_final = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    return best_h, best_c5, n_points
