import numpy as np
import time
import cvxpy as cp
from math import sqrt

try:
    n_points_initial = len(initial_h_values)
except:
    n_points_initial = 45  # Fallback to base resolution if no parent given

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.amax(corr * dx)

def project_l1(h, n_points):
    total = h.sum()
    needed = n_points / 2
    delta = (total - needed) / n_points
    return np.clip(h - delta, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = n_points_initial
    try:
        h = initial_h_values[:]  # Use parent's construction as starting point
    except:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        h = project_l1(h, n_points)

    best_c5 = evaluate_c5(h, n_points)
    best_h = h.copy()
    max_iter = 10000

    t0 = time.time()
    for iter in range(max_iter):
        if time.time() - t0 > budget_s:
            break

        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)

        # Convex subproblem formulation
        h_var = cp.Variable(n_points)
        overlap = cp.correlate(h_var, 1 - h_var, mode='full')[k_max_idx]
        objective = cp.Minimize(overlap * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)

        try:
            prev_h = h.copy()
            problem.solve(solver=cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.OSQP,
                          verbose=False,
                          warm_start=True)
            if h_var.value is not None:
                h = h_var.value  # Project and clip back to feasible region
                h = np.maximum(0.0, np.minimum(1.0, h))
                h = project_l1(h, n_points)
                h = h / (2 * dx) * np.sum(h) / (n_points / 2)  # Maintain goal constraint
        except:
            h = 0.5 * (h + prev_h)
            h = project_l1(h, n_points)

        # Gradient noise & momentum updates to escape local optima
        h += (np.random.normal(0, 0.01, n_points) - 0.05) * (iter % 5 == 0)
        if iter % 10 == 0:
            idxs = np.argsort(np.arange(n_points) % 20)
            h[idxs[:5]] = 1.0 - h[idxs[:5]]

        # Evaluation and constraints
        h = np.clip(h, 0, 1)
        h = project_l1(h, n_points)
        current_c5 = evaluate_c5(h, n_points)

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        if iter % 10 == 0:
            print(f"Iteration {iter}, best C5: {best_c5:.6f}, h: {h[:20]}")

    return best_h, best_c5, n_points
