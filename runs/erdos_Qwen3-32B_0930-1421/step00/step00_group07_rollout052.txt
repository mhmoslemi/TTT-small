import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    initial_n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, copy=True)
    n_points = initial_n_points
    dx = 2.0 / n_points

    def intervals_from_h(h, n_points, dx):
        x = np.linspace(0, 2, n_points)
        bins = []
        current_interval = None
        intervals = []

        for i, val in enumerate(h):
            if val > 0.5 and current_interval is None:
                current_interval = [x[i]]
            elif val < 0.5 and current_interval is not None:
                intervals.append((current_interval[0], x[i]))
                current_interval = None
        if current_interval is not None:
            intervals.append((current_interval[0], x[-1]))
        return intervals

    def h_from_intervals(intervals, n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for a, b in intervals:
            mask = np.logical_and(x >= a, x <= b)
            h += mask
        h = np.clip(h, 0, 1)
        return h

    def compute_overlap(h, dx):
        h1 = h
        h2 = 1 - h
        corrs = correlate(h1, h2, mode='full')
        widths = np.linspace(-2 + dx, 2 - dx, len(corrs))
        valid_mask = (widths >= -1) & (widths <= 1)
        valid_corrs = corrs[valid_mask] * dx
        return np.max(valid_corrs)

    # Convert initial_h_values to intervals for further processing
    intervals = intervals_from_h(h_values, n_points, dx)

    start_time = time.time()
    best_c5 = compute_overlap(h_values, dx)
    best_intervals = intervals.copy()
    
    # Annealing parameters
    T = 0.5
    T_decay = 0.99
    delta = 0.05
    delta_decay = 0.999
    max_trials = 2000
    no_improvement_limit = 50
    no_improvement_count = 0

    while time.time() - start_time < budget_s and no_improvement_count < no_improvement_limit and max_trials > 0:
        max_trials -= 1

        # Propose a change to one interval
        i = np.random.randint(len(intervals))
        a_i, b_i = intervals[i]
        r = np.random.rand()

        if r < 0.33:  # Shift left boundary
            new_a = np.clip(a_i + np.random.uniform(-delta, delta), 0.0, b_i - 1e-6)
            intervals[i] = (new_a, b_i)
        elif r < 0.66:  # Shift right boundary
            new_b = np.clip(b_i + np.random.uniform(-delta, delta), a_i + 1e-6, 2.0)
            intervals[i] = (a_i, new_b)
        else:  # Split/merge interval
            if len(intervals) > 1:
                if np.random.rand() < 0.5:
                    # Reduce this interval, increase next
                    new_len = intervals[i][1] - intervals[i][0] + (np.random.rand() * 0.02)
                    if i < len(intervals) - 1:
                        intervals[i] = (intervals[i][0] + new_len, intervals[i][1])
                        intervals[i + 1] = (intervals[i + 1][0], intervals[i + 1][1] + new_len)
                else:
                    # Increase this interval, reduce next
                    new_len = intervals[i][1] - intervals[i][0] - (np.random.rand() * 0.02)
                    if i < len(intervals) - 1:
                        intervals[i] = (intervals[i][0] - new_len, intervals[i][1])
                        intervals[i + 1] = (intervals[i + 1][0], intervals[i + 1][1] - new_len)

        # Ensure intervals are valid (no overlaps, ordered)
        intervals.sort()
        new_intervals = []
        i = 0
        while i < len(intervals):
            a, b = intervals[i]
            j = i + 1
            while j < len(intervals) and intervals[j][0] <= b:
                j += 1
            new_intervals.append((a, max(b, intervals[j - 1][1])))
            i = j

        intervals = new_intervals

        # If no intervals, reject
        if not intervals:
            intervals = best_intervals.copy()
            continue

        h_new = h_from_intervals(intervals, n_points)
        c5_new = compute_overlap(h_new, dx)

        # Metropolis criterion
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            best_c5 = c5_new
            best_intervals = intervals.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        T *= T_decay
        delta *= delta_decay

    # Final validation
    h_values = h_from_intervals(best_intervals, n_points)
    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1.0) > 1e-6:
        # Normalize to 1
        h_values = h_values / (np.sum(h_values) * dx) * (1 / dx)
        h_values = np.clip(h_values, 0, 1)

    c5_bound = compute_overlap(h_values, dx)
    return h_values, c5_bound, n_points
