import numpy as np
import time
import scipy.fft as fft

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use 84 as base n_points for consistency and compatibility with metadata
    n_points = 84 * 2  # 168 offers richer resolution for optimization
    dx = 2.0 / n_points

    # Initialize with given `initial_h_values` if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Build uniform step function scaled to meet integral constraint
        h = np.ones((n_points,), dtype=float) / n_points * n_points / 2

    # Normalize to satisfy ∫₀² h(x) dx = 1
    h /= np.sum(h) * dx
    h *= n_points / 2  # ensures ∑ h ≈ n_points / 2

    # Optimization parameters
    alpha = 0.01
    patience = 100
    best_c5 = float('inf')
    counter = 0
    start_time = time.time()

    # Main optimization loop
    while time.time() - start_time < budget_s:
        # Fourier transform preparations
        h_fft = fft.fft(h)
        z = np.zeros_like(h_fft)
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(h, -k)
            if k < 0:
                shifted = np.roll(h, -k + n_points)
            overlap = np.dot(h, 1 - shifted) * dx
            idx = n_points + k if k >= 0 else k
            z[idx] = overlap

        overlaps = np.imag(fft.ifft(h_fft * np.conj(fft.fft(1 - h))))

        # Compute current max overlap
        c5 = np.max(overlaps * dx)

        if c5 < best_c5:
            best_c5 = c5
            counter = 0
        else:
            counter += 1

        # Optional: early stopping
        if counter > patience:
            break

        # Gradient update via difference across shifts
        gradient = np.imag(
            fft.ifft(
                h_fft * np.conj(fft.fft(1 - h))
            )
        )
        gradient /= dx
        gradient *= np.random.uniform(0.95, 1.05)
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        # Enforce ∑ h = n_points / 2 (fix integral)
        h *= (n_points / 2) / (np.sum(h) + 1e-12)

    # Final verification
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / (np.sum(h) + 1e-12)

    return (h.tolist(), best_c5, n_points)
