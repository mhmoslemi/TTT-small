import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

# Helper to evaluate the C5 constant
def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.asarray(h, dtype=np.float64)
    h_ = h - h**2
    corr = correlate(h, h_, mode='full')
    c5 = np.max(np.abs(corr)) * dx
    return c5

# Ensure constraints: sum(h) = n_points/2 and h ∈ [0,1]
def project_solution(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s) < 1e-8:
        return h  # prevent division by zero
    h *= n_points / (2 * s)
    return h

    for i in range(pop_size):
        h = best_h + mut_strength * (np.random.rand(n_points) - 0.5)
        h = project_solution(h, n_points)
        pop[i] = h

    start_time = time.time()
    for gen in range(max_gen):
        # Evaluate population
        fitness = []
        for h in pop:
            c5 = evaluate_erdos_solution(h)
            violations = 0.0
            s = np.sum(h) - n_points / 2.0
            if abs(s) > 1e-10:

            violations += np.sum(np.clip(h, 0, 1) - h) ** 2  # Penalize out-of-bounds
            fitness.append(-c5 - 1000.0 * violations)  # Maximize fitness to minimize C5

        # Selection: top 20%
        sorted_indices = np.argsort(fitness)[::-1]
        elites = [pop[i] for i in sorted_indices[:int(pop_size * 0.2)]]
        top_fitness = [fitness[i] for i in sorted_indices[:int(pop_size * 0.2)]]

                parent1 = elites[idx1].copy()
                parent2 = elites[idx2].copy()
                # Arithmetic crossover
                alpha = np.random.rand(n_points)
                child = alpha * parent1 + (1 - alpha) * parent2
                child = project_solution(child, n_points)
                new_pop.append(child)
            else:
                idx = np.random.randint(len(elites))
                parent = elites[idx].copy()
                child = parent + mut_strength * (np.random.rand(n_points) - 0.5)
                child = project_solution(child, n_points)
                new_pop.append(child)

        # Combine elites and new_pop
        pop = np.array(elites + new_pop)
        if time.time() - start_time > budget_s - 2.0:
            break

    # Final post-processing: refine best solution using BFGS
    def cost(h_vec):
        h = h_vec.reshape((n_points,))
        h = project_solution(h, n_points)

    final_h = project_solution(final_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
