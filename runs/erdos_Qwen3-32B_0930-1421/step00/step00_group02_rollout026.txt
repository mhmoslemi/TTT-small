import numpy as np
import time
import cvxpy as cp
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    t0 = time.time()

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        n_points = 45  # start with 45 points
        h_initial = np.zeros(n_points)
        h_initial[n_points // 2 - 2:n_points // 2 + 2] = 0.25
        h_initial = h_initial / h_initial.sum() * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)

    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = np.copy(h_initial)

    def compute_c5(h):
        correlate = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(correlate) * dx
        return max_c5

    def optimize_phase1(h):
        h_vars = cp.Variable(n_points)
        obj = cp.Minimize(cp.Max(cp.abs(cp.fft.fft(h_vars * (1 - h_vars)))))  # minimize max Fourier coefficient
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCIPY, verbose=False)
        h_opt = np.clip(np.ascontiguousarray(h_vars.value), 0, 1)
        return h_opt

    def optimize_phase2(h, iter_steps=1500, learn_rate=0.01):
        h_np = np.copy(h).astype(np.float64)
        c5_best = compute_c5(h_np)
        best_h_np = np.copy(h_np)

        for j in range(iter_steps):
            delta = np.zeros_like(h_np)
            cor = np.correlate(h_np, 1 - h_np, mode='full')
            max_idx = np.argmax(np.abs(cor))
            valid_idx = max_idx - n_points + 1

            if 0 <= valid_idx < n_points:
                delta[valid_idx] = -1
                delta[valid_idx % n_points] = +1

            h_np += delta * learn_rate
            h_np = np.clip(h_np, 0, 1)
            s = h_np.sum()
            h_np = h_np / s * (n_points / 2)

            c5_current = compute_c5(h_np)
            if c5_current < c5_best:
                c5_best = c5_current
                best_h_np = np.copy(h_np)
                if c5_best < best_c5:
                    best_c5 = c5_best
                    best_h[:] = best_h_np
            if time.time() - t0 > budget_s - 5:
                break
        return best_h_np

    best_c5 = compute_c5(h_initial)

    # Phase 1: Frequency domain optimization
    h_fft_opt = optimize_phase1(h_initial)
    h_fft_opt = h_fft_opt / h_fft_opt.sum() * (n_points / 2)
    h_fft_opt = np.clip(h_fft_opt, 0, 1)

    # Phase 2: Time domain refinement
    h_final = optimize_phase2(h_fft_opt)

    # Final evaluation
    c5_final = compute_c5(h_final)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h[:] = h_final

    return (best_h, best_c5, n_points)
