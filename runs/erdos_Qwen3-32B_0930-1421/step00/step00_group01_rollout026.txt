import numpy as np
import scipy.optimize
import time

from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values as init_h

def run(budget_s=1100, **kwargs):
    start_time = time.time()
    max_iter = 1000
    tolerance = 1e-6
    batch_size = 10

    h = np.copy(init_h)
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.copy().reshape(n_points)
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
        h_sum = h.sum()
        h = h / h_sum * (n_points / 2.0)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5

    def constraint_sum(h_flat):
        return h_flat.sum() - n_points / 2.0

    def constraint_bounds(h_flat):
        return np.append(h_flat, -h_flat)

    bounds = np.array([[0.0, 1.0]] * n_points).ravel()

    # Initial guess
    h_flat = h.reshape(n_points)

    # Optimization
    res = scipy.optimize.minimize(
        objective,
        h_flat,
        method="SLSQP",
        bounds=bounds,
        constraints=[
            {"type": "eq", "fun": constraint_sum},
            {"type": "ineq", "fun": lambda x: x},
            {"type": "ineq", "fun": lambda x: 1.0 - x},
        ],
        options={
            "maxiter": max_iter,
            "ftol": tolerance,
            "disp": False,
        },
    )

    if res.success:
        h_opt = res.x.reshape(n_points)
    else:
        h_opt = np.copy(h)

    h_opt[h_opt > 1.0] = 1.0
    h_opt[h_opt < 0.0] = 0.0
    h_opt_sum = h_opt.sum()
    h_opt = h_opt / h_opt_sum * (n_points / 2.0)

    c5_bound = evaluate_erdos_solution(h_opt, visualize=False)

    return (h_opt, c5_bound, n_points)
