import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    kh = np.concatenate((h, h, h, h, h))  # Extended to avoid edge effects
    one_minus_kh = 1 - kh
    corr = correlate(kh, one_minus_kh, mode='valid') * dx
    c5_value = np.max(corr)
    return c5_value

            if i + k_max - 1 < len(h):
