import numpy as np
import scipy.optimize as so
import time
import math

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    # Extend h to handle full convolution with wrap-around
    h_input = np.concatenate((h, np.zeros(n_points - 1)))
    h_ones = np.zeros_like(h_input)
    for i in range(n_points):
        h_ones += h[i] * np.roll(h_input, i)
    overlap = 1 - h_ones[:n_points] - h_input
    overlap_integral = overlap * dx
    return max(overlap_integral)

def objective(h, n_points):
    return compute_c5(h, n_points)

def constraint_sum_one(h, n_points):
    return np.sum(h) - n_points / 2

def bounds_and_constraints(h, n_points):
    # Box bounds [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Linear equality constraint: sum(h) = n_points / 2
    cons = [{'type': 'eq', 'fun': lambda x: constraint_sum_one(x, n_points)}]
    return bounds, cons

def gradient_objective(h, n_points):
    dx = 2.0 / n_points
    # Avoid recomputation
    h_extended = np.concatenate((h, np.zeros(n_points - 1)))
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Compute max overlap at all k
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            c = h_extended[k:] * (1 - h_extended[k:] - (k < n_points))
            overlaps[k] = c[:n_points - k].sum()
        overlaps *= dx
        max_k = overlaps.argmax()
        # Backprop gradient only at the contributing positions
        for shift in range(n_points - max_k):
            grad[i] += (1 - h[max_k + shift]) * dx if i == max_k + shift else h[i] * dx
    return grad

def trust_region_update(h, gradient, trust_radius=0.01, clip=True):
    new_h = h - trust_radius * gradient
    if clip:
        new_h = np.clip(new_h, 0.0, 1.0)
        # Renormalize to satisfy the sum constraint
        scaling_factor = (np.sum(h) / 2.0) / np.sum(new_h)
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h
