import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_full = np.tile(h, 4)
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return np.max(np.array(corrs)) * dx

def smooth_max(values, beta=100):
    return np.log(np.sum(np.exp(beta * values))) / beta

def objective_smooth(h_vec, n_points, beta=100):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    h_full = np.tile(h, 4)
    corrs = []
    for k in range(1, n_points):
        if k >= len(h):
            break
        corr = np.dot(h[:n_points - k], 1 - h[k:k + n_points - k])
        corrs.append(corr)
    for k in range(1, n_points):
        corr = np.dot(h[k:], 1 - h[:n_points - k])
        corrs.append(corr)
    return smooth_max(np.array(corrs) * dx, beta)

def objective_raw(h_vec, n_points):
    return compute_c5(h_vec, n_points)

def enforce_normalization(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / total
    return np.clip(h, 0.0, 1.0)

def initial_variation(h0, n_points, seed):
    np.random.seed(seed)
    noise = np.random.uniform(-0.1, 0.1, n_points)
    h = enforce_normalization(h0 + noise, n_points)
    return h

def perturbation_search(h_best, n_points, budget, improve_func, initial_beta):
    h = h_best.copy()
    h = enforce_normalization(h, n_points)
    best_val = compute_c5(h, n_points)
    beta = initial_beta
    temp = 2.0
    time_start = time.time()
    while time.time() - time_start < budget:
        h_p = h + np.random.normal(0, 0.05, n_points)
        h_p = enforce_normalization(h_p, n_points)
        val_p = compute_c5(h_p, n_points)
        if val_p < best_val or np.random.rand() < np.exp((best_val - val_p) * beta / temp):
            h = h_p.copy()
            best_val = val_p
        beta = min(beta + 0.001, 100)
        temp = max(temp * 0.995, 0.05)
    return h, best_val

def run(seed=42, budget_s=1100, init_h=None):
    np.random.seed(seed)
    if init_h is None:
        init_h = np.random.rand(47)  # Fallback to random if not provided
        init_h = enforce_normalization(init_h, 47)

    n_points = len(init_h)
    time_start = time.time()
    h_init = init_h.copy()
    h_best = h_init.copy()

    # Perturbation-based exploration
    for trial in range(5):
        h_trial = initial_variation(h_init, n_points, seed + trial)
        h_trial = opt.minimize(lambda x: objective_smooth(x, n_points), h_trial,
                               method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                               options={'maxiter': 1000, 'eps': 1e-4}).x
        h_trial = enforce_normalization(h_trial, n_points)
        val = compute_c5(h_trial, n_points)
        if val < compute_c5(h_best, n_points):
            h_best = h_trial.copy()

    # Global search
    h_best, val_best = perturbation_search(h_best, n_points, budget_s - (time.time() - time_start), objective_smooth, 1.0)

    val_best = compute_c5(h_best, n_points)
    return h_best, val_best, n_points
