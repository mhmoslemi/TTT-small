import numpy as np
import time

from scipy.signal import correlate


def evaluate_erdos_c5(h):
    n = len(h)
    dx = 2.0 / n
    h = h / dx  # Convert to unit measure for convolution
    _, _, _, max_vals = correlate_two_seq(h, 1 - h)
    return max_vals * dx


def correlate_two_seq(a, b):
    """Compute the full convolution and max value without scipy dependency.
    Returns (indices where max was found, values of overlaps, and the max)."""
    n_a = len(a)
    n_b = len(b)
    n_out = n_a + n_b - 1
    result = np.zeros(n_out)
    for k in range(n_out):
        for i in range(n_a):
            j = k - i
            if 0 <= j < n_b:
                result[k] += a[i] * b[j]
    max_idx = np.argmax(result)
    max_val = result[max_idx]
    return np.arange(n_out), result, max_idx, max_val

    # Gradient descent loop with restart
    while time.time() - start_time < budget_s and step < max_iter:

        grad = np.zeros_like(h)

            grad[i] = (c5_up - current_c5) / eps

        # Move in the negative gradient direction
        h = h - 0.01 * grad
        h = proj_simplex(h, dx, 1.0)

        # Update best solution
        c5 = evaluate_erdos_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Add candidate to simplex
        simplex_history.append(h.copy())
        if len(simplex_history) >= 5:
            simplex_history = []
            # Restart: pick best from simplex
            simplexes = [s for s in simplex_history]
            results = [evaluate_erdos_c5(s) for s in simplexes]
            best_idx = np.argmin(results)
            h = simplexes[best_idx]
            # Perturb slightly to escape local minima
            h += np.random.uniform(-simplex_radius, simplex_radius, h.shape) * h
            h = np.clip(h, 0, 1)
            h = proj_simplex(h, dx, 1.0)

    return best_h.tolist(), best_c5, n_points


def proj_simplex(x, dx, target_sum):
    """Project x onto [0,1]^n, sum x * dx == target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = x.sum() * dx
    if not np.isclose(current_sum, target_sum):
        s = x.sum()
        scaling = target_sum / dx / s if not s == 0 else 0
        x = np.clip(x * scaling, 0, 1)
    return x
