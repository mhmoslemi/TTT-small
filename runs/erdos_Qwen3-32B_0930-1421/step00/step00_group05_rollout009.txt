import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = float("inf")
    h = h.astype(float)

    learning_rate = 0.1

    while time.time() - start_time < budget_s:
        # Compute overlap integrals
        correlate = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(abs(correlate))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_index = np.argmax(np.abs(correlate))
        k = max_index - (n_points - 1)

        # Compute gradient of the overlap at k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] = (1 - 2 * h[i]) * (1 - h[j])

        # Gradient descent update
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

        # Enforce the integral constraint
        total = h.sum()
        if total > 1e-8:  # Avoid divide-by-zero
            h = h * (n_points / 2) / total

        # Reduce learning rate for small changes
        learning_rate *= 0.997

        # Safety stop
        if learning_rate < 1e-4:
            break

    # Final clip and constraint fix
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / h.sum()
    final_c5 = np.max(np.abs(np.correlate(h, 1 - h, mode="full") * dx))
    return best_h, best_c5, n_points
