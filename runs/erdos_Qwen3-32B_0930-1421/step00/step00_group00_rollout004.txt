import numpy as np
import time
import evaluate_erdos_solution as evaluator

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    scale_factor = target_sum / current_sum
    h *= scale_factor
    return np.clip(h, 0, 1)

def run(budget_s=1100, initial_alpha=1.0, beta=0.5, max_search=10, max_iter=1000):
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    for iter in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

        # Compute cross-correlation for fold and shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        window =corr
        max_value = window[max_corr_idx]

        # Shift k* corresponds to index of max overlap
        k = max_corr_idx - (n_points - 1)

        if n_points <= np.abs(k) >= 2 * n_points:
            continue

        # Compute gradient
        gradient = np.zeros_like(h)
        shift_mask = np.zeros_like(h)
        pos_idx = np.clip(np.arange(n_points) - k, 0, n_points - 1)
        shift_mask[pos_idx] = 1
        gradient = 1 - h[pos_idx] 
        gradient *= (1 - h[pos_idx])  # Ensure direction for descent

        # Line search
        alpha = initial_alpha
        found_step = False
        for _ in range(max_search):
            next_h = h - alpha * gradient
            next_h = project_h(next_h, target_sum)
            next_c5 = compute_c5(next_h, dx)

            if next_c5 < best_c5:
                h = next_h
                best_h = h.copy()
                best_c5 = next_c5
                found_step = True
                break
            alpha *= beta

        # Track progress and finalize
        if not found_step:
            break  # No improvement found, escape loop

    return best_h, best_c5, n_points
