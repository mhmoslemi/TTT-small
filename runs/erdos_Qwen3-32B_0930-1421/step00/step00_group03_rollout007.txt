import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # Load initial backbone
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = h.shape[0]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def c5(h):
        # Evaluate the Erdos constant bound
        neg_h = 1.0 - h
        c = correlate(h, neg_h, mode='full')
        c5 = np.max(c) * dx
        return c5

    def feasibility_projection(h, target_sum):
        h = np.clip(h, 0, 1)
        err = h.sum() - target_sum
        if abs(err) < 1e-9:
            return h
        # Scale the largest values if over, smallest if under
        if err > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        deltat = abs(err) / len(idx)
        h[idx] -= np.sign(err) * deltat
        h = np.clip(h, 0, 1)
        return h

    best_h = h.copy()
    best_c5 = c5(best_h)
    best_time = time.time()

    # Parameters for SA
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    max_sa_steps = 1000
    no_improvement_patience = 100

    # Parameters for GD
    eta = 0.01
    max_gd_steps = 500

    def gradient(h):
        neg_h = 1.0 - h
        full_corr = correlate(h, neg_h, mode='full')
        max_index = np.argmax(full_corr)
        if max_index < n_points:
            shift = max_index
            grad = -neg_h[shift:] + (1 - h[:n_points - shift])  # Derivative for max index case
        else:
            shift = max_index - n_points
            grad = neg_h[:n_points - shift] - (1 - h[shift:])  # Derivative for min index case
        grad = np.pad(grad, (0, n_points - len(grad)), mode='constant') if len(grad) < n_points else grad
        return grad

    for phase in range(2):
        start_time = time.time()
        h = np.clip((best_h + 0.05 * np.random.randn(n_points)), 0, 1)
        h = feasibility_projection(h, target_sum)
        current_c5 = c5(h)
        no_imp = 0

        for i in range(max_sa_steps):
            if time.time() - t_start > budget_s - 30:
                break
            h_pert = h + sigma * np.random.randn(n_points)
            h_pert = feasibility_projection(h_pert, target_sum)
            c5_new = c5(h_pert)
            if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / max(T, 1e-6)):
                h = h_pert
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    best_time = time.time()
                    no_imp = 0
                else:
                    no_imp += 1
                sigma = max(1e-3, sigma * 0.995)
            T *= alpha
            if no_imp > no_improvement_patience:
                h = np.clip(best_h + np.random.randn(n_points) * 0.1, 0, 1)
                h = feasibility_projection(h, target_sum)
                no_imp = 0

        # Gradient descent refinement
        for i in range(max_gd_steps):
            if time.time() - t_start > budget_s - 30:
                break
            grad = gradient(h)
            h = h - eta * grad
            h = feasibility_projection(h, target_sum)
            c5_new = c5(h)
            if c5_new >= current_c5:
                eta *= 0.95
            else:
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    best_time = time.time()

        elapsed = time.time() - start_time

    h_values = np.round(best_h * 16000).astype(np.int32)  # Multiply by 16000 and round to fitting int
    c5_bound = best_c5
    return h_values, c5_bound, n_points
