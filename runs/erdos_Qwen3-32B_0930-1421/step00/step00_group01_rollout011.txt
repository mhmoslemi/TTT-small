import numpy as np
from scipy.optimize import minimize, Bounds
import time

def correlate_1d(h, dx):
    h_flattened = h.flatten()
    one_minus_h = 1.0 - h_flattened
    corr = np.correlate(h_flattened, one_minus_h, mode='full')
    corr *= dx
    return np.max(corr)

def objective(h_vec, n_points):
    h = h_vec.reshape((n_points,))
    dx = 2.0 / n_points
    return correlate_1d(h, dx)

    return np.sum(h_vec) - expected_sum
