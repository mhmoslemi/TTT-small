import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h_values = np.array(initial_h_values)  # Provided by the environment

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)

    # Initialize warm starts
    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    # Convolution expressions for all k overlaps
    max_overlap = 0
    for k in range(1, n_points):
        shifted_h = np.roll(h, -k)
        overlap = cp.sum(h * (1 - shifted_h) * dx)
        max_overlap = cp.Max(max_overlap, overlap)

    obj = cp.Minimize(max_overlap)
    prob = cp.Problem(obj, constraints)

    start_time = time.time()
    prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=50, warm_start=True)
    best_h = np.clip(h.value, 0, 1)
    best_c5 = prob.value

    # Phase 2: Restart and refine
    best_solution = best_h
    best_bound = best_c5
    while time.time() - start_time < budget_s - 10:
        # Perturb and warm start
        perturbation = 0.01 * np.random.randn(n_points)
        h.value = np.clip(best_solution + perturbation, 0, 1)
        h.value = best_solution * 0.99 + h.value * 0.01  # Conserve volume
        h.value = h.value / np.sum(h.value) * (n_points / 2)  # Normalize for constraint

        prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=50, warm_start=True)

        if prob.status in ["optimal", "optimal_inaccurate"]:
            new_h = np.clip(h.value, 0, 1)
            new_c5 = prob.value
            if new_c5 < best_bound:
                best_solution = new_h
                best_bound = new_c5

    final_h = best_solution
    final_c5 = evaluate_erdos_solution(final_h)
    assert abs(np.sum(final_h) - n_points / 2) < 1e-6, "Feasibility violated: integral constraint"
    assert np.all(np.isfinite(final_h)), "NaN detected in solution"
    return (final_h.tolist(), final_c5, n_points)
