import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=42, pop_size=20, n_generations=200, mutation_rate=0.1):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    target_integral = n_points / 2.0

    def compute_c5(h):
        h_cor = np.correlate(h, 1 - h, mode="full")
        return np.max(h_cor) * dx

    def clip_and_renormalize(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h.copy()
        scale = target_integral / current_sum
        h_clipped_scaled = np.clip(h_clipped * scale, 0, 1)
        return h_clipped_scaled

    def initialize_population():
        population = []
        if 'initial_h_values' in globals() and initial_h_values is not None:
            population.append(initial_h_values)
        while len(population) < pop_size:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * target_integral
            h = clip_and_renormalize(h)
            population.append(h)
        return np.array(population)

    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), size=3)
        return np.argmax(fitnesses[indices])

    def crossover(p1, p2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, p1, p2)
        child = clip_and_renormalize(child)
        return child

    def mutate(h):
        indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
        delta = np.random.uniform(-0.1, 0.1, size=len(indices))
        h[indices] += delta
        h = clip_and_renormalize(h)
        return h

    population = initialize_population()
    fitnesses = [compute_c5(h) for h in population]
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = np.min(fitnesses)

    generation = 0
    while generation < n_generations and time.time() - start_time < budget_s:
        new_population = [best_h]
        for _ in range(pop_size - 1):
            idx1 = tournament_selection(population, fitnesses)
            idx2 = tournament_selection(population, fitnesses)
            child = crossover(population[idx1], population[idx2])
            child = mutate(child)
            new_population.append(child)
        population = np.array(new_population)
        fitnesses = [compute_c5(h) for h in population]
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
        generation += 1

    return best_h, best_c5, n_points
