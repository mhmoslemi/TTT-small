import time
import numpy as np
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    # Project to constraint: sum(h) == n_points / 2
    def project_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        h = np.clip(h * scale, 0, 1)
        return h

    best_h = h.copy()
    best_c5 = np.inf

    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]

    while time.time() - t0 < budget_s - 5:
        # Compute the convolution overlaps
        one_minus_h = 1 - h
        corr = scipy.signal.correlate(h, one_minus_h, mode='full')
        corr = corr[n_points - 1: 2 * n_points - 1]
        corr *= dx

        # Find the worst (max) overlap
        k_star = np.argmax(corr)
        c5 = corr[k_star]

        if c5 < best_c5 and np.all(0 <= h) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-6:
            best_c5 = c5
            best_h = h.copy()

        # Build the gradient at the max overlap index k_star
        grad = np.zeros(n_points)
        grad[:-k_star] += one_minus_h[k_star:]
        grad[k_star:] -= h[:-k_star]

        objective = cp.Minimize(cp.sum_squares(h_var - (h - 0.1 * grad)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=50)
            h = h_var.value
            h = np.nan_to_num(h)
        except Exception:
            h = h - 0.1 * grad
        h = project_h(h)

    return best_h, best_c5, n_points
