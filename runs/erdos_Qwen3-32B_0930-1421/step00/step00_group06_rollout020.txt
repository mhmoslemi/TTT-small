import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    np.random.seed(seed)
    h0 = np.array(initial_h_values)

    n_points = len(h0)
    dx = 2.0 / n_points

    def objective_and_constraints(h):
        h = h.reshape(n_points)

        # Ensure non-negativity and normalize
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / np.clip(np.sum(h), 1e-10, np.inf)

        # Compute all overlap integrals
        overlap = np.zeros(2 * n_points)
        for k in range(1, n_points):
            idx0 = k % n_points
            overlap_idx = n_points - k
            idx1 = np.arange(n_points) % n_points
            overlap[overlap_idx] = np.sum(h[idx0] * (1 - h[idx1])) * dx
        C5 = np.max(overlap)
        return C5

    def objective(h):
        return objective_and_constraints(h)

    # Perform optimization
    result = minimize(
        objective,
        x0=h0,
        bounds=[(0.0, 1.0)] * n_points,
        method="SLSQP",
        options={"maxiter": 1000, "ftol": 1e-8}
    )

    h_opt = np.clip(result.x, 0, 1)
    h_opt *= n_points / 2 / np.sum(h_opt)
    final_C5 = objective_and_constraints(h_opt)

    end_time = time.time()
    run_time = end_time - start_time

    if run_time < budget_s:
        print(f"Solution found in {run_time:.2f}s within time budget.")

    return (h_opt, final_C5, n_points)
