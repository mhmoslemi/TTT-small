import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial = np.array(initial_h_values)

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-6:
            h = h / h_sum * target_sum
        h = np.clip(h, 0.0, 1.0)
        ones_minus_h = 1 - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr

    def mutate(h):
        h = np.copy(h)
        n_mut = np.random.randint(1, 6)
        mut_idx = np.random.choice(n_points, size=n_mut, replace=False)
        h[mut_idx] = np.random.rand(n_mut)
        h_sum = h.sum()
        if h_sum > 1e-9:
            h = h / h_sum * target_sum
        return h

    def crossover(p1, p2):
        s = np.random.randint(1, n_points - 1)
        child = np.concatenate((p1[:s], p2[s:]))
        return child

    pop_size = 50
    population = [initial]
    while len(population) < pop_size:
        noise = np.random.randn(n_points) * 0.1
        candidate = np.clip(initial + noise, 0, 1)
        candidate = candidate / candidate.sum() * target_sum
        population.append(candidate)

    best = initial
    best_c5 = evaluate(best)
    elapsed = time.time() - start_time

    while elapsed < budget_s:
        fitness = [evaluate(ind) for ind in population]
        idxs = np.argsort(fitness)
        survivors = [population[i] for i in idxs[: pop_size // 5]]
        next_gen = survivors.copy()

        for _ in range(pop_size // 5):
            p1 = np.random.choice(survivors)
            p2 = np.random.choice(survivors)
            child = crossover(p1, p2)
            child = mutate(child)
            next_gen.append(child)

        population = next_gen

        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best = ind

        elapsed = time.time() - start_time

    return (best.tolist(), best_c5, n_points)
