import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h_values = np.array(h_init, dtype=np.float64)

    def compute_c5(h):
        result = evaluate_erdos_solution(h)
        return result['c5_bound'], result['valid']

    c5_current, valid = compute_c5(h_values)
    if not valid:
        return list(h_values), c5_current, n_points

    best_h = h_values.copy()
    best_c5 = c5_current

    lower = 0.38085
    upper = 0.4
    eps = 1e-6
    max_time = budget_s * 0.8  # reserve time for final checks

    n_points = len(h_values)
    i = np.arange(n_points)
    dx = 2.0 / n_points

    while upper - lower > eps and time.time() - start_time < max_time:
        mid = (lower + upper) / 2
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
        ]
        valid = False

        # Build convolution-based overlap constraints using diagonal overlap matrix
        for k in range(n_points * 2):
            left = np.zeros_like(i)
            right = np.zeros_like(i)

            # Overlap at lag k
            idx1 = i[(i + k >= 0) & (i + k < n_points)]
            idx2 = i[(i - k >= 0) & (i - k < n_points)]
            if idx1.size > 0 and idx2.size > 0:
                left[idx1] = 1
                right[idx2] = 1
                overlap = dx * cp.sum(h * (1 - h @ np.outer(right, left).T))
                constraints.append(overlap <= mid)

        objective = cp.Minimize(mid)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver='GUROBI', verbose=False)

        if h.value is not None and prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h = np.clip(h.value, 0, 1)
            if abs(np.sum(new_h) - target_sum) < 1e-5:
                sol_c5, sol_valid = compute_c5(new_h)
                if sol_valid and sol_c5 < best_c5:
                    best_h = new_h
                    best_c5 = sol_c5
                    print(f"Improved C5: {best_c5}")
                    lower = mid
                else:
                    upper = mid
        else:
            lower = mid

    return list(best_h), best_c5, n_points
