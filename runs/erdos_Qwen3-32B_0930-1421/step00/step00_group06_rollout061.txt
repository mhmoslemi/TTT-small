import numpy as np
import cvxpy as cp
import time
import math

def run(seed=42, budget_s=1100, n_points=95):
    np.random.seed(seed)

    h = initial_h_values.copy()
    n = h.size
    dx = 2.0 / n
    alpha = 0.5
    max_iter = 1000
    improvement_threshold = 1e-4
    h_prev = h.copy()
    start_time = time.time()

    for iter in range(max_iter):
        h_shifted = alpha * h + (1 - alpha) * h_prev

        # Compute the overlap correlation
        corr = np.correlate(h_shifted, 1 - h, mode="full")
        C5 = np.max(corr * dx)

        # Find the largest overlap index
        k_max = np.argmax(corr)

        # Setup the convex optimization problem
        h_var = cp.Variable(n)
        obj = cp.Minimize(h_var[k_max % n] * (1 - h_shifted[(k_max - 1) % n]) * dx)
        for k in range(1, n):
            obj += h_var[(k_max + k) % n] * (1 - h_shifted[(k_max + k - 1) % n]) * dx

        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n / 2]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GUROBI, verbose=False)

        if prob.status != "optimal":
            break

        h_new = h_var.value

        # Update h and h_prev
        h = np.clip(h_new, 0, 1)
        h_prev = h_prev * 0.99 + h * 0.01

        # Dynamically adjust alpha
        if iter % 50 == 0:
            curr_C5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
            if abs(C5 - curr_C5) < improvement_threshold:
                alpha *= 0.9

        # Check time
        if time.time() - start_time > budget_s:
            break

    # Compute and return result
    final_C5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
    h_values = h
    return (h_values, final_C5, n_points)
