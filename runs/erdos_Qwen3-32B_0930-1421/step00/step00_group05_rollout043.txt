import numpy as np
import time
from numpy import correlate

def compute_overlap(h, dx):
    fullcor = correlate(h, 1 - h, mode="full")
    return np.max(fullcor) * dx

def update_h_gradient(h_full, window, direction, h, idx, dx):
    # Compute the derivative of the overlap with respect to h
    h_flat = h_full.flatten()
    
    # Zero out the derivative below the first sample and after the last sample to avoid artifacts
    left = max(0, idx - window)
    right = min(len(h_flat), idx + window + 1)

    # Convolve with the derivative templates
    dL = correlate(h_flat[left:right], np.roll(-np.ones_like(h_flat[left:right]), window)[::-1], mode='valid')
    dR = correlate(np.roll(np.ones_like(h_flat[left:right]), window), h_flat[left:right][::-1], mode='valid')

    # Combine the left and right for the full derivative
    gradient = np.zeros_like(h_flat)
    gradient[left:right] = dR
    gradient[idx:right + len(dR)] += dL
    gradient *= dx

    # Projection: Clip h between 0 and 1
    h_new = h_flat + direction * gradient
    h_new = np.clip(h_new, 0, 1)

    # Ensure the sum constraint is maintained
    delta_mass = (sum(h_flat) - sum(h_new)) * dx
    h_new += delta_mass / len(h_flat)

    # Normalize to keep the sum constant
    h_new = h_new * len(h_flat) / 2
    h_new = np.clip(h_new, 0, 1)

    return h_new

    n_points = len(h)

    dx = 2.0 / n_points

    # Enforce constraints
    h[h > 1] = 1
    h[h < 0] = 0
    h /= np.sum(h)
    h *= n_points / 2
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    c5_history = [best_c5]
    
    learning_rate = 0.5
    window = 5
    max_iter = 10000
    iter_no_improvement = 0

    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break

        # Compute current max overlap
        overlaps = correlate(h, 1 - h, mode="full") * dx  # scalar per shift
        max_index = np.argmax(overlaps)
        max_k = max_index  # shift
        current_c5 = overlaps[max_index]

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iter_no_improvement = 0
            c5_history = c5_history[-50:] + [best_c5]
        else:
            iter_no_improvement += 1

        if iter_no_improvement > 100:
            learning_rate *= 0.9
            iter_no_improvement = 0

        if time.time() - start_time > budget_s * 0.98:
            break

        # Compute gradient at the worst shift and update h
        h = update_h_gradient(h, window, -1, h, max_k, dx)

    h = best_h
    final_c5 = compute_overlap(h, dx)

    return h.tolist(), final_c5, n_points
