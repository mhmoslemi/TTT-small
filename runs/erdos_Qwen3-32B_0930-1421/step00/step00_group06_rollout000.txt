import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = h.shape[0]
    dx = 2.0 / n_points
    
    # Enforce integral constraint (normalized)
    h *= n_points / 2 / h.sum() if h.sum() != 0 else 1.0

    best_c5 = np.inf
    best_h = h.copy()

    t_start = time.time()
    max_iter = 1000
    lr = 0.01
    decay = 0.995
    tol = 1e-6

    start_iter = 0
    for iter in range(start_iter, max_iter):
        t_now = time.time()
        if t_now - t_start >= budget_s:
            break

        # Compute C₅ (objective)
        ones_h = 1 - h
        overlap = correlate(h, ones_h, mode="full")
        overlap *= dx
        c5 = overlap.max()

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Determine the worst lag k
        max_idx = overlap.argmax()
        k = max_idx - n_points + 1  # compute lag from index

        # Compute gradient manually
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = ones_h[i_plus_k] - h[i_minus_k]

        # Gradient ascent (minimize the max overlap => gradient descent of the max function)
        h = h - lr * grad

        # Get the index for the symmetric half (shifted by k)
        # Enforce constraints: [0, 1]
        h = np.clip(h, 0, 1)

        # Normalize to keep ∫₀² h(x) dx = 1
        h *= (n_points / 2) / h.sum() if h.sum() != 0 else 1.0

        # Decay learning rate
        lr *= decay

        # Early stopping
        if iter > 0 and abs(best_c5 - c5) < tol:
            break

    return best_h.tolist(), float(best_c5), n_points
