import time
from scipy.signal import correlate

    n_points = len(h)

    def compute_overlap(h, dx):
        # Compute all overlaps using convolution
        h_flip = 1 - h
        corr_vals = correlate(h, h_flip, mode='full') * dx
        return np.max(corr_vals)

    def project_simplex(v, z=1):
        # Projection onto the simplex: sum(h) = z, h[i] ∈ [0,1]
        n = len(v)
        mu_lb = (v.cumsum() - z) / (1 + np.arange(n))
        mu = max(mu_lb[-1], np.max(np.where(np.arange(n) > 0, (v.cumsum() - z) / (1 + np.arange(n)), v[-1])))
        return np.clip(v - mu, 0, 1)

    def optimize_h(h_init):
        h = h_init.copy()
        dx = 2.0 / len(h)
        best_c5 = compute_overlap(h, dx)
        best_h = h.copy()
        beta = 20  # controls how tight the max approx is
        lr = 0.02  # learning rate
        iter_count = 0
        patience = 50
        no_improve = 0

        while time.time() - start_time < budget_s:
            iter_count += 1

            # Compute all overlaps
            h_flip = 1 - h
            corr_vals = correlate(h, h_flip, mode='full') * dx
            C = corr_vals
            max_idx = np.argmax(C)

            # Compute gradient around the current maximum
            grad = np.zeros_like(h)
            for i in range(n_points):
                for k in range(1, n_points):
                    if (0 <= (i - k)) < n_points:
                        grad[i] -= h_flip[i - k]
                    if (0 <= (i + k)) < n_points:
                        grad[i] += h_flip[i + k]

            # Gradient descent step
            h = h - lr * grad
            h = np.clip(h, 0, 1)

            # Project onto simplex constraint
            h = dx * project_simplex(h / dx, z=1)

            # Zero out very small negative values due to floating-point error
            h = np.clip(h, 0, 1)

            # Evaluate new overlap
            h_flip = 1 - h
            corr_vals = correlate(h, h_flip, mode='full') * dx
            c5 = np.max(corr_vals)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Line search and beta increase
            if no_improve >= patience:
                h = best_h.copy()
                no_improve = 0
                if beta < 100:
                    beta += 1
                if lr > 1e-4:
                    lr *= 0.5

        return best_h, best_c5

    # Run optimization
    optimized_h, final_c5 = optimize_h(h)

    return (optimized_h.tolist(), final_c5, n_points)
