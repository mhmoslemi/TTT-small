import numpy as np
import time
import scipy.signal
import cvxpy as cp
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # --- Setup and Initialization ---
    np.random.seed(seed)
    h0 = initial_h_values  # Provided global initial construction
    n_points = len(h0)
    dx = 2.0 / n_points

    # Ensure h satisfies constraints
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = h0 * (n_points / 2) / np.sum(h0)  # Normalize

    h = h0.copy()

    def evaluate_c5(h):
        # From the provided evaluate_erdos_solution (pre-imported)
        return np.max(scipy.signal.correlate(h, 1 - h, mode="full")) * dx

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    t0 = time.time()

    # --- Phase 1: Frequency-domain optimization ---
    max_phase1_time = 0.6 * budget_s
    while time.time() - t0 < max_phase1_time:
        H = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points, d=dx)

        # Construct the convolution proxy: h * (1 - h)
        h1minus_h = h * (1 - h)
        H1minus_H = np.fft.fft(h1minus_h)

        # Solve a convex optimization to minimize max magnitude of frequency band
        alpha = cp.Variable(H1minus_H.shape)
        w = cp.Parameter(H1minus_H.shape)
        w.value = H1minus_H

        objective = cp.Minimize(cp.norm(alpha, np.inf))
        constraints = [alpha == w]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != cp.OPTIMAL:
            continue

        # Apply result to frequency domain, go back to time domain
        H_filtered = alpha.value
        h_filtered = np.fft.ifft(H_filtered).real
        h_filtered = np.clip(h_filtered, 0.0, 1.0)
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)

        c5_filtered = evaluate_c5(h_filtered)
        if c5_filtered < best_c5:
            best_c5 = c5_filtered
            best_h = h_filtered.copy()

    # --- Phase 2: Time-domain gradient refinement ---
    max_phase2_time = 0.4 * budget_s
    t0 = time.time()

    while time.time() - t0 < max_phase2_time:
        grad = np.gradient(evaluate_c5, h, dx, order=4)
        learning_rate = 0.01 * (1 - (time.time() - t0) / max_phase2_time)
        h = h - learning_rate * grad

        # Enforce constraints and re-normalize
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
