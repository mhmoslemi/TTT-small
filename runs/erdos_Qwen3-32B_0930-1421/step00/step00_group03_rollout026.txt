```python
import time
import numpy as np
import cvxpy as cp
import scipy

# Define evaluate_erdos_solution as a dummy if not defined
try:
    from evaluate_erdos_solution import evaluate_erdos_solution
except ImportError:
    def evaluate_erdos_solution(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        ones_minus_h = 1.0 - h_values
        corr = np.correlate(h_values, ones_minus_h, mode='full')
        return np.max(corr) * dx

try:
    from parent_construction import initial_h_values
except ImportError:
    initial_h_values = None

def run(seed=42, budget_s=1100, n_points=70):
    np.random.seed(seed)
    start_time = time.time()

    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        scaling_factor = n_points / 2 / h_sum
        h *= scaling_factor
        h = np.clip(h, 0.0, 1.0)
    else:
        # Random initialization fulfilling measure constraint
        h = np.random.uniform(0, 1, size=n_points)
        h /= h.sum() * 2.0
        h *= n_points / 2
        h = np.clip(h, 0.0, 1.0)

    # Define problem variables
    h_var = cp.Variable(n_points, nonneg=True)
    C5 = cp.Parameter(nonneg=True)

    # Construct overlap constraints
    overlapping_indices = [k for k in range(-(n_points - 1), n_points)]
    constraints = [h_var >= 0, h_var <= 1.0, cp.sum(h_var) == n_points / 2.0]
    
    def get_overlap_objective(h_var, k):
        dx = 2.0 / n_points
        overlap = 0.0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                overlap += h_var[i] * (1 - h_var[j]) * dx
            elif -n_points < j < 0:
                overlap += h_var[i] * (1 - h_var[j + n_points]) * dx
        return overlap

    for k in range(n_points):
        constraints.append(get_overlap_objective(h_var, k - n_points + 1) <= C5)

    # Solve using CVXOPT with binary search
    C5_low = 0.38085
    C5_high = 0.390
    C5_best = 1.0
    best_h = h
    best_time = time.time()

    while (time.time() - start_time) < budget_s - 5 and C5_high - C5_low > 1e-6:
        C5_mid = (C5_low + C5_high) / 2.0
        C5.value = C5_mid

        problem = cp.Problem(cp.Minimize(C5), constraints)
        problem.solve(solver='CVXOPT', verbose=False)

        if problem.status == cp.OPTIMAL:
            C5_high = C5_mid
            best_h = h_var.value
            C5_best = C5_mid
            best_time = time.time()
        else:
            C5_low = C5_mid

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10:
            break

    # Compute final C5 value using the evaluator
    best_h = np.clip(best_h, 0.0, 1.0)
    final_C5 = evaluate_erdos_solution(best_h)

    # Ensure sum constraint is satisfied
    best_h_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2 / best_h_sum)
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), final_C5, n_points)
```
