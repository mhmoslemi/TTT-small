import numpy as np
import cvxpy as cp
import time
from functools import partial

def run(seed=42, budget_s=1100, n_start=70, n_max=1000, refine=True):
    start_time = time.time()
    np.random.seed(seed)
    dt = 0.1  # Time check interval

    # Extract and initialize from the provided vector
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points

    def evaluate(h):
        h = np.asarray(h).ravel()
        C5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        return C5

    def build_and_evaluate(h_current):
        h = np.clip(h_current, 0, 1)
        C5 = evaluate(h)
        total = h.sum()
        obj_val = C5
        return h, C5, total

    def get_valid_h(h):
        h_valid = np.clip(h, 0, 1)
        h_valid *= n_points / 2 / h_valid.sum()
        return h_valid

    def optimize_with_cvxpy(h0, n_points):
        h = cp.Variable(n_points)
        dx = 2.0 / n_points
        sum_h = cp.sum(h) * dx - 1
        constraints = [
            0 <= h,
            h <= 1,
            sum_h == 0,
        ]
        max_c5 = None
        for k in range(1, n_points - 1):
            shifted_h = np.roll(h, -k).flatten()[:n_points]
            integrand = h[:(n_points - k)] * (1 - shifted_h[:(n_points - k)])
            corr_value = cp.sum(integrand) * dx
            if max_c5 is None:
                max_c5 = corr_value
            else:
                max_c5 = cp.maximum(max_c5, corr_value)
        objective = cp.Minimize(max_c5)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h0, float("inf")
        return h.value, problem.value

    h_best, c5_min, t_total = h0, float("inf"), h0.sum()
    best_h, best_C5 = h_best, c5_min
    if n_points < n_max:
        n_points = min(n_points * 2, n_max)
        h = np.tile(h0, (n_points // len(h0), 1)).flatten()
        h /= np.sum(h) * 2.0 / n_points

    h = get_valid_h(h)
    elapsed = 0
    while elapsed < budget_s:
        if time.time() - start_time > budget_s - dt:
            break
        h, c5_valid, t_total = build_and_evaluate(h)
        if c5_valid < best_C5:
            best_h = h.copy()
            best_C5 = c5_valid
        if not refine:
            break
        h, _ = optimize_with_cvxpy(h, n_points)
        h = get_valid_h(h)
        elapsed = time.time() - start_time

    return best_h, best_C5, n_points
