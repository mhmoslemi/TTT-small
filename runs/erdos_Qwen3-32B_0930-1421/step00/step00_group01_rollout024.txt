import numpy as np
import scipy.optimize
import time

def correlate_max_overlap(h, dx):
    # Compute all correlations h * (1 - h[k + delta]) over shifts
    conv_result = np.correlate(h, 1 - h, mode='full')
    return np.max(conv_result * dx)

def objective_function(h_flat, n_points, dx):
    h = h_flat.reshape(n_points, 1)
    return correlate_max_overlap(h.flat, dx)

def constraint_sum_1(h_flat, n_points):
    return np.sum(h_flat) - n_points / 2

def bounds_h(n_points):
    return [(0, 1)] * n_points

def run(seed=42, budget_s=1100):
    start_time = time.time()
    
    # Use initial_h_values or initialize empty for reproducibility check
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values.values)
    else:
        raise ValueError("initial_h_values is not available.")

    n_points = h_init.size
    dx = 2.0 / n_points
    h_opt = h_init.copy()
    best_c5 = correlate_max_overlap(h_opt, dx)

    def opt_objective(h_flat):
        return correlate_max_overlap(h_flat, dx)

    def opt_grad(h_flat):
        h = h_flat.reshape(n_points, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        grad_flat = np.zeros(n_points)

        # Compute gradient using finite differences
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = np.clip(h_plus[i] + eps, 0, 1)
            val_plus = correlate_max_overlap(h_plus.flat, dx)
            h_minus = h.copy()
            h_minus[i] = np.clip(h_minus[i] - eps, 0, 1)
            val_minus = correlate_max_overlap(h_minus.flat, dx)
            grad_flat[i] = (val_plus - val_minus) / (2 * eps)

        return grad_flat

    # Constraints and bounds
    constraints = [
        {'type': 'eq', 'fun': constraint_sum_1, 'args': (n_points,)}
    ]
    bounds = bounds_h(n_points)

    # Optimization loop
    while time.time() - start_time < budget_s - 20:
        result = scipy.optimize.minimize(
            fun=opt_objective,
            x0=h_opt,
            jac=opt_grad,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-6, 'maxiter': 50, 'disp': False}
        )

        if not result.success:
            break

        new_h = result.x
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / np.sum(new_h) * (n_points / 2)  # Normalize
        new_c5 = correlate_max_overlap(new_h, dx)

        if new_c5 < best_c5:
            best_c5 = new_c5
            h_opt = new_h

    # Final validation
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    c5_bound = correlate_max_overlap(h_opt, dx)
    h_values = h_opt
    return h_values, c5_bound, n_points
