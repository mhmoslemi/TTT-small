import numpy as np
import scipy.optimize
import time

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h1 = np.concatenate([h, np.zeros(n_points - 1)])
    h2 = np.concatenate([np.zeros(n_points - 1), 1 - h])
    overlaps = np.sum(h1 * h2, axis=0) * dx
    return np.max(overlaps)

def objective(x):
    h = x
    h = np.clip(h, 0, 1)
    return compute_c5(h)
