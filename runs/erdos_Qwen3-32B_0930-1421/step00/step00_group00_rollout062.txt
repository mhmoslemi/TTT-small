import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    initial_h = initial_h_values
    n_points = len(initial_h)

    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    const = [h <= 1.0]
    const += [cp.sum(h) == n_points / 2]

    best_h = initial_h.copy()
    best_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    print(f"Initial C5: {best_c5}")

    while time.time() - t0 < budget_s:
        overlaps = correlate(best_h, 1 - best_h, mode="full")
        k_max = np.argmax(overlaps)
        
        objective = overlaps[k_max] * dx
        problem = cp.Problem(cp.Minimize(objective), const)
        problem.solve(solver=cp.OSQP, verbose=False)

        h_sol = h.value
        h_sol = np.clip(h_sol, 0, 1)
        h_sol *= n_points / 2.0 / h_sol.sum()
        h_new = h_sol

        overlaps_new = correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(overlaps_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    return (best_h.tolist(), best_c5, n_points)
