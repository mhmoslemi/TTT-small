import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    """Run optimization for Erdős minimum overlap problem."""
    # start timing
    start_time = time.time()
    
    # parameters
    n_points = 126  # increased for better resolution
    dx = 2.0 / n_points
    total_sum = n_points // 2  # constraint sum(h) == 1 * n_points / 2
    np.random.seed(seed)

    # Initialize h
    if 'initial_h_values' in globals():
        n_initial = len(initial_h_values)
        h = np.zeros(n_points)
        idx = np.round(np.linspace(0, n_initial - 1, n_points)).astype(int)
        h[idx] = initial_h_values
        h = h / np.sum(h) * total_sum
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points)
        h = h / h.sum() * total_sum
        h = np.clip(h, 0, 1)

    # Project onto feasible region
    def project_feasible(h, total_sum):
        h = np.clip(h, 0, 1)
        scaling_factor = total_sum / np.sum(h)
        h = h * scaling_factor
        return h

    best_c5 = np.inf
    best_h = h.copy()

    # core optimization loop
    while time.time() - start_time < budget_s:
        # compute C₅ via convolution
        def evaluate_c5(h, dx):
            # pad h and 1 - h with zeros
            h_padded = np.concatenate((np.zeros(n_points), h, np.zeros(n_points)))
            h1_padded = np.concatenate((np.zeros(n_points), 1 - h, np.zeros(n_points)))
            result = np.correlate(h_padded, h1_padded, mode="full")
            result = result[n_points:-n_points] * dx
            return np.max(result)
        
        c5 = evaluate_c5(h, dx)

        # update best result
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"[Best] C5 = {best_c5:.7f} at t={time.time()-start_time:.2f}s")

        # compute gradient numerically
        eps = 1.0e-2 * (np.sum(h) / n_points)  # step size
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, total_sum)
            c5_plus = evaluate_c5(h_plus, dx)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_feasible(h_minus, total_sum)
            c5_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # project gradient update
        alpha = 0.05  # learning rate
        h = h - alpha * grad
        h = project_feasible(h, total_sum)

    # validation and return
    h = np.round(h, 12)  # numerical robustness
    return best_h, best_c5, n_points
