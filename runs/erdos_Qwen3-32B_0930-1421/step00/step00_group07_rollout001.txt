import numpy as np
import scipy.signal
import time
import math

def run(seed=42, budget_s=2000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Enforce the constraint that the sum of h is n_points / 2
        h *= (n_points / 2) / h.sum()
    else:
        # Construct initial h with 80 points
        n_points = 80
        h = np.zeros(n_points)
        h[n_points // 4:3 * n_points // 4] = 1.0  # Initial guess with middle half = 1
        h *= (n_points / 2) / h.sum()  # Normalize to satisfy the constraint

    dx = 2.0 / n_points
    max_iter = 5000
    lr = 0.02
    patience = 200
    best_h = h.copy()
    best_c5 = float('inf')
    no_improvement = 0

    for it in range(max_iter):
        # Compute (1 - h) for correlation
        one_minus_h = 1 - h

        # Compute cross-correlation between h and (1 - h)
        corr = scipy.signal.correlate(h, one_minus_h, mode='full')

        # Compute the correlation with scaling
        c5 = np.max(corr * dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                lr *= 0.5
                no_improvement = max(no_improvement - patience, 0)
                if lr < 1e-6:
                    break

        # Compute gradient: only the part overlapping at maximum correlation
        lag = np.argmax(corr)
        if lag <= n_points - 1:
            # h[*] * (1 - h[* + k])
            grad = -one_minus_h[:lag]
            grad = np.concatenate([grad, np.zeros(n_points - lag)])
        else:
            # h[* + k] * (1 - h[*])
            grad = -h[lag - n_points:]
            grad = np.concatenate([np.zeros(n_points - (n_points * 2 - lag)), grad])

        grad *= dx  # account for dx scaling in gradient descent
        h -= lr * grad

        # Project h to [0, 1]
        h = np.clip(h, 0, 1)

        # Project back to satisfy the constraint sum(h) = n_points / 2
        total = h.sum()
        h *= (n_points / 2) / total

        # Check elapsed time
        if time.time() - start_time >= budget_s:
            break

    best_h = best_h.tolist()
    return (best_h, best_c5, n_points)
