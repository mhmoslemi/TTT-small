import numpy as np
import time
from copy import deepcopy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Load initial solution and determine n_points
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    desired_total = n_points // 2  # sum(h) must be exactly n_points / 2

    def evaluate(h):
        h_adj = h.clip(0.0, 1.0)
        # Ensure sum constraint
        current_sum = np.sum(h_adj)
        if abs(current_sum - desired_total) > 1e-3:
            correction = (desired_total - current_sum) / n_points
            h_adj += correction
            h_adj = h_adj.clip(0.0, 1.0)
        # Normalize to exactly satisfy sum constraint via rescaling utilities
        utility = h_adj.copy()
        utility -= np.mean(utility)
        utility = np.clip(utility / (np.std(utility) + 1e-8), -1, 1)
        utility += 0.5
        h_adj = utility
        h_adj *= desired_total / np.sum(h_adj)
        h_adj = h_adj.clip(0.0, 1.0)

        # Compute convolutions
        corr = np.correlate(h_adj, 1 - h_adj, mode='full')
        conv_window = np.concatenate([corr[-n_points + 1:], corr[:n_points]])

        c5 = np.max(conv_window * dx)
        return c5, h_adj

    best_c5 = np.inf
    best_h = h.copy()

    # Setup population
    pop_size = 40
    elite_pct = 0.1
    n_elite = int(pop_size * elite_pct)
    population = []
    for _ in range(pop_size):
        if _ == 0:
            indiv = h.copy()
        else:
            indiv = np.random.rand(n_points)  # Random initialization
        indiv = indiv / indiv.max()
        indiv *= desired_total / np.sum(indiv)
        indiv = indiv.clip(0.0, 1.0)
        population.append(indiv)

    # Iteration
    generation = 0
    while time.time() - t0 < budget_s * 0.99:
        generation += 1
        # Evaluate fitness
        scores = []
        for indiv in population:
            c5, _ = evaluate(indiv)
            scores.append(c5)

        # Find best in population
        best_index = np.argmin(scores)
        current_best = population[best_index]
        current_c5 = scores[best_index]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            print(f"New best C5: {best_c5:.6f} (gen {generation})")

        # Select parents
        sorted_indices = np.argsort(scores)
        elites = [population[i] for i in sorted_indices[:n_elite]]
        parents = elites[:]
        while len(parents) < pop_size:
            # Tournament selection
            tourney = np.random.choice(len(population), 2)
            winner = tourney[np.argmin([scores[i] for i in tourney])]
            parents.append(population[winner])

        # Crossover
        new_population = elites[:]
        while len(new_population) < pop_size:
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            split_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split_point], parent2[split_point:]])
            new_population.append(child)

        # Mutation
        for i in range(len(new_population)):
            if i < n_elite:
                continue  # don't mutate elites
            if np.random.rand() < 0.1:  # mutation rate
                mutation_mask = np.random.rand(n_points) < 0.03  # 3% point mutation
                noise = np.random.normal(loc=0, scale=0.1, size=n_points)
                new_population[i][mutation_mask] = np.clip(new_population[i][mutation_mask] + noise[mutation_mask], 0, 1)

        # Replace population
        population = new_population

    # Post-optimization improvement: local line search
    improvements = 0
    for _ in range(3):
        c5, _ = evaluate(best_h)
        if c5 < best_c5:
            best_c5 = c5
            improvements += 1

    return best_h.tolist(), best_c5, n_points
