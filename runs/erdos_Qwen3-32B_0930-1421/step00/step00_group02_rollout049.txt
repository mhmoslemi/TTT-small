import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with initial n_points
    if not hasattr(initial_h_values, '__len__'):
        initial_h_values = None

    if initial_h_values is not None:
        h = np.array(initial_h_values)
        n_points_h = len(h)
        if n_points_h != n_points:
            # Stretch or interpolate to match n_points
            x_old = np.linspace(0, 2, n_points_h)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
    else:
        # Construct h as uniform or mid-point shifted
        h = np.zeros(n_points)
        center = np.argmin(np.abs(np.linspace(0,2,n_points) - 1))
        h[center - 10:center + 10] = 0.5
        h = h / np.sum(h) * n_points / 2
    h = h.astype(np.float64)

    dx = 2 / n_points

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s * 0.98:
        # CVX formulation: minimize max overlap
        h_cvx = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            h_cvx <= 1
        ]

        # h(x)(1 - h(x + k)) for all k
        overlaps = []
        h_rolled = [np.roll(h.CVXPY_compat, -k, axis=0) for k in range(n_points - 1)]
        for k in range(n_points):
            overlaps.append(
                cp.sum(h_cvx * (1 - (np.roll(h_cvx, -k) if k < n_points else 0))) * dx
            )

        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCIPY, verbose=False)
        if h_cvx.value is not None:
            h = np.clip(h_cvx.value, 0, 1)
            h /= np.sum(h) * dx
            c5 = max(np.correlate(h, 1 - h, mode="full") * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                print(f"New best: {best_c5:.6f}")
        else:
            print("CVX failed, repairing")
            h = np.clip(h + np.random.normal(0, 0.01, len(h)), 0, 1)
            h = h * (n_points / 2 / np.sum(h))

    return (best_h.tolist(), best_c5, n_points)
