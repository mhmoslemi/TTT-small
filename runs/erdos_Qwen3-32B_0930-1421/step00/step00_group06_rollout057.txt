def estimate_c5(h):
    """Estimate C5 using numpy correlate."""
    n_points = len(h)
    dx = 2.0 / n_points
    h_combined = np.concatenate([h, 1 - h])
    overlaps = np.correlate(h_combined[:n_points], h_combined[n_points:], mode="full")
    overlaps = overlaps * dx
    return np.max(overlaps)
