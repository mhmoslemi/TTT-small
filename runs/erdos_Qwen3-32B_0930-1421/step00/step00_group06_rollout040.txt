import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Parameters
    n_points = 95
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialization
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.mean() * target_sum

    # Ensure the sum constraint is satisfied
    h = np.clip(h, 0, 1)
    h = h / h.mean() * target_sum

    max_iter = 500
    tol = 1e-4
    epsilon = 1e-6
    prev_c5 = float('inf')

    total_time = 0
    start_time = time.time()

    for it in range(max_iter):
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed >= budget_s:
            break

        # Compute correlation
        overlap = correlate(h, 1 - h, mode='full') * dx
        lag = n_points
        c5 = np.max(overlap[lag:2 * lag])
        nu = np.max((overlap[lag:2 * lag], c5), axis=0)

        if abs((prev_c5 - c5) / prev_c5) < tol:
            break
        prev_c5 = c5

        # Form and solve convex optimization problem
        h_var = cp.Variable(n_points, nonneg=True)
        objective = cp.Minimize(cp.sum(cp.log(nu + epsilon + 1e-10)))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        h = h_var.value
        h = np.clip(h, 0, 1)

        # Project back to feasible sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            scaling = target_sum / current_sum
            h = h * scaling

    # Final validation and return
    h = np.clip(h, 0, 1)
    h = h / h.mean() * target_sum
    overlap = correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(overlap[n_points:2 * n_points])

    return (h, c5_bound, n_points)
