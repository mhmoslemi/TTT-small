import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def create_phase1_model(h, dx):
    h_var = cp.Variable(len(h), nonneg=True)
    constr = [h_var >= 0, h_var <= 1]
    constr += [cp.sum(h_var) * dx == 1]
    obj = cp.Minimize(cp.max(
        cp.Real(cp.ifft(cp.fft(h_var) * cp.conj(cp.fft(1 - h_var))).imag)
    ))
    prob = cp.Problem(obj, constr)
    return prob, h_var
