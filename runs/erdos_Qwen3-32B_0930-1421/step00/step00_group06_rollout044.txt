import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, alpha_start=0.5, alpha_min=0.1, max_iterations=1000, recompute_h_shifted_every=50, tolerance=1e-4):
    np.random.seed(seed)

    # Extract initial solution
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize to have mass 1 in [0, 2]
    h_vals = h / np.sum(h) * n_points / 2
    h_vals = np.clip(h_vals, 0, 1)

    h_prev = np.copy(h_vals)
    alpha = alpha_start
    best_c5 = float("inf")
    best_h = h_vals.copy()

    start_time = time.time()

    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Create frozen h_shifted
        h_shifted = alpha * h_vals + (1 - alpha) * h_prev

        # Compute overlap integrals via correlation
        overlap = np.correlate(h_shifted, 1 - h_vals, mode="full") * dx

        # Find the max overlap
        k_max = int(np.argmax(overlap))

        # Create the convex optimization problem to optimize h for the worst case
        h_opt = cp.Variable(shape=n_points)
        obj = cp.Minimize(overlap[k_max])
        constraints = [h_opt >= 0, h_opt <= 1, cp.sum(h_opt) == n_points / 2]

        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GUROBI, verbose=False)

        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_prev = np.copy(h_vals)
            alpha *= 0.9
            alpha = max(alpha, alpha_min)
            continue

        new_h = h_opt.value
        if new_h is None:
            continue

        new_h = np.clip(new_h, 0, 1)

        # Compute new overlap
        overlap_new = np.correlate(new_h, 1 - new_h, mode="full") * dx
        c5_new = np.max(overlap_new)

        # Improve if it's better
        if c5_new < best_c5 - tolerance:
            best_c5 = c5_new
            best_h = new_h.copy()
            h_vals = new_h.copy()
            alpha = alpha_start
        else:
            alpha *= 0.99
            alpha = max(alpha, alpha_min)
            h_vals = alpha * new_h + (1 - alpha) * h_vals

        h_prev = h_vals.copy()

        if iter % recompute_h_shifted_every == 0:
            curr_overlap = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
            curr_c5 = np.max(curr_overlap)
            print(f"Iter {iter}: C₅ = {curr_c5:.8f}")

    c5_val = np.max(np.correlate(best_h, 1 - best_h, mode="full") * dx)

    return (best_h, c5_val, n_points)
