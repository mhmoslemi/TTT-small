import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_h(h):
    total = np.sum(h)
    target = n_points / 2
    h = h * (target / total)
    h = np.clip(h, 0, 1)
    return h

def tournament_selection(population, fitness, tournament_size=3):
    indices = np.random.choice(len(population), tournament_size)
    best_idx = np.argmin(fitness[indices])
    return population[indices[best_idx]], indices[best_idx]

def crossover(a, b):
    idx = np.random.choice(len(a), size=2, replace=False)
    idx = np.sort(idx)
    left, right = idx
    child = a.copy()
    child[left:right] = b[left:right]
    child = normalize_h(child)
    return child

def mutate(h, mutation_rate=0.05):
    mut_indices = np.random.rand(len(h)) < mutation_rate
    h = h.copy()
    h[mut_indices] = np.random.rand(np.sum(mut_indices))
    h = normalize_h(h)
    return h

def run(seed=42, budget_s=1100, n_points=42, pop_size=20, generations=100, mutation_rate=0.05):
    np.random.seed(seed)

    # Reading initial construction if available
    if 'initial_h_values' in globals():
        h_0 = initial_h_values
        h_sum = int(n_points / 2)
        h_0 = (np.array(h_0)).astype(float) * (h_sum / np.sum(h_0))
        h_0 = np.clip(h_0, 0.0, 1.0)
        h_values = [h_0]
        while len(h_values) < pop_size:
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (h_sum / (h_sum * 2 if h_sum > 0 else 1))
            h = np.clip(h, 0.0, 1.0)
            h = normalize_h(h)
            h_values.append(h)
    else:
        h_values = [normalize_h(np.random.rand(n_points)) for _ in range(pop_size)]

    best_solution = None
    best_C5 = float('inf')
    start_time = time.time()

    for gen in range(generations):
        fitness = [evaluate_C5(h, n_points) for h in h_values]
        idx = np.argmin(fitness)
        if fitness[idx] < best_C5:
            best_C5 = fitness[idx]
            best_solution = h_values[idx]
        if time.time() - start_time > budget_s:
            break

        next_population = []
        for _ in range(pop_size // 2):
            a, a_idx = tournament_selection(h_values, fitness)
            b, b_idx = tournament_selection(h_values, fitness)
            child1 = crossover(a, b)
            child2 = crossover(b, a)
            next_population.extend([child1, child2])

        for i in range(len(next_population)):
            if np.random.rand() < mutation_rate:
                next_population[i] = mutate(next_population[i])

        h_values = next_population

    if best_solution is None:
        best_solution = h_values[np.argmin([evaluate_C5(h, n_points) for h in h_values])]

    return best_solution.tolist(), best_C5, n_points
