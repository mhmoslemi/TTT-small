import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_candidate(solution):
    h, n_points = solution
    dx = 2.0 / n_points
    inv_dx = n_points / 2
    inter = correlate(h, 1 - h, mode="full")
    return inter * dx, True

def feasible(h, n_points, target_c5, dx):
    max_corr = 0.0
    for k in range(1, n_points):
        overlap = np.sum(h[:n_points - k] * (1 - h[k:])) * dx
        if overlap > max_corr:
            max_corr = overlap
    return max_corr <= target_c5

def run(seed=42, budget_s=1100, n_points=70):
    dx = 2.0 / n_points
    np.random.seed(seed)
    best_h = initial_h_values[:] if 'initial_h_values' in globals() else np.zeros(n_points)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = evaluate_candidate((best_h, n_points))[0].max()
    low, high = best_c5, 0.4  # Target C_5 within [0.38085, 0.4]
    best_c5 = min(best_c5, high)
    start_time = time.time()
    total_budget = budget_s

    while high - low > 1e-5 and time.time() - start_time < total_budget:
        mid = (low + high) / 2
        h = cp.Variable(n_points, nonneg=True)
        prob = cp.Problem(
            cp.Minimize(0),
            constraints=[
                cp.sum(h) == n_points / 2,
                h <= 1,
                h >= 0,
            ] + [
                cp.sum([h[i] * (1 - h[i + k]) for i in range(n_points - k)]) * dx <= mid
                for k in range(1, n_points)
            ]
        )

        prob.solve(solver=cp.OSQP, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            candidate_h = np.clip(np.array(h.value).ravel(), 0, 1)
            corr, is_feasible = evaluate_candidate((candidate_h, n_points))
            if is_feasible:
                if corr.max() <= high:
                    best_c5 = corr.max()
                    best_h = candidate_h
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        else:
            low = mid

        if time.time() - start_time > total_budget:
            break

    return (best_h.tolist(), float(best_c5), n_points)
