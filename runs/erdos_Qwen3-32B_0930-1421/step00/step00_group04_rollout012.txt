import numpy as np

def run(seed=42, budget_s=1100):
    start_time = time.time()

    # Load initial solution if available

        n_points = len(h)
    except:
        n_points = 42

    # Ensure constraint is met
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum() if h.sum() != 0 else 0.5

    # Discretization step
    dx = 2.0 / n_points

    def compute_c5(h_values):
        hcorr = scipy.signal.correlate(h_values, 1 - h_values, mode="full")
        return np.max(hcorr * dx)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    step_size = 0.02
    restart_interval = 200
    restart_count = 0
    learning_rate = 1.0
    learning_decay = 0.95
    mass_dir = np.random.rand(n_points)

    # Optimize while within time budget
    while time.time() - start_time < budget_s:
        # Perform gradient-like steps
        for step in range(10):
            if time.time() - start_time >= budget_s:
                break

            grad = np.zeros(n_points)
            for i in range(n_points):
                h[i] += 1e-3
                h_bad = compute_c5(h)
                h[i] -= 2e-3
                h_good = compute_c5(h)
                h[i] += 1e-3
                grad[i] = h_bad - h_good

            h += grad * learning_rate
            h = np.clip(h, 0, 1)

            # Enforce constraint
            h *= n_points / 2 / h.sum() if h.sum() != 0 else 0.5

            # Record best solution found
            c5_current = compute_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()

        learning_rate *= learning_decay

        # Restart mechanism - create a small simplex
        center = h.copy()
        step_size *= 1.1

        for idx in range(1, 10):
            candidate = center.copy()
            direction = np.roll(mass_dir, idx) if idx > 1 else mass_dir

            # Perturbation with direction
            perturbation = step_size * direction - (h * step_size * direction).sum() * h
            candidate += perturbation
            candidate = np.clip(candidate, 0, 1)

            # Normalize to constraint
            candidate *= n_points / 2 / candidate.sum() if candidate.sum() != 0 else 0.5

            # Check if candidate is better
            c5_candidate = compute_c5(candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
                h = candidate.copy()
                break

        restart_count += 1

    return list(best_h), best_c5, n_points
