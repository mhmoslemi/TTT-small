import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h
    best_c5 = None
    
    # Optimization parameters
    alpha = 0.5  # Mixing parameter for h_shifted update
    max_iter = 1000
    improvement_threshold = 1e-4
    patience = 25
    tolerance = 1e-5
    
    # Track time
    start_time = time.time()
    last_improvement_time = start_time
    no_improvement_count = 0
    best_overlap = float('inf')
    
    h_prev = h.copy()
    h_shifted = h.copy()
    
    for it in range(max_iter):
        # Compute overlap with current h and h_shifted
        correlation = np.correlate(h_shifted, 1 - h, mode="full") * dx
        overlap_k = correlation[len(h):2*len(h)]  # Relevant region for k=1 to k=n-1
        c5 = np.max(overlap_k)
        
        if best_c5 is None or c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            last_improvement_time = time.time()
        else:
            no_improvement_count += 1

        if time.time() - start_time > budget_s:
            break
            
        # If no improvement for patience rounds, reduce alpha
        if no_improvement_count >= patience:
            alpha = max(alpha * 0.97, 0.1)

        # Identify the k with maximum overlap
        k_max = np.argmax(overlap_k)

        # Create the index mapping: [(i + k_max) % n_points for i in 0..n-1]
        shifted_idx = (np.arange(n_points) + k_max) % n_points
        
        # Build convex surrogate problem
        h_var = cp.Variable(n_points)
        
        # Objective: minimize target overlap at k_max
        obj = cp.Minimize(
            cp.sum(h_var * (1 - h_shifted[shifted_idx])) * dx
        )
        
        # Equality constraint for the integral (mean constraint)
        const = [h_var >= 0, h_var <= 1,
                 cp.sum(h_var) == n_points / 2]
        
        # Optimization problem
        prob = cp.Problem(obj, const)
        
        # Solve convex surrogate
        try:
            result = prob.solve(solver=cp.GUROBI, verbose=False)
            h = h_var.value
        except:
            # If solver fails, fallback to previous best
            h = best_h.copy()
            continue

        # Ensure result is valid and project to feasible region
        if h is None or np.any(np.isnan(h)):
            h = best_h.copy()
        else:
            h = np.clip(h, 0, 1)
        
        # Update shifted history
        h_shifted = alpha * h + (1 - alpha) * h_prev
        h_prev = h.copy()
        
        # Stop if we have no further improvements or time is exhausted
        if best_c5 is not None and best_c5 <= tolerance:
            break

    # Ensure final output h satisfies all constraints
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.mean(h) * (n_points / 2)
    
    # Final validation
    c5 = np.max(np.correlate(h, 1 - h, mode="full")[len(h):2*len(h)]) * dx
    
    return (h, c5, n_points)
