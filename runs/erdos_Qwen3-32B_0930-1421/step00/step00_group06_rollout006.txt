import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values[:]
    n_points = len(h0)
    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    objective = 0.0
    max_k_lag = 2 * n_points  # full cross-correlation
    impulses = []
    for lag in range(1, n_points):
        if lag >= n_points:
            break
        hh = h[lag:] - h[lag:]**2
        hh = hh + h[:-lag] - h[:-lag]**2
        impulses.append(hh * dx - cp.Constant(1e-8))

    max_imp = cp.Maximize(sum(impulses))
    constrain = [cp.sum(h) * dx == 1, h <= 1]
    prob = cp.Problem(max_imp, constrain)
    prob.solve(solver=cp.ECOS, max_iter=150)
    h_opt = h.value
    h_values = h_opt
    h_norm = h_values / h_values.sum() * n_points / 2
    from evaluate_erdos_solution import evaluate_erdos_solution
    c5_bound = evaluate_erdos_solution(h_norm)
    return h_norm, c5_bound, n_points
