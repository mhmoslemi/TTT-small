import numpy as np
import cvxpy as cp
import time
import math
from evaluate_erdos_solution import evaluate_erdos_solution  # pre-imported

def run(seed=42, budget_s=1100, n_points=42, pop_size=20, generations=200):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Construct an initial candidate with a balanced distribution
        h0 = np.zeros(n_points)
        h0[n_points // 2 - 10:n_points // 2 + 10] = 0.5  # rough balance
        h0 *= (n_points / 2) / np.sum(h0)  # rescale to meet sum constraint

    if not np.isclose(np.sum(h0), n_points / 2):
        raise ValueError("Initial h0 does not meet the area constraint.")

    population = [h0]  # Start with the initial one

    # Generate more diverse initial members to avoid early population similarity
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h *= (n_points / 2) / np.sum(h)  # satisfy the normalization constraint.
        population.append(h)

    population = [np.clip(h, 0, 1) for h in population]  # ensure constraint [0, 1]

    def fitness(h):
        if not np.all((0 <= h) * (h <= 1)):
            return float('inf')
        if not np.isclose(np.sum(h), n_points / 2):
            return float('inf')
        correlations = np.correlate(h, 1 - h, mode="full")
        max_corr = correlations[np.abs(correlations) == np.max(np.abs(correlations))].max()
        return max_corr * dx

    best_h = None
    best_c5 = float('inf')

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break

        # Evaluate the population
        population_scores = [(fitness(ind), ind) for ind in population]
        population_scores.sort()
        population_scores = population_scores[:pop_size]

        # Update best solution
        for s, ind in population_scores:
            if s < best_c5:
                best_c5 = s
                best_h = ind.copy()

        # Select parents using tournament with window size 5
        parents = []
        while len(parents) < pop_size // 2:
            candidates = np.random.choice(population, 5)
            candidates = sorted([(fitness(h), h) for h in candidates])
            parents.append(candidates[0][1])
            parents.append(candidates[1][1])

        # Crossover (greedy uniform crossover)
        next_gen = parents[:]
        while len(next_gen) < pop_size:
            if not parents:
                break
            p1 = parents[np.random.choice(len(parents))]
            p2 = parents[np.random.choice(len(parents))]
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, p1, p2)
            # Enforce the sum constraint
            if np.sum(child) == 0:
                continue
            scaling = (n_points / 2) / np.sum(child)
            child = np.clip(child * scaling, 0, 1)
            next_gen.append(child)
        population = next_gen

        # Mutate (small perturbations)
        for i in range(0, pop_size, 5):
            h = population[i]
            # Flip ~10% of the indices
            m = np.random.choice(n_points, int(0.1 * n_points), replace=False)
            h2 = h.copy()
            h2[m] = 1 - h2[m]
            # Re-normalize to keep the area constraint
            if np.sum(h2) != 0:
                h2 *= (n_points / 2) / np.sum(h2)
            h2 = np.clip(h2, 0, 1)
            population[i] = h2

    # Validate and extract the best solution
    best_h = best_h  # already scaled
    best_c5_list = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(np.abs(best_c5_list)) * dx

    return best_h, best_c5, n_points
