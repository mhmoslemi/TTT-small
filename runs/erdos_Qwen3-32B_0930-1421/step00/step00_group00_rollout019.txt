import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp
import evaluate_erdos_solution

def evaluate_fitness(h, dx, pop_size=50):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(corr) * dx
    # Penalty term
    penalty = 1000.0 * (
        abs(np.sum(h) - (h.size / 2)) +
        np.maximum(0, np.max(h) - 1.0) +
        np.maximum(0, np.min(h))
    )
    return C5 + penalty  # minimize the objective

def project_to_simplex(h):
    total_mass = np.sum(h)
    target_mass = (len(h) / 2)
    if abs(total_mass - target_mass) > 1e-8:
        scale = target_mass / total_mass
        h = h * scale
    h_clipped = np.clip(h, -1e-8, 1.0)
    return h_clipped

def crossover(parent1, parent2, alpha=0.5):
    child = alpha * parent1 + (1 - alpha) * parent2
    return project_to_simplex(child)

def mutate(candidate, mutation_rate=0.05):
    noise = np.random.normal(0, 0.1, size=len(candidate))
    mask = np.random.rand(len(candidate)) < mutation_rate
    candidate[mask] += noise[mask]
    return project_to_simplex(candidate)

def optimize_h(initial_h):
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h_proj = project_to_simplex(initial_h)
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = project_to_simplex(h)
        return evaluate_fitness(h, dx)

    # Use LBFGS to refine the best h after GA
    result = minimize(objective, h_proj, jac=False, method="L-BFGS-B", bounds=[(0, 1) for _ in range(n_points)])
    h_opt = project_to_simplex(result.x.reshape(n_points))
    return h_opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_used = 0

    # Use the initial construction if available
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, copy=True)
    dx = 2.0 / n_points
    population = [h_initial]
    h_best = h_initial
    c5_best = evaluate_fitness(h_best, dx)

    # Generate initial population
    for _ in range(49):
        h = h_best + np.random.normal(0, 0.1, size=n_points)
        h = project_to_simplex(h)
        population.append(h)

    # Genetic algorithm loop
    max_gen = 200
    for generation in range(max_gen):
        # Evaluate fitness of each individual
        fitness_vals = [
            (i, evaluate_fitness(h, dx)) for i, h in enumerate(population)
        ]
        # Sort by fitness (low is better)
        fitness_vals.sort(key=lambda x: x[1])
        idxs, fitnesses = zip(*fitness_vals)

        # Check if time is over
        budget_used = time.time() - start_time
        if budget_used > budget_s * 0.97:
            break

        # Get best from this population
        idx_best = idxs[0]
        if fitness_vals[0][1] < c5_best:
            h_best = population[idx_best]
            c5_best = fitness_vals[0][1]

        # Select parents
        parents = [population[i] for i in idxs[: (len(population) // 5)]]

        # Crossover & Mutation to create new offspring
        new_offspring = []
        while len(new_offspring) < len(population) - len(parents):
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            child = crossover(p1, p2)
            if np.random.rand() < 0.1:
                child = mutate(child)
            new_offspring.append(child)

        population = parents + new_offspring

    # Post-processing: refine the best solution
    h_best = optimize_h(h_best)
    c5_best = evaluate_fitness(h_best, dx)

    # Final evaluation
    h_values = h_best
    c5_bound = c5_best
    n_points = len(h_values)
    return h_values, c5_bound, n_points
