import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.all(0 <= h) and np.all(h <= 1)
    assert np.sum(h) == n_points / 2

    max_iter = 500
    rho = 1.0
    eps_abs = 1e-4
    eps_rel = 1e-4

    H = cp.Parameter(shape=(n_points,), value=h, nonneg=True)
    H.value = h

    # Precompute shifted (1 - h) vectors for all lags
    corr_lags = []
    for k in range(n_points, 3 * n_points):
        if k >= n_points and k < 2 * n_points:
            idx = k - n_points
            corr = np.sum(H * (1 - cp.roll(H, idx)))
            corr_lags.append(corr)

    z = cp.Variable(len(corr_lags))
    obj = cp.Minimize(cp.max(z))
    constraints = [cp.vstack([z - corr for corr in corr_lags]) >= 0,
                   cp.sum(H) == n_points / 2,
                   H >= 0, H <= 1]

    problem = cp.Problem(obj, constraints)

    best_h = h.copy()
    best_c5 = np.inf
    last_improve = 0

    for i in range(max_iter):
        if time.time() - t0 > budget_s - 5:
            break

        H.value = best_h
        problem.solve(solver=cp.ECOS, maxiters=100)

        if problem.status in ['infeasible', 'unbounded']:
            print("Warning: problem status is", problem.status)
            break

        h_update = H.value
        h_update = np.clip(h_update, 0, 1)
        h_update *= (n_points / 2) / np.sum(h_update)

        H_current_correlate = np.correlate(h_update, 1 - h_update, mode='full') * dx
        C5 = np.max(H_current_correlate)

        if C5 < best_c5:
            best_c5 = C5
            best_h = h_update
            last_improve = i

        elif i - last_improve > 30:
            rho *= 2.0
            print(f"Increasing rho to {rho}")
            last_improve = i

    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    return best_h.tolist(), best_c5, n_points
