import numpy as np
import time
from scipy.signal import correlate

def evaluate_solution(h, n_points):
    dx = 2.0 / n_points
    h = h.copy()
    # Normalize to ensure the integral is 1
    h *= n_points / 2 / h.sum()
    # Compute the overlap integrals for all k
    # Compute (1 - h) for convolution
    one_minus_h = 1 - h
    # Full convolution to get all lags
    result_full = correlate(h, one_minus_h, mode='full') * dx
    # Max negative overlap is the maximum convolution result
    c5 = np.max(result_full)
    return c5

def project_to_simplex(h):
    n_points = len(h)
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    scale = n_points / (2 * current_sum)
    h *= scale
    h = np.clip(h, 0, 1)
    return h
