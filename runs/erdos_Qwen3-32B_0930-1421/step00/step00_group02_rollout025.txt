    n_points = len(h_init)

    # Step 2: Optimization parameters
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = np.inf
    best_time = time.time()

    # Rescale to ensure constraint is satisfied
    h[:] = np.clip(h, 0, 1)

    if h_sum != target_sum:
        h *= (target_sum / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = h.sum()

    # Ensure constraint is exactly satisfied after clip
    while h_sum < target_sum:
        idx = np.argmin(h)
        h[idx] += 0.001
        h_sum = h.sum()
        h = np.clip(h, 0, 1)

    # Main optimization loop
    while time.time() - start_time < budget_s - 5:  # leave time for final evaluation
        # Local search: perturb small subset
        indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        proposed_h = h.copy()
        noise = np.random.uniform(-0.01, 0.01, size=proposed_h[indices].shape)
        proposed_h[indices] += noise
        proposed_h = np.clip(proposed_h, 0, 1)

        # Ensure constraint
        proposed_h_sum = proposed_h.sum()
        if abs(proposed_h_sum - target_sum) > 1e-7:
            proposed_h *= target_sum / proposed_h_sum

        # Compute C5
        def compute_c5(optimized_h):
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_h_sum = optimized_h.sum()
            if abs(optimized_h_sum - target_sum) > 1e-7:
                optimized_h *= target_sum / optimized_h_sum

            ones_minus_h = 1.0 - optimized_h
            max_corr = -np.inf
            for k in range(1, n_points):
                if k >= optimized_h.shape[0] - 1:
                    continue
                corr = np.dot(optimized_h[:n_points - k], ones_minus_h[k:])
                max_corr = max(max_corr, corr * dx)

            for k in range(1, n_points):
                if k >= optimized_h.shape[0] - 1:
                    continue
                corr = np.dot(optimized_h[k:], ones_minus_h[:n_points - k])
                max_corr = max(max_corr, corr * dx)

            return max_corr

            break

            print(f"Time: {time.time()-start_time:.2f}s, C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
