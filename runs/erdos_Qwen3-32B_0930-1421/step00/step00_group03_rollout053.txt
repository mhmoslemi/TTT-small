import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Use the provided initial_h_values and configuration
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    assert np.allclose(np.sum(h), target_sum), "Initial h has wrong integral"

    def compute_c5(h):
        return evaluate_erdos_solution(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Simulated Annealing
    temp = 1.0
    cooling_rate = 0.99
    num_sa_steps = 1000
    early_stop_SA = 100

    for sa_iter in range(num_sa_steps):
        if time.time() - start_time > budget_s - 10:
            break

        # Perturb h
        h_candidate = np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)

        # Project to constraint
        current_sum = np.sum(h_candidate)
        diff = current_sum - target_sum
        if abs(diff) > 1e-8:
            idx = np.argsort(h_candidate)
            if diff > 0:  # Too large, decrease largest values
                h_candidate[idx[-int(np.ceil(abs(diff))):]] -= diff / np.ceil(abs(diff))
            else:  # Too small, increase smallest values
                h_candidate[idx[:int(np.ceil(abs(diff)))]] -= diff / np.ceil(abs(diff))
            h_candidate = np.clip(h_candidate, 0, 1)

        # Compute c5
        try:
            c5_candidate = compute_c5(h_candidate)
        except:
            continue

        # Acceptance
        if c5_candidate < best_c5 or np.random.rand() < np.exp((best_c5 - c5_candidate) / temp):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate

        temp *= cooling_rate

    # Gradient Descent
    num_gd_steps = 500
    learning_rate = 0.01
    grad = np.zeros(n_points)

    for gd_iter in range(num_gd_steps):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute gradient of max correlation (approximate)
        c5_vals = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(c5_vals)

        h_vals = h.copy()
        for i in range(n_points):
            # Perturb h slightly to estimate gradient
            h_up = h.copy()
            h_up[i] = np.clip(h[i] + 1e-5, 0, 1)
            c5_up = compute_c5(h_up)

            h_down = h.copy()
            h_down[i] = np.clip(h[i] - 1e-5, 0, 1)
            c5_down = compute_c5(h_down)

            grad[i] = (c5_up - c5_down) / 2e-5

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        current_sum = np.sum(h)
        diff = current_sum - target_sum
        if abs(diff) > 1e-8:
            idx = np.argsort(h)
            if diff > 0:
                h[idx[-int(np.ceil(abs(diff))):]] -= diff / np.ceil(abs(diff))
            else:
                h[idx[:int(np.ceil(abs(diff)))]] -= diff / np.ceil(abs(diff))
            h = np.clip(h, 0, 1)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
