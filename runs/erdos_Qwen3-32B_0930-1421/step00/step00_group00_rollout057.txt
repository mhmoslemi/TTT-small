```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    c_values = np.zeros(2 * n_points)
    for k in range(-n_points + 1, n_points):
        i0 = max(0, -k)
        i1 = min(n_points, n_points - k)
        c_values[k + n_points - 1] = (h[i0:i1] * (1 - h[k + i0:k + i1])).sum() * dx
    return max(c_values)

def initial_h_values():
    h = np.zeros(45)
    h[:23] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    n_points = len(initial_h_values())
    h = initial_h_values().copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    dx = 2.0 / n_points
    alpha = 0.1
    beta = 0.7
    max_iter = 2000
    n_iter = 0

    while n_iter < max_iter and time.time() - t_start < budget_s:
        n_iter += 1
        # Compute cross-correlation
        h_arr = h.reshape(-1)
        ones_minus_h = 1 - h_arr
        corr = correlate(h_arr, ones_minus_h, mode="full")
        corr *= dx
        corr_indices = np.arange(len(corr))
        # Consider only overlaps with shift k ∈ [0, 2]
        valid_corr_indices = (corr_indices >= n_points - 1) & (corr_indices < 3 * n_points - 1)
        valid_corr = corr[valid_corr_indices]
        if len(valid_corr) == 0:
            break
        k_max_index = np.argmax(valid_corr)
        k_max = corr_indices[valid_corr_indices][k_max_index]
        k_offset = k_max - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_offset) % n_points
            grad[i] = ones_minus_h[j] * dx

        # Update h with clipping and projection
        h_proposed = h - alpha * grad
        h_proposed = np.clip(h_proposed, 0, 1)
        excess = 0
        deficit = 0
        excess_mask = h_proposed > 1 - 1e-6
        deficit_mask = h_proposed < 1e-6

        h_proposed[excess_mask] = 1.0
        h_proposed[deficit_mask] = 0.0
        excess = np.sum(h_proposed[excess_mask]) - np.sum(h[excess_mask])
        deficit = np.sum(h[deficit_mask]) - np.sum(h_proposed[deficit_mask])

        if excess < 0:
            deficit += excess
            excess = abs(excess)

        # Redistribute excess/deficit
        if excess > 1e-8 or deficit > 1e-8:
            diff = excess - deficit
            mask = (h_proposed > 1e-8) & (h_proposed < 1 - 1e-8)
            n_points_adj = np.sum(mask)
            if n_points_adj > 0:
                adjustment = diff / n_points_adj
                h_proposed[mask] += adjustment
                h_proposed = np.clip(h_proposed, 0, 1)

        c5_proposed = evaluate_erdos_solution(h_proposed)
        if c5_proposed < best_c5:
            best_h = h_proposed.copy()
            best_c5 = c5_proposed
            h = h_proposed.copy()
        else:
            alpha *= beta

    return best_h, best_c5, n_points
```
