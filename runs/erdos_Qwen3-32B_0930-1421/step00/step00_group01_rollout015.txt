import numpy as np
import time

import cvxpy as cp

from scipy.optimize import minimize, Bounds
from math import exp, log

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = h_values
    correlate_output = np.correlate(h, 1 - h, mode="full")
    vals = correlate_output[n_points - 1: 2 * n_points - 1] * dx
    return np.max(vals)

def compute_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr[len(h) - 1: 2 * len(h) - 1] * dx
    return np.max(vals)

def smooth_max(arr, beta):
    return (1.0 / beta) * log(np.sum(exp(beta * arr)))

def update_h_sum(h):
    s = h.sum()
    target = len(h) / 2
    diff = target - s
    h += diff / len(h)
    return h.clip(0, 1)
