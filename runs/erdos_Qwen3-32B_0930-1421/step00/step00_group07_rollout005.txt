import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    dx = 2.0 / 80
    h = initial_h_values.copy()
    n_points = 80

    def h_to_intervals(h):
        intervals = []
        a = None
        for i in range(n_points):
            x = i * dx
            if h[i] > 0.5 and a is None:
                a = x
            elif h[i] <= 0.5 and a is not None:
                b = x
                if b > a + 1e-10:
                    intervals.append((a, b))
                a = None
        if a is not None and a < 2:
            intervals.append((a, 2.0))
        return intervals

    def intervals_to_h(intervals):
        h = np.zeros(n_points)
        delta_x = 2.0 / n_points
        for a, b in intervals:
            left = int(np.floor(a / delta_x))
            right = int(np.ceil(b / delta_x))
            for i in range(left, right):
                if i >= n_points:
                    break
                h[i] = 1.0
        return h

    def compute_c5(h):
        h_bin = (h > 0.5).astype(float)
        corr = np.correlate(h_bin, 1 - h_bin, mode="full") * dx
        return np.max(corr)

    intervals = h_to_intervals(h)
    best_intervals = intervals.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    iter_count = 0
    step_size = 0.05
    temp = 1.0
    improve_thresh = 100
    no_improvement = 0

    while time.time() - t_start < budget_s - 10:
        iter_count += 1

        if iter_count % 500 == 0:
            print(f"Iteration {iter_count}, best C5: {best_c5}")

        # Sample interval to adjust
        intervals_list = best_intervals
        if not intervals_list:
            break

        i = np.random.randint(len(intervals_list))
        a, b = intervals_list[i]
        delta = np.random.normal(0, step_size / 2)
        prop_type = np.random.choice(['start', 'end'])

        if prop_type == 'start' and i > 0:
            a_prev, b_prev = intervals_list[i - 1]
            new_a = max(a + delta, a_prev + 1e-5)
            new_b = b
        elif prop_type == 'end' and i < len(intervals_list) - 1:
            a_next, b_next = intervals_list[i + 1]
            new_a = a
            new_b = min(b + delta, a_next - 1e-5)
        else:
            # Try to shift both endpoints to widen or narrow
            new_a = a + np.random.normal(0, step_size / 3)
            new_b = b + np.random.normal(0, step_size / 3)
            new_a = max(min(new_a, b - 1e-5), 0.0)
            if i < len(intervals_list) - 1:
                a_next, b_next = intervals_list[i + 1]
                new_b = min(new_b, a_next - 1e-5)
            else:
                new_b = min(new_b, 2.0)

        if new_a >= new_b:
            continue

        # Check overlaps
        valid = True
        for j in range(len(intervals_list)):
            if j == i:
                continue
            a_j, b_j = intervals_list[j]
            if not (new_b <= a_j - 1e-6 or new_a >= b_j + 1e-6):
                valid = False
                break

        if not valid:
            continue

        # Try to insert new a,b in place of the old one
        new_intervals = intervals_list[:i] + [(new_a, new_b)] + intervals_list[i+1:]

        if len(new_intervals) == 0:
            continue

        # Normalize to ensure total measure is 1
        total_measure = sum(b - a for a, b in new_intervals)
        if abs(total_measure - 1.0) > 1e-5:
            expand_ratio = (1.0 - total_measure) / total_measure * 0.5
            expand_a, expand_b = new_intervals[i]
            expand_a += expand_ratio
            expand_b -= expand_ratio
            if expand_b > expand_a + 1e-5:
                new_intervals[i] = (expand_a, expand_b)
            else:
                continue

        new_intervals = new_intervals.copy()
        new_h = intervals_to_h(new_intervals)
        total = np.sum(new_h)
        if abs(total - 40.0) > 0.5:
            continue

        new_c5 = compute_c5(new_h)

        # Accept if better or by metallurgy
        if new_c5 < best_c5:
            best_intervals = new_intervals.copy()
            best_c5 = new_c5
            best_h = new_h.copy()
            no_improvement = 0
        elif np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
            best_intervals = new_intervals.copy()
            best_c5 = new_c5
            best_h = new_h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        temp *= 0.995
        if no_improvement >= 500 and iter_count >= 200:
            step_size *= 0.9
            no_improvement = 0

        if best_c5 < 0.37 and iter_count > 100:
            break

    return best_h, best_c5, n_points
