import numpy as np
import time
from scipy.signal import correlate

def evaluate(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones = np.ones_like(h)
    corr = correlate(h, ones - h, mode='full')
    return np.max(corr * dx)

def normalize_h(h):
    total = np.sum(h)
    target = len(h) / 2
    return h * target / total

def mutate(h, noise_rate=0.03):
    h = h.copy()
    idxs = np.random.choice(len(h), size=int(noise_rate * len(h)), replace=False)
    h[idxs] = np.random.rand(len(idxs))
    h = np.clip(h, 0, 1)
    return normalize_h(h)

def crossover(p1, p2):
    n = len(p1)
    split = np.random.randint(n // 4, 3 * n // 4)
    return normalize_h(np.concatenate((p1[:split], p2[split:])))

def tournament_select(pop, fitness, num=2):
    idxs = np.random.choice(len(pop), size=2 * num, replace=False)
    return np.squeeze(pop[np.argsort(fitness[idxs])[-num:]])
