import numpy as np
import cvxpy as cp
import time

from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=1100):
    start = time.time()

    # Initialize from parent construction or build a new one
    if not hasattr(initial_h_values, "__len__") or len(initial_h_values) == 0:
        raise ValueError("Initial_h_values must be non-empty and defined.")

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return corr * dx

    def objective(h):
        overlap = compute_overlap(h)
        return np.max(overlap)

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:  # Edge-case safety
            return np.zeros_like(h)
        h *= (n_points / 2) / total
        h = np.clip(h, 0, 1)  # Final clip in case scaling overshoots [0,1]
        return h

    h = normalize_h(h)
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_time = 0.0
    last_improvement = 0.0

    MAX_ITER = 5000
    iter = 0

    while time.time() - start < budget_s:
        overlap = compute_overlap(h)
        k_star = int(np.argmax(overlap))  # index of worse shift
        overlap_k = overlap[k_star]

        # Compute gradient for k_star
        P = k_star
        N = len(h)
        gradient = np.zeros_like(h)
        for i in range(N - P):
            gradient[i] -= h[i + P]
        for i in range(N - P):
            gradient[i + P] += h[i]

        # L2 projected update using solver
        dh = cp.Variable(n_points)
        loss = cp.sum_squares(dh - 0.05 * gradient)
        constraints = [
            dh >= -h,
            dh <= 1 - h,
            cp.sum(dh) == 0.0  # keep the sum invariant
        ]
        objective_cvx = cp.Minimize(loss)
        prob = cp.Problem(objective_cvx, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h += dh.value
        except:
            pass
        h = normalize_h(h)

        # Evaluate
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time() - start

        iter += 1
        if iter % 100 == 0:
            print(f"Iter: {iter}, best_c5: {best_c5:.7f}")
        if time.time() - last_improvement > 0.1 * budget_s:
            print("No improvement timeout")
            break

    return (best_h, best_c5, n_points)
