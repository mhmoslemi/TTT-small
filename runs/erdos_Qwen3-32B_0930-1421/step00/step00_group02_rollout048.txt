import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=45):
    import warnings
    warnings.filterwarnings("ignore", category=UserWarning, module="cvxpy")
    import evaluate_erdos_solution as evaluator
    
    # Fix random seed
    np.random.seed(seed)
    
    # Initialize h with provided or default values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # If no initial_h_values provided, use uniform distribution
        h = np.full(n_points, 0.5 / n_points * 2.0)
        idx = np.arange(n_points)
        np.random.shuffle(idx)
        for i in idx[:n_points // 2]:
            h[i] = 1.0
        
    h = h / h.sum() * n_points / 2  # normalize to constraint
    h = np.clip(h, 0, 1)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluator.evaluate_erdos_solution(best_h)

    # Optimization loop
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var <= 1]
        constraints += [cp.sum(h_var) == n_points / 2]

        # Compute overlap integrals as LP softmax approx
        shifted_h = cp.Constant(1 - h)
        objective_terms = []
        
        for k in range(1, n_points - 1):
            if k < n_points // 2:
                overlap = cp.sum(h_var * shifted_h[k:])
                objective_terms.append(overlap)
        
        # Minimize the maximum overlap using log-sum-exp approximation
        objective = cp.Minimize(cp.max(objective_terms))
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = np.clip(h_var.value, 0, 1)
                new_h = new_h / new_h.sum() * n_points / 2
                new_h = np.clip(new_h, 0, 1)
                
                c5 = evaluator.evaluate_erdos_solution(new_h)
                
                if c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = c5
                    h = new_h  # Update for next iteration
        except Exception as e:
            pass

        # Budget management
        if time.time() - start_time >= budget_s:
            break

    # Final clean-up
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * n_points / 2
    best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, n_points
