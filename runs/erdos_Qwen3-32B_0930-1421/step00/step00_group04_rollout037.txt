def evaluate_erdos_solution(h_values):
    h = np.asarray(h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    ones_h = 1.0 - h
    corr = correlate(h, ones_h, mode='full')
    c5 = np.max(corr) * dx
    return c5

    best_c5 = float('inf')

        c5 = evaluate_erdos_solution(ind)
        return c5

            break
        population = np.array(population)
        fitness = np.array([evaluate(ind) for ind in population])
        
        best_index = np.argmin(fitness)
        if fitness[best_index] < best_c5:

            best_h = population[best_index].copy()
        
        # Tournament selection
        selected_indices = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            selected_indices.append(idx1 if fitness[idx1] < fitness[idx2] else idx2)
        selected_indices = np.array(selected_indices)
        selected = population[selected_indices]

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size:
                a, b = selected[i], selected[i+1]
                # Single-point crossover
                idx = np.random.randint(0, n_points)
                child1 = np.concatenate((a[:idx], b[idx:]))
                child2 = np.concatenate((b[:idx], a[idx:]))
                # Normalize to preserve the sum constraint
                child1 = child1 / child1.sum() * (n_points / 2)
                child2 = child2 / child2.sum() * (n_points / 2)
                child1 = np.clip(child1, 0, 1)
                child2 = np.clip(child2, 0, 1)
                offspring.append(child1)
                offspring.append(child2)

        # Mutation
        for ch in offspring:
            if np.random.rand() < mutate_rate:
                mu_idx = np.random.choice(n_points, size=int(mutate_rate * n_points), replace=False)
                ch[mu_idx] += np.random.randn(len(mu_idx)) * 0.1
                ch = np.clip(ch, 0, 1)
                ch = ch / ch.sum() * (n_points / 2)
        
        # Replace population
        population = offspring

    return best_h, best_c5, n_points
