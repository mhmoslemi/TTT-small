import numpy as np
import math
import time

def evaluate_overlap_integral(h, dx):
    h = np.asarray(h)
    kernels = [np.roll(h, shift=k) for k in range(1, len(h))]
    kernels = [1 - k for k in kernels]
    corrs = [np.dot(h, k) * dx for k in kernels]
    return max(corrs)

def project_constraints(h, dx, n_points):
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / (np.sum(h) + 1e-12)
    return h

def evaluate_erdos_solution(h):
    h = np.asarray(h)
    dx = 2.0 / len(h)
    shift_results = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(shift_results[shift_results.shape[0] // 2 - 1 : shift_results.shape[0] // 2 + len(h)])

def run(seed=42, budget_s=1100, n_points=70, max_iter=5000, lr=0.01, min_lr=0.0001, lr_decay=0.998, patience=30):
    np.random.seed(seed)
    start_time = time.time()
    t0 = start_time

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[n_points // 4: n_points // 4 + n_points // 2] = 1
        h *= (n_points / 2) / np.sum(h)

    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_time = time.time() - t0
    no_progress_count = 0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute gradient
        grads = np.zeros(n)
        for k in range(1, n):
            h_k = np.roll(h, -k)
            h_k = h_k[:n - k]
            dh = h[:n - k] * (1 - h_k) * dx
            grads[:n - k] += dh
            h_k = np.roll(h, k)
            h_k = h_k[k:]
            dh = h[k:] * (1 - h_k) * dx
            grads[k:] += dh
        # Normalize gradient
        grads = 2 * h - 2 * h * grads  # Derivative of the max overlap
        # Project gradient ascent direction
        grad_proj = grads
        h_new = h + lr * grad_proj
        h_new = project_constraints(h_new, dx, n)
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_progress_count = 0
        else:
            no_progress_count += 1
        # Check learning rate decay
        if lr > min_lr:
            lr *= lr_decay
        if no_progress_count > patience:
            break
        h[:] = h_new[:]

    return (best_h.tolist(), best_c5, n)
