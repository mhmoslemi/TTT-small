import numpy as np
from scipy.signal import correlate

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    c5 = 0.0
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(h_values, k)
        shifted = shifted[n_points:]  # only the [2, 4] interval is relevant
        overlap = np.sum(h_values[n_points - k:] * (1 - shifted)) * dx
        c5 = max(c5, overlap)
    return c5
