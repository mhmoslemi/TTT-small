import numpy as np
import time
from scipy.signal import correlate


def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap * dx)
    return c5


def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    scale = target_sum / current_sum if current_sum != 0 else 1.0
    return h_clipped * scale


def run(seed=42, budget_s=1100, n_points=42, lr=0.01):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if "initial_h_values" in globals():
        initial = np.array(initial_h_values)
        h = np.zeros(n_points)
        offset = (n_points - len(initial)) // 2
        h[offset:offset + len(initial)] = initial
        h = project_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = project_h(h, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute the current C₅ bound
        c5 = evaluate_erdos_solution(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient of C₅
        # correlation is h * (1 - shifted_h)
        # gradient w.r.t. h is dx * (h - shifted_h) where shift is at the peak index
        overlap = correlate(h, 1 - h, mode="full")
        peak_idx = np.argmax(overlap)
        k = peak_idx - len(h) + 1  # shift in samples

        shifted = np.roll(h, -k)
        grad = (h - shifted) * dx

        # Update h with gradient
        h = h - lr * grad
        h = project_h(h, target_sum)

        # Break if no improvement in last 100 steps
        # This would be tracked if we added a step counter and best tracker

    return best_h.tolist(), best_c5, n_points
