import numpy as np
import cvxpy as cp
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 95  # Match the size of the initial_h_values

    # Use the initial_h_values if provided
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    alpha = 0.5  # Mixing factor for h_shifted
    h_prev = h.copy()
    h_shifted = h.copy()
    best_c5 = float('inf')
    no_improvement_counter = 0
    max_iterations = 2000
    improvement_threshold = 1e-4
    evaluation_stride = 50

    h_var = cp.Variable(n_points)
    dx = 2.0 / n_points

    while time.time() - start_time < budget_s and no_improvement_counter < 200:
        cor = np.correlate(h_shifted, 1 - h, mode='full') * dx
        k_max = np.argmax(cor)
        C5 = cor[k_max]

        if C5 < best_c5:
            best_c5 = C5
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        if no_improvement_counter >= 50:
            alpha = max(alpha * 0.9, 0.1)

        # Form convex problem using current fixed h_shifted
        cor_fixed = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            cor_fixed[i] += h_shifted[j]  # Since 1 - h[j] is variable part

        obj = 1 - cp.sum(cp.multiply(h_var, cor_fixed)) * dx
        constraints = [
            cp.sum(h_var) == n_points / 2,
            cp.Constant(0) <= h_var,
            h_var <= cp.Constant(1)
        ]
        problem = cp.Problem(cp.Minimize(obj), constraints)

        try:
            # Use CVXOPT if available, otherwise fallback
            problem.solve(solver=cp.CVXOPT, verbose=False)
        except:
            problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status != cp.OPTIMAL:
            # Fallback to SCS for stability if needed
            problem.solve(solver=cp.SCIP, verbose=False)

        h = h_var.value
        h = np.clip(h, 0, 1)
        h_shifted = alpha * h + (1 - alpha) * h_prev
        h_prev = h.copy()

        # Periodically check all overlaps to ensure we track global maximum
        if (time.time() - start_time) % evaluation_stride < 1 and time.time() - start_time < budget_s:
            cor_full = np.correlate(h, 1 - h, mode='full') * dx
            best_c5 = np.max(cor_full)

    # Final validation before returning
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    cor_full = np.correlate(h, 1 - h, mode='full') * dx
    c5_final = np.max(cor_full)

    return h, c5_final, n_points
