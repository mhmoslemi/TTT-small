from scipy.signal import correlate

def evaluate_overlap(h, n_points):
    dx = 2.0 / n_points
    h_binary = (h > 0.5).astype(np.float64)
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    return np.max(corr) * dx

    current_sum = np.sum(h)
    while current_sum != target_sum:

    patience = 1000

        indices = np.where(h == 1)[0]
        one_idx = np.random.choice(indices)
        indices = np.where(h == 0)[0]
        zero_idx = np.random.choice(indices)
        candidate = h.copy()
        candidate[one_idx], candidate[zero_idx] = 0, 1

        current_c5 = evaluate_overlap(h, n_points)
        candidate_c5 = evaluate_overlap(candidate, n_points)

        if candidate_c5 < current_c5:
            h = candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                no_improvement = 0
        else:
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = candidate.copy()
            else:
                no_improvement += 1

        T *= alpha
        if T < min_T:
            T = max(min_T, T * 0.5)
            alpha = max(0.999, alpha * 0.999)

        if no_improvement > patience:
            # Reset temperature
            T = max(T * 5, min_T)
            no_improvement = 0
            alpha = max(0.99, alpha * 0.99)

    best_h = np.array(best_h, dtype=int)
    final_c5 = evaluate_overlap(best_h, n_points)
