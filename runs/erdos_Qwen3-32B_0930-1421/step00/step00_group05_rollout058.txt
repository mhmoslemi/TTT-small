import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Retrieve initial h_values if available
    try:
        initial_h_values = initial_h_values.copy()
        n_points = len(initial_h_values)
    except:
        n_points = 70
        initial_h_values = np.zeros(n_points)
        idx = int(n_points / 2)
        initial_h_values[idx - 1:idx + 1] = 0.5
        initial_h_values = initial_h_values / initial_h_values.sum() * n_points / 2
    
    h = np.copy(initial_h_values)
    
    # Normalize the initial solution
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    if h_sum != n_points / 2:
        h = h * (n_points / 2) / h_sum
    
    start_time = time.time()
    best_c5 = np.inf
    best_h = h.copy()
    
    learning_rate = 0.1
    decay_factor = 0.5
    
    dx = 2.0 / n_points
    max_shifts = n_points * 2 - 1
    indices_half_window = max_shifts // 2
    
    while time.time() - start_time < budget_s:
        # Evaluate overlap integrals for all shifts
        C = np.correlate(h, 1 - h, mode="full") * dx
        
        k_max = np.argmax(C)
        
        # Find the gradient (direction to change h to reduce overlap)
        # Build 1-h shifted by k_max
        overlap_shifted = np.zeros_like(h)
        if k_max < n_points:
            overlap_shifted[k_max:] = 1 - h[:n_points - k_max]
        else:
            overlap_shifted[:k_max - n_points + 1] = 1 - h[k_max - n_points + 1:]
        
        # Gradient: -overlap_shifted
        grad = -overlap_shifted
        
        with np.errstate(invalid='ignore'):
            learning_rate = 0.1
            for _ in range(5):
                candidate_h = h + grad * learning_rate
                candidate_h = np.clip(candidate_h, 0.0, 1.0)
                candidate_h_sum = np.sum(candidate_h)
                if candidate_h_sum > 0:
                    candidate_h = candidate_h * (n_points / 2) / candidate_h_sum
                
                # Compute new overlap
                C_new = np.correlate(candidate_h, 1 - candidate_h, mode="full") * dx
                c5_new = np.max(C_new)

                if np.isfinite(c5_new) and c5_new < best_c5:
                    h = candidate_h.copy()
                    best_c5 = c5_new
                    best_h = h.copy()
                    break

                learning_rate *= decay_factor

        # Check convergence
        if best_c5 < C[k_max]:
            break

    return best_h.tolist(), best_c5, n_points
