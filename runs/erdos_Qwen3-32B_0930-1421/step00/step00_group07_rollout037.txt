import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    N = len(initial_h_values)
    h_vals = np.copy(initial_h_values)
    dx = 2.0 / N

    intervals = []

    # Convert initial_h_values into interval list
    i = 0
    while i < N:
        if h_vals[i] > 0.5:
            start = i * dx
            j = i
            while j < N and h_vals[j] > 0.5:
                j += 1
            end = j * dx
            intervals.append((start, end))
            i = j
        else:
            i += 1

    best_intervals = intervals.copy()
    best_c5 = evaluate_configuration(intervals, N, dx)

    # Annealing parameters
    T = 1.0
    cooling_rate = 0.99

    step_size = 0.01
    iterations = 0
    no_improvement = 0
    max_no_improvement = 300
    max_iterations = 500

    while time.time() - start_time < budget_s * 0.995 and iterations < max_iterations and no_improvement < max_no_improvement:
        intervals = best_intervals.copy()
        intervals = adjust_intervals(intervals, step_size, N, dx)
        current_c5 = evaluate_configuration(intervals, N, dx)
        
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / T):
            best_intervals = intervals.copy()
            best_c5 = current_c5
            no_improvement = 0
            if step_size > 1e-3:
                step_size *= 0.95
        else:
            no_improvement += 1
        
        T *= cooling_rate
        iterations += 1

    h_values = generate_h_values(best_intervals, N, dx)
    c5_bound = evaluate_configuration(best_intervals, N, dx)

    return h_values, c5_bound, N

def evaluate_configuration(intervals, n_points, dx):
    h = np.zeros(n_points)
    for (a, b) in intervals:
        start_idx = int(np.floor(a / dx))
        end_idx = int(np.ceil(b / dx))
        h[start_idx:end_idx] = 1
        if end_idx - start_idx == 0:
            h[start_idx] = 1  # ensure smallest non-zero segment

    # Normalize h to have sum(h)*dx = 1
    h_sum = np.sum(h) * dx
    if not np.isclose(h_sum, 1.0):
        scale_factor = 1.0 / h_sum
        h = np.clip(h * scale_factor, 0, 1)

    h_1_h = 1 - h
    corr = np.array([np.correlate(h, h_1_h[k::], mode='valid') for k in range(2 * n_points)]).flatten()
    c5 = np.max(corr * dx)
    return float(c5)

def generate_h_values(intervals, n_points, dx):
    h = np.zeros(n_points)
    for (a, b) in intervals:
        start_idx = int(np.floor(a / dx))
        end_idx = int(np.ceil(b / dx))
        h[start_idx:end_idx] = 1
        if end_idx - start_idx == 0:
            h[start_idx] = 1
    h_sum = np.sum(h) * dx
    if not np.isclose(h_sum, 1.0):
        scale_factor = 1.0 / h_sum
        h = np.clip(h * scale_factor, 0, 1)
    return h.tolist()

def adjust_intervals(intervals, step_size, n_points, dx):
    # Try multiple candidates per interval to explore better configurations
    candidates = []
    for _ in range(2 * len(intervals)):
        intervals_candidate = intervals.copy()
        i = np.random.randint(len(intervals_candidate))
        (a, b) = intervals_candidate[i]
        delta = np.random.normal(0, step_size)
        
        new_int = None
        if np.random.rand() < 0.5:
            # Adjust start
            new_a = max(0.0, min(a + delta, b - dx / 2.0))
            new_b = b
            if i > 0:
                new_a = max(new_a, intervals_candidate[i-1][1] + dx / 2.0)
            if i < len(intervals_candidate) - 1:
                new_b = min(new_b, intervals_candidate[i+1][0] - dx / 2.0)
            new_int = [(new_a, new_b)]
        else:
            # Adjust end
            new_a = a
            new_b = min(2.0, max(b + delta, a + dx / 2.0))
            if i > 0:
                new_a = max(new_a, intervals_candidate[i-1][1] + dx / 2.0)
            if i < len(intervals_candidate) - 1:
                new_b = min(new_b, intervals_candidate[i+1][0] - dx / 2.0)
            new_int = [(new_a, new_b)]
        
        if new_int and (new_int[0][1] - new_int[0][0]) > 0:
            if i >= len(intervals_candidate):
                i = len(intervals_candidate) - 1
            intervals_candidate[i] = new_int[0]
            candidates.append(intervals_candidate)

    # Select candidate with smallest C₅
    results = []
    for candidate in candidates:
        if not is_valid(candidate):
            continue
        c5 = evaluate_configuration(candidate, n_points, dx)
        results.append((candidate, c5))
    
    if not results:
        return intervals
    
    best_candidate, best_c5 = min(results, key=lambda x: x[1])
    return best_candidate

def is_valid(intervals):
    """Check if the intervals are valid and non-overlapping"""
    if not intervals:
        return False
        
    total_length = sum(b - a for a, b in intervals)
    if not (abs(total_length - 1) < 1e-8):
        return False
        
    for i in range(len(intervals) - 1):
        if intervals[i][1] > intervals[i+1][0]:
            return False
    return True
