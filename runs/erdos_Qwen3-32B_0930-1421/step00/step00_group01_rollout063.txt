import numpy as np
import time
from scipy.optimize import minimize, Bounds
from optimize_erdos_solution_with_scipy import evaluate_erdos_solution
import cvxpy as cp

def smooth_max(objs, eta):
    return (1.0 / eta) * np.log(np.sum(np.exp(objs * eta)))

def objective_smooth(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode="full")
    return smooth_max(corr * dx, eta=10.0)

def objective_true(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def project_backend(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2.0) / total
    h *= scale
    return np.clip(h, 0, 1)

def gradient_descent_step(h, n_points):
    def obj(h_flat):
        return objective_smooth(h_flat, n_points)
    res = minimize(obj, h, method="SLSQP", bounds=Bounds(0, 1),
                   constraints={"type": "eq", "fun": lambda x: np.sum(x) - n_points / 2.0})
    return project_backend(res.x, n_points)

def perturb_h(h, scale=0.1):
    n_points = len(h)
    return h + scale * np.random.randn(n_points)

def simulated_annealing_step(h, temp):
    candidate = perturb_h(h)
    o1 = objective_true(h, len(h))
    o2 = objective_true(candidate, len(candidate))
    if o2 < o1 or np.random.random() < np.exp((o1 - o2) / temp):
        return candidate
    return h

def run(seed=42, budget_s=1100, n_points=47):
    np.random.seed(seed)

        h = perturb_h(best_h, scale=0.1)

            budget_used = time.time() - time_start
