import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values
    n_points = len(h0)
    dx = 2.0 / n_points

    # Normalize the initial h vector to satisfy the sum constraint
    h_sum = np.sum(h0)
    desired_sum = n_points / 2
    h0 = h0 * (desired_sum / h_sum)
    h0 = np.clip(h0, 0.0, 1.0)

    t0 = time.time()

    def compute_overlap(h):
        h = np.ascontiguousarray(h)
        hh = h[None, :]
        for k in range(n_points):
            if k == 0:
                corr = np.sum(h * (1 - h))
            else:
                corr0 = np.sum(h[k:] * (1 - h[:n_points - k]))
                corr1 = np.sum(h[:n_points - k] * (1 - h[k:]))
                corr = max(corr0, corr1)
            if k == 0:
                C5 = corr * dx
            else:
                C5 = max(C5, corr * dx)
        return C5

    def objective(h_vec):
        h_vec = h_vec.reshape(n_points)
        return compute_overlap(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Bounds for h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    cons = [{'type': 'eq', 'fun': constraint_sum}]
  
    def callback(x):
        nonlocal best_h, best_C5
        h_candidate = x.reshape(n_points)
        current_C5 = objective(h_candidate)
        if current_C5 < best_C5:
            best_h = h_candidate.copy()
            best_C5 = current_C5
        return True  # Continue optimization

    best_h = h0.copy()
    best_C5 = compute_overlap(best_h)

    # Run optimization within time budget
    t_start = time.time()
    res = scipy.optimize.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        callback=callback,
        tol=1e-6
    )

    # Clip final solution and re-normalize if needed
    if not res.success:
        print("Optimization did not converge, returning best found solution.")
    h_optimal = res.x
    h_optimal = np.clip(h_optimal, 0.0, 1.0)
    h_sum = np.sum(h_optimal)
    h_optimal = h_optimal * (desired_sum / h_sum)
    h_optimal = np.clip(h_optimal, 0.0, 1.0)

    final_C5 = compute_overlap(h_optimal)

    if final_C5 < best_C5:
        best_h = h_optimal
        best_C5 = final_C5

    return best_h, best_C5, n_points
