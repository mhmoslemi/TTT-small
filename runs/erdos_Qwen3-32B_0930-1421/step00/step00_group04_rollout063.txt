import numpy as np

def compute_overlap_cost(h, dx):
    full_correlate = np.correlate(h, 1 - h, mode='full')
    return np.max(full_correlate) * dx

def loss_func(h_flat, n_points):
    h = h_flat.reshape(-1)
    dx = 2.0 / n_points
    # Enforce constraints
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    h_rescaled = h_clipped / total * n_points / 2.0
    return compute_overlap_cost(h_rescaled, dx)

def objective(x, info):
    return loss_func(x, info['n_points'])

def grad_objective(x, info):
    n_points = info['n_points']
    dx = 2.0 / n_points

    h = x.reshape(-1)
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    h_rescaled = h_clipped / total * n_points / 2.0

    # Compute max overlap index
    full_correlate = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
    max_index = np.argmax(full_correlate)

    pad_h = np.zeros(2 * n_points)
    pad_h[:n_points] = h_rescaled

    grad = np.zeros(n_points)

    offset = max_index
    if offset < n_points:
        grad[offset:] = -(1 - h_rescaled[:n_points - offset])
        grad[offset:] *= n_points / 2
    else:
        offset = 2 * n_points - 1 - offset
        grad[:n_points - offset] = (1 - h_rescaled[offset:])

    # Apply Jacobian from clipping and rescaling
    grad *= h_clipped / (n_points / 2.0) * h_rescaled / (1 / h_clipped.mean())

    return grad.flatten()

def constrained_gradient_descent(
    h_start, 
    loss_func,
    grad_func,
    n_points,
    lr=0.05,
    num_steps=300,
    restart_interval=50,
):
    t_start = time.monotonic()
    h_best = h_start.copy()
    c5_best = loss_func(h_start, n_points)
    
    for step in range(num_steps):
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s:
            return h_best, c5_best

        loss = loss_func(h_best, n_points)

        if loss < c5_best:
            c5_best = loss
            h_best = h_best.copy()

        grad = grad_func(h_best, {'n_points': n_points})
        h_candidate = h_best - lr * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate /= h_candidate.sum() * (2.0 / n_points)  # normalize to 1

        if compute_overlap_cost(h_candidate, 2.0 / n_points) < c5_best:
            h_best = h_candidate
        else:
            # If no improvement, try small simplex perturbations
            directions = np.eye(n_points) * 0.01
            for d in directions:
                h_new = h_best + d
                h_new = np.clip(h_new, 0, 1)
                h_new /= h_new.sum() * (2.0 / n_points)
                new_cost = compute_overlap_cost(h_new, 2.0 / n_points)
                if new_cost < c5_best:
                    h_best = h_new
                    c5_best = new_cost
                    break

        if (step + 1) % restart_interval == 0:
            h_best = np.random.uniform(0, 1, n_points)
            h_best /= h_best.sum() * (2.0 / n_points)

    return h_best, c5_best
