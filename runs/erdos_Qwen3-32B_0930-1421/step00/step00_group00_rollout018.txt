import numpy as np
import time
from scipy.optimize import minimize, check_grad
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    h = np.array(h, dtype=float)
    dx = 2.0 / n_points
    h_padded = np.concatenate([h, np.zeros_like(h)])
    one_minus_h = 1.0 - h
    corr = correlate(h_padded, one_minus_h, mode="valid")
    C5 = np.max(corr) * dx
    return C5

def project_to_constraints(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    scaling_factor = n_points / 2.0 / current_sum if current_sum != 0 else 1.0
    h = np.clip(h * scaling_factor, 0.0, 1.0)
    return h

def ga_step(population, n_points):
    new_population = []
    top_idx = np.argsort([evaluate_C5(h, n_points) for h in population])[:int(0.2*len(population))]
    top = [population[i] for i in top_idx]

    for _ in range(len(population)//2 - len(top)):
        parent = top[np.random.choice(len(top))]
        child = parent * (1 - 0.1 * np.random.rand(n_points)) + 0.1 * np.random.rand(n_points)
        child = project_to_constraints(child, n_points)
        new_population.append(child)
    
    return top + new_population

def optimize_C5(h, n_points, budget_s):
    start = time.time()
    ga_pop = [h + 0.05 * np.random.randn(n_points) for _ in range(50)]
    ga_pop = [project_to_constraints(g, n_points) for g in ga_pop]
    best_h = h
    best_C5 = evaluate_C5(h, n_points)

    while time.time() - start < budget_s - 1.0:
        ga_pop = ga_step(ga_pop, n_points)
        current_C5 = [evaluate_C5(g, n_points) for g in ga_pop]
        idx = np.argmin(current_C5)
        current_best = ga_pop[idx]
        current_best_C5 = current_C5[idx]
        if current_best_C5 < best_C5:
            best_h = current_best
            best_C5 = current_best_C5

        # Gradient refinement
        def objective(x):
            return evaluate_C5(x, n_points)
        
        res = minimize(objective, best_h, method="L-BFGS-B", bounds=[(0, 1) for _ in best_h], jac=check_grad(objective, lambda x: np.gradient(objective(x)), best_h))
        refined_h = project_to_constraints(res.x, n_points)
        refined_C5 = evaluate_C5(refined_h, n_points)
        if refined_C5 < best_C5:
            best_h = refined_h
            best_C5 = refined_C5

    return best_h, best_C5
