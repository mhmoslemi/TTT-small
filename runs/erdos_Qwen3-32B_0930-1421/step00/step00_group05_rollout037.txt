import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s * 0.95  # 5% extra slack for termination

    h_init = initial_h_values
    n_points = len(h_init)

    h = cp.Variable(n_points, nonneg=True)
    ones = np.ones(n_points)
    overlap = []

    for k in range(-n_points + 1, n_points):
        idxs = (np.arange(n_points) + k) % n_points
        if k < 0:
            idxs = (np.arange(n_points) + k + n_points) % n_points
        h_k = h[idxs]
        overlap.append(cp.sum(h * (1 - h_k) / n_points * 2.0))

    C5 = cp.Max(overlap)

    # Integral constraint: sum(h) * (2.0 / n_points) = 1
    integrality = 2.0 * cp.sum(h) == n_points

    # Bounds on function values
    bounds = [h <= 1, h >= 0]

    # Objective is to minimize the maximum overlap
    prob = cp.Problem(cp.Minimize(C5), [integrality] + bounds)

    # Disable verbose output
    prob.solve(solver=cp.GUROBI, verbose=False)

    h_values = h.value
    h_values = np.clip(h_values, 0, 1)

    c5_bound = evaluate_erdos_solution(h_values)
    return (h_values, c5_bound, n_points)
