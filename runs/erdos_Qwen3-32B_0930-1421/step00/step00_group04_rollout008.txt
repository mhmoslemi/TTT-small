import numpy as np
import scipy.signal
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Use initial_h_values if present or initialize
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        h *= (n_points / 2) / h.sum()
    else:
        n_points = 84
        h = np.ones(n_points) / n_points
        h *= (n_points / 2)

    # Parameters
    dx = 2.0 / n_points
    max_iter = 2000
    log_interval = 100
    alpha = 0.01
    best_h = h.copy()
    best_c5 = np.inf

    start_time = time.time()

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Normalize area to exactly 1
        h *= (n_points / 2) / h.sum()

        # Evaluate C_5 by correlating h and (1 - h)
        ones_h = np.ones(n_points) - h
        corr = np.correlate(h, ones_h, mode='full')

        # Set negative pills to zero and scale by dx
        corr = np.maximum(corr, 0) * dx

        # Only evaluate the meaningful indices of k in [0, n_points]
        indices = np.arange(n_points)
        C5 = corr[indices].max()
        
        # Improve if better
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
            print(f"Iteration {i}, C₅={best_c5:.8f}")

        # Objective function gradient in frequency domain
        h_fft = np.fft.fft(h)
        ones_h_fft = np.fft.fft(ones_h)
        overlap_fft = h_fft * np.conj(ones_h_fft)
        overlap = np.fft.ifft(overlap_fft).real * dx

        # Create derivative ∇C₅
        dh = h - np.concatenate((h[-1:], h[:-1]))
        dh *= overlap / np.linalg.norm(overlap)

        # Gradient update
        h = h - alpha * dh

        # Project h onto box constraints and normalize area
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

    return best_h.tolist(), best_c5, n_points
