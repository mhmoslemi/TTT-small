            g_minus[i] = val_minus

        candidate_h = h - learn_rate * grad

        candidate_h = np.clip(candidate_h, 0, 1)

        current_norm = np.sum(candidate_h)

        new_c5 = evaluate_erdos_solution(candidate_h)
