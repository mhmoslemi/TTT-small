import time
import numpy as np
import cvxpy as cp
import scipy


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    hh = np.array(h)
    h1m = 1 - hh
    corr = np.correlate(hh, h1m, mode='full') * dx
    return np.max(corr)


def evaluate_feasibility(h, n_points, target_c5):
    dx = 2.0 / n_points
    hh = cp.Parameter(n_points, value=h)
    H = cp.Variable(n_points)

    # Enforce [0,1]
    const_bounds = [0 <= H, H <= 1]

    # Normalize to 1 unit area
    const_area = [cp.sum(H) == n_points / 2]

    # Discretized overlap constraint: y = H conv (1 - shift_k(H)) <= c5
    overlaps = []
    for k in range(1, n_points):  # lags from 1 to n_points - 1
        # Define (1 - shift_k(H)) when shift is valid
        shifted = cp.vstack([H[k:], cp.zeros(k)])
        overlap = cp.sum(H * (1 - shifted)) * dx
        overlaps.append(overlap <= target_c5)

    problem = cp.Problem(cp.Minimize(0), const_bounds + const_area + overlaps)
    try:
        problem.solve(solver='SCIPY', solver_args={"method": "SLSQP"}, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return H.value, True
    except Exception:
        pass
    return h, False
