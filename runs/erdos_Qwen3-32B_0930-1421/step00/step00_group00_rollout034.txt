import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h, dtype=np.float64)
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr * dx)
    return c5

def project_to_feasible_space(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    h = h / h_sum * (n_points / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def fitness(individual):
    c5 = evaluate_c5(individual)
    n_points = len(individual)
    sum_h = np.sum(individual)
    penalization = (
        10.0 * abs(sum_h - n_points / 2.0) +
        100 * max(np.max(individual) - 1.0, 0.0) +
        100 * max(np.min(individual), 0.0)
    )
    return -(c5 + penalization)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_parent)
    else:
        n_points = 45
        h_parent = np.clip(np.random.rand(n_points), 0.0, 1.0)
        h_parent = h_parent / np.sum(h_parent) * (n_points / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)

    pop_size = 50
    max_gen = 200
    mutation_strength = 0.1
    elitism_ratio = 0.2
    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h)
    best_so_far = (best_h, best_c5)

    # Initialize population
    population = [h_parent.copy() + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    for p in population:
        p[:] = project_to_feasible_space(p, n_points)

    # GA loop
    while time.time() - start_time < budget_s and max_gen > 0:
        max_gen -= 1
        pop_fitness = [(p, fitness(p)) for p in population]
        pop_fitness.sort(key=lambda x: x[1])

        # Elitism
        elites = [p for p, fit in pop_fitness[:int(elitism_ratio * pop_size)]]
        for h in elites:
            if evaluate_c5(h) < best_c5:
                best_h = h.copy()
                best_c5 = evaluate_c5(h)
                best_so_far = (best_h, best_c5)

        # Selection
        offspring = []
        while len(offspring) < pop_size:
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            alpha = np.random.rand(n_points)
            child = alpha * parent1 + (1 - alpha) * parent2

            # Mutation
            if np.random.rand() < 0.5:
                child += np.random.normal(0, mutation_strength, n_points)
                child = np.clip(child, 0.0, 1.0)

            child = project_to_feasible_space(child, n_points)
            offspring.append(child)

        population = elites + offspring[:pop_size - len(elites)]

    # Local optimization
    h_best, c5_best = best_so_far
    h_best = cp.Variable(n_points)
    constraint = [cp.sum(h_best) == n_points / 2.0, 0.0 <= h_best, h_best <= 1.0]
    objective = cp.Minimize(
        cp.Max(
            cp.Conv(h_best, cp.reshape(1 - h_best, (n_points,)), mode='full') * (2.0 / n_points)
        )
    )
    prob = cp.Problem(objective, constraint)
    try:
        prob.solve()
        if prob.status == "optimal":
            h_best_val = h_best.value
            final_c5 = evaluate_c5(h_best_val)
            best_c5 = final_c5
            best_h = h_best_val
    except Exception:
        pass

    final_c5 = evaluate_c5(best_h)
    return (best_h.ravel(), final_c5, n_points)
