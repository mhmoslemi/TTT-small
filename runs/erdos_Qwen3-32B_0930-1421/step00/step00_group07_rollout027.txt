import numpy as np
import time
from scipy.signal import correlate as sp_correlate
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()

    # Ensure normalization constraint
    total_area = np.sum(h) * dx
    scaling_factor = 1.0 / total_area
    h *= scaling_factor

    best_h = h.copy()
    best_c5_bound = np.inf
    patience_counter = 0
    MAX_ITER = 1000
    patience = 200
    alpha = 0.02

    def compute_c5(h):
        dh = h * dx
        corr = sp_correlate(dh, 1 - dh, mode='full')
        C = corr / dx  # reverse the scale applied in correlate
        return float(np.max(C))

    best_c5_bound = compute_c5(h)
    best_h = h.copy()

    for it in range(MAX_ITER):
        if time.time() - start_time > budget_s * 0.98:
            break

        dh = h * dx
        corr = sp_correlate(dh, 1 - dh, mode='full')
        C = corr / dx
        max_index = np.argmax(C)
        offset = max_index - (n_points - 1)

        grad = np.zeros_like(h)

        for i in range(n_points):
            i_minus = (i - offset) % n_points
            if i_minus != i:
                grad[i] -= (1 - h[i_minus])
            i_plus = (i + offset) % n_points
            if i_plus != i:
                grad[i] += (1 - h[i_plus])  # h[i_plus] is being subtracted in (1-h(i+k))

        # Normalize gradient
        grad -= np.mean(grad)
        # Update h
        h = h - alpha * grad

        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Rescale to maintain area constraint
        total_area = np.sum(h) * dx
        scaling_factor = 1.0 / total_area
        h *= scaling_factor

        # Update best solution
        c5 = compute_c5(h)
        if c5 < best_c5_bound:
            best_c5_bound = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                alpha *= 0.5
                patience_counter = 0

    c5_bound = evaluate_erdos_solution(best_h)
    return (best_h, c5_bound, n_points)
