import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    def project(h, target_sum):
        # Clip to [0,1]
        h_proj = np.clip(h, 0, 1)
        # Scale to meet constraint sum(h) = target_sum
        sum_val = np.sum(h_proj)
        if abs(sum_val) < 1e-10:
            h_proj[:] = 0.0
        else:
            h_proj *= target_sum / sum_val
        return h_proj

    # Interpolate h with more points for better correlation
    def interpolate_h(h, scale_factor):
        new_n_points = n_points * scale_factor
        x = np.linspace(0, 2, h.shape[0])
        x_new = np.linspace(0, 2, new_n_points)
        from scipy.interpolate import interp1d
        f = interp1d(x, h, kind='linear')
        h_new = f(x_new)
        return h_new

    max_interpolated = 4
    h_1d = h
    for scale in range(1, max_interpolated + 1):
        if scale > 1:
            h_1d = interpolate_h(best_h, scale)
            n_points = len(h_1d)
            dx = 2.0 / n_points
            target_sum = n_points / 2.0

        def compute_c5(h_flat):
            h = h_flat.reshape((n_points,))
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return c5

        # Gradient descent with line search and constraint projection
        h_flat = h_1d.flatten()
        learning_rate = 0.1
        max_grad_steps = 1000
        alpha = 1.0
        beta = 0.5

        for grad_step in range(max_grad_steps):
            if time.time() - start_time > budget_s * 0.98:
                break
            h = h_flat.reshape((n_points,))
            corr = correlate(h, 1 - h, mode='full') * dx
            k_star = np.argmax(corr)

            # Compute valid indices for the correlation
            offset = n_points - 1
            if 0 <= k_star - offset < n_points:
                i = k_star - offset
                grad = (1 - h)[i]
                grad_full = np.zeros(n_points)
                grad_full[i] = grad
            elif 0 <= k_star < n_points:
                i = k_star
                grad = (1 - h)[i]
                grad_full = np.zeros(n_points)
                grad_full[i] = grad
            else:
                print("Invalid k_star found")
                break

            # Perform line search
            current_c5 = compute_c5(h_flat)
            while True:
                new_h_flat = h_flat - learning_rate * grad_full
                new_h = new_h_flat.reshape((n_points,))
                new_h = project(new_h, target_sum)
                new_h_flat = new_h.flatten()
                new_c5 = compute_c5(new_h_flat)
                if new_c5 < current_c5:
                    break
                learning_rate *= beta

            h_flat = new_h_flat
            current_c5 = new_c5

            if current_c5 < best_c5:
                best_h = project(h_flat.reshape((n_points,)), target_sum)
                best_c5 = compute_c5(best_h.flatten())

            # Decrease learning rate
            learning_rate *= 1.0

        # If we haven't updated the best_h, use the projection of the last h_flat
        h = project(h_flat.reshape((n_points,)), target_sum)
        h_flat = h.flatten()

    best_c5_final = evaluate_erdos_solution(best_h)

    return best_h, best_c5_final, len(best_h)
