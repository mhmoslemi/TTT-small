import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=95):
    np.random.seed(seed)
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.mean()

    max_iter = 500
    epsilon = 1e-6
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    for t in range(max_iter + 1):
        # Enforce sum constraint
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if abs(h_sum - n_points / 2) > 1e-6:
            h *= (n_points / 2) / h_sum

        # Compute overlap integrals via correlate
        h1_minus_h = 1.0 - np.roll(h, shift=1, axis=0)
        corr = np.correlate(h, h1_minus_h, mode="full")
        corr = corr * dx
        current_c5 = np.max(corr)

        # Check for improvement and update best result
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Check time and break if exceeded
        if time.time() - start_time > budget_s:
            break

        # Build convex problem using surrogate log-sum objective
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            h_var.sum() == n_points / 2,
        ]
        overlaps = []
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(h, shift=k, axis=0)
            overlap = np.sum(h_var * (1 - shifted)) * dx
            overlaps.append(overlap)
        obj = cp.Minimize(cp.sum(cp.log(overlaps + [epsilon] * len(overlaps))))

        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.OSQP, verbose=False)

        if h_var.value is not None:
            h = h_var.value

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / final_h.mean() * (n_points / 2)
    final_h = np.clip(final_h, 0, 1)
    c5_bound = evaluate_erdos_solution(final_h)
    return (final_h, c5_bound, n_points)
