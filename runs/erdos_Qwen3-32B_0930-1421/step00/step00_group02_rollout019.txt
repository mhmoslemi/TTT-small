import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Best so far
    best_c5 = float('inf')
    best_h = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
        n_points = len(initial)
    else:
        n_points = 45
        initial = np.zeros(n_points)
        idx = int(n_points / 2)
        initial[idx] = 1.0  # Simple guess forcing mean constraint

    dx = 2.0 / n_points
    initial = initial / np.sum(initial) * n_points / 2  # Enforce volume constraint

    h = cp.Variable(n_points, nonneg=True, name="h")
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    def compute_c5_loss(h_vals):
        h_vals = h_vals.reshape(n_points)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr[1:n_points] * dx)  # Avoid the trivial k=0 overlap
        return c5

    best_h = initial.copy()
    best_c5 = compute_c5_loss(best_h)
    print(f"Initial C5: {best_c5:.6f}")

    warm_start = initial.copy()

    while time.time() - start_time < budget_s:
        dk = 1e-1
        # Approximate loss
        fake_loss = cp.Max(
            cp.multiply(dk, cp.remainder(np.arange(1, n_points), 1))[None, :] @ cp.reshape(h, [n_points, 1])
        )

        objective = cp.Minimize(fake_loss)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.OSQP, warm_start=True, max_iter=1000)
        if problem.status != 'optimal':
            print("Fallback...")
            warm_start = best_h.copy()
            continue

        trial_h = h.value
        trial_c5 = compute_c5_loss(trial_h)
        print(f"Intermediate C5: {trial_c5:.6f}")

        if trial_c5 < best_c5:
            best_h = trial_h
            best_c5 = trial_c5
            warm_start = best_h.copy()

        # Progress check
        elapsed = time.time() - start_time
        print(f"Elapsed: {elapsed:.1f} s")
        if elapsed > budget_s - 30:
            break

    return (best_h, best_c5, n_points)
