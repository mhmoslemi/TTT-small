import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, Bounds
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use provided parent construction or form new one
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except:
        n_points = 47
        initial_h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=int(n_points/2), replace=False)
        initial_h[idx] = 1.0

    def smooth_max(objs, eta=1.0):
        return np.log(np.sum(np.exp(eta * (objs - np.max(objs))))) / eta + np.max(objs)

    def evaluate_C5(h_, n=n_points):
        h = np.clip(h_, 0, 1).copy()
        h_sum = np.sum(h)
        scale_factor = (n / 2.0) / h_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)

        dx = 2.0 / n
        h = np.concatenate((h, np.zeros(n - 1)))
        overlaps = np.zeros(n * 2 - 1)
        for k in range(n * 2 - 1):
            valid = np.arange(n)
            valid = valid[(valid + k) < n * 2]
            overlap = np.sum(h[valid] * (1 - h[valid + k]))
            overlaps[k] = overlap * dx

        return np.max(overlaps)

    def evaluate_C5_smooth(h_, n=n_points, eta=50.0):
        h = np.clip(h_, 0, 1).copy()
        h_sum = np.sum(h)
        scale_factor = (n / 2.0) / h_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)

        dx = 2.0 / n
        h = np.concatenate((h, np.zeros(n - 1)))
        overlaps = np.zeros(n * 2 - 1)
        for k in range(n * 2 - 1):
            valid = np.arange(n)
            valid = valid[(valid + k) < n * 2]
            overlap = np.sum(h[valid] * (1 - h[valid + k]))
            overlaps[k] = overlap * dx

        return smooth_max(overlaps, eta)

    def perturb_h(h):
        n = len(h)
        new_h = np.clip(h + 0.1 * np.random.randn(n), 0.0, 1.0)
        new_h = new_h / np.sum(new_h) * (n / 2.0)
        new_h = np.clip(new_h, 0.0, 1.0)
        return new_h

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_C5(best_h)

    start_points = [initial_h]
    for _ in range(4):
        start_points.append(perturb_h(start_points[-1]))

    for starting_point in start_points:
        if time.time() - start_time > budget_s - 60:
            break

        h_opt = starting_point.copy()
        result = minimize(
            evaluate_C5_smooth,
            x0=h_opt,
            args=(n_points, 50),
            method='SLSQP',
            bounds=Bounds(0, 1),
            options={'maxiter': 500, 'ftol': 1e-6}
        )
        optimized_h = result.x

        # Normalize h to satisfy the constraint
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
        optimized_h = optimized_h / np.sum(optimized_h) * (n_points / 2.0)
        optimized_h = np.clip(optimized_h, 0.0, 1.0)

        # Annealing
        current_c5 = evaluate_C5(optimized_h)
        current_h_ = optimized_h.copy()
        temp = 1.0
        for sa_step in range(100):
            if time.time() - start_time > budget_s - 30:
                break
            proposal = perturb_h(current_h_)
            proposal_c5 = evaluate_C5(proposal)
            delta = proposal_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h_ = proposal.copy()
                current_c5 = proposal_c5
            temp *= 0.95

        if evaluate_C5(current_h_) < best_c5:
            best_h = current_h_.copy()
            best_c5 = evaluate_C5(best_h)

    best_h = best_h / np.sum(best_h) * (n_points / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = evaluate_C5(best_h)

    return (best_h.tolist(), final_c5, n_points)
