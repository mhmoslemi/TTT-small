import numpy as np
import time
import scipy.signal
import cvxpy as cp
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # Initialization
    rng = default_rng(seed)
    n_points_start = 42 if 'initial_h_values' in globals() else 42
    n_points = n_points_start
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / h.sum()

    def compute_c5(h):
        dx = 2.0 / len(h)
        ones_minus_h = 1 - h
        overlaps = scipy.signal.correlate(h, ones_minus_h, mode='full')
        overlaps *= dx
        return overlaps.max()

    def normalize_h(h):
        total_mass = h.sum()
        scaling_factor = (len(h) / 2) / total_mass
        h *= scaling_factor
        return np.clip(h, 0, 1)

    def gradient_step(h, lr):
        dx = 2.0 / len(h)
        ones_h = 1 - h
        overlaps = scipy.signal.correlate(h, ones_h, mode='full')
        peak_idx = overlaps.argmax()
        overlap = overlaps[peak_idx]
        k_shift = peak_idx - len(h) + 1
        # Construct shifted h
        shift = int(k_shift)
        shifted_h = np.roll(h, -shift) + 1e-8  # avoid zeros
        # Gradient estimate: derivative of h with -h shifted
        dh = (ones_h - shifted_h) * dx
        new_h = h - lr * dh
        return normalize_h(new_h)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n_points

    # Warmup iterations
    for i in range(40):
        if time.time() - start_time > budget_s - 5:
            break
        h = gradient_step(h, 0.01)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n_points

    # Neighborhood exploration with simplex-like restart
    def sample_neighbors(h_vec, n_points, count=10):
        candidates = []
        for _ in range(count):
            perturbation = rng.normal(0, 0.1, n_points)
            new_h = np.clip(h_vec + perturbation, 0, 1)
            new_h = normalize_h(new_h)
            candidates.append((new_h, compute_c5(new_h)))
        return candidates

    # Iterate with restart
    restart_interval = 20
    current_h = best_h
    current_c5 = best_c5
    current_n = best_n
    lr = 0.01

    while time.time() - start_time < budget_s - 10:
        for i in range(restart_interval):
            if time.time() - start_time >= budget_s - 10:
                break
            current_h = gradient_step(current_h, lr)
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                best_n = current_n
        # Restart
        neighbors = sample_neighbors(best_h, best_n)
        best_neighbor = min(neighbors, key=lambda x: x[1])
        if best_neighbor[1] < best_c5:
            current_h = best_neighbor[0]
            current_c5 = best_neighbor[1]
            best_h = current_h.copy()
            best_c5 = current_c5
            lr *= 0.8

    return best_h, best_c5, best_n
