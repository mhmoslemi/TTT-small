import numpy as np
import time
from math import floor
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    def compute_C5(h, n_points):
        dx = 2.0 / n_points
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    def tournament_selection(population, fitness, tournament_size=5):
        indices = np.random.choice(len(population), tournament_size)
        return population[indices[fitness[indices].argmin()]]

    def mutate(h, mutation_rate=0.08):
        n_points = len(h)
        h = h.copy()
        alt = np.random.rand(n_points) < mutation_rate
        h[alt] = 1 - h[alt]
        # Fix the sum constraint
        total = np.sum(h)
        delta = total - n_points / 2
        if delta > 0:
            h[alt] = np.clip(h[alt] - delta / np.sum(alt), 0, 1)
        elif delta < 0:
            h[~alt] = np.clip(h[~alt] + delta / np.sum(~alt), 0, 1)
        return h

    def crossover(parent1, parent2):
        child = np.where(np.random.rand(len(parent1)) < 0.5, parent1, parent2)
        # Fix the sum constraint
        child = np.clip(child, 0, 1)
        total = np.sum(child)
        n_points = len(child)
        delta = total - n_points / 2
        if delta > 0:
            adjust = np.zeros_like(child)
            adjust[child == 1.0] = -delta / len(child[child == 1.0])
            child += adjust
        elif delta < 0:
            adjust = np.zeros_like(child)
            adjust[child == 0.0] = -delta / len(child[child == 0.0])
            child += adjust
        return np.clip(child, 0, 1)

    # Parameters
    n_points = 42  # Use the seed with n=42
    h_init = initial_h_values.copy()
    pop_size = 40
    generations = 800
    mutation_rate = 0.08
    start_time = time.time()
    elapsed_time = 0.0

    # Initialize population with initial_h_values and random mutations
    population = [mutate(h_init, mutation_rate=0.2) for _ in range(pop_size)]
    fitness = np.array([compute_C5(h, n_points) for h in population])
    best_index = fitness.argmin()
    best_h = population[best_index].copy()
    best_c5 = fitness[best_index]

    gen = 0
    while elapsed_time < budget_s and gen < generations:
        new_population = []

        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            child1 = mutate(crossover(parent1, parent2), mutation_rate)
            child2 = mutate(crossover(parent1, parent2), mutation_rate)
            new_population.extend([child1, child2])

        population = new_population
        fitness = np.array([compute_C5(h, n_points) for h in population])

        new_best_index = fitness.argmin()
        if fitness[new_best_index] < best_c5:
            best_h = population[new_best_index].copy()
            best_c5 = fitness[new_best_index]

        elapsed_time = time.time() - start_time
        gen += 1

    return best_h, best_c5, n_points
