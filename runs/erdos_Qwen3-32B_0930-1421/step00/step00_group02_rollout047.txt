import numpy as np
import cvxpy as cp
import time
from math import isclose

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 45  # can tune later for speed and accuracy
        dx = 2.0 / n_points
        h = np.full(n_points, 0.5)
        h /= h.sum() * dx  # normalize to satisfy constraint

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    max_iter = 10000
    iters_since_improve = 0
    alpha = 0.1

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        correlations = np.correlate(h, 1 - h, mode="full")
        overlap_values = correlations * dx
        max_idx = np.argmax(overlap_values)
        max_overlap = overlap_values[max_idx]
        k_star = max_idx - n_points + 1

        # Compute gradient for shift k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k_star < n_points:
                grad[i] += (1 - h[i + k_star])
            grad[i] -= h[i]  # derivative of h[i]*(1 - h[j]) with respect to h[i] is (1 - h[j]) and -h[i]

        grad *= dx  # apply dx scaling

        # Optimization with constraints
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_var - (h - alpha * grad)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, max_iters=200)

        if prob.status not in ["optimal", "optimal_inaccurate"]:
            iters_since_improve += 1
        else:
            h_new = h_var.value
            h_new = np.clip(h_new, 0, 1)  # finalize clipping
            h_new = h_new / h_new.sum() * n_points / 2  # normalize sum
            c5_new = evaluate_erdos_solution(h_new)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                iters_since_improve = 0
            else:
                iters_since_improve += 1

            h = h_new  # always update for gradient direction

        if iters_since_improve > 30:
            alpha /= 2  # shrink step size
            iters_since_improve = 0

    best_c5 = evaluate_erdos_solution(best_h)
    result_h = np.clip(best_h, 0, 1)
    result_h = result_h / result_h.sum() * n_points / 2  # final enforcement of sum constraint

    return result_h, best_c5, n_points
