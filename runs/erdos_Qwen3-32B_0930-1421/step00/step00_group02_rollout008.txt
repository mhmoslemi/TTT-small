import cvxpy as cp

    dx = 2.0 / n_points

    # Precompute h(1 - h)
    def compute_overlap_integral(h, dx):
        h2 = 1 - h
        corr = np.correlate(h, h2, mode="full")
        corr *= dx
        return np.max(corr)

    # Track best h found
    h_best = h_init.copy()
    c5_best = compute_overlap_integral(h_best, dx)
    start_time = time.time()

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        area = h.sum() * dx
        scale = (1.0 / area)
        h *= scale
        h = np.clip(h, 0, 1)  # Final clip after scaling
        return h

    # Phase 1: Frequency-domain convex optimization
    phase1_time = budget_s * 0.6
    end_phase1 = start_time + phase1_time
    phase_iterations = 0

    while time.time() < end_phase1:
        phase_iterations += 1

        # Enforce constraints
        h_current = clip_and_rescale(h_best)

        # Fourier Analysis
        H = np.fft.fft(h_current)

        # Use real convolution tricks to cast (h*(1-h)) DC term as objective
        # h(1-h) has Fourier transform = (H**2)*ifft([1,0,...])
        # Goal is to minimize max Fourier magnitude at non-DC
        freqs = np.fft.fftfreq(n_points, d=dx)
        mask = (freqs != 0)
        H_non_dc = H[mask].copy()
        coeff_var = cp.Variable(H.size, complex=True)
        objectives = [cp.Maximize(cp.min(cp.abs(coeff_var)))]
        constraints = [
            cp.abs(coeff_var[0]) == 1.,  # Normalize DC term as scale reference
            coeff_var[mask] == H_non_dc,  # Keep non-DC fixed
            coeff_var[~mask] == H[~mask],  # Keep DC term free
            coeff_var.imag >= -1e-8,  # Avoid complex with negative imag part
        ]
        prob = cp.Problem(*objectives + constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            coeff = coeff_var.value
        except Exception:
            continue

        # Inverse FFT with optimized frequencies
        h_new = np.fft.ifft(coeff).real
        h_new = (h_new - h_new.min()) / (h_new.max() - h_new.min() + 1e-10)  # Normalize
        h_new = clip_and_rescale(h_new)

        # Evaluate and update
        c5_new = compute_overlap_integral(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 2: Time domain gradient descent
    phase2_time = budget_s * 0.4
    max_grad_steps = 500
    step_size = 0.1
    grad_beta = 0.9  # Exponentially decaying step
    step_decay = 1.0 - 0.001  # Slow but steady decay
    end_time = start_time + budget_s

    for step in range(max_grad_steps):
        if time.time() > end_time:
            break

        h_current = clip_and_rescale(h_current)
        h_current = h_current.reshape(-1, 1)

        # Compute gradient of C5 w.r.t h
        # Use finite differences with directional smoothing
        grad = np.zeros_like(h_current)
        eps = 1e-7

        for i in range(n_points):
            # Up perturbation
            h_up = h_current.copy()
            h_up[i] = np.clip(h_up[i] + eps, 0, 1)
            h_up = clip_and_rescale(h_up)
            c5_up = compute_overlap_integral(h_up, dx)

            # Down perturbation
            h_down = h_current.copy()
            h_down[i] = np.clip(h_down[i] - eps, 0, 1)
            h_down = clip_and_rescale(h_down)
            c5_down = compute_overlap_integral(h_down, dx)

            grad[i] = (c5_up - c5_down) / (2 * eps)

        # Gradient direction = reverse direction to minimize max overlap
        update = -grad.T / (1e-8 + np.linalg.norm(grad))
        update = update * (step_size / (step + 1))  # Decay as time/step increases
        h_new = h_current + update
        h_new = h_new.reshape(-1)

        # Clip and feasible rescale
        h_new = clip_and_rescale(h_new)
        c5_new = compute_overlap_integral(h_new, dx)

        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

        # Decay step and update timing
        step_size *= step_decay

    return (h_best, c5_best, n_points)
