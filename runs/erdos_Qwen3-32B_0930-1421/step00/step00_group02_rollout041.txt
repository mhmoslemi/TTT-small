import numpy as np
import cvxpy as cp
import time
from math import isfinite


def run(seed=42, budget_s=1100, **kwargs):
    # --- Initialize ---
    np.random.seed(seed)
    h = np.array(initial_h_values)  # Use the pre-imported seed construction
    n_points = len(h)
    dx = 2.0 / n_points

    # --- Objective and constraints ---
    def compute_c5(h):
        # Fast evaluation using convolution
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return min(c5, 0.4)  # Clipping for robustness

    def rescale_h(h):
        s = np.sum(h)
        if np.isclose(s, 0):
            return np.zeros_like(h)
        h = ((0.5 * n_points) / s) * h
        h = np.clip(h, 0.0, 1.0)
        return h

    # --- Optimization settings ---
    total_time = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    t0 = time.time()

    h = cp.Parameter(n_points, nonneg=True)
    objective = cp.Minimize(0)  # Dummy objective; to be replaced
    constraints = [h >= 0, h <= 1]
    prob = cp.Problem(objective, constraints)

    # --- Phase 1: Frequency domain optimization (Convex relaxation) ---
    t_phase1 = time.time()
    while time.time() - t0 < 0.6 * budget_s:
        # Fourier transform surrogate
        H = np.fft.fft(h.value)
        idxs = np.fft.fftfreq(n_points, d=dx)

        # Use convex optimization to reduce max |F{h(1 - h)}|
        # Construct the surrogate: approx h(1-h) as convex around current h
        weight = h.value * (1 - h.value)
        weight = weight / (1e-3 + abs(weight)).clip(1e-3, 1.0)

        surrogate_h = cp.Variable(n_points)
        prob = cp.Problem(
            cp.Minimize(cp.norm(np.fft.fft(surrogate_h), np.inf)),
            [surrogate_h >= 0, surrogate_h <= 1, cp.sum(surrogate_h) == n_points / 2]
        )

        try:
            prob.solve(solver=cp.ECOS, max_iters=200)
            h.value = surrogate_h.value
            if h.value is None:
                break
            h.value = np.clip(h.value, 0, 1)
            h.value = rescale_h(h.value)
            this_c5 = compute_c5(h.value)
            if this_c5 < best_c5 and isfinite(this_c5):
                best_c5 = this_c5
                best_h = h.value.copy()
        except:
            pass

        total_time = time.time() - t0
        if total_time + (time.time() - t_phase1) * 2 > budget_s:
            break

    # --- Phase 2: Refinement in time domain (Gradient descent) ---
    lr = 0.1
    decay = 0.999

    while time.time() - t0 < budget_s:
        h.grad = np.correlate(
            (1 - 2 * h.value) * (1 - h.value),
            1 - h.value,
            mode='same'
        ) * dx
        h.value = (h.value - lr * h.grad).clip(0, 1)
        h.value = rescale_h(h.value)
        this_c5 = compute_c5(h.value)
        if this_c5 < best_c5 and isfinite(this_c5):
            best_c5 = this_c5
            best_h = h.value.copy()
        lr *= decay
        if lr < 1e-8:
            break

    # --- Final validation and return ---
    best_h = rescale_h(best_h)
    c5 = compute_c5(best_h)
    print(f"[TIME {time.time()-t0:.2f}s] Best C₅ = {c5:.8f}")

    return (best_h.tolist(), c5, n_points)
