import time
import numpy as np
import scipy.signal
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    # Load or initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 80
        h = np.full(n_points, 0.5, dtype=np.float64)

    n_points = len(h)
    assert np.allclose(np.sum(h), n_points / 2)

    # Normalize if needed
    h *= n_points / 2 / np.sum(h)

    # Constants
    dx = 2.0 / n_points

    # Evaluator shortcut
    def evaluate_c5(h_iter):
        # Compute cross-correlation for maximization
        corr = scipy.signal.correlate(h_iter, 1 - h_iter, mode="full")
        C5 = np.max(corr) * dx
        return C5

    start_time = time.time()
    patience = 100
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    alpha = 0.02
    improvement_count = 0
    max_iter = 2000

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Quick evaluate for current h
        corr_k = scipy.signal.correlate(h, 1 - h, mode="full")
        C5_k = corr_k * dx
        max_idx = np.argmax(C5_k)

        # Compute gradient at worst shift k
        k = max_idx - np.argmax(h)  # lag shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx_forward = (i - k) % n_points  # wrap around
            idx_backward = (i + k) % n_points
            grad[i] += (1 - h[idx_forward]) - (h[idx_backward])

        h_grad = h - alpha * grad
        h_grad = np.clip(h_grad, 0, 1)
        h_grad = h_grad / np.sum(h_grad) * (n_points / 2)

        c5_new = evaluate_c5(h_grad)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_grad.copy()
            improvement_count = 0
            alpha *= 1.02
        else:
            alpha *= 0.92
            improvement_count += 1

        if improvement_count > patience:
            alpha *= 0.5
            improvement_count = 0

        h = h_grad.copy()

    # Verify constraints
    assert np.all(0 <= best_h) and np.all(best_h <= 1)
    assert np.allclose(np.sum(best_h), n_points * 0.5)

    return best_h, best_c5, n_points
