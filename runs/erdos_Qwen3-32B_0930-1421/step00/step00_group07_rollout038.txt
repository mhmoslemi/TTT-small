import numpy as np
import cvxpy as cp
import time
from typing import List, Tuple

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def h_to_intervals(h):
        in_interval = False
        intervals = []
        for i in range(n_points):
            if h[i] == 1:
                if not in_interval:
                    start = i
                    in_interval = True
            else:
                if in_interval:
                    end = i
                    intervals.append((start * dx, end * dx))
                    in_interval = False
        if in_interval:
            intervals.append((start * dx, 2.0))
        return intervals

    def intervals_to_h(intervals, n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for a, b in intervals:
            indices = np.where((x >= a) & (x < b))[0]
            h[indices] = 1
        return h

    def compute_c5(h):
        h1 = h
        h2 = 1 - h
        corr = np.correlate(h1, h2, mode='full') * dx
        c5 = np.max(corr)
        return c5

    intervals = h_to_intervals(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    T = 1.0
    T_decay = 0.99
    max_iter = 500
    step = 0.005
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > max_time:
            break

        if len(intervals) > 1:
            i = np.random.randint(len(intervals))
            a, b = intervals[i]
            delta = np.random.uniform(-step, step, 2)
            new_a = max(0.0, a + delta[0])
            new_b = min(2.0, b + delta[1])
            if new_a >= new_b:
                continue
            if i > 0:
                new_a = max(new_a, intervals[i - 1][1])
            if i < len(intervals) - 1:
                new_b = min(new_b, intervals[i + 1][0])

            new_intervals = intervals[:i] + [(new_a, new_b)] + intervals[i + 1:]
        else:
            a, b = intervals[0]
            offset = np.random.uniform(-step, step)
            center = (a + b) / 2
            new_a = max(0.0, center - (b - a) / 2 + offset)
            new_b = new_a + (b - a)
            new_intervals = [(new_a, new_b)]

        total_length = np.sum([b - a for a, b in new_intervals])
        if abs(total_length - 1.0) > 1e-6:
            scale = 1.0 / total_length
            new_intervals = [(a * scale, b * scale) for a, b in new_intervals]

        new_h = intervals_to_h(new_intervals, n_points)
        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            T *= 0.995
        else:
            if np.random.random() < np.exp(-max(0.001, new_c5 - best_c5) / T):
                best_h = new_h.copy()
                best_c5 = new_c5

        T *= T_decay

    return (best_h, best_c5, n_points)
