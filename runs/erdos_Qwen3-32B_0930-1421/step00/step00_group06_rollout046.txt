import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = cp.Variable(n_points, nonneg=True)
    one_minus_h = 1 - h
    h_array = initial_h_values.copy()

    best_c5 = float("inf")
    best_h = h_array.copy()
    alpha = 0.5
    max_iterations = 1000
    k_max = None

    # Used for tracking progress
    prev_c5 = None
    patience = 0
    k_history = []

    for it in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        h_shifted = alpha * best_h + (1 - alpha) * h_array

        # Use initial_h_values as initial h_shifted if nothing better found
        if it == 0 and best_h is not None:
            h_shifted = best_h

        corr = np.correlate(h_shifted, 1 - h_array, mode='full') * dx
        k_max = np.argmax(corr)
        k_history.append(k_max)

        if prev_c5 is not None and abs(corr[k_max] - prev_c5) < 1e-6:
            patience += 1
            if patience >= 10:
                alpha = max(alpha - 0.05, 0.1)
        else:
            patience = 0

        # Fix h_shifted and solve
        offset = n_points - k_max
        selected_h_shifted = h_shifted[(n_points - k_max):] if k_max < n_points else h_shifted[:k_max]
        constraint = h * one_minus_h @ selected_h_shifted.T

        # Re-formulate minimization as convex competition for k_max
        objective = cp.Minimize(np.dot(h_array, (1 - h_shifted[(n_points - k_max):])) * dx)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        prob = cp.Problem(objective, constraints)

        try:
            result = prob.solve(solver='ECOS', verbose=False)
            new_h = h.value
            new_h = np.clip(new_h, 0, 1)
            h_array = new_h

            # Compute norm change
            norm_change = np.linalg.norm(new_h - best_h)

            # Compute new overlap
            corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
            current_c5 = np.max(corr)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
                alpha = max(alpha - 0.001, 0.1)

        except Exception as e:
            print(f"Iteration {it} failed with {e}. Skipped.")
            continue

        prev_c5 = current_c5

    # Compute final solution
    corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
    c5_bound = np.max(corr)

    # Project h to satisfy constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.mean(best_h) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)

    return (best_h, c5_bound, n_points)
