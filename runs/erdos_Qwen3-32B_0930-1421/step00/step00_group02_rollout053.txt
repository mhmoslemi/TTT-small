import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        assert sum(h) == n_points / 2, "Initial h violates the integral constraint"
    else:
        # Use a shallow sawtooth pattern
        n_points = 45
        h = np.linspace(0.4, 0.6, n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

    dx = 2.0 / n_points

    # Evaluate initial
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(corr)

    c5_best = compute_c5(h)
    h_best = h.copy()

    # Time budget setup
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute gradients targeting dominant lag
        corr_vals = correlate(h, 1 - h, mode='full') * dx
        best_lag = np.argmax(corr_vals)

        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + best_lag < len(h):
                grad[i] += 1 - h[i + best_lag]
            if i - best_lag >= 0:
                grad[i] += -(1 - h[i - best_lag])  # For negative lags

        # Normalize and project
        grad = grad / max(np.abs(grad).max(), 1e-10)
        step = 0.01
        h_proposed = h + step * grad
        h_proposed = np.clip(h_proposed, 0.0, 1.0)
        h_proposed *= (n_points / 2) / h_proposed.sum()

        # Accept if better
        c5_proposed = compute_c5(h_proposed)
        if c5_proposed < c5_best:
            h_best = h_proposed.copy()
            c5_best = c5_proposed

        h = h_proposed.copy()  # Keep moving

    return h_best.tolist(), c5_best, n_points
