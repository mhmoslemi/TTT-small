import numpy as np
import math
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    rho_admm = 1.0
    max_iter = 1000
    tol_abs = 1e-4
    tol_rel = 1e-4
    start_time = time.time()

    # Initial guesses
    z_k = [0.0 for k in range(n_points)]  # Represents max overlap
    u_k = [0.0 for k in range(n_points)]  # ADMM multipliers for constraints

    prev_h = h.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    converged = False

    def compute_overlap(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode="full")
        conv = conv * dx
        max_c5 = np.max(conv)
        return max_c5

    def feasible_project(h_vec):
        # Project into [0, 1] and normalize to sum to 1
        h_vec = np.clip(h_vec, 0.0, 1.0)
        sum_h = np.sum(h_vec)
        if sum_h > 1e-10:
            h_vec *= (n_points / 2.0) / sum_h
        else:
            h_vec = np.zeros_like(h_vec)
        return h_vec

    target_c5 = 0.38085
    max_time = budget_s

    while time.time() - start_time < max_time and not converged:
        # Compute overlap for all k
        overlap_vals = []
        for k in range(n_points):
            h_shifted = np.roll(h, -k)
            val = np.dot(h, 1 - h_shifted) * dx
            overlap_vals.append(val)

        # Update auxiliary z_k values
        for k in range(n_points):
            z_k[k] = max(overlap_vals[k], z_k[k] - u_k[k] / rho_admm)
            u_k[k] += rho_admm * (overlap_vals[k] - z_k[k])

        # Form linear estimates of the max
        obj = 0.0
        for k in range(n_points):
            obj += cp.pos(cp.Variable()) >= z_k[k]
        max_z = cp.Variable()
        constraints = [cp.Variable() == z_k[k] for k in range(n_points)]
        constraints += [max_z >= z for z in z_k]
        z_optimizer = cp.Problem(cp.Minimize(max_z), [constraints[i] for i in range(len(constraints))])
        z_optimizer.solve(solver=cp.CVXOPT, verbose=False)

        # Now infinitely constrain h to improve C5
        max_c5 = compute_overlap(h)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()

        # Gradient step with line search
        grad_h = np.zeros_like(h)
        for k in range(n_points):
            if overlap_vals[k] == z_k[k]:
                h_shifted = np.roll(h, -k)
                grad_h += 1 - 2 * h_shifted
        grad_h *= dx

        t = 1.0 / np.sqrt(n_points)
        gradient_step = -grad_h * t
        h_propose = h + gradient_step
        h_propose = feasible_project(h_propose)

        h_update_candidate = h_propose.copy()
        overlap_propose = compute_overlap(h_update_candidate)
        if overlap_propose < best_c5:
            h = h_update_candidate.copy()

        h = feasible_project(h)

        # Check convergence
        norm_r = np.sqrt(np.sum((overlap_vals - [z_k[k] - u_k[k]/rho_admm for k in range(n_points)])**2))
        norm_s = np.sqrt(np.sum((np.array([rho_admm * (overlap_vals[k] - z_k[k]) for k in range(n_points)])**2)))
        eps_primal = np.sqrt(n_points) * tol_abs + tol_rel * max(
            np.sqrt(np.sum(np.array(overlap_vals)**2)),
            np.sqrt(np.sum(np.array(z_k)**2)))
        eps_dual = np.sqrt(n_points) * tol_abs + tol_rel * np.sqrt(np.sum(np.array(u_k)**2))

        if norm_r <= eps_primal and norm_s <= eps_dual:
            converged = True

    # Validate the solution
    h = feasible_project(h)
    C_5 = compute_overlap(h)

    return (h.tolist(), C_5, n_points)
