import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    sa_iter = 0
    sa_max_iter = 1000
    gd_iter = 0
    gd_max_iter = 500
    T = 1.0
    T_alpha = 0.99
    sigma = 0.1
    eta = 0.01
    no_improvement = 0
    max_no_improvement = 100
    start_time = time.time()

    def project_to_box(h):
        return np.clip(h, 0.0, 1.0)

    def project_to_sum(h, target_sum):
        h = h.copy()
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) < 1e-6:
            return h
        diff = sum_h - target_sum
        sort_indices = np.argsort(h)
        if diff > 0:
            # need to decrease largest values
            top = sort_indices[::-1]
            to_adjust = top[:10]
            adjustment = -diff / len(to_adjust)
            h[to_adjust] += adjustment
        else:
            # need to increase smallest values
            bottom = sort_indices[:10]
            adjustment = -diff / len(bottom)
            h[bottom] += adjustment
        return h

    def evaluate(h):
        h_unclipped = h.copy()
        h_clipped = project_to_box(h_unclipped)
        h_sumayed = project_to_sum(h_clipped, n_points / 2)
        return evaluate_erdos_solution(h_sumayed), h_sumayed

    def sa_step(h, T, sigma):
        h_proposal = h + sigma * np.random.randn(n_points)
        h_proposal = project_to_box(h_proposal)
        h_proposal = project_to_sum(h_proposal, n_points / 2)
        return h_proposal

    def gd_step(h, eta):
        h_clipped = project_to_box(h.copy())
        corr = np.correlate(h_clipped, 1.0 - h_clipped, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr * dx)

        corr = np.maximum(np.zeros_like(corr), corr).astype(np.float)
        gradient = np.zeros(n_points)
        indices = []

        for i in range(-n_points + 1, n_points):
            idx1 = (i + n_points - 1) % n_points
            idx2 = (i) % n_points
            if np.isclose(corr[i + n_points - 1], c5, atol=1e-5):
                indices.append(i)

        indices = np.array(indices).clip(0, n_points - 1)
        for k in indices:
            h_part1 = h_clipped[k % n_points]
            h_part2 = h_clipped[(k + n_points - 1) % n_points]
            idx = k
            gradient[idx % n_points] += (1 - h_part2)
            idx = k + n_points - 1
            gradient[idx % n_points] -= h_part1
        gradient = project_to_box(h_clipped - eta * gradient)
        gradient = project_to_sum(gradient, n_points / 2)
        return gradient

    while time.time() - start_time < budget_s:
        phase = 'SA' if (sa_iter < sa_max_iter) or (rand := np.random.random()) < 0.5 else 'GD'

        if phase == 'SA':
            h_prop = sa_step(best_h, T, sigma)
            score, h_prop = evaluate(h_prop)
            if score < best_c5:
                best_c5 = score
                best_h[:] = h_prop.copy()
                no_improvement = 0
                T = max(0.1, T * T_alpha)
                sigma = max(0.01, sigma * 0.99)
            else:
                delta = best_c5 - score
                if np.random.random() < np.exp(delta / T):
                    best_c5 = score
                    best_h[:] = h_prop.copy()
                no_improvement += 1
                sigma = max(0.01, sigma * 1.05)
            sa_iter += 1
            if no_improvement >= max_no_improvement:
                T = max(0.5, T * 1.2)
                no_improvement = 0
        else:
            best_h = gd_step(best_h, eta)
            current_c5, _ = evaluate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                no_improvement = 0
                eta = max(0.001, eta * 0.95)
            else:
                no_improvement += 1
            gd_iter += 1
            if no_improvement >= max_no_improvement:
                T = max(0.5, T * 1.2)
                no_improvement = 0

    return best_h, best_c5, n_points
