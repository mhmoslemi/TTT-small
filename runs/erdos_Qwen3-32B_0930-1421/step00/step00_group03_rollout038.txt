import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_n_points = len(initial_h_values)
    n_points = initial_n_points
    h_flat = np.array(initial_h_values)
    assert 0 <= np.min(h_flat) and np.max(h_flat) <= 1
    assert abs(np.sum(h_flat) - n_points / 2) < 1e-6
    dx = 2.0 / n_points

    def project_to_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if abs(total - n_points / 2) < 1e-8:
            return h
        error = n_points / 2 - total
        num_elements = min(100, int(n_points / 2))
        indices = h.argsort(kind='stable')
        if error > 0:
            indices = indices[-num_elements:]  # top indices
            h[indices] += error / num_elements
        else:
            indices = indices[:num_elements]  # bottom indices
            h[indices] -= error / num_elements
        h = np.clip(h, 0.0, 1.0)
        return h

    def compute_c5(h):
        ones_minus_h = 1.0 - h
        full_corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(full_corr) * dx

    best_c5 = current_c5 = compute_c5(h_flat)
    best_h = h_flat.copy()
    start_time = time.time()

    T = 1.0
    alpha = 0.99
    step_size = 0.1
    max_sa_steps = 1000
    max_gd_steps = 500
    sa_step = 0
    grid_step = 0

    while time.time() - start_time < budget_s:
        # SA phase
        if sa_step < max_sa_steps:
            h = project_to_feasible(h_flat + step_size * np.random.randn(n_points))
            new_c5 = compute_c5(h)
            delta = current_c5 - new_c5
            if delta > 0 or np.exp(delta / T) > np.random.rand():
                h_flat[:] = h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            sa_step += 1
            T *= alpha
            if sa_step % 100 == 0:
                step_size *= 0.95

        # GD phase
        elif grid_step < max_gd_steps:
            fc = 1 - h_flat
            corr = np.correlate(h_flat, fc, mode="full")
            max_idx = np.argmax(corr)
            k = max_idx - n_points + 1
            # Compute step in h that reduces corr at position k
            grad = np.zeros_like(h_flat)
            if k >= 0:
                for i in range(n_points):
                    if i + k < n_points:
                        grad[i] = fc[i + k] - h_flat[i]
                    elif i + k >= n_points:
                        if i + k - n_points < n_points:
                            grad[i] = h_flat[i + k - n_points] - h_flat[i]
            else:
                for i in range(n_points):
                    if i - k < n_points:
                        grad[i] = -fc[i - k] - h_flat[i]

            h_flat += 0.005 * grad  # learning rate
            h_flat = project_to_feasible(h_flat)
            current_c5 = compute_c5(h_flat)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_flat.copy()
            grid_step += 1
        # Fallback to SA
        else:
            break

    return best_h.tolist(), best_c5, n_points
