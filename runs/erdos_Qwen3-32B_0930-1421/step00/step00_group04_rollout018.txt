import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use the fixed number of points from the initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 1000  # Fallback to 1000 samples if not provided
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        # Clip to [0,1]
        np.clip(initial_h, 0, 1, out=initial_h)

    dx = 2.0 / n_points
    h = np.copy(initial_h)

    start_time = time.time()

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h=h, n_points=n_points)

    max_iter = 10000
    learning_rate = 0.01
    patience = 100
    no_improvement = 0

    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        current_c5, overlaps = evaluate_erdos_solution(h=h, return_overlaps=True, n_points=n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                learning_rate *= 0.95

        # Compute gradient
        grad = np.zeros_like(h)
        kernel = np.ones((n_points + 1), dtype=np.float64)
        for k in range(1, n_points + 1):
            h_k = np.roll(h, -k)
            grad_k = (h - h_k)
            grad -= np.roll(grad_k, k)

        grad *= dx

        # Gradient step
        h_new = h - learning_rate * grad

        # Projection onto feasible region: [0,1], sum(h) = n_points / 2
        h_new = np.clip(h_new, 0, 1)
        s = np.sum(h_new)
        if s != 0:
            h_new *= (n_points / 2.0) / s
        h = h_new

    return best_h, best_c5, n_points
