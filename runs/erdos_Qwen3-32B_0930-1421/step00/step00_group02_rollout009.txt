import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if available, else create a flat initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 90  # Use 90 points for higher resolution
        h = np.zeros(n_points)
        h[n_points // 2] = n_points / 2  #集中设置保证积分约束
        h = h.clip(0, 1)

    dx = 2.0 / n_points

    # Ensure the sum normalization constraint
    h /= np.sum(h) * dx
    h *= n_points / 2

    best_c5 = float('inf')
    best_h = h.copy()

    def compute_c5(h_vec):
        h_vec = h_vec.clip(0, 1)
        h_vec_sum = np.sum(h_vec) * dx
        h_vec = (h_vec / h_vec_sum) * (n_points / 2)
        h_vec = h_vec.clip(0, 1)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(corr * dx)
        return c5, h_vec

    def refine_h(h_vec):
        h_vec = h_vec.clip(0, 1)
        h_vec_sum = np.sum(h_vec) * dx
        scale_factor = (n_points / 2) / h_vec_sum
        h_vec = h_vec * scale_factor
        h_vec = h_vec.clip(0, 1)
        return h_vec

    def opt_cycle(h_vec):
        h_vec = refine_h(h_vec)
        conv = 1 - h_vec
        corr = correlate(h_vec, conv, mode='same')  # approximate C5
        max_corr = np.max(corr)
        # Solve linear program to minimize weighted corr

        hp = cp.Variable(n_points)
        obj = cp.Minimize(np.max(cp.conv(cp.sum(hp), conv, mode='same')))
        constraints = [
            cp.sum(hp) * dx == 1,
            0 <= hp,
            hp <= 1
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='ECOS')
        return hp.value if prob.status == 'optimal' else h_vec

    while time.time() - start_time < budget_s:
        h = refine_h(h)
        c5, h = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = opt_cycle(h)

    # Final refinement
    best_h = refine_h(best_h)
    return best_h, best_c5, n_points
