import numpy as np
import scipy.optimize as opt
import time

# Pre-imported: evaluate_erdos_solution(), initial_h_values

def objective(h_vec, n_points):
    h = h_vec.reshape(n_points)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def linearized_constraints(n_points):
    # Constraint: sum(h) == n_points / 2
    A_eq = np.ones((1, n_points))
    b_eq = np.array([n_points / 2.0])
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    return A_eq, b_eq, bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start = time.time()
    n_points = len(initial_h_values)
    h0 = np.array(initial_h_values).copy()
    
    # Evaluate initial solution
    c5_initial = objective(h0, n_points)
    
    # Optimization settings
    max_iter = 10000  # Max COBYLA iterations
    max_outer = 2     # Max number of restarts

    best_h = h0.copy()
    best_c5 = c5_initial

    for _ in range(max_outer):
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        A_eq, b_eq, bounds = linearized_constraints(n_points)

        # Define optimization problem
        def f(h_vec):
            return objective(h_vec, n_points)

        result = opt.minimize(
            f,
            h0.flatten(),
            method='SLSQP',  # COBYLA not always available; SLSQP with constraints + bounds
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2.0)},
            bounds=bounds,
            options={'maxiter': max_iter // max_outer}
        )

        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
            h0 = best_h  # next outer-loop restart point
        else:
            break  # early if no improvement

        h0 = best_h / np.sum(best_h) * (n_points / 2.0)  # re-normalize if necessary

    # Final validation
    h_best = best_h.reshape(n_points)
    h_best = np.clip(h_best, 0, 1)  # enforce bounds
    h_best = h_best / np.sum(h_best) * (n_points / 2.0)  # enforce sum constraint

    c5_final = evaluate_erdos_solution(h_best)
    return (h_best.tolist(), c5_final, n_points)
