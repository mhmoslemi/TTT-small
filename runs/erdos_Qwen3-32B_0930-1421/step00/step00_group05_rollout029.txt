import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Reusable normalization
    def normalize_h(weights):
        return weights / np.sum(weights) * (n_points / 2)

    # Create triangular basis functions
    def create_triangular_basis(n, basis_count):
        x = np.linspace(0, 2, n_points, endpoint=False)
        basis = np.zeros((basis_count, n))
        for i in range(basis_count):
            center = 2.0 * i / (basis_count - 1)
            left = np.maximum(0, 1 - np.abs(x - center) * (basis_count - 1) / 2)
            basis[i] = left
        return basis

    basis_count = 100
    basis = create_triangular_basis(n_points, basis_count)

    # Compute initial weights from initial_h_values
    weights_initial = np.random.random(basis_count)
    for _ in range(20):
        h_rec = basis @ normalize_h(weights_initial)
        weights_initial = h_rec.T @ basis / (n_points / 2)
    weights_initial = np.maximum(0, np.minimum(1, weights_initial))
    weights_initial /= weights_initial.sum()
    weights_initial *= (n_points / 2)

    # MICP variables
    w = cp.Variable(basis_count, nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(w) == n_points / 2,
        w >= 0,
        t >= 0
    ]

    # For each k, compute correlation-like constraint
    for k in range(n_points):
        h = basis @ w
        corr_term = np.sum(h * (1 - np.roll(h, -k)) / n_points)
        corr_term = cp.MaxElement(corr_term)  # linear relaxation
        constraints.append(t >= corr_term)

    # Problem formulation
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver='GUROBI', max_iter=100, verbose=True, time_limit=budget_s * 0.98)

    h_optimized = basis @ w.value
    h_optimized /= np.sum(h_optimized) * n_points * dx
    h_optimized *= n_points / 2  # Enforce integral constraint

    h_optimized = np.clip(h_optimized, 0, 1)
    h_optimized /= np.sum(h_optimized) * n_points * dx
    h_optimized *= n_points / 2

    def evaluate_c5(h):
        h_ = h / n_points / dx
        c5 = np.max(correlate(h_, 1 - h_, mode='full') * 2 * dx)
        return c5

    final_c5 = evaluate_c5(h_optimized)

    return h_optimized, final_c5, n_points
