import numpy as np
import time
import cvxpy as cp
import evaluate_erdos_solution
from functools import partial

def run(seed=42, budget_s=1100):
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values).copy()
        n_points = len(h_init)
    else:
        n_points = 200
        h_init = np.zeros(n_points)
        h_init[n_points // 4 : 3 * n_points // 4] = 1 / (n_points // 2) * 2  # rough initialization
        h_init *= h_init.sum() / h_init.sum() * n_points / 2  # enforce sum(h) = n_points / 2

    dx = 2.0 / n_points
    h = h_init.copy()
    h_best = h.copy()
    c5_best = evaluate_erdos_solution(h)

    start_time = time.time()

    def optimize_h(h_in, strength=0.1, bias=0.0):
        h_var = cp.Variable(len(h_in), nonneg=True)
        constraints = [
            cp.sum(h_var) == len(h_in) / 2,
            h_var <= 1,
        ]
        h_plus = (1 - h_in)
        corr_vals = []
        for k in range(1, len(h_in) - 1):
            overlap = cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, shift=k)))
            corr_vals.append(overlap * dx)

        corr_var = cp.Variable(len(corr_vals))
        constraints += [corr_var >= v for v, val in zip(corr_vals, corr_vals)]
        obj = cp.Minimize(corr_var.max())
        prob = cp.Problem(obj, constraints)
        _ = prob.solve(solver=cp.ECOS)
        return h_var.value

    def repair_h(h_in):
        h_in = np.clip(h_in, 0.0, 1.0)
        s = h_in.sum()
        scale = (n_points / 2) / s if s > 0 else 1.0
        h_out = h_in * scale
        h_out = np.clip(h_out, 0.0, 1.0)
        h_out[h_out < 1e-6] = 0.0
        h_out[h_out > 1.0 - 1e-6] = 1.0
        return h_out

    def evaluate(h_in):
        return evaluate_erdos_solution(h_in)

    while time.time() - start_time < budget_s:
        h_opt = optimize_h(h)
        if h_opt is not None:
            h = repair_h(h_opt)
            c5 = evaluate(h)
            if c5 < c5_best:
                h_best = h.copy()
                c5_best = c5
        else:
            h = repair_h(h + np.random.normal(0, 0.01, len(h)))
            h = repair_h(h)
            c5 = evaluate(h)
            if c5 < c5_best:
                h_best = h.copy()
                c5_best = c5

        if abs(c5_best - evaluate(h_best)) < 1e-6:
            break

    h_best = repair_h(h_best)
    c5_best = evaluate(h_best)
    return h_best.tolist(), c5_best, n_points
