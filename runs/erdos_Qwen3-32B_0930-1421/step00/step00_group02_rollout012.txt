import numpy as np
import time
import cvxpy as cp


def evaluate_overlap(h, dx):
    h = np.asarray(h, dtype=np.float64)
    ones_minus_h = 1.0 - h
    full_conv = np.correlate(h, ones_minus_h, mode="full")
    c5_bound = np.max(full_conv * dx)
    return c5_bound


def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)
    t0 = time.time()
    dx = 2.0 / n_points
    target_volume = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_volume
    h = np.clip(h, 0.0, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_overlap(best_h, dx)

    while time.time() - t0 < budget_s:
        h = np.clip(h, 0.0, 1.0)
        h /= np.sum(h) * dx
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h *= target_volume / h_sum

        overlap_conv = np.correlate(h, 1.0 - h, mode='full')
        max_k = np.argmax(overlap_conv)
        k = max_k
        s_k = overlap_conv[k]

        # Build gradient for the max overlap S_k
        grad_h = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad_h[i] += (1.0 - h[i + k]) * dx
            if i - k >= 0:
                grad_h[i] -= (1.0 - h[i - k]) * h[i - k]
        grad_h *= dx

        # Solve constrained update
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(h_var - (h - 0.5 * grad_h)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_volume
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        h = h_var.value

        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    h_values = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(h_values)
    h_values *= target_volume / h_sum
    h_values = np.clip(h_values, 0.0, 1.0)
    return (h_values, best_c5, n_points)
