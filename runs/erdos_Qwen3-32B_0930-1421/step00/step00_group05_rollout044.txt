import numpy as np
import numpy.random as rnd
import math
import time
import cvxpy as cp
from copy import deepcopy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Pre-imported from harness
    import evaluate_erdos_solution
    global initial_h_values

    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_integral = n_points / 2.0

    def construct_overlap_matrix():
        # Construct off-diagonal overlap matrix
        h_array = cp.Parameter(n_points, value=initial_h_values)
        corr_values = cp.Parameter(shape=(n_points * 2 - 1, n_points), value=None)
        for k in range(n_points * 2 - 1):
            shift = n_points - 1 if k > n_points - 1 else 0
            slice_1 = h_array if k <= n_points - 1 else h_array[k - (n_points - 1):]
            slice_2 = h_array[k:] if k <= n_points - 1 else h_array[: k - n_points + 1]
            corr_values[k] = cp,sum(
                slice_1[i] * (1 - slice_2[i]) for i in range(min(len(slice_1), len(slice_2)))
            )
        return corr_values

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

    def cand(h, t):
        return (h, compute_c5(h), len(h))

    def soft_clip(h, eps=1e-8):
        h_clipped = np.clip(h, 0, 1 - eps)
        h_clipped *= (target_integral / np.sum(h_clipped))
        return h_clipped

    h_best = initial_h_values.copy()
    c5_best = compute_c5(h_best)

    # Local search or optimization routine
    steps = 0
    total_rounds = 2000
    while time.time() - start_time < budget_s:
        steps += 1
        # Random perturbation
        h_new = h_best + 0.04 * (2 * rnd.rand(n_points) - 1)
        h_new = soft_clip(h_new)

        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            print(f"Improved C₅: {c5_best:.8f}")
        if steps % 100 == 0:
            print(f"{time.time()-start_time:.2f}s passed, current C₅: {c5_best:.8f}")

    return (h_best.tolist(), c5_best, n_points)
