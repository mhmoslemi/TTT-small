import numpy as np
from scipy.ndimage import correlate
import time
import random

def evaluate_candidate(h, n_points):
    overlap = np.array([correlate(h, 1 - np.roll(h, k), mode='full') for k in range(n_points)])
    overlap = overlap[:, n_points - 1 : 2 * n_points - 1]
    dx = 2.0 / n_points
    overlap_values = overlap.T * dx
    c5 = np.max(overlap_values)
    return c5

def correct_integral_constraint(h, n_points):
    total = np.sum(h)
    target = n_points / 2
    if abs(total - target) < 1e-8:
        return h
    diff = target - total
    indices = np.argsort(h)
    if diff > 0:
        h[indices[-int(abs(diff) * 10):]] += diff / int(abs(diff) * 10)
    else:
        h[indices[:int(abs(diff) * 10)]] -= abs(diff) / int(abs(diff) * 10)
    np.clip(h, 0, 1, out=h)
    return h

def initialize_particle(n_points, base=None, perturbation_scale=0.1):
    if base is not None:
        h = base + np.random.uniform(-perturbation_scale, perturbation_scale, n_points)
    else:
        h = np.random.uniform(0, 1, n_points)
    h = correct_integral_constraint(h, n_points)
    return h

def run(seed=42, budget_s=1100, n_points=70, n_particles=50, n_iterations=600, omega=0.7, c1=1.5, c2=1.5):
    random.seed(seed)

    best_h = correct_integral_constraint(best_h, n_points)
    best_c5 = evaluate_candidate(best_h, n_points)

    particles = [initialize_particle(n_points, initial_h_values) for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    particle_best = [np.copy(p) for p in particles]
    particle_best_c5 = [evaluate_candidate(p, n_points) for p in particles]
    global_best_index = np.argmin(particle_best_c5)
    global_best_h = np.copy(particle_best[global_best_index])
    best_c5 = particle_best_c5[global_best_index]

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 10 and iteration < n_iterations:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                omega * velocities[i]
                + c1 * r1 * (particle_best[i] - particles[i])
                + c2 * r2 * (global_best_h - particles[i])
            )
            new_h = particles[i] + velocities[i]
            new_h = correct_integral_constraint(new_h, n_points)
            particles[i] = new_h
            c5 = evaluate_candidate(particles[i], n_points)
            if c5 < particle_best_c5[i]:
                particle_best[i] = new_h.copy()
                particle_best_c5[i] = c5
        global_best_index = np.argmin(particle_best_c5)
        if particle_best_c5[global_best_index] < best_c5:
            global_best_h = np.copy(particle_best[global_best_index])
            best_c5 = particle_best_c5[global_best_index]
        iteration += 1

    return global_best_h, best_c5, n_points
