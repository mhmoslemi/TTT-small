import time
import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    max_iter = 100
    starting_n = len(initial_h_values)
    h = np.copy(initial_h_values)
    dx = 2.0 / starting_n
    total_integral = 1.0

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    elapsed_time = 0.0
    tolerance = 1e-5
    low = 0.38085
    high = 0.4
    best_c5_result = best_c5

    while elapsed_time < budget_s:
        mid = (low + high) / 2
        h_var = cp.Variable(starting_n)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == starting_n / 2
        ]

        overlaps = []
        for k in range(-starting_n + 1, starting_n):
            if -2 <= k * dx <= 2 and k != 0:
                left = max(0, -k)
                right = min(starting_n, starting_n - k)
                if left < right:
                    overlap = cp.sum(h_var[left:right] * (1 - h_var[left + k:right + k])) * dx
                    overlaps.append(overlap)

        for overlap in overlaps:
            constraints.append(overlap <= mid)

        objective = cp.Minimize(mid)
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_candidate = h_var.value
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5_result:
                    best_c5_result = c5_candidate
                    best_h = h_candidate
                high = mid
            else:
                low = mid
        except Exception:
            low = mid

        elapsed_time = time.time() - start_time

        if high - low < tolerance:
            break

    return (best_h, best_c5_result, starting_n)
