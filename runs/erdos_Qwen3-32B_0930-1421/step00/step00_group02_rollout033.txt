import numpy as np
import cvxpy as cp
import time
import scipy.signal as signal

def evaluate_erdos_solution(h, n_points):
    """Evaluates the maximum overlap integral as per Erdős overlapping problem."""
    dx = 2.0 / n_points
    h_vec = np.array(h).reshape(-1)
    h_step = h_vec
    h_compat = 1.0 - h_step
    corr = signal.correlate64(h_step, h_compat, mode='full')
    corr *= dx
    c5 = np.max(corr)
    return c5
