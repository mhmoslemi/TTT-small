import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    rev_h = 1 - h
    corr = correlate(h, rev_h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_simplex(y):
    """Project onto simplex: sum(h) == n_points / 2, 0 <= h <= 1"""
    m = len(y)
    n_points = m
    s = n_points / 2
    y_flat = y.flatten()
    y_sorted = np.sort(y_flat)
    y_sorted[::-1].sort()
    y_sorted = np.maximum(y_sorted, 0)
    y_sorted = np.minimum(y_sorted, 1)
    y_flat_sorted = y_sorted
    y_flat_sorted[~(y_flat_sorted > 0)] = 0

    y_flat_sorted[~(y_flat_sorted < 1)] = 1

    s_sum = np.cumsum(y_flat_sorted)
    r = np.sum(y_flat_sorted)
    if r <= s:  # Clamp if under
        return np.clip(y, 0, 1)
    
    # Find threshold
    i = np.searchsorted(s_sum, s, side='right')
    t = (s_sum[i - 1] - s) / i if i > 0 else 0
    y_opt = np.clip(y - t, 0, 1)
    return y_opt

    lr = 0.01

    budget_used = 0

    while time.time() - t0 < budget_s:
        rev_h = 1 - h

        # Compute all correlations (overlap integrals C_k)
        corr = correlate(h, rev_h, mode='full')

        # Normalize C_k by dx
        corr *= dx

        # Compute gradient of log-sum-exp approximation of max
        max_c = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_c))
        exp_sum = np.sum(exp_vals)
        grad_weights = (exp_vals / exp_sum)

        # Compute the gradient of C_5 approximation as the convex combination
        corr_diffs = correlate(h, -rev_h, mode='full')  # C_k gradients
        grad_corr = grad_weights * corr_diffs * dx

        # Project the gradient into the input size domain
        grad_h = np.zeros_like(h)
        L = len(grad_corr)

        for k in range(n_points):
            if L >= n_points + k:
                grad_h[k] += np.sum(grad_corr[L - (n_points - k): L - (n_points - k) + k + 1])

        # Perform projected gradient descent
        h -= lr * grad_h
        h = project_to_simplex(h)

        current_c5 = evaluate_erdos_solution(h)

        if current_c5 < best_c5_so_far:
            best_c5_so_far = current_c5
            best_h = h.copy()
            no_progress = 0
        else:
            no_progress += 1

        if no_progress >= max_no_progress:
            lr *= decay_factor
            no_progress = 0

        if (time.time() - t0) // 100 > budget_used:
            budget_used += 100
            beta = min(beta * beta_increase_factor, 1e5)
            beta_increase_factor **= 1.1

    return (best_h.tolist(), best_c5_so_far, n_points)
