import time
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
from itertools import product


def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corr = correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided parent construction if available
    initial_h_values_list = np.array(initial_h_values).tolist() if 'initial_h_values' in globals() else None
    n_points = 70 if initial_h_values_list is not None else 100

    if initial_h_values_list is not None:
        h = np.array(initial_h_values_list)
        n_points = len(h)
        dx = 2.0 / n_points
        assert abs(np.sum(h) - n_points / 2) < 1e-6, "Initial h violates the average constraint"
    else:
        # Generate an initial guess: balanced steps
        h = np.full(n_points, 0.0)
        middle = n_points // 2
        h[:middle] = 1.0
        h[middle:] = 0.0
        h = h / h.sum() * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    low = 0.38085
    high = 0.4
    brackets = int(np.log((high - low) / 1e-6) / np.log(2)) + 1  # number of steps for binary search
    max_time_per_iteration = (budget_s - 10) / brackets  # Ensure there is some buffer

    def feasible(C):
        h_var = cp.Variable(n_points)
        ones_minus_h_var = 1 - h_var
        overlaps = []
        for k in range(1, n_points):
            shifted_h = np.roll(ones_minus_h_var, k)
            overlap = cp.sum(h_var * shifted_h) * (2.0 / n_points)
            overlaps.append(overlap)

        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        for overlap in overlaps:
            constraints.append(overlap <= C)

        problem = cp.Problem(cp.Minimize(0), constraints)
        problem.solve(solver=cp.ECOS, max_iters=100, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
        else:
            return None

    # Binary search for the smallest feasible C
    for idx in range(brackets):
        if time.time() - start_time > budget_s - 30:
            break
        mid = (low + high) / 2
        feasible_h = feasible(mid)
        if feasible_h is not None:
            high = mid
            feasible_h = np.clip(feasible_h, 0.0, 1.0)
            feasible_c5 = evaluate_c5(feasible_h)
            if feasible_c5 < best_c5:
                best_h = feasible_h
                best_c5 = feasible_c5
        else:
            low = mid

    return best_h, best_c5, n_points
