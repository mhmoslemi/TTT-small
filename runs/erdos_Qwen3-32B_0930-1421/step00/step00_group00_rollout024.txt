import numpy as np
import time
from scipy.signal import correlate

def project(h, target_sum):
    n_points = len(h)
    h_proj = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) < 1e-8:
        return h_proj

    indices_nonzero = np.where(h_proj > 0.0)[0]
    indices_not_max = np.where(h_proj < 1.0)[0]

    if diff > 0:
        # Need to increase some elements
        num_increm = len(indices_not_max)
        if num_increm > 0:
            h_proj[indices_not_max] += diff / num_increm
    else:
        # Need to decrease some elements
        num_decrem = len(indices_nonzero)
        if num_decrem > 0:
            h_proj[indices_nonzero] += diff / num_decrem

    h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def evaluate_erdos_integration(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = h * dx
    # Compute np.correlate(h, 1-h, mode='full') efficiently
    ones = np.ones_like(h)
    ones_h = ones - h
    corr = correlate(h, ones_h, mode='full')
    return np.max(corr) / (dx)
