import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Use the initial_h_values from the parent construction
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Because ∫ h(x) dx = 1

    def compute_C5(h):
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h = np.clip(h, 0, 1)  # Enforce bounds
        s = h.sum()
        if s != target_sum:
            h *= target_sum / s  # Normalize to satisfy the integral constraint
        return h

    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    current_h = h.copy()
    start_time = time.time()

    max_iters = 1000
    for iter in range(max_iters):
        if time.time() - start_time >= budget_s:
            break

        ones_minus_h = 1 - current_h
        corr = correlate(current_h, ones_minus_h, mode="valid") * dx
        max_corr = np.max(corr)
        idx = np.argmax(corr)

        # Compute gradient for the dominant shift k
        start_shift = max(0, idx - n_points)
        end_shift = min(n_points, idx + 1)
        grad = np.zeros_like(current_h)
        for i in range(len(grad)):
            j = (i - idx) % n_points
            if j < n_points:
                grad[i] = -ones_minus_h[j % n_points]

        # Backtracking line search parameters
        alpha = 1.0  # Step size
        beta = 0.5   # How much to reduce step
        c = 0.5      # Armijo condition coefficient
        max_line_search = 10

        for _ in range(max_line_search):
            proposed_h = current_h - alpha * grad
            proposed_h = project(proposed_h)
            new_c5 = compute_C5(proposed_h)
            if new_c5 < best_c5:

            alpha *= beta

        if new_c5 < best_c5:
            current_h = proposed_h
            best_c5 = new_c5
            best_h = current_h.copy()

    return (best_h.tolist(), best_c5, n_points)
