import numpy as np
import cvxpy as cp
import time
from math import ceil

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    starting_h = initial_h_values.copy()
    n_points = len(starting_h)

    def evaluate_c5(h):
        dx = 2.0 / n_points
        k_start = -(n_points - 1)
        k_end = n_points - 1
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[(n_points - 1):]
        c5 = np.max(corr * dx)
        return c5

    h = cp.Variable(n_points, nonneg=True)
    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ]

    t_start = time.time()

    # Binary search parameters
    low, high = 0.38085, 0.3815
    best_h = starting_h.copy()
    best_c5 = evaluate_c5(best_h)
    tolerance = 1e-6

    while time.time() - t_start < budget_s - 20 and high - low > tolerance:
        mid = (low + high) / 2
        target = mid
        temp_obj = mid  # Placeholder for objective (we only care about feasibility)
        obj = cp.Minimize(cp.Constant(temp_obj))
        overlap_constraints = []

        for lag in range(1, n_points):
            shifted_h = np.roll(h, -lag)
            shifted_h[:lag] = 0.0
            overlap = cp.sum(h * (1 - shifted_h)) * (2.0 / n_points)
            overlap_constraints.append(overlap <= target)

        prob = cp.Problem(obj, constraints + overlap_constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_c5 = evaluate_c5(h.value)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.value.copy()
            high = mid
        else:
            low = mid

    elapsed_time = time.time() - t_start

    return (best_h, best_c5, n_points)
