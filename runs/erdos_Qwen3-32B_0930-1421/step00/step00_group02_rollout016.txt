import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)[0]

    # Function to evaluate C5
    def compute_c5(h):
        return evaluate_erdos_solution(h)[0]

    # Main optimization loop
    while time.time() - start_time < budget_s:
        h_cvxpy = cp.Variable(n_points, nonneg=True, pos=True)
        constraints = [h_cvxpy <= 1.0, cp.sum(h_cvxpy) == n_points / 2]

        # Approximate the max conv with a single shift step for speed
        best_shift = 0
        best_c5_shift = np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        valid_k = np.argmax(corr) - n_points + 1  # Alignment index k where overlap is max
        if 0 <= valid_k < n_points:
            best_shift = valid_k

        # Build small objective by penalizing single k where overlap is max
        overlap = cp.sum(
            h_cvxpy[0 : n_points - best_shift] *
            (1 - h_cvxpy[best_shift : n_points])
        ) * dx

        prob = cp.Problem(cp.Minimize(overlap), constraints)
        prob.solve(solver=cp.ECOS, kktsolver="robust")

        if prob.status != "optimal":
            h_new = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h_new = h_new * (n_points / 2 / h_new.sum())
        else:
            h_new = h_cvxpy.value

        # Repair mass constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2 / h_new.sum())

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        # Early stopping if feasible improvement
        if best_c5 < 0.380875:
            break

    return best_h, best_c5, n_points
