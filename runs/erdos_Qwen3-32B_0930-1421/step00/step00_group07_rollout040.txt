import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return c5

def project_feasible(h, n_points):
    # Clip values to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Normalize to maintain the integral constraint
    h_sum = np.sum(h_clipped)
    h_normalized = h_clipped / h_sum * (n_points / 2.0)
    return h_normalized
