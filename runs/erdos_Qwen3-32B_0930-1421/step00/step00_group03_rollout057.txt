import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    full_corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(full_corr)

def run(seed=42, budget_s=1100, n_points=70, max_iterations=50, tol=1e-5):
    np.random.seed(seed)
    dx = 2.0 / n_points
    h_init = np.array(initial_h_values)
    assert np.all(h_init >= 0) and np.all(h_init <= 1), "Initial h out of bounds"
    assert np.isclose(np.sum(h_init) * dx, 1.0), "Initial h does not satisfy the integral constraint"

    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    best_h_values = best_h.tolist()
    start_time = time.time()

    # Binary search over C₅
    low = best_c5
    high = 0.4
    iterations = 0

    while time.time() - start_time < budget_s and iterations < max_iterations:
        mid = (low + high) / 2
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h - best_h))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0,
        ]

        lags = np.arange(-n_points + 1, n_points)
        for k in lags:
            if k < 0:
                left = -k
                right = n_points
                overlap = h[left:] @ (1 - h[:n_points - left])
            elif k == 0:
                overlap = h @ (1 - h)
            else:
                left = 0
                right = n_points - k
                overlap = h[:right] @ (1 - h[k:])
            constraints.append(overlap <= mid)

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.array(h.value).flatten()
            # Post-process to clip due to numerical errors
            h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_current = evaluate_erdos_solution(h_opt, n_points)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_opt
                best_h_values = best_h.tolist()
            high = mid
        else:
            low = mid
        iterations += 1

    return (best_h_values, best_c5, n_points)
