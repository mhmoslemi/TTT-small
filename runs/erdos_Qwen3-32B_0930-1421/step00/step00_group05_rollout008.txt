import numpy as np
import time
import math

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure constraint: sum(h) = n_points / 2
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def evaluate_position(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    learning_rate = 0.1
    improvement_threshold = 1e-6
    prev_c5 = None
    no_improvement = 0
    max_no_improvement = 200

    while time.time() - start_time < budget_s:
        c5 = compute_c5(h, dx)
        if prev_c5 is not None and abs(c5 - prev_c5) < improvement_threshold:
            no_improvement += 1
        else:
            no_improvement = 0
        prev_c5 = c5

        if best_c5 > c5:
            best_h[:] = h[:]
            best_c5 = c5

        if no_improvement >= max_no_improvement:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        indices = np.arange(len(h))
        overlap = corr * dx
        k_max = np.nanargmax(overlap)

        # Compute gradient for this k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            l_index = i - k_max + n_points
            l_index %= n_points
            grad[i] = (1 - h[i]) * h[l_index] - h[i] * (1 - h[l_index])

        # Gradient descent update
        h -= learning_rate * grad

        # Projection step
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)

        if learning_rate > 1e-6:
            learning_rate *= 0.995

    return best_h, best_c5, n_points
