import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_h = 1.0 - h
    conv_result = correlate(h, ones_h, mode='full') * dx
    c5 = np.max(conv_result)
    return c5

def run(seed=42, budget_s=1100, n_points=42):
    np.random.seed(seed)
    max_evals = 1_000_000
    patience = 100
    step_size = 0.01
    t_start = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values.astype(float)
        n_points = len(h)
    else:
        h = np.ones(n_points) * (1.0 / n_points) * 2.0

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_progress = 0

    for i in range(max_evals):
        # Gradient step
        ones_h = 1.0 - h
        conv = correlate(h, ones_h, mode='full')
        conv[conv == 0] = np.nan  # prevent 0 gradient
        grad = -np.gradient(conv, axis=1)  # approximate gradient

        grad_h = np.zeros_like(h)
        for k in range(len(h) * 2 - 1):
            start = max(0, k - len(h) + 1)
            end = min(len(h), k + 1)
            grad_h[start:end] += grad[k] / (len(h) * 2 - 1)

        h -= step_size * grad_h

        # Project into [0, 1]
        h = np.clip(h, 0, 1)

        # Enforce integral constraint
        sum_h = h.sum()
        if abs(sum_h - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / sum_h

        # Evaluate improvement
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_progress = 0
        else:
            no_progress += 1

        if time.time() - t_start >= budget_s or no_progress >= patience:
            break

    return best_h, best_c5, n_points
