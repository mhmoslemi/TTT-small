```python
import numpy as np
import cvxpy as cp
import time
from typing import Tuple
import evaluate_erdos_solution as evaluator

def run(seed=42, budget_s=1100, **kwargs) -> Tuple[np.ndarray, float, int]:
    # Seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()

    # Retrieve initial construction from the pre-imported global variable
    h_initial = evaluator.initial_h_values
    n_points = len(h_initial)

    # Discretized partitioning of the domain [0, 2]
    dx = 2.0 / n_points

    # Ensure the initial solution is feasible
    assert np.isclose(np.sum(h_initial) * dx, 1.0), "Initial solution is infeasible."

    # CVXPY variables for h
    h = cp.Variable(shape=(n_points), nonneg=True)

    # Compute max overlap at all shifts
    def compute_overlap(h_vec: np.ndarray):
        correlate = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(correlate * dx)

    # Build expressions for each shifted overlap
    all_overlaps = []
    for shift in range(-n_points + 1, n_points):
        if shift < 0:
            shifted_h = h[shift:]
            basis = 1 - h[:n_points + shift]
        elif shift == 0:
            continue  # h(x)*(1-h(x)) = 0 due to symmetry
        else:
            shifted_h = h[:-shift]
            basis = 1 - h[shift:]
        overlap = cp.sum(shifted_h * basis) * dx
        all_overlaps.append(overlap)

    # Objective: minimize the max overlap
    objective = cp.Minimize(cp.max(all_overlaps))

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,  # Mass constraint
        h <= 1,                     # Bounded above constraint
        h >= 0                      # Bounded below constraint
    ]

    # Formulate and solve problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.GUROBI, verbose=False, timeLimit=max(0, budget_s - (time.time() - start_time)))

    # Failed solve or timeout: return the initial construction
    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = h_initial
        c5_bound = compute_overlap(h_values)
        return h_values, c5_bound, n_points

    h_values = np.clip(h.value, 0, 1)

    # Final evaluation
    c5_bound = compute_overlap(h_values)

    return h_values, c5_bound, n_points
```
