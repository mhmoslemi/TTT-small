import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_c5(h):
        h_dx = h * dx
        max_corr = 0.0
        max_idx = 0
        for k in range(n_points - 1):
            h1 = h_dx
            h2 = (1 - np.roll(h_dx, k + 1))[:len(h1)]
            corr = np.sum(h1 * h2)
            if corr > max_corr:
                max_corr = corr
                max_idx = k
        return max_corr, max_idx

    while time.time() - start_time < budget_s:
        corr, k_star = compute_c5(h)
        if corr < best_c5:
            best_h[:] = h.copy()
            best_c5 = corr

        gradient = np.zeros_like(h)
        h1 = h * dx
        h2 = (1 - np.roll(h_dx, k_star + 1))[:len(h1)]

        # Gradient is h2 - 2*h1*h2 + h1
        grad = (h2 - 2 * h1 * h2 + h1) / dx
        gradient = grad

        h_new = cp.Variable(n_points)

        objective = cp.Minimize(cp.norm(h_new - (h - 0.1 * gradient), 2))

        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2
        ]

        try:
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.ECOS)
            h = np.clip(h_new.value, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        except:
            pass

        if (time.time() - start_time) % 100 < 1:
            curr_c5, _ = compute_c5(h)
            if curr_c5 < best_c5:
                best_h[:] = h.copy()
                best_c5 = curr_c5

    return best_h, best_c5, n_points
