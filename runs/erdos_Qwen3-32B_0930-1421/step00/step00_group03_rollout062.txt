import numpy as np
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    n_particles = 50
    max_iter = 500
    omega = 0.7
    c1, c2 = 1.5, 1.5

    # Initialize particles around the initial solution
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    for i in range(n_particles):
        particles[i] = h_init + (np.random.rand(n_points) - 0.5) * 0.1
        # Enforce bounds and constraints
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = enforce_integral(particles[i], n_points)

    # Initial personal best and global best
    personal_best = particles.copy()
    personal_best_val = np.array([objective(p, n_points, dx) for p in particles])
    global_idx = np.argmin(personal_best_val)
    global_best = particles[global_idx]
    global_best_val = personal_best_val[global_idx]

    start_time = time.time()

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        inertia_decay = 1.0 / np.sqrt(max(1, iteration + 1))
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                inertia_decay * omega * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            particles[i] += velocities[i]

        # Enforce bounds and constraints
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            particles[i] = enforce_integral(particles[i], n_points)

        # Evaluate and update bests
        current_vals = np.array([objective(p, n_points, dx) for p in particles])
        improved = current_vals < personal_best_val
        for i in np.where(improved)[0]:
            personal_best[i] = particles[i].copy()
            personal_best_val[i] = current_vals[i]
        new_global_idx = np.argmin(personal_best_val)
        if personal_best_val[new_global_idx] < global_best_val:
            global_best = personal_best[new_global_idx].copy()
            global_best_val = personal_best_val[new_global_idx]

    return global_best.tolist(), global_best_val.item(), n_points


def enforce_integral(h, n_points):
    target = n_points / 2.0
    current = h.sum()
    if np.abs(current - target) < 1e-6:
        return h
    diff = target - current
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0
        h[mask] += np.abs(diff) / mask.sum()
        h = np.clip(h, 0, 1)
    else:
        # Need to decrease sum
        mask = h > 0.0
        h[mask] -= np.abs(diff) / mask.sum()
        h = np.clip(h, 0, 1)
    return h


def objective(h, n_points, dx):
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode="full")
    C5 = np.max(np.abs(corr * dx))
    return C5
