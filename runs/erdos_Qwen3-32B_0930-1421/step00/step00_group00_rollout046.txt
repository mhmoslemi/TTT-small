import numpy as np
import cvxpy as cp
import time
from math import isclose

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 0.044444444444444446  # dx = 2 / 45
    h0 = initial_h_values
    n_points = len(h0)
    assert isclose(np.sum(h0) * dx, 1, abs_tol=1e-6)
    best_h = h0.copy()
    t0 = time.time()

    def compute_c5(h):
        h_scaled = h / dx  # Normalize to density function on [0,2]
        cross = np.correlate(h_scaled, 1 - h_scaled, mode='full') * dx**2
        return np.max(cross)

    best_c5 = compute_c5(best_h)

    while time.time() - t0 < budget_s:
        # Gradient ascent-like update
        for k in range(2 * n_points - 1):
            # Current cross-correlation at shift k
            iwrap = (np.arange(len(h0)) + k) % n_points
            contrib = best_h * (1 - best_h[iwrap])
            grad = (1 - 2 * best_h) * (1 - best_h[iwrap]) - contrib
            grad[iwrap] -= (1 - 2 * best_h[iwrap]) * best_h

            # scaled grad ascent
            best_h += 0.01 * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h - (np.sum(best_h) - n_points / 2) / n_points

        new_c5 = compute_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        if time.time() - t0 > budget_s - 2:
            break

    return best_h.tolist(), best_c5, n_points
