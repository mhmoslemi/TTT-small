import numpy as np
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h = initial_h_values

    n_points = len(initial_h)
    dx = 2.0 / n_points

    h = np.copy(initial_h)
    h = np.clip(h, 0.0, 1.0)

    # Normalize to ensure sum == n_points / 2
    total = np.sum(h)
    if abs(total) > 1e-10:
        h *= (n_points / 2.0) / total

    best_h = np.copy(h)
    best_c5 = np.inf
    start_time = time.time()

    alpha = 0.02
    patience = 200
    no_improvement_counter = 0
    max_iter = 1000

    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute all overlaps via cross correlation
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        k_max = np.argmax(corr)

        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h[:]
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        # Early stopping if no improvement and reduce learning rate
        if no_improvement_counter >= patience:
            alpha *= 0.5
            no_improvement_counter = 0
            if alpha < 1e-6:
                break

        # Compute gradient for the worst shift k
        grad = np.zeros_like(h)

        for i in range(n_points):
            k = k_max

            # Get the index shifted forward by k
            forward_idx = i + k
            if forward_idx < n_points:
                grad[i] -= 1.0 - h[forward_idx]

            # Get the index shifted backward by k
            backward_idx = i - k
            if backward_idx >= 0:
                grad[i] += h[backward_idx]

        # Update h with gradient descent
        h -= alpha * grad

        # Clip to bounds [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Project onto constraint surface: sum(h) == n_points / 2
        total = np.sum(h)
        if abs(total) > 1e-10:
            h *= (n_points / 2.0) / total

    return (best_h, best_c5, n_points)
