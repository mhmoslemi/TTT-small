import numpy as np
import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    np.random.seed(seed)

    # Use the provided initial solution, or a uniform guess if not available
    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values should be available for this run.")

    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = np.inf
    total_time = 0
    start_time = time.time()

    def compute_C5(h):
        n = len(h)
        dx_inv = n / 2  # since 1/dx = n/2
        max_c5 = -1
        # Compute overlap integrals for all valid overlap shifts
        for k in range(1, n - 1):  # exclude zero offset overlap
            valid_start = max(0, k - n + 1)
            valid_end = min(n - k, n)
            idx = np.arange(valid_start, valid_end)
            y1 = h[idx]
            y2 = h[(idx + k) % n]
            c5 = np.sum(y1 * (1 - y2)) * dx
            max_c5 = max(max_c5, c5)
        return max_c5

    def clip_and_rescale(h):
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
        s = np.sum(h)
        h *= (n_points / 2) / s  # rescale to maintain density
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
        np.clip(h, 0.0, 1.0, out=h)
        return h

    def compute_gradient(h, C5_k_indices):
        n = len(h)
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in C5_k_indices:
            indices = np.arange(0, n - k)
            h1 = h[indices]
            h2 = h[indices + k]
            dh2 = -h1
            dh1 = 1 - h2
            grad[indices + k] += dh2 * dx
            grad[indices] += dh1 * dx
        return grad

    # Initialize
    h[:] = np.clip(h, 0.0, 1.0)
    h = clip_and_rescale(h)
    current_c5 = compute_C5(h)
    best_c5 = current_c5
    best_h[:] = h

    # Determine worst-overlap shifts
    n = len(h)
    c5_values = {}
    for k in range(1, n - 1):
        valid_start = max(0, k - n + 1)
        valid_end = min(n - k, n)
        idx = np.arange(valid_start, valid_end)
        y1 = h[idx]
        y2 = h[(idx + k) % n]
        c5_k = np.sum(y1 * (1 - y2)) * dx
        c5_values[k] = c5_k

    C5_k_indices = [k for k, cv in c5_values.items() if abs(cv - best_c5) < 1e-4]

    # Optimization loop
    alpha = 0.01  # Learning rate
    while total_time < budget_s:
        # Gradient update
        grad = compute_gradient(h, C5_k_indices)
        h += alpha * grad
        h = clip_and_rescale(h)

        # Evaluate current_C5
        current_c5 = compute_C5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h

        # Check time budget
        if time.time() - start_time > budget_s:
            break

        # Alpha schedule (optional decay)
        alpha = max(0.001, alpha * 0.999)

    return (best_h.tolist(), best_c5, n_points)
