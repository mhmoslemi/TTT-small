import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()

    # Use the given initial construction
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure the constraint sum(h) == n_points / 2 (i.e., ∫₀² h(x) dx = 1)
    # Fix if slightly violated
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2):
        h = h * (n_points / 2) / h_sum

    def compute_c5(h, dx):
        # Compute the overlap integral
        h = h.flatten()
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        corr = corr[n_points - 1: -n_points + 1]  # Valid shifts
        c5 = np.max(corr * dx)
        return c5

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    def solve_optimization(h_init):
        h_var = cp.Variable(h_init.shape)
        obj = cp.Minimize(cp.max(
            cp.conv(to_padded(h_var), to_padded(1.0 - h_var))[n_points-1:-n_points+1] * dx
        ))
        constraints = [
            h_var >= 0.0,
            h_var <= 1.0,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False)
        return h_var.value

    def to_padded(x):
        return np.lib.pad(x, (n_points - 1, n_points - 1), 'constant', constant_values=(0, 0))

    while time.time() - start_time < budget_s:
        h = solve_optimization(h)
        if h is None:
            break
        # Ensure feasibility due to numerical error
        h[h < 0.0] = 0.0
        h[h > 1.0] = 1.0
        h_sum = np.sum(h)
        if not np.isclose(h_sum, n_points / 2):
            h = h * (n_points / 2) / h_sum
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return (best_h, best_c5, n_points)
