import numpy as np
import time
import math
import evaluate_erdos_solution as ev
import cvxpy as cp
import scipy.optimize as spo
from scipy.signal import correlate

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-6:
        return h
    # Adjust largest elements if we have surplus
    if diff > 0:
        sort_idx = np.argsort(h)[::-1]
        adjust = h[sort_idx[0]] - 1e-6  # Avoid hitting 1 unless forced
        N = int(np.ceil(diff / adjust))
        h[sort_idx[:N]] -= diff / N
    else:
        # Adjust smallest elements if we have deficit
        sort_idx = np.argsort(h)
        adjust = h[sort_idx[0]] + 1e-6  # Avoid hitting 0 unless forced
        N = int(np.ceil(abs(diff) / adjust))
        h[sort_idx[:N]] += abs(diff) / N
    return h


def eval_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)


def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    np.random.seed(seed)
    h_best = np.array(initial_h_values)
    c5_best = eval_c5(h_best, n_points)
    h_history = [h_best.copy()]
    c5_history = [c5_best]
    start_time = time.time()

    # Simulated Annealing
    T = 1.0
    alpha = 0.995
    max_sa_iter = 4000
    sigma = 0.2
    best_time = 0.0
    while time.time() - start_time < budget_s:
        for sa_iter in range(max_sa_iter):
            if time.time() - start_time > budget_s - 10:
                break
            h = project_to_feasible(h_best + np.random.normal(0, sigma, n_points), n_points)
            c5_new = eval_c5(h, n_points)
            delta = c5_best - c5_new
            if c5_new < c5_best or np.random.rand() < np.exp(delta / T):
                h_best[:] = h
                c5_best = c5_new
                h_history.append(h_best.copy())
                c5_history.append(c5_best)
                best_time = time.time()
            # Cooling schedule
            T *= alpha
            sigma *= np.sqrt(1 - sa_iter / max_sa_iter)

        # Projection if solution deviated
        h_best = project_to_feasible(h_best, n_points)
        c5_best = eval_c5(h_best, n_points)
        # Break if no improvement for a while
        if time.time() - best_time > budget_s * 0.1:
            break

        # Local refinement with gradient projection descent
        max_gd_iter = 200
        for gd_iter in range(max_gd_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # 1 step of projected gradient
            dx = 2.0 / n_points
            corr = correlate(h_best, 1 - h_best, mode='full')
            max_idx = np.argmax(corr)
            max_k = max_idx - n_points + 1
            if max_k < 0 or max_k >= n_points:
                continue
            grad = -2.0 * (1 - h_best[(n_points - 1 + max_k) % n_points]) * dx
            h_best = project_to_feasible(h_best - 0.01 * grad * h_best, n_points)
            c5_new = eval_c5(h_best, n_points)
            if c5_new < c5_best:
                c5_best = c5_new
                h_history.append(h_best.copy())
                c5_history.append(c5_new)

    return h_best, c5_best, n_points
