import numpy as np
import cvxpy as cp
import time

def correlate_overlap(h, dx):
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return overlaps[1:-1]  # exclude self-overlap at index 0 and 2n-2

def optimize_erdos(h0, dx, iterations=1000, alpha=0.5, alpha_decay=0.995,
                   improvement_threshold=1e-5, max_h_shifts=20, verbose=False):
    n_points = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_c5 = np.inf
    start_time = time.time()

    for it in range(iterations):
        delta_time = time.time() - start_time
        if delta_time >= 1100:
            if verbose:
                print(f"Time budget exceeded at iteration {it}")
            break

        h_shifted = alpha * h + (1 - alpha) * h_prev
        overlaps_raw = np.correlate(h_shifted, 1 - h, mode='full') * dx
        overlaps = overlaps_raw[1:-1]  # skip negative and out-of-bound shifts

        if len(overlaps) == 0:
            if verbose:
                print("Zero overlap: breaking")
            break

        k_max = np.argmax(overlaps)
        if k_max >= n_points:
            k_max -= n_points

        # Form CVXPY problem: optimize h for k_max shifted overlap
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(overlaps_raw[k_max+1])
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var <= 1
        ]

        problem = cp.Problem(objective, constraints)
        problem.solve(solver='ECOS')

        if problem.status != 'optimal':
            if verbose:
                print(f"CVXPY failed at iter {it}: {problem.status}")
            break

        new_h = h_var.value
        h = np.clip((new_h + h_shifted) / 2, 0, 1)

        # Projection to constraint for safety
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h / h_sum * (n_points / 2)
        h = np.clip(h, 0, 1)

        current_c5 = np.max(correlate_overlap(h, dx))
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            if verbose:
                print(f"Improved: c5 = {best_c5:.8f} (iter {it})")
        else:
            alpha *= alpha_decay  # Conservative update if no improvement

        h_prev = h.copy()  # Prepare for next h_shifted

    return best_h, best_c5

def run(seed=42, budget_s=1100, n_points=95, verbose=False, max_initial_attempts=10):
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use initial_h_values if available
    use_initial = 'initial_h_values' in globals()
    if use_initial:
        h = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    else:
        # Build initial h with uniform distribution of unit mass
        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2)

    h = np.clip(h, 0, 1)  # Enforce bounds

    h_opt, c5_opt = optimize_erdos(h, dx, verbose=verbose)

    # Validate the result
    assert np.all((0 <= h_opt) & (h_opt <= 1)), "h out of bounds"
    assert abs(np.sum(h_opt) - n_points / 2) < 1e-8, "h sum doesn't match constraint"
    
    return h_opt, c5_opt, n_points
