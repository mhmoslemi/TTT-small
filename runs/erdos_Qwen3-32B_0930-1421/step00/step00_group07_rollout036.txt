import numpy as np
import time
from scipy.signal import correlate

def intervals_to_h(intervals, n_points):
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for a, b in intervals:
        idx_a = int(np.clip(np.floor(a / dx), 0, n_points))
        idx_b = int(np.ceil(b / dx))
        if idx_a > idx_b:
            continue
        h[idx_a:idx_b] = 1
    return h / h.sum() * n_points / 2  # Normalize to have sum(h) == n_points / 2

def compute_overlap(h):
    n_points = len(h)
    dx = 2.0 / n_points
    complement_h = 1 - h
    corr = correlate(complement_h, h, mode='full')
    corr = corr * dx
    return np.max(corr)

            if new_c5 < best_c5:
