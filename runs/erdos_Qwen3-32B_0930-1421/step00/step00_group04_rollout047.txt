def evaluate_c5(h, dx):
    shifted_h = h[::-1]  # Reverse for correct convolution in numpy
    correlate_result = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(correlate_result) * dx
    return c5

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points_options = [84, 42, 168]
    best_h = None
    best_c5 = float('inf')
    best_n_points = None

    if 'initial_h_values' in globals():

        n_points = len(h_initial)
        # Ensure constraint sum(h) == n_points / 2
        scale_factor = (n_points / 2.0) / np.sum(h_initial)
        h_initial *= scale_factor
        h_initial = np.clip(h_initial, 0.0, 1.0)
    else:
        n_points = 84
        h_initial = np.ones(n_points) / n_points
        h_initial = np.clip(h_initial, 0.0, 1.0)

    dx = 2.0 / n_points

    def optimize_step(h, rate, niter=100):
        h = h.copy()
        for _ in range(niter):
            corr = np.correlate(h, 1.0 - h, mode="full")
            grad = rate * (np.correlate(h, -1.0, mode="full") + np.correlate(np.flip(h), 1.0, mode="full"))
            grad = np.clip(np.gradient(corr), -1, 1)
            h_new = h + grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new /= np.sum(h_new)
            h_new *= n_points / 2.0
            h_new = np.clip(h_new, 0.0, 1.0)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < evaluate_c5(h, dx):
                h = h_new
        return h

    def restart_search(h, dx):
        simplex = [h + 0.1 * np.random.randn(len(h)) for _ in range(4)]
        for v in simplex:
            v = np.clip(v, 0.0, 1.0)
            v /= np.sum(v)
            v *= n_points / 2.0
        bestsimplex = min(simplex, key=lambda x: evaluate_c5(x, dx))
        return bestsimplex

    h = h_initial
    c5 = evaluate_c5(h, dx)

    best_h = h
    best_c5 = c5
    best_n_points = n_points

    step_rate = 0.1
    restart_every = 50
    max_iter = 200
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        h = optimize_step(h, step_rate)
        c5 = evaluate_c5(h, dx)

        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points

        if iter_count % restart_every == 0 and iter_count > 0:
            h = restart_search(h, dx)
            step_rate = max(step_rate * 0.5, 0.001)

        iter_count += 1

    return list(best_h.astype(float)), float(best_c5), int(best_n_points)
