import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

np.random.seed(42)

def run(budget_s=1100, initial_population_size=80, generations=1000, mut_rate=0.02):
    start_time = time.time()
    n_points = len(initial_h_values)
    total_sum = n_points / 2

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        divisor = max(1e-8, h_sum)
        h = h * (total_sum / divisor)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = (corr * dx).max()
        return c5, h

    def init_population():
        pop = [np.copy(initial_h_values)]
        for _ in range(initial_population_size - 1):
            noise = np.random.randn(n_points) * 0.1
            candidate = np.clip(initial_h_values + noise, 0, 1)
            candidate_sum = candidate.sum()
            if candidate_sum == 0:
                candidate = np.zeros_like(candidate)
            else:
                candidate = (candidate / candidate_sum) * total_sum
                candidate = np.clip(candidate, 0, 1)
            pop.append(candidate)
        return pop

    def crossover(p1, p2):
        s = np.random.randint(n_points // 4, 3 * n_points // 4)
        child = np.concatenate((p1[:s], p2[s:]))
        return child

    def mutate(child):
        mask = np.random.rand(n_points) < mut_rate
        child[mask] = np.random.rand(sum(mask))
        child_sum = child.sum()
        if child_sum == 0:
            child = np.zeros_like(child)
        else:
            child = (child / child_sum) * total_sum
            child = np.clip(child, 0, 1)
        return child

    pop = init_population()
    pop = [evaluate(chromo)[1] for chromo in pop]

    best_c5 = np.inf
    best_h = np.copy(initial_h_values)

    for gen in range(generations):
        pop = np.array(pop)
        scores = [evaluate(h)[0] for h in pop]
        indices = np.argsort(scores)
        top_indices = indices[:len(pop) // 5]
        top_pop = pop[top_indices]

        next_pop = [best_h]  # carry over the best
        for _ in range(len(pop) - len(next_pop)):
            p1 = p2 = np.copy(np.random.choice(top_pop))
            while np.array_equal(p1, p2):
                p2 = np.copy(np.random.choice(top_pop))
            child = crossover(p1, p2)
            child = mutate(child)
            next_pop.append(child)

        pop = next_pop
        if (gen + 1) % 100 == 0:
            best_c5, best_h = evaluate(best_h)
            print(f"Generation {gen + 1}/{generations} | C5: {best_c5:.8f}")

        if time.time() - start_time > budget_s - 10:
            best_c5, best_h = evaluate(best_h)
            return best_h.tolist(), best_c5, n_points

    best_c5, best_h = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
