import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
        h = initial_h_values.copy().astype(np.float64)
    except:
        n_points = 95
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    # Problem parameters
    dx = 2.0 / n_points

    def compute_max_overlap(h):
        h = h.flatten()
        corrs = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corrs)

    def optimize_h_convex(h_init):
        h_var = cp.Variable(n_points)
        obj_expr = np.max([np.correlate(h_var, 1 - h_var, mode="full")[k] for k in range(n_points)]) * dx
        obj = cp.Minimize(obj_expr)
        constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
        prob = cp.Problem(obj, constraints)
        start = time.time()
        prob.solve(solver=cp.GUROBI if "GUROBI" in cp.installed_solvers() else cp.SCIP)
        elapsed = time.time() - start
        h_sol = h_var.value.copy()
        return h_sol, elapsed, prob.value

    best_c5 = compute_max_overlap(h)
    best_h = h.copy()
    start_time = time.time()
    max_iter = 50
    for i in range(max_iter):
        h_opt, elapsed, c5 = optimize_h_convex(h)
        if h_opt is not None:
            h = h_opt
            current_c5 = c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        if h is not None:
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        if time.time() - start_time >= budget_s:
            break

    # Recheck final unprojected version
    best_c5 = compute_max_overlap(best_h)
    return best_h, best_c5, n_points
