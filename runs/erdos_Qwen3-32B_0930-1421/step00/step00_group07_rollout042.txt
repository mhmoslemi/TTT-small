import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h0 = np.copy(initial_h_values) if 'initial_h_values' in globals() else np.full(80, 0.5)
    n_points = len(h0)
    h = h0.copy()
    best_h = h.copy()
    dx = 2.0 / n_points

    max_iter = 2000
    alpha = 0.02
    patience = 200
    no_improve = 0
    best_c5 = eval_c5(h)

    start_time = time.time()

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute full cross-correlation using direct correlate
        corr = np.correlate(h, 1 - h, mode="full")
        c_overlap = corr * dx

        # Find the worst overlap index
        max_idx = np.argmax(c_overlap)

        # Compute the shift from center of correlation
        k = max_idx - (n_points - 1)

        # Compute gradient only for the worst k
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k + n_points) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]

        # Apply gradient update
        h -= alpha * grad

        # Projection onto feasible set: [0, 1]
        h = np.clip(h, 0, 1)

        # Normalize to satisfy sum constraint
        s = np.sum(h)
        h *= (n_points / 2.0) / s

        # Evaluate new C5
        new_c5 = eval_c5(h)

        # Track best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                alpha *= 0.5
                no_improve = 0

    # Final evaluation of the best solution
    final_c5 = eval_c5(best_h)

    return best_h, final_c5, n_points

def eval_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    c_overlap = corr * dx
    return np.max(c_overlap)
