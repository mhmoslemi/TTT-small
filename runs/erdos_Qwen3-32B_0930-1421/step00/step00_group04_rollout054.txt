import numpy as np
from scipy.signal import correlate
import time
import random
import cvxpy as cp

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    ch = (1 - h).reshape(-1)
    corr = correlate(h, ch, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def crossover(parent1, parent2, n_points):
    indices = np.random.permutation(n_points)
    split = n_points // 2
    child = np.copy(parent1)
    child[indices[:split]] = parent2[indices[:split]]

    current_sum = np.sum(child)
    desired_sum = n_points / 2
    diff_sum = desired_sum - current_sum
    swap_indices = np.argwhere(np.abs(diff_sum) > 1e-8)
    if swap_indices.size > 0:
        idx = random.choice(swap_indices[0])
        child[indices[idx]] -= diff_sum / swap_indices.shape[0]
    return child.clip(0, 1)

def mutate(individual, mutation_rate, n_points):
    indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
    for i in indices:
        individual[i] = random.random()
    # Adjust for constraint
    current_sum = np.sum(individual)
    adjustment = (n_points / 2 - current_sum) / n_points
    individual += adjustment
    return individual.clip(0, 1)

def ga_search(budget_s, n_points=42, pop_size=40, generations=200, mutation_rate=0.08):
    t_start = time.time()
    if 'initial_h_values' in globals():
        population = [np.clip(np.array(initial_h_values), 0, 1)]
    else:
        population = []

    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)
        population.append(h)

    population = [mutate(np.array(p), mutation_rate, n_points) for p in population]
    best_individual = population[0]
    best_fitness = evaluate_C5(best_individual, n_points)

    for generation in range(generations):
        print(f"Generation: {generation + 1}/{generations} | Elapsed: {time.time() - t_start:.1f}s | Best C5: {best_fitness:.6f}")
        if time.time() - t_start >= budget_s:
            break

        # Evaluate fitness
        fitness = np.array([evaluate_C5(p, n_points) for p in population])
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]

        # Select top 20% as elites
        elite_count = max(1, int(pop_size * 0.2))
        elites = population[:elite_count]
        offspring = elites.copy()

        # Crossover and mutate
        parent_pairs = np.random.permutation(pop_size // 2)

        for i in range(pop_size // 2):
            if i >= len(parent_pairs):
                break
            p1 = population[i]
            p2 = population[(i + 1) % pop_size]
            child = crossover(p1, p2, n_points)
            child = mutate(child, mutation_rate, n_points)
            offspring.append(child)

        population = offspring

        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_individual = population[current_best_idx]
            best_fitness = fitness[current_best_idx]

    return best_individual, best_fitness

def run(seed=42, budget_s=1100, n_points=42):
    np.random.seed(seed)
    random.seed(seed)
    
    final_individual, best_C5 = ga_search(budget_s, n_points)
    return (final_individual.tolist(), best_C5, n_points)
