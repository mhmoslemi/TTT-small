import numpy as np
import time
import cvxpy as cp


def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    correlate = np.correlate(h, 1 - h, mode='full')
    # Trim only half of the tails to avoid exclusions based on odd/even length
    start_index = len(h) - 1
    end_index = len(h) + len(h) - 2
    trimmed_correlate = correlate[start_index:end_index+1]
    return np.max(trimmed_correlate) * dx


def initial_h_values():
    n_points = 45
    h = np.zeros(n_points)
    h[:1 + n_points // 2] = 1.0
    # Normalize to have the correct integral
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h_start = initial_h_values()
    h = h_start.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()

    alpha = 1.0  # Initial step size
    beta = 0.5   # Backtracking coefficient
    max_backtrack = 10

    # Precompute a mask that updates the maximum shift at each iteration
    step = 0

    while time.time() - start_time < budget_s:
        # Compute cross-correlate
        correlate = np.correlate(h, 1 - h, mode='full')
        # Index range with overlap
        start_index = n_points - 1
        end_index = n_points + n_points - 2
        trimmed_correlate = correlate[start_index:end_index+1]

        shift_idx = np.argmax(trimmed_correlate)
        max_shift = start_index + shift_idx

        # Convert shift index to shift distance
        k = max_shift - n_points + 1
        k_start = max(0, k)
        k_end = min(n_points - 1, k + n_points - 1)

        # Compute the gradient
        grad = np.zeros_like(h)
        # The gradient is (1 - h[x+k]) if indices are valid
        grad_k = np.zeros_like(h)
        indices = np.arange(n_points)
        valid_k_indices = ((indices + k) >= 0) & ((indices + k) < n_points)
        grad_k[valid_k_indices] = 1 - h[indices[valid_k_indices] + k]
        grad = grad_k

        # Perform line search with backtracking
        current_alpha = alpha
        for backtrack in range(max_backtrack):
            proposed_h = h - current_alpha * grad

            # Projection: clip to [0, 1]
            proposed_h = np.clip(proposed_h, 0, 1)

            # Renormalize to have the right integral
            current_sum = np.sum(proposed_h)
            if current_sum != 0:
                proposed_h *= (target_sum / current_sum)

            # Compute updated C5
            new_c5 = evaluate_erdos_solution(proposed_h)

            if new_c5 < best_c5:
                h = proposed_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break
            else:
                # Reduce step size
                current_alpha *= beta

        step += 1

        # Early stopping if we diverge or cannot improve
        if backtrack == max_backtrack - 1:
            break

    return best_h, best_c5, n_points
