import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = h.size
    target_sum = n_points / 2

    # Normalize to maintain mean constraint
    h_sum = np.sum(h)
    h = h * (target_sum / h_sum)

    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    best_time = 0
    alpha = 0.02
    patience = 0
    max_iterations = 1000
    fft_pad = 2 * n_points

    def compute_overlap(h):
        h_full = np.pad(h, (0, fft_pad - n_points), mode='constant')
        one_h = 1 - h_full
        ccorr = np.correlate(h_full, one_h, mode='full')
        c5_values = ccorr * dx
        c5 = np.max(c5_values[-fft_pad + 1:fft_pad])
        return c5, c5_values

    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1.0:
            break

        # Compute current overlap measure
        c5, ccorr = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_time = time.time() - start_time
            patience = 0
            print(f"Iteration {i}: found new best C₅ = {best_c5:.8f}")
        else:
            patience += 1

        if patience > 500:
            alpha *= 0.5
            print(f"Reducing learning rate to {alpha}")

        # Find the lag with max C₅
        k = np.argmax(ccorr[-n_points + 1:n_points + 1]) - 1
        if k < 0:
            k += n_points

        # Compute gradient for this particular shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            forward = i + k
            if forward < n_points:
                grad[i] -= 1 - h[forward]
            backward = i - k
            if backward >= 0:
                grad[i] += h[backward]

        # Update h
        h = h - h * grad * alpha
        np.clip(h, 0, 1, out=h)
        h -= h.mean()  # Zero center
        h /= h.sum() * 2  # Normalize to have \int h = 1

        if i % 50 == 0 and i > 0:
            # Re-check solution
            h_sum = np.sum(h)
            assert abs(h_sum - target_sum) < 1e-6, "Constraint violated"
            assert np.all((h >= -1e-6) & (h <= 1 + 1e-6)), "Values out of bounds"

    # Final evaluation and constraints
    h_sum = np.sum(h)
    assert abs(h_sum - target_sum) < 1e-6
    h = np.clip(h, 0, 1)

    c5, _ = compute_overlap(h)
    best_h_sum = np.sum(best_h)
    assert abs(best_h_sum - target_sum) < 1e-6
    best_h = np.clip(best_h, 0, 1)

    best_c5, _ = compute_overlap(best_h)

    return (best_h, best_c5, n_points)
