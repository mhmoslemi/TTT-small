import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 95 if 'initial_h_values' in locals() else 95
    dx = 2.0 / n_points
    target_sum = n_points / 2

    try:
        h = initial_h_values.copy().astype(float)
    except Exception:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.mean() * (target_sum)

    max_iter = 500
    tolerance = 1e-4
    reg_eps = 1e-6

    start = time.time()
    for it in range(max_iter):
        overlap = np.correlate(h, 1-h, mode='full') * dx
        lag = np.arange(n_points, 2*n_points)
        overlap = overlap[n_points:2*n_points]
        current_c5 = np.max(overlap)
        if it > 0 and abs(prev_c5 - current_c5) < tolerance:
            break
        prev_c5 = current_c5

        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(cp.maximum(overlap, cp.Constant(0)), p='inf'))

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ]

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)

        if h_var.value is not None:
            new_h = np.clip(h_var.value, 0, 1)
            new_h = new_h / new_h.mean() * (target_sum)
            if 1 <= np.sum(new_h) <= target_sum:
                if time.time() - start > budget_s * 0.9:
                    break
                elif current_c5 > np.max(np.correlate(new_h, 1-new_h, mode='full') * dx):
                    h = new_h
        else:
            break

    overlap = np.correlate(h, 1-h, mode='full') * dx
    lag = np.arange(n_points, 2*n_points)
    final_c5 = np.max(overlap[n_points:2*n_points])

    return h, final_c5, n_points
