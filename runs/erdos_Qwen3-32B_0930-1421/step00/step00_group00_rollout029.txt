import numpy as np
import scipy.signal
import cvxpy as cp
import math
import time

def evaluate_erdos_c5(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = scipy.signal.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n_points / 2.0) > 1e-6:
        h *= (n_points / 2.0) / s
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = project_h(h, n_points)

    best_h = h.copy()
    best_c5 = evaluate_erdos_c5(h, n_points)
    dx = 2.0 / n_points

    iters = 0
    max_iters = 1000
    T = 1100
    best_time = 0

    while time.time() - start_time < max_time:
        iters += 1
        h = project_h(h, n_points)
        # Evaluate overlap for all k
        h_vals = h.copy()
        one_minus_h = 1 - h_vals
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k < n_points:
                if k >= len(h_vals):
                    raise ValueError("Index out of range")
                shifted = np.roll(one_minus_h, shift=k, axis=0)
                valid_overlap = np.sum(h_vals * shifted[n_points-1:])
            else:
                shifted = np.roll(one_minus_h, shift=k, axis=0)
                valid_overlap = np.sum(h_vals * shifted[:n_points])
            overlaps[k] = valid_overlap * dx

        k_max = np.argmax(overlaps)
        k_min = np.argmin(overlaps)

        # Construct shifted version for correlation at k_max
        corr_target = 1 - np.roll(h, shift=k_max - (n_points - 1)) if k_max < n_points else 1 - np.roll(h, shift=k_max - (n_points - 1))
        corr_target = corr_target[:n_points]

        # Define optimization variables
        h_opt = cp.Variable(n_points)
        overlap = cp.conv(h_opt, 1 - np.roll(h_opt, shift=k_max - (n_points - 1)), mode='full')
        constraint = overlap[k_max] * dx <= best_c5 + 1e-5

        # CVXPy formulation
        constraints = [
            0 <= h_opt,
            h_opt <= 1,
            cp.sum(h_opt) - (n_points / 2.0) <= 1e-5,
            cp.sum(h_opt) - (n_points / 2.0) >= -1e-5,
            constraint
        ]
        objective = cp.Minimize(cp.sum(overlap) * dx)

        # Solve
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False)

        if h_opt.value is not None:
            h_proj = project_h(h_opt.value, n_points)
            c5_new = evaluate_erdos_c5(h_proj, n_points)
            if c5_new < best_c5 - 1e-6:
                best_h = h_proj.copy()
                best_c5 = c5_new
                print(f"New best C5: {best_c5} at iter {iters}, time {time.time()-start_time:.2f}s")

        # Use gradient descent step if feasible
        h = project_h(h - 0.01 * (np.roll(h, shift=k_max) * (1 - h)), n_points)
        elapsed = time.time() - start_time
        best_time = elapsed

    h = project_h(h, n_points)
    h = best_h if np.sum(h) != n_points / 2.0 else h
    c5_final = evaluate_erdos_c5(h, n_points)

    return (list(h), c5_final, n_points)
