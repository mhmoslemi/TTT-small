import numpy as np
import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 84
    dx = 2.0 / n_points
    total_mass = n_points / 2.0

    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:int(n_points / 2)] = 1
        h = h * (total_mass / np.sum(h))  # Normalize

    def evaluate_c5(h):
        # Compute 1 - h
        h1 = 1.0 - h
        # Compute all shifts of h1
        h1_padded = np.column_stack((h1, h1, h1))
        # Compute overlap at all k: np.correlate
        corr = np.correlate(h, h1, mode='full') * dx
        return np.max(corr)

    # Ensure initial h meets the sum constraint
    h = h / np.sum(h) * total_mass

    # Optimization loop
    alpha = 0.01
    max_iter = 10000
    patience = 250
    noimprove = 0
    best_c5 = np.inf
    best_h = h.copy()

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute 1 - h
        h1 = 1.0 - h

        # Compute correlations
        corr = np.correlate(h, h1, mode='full') * dx

        # Find the k that maximizes the overlap (zero-lag to n_points-1)
        # Since h is over [0,2], the overlaps of h and 1-h for k from 0 to n_points-1 matter
        max_k = np.argmax(corr[: n_points])

        # Compute the shifted h1
        h_shifted = np.roll(h1, max_k)

        # Gradient is proportional to h_shifted
        grad = h_shifted - h  # Negative gradient direction
        grad = grad / np.max(np.abs(grad))  # Normalize

        # Line search step
        h_new = h + alpha * grad

        # Projection: Ensure h_new ∈ [0,1]
        h_new = np.clip(h_new, 0, 1)

        # Normalize to maintain total_mass
        h_new = h_new / np.sum(h_new) * total_mass

        # Check improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            noimprove = 0
        else:
            noimprove += 1

        # Annealing schedule
        if noimprove >= patience:
            alpha *= 0.95
            noimprove = 0

        h = h_new.copy()

    return (best_h, best_c5, n_points)
