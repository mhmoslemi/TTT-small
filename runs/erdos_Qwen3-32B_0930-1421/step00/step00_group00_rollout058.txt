import time
import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_current = np.array(initial_h_values, dtype=float)
    n_points = len(h_current)
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        shifts = np.arange(-n_points + 1, n_points)
        correlations = correlate(h, 1 - h, mode='full')
        overlaps = correlations * dx
        return np.max(overlaps)

    best_h = h_current.copy()
    best_c5 = evaluate_c5(best_h)
    alpha = 0.1
    beta = 0.7

    start_time = time.time()
    iteration = 0
    max_iterations = 5000

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        # Evaluate correlations and find worst shift
        correlations = correlate(h_current, 1 - h_current, mode='full')
        overlaps = correlations * dx
        k_max = np.argmax(overlaps)
        k_offset = k_max - (n_points - 1)
        h_lag = np.roll(h_current, -k_offset)  # h(x - k)
        grad = (1 - h_lag) * dx
        h_proposed = h_current - alpha * grad

        # Project to [0, 1]
        h_proposed = np.clip(h_proposed, 0, 1)
        h_sum = np.sum(h_proposed)
        target_sum = n_points / 2

        if abs(h_sum - target_sum) > 1e-6:
            # Adjust to sum to target
            indices = h_proposed > 0
            n_valid = np.sum(indices)
            if n_valid > 0:
                scale = (target_sum - np.sum(h_proposed[~indices])) / n_valid
            else:
                scale = 1.0
            h_proposed[indices] += scale

        # Integrity check after projection
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed = h_proposed * (target_sum / np.sum(h_proposed))

        # Evaluate new candidate
        c5_proposed = evaluate_c5(h_proposed)
        if not np.isfinite(c5_proposed):
            # Invalid candidate, try smaller step
            alpha *= beta
            continue

        if c5_proposed < best_c5:
            best_h = h_proposed.copy()
            best_c5 = c5_proposed
            alpha = min(0.1, 2 * alpha)  # Increase step
        else:
            alpha *= beta  # Reduce step size

        # Update current for next iteration
        h_current = best_h.copy()

    return best_h, best_c5, n_points
