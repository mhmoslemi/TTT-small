import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' not in kwargs or kwargs['initial_h_values'] is None:
        # No parent construction: use a uniform baseline
        n_points = 80
        h = np.ones(n_points) / 2
    else:
        h = kwargs['initial_h_values'].copy()
    n_points = len(h)

    dx = 2.0 / n_points
    h = h / h.sum() * (n_points / 2.0)  # Normalize to constraint
    assert np.all(0 <= h) and np.all(h <= 1), "h must stay between 0 and 1"
    assert abs(h.sum() - n_points / 2.0) < 1e-6, "Integral constraint violated"

    # Learning parameters
    alpha = 0.01
    max_iter = 5000
    patience = 200
    decay_factor = 0.5
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute overlap at all lags
        corr = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        lag_idx = np.argmax(corr)

        # Compute gradient: derivative of max correlation w.r.t h
        def grad_overlap(h, lag_idx):
            n = len(h)
            grad = np.zeros(n)
            for i in range(n):
                if 0 <= i - lag_idx < n:
                    grad[i] += - (1 - 2 * h[i]) * h[i - lag_idx]
                if 0 <= i + lag_idx < n:
                    grad[i] += (1 - 2 * h[i]) * h[i + lag_idx]
            return grad

        grad = grad_overlap(h, lag_idx)

        # Gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / h_new.sum() * (n_points / 2.0)

        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        elif it % patience == 0:
            alpha *= decay_factor  # Reduce learning rate

        h = h_new

    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / best_h.sum() * (n_points / 2.0)
    c5_final = evaluate_erdos_solution(best_h)
    return best_h.tolist(), c5_final, n_points
