import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    target_sum = n_points / 2

    # Normalize the initial solution
    h_init = np.clip((initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values)), 0, 1)
    h_init *= (target_sum / h_init.sum())

    def evaluate_c5(h):
        dx = 2.0 / n_points
        h = h * dx
        max_corr = 0.0
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr

    def generate_candidate():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= (target_sum / h.sum())
        return h

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h *= (target_sum / h.sum())
        return h

    def crossover(h1, h2):
        s = np.random.randint(n_points // 4, 3 * n_points // 4)
        h_child = np.concatenate([h1[:s], h2[s:]])
        return normalize_h(h_child)

    def mutate(h, mutation_rate=0.02):
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] = np.random.rand(sum(mask))
        return normalize_h(h)

    population_size = 50
    generations = 1000
    population = [h_init]
    for _ in range(population_size - 1):
        population.append(generate_candidate())

    best_h = h_init
    best_c5 = evaluate_c5(best_h)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
        population = sorted(population, key=lambda h: evaluate_c5(h))
        new_population = population[:20]  # top 20% selected

        for _ in range(population_size - 20):
            parent1, parent2 = np.random.choice(new_population[:10], size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)

        population = new_population

        for h in new_population:
            c5_val = evaluate_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h

    c5_bound = evaluate_c5(best_h)
    h_values = best_h
    return h_values, c5_bound, n_points
