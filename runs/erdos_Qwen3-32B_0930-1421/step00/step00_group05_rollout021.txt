import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def compute_c5(h, dx):
    h_array = np.array(h)
    one_minus_h = 1.0 - h_array
    corr = correlate(h_array, one_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5
