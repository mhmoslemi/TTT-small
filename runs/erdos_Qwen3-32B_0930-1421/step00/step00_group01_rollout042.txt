import numpy as np
import scipy.optimize as opt
import time
from math import floor

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.asarray(h)
    total = np.sum(h)
    assert abs(total * dx - 1.0) < 1e-6, f"Constraint violated: sum(h)*dx = {total*dx}, not 1"
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
