import numpy as np
import cvxpy as cp
import math
import time
import evaluate_erdos_solution
import sys

from typing import Optional, Tuple

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Parse initial_h_values and setup n_points
    initial_h = np.around(np.array(initial_h_values, dtype=float), 8)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Construct basis functions: M triangular basis equally spaced over [0, 2]
    M = int(n_points * 1.2)  # Slightly overcomplete to improve representability
    basis_centers = np.linspace(0 + dx, 2 - dx, M)
    basis_width = 1.2 * dx
    def phi(j, i):
        x = i * dx
        center = basis_centers[j]
        return max(0.0, 1 - math.fabs(x - center) / basis_width)

    Phi = np.zeros((M, n_points))
    for j in range(M):
        for i in range(n_points):
            Phi[j, i] = phi(j, i)

    # Least squares projection of initial h onto basis
    h_flat = initial_h.ravel()
    w_init = np.linalg.lstsq(Phi, h_flat, rcond=None)[0]
    w_init /= np.sum(w_init)  # Normalize
    w_init = np.clip(w_init, 0, 1)
    w_init = w_init / np.sum(w_init)

    # Sample from top basis functions to form initial active set
    active_basis = np.argsort(w_init)[::-1][:3 * n_points // 10]
    z_init = np.zeros(M)
    z_init[active_basis] = 1

    # CVXPY problem formulation
    w = cp.Variable(M, nonneg=True)
    z = cp.Variable(M, boolean=True)
    t = cp.Variable()

    # Define function h(x) as w' * Phi
    h = w @ Phi

    # Define the max overlap objective as t >= convolution for all k
    overlap_constraints = []
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            over = cp.multiply(h, 1 - h[k:])
        else:
            over = cp.multiply(h[-k:], 1 - h[:-k])
        overlap_constraints.append(t >= cp.sum(over * dx))

    C = 1
    # Integral constraint: ∫ h(x) dx = 1
    integral = cp.sum(h) * dx
    overlap_constraints.append(integral == 1)

    # Normalize w
    normalize = cp.sum(w) == 1

    # Dictionary linking w and z through z >= w for sparsity
    sparsity = [w[j] <= z[j] for j in range(M)]

    constraints = overlap_constraints + [normalize] + sparsity

    # Create MICP problem to minimize t
    obj = cp.Minimize(t)

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GUROBI, verbose=False, timeLimit=250, IterationLimit=500, MIPGap=1e-4)

    # Extract solution
    h_values = w.value @ Phi
    h_values = np.clip(h_values, 0, 1)

    # Final checks and evaluation
    def compute_C5():
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx

    c5_bound = compute_C5()
    elapsed_time = time.time() - start_time

    # Final fallback: use initial_h if optimized objective is worse
    if not prob.status in ['optimal', 'optimal_inaccurate']:
        h_values = initial_h
        c5_bound = compute_C5()

    print(f"Finished in {elapsed_time:.2f}s with C5={c5_bound:.8f} for n={n_points}")

    return h_values, c5_bound, n_points
