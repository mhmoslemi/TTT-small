import numpy as np
import time
from numpy import correlate
from scipy.optimize import minimize

def project(h, target_sum):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Scale to match sum constraint
    current_sum = h_clipped.sum()
    if abs(current_sum) < 1e-8:
        return np.zeros_like(h)
    scale_factor = target_sum / current_sum
    h_scaled = h_clipped * scale_factor
    return h_scaled

def compute_c5(h, dx):
    h_padded = np.concatenate([h, h])
    one_minus_h_padded = 1 - h_padded
    corr = correlate(h_padded, one_minus_h_padded, mode='same')
    return (corr[2 * len(h):] * dx).max()

def gradient_desc_step(h, k_index, dx, n_points):
    grad = np.zeros_like(h)
    for i in range(n_points):
        if 0 <= i - k_index + 2 * n_points // n_points < n_points:
            # valid index for overlap
            grad[i] = 1.0 - h[i - k_index + 2 * n_points // n_points]
    return -grad * dx
