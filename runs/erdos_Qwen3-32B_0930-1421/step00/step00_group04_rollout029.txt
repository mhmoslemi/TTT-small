import numpy as np
import time
import scipy.optimize as opt
from cvxpy import Variable, Minimize, Problem, sum as cvxsum, MaxElementwise

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    flat_overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(flat_overlap) * dx  # C5 estimate


def run(seed=42, budget_s=1100, initial_simplex_size=0.03, scale_simplex=True):
    np.random.seed(seed)
    start = time.time()

    if "initial_h_values" in globals():
        h0 = initial_h_values.copy()
    else:
        n_points = 24
        h0 = np.zeros(n_points)
        h0[n_points // 4 : 3 * n_points // 4] = 1.0 * n_points / 4
        h0 = h0 / np.sum(h0) * n_points / 2

    # Adjust n_points to a clean and scalable value
    target_n_points = 84
    h0 = np.interp(np.linspace(0, 1, target_n_points), np.linspace(0, 1, len(h0)), h0)
    h0 = h0 / np.sum(h0) * target_n_points / 2
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, len(h))

    # Optimization parameters
    max_iter = int(budget_s * 30)
    simplex_per = 50  # Run a simplex-based restart every 50 iterations
    restart_step = 0
    learning_rate = 0.01
    simplex_center = h.copy()

    def objective(x):
        x = x.reshape(-1)
        x = np.clip(x, 0, 1)
        dx = 2.0 / len(x)
        x_sum = x.sum()
        if np.abs(x_sum * dx - 1) > 1e-6:
            x = x / x_sum * len(x) / 2
        return evaluate_erdos_solution(x, len(x))

    def constraint_proj(x):
        x = x.reshape(-1)
        if np.abs(np.sum(x) * 2.0 / len(x) - 1) > 1e-6:
            x = x / np.sum(x) * len(x) / 2
        return np.clip(x, 0, 1).flatten()

    for iter in range(max_iter):
        if time.time() - start >= budget_s:
            break

        # Inner gradient descent
        gradient = np.gradient(objective(h - 0.01 * np.eye(len(h)).flatten()))
        grad_step = gradient.reshape(-1)
        h = h - learning_rate * grad_step
        h = constraint_proj(h)
        c5_current = evaluate_erdos_solution(h, len(h))

        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current

        # Simplex restart every `simplex_per` steps
        if iter % simplex_per == 0:
            simplex = [simplex_center + alpha * 0.1 for alpha in np.random.randn(10, len(h))]
            simplex = [constraint_proj(h) for h in simplex]
            c5_simplex = [evaluate_erdos_solution(h, len(h)) for h in simplex]
            best_idx = np.argmin(c5_simplex)
            if c5_simplex[best_idx] < best_c5:
                best_h = simplex[best_idx].copy()
                best_c5 = c5_simplex[best_idx]
            simplex_center = simplex[best_idx].copy()

        restart_step += 1
        if restart_step > 100:
            learning_rate *= 0.9

    return (best_h, best_c5, len(best_h))
