def interval_to_array(intervals: List[Tuple[float, float]], n_points: int, domain):
    dx = (domain[1] - domain[0]) / n_points
    h = np.zeros(n_points)
    for a, b in intervals:
        i0 = max(0, int(np.floor((a - domain[0]) / dx)))
        i1 = min(n_points, int(np.ceil((b - domain[0]) / dx)))
        h[i0:i1] += 1.0
    return h

def intervals_from_array(h, domain):
    n_points = len(h)
    dx = (domain[1] - domain[0]) / n_points
    intervals = []
    in_interval = False
    start_idx = 0
    for i in range(n_points):
        if h[i] > 0.5 and not in_interval:
            in_interval = True
            start_idx = i
        elif h[i] < 0.5 and in_interval:
            in_interval = False
            b = domain[0] + (i - 1) * dx
            intervals.append( (domain[0] + start_idx * dx, b) )
    if in_interval:
        intervals.append( (domain[0] + start_idx * dx, domain[1]) )
    return intervals

@dataclass
class Candidate:
    intervals: list[Tuple[float, float]]
    h: np.ndarray
    c5: float

    def is_valid(self, domain):
        # Check intervals are non-overlapping and contained within domain
        if len(self.intervals) == 0:
            return False
        intervals = sorted(self.intervals)
        total_length = sum(b - a for a, b in intervals)
        if abs(total_length - 1) > 1e-6:
            return False
        if intervals[0][0] < domain[0] - 1e-6 or intervals[-1][1] > domain[1] + 1e-6:
            return False
        for i in range(len(intervals) - 1):
            if intervals[i][1] > intervals[i + 1][0] + 1e-6:
                return False
        return True

def random_initial_intervals(n_intervals, domain=(0, 2)):
    dx = 1.0 / (n_intervals * 2)
    intervals = []
    for i in range(n_intervals):
        a = domain[0] + i * dx
        b = a + dx
        intervals.append((a, b))
    return intervals

def perturb_intervals(intervals: List[Tuple[float, float]], domain=(0, 2), p_remove=0.1, max_shift=0.1):
    intervals = sorted(intervals)
    n = len(intervals)
    if random.random() < p_remove and n > 1:
        idx = random.randint(0, n - 1)
        intervals.pop(idx)
    else:
        idx = random.randint(0, n - 1)
        a, b = intervals[idx]
        shift = random.uniform(-max_shift, max_shift)
        new_a = min(max(a + shift, domain[0]), b - 1e-6)
        new_b = min(max(b + shift, new_a + 1e-6), domain[1])
        intervals[idx] = (new_a, new_b)
    return intervals

def try_merge(intervals: List[Tuple[float, float]]):
    intervals = sorted(intervals)
    merged = []
    for a, b in intervals:
        if not merged:
            merged.append([a, b])
        else:
            last_a, last_b = merged[-1]
            if a <= last_b + 1e-6:  # almost over or touching
                merged[-1][1] = max(last_b, b)
            else:
                merged.append([a, b])
    return [(a, b) for a, b in merged]

def explore_configurations(initial_candidate, domain, domain_discretized, n_points, eval_func, time_budget, temperature=1.0, cooling_rate=0.99):
    best_candidate = initial_candidate
    t_start = time.time()
    no_improvement_counter = 0
    while time.time() - t_start < time_budget:
        intervals = best_candidate.intervals.copy()
        if random.random() < 0.5:
            intervals = perturb_intervals(intervals, domain, p_remove=0.1 if len(intervals) > 2 else 0.0, max_shift=0.05)
        else:
            intervals = try_merge(intervals)
        h = interval_to_array(intervals, n_points, domain_discretized)
        if abs(np.sum(h) / n_points - 1) > 1e-5:
            continue  # constraint violation
        h[np.where(h > 0.99)] = 1.0
        h[np.where(h < 1e-6)] = 0.0
        current_c5 = eval_func(h, domain_discretized, n_points)
        if current_c5 < best_candidate.c5:
            best_candidate = Candidate(intervals, h, current_c5)
            no_improvement_counter = 0
        elif time_budget - (time.time() - t_start) < 5:
            break  # advance to final result prep
        else:
            no_improvement_counter += 1
        if no_improvement_counter > 500:
            break
        temperature *= cooling_rate
    return best_candidate
