import numpy as np

from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Hyperparameters
    max_iter = 800
    lr = 0.01
    lr_decay = 0.992
    tol = 1e-5
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)

    h = initial_h.copy()
    best_c5 = compute_c5(h, dx)

    for i in range(max_iter):
        if time() - start_time > budget_s:
            break
        correl = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(correl[n_points:])  # only consider lags where i + k is valid
        k = idx
        # Compute gradient
        grad = np.zeros_like(h)
        for i_h in range(n_points):
            grad[i_h] = (1 - h[(i_h + k) % n_points]) - h[(i_h - k) % n_points]
        # Gradient step
        h = h - lr * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        # Enforce the integral constraint
        target_sum = n_points / 2
        scale_factor = target_sum / (h.sum())
        h *= scale_factor
        h = np.clip(h, 0, 1)
        # Evaluate new C₅
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        lr *= lr_decay

    h = best_h
    c5_bound = best_c5
    return (h, c5_bound, n_points)

def compute_c5(h, dx):
    correl = np.correlate(h, 1 - h, mode="full") * dx
    c5 = np.max(correl)
    return c5
