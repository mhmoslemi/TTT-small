import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        # Default feasible initialization if no parent construction is available
        n_points = 95
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.mean(h) * (n_points / 2)

    max_iter = 500
    epsilon = 1e-6  # for numerical stability
    best_h = h.copy()
    best_c5 = float("inf")

    for iter in range(max_iter):
        # Compute the overlap values for all lags k
        h_np = np.array(h).ravel()
        correlate_array = np.correlate(h_np, 1 - h_np, mode="full")
        dx = 2.0 / n_points
        corr_values = correlate_array * dx

        # Update the dual variables (nu) as the max between current nu and computed corr
        max_corr = np.max(corr_values)

        # Formulate the convex surrogate objective
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0,
        ]
        proxy_objective = cp.Minimize(cp.sum(cp.log(max(1e-8, max_corr) + epsilon)))
        prob = cp.Problem(proxy_objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False)
        except Exception:
            # Fallback in case of solver failure
            h = best_h.copy()
            continue

        h = np.array(h_var.value).ravel()
        h = np.clip(h, 0, 1)
        h = h / np.mean(h) * (n_points / 2.0)

        # Evaluate C₅ using np.correlate with mode='full'
        h_np = np.array(h).ravel()
        correlate_array = np.correlate(h_np, 1 - h_np, mode="full")
        corr_values = correlate_array * dx
        c5 = np.max(corr_values)

        # Check convergence and budget
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time >= budget_s:
            break

    # Final validation
    dx = 2.0 / n_points
    h = best_h.copy()
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h / np.mean(h) * (n_points / 2.0)
    assert np.all((0 <= h) & (h <= 1)), "Invalid value out of [0, 1] interval"
    assert abs(np.sum(h) - n_points / 2.0) < 1e-6, "Invalid total sum constraint"

    correlate_array = np.correlate(h, 1 - h, mode="full")
    corr_values = correlate_array * dx
    c5 = np.max(corr_values)

    return (h.tolist(), c5, n_points)
