    constraints = [
        cp.sum(w) == 1.0,
        cp.norm1(w) <= len(w)  # Promote sparsity (optional)
    ]

    # Enforce integral constraint
    integral = cp.sum(B @ w * dx * 2.0) == 1.0
    constraints.append(integral)

    # Build convolutions using square grids
    overlap_kernel = np.outer(B, 1 - B[:, None])
    lags = np.arange(n_points - 1)
    total_overlaps = [cp.sum(overlap_kernel[i, :] * w * dx) for i in lags]
    objective = cp.Minimize(cp.max(total_overlaps))

    # Setup problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GUROBI, TMlimit=600)
    h_values = np.dot(B.T, w.value)

    # Clamp h to [0, 1] and maintain integral constraint
    h_values = np.clip(h_values, 0, 1)
    scale = n_points / 2.0 / h_values.sum() if h_values.sum() != 0 else 1.0
    h_values = h_values * scale

    # Final evaluation
    def evaluate(h):
        h_padded = np.pad(h, (0, len(h) - 1), mode='constant', constant_values=0)
        corr = np.correlate(h, h_padded, mode='full')
        mask = np.triu_indices(len(h) * 2 - 1, 1)
        overlaps = corr[mask]
        return np.max(overlaps) * dx

    c5_bound = evaluate(h_values)

    return h_values.tolist(), c5_bound, n_points
