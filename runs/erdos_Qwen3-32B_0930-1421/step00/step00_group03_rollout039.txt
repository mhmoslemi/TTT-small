import time

    best_c5 = evaluate_erdos_solution(h)

    T = 1.0
    alpha = 0.999
    sigma = 0.1

    gd_steps = 500
    decay_rate = 0.9
    patience = 50
    no_improve = 0

    for step in range(sa_steps):
        if time.time() - start_time > budget_s - 10:
            break

        # Perturb
        h_p = h + sigma * np.random.randn(n_points)
        h_p = np.clip(h_p, 0, 1)

        # Project to sum constraint
        s = h_p.sum()
        if s != n_points / 2:
            h_p -= np.sort(np.argsort(h_p)[::-1])[:(100 if s < n_points / 2 else -100)]
            scale = (n_points / 2) / h_p.sum()
            h_p *= scale
            h_p = np.clip(h_p, 0, 1)

        # Evaluate
        c5 = evaluate_erdos_solution(h_p)

        # Acceptance
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / T):
            h = h_p
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

        T *= alpha
        sigma *= decay_rate

        if no_improve >= patience:
            # Restart cooling
            T = 1.0
            no_improve = 0
            h = best_h.copy()

    # Run gradient descent
    h = best_h
    grad = np.zeros_like(h)
    learning_rate = 0.01
    learning_rate_decay = 0.95

    for step in range(gd_steps):
        if time.time() - start_time > budget_s - 10:
            break

        k_argmax = np.argmax(np.abs(correlate(h, 1 - h, mode="full")))

        else:

        grad = -np.zeros_like(h)

            h_p = h.copy()
            h_p[i] += eps
            h_p = np.clip(h_p, 0, 1)
            s = h_p.sum()
            if s != n_points / 2:

                h_p = np.clip(h_p, 0, 1)

            c5 = evaluate_erdos_solution(h)
            grad[i] = (c5_p - c5) / eps

        # Normalize and update
        grad = grad / np.max(np.abs(grad))
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Project to sum constraint
        s = h.sum()
        if s != n_points / 2:
            h -= np.sort(np.argsort(h)[::-1])[:(100 if s < n_points / 2 else -100)]
            scale = (n_points / 2) / h.sum()
            h *= scale
            h = np.clip(h, 0, 1)

        # Track best

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            learning_rate = max(learning_rate * learning_rate_decay, 1e-4)
        else:
            learning_rate = min(learning_rate * 2, 0.2)

    return (best_h.tolist(), best_c5, n_points)
