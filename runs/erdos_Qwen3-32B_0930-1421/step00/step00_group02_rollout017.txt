import numpy as np
import time
import cvxpy as cp
import evaluate_erdos_solution as evaluator

def objective_eval(h, dx):
    hcor = np.correlate(h, 1 - h, mode='full')
    return np.max(hcor * dx)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    opt_h = None
    best_c5 = float("inf")
    n_points = 45  # Use the provided construction

    t_start = time.time()

    h0 = initial_h_values[:]
    h0 = np.array(h0, dtype=np.float64)
    dx = 2.0 / n_points

    # Constraint: sum(h) * dx = 1
    assert abs(np.sum(h0) * dx - 1.0) < 1e-8

    best_c5 = objective_eval(h0, dx)
    opt_h = h0.copy()

    while time.time() - t_start < budget_s - 60:
        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)

        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            cp.sum(h[n_points // 2:] - h[:n_points // 2]) <= 1e3,
            h <= 1
        ]
        # Enforce non-negativity implicitly: cp.Variable(nonNegative=True)

        # Objective is synthetic: minimize upper bound on max overlap
        obj = 0
        for k in range(-n_points + 1, n_points):
            h_shift = np.roll(h0, shift=-k)
            if k < 0:  # Wraparound to match full mode behavior
                h_shift = np.roll(h0, shift=-k)

            overlap = cp.sum(h * (1 - h_shift))
            overlap *= dx
            obj += cp.norm(overlap, p=np.inf) ** 2

        problem = cp.Problem(cp.Minimize(obj), constraints)
        try:
            problem.solve(solver=cp.ECOS, max_iters=150)

            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.clip(h.value, 0, 1)

                if np.allclose(np.sum(h_new) * dx, 1.0, atol=1e-5):
                    c5_new = objective_eval(h_new, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        opt_h = h_new.copy()
                        print(f"Iteration - improved to {best_c5:.6f}")

                # Feasible perturbation
                h0 = np.clip(h_new + 0.05 * np.random.randn(*h_new.shape), 0.0, 1.0)
                h0 = (h0 * n_points) / np.sum(h0)
            else:
                # Use stable fallback
                h0 = opt_h * (1 + 0.01 * np.random.randn(n_points))
                h0 = np.clip(h0, 0, 1)
                h0 /= np.sum(h0) / n_points / 2  # Rescale for energy
        except:
            # Use fallback
            h0 = opt_h * (1 + 0.01 * np.random.randn(n_points))
            h0 = np.clip(h0, 0, 1)
            h0 /= np.sum(h0) / n_points / 2  # Rescale for energy

            if time.time() - t_start < budget_s:
                continue
            else:
                break

    assert opt_h is not None
    opt_h = np.clip(opt_h, 0, 1)
    c5_bound = objective_eval(opt_h, dx)
    opt_h = np.clip(opt_h, 0, 1)

    return opt_h, c5_bound, n_points
