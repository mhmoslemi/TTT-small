import numpy as np
import time
from numpy import correlate
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    h_history = [best_h.copy()]
    best_c5 = evaluate_erdos_solution(h)
    
    max_iter = 10000
    iter_count = 0
    alpha = 1.0
    beta = 0.5
    gamma = 0.9
    max_line_search = 10
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1

        # Compute the cross-correlation using the current h
        def compute_c5(h_vals):
            corr = correlate(h_vals, 1 - h_vals, mode='full') * dx
            return np.max(corr)

        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()
            h_history.append(h.copy())
        
        # Find the shift k* that causes the max overlap
        corr = correlate(h, 1 - h, mode='full') * dx
        k_star = np.argmax(corr)
        
        # Compute gradient
        # Create shifted (1 - h) array for k* and compute exact gradient via correlation
        def get_gradient(h_vals, k):
            n = len(h_vals)
            shifted = np.zeros(n)
            if 0 <= k < n:
                shifted[k:] = 1 - h_vals[:-k]
            elif -n < k < 0:
                shifted[:n+k] = 1 - h_vals[-k:]
            # Compute gradient as upsampled shifted (1-h) vector
            grad = np.zeros(n)
            for i in range(n):
                j = i + k
                if 0 <= j < n:
                    grad[i] = 1 - h_vals[j]
            return grad

        grad = get_gradient(h, k_star)

        # Normalize gradient by dx
        grad *= dx

        # Line search (backtracking)
        for line_search_step in range(max_line_search):
            step_size = alpha * (gamma**line_search_step)

            # Step update
            h_new_unproj = h - step_size * grad

            # Projection onto box constraints [0, 1]
            h_new = np.clip(h_new_unproj, 0, 1)

            # Projection to maintain integral constraint
            current_sum = np.sum(h_new)
            if abs(current_sum - target_sum) > 1e-6:
                scale_factor = target_sum / current_sum
                h_new = np.clip(h_new * scale_factor, 0, 1)

            # Check if improving
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                h[:] = h_new.copy()
                break
            elif line_search_step == max_line_search - 1:
                # If no improvement, reset to best so far
                h[:] = best_h.copy()
        
    # Final cleanup: ensure sum is exactly target_sum and values are in [0,1]
    h_rounded = np.clip(h, 0, 1)
    current_sum = np.sum(h_rounded)
    if abs(current_sum - target_sum) > 1e-6:
        h_rounded = np.clip(h_rounded * (target_sum / current_sum), 0, 1)
    
    # Final evaluation using task evaluator
    c5_bound = evaluate_erdos_solution(h_rounded)

    return (h_rounded.tolist(), c5_bound, n_points)
