import numpy as np
import cvxpy as cp
import time
from math import isclose

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    tic = time.time()

    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    initial = initial_h_values.copy()
    h_init = initial

    # Compute starting bound using given h
    def compute_c5(h, n_points, dx):
        p = np.mean(h)
        h = h.clip(0, 1)
        correl = np.correlate(h, 1 - h, mode="full")
        correl = correl[:(n_points * 2 - 1)] * dx
        return float(np.max(correl)), float(np.mean(correl))

    # Feasibility problem for checking if the overlap constraint holds
    def is_feasible(target_c5):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2,
        ]

        # Compute overlap maxima for all feasible k
        max_overlap = None
        for k in range(-(n_points - 1), n_points):
            # Compute overlap for lag k
            if k < 0:
                idx1 = np.s_[: (n_points + k)]
                idx2 = np.s_[-k:]
            elif k > 0:
                idx1 = np.s_[: (n_points - k)]
                idx2 = np.s_[k:]
            else:
                idx1 = idx2 = np.s_[:]
            obj = cp.sum(h[idx1] * (1 - h[idx2])) * dx
            if max_overlap is None:
                max_overlap = obj
            else:
                max_overlap = cp.maximum(max_overlap, obj)

        # Form problem: minimize maximum overlap, must <= target_c5
        problem = cp.Problem(cp.Minimize(max_overlap))
        problem.solve(solver=cp.ECOS, verbose=False, max_iters=2000)
        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return None, None

        h_vals = np.clip(np.array(h.value).reshape((-1,)), 0, 1)
        return h_vals, max_overlap.value

    best_c5 = float("inf")
    best_h = h_init.copy()
    low, high = 0.38085, 0.4
    iters = 0
    try:
        while (high - low) > 1e-5 and (time.time() - tic) < budget_s * 0.9:
            mid = (low + high) / 2
            h_vals, overlap = is_feasible(mid)
            iters += 1

            if h_vals is None or overlap is None:
                high = mid
                continue

            # Compute true max overlap
            c5, _ = compute_c5(h_vals, n_points, dx)

            if c5 <= mid and h_vals is not None:
                best_c5 = c5
                best_h = h_vals
                high = mid
            else:
                low = mid

    except Exception as e:
        print("Exception during optimization:", e)

    print(f"Best bound found: C5 = {best_c5}, from {iters} iterations")

    # Final verification
    final_c5, _ = compute_c5(best_h, n_points, dx)
    print("Final validation C5 = ", final_c5)

    return best_h, final_c5, n_points
