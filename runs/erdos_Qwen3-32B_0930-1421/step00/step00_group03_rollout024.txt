import numpy as np
import time
import cvxpy as cp
import cvxpy.settings as s


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    np.random.seed(seed)
    
    # Set up
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    max_lags = n_points * 2 - 1
    corr = np.correlate(h_init, 1 - h_init, mode="full")
    initial_c5 = np.max(corr) * dx
    print(f"Initial C5: {initial_c5}")

    # Binary search bounds
    low, high = 0.37, 0.39  # lower than and higher than the record C₅
    tolerance = 1e-5
    best_h = h_init.copy()
    best_c5 = initial_c5

    def solve_feasibility(C5, h_prev=None):
        h = cp.Variable(shape=n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Get positive and negative indices of overlap
        overlaps = []
        for k in range(n_points):
            right = np.roll(h, -k)
            right[k:] = 0
            shift = h[0:n_points - k] * (1 - right[0:n_points - k])
            overlaps.append(cp.sum(shift) * dx)

        for k in range(1, n_points):
            left = np.roll(h, k)
            left[0:n_points - k] = 0
            shift = left[0:n_points - k] * (1 - h[k:n_points])
            overlaps.append(cp.sum(shift) * dx)

        for o in overlaps:
            constraints.append(o <= C5)

        problem = cp.Problem(cp.Minimize(0), constraints)
        problem.solve(solver='MOSEK', verbose=False, mosek_params={"MSK_DPAR_INTPNT_TOL_MU_RED": 1e-9})
        if problem.status not in [s.UNKNOWN, s.INFEASIBLE, s.UNBOUNDED]:
            return h.value, problem.status
        return None, problem.status

    iter = 0
    while (high - low) > tolerance and time.time() - start_time < budget_s - 3:
        iter += 1
        mid = (low + high) / 2
        h_sol, status = solve_feasibility(mid, best_h)
        if h_sol is not None:
            h_sol = np.clip(h_sol, 0, 1)
            # Recalculate actual max overlap
            corr = np.correlate(h_sol, 1 - h_sol, mode="full")
            new_c5 = np.max(corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_sol
            high = mid
            print(f"Iter {iter}: Feasible at C5={mid}, new C5={new_c5}")
        else:
            print(f"Iter {iter}: Infeasible at C5={mid}")
            low = mid

    # Final feasibility solve with best C5
    h_final, status = solve_feasibility(best_c5, best_h)
    if h_final is not None:
        h_values = np.round(h_final, 10)
    else:
        h_values = best_h

    return h_values.tolist(), best_c5, n_points
