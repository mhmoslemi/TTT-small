import numpy as np
import time

def run(seed=42, budget_s=1100, n_points=200):
    np.random.seed(seed)
    start_time = time.time()
    max_iterations = 10000
    lr = 0.05  # Initial learning rate
    patience = 50
    min_lr = 1e-4
    best_c5 = 1.0
    best_h = None

    dx = 2.0 / n_points
    target_sum = n_points // 2

    # Initialize h
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = np.zeros(n_points)
        offset = (n_points - len(initial_h_values)) // 2
        h[offset:offset + len(initial_h_values)] = initial_h_values
        h = h.clip(0, 1)
        h_sum = h.sum()
        factor = target_sum / h_sum if h_sum != 0 else 1.0
        h *= factor
        h = h.clip(0, 1)
    else:
        h = np.random.rand(n_points)
        h = h / h.sum() * target_sum
        h = h.clip(0, 1)

    def objective(h):
        h_arr = h.flatten()
        ones_minus_h = 1.0 - h_arr
        corr = np.correlate(h_arr, ones_minus_h, mode="full")
        return corr.max() * dx

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            h_clipped = np.zeros_like(h)
        else:
            h_clipped *= target_sum / h_sum
        return h_clipped

    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute current value
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_points
            print(f"Iter {iter}, new best C5 = {c5:.7f} at {elapsed:.1f} s")
        else:
            if iter > 0 and lr > min_lr and (iter % patience == 0):
                lr *= 0.75
                print(f"Iter {iter}, reducing LR to {lr:.4f}")

        # Compute gradient
        h_ext = np.concatenate([h, h])
        ones_minus_h = 1.0 - h_ext[n_points:]
        grad = dx * (h_ext[n_points:] - ones_minus_h)
        grad = grad[n_points:]

        # Update and project
        h -= lr * grad
        h = project(h)

    if best_h is None:
        best_h = h.copy()
        best_c5 = objective(best_h)
    return best_h.tolist(), best_c5, best_n
