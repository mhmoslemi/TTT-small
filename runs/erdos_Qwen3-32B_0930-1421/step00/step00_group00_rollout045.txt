import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points

    def is_feasible(h):
        return np.all((0 <= h) & (h <= 1)) and np.isclose(h.sum(), n_points / 2)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if is_feasible(h):
            current_c5 = evaluate_erdos_solution(h, n_points)
            best_h = h.copy()
            best_c5 = current_c5
        else:
            h = np.random.uniform(0, 1, n_points)
            h /= h.sum()
            h *= n_points / 2
    else:
        h = np.random.uniform(0, 1, n_points)
        h /= h.sum()
        h *= n_points / 2
        best_c5 = float("inf")
        best_h = h.copy()

    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = len(overlaps) // 2
        overlaps_kmax = overlaps[k_max]

        v = cp.Variable(n_points)
        constraint = cp.sum(v) == n_points / 2
        constraints = [0 <= v, v <= 1, constraint]
        objective = cp.Minimize(overlaps[k_max]) - cp.Minimize(cp.correlation(v, 1 - v)[k_max] * dx)

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.ECOS, time_limit=1.0)

        h = np.clip(v.value, 0, 1)

        if np.allclose(h, best_h):
            break

        h = h / h.sum() * n_points / 2
        h = np.clip(h, 0, 1)

        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        new_c5 = np.max(overlaps)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    overlaps = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(overlaps)
    return best_h.tolist(), final_c5, n_points
