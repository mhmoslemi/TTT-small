import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Caution: np.correlate is not circular.
    # Compute the overlap integrals for all lags manually.
    c = np.zeros(n_points - 1)
    for lag in range(1, n_points):
        # Only consider positions where x + k is in the domain [0, 2]
        # We will avoid wrapping back to the beginning
        if lag >= n_points:
            break
        indices = np.arange(n_points - lag)
        values = h[indices] * (1 - h[indices + lag])
        c[lag - 1] = np.sum(values) * dx
    return np.max(c)

    best_c5_complete = best_c5
    start_time = time.monotonic()
    max_decay_iterations = 500
    eps = 1e-6

    # Convex proxy objective constants
    nu = np.zeros(n_points - 1)

    for iteration in range(max_decay_iterations):
        elapsed = time.monotonic() - start_time
        if elapsed >= budget_s - 10:
            print(f"Time budget nearly exceeded; returning best h found so far.")
            break

        # Compute overlap integrals
        c = np.zeros(n_points - 1)
        for lag in range(1, n_points):
            if lag >= n_points:
                break
            indices = np.arange(n_points - lag)
            values = h[indices] * (1 - h[indices + lag])
            c[lag - 1] = np.sum(values) * dx

        # Update dual variables
        nu = np.maximum(nu, c)

        # Construct convex problem
        h_var = cp.Variable(n_points, nonneg=True)

        # Proxy objective to minimize an approximation of the max overlap
        proxy_obj = cp.Minimize(cp.sum(cp.log(nu + eps)))
        constraints = [
            h_var <= 1,
            h_var >= 0,
            cp.sum(h_var) == n_points / 2,
        ]

        problem = cp.Problem(proxy_obj, constraints)
        problem.solve(solver=cp.SCIPY, max_iter=100, verbose=False)

        if problem.status in ['infeasible', 'unbounded']:
            # fall back to existing h
            h = best_h.copy()
        else:
            h = h_var.value.ravel()
            h = np.clip(h, 0, 1)
            if np.sum(h) == 0:
                h = best_h.copy()
            else:
                h = h / h.mean() * (n_points / 2)

        # Check constraints and objective
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if iteration % 10 == 0:
            print(f"Iteration {iteration}: best C5 = {best_c5}")
            elapsed = time.monotonic() - start_time
            if elapsed >= budget_s:
                break

    # Final objective and constraint check
    return (best_h, best_c5, n_points)
