import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    max_points = 100
    start_time = time.time()
    
    h_init = initial_h_values
    n_init = len(h_init)
    n_points = max(n_init, 70)
    
    # Enforce sum(h) == n_points / 2 precisely
    def normalize_h(h):
        total = h.sum()
        return h * (n_points / 2) / total

    h = normalize_h(h_init)

    # Define problem as an optimization over binary variables (form a step function)
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        ones = np.ones_like(h)
        one_minus_h = ones - h
        corr = correlate(h, one_minus_h, mode="full")
        return np.max(corr * dx)

    C5 = evaluate_c5(h)

    # Use a simple convex optimization with Gurobi to minimize max overlap
    def construct_basis(n_basis):
        dx = 2.0 / n_basis
        x_centers = np.linspace(0, 2 - dx, n_basis)
        basis = np.zeros((n_basis, n_points))
        for i, x0 in enumerate(x_centers):
            idx = int(np.clip(x0 / (2.0 / n_points), 0, n_points - 1))
            basis[i, idx] = 1
        return basis.T

    # Fine-tune optimization with gridding
    def optimize_h_cd(h0, c5_prev):
        h = h0.copy()
        dx = 2.0 / len(h)
        best_c5 = c5_prev
        best_h = h.copy()
        iters = 0
        max_iter = 500
        while iters < max_iter and time.time() - start_time < budget_s:
            for k in range(1, len(h)):
                idx = k
                corr = h * (1 - np.roll(h, -k))
                overlap = np.sum(corr) * dx

                # Attempt to update one element to reduce target overlap
                for i in np.random.permutation(len(h)):
                    for dy in [-0.05, 0.05]:
                        new_h = h.copy()
                        new_h[i] = np.clip(new_h[i] + dy, 0, 1)
                        new_h = normalize_h(new_h)
                        new_c5 = evaluate_c5(new_h)

                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = new_h.copy()
                            break  # no need to explore further at this step
                    if best_c5 < best_c5:
                        h = best_h.copy()
                        break
                iters += 1
        return best_h, best_c5

    # Use the initial_h_values and optimize it further using coordinate descent
    h_opt, best_c5 = optimize_h_cd(h, C5)

    # Try basis function representation with local convex optimization
    try:
        if time.time() - start_time < budget_s:
            h_basis = construct_basis(100)
            w = cp.Variable(h_basis.shape[0])
            ones = cp.Constant(1.0)

            # Enforce sum to 1 constraint
            constraint_sum = [w @ h_basis.sum(1) == n_points / 2]

            # Ensure values in [0, 1]
            constraint_box = [0 <= w, w <= 1]

            # Construct constraint for overlap across overlaps
            all_conv = []
            for k in range(1, 3):
                overlap = h_basis[:, :len(h_basis) - k] * (ones - np.roll(h_basis, -k, axis=1)[:, :len(h_basis) - k])
                quad_contribution = w.T @ (overlap @ w * (2.0 / n_points))
                all_conv.append(quad_contribution)

            max_contribution = cp.max(all_conv)
            objective = cp.Minimize(max_contribution)

            opt_prob = cp.Problem(objective, constraint_sum + constraint_box)
            sol = opt_prob.solve(solver=cp.MOSEK, max_iters=1000, verbose=False)
            if opt_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_basis_sol = w.value @ h_basis
                h_candidate = normalize_h(h_basis_sol)

                candidate_c5 = evaluate_c5(h_candidate)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    h_opt = h_candidate
    except Exception:
        pass  # fallback to CD

    return h_opt, best_c5, len(h_opt)
