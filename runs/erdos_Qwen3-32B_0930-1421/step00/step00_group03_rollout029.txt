import numpy as np
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values
from scipy.signal import correlate
import cvxpy as cp
import time
import random


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    hnp = np.array(h, dtype=np.float64)
    one_minus_hnp = 1 - hnp
    c = correlate(hnp, one_minus_hnp, mode='full')
    C5 = np.max(c * dx)
    return C5


def constrain_h(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Enforce the integral constraint
    target_sum = n_points / 2
    current_sum = h.sum()
    if current_sum == target_sum:
        return h
    elif current_sum < target_sum:
        # Need to increase some values
        delta_sum = target_sum - current_sum
        # Only affect positions with room to increase
        pos = h < 1
        count = pos.sum()
        if count == 0:
            return h
        h[pos] += delta_sum / count
    else:
        # Need to decrease some values
        delta_sum = current_sum - target_sum
        # Only affect positions with room to decrease
        pos = h > 0
        count = pos.sum()
        if count == 0:
            return h
        h[pos] -= delta_sum / count
    return np.clip(h, 0, 1)


def init_particle_from_seed(h_init, scale):
    h = h_init.copy()
    perturbation = np.random.normal(0, scale, h.shape)
    h += perturbation
    return np.clip(constrain_h(h, h.shape[0]), 0, 1)


def run():
    start_time = time.time()
    max_time = 1100
    max_iter = 500
    n_particles = 50

    h_init = initial_h_values.copy()
    n_points = h_init.shape[0]

    # Population as a list of particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    for _ in range(n_particles):
        h_perturbed = init_particle_from_seed(h_init, 0.1)
        h_clone = h_perturbed.copy()
        v_perturbed = np.random.uniform(-0.05, 0.05, h_clone.shape)
        particles.append(h_clone)
        velocities.append(v_perturbed)
        personal_best.append(h_clone)
        c5 = compute_c5(h_clone, n_points)
        personal_best_scores.append(c5)

    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx]
    best_c5 = personal_best_scores[global_best_idx]
    best_global_c5 = (global_best, best_c5)

    omega = 0.7
    c1 = 1.5
    c2 = 1.5

    iter_no_improve = 0
    patience = 100

    for it in range(max_iter):
        # Check time budget
        if time.time() - start_time > max_time - 10:
            break

        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            v = omega * velocities[i] + \
                c1 * r1 * (personal_best[i] - particles[i]) + \
                c2 * r2 * (global_best - particles[i])
            velocities[i] = v
            particles[i] = particles[i] + v
            particles[i] = constrain_h(particles[i], n_points)

            c5 = compute_c5(particles[i], n_points)

            if c5 < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = c5

        # Update global best
        best_idx = np.argmin(personal_best_scores)
        if personal_best_scores[best_idx] < best_c5:
            global_best = personal_best[best_idx].copy()
            best_c5 = personal_best_scores[best_idx]
            best_global_c5 = (global_best, best_c5)
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        if iter_no_improve >= patience:
            break

    h_final, c5_final = best_global_c5
    return h_final.tolist(), c5_final, n_points
