```python
import numpy as np
import time
from scipy.signal import correlate

initial_h_values = None  # will be provided at runtime
target_C5 = 0.380876


def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    ones_minus_h = 1 - h
    correlations = correlate(h, ones_minus_h, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr


def ensure_feasible(h):
    # Enforce bounds and normalize to the constraint ∑h[i] = n_points / 2
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        h = np.zeros_like(h)
        h[0] = 1.0
    else:
        h *= (len(h) / 2) / total
    return h


def generate_initial_population(pop_size, n_points):
    # Start with uniform samples, optionally add the parent if available
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = ensure_feasible(h)
        population.append(h)
    if initial_h_values is not None:
        population.append(np.copy(initial_h_values))
    return population


def tournament_selection(population, scores, tournament_size=3):
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = min(idxs, key=lambda i: scores[i])
    return np.copy(population[best_idx])


def crossover(parent1, parent2):
    n_points = len(parent1)
    cross_idx = np.random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])
    return ensure_feasible(child)


def mutate(h, mut_rate=0.1):
    n_points = len(h)
    mask = np.random.rand(n_points) < mut_rate
    if not np.any(mask):
        return np.copy(h)
    h[mask] = np.random.rand(np.sum(mask))
    return ensure_feasible(h)


def run(budget_s=1100, pop_size=20, gens=np.inf, mut_rate=0.1, n_points=42):
    start_time = time.time()
    population = generate_initial_population(pop_size, n_points)
    best_h = None
    best_C5 = np.inf

    while time.time() - start_time < budget_s:
        # Evaluate the population
        scores = [evaluate_C5(p) for p in population]
        curr_best_idx = np.argmin(scores)
        curr_best_C5 = scores[curr_best_idx]
        if curr_best_C5 < best_C5:
            best_h = np.copy(population[curr_best_idx])
            best_C5 = curr_best_C5

        # Create the next generation
        new_population = []
        new_population.append(np.copy(best_h))  # elitism

        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, scores)
            parent2 = tournament_selection(population, scores)
            child = crossover(parent1, parent2)
            child = mutate(child, mut_rate)
            new_population.append(child)
            if len(new_population) >= pop_size:
                break

        population = new_population

    return best_h, best_C5, n_points
```
