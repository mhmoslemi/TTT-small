import numpy as np
import time
import cvxpy as cp

dx_cache = None
initial_h = None
n_points_cache = None


def evaluate_c5(h, dx):
    h_corr = np.correlate(h, 1 - h, mode="full")
    return np.max(h_corr * dx)


def project_to_constraints(h, dx):
    h_clipped = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h_clipped) * dx)
    h_projected = h_clipped * scale
    return h_projected


def compute_gradient(h, dx, k_max):
    n = len(h)
    grad = np.zeros_like(h)
    h_k = h[k_max:]
    grad[:n - k_max] -= (1 - 2 * h[:n - k_max]) * h_k
    grad[k_max:] -= (1 - 2 * h_k) * h[:n - k_max]
    grad *= dx
    return grad


def update_h(h, grad, lr):
    h_new = h - lr * grad
    h_new = project_to_constraints(h_new, dx_cache)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    global dx_cache, initial_h, n_points_cache
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h = np.copy(globals()['initial_h_values'])
    n_points_cache = len(h)
    dx_cache = 2.0 / n_points_cache
    best_h = np.copy(h)
    best_c5 = evaluate_c5(h, dx_cache)

    start_time = time.time()
    lr = 0.1
    max_iter = 20000
    iter_limit = max_iter
    for i in range(iter_limit):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx_cache)
        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5
        k_full = np.argmax(corr)
        k_max = k_full % n_points_cache
        grad = compute_gradient(h, dx_cache, k_max)
        h = update_h(h, grad, lr)
        if i % 50 == 0:
            print(f"[iter={i}] c5={c5:.6f} (best={best_c5:.6f})")
    return (best_h, best_c5, n_points_cache)
