import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Get starting vector from global
    n_points = len(initial_h_values)
    h = np.array(initial_h_values).astype(float).copy()
    dx = 2.0 / n_points
    h_sum = np.sum(h)

    # Constraint: h must integrate to 1 over [0,2]
    if abs(h_sum - n_points / 2) > 1e-9:
        h *= (n_points / 2) / h_sum

    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)
    np.fill_diagonal(np.take(np.take(h, idx_k0, axis=0, mode='wrap'), idx_k0, axis=1, mode='wrap'), 0)

    t0 = time.time()
    start_time = t0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    lr = 0.1
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        abs_overlaps = overlaps.copy()

        # Only compute max for valid overlaps k in range(0, n_points * 2 - 1)
        max_k = np.argmax(abs_overlaps)
        max_c = np.max(abs_overlaps)

        # compute gradient at k = max_k
        k = max_k

        idx_k0 = [i % n_points for i in range(n_points)]
        g = np.zeros_like(h)
        for i in range(n_points):
            g[i] = h[i] * (-h[(i + k) % n_points] + h[i] - 1.0)

        # Normalize gradient by n_points
        g /= n_points
        # Descend, but ensure that gradient is unambiguous
        dg = -g
        h += lr * dg
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)

        # Normalize again to satisfy constraints
        if h_sum > 0:
            h *= (n_points / 2) / h_sum

        cur_c5 = evaluate_erdos_solution(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5

    return (best_h, best_c5, n_points)

def idx_k0(n_points):
    return [i % n_points for i in range(n_points)]

def evaluate_erdos_solution(h):
    n_points = len(h)
    if n_points == 0:
        return float('inf')
    overlaps = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
    valid_offsets = [k for k in range(n_points * 2 - 1) if -n_points + 1 < k < n_points]
    overlaps_for_c5 = [overlaps[k] for k in valid_offsets]
    return max(overlaps_for_c5)
