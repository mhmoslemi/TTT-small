import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration parameters
    max_n_points = 100  # Cap to ensure small sequences

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, copy=True)
    else:
        n_points = 47
        # Basic uniform construction with one unit per 2 bins
        h_init = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init[i + i] = 2.0 / n_points
        h_init *= n_points / 2 / h_init.sum()  # Normalize to constraint

    dx = 2.0 / n_points

    # Constraint: sum(h) == n_points / 2
    constraint_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)  # project in [0,1]
        h_sum = h.sum()
        h = h * (constraint_sum / h_sum)  # normalize

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def optimize_h(initial_h, max_iter=1000):
        h_arr = initial_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]

        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            h = h.clip(0, 1)
            h_sum = h.sum()
            h = h * (constraint_sum / h_sum)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        # Run gradient-based refinement
        result = minimize(
            obj_func,
            h_arr,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter}
        )
        h_opt = result.x.reshape(n_points)

        # Manual normalization after optimization
        h_opt = h_opt.clip(0, 1)
        h_sum = h_opt.sum()
        h_opt = h_opt * (constraint_sum / h_sum)

        return h_opt

    def perturb_h(h, noise_factor=0.1):
        h = h.copy()
        noise = np.random.randn(len(h)) * noise_factor
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (constraint_sum / h.sum())
        return h

    best_h = h_init
    best_c5 = objective(best_h)

    # Start with gradient descent
    h = optimize_h(h_init)
    c5 = objective(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Try multiple restarts with perturbation
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        h = perturb_h(best_h, noise_factor=0.1)
        h = optimize_h(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Ensure final solution satisfies constraints
    best_h = best_h.clip(0, 1)
    best_h = best_h * (constraint_sum / best_h.sum())

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (constraint_sum / best_h.sum())
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
