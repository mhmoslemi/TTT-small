import numpy as np
import cvxpy as cp
import math
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()

    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n_points = 45
        h = np.ones(n_points, dtype=np.float64) / n_points * 2.0

    best_h = h.copy()
    best_c5 = np.inf

    def objective_function(h_vec, normalize=True):
        dx = 2.0 / n_points
        h_extended = np.concatenate([h_vec, h_vec, h_vec])
        vals = []
        for k in range(n_points):
            overlap = h_extended[k:k + n_points] * (1 - h_extended[n_points + k:2 * n_points + k])
            integral = np.sum(overlap) * dx
            vals.append(integral)
        c5 = np.max(vals)
        return c5

    def repair_h(h_vec, dx):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        mean = np.sum(h_vec) * dx
        correction = (1.0 - mean) / (2.0 * n_points)
        h_vec += correction
        h_vec = np.clip(h_vec, 0.0, 1.0)
        return h_vec

    best_c5 = objective_function(h)
    last_best_time = time.time()

    while time.time() - start_time < budget_s * 0.98:
        dx = 2.0 / n_points
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var <= 1.0]
        mean_cons = cp.sum(h_var) * dx == 1.0
        constraints.append(mean_cons)

        weights = None
        max_current = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        positional_weights = np.zeros(n_points * 3)
        for i in range(n_points):
            positional_weights[i + n_points] = 1.0
        for k in range(n_points):
            overlap = h[k] * (1 - h[k + n_points]) if k < len(h) and k + n_points < len(h) * 2 else 0.0
            positional_weights[k + len(h)] /= (max_current - overlap + 1e-8)

        weights_ground = positional_weights[n_points:2 * n_points].reshape(-1, 1)

        w = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(positional_weights[n_points:2 * n_points] * (h_var - w) ** 2))
        problem = cp.Problem(objective, constraints=constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status != cp.OPTIMAL:
            continue

        h_proposed = h_var.value
        if h_proposed is None:
            continue

        h_proposed = repair_h(h_proposed, dx)
        current_c5 = objective_function(h_proposed)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_proposed
            last_best_time = time.time()

        if time.time() - last_best_time > budget_s * 0.03:
            break

    dx = 2.0 / n_points
    assert np.all(np.isfinite(best_h))
    assert np.all(0.0 <= best_h)
    assert np.all(best_h <= 1.0)
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6, f"{np.sum(best_h) * dx:.6f} != 1.0"
    final_c5 = objective_function(best_h)

    return best_h.tolist(), final_c5, n_points
