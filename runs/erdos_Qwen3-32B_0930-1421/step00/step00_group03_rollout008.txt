import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points
    h_init = np.array(initial_h_values["h_values"], dtype=np.float64)

    def compute_c5(h):
        h1 = np.concatenate([h, np.zeros(len(h) - 1)])
        h2 = 1 - np.concatenate([np.zeros(len(h) - 1), h])
        overlap = np.correlate(h1, h2, mode="full") * dx
        return float(np.max(overlap))

    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    low = 0.38085
    high = 0.40
    tol = 1e-5

    # Bounds and overlap direction matrix
    def build_overlap_constraints():
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        overlap_integrals = []
        for k in range(-n_points + 1, n_points):
            if k < 0:
                # h(x)(1 - h(x + k)) becomes h(x)(1 - h'(x)) where h' is shifted
                h_shifted = np.roll(h, -k)
                overlap = cp.sum(h[:(n_points + k)] * (1 - h_shifted[:(n_points + k)])) * dx
            elif k > 0:
                h_shifted = np.roll(h, -k)
                overlap = cp.sum(h[k:] * (1 - h_shifted[k:])) * dx
            else:
                overlap = cp.sum(h * (1 - h)) * dx
            overlap_integrals.append(overlap)
        return constraints, overlap_integrals

    def b_search():
        nonlocal best_h, best_c5
        constraints, overlap_integrals = build_overlap_constraints()
        c = cp.Variable()
        max_stacked = cp.Max(cp.hstack(overlap_integrals))
        constraints.append(max_stacked <= c)
        problem = cp.Problem(cp.Minimize(c), constraints)
        remaining_time = budget_s - (time.time() - start_time)
        max_iter = 50
        for it in range(max_iter):
            mid = (low + high) / 2
            problem.solve(warm_start=True, verbose=False, solver=cp.SCS,
                          max_iter=2000, use_indirect=True, time_limit=remaining_time / max_iter)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.array(h.value).flatten()
                h_sol[(h_sol < 1e-5)] = 0.0
                h_sol[(h_sol > 1 - 1e-5)] = 1.0
                h_sol = np.clip(h_sol, 0, 1)
                h_sol *= (n_points / 2) / h_sol.sum()
                c5 = compute_c5(h_sol)
                if c5 <= mid:
                    best_h = h_sol
                    best_c5 = c5
                    high = mid
                else:
                    low = mid
            else:
                low = mid
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time < 1e-3 or (high - low) < tol:
                break
        return best_c5, best_h

    try:
        best_c5, best_h = b_search()
    except:
        pass

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        # Use perturbation strategy
        h = best_h.copy()
        for _ in range(3):
            idx = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            h[idx] = 1 - h[idx]
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / h.sum()
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h.tolist(), best_c5, n_points
