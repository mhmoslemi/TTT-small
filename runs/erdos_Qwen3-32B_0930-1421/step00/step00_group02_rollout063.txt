import numpy as np
import cvxpy as cp
from time import perf_counter

def run(seed=42, budget_s=1100, n_points=45, alpha=0.1, max_iter=1000000):
    np.random.seed(seed)

    h = initial_h_values

    print(f"Final C5: {best_c5}")
    return best_h, best_c5, n_points

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps[n - 1:])

def optimize_step(h, dx):
    n = len(h)
    k_max = -1
    val_max = -float('inf')

    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    max_overlap_idx = np.argmax(overlaps[n - 1:])
    k = max_overlap_idx
    correlations = np.correlate(h, h, mode='full')

    grad = np.zeros(n)
    for i in range(n - k):
        grad[i] -= (1 - h[i + k]) * dx
        grad[i + k] += h[i] * dx

    h_prime = cp.Variable(n)
    constraints = [0 <= h_prime, h_prime <= 1, cp.sum(h_prime) == n / 2]
    objective = cp.Minimize(cp.sum_squares(h_prime - (h - grad * 0.05)))
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        h_new = np.ravel(h_prime.value)
    except:
        raise RuntimeError("Optimization failed")

    h[:] = h_new
    c5_new = evaluate_erdos_solution(h)
    return h, c5_new
