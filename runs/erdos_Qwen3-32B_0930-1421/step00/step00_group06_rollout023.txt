import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

dx = 2.0 / 95
h0 = initial_h_values.copy()
n_points = len(h0)

def evaluate_C5(h):
    h = h - 1.0
    h_padded = np.concatenate([h, h[:2]])
    corr = np.correlate(h_padded, h_padded, mode='full')
    return np.max(corr * dx)

def optimize_ADMM(h0, max_iter=1000, rho=1.0, abstol=1e-4, reltol=1e-4, verbose=False):
    h = cp.Variable(n_points)
    z = cp.Variable()
    lamb = cp.Parameter(n_points, value=np.zeros(n_points), nonneg=True)
    objective = cp.Minimize(z)

    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1,
        z >= cp.sum(h * (1 - cp.roll(h, shift=1))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=2))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=3))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=4))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=5))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=6))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=7))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=8))) * dx,
        z >= cp.sum(h * (1 - cp.roll(h, shift=9))) * dx,
    ]

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.OSQP, verbose=verbose)
    return h.value

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_h = h0.copy()
    best_c5 = evaluate_C5(best_h)

    start_time = time.time()
    max_iter = 1000
    max_inner = 50
    verbose = False

    for _ in range(max_inner):
        h_opt = optimize_ADMM(best_h, verbose=verbose)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)

        c5 = evaluate_C5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
