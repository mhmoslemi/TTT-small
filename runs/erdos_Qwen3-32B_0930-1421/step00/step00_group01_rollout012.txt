```python
import numpy as np
import time
from scipy.optimize import minimize
from itertools import product

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    overlaps = []
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(h, -k)
        shifted = np.where(k <= 0, shifted[n_points + k:], shifted[:n_points - k])
        overlap = np.dot(h, 1 - shifted) * dx
        overlaps.append(overlap)
    return np.max(overlaps)

def smooth_max(values, eta):
    return np.log(np.sum(np.exp(eta * np.array(values)))) / eta

def evaluate_smooth_objective(h, n_points, eta):
    dx = 2.0 / n_points
    h = h.reshape(-1)
    values = []
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(h, -k)
        shifted = np.where(k <= 0, shifted[n_points + k:], shifted[:n_points - k])
        overlap = np.dot(h, 1 - shifted) * dx
        values.append(overlap)
    return smooth_max(values, eta)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        h = np.clip(h, 0, 1)
    return h

def perturb_h(h, perturbation=0.1):
    h = h + np.random.uniform(-perturbation, perturbation, size=h.shape)
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' not in globals():
        # Construct initial vector based on uniform distribution
        n_points = 47
        h0 = np.zeros(n_points)
        h0 = project_to_feasible(h0, n_points)
        initial_h_values = h0

    best_h = initial_h_values.copy()
    best_c5 = evaluate_objective(best_h, len(best_h))
    n_points = len(best_h)

    def objective(h_smoothed):
        return evaluate_smooth_objective(h_smoothed, n_points, eta=10)

    def objective_raw(h_raw):
        return evaluate_objective(h_raw, n_points)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Constraints and bounds
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Generate starting points
    starting_points = [initial_h_values]
    for _ in range(5):
        perturbed = perturb_h(initial_h_values)
        starting_points.append(perturbed)

    for idx, h_start in enumerate(starting_points):
        h_start = h_start.reshape(-1).copy()
        result = minimize(
            fun=objective,
            x0=h_start,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        h_opt = project_to_feasible(result.x, n_points)
        c5 = objective_raw(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        if time.time() - start_time > budget_s - 5:
            break

    return (best_h.tolist(), best_c5, n_points)
```
