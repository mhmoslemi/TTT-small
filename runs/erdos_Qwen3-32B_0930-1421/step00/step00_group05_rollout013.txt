import numpy as np
import time
import cvxpy as cp
from scipy.interpolate import interp1d

def triangular_basis(n_points, n_basis):
    dx = 2.0 / n_points
    x = np.linspace(0.0, 2.0, n_points, endpoint=False)

    basis_values = np.zeros((n_basis, n_points))
    width = 2.0 / n_basis
    for j in range(n_basis):
        center = (j + 0.5) * width
        basis = np.maximum(0.0, 1.0 - np.abs(x - center) / width)
        basis_values[j, :] = basis
    return basis_values

def project_to_basis(h, basis):
    n_basis, n_points = basis.shape
    h_proj = np.linalg.lstsq(basis.T, h, rcond=None)[0]
    return np.clip(h_proj, 0, 1)

def evaluate_c5(h, dx):
    # Compute all overlap integrals C_k
    c5_overlap = -np.correlate(h, 1 - h, mode='full') * dx
    return -np.max(c5_overlap)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    n_points = 70
    dx = 2.0 / n_points

    else:
        h_initial = np.zeros(n_points)
        h_initial[n_points // 2 - 5 : n_points // 2 + 5] = 0.1
        h_initial *= n_points / 2 / np.sum(h_initial)

    h_best = h_initial.copy()
    best_c5 = evaluate_c5(h_best, dx)

    n_basis = 50
    basis = triangular_basis(n_points, n_basis)
    h0_proj = project_to_basis(h_best, basis)

    w_var = cp.Variable(n_basis, nonneg=True)
    constraint_w_sum = [cp.sum(w_var) == 1.0]
    constraint_integral = [cp.sum(w_var * basis @ basis.T @ w_var) * dx == 1.0]
    constraint_values = [w_var <= 1.0]
    obj_integral_var = cp.sum(w_var * basis, axis=0)

    t_var = cp.Variable()
    overlap_obj = -t_var

    prob = cp.Problem(cp.Minimize(0), [t_var >= 1.0])

    partial_overlap = []

    for k in range(-n_points + 1, n_points):
        shifted_basis = np.roll(basis, shift=k, axis=1)
        shifted_basis[:, :max(0, -k)] = 0.0
        shifted_basis[:, n_points + k:] = 0.0
        partial_overlap.append((basis @ w_var) * (1.0 - shifted_basis @ w_var))
    
    overlap = cp.sum(cp.hstack(partial_overlap)) * dx
    constraints_overlap = [t_var >= overlap]

    solver_params = {"solver": cp.MOSEK, "verbose": False, "abstol": 1e-6, "reltol": 1e-6, "feastol": 1e-6}

    # Run optimization
    while time.time() - t0 < budget_s:
        prob = cp.Problem(cp.Minimize(t_var),
                          constraint_w_sum + constraint_values,
                          overlaps=[overlap] + constraints_overlap)
        prob.solve(**solver_params, warm_start=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = basis @ w_var.value
            h_new = np.clip(h_new, 0.0, 1.0)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h_best = h_new
                best_c5 = c5_new

        if prob.solver_stats.solve_time > 30:
            break

    return (h_best, best_c5, n_points)
