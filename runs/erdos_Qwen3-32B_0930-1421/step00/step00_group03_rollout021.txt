import numpy as np
import time
import scipy.signal
import math
import cvxpy as cp

def project_to_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    wanted_sum = n_points / 2
    sum_h = np.sum(h)
    if np.isclose(sum_h, wanted_sum):
        return h
    while abs(sum_h - wanted_sum) > 1e-5:
        diff = wanted_sum - sum_h
        indices = np.argsort(h)
        if diff > 0:
            idx = indices[-np.searchsorted(h[indices][::-1], 1.0) - 1] if 1.0 in h else indices[-1]
            increase_vals = h >= h[idx]
            h[increase_vals] += diff / np.sum(increase_vals)
        else:
            idx = indices[0]
            decrease_vals = h <= h[idx]
            h[decrease_vals] += diff / np.sum(decrease_vals)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
    return h

def compute_c5(h_vals, n_points):
    dh = 2.0 / n_points
    h = h_vals.copy()
    h_relu = h
    h_comp = 1 - h_relu
    corr = scipy.signal.correlate(h_relu, h_comp, mode='full')
    c5 = np.max(corr * dh)
    return c5

    alpha = 0.99
    max_sa = 1000
    max_gd = 500
    sigma = 0.1
    eta = 0.01

    start_time = time.time()
    while (time.time() - start_time) < budget_s:
        # Simulated Annealing
        for sa_i in range(max_sa):
            h_next = best_h + sigma * np.random.randn(n_points)
            h_next = project_to_feasibility(h_next, n_points)
            c5_next = compute_c5(h_next, n_points)
            if c5_next < best_c5 or np.random.rand() < np.exp((best_c5 - c5_next) / T):
                best_h = h_next.copy()
                best_c5 = c5_next
            T = max(T_min, T * alpha)
            if (time.time() - start_time) >= budget_s:
                break

        # Gradient Descent
        for gd_i in range(max_gd):
            h_relu = np.clip(best_h, 0, 1)
            h_comp = 1 - h_relu

            corr = scipy.signal.correlate(h_relu, h_comp, mode='full')
            k_opt = np.argmax(corr)
            grad = np.zeros(n_points)

            for i in range(n_points):
                if k_opt < n_points and i - (k_opt) < n_points:
                    grad[i] += h_comp[i - k_opt]  # from h(x)(1-h(x+k))
                if k_opt <= n_points and i + k_opt < n_points:
                    grad[i] -= h_relu[i + k_opt]  # from (1-h(x+k)) effect on h(x)

            best_h -= eta * grad
            best_h = project_to_feasibility(best_h, n_points)
            c5 = compute_c5(best_h, n_points)
            if c5 < best_c5:
                best_c5 = c5

            if (time.time() - start_time) >= budget_s:
                break

        if best_c5 >= 0.38087:
            h_init = np.random.rand(n_points)
            h_init = project_to_feasibility(h_init, n_points)
            best_h = h_init.copy()
            best_c5 = compute_c5(best_h, n_points)
        T = max(T_min * 10, T)

    return best_h, best_c5, n_points
