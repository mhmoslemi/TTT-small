import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
import random

def evaluate_solution(h):
    n = len(h)
    dx = 2.0 / n
    h_array = np.array(h, dtype=np.float64)
    one_minus_h = 1.0 - h_array

    corr = correlate(h_array, one_minus_h, mode='full')
    # Adjust to match the proper overlap range
    corr_values = corr[n - 1:2 * n - 1]
    max_c5 = np.max(corr_values) * dx
    return max_c5

def repair_solution(h):
    n = len(h)
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    target_sum = n / 2
    diff = target_sum - current_sum
    delta = 0.001
    flip_order = np.argsort(h_clipped)
    if diff > 0:
        for i in flip_order:
            if h_clipped[i] + delta <= 1:
                h_clipped[i] += delta
                diff -= delta
                if abs(diff) < 1e-8:
                    break
            else:
                add = 1 - h_clipped[i]
                h_clipped[i] = 1
                diff -= add
                if abs(diff) < 1e-8:
                    break
    else:
        for i in flip_order[::-1]:
            if h_clipped[i] - delta >= 0:
                h_clipped[i] -= delta
                diff += delta
                if abs(diff) < 1e-8:
                    break
            else:
                sub = h_clipped[i]
                h_clipped[i] = 0
                diff += sub
                if abs(diff) < 1e-8:
                    break
    h_clipped[0] = min(1.0, max(0.0, h_clipped[0]))  # Ensure first point is valid
    h_clipped[-1] = min(1.0, max(0.0, h_clipped[-1]))  # Ensure last point is valid
    return h_clipped

def mutate_by_repair(h):
    h_mutated = h.copy()
    n = len(h_mutated)

    # Random small perturbations in the middle
    positions_to_perturb = np.random.choice(n, int(0.05 * n), replace=False)
    perturbation_amount = np.random.uniform(-0.1, 0.1, size=positions_to_perturb.shape)
    h_mutated[positions_to_perturb] += perturbation_amount
    return repair_solution(h_mutated)

def generate_initial_population(pop_size, n_points, initial_h_values):
    population = []
    for _ in range(pop_size):
        if _ == 0 and initial_h_values is not None:
            population.append(repair_solution(initial_h_values))
        else:
            h_random = np.random.uniform(0, 1, size=n_points)
            h_random = repair_solution(h_random)
            population.append(h_random)
    return population

def tournament_selection(population, fitness, k=3):
    selected = []
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin(fitness[indices])]
    selected.append(population[best_idx])
    return selected[0]

def evolve_population(population, pop_size, n_points):
    fitness = np.array([evaluate_solution(h) for h in population])
    new_population = []
    for _ in range(pop_size):
        parent1 = tournament_selection(population, fitness)
        parent2 = tournament_selection(population, fitness)
        # Naive crossover: weighted average around midpoint
        if np.random.rand() < 0.5:
            midpoint = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:midpoint], parent2[midpoint:]])
        else:
            child = (parent1 + parent2) / 2.0
        child = repair_solution(child)
        if np.random.rand() < 0.2:
            child = mutate_by_repair(child)
        new_population.append(child)
    return new_population

def run(seed=42, budget_s=1100, pop_size=40, generations=20):
    np.random.seed(seed)
    random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() and initial_h_values is not None else None
    n_points = len(initial_h) if initial_h is not None else 42  # default size

    start_time = time.time()
    elapsed = 0.0
    population = generate_initial_population(pop_size, n_points, initial_h)

    best_h = None
    best_c5 = float('inf')

    for gen in range(generations):
        population = evolve_population(population, pop_size, n_points)
        new_fitness = [evaluate_solution(h) for h in population]
        new_best_idx = np.argmin(new_fitness)
        new_best_c5 = new_fitness[new_best_idx]
        new_best_h = population[new_best_idx]

        if new_best_c5 < best_c5:
            best_c5 = new_best_c5
            best_h = new_best_h

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
