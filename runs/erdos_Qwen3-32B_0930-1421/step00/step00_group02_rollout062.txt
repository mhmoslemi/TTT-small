import numpy as np
import cvxpy as cp
import time
from functools import partial

def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    max_iter = 500
    iter_no_improve = 0
    max_no_improve = 100
    
    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iter_no_improve >= max_no_improve:
            break

        # Compute overlap integral for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1:] * dx  # Only lags in [0, n_points - 1]
        
        k_star = np.argmax(corr)
        max_val = corr[k_star]
        
        # Compute gradient of max_val w.r.t h
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_star
            if j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Define optimization problem
        h_new = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_new - (h - 0.1 * grad)))
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2
        ]
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, max_seconds=5)
        
        h_new = np.array(h_new.value).flatten()
        h = h_new
        
        # Projection step to ensure constraints are satisfied
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s != 0:
            h = h * (n_points / 2) / s
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

    return (best_h, best_c5, n_points)
