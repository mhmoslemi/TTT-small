import numpy as np
import time
import numpy.random as npr

def evaluate_h(h):
    n_points = h.size
    dx = 2.0 / n_points
    differ = 1.0 - h
    convol = np.correlate(h, differ, mode="full")
    return np.max(convol * dx)

def enforce_integral_constraint(h):
    n_points = h.size
    target = n_points / 2
    total = h.sum()

    if abs(total - target) < 1e-6:
        return h

    # If constraint is slightly violated, adjust h to satisfy it
    if total > target:
        # Find the largest excess points to reduce
        diff = total - target
        h = h - diff / n_points
    else:
        # Find the largest deficit points to increase
        diff = target - total
        h = h + diff / n_points

    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h
