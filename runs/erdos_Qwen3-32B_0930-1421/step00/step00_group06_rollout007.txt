import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = h.reshape(-1)
    h1 = np.pad(h, (0, 2 * n_points - h.shape[0]), mode="wrap")
    h2 = 1 - h1
    corr = np.correlate(h1, h2, mode="full")
    c5_bound = np.max(corr * dx)
    return c5_bound
