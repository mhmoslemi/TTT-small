import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Initialization
    h_current = initial_h.copy().astype(np.float64)
    best_h = h_current.copy()
    best_c5 = float("inf")

    target_sum = n_points / 2.0
    alpha = 0.1  # Initial step size
    beta = 0.7  # Line search shrink factor
    max_steps = 200  # Prevent infinite loops

    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)

        # Correct the integral constraint
        diff = target_sum - s
        non_saturated = (h_clipped < 1) | (h_clipped > 0)
        active_indices = np.where(non_saturated)[0]

        if len(active_indices) == 0:
            return h_clipped  # Should not happen with valid input

        correction = diff / len(active_indices)
        h_clipped[active_indices] += correction

        # Clip again in case of floating point drift
        return np.clip(h_clipped, 0, 1)

    def compute_c5(h):
        correlate = np.correlate(h, 1 - h, mode="full")
        correlate = correlate * dx
        return np.max(correlate)

    best_c5 = compute_c5(h_current)

    step = 0
    while time.time() - start_time < budget_s - 0.1 and step < max_steps:
        step += 1

        # Compute full cross-correlations
        correlate = np.correlate(h_current, 1 - h_current, mode="full")
        correlate = correlate * dx

        # Find k_max where the overlap is worst
        k_max = int(np.argmax(correlate) - n_points)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            index = (i - k_max) % n_points
            grad[i] = (1 - h_current[index]) * dx

        # Gradient update
        h_proposed = h_current - alpha * grad
        h_proposed = project_h(h_proposed)

        # Evaluate new candidate
        c5_proposed = compute_c5(h_proposed)

        # Line search: accept if better, else backtrack
        if c5_proposed < best_c5:
            # New best solution
            best_c5 = c5_proposed
            h_current = h_proposed
            alpha *= 1.05  # Slight increase for faster convergence
        else:
            # No improvement: backtrack
            alpha *= beta
            continue

    # Post-processing
    final_h = project_h(h_current)
    final_c5 = compute_c5(final_h)

    return final_h.tolist(), final_c5, n_points
