import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    if not hasattr(run, "initial_h_values"):
        raise ValueError("run.initial_h_values must be set before calling run()")

    np.random.seed(seed)
    start_time = time.time()

    n_points = len(run.initial_h_values)
    h_init = np.array(run.initial_h_values).copy()
    h = h_init
    dx = 2.0 / n_points

    # Correct h to satisfy integral constraint
    h = np.clip(h, 0, 1)
    total_mass = np.sum(h)
    h *= (n_points / 2) / total_mass

    epsilon = 1e-6
    max_iter = 1000
    best_h = None
    best_c5 = float("inf")

    def compute_c5(h_flat):
        h_vals = h_flat.reshape(-1)
        overlaps = correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(overlaps)

    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        overlaps = correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(overlaps)

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Surrogate problem
        H = cp.Variable(n_points, nonneg=True)
        constraints = [H <= 1, cp.sum(H) == n_points / 2]

        proxy_objective = current_c5 + cp.sum((H - h) ** 2)
        objective = cp.Minimize(proxy_objective)
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, max_iters=150, verbose=False)
            h_new = np.clip(H.value, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new *= (n_points / 2) / h_new_sum
            h = h_new
        except:
            h = best_h.copy() if best_h is not None else h_init
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h *= (n_points / 2) / h_sum

    h = best_h
    c5_bound = compute_c5(h)

    return (h, c5_bound, n_points)
