import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def evaluate_objective(h, dx, n_points):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h = h / h_sum * (n_points / 2)
    h = np.clip(h, 0, 1)
    h_array = np.concatenate([h, np.zeros(n_points - 1)])
    ones_minus_h = 1 - h
    ones_minus_h_array = np.concatenate([ones_minus_h, np.zeros(n_points - 1)])
    correlations = np.correlate(h_array, ones_minus_h_array, mode='full') * dx
    C5 = np.max(correlations[1:-1])
    return C5


def perturb_h(h, scale=0.1, n_points=None):
    if n_points is None:
        n_points = len(h)
    perturbed = h + np.random.normal(0, scale, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed_sum = perturbed.sum()
    perturbed = perturbed / perturbed_sum * (n_points / 2)
    return np.clip(perturbed, 0, 1)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(globals()['initial_h_values']).copy()
    else:
        h = np.zeros(n_points)
        h[0] = 1.0
    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0, 1)

    best_c5 = evaluate_objective(h, dx, n_points)
    best_h = h.copy()

    start_positions = []
    if 'initial_h_values' in globals():
        start_positions.append(globals()['initial_h_values'].copy())
        for _ in range(3):
            hp = perturb_h(globals()['initial_h_values'])
            start_positions.append(hp.copy())

    for idx in range(10):
        if time.time() - start_time >= budget_s:
            break

        if idx < len(start_positions):
            h = start_positions[idx]
        else:
            h = np.random.rand(n_points)
            h = h / h.sum() * (n_points / 2)
            h = np.clip(h, 0, 1)

        def objective(x):
            h_test = x.reshape(-1)
            h_test = np.clip(h_test, 0, 1)
            h_test_sum = h_test.sum()
            h_test = h_test / h_test_sum * (n_points / 2)
            h_test = np.clip(h_test, 0, 1)
            h_array = np.concatenate([h_test, np.zeros(n_points - 1)])
            ones_minus_h = 1 - h_test
            ones_minus_h_array = np.concatenate([ones_minus_h, np.zeros(n_points - 1)])
            correlations = np.correlate(h_array, ones_minus_h_array, mode='full') * dx
            C5 = np.max(correlations[1:-1])
            return C5

        # Constraint: sum(x) = n_points / 2
        def constraint_sum(x):
            x = x.reshape(-1)
            x = np.clip(x, 0, 1)
            return x.sum() - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]

        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            h_post = np.clip(result.x, 0, 1)
            h_post_sum = h_post.sum()
            h_post = h_post / h_post_sum * (n_points / 2)
            h_post = np.clip(h_post, 0, 1)
            c5_val = evaluate_objective(h_post, dx, n_points)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_post.copy()

        # Simulated annealing step
        if time.time() - start_time < budget_s:
            temp = 10.0
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                temp *= 0.9
                h_candidate = perturb_h(best_h)
                current_c5 = best_c5
                candidate_c5 = evaluate_objective(h_candidate, dx, n_points)
                if candidate_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - candidate_c5) / temp):
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    best_h = best_h / best_h_sum * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_objective(best_h, dx, n_points)
    return (best_h, final_c5, n_points)
