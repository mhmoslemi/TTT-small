```python
import numpy as np
import time
import math
from scipy.signal import correlate

def binary_flip(h):
    # Find indices where h is 0 and 1
    idx0 = np.where(h == 0)[0]
    idx1 = np.where(h == 1)[0]
    i = np.random.choice(idx1)
    j = np.random.choice(idx0)
    h2 = h.copy()
    h2[i] = 0
    h2[j] = 1
    return h2

def evaluate_overlap(h, n_points):
    dx = 2.0 / n_points
    ones_count = int(n_points / 2)
    # Convert h to a binary vector of 0/1
    h = (h > 0.5).astype(np.float64)
    # Ensure mean is exactly half
    while h.sum() != ones_count:
        idx_below = np.where(h < 0.5)[0]
        idx_above = np.where(h > 0.5)[0]
        if h.sum() < ones_count:
            h[np.random.choice(idx_below)] = 1
        else:
            h[np.random.choice(idx_above)] = 0
    # Compute full correlation with mode="full"
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return c5, h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = (initial_h_values > 0.5).astype(np.float64)
        n_points = len(h)
        ones_count = int(n_points / 2)
    else:
        n_points = 80
        h = np.random.choice([1, 0], size=n_points, p=[0.5, 0.5])
        h[h.sum() < ones_count] = 1  # repair constraint
        h[:ones_count - h.sum()] = 1  # ensure exactly half are 1s

    best_h = h.copy()
    best_c5, best_h = evaluate_overlap(h, n_points)
    
    dx = 2.0 / n_points
    current_c5 = best_c5
    T0 = 1.0
    T = T0
    alpha = 0.99
    min_temp = 1e-5
    no_improve = 0

    while time.time() - start_time < budget_s:
        candidate_h = binary_flip(best_h)
        candidate_c5, _ = evaluate_overlap(candidate_h, n_points)

        if candidate_c5 < best_c5 or math.exp((best_c5 - candidate_c5) / T) > np.random.rand():
            best_c5 = candidate_c5
            best_h = candidate_h.copy()
            no_improve = 0
        else:
            no_improve += 1

        T *= alpha
        if T < min_temp or no_improve >= 1000:
            T = max(T * 2, min_temp)
            alpha *= 0.995  # reduce cooling rate if not improving
            no_improve = 0

    c5_bound = best_c5
    h_values = best_h

    return h_values, c5_bound, n_points
```
