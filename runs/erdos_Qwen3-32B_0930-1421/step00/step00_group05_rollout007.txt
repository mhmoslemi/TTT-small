import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    n = len(h)
    k_ranges = np.arange(-n + 1, n)
    max_c = 0.0
    refl_h = 1 - h
    c = correlate(h, refl_h, mode='full')
    c = c * dx
    max_c = np.max(c)
    return max_c

    h = cp.Variable(n_points, nonneg=True)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

    # Precompute indices for all k in convolution
    max_shift = n_points - 1
    overlap = []
    for k in range(1, max_shift):

        overlap.append(cp.sum(h[i] * (1 - h[i + k])) * dx)
    obj = cp.Minimize(cp.maximum(*overlap))
    prob = cp.Problem(obj, constraints)

    # Convert problem to a form friendly to solvers with many objectives
    prob.solve(solver=cp.GLPK_MI, verbose=False)

    h_values = np.clip(h.value, 0, 1)
    c5 = evaluate_C5(h_values, dx)

    best_c5 = c5
    best_h = h_values.copy()

    # Simple greedy local search for refinement
    while time.time() - start_time < budget_s - 30:  # leave 30s for final conversion and return
        h_greedy = best_h.copy()
        idx = np.random.choice(n_points)
        # try toggle h[i] to 0 or 1
        for val in [0.0, 1.0]:
            h_try = h_greedy.copy()
            h_try[idx] = val
            # adjust other point to maintain integral
            total = np.sum(h_try)
            if total < n_points / 2:
                extra = (n_points / 2) - total
                h_add_idx = np.random.choice(n_points)
                h_try[h_add_idx] = min(1.0, h_try[h_add_idx] + extra)
            elif total > n_points / 2:
                extra = total - (n_points / 2)
                h_sub_idx = np.random.choice(n_points)
                h_try[h_sub_idx] = max(0.0, h_try[h_sub_idx] - extra)
            h_try = np.clip(h_try, 0, 1)
            c5_new = evaluate_C5(h_try, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_try.copy()

    return best_h, best_c5, n_points
