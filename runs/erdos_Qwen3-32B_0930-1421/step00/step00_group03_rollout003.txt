import numpy as np
import evaluate_erdos_solution as evaluator
import time
from scipy.signal import correlate

def featurize_overlap(h, dx):
    n = len(h)
    h_tilde = 1 - h
    corr = correlate(h, h_tilde, mode='full') * dx
    idx_center = n - 1
    return max(corr[idx_center - 4:idx_center + 5])

def project_sum_constraint(h, dx, target_sum):
    current_sum = h.sum()
    scale_factor = target_sum / current_sum
    return np.clip(h * scale_factor, 0, 1)

def compute_feasible_h(h, dx):
    h_clipped = np.clip(h, 0, 1)
    target_sum = len(h) / 2
    h_feasible = project_sum_constraint(h_clipped, dx, target_sum)
    return h_feasible

def compute_gradient(h, n_points, dx):
    corr = correlate(h, 1 - h, mode='full')
    idx_center = n_points - 1
    overlap_window = corr[idx_center - 4:idx_center + 5]
    grad = np.zeros_like(h)
    for i in range(1, 5):  # k = 1 to 4
        for j in range(n_points - i):
            grad[j] += -(1 - h[j + i]) * dx
            grad[j + i] += -h[j] * dx
    grad_window = grad[idx_center - 4:idx_center + 5]
    return grad_window

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = evaluator.initial_h_values

    n_points = len(h0)
    dx = 2.0 / n_points
    h = h0.copy()
    best_h = h.copy()
    best_c5 = featurize_overlap(h, dx)

    start_time = time.time()
    patience = 0
    improvement_threshold = 1e-5
    learning_rate = 0.01
    max_patience = 100
    total_iterations = 2000

    for iter_ in range(total_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        h = compute_feasible_h(h - learning_rate * compute_gradient(h, n_points, dx), dx)
        current_c5 = featurize_overlap(h, dx)

        if current_c5 < best_c5 - improvement_threshold:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            learning_rate *= 0.5
            patience = 0
            if learning_rate < 1e-4:
                break

    return best_h.tolist(), best_c5, n_points
