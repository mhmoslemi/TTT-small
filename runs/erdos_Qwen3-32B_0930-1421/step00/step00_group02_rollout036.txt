import numpy as np
import scipy.signal
import time
import cvxpy as cp


def evaluate_erdos_solution(h_values, n_points, exact=True):
    h = np.reshape(h_values, (-1,))
    dx = 2.0 / n_points
    # Validate constraints
    if not np.all(np.isfinite(h)):
        return float('inf'), h
    if np.any((h < -1e-6) | (h > 1 + 1e-6)):
        return float('inf'), h
    if abs(np.sum(h) - n_points / 2) > 1e-6:
        return float('inf'), h
    if exact:
        c5_values = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(c5_values)
    else:
        c5 = float('inf')
    return c5, h


def project_probability_mass(h, n_points):
    h = np.clip(h, 0, 1)
    sum_h = h.sum()
    adjustment = (n_points / 2 - sum_h) / n_points
    delta = np.zeros_like(h)
    indices = h < 0.5
    indices.reverse()
    count = np.count_nonzero(indices)
    if count > 0:
        delta[indices] = adjustment / count
    else:
        indices = h > 0.5
        indices.reverse()
        count = np.count_nonzero(indices)
        delta[indices] = adjustment / count
    h += delta
    h = np.clip(h, 0, 1)
    return h


def iterate_h(h, n_points, lr=5.0, momentum=0.8, dampening=0.9, beta=0.5):
    dx = 2.0 / n_points
    grad = np.zeros_like(h)

    # For each lag k
    for k in range(n_points):
        # Create delayed versions of h
        h_k = hk = np.roll(h, -k)
        ones_minus_h_k = ohk = 1 - hk
        inner = h * ohk
        grad_k = np.correlate(inner, np.ones(1), mode="same") / dx
        grad_k[0] = inner.sum() / np.sqrt(n_points)

        grad += grad_k / (1 + k ** beta)

    # Normalize the gradient
    grad = grad / (1e-12 + np.max(np.abs(grad)))
    grad = (1 - momentum) * grad + momentum * grad[:-1]

    # Update h with gradient step
    h_update = h - lr * grad

    # Project onto feasible set
    h = project_probability_mass(h_update, n_points)

    return h


def run(seed=42, budget_s=1100, max_iter=1000, n_points=45):
    np.random.seed(seed)

    if "initial_h_values" in globals():
        h = np.array(globals()["initial_h_values"], dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = project_probability_mass(h, n_points)

    best_h = h.copy()
    best_score, _ = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    dx = 2.0 / n_points

    while time.time() - start_time < budget_s and max_iter > 0:
        max_iter -= 1

        h = iterate_h(h, n_points)
        new_score, _ = evaluate_erdos_solution(h, n_points)

        if new_score < best_score:
            best_h = h.copy()
            best_score = new_score

    if best_score == float('inf'):
        best_score, _ = evaluate_erdos_solution(best_h, n_points)

    return best_h, best_score, n_points
