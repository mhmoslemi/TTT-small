            if h[i] == 1:
                if not in_one:
                    start = x[i]
                    in_one = True
            else:
                if in_one:
                    end = x[i]
                    intervals.append((start, end))
                    in_one = False
        if in_one:
            end = 2.0
            intervals.append((start, end))
        return intervals

    def intervals_to_h(intervals, dx, n_points):
        h = np.zeros(n_points, dtype=np.float64)
        x = np.linspace(0, 2, n_points)

        x_left = x - dx/2
        for a, b in intervals:
            mask = np.logical_and(x_right >= a, x_left <= b)
            h[mask] += 1
        return h

    def compute_c5(h, dx):
        return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

    def feasibility(intervals):
        measure = sum(b - a for a, b in intervals)
        return np.isclose(measure, 1.0)
