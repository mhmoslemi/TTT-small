import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # For reproducibility
    np.random.seed(seed)

    # Use starting construction if available
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        n_points = len(h_start)
    else:
        # Starting from uniform distribution over [0, 1]
        n_points = 95  # Default, can be adjusted easily
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start / h_start.mean() * (n_points / 2)
        h_start = np.clip(h_start, 0, 1)

    dx = 2.0 / n_points
    max_iter = 500
    eps = 1e-6  # Regularization for log objective

    h_vals = cp.Variable(shape=(n_points), nonneg=True)
    h = h_vals

    # Precompute correlation lags
    lags = []
    for k in range(1, n_points):
        offsets = (np.arange(n_points) + k) % n_points
        lag = cp.Parameter((n_points,), value=np.arange(n_points))
        lags.append(lag)

    obj_vals = [h[i] * (1 - h[offsets[i]]) for i in range(n_points)]
    obj = cp.sum(obj_vals) * dx

    constraints = [
        h[i] <= 1 for i in range(n_points)
    ] + [
        h[i] >= 0 for i in range(n_points)
    ] + [
        cp.sum(h) == n_points / 2
    ]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    h_curr = h_start.copy()
    best_h = h_curr.copy()
    best_c5 = np.inf
    t_start = time.time()

    for iter_num in range(max_iter):
        if time.time() - t_start > budget_s - 5:
            break

        u_k = []

        for k in range(1, n_points):
            shifts = (np.arange(n_points) + k) % n_points
            val = np.mean(h_curr * (1 - h_curr[shifts])) * 2.0

            # Keep track of worst-case correlations
            u_k.append(val)

        C5_current = max(u_k)
        if C5_current < best_c5:
            best_c5 = C5_current
            best_h = h_curr.copy()

        # Update optimization objective
        obj = 0
        max_corr = 0
        for k in range(1, n_points):
            shifts = (np.arange(n_points) + k) % n_points
            corris = h * (1 - h[shifts])
            obj += corris.sum() * dx + eps  # Minimal smoothing

        # Reconstruct and optimize
        obj = cp.Minimize(obj)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)

        h_curr = np.clip(h_vals.value, 0, 1).flatten()

        # Restore integral constraint
        h_curr /= h_curr.mean()
        h_curr *= n_points / 2
        h_curr = np.clip(h_curr, 0, 1)

        # Early stopping criterion
        if iter_num > 50 and abs(C5_current - min(u_k)) < 1e-4:
            break

    # Final C₅ evaluation
    C5 = 0
    for k in range(1, n_points):
        shifts = (np.arange(n_points) + k) % n_points
        val = (best_h * (1 - best_h[shifts])).sum() * dx
        C5 = max(C5, val)

    return (best_h, C5, n_points)
