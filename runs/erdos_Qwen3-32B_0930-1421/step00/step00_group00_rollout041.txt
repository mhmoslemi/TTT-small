import numpy as np
import time
import scipy

def evaluate_erdos_solution(h_values):
    h = np.array(h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    h = h * dx
    h_correlation = np.correlate(h, 1 - h, mode='full')
    result = np.max(h_correlation) / dx
    return float(result)

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-8:
        return h
    if diff > 0:
        nonzero = h < 1
        if np.sum(nonzero) > 0:
            h[nonzero] += (diff / np.sum(nonzero))
    else:
        nonzero = h > 0
        if np.sum(nonzero) > 0:
            h[nonzero] += (diff / np.sum(nonzero))
    return h
