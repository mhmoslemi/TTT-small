import numpy as np
import scipy.optimize
import time

def evaluate_C5(h_vec, dx):
    n_points = len(h_vec)
    h = h_vec.reshape(-1)
    max_c5 = 0.0
    for k in range(1, n_points):
        overlap = np.dot(h, np.roll(1 - h, k))
        overlap *= dx
        max_c5 = max(max_c5, overlap)
    return max_c5

def objective_C5(x, dx):
    return evaluate_C5(x, dx)

def bounds_and_constraints(n_points):
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    def constraint_sum(x):
        return np.sum(x) - (n_points / 2.0)
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    return bounds, constraints

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        n_points = len(h_init)
    else:
        n_points = 47
        h_init = np.zeros(n_points)
        h_init[:n_points // 2] = 1.0

    dx = 2.0 / n_points

    def objective(x):
        return evaluate_C5(x, dx)

    # Prepare constraints
    bounds, constraints = bounds_and_constraints(n_points)

    # Initial projection to satisfy the constraint
    h_proj = h_init.copy()
    h_proj[h_proj < 0.0] = 0.0
    h_proj[h_proj > 1.0] = 1.0
    sum_h = np.sum(h_proj)
    while abs(sum_h - n_points / 2.0) > 1e-6:
        if sum_h < n_points / 2.0:
            idx = np.argmin(h_proj)
            add = min(1.0 - h_proj[idx], (n_points / 2.0 - sum_h))
            h_proj[idx] += add
        else:
            idx = np.argmax(h_proj)
            sub = min(h_proj[idx], sum_h - n_points / 2.0)
            h_proj[idx] -= sub
        sum_h = np.sum(h_proj)

    result = None
    opt_result = scipy.optimize.minimize(
        objective,
        h_proj,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-10}
    )

    if opt_result.success:
        result = opt_result.x
    else:
        result = h_proj

    # Final check and projection
    result = np.clip(result, 0.0, 1.0)
    sum_h = np.sum(result)
    if abs(sum_h - n_points / 2.0) > 1e-5:
        delta = sum_h - (n_points / 2.0)
        if delta > 0:
            indices = np.where(result > 0.0)[0]
            result[indices] -= delta / len(indices)
        else:
            indices = np.where(result < 1.0)[0]
            result[indices] -= delta / len(indices)
        result = np.clip(result, 0.0, 1.0)

    best_c5 = evaluate_C5(result, dx)
    elapsed = time.time() - start_time
    return (result, best_c5, n_points)
