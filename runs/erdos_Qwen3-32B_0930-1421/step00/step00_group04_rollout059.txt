import numpy as np
from scipy.fft import fft, ifft
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=84):
    np.random.seed(seed)

    # Use initial_h_values provided by the harness
    if 'initial_h_values' not in globals():
        # Create uniform distribution, scaled to ∫h = 1
        h = np.full(n_points, 0.5)
        dx = 2.0 / n_points
        h *= n_points / (2 * h.sum())
    else:
        h = np.array(initial_h_values[:n_points])

    dx = 2.0 / n_points
    h = h / h.sum() * (n_points / 2)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    max_iter = 2000
    alpha = 0.05
    alpha_decay = 0.995
    patience = 50
    no_improvement = 0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute overlaps using FFT-based convolution
        fft_h = fft(h)
        fft_1h = fft(1 - h)
        fft_overlap = fft_h * np.conj(fft_1h)

        overlap = ifft(fft_overlap).real * dx

        k_max = int(np.argmax(overlap))
        gradient = h - np.roll(h, shift=k_max)

        # Gradient ascent to minimize C₅
        h = h - alpha * gradient

        # Projection to maintain constraints
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)

        c5 = evaluate_erdos_solution(h)

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Early stopping
        if no_improvement >= patience:
            break

        # Reduce learning rate
        alpha *= alpha_decay

    return best_h.tolist(), best_c5, n_points
