import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    ones = np.ones_like(h)

    def compute_max_overlap():
        h_array = h * dx
        one_minus_h = (ones - h) * dx
        corr = correlate(h_array, one_minus_h, mode='full')
        overlap = np.max(corr)
        return overlap

    C5 = compute_max_overlap()
    return C5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Start from the initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure the sum constraint
    desired_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - desired_sum) > 1e-6:
        h = h * (desired_sum / current_sum)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_t = time.time()

    max_iter = 100000
    iteration = 0
    no_improve_thresh = 60
    patience = 0
    learning_rate = 0.01
    lr_decay = 0.995

    def project_and_renormalize(h_val):
        h_val = np.clip(h_val, 0.0, 1.0)
        current_sum = np.sum(h_val)
        target_sum = n_points / 2.0
        h_val *= (target_sum / current_sum)
        h_val = np.clip(h_val, 0.0, 1.0)
        return h_val

    start_time = time.time()

    while time.time() - start_time < budget_s and iteration < max_iter and patience < no_improve_thresh:
        iteration += 1

        # Compute overlaps for all k
        k_min = -n_points + 1
        k_max = n_points - 1
        overlaps = []

        for k in range(k_min, k_max + 1):
            if k < 0:
                a = h[-k:]
                b = h[:len(a)]
            elif k >= 0:
                a = h[:len(h) - k]
                b = h[k:]
            else:
                continue
            if len(a) == 0 or len(b) == 0:
                continue
            overlaps.append(np.sum(a * (1 - b)) * dx)
        overlaps = np.array(overlaps)
        worst_k = np.argmax(overlaps)
        worst_overlap = overlaps[worst_k]

        # Approximate gradient for worst_k and update h
        if worst_k < n_points:
            grad = h[worst_k:] * (1 - h[n_points:]) - h[worst_k:] ** 2
            h[:n_points - worst_k] -= learning_rate * grad
        else:
            grad = h[:-worst_k] * (1 - h[worst_k:]) - h[:-worst_k] ** 2
            h[worst_k:] -= learning_rate * grad

        h = project_and_renormalize(h)

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_t = time.time()
            patience = 0
        else:
            patience += 1

        learning_rate *= lr_decay

    return best_h, best_c5, n_points
