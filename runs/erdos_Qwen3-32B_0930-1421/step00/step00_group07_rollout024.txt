import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize h to fulfill sum(h) == n_points/2
    current_sum = h.sum()
    h = h * (n_points / 2) / current_sum

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    last_improvement_time = time.time()
    patience = 30  # seconds without improvement before breaking
    alpha = 0.02
    halve_after = 500
    counter = 0
    max_iter = 5000

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        C = corr * dx
        return C

    start_time = time.time()

    for it in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        C = compute_c5(h)
        current_c5 = C.max()
        max_idx = C.argmax()

        # Compute gradient for current max KL
        grad = np.zeros_like(h)
        for i in range(n_points):
            k = max_idx - (n_points - 1)
            if 0 <= i - k < n_points:
                grad[i] += 1.0 - h[i - k]
            if 0 <= i + k < n_points:
                grad[i] -= h[i + k]  # grad = (1 - h_{i+k}) - h_{i-k}

        update = h - alpha * grad
        # Clip and project to [0, 1]
        h_clipped = np.clip(update, 0, 1)
        # Normalize to satisfy sum(h) == n_points/2 (project into feasible set)
        h_sum = h_clipped.sum()
        if h_sum > 0:
            h = h_clipped * (n_points / 2) / h_sum
        else:
            h = np.zeros_like(h_clipped)

        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement_time = time.time()

        # Adjust alpha
        counter += 1
        if counter >= halve_after and c5 >= best_c5:
            alpha *= 0.5
            counter = 0

    # Final validation
    assert np.all(best_h <= 1.0 + 1e-8)
    assert np.all(best_h >= -1e-8)
    assert abs(best_h.sum() - n_points / 2) < 1e-8

    return (best_h, best_c5, n_points)
