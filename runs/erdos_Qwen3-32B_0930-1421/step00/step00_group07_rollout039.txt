import numpy as np
import time
import random


def run(seed=42, budget_s=1100, **kwargs):
    # Set seed
    np.random.seed(seed)
    random.seed(seed)

    # Start with the given initial_h_values
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Check that measure constraint is satisfied
    assert np.isclose(np.sum(h) * dx, 1), "Initial h violates the measure constraint."

    # Convert h into a list of intervals (start, end)
    intervals = []
    inside = False
    interval_start = 0
    for i in range(n_points):
        if h[i] > 0.5 and not inside:
            inside = True
            interval_start = i
        elif h[i] < 0.5 and inside:
            inside = False
            interval_end = i
            # Map index to position in [0,2] interval
            a = interval_start * dx
            b = interval_end * dx
            intervals.append((a, b))
    if inside:
        interval_end = n_points
        a = interval_start * dx
        b = interval_end * dx
        intervals.append((a, b))

    # Helper: compute C5 from intervals
    def compute_c5(intervals, n_points=n_points, dx=dx):
        # Map intervals to the h_values
        h_candidate = np.zeros(n_points)
        for a, b in intervals:
            start = int(np.floor(a / dx))
            end = int(np.ceil(b / dx))
            h_candidate[start:end] = 1

        # Compute max overlap (C5) using cross-correlation
        c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode="full") * dx)
        return h_candidate, c5

    # Initial C5 evaluation
    h_best, c5_best = compute_c5(intervals)
    best_intervals = intervals.copy()

    # Optimization parameters
    n_intervals = len(intervals)
    max_intervals = 20
    T_start = 2.0
    T_end = 0.001
    cooling_rate = 0.999
    max_iterations = 1000
    step_size = 0.05
    max_no_improvement = 100
    stalling_count = 0

    start_time = time.time()
    best_time = start_time

    for iter_num in range(max_iterations):
        current_time = time.time()
        if current_time - start_time >= budget_s:
            break
        T = T_start * np.exp(np.log(T_end / T_start) * (current_time - start_time) / (budget_s))
        if T < 1e-6:
            T = 1e-6

        # Randomly pick an interval to perturb (index choice and position)
        m = len(intervals)
        if m == 0:
            break

        interval_idx = np.random.randint(m)
        a, b = intervals[interval_idx]

        # Decide what to perturb
        perturb_start = random.random() < 0.5
        if perturb_start:
            # Perturb start of interval
            delta = np.random.uniform(-step_size, step_size)
            new_a = a + delta
            new_a = max(new_a, 0)  # ensure within [0,2]
            if new_a >= b:
                new_a = b - 1e-5
            if interval_idx > 0 and intervals[interval_idx-1][1] >= new_a:
                new_a = intervals[interval_idx-1][1]
            new_interval = (new_a, b)
        else:
            # Perturb end of interval
            delta = np.random.uniform(-step_size, step_size)
            new_b = b + delta
            new_b = min(new_b, 2)  # ensure within [0,2]
            if new_b <= a:
                new_b = a + 1e-5
            if interval_idx < m - 1 and intervals[interval_idx+1][0] <= new_b:
                new_b = intervals[interval_idx+1][0]
            if new_b > 2:
                new_b = 2
            new_interval = (a, new_b)

        # Create new intervals
        new_intervals = intervals.copy()
        new_intervals[interval_idx] = new_interval

        # Ensure no overlap
        if any(new_intervals[i][1] > new_intervals[i+1][0] for i in range(len(new_intervals)-1)):
            # Overlap occurs, skip this candidate
            continue

        # Ensure total measure is 1
        total_length = sum(b - a for a, b in new_intervals)
        scale_factor = 1.0 / total_length
        if np.isnan(scale_factor):
            continue
        scaled_intervals = [(a * scale_factor, b * scale_factor) for a, b in new_intervals]
        if any(scaled_intervals[i][1] > scaled_intervals[i+1][0] for i in range(len(scaled_intervals)-1)):
            continue

        # Evaluate candidate
        h_candidate, c5_candidate = compute_c5(scaled_intervals)

        # Accept if better or with probability in Boltzmann chain
        if c5_candidate < c5_best or np.random.random() < np.exp((c5_best - c5_candidate) / T):
            intervals = scaled_intervals.copy()
            h_best, c5_best = h_candidate, c5_candidate
            best_intervals = intervals.copy()
            best_time = time.time()
            stalling_count = 0
        else:
            stalling_count += 1

        # Dynamically adjust step size or early terminate
        if stalling_count > max_no_improvement and T < 0.1:
            step_size *= 0.5

        if iter_num % 10 == 0:
            print(f"Iter {iter_num}, C5: {c5_best:.5f}, T: {T:.3f}, Time: {current_time - start_time:.1f}s")

    h_values = h_best
    return h_values, c5_best, n_points
