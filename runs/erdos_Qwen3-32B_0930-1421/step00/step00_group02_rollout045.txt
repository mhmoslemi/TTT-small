```python
from typing import Tuple, Optional
import numpy as np
import cvxpy as cp
from time import time

def compute_overlap_integral(h, dx):
    ones_minus_h = 1.0 - h
    S = np.correlate(h, ones_minus_h, mode="full")
    return S * dx  # This is the C₅ candidate values for each shift

def project_to_constraints(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to satisfy the constraint sum h = n_points / 2
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-3:
        h = h * (n_points / 2) / h_sum
    return h

def approximate_overlap_gradient(h, k, dx):
    n = len(h)
    if k < 0 or k >= n:
        return None
    gradient = np.zeros(n)
    for i in range(n - k):
        gradient[i] -= h[i + k] * dx
        gradient[i + k] -= h[i] * dx
    return gradient

def run(seed=42, budget_s=1100, n_points=45, max_iterations=5000, step_size=0.1):
    np.random.seed(seed)
    start_time = time()
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.full(n_points, 0.5 / dx)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to constraint
    best_h = h.copy()
    best_c5 = float('inf')
    
    for it in range(max_iterations):
        if (time() - start_time) > budget_s - 10:
            break
        overlap_values = compute_overlap_integral(h, dx)
        if np.max(overlap_values) < best_c5:
            best_c5 = np.max(overlap_values)
            best_h = h.copy()

        k_star = np.argmax(overlap_values)
        gradient = approximate_overlap_gradient(h, k_star, dx)

        if gradient is None:
            h = best_h.copy()
            continue

        # Constrained optimization (projection step)
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_var - h + step_size * gradient))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_var.value
            h = project_to_constraints(h, n_points)
        else:
            # Fallback in case of failure
            h = best_h.copy()
    
    # Final projection to make it safe
    h = project_to_constraints(h, n_points)
    overlap_values = compute_overlap_integral(h, dx)
    c5_bound = np.max(overlap_values)
    
    return h, c5_bound, n_points
```
