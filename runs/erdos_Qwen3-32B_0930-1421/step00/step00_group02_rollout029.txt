import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t0 = time.time()
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h / h.sum() * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

    best_h = np.copy(h)
    best_c5 = float('inf')

    while time.time() - t0 < budget_s:
        # Compute convolutions
        h1 = np.array(h)
        h0_h1 = 1 - h1
        corr = np.correlate(h1, h0_h1, mode='full')
        corr = corr[n_points - 1:-n_points + 1] * dx
        max_loc = np.argmax(corr)
        k_star = max_loc

        # Compute gradient for k_star
        grad = np.zeros_like(h)
        grad[k_star:] -= h0_h1[:-k_star]
        grad[:-k_star] += h1[k_star:]

        # Form optimization problem
        h_new = cp.Variable(n_points)
        constraints = [h_new >= 0, h_new <= 1, cp.sum(h_new) == n_points / 2]
        objective = cp.Minimize((h_new - (h - 0.1 * grad)) @ (h_new - (h - 0.1 * grad)))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver='ECOS')

        if problem.status != cp.OPTIMAL:
            # If infeasible, project
            h_proj = np.clip(h - 0.1 * grad, 0, 1)
            h_proj *= (n_points / 2) / h_proj.sum()
            h = h_proj
        else:
            h = h_new.value

        # Evaluate and update best solution
        h0_h1 = 1 - h
        corr = np.correlate(h, h0_h1, mode='full')
        c5 = np.max(corr[n_points - 1:-n_points + 1]) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
