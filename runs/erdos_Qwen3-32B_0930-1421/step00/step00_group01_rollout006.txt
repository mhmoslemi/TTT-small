import numpy as np

def compute_c5(h, n_points):
    """Compute C₅ for given h using correlation."""
    h = np.minimum(1.0, np.maximum(0.0, h))
    dx = 2.0 / n_points
    sum_h = h.sum()
    if not np.isclose(sum_h, n_points / 2.0):
        h += (n_points / 2.0 - sum_h) / n_points
        h = np.minimum(1.0, np.maximum(0.0, h))
    c5_values = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c5_values.clip(0.0, 1.0))

def normalize_h(h):
    """Ensure h satisfies the constraints."""
    h = np.minimum(1.0, np.maximum(0.0, h))
    total = h.sum()
    if np.isclose(total, 0):
        return np.zeros_like(h)
    correction = (h.shape[0] / 2.0 - total) / h.shape[0]
    h += correction
    h = np.minimum(1.0, np.maximum(0.0, h))
    return h

def swap_mutate(h):
    """Perform a swap mutation by flipping some values."""
    n_points = len(h)
    mask = np.random.rand(n_points) < 0.025
    h[mask] = np.random.rand(np.sum(mask))
    return normalize_h(h)

def crossover(parent1, parent2):
    """Crossover two parents by slicing in the middle."""
    n = len(parent1)
    split = n // 4 + np.random.randint(n // 2)
    child = np.concatenate([parent1[:split], parent2[split:]])
    return normalize_h(child)
