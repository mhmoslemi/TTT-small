import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_objective(h_vals):
        ones_hx = 1 - h_vals
        corrs = []
        for k in range(-n_points + 1, n_points):
            valid = (h_vals.shape[0] + k <= h_vals.shape[0]) and (h_vals.shape[0] + k >= 1)
            if not valid:
                continue
            corr = np.sum(h_vals * ones_hx[k:])
            corrs.append(corr * dx)
        return max(corrs)

    def rescale_to_mass(h_vals):
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2 / mass)
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals

    learning_rate = 0.01
    num_iter = 0
    patience = 20
    no_improve = 0

    while time.time() - start_time < budget_s - 1e-4:
        dh = 0.0

        for k in range(-n_points + 1, n_points):
            idxs = np.arange(n_points)
            valid = (idxs + k) < n_points
            if not valid.any():
                continue
            mask = np.zeros_like(h)
            mask[:-k] = 1
            contrib = -h[k:] if k < 0 else -h[:n_points - k]
            dh += contrib * mask

        grad = dh
        h_grad = (1 - h) * h * grad
        h = h - learning_rate * (h_grad / (1e-7 + np.abs(h_grad).mean()))
        h = rescale_to_mass(h)
        h = np.clip(h, 0, 1)

        current_c5 = compute_objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
            learning_rate = max(learning_rate * 1.02, 0.005)
        else:
            no_improve += 1
            learning_rate = max(learning_rate * 0.98, 0.001)

        if no_improve >= patience:
            break

        num_iter += 1

    return best_h, best_c5, n_points
