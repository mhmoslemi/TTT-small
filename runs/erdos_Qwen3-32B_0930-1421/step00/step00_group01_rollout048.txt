import scipy.optimize
import time

def compute_c5(h, dx):
    n_points = len(h)
    h = np.asarray(h)
    max_overlap = -np.inf
    for k in range(-n_points + 1, n_points):
        if -2 <= k <= 1:
            shifted = np.roll(h, -k) if k < 0 else h if k == 0 else np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted))
            max_overlap = max(max_overlap, overlap * dx)
    return max_overlap
