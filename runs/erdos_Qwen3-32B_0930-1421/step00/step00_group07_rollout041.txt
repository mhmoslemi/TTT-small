def run(seed=42, budget_s=1100):

    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.repeat(0.5, n_points)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    alpha = 0.02
    patience = 500
    no_improvement = 0
    max_iter = 5000

    while time.time() - start_time < budget_s:
        for iter in range(max_iter):
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            c5_vals = corr * dx
            max_index = np.argmax(c5_vals)
            k = max_index - (n_points - 1)

            # Compute relevant indices
            low = max(0, -k)
            high = min(n_points, n_points - k)

            # Compute gradient for this shift only
            grad = np.zeros(n_points)
            grad[low:high] = (1 - h[(low + k):]) - h[:(high - low)]

            # Update h
            h = h - alpha * grad

            # Project to feasible region [0, 1]
            h = np.clip(h, 0, 1)

            # Rescale to ensure the constraint is satisfied
            h *= (n_points / 2) / h.sum()

            # Evaluate current C5
            corr = np.correlate(h, 1 - h, mode="full")
            c5_vals = corr * dx
            current_c5 = np.max(c5_vals)

            # Track best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1

            # Adjust learning rate
            if no_improvement >= patience:
                alpha *= 0.5
                no_improvement = 0

            # Early stop if on track
            if best_c5 <= 0.38085:
                break

        # Exit loop based on time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
