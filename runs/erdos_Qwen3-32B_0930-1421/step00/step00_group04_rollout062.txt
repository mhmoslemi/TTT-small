import numpy as np
import cvxpy as cp
import time
from math import floor, ceil
from evaluate_erdos_solution import evaluate_erdos_solution

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_h(h, n_points):
    # Clip values to [0,1]
    h_proj = np.clip(h, 0.0, 1.0)
    # Normalize to satisfy the area constraint
    area = h_proj.sum() * (2.0 / n_points)
    if area == 0:
        return h_proj
    scale = 1.0 / area
    h_proj *= scale / n_points * 2.0
    return h_proj

def compute_grad(h, dx, n_points):
    k_values = np.arange(-n_points + 1, n_points)
    dh_dp = np.zeros((len(k_values), n_points))
    for i, k in enumerate(k_values):
        sh_k = (n_points + k) % n_points
        dh_dp[i, :sh_k] = h[sh_k:]
        dh_dp[i, sh_k:] = h[:n_points - sh_k]
    dh_dp = dh_dp - 2 * h[np.newaxis, :]
    dh_dp = dh_dp * dx

    overlap = np.correlate(h, 1 - h, mode="full") * dx
    max_idx = np.argmax(overlap)
    grad = dh_dp[max_idx]
    return grad

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = h.shape[0]
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    initial_c5 = best_c5
    print(f"Initial C5: {best_c5}")
    
    # Optimize
    max_iter = 10_000
    learning_rate = 0.01
    restart_period = 100
    iter_counter = 0
    
    while time.time() - start_time < budget_s and iter_counter < max_iter:
        grad = compute_grad(h, dx, n_points)
        updated_h = h - learning_rate * grad
        updated_h = project_h(updated_h, n_points)
        updated_c5 = evaluate_c5(updated_h, n_points)
        
        if updated_c5 < best_c5:
            h = updated_h
            best_h = h.copy()
            best_c5 = updated_c5
            print(f"Improved C5: {best_c5} at iter {iter_counter}")
        else:
            # If no improvement, try to generate random perturbation
            h = best_h.copy()
            h = project_h(h + np.random.randn(n_points) * 0.05, n_points)
        
        iter_counter += 1
        
        # Every restart_period steps, try a simplex move
        if iter_counter % restart_period == 0:
            simplex_size = 0.1
            simplex = np.array([best_h + simplex_size * (np.random.rand(n_points) - 0.5) for _ in range(3)])
            simplex = np.vstack([simplex, best_h])
            simplex = np.array([project_h(h_point, n_points) for h_point in simplex])
            c5_values = np.array([evaluate_c5(h_point, n_points) for h_point in simplex])
            best_simplex = simplex[np.argmin(c5_values)]
            if evaluate_c5(best_simplex, n_points) < best_c5:
                h = best_simplex
                best_h = h.copy()
                best_c5 = evaluate_c5(h, n_points)
                print(f"Simplex improved C5: {best_c5}")

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
