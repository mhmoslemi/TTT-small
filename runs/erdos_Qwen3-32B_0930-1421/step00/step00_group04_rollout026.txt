import numpy as np
import scipy.fft as sf
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / 256
    n_points = 256
    h = initial_h_values if hasattr(run, 'initial_h_values') else np.ones(n_points) / n_points * n_points / 2
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (n_points / 2) / total

    def objective(h_vec):
        h = np.reshape(h_vec, n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scaling = (n_points / 2) / total
        h = h * scaling
        H = sf.fft(h)
        K = sf.fft(1 - h)
        cross_correlation = sf.ifft(H * K.conj()).real * dx
        maximum = cross_correlation.max()
        return maximum

    best_h = h.copy()
    best_val = objective(best_h)

    iters = 300
    alpha = 0.05
    decay = 0.97
    for i in range(iters):
        if time.time() - start_time > budget_s - 1:
            break

        grad = np.random.randn(n_points) - 0.5
        grad = grad / np.abs(grad).max()

        grad_h = grad * alpha
        h_update = best_h + grad_h
        h_update = np.clip(h_update, 0, 1)
        total = np.sum(h_update)
        scaling = (n_points / 2) / total if total > 0 else 1.0
        h_update = h_update * scaling

        val = objective(h_update)
        if val < best_val:
            best_val = val
            best_h = h_update
            alpha *= 1.1
        else:
            alpha *= decay

    # Additional adaptive frequency-domain step
    H = sf.fft(best_h)
    for k in range(3):
        freq_mask = np.zeros_like(H)
        mask = np.concatenate([np.ones(2), np.zeros(n_points - 4), np.ones(2)])
        freq_mask = H * mask
        best_h = sf.ifft(freq_mask).real
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        best_h = best_h * (n_points / 2) / total
        best_val = objective(best_h)

    return (best_h, best_val, n_points)
