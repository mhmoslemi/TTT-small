import numpy as np
import scipy.signal
import cvxpy as cp
from time import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    t0 = time()
    budget_s = min(budget_s, 1100)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 100
        h = np.zeros(n_points, dtype=np.float64)
        h[:25] = 0.5
        h[25:50] = 0.3
        h[50:75] = 0.6
        h[75:] = 0.6
        h *= (n_points / 2) / h.sum()
    
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    n_outer_loop = 0
    t_phase1 = 0.6 * budget_s
    t_phase2 = 0.4 * budget_s

    # PHASE 1: FREQUENCY-DOMAIN APPROXIMATION
    while time() - t0 < t_phase1:
        n_outer_loop += 1

        # Frequency domain cross-correlation setup
        hh = h * (1 - h)
        corr = scipy.signal.correlate(hh, np.ones_like(hh), mode="full") * dx
        max_k_idx = np.argmax(np.abs(corr))
        max_k = max_k_idx - len(corr) // 2

        # Convex reparametrization
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(np.linalg.norm(cp.fft.fft(h_var * (1 - h_var)), np.inf))
        constraints = [
            0 <= h_var, h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_candidate = np.clip(h_var.value, 0, 1)
            h_candidate /= h_candidate.sum() * dx
            h_candidate *= n_points / 2
            h = h_candidate.copy()
        except:
            continue

        h = h_candidate.copy()
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # PHASE 2: TIME-DOMAIN REFINEMENT
    t_start = time()
    alpha = 0.01
    while time() - t_start < t_phase2:
        cost = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(np.abs(cost))
        offset = max_idx - len(cost) // 2

        grad = -scipy.signal.correlate(1 - h, h, mode='full')[len(h) + offset] * dx
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
