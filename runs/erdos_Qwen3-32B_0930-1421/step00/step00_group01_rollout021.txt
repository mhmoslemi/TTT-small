import numpy as np
import time
import random
from math import floor, ceil

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Initial parameters
    n_points = len(initial_h_values)
    h_init = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, mutation_strength=0.05):
        # Create a copy to avoid modifying the original
        h_mutated = h.copy()
        idxs = np.random.choice(n_points, size=floor(n_points * mutation_strength), replace=False)
        h_mutated[idxs] = np.random.rand(len(idxs))
        return clip_and_normalize(h_mutated)

    def crossover(parent1, parent2):
        split_point = random.randint(1, n_points - 2)
        child = np.concatenate((parent1[:split_point], parent2[split_point:]))
        return clip_and_normalize(child)

    def tournament_selection(population, fitness, k=3):
        indices = np.random.choice(len(population), k)
        return population[np.argmin(fitness[indices])]

    # Initialize population
    population_size = 50
    pop = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init.copy()
        h += np.random.normal(0, 0.1, h.shape)
        pop.append(clip_and_normalize(h))
    pop = [clip_and_normalize(x) for x in pop]

    # Evaluate initial fitness
    fitness = [compute_c5(h) for h in pop]
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx]
    best_c5 = fitness[best_idx]
    start_time = time.time()

    # Main loop
    while time.time() - start_time < budget_s:
        # Select parents
        parents = []
        for _ in range(population_size // 2):
            p1 = tournament_selection(pop, fitness)
            p2 = tournament_selection(pop, fitness)
            parents.extend([p1, p2])

        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i + 1]
            child1 = crossover(p1, p2)
            child2 = mutate(child1, mutation_strength=0.02)
            offspring.extend([child1, child2])

        # Replace worst with new offspring
        combined = pop + offspring
        combined_fitness = [compute_c5(h) for h in combined]
        best_indices = np.argsort(combined_fitness)[:population_size]
        pop = [combined[i] for i in best_indices]
        fitness = [combined_fitness[i] for i in best_indices]

        # Update best solution
        current_best = min(fitness)
        if current_best < best_c5:
            best_idx = np.argmin(fitness)
            best_h = pop[best_idx]
            best_c5 = min(fitness)

    # Final evaluations
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
