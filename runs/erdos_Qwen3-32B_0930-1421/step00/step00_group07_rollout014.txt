```python
import numpy as np
import math
import time
from scipy.signal import correlate

T0 = 1.0
alpha = 0.999
min_T = 1e-5
patience = 1000

def evaluate_overlap(h, dx):
    h_arr = np.array(h, dtype=float)
    one_minus_h = 1.0 - h_arr
    corr = correlate(h_arr, one_minus_h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, starting_n=80):
    start_time = time.time()
    best_c5 = float('inf')
    n_points = starting_n
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h_bin = (h > 0.5).astype(int)
        # Force the sum constraint
        ones_needed = n_points // 2
        if np.sum(h_bin) != ones_needed:
            diff = np.sum(h_bin) - ones_needed
            idx_ones = np.where(h_bin == 1)[0]
            idx_zeros = np.where(h_bin == 0)[0]
            if diff > 0:
                np.random.shuffle(idx_ones)
                h_bin[idx_ones[:diff]] = 0
            else:
                np.random.shuffle(idx_zeros)
                h_bin[idx_zeros[:abs(diff)]] = 1
    else:
        h_bin = np.zeros(n_points, dtype=int)
        ones_needed = n_points // 2
        idx = np.random.choice(n_points, ones_needed, replace=False)
        h_bin[idx] = 1

    dx = 2.0 / n_points
    current_c5 = evaluate_overlap(h_bin, dx)
    best_c5 = current_c5
    best_h = h_bin.copy()

    T = T0
    no_improvement = 0

    while time.time() - start_time < budget_s:
        # Propose a new candidate by flipping two bits
        idx_1s = np.where(h_bin == 1)[0]
        idx_0s = np.where(h_bin == 0)[0]
        if len(idx_1s) == 0 or len(idx_0s) == 0:
            break
        i1 = np.random.choice(idx_1s)
        i2 = np.random.choice(idx_0s)
        h_new = h_bin.copy()
        h_new[i1] = 0
        h_new[i2] = 1

        candidate_c5 = evaluate_overlap(h_new, dx)
        delta = candidate_c5 - current_c5

        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            h_bin = h_new
            current_c5 = candidate_c5
            no_improvement = 0
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h_new.copy()
        else:
            no_improvement += 1

        if T > min_T:
            T *= alpha
        else:
            T = max(T, min_T)

        if no_improvement > patience:
            T = max(T * 10, min_T)
            no_improvement = 0
            alpha = min(alpha * 1.005, 0.99999)

    return (best_h.tolist(), best_c5, n_points)
```
