import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    h_init = initial_h_values  # Use pre-determined starting vector
    h = h_init.copy()

    n_points = h.shape[0]
    dx = 2.0 / n_points

    best_c5 = np.inf
    best_h = h.copy()

    def evaluate_c5(h_values):
        h_vals = h_values.flatten()
        k_max = n_points - 1
        overlaps = np.zeros(k_max)
        for k in range(1, k_max):
            if k >= h_vals.shape[0]:
                break
            overlap = np.sum(h_vals[:n_points - k] * (1.0 - h_vals[k:]) * dx)
            overlaps[k] = overlap
        return np.max(overlaps)

    def project_h(h_values):
        h_proj = np.clip(h_values, 0, 1).copy()
        current_total = np.sum(h_proj)
        correction = (n_points / 2.0 - current_total) / n_points
        h_proj += correction
        h_proj = np.clip(h_proj, 0, 1)
        return h_proj

    while time.time() - start_time < budget_s:
        h = project_h(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        overlaps = np.zeros(n_points)
        for k in range(1, n_points):
            if k >= h.shape[0]:
                break
            overlaps[k] = np.sum(h[:n_points - k] * (1 - h[k:]) * dx)

        k_star = np.argmax(overlaps[1:])
        grad = np.zeros_like(h)

        for i in range(n_points - k_star):
            grad[i] -= h[i + k_star] * dx
            grad[i + k_star] -= h[i] * dx

        grad = -grad

        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.norm(h_var - (h - 0.1 * grad)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0,
        ]
        prob = cp.Problem(obj, constraints)

        prob.solve(solver=cp.ECOS, max_iters=100)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_var.value

        if (time.time() - start_time) >= budget_s:
            break

    best_h = project_h(best_h)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_points
