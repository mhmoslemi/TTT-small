import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, dx):
    n = len(h)
    h_extended = np.concatenate([h, h])
    one_minus_h = 1.0 - h_extended
    corrs = np.zeros(n)
    for k in range(n):
        c = np.sum(h_extended[:n] * one_minus_h[k:k+n]) * dx
        corrs[k] = c
    return np.max(corrs)

def objective(h, n_points, dx):
    h = h.copy()
    total = np.sum(h)
    if abs(total - n_points / 2.0) > 1e-6:
        return float('inf')  # Penalize constraint violations for sum
    h_clipped = np.clip(h, 0, 1)
    if abs(np.sum(h_clipped) - n_points / 2.0) > 1e-6:
        return float('inf')
    val = compute_c5(h_clipped, dx)
    return val

def gradient(h, n_points, dx):
    h = h.copy()
    h_clipped = np.clip(h, 0, 1)
    H = h_clipped
    one_minus_H = 1.0 - h_clipped
    grads = np.zeros_like(h)
    dx_return = dx
    for k in range(1, n_points - 1):
        corr = h[: n_points - k] * one_minus_H[k : n_points]
        g1 = 2 * H[: n_points - k] * H[k : n_points] - H[: n_points - k]
        g2 = 2 * H[k : n_points] * H[: n_points - k] - H[k : n_points]
        grads[: n_points - k] += np.sum(g1, axis=0) * dx_return
        grads[k : n_points] += np.sum(g2, axis=0) * dx_return
    return grads

def optimize_from_initial(initial, n_points, dx, max_iter=1000):
    best_h = initial.copy()
    best_c5 = compute_c5(best_h, dx)

    def callback(xk):
        nonlocal best_h, best_c5
        hclipped = np.clip(xk, 0, 1)
        if abs(np.sum(hclipped) - n_points / 2.0) < 1e-6:
            c5_val = compute_c5(hclipped, dx)
            if c5_val < best_c5:
                best_h[:] = hclipped[:]
                best_c5 = c5_val

    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0})
    bounds = [(0, 1)] * len(initial)

    res = opt.minimize(
        lambda x: objective(x, n_points, dx),
        x0=initial,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',  # better for nonlinear constraints
        callback=callback,
        options={'maxiter': max_iter}
    )
    return best_h, best_c5
