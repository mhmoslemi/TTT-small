import numpy as np
import evaluate_erdos_solution as evaluator
import time
import random
import math
import cvxpy as cp
from scipy.signal import correlate
import copy

def evaluate_overlap(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode="full")
    return np.max(corr * dx)

def create_binary_initial(n_points, seed=42):
    random.seed(seed)
    total_mass = n_points // 2
    h = np.zeros(n_points, dtype=int)
    indices = random.sample(range(n_points), total_mass)
    h[indices] = 1
    return h

def flip_two_bits(h):
    n_points = len(h)
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) == n_points // 2:
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
    elif len(zeros_idx) == n_points // 2:
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
    else:
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
    h_new = copy.copy(h)
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def is_valid_solution(h, n_points):
    return int(np.sum(h)) == n_points // 2 and np.all(h == h.astype(int))

def run(seed=42, budget_s=1100, patience=1000):
    random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_float = initial_h_values
        h_bin = np.where(h_float > 0.5, 1, 0).astype(int)
        while np.sum(h_bin) != n_points // 2:
            flip_one_replacement = np.where(np.random.rand(n_points) < 0.5, h_bin, 1 - h_bin)[0]
            coin = random.random()
            if coin < 0.5:
                ones_idx = np.where(h_bin == 1)[0]
                i = random.choice(ones_idx)
                h_bin[i] = 0
            else:
                zeros_idx = np.where(h_bin == 0)[0]
                i = random.choice(zeros_idx)
                h_bin[i] = 1
    else:
        h_bin = create_binary_initial(n_points, seed=seed)

    while not is_valid_solution(h_bin, n_points):
        h_bin = create_binary_initial(n_points)

    best_h = h_bin.copy()
    best_c5 = evaluate_overlap(best_h, n_points)

    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    start_time = time.time()
    iter_no_improve = 0
    max_iter = int(budget_s * 10)  # Max iterations based on budget
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        h_candidate = flip_two_bits(h_bin)
        c_candidate = evaluate_overlap(h_candidate, n_points)
        c_current = evaluate_overlap(h_bin, n_points)

        if c_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c_candidate
            iter_no_improve = 0
        elif random.random() < math.exp((c_current - c_candidate) / T):
            h_bin = h_candidate.copy()
        else:
            h_bin = h_bin.copy()

        T *= alpha
        if T < min_T:
            T = max_T
            alpha *= 0.995
            iter_no_improve = 0
        iter_no_improve += 1

        if best_h is None or best_c5 is None:
            best_h = h_candidate
            best_c5 = c_candidate

    return (best_h, best_c5, n_points)
