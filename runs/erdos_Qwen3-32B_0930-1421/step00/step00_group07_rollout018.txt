import numpy as np
import math
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    shifted_h = h * (n_points - 1)
    ones_sub_h = (1.0 - h) * (n_points - 1)
    corr = np.correlate(shifted_h, ones_sub_h, mode="full") / ((n_points - 1) ** 2)
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    h *= n_points / 2.0 / h.sum()
    return h

def compute_gradient(h, dx):
    n_points = len(h)
    shifted_h = h * (n_points - 1)
    ones_sub_h = (1.0 - h) * (n_points - 1)
    corr = np.correlate(shifted_h, ones_sub_h, mode="full") / ((n_points - 1) ** 2)
    c5_lags = np.argmax(corr)
    grad = np.zeros(n_points)
    k = c5_lags
    for i in range(n_points):
        if i + k < n_points:
            grad[i] += h[i] * (1.0 - h[i + k])
        if i - k >= 0:
            grad[i] += h[i] * (1.0 - h[i - k])
    return grad * dx
