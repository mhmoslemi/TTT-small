import numpy as np
import scipy.fft as fft
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2 / 42  # Use the finest scalable sampling matching initial_h_values
    n_points = len(initial_h_values)

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.concatenate((h, h, h))  # Repeat for convolution
        k_vals = []
        for k in range(1 - n_points + 1, n_points):
            shifted = np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted[1: n_points + 1])) * dx
            k_vals.append(overlap)
        return max(k_vals)

    def bounds_and_constraint(h_shape=n_points):
        return Bounds(0, 1), {'type': 'eq', 'fun': lambda h: np.sum(h) - h_shape / 2}

    def project(h):
        h = np.clip(h, 0, 1)
        h *= (h_shape / 2) / np.sum(h)
        return h

    h_shape = n_points
    h_sol = np.array(initial_h_values).copy()
    h_hist = [h_sol]
    obj_hist = [objective(h_sol)]
    best_obj =obj_hist[-1]
    h_best = h_sol.copy()

    start_time = time.time()
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        bounds, constraint = bounds_and_constraint(h_shape)
        result = minimize(objective, h_sol, bounds=bounds, constraints=constraint, method='L-BFGS-B')
        if result.success:
            h_sol = result.x
            h_sol = project(h_sol)
            obj_val = result.fun
            h_hist.append(h_sol)
            obj_hist.append(obj_val)
            if obj_val < best_obj:
                best_obj = obj_val
                h_best = h_sol
    c5_bound = best_obj
    return (h_best, c5_bound, n_points)
