import numpy as np
import time
from scipy.optimize import minimize, Bounds, BFGS
import evaluate_erdos_solution

def compute_c5(h):
    dx = 2.0 / len(h)
    h_corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(h_corr) * dx
    return c5

def project_to_feasible_space(h):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if total > 0:
        h_clipped *= (len(h) / 2) / total
    return h_clipped

def perturb_h(parent_h, scale=0.1):
    h = parent_h + np.random.normal(0, scale, len(parent_h))
    return project_to_feasible_space(h)

def crossover(p1, p2):
    alpha = np.random.rand()
    return project_to_feasible_space(alpha * p1 + (1 - alpha) * p2)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    initial_h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Genetic Algorithm Parameters
    pop_size = 50
    max_gen = 500
    mutation_rate = 0.1
    elite_ratio = 0.2
    
    # Initialize population
    population = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size // 2)]
    population.extend([perturb_h(initial_h, scale=0.1) for _ in range(pop_size // 2)])
    # Enforce feasibility
    population = [project_to_feasible_space(h) for h in population]
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Genetic Algorithm Loop
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.9:  # Early exit if close to budget
            break
            
        # Evaluate fitness
        population = [project_to_feasible_space(h) for h in population]
        c5_scores = np.array([compute_c5(h) for h in population])
        fevals = len(c5_scores)
        
        best_idx = np.argmin(c5_scores)
        if c5_scores[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = c5_scores[best_idx]
        
        # Select elites
        sorted_indices = np.argsort(c5_scores)
        elite_count = max(1, int(pop_size * elite_ratio))
        elites = [population[i] for i in sorted_indices[:elite_count]]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            p1_idx, p2_idx = np.random.choice(sorted_indices[:pop_size//2], size=2, replace=False)
            child = crossover(population[p1_idx], population[p2_idx])
            if np.random.rand() < mutation_rate:
                child = perturb_h(child, scale=0.1)
            offspring.append(project_to_feasible_space(child))
        
        # Next generation
        population = elites + offspring
    
    # Post-processing with local gradient descent
    def objective(h):
        return compute_c5(h)
    
    bounds = Bounds(0, 1)
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50}
    )
    
    final_h = project_to_feasible_space(result.x)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, n_points
