import numpy as np
import time
from scipy.optimize import minimize, dual_annealing
import evaluate_erdos_solution as evaluator

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    total_budget = 1100  # fixed budget via problem statement
    
    # Use initial_h_values if provided, and enhance with perturbations
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values)
    else:
        n = 47
        h0 = np.zeros(n)
        h0[n // 2 - 5 : n // 2 + 5] = 0.5  # symmetric seed

    class Timer:
        def __init__(self):
            self.start_time = time.time()
            self.timeout_s = budget_s

        def consume(self, max_frac=0.99):
            now = time.time()
            elapsed = now - self.start_time
            if elapsed > self.timeout_s:
                return True, elapsed
            else:
                remaining_s = self.timeout_s - elapsed
                allowed_frac = max_frac if elapsed < self.timeout_s / 2 else 0.5
                max_allowed = min(allowed_frac * remaining_s, 60)
                return False, max_allowed

    def compute_C5(h, n_points):
        dx = 2.0 / n_points
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(conv)

    def smooth_C5(h, n_points, smooth_weight=100.0):
        dx = 2.0 / n_points
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        return (1.0 / smooth_weight) * np.log(np.sum(np.exp(smooth_weight * conv)))

    def objective(h, n_points):
        return compute_C5(np.clip(h, 0.0, 1.0), n_points)

    def constrained_objective(h, n_points):
        h_clipped = np.clip(h, 0.0, 1.0)
        mean_needed = 0.5
        offset = (mean_needed * n_points - h_clipped.sum()) / n_points
        h_normalized = h_clipped + offset
        h_normalized = np.clip(h_normalized, 0.0, 1.0)
        return compute_C5(h_normalized, n_points)

    def perturb(h_values, scale=0.05):
        h = h_values.copy()
        pert = np.random.normal(0, scale, h.shape)
        h += pert
        mean_needed = 0.5
        h = np.clip(h, 0.0, 1.0)
        h = h * (mean_needed * h.shape[0] / h.sum())
        return h

    best_h = h0.copy()
    best_C5 = compute_C5(best_h, n)
    print(f"Initial C5: {best_C5}")
    best_time = 0.0

    trials = 5
    num_candidates = 5

    for trial in range(trials):
        trial_timer = Timer()
        timeout, allowed_time = trial_timer.consume()

        candidates = [best_h]
        for _ in range(num_candidates):
            candidate = perturb(candidates[-1])
            candidates.append(candidate)

        for h_candidate in candidates:
            if time.time() - start_time > budget_s - 5:
                break
            print(f"Evaluating perturbed candidate {candidates.index(h_candidate)}")
            def wrap_obj(x):
                return constrained_objective(x, n)
            result = minimize(
                wrap_obj,
                x0=h_candidate,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                options={"maxiter": 100, "ftol": 1e-6}
            )
            h_opt = np.clip(result.x, 0.0, 1.0)
            mean_needed = 0.5
            h_opt = h_opt * (mean_needed * n / h_opt.sum())
            current_C5 = compute_C5(h_opt, n)
            if current_C5 < best_C5:
                best_h = h_opt
                best_C5 = current_C5
                best_time = time.time() - start_time
            print(f"Improved C5 after refinement: {best_C5:.6f} at time {best_time:.2f}s")

            da_result = dual_annealing(
                wrap_obj,
                bounds=[(0.0, 1.0)] * n,
                maxiter=50,
                initial_temp=20,
                seed=trial * 42 + seed
            )
            h_da = np.clip(da_result.x, 0.0, 1.0)
            h_da = h_da * (mean_needed * n / h_da.sum())
            current_C5_da = compute_C5(h_da, n)
            if current_C5_da < best_C5:
                best_h = h_da
                best_C5 = current_C5_da
                best_time = time.time() - start_time
            print(f"Improved C5 after SA: {best_C5:.6f} at time {best_time:.2f}s")

        if time.time() - start_time > budget_s - 5:
            break

    # Final validation and return
    mean_needed = 0.5
    best_h = np.clip(best_h, 0.0, 1.0)
    h_normalized = best_h * (mean_needed * n / best_h.sum())
    h_normalized = np.clip(h_normalized, 0.0, 1.0)

    final_C5 = compute_C5(h_normalized, n)
    print(f"Final C5: {final_C5}")
    print(f"Time used: {time.time() - start_time:.2f}s")

    return (
        h_normalized.tolist(),
        final_C5,
        len(h_normalized)
    )
