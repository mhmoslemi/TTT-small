import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if provided
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 80
        h = np.full(n_points, 0.5, dtype=np.float64)
    h /= h.sum() * 2  # Normalize to satisfy ∫h dx = 1

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    patience = 200
    no_improvement = 0
    max_iter = 10_000
    alpha = 0.005
    start_time = time.time()
    prev_time = start_time
    shift_cache = {}

    def compute_overlap(h):
        h1 = h
        h2 = 1 - h
        corr = np.correlate(h1, h2, mode="full") * dx
        return np.max(corr)

    def normalize_h(h, target_sum=n_points / 2):
        h_clipped = np.clip(h, 0.0, 1.0)
        s = np.sum(h_clipped)
        if s != 0:
            return h_clipped * target_sum / s
        else:
            return h_clipped

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        if no_improvement >= patience:
            break

        # Compute correlation
        h1 = h
        h2 = 1 - h

        corr = np.correlate(h1, h2, mode='full') * dx
        k_max = np.argmax(corr)
        c5 = corr[k_max]

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Compute gradient for the worst shift k_max
        # gradient = (1 - h[i + k]) - h[i - k]
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx_plus = (i + k_max) % n_points
            if idx_plus != i:
                grad[i] += (1.0 - h[idx_plus]) / 2.0
            idx_minus = (i - k_max) % n_points
            if idx_minus != i:
                grad[i] -= h[idx_minus] / 2.0

        # Update h with gradient
        h_new = h - alpha * grad
        h = normalize_h(h_new)

    return best_h, best_c5, n_points
