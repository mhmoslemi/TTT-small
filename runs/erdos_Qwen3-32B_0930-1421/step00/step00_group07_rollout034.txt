            learning_rate *= 0.5
            patience_counter = 0

        # Evaluate the cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx_max = np.argmax(np.abs(corr))
        # Identify the two overlapping segments based on the lag
        if idx_max < n_points:
            # Lag corresponds to shifting h to the right (1 - h is to the left)
            corr_segment = np.sum(h[: idx_max] * (1 - h[idx_max:]))
            # Gradient is -(1-h) for the overlap part
            grad_contrib = np.zeros_like(h)
            if idx_max > 0:
                grad_contrib[: idx_max] -= (1 - h[idx_max:])
                grad_contrib[idx_max:] -= -h[: idx_max]
        else:
            # Lag corresponds to shifting 1-h to the right (h is to the left)
            offset = idx_max - (n_points - 1)
            corr_segment = np.sum(h[offset:] * (1 - h[: -offset]))
            grad_contrib = np.zeros_like(h)
            if offset < n_points:
                grad_contrib[offset:] -= (1 - h[: -offset])
                grad_contrib[: -offset] -= -h[offset:]

        # Normalize gradient
        grad = grad_contrib / (1 + abs(grad_contrib).sum())
        # Update with learning rate
        h += learning_rate * grad

        # Clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Restore constraint sum(h) == n_points / 2
        h_sum = h.sum()
        if h_sum != 0:
            h *= target_sum / h_sum

    if 'updated_c5' not in locals():
        best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
