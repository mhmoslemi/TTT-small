import numpy as np
import time
from math import ceil
import cvxpy as cp


def compute_overlap(h, dx):
    ones = np.ones_like(h)
    ones_h = 1 - h
    corr = np.correlate(h, ones_h, mode='full')
    return np.max(corr * dx)


def crop_and_reflect(h, n_points, dx):
    """Pad and crop h to have length K * n_points"""
    pad_width = ceil(2 / dx) - n_points
    if pad_width > 0:
        h = np.concatenate([h, np.flip(h)])
    return h[:n_points]


def project_h(feasible_h, n_points):
    """Project h to [0,1]^n and normalize to meet sum(h) = n / 2"""
    feasible_h = np.clip(feasible_h, 0, 1)
    norm = feasible_h.sum()
    if norm == 0:
        return np.zeros(n_points)
    beta = n_points / (2 * norm)
    return beta * feasible_h


def gradient(h, index_k, n_points, dx):
    """Compute gradient for the shift index_k"""
    grad = np.zeros(n_points)
    positions = np.arange(n_points)
    right = positions + index_k
    right_clipped = right[right < n_points]
    left = positions - index_k
    left_clipped = left[left >= 0]
    h_right = h[right_clipped]
    h_left = h[left_clipped]

    grad[right_clipped] -= (1 - h_right)
    grad[left_clipped] -= h_left
    return grad


def run(seed=42, budget_s=1100, patience=200, learning_rate=0.05, n_points=80):
    if n_points % 2 != 0:
        n_points += 1  # Ensure even for symmetric padding
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Use provided initial_h_values if it exists
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.array([0.5] * n_points)
    initial_h = initial_h[:n_points]
    h = project_h(initial_h, n_points)

    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    best_time = time.time()
    current_h = h.copy()
    no_improvement_count = 0

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute correlation and find max_k
        h_corr = project_h(current_h, n_points)
        corr = np.correlate(h_corr, 1 - h_corr, mode='full')
        C = corr * dx
        index_k_argmax = np.argmax(C) - n_points + 1
        # Avoid edge issues, search local neighborhood
        kt = np.argwhere(C == np.max(C[:n_points])).ravel()
        for mi in kt:
            if mi == n_points:
                break
            index_k = n_points - mi
            break

        # Compute gradient for worst shift
        grad = gradient(h_corr, index_k, n_points, dx)

        # Gradient step
        feasible_h = h_corr - learning_rate * grad
        feasible_h = project_h(feasible_h, n_points)

        # Update C₅
        new_c5 = compute_overlap(feasible_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = feasible_h.copy()
            best_time = time.time()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        h_corr = feasible_h.copy()

        if no_improvement_count > patience:
            learning_rate *= 0.5
            no_improvement_count = 0
            if learning_rate < 1e-5:
                break  # Exit if no progress and too small learning rate

    return best_h.tolist(), best_c5, n_points
