        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'disp': False}
    )
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        best_h = result.x
        c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        return (best_h, c5_bound, n_points)

    remaining_time = budget_s - elapsed
    if remaining_time > 0:
        for _ in range(3):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand / np.sum(h_rand) * target_sum
            h_rand = np.clip(h_rand, 0, 1)
            result_rand = scipy.optimize.minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'disp': False}
            )
            if result_rand.success and result_rand.fun < result.fun:
                result = result_rand
            elapsed_rand = time.time() - start_time
            if elapsed_rand >= budget_s:
                break

    best_h = result.x
    c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, c5_bound, n_points)
