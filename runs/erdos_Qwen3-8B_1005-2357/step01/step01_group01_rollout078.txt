            return np.max(corr) * dx
        
        def constraint_slsqp_high(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds_slsqp_high = [(0, 1)] * n_points
        constraints_slsqp_high = [{'type': 'eq', 'fun': constraint_slsqp_high}]
        
        try:
            result_slsqp_high = opt.minimize(
                fun=objective_slsqp_high,
                x0=best_h,
                method='SLSQP',
                bounds=bounds_slsqp_high,
                constraints=constraints_slsqp_high,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_h = result_slsqp_high.x
        except:
            best_h = best_h
        
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        scale = (n_points / 2) / total
        best_h = best_h * scale
        c5_bound = objective_slsqp_high(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    return (best_h, c5_bound, n_points)
