import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

    start_time = time.time()
    elapsed = 0.0

    def objective_de(h_vec):
        nonlocal elapsed
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed > budget_s * 0.8:
            raise KeyboardInterrupt
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e5 * abs(np.sum(h_vec) - n_points / 2)
        return max_corr + penalty

    bounds = [(0, 1)] * n_points
    try:
        result_de = opt.differential_evolution(
            fun=objective_de,
            bounds=bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-6,
            maxiter=500,
            seed=seed
        )
    except:
        result_de = opt.OptimizeResult(x=h)
