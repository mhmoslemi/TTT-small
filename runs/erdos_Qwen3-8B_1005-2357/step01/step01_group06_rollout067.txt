import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 100
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.zeros(n_points)
            h[0] = 1.0
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_flat):

        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = target_sum / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.7, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False
    )
    elapsed = time.time() - start_time

    best_h = result_de.x
    if elapsed < budget_s:
        def objective_slsqp(h_flat):
            h = np.clip(h_flat, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                return np.inf
            scale = target_sum / sum_h
            h_scaled = np.clip(h * scale, 0, 1)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx

            h = np.clip(h_flat, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                return np.inf
            scale = target_sum / sum_h
            h_scaled = np.clip(h * scale, 0, 1)
            return np.sum(h_scaled) - target_sum

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

        result_slsqp = scipy.optimize.minimize(
