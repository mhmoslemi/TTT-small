        if sum_clipped == 0:
            return np.zeros_like(h)
        factor = 1.0 / sum_clipped
        return h_clipped * factor

    def objective_de(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.3:
        best_h = initial_h
    else:
        try:
            result_de = differential_evolution(objective_de, bounds, strategy='best1bin',
                                              popsize=50, mutation=(0.8, 1.0), recombination=0.9,
                                              tol=1e-6, maxiter=100, seed=seed, disp=False)
            best_h = result_de.x
        except:
            best_h = initial_h

    def objective_slsqp(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    def gradient_slsqp(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            if 0 <= i_plus_k < n_points and 0 <= i_minus_k < n_points:
                grad[i] = dx * ((1 - h_repaired[i_plus_k]) - h_repaired[i_minus_k])
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1})
    bounds_slsqp = [(0, 1)] * n_points

    best_h_repaired = repair(best_h)
    result_slsqp = minimize(objective_slsqp, best_h_repaired, method='SLSQP',
                            jac=gradient_slsqp, bounds=bounds_slsqp, constraints=cons,
                            tol=1e-6, options={'maxiter': 100})

    best_h = result_slsqp.x
    best_c5 = objective_slsqp(best_h)

    best_h_repaired = repair(best_h)
    best_sum = np.sum(best_h_repaired) * dx
    if abs(best_sum - 1) > 1e-5:
        factor = 1 / best_sum
        best_h_repaired = best_h_repaired * factor

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_repaired, best_c5, n_points

    return best_h_repaired, best_c5, n_points
