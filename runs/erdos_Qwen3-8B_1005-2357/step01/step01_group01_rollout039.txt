    try:
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
    except:
        best_h = h
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    c5_bound = objective(best_h)
    
    if elapsed < budget_s * 0.9:
        n_points = 100
        dx = 2.0 / n_points
        x45 = np.linspace(0, 2, 45)
        x100 = np.linspace(0, 2, 100)
        h100_init = np.interp(x100, x45, best_h)
        total = np.sum(h100_init)
        scale = (n_points / 2) / total
        h100_init = h100_init * scale
        h100_init = np.clip(h100_init, 0, 1)
        
        def objective100(h_vec):
            nonlocal elapsed
            current_time = time.time()
            elapsed = current_time - start_time
            if elapsed > budget_s:
                raise KeyboardInterrupt
            k = 1 - h_vec
            corr = np.correlate(h_vec, k, mode='full')
            return np.max(corr) * dx
        
        def constraint100(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds100 = [(0, 1)] * n_points
        constraints100 = [{'type': 'eq', 'fun': constraint100}]
        
        try:
            result100 = opt.minimize(
                fun=objective100,
                x0=h100_init,
                method='SLSQP',
                bounds=bounds100,
                constraints=constraints100,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_h = result100.x
        except:
            best_h = h100_init
        
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        scale = (n_points / 2) / total
        best_h = best_h * scale
        c5_bound = objective100(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    return (best_h, c5_bound, n_points)
