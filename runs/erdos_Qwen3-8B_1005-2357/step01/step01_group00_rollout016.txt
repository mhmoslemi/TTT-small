    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip = h_clip * scale
        return h_clip

    def objective(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx

    def gradient(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cor)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if -1 <= j < n_points:
                grad[i] = 1 - h[j]
        return grad * dx

    def generate_initial(n_points, type='dirichlet'):
        if initial_h is not None and len(initial_h) == n_points:
            return initial_h.copy()
        if type == 'dirichlet':
            return np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        elif type == 'two_peak':
            width = min(10, n_points // 4)
            center1 = 0.5
            center2 = 1.5
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * 2.0 / n_points
                if abs(x - center1) < width or abs(x - center2) < width:
                    h[i] = 1.0
            return h
        elif type == 'uniform':
            return np.full(n_points, n_points / 2) / n_points
        elif type == 'periodic':

            return np.random.rand(n_points) * (n_points / 2)

            constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
