from scipy.optimize import minimize
import time

    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    def constraint_sum(h):
        return np.sum(h) - n / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n)]

    for _ in range(20):  # Increased iterations
        result = minimize(objective, initial_h, method='L-BFGS-B', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:

            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
