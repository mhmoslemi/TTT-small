        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k_max
            if 0 <= i_plus_k < n_points:
                grad[i] += (1 - h_scaled[i_plus_k])
            i_minus_k = i - k_max
            if 0 <= i_minus_k < n_points:
                grad[i] -= h_scaled[i_minus_k]
        return grad * dx

    # SLSQP constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Bounds for SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    # Initial guess is the best from DE
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', bounds=bounds_slsqp, 
                           constraints=cons, jac=grad_slsqp, tol=1e-6, options={'maxiter': 100})

    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = objective_slsqp(best_h_slsqp)

    # Repair the final candidate
    best_h = np.clip(best_h_slsqp, 0, 1)
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1) > 1e-5:
        best_h = best_h * (1 / sum_h)
    # Ensure sum is correct
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1) > 1e-5:
        best_h = best_h * (1 / sum_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5_slsqp, n_points

    return best_h, best_c5_slsqp, n_points
