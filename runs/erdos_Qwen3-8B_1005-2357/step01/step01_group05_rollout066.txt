import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired
    
    def upsample(h_prev, n_prev, n_new):
        x_prev = np.linspace(0, 2, n_prev)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_prev, h_prev)
        return h_new
    
    def objective(h):
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * (2.0 / len(h))
    
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    best_h = initial_h_values if initial_h_values is not None else np.random.rand(47)
    best_h = repair_vector(best_h, 47)
    best_c5 = objective(best_h)
    
    for i, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        if i == 0:
            initial_guess = best_h.copy()
        else:
            prev_best = best_h
            initial_guess = upsample(prev_best, n_values[i-1], n)
            initial_guess = repair_vector(initial_guess, n)
        
        num_restarts = 5
        for _ in range(num_restarts):
            if time_used >= budget_s:
                break
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100}
            )
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            
            h_opt = result.x
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * (2.0 / n)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired
    
    return best_h, best_c5, len(best_h)
