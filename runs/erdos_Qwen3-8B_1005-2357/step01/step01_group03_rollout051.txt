from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():

        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-5:
            factor = 1 / sum_h
            h = h * factor
        return h

    def objective_sym(h_sym):

        h_full = repair(h_full)
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx

        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx
