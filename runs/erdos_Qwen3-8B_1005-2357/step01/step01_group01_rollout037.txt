    h = h * scale
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    bounds = [(0, 1)] * n_points
    
    def objective_de(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1',
        mutation=0.9,
        recombination=0.7,
        maxiter=100,
        popsize=15,
        tol=1e-6,
        vectorize=True,
        polish=True,
        seed=seed
    )
    best_h = result_de.x
    
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h
    best_h = best_h * scale
    best_h = np.clip(best_h, 0, 1)
    
    def objective_slsqp(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_slsqp(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds_slsqp = [(0, 1)] * n_points
    constraints_slsqp = [{'type': 'eq', 'fun': constraint_slsqp}]
    
    result_slsqp = minimize(
        fun=objective_slsqp,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraints_slsqp,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    best_h = result_slsqp.x
    
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    best_h = np.clip(best_h, 0, 1)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
