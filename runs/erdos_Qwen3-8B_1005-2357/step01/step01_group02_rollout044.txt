        pop[worst_idx] = child
    
    best_bin = min(pop, key=fitness)
    best_h = best_bin.astype(float)
    best_c5 = compute_c5(best_h, n_points)
    
    # Continuous optimization for 45 points
    def optimize(h_start):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(lambda x: compute_c5(x, n_points), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h_start
    
    candidates = [best_h, initial_h]
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        h = candidates[-1] + perturb
        h = np.clip(h, 0, 1)
        delta = target_sum - np.sum(h)
        h[-1] += delta
        h = np.clip(h, 0, 1)
        candidates.append(h)
    
    for h in candidates:
        h_opt = optimize(h)
        c5 = compute_c5(h_opt, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Continuous optimization for 90 points
    n_points_90 = 90
    target_sum_90 = n_points_90 / 2
    dx_90 = 2.0 / n_points_90
    
    def optimize_90(h_start):
        bounds = [(0, 1)] * n_points_90
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_90}]
        res = minimize(lambda x: compute_c5(x, n_points_90), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h_start
    
    starts = []
    for _ in range(5):
        h = np.random.rand(n_points_90)
        delta = target_sum_90 - np.sum(h)
        h[-1] += delta
        h = np.clip(h, 0, 1)
        starts.append(h)
    
    for h in starts:
        h_opt = optimize_90(h)
        c5 = compute_c5(h_opt, n_points_90)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            n_points = n_points_90
    
    return (best_h, best_c5, n_points)
