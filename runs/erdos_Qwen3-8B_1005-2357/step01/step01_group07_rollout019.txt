import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(45)
        initial_h[:22] = 1.0
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / len(initial_h))
    start_time = time.time()
    
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    grid_sizes = [50, 60, 70, 80, 90, 100, 120, 140, 160, 200]
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        initial_conditions = []
        h_step = np.zeros(n)
        h_step[:n//2] = 1.0
        initial_conditions.append(h_step)
        h_perm = h_step.copy()
        np.random.seed(seed)
        np.random.shuffle(h_perm)
        initial_conditions.append(h_perm)
        h_binary = np.zeros(n)
        h_binary[::2] = 1.0
        if n % 2 == 1:
            h_binary[-1] = 0.5
        initial_conditions.append(h_binary)
        for h_initial in initial_conditions:
            def objective(h):
                return compute_c5(h)
            def gradient(h):
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
                return grad
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            bounds = [(0, 1) for _ in range(n)]
            result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                current_h = result.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        if elapsed + 300 <= budget_s:
            def objective_de(h):
                return compute_c5(h)
            bounds_de = [(0, 1) for _ in range(n)]
            cons_de = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            result_de = differential_evolution(objective_de, bounds_de, constraints=cons_de, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
            if result_de.success:
                current_h = result_de.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
