import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 90
    target_sum = n_points / 2
    initial_h = initial_h_values
    
    # Rescale initial_h to n_points
    if len(initial_h) != n_points:
        initial_h = np.repeat(initial_h, 2)
        initial_h = initial_h[:n_points]
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, n_points)
    
    generations = 20
    for _ in range(generations):
        candidates = []
        # Perturb best_h
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        delta = target_sum - current_sum
        perturbed[-1] += delta
        perturbed = np.clip(perturbed, 0, 1)
        candidates.append(perturbed)
        # Generate 4 random candidates
        for _ in range(4):
            rand_h = np.random.uniform(0, 1, n_points)
            current_sum = np.sum(rand_h)
            delta = target_sum - current_sum
            rand_h[-1] += delta
            rand_h = np.clip(rand_h, 0, 1)
            candidates.append(rand_h)
        
        for candidate in candidates:
            def objective(h):
                return compute_c5(h, n_points)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            
            result = minimize(objective, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
            if result.success:

                current_c5 = compute_c5(current_h, n_points)
                if current_c5 < best_c5:
