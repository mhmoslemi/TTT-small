        dx = 2.0 / n
        if n == 45:
            h_initial = initial_h.copy()
        else:
            k = n // 2
            h_initial = np.zeros(n)
            h_initial[:k] = 1.0
            if n % 2 == 1:
                h_initial[-1] += 0.5
        bounds = [(0.0, 1.0) for _ in range(n)]
        def objective_de(h):
            conv = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            penalty = 1e4 * abs(np.sum(h) - n / 2)
            return c5 + penalty
        result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.8, tol=1e-6, maxiter=100, seed=seed)
        h_de = result_de.x
        if np.abs(np.sum(h_de) - n / 2) > 1e-5:
            scale = (n / 2) / np.sum(h_de)
            h_de = h_de * scale
        conv = np.correlate(h_de, 1 - h_de, mode='full')
        c5_de = np.max(conv) * dx
        if c5_de < best_c5:
            best_h = h_de.copy()
            best_c5 = c5_de
        def objective_sl(h):
            conv = np.correlate(h, 1 - h, mode='full')
            return np.max(conv) * dx
        def grad_sl(h):
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                grad[i] = (objective_sl(h_plus) - objective_sl(h_minus)) / (2 * eps)
            return grad
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
        bounds_sl = [(0.0, 1.0) for _ in range(n)]
        result_sl = minimize(objective_sl, h_de, method='SLSQP', jac=grad_sl, bounds=bounds_sl, constraints=cons, tol=1e-6)
        if result_sl.success:
            h_sl = result_sl.x
            conv = np.correlate(h_sl, 1 - h_sl, mode='full')
            c5_sl = np.max(conv) * dx
            if c5_sl < best_c5:
                best_h = h_sl.copy()
                best_c5 = c5_sl
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, len(best_h))
