        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)

            scale = (n_points/2) / total
            h_clip *= scale
        return h_clip

        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        return np.max(cor) * dx
    
    def gradient(h, n_points):
        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        max_val = np.max(cor)
        max_indices = np.where(np.isclose(cor, max_val, atol=1e-5))[0]
        grad = np.zeros(n_points)
        for idx in max_indices:
            k0 = idx - (n_points-1)
            for i in range(n_points):
                term1 = 1 - h[(i + k0) % n_points] if (0 <= i + k0 < n_points) else 0

                grad[i] += dx * (term1 + term2)
        return grad
