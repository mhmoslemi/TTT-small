    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Objective function for DE
    def objective_de(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]

    # DE phase
    result_de = differential_evolution(objective_de, bounds, strategy='best1', 
                                      mutation=0.8, recombination=0.9, 
                                      popsize=50, tol=1e-6, 
                                      maxiter=100, disp=False)
    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)

    # Repair the best candidate from DE
    best_h_de = np.clip(best_h_de, 0, 1)
    sum_de = np.sum(best_h_de) * dx
    if abs(sum_de - 1) > 1e-5:
        best_h_de = best_h_de * (1 / sum_de)

    # SLSQP phase
    def objective_slsqp(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def gradient_slsqp(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        m_max = np.argmax(corr)
        k_max = m_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            # term1: (1 - h[i + k_max]) * dx if i + k_max is within the array
            if 0 <= i + k_max < n_points:
                term1 = (1 - h[i + k_max]) * dx
            else:
                term1 = 0
            # term2: -h[i - k_max] * dx if i - k_max is within the array
            if 0 <= i - k_max < n_points:
                term2 = -h[i - k_max] * dx
            else:
                term2 = 0
            grad[i] = term1 + term2
        return grad

    # Constraints for SLSQP: sum(h) * dx = 1
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1})

    # Bounds for SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    # Initial guess from DE
    result_slsqp = minimize(objective_slsqp, best_h_de, 
                           method='SLSQP', bounds=bounds_slsqp, 
                           constraints=cons, jac=gradient_slsqp, 
                           tol=1e-6, maxiter=100)

    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = objective_slsqp(best_h_slsqp)

    # Repair the final solution
    best_h_slsqp = np.clip(best_h_slsqp, 0, 1)
    sum_final = np.sum(best_h_slsqp) * dx
    if abs(sum_final - 1) > 1e-5:
        best_h_slsqp = best_h_slsqp * (1 / sum_final)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_slsqp, best_c5_slsqp, n_points

    return best_h_slsqp, best_c5_slsqp, n_points
