    result_sym = differential_evolution(objective_sym, bounds_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_c5_sym = result_sym.fun
    best_h_sym = result_sym.x
    best_h_full_sym = np.concatenate([best_h_sym, best_h_sym[::-1]])

    elapsed_sym = time.time() - start_time
    if elapsed_sym > budget_s:
        return best_h_full_sym, best_c5_sym, n_points

    # Non-symmetric optimization
    bounds_non_sym = [(0, 1)] * n_points
    result_non_sym = differential_evolution(objective_non_sym, bounds_non_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_c5_non_sym = result_non_sym.fun
    best_h_full_non_sym = result_non_sym.x

    elapsed_non_sym = time.time() - start_time - elapsed_sym
    if elapsed_non_sym > budget_s - elapsed_sym:
        best_h_full = best_h_full_sym
        best_c5 = best_c5_sym
    else:
        if best_c5_non_sym < best_c5_sym:
            best_h_full = best_h_full_non_sym
            best_c5 = best_c5_non_sym
        else:
            best_h_full = best_h_full_sym
            best_c5 = best_c5_sym

    # Final repair
    best_h_full = repair(best_h_full, target_sum_non_sym, n_points)
    best_sum = np.sum(best_h_full) * dx
    if abs(best_sum - 1) > 1e-5:
        factor = 1 / best_sum
        best_h_full = best_h_full * factor

    return best_h_full, best_c5, n_points
