    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():

    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_initial = np.clip(h_initial, 0, 1)

    # Repair function
    def repair(h):

        return h * scale

        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx
