import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_random_feasible(n_points, target_sum):
    h = np.random.rand(n_points)
    h += (target_sum - np.sum(h)) / n_points
    h = np.clip(h, 0, 1)
    h[-1] += (target_sum - np.sum(h))
    h = np.clip(h, 0, 1)
    return h

def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    delta = target_sum - np.sum(h)
    if delta:
        h[-1] += delta
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2

    population = [initial_h_values]
    for _ in range(19):
        population.append(generate_random_feasible(n_points, target_sum))

    best_h = initial_h_values
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()

    for gen in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, n_points) for h in population]
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx]

        new_population = []
        for _ in range(20):
            tournament = random.sample(population, 3)
            winner = min(tournament, key=lambda h: compute_c5(h, n_points))
            new_population.append(winner)

        for i in range(0, 20, 2):
            if random.random() < 0.8:
                p1, p2 = new_population[i], new_population[i+1]
                split = np.random.randint(1, n_points-1)
                c1, c2 = np.concatenate([p1[:split], p2[split:]]), np.concatenate([p2[:split], p1[split:]])
                new_population[i], new_population[i+1] = c1, c2
