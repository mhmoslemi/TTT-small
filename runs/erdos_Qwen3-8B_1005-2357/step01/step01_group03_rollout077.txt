    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Repair function
    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if abs(sum_h - target_sum) < 1e-5:
            return h_clipped
        factor = target_sum / sum_h
        return h_clipped * factor

    # Objective function for DE
    def objective_de(h):
        h_repaired = repair(h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]

    # DE Phase
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        mutation=(0.8, 1.0),
        recombination=0.9,
        popsize=50,
        tol=1e-6,
        maxiter=1000,
        seed=seed
    )
    best_h_de = result_de.x

    # SLSQP Phase
    def objective_slsqp(h):
        h_repaired = repair(h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    def gradient_slsqp(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < n_points:
                grad[i] -= h[i - k]
        grad *= dx
        return grad

    # Initial guess for SLSQP
    result_slsqp = minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        jac=gradient_slsqp,
        bounds=bounds,
        tol=1e-6
    )

    best_h = result_slsqp.x
    best_c5 = objective_slsqp(best_h)
    best_h = repair(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
