from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    tournament_size = 3

    time_limit = 250  # seconds

    # Generate initial population
    population = []
    for _ in range(pop_size):
        num_ones = np.random.choice([22, 23])
        h = np.zeros(n_points, dtype=int)
        idxs = np.random.choice(n_points, size=num_ones, replace=False)
        h[idxs] = 1
        population.append(h)

    # Evaluate initial population
    fitness = [compute_c5(h.astype(float), n_points) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].astype(float)
    best_c5 = fitness[best_idx]

    start_time = time.time()
    while time.time() - start_time < time_limit and gen_limit > 0:
        # Tournament selection
        parents = []
        for _ in range(pop_size // 2):
            t = np.random.choice(pop_size, tournament_size)
            sel = t[np.argmin([fitness[i] for i in t])]
            parents.append(population[sel])
        
        # Crossover
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < crossover_prob:
                pt = np.random.randint(1, n_points)
                c1 = np.concatenate([p1[:pt], p2[pt:]])
                c2 = np.concatenate([p2[:pt], p1[pt:]])
                new_pop.extend([c1, c2])
            else:
                new_pop.extend([p1, p2])
        
        # Mutation and repair
        for i in range(len(new_pop)):
            h = new_pop[i]
            if np.random.rand() < mutation_prob:
                idx = np.random.randint(n_points)
                h[idx] ^= 1
            ones = np.sum(h)
            if ones < 22:
                idx = np.random.randint(n_points)
                h[idx] = 1
            elif ones > 23:
                idx = np.random.randint(n_points)
                h[idx] = 0
            new_pop[i] = h
        
        # Evaluate new population
        new_fitness = [compute_c5(h.astype(float), n_points) for h in new_pop]
        population = new_pop
        fitness = new_fitness
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].astype(float)
        best_c5 = fitness[best_idx]
        gen_limit -= 1

    # Continuous optimization
    def objective(h):
        return compute_c5(h, n_points)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Run on best binary and initial
    result_binary = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    result_initial = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    final_c5 = min(result_binary.fun, result_initial.fun)
    final_h = result_binary.x if result_binary.success else result_initial.x

    return (final_h, final_c5, n_points)
