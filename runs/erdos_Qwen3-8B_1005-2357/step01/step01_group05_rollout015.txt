import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    for idx, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        if idx == 0 and initial_h_values is not None:
            initial_guess = initial_h_values.copy()
        else:
            if idx == 0:
                initial_guess = np.random.rand(n)
            else:
                prev_n = n_values[idx-1]
                prev_h = best_h
                if prev_h is None:
                    initial_guess = np.random.rand(n)
                else:
                    initial_guess = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, prev_n), prev_h)
            initial_guess = np.clip(initial_guess, 0, 1)
            s = np.sum(initial_guess)
            if s == 0:
                scale = 0
            else:
                scale = (n / 2) / s
            initial_guess = initial_guess * scale
            initial_guess = repair_vector(initial_guess, n)
        
        dx = 2.0 / n
        def objective(h):
            cor = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(cor) * dx
            return max_overlap
        
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
        bounds = [(0, 1) for _ in range(n)]
        
        if n <= 100:
            method = 'L-BFGS-B'
            options = {'maxiter': 200}
        else:
            method = 'SLSQP'
            options = {'maxiter': 100}
        
        start_time = time.time()
        result = minimize(
            fun=objective,
            x0=initial_guess,
            method=method,
            bounds=bounds,
            constraints=constraints,
            jac=None,
            options=options
        )
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
        
        h_opt = result.x
        h_repaired = repair_vector(h_opt, n)
        cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        c5 = np.max(cor) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_repaired
    
    return best_h, best_c5, len(best_h)
