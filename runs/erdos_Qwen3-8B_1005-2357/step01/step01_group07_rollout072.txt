            else:
                h_step[:n//2] = 1.0
                h_step[n//2] = 0.5
            initial_guess = h_step
        def objective_de(h):
            c5 = compute_c5(h)
            penalty = 1e4 * abs(np.sum(h) - n/2)
            return c5 + penalty
        bounds = [(0, 1) for _ in range(n)]
        try:
            result = differential_evolution(objective_de, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        except:
            result = differential_evolution(objective_de, bounds, strategy='best1bin', maxiter=50, popsize=10, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        best_candidate = result.x
        def objective_slsqp(h):
            return compute_c5(h)
        def constraint_sum(h):
            return np.sum(h) - n/2
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds_slsqp = [(0, 1) for _ in range(n)]
        try:
            result_slsqp = minimize(objective_slsqp, best_candidate, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-6)
        except:
            result_slsqp = minimize(objective_slsqp, best_candidate, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-4)
        refined_h = result_slsqp.x
        current_c5 = compute_c5(refined_h)
        if current_c5 < best_c5:
            best_h = refined_h
            best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, len(best_h))
