    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(100)
        h = h / np.sum(h) * 50
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def repair(h_vec, n_points):
        h = np.clip(h_vec, 0, 1)

            h = h * scale

        corr = np.correlate(h, 1 - h, mode='full')
