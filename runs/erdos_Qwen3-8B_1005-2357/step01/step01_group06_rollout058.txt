import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 100
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():

        return np.max(corr) * dx
