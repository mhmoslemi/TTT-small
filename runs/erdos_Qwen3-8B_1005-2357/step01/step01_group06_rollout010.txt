import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():

        h = np.random.rand(n_points)

        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)

    # Multi-start optimization
    for _ in range(5):
        # Perturb the current best solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * target_sum

        result = scipy.optimize.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elif result.status == 1:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, n_points)
