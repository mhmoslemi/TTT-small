import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_list = [100, 200, 400]
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def objective(h_vec, n, dx, target_sum):
        h = np.clip(h_vec, 0, 1)
        s = np.sum(h)
        if abs(s - target_sum) > 1e-6:
            h *= target_sum / s
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vec, target_sum):
        return np.sum(h_vec) - target_sum

    for n in n_list:
        if time.time() - start_time > budget_s:
            break

        if n == 100 and 'initial_h_values' in globals():

                prev_h = best_h if best_h is not None else np.random.rand(prev_n)
                x_prev = np.linspace(0, 2, prev_n)
                x_new = np.linspace(0, 2, n)
                h = np.interp(x_new, x_prev, prev_h)
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)

        bounds = [(0, 1)] * n
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, target_sum)}]
        result = opt.minimize(
