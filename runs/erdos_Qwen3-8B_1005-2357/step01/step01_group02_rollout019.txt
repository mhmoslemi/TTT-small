import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_random_feasible(n_points, target_sum):
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    delta = target_sum - np.sum(h)
    h[-1] += delta
    h = np.clip(h, 0, 1)
    h[-1] = np.clip(h[-1], 0, 1)
    h[-1] += (target_sum - np.sum(h))  # Final adjustment
    return h

def repair(h, target_sum):

    delta = target_sum - np.sum(h)

    h[-1] += delta
    h = np.clip(h, 0, 1)
    return h
