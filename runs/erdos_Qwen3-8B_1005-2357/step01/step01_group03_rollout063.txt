from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):

    def objective(h):

                                      recombination=0.9, tol=1e-6, maxiter=1000,

    ga_elapsed = time.time() - ga_start
