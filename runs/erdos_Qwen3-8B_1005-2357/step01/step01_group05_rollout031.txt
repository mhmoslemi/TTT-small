                return np.repeat(prev_h, 2)
        x = np.random.rand(n)
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            scale = 0
        else:
            scale = (n / 2) / s
        x_scaled = x * scale
        x_repaired = repair_vector(x_scaled, n)
        return x_repaired
    
    prev_best_h = None
    for n in n_values:
        if time_used >= budget_s:
            break
        if n == 47:
            if initial_h_values is not None:
                initial_guess = initial_h_values.copy()
            else:
                initial_guess = generate_initial_guess(n)
        else:
            initial_guess = generate_initial_guess(n, prev_best_h)
        
        best_h_for_this_grid = None
        best_c5_for_this_grid = float('inf')
        num_restarts = 5
        for _ in range(num_restarts):
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='trust-constrained',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 100}
            )
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            
            h_opt = result.x
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx
            
            if c5 < best_c5_for_this_grid:
                best_c5_for_this_grid = c5
                best_h_for_this_grid = h_repaired
        
        if best_c5_for_this_grid < best_c5:
            best_c5 = best_c5_for_this_grid
            best_h = best_h_for_this_grid
        
        prev_best_h = best_h_for_this_grid
    
    return best_h, best_c5, len(best_h)
