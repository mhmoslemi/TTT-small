import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial_non_sym = initial_h_values
        h_initial_sym = h_initial_non_sym[:n_points//2]
    else:
        h_initial_sym = np.random.rand(n_points//2)
        h_initial_sym = h_initial_sym / np.sum(h_initial_sym) * (n_points / 4)

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h) * dx
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        return h * scale

    def objective(x):
        if len(x) == 50:
            h_full = np.concatenate([x, x[::-1]])
        else:
            h_full = x
        h_repaired = repair(h_full)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    bounds_sym = [(0,1)] * 50
    result_sym = differential_evolution(objective, bounds_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h_sym = result_sym.x
    best_c5_sym = objective(best_h_sym)
    best_h_full_sym = np.concatenate([best_h_sym, best_h_sym[::-1]])
    best_h_full_sym = repair(best_h_full_sym)

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        bounds_non_sym = [(0,1)] * n_points
        result_non_sym = differential_evolution(objective, bounds_non_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        best_h_non_sym = result_non_sym.x
        best_c5_non_sym = objective(best_h_non_sym)
        best_h_full_non_sym = repair(best_h_non_sym)

        best_h_full = best_h_full_sym if best_c5_sym <= best_c5_non_sym else best_h_full_non_sym
        best_c5 = best_c5_sym if best_c5_sym <= best_c5_non_sym else best_c5_non_sym
    else:
        best_h_full = best_h_full_sym
        best_c5 = best_c5_sym

    best_h_full = repair(best_h_full)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_full, best_c5, n_points

    return best_h_full, best_c5, n_points
