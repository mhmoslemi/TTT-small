import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(h, n):
        s = np.sum(h)
        target = n / 2
        delta = target - s
        h_repaired = h + delta / n
        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired
    
    def upsample_vector(old_h, old_n, new_n):
        x_new = np.linspace(0, 2, new_n)
        x_old = np.linspace(0, 2, old_n)
        new_h = np.interp(x_new, x_old, old_h)
        return repair_vector(new_h, new_n)
    
    for n in n_values:
        if time_used >= budget_s:
            break
        num_restarts = 5
        initial_guesses = []
        if n == 47 and initial_h_values is not None:
            initial_guesses.append(initial_h_values)
        else:
            if best_h is not None and len(best_h) < n:
                initial_guesses.append(upsample_vector(best_h, len(best_h), n))

                    scale = 0
                else:
                    scale = (n / 2) / s
                x_scaled = x * scale
                x_repaired = repair_vector(x_scaled, n)
                initial_guesses.append(x_repaired)
        
        for guess in initial_guesses:
            if time_used >= budget_s:
                break
            guess = repair_vector(guess, n)
            dx = 2.0 / n
            
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                return np.max(cor) * dx
            
            def constraint(h):
                return np.sum(h) - n / 2
            
            bounds = [(0, 1)] * n
            constraints = [{'type': 'eq', 'fun': constraint}]
            
            if n <= 100:
                method = 'trust-constrained'
                options = {'maxiter': 50}
            else:
                method = 'SLSQP'
                options = {'maxiter': 100}
            
            start_time = time.time()
            try:

            time_used += time.time() - start_time
