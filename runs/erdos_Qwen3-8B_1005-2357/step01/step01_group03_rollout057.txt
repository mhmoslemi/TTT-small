from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):

    target_sum_sym = n_points / 4

        h_non_sym = np.concatenate([h_sym, h_sym[::-1]])

    def repair_sym(sym_h):
        sym_clipped = np.clip(sym_h, 0, 1)
        if np.sum(sym_clipped) == 0:
            return np.zeros_like(sym_clipped)
        scale = target_sum_sym / np.sum(sym_clipped)
        return sym_clipped * scale
