import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_random_feasible(n_points, target_sum):
    h = np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    h[-1] += delta
    h = np.clip(h, 0, 1)

    delta = target_sum - current_sum
    h[-1] += delta
    h = np.clip(h, 0, 1)
    return h
