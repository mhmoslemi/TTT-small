                    # Refine with SLSQP
                    def objective_slsqp(h):
                        return compute_c5(h)
                    def constraint_sum(h):
                        return np.sum(h) - n / 2
                    cons = [{'type': 'eq', 'fun': constraint_sum}]
                    bounds_slsqp = [(0.0, 1.0) for _ in range(n)]
                    result_slsqp = minimize(
                        objective_slsqp,
                        adjusted_h,
                        method='SLSQP',
                        bounds=bounds_slsqp,
                        constraints=cons,
                        tol=1e-6
                    )
                    if result_slsqp.success:
                        refined_h = result_slsqp.x
                        refined_c5 = compute_c5(refined_h)
                        if refined_c5 < best_c5:
                            best_c5 = refined_c5
                            best_h = refined_h.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, len(best_h))
