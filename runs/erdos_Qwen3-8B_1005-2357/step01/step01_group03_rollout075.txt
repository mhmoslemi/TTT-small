                                      vectorized=True)
    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.7:
        # Not enough time for SLSQP, return DE result
        best_h_de = np.clip(best_h_de, 0, 1)
        best_h_de = best_h_de * (target_sum / np.sum(best_h_de))
        return best_h_de, best_c5_de, n_points

    # Repair the DE result
    best_h_de = np.clip(best_h_de, 0, 1)
    best_h_de = best_h_de * (target_sum / np.sum(best_h_de))

    # SLSQP phase
    def objective_slsqp(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def gradient_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        s = m - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + s < len(h):
                grad[i] = (1 - h[i + s]) * dx
            else:
                grad[i] = 0
        return grad

    # Constraints for SLSQP: sum(h) = target_sum
    cons_slsqp = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Bounds for SLSQP
    bounds_slsqp = [(0, 1)] * n_points

    # Optimization
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', 
                           jac=gradient_slsqp, bounds=bounds_slsqp, 
                           constraints=cons_slsqp, tol=1e-6)

    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = objective_slsqp(best_h_slsqp)

    # Repair the final result
    best_h_slsqp = np.clip(best_h_slsqp, 0, 1)
    best_h_slsqp = best_h_slsqp * (target_sum / np.sum(best_h_slsqp))

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_slsqp, best_c5_slsqp, n_points

    return best_h_slsqp, best_c5_slsqp, n_points
