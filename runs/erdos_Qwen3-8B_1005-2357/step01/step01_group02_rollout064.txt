import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    best_c5 = float('inf')
    best_h = initial_h.copy()
    time_start = time.time()
    
    for _ in range(20):  # 20 generations
        current_population = [best_h]
        for _ in range(4):  # 4 new candidates per generation
            perturbed = best_h + np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_population.append(perturbed)
        
        for candidate in current_population:

                return compute_c5(h, n_points)
