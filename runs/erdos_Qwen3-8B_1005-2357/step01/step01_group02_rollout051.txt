import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_random_feasible(n_points, target_sum):
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    delta = target_sum - np.sum(h)
    h[-1] += delta
    h = np.clip(h, 0, 1)
    h[-1] = np.clip(h[-1], 0, 1)
    h[-1] += (target_sum - np.sum(h))  # Final adjustment
    return h

def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    delta = target_sum - np.sum(h)
    if abs(delta) < 1e-6:
        return h
    h[-1] += delta
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = repair(initial_h, target_sum)
    
    population = [initial_h]
    for _ in range(19):
        h = generate_random_feasible(n_points, target_sum)
        population.append(h)
    
    best_h = initial_h
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    
    for gen in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, n_points) for h in population]
        ranked = sorted(zip(fitness, population), key=lambda x: x[0])
        selected = [h for _, h in ranked[:10]]
        
        new_population = []
        for _ in range(10):
            a, b = random.sample(selected, 2)
            alpha = np.random.rand()
            child = alpha * a + (1 - alpha) * b
            child = repair(child, target_sum)
            new_population.append(child)
        
        for h in new_population:
            def objective(h_vec):
                return compute_c5(h_vec, n_points)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                new_h = result.x
                new_c5 = compute_c5(new_h, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
