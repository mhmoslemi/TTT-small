import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_random_feasible(n_points, target_sum):
    h = np.random.uniform(0, 1, n_points)
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return np.zeros(n_points)
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def repair(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped
    delta = target_sum - current_sum
    h_clipped[-1] += delta
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped
