import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 100
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h
    best_c5 = objective(h)

    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the current best solution
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(h + perturb, 0, 1)
        # Adjust sum to target
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed != target_sum:
            scale = target_sum / sum_perturbed
            perturbed_h *= scale
        # Run optimization
        result = scipy.optimize.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            current_h = perturbed_h
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    # Final evaluation
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
