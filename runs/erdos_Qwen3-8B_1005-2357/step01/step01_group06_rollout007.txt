import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 100
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'ftol': 1e-8, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = result.x
        c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        return (best_h, c5_bound, n_points)

    # Fallback to initial if optimization stalls
    best_h = result.x
    c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, c5_bound, n_points)
