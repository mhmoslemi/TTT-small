    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip = h_clip * scale
        return h_clip
    
    def objective(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx
    
    def gradient(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cor)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if -1 <= j < n_points:
                grad[i] = 1 - h[j]
        return grad * dx
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break

            h_initial = initial_h.copy()
        else:
            h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        
        def objective_func(h):
            h = repair(h, n_points)
            return objective(h, n_points)
        
        def grad_func(h):
            h = repair(h, n_points)
            return gradient(h, n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        try:
            result = opt.minimize(
                fun=objective_func,
                x0=h_initial,
                method='trust-constr',

            )
        except:
