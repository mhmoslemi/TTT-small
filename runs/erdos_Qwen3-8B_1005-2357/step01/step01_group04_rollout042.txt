            generations_new = 100
            population_new = np.random.rand(pop_size_new, n_points_new)
            population_new = np.apply_along_axis(lambda x: x * target_sum_new / np.sum(x), axis=1, arr=population_new)
            for gen in range(generations_new):
                fitness_new = np.array([objective_new(ind) for ind in population_new])
                parents_new = []
                for _ in range(pop_size_new):
                    idxs = np.random.choice(pop_size_new, 3, replace=False)
                    best_idx = idxs[np.argmin(fitness_new[idxs])]
                    parents_new.append(population_new[best_idx])
                new_population_new = []
                for i in range(0, pop_size_new, 2):
                    if i + 1 < pop_size_new:
                        parent1 = parents_new[i]
                        parent2 = parents_new[i+1]
                        child1 = np.where(np.random.rand(n_points_new) < 0.5, parent1, parent2)
                        child2 = np.where(np.random.rand(n_points_new) < 0.5, parent2, parent1)
                        new_population_new.extend([child1, child2])
                    else:
                        new_population_new.append(parents_new[i])
                for i in range(len(new_population_new)):
                    if np.random.rand() < 0.1:
                        mutation = np.random.normal(0, 0.1, n_points_new)
                        new_population_new[i] += mutation
                        new_population_new[i] = repair(new_population_new[i])
                population_new = new_population_new
            best_idx_new = np.argmin([objective_new(ind) for ind in population_new])
            best_h_new = population_new[best_idx_new]
            bounds_new = [(0, 1)] * n_points_new
            cons_new = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_new}]
            result_new = opt.minimize(objective_new, best_h_new, method='L-BFGS-B', bounds=bounds_new, constraints=cons_new)
            best_h_new = result_new.x
            c5_new = objective_new(best_h_new)
            if c5_new < c5:
                c5 = c5_new
                best_h = best_h_new
                n_points = n_points_new
    best_h = repair(best_h)
    return (best_h, c5, n_points)
