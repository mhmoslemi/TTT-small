import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Use provided initial_h_values if available
    if 'initial_h_values' in locals():
        initial_h = initial_h_values
    else:
        # Fallback to a simple uniform distribution
        initial_h = np.zeros(94)
        initial_h[::2] = 1.0  # Example initial guess
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h):
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx
    
    # Gradient function
    def gradient(h):
        cor = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(cor)
        k0 = idx - (n_points - 1)
        j = np.arange(n_points) + k0
        mask = (j >= 0) & (j < n_points)
        grad = np.zeros(n_points)
        grad[mask] = 1.0 - h[j[mask]]
        grad[~mask] = 1.0
        return grad * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_h,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        jac=gradient,
        options={'maxiter': 1000, 'gtol': 1e-5}
    )
    
    # Repair solution
    h = result.x
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-5:
        scale = (n_points / 2) / total
        h *= scale
    h = np.clip(h, 0, 1)
    
    # Compute C5
    cor = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(cor) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
        best_n = n_points
    
    return (best_h, best_c5, best_n)
