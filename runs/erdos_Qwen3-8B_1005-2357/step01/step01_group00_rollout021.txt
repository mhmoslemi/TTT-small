import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    best_c5 = float('inf')

    best_n = 0

    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip *= scale
        return h_clip

    def objective(h):
        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        return np.max(cor) * dx

    def gradient(h):
        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        max_idx = np.argmax(cor)
        k0 = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k0
            if -1 < j < n_points:
                grad[i] += dx * (1 - h[j])
            j = i - k0
            if -1 < j < n_points:
                grad[i] += dx * (-h[j])
        return grad

    for n_points in grid_sizes:
        if time.time() - start_time > budget_s:
            break
        if initial_h is not None and len(initial_h) == n_points:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        h_initial = repair(h_initial, n_points)

        def de_objective(h):
            return objective(h)

        bounds = [(0, 1)] * n_points
        result = opt.differential_evolution(de_objective, bounds, strategy='best1bin',

                                            popsize=15, maxiter=100, tol=1e-4, 
                                            recombination=0.5, seed=seed)
        h_de = repair(result.x, n_points)

        def trust_objective(h):
            return objective(h)

        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        result_trust = opt.minimize(trust_objective, h_de, method='trust-constr',
                                    bounds=bounds, constraints=cons, jac=gradient,
