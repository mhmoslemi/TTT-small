        h_binary[:n//2] = 1.0
        current_sum = n//2
        target_sum = n / 2
        if current_sum < target_sum:
            h_binary[-1] += 0.5
        elif current_sum > target_sum:
            h_binary[-1] -= 0.5
        h_permuted = np.random.permutation(h_binary)
        return [h_binary, h_permuted]

    for n in grid_sizes:
        if elapsed >= budget_s:
            break
        initial_conditions = generate_initial_conditions(n)
        for h_initial in initial_conditions:
            bounds = [(0.0, 1.0) for _ in range(n)]
            result = differential_evolution(compute_c5, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
            h_de = result.x
            current_sum = np.sum(h_de)
            scale_factor = (n / 2) / current_sum
            h_de = np.clip(h_de * scale_factor, 0.0, 1.0)
            def objective(h):
                return compute_c5(h)
            def gradient(h):
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
                return grad
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            result_slsqp = minimize(objective, h_de, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
            h_refined = result_slsqp.x
            current_c5 = compute_c5(h_refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined
                n_points = n
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n_points)
