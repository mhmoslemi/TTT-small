    )
    if result_slsqp.success:
        current_h = result_slsqp.x
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    # Check time remaining for larger grid
    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:
        n = 100
        dx = 2.0 / n
        initial_h_large = np.zeros(n)
        initial_h_large[:50] = 0.5
        bounds_large = [(0, 1) for _ in range(n)]

        def objective_penalty_large(h):
            return objective_penalty(h, n, dx)

        result_de_large = differential_evolution(
            objective_penalty_large,
            bounds_large,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        if result_de_large.success:
            current_h_large = result_de_large.x
            current_c5_large = compute_c5(current_h_large)
            if current_c5_large < best_c5:
                best_h = current_h_large
                best_c5 = current_c5_large

    return (best_h, best_c5, len(best_h))
