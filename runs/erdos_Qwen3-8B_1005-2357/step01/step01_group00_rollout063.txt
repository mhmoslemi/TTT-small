import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    initial_h = initial_h_values if 'initial_h_values' in locals() else None

    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip = h_clip * scale
        return h_clip

    def objective(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = cor * dx
        return np.max(overlap)

    def gradient(h, n_points):
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cor)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if -1 <= j < n_points:
                grad[i] = 1 - h[j]
        return grad * dx

    n_points_list = [94, 188, 376, 752]
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break

            h_initial = initial_h.copy()
        else:
            dirichlet = np.random.dirichlet([1.0]*n_points)

        c5 = np.max(overlap)
