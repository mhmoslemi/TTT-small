from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    def objective(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        factor = target_sum / sum_clipped
        return h_clipped * factor

    # Differential Evolution Phase
    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(
        objective, bounds, strategy='best1', mutation=0.8,
        recombination=0.9, popsize=50, tol=1e-6,
        maxiter=1000, disp=False, polish=False,
        callback=lambda x: None
    )
    best_h_de = repair(result_de.x)
    best_c5_de = objective(best_h_de)
    elapsed_de = time.time() - start_time
    if elapsed_de > budget_s:
        return best_h_de, best_c5_de, n_points

    # SLSQP Phase with analytical gradient
    def gradient(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        max_index = np.argmax(corr)
        N = len(h)
        k = max_index - (N - 1)
        grad = np.zeros_like(h)
        for i in range(N):
            if 0 <= i + k < N:
                grad[i] = g[i + k]
        grad *= dx
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_slsqp = [(0, 1)] * n_points
    result_slsqp = minimize(

        bounds=bounds_slsqp, constraints=cons,
        jac=gradient, tol=1e-6, options={'maxiter': 100}
    )
    best_h_slsqp = repair(result_slsqp.x)
    best_c5_slsqp = objective(best_h_slsqp)
    elapsed_total = time.time() - start_time
    if elapsed_total > budget_s:
