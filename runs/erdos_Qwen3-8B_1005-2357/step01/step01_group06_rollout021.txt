import scipy.optimize
import time

        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time.time()
    best_h = h
    best_c5 = objective(h)

    for _ in range(5):  # Multi-start optimization
        if time.time() - start_time > budget_s * 0.8:
            break
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        result = scipy.optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
