import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h_values = initial_h_values
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * (2.0 / len(initial_h_values))
    start_time = time.time()
    elapsed = 0
    
    grid_sizes = [45, 50, 60, 70, 80, 90, 100, 120, 140, 160, 200]
    
    for n in grid_sizes:
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        # Generate initial guesses
        initial_guesses = []
        # Step function
        step = np.zeros(n)
        step[:n//2] = 0.5
        step[n//2:] = 0.5
        initial_guesses.append(step)
        # Random guess
        rand = np.random.rand(n)
        rand = rand * (n / 2) / np.sum(rand)
        initial_guesses.append(rand)
        # Binary guess
        bin_guess = np.zeros(n)
        bin_guess[:n//2] = 1.0
        bin_guess[n//2:] = 0.0
        initial_guesses.append(bin_guess)
        # Use initial_h_values if n matches
        if len(initial_h_values) == n:
            initial_guesses.append(initial_h_values)
        
        for guess in initial_guesses:
            # Define objective with scaling
            def objective(h):
                h_scaled = h * (n / 2) / np.sum(h)
                conv = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                return np.max(conv) * dx
            bounds = [(0, 1) for _ in range(n)]
            # Run DE
            try:
                result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, maxiter=100, seed=seed)
                elapsed = time.time() - start_time
                if elapsed >= budget_s:
                    break
                # Refine with SLSQP
                def obj_slsqp(h):
                    h_scaled = h * (n / 2) / np.sum(h)
                    conv = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                    return np.max(conv) * dx
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
                res_slsqp = minimize(obj_slsqp, result.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if res_slsqp.success:
                    current_h = res_slsqp.x
                    current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
            except:
                pass
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return (best_h, best_c5, len(best_h))
