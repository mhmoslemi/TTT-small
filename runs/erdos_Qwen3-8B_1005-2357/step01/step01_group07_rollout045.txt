import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()
    elapsed = 0
    best_n = n

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - len(h) / 2

    def generate_initial(n, seed=None):
        if seed is not None:
            np.random.seed(seed)
        h = np.zeros(n)
        h[:n//2] = 0.5
        np.random.shuffle(h)
        return h

    def optimize_grid(n, initial=None):
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        if initial is None:
            initial = generate_initial(n, seed)
        result = minimize(objective, initial, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            return result.x
        return initial

    def global_optimize(n, initial):
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = differential_evolution(objective, bounds, constraints=cons, strategy='best1bin', popsize=15, tol=1e-6, maxiter=100)
        return result.x

    for n in [45, 50, 60, 70, 80, 90, 100, 120, 140, 160, 200]:
        if elapsed >= budget_s - 100:
            break
        initial = generate_initial(n, seed)
        local_h = optimize_grid(n, initial)
        local_c5 = compute_c5(local_h)
        if local_c5 < best_c5:
            best_h = local_h
            best_c5 = local_c5
            best_n = n
        elapsed = time.time() - start_time
        if elapsed < budget_s - 100:
            global_h = global_optimize(n, initial)
            global_c5 = compute_c5(global_h)
            if global_c5 < best_c5:
                best_h = global_h
                best_c5 = global_c5
                best_n = n
        elapsed = time.time() - start_time

    return (best_h, best_c5, best_n)
