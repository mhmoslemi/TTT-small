import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_initial = len(initial_h)
    dx = 2.0 / n_initial
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def generate_initial_conditions(n, seed):
        np.random.seed(seed)
        binary = np.zeros(n)
        binary[:n//2] = 1
        if n % 2 == 1:
            binary[-1] = 0.5
        np.random.shuffle(binary)
        step = np.zeros(n)
        step[n//2] = 1
        if n == n_initial:
            perm = np.random.permutation(initial_h)
        else:
            perm = None
        return [binary, step, perm]

    n_values = [45, 100]
    for n in n_values:
        initial_conditions = generate_initial_conditions(n, seed)
        for init in initial_conditions:
            bounds = [(0, 1) for _ in range(n)]
            def objective(h):
                return compute_c5(h)
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            result = minimize(objective, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                current_h = result.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, len(best_h))
