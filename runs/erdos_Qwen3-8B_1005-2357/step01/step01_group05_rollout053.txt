from scipy.optimize import minimize, CMAES
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        delta = target - s
        if np.abs(delta) < 1e-8:
            return x_clipped
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired
    
    for idx, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        time_budget = [500, 300, 250, 50][idx]
        if idx == 0 and initial_h_values is not None:
            initial_guess = initial_h_values.copy()
        else:
            if idx == 0:
                initial_guess = np.random.rand(n)
            else:
                prev_n = n_values[idx-1]
                prev_h = best_h
                if prev_h is None:
                    initial_guess = np.random.rand(n)
                else:
                    initial_guess = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, prev_n), prev_h)
            initial_guess = repair_vector(initial_guess, n)
        
        dx = 2.0 / n
        def objective(h):
            cor = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(cor) * dx
            return max_overlap
        
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
        bounds = [(0, 1) for _ in range(n)]
        
        if n <= 100:
            method = 'CMAES'
            options = {'maxiter': 100, 'popsize': 10}
        else:
            method = 'SLSQP'
            options = {'maxiter': 100}
        
        start_time = time.time()
        if method == 'CMAES':
            result = CMAES.optimize(objective, x0=initial_guess, bounds=bounds, constraints=constraints, 
                                   options=options, ftol=1e-8, maxiter=100)
        else:
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method=method,
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options=options
            )
        time_used += time.time() - start_time

        h_opt = result.x
