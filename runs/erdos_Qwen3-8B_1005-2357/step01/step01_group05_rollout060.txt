from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    for idx, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        num_restarts = 5
        restarts = []
        if idx == 0 and initial_h_values is not None:

        else:
            for _ in range(num_restarts):
                if idx == 0 and initial_h_values is not None:
                    x = initial_h_values.copy()

                    x = np.random.rand(n)
                    x = np.clip(x, 0, 1)
                    s = np.sum(x)
                    if s == 0:
                        scale = 0
                    else:
                        scale = (n / 2) / s
                    x_scaled = x * scale
                    x_repaired = repair_vector(x_scaled, n)
                    x = x_repaired
                restarts.append(x)
        
        for guess in restarts:
            if time_used >= budget_s:
                break
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )

            if time_used >= budget_s:

            h_opt = result.x
