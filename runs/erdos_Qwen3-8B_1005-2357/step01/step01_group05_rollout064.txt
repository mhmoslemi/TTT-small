import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    prev_h = initial_h_values if initial_h_values is not None else None

    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    def upsample(h_prev, n_prev, n_new):
        x_prev = np.linspace(0, 2, n_prev)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_prev, h_prev)
        return h_new

    for n in n_values:
        if time_used >= budget_s:
            break
        num_restarts = 5
        for _ in range(num_restarts):
            if prev_h is not None and n == 47:
                initial_guess = prev_h.copy()
            else:
                if prev_h is not None and n > 47:
                    initial_guess = upsample(prev_h, len(prev_h), n)
                else:
                    x = np.random.rand(n)
                    x = np.clip(x, 0, 1)
                    s = np.sum(x)
                    if s == 0:
                        scale = 0
                    else:
                        scale = (n / 2) / s
                    x_scaled = x * scale
                    initial_guess = repair_vector(x_scaled, n)
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 100}
            )
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            h_opt = result.x
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired
        prev_h = best_h if best_h is not None else initial_guess
    return best_h, best_c5, len(best_h)
