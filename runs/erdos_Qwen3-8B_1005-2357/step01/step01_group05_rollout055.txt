from scipy.optimize import minimize

    def repair_vector(h, n):

        target = n / 2

        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired
    
    def upsample_vector(old_h, old_n, new_n):
        x_old = np.linspace(0, 2, old_n)
        x_new = np.linspace(0, 2, new_n)
        new_h = np.interp(x_new, x_old, old_h)
        new_h = repair_vector(new_h, new_n)
        return new_h
    
    for i, n in enumerate(n_values):

                initial_guesses = [upsample_vector(best_h, len(best_h), n)] + [np.random.rand(n) for _ in range(4)]

                initial_guesses = [np.random.rand(n) for _ in range(5)]

                cor = np.correlate(h, 1 - h, mode='full')
                return np.max(cor) * dx
            
            def constraint(h):

            bounds = [(0, 1)] * n
            constraints = [{'type': 'eq', 'fun': constraint}]

                method = 'SLSQP'
