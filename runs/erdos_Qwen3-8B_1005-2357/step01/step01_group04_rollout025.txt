import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def repair_individual(ind, target_sum):
        ind = np.clip(ind, 0, 1)
        s = np.sum(ind)
        if abs(s - target_sum) > 1e-6:
            scale = target_sum / s
            ind *= scale
        return ind

    def evaluate(h):
        h = repair_individual(h, target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def ga_optimize(n_points, target_sum, time_budget, initial_pop=None):
        pop_size = max(50, int(n_points / 2))
        pop = np.random.rand(pop_size, n_points) if initial_pop is None else initial_pop
        pop = np.array([repair_individual(ind, target_sum) for ind in pop])
        best_ind = pop[0]
        best_val = evaluate(best_ind)
        start_time = time.time()
        while time.time() - start_time < time_budget:
            inds = np.random.choice(pop_size, 3, replace=False)
            parents = pop[inds]
            offspring = []
            for p1, p2 in zip(parents[::2], parents[1::2]):
                if np.random.rand() < 0.7:
                    child = np.random.rand(n_points)
                    child = np.clip(child, 0, 1)
                    child = np.clip(p1 * np.random.rand(n_points) + p2 * (1 - np.random.rand(n_points)), 0, 1)
                else:
                    child = np.random.rand(n_points)
                    child = np.clip(child, 0, 1)
                child = repair_individual(child, target_sum)
                offspring.append(child)
            pop = np.vstack([pop, offspring])
            pop = np.vstack([pop[:pop_size], pop[pop_size:]])
            for i in range(pop_size):
                if evaluate(pop[i]) < best_val:
                    best_val = evaluate(pop[i])
                    best_ind = pop[i]
        return best_ind

    if 'initial_h_values' in globals():
        initial_pop = np.array(initial_h_values, dtype=np.float64)
        initial_pop = repair_individual(initial_pop, target_sum)
        best_h = initial_pop
        best_c5 = evaluate(best_h)
    else:
        initial_pop = np.random.rand(n_points)
        initial_pop = repair_individual(initial_pop, target_sum)
        best_h = initial_pop
        best_c5 = evaluate(best_h)

    def local_refine(ind, n_points, target_sum):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = opt.minimize(lambda x: evaluate(x), ind, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return repair_individual(result.x, target_sum)

    elapsed = 0
    for n_points in [100, 200, 400]:
        if elapsed >= budget_s:
            break
        target_sum = n_points / 2
        dx = 2.0 / n_points
        if n_points == 100 and 'initial_h_values' in globals():
            initial_pop = np.array(initial_h_values, dtype=np.float64)
            initial_pop = repair_individual(initial_pop, target_sum)
        else:
            initial_pop = np.random.rand(n_points)
            initial_pop = repair_individual(initial_pop, target_sum)
        start_time = time.time()
        ga_pop = ga_optimize(n_points, target_sum, budget_s - elapsed, initial_pop)
        ga_pop = local_refine(ga_pop, n_points, target_sum)
        c5 = evaluate(ga_pop)
        if c5 < best_c5:
            best_c5 = c5
            best_h = ga_pop
        elapsed = time.time() - start_time

    best_h = repair_individual(best_h, target_sum)
    return (best_h, best_c5, n_points)
