import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * 2.0 / len(initial_h)
    start_time = time.time()
    elapsed = 0
    
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / len(h)
    
    grid_sizes = [45, 50, 60, 70, 80, 90, 100, 120, 140, 160, 200]
    for n in grid_sizes:
        if elapsed >= budget_s - 100:
            break
        dx = 2.0 / n
        h = np.zeros(n)
        h[:n//2] = 1.0
        h = h.astype(float)
        
        def objective(x):
            x = x.reshape(n)
            return compute_c5(x)
        
        def constraint(x):
            x = x.reshape(n)
            return np.sum(x) - n / 2
        
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': constraint}]
        
        # Differential evolution for global search
        result_de = differential_evolution(objective, bounds, constraints=cons, popsize=15, tol=1e-5, maxiter=50, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result_de.success:
            h_de = result_de.x
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de
        
        # Local refinement with SLSQP
        result_slsqp = minimize(objective, h_de, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result_slsqp.success:
            h_slsqp = result_slsqp.x
            c5_slsqp = compute_c5(h_slsqp)
            if c5_slsqp < best_c5:
                best_h = h_slsqp.copy()
                best_c5 = c5_slsqp
        
        elapsed = time.time() - start_time
    
    return (best_h, best_c5, len(best_h))
