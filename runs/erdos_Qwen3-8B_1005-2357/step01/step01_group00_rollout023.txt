import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_n = 0

    grid_sizes = [94, 188, 376, 752]

    for n_points in grid_sizes:
        if time.time() - start_time > budget_s:
            break
        initial_conditions = []
        for _ in range(5):
            if initial_h is not None and len(initial_h) == n_points:
                h_initial = initial_h.copy()
            else:
                typ = np.random.choice(initial_types)
                if typ == 'dirichlet':
                    h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
                elif typ == 'two_peak':
                    h_initial = np.zeros(n_points)
                    mid = n_points // 2
                    h_initial[mid-10:mid+10] = 1.0
                    h_initial[mid+10:mid+20] = 1.0
                elif typ == 'uniform':
                    h_initial = np.zeros(n_points)
                    h_initial[:n_points//2] = 1.0
                elif typ == 'periodic':
                    h_initial = np.tile([1.0, 0.0], (n_points//2, ))[:n_points]
                else:
                    h_initial = np.random.rand(n_points) * (n_points / 2)
            initial_conditions.append(h_initial)

        for h_initial in initial_conditions:
            dx = 2.0 / n_points
            def obj(h):
                cor = np.correlate(h, 1 - h, mode='full')
                return np.max(cor * dx)
            def grad(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_idx = np.argmax(cor)
                lag = max_idx - (n_points - 1)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + lag
                    if -1 <= j < n_points:
                        grad[i] = 1 - h[j]
                return grad * dx

            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

            try:
                res = opt.minimize(obj, h_initial, method='trust-constr',
