    def repair(h, n):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        target = n / 2
        if s < target:
            h = h * (target / s)
        else:
            h = h * (target / s)
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * (2.0 / len(h))

    def ga_step(h, n, max_gen=100, pop_size=50, tol=1e-4):

        pop = repair(pop, n)
        for _ in range(max_gen):
            fitness = np.array([compute_c5(ind) for ind in pop])
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < tol:
                break
            parents = np.random.choice(pop_size, size=pop_size, replace=True)

                child = np.random.uniform(a, b, n)
                child = repair(child, n)
                children.append(child)
            pop = np.array(children)
        return pop[np.argmin([compute_c5(ind) for ind in pop])]

    for n in grid_sizes:
