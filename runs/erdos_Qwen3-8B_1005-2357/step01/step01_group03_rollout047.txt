import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Objective function
    def objective(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # Gradient function
    def gradient(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        k0 = np.argmax(corr)
        L = k0 - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + L < n_points:
                grad[i] = (1 - h[i + L]) - h[i]
        return grad * dx

    # Constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, jac=gradient, tol=1e-6)

    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h)

    # Ensure constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h) * dx
    if abs(best_sum - 1) > 1e-5:
        factor = 1 / best_sum
        best_h = best_h * factor

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
