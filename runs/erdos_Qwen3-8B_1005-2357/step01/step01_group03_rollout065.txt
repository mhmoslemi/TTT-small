import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)

        if sum_clipped == 0:
            return np.zeros_like(h_candidate)
        factor = target_sum / sum_clipped
        return h_clipped * factor

    def objective_de(h_candidate):
        h_repaired = repair(h_candidate)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(objective_de, bounds, strategy='best1bin',

    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)

    def objective_local(h_candidate):
        h_repaired = repair(h_candidate)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    def objective_local_func(h):
        return objective_local(h)
