            h_clip = h_clip * scale
        return h_clip

    def objective(h):

        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx

    def gradient(h):
        n = len(h)
        dx = 2.0 / n
        cor = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cor)
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j = i + lag
            if -1 <= j < n:
                grad[i] = 1 - h[j]
        return grad * dx
