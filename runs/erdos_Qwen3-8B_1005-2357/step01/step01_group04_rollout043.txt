import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_pop = None

    if 'initial_h_values' in globals():
        initial_pop = [np.array(initial_h_values, dtype=np.float64)]

    def repair(ind):
        ind = np.clip(ind, 0, 1)
        s = np.sum(ind)
        if abs(s - target_sum) > 1e-6:
            scale = target_sum / s
            ind *= scale
            ind = np.clip(ind, 0, 1)
            if abs(np.sum(ind) - target_sum) > 1e-6:
                ind[-1] += target_sum - np.sum(ind)
        return ind

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def ga_optimize(n_points, target_sum, initial_pop, time_budget):
        pop_size = 50
        if initial_pop is None:
            h_pop = np.random.rand(pop_size, n_points)
            h_pop = h_pop / np.sum(h_pop, axis=1, keepdims=True) * target_sum
            h_pop = np.clip(h_pop, 0, 1)
        else:
            h_pop = np.array([np.array(p) for p in initial_pop])
            h_pop = np.array([repair(p) for p in h_pop])
        start = time.time()
        while time.time() - start < time_budget:
            tournament = np.random.choice(pop_size, 3, replace=False)
            parents = h_pop[tournament]
            offspring = []
            for i in range(0, pop_size, 2):
                if np.random.rand() < 0.7:
                    p1, p2 = parents[i//2], parents[i//2 + 1]
                    c1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                    c2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
                    offspring.extend([c1, c2])
                else:
                    offspring.extend([parents[i//2], parents[i//2 + 1]])
            for j in range(len(offspring)):
                if np.random.rand() < 0.1:
                    offspring[j] += np.random.normal(0, 0.1, n_points)
                    offspring[j] = repair(offspring[j])
            fitness = np.array([compute_c5(p) for p in h_pop])

            h_pop = new_pop

        best_idx = np.argmin(fitness)

        return res.x
