    def constraint_start(params, j):
        s = params[::2]
        l = params[1::2]
        return s[j] + l[j] - n_points

    def constraint_disjoint(params, j):
        s = params[::2]
        l = params[1::2]
        if j >= len(s) - 1:
            return 0.0
        return s[j] + l[j] - s[j+1]

    initial_params = []
    for j in range(m):
        initial_params.append(initial_starts[j])
        initial_params.append(initial_lengths[j])
    initial_params = np.array(initial_params)

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    for j in range(m):
        constraints.append({'type': 'ineq', 'fun': lambda p, j=j: constraint_start(p, j)})
    for j in range(m-1):
        constraints.append({'type': 'ineq', 'fun': lambda p, j=j: constraint_disjoint(p, j)})

    try:
        result = opt.minimize(
            fun=objective,
            x0=initial_params,
            method='SLSQP',
            bounds=[(0, n_points - 1) for _ in range(2*m)],
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_params = result.x
        best_h = build_h(best_params[::2], best_params[1::2])
        best_c5 = objective(best_params)
    except:
        best_h = h
        best_c5 = float('inf')

    if elapsed < budget_s * 0.9 and m < 5:
        m += 1
        total_length = n_points / 2
        remaining_length = total_length - m
        initial_lengths = np.ones(m) + remaining_length / m
        initial_starts = np.arange(m)
        initial_params = []
        for j in range(m):
            initial_params.append(initial_starts[j])
            initial_params.append(initial_lengths[j])
        initial_params = np.array(initial_params)
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        for j in range(m):
            constraints.append({'type': 'ineq', 'fun': lambda p, j=j: constraint_start(p, j)})
        for j in range(m-1):
            constraints.append({'type': 'ineq', 'fun': lambda p, j=j: constraint_disjoint(p, j)})
        try:
            result = opt.minimize(
                fun=objective,
                x0=initial_params,
                method='SLSQP',
                bounds=[(0, n_points - 1) for _ in range(2*m)],
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_params = result.x
            best_h = build_h(best_params[::2], best_params[1::2])
            best_c5 = objective(best_params)
        except:
            best_h = h
            best_c5 = float('inf')

    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    c5_bound = objective(best_params)
    return (best_h, c5_bound, n_points)
