            grad = np.zeros(n_points)
            for idx in max_indices:
                k0 = idx - (n_points - 1)
                for i in range(n_points):
                    if 0 <= i + k0 < n_points:
                        term1 = 1 - h[i + k0]
                    else:
                        term1 = 0
                    if 0 <= i - k0 < n_points:
                        term2 = h[i - k0]
                    else:
                        term2 = 0
                    grad[i] += dx * (term1 - term2)
            return grad
        
        result_trust = opt.minimize(
            fun=objective,
            x0=h_repaired,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            jac=gradient,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        h = result_trust.x
        h = np.clip(h, 0, 1)
        h = repair(h, n_points)
        
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = cor * dx
        c5 = np.max(overlap)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points
    
    return (best_h, best_c5, best_n)

def repair(h, n_points):
    h_clip = np.clip(h, 0, 1)
    total = np.sum(h_clip)
    if abs(total - n_points / 2) > 1e-5:
        scale = (n_points / 2) / total
        h_clip *= scale
    return h_clip
