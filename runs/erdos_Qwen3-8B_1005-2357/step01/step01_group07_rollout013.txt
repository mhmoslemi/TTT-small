import numpy as np
from scipy.optimize import minimize
import time
from skopt import BayesSearchCV
from skopt.utils import use_named_args

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    grid_sizes = [45, 50, 60, 70, 80, 90, 100]
    repair_threshold = 1e-6

    def repair_h(h, n):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        target = n / 2
        if abs(sum_h - target) > repair_threshold:
            excess = sum_h - target
            if excess > 0:
                indices = np.argsort(h)
                for i in range(n):
                    if excess <= 0:
                        break
                    if h[indices[i]] > 0:
                        reduction = min(h[indices[i]], excess)
                        h[indices[i]] -= reduction
                        excess -= reduction
            else:
                indices = np.argsort(h)
                for i in range(n):
                    if abs(excess) <= repair_threshold:
                        break
                    if h[indices[i]] < 1:
                        increase = min(1 - h[indices[i]], -excess)
                        h[indices[i]] += increase
                        excess += increase
        return h

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * (2.0 / len(h))

    for n in grid_sizes:
        if time.time() - start_time > budget_s - 100:

            h = repair_h(h, n)
        bounds = [(0.0, 1.0) for _ in range(n)]
        @use_named_args
