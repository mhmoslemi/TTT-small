import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n)]

            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                initial_h = current_h
        else:
            break
