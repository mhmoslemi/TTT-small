import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        starts = [initial_h_values]
    else:
        starts = []
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_start():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        return h

    for _ in range(5):
        if 'initial_h_values' not in globals():
            h = generate_start()
        else:
            h = initial_h_values.copy()
            h += np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * target_sum
        starts.append(h)

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h_start in starts:
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        if result.success:
            current_c5 = np.max(np.correlate(result.x, 1 - result.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
