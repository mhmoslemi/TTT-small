    dx = 2.0 / n_points
    start_time = time.time()
    elapsed = 0

    def generate_h(s, l):
        h = np.zeros(n_points, dtype=np.float64)
        for j in range(m):
            start = int(s[j])
            length = int(l[j])
            end = start + length - 1
            if end >= n_points:
                end = n_points - 1
            h[start:end+1] = 1.0
        return h

    if 'initial_h_values' in globals():
        intervals = []
        current_start = 0
        for i in range(n_points):
            if h[i] == 1 and (i == 0 or h[i-1] == 0):
                current_start = i
            elif h[i] == 0 and (i == 0 or h[i-1] == 1):
                intervals.append((current_start, i))
        if h[-1] == 1:
            intervals.append((current_start, n_points - 1))
        s = np.array([interval[0] for interval in intervals])
        l = np.array([interval[1] - interval[0] + 1 for interval in intervals])
    else:
        s = np.random.randint(0, n_points, size=m)
        l = np.random.randint(1, n_points, size=m)
        total_length = np.sum(l)
        desired_length = n_points / 2
        scale = desired_length / total_length
        l = l * scale
        l = np.round(l).astype(int)
        while np.sum(l) != desired_length:
            l += np.random.randint(-1, 2, size=m)

    def objective(params):
        nonlocal elapsed
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed > budget_s * 0.9:
            raise KeyboardInterrupt
        s_params = params[:m]
        l_params = params[m:]
        h_array = generate_h(s_params, l_params)
        k = 1 - h_array
        corr = np.correlate(h_array, k, mode='full')
        return np.max(corr) * dx

    def constraint_sum(params):
        return np.sum(params[m:]) - n_points / 2

    def constraint_overlap(params):
        s_params = params[:m]
        l_params = params[m:]
        for j in range(m-1):
            if s_params[j+1] < s_params[j] + l_params[j]:
                return s_params[j+1] - s_params[j] - l_params[j]
        return 0.0

    bounds = [(0, n_points - 1)] * m + [(1, n_points - 1)] * m
    constraints = [{'type': 'eq', 'fun': constraint_sum}, {'type': 'ineq', 'fun': constraint_overlap}]
    params_initial = np.concatenate([s, l])

    try:
        result = opt.minimize(
            fun=objective,
            x0=params_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_params = result.x
    except:
        best_params = params_initial

    s_best = best_params[:m]
    l_best = best_params[m:]
    h_best = generate_h(s_best, l_best)
    h_best = np.clip(h_best, 0, 1)
    total = np.sum(h_best)
    scale = (n_points / 2) / total
    h_best = h_best * scale
    c5_bound = objective(best_params)

    return (h_best, c5_bound, n_points)
