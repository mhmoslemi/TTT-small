    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():

        h = h / np.sum(h) * target_sum

        h = np.clip(h_vec, 0, 1)
        s = np.sum(h)
        if abs(s - target_sum) > 1e-6:
            h *= target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    remaining_time = budget_s - (time.time() - start_time)
