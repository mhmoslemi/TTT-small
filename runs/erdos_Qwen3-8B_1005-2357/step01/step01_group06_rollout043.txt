    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective_de(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        sum_initial = np.sum(h)
        scale_factor = target_sum / sum_initial
        h_adjusted = h * scale_factor
        h_adjusted = np.clip(h_adjusted, 0, 1)
        def objective_slsqp(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            return np.max(corr) * dx
        def constraint_sum(h_values):
            return np.sum(h_values) - target_sum
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        result_slsqp = scipy.optimize.minimize(
            objective_slsqp, h_adjusted,
            method='SLSQP', bounds=bounds_slsqp, constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        best_h = result_slsqp.x
        c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        return (best_h, c5_bound, n_points)
    else:
        result_de = scipy.optimize.differential_evolution(
            objective_de, bounds,
            strategy='best1bin', maxiter=100, popsize=20, tol=1e-6,
            mutation=(0.5, 1), recombination=0.9, seed=seed
        )
        h_de = result_de.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            sum_de = np.sum(h_de)
            scale_factor = target_sum / sum_de
            h_adjusted = h_de * scale_factor
            h_adjusted = np.clip(h_adjusted, 0, 1)
            c5_bound = np.max(np.correlate(h_adjusted, 1 - h_adjusted, mode='full')) * dx
            return (h_adjusted, c5_bound, n_points)
        else:
            sum_de = np.sum(h_de)
            scale_factor = target_sum / sum_de
            h_adjusted = h_de * scale_factor
            h_adjusted = np.clip(h_adjusted, 0, 1)
            def objective_slsqp(h_values):
                corr = np.correlate(h_values, 1 - h_values, mode='full')
                return np.max(corr) * dx

            cons = [{'type': 'eq', 'fun': constraint_sum}]
            bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
            result_slsqp = scipy.optimize.minimize(
