from scipy.optimize import minimize
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        current_sum = np.sum(h_perturbed)
        delta = target_sum - current_sum
        h_perturbed[-1] += delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        starts.append(h_perturbed)
    
    best_c5 = float('inf')
    best_h = initial_h
    
    for h_start in starts:
        def objective(h):
            return compute_c5(h, n_points)
