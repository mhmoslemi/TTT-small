            grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
        return grad
    
    grid_sizes = [45, 50, 60, 70, 80, 90, 100, 120, 140, 160, 200]
    
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        # Step function
        if n % 2 == 0:
            step_h = np.zeros(n)
            step_h[n//2:] = 1.0
        else:
            step_h = np.zeros(n)
            step_h[n//2:] = 1.0
            step_h[-1] = 0.5
        # Random permutation
        if n % 2 == 0:
            binary = np.zeros(n)
            binary[:n//2] = 1.0
            binary = np.random.permutation(binary)
        else:
            binary = np.zeros(n)
            binary[:n//2] = 1.0
            binary[-1] = 0.5
            binary = np.random.permutation(binary)
        # Optimize step_h
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
        result = minimize(compute_c5, step_h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Optimize binary
        result = minimize(compute_c5, binary, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    return (best_h, best_c5, len(best_h))
