    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    # Initialize previous solution
    prev_h = initial_h_values.copy() if initial_h_values is not None else None
    
    # Time budgets per grid
    time_budgets = [500, 300, 250, 50]  # for n=47, 94, 188, 376

    for idx, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        budget = time_budgets[idx]
        num_restarts = 5
        
        # Generate initial guess
        if n == 47:
            if initial_h_values is None:
                # Fallback: create a random initial guess
                x = np.random.rand(n)
                x = np.clip(x, 0, 1)
                s = np.sum(x)
                scale = (n / 2) / s if s != 0 else 0
                x_scaled = x * scale
                initial_guess = repair_vector(x_scaled, n)
            else:
                initial_guess = initial_h_values.copy()
        else:
            # Generate initial guess by interpolating from prev_h
            # prev_h is the best solution from the previous grid
            initial_guess = np.repeat(prev_h, 2)
        
        # Perform optimization with multiple restarts
        for _ in range(num_restarts):
            if time_used >= budget_s:
                break
            
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 100}
            )
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            
            h_opt = result.x
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired
        
        # Update prev_h for the next grid
        prev_h = best_h.copy() if best_h is not None else prev_h
    
    return best_h, best_c5, len(best_h)
