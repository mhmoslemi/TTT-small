import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Objective for DE
    def objective_de(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        factor = target_sum / sum_h
        return h * factor

    # DE phase
    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(
        objective_de, bounds, strategy='best1',
        mutation=0.8, recombination=0.9, popsize=50,
        tol=1e-6, maxiter=100, seed=seed, polish=False
    )
    best_h_de = repair(result_de.x)

    # SLSQP phase
    def objective_slsqp(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def grad_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k < n_points:
                term1 = (1 - h[i + k]) * dx
            term2 = 0.0
            if 0 <= i - k < n_points:
                term2 = -h[i - k] * dx
            grad[i] += term1 + term2
        return grad

    result_slsqp = minimize(
        objective_slsqp, best_h_de, method='SLSQP',
        jac=grad_slsqp, bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        tol=1e-6, options={'maxiter': 100}
    )

    best_h_final = repair(result_slsqp.x)
    best_c5 = objective_slsqp(best_h_final)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_final, best_c5, n_points

    return best_h_final, best_c5, n_points
