            best_h_full = np.concatenate([result_sym.x, result_sym.x[::-1]])
        else:
            best_h_full = result_non_sym.x
        return best_h_full, best_c5, n_points
    else:
        best_h_full = np.concatenate([result_sym.x, result_sym.x[::-1]])
        best_c5 = result_sym.fun
        return best_h_full, best_c5, n_points
