import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    initial_h = initial_h_values if 'initial_h_values' in locals() else None

    n_points_list = [94, 188, 376, 752]
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        if initial_h is not None and len(initial_h) == n_points:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        
        def objective(h):
            dx = 2.0 / n_points
            cor = np.correlate(h, 1 - h, mode='full')
            return np.max(cor * dx)
        
        def gradient(h):
            dx = 2.0 / n_points
            cor = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(cor)
            lag = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + lag
                if -1 <= j < n_points:
                    grad[i] = 1 - h[j]
            return grad * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            jac=gradient,
            options={'maxiter': 100}
        )
        
        h = result.x
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h = h * scale
        else:
            h = h
        
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cor * dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points

    return (best_h, best_c5, best_n)
