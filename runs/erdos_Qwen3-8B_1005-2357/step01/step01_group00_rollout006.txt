import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points_list = [94, 188, 376, 752]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    initial_h = initial_h_values if 'initial_h_values' in locals() else None
    
    def generate_initial(n_points):
        if initial_h is not None and len(initial_h) == n_points:
            return initial_h.copy()
        dirichlet = np.random.dirichlet([1.0]*n_points)
        h = dirichlet * (n_points / 2)
        return h
    
    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip *= scale
        return h_clip
    
    def objective(h):
        dx = 2.0 / len(h)
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        return np.max(cor) * dx
    
    def gradient(h):
        dx = 2.0 / len(h)
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        max_idx = np.argmax(cor)
        k0 = max_idx - (len(h)-1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = i + k0
            if -1 < j < len(h):
                grad[i] += dx * (1 - h[j])
            j = i - k0
            if -1 < j < len(h):
                grad[i] += dx * (-h[j])
        return grad
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        h_initial = generate_initial(n_points)
        h_initial = repair(h_initial, n_points)
        
        bounds = [(0, 1)] * n_points
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            polish=True,
            disp=False
        )
        
        h_de = result.x
        h_de = repair(h_de, n_points)
        
        result = opt.minimize(
            objective,
            h_de,
            method='trust-constr',
            bounds=bounds,
            constraints=constraint,
            jac=gradient,
            options={'maxiter': 200, 'gtol': 1e-5, 'eps': 1e-8}
        )
        
        h = result.x
        h = repair(h, n_points)
        total = np.sum(h)
        if total != 0:
            h = h / total * (n_points / 2)
        else:
            h = np.zeros(n_points)
        
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = cor * dx
        c5 = np.max(overlap)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points
    
    return (best_h, best_c5, best_n)
