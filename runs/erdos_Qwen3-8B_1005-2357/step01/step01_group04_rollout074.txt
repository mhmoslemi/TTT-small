import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
    else:
        h_initial = np.random.rand(100)
        h_initial = h_initial / np.sum(h_initial) * 100 / 2

    best_h = h_initial

    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_vec):

        s = np.sum(h)
        if abs(s - target_sum) > 1e-6:

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    def local_optimize(h0):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = opt.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x

    start_time = time.time()

    def optimize_level(n):

        target_sum = n / 2
        dx = 2.0 / n

        h_new = np.clip(h_new, 0, 1)

        best_h = local_optimize(best_h)
        best_c5 = objective(best_h)
