    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        method='trust-constr',
        jac=grad,
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 1000, 'gtol': 1e-5, 'eps': 1e-8}
    )

    # Repair the result
    h = result.x
    h = repair(h)

    # Compute C5
    dx = 2.0 / n_points
    cor = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(cor) * dx

    # Update best solution
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
        best_n = n_points

    # Check if time is up
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, best_n)

    # Try other initializations if time permits
    # For example, two-peak or uniform
    # But given time constraints, maybe not

    return (best_h, best_c5, best_n)
