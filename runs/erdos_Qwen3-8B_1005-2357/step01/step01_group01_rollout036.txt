        k = 1 - h_repaired
        corr = np.correlate(h_repaired, k, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    try:
        result_de = opt.differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        best_h_de = result_de.x
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            raise KeyboardInterrupt
    except:
        best_h_de = h
    
    best_h_de_repaired = repair_h(best_h_de)
    
    def objective_slsqp(h_vec):
        k = 1 - h_vec
        corr = np.correlate(h_vec, k, mode='full')
        return np.max(corr) * dx
    
    def constraint_slsqp(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds_slsqp = [(0, 1)] * n_points
    constraints_slsqp = [{'type': 'eq', 'fun': constraint_slsqp}]
    
    try:
        result_slsqp = opt.minimize(
            fun=objective_slsqp,
            x0=best_h_de_repaired,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints_slsqp,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result_slsqp.x
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            raise KeyboardInterrupt
    except:
        best_h = best_h_de_repaired
    
    best_h = repair_h(best_h)
    c5_bound = objective_slsqp(best_h)
    
    if (time.time() - start_time) < budget_s * 0.9:
        n_points_new = 200
        dx_new = 2.0 / n_points_new
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new_init = np.interp(x_new, x_old, best_h)
        total = np.sum(h_new_init)
        scale = (n_points_new / 2) / total
        h_new_init = h_new_init * scale
        h_new_init = np.clip(h_new_init, 0, 1)
        
        def objective_de_new(h_vec):
            h_repaired = repair_h(h_vec)
            k = 1 - h_repaired
            corr = np.correlate(h_repaired, k, mode='full')
            return np.max(corr) * dx_new
        
        bounds_new = [(0, 1)] * n_points_new
        
        try:
            result_de_new = opt.differential_evolution(
                objective_de_new,
                bounds_new,
                strategy='best1bin',
                maxiter=50,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                disp=False
            )
            best_h_new = result_de_new.x
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.9:
                raise KeyboardInterrupt
        except:
            best_h_new = h_new_init
        
        best_h_new_repaired = repair_h(best_h_new)
        
        def objective_slsqp_new(h_vec):
            k = 1 - h_vec
            corr = np.correlate(h_vec, k, mode='full')
            return np.max(corr) * dx_new
        
        constraints_slsqp_new = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_new / 2}]
        bounds_slsqp_new = [(0, 1)] * n_points_new
        
        try:
            result_slsqp_new = opt.minimize(
                fun=objective_slsqp_new,
                x0=best_h_new_repaired,
                method='SLSQP',
                bounds=bounds_slsqp_new,
                constraints=constraints_slsqp_new,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_h = result_slsqp_new.x
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.9:
                raise KeyboardInterrupt
        except:
            best_h = best_h_new_repaired
        
        best_h = repair_h(best_h)
        c5_bound = objective_slsqp_new(best_h)
        n_points = n_points_new
    
    best_h = repair_h(best_h)
    c5_bound = objective_slsqp(best_h)
    return (best_h, c5_bound, n_points)
