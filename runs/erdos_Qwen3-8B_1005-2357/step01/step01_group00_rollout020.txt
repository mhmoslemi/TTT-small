import numpy as np
import scipy.optimize as opt
import time

def repair(h, n_points):
    h_clip = np.clip(h, 0, 1)
    total = np.sum(h_clip)
    if abs(total - n_points / 2) > 1e-5:
        scale = (n_points / 2) / total
        h_clip = h_clip * scale
    return h_clip

def objective(h, n_points):
    dx = 2.0 / n_points
    cor = np.correlate(h, 1 - h, mode='full')
    return np.max(cor) * dx

def gradient(h, n_points):
    dx = 2.0 / n_points
    cor = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(cor)
    lag = max_idx - (n_points - 1)
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + lag
        if -1 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    n_points_list = [94, 188, 376, 752]
    
    initial_h = initial_h_values if 'initial_h_values' in locals() else None
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        if initial_h is not None and len(initial_h) == n_points:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.rand(n_points) * 2 - 1
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def obj_func(h):
            return objective(h, n_points)
        
        def grad_func(h):
            return gradient(h, n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        try:
            result = opt.minimize(
                fun=obj_func,
                x0=h_initial,
                method='trust-constr',
                jac=grad_func,
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 100}
            )
        except:
            result = opt.minimize(
                fun=obj_func,
                x0=h_initial,
                method='COBYLA',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 100}
            )
        
        h = result.x
        h = repair(h, n_points)
        dx = 2.0 / n_points
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = cor * dx
        c5 = np.max(overlap)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points
    
    return (best_h, best_c5, best_n)
