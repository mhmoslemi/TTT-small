import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped) * dx
        if total == 0:
            return np.random.rand(len(h)) * 0.5
        return h_clipped * (1.0 / total)

    def obj_ga(h):
        h_repaired = repair(h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    result_ga = differential_evolution(obj_ga, bounds, strategy='best1bin', popsize=50, mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=500)
    best_h = result_ga.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, obj_ga(best_h), n_points

    def obj_local(h):
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def cons(h):
        return np.sum(h) * dx - 1

    cons = ({'type': 'eq', 'fun': cons})
    result_local = minimize(obj_local, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter': 100})

    best_h = result_local.x
    best_c5 = obj_local(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
