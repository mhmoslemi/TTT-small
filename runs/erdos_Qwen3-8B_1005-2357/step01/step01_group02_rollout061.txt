import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 90
    target_sum = n_points / 2
    initial_h = initial_h_values
    
    def generate_candidates(base, num=10, std=0.1):
        candidates = []
        for _ in range(num):
            perturb = np.random.normal(0, std, n_points)
            h_perturb = base + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            current_sum = np.sum(h_perturb)
            delta = target_sum - current_sum
            h_perturb[-1] += delta
            h_perturb = np.clip(h_perturb, 0, 1)
            candidates.append(h_perturb)
        return candidates
    
    def optimize_candidate(h_start):
        def objective(h):
            return compute_c5(h, n_points)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, maxiter=50)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_opt = np.sum(h_opt)
            if np.abs(sum_opt - target_sum) > 1e-6:
                scale = target_sum / sum_opt
                h_opt = np.clip(h_opt * scale, 0, 1)
            return h_opt
        return h_start
    
    best_h = initial_h
    best_c5 = compute_c5(best_h, n_points)
    
    start_time = np.datetime64('now')
    elapsed = 0
    
    for gen in range(20):
        if elapsed >= budget_s:
            break
        candidates = generate_candidates(best_h, num=10, std=0.1)
        for h_start in candidates:
            h_opt = optimize_candidate(h_start)
            current_c5 = compute_c5(h_opt, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
        elapsed = np.datetime64('now') - start_time
    
    return (best_h, best_c5, n_points)
