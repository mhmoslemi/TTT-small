import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_guesses = [initial_h_values]
    else:
        initial_guesses = []

            h = h / np.sum(h) * 50.0

            initial_guesses.append(h)

        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    for guess in initial_guesses:
        result = scipy.optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
