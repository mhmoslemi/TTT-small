import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0

    def repair_vector(h, n):
        s = np.sum(h)
        target = n / 2
        delta = target - s
        h_repaired = h + delta / n
        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired

    def upsample_vector(old_h, old_n, new_n):
        x_old = np.linspace(0, 2, old_n)
        x_new = np.linspace(0, 2, new_n)
        new_h = np.interp(x_new, x_old, old_h)
        new_h = repair_vector(new_h, new_n)
        return new_h

    for i, n in enumerate(n_values):
        if time_used >= budget_s:
            break
        if i == 0 and initial_h_values is not None:

                initial_guesses = [upsample_vector(best_h, len(best_h), n)] + [np.random.rand(n) for _ in range(4)]

                initial_guesses = [np.random.rand(n) for _ in range(5)]

        for j, guess in enumerate(initial_guesses):
            if time_used >= budget_s:
                break
            guess = repair_vector(guess, n)
            dx = 2.0 / n

            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                return np.max(cor) * dx

            def constraint(h):
                return np.sum(h) - n / 2

            bounds = [(0, 1)] * n
            constraints = [{'type': 'eq', 'fun': constraint}]

            start_time = time.time()
            if n <= 100:
                method = 'trust-constrained'
                options = {'maxiter': 50}
            else:
                method = 'SLSQP'
                options = {'maxiter': 100}

            try:
                result = minimize(objective, guess, method=method, bounds=bounds, constraints=constraints, options=options)
