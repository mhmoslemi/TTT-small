import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 100
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

        h_norm = h_values / np.sum(h_values) * target_sum
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result_de = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    best_h = result_de.x
    best_c5 = objective(best_h)
    
    def slsqp_objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1}]
    result_slsqp = scipy.optimize.minimize(
        slsqp_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'disp': False}
    )
    
    best_h = result_slsqp.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
