    if elapsed_sym < budget_s:
        bounds_non_sym = [(0, 1)] * n_points
        if 'initial_h_values' in globals():
            h_initial_non_sym = initial_non_sym.copy()
        else:
            h_initial_non_sym = np.random.rand(n_points)
        start_time_non = time.time()
        result_non_sym = differential_evolution(objective_non_sym, bounds_non_sym, x0=h_initial_non_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.9, seed=seed)
        elapsed_non = time.time() - start_time_non
        total_elapsed = elapsed_sym + elapsed_non
    else:
        result_non_sym = None
        total_elapsed = elapsed_sym

    if result_non_sym is not None and result_non_sym.success:
        h_full_non = result_non_sym.x
        h_repaired_non = repair(h_full_non)
        c5_non = objective_non_sym(h_full_non)
        if c5_non < best_c5:
            best_c5 = c5_non
            best_h = h_repaired_non

    # Fallback if no solution found
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = initial_h_values
        else:
            best_h = np.random.rand(n_points)
            best_h = repair(best_h)

    return best_h, best_c5, n_points
