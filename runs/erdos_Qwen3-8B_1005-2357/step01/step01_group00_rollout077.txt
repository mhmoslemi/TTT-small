import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points_list = [94, 188, 376, 752]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    initial_h = initial_h_values if 'initial_h_values' in locals() else None
    
    def repair(h, n_points):

        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip *= scale
        return h_clip
    
    def objective(h, n_points):

        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        return np.max(cor) * dx
    
    def gradient(h, n_points):

        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        max_val = np.max(cor)

        for idx in max_indices:
            k0 = idx - (n_points - 1)
            for i in range(n_points):
                term1 = 1 - h[i + k0] if (0 <= i + k0 < n_points) else 0
                term2 = -h[i - k0] if (0 <= i - k0 < n_points) else 0
                grad[i] += dx * (term1 + term2)
        return grad
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        if initial_h is not None and len(initial_h) == n_points:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        
        def objective_de(h):
