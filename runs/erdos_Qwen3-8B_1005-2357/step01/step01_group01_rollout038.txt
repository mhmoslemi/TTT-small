    bounds_slsqp = [(0, 1)] * n_points
    constraints_slsqp = [{'type': 'eq', 'fun': constraint_slsqp}]
    result_slsqp = opt.minimize(
        fun=objective_slsqp,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraints_slsqp,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    best_h = result_slsqp.x
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        n_points_up = 200
        x100 = np.linspace(0, 2, n_points)
        x200 = np.linspace(0, 2, n_points_up)
        h200_init = np.interp(x200, x100, best_h)
        sum_h200 = np.sum(h200_init)
        scale = (n_points_up / 2) / sum_h200
        h200_init = h200_init * scale
        h200_init = np.clip(h200_init, 0, 1)
        
        def objective_de_up(x):
            x_clipped = np.clip(x, 0, 1)
            sum_clipped = np.sum(x_clipped)
            scale = (n_points_up / 2) / sum_clipped
            x_scaled = x_clipped * scale
            k = 1 - x_scaled
            corr = np.correlate(x_scaled, k, mode='full')
            return np.max(corr) * (2.0 / n_points_up)
        
        bounds_up = [(0, 1)] * n_points_up
        result_de_up = opt.differential_evolution(
            fun=objective_de_up,
            bounds=bounds_up,
            popsize=15,
            strategy='best1bin',
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-6,
            maxiter=1000,
            polish=True,
            initial_population=[h200_init] if 'initial_h_values' in globals() else None
        )
        best_h = result_de_up.x
        
        def objective_slsqp_up(x):
            k = 1 - x
            corr = np.correlate(x, k, mode='full')
            return np.max(corr) * (2.0 / n_points_up)
        
        def constraint_slsqp_up(x):
            return np.sum(x) - n_points_up / 2
        
        bounds_slsqp_up = [(0, 1)] * n_points_up
        constraints_slsqp_up = [{'type': 'eq', 'fun': constraint_slsqp_up}]
        result_slsqp_up = opt.minimize(
            fun=objective_slsqp_up,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_slsqp_up,
            constraints=constraints_slsqp_up,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result_slsqp_up.x
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    c5_bound = objective_slsqp(best_h)
    
    return (best_h, c5_bound, n_points)
