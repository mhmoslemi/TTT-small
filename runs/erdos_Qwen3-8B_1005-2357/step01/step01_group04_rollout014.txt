import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def repair(x, n):
        h = np.clip(x, 0, 1)
        s = np.sum(h) - n / 2
        if abs(s) < 1e-6:
            return h
        scale = (n / 2) / s if s != 0 else 1
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    def objective(h_vec, n, dx):
        h = repair(h_vec, n)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def get_initial_h(n):
        if 'initial_h_values' in globals():
            x_old = np.linspace(0, 2, 100)
            x_new = np.linspace(0, 2, n)
            h_new = np.interp(x_new, x_old, initial_h_values)
            h_new = repair(h_new, n)
            return h_new
        return np.full(n, 0.5)
    
    best_h = None
    best_c5 = float('inf')
    elapsed = 0
    
    for n in [100, 200, 400]:
        if elapsed >= budget_s:
            break
        target_sum = n / 2
        dx = 2.0 / n
        h = get_initial_h(n)
        bounds = [(0, 1)] * n
        
        start_time = time.time()
        result = opt.differential_evolution(
            lambda x: objective(x, n, dx),
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=True,
            disp=False
        )
        elapsed = time.time() - start_time
        
        current_h = result.x
        current_c5 = objective(current_h, n, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    best_h = repair(best_h, n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h *= target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
