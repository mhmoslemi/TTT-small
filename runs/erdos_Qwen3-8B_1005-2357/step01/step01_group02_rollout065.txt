import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_sum = np.sum(initial_h)
    delta = target_sum - initial_sum
    initial_h[-1] += delta
    initial_h = np.clip(initial_h, 0, 1)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        for _ in range(5):
            noise = np.random.normal(0, 0.1, n_points)
            h_perturbed = best_h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            delta = target_sum - current_sum
            h_perturbed[-1] += delta
            h_perturbed = np.clip(h_perturbed, 0, 1)
            def objective(h):
                return compute_c5(h, n_points)
            def constraint_sum(h):
                return np.sum(h) - target_sum
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:

                current_c5 = compute_c5(current_h, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
    return (best_h, best_c5, n_points)
