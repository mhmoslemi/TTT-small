import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)

        h_initial = h_initial / np.sum(h_initial) * target_sum

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        L = k0 - (len(h) - 1)
        j = np.arange(len(h)) + L

        grad = np.zeros_like(h)
        grad[mask] = (1 - h[j[mask]]) - h[mask]
        return grad * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)

    best_h = result.x
    best_c5 = objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h) * dx

        factor = 1 / best_sum
        best_h = best_h * factor
