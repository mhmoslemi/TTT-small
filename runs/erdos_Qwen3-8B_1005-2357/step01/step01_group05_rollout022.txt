import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0

    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    for n in n_values:
        if time_used >= budget_s:
            break
        num_restarts = 5
        for _ in range(num_restarts):
            if initial_h_values is not None and n == 47:
                initial_guess = initial_h_values.copy()
            else:
                if initial_h_values is not None and n > 47:
                    x_initial = np.linspace(0, 2, len(initial_h_values))
                    x_new = np.linspace(0, 2, n)
                    h_interpolated = np.interp(x_new, x_initial, initial_h_values)
                    initial_guess = h_interpolated
                else:
                    x = np.random.rand(n)
                    x = np.clip(x, 0, 1)
                    s = np.sum(x)
                    if s == 0:
                        scale = 0
                    else:
                        scale = (n / 2) / s
                    x_scaled = x * scale
                    x_repaired = repair_vector(x_scaled, n)
                    initial_guess = x_repaired
            initial_guess = repair_vector(initial_guess, n)
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                return np.max(cor) * dx
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1)] * n
            start_time = time.time()
            if n <= 100:
                result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
                h_opt = result.x
            else:
                result = minimize(fun=objective, x0=initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
                h_opt = result.x
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired
    return best_h, best_c5, len(best_h)
