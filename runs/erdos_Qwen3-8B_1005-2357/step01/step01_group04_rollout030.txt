    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum_new / np.sum(best_h)
        return (best_h, c5, n_points_new)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum_new / np.sum(best_h)
    return (best_h, c5, n_points_new)
