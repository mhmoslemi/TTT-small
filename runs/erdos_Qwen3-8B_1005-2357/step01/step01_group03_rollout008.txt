import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(candidate, is_symmetric=True):
        candidate = np.clip(candidate, 0, 1)
        if is_symmetric:
            target = n_points / 4
        else:
            target = target_sum
        if np.sum(candidate) == 0:
            return candidate
        scale = target / np.sum(candidate)
        return candidate * scale

    def obj_sym(x):
        x = repair(x, True)
        h = np.concatenate([x, x[::-1]])
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    def obj_non(x):
        x = repair(x, False)
        g = 1 - x
        corr = np.correlate(x, g, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_sym_initial = h_initial[:n_points//2]
    else:
        h_sym_initial = np.random.rand(n_points//2)
        h_sym_initial = h_sym_initial / np.sum(h_sym_initial) * (n_points/4)

    bounds_sym = [(0, 1)] * (n_points//2)
    bounds_non = [(0, 1)] * n_points

    result_sym = None
    result_non = None

    try:
        result_sym = differential_evolution(obj_sym, bounds_sym, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    except:
        pass

    if time.time() - start_time < budget_s:
        try:
            result_non = differential_evolution(obj_non, bounds_non, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        except:
            pass

    best_c5 = float('inf')

    best_h = best_h / np.sum(best_h) * target_sum
