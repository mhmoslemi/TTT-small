import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 100
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum  # Ensure sum is correct

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h
    best_c5 = float('inf')
    start_time = time.time()

    for _ in range(5):  # Multi-start optimization
        if 'initial_h_values' not in globals():
            h_new = np.random.rand(n_points)
            h_new = h_new / np.sum(h_new) * target_sum
            h_new = np.clip(h_new, 0, 1)
        else:
            h_new = h.copy()

        result = scipy.optimize.minimize(
            objective,
            h_new,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(3):  # Additional multi-starts if time permits
            if 'initial_h_values' not in globals():
                h_new = np.random.rand(n_points)
                h_new = h_new / np.sum(h_new) * target_sum
                h_new = np.clip(h_new, 0, 1)
            else:
                h_new = h.copy()

            result = scipy.optimize.minimize(
                objective,
                h_new,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'disp': False}
            )

            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
