import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    def constraint_sum(h):
        return np.sum(h) - n / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n)]

    def generate_initial_guesses(n):
        guesses = []
        # Binary step function
        if n % 2 == 0:
            binary = np.zeros(n)
            binary[:n//2] = 1
            guesses.append(binary)
        # Random permutation
        perm = initial_h.copy()
        np.random.shuffle(perm)
        guesses.append(perm)
        # Random binary pattern
        if n % 2 == 0:
            binary = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
            binary = binary.astype(float)
            binary = binary - np.sum(binary) / n * 2
            guesses.append(binary)
        return guesses

    grid_sizes = [45, 60, 80, 100]
    for size in grid_sizes:
        if time.time() - start_time > budget_s - 100:
            break
        if size != n:
            n_new = size
            dx_new = 2.0 / n_new
            h_new = np.zeros(n_new)
            # Generate initial guesses
            guesses = generate_initial_guesses(n_new)
            for guess in guesses:
                # Ensure sum is correct
                guess = guess * (n_new / 2) / np.sum(guess)
                guess = np.clip(guess, 0, 1)
                # Optimize
                result = minimize(objective, guess, method='L-BFGS-B', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
                if result.success:
                    current_h = result.x
                    current_c5 = compute_c5(current_h)
                    if current_c5 < best_c5:
                        best_h = current_h
                        best_c5 = current_c5
                        n = n_new
                        dx = dx_new
            if n == n_new:
                n = n_new
                dx = dx_new
    return (best_h, best_c5, len(best_h))
