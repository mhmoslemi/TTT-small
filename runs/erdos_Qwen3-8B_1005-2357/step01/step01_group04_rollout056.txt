    def repair(x, n):
        h = np.clip(x, 0, 1)
        s = np.sum(h) - n / 2
        if abs(s) < 1e-6:
            return h
        scale = (n / 2) / s if s != 0 else 1
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h_vec, n, dx):
        h = repair(h_vec, n)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def local_optimize(h0, n, dx):
        bounds = [(0, 1)] * n
        result = opt.minimize(lambda x: objective(x, n, dx), h0, method='L-BFGS-B', bounds=bounds)
        return result.x

    def evaluate(h, n, dx):
        h = repair(h, n)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for n in n_points_list:
        if remaining_time <= 0:
            break
        dx = 2.0 / n
        target_sum = n / 2
        if initial_h is not None and best_h is None:

        h_opt = result.x
