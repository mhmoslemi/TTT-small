        result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        if result_de.success:
            current_h = result_de.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

        result_slsqp = minimize(objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-6)
        if result_slsqp.success:
            current_h = result_slsqp.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    return (best_h, best_c5, n)
