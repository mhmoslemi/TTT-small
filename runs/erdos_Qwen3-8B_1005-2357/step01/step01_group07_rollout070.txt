    if elapsed < budget_s - 100:
        n = 200
        dx = 2.0 / n
        bounds = [(0, 1) for _ in range(n)]
        h_step = np.zeros(n)
        h_step[:n//2] = 0.5
        h_rand = np.random.permutation(h_step)
        h_binary = np.random.randint(0, 2, n)
        h_binary = h_binary / np.sum(h_binary) * n / 2
        for h_initial in [h_step, h_rand, h_binary]:
            h_initial = h_initial.copy()
            if np.abs(np.sum(h_initial) - n/2) > 1e-6:
                h_initial = h_initial / np.sum(h_initial) * (n/2)
            result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
            h_de = result_de.x
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
            result_slsqp = minimize(compute_c5, h_de, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result_slsqp.success:
                h_slsqp = result_slsqp.x
                c5_slsqp = compute_c5(h_slsqp)
                if c5_slsqp < best_c5:
                    best_h = h_slsqp
                    best_c5 = c5_slsqp

    return (best_h, best_c5, len(best_h))
