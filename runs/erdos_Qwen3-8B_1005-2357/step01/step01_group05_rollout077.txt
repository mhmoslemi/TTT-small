import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired
    
    best_solutions = []
    for n in n_values:
        if time_used >= budget_s:
            break
        # Generate initial guess
        if len(best_solutions) > 0:
            prev_h = best_solutions[-1]
            x_prev = np.linspace(0, 2, len(prev_h))
            x_new = np.linspace(0, 2, n)
            h_upsampled = np.interp(x_new, x_prev, prev_h)
            initial_guess = repair_vector(h_upsampled, n)
        else:
            if initial_h_values is not None:
                initial_guess = initial_h_values.copy()
            else:
                x = np.random.rand(n)

                    scale = 0
                else:
                    scale = (n / 2) / s
                x_scaled = x * scale
                x_repaired = repair_vector(x_scaled, n)
                initial_guess = x_repaired
        
        best_c5_n = float('inf')
        best_h_n = None
        num_restarts = 5
        for _ in range(num_restarts):
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            start_time = time.time()
            result = minimize(
                fun=objective,
                x0=initial_guess,
