import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n / 2

    grid_sizes = [45, 50, 60, 70, 80, 90, 100, 120, 140, 160, 200]
    for grid_size in grid_sizes:
        if time.time() - start_time > budget_s - 100:
            break
        n = grid_size
        dx = 2.0 / n
        bounds = [(0, 1) for _ in range(n)]
        # Step function
        h_step = np.full(n, 0.5)
        # Random permutation
        h_rand = np.random.permutation(h_step)
        # Binary initial condition
        h_bin = np.zeros(n)
        h_bin[:n//2] = 1
        # Run differential evolution
        result = differential_evolution(objective, bounds, constraints={'type': 'eq', 'fun': constraint_sum}, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Run local optimization
        result = minimize(objective, h_step, method='SLSQP', jac=None, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Run local optimization on random permutation
        result = minimize(objective, h_rand, method='SLSQP', jac=None, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Run local optimization on binary
        result = minimize(objective, h_bin, method='SLSQP', jac=None, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    return (best_h, best_c5, len(best_h))
