import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    def generate_initial_conditions(n):
        if n % 2 == 0:
            step = np.zeros(n)
            step[:n//2] = 1.0
        else:
            step = np.zeros(n)
            step[:n//2] = 1.0
            step[-1] = 0.5
        step_perm = step.copy()
        np.random.shuffle(step_perm)
        random_h = np.random.rand(n)
        random_h = random_h / np.sum(random_h) * (n / 2)
        return [step, step_perm, random_h]

    grid_sizes = [45, 60, 80, 100, 120, 140, 160, 200]
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        initial_conditions = generate_initial_conditions(n)
        for h_initial in initial_conditions:
            bounds = [(0, 1) for _ in range(n)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                current_h = result.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            if elapsed >= budget_s - 100:
                break
        if elapsed >= budget_s - 100:
            break

    return (best_h, best_c5, len(best_h))
