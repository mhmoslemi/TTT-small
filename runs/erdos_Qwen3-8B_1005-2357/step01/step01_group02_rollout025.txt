from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):

    initial_h = initial_h_values
    
    starts = []
    for _ in range(15):
        h_perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        delta = target_sum - current_sum
        h_perturbed[-1] += delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        starts.append(h_perturbed)
    
    best_c5 = float('inf')
    best_h = initial_h
    start_time = time.time()
    
    for h_start in starts:
        def objective(h):
            return compute_c5(h, n_points)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
