import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(100)
    initial_h = np.clip(initial_h, 0, 1)
    target_sum = 100 / 2
    initial_h *= target_sum / np.sum(initial_h)
    
    def repair(x, n):
        h = np.clip(x, 0, 1)
        s = np.sum(h) - n / 2
        if abs(s) < 1e-6:
            return h
        scale = (n / 2) / s if s != 0 else 1
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    def objective(h_vec, n):
        h = repair(h_vec, n)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    
    levels = [100, 200, 400]
    best_h = initial_h.copy()
    best_c5 = objective(best_h, 100)
    elapsed = 0
    
    for i, n in enumerate(levels):
        if elapsed >= budget_s:
            break
        if i > 0:
            x_old = np.linspace(0, 2, levels[i-1])
            x_new = np.linspace(0, 2, n)
            h_new = np.interp(x_new, x_old, best_h)
            h_new = repair(h_new, n)
        else:
            h_new = initial_h.copy()
        
        start = time.time()
        result = opt.basinhopping(lambda x: objective(x, n), h_new, niter=50, T=1.0, stepsize=0.5, 
                                  minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1)] * n})
        elapsed = time.time() - start
        h_opt = result.x
        c5 = objective(h_opt, n)
        
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        
        if i < len(levels) - 1:
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, levels[i+1])
            best_h = np.interp(x_new, x_old, best_h)
            best_h = repair(best_h, levels[i+1])
    
    best_h = np.clip(best_h, 0, 1)
    best_h *= (len(best_h) / 2) / np.sum(best_h)
    return (best_h, best_c5, len(best_h))
