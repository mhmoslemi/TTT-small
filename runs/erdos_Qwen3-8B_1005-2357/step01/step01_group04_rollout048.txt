import scipy.optimize as opt
import time

        h = np.clip(h_vec, 0, 1)
        s = np.sum(h)
        if abs(s - target_sum) > 1e-6:
            h *= target_sum / s
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def local_optimize(h0, n, target_sum, dx):
        bounds = [(0, 1)] * n
        result = opt.minimize(
            lambda x: objective(x, n, target_sum, dx),

            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        return result.x
