import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 100
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time.time()
    best_h = h
    best_c5 = objective(best_h)

    # First optimization
    result = scipy.optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'disp': False}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time

    # Additional optimizations
    num_starts = 4
    for _ in range(num_starts):
        if elapsed >= budget_s:
            break
        new_h = np.random.rand(n_points)
        new_h = new_h / np.sum(new_h) * target_sum
        new_h = np.clip(new_h, 0, 1)
        result = scipy.optimize.minimize(
            objective,
            new_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        elapsed = time.time() - start_time

    return (best_h, best_c5, n_points)
