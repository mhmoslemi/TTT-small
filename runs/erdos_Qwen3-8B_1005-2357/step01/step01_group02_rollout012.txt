import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

            h = np.zeros(n_points)
        else:
            scale = target_sum / current_sum

        h = np.clip(h, 0, 1)
        population.append(h)
