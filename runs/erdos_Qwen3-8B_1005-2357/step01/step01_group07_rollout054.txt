import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = [(0, 1) for _ in range(n)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

        cond = [h.copy()]
        np.random.shuffle(cond[0])
        cond.append(np.zeros(n))
        cond[-1][:n//2] = 1
        np.random.shuffle(cond[-1])

        noisy = np.clip(noisy, 0, 1)
        noisy = noisy * (n / 2) / np.sum(noisy)
        cond.append(noisy)
        return cond

    initial_conditions = generate_initial_conditions(initial_h)
    for h_initial in initial_conditions:
        result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:

    elapsed = time.time() - start_time
