import numpy as np
from scipy.optimize import minimize, CMAES
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    budgets = [500, 300, 250, 50]
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired
    
    for i, (n, budget) in enumerate(zip(n_values, budgets)):
        if time_used >= budget_s:
            break
        num_restarts = 5
        for _ in range(num_restarts):
            if i == 0 and initial_h_values is not None:
                initial_guess = initial_h_values.copy()
            else:
                if i == 0:
                    x = np.random.rand(n)
                else:
                    prev_n = n_values[i-1]
                    prev_h = best_h if best_h is not None else initial_h_values
                    x = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, prev_n), prev_h)
                x = np.clip(x, 0, 1)
                s = np.sum(x)
                if s == 0:
                    scale = 0
                else:
                    scale = (n / 2) / s
                x_scaled = x * scale
                initial_guess = repair_vector(x_scaled, n)
            
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            start_time = time.time()
            if n <= 100:
                result = CMAES.optimize(objective, bounds, seed=seed, popsize=10, maxiter=100)
                h_opt = result.x
            else:
                result = minimize(
                    fun=objective,
                    x0=initial_guess,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    jac=None,
                    options={'maxiter': 100}
                )
                h_opt = result.x
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
            
            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx
            
            if c5 < best_c5:

                best_h = h_repaired
