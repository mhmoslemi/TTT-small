        total = np.sum(h_clip)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_clip *= scale
        return h_clip
    
    for n_points in grid_sizes:
        if time.time() - start_time > budget_s:
            break
        
        if initial_h is not None and len(initial_h) == n_points:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.dirichlet([1.0]*n_points) * (n_points / 2)
        
        w = min(10, n_points // 4)
        block = np.zeros(n_points)
        start = (n_points // 2) - w // 2
        end = start + w
        block[start:end] = 1.0
        start2 = (n_points // 2) + (n_points // 2) - w // 2
        end2 = start2 + w
        block[start2:end2] = 1.0
        total = np.sum(block)
        scale = (n_points / 2) / total
        h_two_peak = block * scale
        
        h_uniform = np.zeros(n_points)
        h_uniform[:n_points//2] = 1.0
        total = np.sum(h_uniform)
        scale = (n_points / 2) / total
        h_uniform = h_uniform * scale
        
        dx = 2.0 / n_points
        p = np.random.uniform(0.5, 1.5)
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if (x - p) % 1.0 < 0.5:
                h_periodic[i] = 1.0
            else:
                h_periodic[i] = 0.0
        total = np.sum(h_periodic)
        scale = (n_points / 2) / total
        h_periodic = h_periodic * scale
        
        initial_conditions = [
            h_initial,
            h_two_peak,
            h_uniform,
            h_periodic
        ]
        
        for h_initial in initial_conditions:
            bounds = [(0, 1)] * n_points
            
            def objective(h):
                dx = 2.0 / n_points
                v = 1 - h
                cor = np.correlate(h, v, mode='full')
                return np.max(cor) * dx
            
            result_de = opt.differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                maxiter=100,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.9,
                seed=seed
            )
            
            h_de = result_de.x
            h_de = repair(h_de, n_points)
            
            def gradient(h):
                dx = 2.0 / n_points
                v = 1 - h
                cor = np.correlate(h, v, mode='full')
                max_idx = np.argmax(cor)
                k0 = max_idx - (n_points - 1)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + k0
                    if -1 < j < n_points:
                        grad[i] += dx * (1 - h[j])
                    j = i - k0
                    if -1 < j < n_points:
                        grad[i] += dx * (-h[j])
                return grad
            
            result_tn = opt.minimize(
                fun=objective,
                x0=h_de,
                method='trust-constr',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
                jac=gradient,
                options={'maxiter': 100, 'gtol': 1e-6, 'ftol': 1e-6}
            )
            
            h_tn = result_tn.x
            h_tn = repair(h_tn, n_points)
            
            dx = 2.0 / n_points
            v = 1 - h_tn
            cor = np.correlate(h_tn, v, mode='full')
            c5 = np.max(cor) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tn
                best_n = n_points
    
    return (best_h, best_c5, best_n)
