import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):

    def repair(h, n_points):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - n_points/2) > 1e-5:
            scale = (n_points/2) / total
            h_clip *= scale
        return h_clip
    
    def objective(h, n_points):
        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        return np.max(cor) * dx
    
    def gradient(h, n_points):
        dx = 2.0 / n_points
        v = 1 - h
        cor = np.correlate(h, v, mode='full')
        max_val = np.max(cor)
        max_indices = np.where(np.isclose(cor, max_val, atol=1e-5))[0]
        grad = np.zeros(n_points)
        for idx in max_indices:
            k0 = idx - (n_points-1)
            for i in range(n_points):
                term1 = 1 - h[i+k0] if (0 <= i+k0 < n_points) else 0
                term2 = -h[i-k0] if (0 <= i-k0 < n_points) else 0
                grad[i] += dx * (term1 + term2)
        return grad
    
    for n_points in n_points_list:

            return objective(h, n_points)
        
        def gradient_wrapper(h):
            return gradient(h, n_points)
        
        result = opt.differential_evolution(

            bounds,
            strategy='best1bin',
