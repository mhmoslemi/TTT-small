import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
        # Count intervals
        m = 0
        for i in range(n_points - 1):
            if h[i] != h[i+1]:
                m += 1
        m = max(1, m)
    else:
        n_points = 100
        m = 2
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        n_points = len(h)

    def generate_initial_params(m, n_points):
        l = np.zeros(m, dtype=int)
        s = np.zeros(m, dtype=int)
        total_length = n_points // 2
        l.fill(total_length // m)
        remaining = total_length % m
        for i in range(remaining):
            l[i] += 1
        s[0] = 0
        for i in range(1, m):
            s[i] = s[i-1] + l[i-1]
        if s[-1] + l[-1] > n_points:
            s[-1] = n_points - l[-1]
            l[-1] = n_points - s[-1]
        return s, l

    s_initial, l_initial = generate_initial_params(m, n_points)
    h_initial = np.zeros(n_points)
    for i in range(m):
        start = s_initial[i]
        length = l_initial[i]
        h_initial[start:start+length] = 1

    def objective(params):
        s = params[::2]
        l = params[1::2]
        h = np.zeros(n_points)
        for i in range(m):
            start = s[i]
            length = l[i]
            if start + length > n_points:
                length = n_points - start
            h[start:start+length] = 1
        k = 1 - h
        corr = np.correlate(h, k, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(params):
        s = params[::2]
        l = params[1::2]
        return np.sum(l) - n_points / 2

    def constraint_disjoint(params):
        s = params[::2]
        l = params[1::2]
        constraints = []
        for j in range(m-1):
            constraints.append(s[j+1] - s[j] - l[j])
        return constraints

    bounds = []
    for _ in range(m):
        bounds.extend([(0, n_points - 1), (1, n_points)])
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    for c in constraint_disjoint([0]*len(bounds)):
        constraints.append({'type': 'ineq', 'fun': lambda x, c=c: c})

    try:
        result = opt.minimize(
            fun=objective,
            x0=np.concatenate([s_initial, l_initial]),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_params = result.x
    except:
        best_params = np.concatenate([s_initial, l_initial])

    s = best_params[::2]
    l = best_params[1::2]
    h = np.zeros(n_points)
    for i in range(m):
        start = s[i]
        length = l[i]
        if start + length > n_points:
            length = n_points - start
        h[start:start+length] = 1
    c5_bound = objective(best_params)
    return (h, c5_bound, n_points)
