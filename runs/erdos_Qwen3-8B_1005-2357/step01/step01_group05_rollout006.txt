from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    def upsample_vector(old_h, old_n, new_n):
        dx = 2.0 / old_n
        new_dx = 2.0 / new_n
        interp = np.linspace(0, 2, new_n)
        old_x = np.linspace(0, 2, old_n)
        old_h_interp = np.interp(interp, old_x, old_h)
        return repair_vector(old_h_interp, new_n)

    for i, n in enumerate(n_values):

            break

                fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n),

                constraints=constraints,

                mutation=(0.5, 1),

                tol=1e-6,
                maxiter=100,
                polish=True,
                disp=False
            )

            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * (2.0 / n)

                best_c5 = c5
                best_h = h_repaired
            time_used += time.time() - time.time()
