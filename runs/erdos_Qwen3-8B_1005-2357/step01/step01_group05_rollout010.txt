import numpy as np
from scipy.optimize import minimize, CMAES
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    
    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    def upsample_vector(x, n_coarse, n_fine):
        if n_fine == n_coarse:
            return x
        x_interpolated = np.zeros(n_fine)
        for i in range(n_fine):
            x_coarse = i * n_coarse / n_fine
            idx = int(x_coarse)
            if idx < n_coarse:
                x_interpolated[i] = x[idx]
            else:
                x_interpolated[i] = x[-1]
        return repair_vector(x_interpolated, n_fine)
    
    for n in n_values:
        if time_used >= budget_s:
            break
        num_restarts = 5
        for _ in range(num_restarts):
            if initial_h_values is not None and n == 47:
                initial_guess = initial_h_values.copy()
            else:
                if n == 47 and initial_h_values is not None:
                    initial_guess = upsample_vector(initial_h_values, 47, n)
                else:
                    x = np.random.rand(n)
                    x = np.clip(x, 0, 1)
                    s = np.sum(x)
                    if s == 0:
                        scale = 0
                    else:
                        scale = (n / 2) / s
                    x_scaled = x * scale
                    x_repaired = repair_vector(x_scaled, n)
                    initial_guess = x_repaired
            
            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
            
            if n <= 100:
                start_time = time.time()
                result = CMAES.optimize(objective, x0=initial_guess, bounds=bounds, constraints=constraints, 
                                    eval_budget=100, popsize=20, seed=seed, 
                                    constraints_as_dict=True, 
                                    options={'ftol': 1e-8, 'gtol': 1e-8})
                time_used += time.time() - start_time
            else:
                start_time = time.time()
                result = minimize(
                    fun=objective,
                    x0=initial_guess,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    jac=None,
                    options={'maxiter': 100}
                )
