import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 100
        h = np.random.rand(n_points)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h = h * scale
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    elapsed = 0
    
    def repair(h_vec):
        s = np.sum(h_vec)
        if s == 0:
            return h_vec
        scale = 50.0 / s
        h_repaired = h_vec * scale
        return np.clip(h_repaired, 0, 1)
    
    def objective_de(h_vec):
        h_repaired = repair(h_vec)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    
    # Differential Evolution Phase
    try:
        result_de = opt.differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            mutation=0.9,
            recombination=0.7,
            popsize=150,
            tol=1e-8,
            maxiter=100,
            seed=seed,
            polish=True,
            disp=False
        )
        best_h = result_de.x
    except:
        best_h = h
    
    # Time check after DE
    current_time = time.time()
    elapsed = current_time - start_time
    if elapsed > budget_s * 0.7:
        raise KeyboardInterrupt
    
    # Local Refinement with SLSQP
    def objective_slsqp(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - 50}]
    bounds_slsqp = [(0, 1)] * n_points
    
    try:
        result_slsqp = opt.minimize(
            fun=objective_slsqp,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result_slsqp.x
    except:
        best_h = best_h
    
    # Final time check
    current_time = time.time()
    elapsed = current_time - start_time
    if elapsed > budget_s * 0.9:
        raise KeyboardInterrupt
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    c5_bound = objective_slsqp(best_h)
    
    return (best_h, c5_bound, n_points)
