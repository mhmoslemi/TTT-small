                polish=True,
                disp=False,
                workers=-1
            )
            best_h = result_new.x
            n_points = n_points_new
            dx = dx_new
        except:
            best_h = h_new_init
            n_points = n_points_new
            dx = dx_new
    
    def final_objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h_scaled = h * scale
        k = 1 - h_scaled
        corr = np.correlate(h_scaled, k, mode='full')
        return np.max(corr) * dx
    
    c5_bound = final_objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    scale = (n_points / 2) / total
    best_h = best_h * scale
    
    return (best_h, c5_bound, n_points)
