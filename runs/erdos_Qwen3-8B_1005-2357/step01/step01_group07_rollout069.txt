            new_constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            result = minimize(compute_c5, h_initial, method='SLSQP', jac=gradient, bounds=new_bounds, constraints=new_constraints, tol=1e-6)
            if result.success:
                current_c5 = compute_c5(result.x)
                if current_c5 < best_c5:
                    best_h = result.x
                    best_c5 = current_c5
            elapsed = time.time() - start_time
            if elapsed >= budget_s - 100:
                break
    
    return (best_h, best_c5, len(best_h))
