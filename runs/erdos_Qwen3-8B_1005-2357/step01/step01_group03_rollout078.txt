def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) * dx = 1 => sum(h) = 1/dx = n_points/2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Objective function for GA
    def objective(h_candidate):
        h_repaired = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_repaired) * dx
        if sum_h == 0:
            return np.inf
        h_repaired = h_repaired * (target_sum / sum_h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    # Bounds for GA
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # GA phase
    result_ga = differential_evolution(objective, bounds, strategy='best1bin', 
                                      popsize=15, mutation=(0.8, 1.0), 
                                      recombination=0.9, tol=1e-6, 
                                      maxiter=500, 
                                      disp=False, 
                                      polish=True, 
                                      seed=seed)

    best_h_ga = result_ga.x

    # Local optimization phase
    def objective_local(h_candidate):
        h_repaired = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_repaired) * dx
        if sum_h == 0:
            return np.inf
        h_repaired = h_repaired * (target_sum / sum_h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    # Initial guess for local optimization
    h_local = best_h_ga

    # Use L-BFGS-B for local optimization
    result_local = minimize(objective_local, h_local, method='L-BFGS-B', 
                            bounds=bounds, tol=1e-6, 
                            options={'maxiter': 100, 'disp': False})

    best_h_local = result_local.x

    # Final repair
    h_final = np.clip(best_h_local, 0, 1)
    sum_h = np.sum(h_final) * dx
    if sum_h != 0:
        h_final = h_final * (target_sum / sum_h)

    # Compute final C5
    g = 1 - h_final
    corr = np.correlate(h_final, g, mode='full')
    c5_bound = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_final, c5_bound, n_points

    return h_final, c5_bound, n_points
