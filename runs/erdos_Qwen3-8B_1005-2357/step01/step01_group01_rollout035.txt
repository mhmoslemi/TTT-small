        x0 = np.array(x0)
    else:
        x0 = [0, n_points//2]

    def objective(x):
        s = []
        l = []
        for i in range(m):
            s.append(x[2*i])
            l.append(x[2*i + 1])
        h = np.zeros(n_points)
        for j in range(m):
            start = s[j]
            length = l[j]
            if start + length <= n_points:
                h[start:start+length] = 1
        k = 1 - h
        corr = np.correlate(h, k, mode='full')
        return np.max(corr) * dx

    constraints = []
    def sum_constraint(x):
        s = []
        l = []
        for i in range(m):
            s.append(x[2*i])
            l.append(x[2*i + 1])
        return np.sum(l) - n_points / 2
    constraints.append({'type': 'eq', 'fun': sum_constraint})

    for j in range(m):
        def lower_bound_j(x, j=j):
            return x[2*j]
        constraints.append({'type': 'ineq', 'fun': lower_bound_j})

    for j in range(m):
        def upper_bound_j(x, j=j):
            s_j = x[2*j]
            l_j = x[2*j + 1]
            return n_points - l_j - s_j
        constraints.append({'type': 'ineq', 'fun': upper_bound_j})

    for j in range(m-1):
        def disjoint_j(x, j=j):
            s_j = x[2*j]
            l_j = x[2*j + 1]
            s_j1 = x[2*(j+1)]
            return s_j1 - s_j - l_j
        constraints.append({'type': 'ineq', 'fun': disjoint_j})

    bounds = []
    for i in range(2*m):
        if i % 2 == 0:
            bounds.append((0, n_points))
        else:
            bounds.append((1, n_points))

    try:
        result = opt.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
    except:
        result = opt.minimize(
            fun=objective,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )

    best_x = result.x
    s = []
    l = []
    for i in range(m):
        s.append(best_x[2*i])
        l.append(best_x[2*i + 1])
    h = np.zeros(n_points)
    for j in range(m):
        start = s[j]
        length = l[j]
        if start + length <= n_points:
            h[start:start+length] = 1
    k = 1 - h
    corr = np.correlate(h, k, mode='full')
    c5_bound = np.max(corr) * dx
    return (h, c5_bound, n_points)
