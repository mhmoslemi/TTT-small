        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max < n_points:
                if i + k_max < n_points:
                    grad[i] += (1 - h_repaired[i + k_max]) * dx
                if i - k_max >= 0:
                    grad[i] -= h_repaired[i - k_max] * dx
            else:
                if i - k_max >= 0:
                    grad[i] -= h_repaired[i - k_max] * dx
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    result_slsqp = minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        jac=gradient_slsqp,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )

    best_h = result_slsqp.x
    best_c5 = objective_slsqp(best_h)

    best_h = repair(best_h)
    best_sum = np.sum(best_h) * dx
    if abs(best_sum - 1) > 1e-5:
        factor = 1 / best_sum
        best_h = best_h * factor

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
