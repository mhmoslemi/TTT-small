import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum

    # Repair function
    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.random.rand(n_points)
        else:
            return h * (target_sum / s)

    # Objective function for GA
    def objective(h):
        h_repaired = repair(h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    # Run differential evolution for global search
    bounds = [(0.0, 1.0)] * n_points
    result_ga = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, tol=1e-6,
        mutation=(0.5, 1.0), recombination=0.8, maxiter=500, disp=False
    )

    best_h_ga = result_ga.x

    # Local optimization with SLSQP
    def objective_local(h):
        h_repaired = repair(h)
        g = 1 - h_repaired
        corr = np.correlate(h_repaired, g, mode='full')
        return np.max(corr) * dx

    bounds_local = [(0.0, 1.0)] * n_points
    result_local = minimize(
        objective_local, best_h_ga, method='SLSQP', bounds=bounds_local,
        tol=1e-6, options={'maxiter': 100}
    )

    best_h_local = result_local.x
    best_c5 = objective_local(best_h_local)
    best_h_local = repair(best_h_local)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_local, best_c5, n_points

    return best_h_local, best_c5, n_points
