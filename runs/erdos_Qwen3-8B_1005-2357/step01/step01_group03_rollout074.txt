        h_full = np.concatenate([h_sym * scale, h_sym[::-1] * scale])
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx

    def objective_non_sym(h_full):
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full)
        scale = target_sum_non_sym / current_sum
        h_full = h_full * scale
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx

    bounds_sym = [(0, 1)] * (n_points // 2)
    result_sym = differential_evolution(objective_sym, bounds_sym, x0=h_sym_initial, maxiter=500, tol=1e-4, popsize=50, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h_sym = result_sym.x
    best_c5_sym = objective_sym(best_h_sym)

    elapsed_sym = time.time() - start_time
    if elapsed_sym >= budget_s:
        best_h_full = np.concatenate([best_h_sym, best_h_sym[::-1]])
        best_sum = np.sum(best_h_full) * dx
        if abs(best_sum - 1) > 1e-5:
            factor = 1 / best_sum
            best_h_full = best_h_full * factor
        return best_h_full, best_c5_sym, n_points

    bounds_non_sym = [(0, 1)] * n_points
    result_non_sym = differential_evolution(objective_non_sym, bounds_non_sym, x0=h_non_sym_initial, maxiter=250, tol=1e-4, popsize=50, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h_non_sym = result_non_sym.x
    best_c5_non_sym = objective_non_sym(best_h_non_sym)

    best_c5 = min(best_c5_sym, best_c5_non_sym)
    if best_c5 == best_c5_sym:
        best_h_full = np.concatenate([best_h_sym, best_h_sym[::-1]])
    else:
        best_h_full = best_h_non_sym

    best_sum = np.sum(best_h_full) * dx
    if abs(best_sum - 1) > 1e-5:
        factor = 1 / best_sum
        best_h_full = best_h_full * factor

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_full, best_c5, n_points

    return best_h_full, best_c5, n_points
