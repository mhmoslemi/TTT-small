        s = np.sum(h_clipped)
        target = n / 2
        if np.isclose(s, target):
            return h_clipped
        delta = target - s
        h_repaired = h_clipped + delta / n
        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired
    
    best_solutions = []
    for n in n_values:
        if time_used >= budget_s:

                    scale = 0

            initial_guess = repair_vector(initial_guess, n)

                max_overlap = np.max(cor) * dx
                return max_overlap
            
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]
