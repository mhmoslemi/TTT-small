import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n / 2

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    grid_sizes = [45, 60, 80, 100, 120, 140, 160, 200]
    for grid_size in grid_sizes:
        if time.time() - start_time > budget_s - 100:
            break
        # Generate initial conditions
        h_uniform = np.full(grid_size, 0.5)
        h_binary = np.zeros(grid_size)
        indices = np.random.choice(grid_size, size=grid_size // 2, replace=False)
        h_binary[indices] = 1.0
        sum_h = np.sum(h_binary)
        if sum_h < grid_size / 2:
            h_binary[-1] += (grid_size / 2 - sum_h)
        elif sum_h > grid_size / 2:
            h_binary[-1] -= (sum_h - grid_size / 2)
        h_binary = np.clip(h_binary, 0, 1)
        # Run optimizations
        for h_initial in [h_uniform, h_binary]:
            if np.all(h_initial >= 0) and np.all(h_initial <= 1):
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - grid_size / 2}]
                bounds = [(0, 1) for _ in range(grid_size)]
                result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
                if result.success:
                    current_h = result.x
                    current_c5 = compute_c5(current_h)
                    if current_c5 < best_c5:
                        best_h = current_h
                        best_c5 = current_c5
                        n = grid_size
    return (best_h, best_c5, len(best_h))
