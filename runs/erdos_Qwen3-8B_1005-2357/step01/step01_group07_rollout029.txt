import numpy as np
from scipy.optimize import minimize
from skopt import BayesSearch
from skopt.space import Real
import time

def repair(h):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    t = len(h) / 2
    if abs(s - t) > 1e-6:
        idx = np.argsort(h)
        diff = (s - t) / len(h)
        h[idx] -= diff
    return h

def compute_c5(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n = len(initial_h)
    best_h = initial_h.copy()
