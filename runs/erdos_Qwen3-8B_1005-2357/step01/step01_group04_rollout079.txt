        result = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x

    def basinhopping_optimize(h0, n, target_sum):
        def obj(h_vec):
            h = np.clip(h_vec, 0, 1)
            s = np.sum(h)
            if abs(s - target_sum) > 1e-6:
                h *= target_sum / s
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n)
        result = opt.basinhopping(obj, h0, niter=50, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1)] * n})
        return result.x

    h = basinhopping_optimize(h, n_points, target_sum)
    h = local_optimize(h, n_points, target_sum)
    best_c5 = np.max(np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode='full')) * (2.0 / n_points)
    best_h = h

    for n in [200, 400]:
        if time.time() - start_time >= budget_s:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = np.clip(h_new, 0, 1)
        h_new *= n / 2 / np.sum(h_new)
        target_sum = n / 2
        dx = 2.0 / n
        h = basinhopping_optimize(h_new, n, target_sum)
        h = local_optimize(h, n, target_sum)
        current_c5 = np.max(np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode='full')) * (2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

    best_h = np.clip(best_h, 0, 1)
    best_h *= target_sum / np.sum(best_h)
    elapsed = time.time() - start_time
    return (best_h, best_c5, n)
