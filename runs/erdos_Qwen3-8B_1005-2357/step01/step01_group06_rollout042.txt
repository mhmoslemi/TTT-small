    def objective_de(params, m, n_points, dx):
        breakpoints = params[:m-1]
        heights = params[m-1:]
        breakpoints = np.sort(breakpoints)
        breakpoints = np.concatenate(([0.0], breakpoints, [2.0]))
        if np.any(breakpoints < 0) or np.any(breakpoints > 2):
            return np.inf
        widths = np.diff(breakpoints)
        integral = np.sum(heights * widths)
        if integral == 0:
            return np.inf
        scaled_heights = heights * (1.0 / integral)
        scaled_heights = np.clip(scaled_heights, 0, 1)
        x = np.linspace(0, 2, n_points)
        indices = np.searchsorted(breakpoints, x)
        indices = np.clip(indices, 1, len(breakpoints)-1)
        h_values = scaled_heights[indices-1]
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    m = 10
    bounds = [(0.0, 2.0) for _ in range(m-1)] + [(0.0, 1.0) for _ in range(m)]
    result_de = scipy.optimize.differential_evolution(
        lambda params: objective_de(params, m, n_points, dx),
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.7, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=500,
        seed=seed
    )
    best_params = result_de.x

    breakpoints = best_params[:m-1]
    heights = best_params[m-1:]
    breakpoints = np.sort(breakpoints)
    breakpoints = np.concatenate(([0.0], breakpoints, [2.0]))
    widths = np.diff(breakpoints)
    integral = np.sum(heights * widths)
    scaled_heights = heights * (1.0 / integral)
    scaled_heights = np.clip(scaled_heights, 0, 1)
    x = np.linspace(0, 2, n_points)
    indices = np.searchsorted(breakpoints, x)
    indices = np.clip(indices, 1, len(breakpoints)-1)
    h_values = scaled_heights[indices-1]

    def objective_slsqp(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = scipy.optimize.minimize(
        objective_slsqp,
        h_values,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 500, 'disp': False}
    )
    h_values = result_slsqp.x
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_values, c5_bound, n_points)
