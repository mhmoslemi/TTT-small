    # Repair function for symmetric part
    def repair_symmetric(h_sym):
        sum_h = np.sum(h_sym)
        target = target_sum
        if abs(sum_h - target) < 1e-5:
            return h_sym
        h_clipped = np.clip(h_sym, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_sym
        factor = target / sum_clipped
        h_scaled = h_clipped * factor
        if np.any(h_scaled > 1):
            h_scaled = np.clip(h_scaled, 0, 1)
            sum_scaled = np.sum(h_scaled)
            factor = target / sum_scaled
            h_scaled = h_scaled * factor
        return h_scaled

    # Objective function for GA
    def objective_symmetric(h_sym):
        h_repaired = repair_symmetric(h_sym)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx

    # GA phase
    bounds = [(0, 1) for _ in range(n_points // 2)]
    result_ga = differential_evolution(
        objective_symmetric, bounds, strategy='best1bin', 
        popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.9, 
        maxiter=500, seed=seed
    )
    best_h_sym_ga = result_ga.x

    # Local optimization phase
    def objective_local(h_sym):
        h_repaired = repair_symmetric(h_sym)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return np.max(corr) * dx

    def grad_objective_local(h_sym):
        eps = 1e-6
        grad = np.zeros_like(h_sym)
        for i in range(len(h_sym)):
            h_plus = h_sym.copy()
            h_plus[i] += eps
            h_repaired_plus = repair_symmetric(h_plus)
            h_full_plus = np.concatenate([h_repaired_plus, h_repaired_plus[::-1]])
            g_plus = 1 - h_full_plus
            corr_plus = np.correlate(h_full_plus, g_plus, mode='full')
            obj_plus = np.max(corr_plus) * dx

            h_minus = h_sym.copy()
            h_minus[i] -= eps
            h_repaired_minus = repair_symmetric(h_minus)
            h_full_minus = np.concatenate([h_repaired_minus, h_repaired_minus[::-1]])
            g_minus = 1 - h_full_minus
            corr_minus = np.correlate(h_full_minus, g_minus, mode='full')
            obj_minus = np.max(corr_minus) * dx

            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_local = [(0, 1) for _ in range(n_points // 2)]

    result_local = minimize(
        objective_local, best_h_sym_ga, method='SLSQP', 
        bounds=bounds_local, constraints=cons, tol=1e-6, 
        jac=grad_objective_local, options={'maxiter': 100}
    )

    best_h_sym = result_local.x
    h_full = np.concatenate([best_h_sym, best_h_sym[::-1]])
    best_c5 = objective_local(best_h_sym)

    # Ensure constraints are met
    h_full = np.clip(h_full, 0, 1)
    sum_h_full = np.sum(h_full) * dx
    if abs(sum_h_full - 1) > 1e-5:
        factor = 1 / sum_h_full
        h_full = h_full * factor

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_full, best_c5, n_points

    return h_full, best_c5, n_points
