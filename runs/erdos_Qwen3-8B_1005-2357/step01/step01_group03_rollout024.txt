import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = 50.0

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        h_sym_initial = h_initial[:n_points//2]
    else:
        h_sym_initial = np.random.rand(n_points // 2)
        h_sym_initial = h_sym_initial / np.sum(h_sym_initial) * (target_sum / 2)

    def repair(h_full):
        h_full = np.clip(h_full, 0, 1)
        sum_h = np.sum(h_full)
        if sum_h == 0:
            return h_full
        factor = target_sum / sum_h
        return h_full * factor

    def objective_sym(h_sym):
        h_full = np.concatenate([h_sym, h_sym[::-1]])
        h_full = repair(h_full)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective_non_sym(h_full):
        h_full = repair(h_full)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_sym = [(0, 1) for _ in range(n_points // 2)]
    bounds_non_sym = [(0, 1) for _ in range(n_points)]

    result_sym = differential_evolution(objective_sym, bounds_sym, strategy='best1bin',
                                       popsize=100, tol=1e-6, mutation=(0.5, 1),
                                       recombination=0.9, maxiter=1000, disp=False, polish=True, seed=seed)
    best_h_sym = result_sym.x
    best_h_full_sym = np.concatenate([best_h_sym, best_h_sym[::-1]])
    best_h_full_sym = repair(best_h_full_sym)
    best_c5_sym = objective_sym(best_h_sym)

    result_non_sym = differential_evolution(objective_non_sym, bounds_non_sym, strategy='best1bin',
                                           popsize=100, tol=1e-6, mutation=(0.5, 1),

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_values, best_c5, n_points

    return best_h_values, best_c5, n_points
