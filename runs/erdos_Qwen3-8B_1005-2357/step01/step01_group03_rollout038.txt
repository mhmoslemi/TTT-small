        corr = np.correlate(x, g, mode='full')
        return np.max(corr) * dx

    bounds_sym = [(0, 1)] * (n_points // 2)
    bounds_non_sym = [(0, 1)] * n_points

    result_sym = differential_evolution(objective_sym, bounds_sym, x0=h_initial_sym, maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.9, strategy='best1bin', polish=True, disp=False)
    elapsed_sym = time.time() - start_time

    result_non_sym = None
    if elapsed_sym < budget_s:
        result_non_sym = differential_evolution(objective_non_sym, bounds_non_sym, x0=h_initial_non_sym, maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.9, strategy='best1bin', polish=True, disp=False)
        elapsed_non_sym = time.time() - start_time
    else:
        elapsed_non_sym = 0

    best_c5 = float('inf')
    best_h = None
    if result_sym.success:
        h_sym = result_sym.x
        h_full_sym = np.concatenate([h_sym, h_sym[::-1]])
        c5_sym = objective_sym(h_sym)
        best_c5 = c5_sym
        best_h = h_full_sym
    if result_non_sym is not None and result_non_sym.success:
        h_full_non_sym = result_non_sym.x
        c5_non_sym = objective_non_sym(h_full_non_sym)
        if c5_non_sym < best_c5:
            best_c5 = c5_non_sym
            best_h = h_full_non_sym

    if best_h is None:
        best_h = h_initial_non_sym
        best_c5 = objective_non_sym(best_h)

    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h) * dx
    scale = 50.0 / current_sum if current_sum != 0 else 1.0
    best_h = best_h * scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
