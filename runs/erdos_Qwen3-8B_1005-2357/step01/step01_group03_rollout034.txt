import numpy as np

    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():

        h_initial = np.clip(h_initial, 0, 1)

        h_initial = h_initial / np.sum(h_initial) * target_sum

    def repair(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if abs(sum_clipped - target_sum) < 1e-6:
            return x_clipped
        scale = target_sum / sum_clipped if sum_clipped != 0 else 1.0
        return x_clipped * scale

        corr = np.correlate(x_repaired, 1 - x_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
