        n_points = len(initial_h)
        initial_guesses = [initial_h]
    else:
        n_points = 100
        initial_guesses = []
        for _ in range(5):
            h = np.random.rand(n_points)
            target_sum = n_points / 2
            h = h / np.sum(h) * target_sum
            h = np.clip(h, 0, 1)
            initial_guesses.append(h)
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    target_sum = n_points / 2
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    
    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    for guess in initial_guesses:
        result = scipy.optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        current_h = result.x
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n_points)
