import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        s = np.sum(h)
        if abs(s - target_sum) > 1e-6:
            h = h * (target_sum / s)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    def local_optimize(h0):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = opt.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        return result.x

    start_time = time.time()
    result_global = opt.basinhopping(objective, h, niter=50, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1)] * n_points})
    best_h = result_global.x
    best_h = local_optimize(best_h)
    c5 = objective(best_h)

    if time.time() - start_time < budget_s:
        n_points = 300
        target_sum = n_points / 2
        dx = 2.0 / n_points
        x_old = np.linspace(0, 2, 200)
        x_new = np.linspace(0, 2, 300)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = np.clip(h_new, 0, 1)
        h_new *= target_sum / np.sum(h_new)

        def obj_300(h_vec):
            h = np.clip(h_vec, 0, 1)
            s = np.sum(h)
            if abs(s - target_sum) > 1e-6:
                h = h * (target_sum / s)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        result_global_300 = opt.basinhopping(obj_300, h_new, niter=50, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1)] * n_points})
        best_h_300 = result_global_300.x
