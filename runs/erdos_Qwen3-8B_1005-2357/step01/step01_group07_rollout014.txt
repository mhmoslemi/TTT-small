from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n

    start_time = time.time()
    elapsed = 0

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def repair(h, n):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s < n / 2:
            diff = (n / 2 - s) / (n - np.sum(h == 0))
            h[h > 0] += diff
        elif s > n / 2:
            diff = (s - n / 2) / (n - np.sum(h == 1))
            h[h < 1] += diff
        return h

    def blx_alpha(parent1, parent2, alpha=0.5):
        low = np.minimum(parent1, parent2) - alpha * np.abs(parent1 - parent2)
        high = np.maximum(parent1, parent2) + alpha * np.abs(parent1 - parent2)
        return np.random.uniform(low=low, high=high, size=len(parent1))

    def polynomial_mutation(h, eta=1.0):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < 1.0 / len(h):
                u = np.random.rand()
                if u < 0.5:
                    h[i] += (2 * u) ** (1 / (eta + 1)) * (1 - h[i])
                else:
                    h[i] -= (2 * (1 - u)) ** (1 / (eta + 1)) * h[i]
                h[i] = np.clip(h[i], 0, 1)
        return h

    def ga(n_points, pop_size=50, generations=100, seed=42):
        np.random.seed(seed)
        pop = []
        for _ in range(pop_size):
            if np.random.rand() < 0.3:

            pop.append(h)
        pop = np.array(pop)

            eps = 1e-4

                x_plus = x.copy()
                x_plus[i] += eps
                x_minus = x.copy()
                x_minus[i] -= eps
                grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
