    initial_h = initial_h_values

    def compute_c5(h, n):
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    # Function to generate binary pattern for a given n

            np.random.shuffle(binary)

            binary[:n//2] = 1.0
            binary[-1] += 0.5
        return binary

    # Function to generate step function for a given n
    def generate_step(n):
        h_step = np.zeros(n)
        h_step[n//2:] = 1.0
        return h_step

    # Function to generate random permutation of initial_h (if available)
    def generate_permutation(n):
        if n_initial == n:
            return initial_h.copy()
        else:
            return np.random.permutation(np.zeros(n))

    for n in grid_sizes:

        h_binary = generate_binary(n)
