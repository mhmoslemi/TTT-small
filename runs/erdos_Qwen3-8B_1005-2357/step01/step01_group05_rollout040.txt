            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired

            best_h_prev = h_repaired

    return best_h, best_c5, len(best_h)
