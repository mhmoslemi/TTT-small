from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_values = [47, 94, 188, 376]

    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2

        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    for n in n_values:

                s = np.sum(x)
                if s == 0:
                    scale = 0
                else:
                    scale = (n / 2) / s
                x_scaled = x * scale

                initial_guess = repair_vector(initial_guess, n)
        # Run optimization
        dx = 2.0 / n
        def objective(h):
            cor = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(cor) * dx
            return max_overlap
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
        bounds = [(0, 1) for _ in range(n)]
        # Allocate time for this grid
        grid_time = 0

        elif n == 94:
            grid_time = 300
        elif n == 188:
            grid_time = 250
        elif n == 376:
            grid_time = 50
        for _ in range(num_restarts):
            if time_used + grid_time > budget_s:
                break
            start_time = time.time()
            result = minimize(

                x0=initial_guess,
