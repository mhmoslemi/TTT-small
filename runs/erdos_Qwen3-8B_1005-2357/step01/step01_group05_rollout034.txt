            max_overlap = np.max(cor) * dx
            return max_overlap
        
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
        bounds = [(0, 1) for _ in range(n)]
        
        if n == 47:
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 100}
            )
        else:
            result = minimize(
                fun=objective,
                x0=initial_guess,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                jac=None,
                options={'maxiter': 100}
            )
        
        time_used += time.time() - start_time
        
        h_opt = result.x
        h_repaired = repair_vector(h_opt, n)
        cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        c5 = np.max(cor) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_repaired
    
    return best_h, best_c5, len(best_h)
