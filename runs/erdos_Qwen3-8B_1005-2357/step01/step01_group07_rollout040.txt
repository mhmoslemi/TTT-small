def run(seed=42, budget_s=1100, **kwargs):

        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * (2.0 / len(h))
    
    def objective(h):
        return compute_c5(h)
    
    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

        if grid_size == current_grid:
            continue
