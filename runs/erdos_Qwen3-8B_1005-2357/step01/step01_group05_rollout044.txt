import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_values = [47, 94, 188, 376]
    time_used = 0.0
    prev_n = 0

    def repair_vector(x, n):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        target = n / 2
        if np.isclose(s, target):
            return x_clipped
        delta = target - s
        x_repaired = x_clipped + delta / n
        x_repaired = np.clip(x_repaired, 0, 1)
        return x_repaired

    for n in n_values:
        if time_used >= budget_s:
            break
        num_restarts = 5
        for _ in range(num_restarts):
            if prev_n == 0:
                if initial_h_values is not None and n == 47:
                    initial_guess = initial_h_values.copy()
                else:
                    x = np.random.rand(n)
                    x = np.clip(x, 0, 1)
                    s = np.sum(x)
                    if s == 0:
                        scale = 0
                    else:
                        scale = (n / 2) / s
                    x_scaled = x * scale
                    x_repaired = repair_vector(x_scaled, n)
                    initial_guess = x_repaired
            else:
                new_h = np.repeat(best_h, n // prev_n)
                new_h = repair_vector(new_h, n)
                initial_guess = new_h

            dx = 2.0 / n
            def objective(h):
                cor = np.correlate(h, 1 - h, mode='full')
                max_overlap = np.max(cor) * dx
                return max_overlap

            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
            bounds = [(0, 1) for _ in range(n)]

            if n == 47:
                start_time = time.time()
                result = differential_evolution(
                    objective,
                    bounds,
                    strategy='best1bin',
                    popsize=15,
                    maxiter=100,
                    tol=1e-6,
                    mutation=(0.5, 1),
                    recombination=0.8,
                    seed=seed
                )
                h_opt = result.x
                time_used += time.time() - start_time
            else:
                start_time = time.time()
                result = minimize(
                    fun=objective,
                    x0=initial_guess,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    jac=None,
                    options={'maxiter': 100}
                )
                h_opt = result.x
                time_used += time.time() - start_time

            h_repaired = repair_vector(h_opt, n)
            cor = np.correlate(h_repaired, 1 - h_repaired, mode='full')
            c5 = np.max(cor) * dx

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_repaired

        prev_n = n

    return best_h, best_c5, len(best_h)
