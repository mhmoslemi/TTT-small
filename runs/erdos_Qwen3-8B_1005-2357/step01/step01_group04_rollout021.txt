import numpy as np
import scipy.optimize as opt
import time

def repair(x, n):
    h = np.clip(x, 0, 1)
    s = np.sum(h) - n / 2
    if abs(s) < 1e-6:
        return h
    scale = (n / 2) / s if s != 0 else 1
    h = h * scale
    h = np.clip(h, 0, 1)
    return h

def objective(h_vec, n, dx):
    h = repair(h_vec, n)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    n_points = 100
    resolutions = [100, 200, 400]
    elapsed_time = 0.0
    start_time = time.time()
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        h_start = initial_h
    else:
        h_start = np.random.rand(100)
        h_start = h_start / np.sum(h_start) * 50

    for n in resolutions:
        if elapsed_time >= budget_s:
            break
        dx = 2.0 / n
        target_sum = n / 2
        if n == 100:
            h_initial = h_start
        else:
            if best_h is None:
                h_initial = np.random.rand(n)
                h_initial = h_initial / np.sum(h_initial) * target_sum

                x_old = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n)
                h_initial = np.interp(x_new, x_old, best_h)

        bounds = [(0, 1)] * n
        resolution_time = min(budget_s - elapsed_time, 300)
