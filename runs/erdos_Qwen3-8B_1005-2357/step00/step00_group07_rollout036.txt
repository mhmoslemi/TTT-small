            h_clip = np.clip(h_candidate, 0, 1)
            total = np.sum(h_clip)
            h_scaled = h_clip * (n_points / 2) / total
            corr_candidate = np.correlate(h_scaled, 1 - h_scaled, mode="full")
            c_candidate = np.max(corr_candidate) * dx
            if c_candidate < best_alpha_c:
                best_alpha = alpha
                best_alpha_c = c_candidate
        
        h = h + best_alpha * g
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        
        corr_current = np.correlate(h, 1 - h, mode="full")
        c_current = np.max(corr_current) * dx
        if c_current < best_c:
            best_c = c_current
            best_h = h.copy()
        
        if time.time() - time_start > budget_s:
            break
    
    return best_h, best_c, n_points
