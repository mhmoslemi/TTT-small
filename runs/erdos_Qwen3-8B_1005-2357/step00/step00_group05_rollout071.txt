        n_points = 100
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guess_100 = h

        def obj_100(h_flat):
            h = h_flat.reshape(-1)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / len(h))
            return c5

        def cons_sum_100(h_flat):
            return np.sum(h_flat) - len(h_flat)/2

        bounds_100 = Bounds(0, 1)
        cons_100 = [{'type': 'eq', 'fun': cons_sum_100}]
        result_100 = minimize(obj_100, initial_guess_100, method='COBYLA', bounds=bounds_100, constraints=cons_100, options={'maxiter': 100})
        if result_100.success:
            best_h_100 = result_100.x
            best_c5_100 = result_100.fun
            if best_c5_100 < best_c5:
                best_h = best_h_100
                best_c5 = best_c5_100
        else:
            best_h_100 = initial_guess_100
            best_c5_100 = obj_100(best_h_100)
            if best_c5_100 < best_c5:
                best_h = best_h_100
                best_c5 = best_c5_100

    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        delta = (n_points / 2 - sum_h) / n_points
        best_h += delta
        best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
