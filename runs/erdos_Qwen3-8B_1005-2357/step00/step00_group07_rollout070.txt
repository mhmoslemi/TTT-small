import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    initial_h = initial_h_values

    population = [initial_h.copy()]

        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        population.append(h)
    
    best_h = initial_h.copy()

    start_time = time.time()
    
    for gen in range(1000):

        mutant = np.clip(mutant, 0, 1)
        total_mutant = np.sum(mutant)

        mutant *= scale_mutant

        if c5 < best_c5:
            best_c5 = c5
            best_h = mutant.copy()
        worst_idx = np.argmax([evaluate_erdos_solution(h) for h in population])
        population[worst_idx] = mutant
