import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    g = 1 - h
    corr = np.correlate(h, g, mode='full')
    return dx * np.max(corr)

def repair_solution(h_sym, n_points, dx, target_sum):
    h_full = np.concatenate([h_sym, h_sym[::-1]])
    h_full = np.clip(h_full, 0, 1)
    current_sum = np.sum(h_full) * dx
    if abs(current_sum - target_sum) > 1e-6:
        factor = target_sum / current_sum
        h_full = h_full * factor
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full) * dx
        if current_sum < target_sum:
            available_increase = (1 - h_full).sum() * dx
            if available_increase > 0:
                factor = (target_sum - current_sum) / available_increase
                h_full = np.minimum(h_full + factor, 1)
    return h_full[:n_points//2]
