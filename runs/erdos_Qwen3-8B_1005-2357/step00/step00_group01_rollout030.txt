        else:
            initial_h = generate_initial(n)

        dx = 2.0 / n

        def objective_func(h_flat):
            h = h_flat.reshape(n)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def constraint_sum(h_flat):
            return np.sum(h_flat) - n / 2

        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]

        max_restarts = 5
        best_c5_n = float('inf')
        best_h_n = initial_h.copy()

        for restart in range(max_restarts):
            if restart == 0:
                h_initial = initial_h
            else:
                h_initial = generate_initial(n)
            result = minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='trust-constr')
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                sum_h = np.sum(h_candidate)
                if abs(sum_h - n/2) > 1e-6:
                    h_candidate = h_candidate * (n/2) / sum_h
                    h_candidate = np.clip(h_candidate, 0, 1)
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5_n:
                    best_c5_n = c5
                    best_h_n = h_candidate

        if best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n

        elapsed_time += time.time() - start_time

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - len(best_h)/2) > 1e-6:
        best_h = best_h * (len(best_h)/2) / sum_h
        best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, len(best_h))
