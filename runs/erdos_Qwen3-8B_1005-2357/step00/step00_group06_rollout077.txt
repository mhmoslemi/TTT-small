    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum

    # Function to project integral constraint
    def project_integral(h):
        current_sum = np.sum(h)
        target = target_sum
        diff = target - current_sum
        if abs(diff) < 1e-10:
            return h
        # Adjust by distributing the difference evenly
        h += diff / n_points
        h = np.clip(h, 0, 1)
        # Final adjustment to ensure sum is correct
        h = h / np.sum(h) * target_sum
        return h

    # Define the objective function with penalty
    def objective(h, mu):
        conv = np.correlate(h, 1 - h, mode='full')
        max_conv = np.max(conv)
        penalty = mu * (np.sum(h) - target_sum)**2
        return max_conv * dx + penalty

    # Define the gradient function
    def gradient(h, mu):
        conv = np.correlate(h, 1 - h, mode='full')
        max_conv = np.max(conv)
        max_indices = np.where(conv == max_conv)[0]
        grad = np.zeros_like(h)
        for k in max_indices:
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if 0 <= i - k < n_points:
                    grad[i] -= h[i - k] * dx
        grad += 2 * mu * (np.sum(h) - target_sum)
        return grad

    # Optimization parameters
    mu_values = [0.0, 0.1, 0.2]
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    for mu in mu_values:
        h_perturbed = h.copy() + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = project_integral(h_perturbed)

        def objective_with_mu(h_vec):
            return objective(h_vec, mu)

        def gradient_with_mu(h_vec):
            return gradient(h_vec, mu)

        result = minimize(
            fun=objective_with_mu,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=gradient_with_mu,
            options={'maxiter': 10000, 'ftol': 1e-6}
        )

        if result.success:
            current_h = result.x
            current_h = project_integral(current_h)
            current_c5 = objective(current_h, mu)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = project_integral(best_h)
    conv = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv) * dx
    return (best_h.tolist(), final_c5, n_points)
