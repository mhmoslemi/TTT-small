        if n == n_values[0]:
            if 'initial_h_values' in globals():
                guess = initial_h_values
            else:
                guess = np.full(n, 0.5)
        else:
            guess = np.full(n, 0.5)

        start = time.time()
        dx = 2.0 / n
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            return np.max(corr) * dx
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n/2})
        bounds = [(0.0, 1.0) for _ in range(n)]
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, options={'gtol': 1e-8, 'maxiter': 1000})
        elapsed = time.time() - start
        elapsed_time += elapsed

        if res.success:
            h = np.clip(res.x, 0.0, 1.0)
            s = np.sum(h)
            if abs(s - n/2) > 1e-6:
                h *= (n/2) / s
            c5 = res.fun
        else:
            h = np.clip(guess, 0.0, 1.0)
            s = np.sum(h)
            h *= (n/2) / s
            c5 = obj(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = initial_h_values
        else:
            best_h = np.full(47, 0.5)
        best_c5 = obj(best_h)

    return best_h, best_c5, len(best_h)
