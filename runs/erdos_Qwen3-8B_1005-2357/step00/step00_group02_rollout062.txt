import numpy as np
from scipy.optimize import minimize
import time

def objective(h, n, dx):
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def grad_objective(h, n, dx):
    grad = np.zeros(n)
    delta = 1e-5
    for i in range(n):
        h_p = h.copy()
        h_p[i] = np.clip(h[i]+delta, 0, 1)
        obj_p = objective(h_p, n, dx)
        h_m = h.copy()
        h_m[i] = np.clip(h[i]-delta, 0, 1)
        obj_m = objective(h_m, n, dx)
        grad[i] = (obj_p - obj_m) / (2*delta)
    return grad

def constraint(h, n):
    return n/2 - np.sum(h)
