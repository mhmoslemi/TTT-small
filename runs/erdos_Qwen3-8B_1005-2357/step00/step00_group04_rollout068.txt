import numpy as np
import time
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern
import scipy.stats

def project_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    target_sum = n_points / 2
    scale = target_sum / sum_h
    h = h * scale
    return h

def evaluate_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = project_feasibility(h, n_points)
    c5_initial = evaluate_overlap(h, dx)
    best_c5 = c5_initial
    best_h = h.copy()
    
    X_initial = [h]
    y_initial = [c5_initial]
    kernel = Matern(nu=1.5)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
    gp.fit(np.array(X_initial), np.array(y_initial))
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        def acquisition(x):
            x_proj = project_feasibility(x, n_points)
            mean, std = gp.predict([x_proj], return_std=True)
            if std == 0:
                ei = 0.0
            else:
                z = (best_c5 - mean) / std
                ei = (best_c5 - mean) * scipy.stats.norm.cdf(z) + std * scipy.stats.norm.pdf(z)
            return -ei
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(acquisition, x0=best_h, method='L-BFGS-B', bounds=bounds)
        h_candidate = result.x
        h_proj = project_feasibility(h_candidate, n_points)
        c5_candidate = evaluate_overlap(h_proj, dx)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_proj.copy()
        
        X_new = [h_proj]
        y_new = [c5_candidate]
        gp.fit(np.array(X_initial + X_new), np.array(y_initial + y_new))
    
    return (best_h, best_c5, n_points)
