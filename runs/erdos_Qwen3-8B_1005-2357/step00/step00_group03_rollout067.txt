import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial local optimization
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
    best_h = result.x
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-5:
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    
    # Refine with differential evolution
    def de_objective(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    result_de = differential_evolution(de_objective, bounds, strategy='best1bin',

    best_h_de = result_de.x
