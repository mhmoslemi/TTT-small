import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Global optimization with COBYLA
    result = minimize(objective, initial_h, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 100})
    if result.success:
        h_opt = result.x
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Local refinement with trust-constrained
    result = minimize(objective, best_h, method='trust-constrained', bounds=bounds, constraints=cons, options={'maxiter': 50})
    if result.success:
        h_opt = result.x
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Check time and attempt higher resolution if possible
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        n_points_new = 100
        dx_new = 2.0 / n_points_new
        np.random.seed(seed)
        h_new = np.random.rand(n_points_new)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        def obj_new(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_new
        cons_new = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_new / 2}]
        result = minimize(obj_new, h_new, method='COBYLA', bounds=[(0,1)]*n_points_new, constraints=cons_new, options={'maxiter': 50})
        if result.success:
            h_opt = result.x
            c5 = obj_new(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
                n_points = n_points_new
        else:
            n_points = len(best_h)

    return (best_h, best_c5, n_points)
