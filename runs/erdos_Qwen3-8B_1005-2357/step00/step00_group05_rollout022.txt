import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize from parent if available
    if 'initial_h_values' in globals():

        n_points = len(h_initial)

        h_initial = np.random.rand(n_points)

        h_initial = np.clip(h_initial, 0, 1)
    
    def obj_func(x):
        return evaluate_erdos_solution(x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0, 1)] * n_points
