        time_budget = time_budgets[n]
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < time_budget:
            time_budget = remaining
        while time.time() - start_time < budget_s:
            offspring = []
            for parent in population:
                noise = np.random.normal(0, 0.1, n_points)
                mutated = parent + noise
                mutated = repair(mutated, n_points)
                offspring.append(mutated)
            c5_values = []
            for h in offspring:
                corr = np.correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                dx = 2.0 / n_points
                c5 = max_corr * dx
                c5_values.append(c5)
            indices = np.argsort(c5_values)
            selected = [offspring[i] for i in indices[:50]]
            population = selected
            for h in population:
                corr = np.correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                dx = 2.0 / n_points
                c5 = max_corr * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                    best_n = n_points
            if time.time() - start_time >= budget_s:
                break

    return (best_h, best_c5, best_n)
