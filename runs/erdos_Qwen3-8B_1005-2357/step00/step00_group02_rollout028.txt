import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h_flat):
        h = h_flat.copy()
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        k = max_corr_idx
        grad = np.zeros(n)
        for i in range(n):
            term1 = (1 - h[i + k]) * dx if i + k < n else 0.0
            term2 = h[i - k] * dx if i - k >= 0 else 0.0
            grad[i] = term1 - term2
        return grad

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    bounds = [(0, 1) for _ in range(n_points)]

    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    best_h = np.clip(result.x, 0, 1)
    current_sum = np.sum(best_h)
    if abs(current_sum - n_points / 2) > 1e-6:
        diff = (n_points / 2) - current_sum
        best_h[0] += diff
        best_h = np.clip(best_h, 0, 1)

    return (best_h, objective(best_h), n_points)
