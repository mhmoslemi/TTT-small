import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    h = np.array([], dtype=float)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)

        max_corr = np.max(corr)
        return max_corr * dx

    def clip_and_scale(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        scale = (n_points / 2) / total
        return h_clip * scale

    def generate_feasible():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * (n_points / 2) / total

    best_h = h.copy()

    T_start = 10.0
    cooling_rate = 0.99
    start_time = time.time()

    for _ in range(100000):

        if elapsed > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            jf = i + s
            jb = i - s
            t1 = 1 - h[jf] if 0 <= jf < n_points else 0
