    initial_h = initial_h_values
    target_sum = n_points / 2
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - target_sum) > 1e-3:
        initial_h = initial_h * (target_sum / initial_sum)
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    best_h = result.x
    best_c5 = objective(best_h)
    
    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-3:
        best_h = best_h * (target_sum / sum_h)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        n_new = 100
        h_transferred = transfer_solution(best_h, n_points, n_new)
        target_sum_new = n_new / 2
        dx_new = 2.0 / n_new
        
        def objective_new(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_new
        
        def constraint_new(h):
            return np.sum(h) - target_sum_new
        
        bounds_new = [(0, 1)] * n_new
        result_new = minimize(objective_new, h_transferred, method='L-BFGS-B', bounds=bounds_new, constraints={'type': 'eq', 'fun': constraint_new})
        best_h_new = result_new.x
        sum_h_new = np.sum(best_h_new)
        if abs(sum_h_new - target_sum_new) > 1e-3:
            best_h_new = best_h_new * (target_sum_new / sum_h_new)
            best_h_new = np.clip(best_h_new, 0, 1)
        best_c5_new = objective_new(best_h_new)
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
            n_points = n_new
    
    return (best_h, best_c5, n_points)
