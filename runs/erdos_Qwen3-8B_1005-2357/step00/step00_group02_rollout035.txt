        h = np.array(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h_vec):
        h = np.array(h_vec)
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            obj_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if result.success:
        h = result.x
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            delta = (n_points / 2 - total) / n_points
            h += delta
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    else:
        h = initial_h_values.copy()
        c5 = objective(h)
        best_c5 = c5
        best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:
        n_points_new = 100
        dx_new = 2.0 / n_points_new
        h_new = np.zeros(n_points_new)
        h_new[:n_points] = initial_h_values
        h_new[n_points:] = initial_h_values
        h_new = np.repeat(h_new, 2)
        h_new = h_new[:n_points_new]
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new)
        delta = (n_points_new / 2 - total) / n_points_new
        h_new += delta
        
        def objective_new(h_vec):
            h = np.array(h_vec)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_new
        
        def gradient_new(h_vec):
            h = np.array(h_vec)
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                obj_plus = objective_new(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                obj_minus = objective_new(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * eps)
            return grad
        
        cons_new = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_new / 2}]
        bounds_new = [(0, 1) for _ in range(n_points_new)]
        
        result_new = minimize(
            fun=objective_new,
            x0=h_new,
            method='SLSQP',
            jac=gradient_new,
            bounds=bounds_new,
            constraints=cons_new,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result_new.success:
            h_new = result_new.x
            h_new = np.clip(h_new, 0, 1)
            total = np.sum(h_new)
            if abs(total - n_points_new / 2) > 1e-6:
                delta = (n_points_new / 2 - total) / n_points_new
                h_new += delta
            c5_new = objective_new(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                n_points = n_points_new
    
    return (best_h, best_c5, n_points)
