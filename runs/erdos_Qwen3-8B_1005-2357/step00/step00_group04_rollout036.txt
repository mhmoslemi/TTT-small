        dx_new = 2.0 / n_points
        h_upscaled = np.repeat(best_h, 2)
        total = np.sum(h_upscaled) * dx_new
        h_upscaled = h_upscaled * (1.0 / total)
        best_c5 = evaluate_overlap(h_upscaled)
        best_h = h_upscaled.copy()
        
        max_iter = 1000
        step_size = 0.01
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            k0 = np.argmax(corr)
            grad = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= j + k0 < n_points:
                    grad[j] += (1 - best_h[j + k0]) * dx_new
                if 0 <= j - k0 < n_points:
                    grad[j] -= best_h[j - k0] * dx_new
            best_h -= step_size * grad
            best_h = np.clip(best_h, 0, 1)
            total = np.sum(best_h) * dx_new
            if abs(total) < 1e-6:
                best_h = np.ones_like(best_h) * 0.5
            else:
                best_h = best_h * (1.0 / total)
            current_c5 = evaluate_overlap(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
            else:
                break
    
    return (best_h, best_c5, n_points)
