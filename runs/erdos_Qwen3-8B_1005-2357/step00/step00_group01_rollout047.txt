import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    best_h = None
    best_c5 = float('inf')
    best_n = 47
    time_used = 0.0

    def repair(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n / 2) < 1e-6:
            return h
        scale = (n / 2) / total if total != 0 else 1.0
        h = h * scale
        return h

    def objective(h, n):
        dx = 2.0 / n
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = np.max(cor * dx)
        return overlap

    def constraint_sum(h, n):
        return np.sum(h) - n / 2

    def bounds(n):
        return [(0, 1) for _ in range(n)]

    def generate_initial(n, parent=None):
        if parent is not None:
            h = np.copy(parent)
            h += np.random.uniform(-0.1, 0.1, n)
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if abs(total - n / 2) > 1e-3:
                h = h * (n / 2) / total
            return h
        else:
            return np.random.dirichlet(np.ones(n), size=1)[0]

    def evaluate(h, n):
        dx = 2.0 / n
        cor = np.correlate(h, 1 - h, mode='full')
        overlap = np.max(cor * dx)
        return overlap

    def optimize(n, parent=None):
        h = generate_initial(n, parent)
        h = repair(h, n)
        res = opt.minimize(
            lambda x: objective(x, n),
            x0=h,
            method='SLSQP',
            bounds=bounds(n),
            constraints={'type': 'eq', 'fun': constraint_sum, 'args': (n,)},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            h_opt = repair(res.x, n)
            c5 = evaluate(h_opt, n)
            return h_opt, c5
        return h, float('inf')

    for n in [47, 94, 188, 376]:
        if time_used >= budget_s:
            break
        if n < 47:
            continue
        for _ in range(5):
            if time_used >= budget_s:
                break
            start = time.time()
            h, c5 = optimize(n, initial_h if n == 47 else None)
            time_used += time.time() - start
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = n
    return best_h, best_c5, best_n
