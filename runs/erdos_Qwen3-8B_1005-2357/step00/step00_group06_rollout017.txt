        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
        lags = [idx - (n-1) for idx in max_indices]
        grad = np.zeros(n)
        for i in range(n):
            for k in lags:
                term1 = 1.0 - h[i + k] if (i + k < n) else 0.0
                term2 = h[i - k] if (i - k >= 0) else 0.0
                grad[i] += (term1 - term2)
        return grad * dx
    
    for n in n_points_list:
        if initial_h is not None and n == 95:
            h = initial_h.copy()
        else:
            h = np.random.rand(n)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
        result = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n, options={'maxiter': 1000})
        if time.time() - start_time > budget_s:
            break
        h_result = result.x
        h_result = np.clip(h_result, 0, 1)
        total = np.sum(h_result)
        scale = (n / 2) / total if total != 0 else 1.0
        h_result *= scale
        dx = 2.0 / n
        g = 1.0 - h_result
        corr = np.correlate(h_result, g, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_result
            best_n = n
    return (best_h, best_c5, best_n)
