import numpy as np
import time

def evaluate_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasibility(h, dx):
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped) * dx
    if abs(total - 1) > 1e-6:
        scale = 1.0 / total
        h_clipped *= scale
    return h_clipped
