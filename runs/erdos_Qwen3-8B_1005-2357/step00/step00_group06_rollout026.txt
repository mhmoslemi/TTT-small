        for i in range(n):
            if 0 <= i + k < n:
                grad[i] += (1 - h_vec[i + k]) * dx
                grad[i + k] -= h_vec[i] * dx
        grad += 2 * mu * (np.sum(h_vec) - target_sum) * np.ones(n)
        return grad
    
    for mu in mu_values:
        h_perturbed = h.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        delta = (target_sum - current_sum) / n_points
        h_perturbed += delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        start_time = time.time()
        result = minimize(
            lambda x: objective(x, mu),
            h_perturbed,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=lambda x: gradient(x, mu),
            options={'maxiter': 1000, 'ftol': 1e-6}
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        h_opt = result.x
        current_sum = np.sum(h_opt)
        if abs(current_sum - target_sum) > 1e-5:
            delta = (target_sum - current_sum) / n_points
            h_opt += delta
            h_opt = np.clip(h_opt, 0, 1)
        
        conv = np.correlate(h_opt, 1 - h_opt, mode='full')
        current_c5 = np.max(conv) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    current_sum = np.sum(best_h)
    delta = (target_sum - current_sum) / n_points
    best_h += delta
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, n_points
