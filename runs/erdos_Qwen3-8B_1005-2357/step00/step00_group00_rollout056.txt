import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2
    current_sum = np.sum(initial_h)
    delta = target_sum - current_sum
    initial_h[-1] += delta
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()
    num_starts = 5

        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.1, size=n_points)
        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        current_sum = np.sum(h_perturb)
        delta = target_sum - current_sum
        h_perturb[-1] += delta
        def objective(x):
            h_clipped = np.clip(x, 0, 1)
            sum_clipped = np.sum(h_clipped)
            delta_sum = target_sum - sum_clipped
            h_clipped += delta_sum / n_points
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            max_corr = np.max(corr)

        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_sum = np.sum(current_h)
            delta = target_sum - current_sum
            current_h += delta / n_points
            corr = np.correlate(current_h, 1 - current_h, mode='full')

            c5 = max_corr * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
