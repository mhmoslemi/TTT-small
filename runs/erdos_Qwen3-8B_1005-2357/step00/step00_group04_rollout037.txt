    dx = 2.0 / n_points
    max_iterations = 1000
    step_size = 0.01
    elapsed_time = 0
    start_time = time.time()
    
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k0 = m - (n_points - 1)
        i_plus_k0 = np.arange(n_points) + k0
        valid = (i_plus_k0 >= 0) & (i_plus_k0 < n_points)
        gradient = np.zeros_like(h)
        gradient[valid] = dx * (1 - h[i_plus_k0[valid]])
        h -= step_size * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 21 / sum_h
        h = h * scale
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_corr) * dx
