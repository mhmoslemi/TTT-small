        if abs(sum_h - n_points / 2) > 1e-6:
            h_initial = np.full(n_points, 0.5)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n_points)
    n_points = 47

    for _ in range(10):
        h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_candidate *= scale
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)
        if abs(sum_h - n_points / 2) > 1e-6:
            delta = (n_points / 2) - sum_h
            h_candidate += delta / n_points
            h_candidate = np.clip(h_candidate, 0, 1)
        result = minimize(lambda x: compute_c5(x, n_points), h_candidate, method='trust-constr',
                          bounds=Bounds(0, 1), constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                          options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    return (best_h, best_c5, n_points)
