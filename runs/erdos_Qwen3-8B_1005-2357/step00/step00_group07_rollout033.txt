import time

def run(seed=42, budget_s=1100, **kwargs):

    else:

            h = np.random.uniform(0, 1, n_points)

            total = np.sum(h_clip)
            return h_clip * (n_points / 2) / total
        h_pop = [gen_sol() for _ in range(population_size)]
    best_h = h_pop[0]
    best_c5 = compute_C5(best_h)
    start_time = time.time()

        if time.time() - start_time > budget_s:

                best_c5 = c5
                best_h = h.copy()
        h = best_h.copy()
        corr = np.correlate(h, 1 - h, mode='full')

        k = max_idx - (n_points - 1)

        mask_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
        term1 = np.where(mask_plus, 1 - h[i_plus_k], 0)
        i_minus_k = np.arange(n_points) - k
        mask_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
        term2 = np.where(mask_minus, h[i_minus_k], 0)
        grad = term1 - term2
        alpha = [0.01, 0.1, 0.5, 1.0]
        best_a = 0.0
        best_c = float('inf')

                best_a = a
