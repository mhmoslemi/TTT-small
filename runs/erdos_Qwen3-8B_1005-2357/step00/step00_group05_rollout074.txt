        return (result_cobyla.x, objective(result_cobyla.x), n_points)

    # Local refinement with trust-constr
    result_trust = minimize(objective, result_cobyla.x, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (result_trust.x, objective(result_trust.x), n_points)

    # Check time for higher resolution
    if elapsed < budget_s:
        n_points = 100
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = h * scale

        def objective_100(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            return np.max(corr) * dx

        cons_100 = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds_100 = Bounds(0, 1)

        result_cobyla_100 = minimize(objective_100, h, method='COBYLA', bounds=bounds_100, constraints=cons_100, tol=1e-6, options={'maxiter': 100})
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return (result_cobyla_100.x, objective_100(result_cobyla_100.x), n_points)

        result_trust_100 = minimize(objective_100, result_cobyla_100.x, method='trust-constr', bounds=bounds_100, constraints=cons_100, tol=1e-6, options={'maxiter': 100})
        elapsed = time.time() - start_time

        c5_70 = objective(result_trust.x)
        c5_100 = objective_100(result_trust_100.x)
        if c5_100 < c5_70:
            best_h = result_trust_100.x
            best_c5 = c5_100
            best_n = n_points
        else:
            best_h = result_trust.x
            best_c5 = c5_70
            best_n = n_points

    else:
        best_h = result_trust.x
        best_c5 = objective(result_trust.x)
        best_n = n_points

    # Ensure sum constraint is met
    sum_h = np.sum(best_h)
    if abs(sum_h - best_n / 2) > 1e-3:
        scale = (best_n / 2) / sum_h
        best_h = best_h * scale
        best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, best_n)
