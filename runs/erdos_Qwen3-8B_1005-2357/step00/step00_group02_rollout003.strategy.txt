   - Constraints: Box constraints \([0, 1]\) and equality constraint \(\sum h_i = 22.5\).

   - Transfer the best n=45 solution to n=100 via interpolation and scaling.

   - Ensure integral constraint is satisfied iteratively.

4. **Validation**:
   - Verify \(0 \leq h_i \leq 1\) and \(\sum h_i \times \Delta x \approx 1\).
   - Compute \(C_5\) for the best solution.

### Search Choices and Variations
- **Gradient Perturbation**: \(\delta = 1e-5\).
- **Time Budget**:
  - n=45: 90% of total time.
  - n=100: Up to 100 seconds.
- **Fallback**: Use initial solution if optimization fails.

### Compute Budget and Robustness
- **Complexity**: \(O(n^2)\) per gradient evaluation.
- **n=45**: ~20 million operations, feasible in 990 seconds.
- **n=100**: ~200 million operations, feasible in 100 seconds.
- **Fallback**: Return best solution found.

### Validation and Return Contract
- **Feasibility**: \(0 \leq h_i \leq 1\) and \(\sum h_i \times \Delta x = 1\).
- **Objective**: \(C_5 = \max(\text{np.correlate}(h, 1-h, \text{mode}="full") \times \Delta x\).
- **Output**: `(h_values, c5_bound, n_points)`.

import numpy as np
from scipy.optimize import minimize
import time

def objective(h, n, dx):
    k = 1 - h
    corr = np.correlate(h, k, mode='full')
    return np.max(corr) * dx

def grad_objective(h, n, dx):
    n_val = len(h)
    grad = np.zeros(n_val)
    delta = 1e-5
    for i in range(n_val):
        h_plus = h.copy()
        h_plus[i] = min(1, h[i] + delta)
        h_minus = h.copy()
        h_minus[i] = max(0, h[i] - delta)
        obj_plus = objective(h_plus, n_val, dx)
        obj_minus = objective(h_minus, n_val, dx)
        grad[i] = (obj_plus - obj_minus) / (2 * delta)
    return grad

def constraint(h, n):
    return n/2 - np.sum(h)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    # Use provided initial solution if available
    if 'initial_h_values' in globals():
        h0 = initial_h_values
    else:
        # Fallback: Construct random feasible solution for n=45
        n_val = 45
        h0 = np.random.rand(n_val)
        total = np.sum(h0)
        scale = 22.5 / total
        h0 = h0 * scale
        h0 = np.clip(h0, 0, 1)
        total_clip = np.sum(h0)
        scale2 = 22.5 / total_clip
        h0 = h0 * scale2

    # Compute initial C5
    n1 = 45
    dx1 = 2.0 / n1
    c5_initial = objective(h0, n1, dx1)
    best_h = h0.copy()
    best_c5 = c5_initial
    n_points = n1

    # Phase 1: n=45 optimization
    start_time = time.time()
    max_time_phase1 = budget_s * 0.9

    def objective_time(h):
        current_time = time.time()
        if current_time - start_time > max_time_phase1:
            raise KeyboardInterrupt
        return objective(h, n1, dx1)

    def grad_objective_time(h):
        current_time = time.time()
        if current_time - start_time > max_time_phase1:
            raise KeyboardInterrupt
        return grad_objective(h, n1, dx1)

    bounds = [(0, 1)] * n1
    cons = {'type': 'eq', 'fun': lambda h: constraint(h, n1)}

    try:
        res = minimize(objective_time, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad_objective_time, options={'maxiter': 10000})
    except KeyboardInterrupt:
        pass

    if 'x' in res and res.success:
        h_candidate = res.x
        c5_candidate = objective(h_candidate, n1, dx1)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            n_points = n1

    # Phase 2: n=100 optimization if time permits
    time_remaining = budget_s - time.time()
    if time_remaining > 100:
        n2 = 100
        dx2 = 2.0 / n2

        # Transfer solution from n=45 to n=100
        x45 = np.linspace(0, 2, 45)
        x100 = np.linspace(0, 2, 100)
        h100_init = np.interp(x100, x45, best_h)
        current_sum = np.sum(h100_init)
        scale = 50 / current_sum
        h100_init = h100_init * scale

        # Iterative clipping and scaling to satisfy constraints
        max_iter = 5
        for _ in range(max_iter):
            h100_init = np.clip(h100_init, 0, 1)
            current_sum = np.sum(h100_init)
            if np.abs(current_sum - 50) < 1e-5:
                break
            scale = 50 / current_sum
            h100_init = h100_init * scale

        start_time_phase2 = time.time()
        max_time_phase2 = 100

        def objective_time2(h):
            current_time = time.time()
            if current_time - start_time_phase2 > max_time_phase2:
                raise KeyboardInterrupt
            return objective(h, n2, dx2)

        def grad_objective_time2(h):
            current_time = time.time()
            if current_time - start_time_phase2 > max_time_phase2:
                raise KeyboardInterrupt
            return grad_objective(h, n2, dx2)

        bounds2 = [(0, 1)] * n2
        cons2 = {'type': 'eq', 'fun': lambda h: constraint(h, n2)}

        try:
            res2 = minimize(objective_time2, h100_init, method='SLSQP', bounds=bounds2, constraints=cons2, jac=grad_objective_time2, options={'maxiter': 1000})
        except KeyboardInterrupt:
            pass

        if 'x' in res2 and res2.success:
            h_candidate = res2.x
            c5_candidate = objective(h_candidate, n2, dx2)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                n_points = n2

    # Validate solution
    if n_points == 45:
        dx45 = 2.0 / 45
        assert 0 <= min(best_h) and max(best_h) <= 1, "h values out of bounds"
        assert np.isclose(np.sum(best_h) * dx45, 1.0, atol=1e-5), "Integral constraint not satisfied"
    elif n_points == 100:
        dx100 = 2.0 / 100
        assert 0 <= min(best_h) and max(best_h) <= 1, "h values out of bounds"
        assert np.isclose(np.sum(best_h) * dx100, 1.0, atol=1e-5), "Integral constraint not satisfied"

    return (best_h, best_c5, n_points)
