import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    n_list = [47, 94, 188, 376]
    time_per_n = [550, 300, 200, 50]
    for idx, n in enumerate(n_list):

        for h_initial in initial_points:
            dx = 2.0 / n
            def obj(x):
                h = np.clip(x, 0, 1)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            def cons(x):
                return np.sum(x) - n / 2
            bounds = [(0.0, 1.0) for _ in range(n)]
