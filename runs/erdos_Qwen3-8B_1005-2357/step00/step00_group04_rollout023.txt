    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    step_size = 0.01
    max_iterations = 1000
    for _ in range(max_iterations):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(cross_corr)
        j = np.argmax(cross_corr)
        k0 = j - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k0 = i + k0
            i_minus_k0 = i - k0
            term1 = 0.0
            if 0 <= i_plus_k0 < n_points:
                term1 = 1.0 - h[i_plus_k0]
            term2 = 0.0
            if 0 <= i_minus_k0 < n_points:
                term2 = h[i_minus_k0]
            gradient[i] = dx * (term1 - term2)
