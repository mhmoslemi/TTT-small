def evaluate_erdos_solution(h, n, dx=2.0/n):
    """Compute the Erdos overlap constant C5 for given h"""
    # Compute 1-h
    h_comp = 1 - h
    # Compute full correlation
    corr = np.zeros(2*n-1)
    for lag in range(-(n-1), n):
        lag_idx = lag + (n-1)
        c_val = 0.0
        # For each valid i where i+lag is in [0, n-1]
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                c_val += h[i] * h_comp[j]
        corr[lag_idx] = c_val
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n):
    """Project h onto feasible set [0,1]^n with sum(h)=n/2"""
    # Clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Adjust sum to n/2
    total = np.sum(h_clipped)
    if total == 0:
        # If all zeros, set one element to 0.5
        h_clipped[0] = 0.5
        return h_clipped
    scale = (n/2) / total
    h_scaled = h_clipped * scale
    # Clip again to avoid exceeding bounds
    h_final = np.clip(h_scaled, 0, 1)

    if np.any(initial_h_values < 0) or np.any(initial_h_values > 1):
        initial_h_values = np.clip(initial_h_values, 0, 1)
        total = np.sum(initial_h_values)
        scale = 45/2 / total
        initial_h_values = initial_h_values * scale
    
    # Best solution tracker
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, 45)
    best_n = 45
    
    # Time tracking
    start_time = time.time()
    elapsed = 0
    
    # Optimization parameters
    n_runs_per_resolution = 10
    n_list = [45, 100]
    n_per_run = 1  # Number of optimization runs per resolution
    
    # Process each resolution
    for n in n_list:
        if elapsed >= budget_s:
            break
            
        # Time allocation
        if n == 45:
            time_per_run = 100  # 100 seconds per run
            max_runs = min(n_runs_per_resolution, (budget_s - elapsed) // time_per_run)
        else:  # n=100
            time_per_run = 10   # 10 seconds per run
            max_runs = min(n_runs_per_resolution, (budget_s - elapsed) // time_per_run)
        
        # Compute dx
        dx_val = 2.0 / n
        
        # Prepare initial guess for this resolution
        if n == 45:
            # Use initial_h_values directly
            h0 = initial_h_values
        else:
            # For n=100, upsample from n=45
            x_initial = np.linspace(0, 2, 45)
            x_new = np.linspace(0, 2, 100)
            # Linear interpolation
            h0 = np.interp(x_new, x_initial, initial_h_values)
            # Clip and adjust sum
            h0 = np.clip(h0, 0, 1)
            total = np.sum(h0)
            if abs(total - 100/2) > 1e-5:
                scale = (100/2) / total
                h0 = h0 * scale
        
        # Optimization function definitions
        def objective(h_vec):
            h_vec = np.asarray(h_vec)
            if len(h_vec) != n:
                raise ValueError(f"Expected length {n}, got {len(h_vec)}")
            return evaluate_erdos_solution(h_vec, n, dx_val)
        
        def grad_objective(h_vec):
            h_vec = np.asarray(h_vec)
            if len(h_vec) != n:
                raise ValueError(f"Expected length {n}, got {len(h_vec)}")
            # Finite-differences gradient
            grad = np.zeros(n)
            eps = 1e-4
            for i in range(n):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                obj_plus = objective(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                obj_minus = objective(h_minus)
                
                grad[i] = (obj_plus - obj_minus) / (2 * eps)
            return grad
        
        # Constraints
        def constraint(h_vec):
            h_vec = np.asarray(h_vec)
            return np.sum(h_vec) - n/2
        
        # Bounds and constraints
        bounds = [(0, 1)] * n
        constraints = [{'type': 'eq', 'fun': constraint}]
        
        # Optimization loop
        for run_idx in range(max_runs):
            if elapsed >= budget_s:
                break
                
            # Perform one optimization run
            try:
                result = minimize(
                    objective,
                    h0,
                    method='SLSQP',
                    jac=grad_objective,
                    bounds=bounds,
                    constraints=constraints,
                    options={'disp': False}
                )
                
                # Check if solution is feasible
                if np.any(np.isnan(result.x)) or np.any(result.x < 0) or np.any(result.x > 1):
                    # Infeasible solution, try projected version
                    h_proj = project_to_feasible(result.x, n)
                    c5_proj = objective(h_proj)
                    if c5_proj < best_c5:
                        best_c5 = c5_proj
                        best_h = h_proj.copy()
                        best_n = n
                    continue
                
                # Check feasibility and objective
                h_candidate = result.x
                total_sum = np.sum(h_candidate)
                if abs(total_sum - n/2) > 1e-5:
                    # Infeasible, project and check
                    h_proj = project_to_feasible(h_candidate, n)
                    c5_proj = objective(h_proj)
                    if c5_proj < best_c5:
                        best_c5 = c5_proj
                        best_h = h_proj.copy()
                        best_n = n
                    continue
                
                c5_candidate = result.fun
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    best_n = n
                
            except Exception as e:
                # Optimization failed, try projected version
                h_proj = project_to_feasible(h0, n)
                c5_proj = objective(h_proj)
                if c5_proj < best_c5:
                    best_c5 = c5_proj
                    best_h = h_proj.copy()
                    best_n = n
            
            # Update elapsed time
            elapsed = time.time() - start_time
            h0 = result.x  # Use last result as new starting point
        
        # Update elapsed time
        elapsed = time.time() - start_time
    
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    total_sum = np.sum(best_h)
    if abs(total_sum - best_n/2) > 1e-5:
        best_h = project_to_feasible(best_h, best_n)
    
    # Compute final C5
    final_c5 = evaluate_erdos_solution(best_h, best_n)
    
    # Return best solution
    return best_h, final_c5, best_n
