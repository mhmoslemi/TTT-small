import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    h_initial = initial_h_values if 'initial_h_values' in kwargs else None
    
    if h_initial is None:
        h_values = np.random.uniform(0, 1, n_points)
        target_sum = n_points / 2
        h_values += (target_sum - np.sum(h_values)) / n_points
    else:
        h_values = h_initial.copy()
    
    def objective(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(objective, h_values, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
