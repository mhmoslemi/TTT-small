            if total_time >= budget_s:
                break
        # Final check of current population
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    if best_h is None:
        if initial_h is not None:
            best_h = repair_h(np.array(initial_h), len(initial_h))
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            best_c5 = np.max(corr) * (2.0 / len(best_h))
        else:
            n = 47
            h = np.random.rand(n)
            h = repair_h(h, n)
            corr = np.correlate(h, 1 - h, mode='full')
            best_c5 = np.max(corr) * (2.0 / n)
            best_h = h
    return (best_h, best_c5, len(best_h))
