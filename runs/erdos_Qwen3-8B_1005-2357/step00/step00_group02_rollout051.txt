from scipy.optimize import minimize
import time

def objective(h, n, dx):
    k = 1 - h
    corr = np.correlate(h, k, mode='full')
    return np.max(corr) * dx

def grad_objective(h, n, dx):
    grad = np.zeros(n)
    delta = 1e-5
    for i in range(n):
