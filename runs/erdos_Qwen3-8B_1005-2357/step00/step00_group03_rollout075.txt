    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if abs(candidate_sum - target_sum) > 1e-5:
            scale = target_sum / candidate_sum
            candidate = candidate * scale
        population.append(candidate)
    
    for gen in range(gens):
        if time.time() - start_time > time_left:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + np.random.uniform(-1, 1, n_points) * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            if abs(mutant_sum - target_sum) > 1e-5:
                scale = target_sum / mutant_sum
                mutant = mutant * scale
            trial = np.where(np.random.rand(n_points) < 0.5, mutant, best_h)
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if abs(trial_sum - target_sum) > 1e-5:
                scale = target_sum / trial_sum
                trial = trial * scale
            new_pop.append(trial)
        population = new_pop
    
    # Evaluate best from DE
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # SLSQP refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        return grad
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(objective, best_h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if abs(best_sum - target_sum) > 1e-5:
        scale = target_sum / best_sum
        best_h = best_h * scale
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
