        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        max_idx = np.where(conv == max_val)[0]
        grad = np.zeros_like(h_vec)
        for k in max_idx:
            for i in range(n_points):
                j_plus_k = i + k
                if 0 <= j_plus_k < n_points:
                    grad[i] += (1 - h_vec[j_plus_k])
                j = i - k
                if 0 <= j < n_points:
                    grad[i] -= h_vec[j]
        grad *= dx
        return grad
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds, tol=1e-8, options={'maxiter': 1000})
    h_values = result.x if result.success else h
    
    sum_h = np.sum(h_values)
    target_sum = n_points / 2
    if abs(sum_h - target_sum) > 1e-6:
        delta = (target_sum - sum_h) / n_points
        h_values[-1] += delta
    
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(conv) * dx
    return (h_values, c5_bound, n_points)
