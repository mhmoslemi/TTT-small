        constraint_high = LinearConstraint(np.ones(n_new), [n_new/2], [n_new/2])
        res_high = minimize(objective_high, h_initial_high, method='SLSQP', bounds=bounds_high, constraints=constraint_high, jac=gradient_high, options={'maxiter': 1000})
        best_solution_high = res_high.x
        best_objective_high = res_high.fun

        if best_objective_high < best_objective:
            best_solution = best_solution_high
            best_objective = best_objective_high

    final_corr = np.correlate(best_solution, 1 - best_solution, mode='full')
    final_c5 = np.max(final_corr) * dx
    sum_final = np.sum(best_solution) - n / 2
    if abs(sum_final) > 1e-5:
        adjustment = -sum_final / n
        best_solution += adjustment
        best_solution = np.clip(best_solution, 0, 1)
        sum_final = np.sum(best_solution) - n / 2

    return best_solution, final_c5, n
