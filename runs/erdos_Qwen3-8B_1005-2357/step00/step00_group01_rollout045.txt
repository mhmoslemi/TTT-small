import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_values = [47, 94, 188, 376]
    time_budgets = {47: 550, 94: 300, 188: 200, 376: 50}
    best_c5 = float('inf')
    best_h = None
    best_n = 47

    # Validate initial_h_values if available
    if 'initial_h_values' in globals():

        if len(initial_h) == n_values[0]:

            initial_h[initial_h < 0] = 0
            initial_h[initial_h > 1] = 1
            if np.abs(np.sum(initial_h) - n_values[0]/2) > 1e-6:
                initial_h = (initial_h / np.sum(initial_h)) * (n_values[0]/2)
            initial_h = initial_h.clip(0, 1)
        else:
            initial_h = None
    else:
        initial_h = None

    start_time = time.time()

    for n in n_values:
        if time.time() - start_time > budget_s:
            break
        if n not in time_budgets:
            continue
        budget = time_budgets[n]
        if time.time() - start_time >= budget:
            continue

        # Generate initial guess
        if initial_h is not None and len(initial_h) == n:
            x0 = initial_h.copy()
        else:
            # Dirichlet distribution for random initial guess
            x0 = np.random.default_rng(seed).dirichlet(np.ones(n), size=1)[0]
            x0 = x0 * (n / 2)
            x0 = np.clip(x0, 0, 1)

        # Optimization
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * (n / 2) / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)

        def constraint(x):
            return np.sum(x) - n / 2

        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': constraint}

        result = opt.minimize(
            objective,
            x0,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'gtol': 1e-8, 'maxiter': 1000}
        )

        if result.success:
            current_h = np.clip(result.x, 0, 1)
            current_h = current_h * (n / 2) / np.sum(current_h)
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n = n
        else:
            # Fallback to initial guess if optimization fails
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * (n / 2) / np.sum(x0)
            current_c5 = objective(x0)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x0
                best_n = n

        # Update initial_h for next n if needed
