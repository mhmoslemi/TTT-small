        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(corr) * dx
        integral_penalty = 1000 * abs(sum_h - target_sum)
        return c5 + integral_penalty

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    result = basinhopping(objective, h_initial, niter=100, T=1.0, stepsize=0.5,
                          minimizer_kwargs={"method": "L-BFGS-B", "bounds": bounds})
    elapsed = time.time() - start_time

    h_best = result.x
    h_clipped = np.clip(h_best, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    c5 = np.max(corr) * dx

    if elapsed >= budget_s:
        return (h_best, c5, n_points)

    local_result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds)
    h_best = local_result.x
    h_clipped = np.clip(h_best, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_best, c5, n_points)

    new_n_points = 100
    new_dx = 2.0 / new_n_points
    new_x = np.linspace(0, 2, new_n_points)
    old_x = np.linspace(0, 2, n_points)
    h_upscaled = np.interp(new_x, old_x, h_best)
    h_upscaled = np.clip(h_upscaled, 0, 1)
    target_sum_new = new_n_points / 2.0
    current_sum = np.sum(h_upscaled)
    scale_factor = target_sum_new / current_sum
    h_upscaled = h_upscaled * scale_factor

    def objective_upscaled(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(corr) * new_dx
        integral_penalty = 1000 * abs(sum_h - target_sum_new)
        return c5 + integral_penalty

    new_bounds = [(0, 1)] * new_n_points
    result_upscaled = basinhopping(objective_upscaled, h_upscaled, niter=50, T=1.0, stepsize=0.5,
                                  minimizer_kwargs={"method": "L-BFGS-B", "bounds": new_bounds})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h_clipped = np.clip(result_upscaled.x, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5_upscaled = np.max(corr) * new_dx
        return (result_upscaled.x, c5_upscaled, new_n_points)

    local_result_upscaled = minimize(objective_upscaled, result_upscaled.x, method='L-BFGS-B', bounds=new_bounds)
    h_best_upscaled = local_result_upscaled.x
    h_clipped = np.clip(h_best_upscaled, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    c5_upscaled = np.max(corr) * new_dx

    if c5 < c5_upscaled:
        return (h_best, c5, n_points)
    else:
        return (h_best_upscaled, c5_upscaled, new_n_points)
