    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        sum_penalty = penalty_int * (np.sum(h) - n/2)**2
        box_penalty = penalty_box * np.sum(np.maximum(h - 1, 0) + np.maximum(0 - h, 0))
        return c5 + sum_penalty + box_penalty
    
    population = [initial_h.copy() for _ in range(population_size)]
    fitness = [evaluate_fitness(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        tournament = np.random.choice(range(population_size), size=tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament]
        parent_idx = tournament[np.argmin(tournament_fitness)]
        parent = population[parent_idx].copy()
        
        if np.random.rand() < crossover_rate:
            other_idx = np.random.choice(range(population_size), size=1, replace=False)[0]
            other = population[other_idx].copy()
            child = (parent + other) / 2
        else:
            child = parent.copy()
        
        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, 0.05, size=n)
            child += noise
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child != 0:
                child = (child * (n / 2)) / sum_child
            child = np.clip(child, 0, 1)
        
        child_fitness = evaluate_fitness(child)
        worst_idx = np.argmax(fitness)
        if child_fitness < fitness[worst_idx]:
            population[worst_idx] = child.copy()
            fitness[worst_idx] = child_fitness
            if child_fitness < best_c5:
                best_c5 = child_fitness
                best_h = child.copy()
    
    def objective(x):
        h = x.reshape(n)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(x):
        h = x.reshape(n)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag_index = max_corr_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j = i + lag_index
            if 0 <= j < n:
                grad[i] = (1 - h[j]) * dx
        return grad.flatten()
    
    cons = (
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2},
        {'type': 'ineq', 'fun': lambda x: x - 1},
        {'type': 'ineq', 'fun': lambda x: 1 - x}
    )
    
    result = minimize(objective, best_h, method='SLSQP', jac=gradient, constraints=cons, tol=1e-6, options={'maxiter': 500})
    best_h = result.x
    
    sum_h = np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    if abs(sum_h - n / 2) > 1e-5:
        scale_factor = (n / 2) / sum_h
        best_h = best_h * scale_factor
        best_h = np.clip(best_h, 0, 1)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_corr = np.max(corr)
    best_c5 = max_corr * dx
    
    return best_h, best_c5, n
