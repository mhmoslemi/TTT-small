import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values

    best_c5 = float('inf')

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')

        h = np.clip(h, 0, 1)

        delta = target_sum - current_sum
        if delta:

            h[idx] += delta
            h = np.clip(h, 0, 1)
