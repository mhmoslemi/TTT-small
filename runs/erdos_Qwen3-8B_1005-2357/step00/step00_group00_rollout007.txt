import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values

    def obj_func(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    best_c5 = float('inf')
    best_h = initial_h.copy()

    start_time = time.time()

    def perturb(h):
        h_perturbed = h.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        for i in indices:
            h_perturbed[i] += np.random.normal(0, 0.1)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        delta = (n_points / 2) - np.sum(h_perturbed)
        h_perturbed[-1] += delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed

    for _ in range(5):
        h_start = perturb(initial_h)
        result = minimize(obj_func, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_c5 = obj_func(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if obj_func(initial_h) < best_c5:
        best_c5 = obj_func(initial_h)
        best_h = initial_h

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
