import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / len(initial_h))

    n_points = len(initial_h)
    dx = 2.0 / n_points

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            obj_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    def optimize(n, h0):
        nonlocal time_used, best_h, best_c5, best_n
        start_time = time.time()

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
