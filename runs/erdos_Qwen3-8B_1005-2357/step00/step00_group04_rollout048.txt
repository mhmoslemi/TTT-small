import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 1000
    pop_size = 100
    crossover_prob = 0.9
    mutation_prob = 0.1
    tournament_size = 3
    grad_steps = 100
    grad_step_size = 0.01
    time_limit = budget_s
    start_time = time.time()
    
    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped) * dx
        if abs(total) < 1e-6:
            return np.full_like(h, 0.5)
        return h_clipped * (1.0 / total)
    
    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    def compute_gradient(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            lag = k0 - (len(h) - 1)
            if 0 <= i - lag < len(h):
                grad[i] = (1 - h[i - lag]) * dx
        return grad
    
    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), tournament_size)
        best_idx = np.argmin(fitnesses[indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(parent1)):
            if np.random.rand() < crossover_prob:
                child[i] = parent2[i]
        return child
    
    def mutate(h):
        for i in range(len(h)):
            if np.random.rand() < mutation_prob:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    if initial_h_values is not None:
        h_pop = [np.array(initial_h_values, dtype=np.float64)]
    else:
        h_pop = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    
    h_pop = [repair(h) for h in h_pop]
    best_h = h_pop[0]
    best_c5 = evaluate_overlap(best_h)
    
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        fitnesses = [evaluate_overlap(h) for h in h_pop]
        new_pop = []
        for _ in range(pop_size):
            parent1 = tournament_selection(h_pop, fitnesses)
            parent2 = tournament_selection(h_pop, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = repair(child)
            new_pop.append(child)
        h_pop = new_pop
        current_c5 = evaluate_overlap(h_pop[0])
        if current_c5 < best_c5:
            best_h = h_pop[0]
            best_c5 = current_c5
    
    if time.time() - start_time < time_limit - 500:

    return best_h, final_c5, n_points
