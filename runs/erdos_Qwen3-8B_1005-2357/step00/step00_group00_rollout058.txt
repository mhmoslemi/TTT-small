from scipy.optimize import minimize
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    initial_h = initial_h_values
    
    starts = []

        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)

        h_perturbed[-1] += delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        starts.append(h_perturbed)
    
    best_c5 = float('inf')
    best_h = initial_h
