import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()
    
    n_values = [47, 94, 188, 376]
    time_budgets = [550, 300, 200, 50]
    
    for i, n in enumerate(n_values):
        elapsed = time.time() - start_time
        if i > 0 and elapsed >= budget_s:
            break
        max_time = time_budgets[i]
        if i > 0:
            remaining = budget_s - elapsed
            if remaining < max_time:
                max_time = remaining
        
        for _ in range(5):
            if n == 47:
                h0 = initial_h_values.copy()
            else:
                dirichlet = np.random.dirichlet([1]*n)
                h0 = dirichlet * (n / 2)
            
            def obj(h_flat):
                h = h_flat.reshape(n)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * (2.0 / n)
            
            cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n)) - n/2})
            bounds = [(0, 1) for _ in range(n)]
            
            result = minimize(obj, h0, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 100})
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_n = n
    
    return (best_h, best_c5, best_n)
