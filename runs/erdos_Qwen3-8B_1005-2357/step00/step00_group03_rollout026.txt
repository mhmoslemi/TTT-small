        n_points_new = n_points * 2
        h_new = np.repeat(h_opt, 2)
        h_new = repair(h_new, n_points_new)
        dx_new = 2.0 / n_points_new
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx_new
        if c5_new < c5:
            h_opt = h_new
            c5 = c5_new
            n_points = n_points_new
    
    return h_opt, c5, n_points
