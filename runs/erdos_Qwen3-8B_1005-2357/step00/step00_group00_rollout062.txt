    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(h) for h in population]
        sorted_pop = [p for _, p in sorted(zip(fitness, population))]
        selected = sorted_pop[:int(len(population) * 0.2)]
        new_pop = []
        for i in range(0, len(selected), 2):
            p1, p2 = selected[i], selected[i+1]
            c1 = crossover(p1, p2)
            c2 = crossover(p2, p1)
            c1 = mutate(c1)
            c2 = mutate(c2)
            c1 = repair_sum(c1)
            c2 = repair_sum(c2)
            new_pop.extend([c1, c2])
        population = new_pop
        best_idx = np.argmin(fitness)
        optimized = local_optimize(population[best_idx])
        population[best_idx] = optimized
        current_c5 = evaluate_c5(optimized)
        if current_c5 < best_c5:
            best_h = optimized
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
