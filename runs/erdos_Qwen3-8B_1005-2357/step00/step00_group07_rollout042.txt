import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = clip_and_scale(h)
    best_h = h.copy()
    best_c5 = compute_C5(h)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        best_alpha = 0.01
        best_c5_new = np.inf
        for alpha in [0.01, 0.1, 0.5, 1.0]:
            new_h = h - alpha * grad
            new_h = clip_and_scale(new_h)
            c5_new = compute_C5(new_h)
            if c5_new < best_c5_new:
                best_alpha = alpha
                best_c5_new = c5_new
        h = h - best_alpha * grad
        h = clip_and_scale(h)
        current_c5 = compute_C5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = clip_and_scale(h)
    return best_h, best_c5, n_points

def clip_and_scale(h):
    h_clip = np.clip(h, 0, 1)
    total = np.sum(h_clip)
    return h_clip * (n_points / 2) / total

def compute_C5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h):
    n_points = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n_points - 1)
    g = np.zeros(n_points)
    for i in range(n_points):
        i_plus_k = i + k
        term1 = 1 - h[i_plus_k] if 0 <= i_plus_k < n_points else 0.0
        i_minus_k = i - k
        term2 = h[i_minus_k] if 0 <= i_minus_k < n_points else 0.0
        g[i] = term1 - term2
    return g
