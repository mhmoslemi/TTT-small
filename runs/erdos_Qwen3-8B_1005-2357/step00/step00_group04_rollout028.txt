        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5 = np.max(corr) * dx
        return -c5

    result = differential_evolution(fitness, [(0.0, 1.0)] * n_points,
                                   strategy='best1bin', popsize=50,
                                   mutation=(0.8, 1.0), recombination=0.9,
                                   tol=1e-6, maxiter=1000, seed=seed, disp=False)

    best_h = result.x

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    active_k_index = np.argmax(corr)
    m = active_k_index
    k_offset = m - (n_points - 1)
    gradient = np.zeros(n_points)
    for i in range(n_points):
        j = i + k_offset
        if 0 <= j < n_points:
            gradient[i] = (1 - best_h[j]) * dx
        else:
            gradient[i] = 0.0

    learning_rate = 0.01
    for _ in range(100):
        best_h = best_h - learning_rate * gradient
        best_h = project_feasibility(best_h, dx)

    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr_final) * dx
    return (best_h, c5_bound, n_points)
