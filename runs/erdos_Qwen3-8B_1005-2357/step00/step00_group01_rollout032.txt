        overlap = cor * dx
        c5 = np.max(overlap)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, best_n)
