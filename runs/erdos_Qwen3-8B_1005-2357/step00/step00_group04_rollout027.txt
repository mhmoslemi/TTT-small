            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    pop_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_generations = 1000
    time_budget = budget_s
    
    population = [project_feasibility(np.random.rand(n_points)) for _ in range(pop_size)]
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > time_budget:
            break
        
        fitness = [-evaluate_c5(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        
        new_pop = []
        for i in range(pop_size):
            a, b, c = random.sample(range(pop_size), 3)
            mutant = sorted_pop[c] + mutation_factor * (sorted_pop[b] - sorted_pop[a])
            trial = sorted_pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            trial = project_feasibility(trial)
            new_pop.append(trial)
        
        population = new_pop
    
    best_h = population[np.argmin([-evaluate_c5(h) for h in population])]
    best_c5 = evaluate_c5(best_h)
    
    learning_rate = 0.01
    max_steps = 100
    for _ in range(max_steps):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        i_active = np.argmax(corr)
        k_active = i_active - (n_points - 1)
        gradient = np.zeros_like(best_h)
        for j in range(n_points):
            shifted_j = j + k_active
            if 0 <= shifted_j < n_points:
                gradient[j] = dx * (1 - best_h[shifted_j])
        best_h -= learning_rate * gradient
        best_h = project_feasibility(best_h)
        current_c5 = evaluate_c5(best_h)
        if current_c5 >= best_c5:
            break
        best_c5 = current_c5
    
    return best_h, best_c5, n_points
