from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.random.uniform(0, 1, n_points)
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - n_points / 2) > 1e-3:
        initial_h = initial_h * (n_points / 2) / initial_sum
    best_h = initial_h
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    best_n = n_points
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    for _ in range(3):
        result = minimize(objective, best_h, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 500})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            best_n = n_points
        else:
            break
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if np.random.rand() < 0.3:
            new_h = np.random.uniform(0, 1, n_points)
            new_sum = np.sum(new_h)
            new_h = new_h * (n_points / 2) / new_sum
            best_h = new_h
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0 and n_points < 100:

        bounds_100 = Bounds(0, 1)
