            return float('inf')
        scale = (n_points / 2) / total
        h_scaled = h_clipped * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
