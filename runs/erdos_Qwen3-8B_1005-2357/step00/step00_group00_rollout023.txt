import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    candidates = [initial_h_values.copy()]
    uniform = np.full(n_points, 0.5)
    candidates.append(uniform)
    for _ in range(3):
        h_perturbed = initial_h_values.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        for i in indices:
            h_perturbed[i] = 1 - h_perturbed[i]
        current_sum = np.sum(h_perturbed)
        scale = target_sum / current_sum
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        candidates.append(h_perturbed)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    for h_start in candidates:
        if abs(np.sum(h_start) - target_sum) > 1e-6:
            h_start = np.clip(h_start * (target_sum / np.sum(h_start)), 0, 1)
        def objective(x):
            return evaluate_erdos_solution(x)
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
