import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = (initial_h / sum_h) * (n_points / 2)
    initial_guess = initial_h.copy()

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def eq_constraint(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = [{'type': 'eq', 'fun': eq_constraint}]

    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        method='trust-constr',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
    )
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        return (initial_guess, np.max(np.correlate(initial_guess, 1 - initial_guess, mode='full')) * dx, n_points)

    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
