    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    learning_rate = 0.01
    max_iterations = 1000
    improvement_threshold = 1e-4
    gradient_norm_threshold = 1e-6
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        k_actual = max_index - (n_points - 1)
        c5 = cross_corr[max_index] * dx
        if c5 < best_c5 - improvement_threshold:
            best_c5 = c5
            best_h = h.copy()
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = i + k_actual
            if 0 <= j < len(h):
                gradient[i] = (1 - h[j]) * dx
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = h * scale
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < gradient_norm_threshold:

        sum_h_new = np.sum(h_interpolated) * dx_new
        scale = (n_points_new / 2) / sum_h_new if sum_h_new != 0 else 1.0
        h_interpolated = h_interpolated * scale
        h = h_interpolated.copy()
        n_points = n_points_new
        dx = dx_new
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            cross_corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(cross_corr)
            k_actual = max_index - (n_points - 1)
            c5 = cross_corr[max_index] * dx
            if c5 < best_c5 - improvement_threshold:
