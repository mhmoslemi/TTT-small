        # Try with n_points=100

        target_sum = n_points / 2
        dx = 2.0 / n_points
        h_initial_100 = np.random.uniform(0, 1, n_points)
        h_initial_100 = h_initial_100 / np.sum(h_initial_100) * target_sum

        def objective_100(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty

        bounds_100 = [(0.0, 1.0) for _ in range(n_points)]
        result_100 = minimize(objective_100, h_initial_100, method='COBYLA', bounds=bounds_100, options={'maxiter': 100})
