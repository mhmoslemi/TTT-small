            if not np.isclose(sum_h_opt_new, n_new / 2):
                diff_new = n_new / 2 - sum_h_opt_new
                sorted_idx_new = np.argsort(h_opt_new)
                for i in sorted_idx_new:
                    if diff_new <= 0:
                        break
                    max_add_new = min(1.0 - h_opt_new[i], diff_new)
                    h_opt_new[i] += max_add_new
                    diff_new -= max_add_new
                for i in reversed(sorted_idx_new):
                    if diff_new <= 0:
                        break
                    max_sub_new = min(h_opt_new[i], diff_new)
                    h_opt_new[i] -= max_sub_new
                    diff_new -= max_sub_new
            c5_new = objective_new(h_opt_new)
            if c5_new < best_c5:
                best_h = h_opt_new
                best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n if n == len(best_h) else 100)
