            if np.random.rand() < mutation_rate:
                h = offspring[i]
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        h[j] = np.clip(h[j] + np.random.rand(), 0, 1)
                    else:
                        h[j] = np.clip(h[j] - np.random.rand(), 0, 1)
                offspring[i] = repair(h)
        
        population = offspring
        
        sorted_idx = np.argsort(fitness)
        top_idx = sorted_idx[:5]
        for i in top_idx:
            h = population[i]
            def obj(x):
                return evaluate_c5(x)
            cons = LinearConstraint(np.ones(n_points), [n_points/2], [n_points/2])
            bounds = Bounds(0, 1)
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
            if res.success:
                population[i] = res.x
        
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
