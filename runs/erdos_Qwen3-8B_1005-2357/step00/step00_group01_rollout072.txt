import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_values = [47, 94, 188, 376]
    time_budget = budget_s
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    for n in n_values:
        if time_budget <= 0:
            break
        if n == 47:
            initial_guess = initial_h_values
        else:
            random_vec = np.random.rand(n)
            sum_rand = np.sum(random_vec)
            scaling = (n / 2) / sum_rand
            scaled_vec = np.clip(random_vec * scaling, 0, 1)
            initial_guess = scaled_vec.tolist()

        def objective(h):
            return evaluate_erdos_solution(h)

        bounds = [(0, 1) for _ in range(n)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]

        for restart in range(5):
            if time_budget <= 0:
                break
            start_time = time.time()
            result = opt.minimize(
                objective,
                initial_guess,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'gtol': 1e-6, 'maxiter': 100}
            )
            elapsed_time = time.time() - start_time
            time_budget -= elapsed_time

            if result.success:
                current_h = result.x

                    best_h = current_h
                    best_n = n

                continue

        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
