        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def repair_solution(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec)
        if abs(total - n_points / 2) > 1e-5:
            scale = (n_points / 2) / total
            h_vec = h_vec * scale
        return h_vec

    def compute_gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_opt = np.argmax(corr)
        lag = k_opt - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = i + lag
            if 0 <= shifted_i < n_points:
                grad[i] = (1 - h_vec[shifted_i]) * dx
        return grad

    pop_size = 50
    generations = 100
    mutation_factor = 0.9
    crossover_rate = 0.7
    population = [h.copy() for _ in range(pop_size)]
    best_h = h.copy()

    start_time = time.time()

    for gen in range(generations):

        fitness = [evaluate_c5(ind) for ind in population]

        tournament = np.random.choice(population, size=5, replace=False)
        tournament_fitness = [evaluate_c5(ind) for ind in tournament]
        best_tourn = tournament[np.argmin(tournament_fitness)]
        new_population = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            if np.random.rand() < crossover_rate:
                mutant = np.where(np.random.rand(n_points) < 0.5, mutant, b)
            mutant = repair_solution(mutant)
            new_population.append(mutant)
        population = new_population

    fitness = [evaluate_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]

    def objective(x):
        h_vec = x.copy()
        h_vec = repair_solution(h_vec)
        return evaluate_c5(h_vec)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = best_h.copy()
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, jac=compute_gradient, options={'maxiter': 500})
    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 500:
        n_points = 140
        dx = 2.0 / n_points
        h_interpolated = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 70), best_h)
        h_interpolated = repair_solution(h_interpolated)

        def grad2(x):

        result2 = minimize(objective2, x0_interpolated, method='SLSQP', bounds=bounds2, constraints=cons2, jac=grad2, options={'maxiter': 500})
        best_h = result2.x
        best_c5 = evaluate_c5(best_h)

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 500:
            n_points = 280
            dx = 2.0 / n_points
            h_interpolated2 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 140), best_h)
            h_interpolated2 = repair_solution(h_interpolated2)
