import numpy as np
from scipy.optimize import minimize
import time

def objective(h, n, dx):
    k = 1 - h
    corr = np.correlate(h, k, mode='full')
    return np.max(corr) * dx

def grad_objective(h, n, dx):
    grad = np.zeros(n)
    delta = 1e-5
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = np.clip(h[i] + delta, 0, 1)
        obj_plus = objective(h_plus, n, dx)
        h_minus = h.copy()
        h_minus[i] = np.clip(h[i] - delta, 0, 1)
        obj_minus = objective(h_minus, n, dx)
        grad[i] = (obj_plus - obj_minus) / (2 * delta)
    return grad

def constraint(h, n):
    return n/2 - np.sum(h)

def perturb(h, n, seed=42):
    rng = np.random.RandomState(seed)
    h_pert = h + rng.uniform(-0.1, 0.1, n)
    h_pert = np.clip(h_pert, 0, 1)
    total = np.sum(h_pert)
    scale = (n/2) / total
    h_pert = h_pert * scale
    h_pert = np.clip(h_pert, 0, 1)
    return h_pert

def transfer_solution(h, n1, n2):

    h2_init = np.interp(x2, x1, h)

    h2_init = h2_init * scale
