import numpy as np
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values

                h_perturb = np.clip(h_perturb, 0, 1)

            return compute_c5(x)

        c5 = compute_c5(optimized)
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimized
