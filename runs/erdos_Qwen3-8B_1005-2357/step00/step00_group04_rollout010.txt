import numpy as np
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap C5."""
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_feasibility(h, dx):
    """Clip h to [0,1] and adjust to satisfy integral constraint."""
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped) * dx
    if abs(total - 1) > 1e-6:
        scale = 1.0 / total
        h_clipped *= scale
    return h_clipped
