        h = np.full(n_points, 0.5)
    else:
        initial_h_values = globals()['initial_h_values']
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    target_sum = n_points / 2
    candidates = [h.copy()]
    uniform_h = np.full(n_points, 0.5)
    candidates.append(uniform_h)
    for _ in range(3):
        indices = np.random.choice(n_points, 5, replace=False)
        h_perturb = h.copy()
        for i in indices:
            h_perturb[i] = 1 - h_perturb[i]
        current_sum = np.sum(h_perturb)
        scale = target_sum / current_sum
        h_perturb = np.clip(h_perturb * scale, 0, 1)
        candidates.append(h_perturb)
    best_c5 = float('inf')
    best_h = None
    dx = 2.0 / n_points
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [
        {'type': 'ineq', 'fun': lambda x: np.sum(x) - target_sum},
        {'type': 'ineq', 'fun': lambda x: target_sum - np.sum(x)}
    ]
    start_time = time.time()
    for candidate in candidates:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        result = minimize(
            lambda x: evaluate_erdos_solution(x),
            candidate,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500}
        )
        if result.success:
            current_h = result.x
            current_sum = np.sum(current_h)
            scale = target_sum / current_sum
            current_h = np.clip(current_h * scale, 0, 1)
            c5 = evaluate_erdos_solution(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h
        else:
            c5 = evaluate_erdos_solution(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
    return (best_h, best_c5, n_points)
