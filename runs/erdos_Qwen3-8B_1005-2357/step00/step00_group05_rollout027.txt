            bounds=bounds_new,
            constraints=[linear_constraint_new],
            options={'maxiter': 500, 'disp': False}
        )
        elapsed_cobyla_new = time.time() - start_time
        if elapsed_cobyla_new >= budget_s:
            # Time up, compare with best_result
            if result_cobyla_new.fun < best_result.fun:
                best_result = result_cobyla_new
        else:
            # Perform local refinement with SLSQP
            result_slsqp_new = minimize(
                fun=objective_new,
                x0=result_cobyla_new.x,
                method='SLSQP',
                bounds=bounds_new,
                constraints=[linear_constraint_new],
                options={'maxiter': 500, 'disp': False}
            )
            elapsed_slsqp_new = time.time() - start_time
            if result_slsqp_new.fun < best_result.fun:
                best_result = result_slsqp_new
    return (best_result.x, best_result.fun, n_points)
