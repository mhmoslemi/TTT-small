def run(seed=42, budget_s=1100, **kwargs):

        g = np.zeros(n_points)

            i_forward = i + k
            i_backward = i - k
            term1 = 0.0 if i_forward < 0 or i_forward >= n_points else 1 - h[i_forward]
            term2 = 0.0 if i_backward < 0 or i_backward >= n_points else h[i_backward]
            g[i] = term1 - term2

        best_c_candidate = best_c
