def slsqp_optimize(h, n, dx, penalty_box, penalty_int, max_iter=500):
    # Objective function
    def objective(x):
        h_current = x.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        c5 = np.max(corr) * dx
        # Compute penalties
        penalty_box_val = penalty_box * (np.sum(np.maximum(0, h_current - 1) + np.maximum(0, -h_current)))
