        return np.full_like(h, 0.5)
    scale = (n_points / 2) / sum_clipped
    return h_clipped * scale

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    initial_h = initial_h_values
    h = project_feasibility(initial_h, dx, n_points)
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h, best_c5, n_points)
