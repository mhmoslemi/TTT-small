    result = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed, maxiter=100)
    elapsed = time.time() - start_time
    
    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2 - sum_best) / n_points
    best_h += delta
    best_h = np.clip(best_h, 0.0, 1.0)
    
    if elapsed < budget_s:
        def objective_slsqp(h):
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        result_slsqp = minimize(objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp, constraints={'type': 'eq', 'fun': constraint_sum})
        best_h = result_slsqp.x
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    delta = (n_points / 2 - sum_best) / n_points
    best_h += delta
    best_h = np.clip(best_h, 0.0, 1.0)
    
    dx = 2.0 / n_points
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
