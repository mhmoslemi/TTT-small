import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def repair(h):
        total = np.sum(h)
        if abs(total - target_sum) < 1e-6:
            return h
        scale = target_sum / total
        h_scaled = h * scale
        return np.clip(h_scaled, 0, 1)

    candidates = [initial_h_values, np.full(n_points, 0.5)]
    for _ in range(3):
        perturbed = initial_h_values.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        for i in indices:
            perturbed[i] = 1 - perturbed[i]
        perturbed = repair(perturbed)
        candidates.append(perturbed)

    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    for candidate in candidates:
        h0 = np.array(candidate)
        if abs(np.sum(h0) - target_sum) > 1e-6:
            h0 = repair(h0)
        def obj(h):
            return evaluate_erdos_solution(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, h0, method='COBYLA', bounds=bounds, tol=1e-6, options={'maxiter': 1000})
        if res.success:
            h_opt = repair(res.x)
            current_c5 = evaluate_erdos_solution(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
