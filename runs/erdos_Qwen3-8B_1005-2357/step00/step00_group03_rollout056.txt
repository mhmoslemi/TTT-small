import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 70
    if initial_h is not None:
        h = np.array(initial_h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - n_points / 2) > 1e-6:
            h *= (n_points / 2) / s
        return h
    
    def de_step(pop, gen, budget):
        nonlocal h
        start = time.time()
        for _ in range(gen):
            if time.time() - start > budget:
                return pop
            for i in range(len(pop)):
                a, b, c = np.random.choice(len(pop), 3, replace=False)
                mutant = pop[a] + 0.9 * (pop[b] - pop[c])
                trial = pop[i].copy()
                for j in range(n_points):
                    if np.random.rand() < 0.7:
                        trial[j] = mutant[j]
                trial = repair(trial)
                if evaluate_c5(trial) < evaluate_c5(pop[i]):
                    pop[i] = trial
        return pop
    
    def grad_func(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k = np.argmax(corr) - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] = (1 - x[i + k]) * (2.0 / n_points)
        return grad
    
    def local_refine(x):
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(evaluate_c5, x, method='SLSQP', jac=grad_func, constraints=cons)
        return res.x
    
    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    for _ in range(100):
        if time.time() > budget_s - 10:
            break
        pop = de_step(pop, 1, budget_s - 10)
    
    best = min(pop, key=evaluate_c5)
    best = local_refine(best)
    best = repair(best)
    final_c5 = evaluate_c5(best)
    
    return best, final_c5, n_points
