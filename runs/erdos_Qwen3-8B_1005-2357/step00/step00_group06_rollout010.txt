        dx = 2.0 / n

        h = np.clip(h, 0, 1)
        def obj(x):
            g = 1 - x
            corr = np.correlate(x, g, 'full')
            return dx * np.max(corr)
        def grad(x):
            g = 1 - x
            corr = np.correlate(x, g, 'full')
            m = np.max(corr)
            idx = np.argmax(corr)
            res = np.zeros_like(x)
            for i in range(n):
                for j in range(n):
                    if idx - j >= 0 and idx - j < n:
                        res[i] += g[idx - j]
            return dx * res
        bounds = [(0,1)]*n
