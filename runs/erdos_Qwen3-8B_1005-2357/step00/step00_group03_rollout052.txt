import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        L = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i - L < n_points:
                grad[i] = (1 - h[i - L]) * dx
        return grad

        return compute_c5(h)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP', jac=compute_gradient, bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    c5_bound = compute_c5(best_h)
