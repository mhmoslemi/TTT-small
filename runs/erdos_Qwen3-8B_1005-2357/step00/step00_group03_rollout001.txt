import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):

    n = len(initial_h_values)
    dx = 2.0 / n

    if abs(sum_initial) > 1e-5:
        adjustment = -sum_initial / n
        initial_h_values += adjustment

        sum_initial = np.sum(initial_h_values) - n / 2

    def compute_objective(h_flat, n, dx, penalty):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1 - h, mode='full')
        c5_val = np.max(corr) * dx
        sum_val = np.sum(h) - n / 2
        return c5_val + penalty * sum_val**2

    def compute_gradient(h_flat, n, dx, penalty):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1 - h, mode='full')
        active_lag_index = np.argmax(corr)
        k = active_lag_index - (n - 1)
        grad_active = np.zeros(n)
        for i in range(n):
            if 0 <= i - k < n:
                grad_active[i] = (1 - h[i - k]) * dx
        sum_val = np.sum(h) - n / 2
        grad_penalty = 2 * penalty * sum_val * np.ones(n)
        return grad_active + grad_penalty

    def objective_minimize(h_flat, n, dx, penalty):
        return compute_objective(h_flat, n, dx, penalty)

    def gradient_minimize(h_flat, n, dx, penalty):
        return compute_gradient(h_flat, n, dx, penalty).flatten()

    h0 = initial_h_values.copy()
    bounds = Bounds([0.0]*n, [1.0]*n)
    constraint = LinearConstraint(np.ones(n), [n/2], [n/2])
    penalty = 1000.0
    start_time = time.time()

    res = minimize(objective_minimize, h0, method='SLSQP',
                   args=(n, dx, penalty), bounds=bounds,

    best_solution = res.x
