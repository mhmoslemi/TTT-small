        n_points = len(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with L-BFGS-B
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})

    best_c5 = objective(best_h)

    if best_c5 < best_c5:

        best_c5 = best_c5
