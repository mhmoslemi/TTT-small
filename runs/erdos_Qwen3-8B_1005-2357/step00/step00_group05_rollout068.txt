        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x_old, h_scaled)
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        scaling_new = target_sum_new / sum_h_new
        h_new_scaled = np.clip(h_new * scaling_new, 0, 1)

        def obj_new(h):
            h_clipped = np.clip(h, 0, 1)
            sum_h = np.sum(h_clipped)
            corr = correlate(h_clipped, 1 - h_clipped, mode='full')
            max_corr = np.max(corr)
            box_penalty = 1e6 * np.sum((h < 0) | (h > 1))
            integral_penalty = 1e3 * np.abs(sum_h - target_sum_new)
            return max_corr * dx_new + box_penalty + integral_penalty

        h_initial_new = h_new_scaled.copy()
        result_new = basinhopping(obj_new, h_initial_new, niter=50, T=1.0, stepsize=0.5, disp=False)
        best_h_new = result_new.x

        def local_obj_new(h):
            h_clipped = np.clip(h, 0, 1)
            corr = correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx_new

        bounds_new = [(0, 1) for _ in range(n_points_new)]
        local_res_new = minimize(local_obj_new, best_h_new, method='L-BFGS-B', bounds=bounds_new, tol=1e-6, options={'maxiter': 500})
        best_h_local_new = local_res_new.x

        h_clipped_new = np.clip(best_h_local_new, 0, 1)
        sum_h_new = np.sum(h_clipped_new)
        scaling_new = target_sum_new / sum_h_new
        h_final_new = np.clip(h_clipped_new * scaling_new, 0, 1)
        corr_new = correlate(h_final_new, 1 - h_final_new, mode='full')
        c5_bound_new = np.max(corr_new) * dx_new

        if c5_bound_new < c5_bound:
            c5_bound = c5_bound_new
            h_scaled = h_final_new
            n_points = n_points_new

    return h_scaled, c5_bound, n_points
