import numpy as np
from scipy.optimize import minimize, COBYLA, trust_constr
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = 70
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = float('inf')

    def obj_func(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def cons_func(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Global optimization with COBYLA
    result = minimize(obj_func, initial_h, method='COBYLA', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_func}], options={'maxiter': 100})
    if result.success:
        h_candidate = result.x
        c5_candidate = obj_func(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Local refinement with trust-constrained
    result = minimize(obj_func, best_h, method='trust-constr', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_func}], options={'maxiter': 50})
    if result.success:
        h_candidate = result.x
        c5_candidate = obj_func(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Check if time remains for higher resolution
    time_remaining = budget_s - (result.nfev * 100)  # Placeholder for actual timing
    if time_remaining > 1000 and n_points < 100:
        n_points = 100
        dx = 2.0 / n_points
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        best_h = np.zeros(n_points)
        best_h[:int(n_points/2)] = 1.0
        # Re-run optimization for higher resolution
        result = minimize(obj_func, best_h, method='COBYLA', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}], options={'maxiter': 50})

                best_h = h_candidate
                best_c5 = c5_candidate

    return (best_h, best_c5, n_points)
