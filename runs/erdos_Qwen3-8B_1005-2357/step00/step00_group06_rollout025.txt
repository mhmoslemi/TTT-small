    def project_integral(h, n_points):
        target = n_points / 2
        current_sum = np.sum(h)
        diff = current_sum - target
        if abs(diff) < 1e-8:
            return h.copy()
        h_proj = h.copy()
        for i in range(n_points):
            if diff > 0:
                if h_proj[i] > 0:
                    delta = min(h_proj[i], diff)
                    h_proj[i] -= delta
                    diff -= delta
            else:
                if h_proj[i] < 1:
                    delta = min(1 - h_proj[i], -diff)
                    h_proj[i] += delta
                    diff += delta
            if abs(diff) < 1e-8:
                break
        return h_proj

    def objective_func(h_vec):

        max_val = np.max(conv)

        conv = np.correlate(h_vec, 1 - h_vec, mode='full')

        grad = np.zeros_like(h_vec)
