        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        if np.sum(h) != 0:
            h = h * (target_sum / np.sum(h))
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective_basinhopping(h):
        c5 = compute_c5(h)
        box_penalty = 1e5 * np.sum((h < 0) | (h > 1))
        sum_penalty = 1e3 * np.abs(np.sum(h) - target_sum)
        return c5 + box_penalty + sum_penalty

    res_bh = basinhopping(objective_basinhopping, h, niter=100, T=1.0, stepsize=0.5, niter_success=10)
    h_bh = res_bh.x

    h_clipped = np.clip(h_bh, 0, 1)
    if np.sum(h_clipped) != 0:
        h_clipped = h_clipped * (target_sum / np.sum(h_clipped))
    current_c5 = compute_c5(h_clipped)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_clipped
        best_n = n_points

    bounds = [(0, 1)] * n_points
    res_local = minimize(compute_c5, h_clipped, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 1000})
    h_local = res_local.x

    h_local_clipped = np.clip(h_local, 0, 1)
    if np.sum(h_local_clipped) != 0:
        h_local_clipped = h_local_clipped * (target_sum / np.sum(h_local_clipped))
    current_c5 = compute_c5(h_local_clipped)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_local_clipped
        best_n = n_points

    if time.time() - start_time < budget_s - 500:
        n_points_new = 100
        dx_new = 2.0 / n_points_new
        target_sum_new = n_points_new / 2

        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        interp_func = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_new = interp_func(x_new)
        h_new = np.clip(h_new, 0, 1)
        if np.sum(h_new) != 0:
            h_new = h_new * (target_sum_new / np.sum(h_new))

        def objective_basinhopping_new(h):
            c5 = compute_c5(h)
            box_penalty = 1e5 * np.sum((h < 0) | (h > 1))
            sum_penalty = 1e3 * np.abs(np.sum(h) - target_sum_new)
            return c5 + box_penalty + sum_penalty

        res_bh_new = basinhopping(objective_basinhopping_new, h_new, niter=100, T=1.0, stepsize=0.5, niter_success=10)
        h_bh_new = res_bh_new.x

        h_clipped_new = np.clip(h_bh_new, 0, 1)
        if np.sum(h_clipped_new) != 0:
            h_clipped_new = h_clipped_new * (target_sum_new / np.sum(h_clipped_new))
        current_c5_new = compute_c5(h_clipped_new)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = h_clipped_new
            best_n = n_points_new

        bounds_new = [(0, 1)] * n_points_new
        res_local_new = minimize(compute_c5, h_clipped_new, method='L-BFGS-B', bounds=bounds_new, tol=1e-6, options={'maxiter': 1000})
        h_local_new = res_local_new.x

        h_local_clipped_new = np.clip(h_local_new, 0, 1)
        if np.sum(h_local_clipped_new) != 0:
            h_local_clipped_new = h_local_clipped_new * (target_sum_new / np.sum(h_local_clipped_new))
        current_c5_new = compute_c5(h_local_clipped_new)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = h_local_clipped_new
            best_n = n_points_new

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, best_n)
    else:
        return (best_h, best_c5, best_n)
