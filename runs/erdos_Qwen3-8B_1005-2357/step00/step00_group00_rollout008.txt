import numpy as np
from scipy.optimize import COBYLA
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()

    def objective(x):
        return evaluate_erdos_solution(x)

    def repair(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if abs(s - target_sum) > 1e-6:
            scale = target_sum / s
            x = x * scale
        return x

    candidates = [initial_h_values]
    for _ in range(3):
        h = initial_h_values.copy()
        idx = np.random.choice(n_points, 5, replace=False)
        h[idx] = 1 - h[idx]
        h = repair(h)
        candidates.append(h)
    candidates.append(np.full(n_points, 0.5))

    for candidate in candidates:
        h = np.array(candidate)
        h = repair(h)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = COBYLA(x0=h, bounds=bounds, constraints=cons, maxiter=1000, tol=1e-6)

            if c5 < best_c5:

                best_h = h_opt

    best_h = repair(best_h)
    return (best_h, best_c5, n_points)
