import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    initial_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (initial_sum / np.sum(h))

    def project_feasibility(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        sum_clipped = np.sum(h_clipped) * dx
        if sum_clipped == 0:
            return h_clipped
        scaling = initial_sum / sum_clipped
        return h_clipped * scaling

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def objective(h_vec):
        return compute_c5(project_feasibility(h_vec))

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - initial_sum}]

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(project_feasibility(best_h))

    return (best_h, best_c5, n_points)
