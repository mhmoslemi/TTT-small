def run(seed=42, budget_s=1100, **kwargs):

            i_forward = i + k0
            term1 = 1.0 - h[i_forward] if 0 <= i_forward < n_points else 0.0
            i_backward = i - k0
            term2 = h[i_backward] if 0 <= i_backward < n_points else 0.0
            g[i] = term1 - term2

        for alpha in [0.01, 0.1, 0.5, 1.0]:
            h_candidate = h + alpha * g
            h_clip = np.clip(h_candidate, 0.0, 1.0)

                continue

            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            c_candidate = np.max(corr) * dx
            if c_candidate < best_c_candidate:
                best_alpha = alpha
                best_c_candidate = c_candidate

        h = h + best_alpha * g
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total != 0:
            h = h * (n_points / 2.0) / total

        corr = np.correlate(h, 1 - h, mode='full')
        c_current = np.max(corr) * dx
        if c_current < best_c:
