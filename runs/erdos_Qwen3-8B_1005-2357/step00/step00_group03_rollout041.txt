            lag = max_index - (n - 1)
            grad = np.zeros(n)
            dx = 2.0 / n
            for i in range(n):
                x_i = i * dx
                x_i_plus_lag = x_i + lag
                if 0 <= x_i_plus_lag <= 2:
                    j = int(np.round(x_i_plus_lag / dx))
                    if 0 <= j < n:
                        grad[i] = (1 - h_clipped[j]) * dx
            return grad
        
        result_slsqp = minimize(slsqp_objective, best_candidate, method='SLSQP', jac=slsqp_gradient, constraints=cons, tol=1e-6)
        best_candidate_slsqp = result_slsqp.x
        c5_candidate_slsqp = compute_c5(best_candidate_slsqp)
        
        if c5_candidate_slsqp < best_c5:
            best_h = best_candidate_slsqp
            best_c5 = c5_candidate_slsqp
            best_n = n
    
    final_h = best_h.copy()
    final_h = final_h.clip(0, 1)
    sum_h = np.sum(final_h)
    if sum_h != best_n / 2:
        scale = (best_n / 2) / sum_h
        final_h = final_h * scale
    final_c5 = compute_c5(final_h)
    return (final_h, final_c5, best_n)
