    n_points = 95
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Load initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.random.rand(n_points)

        h = h / np.sum(h) * target_sum

        h = np.clip(h, 0, 1)

    mu_values = [0.0, 0.1, 0.2]

        max_val = np.max(conv)
        c5 = max_val * dx
        penalty = mu * (np.sum(h) - target_sum) ** 2
        return c5 + penalty

    def gradient(h_vec, mu):

        conv = np.correlate(h, 1 - h, mode='full')

        lags = [m - (n_points - 1) for m in max_indices]
        grad = np.zeros(n_points)

                    term = (1 - h[i_plus_k]) - h[i_minus_k]

        grad += 2 * mu * (np.sum(h) - target_sum)

        h = h_vec.copy()

            # Need to decrease the sum
            h[0] -= diff

            h[0] += -diff
        h = np.clip(h, 0, 1)

    start_time = time.time()
