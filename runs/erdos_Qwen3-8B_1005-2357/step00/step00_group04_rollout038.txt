            k = active_k - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if 0 <= i - k < n_points:
                    grad[i] -= h[i - k] * dx
            h += step_size * grad
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            scale = (n_points / 2) / total if total != 0 else 1
            h *= scale
        return h
    
    best_h = gradient_descent(best_h, dx, max_iter=100, step_size=0.01)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h.tolist(), c5_bound, n_points)
