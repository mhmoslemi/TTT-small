    for gen in range(100):
        parents = []
        for _ in range(50):
            tournament = np.random.choice(population, 5, replace=False)
            best = min(tournament, key=lambda x: evaluate_fitness(x))
            parents.append(best)
        
        new_population = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < 50 else parents[i]
            c1, c2 = (p1 + p2) / 2, (p1 + p2) / 2
            if np.random.rand() < 0.1:
                c1 += np.random.normal(0, 0.05, len(c1))
                c1 = np.clip(c1, 0, 1)
                c1 = c1 / np.sum(c1) * (len(c1) / 2)
            if np.random.rand() < 0.1:
                c2 += np.random.normal(0, 0.05, len(c2))
                c2 = np.clip(c2, 0, 1)
                c2 = c2 / np.sum(c2) * (len(c2) / 2)
            new_population.extend([c1, c2])
        
        population = new_population
        fitnesses = [evaluate_fitness(ind) for ind in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / len(h)
    
    def grad(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr) - len(h) + 1
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i - lag < len(h):
                grad[i] = (1 - h[i - lag]) * 2.0 / len(h)
        return grad
    
    bounds = [(0, 1) for _ in range(len(best_h))]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}]
    result = minimize(objective, best_h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, tol=1e-6)
    best_h_slsqp = result.x
    
    start_time = time.time()
    if time.time() - start_time < budget_s - 300:
        n_new = 140
        x = np.linspace(0, 2, len(best_h_slsqp))
        x_new = np.linspace(0, 2, n_new)
        interpolated = np.interp(x_new, x, best_h_slsqp)
        sum_int = np.sum(interpolated)
        scale = (n_new / 2) / sum_int
        best_h_slsqp = interpolated * scale
    
    dx = 2.0 / len(best_h_slsqp)
    corr = np.correlate(best_h_slsqp, 1 - best_h_slsqp, mode='full')
    c5_bound = np.max(corr) * dx
    return (best_h_slsqp, c5_bound, len(best_h_slsqp))
