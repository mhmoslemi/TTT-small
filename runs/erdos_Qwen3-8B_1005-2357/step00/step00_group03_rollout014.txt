import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def repair_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        target = n_points / 2
        delta = target - sum_h
        if abs(delta) > 1e-6:
            h[-1] += delta
            h = np.clip(h, 0, 1)
        return h

    initial_h = repair_h(initial_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - n_points / 2)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                      popsize=50, mutation=(0.5, 1), recombination=0.9, 
                                      tol=1e-6, maxiter=100, seed=seed)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, objective(initial_h), n_points)

    best_h_de = result_de.x
    best_h_de = repair_h(best_h_de)

    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', 
                           bounds=bounds_slsqp, constraints=cons, 
                           tol=1e-6, options={'maxiter': 100})

    best_h_slsqp = result_slsqp.x
    best_h_slsqp = repair_h(best_h_slsqp)

    corr = np.correlate(best_h_slsqp, 1 - best_h_slsqp, mode='full')
    c5_bound = np.max(corr) * dx
