import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            scale = target_sum / current_sum
            h *= scale
        return h
    
    np.random.seed(seed)
    candidates = [initial_h_values]
    candidates.append(np.full(n_points, 0.5))
    
    for _ in range(3):
        perturbed = initial_h_values.copy()

        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum != 0:
