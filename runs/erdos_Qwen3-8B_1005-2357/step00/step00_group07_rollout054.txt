def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = h * (n_points / 2) / total
    best_c = np.correlate(h, 1 - h, mode='full').max() * dx
    best_h = h.copy()
    T_start = 10.0
    cooling_rate = 0.995
    max_iter = 10000
    start_time = time.time()
    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            jf = i + s
            jb = i - s
            t1 = 1.0 if 0 <= jf < n_points else 0.0
            t2 = h[jb] if 0 <= jb < n_points else 0.0
            grad[i] = t1 - t2
        h1 = h + 0.01 * grad
        h1 = np.clip(h1, 0, 1)
        total1 = np.sum(h1)
        h1 = h1 * (n_points / 2) / total1
