                c2 = crossover(parents[i], parents[i+1])
                offspring.append(c1)
                offspring.append(c2)
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i], 0.1, n_points)
        population = offspring
    
    def objective(h):
        return evaluate_fitness(h, n_points)
    
    def grad_objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        L = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i - L < n_points:
                grad[i] = (1 - h[i - L]) * dx
        return grad
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(objective, best_h, method='SLSQP', jac=grad_objective, bounds=bounds, constraints=cons, options={'maxiter': 500})
    best_h = res.x
    best_c5 = evaluate_fitness(best_h, n_points)
    
    if time.time() - start_time < budget_s - 300:
        n_points = 140
        dx = 2.0 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 70), best_h)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        best_h = h
        best_c5 = evaluate_fitness(best_h, n_points)
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-5:
        best_h = best_h * (n_points / 2) / sum_h
        best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_fitness(best_h, n_points)
    return best_h, best_c5, n_points
