import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 42
        h = np.random.rand(n_points)
        dx = 2.0 / n_points
        h = h * (1.0 / (np.sum(h) * dx))
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h = h * (1.0 / (np.sum(h) * dx))
    
    def evaluate_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_feasibility(h, dx):
        h = np.clip(h, 0, 1)
        scale = 1.0 / (np.sum(h) * dx)
        return h * scale
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:

        perturbation = np.random.normal(0, 0.1, len(h))
        h_candidate = h + perturbation
        h_candidate = project_feasibility(h_candidate, dx)
        c5_candidate = evaluate_overlap(h_candidate, dx)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()

    return best_h, best_c5, len(best_h)
