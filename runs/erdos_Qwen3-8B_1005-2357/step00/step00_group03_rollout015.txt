        n_points = 70
        h = np.full(n_points, 0.5)

        return max_corr * dx

    def objective(h):

        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            best_h = result.x
            best_c5 = compute_c5(best_h)
    except:
        pass

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    def objective_slsqp(h):
        return compute_c5(h)

    initial_guess = best_h
    try:
        start_time_slsqp = time.time()

            objective_slsqp,
