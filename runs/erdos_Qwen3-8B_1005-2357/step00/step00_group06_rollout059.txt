    target_sum = n_points / 2
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()

    def objective(h, mu):
        conv = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(conv)
        return dx * max_val + mu * (np.sum(h) - target_sum)**2

    def gradient(h, mu):
        conv = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(conv)
        max_lags = np.where(conv == max_val)[0]
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in max_lags:
                if 0 <= i + k < n_points:
                    term1 = (1 - h[i + k])
                else:
                    term1 = 0
                if 0 <= i - k < n_points:
                    term2 = h[i - k]
                else:
                    term2 = 0
                grad[i] += dx * (term1 - term2)
        grad += 2 * mu * (np.sum(h) - target_sum)
        return grad

        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-6:
            return h

        h[-1] += diff
        h = np.clip(h, 0, 1)
        return h

    for mu in mu_values:

        h_start = best_h.copy()

        result = minimize(

            x0=h_start,
            method='L-BFGS-B',

            options={'maxiter': 1000, 'ftol': 1e-6}
