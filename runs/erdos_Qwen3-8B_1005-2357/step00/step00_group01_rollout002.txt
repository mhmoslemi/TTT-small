import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        n_points = 47
        h = np.random.dirichlet([1]*n_points) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(x):
        h_flat = x.reshape(n_points)
        h_ones = 1 - h_flat
        corr = np.correlate(h_flat, h_ones, mode='full')
        return np.max(corr * dx)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8})
    return (result.x, result.fun, n_points)
