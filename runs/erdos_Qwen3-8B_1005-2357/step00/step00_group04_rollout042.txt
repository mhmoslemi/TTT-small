        c5 = np.max(corr * dx)
        constraint_violation = np.abs(np.sum(h) * dx - 1.0)
        return c5 + 1e3 * constraint_violation

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()

    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
