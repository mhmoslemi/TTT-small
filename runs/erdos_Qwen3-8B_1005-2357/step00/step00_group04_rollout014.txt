import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_overlap(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = h * scaling
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_feasibility(h_vec):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h if sum_h != 0 else 1
        return h * scaling
    
    def objective(h_vec):
        return evaluate_overlap(h_vec)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)})
    
    start_time = time.time()
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    elapsed = time.time() - start_time
    
    best_h = res.x
    best_c5 = evaluate_overlap(best_h)
    
    if elapsed < budget_s:
        n_points = 84
        dx = 2.0 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 42), best_h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = evaluate_overlap(h)
    
    return best_h, best_c5, n_points
