import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 42
        h = np.random.rand(n_points)
        total = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / total)
        h = np.clip(h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped) * dx
        if total == 0:
            return np.inf
        h_scaled = h_clipped * (1.0 / total)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def project_feasibility(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped) * dx
        if total == 0:
            return h_clipped
        scaling = 1.0 / total
        return h_clipped * scaling

    def acquisition(x):
        return -evaluate_c5(x)

    def optimize():
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(acquisition, h, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 100})
        return res.x

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    for _ in range(200):
        if time.time() - start_time > budget_s:

        h = optimize()
        h = project_feasibility(h)
        c5 = evaluate_c5(h)
