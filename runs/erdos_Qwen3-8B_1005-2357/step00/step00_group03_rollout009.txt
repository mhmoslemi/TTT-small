import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    def repair(h, n):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
