        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < allocated_time:
            allocated_time = remaining_time
        initial_guesses = []
        if n == 47:
            initial_guesses.append(initial_h)
        else:
            initial_guesses.append(np.full(n, 0.5))
            for _ in range(2):
                h = np.random.rand(n)
                sum_h = np.sum(h)
                scale = (n / 2) / sum_h if sum_h != 0 else 1
                h = h * scale
                h = np.clip(h, 0, 1)
                initial_guesses.append(h)
        for h_guess in initial_guesses:
            def objective(h_flat):
                h = h_flat.reshape(n, 1)
                corr = np.correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                dx = 2.0 / n
                return max_corr * dx
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n/2})
            bounds = [(0, 1) for _ in range(n)]
            result = minimize(objective, h_guess, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 100})
            if result.success:
                current_h = result.x
                current_c5 = evaluate_erdos_solution(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != len(best_h)/2:
        scale = (len(best_h)/2) / sum_best if sum_best != 0 else 1
        best_h = best_h * scale
        best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, len(best_h))
