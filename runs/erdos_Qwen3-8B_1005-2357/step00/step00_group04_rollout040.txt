    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 42
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    def repair(h, dx):
        h_clipped = np.clip(h, 0, 1)
        total_clipped = np.sum(h_clipped) * dx
        if abs(total_clipped) < 1e-6:
            h_clipped = np.ones_like(h) * 0.5
        else:
            h_clipped = h_clipped * (1.0 / total_clipped)
        return h_clipped
    
    def evaluate_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    if initial_h_values is not None:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * sum_required / np.sum(h_initial)
    h_initial = repair(h_initial, dx)
    
    population_size = 100
    population = [h_initial.copy() for _ in range(population_size)]
    best_h = h_initial.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    time_start = time.time()
    
    for gen in range(1000):
        if time.time() - time_start > budget_s:
            break
        
        parents = []
        for _ in range(population_size):
            tournament = random.sample(population, 3)
            winner = min(tournament, key=lambda x: evaluate_overlap(x, dx))
            parents.append(winner)
        
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                p1, p2 = parents[i], parents[i+1]
                if random.random() < 0.9:
                    c1 = (p1 + p2) / 2
                    c2 = (p1 + p2) / 2
                else:
                    c1, c2 = p1.copy(), p2.copy()
                for c in [c1, c2]:
                    for j in range(len(c)):
                        if random.random() < 0.1:
                            c[j] += np.random.uniform(-0.1, 0.1)
                            c[j] = np.clip(c[j], 0, 1)
                c1 = repair(c1, dx)
                c2 = repair(c2, dx)
                new_population.extend([c1, c2])
            else:
                new_population.append(parents[i].copy())
        
        population = new_population
        for h in population:
            c5 = evaluate_overlap(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    time_remaining = budget_s - (time.time() - time_start)
    if time_remaining > 500:
        n_points = 84
        dx_new = 2.0 / n_points
        h_84 = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx_new
            idx1 = int(x / dx)
            if idx1 >= n_points:
                idx1 = n_points - 1
            if i % 2 == 0:
                h_84[i] = best_h[idx1]
            else:
                idx2 = idx1 + 1
                if idx2 >= n_points:
                    idx2 = n_points - 1
                frac = (x - idx1 * dx) / dx
                h_84[i] = best_h[idx1] * (1 - frac) + best_h[idx2] * frac
        h_84 = repair(h_84, dx_new)
        current_c5 = evaluate_overlap(h_84, dx_new)
        best_h_84 = h_84.copy()
        best_c5_84 = current_c5
        
        for _ in range(1000):
            if time.time() - time_start > budget_s:
                break
            corr = np.correlate(best_h_84, 1 - best_h_84, mode='full')
            k0 = np.argmax(corr)
            grad = np.zeros_like(best_h_84)
            for i in range(len(best_h_84)):
                lag = k0 - (len(best_h_84) - 1)
                if 0 <= i - lag < len(best_h_84):
                    grad[i] = (1 - best_h_84[i - lag]) * dx_new
            best_h_84 -= grad * 0.01
            best_h_84 = repair(best_h_84, dx_new)
            current_c5 = evaluate_overlap(best_h_84, dx_new)
            if current_c5 < best_c5_84:
                best_c5_84 = current_c5
                best_h_84 = best_h_84.copy()
        
        if best_c5_84 < best_c5:
            best_c5 = best_c5_84
            best_h = best_h_84.copy()
    
    return (best_h, best_c5, n_points)
