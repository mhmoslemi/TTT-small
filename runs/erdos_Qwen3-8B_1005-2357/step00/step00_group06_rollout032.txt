    n_points = 95
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def objective(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(conv) * dx
        penalty = 0.2 * (np.sum(h_vec) - target_sum) ** 2
        return c5 + penalty

    def gradient(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        max_idx = np.argmax(conv)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j_plus = i + lag
            val_plus = h_vec[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - lag
            val_minus = h_vec[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * ((1 - val_plus) - val_minus) + 2 * 0.2 * (np.sum(h_vec) - target_sum)
        return grad

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_perturbed = h.copy() + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        h_perturbed *= scale
        res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=gradient, options={'maxiter': 1000, 'ftol': 1e-6})
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        h = best_h.copy()

    # Final projection
    final_sum = np.sum(best_h)
    scale = target_sum / final_sum if final_sum != 0 else 1.0
    best_h *= scale
    best_h = np.clip(best_h, 0, 1)
    conv = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv) * dx

    return (best_h, final_c5, n_points)
