    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    elapsed = time.time() - start_time
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n_points = n_points
    else:
        current_c5 = objective(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_initial
            best_n_points = n_points

    # Check if time remains for n_points=100
    if elapsed < budget_s:
        n_points = 100
        if 'initial_h_values' in globals():
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        h_initial = np.array(h_initial, dtype=np.float64)

        def objective_100(h_vec):
            h = h_vec.reshape(n_points)
            one_minus_h = 1 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            return max_corr * dx

        def constraint_100(h_vec):
            return np.sum(h_vec) - n_points / 2

        bounds_100 = [(0.0, 1.0) for _ in range(n_points)]

        result_100 = minimize(objective_100, h_initial, method='L-BFGS-B', bounds=bounds_100, constraints={'type': 'eq', 'fun': constraint_100})
        elapsed_100 = time.time() - start_time
        if result_100.success:
            current_c5 = result_100.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_100.x
                best_n_points = n_points
        else:
            current_c5 = objective_100(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
                best_n_points = n_points

    return (best_h.tolist(), best_c5, best_n_points)
