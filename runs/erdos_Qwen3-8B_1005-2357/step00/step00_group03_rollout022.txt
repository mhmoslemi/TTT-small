            best_h_refined_new = np.clip(best_h_refined_new, 0, 1)
        
        corr_new = np.correlate(best_h_refined_new, 1 - best_h_refined_new, mode='full')
        c5_bound_new = np.max(corr_new) * dx_new
        if c5_bound_new < c5_bound:
            best_h_refined = best_h_refined_new
            c5_bound = c5_bound_new
            n = n_new
    
    return (best_h_refined, c5_bound, n)
