            h_test = h + alpha * grad
            h_test = _clip_and_scale(h_test)
            c5_test = _compute_C5(h_test)
            if c5_test < best_c5_alpha:
                best_alpha = alpha
                best_c5_alpha = c5_test
        h = h + best_alpha * grad
        h = _clip_and_scale(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.random.rand() < 0.1:
            noise = 0.01 * np.random.randn(n_points)
            h = h + noise
            h = _clip_and_scale(h)
    return (best_h, best_c5, n_points)
