import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Assume provided by the environment

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points_list = [95, 100, 150, 200]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_gradient(h, n_points):
        dx = 2.0 / n_points
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        max_corr = np.max(corr)
        max_indices = np.where(corr == max_corr)[0]
        grad = np.zeros_like(h)
        for idx in max_indices:
            lag = idx - (n_points - 1)
            if lag < 0:
                for i in range(n_points):
                    if i + lag >= 0 and i + lag < n_points:
                        grad[i] += (1 - h[i + lag]) - h[i - lag]
            else:
                for i in range(n_points):
                    if i - lag >= 0 and i - lag < n_points:
                        grad[i] += (1 - h[i - lag]) - h[i + lag]
        return grad * dx
    
    def repair_solution(h, n_points):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) < 1e-6:
            return h
        scale = (n_points / 2) / total
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

            h = np.random.rand(n_points)
            h = repair_solution(h, n_points)
        
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            try:
                result = opt.minimize(
                    lambda x: compute_c5(x, n_points),
                    h,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    jac=lambda x: compute_gradient(x, n_points),
                    options={'maxiter': 1000, 'ftol': 1e-8}
                )
