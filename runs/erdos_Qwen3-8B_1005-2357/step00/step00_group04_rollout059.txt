import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    initial_h = initial_h_values
    h = initial_h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    start_time = time.time()
    for _ in range(1000):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        total = np.sum(h_perturbed) * dx
        if total == 0:
            continue
        scaling = (n_points / 2) / np.sum(h_perturbed)
        h_perturbed *= scaling
        current_c5 = compute_c5(h_perturbed, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
