                    scale = (n / 2) / sum_h
                    h_opt = h_opt * scale
                h_arr = np.array(h_opt)
                one_minus_h = 1.0 - h_arr
                corr = np.correlate(h_arr, one_minus_h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n

            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break

    if best_h is None:
        best_h = initial_h_values
        h_arr = np.array(best_h)
        one_minus_h = 1.0 - h_arr
        corr = np.correlate(h_arr, one_minus_h, mode='full')
        dx = 2.0 / len(best_h)
        best_c5 = np.max(corr) * dx

    return (best_h, best_c5, best_n)
