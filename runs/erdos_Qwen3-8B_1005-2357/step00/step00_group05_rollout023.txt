            val = a_current[0]
            for k in range(1, N + 1):
                val += a_current[k] * np.cos(np.pi * k * x[i])
                val += b_current[k - 1] * np.sin(np.pi * k * x[i])
            h_vals[i] = val
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    bounds = [(-10, 10) for _ in range(2 * N)]
    result = minimize(objective, coeffs, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
    best_coeffs = result.x
    a_best = np.zeros(N + 1)
    a_best[0] = 0.5
    for i in range(1, N + 1):
        a_best[i] = best_coeffs[i - 1]
    b_best = np.zeros(N)
    for i in range(N):
        b_best[i] = best_coeffs[N + i]
    h_best = np.zeros(n_points)
    for i in range(n_points):
        val = a_best[0]
        for k in range(1, N + 1):
            val += a_best[k] * np.cos(np.pi * k * x[i])
            val += b_best[k - 1] * np.sin(np.pi * k * x[i])
        h_best[i] = val
    h_best = np.clip(h_best, 0, 1)
    corr = np.correlate(h_best, 1 - h_best, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_best, c5_bound, n_points)
