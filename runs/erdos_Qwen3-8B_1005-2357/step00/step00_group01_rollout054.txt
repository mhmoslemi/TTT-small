import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_values = [47, 94, 188, 376]
    time_budgets = {47: 550, 94: 300, 188: 200, 376: 50}
    best_c5 = float('inf')
    best_h = None
    total_time = 0.0

    def compute_c5(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial(n, initial_h=None):
        if initial_h is not None and len(initial_h) == n:
            return initial_h.copy()
        else:
            dirichlet_samples = np.random.default_rng(seed).dirichlet([1.0]*n)
            return dirichlet_samples * (n / 2)

    for n in n_values:
        if total_time >= budget_s:
            break
        time_limit = time_budgets.get(n, 0)
        if time_limit == 0:
            continue
        if total_time + time_limit > budget_s:
            time_limit = budget_s - total_time
        h_initial = generate_initial(n, initial_h if n == 47 else None)
        bounds = [(0.0, 1.0) for _ in range(n)]
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n / 2
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        start_time = time.time()
        result = minimize(lambda x: compute_c5(x, n), h_initial, method='trust-constr',
                          bounds=bounds, constraints=cons, options={'maxiter': 100})
        elapsed = time.time() - start_time
        total_time += elapsed
        if total_time >= budget_s:
            break
        h_opt = np.clip(result.x, 0.0, 1.0)
        current_sum = np.sum(h_opt)
        delta = (n / 2) - current_sum
        if delta != 0:
            h_opt[0] += delta
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5 = compute_c5(h_opt, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    if best_h is None and initial_h is not None:
        best_h = initial_h
        best_c5 = compute_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
