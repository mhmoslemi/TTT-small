            x = np.atleast_2d(x)
            mean, std = gp.predict(x, return_std=True)
            if std == 0:
                ei = 0
            else:
                z = (mean - best_c5) / std
                ei = (mean - best_c5) * np.cumsum(z) + std * np.cumsum(np.exp(-z**2 / 2) / np.sqrt(2 * np.pi))
            return -ei
        
        x0 = np.random.rand(42)
        x0 = project_feasibility(x0, 42)
        result = minimize(acquisition, x0, method='L-BFGS-B', bounds=[(0, 1)] * 42)
        x_new = result.x
        x_new = project_feasibility(x_new, 42)
        
        c5_new = evaluate_overlap(x_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
        
        X_new = np.array([x_new]).reshape(1, -1)
        y_new = np.array([c5_new])
        gp.fit(np.vstack((X_initial, X_new)), np.hstack((y_initial, y_new)))
    
    if best_c5 < 0.380876:
        n_points = 84
        dx = 2.0 / n_points
        h_upscaled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 42), best_h)
        h_upscaled = project_feasibility(h_upscaled, n_points)
        c5_new = evaluate_overlap(h_upscaled, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_upscaled
    
    return (best_h, best_c5, n_points)
