                idx = np.where(h_clipped < 1)[0]
                if len(idx) > 0:
                    remaining = D - S_clipped
                    h_clipped[idx] += remaining / len(idx)
                else:
                    h_clipped = np.full(n, D / n)
            h_initial = h_clipped

        dx = 2.0 / n
        def obj(h):
            cor = np.correlate(h, 1 - h, mode='full')
            return np.max(cor * dx)
        bounds = [(0., 1.) for _ in range(n)]
        def cons(h):
            return np.sum(h) - n / 2
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons})
        h_res = np.clip(res.x, 0, 1)
        sum_h = np.sum(h_res)
        if abs(sum_h - n / 2) > 1e-6:
            h_res *= (n / 2) / sum_h
        cor = np.correlate(h_res, 1 - h_res, mode='full')
        c5 = np.max(cor * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res
    return best_h, best_c5, len(best_h)
