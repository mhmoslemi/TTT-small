import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    from scipy.signal import correlate
    from scipy.optimize import Bounds, minimize
    
    # Assume initial_h_values is provided as a global variable
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Feasibility projection
    def project_feasibility(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped) * dx
        scale = 1.0 / total if total != 0 else 1.0
        return h_clipped * scale
    
    # Initial guess
    h_initial = project_feasibility(initial_h_values)
    
    # Optimization
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    elapsed_time = time.time() - start_time
    
    best_h = result.x
    c5 = objective(best_h)
    
    return best_h, c5, n_points
