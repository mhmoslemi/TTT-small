import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_list = [47, 94, 188, 376]
    time_budgets = {47: 550, 94: 300, 188: 200, 376: 50}
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    def repair(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        target = n / 2
        if abs(total - target) < 1e-6:
            return h
        if total > target:
            scale = target / total
            h = h * scale
        else:
            deficit = target - total
            h = h + deficit * (1 - h) / (np.sum(1 - h) + 1e-12)
        return h
    
    def evaluate(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    for n in n_list:
        if time.time() - start_time > budget_s:
            break
        if n not in time_budgets:
            continue
        if time.time() - start_time > time_budgets[n]:
            continue
        if initial_h is not None and n == 47:
            h_pop = np.array(initial_h, dtype=np.float64)
            h_pop = repair(h_pop, n)
        else:
            h_pop = np.random.uniform(0, 1, n)
            h_pop = repair(h_pop, n)
        pop_size = 50
        generations = 0
        while time.time() - start_time < time_budgets[n]:
            generations += 1
            offspring = []
            for _ in range(100):
                parent = np.random.choice(h_pop)
                noise = np.random.normal(0, 0.1, n)
                child = parent + noise
                child = repair(child, n)
                offspring.append(child)
            offspring = np.array(offspring)
            scores = np.array([evaluate(h, n) for h in offspring])
            indices = np.argsort(scores)
            h_pop = offspring[indices[:pop_size]]
            current_best = evaluate(h_pop[0], n)
            if current_best < best_c5:
                best_c5 = current_best
                best_h = h_pop[0].copy()
    
    if best_h is None:
        if initial_h is not None:
            best_h = initial_h
        else:
            best_h = np.random.uniform(0, 1, 47)
            best_h = repair(best_h, 47)
        best_c5 = evaluate(best_h, 47)
    return (best_h, best_c5, len(best_h))
