    start_time = time.time()
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        max_indices = np.where(corr == max_corr)[0]
        grad = np.zeros(n_points)
        for idx in max_indices:
            k = idx - (n_points - 1)

                term1 = 0

                    term1 = 1 - x[j + k]
                term2 = 0
                if 0 <= j - k < n_points:
                    term2 = -x[j - k]
                grad[j] += term1 + term2
        return grad
