            child = repair_sum(child, target_sum)
        else:
            child = parent1.copy()

        if np.random.rand() < mutation_rate:
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] = np.random.rand()
            child = repair_sum(child, target_sum)

        worst_idx = np.argmax(fitness)
        population[worst_idx] = child.copy()

        def objective_func(h_vec):
            h = h_vec.copy()
            h = repair_sum(h, target_sum)
            return compute_C5(h)

        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective_func, current_best, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            optimized_h = result.x
            optimized_c5 = compute_C5(optimized_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h.copy()
                best_c5 = optimized_c5

    return (best_h, best_c5, n_points)
