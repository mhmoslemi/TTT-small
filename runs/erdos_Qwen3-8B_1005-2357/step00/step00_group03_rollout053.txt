import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5
    sum_h = np.sum(h)
    scale = (n_points / 2.0) / sum_h
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def find_active_lag(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        lag = idx - (n_points - 1)
        return lag
    
    def compute_gradient(h_vec, L):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if 0 <= i - L < len(h_vec):
                grad[i] = (1 - h_vec[i - L]) * dx
        return grad
    
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def gradient(h_vec):
        L = find_active_lag(h_vec)
        return compute_gradient(h_vec, L)
    
    cons = (
        {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2},
        {'type': 'ineq', 'fun': lambda h_vec: h_vec},
        {'type': 'ineq', 'fun': lambda h_vec: 1 - h_vec},
    )
    
    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        jac=gradient,
        constraints=cons,
        options={'maxiter': 100}
    )
    
    best_h = result.x
