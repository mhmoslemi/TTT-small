import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    assert np.isclose(np.sum(initial_h), n_points / 2), "Initial sum invalid"
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    start_time = time.time()
    
    # Generate perturbations
    n_starts = 10
    perturbations = [initial_h.copy() for _ in range(n_starts)]
    for i in range(n_starts):
        h = perturbations[i].copy()
        idxs = np.random.choice(n_points, 5, replace=False)
        h[idxs] += np.random.uniform(-0.1, 0.1, size=5)
        h = np.clip(h, 0, 1)
        delta = (n_points / 2) - np.sum(h)
        if delta != 0:
            idx = np.random.randint(n_points)
            h[idx] += delta
            h = np.clip(h, 0, 1)
        perturbations[i] = h
    
    for h in perturbations:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if c5_initial < best_c5:
