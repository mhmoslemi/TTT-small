        n_points = 42
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (1.0 / (dx * np.sum(h)))
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h):
        corr = correlate(h, h, mode='full')
        max_corr = np.max(corr)
        return 1.0 - max_corr * dx

    def constraint_sum(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'ftol': 1e-6, 'maxiter': 1000})

    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
