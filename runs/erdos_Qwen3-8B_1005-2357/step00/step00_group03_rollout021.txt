        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points - 1)
                child1, child2 = np.concatenate([p1[:point], p2[point:]]), np.concatenate([p2[:point], p1[point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            new_population.extend([child1, child2])
        
        for h in new_population:
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                h[idx] += np.random.uniform(-0.1, 0.1)
                h[idx] = np.clip(h[idx], 0, 1)
        
        for h in new_population:
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-5:
                scale = (n_points / 2) / h_sum
                h = h * scale
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-5:
                diff = (n_points / 2) - h_sum
                idx = np.random.randint(n_points)
                h[idx] += diff
                h[idx] = np.clip(h[idx], 0, 1)
        
        population = new_population
        for h in population:
            if fitness(h) < best_fitness:
                best_fitness = fitness(h)
                best_h = h.copy()
    
    def objective(h):
        return compute_c5(h)
    
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
        {'type': 'ineq', 'fun': lambda h: h},
        {'type': 'ineq', 'fun': lambda h: 1 - h}
    ]
    
    def objective_grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        l = max_idx
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + l < n_points:
                grad[i] = (1 - h[i + l]) * dx
        return grad
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        jac=objective_grad,
        constraints=constraints,
        options={'maxiter': 500}
    )
    
    best_h = result.x
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2) > 1e-5:
        scale = (n_points / 2) / final_sum
        best_h = best_h * scale
    
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
