import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_points_list = [47, 94, 188, 376]
    time_budgets = {47: 550, 94: 300, 188: 200, 376: 50}
    total_time = 0

    def objective(h_flat):
        h = h_flat.reshape(n_points, 1)
        corr = np.correlate(h.flatten(), 1 - h.flatten(), mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    def clip_and_scale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scale = (n_points / 2) / sum_clipped
        return h_clipped * scale

    for n_points in n_points_list:
        if total_time >= budget_s:
            break
        time_remaining = budget_s - total_time
        if n_points not in time_budgets or time_budgets[n_points] > time_remaining:
            continue
        time_for_n = time_budgets[n_points]
        dx = 2.0 / n_points
        if initial_h_values is not None and len(initial_h_values) == n_points:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        h_initial = clip_and_scale(h_initial)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        result = opt.minimize(
            objective,
            h_initial,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'disp': False}
        )
        h_candidate = result.x
        h_candidate = clip_and_scale(h_candidate)
        if np.all(0 <= h_candidate <= 1) and np.isclose(np.sum(h_candidate), n_points / 2):
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
        total_time += time_for_n

    if best_h is None:
        if initial_h_values is not None:
            h_final = initial_h_values.copy()
        else:
            h_final = np.random.dirichlet(np.ones(47)) * (47 / 2)
        h_final = clip_and_scale(h_final)
        c5_final = objective(h_final)
        return (h_final, c5_final, 47)
    else:
        return (best_h, best_c5, len(best_h))
