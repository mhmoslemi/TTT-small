    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', jac=grad_func, constraints=cons, bounds=bounds, options={'ftol': 1e-8})
    best_h = result.x
    
    best_h = repair(best_h)
    c5_bound = evaluate_c5(best_h, n_points, 2.0 / n_points)
    return (best_h, c5_bound, n_points)
