import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    time_remaining = budget_s
    start_time = time.time()

    n_values = [47, 94, 188, 376]
    time_budgets = [550, 300, 200, 50]

    def generate_initial(n):
        if n == 47 and hasattr(globals(), 'initial_h_values'):
            return initial_h_values.copy()
        dirichlet = np.random.dirichlet(np.ones(n))
        dirichlet *= (n/2)
        dirichlet = np.clip(dirichlet, 0, 1)
        current_sum = np.sum(dirichlet)
        diff = (n/2) - current_sum
        dirichlet[-1] += diff
        dirichlet = np.clip(dirichlet, 0, 1)
        return dirichlet

    for i, n in enumerate(n_values):
        if time_remaining <= 0:
            break
        time_per_restart = time_budgets[i] / 5
        initial_h = generate_initial(n)
        for _ in range(5):
            h = initial_h.copy()
            def objective(x):
                h_vec = x
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                dx = 2.0 / n
                return np.max(corr) * dx
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
            bounds = [(0, 1) for _ in range(n)]
            try:
                result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
                if result.success:
                    current_h = result.x
                    current_h = np.clip(current_h, 0, 1)
                    current_sum = np.sum(current_h)
                    diff = (n/2) - current_sum
                    if abs(diff) > 1e-6:
                        current_h[-1] += diff
                        current_h = np.clip(current_h, 0, 1)
                    corr = np.correlate(current_h, 1 - current_h, mode='full')
                    dx = 2.0 / n
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = current_h
                        best_n = n
            except:
                pass
            elapsed = time.time() - start_time
            time_remaining = budget_s - elapsed
            if time_remaining <= 0:
                break
    return (best_h, best_c5, best_n)
