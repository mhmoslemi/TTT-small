import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c = compute_C5(h)
    start_time = time.time()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def clip_and_scale(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        scale = (n_points / 2) / total
        return h_clip * scale
    
    for _ in range(1000):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k0 = max_idx - (n_points - 1)
        
        g = np.zeros(n_points)
        for i in range(n_points):
            i_forward = i + k0
            term1 = 1 - h[i_forward] if 0 <= i_forward < n_points else 0.0
            i_backward = i - k0
            term2 = h[i_backward] if 0 <= i_backward < n_points else 0.0
            g[i] = term1 - term2
        
        best_alpha = 0.0
        best_c_candidate = best_c
        for alpha in [0.01, 0.1, 0.5, 1.0]:
            h_candidate = h + alpha * g
            h_clip = np.clip(h_candidate, 0, 1)
            total = np.sum(h_clip)
            scale = (n_points / 2) / total
            h_scaled = h_clip * scale
            c_candidate = compute_C5(h_scaled)
            if c_candidate < best_c_candidate:
                best_alpha = alpha
                best_c_candidate = c_candidate
        
        h = h + best_alpha * g
        h = clip_and_scale(h)
        current_c = compute_C5(h)
