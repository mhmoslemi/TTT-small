    result_bh = basinhopping(objective, h, niter=50, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0,1)]*len(h)})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = result_bh.x
        best_c5 = objective(best_h)
        return (best_h, best_c5, n_points)

    result_local = minimize(objective, result_bh.x, method='L-BFGS-B', bounds=[(0,1)]*len(h), options={'maxiter': 100})
    best_h = result_local.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    n_points_new = 100
    dx_new = 2.0 / n_points_new
    x_old = np.linspace(0, 2, n_points)
    x_new = np.linspace(0, 2, n_points_new)
    h_new = np.interp(x_new, x_old, best_h)
    h_new = np.clip(h_new, 0, 1)
    sum_h_new = np.sum(h_new)
    target_sum_new = n_points_new / 2
    scale = target_sum_new / sum_h_new
    h_new *= scale

    def objective_new(h_vec):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        penalty = 1e3 * np.abs(sum_h - target_sum_new)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_new + penalty

    result_bh_new = basinhopping(objective_new, h_new, niter=30, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0,1)]*len(h_new)})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_new, objective_new(h_new), n_points_new)

    result_local_new = minimize(objective_new, result_bh_new.x, method='L-BFGS-B', bounds=[(0,1)]*len(h_new), options={'maxiter': 100})
    best_h_new = result_local_new.x
    best_c5_new = objective_new(best_h_new)
    if best_c5_new < best_c5:
        best_h = best_h_new
        best_c5 = best_c5_new
        n_points = n_points_new

    return (best_h, best_c5, n_points)
