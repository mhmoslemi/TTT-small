import numpy as np
import time

def evaluate_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasibility(h, dx):
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if abs(total - 1) > 1e-6:
        h *= 1.0 / total
    return h
