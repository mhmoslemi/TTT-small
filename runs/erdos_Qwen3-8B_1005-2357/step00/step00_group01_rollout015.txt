import numpy as np
import time

def repair(h, n):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    t = n / 2
    if np.isclose(s, t):
        return h
    if s > t:
        h *= t / s
    else:
        d = t - s
        a = np.sum(1 - h)
        if a > 0:
            h += d * (1 - h) / a
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evolve(n, initial_pop, time_budget, seed):
    np.random.seed(seed)
    pop = initial_pop.copy()
    best = pop[0]
    best_c5 = compute_c5(best, n)
    start = time.time()
    while time.time() - start < time_budget:
        off = []
        for _ in range(100):
            p = np.random.choice(pop)
            m = p + np.random.normal(0, 0.1, n)
            m = repair(m, n)
            off.append(m)
        off_c5 = [compute_c5(h, n) for h in off]
        combined = list(zip(off, off_c5))
        combined.sort(key=lambda x: x[1])
        new_pop = [h for h, _ in combined[:50]]
        pop = new_pop
        for h in pop:
            c = compute_c5(h, n)
            if c < best_c5:
                best_c5 = c
                best = h
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    n_list = [47, 94, 188, 376]
    time_budgets = {47: 550, 94: 300, 188: 200, 376: 50}
    total_time = 0
    best_h, best_c5, best_n = None, float('inf'), 47

    if 'initial_h_values' in globals():
        h0 = initial_h_values
        n = len(h0)
        if n == 47:
            h0 = repair(h0, n)
            if np.isclose(np.sum(h0), n / 2):
                pop = [h0]
                for _ in range(49):
                    h = np.random.uniform(0, 1, n)
                    h = repair(h, n)
                    pop.append(h)
                h, c = evolve(n, pop, time_budgets[n], seed)
                if c < best_c5:
                    best_h, best_c5, best_n = h, c, n
                total_time += time_budgets[n]

    if total_time < budget_s:
        n = 47
        pop = [np.random.uniform(0, 1, n) for _ in range(50)]
        for i in range(50):
            pop[i] = repair(pop[i], n)
        h, c = evolve(n, pop, time_budgets[n], seed)
        if c < best_c5:
            best_h, best_c5, best_n = h, c, n
        total_time += time_budgets[n]
