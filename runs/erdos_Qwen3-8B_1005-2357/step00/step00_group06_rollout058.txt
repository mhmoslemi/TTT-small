import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial symmetric h
    h_sym = np.random.rand(n_points // 2)
    h_full = np.concatenate([h_sym, h_sym[::-1]])

    # Repair function
    def repair(h_full):
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full) * dx
        if abs(current_sum - target_sum) > 1e-5:
            factor = target_sum / current_sum
            h_full = h_full * factor
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full) * dx
            if current_sum < target_sum:
                available_increase = (1 - h_full).sum() * dx
                if available_increase > 0:
                    factor = (target_sum - current_sum) / available_increase
                    h_full = np.minimum(h_full + factor, 1)
        return h_full

    h_full = repair(h_full)

    # Objective function
    def objective(x):
        h_full = x
        g = 1 - h_full
        corr = np.correlate(h_full, g, mode='full')
        return dx * np.max(corr)

    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - target_sum}

    # Optimization
    result = minimize(objective, h_full, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})

    # Final repair
    best_h_full = repair(result.x)
    best_c5 = objective(best_h_full)

    return best_h_full, best_c5, n_points
