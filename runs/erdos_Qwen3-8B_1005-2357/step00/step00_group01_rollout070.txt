            bounds = [(0, 1) for _ in range(n_points)]

            def constraint_func(h):
                return np.sum(h) - n_points / 2

            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
            )

            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            T = n_points / 2
            if sum_h > T:
                h_opt *= T / sum_h

            cor = np.correlate(h_opt, 1 - h_opt, mode='full')
            overlap = cor * dx
            c5 = np.max(overlap)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n_points

    return (best_h, best_c5, best_n)
