        initial_h = np.clip(initial_h * scale, 0, 1)
    h = initial_h

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def objective(h_vec):
        c5 = compute_c5(h_vec)
        sum_penalty = 1e6 * (np.sum(h_vec) - n_points / 2)**2
        box_penalty = 1e6 * np.sum(np.maximum(0, h_vec - 1) + np.maximum(0, -h_vec))
        return c5 + sum_penalty + box_penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=50, mutation=(0.5, 1), recombination=0.9,
                                   tol=1e-6, maxiter=100, seed=seed)
    elapsed = time.time() - start_time
    h_opt = result.x
    c5_bound = compute_c5(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if abs(sum_h - n_points / 2) > 1e-5:
        scale = (n_points / 2) / sum_h
        h_opt = np.clip(h_opt * scale, 0, 1)

    if elapsed < budget_s:
        n_new = n_points * 2
        h_interp = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_points), h_opt)
        sum_h_interp = np.sum(h_interp)
        scale = (n_new / 2) / sum_h_interp
        h_interp = np.clip(h_interp * scale, 0, 1)
        dx_new = 2.0 / n_new

        def objective_new(h_vec):
            c5 = compute_c5(h_vec)
            sum_penalty = 1e6 * (np.sum(h_vec) - n_new / 2)**2
            box_penalty = 1e6 * np.sum(np.maximum(0, h_vec - 1) + np.maximum(0, -h_vec))
            return c5 + sum_penalty + box_penalty

        bounds_new = [(0.0, 1.0) for _ in range(n_new)]
        result_new = differential_evolution(objective_new, bounds_new, strategy='best1bin',
                                            popsize=50, mutation=(0.5, 1), recombination=0.9,
                                            tol=1e-6, maxiter=50, seed=seed)
        elapsed_new = time.time() - start_time
        if elapsed_new < budget_s:
            h_opt = result_new.x
            c5_bound = compute_c5(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            if abs(sum_h - n_new / 2) > 1e-5:
                scale = (n_new / 2) / sum_h
                h_opt = np.clip(h_opt * scale, 0, 1)
            n_points = n_new

    return (h_opt, c5_bound, n_points)
