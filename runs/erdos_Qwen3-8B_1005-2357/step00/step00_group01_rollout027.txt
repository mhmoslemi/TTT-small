                        child[k] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                required = n_points / 2
                diff = required - total
                if abs(diff) > 1e-6:
                    child[0] += diff
                    child = np.clip(child, 0, 1)
                new_population.append(child)
        for _ in range(10):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            new_population.append(h)
        population = selected + new_population
        population = population[:population_size]
        for c5, h in evaluations:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    return (best_h, best_c5, n_points)
