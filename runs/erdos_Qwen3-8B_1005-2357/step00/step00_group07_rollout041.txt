```python
import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        return h * target_sum / total
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    initial_h = initial_h_values
    population = [initial_h]
    for _ in range(49):
        h = np.random.uniform(0, 1, n_points)
        h = clip_and_scale(h)
        population.append(h)
    
    fitness = [evaluate(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    pop_size = 50
    max_gen = 1000
    mu = 0.5
    cr = 0.9
    mu_min, mu_max = 0.1, 1.0
    cr_min, cr_max = 0.5, 1.0
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            a, b, c = random.sample(population, 3)
            mutant = a + mu * (b - c)
            mutant = clip_and_scale(mutant)
            target = random.choice(population)
            mask = np.random.rand(n_points) < cr
            offspring = np.copy(target)
            offspring[mask] = mutant[mask]
            new_pop.append(offspring)
        population = new_pop
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best]
        if fitness[current_best] < best_c5:
            mu = max(mu - 0.01, mu_min)
            cr = max(cr - 0.01, cr_min)
        else:
            mu = min(mu + 0.01, mu_max)
            cr = min(cr + 0.01, cr_max)
    
    return (best_h, best_c5, n_points)
```
