            best_c5 = best_c5_new
            n_points = n_points_new

    sum_best = np.sum(best_h)
    scale = target_sum / sum_best if sum_best != 0 else 1.0
    best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, best_c5, n_points)
