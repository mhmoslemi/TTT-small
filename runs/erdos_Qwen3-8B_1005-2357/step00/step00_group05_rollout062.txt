import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 70
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2 - 1) * 0.5 + 0.5
        h_initial[-1] += (70 / 2 - np.sum(h_initial))
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_eq(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq}],
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    elapsed = time.time() - start_time

    if result.success and elapsed < budget_s:
        h_values = result.x
        c5_bound = objective(h_values)
        n_points_used = n_points
        return (h_values, c5_bound, n_points_used)

    if elapsed >= budget_s:
        return (h_initial, objective(h_initial), n_points)

    # Try differential evolution with penalty
    def objective_with_penalty(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - n_points / 2.0)**2
        return max_corr * dx + penalty

    result_de = differential_evolution(
        objective_with_penalty,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed_de = time.time() - start_time
    if elapsed_de < budget_s:
        h_values = result_de.x
        c5_bound = objective(h_values)
        n_points_used = n_points
        return (h_values, c5_bound, n_points_used)

    # Fallback to initial solution
    return (h_initial, objective(h_initial), n_points)
