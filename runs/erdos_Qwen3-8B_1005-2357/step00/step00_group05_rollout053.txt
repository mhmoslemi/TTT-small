import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 70
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    result = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}])
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    result = minimize(obj, best_h, method='trust-constrained', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}])
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        n_points_new = 100
        dx_new = 2.0 / n_points_new
        h_new = np.random.uniform(0, 1, n_points_new)
        h_new = h_new / np.sum(h_new) * (n_points_new / 2)
        result = minimize(obj, h_new, method='COBYLA', bounds=Bounds(0, 1), constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_new / 2}])
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5

    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-5:
        scale = (n_points / 2) / sum_h
        best_h = best_h * scale

    return (best_h, best_c5, n_points)
