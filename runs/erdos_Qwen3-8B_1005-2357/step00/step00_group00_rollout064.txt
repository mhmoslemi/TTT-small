    idx = np.random.choice(len(h_p), n_flips, replace=False)
    h_p[idx] += np.random.uniform(-noise_scale, noise_scale, size=n_flips)
    h_p = np.clip(h_p, 0, 1)
    target = len(h_p) / 2
    delta = target - np.sum(h_p)
    h_p[-1] += delta
    h_p = np.clip(h_p, 0, 1)
    return h_p

def optimize_candidate(h_initial, target_sum, n_points):
    def obj(h):
        return compute_c5(h)
    bounds = [(0,1)]*n_points
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    res = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons)
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    target_sum = n / 2
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(10):
        h_perturbed = perturb(initial_h)
        try:
            optimized = optimize_candidate(h_perturbed, target_sum, n)
            current_c5 = compute_c5(optimized)
            if current_c5 < best_c5:
                best_h = optimized
                best_c5 = current_c5
        except:
            pass
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n)
