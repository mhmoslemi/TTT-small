            delta = 1e-5
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] = np.clip(h_plus[i] + delta, 0, 1)
                obj_plus = objective100(h_plus)
                h_minus = h.copy()
                h_minus[i] = np.clip(h_minus[i] - delta, 0, 1)
                obj_minus = objective100(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * delta)
            return grad

        bounds100 = [(0, 1)] * 100
        cons100 = {'type': 'eq', 'fun': constraint100}
        result100 = optimize.minimize(
            fun=objective100,
            x0=h100,
            method='SLSQP',
            bounds=bounds100,
            constraints=cons100,
            jac=grad_objective100,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result100.success:
            best_h = result100.x
        else:
            best_h = h100
        sum_h = np.sum(best_h)
        if abs(sum_h - 100 / 2) > 1e-6:
            scale = (100 / 2) / sum_h
            best_h = np.clip(best_h * scale, 0, 1)
        c5_bound = objective100(best_h)
        n_points = 100

    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, c5_bound, n_points)
