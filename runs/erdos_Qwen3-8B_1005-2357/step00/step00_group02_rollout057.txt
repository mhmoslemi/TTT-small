import numpy as np
from scipy.optimize import minimize
import time

def objective(h, n, dx):
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def grad_objective(h, n, dx):
    grad = np.zeros(n)
    delta = 1e-5
    for i in range(n):
        h_p = h.copy()
        h_p[i] = np.clip(h[i] + delta, 0, 1)
        obj_p = objective(h_p, n, dx)
        h_m = h.copy()
        h_m[i] = np.clip(h[i] - delta, 0, 1)
        obj_m = objective(h_m, n, dx)
        grad[i] = (obj_p - obj_m) / (2 * delta)
    return grad

def constraint(h, n):
    return n/2 - np.sum(h)

def perturb(h, n, seed):
    rng = np.random.RandomState(seed)
    h_p = h + rng.uniform(-0.1, 0.1, n)
    h_p = np.clip(h_p, 0, 1)
    scale = (n/2) / np.sum(h_p)
    h_p = h_p * scale
    h_p = np.clip(h_p, 0, 1)
    return h_p

def transfer_solution(h, n1, n2):

    h2 = np.interp(x2, x1, h)
    scale = (n2/2) / np.sum(h2)
    h2 = h2 * scale
    h2 = np.clip(h2, 0, 1)
    return h2

    for _ in range(5):
        h = perturb(h0, n1, seed)
        res = minimize(lambda x: objective(x, n1, dx1), h, method='SLSQP',
