        max_corr = np.max(corr)
        c5 = max_corr * dx
        sum_penalty = 1e6 * (np.sum(h_vec) - n_points / 2)**2
        return c5 + sum_penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(objective, bounds, seed=seed, maxiter=100)
    elapsed = time.time() - start_time
    best_h = result.x
    
    def objective_refine(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds_refine = [(0.0, 1.0) for _ in range(n_points)]
    result_refine = minimize(objective_refine, best_h, method='SLSQP', bounds=bounds_refine, constraints=cons, options={'ftol': 1e-6})
    best_h_refine = result_refine.x
    
    corr = np.correlate(best_h_refine, 1 - best_h_refine, mode='full')
    max_corr = np.max(corr)
    c5_bound = max_corr * dx
    return (best_h_refine, c5_bound, n_points)
