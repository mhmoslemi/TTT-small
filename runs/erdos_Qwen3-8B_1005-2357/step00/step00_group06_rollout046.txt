    def project_integral(h):
        target = n_points / 2
        diff = np.sum(h) - target
        if abs(diff) < 1e-6:
            return h

        available[:] = 1 - h
        available *= diff
        available = np.abs(available)
        idx = np.argsort(available)
        for i in idx:
            if diff == 0:
                break
            change = min(available[i], abs(diff))
            if h[i] < 1:
                h[i] += change * np.sign(diff)
                diff -= change
            elif h[i] > 0:
                h[i] -= change * np.sign(diff)
                diff += change
        return np.clip(h, 0, 1)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx

        return c5 + penalty
    
    def gradient(h_flat):
        h = h_flat.reshape(n_points)

        k = max_idx - n_points + 1

            if i + k < 0 or i + k >= n_points:
                continue
            grad[i] += (1 - h[i + k]) - h[i - k]
        grad *= dx
        grad += 2 * 1e4 * (np.sum(h) - n_points / 2)
        return grad.flatten()
    
    best_h = h.copy()

    for mu in [0.0, 0.1, 0.2]:
