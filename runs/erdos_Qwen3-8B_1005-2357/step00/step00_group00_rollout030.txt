            h = x.reshape(n_points)
            return compute_c5(h, n_points)
        bounds = [(0,1)]*n_points
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter':100})
        best_h = res.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n_points)
