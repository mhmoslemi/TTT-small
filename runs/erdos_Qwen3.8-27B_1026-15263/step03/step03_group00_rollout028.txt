import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:n] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha, top_k=50):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = m_val + np.log(total) / alpha
    g = np.zeros(n)
    # Only compute gradients for lags with significant weight
    active = np.where(w > 1e-10)[0]
    # Sort by weight and take top_k
    if len(active) > top_k:
        active = active[np.argsort(w[active])[::-1][:top_k]]
    for idx in active:
        lag = int(idx) - (n - 1)
        g += w[idx] * corr_gradient_single(h, lag)
    return obj, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500, top_k=50):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': ones_jac}]
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x):
        return smooth_max_and_grad(x, alpha, top_k)[0]
    
    def grad_func(x):
        return smooth_max_and_grad(x, alpha, top_k)[1]
    
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def ones_jac(x):
    return np.ones(len(x))

def coordinate_descent(h, target_sum, max_iters=10000, top_k=3, step_fracs=None):
    n = len(h)
    if step_fracs is None:
        step_fracs = [0.5, 0.3, 0.15, 0.08, 0.04]
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        # Get top k indices
        top_idx = np.argsort(corr)[::-1][:top_k]
        # Composite gradient
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += corr_gradient_single(h, int(lag))
        g /= top_k
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-14:
            stale += 1
            if stale > 100:
                break
            continue
        stale = 0
        max_move = min(1.0 - h[i_inc], h[i_dec])
        improved = False
        for frac in step_fracs:
            move = frac * max_move
            if move < 1e-12:
                continue
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                improved = True
                break
            else:
                h[i_inc] -= move
                h[i_dec] += move
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
        if not improved:
            stale += 1
            if stale > 200:
                break
    return h, c5

def projected_subgrad(h, target_sum, alpha, n_iters, step_init=0.01, decay=0.999):

    step = step_init
    best_c5 = compute_c5(h)

            best_h = h_new.copy()
        h = h_new
        step *= decay
        if step < 1e-8:

    return best_h, best_c5

        return budget_s - (time.time() - t_start)
