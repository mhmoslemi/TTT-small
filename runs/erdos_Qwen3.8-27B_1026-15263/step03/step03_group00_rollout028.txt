    # h[j-m+n-1] * (1-h[j]), i.e., h[j'] * (1-h[j]) where j' = j-m+n-1

    valid_s = (second_offset >= 0) & (second_offset < n)

            max_move = min(1.0 - h[i_inc], h[i_dec])
            step = rng.uniform(0.05, 0.5) * max_move
            h[i_inc] += step
            h[i_dec] -= step
        elif r < 0.8:
            ci = np.where(can_inc)[0]
            cd = np.where(can_dec)[0]
            i_inc = int(ci[rng.randint(len(ci))])
            i_dec = int(cd[rng.randint(len(cd))])
            if i_inc == i_dec:
                t *= cool_rate
                continue
            max_move = min(1.0 - h[i_inc], h[i_dec])
            step = rng.uniform(0.05, 0.6) * max_move
            h[i_inc] += step
            h[i_dec] -= step
        else:
            bs = max(3, n // 15)
            s1 = int(rng.randint(0, n - bs))
            s2 = int(rng.randint(0, n - bs))
            if abs(s1 - s2) < bs:
                s2 = (s2 + n // 2) % (n - bs)
            transfer = rng.uniform(-0.4, 0.4) * bs * 0.3
            h[s1:s1 + bs] += transfer / bs
            h[s2:s2 + bs] -= transfer / bs
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h = h_saved
        t *= cool_rate
        if it > 0 and it % 5000 == 0 and c5 > best_c5 + 5 * t:
            h = best_h.copy()
            c5 = best_c5
    return best_h, best_c5

def multi_scale_perturb(h, rng, n, scale_levels):
    hp = h.copy()
    for (bs, amp) in scale_levels:
        n_blocks = n // bs
        for _ in range(max(1, n_blocks // 4)):
            start = int(rng.randint(0, n - bs))
            amp_val = rng.uniform(-amp, amp)
            hp[start:start + bs] += amp_val
    hp = np.clip(hp, 0.0, 1.0)
    return project_to_sum(hp, n / 2.0)

def ilp_step(h, target_sum, rng, n, perturb_scales, cd_iters, slqp_iters, alpha):
    hp = multi_scale_perturb(h, rng, n, perturb_scales)
    hc, c5c = coordinate_descent(hp, target_sum, max_iters=cd_iters, step_frac=0.12)
    if slqp_iters > 0:
        hs = slsqp_local(hc, n, target_sum, alpha, maxiter=slqp_iters)
        c5s = compute_c5(hs)
        if c5s < c5c:
            hc, c5c = hs, c5s
    return hc, c5c

def genetic_algorithm(h_init, target_sum, rng, n, pop_size=6, n_gen=30, t_start=0.0, budget_s=1300):
    ts = target_sum
    population = []
    population.append((h_init.copy(), compute_c5(h_init)))
    for i in range(pop_size - 1):
        noise = 0.04 + 0.06 * (i / float(max(pop_size - 1, 1)))
        hm = h_init + rng.uniform(-noise, noise, n)
        hm = project_to_sum(hm, ts)
        hc, c5c = coordinate_descent(hm, ts, max_iters=1500, step_frac=0.12)
        population.append((hc, c5c))
    best_h = min(population, key=lambda x: x[1])[0].copy()
    best_c5 = min(population, key=lambda x: x[1])[1]
    for gen in range(n_gen):
        elapsed = time.time() - t_start
        if elapsed > budget_s - 80:
            break
        new_inds = []
        for ci in range(pop_size // 2):
            idx_a = int(rng.randint(len(population)))
            idx_b = int(rng.randint(len(population)))
            if idx_a == idx_b:
                idx_b = (idx_b + 1) % len(population)
            mask = rng.random(n) < 0.5
            child = np.where(mask, population[idx_a][0], population[idx_b][0])
            child = project_to_sum(child, ts)
            child = multi_scale_perturb(child, rng, n, [(5, 0.08), (20, 0.04)])
            hc, c5c = coordinate_descent(child, ts, max_iters=2000, step_frac=0.12)
            if gen % 4 == 0:
                hs = slsqp_local(hc, n, ts, 800, maxiter=300)
                c5s = compute_c5(hs)
                if c5s < c5c:
                    hc, c5c = hs, c5s
            new_inds.append((hc, c5c))
        population.sort(key=lambda x: x[1])
        new_pop = [population[0]]
        for ind in new_inds:
            new_pop.append(ind)
        for ind in population[1:]:
            if len(new_pop) >= pop_size:
                break
            new_pop.append(ind)
        population = new_pop[:pop_size]
        gb = min(population, key=lambda x: x[1])
        if gb[1] < best_c5:
            best_c5 = gb[1]
            best_h = gb[0].copy()
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Quick SLSQP polish
    for alpha in [200, 1000, 5000]:
        if time_left() < 60:
            break
        h_new = slsqp_local(best_h, n, ts, alpha, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, ts, max_iters=5000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Simulated annealing (multiple rounds with different parameters)
    sa_configs = [
        (25000, 0.05, 0.0005),
        (25000, 0.08, 0.001),
        (20000, 0.1, 0.002),
        (20000, 0.04, 0.0003),
    ]
    for (n_iter, t_i, t_f) in sa_configs:
        if time_left() < 120:
            break
        h_sa, c5_sa = simulated_annealing(best_h, ts, rng, n_iter=n_iter, t_init=t_i, t_final=t_f)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        if time_left() < 60:
            break
        h_pol = slsqp_local(best_h, n, ts, 3000, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_cd, c5_cd = coordinate_descent(best_h, ts, max_iters=3000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Iterated local search with adaptive perturbation
    if time_left() > 150:
        n_ils = 0
        max_ils = 25
        perturb_amp = 0.06
        for ils_iter in range(max_ils):
            if time_left() < 60:
                break
            scales = [(5, perturb_amp * 0.8), (15, perturb_amp * 0.5), (40, perturb_amp * 0.3)]
            hp = multi_scale_perturb(best_h, rng, n, scales)
            hc, c5c = coordinate_descent(hp, ts, max_iters=2500, step_frac=0.12)
            if time_left() > 40:
                hs = slsqp_local(hc, n, ts, 2000, maxiter=350)
                c5s = compute_c5(hs)
                if c5s < c5c:
                    hc, c5c = hs, c5s
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
                perturb_amp = max(0.03, perturb_amp * 0.85)
            else:
                perturb_amp = min(0.2, perturb_amp * 1.15)
            n_ils += 1

    # Phase 4: Genetic algorithm
    if time_left() > 200:
        h_ga, c5_ga = genetic_algorithm(best_h, ts, rng, n,
                                         pop_size=6, n_gen=20,
                                         t_start=t_start, budget_s=budget_s)
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga.copy()
        h_pol = slsqp_local(best_h, n, ts, 5000, maxiter=500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Multi-resolution
    if time_left() > 120:
        for n_r in [n, 480, 600]:
            if time_left() < 60:
                break
            if n_r == n:
                continue
            ts_r = n_r / 2.0
            h_r = interp_to_n(best_h, n_r)
            h_r = project_to_sum(h_r, ts_r)
            h_r, c5_r = coordinate_descent(h_r, ts_r, max_iters=3000, step_frac=0.1)
            if time_left() > 40:
                h_r2 = slsqp_local(h_r, n_r, ts_r, 2000, maxiter=400)
                c5_r2 = compute_c5(h_r2)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_r

    # Phase 6: More SA rounds if time allows
    if time_left() > 100:
        for (n_iter, t_i, t_f) in [(15000, 0.03, 0.0002), (15000, 0.06, 0.001)]:
            if time_left() < 60:
                break
            h_sa, c5_sa = simulated_annealing(best_h, ts, rng, n_iter=n_iter, t_init=t_i, t_final=t_f)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
            h_cd, c5_cd = coordinate_descent(best_h, ts, max_iters=3000, step_frac=0.08)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 7: Final intensive polishing
    for alpha in [5000, 20000, 80000]:
        if time_left() < 30:
            break
        h_pol = slsqp_local(best_h, n, ts, alpha, maxiter=600)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 20:
        h_cd, c5_cd = coordinate_descent(best_h, ts, max_iters=20000, step_frac=0.05)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 15:
        h_final = slsqp_local(best_h, n, ts, 100000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
