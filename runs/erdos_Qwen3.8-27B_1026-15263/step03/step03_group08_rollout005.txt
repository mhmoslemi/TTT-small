def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = np.sum(mask)

def lag_gradient(h, l):

def top_lags_gradient(h, m, extra_margin=0.0):

            break

        v = beta * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)

            v *= 0.3
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5

def slsqq_smooth(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sh = np.clip(alpha * (corr - m), -500.0, 500.0)
        ev = np.exp(sh)
        s = np.sum(ev)
        return m + np.log(s) / alpha
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sh = np.clip(alpha * (corr - m), -500.0, 500.0)
        ev = np.exp(sh)
        s = np.sum(ev)
        w = ev / s
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            l = k - (n - 1)
            g += wk * lag_gradient(h, l)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
