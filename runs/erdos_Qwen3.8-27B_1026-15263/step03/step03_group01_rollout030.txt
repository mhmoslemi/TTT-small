import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha

    # corr[k] = sum_i h[i] * (1-h[i-k+n-1]) for valid i
    # d corr[k] / d h[j] = (1-h[j-k+n-1]) * [valid] - h[j+k-n+1] * [valid]
    # So grad[j] = sum_k w[k] * [(1-h[j-k+n-1]) * valid1 - h[j+k-n+1] * valid2]

    indices = np.arange(n_corr)
    # For corr[k] = sum_i h[i] * (1-h[i-(k-n+1)])

    #          j appears in second factor when i-(k-n+1)=j, i.e., i=j+k-n+1
    #          contributing -h[j+k-n+1]

    grad = np.sum(valid1 * (1.0 - h[inner1_clipped]) * w[None, :] - 
                  valid2 * h[inner2_clipped] * w[None, :], axis=1)
    return obj, grad

def weighted_obj_grad(h, w):
    """Objective with explicit weights on lags. w has length 2n-1."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = float(np.sum(w * corr))
    
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    inner1 = j_idx - k_idx + n - 1

    inner1_clipped = np.clip(inner1, 0, n-1)
    inner2 = j_idx + k_idx - n + 1
    valid2 = (inner2 >= 0) & (inner2 < n)

    grad = np.sum(valid1 * (1.0 - h[inner1_clipped]) * w[None, :] -

    return obj, grad

def slsqp_opt(h0, n, ts, objective_fn, maxiter=1000):
    def f(x):
        return objective_fn(x)[0]
    def g(x):
        return objective_fn(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_opt(h0, n, ts, objective_fn, n_iter=5000, lr_init=0.05):
    h = h0.copy()
    lr = lr_init

    stall = 0

        _, grad = objective_fn(h)

        obj_new, _ = objective_fn(h_new)

            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.5
            stall += 1

                break
    return h

One key insight: for the zero lag (k = n-1 in the full correlate), the value is sum(h[i] * (1-h[i])) = sum(h) - sum(h^2) = n/2 - sum(h^2). To minimize this, we want to maximize sum(h^2), i.e., make h as binary as possible.

But for other lags, we want h and its shifted version to be complementary.

The optimal solution likely has a structure where h is close to 0 or 1 (binary-like) but with smooth transitions to handle the trade-off between different lags.

Let me think about a **binary search on the target** approach more carefully:

For a target t, we want: min over h of max_k f_k(h) ≤ t
Equivalent: find h such that f_k(h) ≤ t for all k

We can use the feasible direction approach:
- Start with current h
- Compute f_k(h) for all k
- Find the "worst" lag k*
- Try to reduce f_{k*}(h) while not increasing others too much
- This is like a cutting plane / active set method

Actually, let me try a different approach: **subgradient method with step-size rule**.

The subgradient of g(h) = max_k f_k(h) at h is:
∇g(h) = sum_{k in K*} w_k * ∇f_k(h)
where K* = {k : f_k(h) = g(h)} and w_k ≥ 0, sum w_k = 1.

A natural choice is w_k = 1/|K*| for k in K*.

Then we do:
h_{t+1} = project(h_t - eta_t * ∇g(h_t))

With eta_t = alpha / ||∇g(h_t)||^2 (standard subgradient step size).

This is guaranteed to converge (sublinearly) for convex problems, but our problem is non-convex. Still, it might work well in practice.

Let me also think about the **structure** of the optimal solution. 

For the minimum overlap problem on [0,2] with ∫h = 1:
- If we consider h(x) = 1 for x in [0,1] and h(x) = 0 for x in (1,2], then:
  - Zero lag: 1 - 1 = 0 (since h is 0-1)
  - Lag 1 (shift by 1): h(x)(1-h(x+1)) = 0 for x in [0,1] since h(x+1)=0
  - Lag -1: h(x)(1-h(x-1)) = 0 for x in (1,2] since h(x)=0... wait

Actually for the discrete case with n points:
- h = [1, 1, ..., 1, 0, 0, ..., 0] with n/2 ones and n/2 zeros
- Zero lag (k=n-1): sum h[i]*(1-h[i]) = 0
- Lag +1 (shift by 1 position): sum h[i]*(1-h[i+1]) - this is the boundary
- Lag n/2 - 1 (shift by n/2 positions = 1 unit): completely complementary!

So the issue is the "intermediate" lags. The shift by 1 position and shift by n/2-1 positions will have some overlap.

For the binary step function, the maximum overlap is at small lags (shifts of 1, 2, 3, ... positions), where the boundary between 1s and 0s creates partial overlap.

To reduce this, we need to "spread out" the transition, which is what the continuous optimization does.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    inner1 = j_idx - k_idx + n - 1
    valid1 = (inner1 >= 0) & (inner1 < n)
    inner1c = np.clip(inner1, 0, n-1)
    inner2 = j_idx + k_idx - n + 1
    valid2 = (inner2 >= 0) & (inner2 < n)
    inner2c = np.clip(inner2, 0, n-1)
    
    grad = np.sum(valid1 * (1.0 - h[inner1c]) * w[None, :] - 
                  valid2 * h[inner2c] * w[None, :], axis=1)
    return obj, grad

def weighted_obj_grad(h, w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = float(np.sum(w * corr))
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    inner1 = j_idx - k_idx + n - 1
    valid1 = (inner1 >= 0) & (inner1 < n)
    inner1c = np.clip(inner1, 0, n-1)
    inner2 = j_idx + k_idx - n + 1
    valid2 = (inner2 >= 0) & (inner2 < n)
    inner2c = np.clip(inner2, 0, n-1)
    grad = np.sum(valid1 * (1.0 - h[inner1c]) * w[None, :] - 
                  valid2 * h[inner2c] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, obj_fn, maxiter=1000):
    def f(x):
        return obj_fn(x)[0]
    def g(x):
        return obj_fn(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_opt(h0, n, ts, obj_fn, n_iter=5000, lr_init=0.03):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = obj_fn(h)
    stall = 0
    for i in range(n_iter):
        _, grad = obj_fn(h)
        # Adaptive step size (Barzilai-Borwein-like)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = obj_fn(h_new)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 500 or lr < 1e-14:
                break
    return h

def subgradient_opt(h0, n, ts, n_iter=5000, alpha0=0.01):
    """Subgradient method for the actual max objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        # Active set
        tol = 1e-8
        active = corr > m_val - tol
        n_active = np.sum(active)
        if n_active == 0:
            active = np.zeros(n_corr, dtype=bool)
            active[np.argmax(corr)] = True
            n_active = 1
        w = np.zeros(n_corr)
        w[active] = 1.0 / n_active
        
        _, grad = weighted_obj_grad(h, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        eta = alpha0 / (it + 1)
        h_new = project_to_sum(h - eta * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still move, subgradient method doesn't require descent
    
    return best_h, best_c5

def iterative_reweighting(h0, n, ts, n_iter=50, eta=10.0, inner_iters=300):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    w = np.ones(n_corr) / n_corr

        def obj_fn(x):
            return weighted_obj_grad(x, w)
        h = proj_grad_opt(h, n, ts, obj_fn, n_iter=inner_iters, lr_init=0.02)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Update weights
        overlap_vals = np.correlate(h, 1.0 - h, mode="full")
        t_cur = np.max(overlap_vals)
        if t_cur < 1e-10:
            break
        w *= np.exp(eta * overlap_vals / t_cur)
        w /= np.sum(w)
    
    return best_h, best_c5

    n_old = len(h_old)

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    inner1 = j_idx - k_idx + n - 1
    valid1 = (inner1 >= 0) & (inner1 < n)
    inner1c = np.clip(inner1, 0, n - 1)
    inner2 = j_idx + k_idx - n + 1
    valid2 = (inner2 >= 0) & (inner2 < n)
    inner2c = np.clip(inner2, 0, n - 1)
    grad = np.sum(valid1 * (1.0 - h[inner1c]) * w[None, :] - valid2 * h[inner2c] * w[None, :], axis=1)
    return obj, grad

def weighted_obj_grad(h, w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = float(np.dot(w, corr))
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    inner1 = j_idx - k_idx + n - 1
    valid1 = (inner1 >= 0) & (inner1 < n)
    inner1c = np.clip(inner1, 0, n - 1)
    inner2 = j_idx + k_idx - n + 1
    valid2 = (inner2 >= 0) & (inner2 < n)
    inner2c = np.clip(inner2, 0, n - 1)
    grad = np.sum(valid1 * (1.0 - h[inner1c]) * w[None, :] - valid2 * h[inner2c] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, obj_fn, maxiter=1000):
    def f(x):
        return obj_fn(x)[0]
    def g(x):
        return obj_fn(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_opt(h0, n, ts, obj_fn, n_iter=5000, lr_init=0.03):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = obj_fn(h)
    stall = 0
    for i in range(n_iter):
        _, grad = obj_fn(h)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = obj_fn(h_new)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 500 or lr < 1e-14:
                break
    return h

def subgradient_opt(h0, n, ts, n_iter=5000, alpha0=0.005):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        tol = max(1e-8, 1e-6 * m_val)
        active = corr > m_val - tol
        n_active = np.sum(active)
        if n_active == 0:
            active = np.zeros(n_corr, dtype=bool)
            active[int(np.argmax(corr))] = True
            n_active = 1
        w = np.zeros(n_corr)
        w[active] = 1.0 / n_active
        _, grad = weighted_obj_grad(h, w)
        eta = alpha0 / (1.0 + 0.01 * it)
        h_new = project_to_sum(h - eta * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def iterative_reweighting(h0, n, ts, n_iter=40, eta=12.0, inner_iters=200):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    w = np.ones(n_corr) / n_corr
    for it in range(n_iter):
        def obj_fn(x):
            return weighted_obj_grad(x, w)
        h = proj_grad_opt(h, n, ts, obj_fn, n_iter=inner_iters, lr_init=0.02)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        overlap_vals = np.correlate(h, 1.0 - h, mode="full")
        t_cur = np.max(overlap_vals)
        if t_cur < 1e-10:
            break
        w *= np.exp(eta * overlap_vals / t_cur)
        w /= np.sum(w)
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def grad_guided_stochastic(h, n, ts, rng, n_steps=60000, alpha=20000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    temp = 0.3
    for step in range(n_steps):
        temp = 0.3 * np.exp(-step / (n_steps * 0.7))
        _, grad = smooth_max_obj_grad(h, alpha)
        r = rng.random()
        if r < 0.5:
            pos_mask = grad > 1e-8
            neg_mask = grad < -1e-8
            if not np.any(pos_mask) or not np.any(neg_mask):
                continue
            i = int(rng.choice(np.where(pos_mask)[0]))
            j = int(rng.choice(np.where(neg_mask)[0]))
            scale = abs(grad[i] - grad[j]) / max(np.max(np.abs(grad)), 1e-10)
            delta = rng.uniform(0.02, 0.2) * min(1.0, scale + 0.3)
            delta = min(delta, h[i] * 0.8, (1.0 - h[j]) * 0.8)
            if delta < 1e-5:
                continue
            h[i] -= delta
            h[j] += delta
        elif r < 0.8:
            blk = int(rng.randint(3, max(4, n // 15)))
            pos = int(rng.randint(0, n - blk))
            bg = grad[pos:pos + blk]
            noise = rng.randn(blk) * 0.04 - 0.015 * bg
            h[pos:pos + blk] = np.clip(h[pos:pos + blk] + noise, 0.0, 1.0)
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta = rng.uniform(0.01, 0.2)
            delta = min(delta, h[i] * 0.7, (1.0 - h[j]) * 0.7)
            if delta < 1e-5:
                continue
            h[i] -= delta
            h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(temp, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0

    # Phase 1: Sharp smooth max SLSQP at current resolution
    for alpha in [3000, 15000, 60000]:
        if time_left() < 50:
            break
        def obj_fn_a(x, a=alpha):
            return smooth_max_obj_grad(x, a)
        h_new = slsqp_opt(best_h, n1, ts1, obj_fn_a, maxiter=2500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Projected gradient with very sharp max
    if time_left() > 60:
        def obj_fn_b(x):
            return smooth_max_obj_grad(x, 100000)
        h_pg = proj_grad_opt(best_h, n1, ts1, obj_fn_b, n_iter=8000, lr_init=0.01)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Iterative reweighting
    if time_left() > 150:
        h_rw, c5_rw = iterative_reweighting(best_h, n1, ts1, n_iter=35, eta=12.0, inner_iters=150)
        if c5_rw < best_c5:
            best_c5 = c5_rw
            best_h = h_rw.copy()

    # Phase 4: Subgradient on exact max
    if time_left() > 100:
        h_sg, c5_sg = subgradient_opt(best_h, n1, ts1, n_iter=6000, alpha0=0.004)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Gradient-guided stochastic
    if time_left() > 130:
        h_gs, c5_gs = grad_guided_stochastic(best_h, n1, ts1, rng, n_steps=70000, alpha=30000)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()

    # Phase 6: Polish after stochastic
    if time_left() > 50:
        def obj_fn_c(x):
            return smooth_max_obj_grad(x, 150000)
        h_pol = proj_grad_opt(best_h, n1, ts1, obj_fn_c, n_iter=5000, lr_init=0.008)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Try higher resolution
    for n_tr in [600, 720]:
        if time_left() < 130:
            break
        ts_tr = n_tr / 2.0
        h_up = upsample(best_h, n_tr)
        h_up = project_to_sum(h_up, ts_tr)
        for alpha in [5000, 30000]:
            if time_left() < 50:
                break
            def obj_fn_d(x, a=alpha):
                return smooth_max_obj_grad(x, a)
            h_opt = slsqp_opt(h_up, n_tr, ts_tr, obj_fn_d, maxiter=1500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_up = h_opt
        if time_left() > 50:
            def obj_fn_e(x):
                return smooth_max_obj_grad(x, 80000)
            h_pg2 = proj_grad_opt(h_up, n_tr, ts_tr, obj_fn_e, n_iter=4000, lr_init=0.008)
            c5_pg2 = compute_c5(h_pg2)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
                best_n = n_tr

    # Phase 8: Second reweighting at best resolution
    if time_left() > 100:
        n_b = best_n
        ts_b = n_b / 2.0
        h_rw2, c5_rw2 = iterative_reweighting(best_h, n_b, ts_b, n_iter=25, eta=15.0, inner_iters=100)
        if c5_rw2 < best_c5:
            best_c5 = c5_rw2
            best_h = h_rw2.copy()

    # Phase 9: Final stochastic refinement
    if time_left() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        h_fs, c5_fs = grad_guided_stochastic(best_h, n_f, ts_f, rng, n_steps=35000, alpha=50000)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Phase 10: Ultimate polish
    if time_left() > 30:
        n_u = best_n
        ts_u = n_u / 2.0
        for alpha in [150000, 300000]:
            if time_left() < 20:
                break
            def obj_fn_f(x, a=alpha):
                return smooth_max_obj_grad(x, a)
            h_u = proj_grad_opt(best_h, n_u, ts_u, obj_fn_f, n_iter=4000, lr_init=0.004)
            c5_u = compute_c5(h_u)
            if c5_u < best_c5:
                best_c5 = c5_u
                best_h = h_u.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    for _ in range(1000):
        diff = target - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
