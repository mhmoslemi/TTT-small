   def compute_c5(h):
       n = len(h)

       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)
   
   def project_to_sum(h, target_sum):
       h = h.astype(np.float64).copy()
       lo = h.min() - 1.0
       hi = h.max() + 1.0
       for _ in range(80):
           mid = (lo + hi) * 0.5
           s = np.clip(h - mid, 0.0, 1.0).sum()
           if s > target_sum:
               lo = mid
           else:
               hi = mid
       result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
       # Final correction
       diff = target_sum - result.sum()
       if abs(diff) > 1e-14:
           mask = (result > 1e-10) & (result < 1.0 - 1e-10)
           if mask.sum() > 0:
               result[mask] += diff / mask.sum()
               result = np.clip(result, 0.0, 1.0)
       return result
   
   class GradContext:
       def __init__(self, n):
           self.n = n
           self.ij = np.arange(n)[:, None]
           self.lm = (np.arange(2*n-1)[None, :] - (n-1)).astype(np.int32)
           self.v1 = (self.ij - self.lm >= 0) & (self.ij - self.lm < n)
           self.idx1 = np.clip(self.ij - self.lm, 0, n-1)
           self.v2 = (self.ij + self.lm >= 0) & (self.ij + self.lm < n)
           self.idx2 = np.clip(self.ij + self.lm, 0, n-1)
           # Store as int32 for memory efficiency
           self.v1 = self.v1.astype(np.float32)
           self.v2 = self.v2.astype(np.float32)
           self.idx1 = self.idx1.astype(np.int32)
           self.idx2 = self.idx2.astype(np.int32)
       
       def smooth_max_grad(self, h, alpha, beta=0.0):
           n = self.n
           corr = np.correlate(h, 1.0 - h, mode="full")
           m = np.max(corr)
           shifted = alpha * (corr - m)
           exp_v = np.exp(shifted - np.max(shifted))
           w = exp_v / exp_v.sum()
           obj = m + np.log(exp_v.sum()) / alpha
           
           # Gradient of smooth max
           A = ((1.0 - h[self.idx1]) * self.v1 * w[None, :]).sum(axis=1)
           B = (h[self.idx2] * self.v2 * w[None, :]).sum(axis=1)
           grad = A - B
           
           # Binary penalty gradient
           if beta > 0:
               obj -= beta * np.sum(h * (1.0 - h)) / n
               grad -= beta * (1.0 - 2.0 * h) / n
           
           return obj, grad
