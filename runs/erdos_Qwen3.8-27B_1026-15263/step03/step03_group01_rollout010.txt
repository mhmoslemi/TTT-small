                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(np.clip(r.x, 0, 1), ts)
    except Exception:
        return project_to_sum(h0, ts)

def lbfgs_pen_opt(h0, n, ts, alpha, maxiter=3000, penalty=100.0):
    def f(x):
        obj, _ = softmax_obj_grad(x, alpha)
        return obj + penalty * (np.sum(x) - ts) ** 2
    def g(x):
        _, grad = softmax_obj_grad(x, alpha)
        return grad + 2.0 * penalty * (np.sum(x) - ts)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='L-BFGS-B', jac=g, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-13, 'maxls': 40})
        return project_to_sum(np.clip(r.x, 0, 1), ts)
    except Exception:
        return project_to_sum(h0, ts)

def subgrad_step(h, ts, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    tol = m_val * 0.01 + 1e-10
    active = np.where(corr >= m_val - tol)[0]
    nj = np.arange(n)[:, None]
    nm = active[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) - vs * h[sc], axis=1) / len(active)
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, ts)
    return h_new, grad

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def structured_perturb(h, rng, n_comp, amp):
    n = len(h)
    x = np.linspace(0, 2, n, endpoint=False)
    perturb = np.zeros(n)
    for _ in range(n_comp):
        f = rng.randint(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        perturb += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    h_new = h + perturb
    return project_to_sum(h_new, n / 2.0)

def stochastic_cd(h, n, ts, rng, n_steps=40000, delta_lo=0.01, delta_hi=0.25):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.25
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        ones_idx = np.where(h > 0.02)[0]
        zeros_idx = np.where(h < 0.98)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            acc = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < acc:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(5e-6, 0.25 * np.exp(-step / max(n_steps // 3, 1000)))
    return best_h, best_c5

def greedy_pair_search(h, n, ts, n_trials=20000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.15)
        if h[i] < delta or (1.0 - h[j]) < delta:
            continue
        h[i] -= delta
        h[j] += delta
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            c5_cur = c5_new
        else:
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n0 = best_n
    ts0 = n0 / 2.0

    # Phase 1: Continuation optimization at native resolution
    alphas1 = [50, 150, 500, 1500, 5000, 15000, 50000]
    for alpha in alphas1:
        if time_left() < 25:
            break
        h_new = slsqp_opt(best_h, n0, ts0, alpha, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: L-BFGS-B with penalty at high alpha
    for alpha in [5000, 20000, 80000]:
        if time_left() < 30:
            break
        h_new = lbfgs_pen_opt(best_h, n0, ts0, alpha, maxiter=4000, penalty=200.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Subgradient steps on exact objective
    if time_left() > 60:
        h_sg = best_h.copy()
        ts_sg = n0 / 2.0
        sg = 0.05
        for it in range(5000):
            if time_left() < 20:
                break
            h_sg, _ = subgrad_step(h_sg, ts_sg, sg)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            sg *= 0.995
            if sg < 1e-6:
                break

    # Phase 4: Basin hopping with structured perturbations
    for trial in range(80):
        if time_left() < 30:
            break
        amp = 0.04 * (1.0 - trial / 100.0) + 0.01
        if trial % 3 == 0:
            h_trial = freq_perturb(best_h, rng, max(5, n0 // 30), amp)
        elif trial % 3 == 1:
            h_trial = structured_perturb(best_h, rng, rng.randint(2, 6), amp)
        else:
            idx = rng.choice(n0, size=rng.randint(10, 50), replace=False)
            h_trial = best_h.copy()
            h_trial[idx] += rng.uniform(-amp, amp, size=len(idx))
            h_trial = project_to_sum(h_trial, n0 / 2.0)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        for alpha in [2000, 10000]:
            if time_left() < 15:
                break
            h_opt = slsqp_opt(h_trial, n0, n0 / 2.0, alpha, maxiter=500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_trial = h_opt

    # Phase 5: Try n=512 upsampling
    if time_left() > 120:
        n_up = 512
        ts_up = n_up / 2.0
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts_up)
        c5_up = compute_c5(h_up)
        current_best_n = n0
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
            current_best_n = n_up
        for alpha in [100, 1000, 5000, 20000]:
            if time_left() < 25:
                break
            h_opt = slsqp_opt(h_up, n_up, ts_up, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_up
                current_best_n = n_up
            h_up = h_opt
        if time_left() > 40:
            h_lb = lbfgs_pen_opt(h_up, n_up, ts_up, 20000, maxiter=3000, penalty=150.0)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
                best_n = n_up

    # Phase 6: Stochastic CD fine tuning
    n_cd = best_n
    ts_cd = n_cd / 2.0
    if time_left() > 100:
        h_cd, c5_cd = stochastic_cd(best_h, n_cd, ts_cd, rng, n_steps=60000, delta_lo=0.005, delta_hi=0.2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: SLSQP polish after stochastic
    if time_left() > 50:
        for alpha in [10000, 50000, 200000]:
            if time_left() < 20:
                break
            h_pol = slsqp_opt(best_h, n_cd, ts_cd, alpha, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 8: Greedy pair search
    if time_left() > 60:
        h_gp, c5_gp = greedy_pair_search(best_h, n_cd, ts_cd, n_trials=30000, rng=rng)
        if c5_gp < best_c5:
            best_c5 = c5_gp
            best_h = h_gp.copy()

    # Phase 9: Final L-BFGS-B polish
    if time_left() > 40:
        h_fin = lbfgs_pen_opt(best_h, n_cd, ts_cd, 50000, maxiter=5000, penalty=300.0)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Tiny steps
    if time_left() > 30:
        h_tiny, c5_tiny = stochastic_cd(best_h, n_cd, ts_cd, rng, n_steps=30000, delta_lo=0.002, delta_hi=0.08)
        if c5_tiny < best_c5:
            best_c5 = c5_tiny
            best_h = h_tiny.copy()
        if time_left() > 20:
            h_last = slsqp_opt(best_h, n_cd, ts_cd, 100000, maxiter=2000)
            c5_last = compute_c5(h_last)
            if c5_last < best_c5:
                best_c5 = c5_last
                best_h = h_last.copy()

    # Final validation and cleanup
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
