  But h[k:] has length n-k. And g[:n-k] also has length n-k. ✓

Let me verify: n=4, k=1
g[1:] += (1-h[:3]) → g[1] += 1-h[0], g[2] += 1-h[1], g[3] += 1-h[2]
g[:3] -= h[1:] → g[0] -= h[1], g[1] -= h[2], g[2] -= h[3]

So: g[0] = -h[1], g[1] = 1-h[0]-h[2], g[2] = 1-h[1]-h[3], g[3] = 1-h[2]

Check with manual: c_1 = h[1](1-h[0]) + h[2](1-h[1]) + h[3](1-h[2])
∂/∂h[0] = -h[1] ✓
∂/∂h[1] = (1-h[0]) - h[2] ✓
∂/∂h[2] = (1-h[1]) - h[3] ✓
∂/∂h[3] = (1-h[2]) ✓

For k<0, let m=-k>0:
c_{-m} = Σ_{i=0}^{n-1-m} h[i](1-h[i+m])
∂/∂h[j] = (1-h[j+m]) for j in [0, n-1-m] and -h[j-m] for j in [m, n-1]

In numpy:
g[:n-m] += (1-h[m:])  # h[m:] has length n-m, g[:n-m] has length n-m ✓
g[m:] -= h[:n-m]      # h[:n-m] has length n-m, g[m:] has length n-m ✓

Verify: n=4, m=1 (k=-1)
g[:3] += (1-h[1:]) → g[0]+=1-h[1], g[1]+=1-h[2], g[2]+=1-h[3]
g[1:] -= h[:3] → g[1]-=h[0], g[2]-=h[1], g[3]-=h[2]

So: g[0]=1-h[1], g[1]=1-h[2]-h[0], g[2]=1-h[3]-h[1], g[3]=-h[2]

Check: c_{-1} = h[0](1-h[1]) + h[1](1-h[2]) + h[2](1-h[3])
∂/∂h[0] = (1-h[1]) ✓
∂/∂h[1] = -h[0] + (1-h[2]) ✓
∂/∂h[2] = -h[1] + (1-h[3]) ✓
∂/∂h[3] = -h[2] ✓

Now for the main algorithm. Let me think about the step size for the subgradient method.

The gradient of c_k has magnitude about O(n-k) (sum of about n-k terms each of magnitude ≤1). So for large n, the gradient is large. The step size should scale as O(1/n) or O(1/√n).

Actually, since we're minimizing c_k (we want to reduce the max), and the gradient tells us the direction of increase, we want to step in the -gradient direction. But we need to be careful: reducing c_{k*} might increase c_k for other k.

A simple approach: use the negative gradient of the worst lag with a small step size, and check if the overall max decreased. If not, try a smaller step.

For better convergence, I'll use:
- Top M lags (e.g., M=5)
- Average their gradients (weighted by how close they are to the max)
- Adaptive step size with backtracking

Let me now write the complete code:

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5_val(h):
    n = len(h)
    corr = compute_corr(h)
    return float(np.max(corr) * 2.0 / n)

def project_sum(h, ts):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = ts - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0, 1)
    return h

def grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        m = -k
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def top_lag_grad(h, n, M=5):
    corr = compute_corr(h)
    # Center of full correlation is at index n-1
    # Lag k corresponds to index n-1+k
    idxs = np.argsort(corr)[-M:]
    lags = idxs - (n - 1)
    weights = np.exp((corr[idxs] - corr.max()) * 10)
    weights /= weights.sum()
    total_grad = np.zeros(n)
    for w, lag in zip(weights, lags):
        total_grad += w * grad_at_lag(h, int(lag))
    return total_grad, corr, lags

def active_lag_descend(h, n, ts, n_iter, time_limit, t_start, M=5):
    h = h.copy()
    best_local = compute_c5_val(h)
    lr = 1.0 / (n * 0.5)
    stall = 0
    for i in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        grad, corr, _ = top_lag_grad(h, n, M)
        h_new = project_sum(h - lr * grad, ts)
        c5_new = compute_c5_val(h_new)
        if c5_new < best_local - 1e-15:
            h = h_new
            best_local = c5_new
            stall = 0
            lr = min(lr * 1.1, 2.0 / n)
        else:
            lr *= 0.7
            stall += 1
            if stall > 100 or lr < 1e-8:
                break
    return h, best_local

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_v = np.exp(shifted - np.max(shifted))
    w = exp_v / exp_v.sum()
    smooth_val = (m_val + np.log(exp_v.sum()) + np.max(shifted)) / alpha
    total_grad = np.zeros(n)
    nz = np.nonzero(w > 1e-15)[0]
    for idx in nz:
        lag = int(idx - (n - 1))
        total_grad += w[idx] * grad_at_lag(h, lag)
    return smooth_val, total_grad

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [
        {
            'type': 'eq',
            'fun': lambda x: np.sum(x) - ts,
            'jac': lambda x: np.ones(n),
        }
    ]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method="SLSQP", jac=g, bounds=bnds,
                     constraints=cons, options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-12})
        return project_sum(r.x, ts)
    except Exception:
        return project_sum(h0, ts)

def stochastic_cd(h, n, ts, rng, n_steps, t_start, time_limit, delta_lo=0.01, delta_hi=0.3):
    h = h.copy()
    c5_cur = compute_c5_val(h)
    temp = 0.2
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        oi = np.where(h > 0.05)[0]
        zi = np.where(h < 0.95)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = rng.choice(oi)
        j = rng.choice(zi)
        if i == j:
            continue
        d = rng.uniform(delta_lo, delta_hi)
        if h[i] < d:
            d = h[i] * 0.5
        if 1 - h[j] < d:
            d = (1 - h[j]) * 0.5
        if d < 1e-5:
            continue
        h[i] -= d
        h[j] += d
        h = np.clip(h, 0, 1)
        h = project_sum(h, ts)
        c5_new = compute_c5_val(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            if rng.random() < np.exp(-(c5_new - c5_cur) / temp):
                c5_cur = c5_new
            else:
                h[i] += d
                h[j] -= d
                h = np.clip(h, 0, 1)
                h = project_sum(h, ts)
        temp = max(1e-5, 0.2 * np.exp(-step / 5000.0))
    return h, c5_cur

def upsample(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    return h_up

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mn = np.mean(h)
    H = np.fft.rfft(h - mn)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mn
    return project_sum(h_new, n // 2.0)
