def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def single_lag_gradient(h, L):
    n = len(h)
    g = np.zeros(n)
    k = abs(int(L))
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

def get_top_lags(corr, n_lags):
    n = (len(corr) - 1) // 2 + 1
    idx = np.argsort(corr)[-n_lags:]
    lags = idx - (n - 1)
    return lags

def active_set_smooth_max(h, active_lags, alpha):
    n = len(h)
    corr_all = compute_corr(h)
    active_idx = active_lags.astype(int) + (n - 1)
    vals = corr_all[active_idx]
    m_val = np.max(vals)
    shifted = alpha * (vals - m_val)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    obj = m_val + np.log(np.sum(exps)) / alpha
    grad = np.zeros(n)
    for i in range(len(active_lags)):
        if w[i] < 1e-14:
            continue
        grad += w[i] * single_lag_gradient(h, active_lags[i])
    return obj, grad, corr_all

def full_smooth_max_obj(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    obj = m_val + np.log(np.sum(exps)) / alpha
    return obj, w

def gradient_from_top_weighted(h, w, top_k=15):
    n = len(h)
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        if w[m] < 1e-12:
            continue
        lag = m - (n - 1)
        grad += w[m] * single_lag_gradient(h, lag)
    return grad

def _slsqp_obj(x):
    obj, w = full_smooth_max_obj(x, _slsqp_alpha)
    return obj

def _slsqp_grad(x):
    obj, w = full_smooth_max_obj(x, _slsqp_alpha)
    return gradient_from_top_weighted(x, w, _slsqp_top_k)

    return np.sum(x) - _slsqp_target_sum

def slsqp_optimize(h0, alpha, top_k, maxiter):

    _slsqp_alpha = alpha
    _slsqp_top_k = top_k
    _slsqp_n = len(h0)
    _slsqp_target_sum = _slsqp_n / 2.0
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]

def smooth_perturbation(h, rng, noise_level, bandwidth=10):
    n = len(h)
    noise = rng.normal(0, noise_level, n)
    kernel = np.zeros(n)
    for i in range(n):
        d = min(i, n - i, abs(i - n // 2))
        if d < bandwidth:
            kernel[i] = math.exp(-0.5 * (d / (bandwidth / 3.0)) ** 2)
    from numpy.fft import rfft, irfft
    noise_f = rfft(noise)
    kernel_f = rfft(kernel)
    smooth = irfft(noise_f * kernel_f, n)
    h_pert = h + smooth
    return project_to_sum(h_pert, n / 2.0)

def block_perturbation(h, rng, n_blocks=5):
    n = len(h)
    h_pert = h.copy()
    block_size = n // n_blocks
    for _ in range(3):
        b1 = rng.randint(0, n_blocks)
        b2 = rng.randint(0, n_blocks)
        if b1 == b2:
            continue
        s1 = b1 * block_size
        s2 = b2 * block_size
        e1 = s1 + block_size
        e2 = s2 + block_size
        h_pert[s1:e1], h_pert[s2:e2] = h_pert[s2:e2].copy(), h_pert[s1:e1].copy()
    return project_to_sum(h_pert, n / 2.0)
