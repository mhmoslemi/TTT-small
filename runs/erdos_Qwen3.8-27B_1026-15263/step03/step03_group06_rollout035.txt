import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(mask)
        if n_free > 0 and abs(diff) <= n_free * 0.49:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_set(corr, T):
    """Get indices of top T correlation values."""
    return np.argsort(corr)[::-1][:T]

def smooth_max_active(corr, active_idx, alpha):
    """Smooth max over active indices only."""
    c = corr[active_idx]
    m = np.max(c)
    shifted = np.clip(alpha * (c - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    return sm_val, w

def grad_active(h, active_idx, w, n):
    """Gradient of active-set smooth max."""
    g = np.zeros(n)
    for t, k in enumerate(active_idx):
        wt = w[t]
        if wt < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wt * (1.0 - h[:n-m])
            g[:n-m] += -wt * h[m:]
        else:
            q = -m
            g[:n-q] += wt * (1.0 - h[q:])
            g[q:] += -wt * h[:n-q]
    return g

def slsqp_active(h0, n, target_sum, T, alpha, maxiter):
    corr0 = compute_corr(h0)
    active_idx = get_active_set(corr0, T)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        v, _ = smooth_max_active(c, active_idx, alpha)
        return v
    
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        _, w = smooth_max_active(c, active_idx, alpha)
        return grad_active(ht, active_idx, w, n)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except Exception:
        pass
    
    return best_h, best_c5

def nesterov_active(h0, n, target_sum, T, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h0.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    c5_cur = c5_best
    stall_count = 0
    
    for it in range(n_iters):
        corr_y = compute_corr(y)
        active_idx = get_active_set(corr_y, T)
        _, w = smooth_max_active(corr_y, active_idx, alpha)
        g = grad_active(y, active_idx, w, n)
        
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new > c5_cur * 1.005 + 1e-6:
            lr *= 0.5
            stall_count += 1
            if stall_count > 50:
                break
            continue
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mu = (t_val - 1.0) / t_new
        y = h_new + mu * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = c5_new
        stall_count = 0
        
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.02, lr_init * 2)
        else:
            lr *= 0.95
    
    return h_best, c5_best

def coord_descent(h0, n, target_sum, n_rounds, rng, T=5):
    """Coordinate descent with active set awareness."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(n_rounds):
        improved = False
        corr = compute_corr(h)
        top_k = np.argmax(corr)
        
        # Focus on indices that affect the top lag most
        lag = top_k - (n - 1)
        if lag >= 0:
            hot = np.concatenate([np.arange(max(0, lag-20), min(n, lag+20)),
                                  np.arange(max(0, -20), min(n, n-lag+20))])
        else:
            q = -lag
            hot = np.concatenate([np.arange(0, min(n, n-q+20)),
                                  np.arange(max(0, q-20), n)])
        hot = np.unique(hot)
        
        idx_order = hot[rng.permutation(len(hot))]
        
        for i in idx_order:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-6:
                continue
            
            # Try a few candidate values
            candidates = np.linspace(lo, hi, 7)
            best_val = c5_cur
            best_cand = h[i]
            for v in candidates:
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val - 1e-12:
                    best_val = c5_tmp
                    best_cand = v
            
            if best_cand != h[i]:
                h[i] = best_cand
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(mask)
        if n_free > 0 and abs(diff) <= n_free * 0.49:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_top_indices(corr, T):
    return np.argsort(corr)[::-1][:T]

def smooth_max_active(corr, active_idx, alpha):
    c = corr[active_idx]
    m = np.max(c)
    shifted = np.clip(alpha * (c - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    return sm_val, w

def grad_active_vec(h, active_idx, w, n):
    g = np.zeros(n)
    for t, k in enumerate(active_idx):
        wt = w[t]
        if wt < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wt * (1.0 - h[:n - m])
            g[:n - m] -= wt * h[m:]
        else:
            q = -m
            g[:n - q] += wt * (1.0 - h[q:])
            g[q:] -= wt * h[:n - q]
    return g

def nesterov_active(h0, n, target_sum, T, alpha, n_iters, lr_init, time_limit=None, t_start=None):
    h = h0.copy()
    y = h0.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    c5_cur = c5_best
    stall = 0
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        corr_y = compute_corr(y)
        active_idx = get_top_indices(corr_y, T)
        _, w = smooth_max_active(corr_y, active_idx, alpha)
        g = grad_active_vec(y, active_idx, w, n)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_cur * 1.002 + 1e-5:
            lr *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mu = (t_val - 1.0) / t_new
        y = h_new + mu * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = c5_new
        stall = 0
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.015, lr_init * 3.0)
        else:
            lr *= 0.97
    return h_best, c5_best

def slsqp_active(h0, n, target_sum, T, alpha, maxiter):
    corr0 = compute_corr(h0)
    active_idx = get_top_indices(corr0, T)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        v, _ = smooth_max_active(c, active_idx, alpha)
        return v

    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        _, w = smooth_max_active(c, active_idx, alpha)
        return grad_active_vec(ht, active_idx, w, n)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except Exception:
        pass
    return best_h, best_c5

def slsqp_full(h0, n, target_sum, alpha, maxiter):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        m = np.max(c)
        shifted = np.clip(alpha * (c - m), -500.0, 500.0)
        ev = np.exp(shifted)
        s = ev.sum()
        return m + np.log(s / len(c)) / alpha

    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        c = compute_corr(ht)
        m = np.max(c)
        shifted = np.clip(alpha * (c - m), -500.0, 500.0)
        ev = np.exp(shifted)
        s = ev.sum()
        w = ev / s
        return grad_active_vec(ht, np.arange(len(c)), w, n)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except Exception:
        pass
    return best_h, best_c5

def coord_descent_smart(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        corr = compute_corr(h)
        top_k = np.argmax(corr)
        lag = top_k - (n - 1)
        span = max(20, n // 20)
        if lag >= 0:
            hot_a = np.arange(max(0, lag - span), min(n, lag + span))
            hot_b = np.arange(0, min(n, n - lag + span))
        else:
            q = -lag
            hot_a = np.arange(0, min(n, n - q + span))
            hot_b = np.arange(max(0, q - span), n)
        hot = np.unique(np.concatenate([hot_a, hot_b]))
        idx_order = hot[rng.permutation(len(hot))]
        for i in idx_order:
            lo = max(0.0, h[i] - 0.4)
            hi = min(1.0, h[i] + 0.4)
            if hi - lo < 1e-6:
                continue
            candidates = np.concatenate([[lo, hi], np.linspace(lo, hi, 5)])
            best_val = c5_cur
            best_cand = h[i]
            for v in candidates:
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val - 1e-13:
                    best_val = c5_tmp
                    best_cand = v
            if best_cand != h[i]:
                h[i] = best_cand
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_init(n, rng, n_modes=8):
    h = np.full(n, 0.5)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(k * np.pi * np.arange(n) / (n / 2.0) + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    return h

def structured_perturb(h, n, rng, strength=0.05):
    h_pert = h.copy()
    n_blocks = rng.randint(3, 8)
    block_size = n // (2 * n_blocks)
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        h_pert[start:start + block_size] += rng.uniform(-strength, strength)
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_to_sum(h_pert, n / 2.0)
    return h_pert

            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
            h_opt, c5_opt = nesterov_active(h_f, n_cur, ts_cur, T=10, alpha=200.0,
                                            n_iters=min(1500, int(time_left() * 1.2)),
                                            lr_init=0.1,
                                            time_limit=time_left() - 40, t_start=t_start)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            if time_left() > 20:
                h_opt2, c5_opt2 = slsqp_active(h_opt, n_cur, ts_cur, T=10,
                                               alpha=2000.0,
                                               maxiter=min(200, int(time_left() * 0.8)))
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()

    # Phase 8: Final polish
    if best_h is not None and time_left() > 20:
        n_p = len(best_h)
        ts_p = n_p / 2.0
        h_pol, c5_pol = slsqp_active(best_h, n_p, ts_p, T=5, alpha=5000.0,
                                     maxiter=min(200, int(time_left() * 1.5)))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if time_left() > 15:
            h_cd, c5_cd = coord_descent_smart(best_h, n_p, ts_p, n_rounds=3, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final projection and validation
    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, final_n)
