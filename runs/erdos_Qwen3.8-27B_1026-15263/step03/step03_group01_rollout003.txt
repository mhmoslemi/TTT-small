def binarize_and_refit(h, n):
    """Push h towards 0-1 valued while maintaining sum constraint."""
    ts = n / 2.0
    h_b = h.copy()
    # Sort by value from high to low
    order = np.argsort(-h_b)
    h_new = np.zeros(n)
    # Assign 1 to top ts positions
    h_new[order[:int(ts)]] = 1.0
    if ts != int(ts):
        h_new[order[int(ts)]] = ts - int(ts)
    return h_new

def lag_aware_perturb(h, rng, n_ts, n_perturb=20):
    """Perturb h at positions that contribute to the worst lags."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    # Find top lags
    top_lags = np.argsort(-corr)[:n_ts]
    # For each top lag, find positions with high contribution
    # corr[m] = sum_i h[i] * (1-h[i+m-n+1]) (approximately)
    # The "worse" positions are where h[i] is high AND h[i+k] is high (both contribute to overlap)
    for lag in top_lags:
        k = lag - n + 1  # actual shift
        # positions where h is high and h[i+k] is also high
        for _ in range(n_perturb // n_ts):
            i = rng.randint(0, n)
            j = i + k
            if 0 <= j < n:
                # Transfer mass from the higher of h[i], h[j] to reduce product
                if h[i] > 0.05 and h[j] > 0.05:
                    delta = rng.uniform(0.02, 0.1) * min(h[i], h[j])

                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    h = np.clip(h, 0.0, 1.0)
    return project_to_sum(h, n_ts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def lbfgsb_auglag(h0, n, ts, alpha, maxiter=1500, rho0=10.0):
    h = h0.copy()
    mu_aug = 0.0
    rho = rho0
    bnds = [(0.0, 1.0)] * n
    for outer in range(8):
        def f_aug(x):
            obj, grad = smooth_max_obj_grad(x, alpha)
            s = np.sum(x) - ts
            obj_a = obj + rho * s * s * 0.5 + mu_aug * s
            grad_a = grad + (rho * s + mu_aug) * np.ones(n)
            return obj_a, grad_a
        try:
            r = minimize(f_aug, h.copy(), method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
            h = r.x
        except Exception:
            break
        s = np.sum(h) - ts
        mu_aug = mu_aug + rho * s
        if abs(s) < 1e-7:
            rho *= 0.5
        else:
            rho *= 2.0
        if abs(s) < 1e-12:
            break
    return project_to_sum(h, ts)

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f_fn(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g_fn(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def c_fun(x):
        return np.sum(x) - ts
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f_fn, h0.copy(), method='SLSQP', jac=g_fn, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    stall = 0
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-14:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= 0.5
            stall += 1
            if stall > 120 or lr < 1e-14:
                break
    return h

def lag_aware_perturb(h, rng, n_ts, n_delta=30):

                best_h = h_pol.copy()

    # Phase 6: Diverse frequency restarts
    if time_left() > 100:
        n6 = best_n
        ts6 = n6 / 2.0
        x = np.linspace(0, 2, n6, endpoint=False)
        for trial in range(15):
            if time_left() < 30:
                break
            nf = rng.randint(3, 10)
            freqs = rng.choice(np.arange(1, 25), size=nf, replace=False)
            phases = rng.uniform(0, 2 * np.pi, nf)
            amps = rng.uniform(0.05, 0.25, nf)
            h_s = np.full(n6, 0.5)
            for i in range(nf):
                h_s += amps[i] * np.sin(2 * np.pi * freqs[i] * x / 2.0 + phases[i])
            h_s = project_to_sum(h_s, ts6)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_o = lbfgsb_auglag(h_s, n6, ts6, 200, maxiter=600)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            h_o = lbfgsb_auglag(h_o, n6, ts6, 2000, maxiter=500)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Phase 7: Another round of lag-aware + optimize
    if time_left() > 60:
        n7 = best_n
        for trial in range(20):
            if time_left() < 25:
                break
            h_p = lag_aware_perturb(best_h, rng, n_ts=12, n_delta=25)
            h_o = lbfgsb_auglag(h_p, n7, n7 / 2.0, 2000, maxiter=600)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Phase 8: Final high-alpha polish
    if time_left() > 40:
        n8 = best_n
        for alpha in [8000, 20000]:
            if time_left() < 25:
                break
            h_f = lbfgsb_auglag(best_h, n8, n8 / 2.0, alpha, maxiter=1500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 9: Projected gradient final squeeze
    if time_left() > 30:
        n9 = best_n
        h_pg = proj_grad(best_h, n9, n9 / 2.0, 15000, n_iter=4000, lr_init=0.003)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
