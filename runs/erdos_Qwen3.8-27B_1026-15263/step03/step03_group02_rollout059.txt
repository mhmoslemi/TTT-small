        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

        grad[a_range] += wm * (1.0 - h[b_range])
        grad[b_range] -= wm * h[a_range]

        h_new = project_to_sum(h_new, target_sum)

            h = h_new
            t = min(t * 1.05, 0.1)

        # Gradient with line search
        for _ in range(5):
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) + 1e-10:
                h = h_new
                break
            lr *= 0.5
        else:
            lr *= 0.5
            if lr < 1e-7:
                break
        
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-10:
            best_c5 = c5_now
            best_h = h.copy()
        lr = min(lr * 1.02, 0.1)
    
    return best_h, best_c5

def binary_sa_directed(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    overlaps_curr = compute_overlaps(h)
    T = 0.005
    min_T = 0.0003
    no_improve = 0
    iter_count = 0
    
    while time.time() < t_end:
        iter_count += 1
        h_save = h.astype(np.float32).copy()  # save for revert
        c5_save = c5_curr
        
        max_val = np.max(overlaps_curr)
        
        # Determine move
        did_move = False
        if max_val > 0:
            hot_frac = 0.85 + 0.10 * min(iter_count / 50000.0, 1.0)  # gradually focus more
            hot_mask = overlaps_curr > hot_frac * max_val
            hot_indices = np.where(hot_mask)[0]
            
            if len(hot_indices) > 0:
                k = hot_indices[rng.randint(len(hot_indices))]
                lag = k - (n - 1)
                if lag >= 0:
                    valid = np.where((h[:n-lag] == 0) & (h[lag:] == 1))[0]
                else:
                    abs_lag = -lag
                    valid = np.where((h[abs_lag:] == 0) & (h[:n-abs_lag] == 1))[0]
                
                if len(valid) > 0:
                    i = valid[rng.randint(len(valid))]
                    if lag >= 0:
                        pos0, pos1 = i, i + lag
                    else:
                        pos1, pos0 = i, i + lag
                    # Wait, for negative lag:
                    # lag < 0, so the "full" correlation index is n-1+lag < n-1
                    # This corresponds to: sum_i h[i] * (1-h[i+lag]) for valid i
                    # = sum_i h[i] * (1-h[i-abs(lag)])
                    # So h[i]=1 and h[i-abs(lag)]=0, i.e., 1→0 at distance abs(lag)
                    # pos1 = i (where h=1), pos0 = i-abs_lag (where h=0)
                    # But wait: i ranges over [abs_lag, n-1]
                    # h[i]=1, h[i-abs_lag]=0
                    # So pos1 = i, pos0 = i - abs_lag = i + lag (since lag = -abs_lag)
                    # That's what I have above. ✓
                    
                    if rng.random() < 0.5:
                        # Move the 1 at pos1 to a random 0 position
                        zeros = np.where(h == 0)[0]
                        zeros_mask = zeros != pos0
                        zeros = zeros[zeros_mask]
                        if len(zeros) > 0:
                            j = zeros[rng.randint(len(zeros))]
                            h[pos1], h[j] = 0.0, 1.0
                            did_move = True
                    if not did_move:
                        # Move the 0 at pos0 to a random 1 position
                        ones = np.where(h == 1)[0]
                        ones_mask = ones != pos1
                        ones = ones[ones_mask]
                        if len(ones) > 0:
                            j = ones[rng.randint(len(ones))]
                            h[pos0], h[j] = 1.0, 0.0
                            did_move = True
        
        if not did_move:
            # Random swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                T = max(T * 0.999998, min_T)
                continue
            h[i], h[j] = h[j], h[i]
        
        new_overlaps = compute_overlaps(h)
        c5_new = np.max(new_overlaps)
        delta = c5_new - c5_curr
        
        if delta < -1e-14 or (delta / (T + 1e-15)) + np.log(rng.uniform() + 1e-15) < 0:
            c5_curr = c5_new
            overlaps_curr = new_overlaps
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[:] = h_save
            c5_curr = c5_save
        
        if no_improve > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            overlaps_curr = compute_overlaps(h)
            T = min(T * 1.8, 0.010)
            no_improve = 0
        T = max(T * 0.999997, min_T)
    
    return best_h, best_c5
