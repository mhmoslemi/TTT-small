    return (h + h[::-1].copy()) / 2.0

def make_initial_sym(n, mode, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = 0.5 + 0.3 * np.cos(np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif mode == 3:
        h = np.where(x < 1.0, 0.8, 0.2)
    elif mode == 4:
        h = 0.5 + 0.25 * np.sin(3 * np.pi * x)
    elif mode == 5:
        h = np.clip(0.5 + rng.randn(n) * 0.1, 0, 1)
    else:
        h = 0.5 + 0.2 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
    h = make_symmetric(h)
    return np.clip(h, 0, 1)

def gradient_boost_step(h, n, target_sum, lr, alpha):
    _, g = smooth_max_and_grad(h, alpha)
    h_new = np.clip(h - lr * g, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_work = best_n
    ts_work = n_work / 2.0

    if time_left() > 100:
        n_try = 360
        ts_try = n_try / 2.0
        for mode in range(4):
            if time_left() < 50:
                break
            h_init = make_initial_sym(n_try, mode, rng)
            h_init = project_to_sum(h_init, ts_try)
            c5_init = compute_c5(h_init)
            for alpha in [30.0, 100.0, 300.0]:
                if time_left() < 20:
                    break
                h_opt = l_bfgs_b_optimize(h_init, n_try, ts_try, alpha, 300, lam=500.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < c5_init:
                    h_init = h_opt
                    c5_init = c5_opt
            if c5_init < best_c5:
                x_old = np.linspace(0, 2, n_try, endpoint=False)
                x_new = np.linspace(0, 2, n_work, endpoint=False)
                h_scaled = np.interp(x_new, x_old, h_init)
                h_scaled = project_to_sum(h_scaled, ts_work)
                c5_scaled = compute_c5(h_scaled)
                if c5_scaled < best_c5:
                    best_c5 = c5_scaled
                    best_h = h_scaled.copy()
                    best_n = n_work

    if time_left() > 60:
        h_cur = best_h.copy()
        for alpha in [50.0, 100.0, 200.0, 500.0]:
            if time_left() < 30:
                break
            h_opt = l_bfgs_b_optimize(h_cur, n_work, ts_work, alpha, 400, lam=500.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_work
            h_cur = h_opt

    if time_left() > 60:
        h_cur = best_h.copy()
        for alpha in [1000.0, 2000.0, 5000.0]:
            if time_left() < 25:
                break
            h_opt = l_bfgs_b_optimize(h_cur, n_work, ts_work, alpha, 300, lam=500.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_work
            h_cur = h_opt

    if time_left() > 40:
        h_cur = best_h.copy()
        n_iters = min(3000, int(time_left() * 3))
        h_opt, c5_opt = momentum_descent(h_cur, n_work, ts_work, 1000.0, n_iters, lr_init=0.02, momentum=0.85)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_work
        h_cur = h_opt

    if time_left() > 40:
        h_cur = best_h.copy()
        h_opt, c5_opt = active_set_refine(h_cur, n_work, ts_work, 2000.0, 2000, lr=0.03)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_work
        h_cur = h_opt

    if time_left() > 80:
        n_hi = n_work * 2
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, n_work, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
        for alpha in [50.0, 200.0, 500.0, 1000.0]:
            if time_left() < 30:
                break
            h_opt = l_bfgs_b_optimize(h_hi, n_hi, ts_hi, alpha, 300, lam=500.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hi
            h_hi = h_opt

    if time_left() > 30:
        h_cur = best_h.copy()
        n_iters = min(5000, int(time_left() * 4))
        h_opt, c5_opt = momentum_descent(h_cur, best_n, best_n / 2.0, 2000.0, n_iters, lr_init=0.01, momentum=0.8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n
        h_cur = h_opt

    if time_left() > 30:
        h_cur = best_h.copy()
        h_opt, c5_opt = active_set_refine(h_cur, best_n, best_n / 2.0, 5000.0, 3000, lr=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n
        h_cur = h_opt

    if time_left() > 50:
        n_final = best_n
        ts_final = n_final / 2.0
        h_final = best_h.copy()
        h_opt, c5_opt = coordinate_refine(h_final, n_final, ts_final, min(40, int(time_left() // 2)), rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_final

    if time_left() > 20:
        h_final = best_h.copy()
        n_final = best_n
        ts_final = n_final / 2.0
        h_opt = slsqp_constrained(h_final, n_final, ts_final, 5000.0, 100)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_final

    if best_h is not None:
        n_out = len(best_h)
        ts_out = n_out / 2.0
        best_h = project_to_sum(best_h, ts_out)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_out)
