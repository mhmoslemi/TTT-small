def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_corr(h, corr, k_offset):
    """Gradient of corr[k_offset] with respect to h.
    corr[k] = sum_i h[i] * (1-h[i+k]) in the shifted index system.
    """
    n = len(h)
    # corr has length 2n-1, indexed from 0 to 2n-2
    # corr[m] = sum_i h[i] * (1-h[i+m-(n-1)])
    # Let shift = m - (n-1), so corr[m] = sum_i h[i]*(1-h[i+shift])
    shift = k_offset - (n - 1)
    grad = np.zeros(n)
    # d/dh[j] of sum_i h[i]*(1-h[i+shift])
    # = (1 - h[j+shift]) [from the h[i] term when i=j]
    #   + sum_i h[i] * (-1) * [i+shift = j]  [from the (1-h[i+shift]) term]
    #   = (1 - h[j+shift]) - h[j-shift]  (with appropriate boundary handling)
    for j in range(n):

        t1 = 0.0
        jp = j + shift
        if 0 <= jp < n:

        t2 = 0.0
        jm = j - shift

def gradient_of_max_corr(h, corr, m_active=5):

    n = len(h)
    top_idx = np.argsort(corr)[-m_active:]
    # Weight by softmax
    vals = corr[top_idx]
    weights = np.exp(100 * (vals - vals.max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)

        grad += w * gradient_of_corr(h, corr, int(k))

def pair_descent_single(h, j, k, corr, n):

def solve_minimax_1d(corr, D, lo, hi):
    """Find delta in [lo,hi] that minimizes max(corr + delta*D)."""
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    
    # Coarse grid
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    
    # Bisection
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    
    return (lo_r + hi_r) * 0.5
