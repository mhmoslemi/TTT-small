    if best_idx == len(t_grid) - 1:

        else:

                best_delta = delta_opt

            h[best_k] = np.clip(h[best_k], 0.0, 1.0)

def pair_descent_full(h, corr, rng):

                best_delta = delta_opt
                best_D = D
                best_k = k

            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

For a piecewise linear function f(t) = max_i (a_i + b_i * t), the minimum over t is at a point where at least two lines are tied for the maximum.

This means: the optimal delta_k is at a point where corr[m1] + delta_k * D_k[m1] = corr[m2] + delta_k * D_k[m2] for some m1 ≠ m2, and this common value is the minimum of max(corr + delta * D_k).

Unfortunately, this doesn't easily vectorize over k because the optimal delta depends on k.

However, we CAN vectorize the evaluation: for a given delta, max_m (corr[m] + delta * D_k[m]) = max_m (corr[m] + delta * (B_j[m] - B_k[m])).

If we precompute B_all (the n × (2n-1) matrix), then:
D_all = B_j[None, :] - B_all  # shape (n, 2n-1)
For a given delta: values = corr[None, :] + delta * D_all  # shape (n, 2n-1)
max_over_m = np.max(values, axis=1)  # shape (n,)

But the optimal delta is different for each k! So we can't just evaluate at one delta.

One approach: use a **sample-based** evaluation. For each k, evaluate at a few candidate deltas (e.g., 5 values spread over [lo_k, hi_k]), and pick the best. This is less accurate but much faster.

Actually, for the initial phases, the accuracy of the 1D solve isn't critical. What matters is finding the right k. Let me use a coarse 1D solve (fewer iterations) for the full scan, and a fine 1D solve for the final polish.

Let me use 8 grid points + 20 bisection iterations for the full scan, and 16 + 45 for the polish.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start1 = max(0, j)
    k_end1 = min(2 * n - 2, j + nm1)
    if k_start1 <= k_end1:
        idx = np.arange(k_start1, k_end1 + 1)
        B[idx] += (1.0 - h[j - idx + nm1])
    k_start2 = max(0, nm1 - j)
    k_end2 = min(2 * n - 2, 2 * n - 2 - j)
    if k_start2 <= k_end2:
        idx = np.arange(k_start2, k_end2 + 1)
        B[idx] += (-h[j + idx - nm1])
    return B

def solve_1d_max(A, D, lo, hi, iters=40):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(iters):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent(h, corr, rng, mode='full', n_cand=20, iters=40):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    n_imp = 0
    for j in order:
        hj = h[j]
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        if mode == 'full':
            candidates = range(n)
        else:
            cands = [(j + 1) % n, (j - 1 + n) % n, (j + 2) % n, (j - 2 + n) % n]
            hot_idx = np.argsort(corr)[-15:]
            for hk in hot_idx:
                cands.append((j + hk) % n)
                cands.append((j - hk + n) % n)
            for _ in range(max(0, n_cand - len(cands))):
                cands.append(rng.randint(n))
            candidates = cands
        seen = set()
        for k in candidates:
            if k == j or k in seen:
                continue
            seen.add(k)
            hk_val = h[k]
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-hj, hk_val - 1.0)
            delta_hi = min(1.0 - hj, hk_val)
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_max(corr, D, delta_lo, delta_hi, iters)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr += best_delta * best_D
            h[j] = np.clip(hj + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            n_imp += 1
    return h, corr, n_imp

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t_start

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Full-scan pair descent (50% budget)
    t_end1 = t_start + 0.50 * budget_s
    h_w = best_h.copy()
    n_w = len(h_w)
    corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
    stall = 0
    noise_level = 0.0

    while elapsed() < t_end1:
        h_w, corr_w, n_imp = pair_descent(h_w, corr_w, rng, mode='full', iters=40)
        c5_w = float(np.max(corr_w) * (2.0 / n_w))
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_w.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall >= 30:
            noise_level = min(0.15, 0.02 * (1.0 + stall / 30.0))
            h_w = project_to_sum(h_w + rng.uniform(-noise_level, noise_level, n_w), n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            stall = 0
            noise_level = 0.0

    # Phase 2: Alternating full/limited scans (15% budget)
    t_end2 = t_start + 0.65 * budget_s
    h_w = best_h.copy()
    n_w = len(h_w)
    corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
    stall2 = 0
    scan_count = 0

    while elapsed() < t_end2:
        if scan_count % 3 == 0:
            h_w, corr_w, _ = pair_descent(h_w, corr_w, rng, mode='full', iters=35)
        else:
            h_w, corr_w, _ = pair_descent(h_w, corr_w, rng, mode='limited', n_cand=30, iters=35)
        scan_count += 1
        c5_w = float(np.max(corr_w) * (2.0 / n_w))
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_w.copy()
            best_n = n_w
            stall2 = 0
        else:
            stall2 += 1
        if stall2 >= 40:
            h_w = project_to_sum(h_w + rng.uniform(-0.05, 0.05, n_w), n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            stall2 = 0

    # Phase 3: Multi-resolution exploration (20% budget)
    t_end3 = t_start + 0.85 * budget_s
    x_old = np.linspace(0, 2, best_n, endpoint=False)
    resolutions = [320, 400, 480, 360]

    for n_tr in resolutions:
        if elapsed() >= t_end3:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        starts = [project_to_sum(h_interp, ts_tr)]
        starts.append(project_to_sum(h_interp + rng.normal(0, 0.08, n_tr), ts_tr))
        blk = n_tr // 8
        hb = np.zeros(n_tr)
        for i in range(0, n_tr, 2 * blk):
            hb[i:min(i + blk, n_tr)] = 1.0
        starts.append(project_to_sum(hb, ts_tr))

        for h_st in starts:
            if elapsed() >= t_end3:
                break
            h_st = project_to_sum(h_st, ts_tr)
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            for _ in range(20):
                if elapsed() >= t_end3:
                    break
                h_st, corr_st, _ = pair_descent(h_st, corr_st, rng, mode='limited', n_cand=15, iters=30)
            c5_st = float(np.max(corr_st) * (2.0 / n_tr))
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 4: Final full-scan polish (12% budget)
    t_end4 = t_start + budget_s - 3
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall4 = 0

    while elapsed() < t_end4:
        h_fin, corr_fin, n_imp = pair_descent(h_fin, corr_fin, rng, mode='full', iters=45)
        c5_fin = float(np.max(corr_fin) * (2.0 / n_fin))
        if c5_fin < best_c5 - 1e-11:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall4 = 0
        else:
            stall4 += 1
        if stall4 >= 20:
            if n_imp == 0:
                break
            h_fin = project_to_sum(h_fin + rng.uniform(-0.02, 0.02, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall4 = 0

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
