    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    n = len(h)
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h + mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h + lam, 0.0, 1.0)
    # Fix any tiny numerical error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-14:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.any(mask):
            x[mask] += diff / np.count_nonzero(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def corr_fft(h, N_fft):
    n = len(h)
    a = np.zeros(N_fft)
    a[:n] = h
    b = np.zeros(N_fft)
    b[:n] = 1.0 - h
    R = np.fft.irfft(np.fft.rfft(a) * np.conj(np.fft.rfft(b)), N_fft)
    return np.concatenate([R[N_fft-(n-1):], R[:n]])

def subgrad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if n > lag:
            g[:n-lag] += 1.0 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        L = -lag
        if L < n:
            g[L:] += 1.0 - h[:n-L]
            g[:n-L] -= h[L:]
    return g

def coord_descent_pass(h, n, target_sum, rng, n_rounds=3, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.03, 0.015, 0.007, 0.003]
    c5_cur = compute_c5(h)
    N_fft = 1
    while N_fft < 2*n-1:
        N_fft *= 2
    
    for rnd in range(n_rounds):
        improved = False
        for i in rng.permutation(n):
            best_d = 0.0
            best_c5 = c5_cur
            for d in step_sizes:
                for sign in [1, -1]:
                    dd = d * sign
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + dd, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    corr = corr_fft(h_new, N_fft)
                    c5_new = np.max(corr) * (2.0/n)
                    if c5_new < best_c5 - 1e-13:
                        best_c5 = c5_new
                        best_d = dd
                        best_h = h_new
            if best_d != 0.0:
                h = best_h
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur
