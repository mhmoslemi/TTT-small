    for alpha in phase1_alphas:
        if time_left() < 50:
            break
        h_opt, c5_opt = fista_projected(h_cur, alpha, target,
                                         n_iters=min(3000, int(time_left() * 3)),
                                         lr_init=0.1, time_limit=min(50, time_left() * 0.6),
                                         t_start=t_start)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    if time_left() > 80:
        h_opt2, c5_opt2 = multi_alpha_slsqp(h_cur, target, [100, 500, 2000], 80, t_start)
        if c5_opt2 < best_c5 - 1e-12:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            h_cur = h_opt2

    if time_left() > 100:
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if time_left() < 40:
                break
            h_bin = binary_quantize(h_cur, target, thresh, 1.0 - thresh)
            c5_bin = compute_c5(h_bin)
            h_optb, c5_optb = fista_projected(h_bin, 2000, target,
                                               n_iters=2000, lr_init=0.05,
                                               time_limit=min(30, time_left() * 0.3),
                                               t_start=t_start)
            if c5_optb < best_c5 - 1e-12:
                best_c5 = c5_optb
                best_h = h_optb.copy()
            if c5_bin < best_c5 - 1e-12:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            h_cur = best_h.copy()

    if time_left() > 80:
        h_blk, c5_blk = block_perturb(h_cur, target, rng,
                                       n_blocks=min(1000, int(time_left() * 2)),
                                       block_frac=0.05)
        if c5_blk < best_c5 - 1e-12:
            best_c5 = c5_blk
            best_h = h_blk.copy()
        h_opt3, c5_opt3 = fista_projected(h_blk, 3000, target,
                                           n_iters=3000, lr_init=0.08,
                                           time_limit=min(40, time_left() * 0.5),
                                           t_start=t_start)
        if c5_opt3 < best_c5 - 1e-12:
            best_c5 = c5_opt3
            best_h = h_opt3.copy()
            h_cur = h_opt3

    n2 = n * 2
    if time_left() > 120 and n2 <= 2880:
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = project_to_sum(h_interp, target2)
        h_hi = h_interp
        for alpha in [200, 1000, 5000, 10000]:
            if time_left() < 50:
                break
            h_opt_hi, c5_opt_hi = fista_projected(h_hi, alpha, target2,
                                                   n_iters=min(4000, int(time_left() * 2.5)),
                                                   lr_init=0.05,
                                                   time_limit=min(45, time_left() * 0.5),
                                                   t_start=t_start)
            if c5_opt_hi < best_c5 - 1e-12:
                best_c5 = c5_opt_hi
                best_h = h_opt_hi.copy()
            h_hi = h_opt_hi

        if time_left() > 60:
            h_opt_hi2, c5_opt_hi2 = multi_alpha_slsqp(h_hi, target2, [500, 3000], 60, t_start)
            if c5_opt_hi2 < best_c5 - 1e-12:
                best_c5 = c5_opt_hi2
                best_h = h_opt_hi2.copy()

    if time_left() > 40:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_fin = best_h.copy()
        for alpha in [5000, 10000, 20000]:
            if time_left() < 15:
                break
            h_fin, c5_fin = fista_projected(h_fin, alpha, t_fin,
                                             n_iters=5000, lr_init=0.03,
                                             time_limit=min(20, time_left() * 0.4),
                                             t_start=t_start)
            if c5_fin < best_c5 - 1e-12:
                best_c5 = c5_fin
                best_h = h_fin.copy()

        if time_left() > 15:
            h_fin = binary_quantize(best_h, t_fin, 0.45, 0.55)
            h_fin, c5_fin = fista_projected(h_fin, 5000, t_fin,
                                             n_iters=3000, lr_init=0.03,
                                             time_limit=min(12, time_left() * 0.3),
                                             t_start=t_start)
            if c5_fin < best_c5 - 1e-12:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
