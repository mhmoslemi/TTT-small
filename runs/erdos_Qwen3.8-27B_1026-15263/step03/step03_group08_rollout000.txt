import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_of_active_lags(h, active_lags):
    n = len(h)
    g = np.zeros(n)
    for l in active_lags:
        lo = 0
        hi = n - l if l >= 0 else 0
        if l >= 0:
            g[:n - l] += (1.0 - h[l:n])
            g[l:] -= h[:n - l]
        else:
            g[-l:] += (1.0 - h[:n + l])
            g[:-l] -= h[-l:]
    return g

def nesterov_subgrad(h0, n, target_sum, n_iters, lr_init, top_k, momentum0, momentum1, t_start, t_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr_init
    dt = 2.0 / n
    stall = 0
    for it in range(n_iters):
        if time.time() > t_deadline - 30:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        active_lags = [int(k - (n - 1)) for k in top_idx]
        g = subgrad_of_active_lags(h, active_lags)
        g *= dt / len(active_lags)
        t_val = it + 1
        mom = momentum0 + (momentum1 - momentum0) * min(1.0, t_val / max(1, n_iters * 0.5))
        h_new = h + mom * v - lr * g
        v = h_new - h
        h = np.clip(h_new, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        if it % 8 == 0 or it == n_iters - 1:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-7:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                if stall > 6:
                    lr *= 0.8
                    v *= 0.3
                    stall = 0
    return best_h, best_c5

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:

        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:

        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def slsqp_polish(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_obj_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_obj_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        return h0.copy()

                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    t_deadline = t_start + budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        x = np.linspace(0, 2, best_n, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / 2.0) + 0.1 * np.sin(2 * np.pi * 7 * x / 2.0 + 1.0)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    resolutions = [best_n]
    if best_n <= 480:
        resolutions.extend([720, 960, 1440])
    elif best_n <= 720:
        resolutions.extend([960, 1440])
    elif best_n <= 960:
        resolutions.extend([1440, 1920])
    else:
        resolutions.append(best_n + 480)
    resolutions = sorted(set(resolutions))
    resolutions = [r for r in resolutions if r % 2 == 0 and r <= 2400]

    for n in resolutions:
        if time.time() > t_deadline - 120:
            break
        ts = n / 2.0

        if best_n == n:
            h_work = best_h.copy()
        else:
            h_work = interpolate_h(best_h, n)
            h_work = project_to_sum(h_work, ts)

        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n

        n_iter1 = min(8000, max(2000, int((t_deadline - time.time()) * 1.5)))
        h_work, c5_w = nesterov_subgrad(h_work, n, ts, n_iter1, 3.0, 15, 0.7, 0.95, t_start, t_deadline)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n

        if time.time() < t_deadline - 60:
            n_iter2 = min(5000, max(1000, int((t_deadline - time.time()) * 1.2)))
            h_work, c5_w = nesterov_subgrad(h_work, n, ts, n_iter2, 1.5, 5, 0.9, 0.97, t_start, t_deadline)
            c5_w = compute_c5(h_work)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_work.copy()
                best_n = n

        if time.time() < t_deadline - 50:
            for alpha in [2000.0, 5000.0, 10000.0]:
                if time.time() > t_deadline - 40:
                    break
                h_pol = slsqp_polish(h_work, n, ts, alpha, 400)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n
                h_work = h_pol

        if time.time() < t_deadline - 40:
            base_steps = [0.03, 0.01, 0.005, -0.03, -0.01, -0.005]
            max_rnd = min(20, max(5, int((t_deadline - time.time()) // 8)))
            h_work, c5_cd = coordinate_descent(h_work, n, ts, max_rnd, base_steps, t_deadline)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_work.copy()
                best_n = n

        if time.time() < t_deadline - 30:
            h_bin, c5_bin = binary_local_search(h_work, n, ts, 500, t_deadline)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n

    if best_h is not None and time.time() < t_deadline - 40:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha in [5000.0, 10000.0, 20000.0, 50000.0]:
            if time.time() > t_deadline - 30:
                break
            h_pol = slsqp_polish(best_h, n_f, ts_f, alpha, 600)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time.time() < t_deadline - 25:
            n_iter_f = min(4000, max(500, int((t_deadline - time.time()) * 0.8)))
            h_f, c5_f = nesterov_subgrad(best_h, n_f, ts_f, n_iter_f, 0.8, 5, 0.93, 0.98, t_start, t_deadline)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        if time.time() < t_deadline - 15:
            base_steps = [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 30, base_steps, t_deadline)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time.time() < t_deadline - 10:
            h_bin, c5_bin = binary_local_search(best_h, n_f, ts_f, 300, t_deadline)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
