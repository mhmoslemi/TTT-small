def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, penalty=0.0):
    """Compute smooth max of correlations and its gradient, plus sum constraint penalty."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    obj = m_val + np.log(np.sum(exp_v)) / alpha
    
    # Gradient of smooth max
    grad = np.zeros(n)
    # Positive lags: k = 0, ..., n-1, corresponding to corr index n-1+k
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-12:
            continue
        # ∂c_k/∂h[j] = (1-h[j-k])*[j>=k] - h[j+k]*[j+k<n]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            term = np.zeros(n)
            term[k:] = 1.0 - h[:n-k]
            term[:n-k] -= h[k:]
            grad += wk * term
    # Negative lags: L = 1, ..., n-1, corresponding to corr index n-1-L
    for L in range(1, n):
        wL = w[n - 1 - L]
        if wL < 1e-12:
            continue
        # ∂c_{-L}/∂h[j] = (1-h[j+L])*[j<=n-1-L] - h[j-L]*[j>=L]
        term = np.zeros(n)
        term[:n-L] = 1.0 - h[L:]
        term[L:] -= h[:n-L]
        grad += wL * term
    
    # Add penalty gradient
    if penalty > 0:
        s = np.sum(h) - n / 2.0
        obj += penalty * s * s
        grad += 2.0 * penalty * s
    
    return obj, grad

def lbfgs_optimize(h0, alpha, penalty, maxiter=300):
    """Optimize using L-BFGS-B."""
    n = len(h0)
    
    def fun(x):
        return smooth_max_obj_grad(x, alpha, penalty)[0]
    
    def jac(x):
        return smooth_max_obj_grad(x, alpha, penalty)[1]
    
    bnds = [(0.0, 1.0)] * n
    result = minimize(fun, h0, method='L-BFGS-B', jac=jac, bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
    return result.x

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

    n = len(h)
