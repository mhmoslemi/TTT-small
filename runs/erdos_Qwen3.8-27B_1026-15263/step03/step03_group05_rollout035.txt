    n = len(h)
    h_new = h.copy()
    if move_type == 0:  # Single point perturbation
        i = rng.randint(n)
        delta = rng.uniform(-0.3, 0.3)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    elif move_type == 1:  # Block complement (short block)
        L = rng.randint(2, max(3, n // 20))
        p = rng.randint(0, n - L)
        h_new[p:p+L] = 1.0 - h[p:p+L]
    elif move_type == 2:  # Segment swap
