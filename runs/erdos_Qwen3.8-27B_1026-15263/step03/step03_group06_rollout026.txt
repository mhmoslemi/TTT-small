def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    """Compute smooth max of corr values and its gradient w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n - 1)) / alpha
    
    # Gradient computation via FFT
    # g[i] = sum_u w[i+n-1-u] * (1-h[u]) - sum_v w[n-1-i+v] * h[v]
    
    N = 1
    while N < 2 * (2 * n - 1):
        N *= 2
    
    w_fft = np.fft.fft(w, N)
    
    # First part: conv(w, 1-h) at positions [n-1, 2n-2]
    conv1 = np.real(np.fft.ifft(w_fft * np.fft.fft(1.0 - h, N)))
    
    # Second part: corr(w, h) at positions [0, n-1] (reversed)
    # corr(w,h)[k] = sum_v w[k+v] * h[v]
    # We want positions n-1, n-2, ..., 0
    # corr(w,h) = conv(flip(w), h)
    flip_w = w[::-1]
    conv2 = np.real(np.fft.ifft(np.fft.fft(flip_w, N) * np.fft.fft(h, N)))
    # conv2[p] = sum_j flip_w[j] * h[p-j] = sum_j w[2n-2-j] * h[p-j]
    # Let u = 2n-2-j: = sum_u w[u] * h[p-2n+2+u]
    # We want sum_v w[k+v] * h[v], let u = k+v: = sum_u w[u] * h[u-k]
    # So p-2n+2+u = u-k → p = 2n-2-k
    # So corr(w,h)[k] = conv2[2n-2-k]
    
    g = np.zeros(n)
    # First part at position i: conv1[i+n-1]
    g += conv1[n-1:2*n-1]
    # Second part at position i: corr(w,h)[n-1-i] = conv2[2n-2-(n-1-i)] = conv2[n-1+i]
    g -= conv2[n-1:2*n-1]
    
    return sm_val, g

def projected_grad_desc(h0, n, target_sum, alpha, n_iters, lr_init, t_budget):
    """Projected gradient descent with step size adaptation."""
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for it in range(n_iters):
        if time.time() - t0 > t_budget:
            break
        sm_val, g = smooth_max_obj_and_grad(h, alpha)
        
        h_new = h - lr * g
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
        
        h = h_new
        if lr < 1e-8:
            break
    
    return best_h, best_c5
