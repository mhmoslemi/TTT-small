```python
            # j+shift ranges from j_lo+shift to j_hi-1+shift
            h_idx_lo = j_lo + shift
            h_idx_hi = j_hi + shift  # exclusive
            grad[j_lo:j_hi] -= wk * h[h_idx_lo:h_idx_hi]
        
        # Term: -wk * h[j - shift] for 0 <= j - shift < n
        # j - shift >= 0 => j >= shift
        # j - shift < n => j < n + shift
        j_lo2 = max(0, shift)
        j_hi2 = min(n, n + shift)  # exclusive
        if j_lo2 < j_hi2:
            h_idx_lo2 = j_lo2 - shift
            h_idx_hi2 = j_hi2 - shift  # exclusive
            grad[j_lo2:j_hi2] -= wk * h[h_idx_lo2:h_idx_hi2]
    
    return grad
```
