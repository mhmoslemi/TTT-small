import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_j(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Grid + bisection
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_sweep(h, corr, n, rng, n_cand=8, top_lags=5):
    """One full sweep of pair coordinate descent."""
    order = np.arange(n)
    rng.shuffle(order)
    # Focus on positions that contribute to the top lags
    max_corr = np.max(corr)
    top_indices = np.argsort(corr)[-top_lags:]
    
    improved = False
    for j in order:
        B_j = compute_B_j(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Candidate selection:
        # 1. Neighbors
        # 2. Positions that affect the top lags
        # 3. Random
        cands = set()
        cands.add((j + 1) % n)
        cands.add((j - 1 + n) % n)
        cands.add((j + n // 2) % n)
        cands.add((j + n // 3) % n)
        cands.add((j + 2 * n // 3) % n)
        # Add random candidates
        for _ in range(n_cand - 5):
            cands.add(rng.randint(n))
        # Add candidates based on top lags (positions j ± d for top lags)
        for li in top_indices:
            d = li - (n - 1)
            c1 = (j + d) % n
            c2 = (j - d + n) % n
            cands.add(c1)
            cands.add(c2)
        
        for k in cands:
            if k == j:
                continue
            B_k = compute_B_j(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0, 1)
            h[best_k] = np.clip(h[best_k] - best_delta, 0, 1)
            improved = True
    
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    """Compute smooth max of corr and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha)
    
    # Gradient: d/dh[j] of Σ_m w[m] * corr[m]
    # = Σ_m w[m] * B_j[m]
    # where B_j[m] = d(corr[m])/dh[j]
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        # B_j[m] for all m, weighted by w[m]
        k_start = max(0, nm1 - j)
        k_end = min(2 * n - 2, 2 * nm1 - j)
        if k_start <= k_end:
            idx = np.arange(k_start, k_end + 1)
            vals = 1.0 - h[j - nm1 + idx]
            grad[j] += np.sum(w[k_start:k_end+1] * vals)
        k_start2 = max(0, j)
        k_end2 = min(2 * n - 2, j + nm1)
        if k_start2 <= k_end2:
            idx = np.arange(k_start2, k_end2 + 1)
            vals = h[j + nm1 - idx]
            grad[j] -= np.sum(w[k_start2:k_end2+1] * vals)
    return obj, grad

def subgrad_descent(h0, n, target_sum, alpha, lr, n_iter, momentum=0.9):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        obj, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Adaptive learning rate
        if it % 50 == 0:
            lr *= 0.95
    return best_h, best_c5

def binary_local_search(h0, n, target_sum, rng, n_iters=5000):
    """Try to find a good binary (or near-binary) solution."""
    h = h0.copy()
    # First, try to binarize
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = project_to_sum(h_bin, target_sum)
    # Now do local search: flip pairs
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    
    for _ in range(n_iters):
        i, j = rng.randint(n, 2)

            h_bin[j] = 1.0 - h_bin[j]
        else:
            best_h = h_bin.copy()
            best_c5 = c5_new
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_j(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]

        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid

def pair_sweep(h, corr, n, rng, n_cand=12, focus_top=5):
    order = np.arange(n)
    rng.shuffle(order)
    top_lags = np.argsort(corr)[-focus_top:]
    improved = False
    
    for j in order:
        B_j = compute_B_j(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        cands = set()
        cands.update([(j + d) % n for d in [1, -1, n//2, n//3, 2*n//3, n//4, 3*n//4]])
        for _ in range(max(0, n_cand - 7)):
            cands.add(rng.randint(n))
        for li in top_lags:
            d = li - (n - 1)
            cands.add((j + d) % n)
            cands.add((j - d) % n)
        
        for k in cands:
            if k == j:
                continue
            B_k = compute_B_j(h, k, n)
            D = B_j - B_k
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-13:
                continue
            delta = solve_1d(corr, D, dlo, dhi)
            val = np.max(corr + D * delta)
            if val < best_val - 1e-14:
                best_val = val
                best_delta = delta
                best_D = D
                best_k = k

            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    
    return h, corr, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))

        if ks2 <= ke2:
            idx = np.arange(ks2, ke2+1)
            grad[j] -= np.dot(w[ks2:ke2+1], h[j + nm1 - idx])
    return obj, grad
