        if use_nesterov:
            t_mom = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
            y_new = y + (it / (it + 1.0 + t_mom)) * (y - h)

        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts_init = best_n / 2.0
            best_h = project_to_sum(best_h, ts_init)
            best_c5 = compute_c5(best_h)
        except (NameError, Exception):
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    alpha_schedule = [10.0, 30.0, 80.0, 200.0, 500.0]
    for alpha in alpha_schedule:
        if time_left() < 50:
            break
        maxiter = max(80, min(400, int(time_left() * 0.25)))
        h_opt = slsqp_opt(h_cur, n, ts, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        h_cur = h_opt

    if time_left() > 50:
        c5_chk, corr_chk, max_corr = compute_c5_info(h_cur)
        threshold = 0.995 * max_corr
        active = np.where(corr_chk >= threshold)[0]
        if 0 < len(active) < n:
            for alpha_a in [500.0, 1500.0]:
                if time_left() < 25:
                    break
                h_act = slsqp_active(h_cur, n, ts, active, alpha_a, 250)
                c5_act = compute_c5(h_act)
                if c5_act < best_c5:
                    best_c5 = c5_act
                    best_h = h_act.copy()
                    best_n = n
                h_cur = h_act
                _, corr_new, max_new = compute_c5_info(h_cur)
                active = np.where(corr_new >= 0.995 * max_new)[0]
                if len(active) <= 1:
                    break

    if time_left() > 100:
        n_hi = n * 2
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_cur)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        for alpha in [50.0, 200.0, 800.0]:
            if time_left() < 35:
                break
            maxiter = max(50, min(250, int(time_left() * 0.25)))
            h_hi_opt = slsqp_opt(h_hi, n_hi, ts_hi, alpha, maxiter)
            c5_hi_opt = compute_c5(h_hi_opt)
            if c5_hi_opt < best_c5:
                best_c5 = c5_hi_opt
                best_h = h_hi_opt.copy()
                best_n = n_hi
            h_hi = h_hi_opt
        if time_left() > 30:
            _, corr_hi, max_hi = compute_c5_info(h_hi)
            active_hi = np.where(corr_hi >= 0.995 * max_hi)[0]
            if 0 < len(active_hi) < n_hi:
                h_act_hi = slsqp_active(h_hi, n_hi, ts_hi, active_hi, 1000.0, 200)
                c5_act_hi = compute_c5(h_act_hi)
                if c5_act_hi < best_c5:
                    best_c5 = c5_act_hi
                    best_h = h_act_hi.copy()
                    best_n = n_hi
                h_hi = h_act_hi

    if time_left() > 40:
        n_cur = best_n
        ts_cur = n_cur / 2.0
        h_sg, c5_sg = projected_subgrad(best_h, n_cur, ts_cur,
                                         min(3000, int(time_left() * 2)), 0.02, 300.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n_cur

    if time_left() > 30:
        n_cur = best_n
        ts_cur = n_cur / 2.0
        h_cd, c5_cd = coord_descent(best_h, n_cur, ts_cur, 15, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_cur

    if time_left() > 50:
        n_cur = best_n
        ts_cur = n_cur / 2.0
        h_ms, c5_ms = multi_start_refine(best_h, n_cur, ts_cur, rng, 5, time_left() * 0.4)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            best_n = n_cur

    if time_left() > 25:
        n_cur = best_n
        ts_cur = n_cur / 2.0
        for alpha in [3000.0, 8000.0]:
            if time_left() < 20:
                break
            h_pol = slsqp_opt(best_h, n_cur, ts_cur, alpha, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_cur

    if time_left() > 20:
        n_cur = best_n
        ts_cur = n_cur / 2.0
        h_cd2, c5_cd2 = coord_descent(best_h, n_cur, ts_cur, 30, rng,
                                       deltas=[-0.002, -0.001, -0.0005, 0.0005, 0.001, 0.002])
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n_cur

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
