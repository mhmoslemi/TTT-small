    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(diff) < 1e-13:

        step = diff / len(h)
        h = np.clip(h + step, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-12:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_descent(h0, n, target_sum, alpha, n_iters, lr_init=0.05):
    """Nesterov accelerated gradient descent for smooth max."""
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t = 1.0
    
    for it in range(n_iters):
        sm_val, g = smooth_max_and_grad(y, alpha)
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        
        # Gradient step with backtracking
        eta = lr
        accepted = False
        for _ in range(20):
            y_trial = y - eta * g
            y_trial = project_to_sum(y_trial, target_sum)
            c5_trial = compute_c5(y_trial)
            if c5_trial <= best_c5 + 0.01 * eta * np.dot(g, y - y_trial):
                h = y_trial
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            h = project_to_sum(y - lr * 0.01 * g, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.8
        
        # Nesterov extrapolation
        y = h + beta * (h - y)
        y = project_to_sum(y, target_sum)
        t = t_new
    
    return best_h, best_c5

def polyak_subgradient(h0, n, target_sum, alpha, n_iters, U_init=None):
    """Subgradient method with Polyak step size."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if U_init is None:
        U = best_c5 - 0.05  # Upper bound on optimal (assumed)
    else:
        U = U_init
    
    for it in range(n_iters):
        sm_val, g = smooth_max_and_grad(h, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        gap = sm_val - U
        if gap <= 0:
            eta = 0.01
        else:
            eta = gap / (g_norm * g_norm)
            eta = min(eta, 1.0)  # Cap the step size
        
        h_new = h - eta * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            U = max(U, c5_new - 0.001)  # Update upper bound
        elif c5_new < sm_val:
            h = h_new
        else:
            h = h_new  # Still move, but don't update best
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            for d in base_steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_val = h_new[i]
            if best_c5 < c5_cur - 1e-14:
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-12:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def _slsqp_obj(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _gp_alpha)
    return v

def _slsqp_grad(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _gp_alpha)
    return g

def _slsqp_con_fun(x):
    return np.sum(x) - _gp_target

def _slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _gp_alpha, _gp_target
    _gp_alpha = alpha
    _gp_target = target_sum
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h0.copy(), jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_descent(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t = 1.0
    c5_cur = best_c5
    
    for it in range(n_iters):
        sm_val, g = smooth_max_and_grad(y, alpha)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        
        h_new = y - lr * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new < c5_cur - 1e-15:
            lr = min(lr * 1.04, 5.0)
        else:
            lr = max(lr * 0.82, 0.00005)
        
        h = h_new
        c5_cur = c5_new
        y = h + beta * (h - y)
        y = project_to_sum(y, target_sum)
        t = t_new
        
        if lr < 0.0001:
            break
    
    return best_h, best_c5

def subgradient_descent(h0, n, target_sum, alpha, n_iters, lr_init, U):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(n_iters):
        sm_val, g = smooth_max_and_grad(h, alpha)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step
        gap = sm_val - U
        if gap > 0:
            eta = gap / g_norm_sq
        else:
            eta = 0.01 / (1.0 + it * 0.01)
        eta = min(eta, 1.0)
        
        h_new = h - eta * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            U = max(0.1, best_c5 - 0.002)
        else:
            if c5_new < sm_val:
                h = h_new
            # else: keep h, don't move
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_local = c5_cur
            best_h_local = h.copy()
            for d in base_steps:
                h_new = h.copy()
                nv = h[i] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h_new[i] = nv
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_local - 1e-14:
                    best_local = c5_new
                    best_h_local = h_new
            if best_local < c5_cur - 1e-14:
                h = best_h_local
                c5_cur = best_local
                improved = True
        if not improved:
            break
    return h, c5_cur

    n = best_n
    ts = n / 2.0

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP multi-start with moderate alpha
    if remaining() > 120:
        starts = [best_h.copy()]
        for delta in [0.03, 0.06, 0.10]:
            h_p = best_h + rng.uniform(-delta, delta, n)
            h_p = project_to_sum(h_p, ts)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
            starts.append(h_p)

        x_grid = np.linspace(0, 2, n, endpoint=False)
        for freq in [2, 3, 5]:
            for amp in [0.15, 0.25]:
                h_s = best_h + amp * np.sin(2.0 * np.pi * freq * x_grid / 2.0)
                h_s = project_to_sum(h_s, ts)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                if c5_s < best_c5 + 0.02:
                    starts.append(h_s)

        for h_s in starts[:5]:
            if remaining() < 50:
                break
            h_opt = slsqp_optimize(h_s, n, ts, 200.0, 250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: SLSQP with increasing alpha
    if remaining() > 100:
        for alpha in [500.0, 1000.0, 2000.0]:
            if remaining() < 40:
                break
            maxit = max(100, min(350, int(remaining() * 0.6)))
            h_opt = slsqp_optimize(best_h, n, ts, alpha, maxit)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Nesterov with increasing alpha (main workhorse)
    if remaining() > 150:
        alpha_schedule = [300.0, 700.0, 1500.0, 3000.0, 6000.0, 12000.0, 25000.0, 50000.0]
        for alpha in alpha_schedule:
            rem = remaining()
            if rem < 50:
                break
            n_iters = max(300, min(4000, int(rem * 0.5)))
            lr_init = min(0.5, max(0.002, 100.0 / alpha))
            h_nest, c5_nest = nesterov_descent(best_h, n, ts, alpha, n_iters, lr_init)
            if c5_nest < best_c5:
                best_c5 = c5_nest
                best_h = h_nest.copy()

    # Phase 4: Subgradient with Polyak steps
    if remaining() > 80:
        U = max(0.1, best_c5 - 0.002)
        for alpha in [2000.0, 8000.0, 20000.0]:
            rem = remaining()
            if rem < 30:
                break
            n_iters = max(500, min(5000, int(rem * 1.2)))
            h_sub, c5_sub = subgradient_polyak(best_h, n, ts, alpha, n_iters, U)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
            U = max(0.1, best_c5 - 0.001)

    # Phase 5: Fine SLSQP polish
    if remaining() > 35:
        for alpha in [15000.0, 40000.0]:
            if remaining() < 18:
                break
            h_fin = slsqp_optimize(best_h, n, ts, alpha, min(300, int(remaining() * 2)))
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 6: Coordinate descent with moderate steps
    if remaining() > 25:
        steps = [0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005]
        max_rnd = max(5, min(25, int(remaining() // 8)))
        h_cd, c5_cd = coordinate_descent(best_h, n, ts, max_rnd, steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final Nesterov push with very high alpha
    if remaining() > 30:
        n_iters = max(200, min(3000, int(remaining() * 1.5)))
        h_nest2, c5_nest2 = nesterov_descent(best_h, n, ts, 30000.0, n_iters, 0.001)
        if c5_nest2 < best_c5:
            best_c5 = c5_nest2
            best_h = h_nest2.copy()

    # Phase 8: Final coordinate descent with small steps
    if remaining() > 12:
        steps = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
        max_rnd = max(3, min(15, int(remaining() // 5)))
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, ts, max_rnd, steps)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 9: Very fine coordinate descent
    if remaining() > 8:
        steps = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]
        max_rnd = max(2, min(10, int(remaining() // 4)))
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, ts, max_rnd, steps)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
