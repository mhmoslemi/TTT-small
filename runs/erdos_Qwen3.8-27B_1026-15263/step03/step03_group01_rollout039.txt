                                            max_iters_per_alpha=2000)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Upsample to 960 and optimize
    if time_left() > 200:
        n_up = 960
        ts_up = n_up / 2.0
        h_up = upsample(best_h, best_n, n_up)
        h_up = project_to_sum(h_up, ts_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
        else:
            best_n = n_up
            best_h = h_up.copy()

        h_opt, c5_opt = optimize_at_resolution(best_h, best_n, best_n / 2.0, time_left,
                                                max_alphas=[30, 100, 500, 2000],
                                                max_iters_per_alpha=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Frequency-domain perturbations and re-optimization
    n_b = best_n
    ts_b = n_b / 2.0
    for trial in range(30):
        if time_left() < 40:
            break
        amp = 0.06 * (1.0 - trial / 40.0) + 0.01
        nf = max(3, n_b // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        alpha = [100, 500][trial % 2]
        h_pg, _ = projected_grad_opt(h_sh, n_b, ts_b, alpha, n_iter=1500, lr_init=0.005, time_fn=time_left)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: Diverse restarts
    n_d = best_n
    ts_d = n_d / 2.0
    for trial in range(15):
        if time_left() < 50:
            break
        starts = make_diverse_starts(n_d, ts_d, best_h, best_n)
        for h_s in starts[:5]:
            if time_left() < 25:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_pg, _ = projected_grad_opt(h_s, n_d, ts_d, 200, n_iter=1000, lr_init=0.01, time_fn=time_left)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 5: Stochastic refinement
    if time_left() > 100:
        n_s = best_n
        ts_s = n_s / 2.0
        h_st, c5_st = stochastic_refine(best_h, n_s, ts_s, rng, 40000, 0.01, 0.3, time_left)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
        if time_left() > 40:
            h_pg, _ = projected_grad_opt(best_h, n_s, ts_s, 5000, n_iter=2000, lr_init=0.002, time_fn=time_left)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 6: High-alpha polish
    if time_left() > 60:
        n_p = best_n
        ts_p = n_p / 2.0
        for alpha in [5000, 20000, 100000]:
            if time_left() < 30:
                break
            h_pg, _ = projected_grad_opt(best_h, n_p, ts_p, alpha, n_iter=2000, lr_init=0.001, time_fn=time_left)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            if time_left() > 25:
                h_sl = slsqp_smooth(best_h, n_p, ts_p, alpha, maxiter=1000)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()

    # Phase 7: Second stochastic pass with smaller steps
    if time_left() > 60:
        n_s2 = best_n
        ts_s2 = n_s2 / 2.0
        h_st2, c5_st2 = stochastic_refine(best_h, n_s2, ts_s2, rng, 30000, 0.003, 0.1, time_left)
        if c5_st2 < best_c5:
            best_c5 = c5_st2
            best_h = h_st2.copy()

    # Phase 8: Final SLSQP polish
    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = slsqpSmooth = slsqp_smooth(best_h, n_f, ts_f, 50000, maxiter=2000)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: Tiny stochastic nudge
    if time_left() > 25:
        n_t = best_n
        ts_t = n_t / 2.0
        h_t, c5_t = stochastic_refine(best_h, n_t, ts_t, rng, 15000, 0.002, 0.05, time_left)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
