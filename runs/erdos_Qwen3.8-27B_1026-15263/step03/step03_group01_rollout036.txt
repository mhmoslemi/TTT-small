    return starts

    t_start = time.time()
    global _grad_tmpl
    _grad_tmpl = None

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            iv = globals().get('initial_h_values', None)
            if iv is not None:
                best_h = np.array(iv, dtype=float)
                best_n = len(best_h)
                best_h = project_to_sum(best_h, best_n / 2.0)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: Upsample to n=960
    target_n = 960
    if target_n != best_n and tl() > 100:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, target_n / 2.0)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
            best_n = target_n
        else:
            best_h = h_up
            best_n = target_n

    n = best_n
    ts = n / 2.0

    # Phase 2: SLSQP continuation at target resolution
    for alpha in [20, 80, 300, 1000]:
        if tl() < 40:
            break
        h_new = slsqp_opt(best_h, n, ts, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Projected gradient (long run)
    if tl() > 60:
        h_pg = proj_grad_opt(best_h, 3000, n_iter=5000, lr_init=0.03)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: Frank-Wolfe for structure exploration
    if tl() > 50:
        for alpha in [50, 200, 800]:
            if tl() < 30:
                break
            h_fw = frank_wolfe(best_h, alpha, n_iter=200)
            c5_fw = compute_c5(h_fw)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw.copy()
        # Polish after FW
        if tl() > 30:
            h_pol = slsqp_opt(best_h, n, ts, 1000, maxiter=800)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Fourier Nelder-Mead for global exploration
    if tl() > 80:
        for n_modes in [15, 25, 40]:
            if tl() < 40:
                break
            h_nm = fourier_nm_opt(best_h, n_modes, maxiter=3000)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()
            # Polish the Fourier result
            if tl() > 30:
                h_pol = slsqp_opt(best_h, n, ts, 500, maxiter=600)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 6: Lag-focused optimization
    if tl() > 60:
        for n_top in [8, 15, 25]:
            if tl() < 30:
                break
            h_lf = lag_focused_opt(best_h, 800, n_top=n_top, maxiter=500, n_rounds=3)
            c5_lf = compute_c5(h_lf)
            if c5_lf < best_c5:
                best_c5 = c5_lf
                best_h = h_lf.copy()

    # Phase 7: Diverse restarts
    if tl() > 80:
        starts = make_structured_starts(n, ts, rng, best_h, best_n)
        for h_s in starts[:8]:
            if tl() < 20:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for alpha in [50, 300]:
                if tl() < 15:
                    break
                h_o = slsqp_opt(h_s, n, ts, alpha, maxiter=500)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_s = h_o

    # Phase 8: Stochastic CD (large steps)
    if tl() > 80:
        h_cd, c5_cd = stochastic_cd(best_h, rng, n_steps=60000, delta_range=(0.01, 0.4))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if tl() > 30:
            h_pol = slsqp_opt(best_h, n, ts, 2000, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 9: Projected gradient (second round, small lr)
    if tl() > 40:
        h_pg2 = proj_grad_opt(best_h, 5000, n_iter=3000, lr_init=0.01)
        c5_pg2 = compute_c5(h_pg2)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()

    # Phase 10: Stochastic CD (fine steps)
    if tl() > 50:
        h_cd2, c5_cd2 = stochastic_cd(best_h, rng, n_steps=40000, delta_range=(0.005, 0.15))
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        if tl() > 30:
            h_pol = slsqp_opt(best_h, n, ts, 5000, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 11: FW on top lags only
    if tl() > 40:
        corr = np.correlate(best_h, 1.0 - h, mode="full")
        top_lags = np.argsort(corr)[-20:]
        h_fw2 = frank_wolfe(best_h, 2000, n_iter=300, lag_subset=top_lags)
        c5_fw2 = compute_c5(h_fw2)
        if c5_fw2 < best_c5:
            best_c5 = c5_fw2
            best_h = h_fw2.copy()
        if tl() > 20:
            h_pol = slsqp_opt(best_h, n, ts, 3000, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 12: Final high-alpha SLSQP polish
    if tl() > 25:
        for alpha in [5000, 15000, 50000]:
            if tl() < 15:
                break
            h_f = slsqp_opt(best_h, n, ts, alpha, maxiter=2000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 13: Tiny stochastic refinements
    if tl() > 25:
        h_t, c5_t = stochastic_cd(best_h, rng, n_steps=20000, delta_range=(0.003, 0.08))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final validation and cleanup
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
