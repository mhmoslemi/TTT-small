import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_all(h, n):
    nm1 = n - 1
    B_all = np.zeros((n, 2 * n - 1))
    for m in range(n):
        kc1s = max(0, nm1 - m)
        kc1e = min(2 * n - 2, 2 * nm1 - m)
        if kc1s <= kc1e:
            h_start = kc1s + m - n + 1
            h_end = kc1e + m - n + 1
            B_all[m, kc1s:kc1e + 1] += 1.0 - h[h_start:h_end + 1]
        kc2s = max(0, m)
        kc2e = min(2 * n - 2, m + nm1)
        if kc2s <= kc2e:
            h_start = m + nm1 - kc2e
            h_end = m + nm1 - kc2s + 1
            B_all[m, kc2s:kc2e + 1] += -h[h_start:h_end][::-1]
    return B_all

def update_B_all_fast(B_all, j, delta, n):
    nm1 = n - 1
    m = np.arange(n)
    k1 = j - m + nm1
    valid1 = (k1 >= 0) & (k1 < 2 * n - 1)
    B_all[m[valid1], k1[valid1]] -= delta
    k2 = m - j + nm1
    valid2 = (k2 >= 0) & (k2 < 2 * n - 1)
    B_all[m[valid2], k2[valid2]] -= delta

def all_pairs_sweep(h, corr, B_all, n, rng, n_grid=10, n_bisect=20, top_lags=12):
    nm1 = n - 1
    improved = False
    curr_max = np.max(corr)
    order = np.arange(n)
    rng.shuffle(order)

    for j in order:
        D_all = B_all[j][None, :] - B_all
        delta_lo = np.maximum(-h[j], h - 1.0)
        delta_hi = np.minimum(1.0 - h[j], h)
        feasible = (delta_hi - delta_lo > 1e-13)
        feasible[j] = False
        if not np.any(feasible):
            continue

        k_feas = np.where(feasible)[0]
        nf = len(k_feas)
        d_lo_f = delta_lo[k_feas]
        d_hi_f = delta_hi[k_feas]
        D_f = D_all[k_feas]

        fracs = np.linspace(0.0, 1.0, n_grid)
        delta_grid = d_lo_f[:, None] + (d_hi_f - d_lo_f)[:, None] * fracs[None, :]

        top_idx = np.argsort(-corr)[:top_lags]
        D_top = D_f[:, top_idx]
        corr_top = corr[top_idx]
        change_top = delta_grid[:, :, None] * D_top[:, None, :]
        new_max_top = np.max(corr_top[None, None, :] + change_top, axis=2)
        quick_best = new_max_top.min(axis=1)
        threshold = curr_max - 1e-11
        cands = np.where(quick_best < threshold)[0]

        if len(cands) == 0:
            continue

        D_cand = D_f[cands]
        delta_cand = delta_grid[cands]
        change_full = delta_cand[:, :, None] * D_cand[:, None, :]
        new_max_full = np.max(corr[None, None, :] + change_full, axis=2)
        flat_idx = int(np.argmin(new_max_full))
        best_ci = flat_idx // n_grid
        best_g = flat_idx % n_grid
        best_val = float(new_max_full[best_ci, best_g])

        if best_val >= curr_max - 1e-11:
            continue

        best_k = int(k_feas[cands[best_ci]])
        D_best = D_all[best_k]

        lo = float(delta_cand[best_ci, max(0, best_g - 1)]) if best_g > 0 else float(d_lo_f[cands[best_ci]])
        hi = float(delta_cand[best_ci, min(n_grid - 1, best_g + 1)]) if best_g < n_grid - 1 else float(d_hi_f[cands[best_ci]])

        for _ in range(n_bisect):
            mid = (lo + hi) * 0.5
            if np.max(corr + D_best * mid) <= np.max(corr + D_best * lo):
                lo = mid
            else:
                hi = mid

        best_delta = (lo + hi) * 0.5
        new_max_final = float(np.max(corr + D_best * best_delta))

        if new_max_final < curr_max - 1e-12:
            corr += best_delta * D_best
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0.0:
                h[j] = 0.0
            elif h[j] > 1.0:
                h[j] = 1.0
            if h[best_k] < 0.0:
                h[best_k] = 0.0
            elif h[best_k] > 1.0:
                h[best_k] = 1.0
            update_B_all_fast(B_all, j, best_delta, n)
            update_B_all_fast(B_all, best_k, -best_delta, n)
            curr_max = new_max_final
            improved = True

    return h, corr, B_all, improved

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -400.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha

    nm1 = n - 1
    j_idx = np.arange(n)
    k_idx = np.arange(2 * n - 1)
    j_row = j_idx[:, None]
    k_col = k_idx[None, :]
    idx1 = j_row + k_col - nm1
    idx2 = j_row - k_col + nm1
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    cl1 = np.clip(idx1, 0, n - 1)
    cl2 = np.clip(idx2, 0, n - 1)
    term1 = valid1 * (1.0 - h[cl1])
    term2 = valid2 * h[cl2]
    grad = dx * np.sum(w[None, :] * (term1 - term2), axis=1)
    return obj, grad

def nesterov_search(h0, n, target_sum, alpha_start, alpha_end, lr_start, n_steps):
    h = h0.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_start
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        _, grad = smooth_grad(h, alpha)
        h_proposed = h_prev - lr * (grad + 0.5 * (h - h_prev))
        h_new = project_to_sum(h_proposed, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        lr *= 0.998
    return best_h, best_c5

def pair_descent_sweep(h, corr, B_all, n, rng, n_cand=10):
    nm1 = n - 1
    improved = False
    best_val = np.max(corr)
    order = np.arange(n)
    rng.shuffle(order)
    for j in order:
        B_j = B_all[j]
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(int(rng.randint(n)))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            D = B_j - B_all[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            t_grid = np.linspace(delta_lo, delta_hi, 8)
            vals = np.max(corr[None, :] + t_grid[:, None] * D[None, :], axis=1)
            bi = int(np.argmin(vals))
            lo_r = t_grid[max(0, bi - 1)]
            hi_r = t_grid[min(len(t_grid) - 1, bi + 1)]
            for _ in range(20):
                mid = (lo_r + hi_r) * 0.5
                if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
                    lo_r = mid
                else:
                    hi_r = mid
            delta_opt = (lo_r + hi_r) * 0.5
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = max(0.0, min(1.0, h[j]))
            h[best_k] = max(0.0, min(1.0, h[best_k]))
            update_B_all_fast(B_all, j, best_delta, n)
            update_B_all_fast(B_all, best_k, -best_delta, n)
            improved = True
    return h, corr, B_all, improved

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: All-pairs vectorized descent (50% budget)
    t_p1 = t_start + 0.50 * budget_s
    h_w = best_h.copy()
    n_w = len(h_w)
    corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
    B_all_w = compute_B_all(h_w, n_w)
    stall = 0
    while time.time() < t_p1:
        h_w, corr_w, B_all_w, imp = all_pairs_sweep(h_w, corr_w, B_all_w, n_w, rng, n_grid=10, n_bisect=20, top_lags=12)
        c5_w = compute_c5(h_w)
        if c5_w < best_c5 - 1e-12:
            best_c5 = c5_w
            best_h = h_w.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall >= 150:
            noise = 0.04
            h_w = project_to_sum(h_w + rng.uniform(-noise, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            B_all_w = compute_B_all(h_w, n_w)
            stall = 0
        elif stall >= 80:
            noise = 0.015
            h_w = project_to_sum(h_w + rng.uniform(-noise, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            B_all_w = compute_B_all(h_w, n_w)

    # Phase 2: Nesterov smooth gradient (15% budget)
    t_p2 = t_start + 0.65 * budget_s
    n_nm = len(best_h)
    ts_nm = n_nm / 2.0
    n_steps_est = int((t_p2 - time.time()) / 0.003)
    n_steps_est = max(50, min(5000, n_steps_est))
    h_nm, c5_nm = nesterov_search(best_h, n_nm, ts_nm, 30.0, 300.0, 0.02, n_steps_est)
    if c5_nm < best_c5:
        best_c5 = c5_nm
        best_h = h_nm.copy()
        best_n = n_nm

    # Phase 3: Quick pair descent after Nesterov (10% budget)
    t_p3 = t_start + 0.75 * budget_s
    h_q = best_h.copy()
    n_q = len(h_q)
    corr_q = np.correlate(h_q, 1.0 - h_q, mode="full")
    B_all_q = compute_B_all(h_q, n_q)
    while time.time() < t_p3:
        h_q, corr_q, B_all_q, _ = pair_descent_sweep(h_q, corr_q, B_all_q, n_q, rng, n_cand=12)
        c5_q = compute_c5(h_q)
        if c5_q < best_c5 - 1e-12:
            best_c5 = c5_q
            best_h = h_q.copy()
            best_n = n_q

    # Phase 4: Multi-resolution exploration (15% budget)
    t_p4 = t_start + 0.90 * budget_s
    resolutions = [160, 200, 280, 320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_p4:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_st = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
        B_all_st = compute_B_all(h_st, n_tr)
        max_sweeps = max(3, int((t_p4 - time.time()) / 3.0))
        for _ in range(max_sweeps):
            if time.time() >= t_p4:
                break
            h_st, corr_st, B_all_st, _ = pair_descent_sweep(h_st, corr_st, B_all_st, n_tr, rng, n_cand=8)
        c5_st = compute_c5(h_st)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            best_n = n_tr

    # Phase 5: Final all-pairs polish (10% budget)
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    B_all_fin = compute_B_all(h_fin, n_fin)
    t_p5 = t_start + budget_s - 5
    while time.time() < t_p5:
        h_fin, corr_fin, B_all_fin, _ = all_pairs_sweep(h_fin, corr_fin, B_all_fin, n_fin, rng, n_grid=8, n_bisect=16, top_lags=10)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
