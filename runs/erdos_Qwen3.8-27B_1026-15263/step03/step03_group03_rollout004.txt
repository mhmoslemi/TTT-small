        worst_lags = np.argsort(corr)[-top_k_lags:]
        for wl in worst_lags:
            k_set.append((j + wl) % n)
        
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k]-1.0)
            delta_hi = min(1.0-h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            # Solve 1D: min delta of max(corr + delta*D)
            # Use ternary search or grid
            t_grid = np.linspace(delta_lo, delta_hi, 16)
            vals = np.max(corr[None,:] + D[None,:]*t_grid[:,None], axis=1)
            best_idx = int(np.argmin(vals))
            # Refine
            if best_idx == 0:
                delta_opt = delta_lo
            elif best_idx == len(t_grid)-1:
                delta_opt = delta_hi
            else:
                lo_r = t_grid[best_idx-1]
                hi_r = t_grid[best_idx+1]
                for _ in range(24):
                    mid = (lo_r+hi_r)*0.5
                    if np.max(corr + D*mid) <= np.max(corr + D*lo_r):
                        lo_r = mid
                    else:
                        hi_r = mid
                delta_opt = (lo_r+hi_r)*0.5
            val_opt = np.max(corr + D*delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def momentum_grad_descent(h0, n, target_sum, budget_s, rng):
    """Momentum-based gradient descent on smooth max of correlation."""
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    lr = 0.005
    lr_decay = 0.999
    alpha = 30  # smooth max parameter
    alpha_growth = 1.002
    t_start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    while time.time() - t_start < budget_s:
        corr = np.correlate(h, 1.0-h, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted = np.clip(shifted, -400, 0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient
        nm1 = n - 1
        grad = np.zeros(n)
        # For each significant weight, compute contribution
        # Use top-K for efficiency
        K = min(50, 2*n-1)
        if K < 2*n-1:
            top_idx = np.argpartition(w, -K)[-K:]
        else:
            top_idx = np.arange(2*n-1)
        
        for ki in top_idx:
            wi = w[ki]
            if wi < 1e-10:
                continue
            l = ki - nm1  # lag
            if l >= 0:
                # First term: (1-h[i+l]) for i in [0, n-1-l]
                grad[:n-l] += wi * (1.0 - h[l:])
                # Second term: -h[i-l] for i in [l, n-1] (since i-l in [0, n-1])
                if l < n:
                    grad[l:] -= wi * h[:n-l]
            else:
                ll = -l
                # First term: (1-h[i-ll]) for i in [ll, n-1]
                grad[ll:] += wi * (1.0 - h[:n-ll])
                # Second term: -h[i+ll] for i in [0, n-1-ll]
                grad[:n-ll] -= wi * h[ll:]
        
        # Update with momentum
        v = beta * v - lr * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        
        # Anneal
        lr *= lr_decay
        alpha *= alpha_growth
    
    return best_h, best_c5

For h = 1/2 everywhere: corr[k] = sum (1/2)(1/2) = n/4 for all k. C₅ = (n/4) * (2/n) = 1/2.

For h being a "block" function: h = 1 on [0,1], h = 0 on [1,2]. Then:
- At shift 0: corr = sum h(h-1) = 0 (h is 0 or 1)

Actually, let me use a simpler but effective approach: **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)** adapted for our constraint. FISTA uses Nesterov acceleration and is O(1/k^2) convergent for convex problems. Our problem isn't convex (max of non-convex functions), but FISTA still works well in practice.

The key is: at each step, compute the gradient of the smooth objective, take a step, then project onto the constraint set {0≤h≤1, sum(h)=n/2}.

For the projection, the exact algorithm is:
1. clip to [0,1]
2. If sum < n/2: add (n/2-sum)/n to all, clip to [0,1], repeat
3. If sum > n/2: subtract (sum-n/2)/n from all, clip to [0,1], repeat

This converges in at most 2 iterations in practice.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            h += diff / len(h)
        else:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    lo1 = max(0, nm1 - j)
    hi1 = min(2 * n - 2, 2 * nm1 - j)
    if lo1 <= hi1:
        idx = j + np.arange(lo1, hi1 + 1) - nm1
        B[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, j)
    hi2 = min(2 * n - 2, j + nm1)
    if lo2 <= hi2:
        idx = j - np.arange(lo2, hi2 + 1) + nm1
        B[lo2:hi2 + 1] -= h[idx]
    return B


def pair_descent_sweep(h, corr, rng, n_cand=10, top_k_lags=5):

        if c5_4 < best_c5 - 1e-12:
            best_c5 = c5_4
            best_h = h4.copy()
            best_n = n4
            stall = 0
        else:
            stall += 1
        if stall == 50:
            noise_level = 0.02
        elif stall == 150:
            noise_level = 0.05
        elif stall == 300:
            noise_level = 0.10
        elif stall == 500:
            noise_level = 0.15
        if stall > 0 and stall % 50 == 0:
            h4 = project_to_sum(h4 + rng.uniform(-noise_level, noise_level, n4), n4 / 2.0)
            corr4 = np.correlate(h4, 1.0 - h4, mode="full")
        if stall > 600:
            h4 = best_h.copy()
            corr4 = np.correlate(h4, 1.0 - h4, mode="full")
            stall = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
