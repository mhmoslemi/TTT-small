_g_alpha = 1.0

def _c_fun(x):
    return np.sum(x) - _g_target_sum

def _c_jac(x):
    return np.ones(_g_n_pts)

def _obj(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _g_alpha)
    return v

def _grad(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _g_alpha)
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kmax1 = min(j, n - 1)
        if kmax1 >= 0:
            ks = np.arange(kmax1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kmax2 = n - j
        if kmax2 > 0:
            ks2 = np.arange(kmax2)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _g_target_sum, _g_n_pts, _g_alpha
    _g_target_sum = target_sum
    _g_n_pts = n
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def min_max_linears(a, b, n_eval=200):
    """Find t in [0,1] minimizing max_k (a[k] + b[k]*t).
    a and b are arrays of length m (number of lags).
    Returns optimal t and the value.
    """
    # Use ternary search since the function is convex
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a + b * m1)
        f2 = np.max(a + b * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    f_opt = np.max(a + b * t_opt)
    return t_opt, f_opt

def exact_coord_descent(h, n, target_sum, max_rounds=200, time_limit=None, t_start=None):
    """Exact cyclic coordinate descent."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * (2.0 / n)
    n_lags = 2 * n - 1
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        for i in range(n):
            # d_k = (1-h[i+k-n+1])[valid] - h[i-k+n-1][valid]
            # k ranges from 0 to 2n-2, lag = k-n+1 ranges from -(n-1) to (n-1)
            
            # For the h[i] term in corr[k]:
            # j=i contributes h[i]*(1-h[i+lag]) where lag=k-n+1
            #   valid if 0 <= i+lag < n, i.e., 0 <= i+k-n+1 < n
            # j+lag=i, i.e., j=i-lag=i-k+n-1, contributes h[i-k+n-1]*(1-h[i])
            #   valid if 0 <= i-k+n-1 < n
            
            lags = np.arange(n_lags) - (n - 1)  # lags[k] = k - n + 1
            idx1 = i + lags  # for (1-h[i+lag])
            idx2 = i - lags  # for h[i-lag]
            
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            # d_k = (1-h[i+lag])*valid1 - h[i-lag]*valid2
            h_at_idx1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
            h_at_idx2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
            
            d = (1.0 - h_at_idx1) * valid1 - h_at_idx2 * valid2
            
            # a_k = corr[k] - h[i] * d_k
            a = corr - h[i] * d
            
            # Find t* minimizing max_k (a_k + t * d_k)
            t_opt, _ = min_max_linears(a, d, n_eval=60)
            
            if abs(t_opt - h[i]) > 1e-12:
                h[i] = t_opt
                corr = a + t_opt * d  # update corr efficiently
                improved = True
        
        if not improved:
            break
    
    h = project_to_sum(h, target_sum)
    c5 = compute_c5(h)
    return h, c5

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_start=None, time_limit=None):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for _ in range(n_iters):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if c5_new > best_c5:
            lr *= 0.95
    return best_h, best_c5

    return v

def _smg(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    sm = m + np.log(s) / alpha - np.log(len(corr)) / alpha
    g = np.zeros(n)
    for j in range(n):
        k1 = min(j, n - 1)
        if k1 >= 0:
            ks = np.arange(k1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        k2 = n - j
        if k2 > 0:
            ks2 = np.arange(k2)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        l1 = n - j - 1
        if l1 >= 1:
            ls = np.arange(1, l1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        l2 = j
        if l2 >= 1:
            ls2 = np.arange(1, l2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def proj(h, ts):

    for _ in range(100):
        diff = ts - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp(h0, n, ts, alpha, maxit):
    global _g_ts, _g_np, _g_al
    _g_ts = ts
    _g_np = n
    _g_al = alpha
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]

        res = minimize(_of, h0.copy(), jac=_gf, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-15})
        return proj(res.x, ts)
    except Exception:
        return proj(h0, ts)

def _mml(a, b):
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = a + b * mid
        fm = np.max(vals)
        tol = 1e-13 * max(1.0, abs(fm))
        act = vals > fm - tol
        if not np.any(act):
            act = vals == fm
        bs = b[act]
        if np.min(bs) > 1e-12:
            hi = mid
        elif np.max(bs) < -1e-12:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def excd(h, n, ts, maxrd, t0, dl):

    lags = np.arange(nl) - (n - 1)

        if time.time() > dl:
            break
        order = np.arange(n) if rnd % 2 == 0 else np.arange(n - 1, -1, -1)
        for i in order:
            idx1 = i + lags
            idx2 = i - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            h1 = np.where(v1, h[np.clip(idx1, 0, n - 1)], 0.0)
            h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
            d = (1.0 - h1) * v1.astype(float) - h2 * v2.astype(float)
            a = corr - h[i] * d
            topt = _mml(a, d)
            delta = topt - h[i]

                h[i] = topt
                corr = corr + delta * d
        h = proj(h, ts)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5c = float(np.max(corr) * (2.0 / n))
        if c5c < c5b - 1e-14:
            c5b = c5c
            hb = h.copy()
    return hb, c5b

def adam_run(h0, n, ts, alpha, niter, lr0, t0, dl):

    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0

            bc5 = c5n
            bh = hn.copy()
        h = hn
        if c5n > bc5:
            lr *= 0.95
    return bh, bc5
