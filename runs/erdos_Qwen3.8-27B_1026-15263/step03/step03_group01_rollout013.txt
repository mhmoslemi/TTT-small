def adam_proj(h0, n, ts, alpha, n_iter, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):

def slsqp_opt(h0, n, ts, alpha, maxiter=500):

        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,

        return project_to_sum(r.x, ts)
    except Exception:

        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at current resolution with continuation
    n1 = best_n
    ts1 = n1 / 2.0
    alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if time_left() < 40:
            break
        h_new, _ = proj_grad_with_momentum(best_h, n1, ts1, alpha, n_iter=3000, lr_init=0.03, momentum=0.85)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Adam at the best resolution
    if time_left() > 120:
        h_adam, _ = adam_proj(best_h, n1, ts1, 5000, n_iter=5000, lr=0.005)
        c5_adam = compute_c5(h_adam)
        if c5_adam < best_c5:
            best_c5 = c5_adam
            best_h = h_adam.copy()

    # Phase 3: Upsample to higher resolution
    target_ns = [720, 960, 1200]
    for n_tr in target_ns:
        if time_left() < 100:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        h_up = upsample_h(best_h, n_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr
        else:
            best_n = n_tr
            best_h = h_up.copy()
        for alpha in [50, 200, 1000, 5000]:
            if time_left() < 50:
                break
            h_opt, _ = proj_grad_with_momentum(h_up, n_tr, ts_tr, alpha, n_iter=2000, lr_init=0.02, momentum=0.85)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_up = h_opt
        if time_left() > 60:
            h_adv, _ = adam_proj(h_up, n_tr, ts_tr, 3000, n_iter=3000, lr=0.003)
            c5_adv = compute_c5(h_adv)
            if c5_adv < best_c5:
                best_c5 = c5_adv
                best_h = h_adv.copy()
                best_n = n_tr

    # Phase 4: Frequency perturbation and re-optimization
    n_b = best_n
    ts_b = n_b / 2.0
    n_trials = max(10, int(time_left() / 40))
    for trial in range(n_trials):
        if time_left() < 35:
            break
        amp = 0.06 * (1.0 - trial / max(n_trials, 1)) + 0.01
        nf = max(3, n_b // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        for alpha in [200, 2000]:
            if time_left() < 25:
                break
            h_ot, _ = proj_grad_with_momentum(h_sh, n_b, ts_b, alpha, n_iter=1500, lr_init=0.015, momentum=0.85)
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            h_sh = h_ot

    # Phase 5: Diverse restarts at current resolution
    n_e = best_n
    ts_e = n_e / 2.0
    n_restarts = max(5, int(time_left() / 60))
    for trial in range(n_restarts):
        if time_left() < 40:
            break
        starts = make_structured_starts(n_e, ts_e, rng, best_h, best_n)
        for h_s in starts[:5]:
            if time_left() < 25:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_e
            for alpha in [100, 1000]:
                if time_left() < 20:
                    break
                h_o, _ = proj_grad_with_momentum(h_s, n_e, ts_e, alpha, n_iter=1000, lr_init=0.02, momentum=0.85)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                    best_n = n_e
                h_s = h_o

    # Phase 6: SLSQP polish at high alpha
    if time_left() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        if n_f <= 960:
            for alpha in [3000, 10000]:
                if time_left() < 30:
                    break
                h_f = slsqp_opt(best_h, n_f, ts_f, alpha, maxiter=1000)
                c5_f = compute_c5(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()

    # Phase 7: Coordinate descent on the exact objective
    if time_left() > 80:
        n_cd = min(best_n, 960)
        if n_cd != best_n:
            h_cd = upsample_h(best_h, n_cd)
        else:
            h_cd = best_h.copy()
        ts_cd = n_cd / 2.0
        n_steps = max(20000, int((time_left() - 40) * 400))
        h_cd, c5_cd = coord_descent_c5(h_cd, n_cd, ts_cd, rng, n_steps=n_steps, delta_max=0.2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_cd

    # Phase 8: Final fine coordinate descent
    if time_left() > 50:
        n_cd2 = best_n
        ts_cd2 = n_cd2 / 2.0
        n_steps2 = max(10000, int((time_left() - 30) * 300))
        h_cd2, c5_cd2 = coord_descent_c5(best_h, n_cd2, ts_cd2, rng, n_steps=n_steps2, delta_max=0.1)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 9: Final gradient polish
    if time_left() > 30:
        n_p = best_n
        ts_p = n_p / 2.0
        h_p, _ = proj_grad_with_momentum(best_h, n_p, ts_p, 10000, n_iter=2000, lr_init=0.005, momentum=0.9)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 10: Tiny ADMM-like refinement
    if time_left() > 25:
        n_t = best_n
        ts_t = n_t / 2.0
        h_t, c5_t = coord_descent_c5(best_h, n_t, ts_t, rng, n_steps=10000, delta_max=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
