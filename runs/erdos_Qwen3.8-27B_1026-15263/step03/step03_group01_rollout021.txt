def topk_smooth_obj_grad(h, k, alpha):
    """Smooth top-k objective using softmax weighting."""
    n = len(h)
    corr = compute_corr(h)
    nc = len(corr)
    
    # Soft top-k weighting
    idx = np.argpartition(-corr, k - 1)[:k]
    threshold = np.min(corr[idx])
    w = 1.0 / (1.0 + np.exp(-alpha * (corr - threshold)))
    w = np.maximum(w, 1e-15)
    w = w / np.sum(w)
    
    obj = np.dot(w, corr)
    
    # Same gradient computation as topk_grad but with weights
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    active = w > 1e-8
    if not np.any(active):
        return obj, grad
    active_idx = np.where(active)[0]
    for j in active_idx:
        if w[j] < 1e-10:
            continue
        d = j - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n - 1, n - 1 + d)
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        
        if lo1 <= hi1:
            m = np.arange(lo1, hi1 + 1)
            grad[m] += w[j] * (1.0 - h[m - d])
        if lo2 <= hi2:
            m = np.arange(lo2, hi2 + 1)
            grad[m] -= w[j] * h[m + d]
    
    return obj, grad

    lr = lr_init
    obj_prev = float('inf')
    stall = 0
    
    for i in range(n_iter):
        grad = topk_grad(h, k)
        v = momentum * v - lr * grad
        h_new = project_to_sum(h + v, ts)

        corr_new = compute_corr(h_new)
        idx = np.argpartition(-corr_new, k - 1)[:k]
        obj_new = np.mean(corr_new[idx])

            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    
    return h

def active_set_newton(h0, n, ts, tol=1e-10, max_iter=50):
    """Newton refinement using active set identification."""
    h = h0.copy()
    for iteration in range(max_iter):
        corr = compute_corr(h)
        nc = len(corr)
        max_val = np.max(corr)
        # Active set: all lags within tol of max
        active = np.where(corr >= max_val - tol)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient of mean of active correlations
        grad = np.zeros(n)
        for j in active:
            d = j - (n - 1)
            lo1 = max(0, d)
            hi1 = min(n - 1, n - 1 + d)
            lo2 = max(0, -d)
            hi2 = min(n - 1, n - 1 - d)
            if lo1 <= hi1:
                m = np.arange(lo1, hi1 + 1)
                grad[m] += (1.0 - h[m - d])
            if lo2 <= hi2:
                m = np.arange(lo2, hi2 + 1)
                grad[m] -= h[m + d]
        grad /= len(active)
        
        # Compute Hessian (approximate) - this is the Gram matrix of the active correlations
        # H = (1/|A|) * sum_{j in A} J_j^T J_j where J_j is the Jacobian of corr[j]
        # For bilinear corr[j] = sum h[a]*h[b] - sum h[a], the Hessian is sparse
        
        # Actually, let's use a simple line search with the negative gradient
        # (since exact Hessian is expensive)
        
        # Line search
        lr = 0.01
        best_h = h.copy()
        best_obj = np.mean(corr[active])
        for ls in range(20):
            h_trial = project_to_sum(h - lr * grad, ts)
            corr_trial = compute_corr(h_trial)
            obj_trial = np.mean(corr_trial[active])
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_h = h_trial.copy()
            else:
                lr *= 0.5
                if lr < 1e-15:
                    break
        
        if np.max(best_obj - corr[active]) < tol and best_obj < np.max(corr):
            h = best_h
        else:
            h = best_h
            if np.linalg.norm(best_h - h) < 1e-12:
                break
    
    return h

            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
    
    return h, c5_cur

def multiobjective_local_search(h0, n, ts, rng, n_steps=10000):

    if time_left() > 80:
        n3 = best_n
        ts3 = n3 / 2.0
        h3, c5_3, _ = targeted_2swap(best_h, n3, ts3, n_steps=50000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Upsample to higher resolution and optimize
    for n_target in [600, 720, 840]:
        if time_left() < 100:
            break
        if n_target % 2 != 0:
            continue
        ts_t = n_target / 2.0
        h_up = upsample_h(best_h, n_target)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_target
        else:
            best_n = n_target

        # Optimize at this resolution
        for k in [1, 3, 5]:
            if time_left() < 40:
                break
            h_opt = projected_grad_opt(h_up, n_target, ts_t, k, n_iter=3000, lr_init=0.02, momentum=0.9)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_up = h_opt

        if time_left() > 50:
            h_sm = smooth_grad_opt(h_up, n_target, ts_t, 5, 50.0, n_iter=1000, lr_init=0.01)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()
            h_up = h_sm

        if time_left() > 40:
            h_ts, c5_ts, _ = targeted_2swap(h_up, n_target, ts_t, n_steps=30000)
            if c5_ts < best_c5:
                best_c5 = c5_ts
                best_h = h_ts.copy()
            h_up = h_ts

        best_n = n_target

    # Phase 5: Fine stochastic search at best resolution
    if time_left() > 60:
        n5 = best_n
        ts5 = n5 / 2.0
        h5, c5_5 = fine_stochastic(best_h, n5, ts5, rng, n_steps=40000, delta_max=0.02)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Another round of targeted search + gradient
    if time_left() > 80:
        n6 = best_n
        ts6 = n6 / 2.0
        h6, c5_6, _ = targeted_2swap(best_h, n6, ts6, n_steps=40000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        if time_left() > 40:
            h6b = projected_grad_opt(best_h, n6, ts6, 1, n_iter=3000, lr_init=0.02, momentum=0.9)
            c5_6b = compute_c5(h6b)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Phase 7: SLSQP polish with smooth max
    if time_left() > 50:
        n7 = best_n
        ts7 = n7 / 2.0
        for alpha in [100, 300]:
            if time_left() < 30:
                break
            h7 = slsqp_polish(best_h, n7, ts7, alpha, maxiter=800)
            c5_7 = compute_c5(h7)
            if c5_7 < best_c5:
                best_c5 = c5_7
                best_h = h7.copy()

    # Phase 8: Very fine stochastic
    if time_left() > 40:
        n8 = best_n
        ts8 = n8 / 2.0
        h8, c5_8 = fine_stochastic(best_h, n8, ts8, rng, n_steps=30000, delta_max=0.01)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Final targeted micro-optimization
    if time_left() > 30:
        n9 = best_n
        ts9 = n9 / 2.0
        h9, c5_9, _ = targeted_2swap(best_h, n9, ts9, n_steps=20000)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Phase 10: Final gradient polish
    if time_left() > 20:
        n10 = best_n
        ts10 = n10 / 2.0
        h10 = projected_grad_opt(best_h, n10, ts10, 1, n_iter=2000, lr_init=0.01, momentum=0.95)
        c5_10 = compute_c5(h10)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    # Final validation and cleanup
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
