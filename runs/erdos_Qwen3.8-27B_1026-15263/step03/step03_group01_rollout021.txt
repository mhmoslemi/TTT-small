        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        w = np.exp(eta * corr) 
        w /= np.sum(w)

        nj = np.arange(n)[:, None]
        nm = np.arange(n_corr)[None, :]

        vs = (si >= 0) & (si < n)

            h = h_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.7
    
    return h

Now let me also think about the "targeted subgradient" approach. The idea is:
1. Find k* = argmax c_k
2. Compute ∂c_{k*}/∂h (this is a sparse vector!)
3. Take a step in the -∂c_{k*}/∂h direction

The gradient of c_{k*} is sparse because c_{k*} only involves a specific pair of shifted indices. For lag d = k* - (n-1):
- If d ≥ 0: c_d = ∑_{i=0}^{n-1-d} h[i+d](1-h[i]) * dx
  - ∂c_d/∂h[j] = dx * (1 - h[j-d]) for j = d, ..., n-1 (from the h[i+d] term)
  - ∂c_d/∂h[j] -= dx * h[j+d] for j = 0, ..., n-1-d (from the -h[j]h[j+d] term)
  
  So ∂c_d/∂h[j] = dx * [(1-h[j-d]) * 1_{j≥d} - h[j+d] * 1_{j≤n-1-d}]

This is O(n) to compute and very sparse in a sense (only n nonzero entries, but they're all potentially nonzero).

For the "subgradient with smoothing" approach, I can:
1. Find the top m lags (say m=5)
2. Weight them by exp(α * c_k)
3. Compute the weighted gradient
4. Take a projected step

This is essentially what the log-sum-exp gradient does, but I can make it more aggressive by using a larger effective alpha or by only considering the top m.

        h_up3 = upsample_h(best_h, best_n, n3)
        c5_up3 = compute_c5(h_up3)
        if c5_up3 < best_c5:
            best_c5 = c5_up3
            best_h = h_up3.copy()
            best_n = n3
        for alpha in [50, 200, 800, 3000]:
            if time_left() < 35:
                break
            h_opt3 = lbfgsb_opt(best_h, n3, ts3, alpha, maxiter=1500)
            c5_opt3 = compute_c5(h_opt3)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
            best_h_for_dns = h_opt3
        if time_left() > 50:
            h_da3, c5_da3 = dual_ascent_opt(best_h, n3, t_start, budget_remaining=55, eta0=100, n_iter=4000)
            if c5_da3 < best_c5:
                best_c5 = c5_da3
                best_h = h_da3.copy()
                best_n = n3

    # Phase 4: Targeted subgradient on best resolution
    n4 = best_n
    if time_left() > 40:
        h_tg, c5_tg = targeted_subgrad(best_h, n4, t_start, budget_remaining=60, n_iter=3000)
        if c5_tg < best_c5:
            best_c5 = c5_tg
            best_h = h_tg.copy()
            best_n = n4

    # Phase 5: Stochastic coordinate descent (long phase)
    n5 = best_n
    if time_left() > 60:
        h_cd, c5_cd = stochastic_cd(best_h, n5, t_start, budget_remaining=120, delta_range=(0.005, 0.3))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n5

    # Phase 6: L-BFGS-B polish after stochastic
    n6 = best_n
    ts6 = n6 / 2.0
    if time_left() > 40:
        for alpha in [1000, 5000]:
            if time_left() < 25:
                break
            h_pol = lbfgsb_opt(best_h, n6, ts6, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            best_h_for_dns = h_pol

    # Phase 7: Second stochastic pass with smaller deltas
    n7 = best_n
    if time_left() > 50:
        h_cd2, c5_cd2 = stochastic_cd(best_h, n7, t_start, budget_remaining=80, delta_range=(0.003, 0.12))
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n7

    # Phase 8: Final dual ascent
    n8 = best_n
    if time_left() > 40:
        h_da8, c5_da8 = dual_ascent_opt(best_h, n8, t_start, budget_remaining=50, eta0=200, n_iter=3000)
        if c5_da8 < best_c5:
            best_c5 = c5_da8
            best_h = h_da8.copy()
            best_n = n8

    # Phase 9: Final L-BFGS-B polish
    n9 = best_n
    ts9 = n9 / 2.0
    if time_left() > 25:
        h_f = lbfgsb_opt(best_h, n9, ts9, 5000, maxiter=2000)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n9

    # Phase 10: Tiny stochastic refinements
    n10 = best_n
    if time_left() > 25:
        h_t, c5_t = stochastic_cd(best_h, n10, t_start, budget_remaining=30, delta_range=(0.002, 0.06))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n10

    # Final validation and cleanup
    best_n = len(best_h)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
