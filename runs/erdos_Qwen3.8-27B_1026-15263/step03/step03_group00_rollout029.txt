def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def nesterov_smooth_max(h0, target_sum, alpha, max_iters, lr, momentum):
    n = len(h0)
    x = h0.copy()
    v = np.zeros(n)
    m = momentum
    best_x = x.copy()

            best_obj = obj
            best_x = x.copy()
        
        v = m * v - lr * grad

        x_new = project_to_sum(x_new, target_sum)
        
        x = x_new

    return best_x

    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[:n - k] += (1.0 - h[k:])
                g[k:] -= h[:n - k]
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def active_set_cd(h, target_sum, max_iters, top_m=3, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        top_lags = np.argsort(-corr)[:top_m]
        g = np.zeros(n)
        for lag_idx in top_lags:
            lag = int(lag_idx) - (n - 1)
            g += corr_gradient_single(h, lag)
        g /= top_m
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 30:
                step_frac *= 0.7
                stale = 0
                if step_frac < 0.01:
                    break
            else:
                move *= 0.5
                if move < 1e-10:
                    break
                h[i_inc] += move
                h[i_dec] -= move
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new >= c5 - 1e-14:
                    h[i_inc] -= move
                    h[i_dec] += move
                    h = np.clip(h, 0.0, 1.0)
                    h = project_to_sum(h, target_sum)
                    break
        c5 = c5_new
    return h, c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, target_sum, alpha, maxiter=500):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_max_and_grad(x, alpha)[0],
                       h0.copy(), method='SLSQP',
                       jac=lambda x: smooth_max_and_grad(x, alpha)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def ga_step(population, fitnesses, target_sum, n, rng, generation, total_gens):
    pop_size = len(population)
    new_pop = [population[0].copy()]
    progress = generation / max(1, total_gens)
    base_noise = 0.04 * (1.0 - 0.7 * progress) + 0.01
    for _ in range(pop_size - 1):
        cand = rng.choice(pop_size, 3, replace=False)
        p1_idx = cand[np.argmin([fitnesses[c] for c in cand])]
        if rng.random() < 0.65:
            cand2 = rng.choice(pop_size, 2, replace=False)
            p2_idx = cand2[np.argmin([fitnesses[c] for c in cand2])]
            alpha = 0.3 + 0.4 * rng.random()
            child = alpha * population[p1_idx] + (1.0 - alpha) * population[p2_idx]
        else:
            child = population[p1_idx].copy()
        child = project_to_sum(child, target_sum)
        noise = base_noise * (0.5 + rng.random())
        child = child + rng.normal(0, noise, n)
        child = project_to_sum(child, target_sum)
        new_pop.append(child)
    return new_pop

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 300
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    if time_left() < 30:
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_h = np.clip(best_h, 0.0, 1.0)
        return (best_h, compute_c5(best_h), best_n)

    h_cur = best_h.copy()
    ts_cur = ts_parent
    n_cur = n_parent

    for alpha in [200, 1000, 5000]:
        if time_left() < 50:
            break
        h_opt = slsqp_optimize(h_cur, ts_cur, alpha, maxiter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_cur
        h_cd, c5_cd = active_set_cd(h_opt, ts_cur, max_iters=8000, top_m=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_cur
        h_cur = best_h.copy()
        ts_cur = best_n / 2.0
        n_cur = best_n

    for n_tr in [n_cur + 60, n_cur + 120, n_cur + 200, n_cur + 300]:
        if n_tr % 2 != 0:
            n_tr += 1
        if time_left() < 100:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        h_cd, c5_cd = active_set_cd(h_interp, ts_tr, max_iters=8000, top_m=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_tr
        if time_left() > 60:
            h_opt = slsqp_optimize(best_h, best_n / 2.0, 1000, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = int(best_n)
            h_cd2, c5_cd2 = active_set_cd(best_h, best_n / 2.0, max_iters=5000, top_m=3)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    n_best = best_n
    ts_best = n_best / 2.0

    if time_left() > 200:
        pop_size = 18
        total_gens = 150
        population = [best_h.copy()]
        for i in range(1, pop_size):
            noise = 0.02 + 0.10 * rng.random()
            h_mut = best_h + rng.normal(0, noise, n_best)
            population.append(project_to_sum(h_mut, ts_best))
        fitnesses = [compute_c5(h) for h in population]

        for gen in range(total_gens):
            if time_left() < 80:
                break
            order = np.argsort(fitnesses)
            population = [population[i] for i in order]
            fitnesses = [fitnesses[i] for i in order]
            if fitnesses[0] < best_c5:
                best_c5 = fitnesses[0]
                best_h = population[0].copy()
            population[0] = best_h.copy()
            fitnesses[0] = best_c5

            if time_left() > 40:
                h_ref, c5_ref = active_set_cd(population[0], ts_best, max_iters=3000, top_m=3)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                population[0] = best_h.copy()
                if len(population) > 1 and time_left() > 40:
                    h_ref2, c5_ref2 = active_set_cd(population[1], ts_best, max_iters=2000, top_m=3)
                    if c5_ref2 < best_c5:
                        best_c5 = c5_ref2
                        best_h = h_ref2.copy()
                    population[1] = h_ref2.copy()

            population = ga_step(population, fitnesses, ts_best, n_best, rng, gen, total_gens)
            fitnesses = [compute_c5(h) for h in population]

        h_final, c5_final = active_set_cd(best_h, ts_best, max_iters=10000, top_m=3)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    n_shakes = 10
    for i in range(n_shakes):
        if time_left() < 50:
            break
        noise = 0.015 + 0.025 * (1.0 - i / float(n_shakes))
        h_shake = best_h + rng.normal(0, noise, n_best)
        h_shake = project_to_sum(h_shake, ts_best)
        h_cd, c5_cd = active_set_cd(h_shake, ts_best, max_iters=5000, top_m=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() < 40:
            break
        h_opt = slsqp_optimize(best_h, ts_best, 2000, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cd2, c5_cd2 = active_set_cd(best_h, ts_best, max_iters=3000, top_m=3)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    if time_left() > 60:
        for alpha in [5000, 20000, 100000]:
            if time_left() < 30:
                break
            h_pol = slsqp_optimize(best_h, ts_best, alpha, maxiter=800)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            h_cd, c5_cd = active_set_cd(best_h, ts_best, max_iters=5000, top_m=2)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    if time_left() > 30:
        max_cd = min(30000, int(time_left() * 80))
        h_s, c5_s = active_set_cd(best_h, ts_best, max_iters=max_cd, top_m=2, step_frac=0.08)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    if time_left() > 15:
        h_s2, c5_s2 = active_set_cd(best_h, ts_best, max_iters=5000, top_m=4, step_frac=0.05)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
