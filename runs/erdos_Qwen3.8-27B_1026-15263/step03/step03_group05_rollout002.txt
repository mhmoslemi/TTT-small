def compute_weights(h, alpha, topk_frac):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * (2.0 / n)
    topk = max(3, int(topk_frac * len(scaled)))
    if topk >= len(scaled):
        topk = len(scaled)
    idx = np.argpartition(-scaled, topk)[:topk]
    topk_vals = scaled[idx]
    m_val = np.max(topk_vals)
    exp_vals = np.exp(alpha * (topk_vals - m_val))
    w = np.zeros(len(corr))

        if time_fn() < 3:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_t = compute_c5(h)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_pair = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = best_c5
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov with top-k, multiple alpha
    if time_left() > 60:
        n1 = best_n
        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
        for alpha in alphas:
            if time_left() < 40:
                break
            lr = 0.05 / (1.0 + 0.05 * alpha)
            n_iter = min(2000, max(300, 3000 // (1 + alpha // 100)))
            h_new, c5_new = nesterov_descent(best_h, n1, n_iter, lr, 0.6, time_left)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 2: Fourier-space L-BFGS
    if time_left() > 120:
        for n_f in [best_n, min(360, best_n * 2) if best_n < 300 else best_n]:
            if time_left() < 60:
                break
            if n_f % 2 != 0:
                n_f += 1
            ts_f = n_f / 2.0
            if n_f != best_n:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n_f, endpoint=False)
                h_init = np.interp(x_new, x_old, best_h, period=2.0)
            else:
                h_init = best_h.copy()
            h_init = project_box_sum(h_init, ts_f)
            n_harm = min(40, max(12, n_f // 6))
            coeffs = grid_to_fourier(h_init, n_f, n_harm)
            for alpha in [50, 200, 800, 3000]:
                if time_left() < 30:
                    break
                try:
                    res = minimize(
                        fourier_obj, coeffs,
                        args=(n_f, n_harm, alpha),
                        method='L-BFGS-B',
                        options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-12}
                    )
                    h_opt = fourier_to_grid(res.x, n_f, n_harm)
                    h_opt = project_box_sum(h_opt, ts_f)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n_f
                        coeffs = grid_to_fourier(h_opt, n_f, n_harm)
                except Exception:
                    pass

    # Phase 3: Nesterov again at possibly new resolution
    if time_left() > 80:
        n3 = best_n
        alphas3 = [50, 200, 1000, 3000, 8000]
        for alpha in alphas3:
            if time_left() < 30:
                break
            lr = 0.03 / (1.0 + 0.03 * alpha)
            h_new, c5_new = nesterov_descent(best_h, n3, 1500, lr, 0.5, time_left)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Binary flip improvement
    if time_left() > 60:
        n4 = best_n
        h_bf, c5_bf = binary_flip_improve(best_h, n4, rng, 80, time_left)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Phase 5: Coordinate descent
    if time_left() > 40:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, 10, time_left)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Shake and re-optimize
    if time_left() > 100:
        n_sh = best_n
        for trial in range(15):
            if time_left() < 30:
                break
            noise_scale = 0.15 * (1.0 - trial / 15.0) + 0.03
            h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n_sh)
            h_shake = project_box_sum(h_shake, n_sh / 2.0)
            for lr_try in [0.03, 0.01]:
                if time_left() < 15:
                    break
                h_opt, c5_opt = nesterov_descent(h_shake, n_sh, 1000, lr_try, 0.6, time_left)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_shake = h_opt

    # Phase 7: High-resolution upsample
    if time_left() > 120:
        n_up = min(480, max(360, best_n * 2))
        if n_up % 2 != 0:
            n_up += 1
        if n_up != best_n:
            ts_up = n_up / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_up = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_up, x_old, best_h, period=2.0)
            h_up = project_box_sum(h_up, ts_up)
            for alpha in [100, 500, 2000]:
                if time_left() < 30:
                    break
                lr = 0.02 / (1.0 + 0.02 * alpha)
                h_opt, c5_opt = nesterov_descent(h_up, n_up, 2000, lr, 0.5, time_left)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_up

    # Phase 8: Final fine-tuning with very small steps
    if time_left() > 30:
        n_fin = best_n
        for lr_fin in [0.005, 0.002, 0.001]:
            if time_left() < 10:
                break
            h_fin, c5_fin = nesterov_descent(best_h, n_fin, 3000, lr_fin, 0.4, time_left)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
            h_cd2, c5_cd2 = coordinate_descent(best_h, n_fin, 5, time_left)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 9: Final binary flips
    if time_left() > 20:
        h_bf2, c5_bf2 = binary_flip_improve(best_h, best_n, rng, 30, time_left)
        if c5_bf2 < best_c5:
            best_c5 = c5_bf2
            best_h = h_bf2.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
