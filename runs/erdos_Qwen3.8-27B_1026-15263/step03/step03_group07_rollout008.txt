```python
    lr = lr_scale * 0.01

        if time_fn is not None and t % 100 == 0 and time_fn() < 5:
            break
        if t % recompute_active_every == 0:
            active = get_active_lags(h, active_frac)
        sm, g = smooth_max_and_grad_active(h, alpha, active)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1_hat = m1 / (1 - b1**t)
        m2_hat = m2 / (1 - b2**t)
        step = lr * m1_hat / (np.sqrt(m2_hat) + eps)
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
```

Hmm, but the issue with Adam is that it might oscillate. Let me add a line search or use a decreasing learning rate schedule.

Actually, let me use a different approach. Since the problem is smooth (with the soft-max approximation), I can use L-BFGS-B from scipy. The key is to handle the sum constraint.

With L-BCGS-B:
- Bounds: [(0,1)]*n
- Constraint: sum(h) = n/2

scipy's minimize with method='L-BFGS-B' does NOT support equality constraints! It only supports bounds.

So I need to either:
1. Use SLSQP (supports equality constraints but is slow for large n)
2. Use a penalty method with L-BFGS-B
3. Absorb the constraint into the parameterization

Let me use approach 3: parameterize h = 0.5 + P*z where P is a projection matrix that ensures sum(h) = n/2.

The constraint sum(h) = n/2 means sum(h - 0.5) = 0. So sum(z) = 0 (after the P projection).

P = I - (1/n)*1*1^T (the projection onto the subspace sum(x)=0)

But then h = 0.5 + P*z, and we need 0 ≤ h ≤ 1, i.e., -0.5/||P|| ≤ z ≤ 0.5/||P|| approximately. This doesn't give clean bounds.

Alternative: just use a penalty. f(h) = SM(corr)*dx + (1e6)*(sum(h) - n/2)². With a large enough penalty, the constraint is approximately satisfied.

Let me try this with L-BFGS-B:

```python
def lbfgs_optimize(h0, target_sum, alpha, maxiter, active_frac=0.95):
    n = len(h0)
    h0 = project_to_sum(h0, target_sum)
    
    def objective(z):
        h = np.clip(z, 0.0, 1.0)
        active = get_active_lags(h, active_frac)
        sm, _ = smooth_max_and_grad_active(h, alpha, active)
        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return sm + penalty
    
    def gradient(z):
        h = np.clip(z, 0.0, 1.0)
        active = get_active_lags(h, active_frac)
        _, g = smooth_max_and_grad_active(h, alpha, active)
        g += 2 * 1e6 * (np.sum(h) - target_sum)
        # Zero out gradient where h is at boundary
        g = np.where((z > 0) & (z < 1), g, 0)
        return g
    
    res = minimize(objective, h0, jac=gradient, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
    return project_to_sum(res.x, target_sum)
```

Wait, but the gradient of the clipped objective is tricky. If z < 0, then h = 0, and the gradient of the objective with respect to z should be 0 (since h doesn't change). Similarly for z > 1.

Actually, the standard approach is to NOT clip inside the objective/gradient, and instead let the bounds handle it. L-BFGS-B naturally respects bounds. So:

```python
def lbfgs_optimize(h0, target_sum, alpha, maxiter, active_frac=0.95):
    n = len(h0)
    h0 = np.clip(project_to_sum(h0, target_sum), 1e-10, 1-1e-10)
    
    def objective(z):
        h = np.clip(z, 0.0, 1.0)  # ensure validity
        active = get_active_lags(h, active_frac)
        sm, _ = smooth_max_and_grad_active(h, alpha, active)
        penalty = 1e8 * (np.sum(h) - target_sum)**2
        return sm + penalty
    
    def gradient(z):
        h = np.clip(z, 0.0, 1.0)
        active = get_active_lags(h, active_frac)
        _, g = smooth_max_and_grad_active(h, alpha, active)
        g += 2 * 1e8 * (np.sum(h) - target_sum)
        return g
    
    res = minimize(objective, h0, jac=gradient, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10})
    return project_to_sum(res.x, target_sum)
```

The penalty of 1e8 should be strong enough to enforce the constraint. And L-BFGS-B is much faster than SLSQP for large n.

But wait - the issue is that get_active_lags is called at every evaluation, which requires computing the full correlation (O(n²)). For n=1440, that's 2M ops per call. With 1000 iterations, that's 2B ops. In NumPy C code, this should take about 1-2 seconds. The gradient computation adds another O(n²) per call. So total per iteration: ~5M ops. 1000 iterations: 5B ops ≈ 3-5 seconds. That's very feasible!

Let me also think about whether to use the full smooth max or just the top few lags. If I use just the top 50 lags (out of 2n-1 ≈ 3000 for n=1440), the gradient loop is much faster.

Let me write the complete solution now:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
    return h

def get_active_lags(corr, frac=0.95, top_k=100):
    m = np.max(corr)
    active = np.where(corr >= frac * m)[0]
    if len(active) > top_k:
        idx = np.argsort(corr[active])[::-1][:top_k]
        active = active[idx]
    return active

def smooth_max_active(corr, active_idx, alpha):
    ac = corr[active_idx]
    m = np.max(ac)
    shifted = np.clip(alpha * (ac - m), -700, 700)
    e = np.exp(shifted)
    w = e / np.sum(e)
    sm_val = m + np.log(np.sum(e) / len(e)) / alpha
    return sm_val, w

def grad_from_weights(h, w, active_idx):
    n = len(h)
    grad = np.zeros(n)
    for ip, mi in enumerate(active_idx):
        wi = w[ip]
        if wi < 1e-14:
            continue
        k = mi - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            kn = -k
            grad[:n-kn] += wi * (1.0 - h[kn:])
            grad[kn:] -= wi * h[:n-kn]
    return grad

def objective_and_grad(h, target_sum, alpha, active_frac, top_k, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = get_active_lags(corr, active_frac, top_k)
    sm_val, w = smooth_max_active(corr, active, alpha)
    g = grad_from_weights(h, w, active) * dx
    # Add penalty gradient
    g += 2.0 * penalty * (np.sum(h) - target_sum)
    val = sm_val + penalty * (np.sum(h) - target_sum)**2
    return val, g

def lbfgs_step(h0, target_sum, alpha, maxiter, active_frac=0.95, top_k=80, penalty=1e6):
    n = len(h0)
    h_start = project_to_sum(h0, target_sum)
    
    def obj(z):
        h = np.clip(z, 0.0, 1.0)
        v, _ = objective_and_grad(h, target_sum, alpha, active_frac, top_k, penalty)
        return v
    
    def grad(z):
        h = np.clip(z, 0.0, 1.0)
        _, g = objective_and_grad(h, target_sum, alpha, active_frac, top_k, penalty)
        return g
    
    try:
        res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        return project_to_sum(h_start, target_sum)
```
