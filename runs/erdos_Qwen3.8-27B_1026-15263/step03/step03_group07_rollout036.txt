import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_indices(n):
    j_idx = np.arange(n)
    l_idx = np.arange(n)
    L1 = n - 1 + l_idx[None, :] - j_idx[:, None]
    L2 = n - 1 + j_idx[:, None] - l_idx[None, :]

    valid2 = (L2 >= 0) & (L2 <= 2*n-2)

    mask1 = valid1.astype(np.float64)
    mask2 = valid2.astype(np.float64)

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)

    s = np.sum(e)

def smooth_max_and_grad(h, alpha, W1_idx, mask1, W2_idx, mask2):

    corr = np.correlate(h, 1.0 - h, mode='full')

    sm_val = m + np.log(s / (2.0*n - 1.0)) / alpha
    W1 = w[W1_idx] * mask1
    W2 = w[W2_idx] * mask2
    grad = W1 @ (1.0 - h) - W2 @ h
    return sm_val, grad

def auglag_objective(x, alpha, target_sum, lam, mu, W1_idx, mask1, W2_idx, mask2):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = np.sum(e)
    val = m + np.log(s / (2.0*n - 1.0)) / alpha
    constraint_viol = np.sum(x) - target_sum
    val += lam * constraint_viol + mu * constraint_viol ** 2
    return val

def auglag_gradient(x, alpha, target_sum, lam, mu, W1_idx, mask1, W2_idx, mask2):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    W1 = w[W1_idx] * mask1
    W2 = w[W2_idx] * mask2
    g = W1 @ (1.0 - x) - W2 @ x
    pen = 2.0 * mu * (np.sum(x) - target_sum) + lam
    g = g + pen
    return g

def run_lbfgsb(h0, n, target_sum, alpha, lam, mu, maxiter,

                max_c = np.max(corr_cur)

    if time_left() > 80:
        for alpha_hi in [10000.0, 30000.0, 80000.0]:
            if time_left() < 30:
                break
            maxiter = max(50, min(500, int(time_left() * 0.5)))
            mu_hi = 10.0
            lam_hi = np.clip(np.sum(h_cur) - ts, -10.0, 10.0)
            h_new = run_lbfgsb(h_cur, n, ts, alpha_hi, lam_hi, mu_hi,
                               maxiter, W1_idx, mask1, W2_idx, mask2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
            h_cur = h_new

    # Phase 4: Projected subgradient with momentum
    if time_left() > 40:
        n_iters = min(5000, int(time_left() * 3))
        h_pg, c5_pg = projected_subgrad_momentum(
            h_cur, n, ts, 5000.0, n_iters, 0.03, 0.9,
            W1_idx, mask1, W2_idx, mask2)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n
        h_cur = h_pg

    # Phase 5: Coordinate descent
    if time_left() > 60:
        n_cd = min(80, int(time_left() * 0.5))
        h_cd, c5_cd = coordinate_descent_fast(
            h_cur, n, ts, n_cd, rng,
            W1_idx, mask1, W2_idx, mask2, 5000.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        h_cur = h_cd

    # Phase 6: Try higher resolution n=1440
    if time_left() > 150 and n <= 720:
        n2 = 1440
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_cur)
        h_hi = project_to_sum(h_hi, ts2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n2

        W1_idx2, mask1_2, W2_idx2, mask2_2 = precompute_indices(n2)

        for alpha_hi in [100.0, 500.0, 2000.0, 8000.0, 30000.0]:
            if time_left() < 30:
                break
            maxiter = max(30, min(200, int(time_left() * 0.5)))
            mu_hi = 5.0
            lam_hi = 0.0
            h_hi_opt = run_lbfgsb(h_hi, n2, ts2, alpha_hi, lam_hi, mu_hi,
                                  maxiter, W1_idx2, mask1_2, W2_idx2, mask2_2)
            c5_hi_opt = compute_c5(h_hi_opt)
            if c5_hi_opt < best_c5:
                best_c5 = c5_hi_opt
                best_h = h_hi_opt.copy()
                best_n = n2
            h_hi = h_hi_opt

        if time_left() > 40:
            corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode='full')
            max_hi = np.max(corr_hi)
            for thresh in [0.99, 0.96]:
                if time_left() < 20:
                    break
                active_hi = np.where(corr_hi >= thresh * max_hi)[0]
                if len(active_hi) < 3:
                    continue
                n_act = min(100, max(30, int(time_left() * 0.3)))
                h_act_hi = active_lag_optimize(h_hi, n2, ts2, 5000.0,
                                               active_hi, n_act,
                                               W1_idx2, mask1_2,
                                               W2_idx2, mask2_2)
                c5_act_hi = compute_c5(h_act_hi)
                if c5_act_hi < best_c5:
                    best_c5 = c5_act_hi
                    best_h = h_act_hi.copy()
                    best_n = n2
                h_hi = h_act_hi

        if time_left() > 30:
            h_cd2, c5_cd2 = coordinate_descent_fast(
                h_hi, n2, ts2, 40, rng,
                W1_idx2, mask1_2, W2_idx2, mask2_2, 5000.0)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n2

    # Phase 7: Final polish on best solution
    if best_h is not None and time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        W1f, m1f, W2f, m2f = precompute_indices(n_f)

        for alpha_f in [50000.0, 200000.0]:
            if time_left() < 20:
                break
            maxiter_f = max(30, min(300, int(time_left() * 0.6)))
            mu_f = 20.0
            lam_f = np.clip(np.sum(h_f) - ts_f, -5.0, 5.0)
            h_pol = run_lbfgsb(h_f, n_f, ts_f, alpha_f, lam_f, mu_f,
                               maxiter_f, W1f, m1f, W2f, m2f)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f
            h_f = h_pol

        if time_left() > 25:
            n_cd_f = min(60, int(time_left() * 0.5))
            h_cdf, c5_cdf = coordinate_descent_fast(
                h_f, n_f, ts_f, n_cd_f, rng,
                W1f, m1f, W2f, m2f, 10000.0)
            if c5_cdf < best_c5:
                best_c5 = c5_cdf
                best_h = h_cdf.copy()
                best_n = n_f

        if time_left() > 20:
            n_pg_f = min(3000, int(time_left() * 2))
            h_pgf, c5_pgf = projected_subgrad_momentum(
                h_f, n_f, ts_f, 10000.0, n_pg_f, 0.01, 0.92,
                W1f, m1f, W2f, m2f)
            if c5_pgf < best_c5:
                best_c5 = c5_pgf
                best_h = h_pgf.copy()
                best_n = n_f

    # Final projection and evaluation
    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
