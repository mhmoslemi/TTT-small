Wait, np.correlate(h, 1-h, mode="full") gives:

Let me just use the definition: for arrays a, b of length n:

So corr = np.correlate(h, 1-h, mode="full"):
corr[k] = ∑_i h[i] * (1-h)[i + k - (n-1)]

Let m = k - (n-1). Then corr[k] = ∑_i h[i] * (1-h)[i + m] for valid i.

When we change h[i] from 1 to 0:
- For each k where i + m is in [0, n-1], the term h[i] * (1-h)[i+m] changes from (1-h)[i+m] to 0
- So corr[k] decreases by (1-h)[i+m]

When we change h[j] from 0 to 1:
- For each k where j + m is in [0, n-1], the term h[j] * (1-h)[j+m] changes from 0 to (1-h)[j+m]
- So corr[k] increases by (1-h)[j+m]

But we also need to consider that changing h[i] affects other correlations where h[i] appears as the "second" argument:
- corr[k] = ∑_i h[i] * (1-h[i+m])
- If h[i] appears as (1-h[i+m]) in some other term, that changes too.

Specifically, h[i] appears in:
1. As the first argument: term i in corr[k] for k where i + m ∈ [0,n-1]
2. As the second argument: term (i-m) in corr[k] for k where i-m ∈ [0,n-1]

For case 1: h[i] goes from 1 to 0, so corr[k] at the corresponding k decreases by (1-h[i+m])
For case 2: (1-h[i]) goes from 0 to 1, so corr[k] at the corresponding k increases by h[i-m]

So the total change in corr when flipping h[i] from 1 to 0:
- corr[k_i_first] -= (1-h[i+m_i])  [where k_i_first = n-1 + m_i, m_i such that i+m_i is valid]
- corr[k_i_second] += h[i-m_i']  [where i-m_i' is valid]

```python
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        n = len(h)
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        perm = rng.permutation(len(ones_idx))[:excess]
        h_bin[ones_idx[perm]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        perm = rng.permutation(len(zeros_idx))[:deficit]
        h_bin[zeros_idx[perm]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha, target_sum, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    obj = m_val + np.log(s) / alpha
    w = exp_vals / s
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    sum_diff = float(np.sum(h)) - target_sum
    obj += 0.5 * penalty * sum_diff * sum_diff
    grad = grad + penalty * sum_diff
    return obj, grad

def sm_obj(h, alpha, target_sum, penalty):
    return smooth_max_and_grad(h, alpha, target_sum, penalty)[0]

def sm_grad(h, alpha, target_sum, penalty):
    return smooth_max_and_grad(h, alpha, target_sum, penalty)[1]

def l_bfgs_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_obj, h0.copy(), method='L-BFGS-B', jac=sm_grad,
                       args=(alpha, target_sum, penalty),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def sum_eq_fun(h):
    return np.array([np.sum(h)])

def sum_eq_jac(h):
    return np.ones((1, len(h)))

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    bounds = [(0.0, 1.0)] * n
    cons = NonlinearConstraint(sum_eq_fun, target_sum - 1e-10, target_sum + 1e-10, jac=sum_eq_jac)
    try:
        res = minimize(sm_obj, h0.copy(), method='SLSQP', jac=sm_grad,
                       args=(alpha, target_sum, penalty),
                       bounds=bounds, constraints=[cons],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_gradient_step(h, grad, lr):
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def projected_gradient_descent(h0, n, target_sum, alpha, t_end, penalty):
    h = h0.copy()
    h = project_to_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 1.0 / (alpha + penalty + 1)
    patience = 200
    no_improve = 0
    
    while time.time() < t_end:
        _, grad = smooth_max_and_grad(h, alpha, target_sum, penalty)
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        h_new = h - lr * grad / g_norm
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.1, 1.0)
        else:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
        h = h_new
        if no_improve > patience:
            h = best_h.copy()
            no_improve = 0
            lr = 0.01
    
    return best_h, best_c5

def binary_sa(h_start, n, target_sum, t_end, rng, T_init=0.015, T_min=0.002, block_prob=0.3):
    h = h_start.copy()
    corr_curr = compute_corr(h)
    c5_curr = float(np.max(corr_curr))
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    
    while time.time() < t_end:
        move_type = rng.random()
        if move_type < (1.0 - block_prob):
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform(1e-300, 1.0)) > delta / T:
                c5_curr = c5_new
                corr_curr = corr_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            # Block swap
            block_len = rng.randint(2, min(n // 5, 15) + 1)
            i = rng.randint(n - block_len)
            j = rng.randint(n - block_len)
            if abs(i - j) < block_len:
                continue
            block_i = h[i:i+block_len]
            block_j = h[j:j+block_len]
            if np.sum(block_i) != np.sum(block_j):
                continue
            h[i:i+block_len] = block_j
            h[j:j+block_len] = block_i
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform(1e-300, 1.0)) > delta / T:
                c5_curr = c5_new
                corr_curr = corr_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i:i+block_len] = block_i
                h[j:j+block_len] = block_j
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            corr_curr = compute_corr(h)
            T = min(T * 2.5, T_init * 2.0)
            no_improve = 0
        T = max(T * 0.999997, T_min)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)
```

```python
        perm = rng.permutation(len(ones_idx))[:excess]
        h_bin[ones_idx[perm]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        perm = rng.permutation(len(zeros_idx))[:deficit]
        h_bin[zeros_idx[perm]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha, target_sum, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    obj = m_val + np.log(s) / alpha
    w = exp_vals / s
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    sum_diff = float(np.sum(h)) - target_sum
    obj += 0.5 * penalty * sum_diff * sum_diff
    grad = grad + penalty * sum_diff
    return obj, grad

def sm_obj(h, alpha, target_sum, penalty):
    return smooth_max_and_grad(h, alpha, target_sum, penalty)[0]

def sm_grad(h, alpha, target_sum, penalty):
    return smooth_max_and_grad(h, alpha, target_sum, penalty)[1]

def sum_eq_fun(h):
    return np.array([np.sum(h)])

def sum_eq_jac(h):
    return np.ones((1, len(h)))

def l_bfgs_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_obj, h0.copy(), method='L-BFGS-B', jac=sm_grad,
                       args=(alpha, target_sum, penalty),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    bounds = [(0.0, 1.0)] * n
    cons = NonlinearConstraint(sum_eq_fun, target_sum - 1e-10, target_sum + 1e-10, jac=sum_eq_jac)
    try:
        res = minimize(sm_obj, h0.copy(), method='SLSQP', jac=sm_grad,
                       args=(alpha, target_sum, penalty),
                       bounds=bounds, constraints=[cons],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa_informed(h_start, n, target_sum, t_end, rng, T_init=0.018, T_min=0.001):
    h = h_start.copy()
    corr_curr = compute_corr(h)
    c5_curr = float(np.max(corr_curr))
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    informed_prob = 0.4
    block_prob = 0.25
    
    while time.time() < t_end:
        r = rng.random()
        if r < informed_prob:
            worst_k = int(np.argmax(corr_curr))
            shift = worst_k - (n - 1)
            if abs(shift) == 0:
                shift = 1
            if shift > 0:
                i = rng.randint(max(1, n - shift))
                j = i + shift
            else:
                j = rng.randint(max(1, -shift))
                i = j - shift
            if i < 0 or i >= n or j < 0 or j >= n:
                i = rng.randint(n)
                j = rng.randint(n)
            if h[i] == h[j]:
                T = max(T * 0.999997, T_min)
                continue
            h[i], h[j] = h[j], h[i]
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform(1e-300, 1.0)) > delta / T:
                c5_curr = c5_new
                corr_curr = corr_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        elif r < informed_prob + block_prob:
            block_len = rng.randint(2, min(n // 6, 12) + 1)
            i = rng.randint(n - block_len)
            j = rng.randint(n - block_len)
            if abs(i - j) < block_len:
                T = max(T * 0.999997, T_min)
                continue
            block_i = h[i:i + block_len]
            block_j = h[j:j + block_len]
            if np.sum(block_i) != np.sum(block_j):
                T = max(T * 0.999997, T_min)
                continue
            h[i:i + block_len] = block_j
            h[j:j + block_len] = block_i
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform(1e-300, 1.0)) > delta / T:
                c5_curr = c5_new
                corr_curr = corr_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i:i + block_len] = block_i
                h[j:j + block_len] = block_j
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                T = max(T * 0.999997, T_min)
                continue
            h[i], h[j] = h[j], h[i]
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform(1e-300, 1.0)) > delta / T:
                c5_curr = c5_new
                corr_curr = corr_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        
        if no_improve > 25000:
            h = best_h.copy()
            c5_curr = best_c5
            corr_curr = compute_corr(h)
            T = min(T * 3.0, T_init * 2.0)
            no_improve = 0
        T = max(T * 0.999997, T_min)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def greedy_improve_binary(h, n, target_sum, max_rounds=50):
    h = h.copy()
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-14
        best_i = -1
        best_j = -1
        n_check = min(len(ones) * len(zeros), 20000)
        for _ in range(n_check // (len(ones) * len(zeros)) + 1):
            idx_i = rng_idx = np.random.randint(len(ones))
            idx_j = np.random.randint(len(zeros))
            i = int(ones[idx_i])
            j = int(zeros[idx_j])
            h[i], h[j] = 0.0, 1.0
            c5_test = float(np.max(compute_corr(h)))
            delta = c5_test - c5_curr
            h[i], h[j] = 1.0, 0.0
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            c5_curr += best_delta
        else:
            break
    return h, c5_curr

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n_main = best_n
    ts_int = int(n_main / 2.0)
    ts_float = n_main / 2.0
    
    h_bin = make_binary(best_h, n_main, ts_int, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    sa_end1 = t_start + 0.30 * budget_s
    h_sa, c5_sa = binary_sa_informed(h_bin, n_main, ts_int, sa_end1, rng, T_init=0.020, T_min=0.001)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    
    cont_end = t_start + 0.55 * budget_s
    alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    for i_a, alpha in enumerate(alphas):
        if time.time() > cont_end:
            break
        pen = alpha * 0.5
        h_try = best_h.copy()
        h_try = project_to_sum(h_try, ts_float)
        h_opt = l_bfgs_optimize(h_try, n_main, ts_float, alpha, maxiter=2000, penalty=pen)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_bin = make_binary(h_opt, n_main, ts_int, rng)
    
    sa_end2 = t_start + 0.72 * budget_s
    h_bin = make_binary(best_h, n_main, ts_int, rng)
    h_sa2, c5_sa2 = binary_sa_informed(h_bin, n_main, ts_int, sa_end2, rng, T_init=0.012, T_min=0.0008)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()
    
    ref_end = t_start + 0.85 * budget_s
    for alpha, pen in [(2000, 1000), (5000, 3000), (10000, 5000), (20000, 10000)]:
        if time.time() > ref_end:
            break
        h_try = best_h.copy()
        h_try = project_to_sum(h_try, ts_float)
        h_opt = slsqp_optimize(h_try, n_main, ts_float, alpha, maxiter=3000, penalty=pen)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    for n_try in [200, 280, 320]:
        if time.time() > t_start + 0.96 * budget_s:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = h_interp
        for alpha, pen in [(50, 20), (300, 100), (1000, 500)]:
            if time.time() > t_start + 0.97 * budget_s:
                break
            h_try = l_bfgs_optimize(h_try, n_try, ts_try, alpha, maxiter=1500, penalty=pen)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
    
    if time.time() < t_start + 0.99 * budget_s:
        h_final = best_h.copy()
        h_final = project_to_sum(h_final, best_n / 2.0)
        h_final = np.clip(h_final, 0.0, 1.0)
        h_final = l_bfgs_optimize(h_final, best_n, best_n / 2.0, 20000, maxiter=5000, penalty=10000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
```
