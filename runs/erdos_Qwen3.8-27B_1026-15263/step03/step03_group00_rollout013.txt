        if abs(diff) < 1e-14:
            break

def corr_gradient_single(h, lag):
    """Compute gradient of corr[lag_index] w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:

        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

    n = len(h)
    corr = compute_corr(h)

                                pop_size, F_init, F_final, CR):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    pop = np.zeros((pop_size, n))
    pop[0] = best_h
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.15:
            pop[i] = best_h + rng.uniform(-0.01, 0.01, n)
        elif r < 0.3:
            pop[i] = best_h + rng.uniform(-0.05, 0.05, n)
        elif r < 0.45:
            pop[i] = best_h + rng.uniform(-0.15, 0.15, n)
        elif r < 0.6:
            pop[i] = best_h + rng.uniform(-0.3, 0.3, n)
        elif r < 0.75:
            pop[i] = rng.random(n)
        elif r < 0.85:
            pop[i] = (rng.random(n) < 0.5).astype(float)
        elif r < 0.92:
            freq = 1 + rng.randint(0, 4)
            phase = rng.uniform(0, 2.0 * np.pi)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            pop[i] = 0.5 + 0.3 * np.sin(freq * np.pi * x + phase)
        else:
            pop[i] = 1.0 - best_h + rng.uniform(-0.03, 0.03, n)
        pop[i] = project_to_sum(pop[i], target_sum)
    pop_c5 = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_idx]
    best_h = pop[best_idx].copy()
    no_improve = 0
    no_improve_limit = 300
    gen = 0
    while gen < 200000:
        if time.time() - t_start > time_limit:
            break
        progress = min(gen / 50000.0, 1.0)
        F = F_init + (F_final - F_init) * progress
        CR_adapt = CR - 0.2 * progress
        for i in range(pop_size):
            if time.time() - t_start > time_limit:
                break
            if rng.random() < 0.25 and gen > 100:
                while True:
                    b = rng.randint(pop_size)
                    if b != i:
                        break
                v = pop[best_idx] + F * (pop[best_idx] - pop[b])
            else:
                while True:
                    a = rng.randint(pop_size)
                    if a != i:
                        break
                while True:
                    b = rng.randint(pop_size)
                    if b != i and b != a:
                        break
                v = pop[a] + F * (pop[a] - pop[b])
            j_rand = rng.randint(n)
            crosser = rng.random(n) < CR_adapt
            crosser[j_rand] = True
            trial = np.where(crosser, v, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_sum(trial, target_sum)
            trial_c5 = compute_c5(trial)
            if trial_c5 <= pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = trial_c5
                if trial_c5 < best_c5 - 1e-12:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
        gen += 1
        if no_improve > no_improve_limit:
            for i in range(pop_size // 3):
                idx = int(rng.randint(pop_size))
                if idx != best_idx:
                    pop[idx] = best_h + rng.uniform(-0.1, 0.1, n)
                    pop[idx] = project_to_sum(pop[idx], target_sum)
                    pop_c5[idx] = compute_c5(pop[idx])
            no_improve = 0
    return best_h, best_c5

def stochastic_local_search(h_init, target_sum, rng, max_iters, noise_scale):
    n = len(h_init)
    h = h_init.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        noise = rng.uniform(-noise_scale, noise_scale, n)
        h_trial = h + noise
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = project_to_sum(h_trial, target_sum)
        h_cd, c5_trial = coordinate_descent(h_trial, target_sum, max_iters=500, step_frac=0.1)
        if c5_trial < c5:
            h = h_cd
            c5 = c5_trial
        else:
            noise_scale *= 0.95
            if noise_scale < 1e-6:
                break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    target_sum = n / 2.0

    def time_left():
        return budget_s - (time.time() - t_start)

    h_init = best_h.copy()

    if time_left() > 200:
        de_time = min(0.45 * budget_s, time_left() - 300)
        h_de, c5_de = run_differential_evolution(
            h_init, n, target_sum, rng, t_start, de_time,
            pop_size=70, F_init=0.9, F_final=0.3, CR=0.85
        )
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()

    if time_left() > 80:
        h_cur = best_h.copy()
        for alpha in [30, 100, 500, 2000, 10000, 50000, 200000]:
            if time_left() < 30:
                break
            h_pg, c5_pg = projected_gradient_descent(
                h_cur, target_sum, alpha, lr_init=0.05, max_steps=500
            )
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_cur = h_pg.copy()
            else:
                h_cur = h_pg.copy()
                if c5_pg < best_c5 + 0.001:
                    h_cur = h_pg

    if time_left() > 60:
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, max_iters=8000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 50:
        h_mcd, c5_mcd = multi_lag_coord_descent(best_h, target_sum, max_iters=5000, step_frac=0.08, n_lags=3)
        if c5_mcd < best_c5:
            best_c5 = c5_mcd
            best_h = h_mcd.copy()

    if time_left() > 40:
        h_mcd2, c5_mcd2 = multi_lag_coord_descent(best_h, target_sum, max_iters=5000, step_frac=0.08, n_lags=5)
        if c5_mcd2 < best_c5:
            best_c5 = c5_mcd2
            best_h = h_mcd2.copy()

    if time_left() > 60:
        n_restarts = 8
        for r in range(n_restarts):
            if time_left() < 30:
                break
            noise = 0.02 + 0.06 * (1.0 - r / float(n_restarts))
            h_shake = best_h + rng.uniform(-noise, noise, n)
            h_shake = project_to_sum(h_shake, target_sum)
            h_sls, c5_sls = stochastic_local_search(
                h_shake, target_sum, rng, max_iters=300, noise_scale=0.03
            )
            if c5_sls < best_c5:
                best_c5 = c5_sls
                best_h = h_sls.copy()

    if time_left() > 40:
        for alpha in [50000, 200000, 1000000]:
            if time_left() < 20:
                break
            h_pg, c5_pg = projected_gradient_descent(
                best_h, target_sum, alpha, lr_init=0.02, max_steps=300
            )
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    if time_left() > 30:
        h_cd2, c5_cd2 = coordinate_descent(best_h, target_sum, max_iters=10000, step_frac=0.05)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    if time_left() > 20:
        h_mcd3, c5_mcd3 = multi_lag_coord_descent(best_h, target_sum, max_iters=8000, step_frac=0.03, n_lags=5)
        if c5_mcd3 < best_c5:
            best_c5 = c5_mcd3
            best_h = h_mcd3.copy()

    if time_left() > 15:
        h_cd3, c5_cd3 = coordinate_descent(best_h, target_sum, max_iters=15000, step_frac=0.02)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
