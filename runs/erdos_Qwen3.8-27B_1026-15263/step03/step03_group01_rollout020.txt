    """
    Compute smooth max of circular correlation and its gradient using FFT.
    The circular correlation approximates the central part of the linear correlation.
    """

    OnH = np.fft.rfft(1.0 - h)
    circ = np.fft.irfft(H * np.conj(OnH), n=n)

    m = np.max(circ)
    shifted = alpha * (circ - m)
    log_sum_exp = m + np.log(np.sum(np.exp(shifted))) / alpha
    w = np.exp(shifted) / np.sum(np.exp(shifted))

    W = np.fft.rfft(w)
    grad = 1.0 - np.fft.irfft(W * H, n=n) - np.fft.irfft(np.conj(W) * H, n=n)
    grad *= dx

The objective max_k ∫h(x)(1-h(x+k))dx can be rewritten as:
minimize t
subject to ∫h(x)(1-h(x+k))dx ≤ t for all k

           0 ≤ h ≤ 1

This is a non-convex problem (the constraints ∫h(x)(1-h(x+k))dx are quadratic in h). So SDP/QP won't directly apply.

But we can use a **successive linearization** approach:
1. Start with current h
2. Linearize the constraints at the current point
3. Solve the resulting LP/QP
4. Update h
5. Repeat

This is essentially the **cutting plane** or **outer approximation** method. Each subproblem is a QP (since the linearization of a quadratic gives a linear constraint, and we have box constraints + sum constraint).

Wait, the linearization of ∫h(x)(1-h(x+k))dx at h⁰ is:
∫h⁰(x)(1-h⁰(x+k))dx + ∫[(1-h⁰(x+k))(h(x)-h⁰(x)) - h⁰(x)(h(x+k)-h⁰(x+k))]dx

This is linear in h. So the subproblem is:
minimize t
subject to [linearized constraint k] ≤ t for all k
           ∫h = 1
           0 ≤ h ≤ 1

This is an LP! We can solve it with any LP solver.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    x = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-13:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def fft_smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h)
    OnH = np.fft.rfft(1.0 - h)
    circ = np.fft.irfft(H * np.conj(OnH), n=n)
    m = np.max(circ)
    shifted = alpha * (circ - m)
    S = np.sum(np.exp(shifted))
    obj = m + np.log(S) / alpha
    w = np.exp(shifted) / S
    W = np.fft.rfft(w)
    term1 = np.fft.irfft(W * H, n=n)
    term2 = np.fft.irfft(np.conj(W) * H, n=n)
    grad = (1.0 - term1 - term2) * dx
    return obj, grad

def proj_grad_opt(h0, n, target_sum, alpha, max_steps, t_start, deadline, lr_init=0.1):
    h = h0.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        _, grad = fft_smooth_max_grad(h, alpha)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / (gnorm + 1e-30)
        improved = False
        for bt in range(25):
            h_new = project_feasible(h - lr * gnorm * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
                improved = True
                break
            lr *= 0.4
            if lr < 1e-15:
                break
        if improved:
            lr = min(lr * 2.0, 5.0)
        else:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def slsqp_fft(h0, n, ts, alpha, maxiter, t_start, deadline):
    def obj_func(x):
        return fft_smooth_max_grad(x, alpha)[0]
    def grad_func(x):
        return fft_smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        if r.success or r.fun < obj_func(h0):
            return project_feasible(r.x, ts)
    except Exception:
        pass
    return project_feasible(h0, ts)

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new / 2.0)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_feasible(h_new, n // 2.0)

def stochastic_cd(h, n, ts, rng, n_steps, t_start, deadline, delta_range):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    temp = 0.15
    for step in range(n_steps):
        if time.time() > deadline:
            break
        ones_idx = np.where(h > 0.03)[0]
        zeros_idx = np.where(h < 0.97)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.4
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.4
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = project_feasible(h, ts)
        temp = max(1e-5, 0.15 * np.exp(-step / 8000.0))
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: Continuation on FFT-based projected gradient at current resolution
    alphas1 = [20, 50, 100, 300, 800, 2000]
    for a in alphas1:
        if time.time() > deadline - 120:

        h_new, c5_new = proj_grad_opt(best_h, n, ts, a, max_steps=8000,
                                       t_start=t_start, deadline=deadline - 120, lr_init=0.08)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: SLSQP polish at current resolution
    for a in [500, 2000, 8000]:
        if time.time() > deadline - 80:
            break
        h_s = slsqp_fft(best_h, n, ts, a, maxiter=1500, t_start=t_start, deadline=deadline - 80)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Upsample and optimize at higher resolutions
    for n_tr in [600, 720, 960, 1200]:
        if time.time() > deadline - 150:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0

        c5_up = compute_c5(h_up)

                                       t_start=t_start, deadline=deadline - 80, lr_init=0.06)
            if c5_g < best_c5:

                best_h = h_g.copy()
                best_n = n_tr
            h_up = h_g
        if time.time() > deadline - 60:
            break
        h_s2 = slsqp_fft(h_up, n_tr, ts_tr, 3000, maxiter=800, t_start=t_start, deadline=deadline - 40)
        c5_s2 = compute_c5(h_s2)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()
            best_n = n_tr

    # Phase 4: Frequency perturbations and re-optimization
    n_b = best_n
    ts_b = n_b / 2.0
    for trial in range(40):
        if time.time() > deadline - 60:
            break
        amp = 0.06 * (1.0 - trial / 50.0) + 0.01
        nf = max(3, n_b // 20)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        for a in [100, 500, 2000]:
            if time.time() > deadline - 40:
                break
            h_o, c5_o = proj_grad_opt(h_sh, n_b, ts_b, a, max_steps=5000,
                                       t_start=t_start, deadline=deadline - 40, lr_init=0.05)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            h_sh = h_o

    # Phase 5: Stochastic coordinate descent
    if time.time() < deadline - 120:
        h_cd, c5_cd = stochastic_cd(best_h, best_n, best_n / 2.0, rng,
                                     n_steps=60000, t_start=t_start,
                                     deadline=deadline - 80, delta_range=(0.005, 0.25))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        # Polish after stochastic
        if time.time() < deadline - 50:
            for a in [1000, 5000]:
                if time.time() > deadline - 30:

                                           max_steps=6000, t_start=t_start,

                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Phase 6: Second stochastic round with smaller steps
    if time.time() < deadline - 80:
        h_cd2, c5_cd2 = stochastic_cd(best_h, best_n, best_n / 2.0, rng,
                                       n_steps=40000, t_start=t_start,

    # Phase 7: Final high-alpha polish

                                       deadline=deadline - 10, lr_init=0.02)

                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 8: SLSQP final polish
    if time.time() < deadline - 20:
        h_fin = slsqp_fft(best_h, best_n, best_n / 2.0, 10000, maxiter=2000,
                          t_start=t_start, deadline=deadline - 5)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
