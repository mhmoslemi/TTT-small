```python
        m = k - (n - 1)
        if m >= 0:
            # Positive shift m
            # (1-h[j-m]) for j=m..n-1
            grad[m:] += w * (1.0 - h[:n-m])
            # -h[j+m] for j=0..n-1-m
            grad[:n-m] -= w * h[m:]
        else:
            # Negative shift, m' = -m > 0
            m_prime = -m
            # (1-h[j+m']) for j=0..n-1-m'
            grad[:n-m_prime] += w * (1.0 - h[m_prime:])
            # -h[j-m'] for j=m'..n-1
            grad[m_prime:] -= w * h[:n-m_prime]
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def get_active_lags(corr, n_full, top_k=30, frac=0.99):
    max_val = np.max(corr)
    if max_val <= 0:
        return np.array([n_full // 2])
    active = np.where(corr >= frac * max_val)[0]
    if len(active) == 0:
        active = np.array([n_full // 2])
    # Also include top_k by value if we have fewer than top_k
    if len(active) < top_k:
        idx_sorted = np.argsort(corr)[::-1][:top_k]
        active = np.sort(np.unique(np.concatenate([active, idx_sorted])))
    return active

def smooth_max_weighted(corr, active, alpha, dx):
    """Compute smooth max approximation and weights, scaled by dx."""
    ac = corr[active]
    m = np.max(ac)
    shifted = alpha * (ac - m)
    shifted = np.clip(shifted, -500, 500)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    val = (m + np.log(s / len(ev)) / alpha) * dx
    return val, w

def grad_from_weights(h, active, w):
    """Vectorized gradient of smooth max (unscaled, multiply by dx outside)."""
    n = len(h)
    nm1 = n - 1
    g = np.zeros(n)
    for idx in range(len(active)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = active[idx]
        m = k - nm1
        if m >= 0:
            if m < n:
                g[m:] += wi * (1.0 - h[:n-m])
                g[:n-m] -= wi * h[m:]
        else:
            mp = -m
            if mp < n:
                g[:n-mp] += wi * (1.0 - h[mp:])
                g[mp:] -= wi * h[:n-mp]
    return g

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def pg_optimize(h0, n, target, n_iter, alpha0=50, alpha1=2000, lr=0.05, top_k=25, frac=0.99):
    """Projected gradient with adaptive alpha and active lag updates."""
    h = h0.copy()
    dx = 2.0 / n
    c5 = compute_c5(h)
    h_best = h.copy()
    curr = compute_corr(h)
    active = get_active_lags(curr, 2*n-1, top_k, frac)
    
    for it in range(n_iter):
        t = it / max(n_iter - 1, 1)
        al = alpha0 * (alpha1 / alpha0) ** t
        _, w = smooth_max_weighted(curr, active, al, dx)
        g = grad_from_weights(h, active, w)
        
        h_new = h - lr * g
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.7
        
        if (it+1) % 100 == 0:
            curr = compute_corr(h)
            active = get_active_lags(curr, 2*n-1, top_k, frac)
    
    return h_best, c5

def slsqp_opt(h0, n, target, alpha, maxiter, top_k=20, frac=0.995):
    dx = 2.0 / n
    def f(x):
        c = compute_corr(x)
        act = get_active_lags(c, 2*n-1, top_k, frac)
        v, _ = smooth_max_weighted(c, act, alpha, dx)
        return v
    def jf(x):
        c = compute_corr(x)
        act = get_active_lags(c, 2*n-1, top_k, frac)
        _, w = smooth_max_weighted(c, act, alpha, dx)
        g = grad_from_weights(x, act, w) * dx
        return g
    cons = [{'type':'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), jac=jf, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_sum(r.x, target)
    except:
        return project_sum(h0.copy(), target)

def coord_descent(h0, n, target, max_rounds, rng, step_sizes=(0.01, 0.003, 0.001)):
    h = h0.copy()
    c5 = compute_c5(h)
    for rd in range(max_rounds):
        imp = False
        order = rng.permutation(n)
        for i in order:
            for s in step_sizes:
                for sign in (1, -1):
                    hn = h.copy()
                    hn[i] = np.clip(hn[i] + sign * s, 0, 1)
                    hn = project_sum(hn, target)
                    c5n = compute_c5(hn)
                    if c5n < c5 - 1e-14:
                        h = hn
                        c5 = c5n
                        imp = True
                        break
                if imp:
                    break
        if not imp:
            break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    tl = lambda: budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = initial_h_values.copy().astype(float)
            best_n = len(best_h)
            best_h = project_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except:
            pass
    
    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n = len(best_h)
    target = n / 2.0
    
    # Stage 1: Projected gradient (fast, many iterations)
    if tl() > 120:
        h, c5 = pg_optimize(best_h, n, target, 3000, 50, 2000, 0.08, 25, 0.99)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Stage 2: SLSQP refinement with increasing alpha
    for al in [500, 2000, 8000, 20000]:
        if tl() < 80:
            break
        mi = min(400, max(50, int(tl() * 0.25)))
        h = slsqp_opt(best_h, n, target, al, mi, 20, 0.995)
