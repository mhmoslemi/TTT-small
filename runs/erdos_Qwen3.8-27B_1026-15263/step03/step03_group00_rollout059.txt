        zeros = np.where(h == 0)[0]

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum(h, target_sum):
    h = h.astype(float).copy()
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_v = np.exp(shifted)
    return m + np.log(exp_v.sum()) / alpha, exp_v / exp_v.sum()

def smooth_max_grad(h, w, alpha):

    # w has 2n-1 elements (weights for each correlation lag)
    # grad_i = sum_k w[k] * (d corr_k / d h_i)
