    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(corr, h, alpha, n, active_idx=None):
    if active_idx is not None and len(active_idx) > 0:
        ac = corr[active_idx]
    else:
        ac = corr
        active_idx = np.arange(len(corr))
    m = np.max(ac)
    sh = alpha * (ac - m)
    sh = np.clip(sh, -700.0, 700.0)
    ev = np.exp(sh)
    s = np.sum(ev)
    w = ev / s
    val = m + np.log(s / len(ev)) / alpha
    g = np.zeros(n)
    nrm1 = n - 1
    for p in range(len(active_idx)):
        wm = w[p]
        if wm < 1e-14:
            continue
        k = active_idx[p] - nrm1
        if k > 0:
            g[:n-k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n-k]
        elif k < 0:
            k2 = -k
            g[k2:] += wm * (1.0 - h[:n-k2])
            g[:n-k2] -= wm * h[k2:]
        else:
            g += wm * (1.0 - 2.0 * h)
    return val, g

def slsqp_opt(h0, n, ts, alpha, mxit, active_idx=None):
    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        c = np.correlate(hx, 1.0 - hx, mode='full')
        v, _ = smooth_max_grad(c, hx, alpha, n, active_idx)
        return v
    def grd(x):
        hx = np.clip(x, 0.0, 1.0)
        c = np.correlate(hx, 1.0 - hx, mode='full')
        _, g = smooth_max_grad(c, hx, alpha, n, active_idx)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, np.clip(h0, 0, 1).copy(), jac=grd, method='SLSQP',
                     bounds=bnds, constraints=cons, options={'maxiter': mxit, 'ftol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(np.clip(h0, 0, 1), ts)

def get_active(corr, n, thresh_frac=0.998, exclude_zero_lag=False):
    mx = np.max(corr)
    if mx <= 0:
        return np.array([n - 1])
    act = np.where(corr >= thresh_frac * mx)[0]
    if exclude_zero_lag:
        act = act[act != (n - 1)]
    if len(act) == 0:
        act = np.where(corr >= 0.99 * mx)[0]
    return act

def binary_push(h, n, ts, n_iter, strength_init=0.08):
    h = h.copy()
    c5 = compute_c5(h)
    strength = strength_init
    for _ in range(n_iter):
        h2 = h + strength * (2.0 * h - 1.0)
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_to_sum(h2, ts)
        c52 = compute_c5(h2)
        if c52 < c5 - 1e-16:
            h = h2
            c5 = c52
        else:
            h_s = h + strength * 0.3 * (2.0 * h - 1.0)
            h_s = np.clip(h_s, 0.0, 1.0)
            h_s = project_to_sum(h_s, ts)
            c5s = compute_c5(h_s)
            if c5s < c5 - 1e-16:
                h = h_s
                c5 = c5s
        strength *= 0.92
        if strength < 1e-5:
            break
    return h, c5

def coord_fine(h, n, ts, max_rounds, rng):
    h = h.copy()
    c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = get_active(corr, n, 0.995, True)
        improved = False
        for i in rng.permutation(n):
            best_c5 = c5
            best_d = 0.0
            for d in [0.005, -0.005, 0.02, -0.02, 0.001, -0.001]:
                h2 = h.copy()
                h2[i] = np.clip(h2[i] + d, 0.0, 1.0)
                h2 = project_to_sum(h2, ts)
                corr2 = np.correlate(h2, 1.0 - h2, mode='full')
                c52 = float(np.max(corr2[active]) * dx)
                if c52 < best_c5 - 1e-14:
                    best_c5 = c52
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5 = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, c5

def projected_subgrad(h, n, ts, n_iter, lr_init, alpha):
    h = h.copy()
    c5 = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        act = get_active(corr, n, 0.998, True)
        _, g = smooth_max_grad(corr, h, alpha, n, act)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            h_best = h.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.6
            if lr < 1e-7:
                break
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def tleft():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    if tleft() > 100:
        h_cur, c5_bp = binary_push(h_cur, n, ts, 30, 0.05)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_cur.copy()

    alphas_p1 = [5.0, 20.0, 60.0, 200.0]
    for alpha in alphas_p1:
        if tleft() < 80:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        act = get_active(corr, n, 0.995, False)
        mxit = max(100, min(350, int(tleft() * 0.25)))
        h_cur = slsqp_opt(h_cur, n, ts, alpha, mxit, act)
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
            best_n = n

    if tleft() > 80:
        h_cur, c5_bp = binary_push(h_cur, n, ts, 40, 0.06)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_cur.copy()

    alphas_p2 = [400.0, 1000.0, 3000.0]
    for alpha in alphas_p2:
        if tleft() < 70:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        act = get_active(corr, n, 0.998, True)
        if len(act) < 3:
            act = get_active(corr, n, 0.995, True)
        mxit = max(100, min(350, int(tleft() * 0.2)))
        h_cur = slsqp_opt(h_cur, n, ts, alpha, mxit, act)
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
            best_n = n

    if tleft() > 60:
        h_cur, c5_bp = binary_push(h_cur, n, ts, 50, 0.04)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_cur.copy()

    if tleft() > 100 and best_n < 1440:
        n_hi = 1440
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
        for alpha in [100.0, 500.0, 2000.0]:
            if tleft() < 60:
                break
            corr = np.correlate(h_hi, 1.0 - h_hi, mode='full')
            act = get_active(corr, n_hi, 0.997, True)
            if len(act) < 3:
                act = get_active(corr, n_hi, 0.994, True)
            mxit = max(60, min(250, int(tleft() * 0.2)))
            h_hi = slsqp_opt(h_hi, n_hi, ts_hi, alpha, mxit, act)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
        if tleft() > 40:
            h_hi, c5_hp = binary_push(h_hi, n_hi, ts_hi, 30, 0.04)
            if c5_hp < best_c5:
                best_c5 = c5_hp
                best_h = h_hi.copy()
        h_cur = h_hi
        n = n_hi
        ts = ts_hi

    if tleft() > 60:
        n = best_n
        ts = n / 2.0
        h_cur = best_h.copy()
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        act = get_active(corr, n, 0.999, True)
        if len(act) < 2:
            act = get_active(corr, n, 0.997, True)
        mxit = max(150, min(500, int(tleft() * 0.4)))
        h_pol = slsqp_opt(h_cur, n, ts, 8000.0, mxit, act)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if tleft() > 40:
        h_final, c5_pf = projecte_proj = project_to_sum(best_h, best_n / 2.0)
        h_final, c5_pf = binary_push(h_final, best_n, best_n / 2.0, 60, 0.03)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_final.copy()

    if tleft() > 50:
        n = best_n
        ts = n / 2.0
        h_cd, c5_cd = coord_fine(best_h, n, ts, 30, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if tleft() > 30:
        n = best_n
        ts = n / 2.0
        n_pg = min(2000, max(200, int(tleft() * 2)))
        h_pg, c5_pg = projected_subgrad(best_h, n, ts, n_pg, 0.03, 2000.0)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
