import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h = h.copy()
            h[mask] += diff / np.sum(mask)
        else:
            h = h.copy()
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_gradient(h, alpha, n_top=50):
    """Compute smooth max value and gradient using only top n_top correlation values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    # Get top indices
    if n_top < len(corr):
        top_idx = np.argsort(corr)[-n_top:]
        top_corr = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        top_corr = corr
    
    shifted = alpha * (top_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Compute gradient
    g = np.zeros(n)
    for idx, k in enumerate(top_idx):
        wi = w[idx]
        if wi < 1e-12:
            continue
        # corr[k] = sum_j h[j] * (1-h[j+k-n+1])
        # d/dh[i] = (1-h[i+k-n+1])*1{i+k-n+1 in [0,n)} - h[i-k+n-1]*1{i-k+n-1 in [0,n)}
        
        # Direct term: i+k-n+1 in [0, n-1] => i in [n-1-k, 2n-2-k]
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 2 - k)
        if i_lo <= i_hi:
            i_range = np.arange(i_lo, i_hi + 1)
            j_range = i_range + k - n + 1
            g[i_lo:i_hi+1] += wi * (1.0 - h[j_range])
        
        # Cross term: i-k+n-1 in [0, n-1] => i in [k-n+1, k+n-1]
        i_lo2 = max(0, k - n + 1)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            i_range2 = np.arange(i_lo2, i_hi2 + 1)
            j_range2 = i_range2 - k + n - 1
            g[i_lo2:i_hi2+1] -= wi * h[j_range2]
    
    return sm_val, g

def projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init, n_top=50):
    """Projected gradient with top-k gradient."""
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    lr = lr_init
    y = h.copy()
    t_val = 1.0
    n_iter = 0
    for it in range(n_iters):
        n_iter += 1
        sm_val, g = topk_gradient(y, alpha, n_top)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_best * 1.02 + 0.01:
            lr *= 0.5
            if lr < 1e-6:
                break
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.95
    return h_best, c5_best

def coordinate_descent_fast(h0, n, target_sum, n_rounds, rng, n_top=50):
    """Coordinate descent using topk gradient for line search evaluation."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-6:
                continue
            # Try a few values
            candidates = np.linspace(lo, hi, 7)
            best_val = c5_cur
            best_v = h[i]
            for v in candidates:
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val:
                    best_val = c5_tmp
                    best_v = v
            if best_v != h[i]:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_init(n, rng, n_modes=20):

    x = np.linspace(0, 2, n, endpoint=False)

    freqs = np.arange(1, n_modes + 1)

        amp = rng.uniform(0.01, 0.1) / np.sqrt(f)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    h = project_to_sum(h, n / 2.0)
    return h

        else:
            n_top_cur = max(30, n_top)
        sm_val, g = topk_gradient(y, alpha, n_top_cur)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_best * 1.03 + 0.005:
            lr *= 0.5
            if lr < 1e-7:
                break
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.04, 2.0)
        elif it % 10 == 0:
            lr *= 0.97
    return h_best, c5_best

def coord_descent_hybrid(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.4)
            hi = min(1.0, h[i] + 0.4)
            if hi - lo < 1e-7:
                continue
            candidates = np.linspace(lo, hi, 9)
            best_val = c5_cur
            best_v = h[i]
            for v in candidates:
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val:
                    best_val = c5_tmp
                    best_v = v
            if best_v != h[i] and abs(best_v - h[i]) > 1e-8:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def coord_descent_fine(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.05)
            hi = min(1.0, h[i] + 0.05)
            if hi - lo < 1e-8:
                continue
            candidates = np.linspace(lo, hi, 13)
            best_val = c5_cur
            best_v = h[i]
            for v in candidates:
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val - 1e-14:
                    best_val = c5_tmp
                    best_v = v
            if best_v != h[i] and abs(best_v - h[i]) > 1e-9:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_init(n, rng, n_modes=20):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    freqs = np.arange(1, n_modes + 1)
    for f in freqs:
        amp = rng.uniform(0.02, 0.15) / np.sqrt(f)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    h = project_to_sum(h, n / 2.0)
    return h

def binary_pattern_init(n, rng):
    h = np.zeros(n)
    block = rng.randint(3, 12)
    for i in range(n):
        h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    h = h + rng.normal(0, 0.1, size=n)
    h = project_to_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            n_init = len(best_h)
            best_h = project_to_sum(best_h, n_init / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        n_init = 720
        best_h = fourier_init(n_init, rng, n_modes=20)
        best_c5 = compute_c5(best_h)
        project_to_sum(best_h, n_init / 2.0)

    n_base = len(best_h)
    ts_base = n_base / 2.0

    if time_left() < 30:
        return (best_h, best_c5, n_base)

    # Phase 1: Projected gradient from initial, moderate alpha
    h_work = best_h.copy()
    alpha_schedule_p1 = [50.0, 100.0, 200.0, 400.0]
    for alpha in alpha_schedule_p1:
        if time_left() < 60:
            break
        n_iters = min(400, max(100, int(time_left() * 0.12)))
        h_pg, c5_pg = projected_gradient_nesterov(h_work, n_base, ts_base,
                                                   alpha=alpha, n_iters=n_iters,
                                                   lr_init=0.3, n_top=80)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
        h_work = h_pg

    # Phase 2: Higher alpha for refinement
    if time_left() > 120:
        h_work = best_h.copy()
        alpha_schedule_p2 = [800.0, 1500.0, 3000.0]
        for alpha in alpha_schedule_p2:
            if time_left() < 50:
                break
            n_iters = min(300, max(80, int(time_left() * 0.1)))
            h_pg, c5_pg = projected_gradient_nesterov(h_work, n_base, ts_base,
                                                       alpha=alpha, n_iters=n_iters,
                                                       lr_init=0.15, n_top=50)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            h_work = h_pg

    # Phase 3: Coordinate descent
    if time_left() > 100:
        n_rounds = min(8, max(3, int(time_left() // 12)))
        h_cd, c5_cd = coord_descent_hybrid(best_h, n_base, ts_base, n_rounds, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Multiple restarts from different starting points
    if time_left() > 150:
        n_starts = min(5, int(time_left() // 45))
        for s in range(n_starts):
            if time_left() < 50:
                break
            if s == 0:
                h_start = best_h + rng.normal(0, 0.04, size=n_base)
            elif s == 1:
                h_start = best_h + rng.normal(0, 0.08, size=n_base)
            elif s == 2:
                h_start = fourier_init(n_base, rng, n_modes=15 + s * 3)
            elif s == 3:
                h_start = binary_pattern_init(n_base, rng)
            else:
                h_start = best_h + rng.normal(0, 0.12, size=n_base)
            h_start = project_to_sum(h_start, ts_base)
            n_iters = min(250, max(80, int(time_left() * 0.08)))
            h_pg, c5_pg = projected_gradient_nesterov(h_start, n_base, ts_base,
                                                       alpha=300.0, n_iters=n_iters,
                                                       lr_init=0.25, n_top=70)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_work = h_pg
            else:
                h_work = h_pg

    # Phase 5: Fine coordinate descent
    if time_left() > 60:
        h_work = best_h.copy()
        n_rounds = min(15, max(5, int(time_left() // 8)))
        h_cd, c5_cd = coord_descent_fine(h_work, n_base, ts_base, n_rounds, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Final gradient polish
    if time_left() > 30:
        h_work = best_h.copy()
        for alpha in [5000.0, 10000.0]:
            if time_left() < 20:
                break
            n_iters = min(200, max(50, int(time_left() * 0.3)))
            h_pg, c5_pg = projected_gradient_nesterov(h_work, n_base, ts_base,
                                                       alpha=alpha, n_iters=n_iters,
                                                       lr_init=0.05, n_top=30)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            h_work = h_pg

    # Phase 7: Very fine coordinate descent
    if time_left() > 25:
        h_work = best_h.copy()
        h_cd, c5_cd = coord_descent_fine(h_work, n_base, ts_base, 5, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Tiny perturbation + re-optimize
    if time_left() > 50:
        for _ in range(3):
            if time_left() < 25:
                break
            h_pert = best_h + rng.normal(0, 0.005, size=n_base)
            h_pert = project_to_sum(h_pert, ts_base)
            h_pg, c5_pg = projected_gradient_nesterov(h_pert, n_base, ts_base,
                                                       alpha=5000.0, n_iters=100,
                                                       lr_init=0.03, n_top=30)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
