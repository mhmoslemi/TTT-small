```python
    return np.correlate(h, 1.0 - h, mode="full")
```

```python
            one_idx = np.where(h == 1)[0]
            to_flip = rng.choice(one_idx, ones_count - n_ones, replace=False)
            h[to_flip] = 0
        elif ones_count < n_ones:
            zero_idx = np.where(h == 0)[0]
            to_flip = rng.choice(zero_idx, n_ones - ones_count, replace=False)
            h[to_flip] = 1

        h[rng.choice(n, n_ones, replace=False)] = 1
```

```python
        T = T_start * (T_end / T_start) ** (it / max_iter)
        
        # Random swap (always valid)
        i = ones_arr[rng.randint(len(ones_arr))]
        j = zeros_arr[rng.randint(len(zeros_arr))]
        
        h[i] = 0
        h[j] = 1
        c5_new = compute_c5(h.astype(float))
        
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            # Update arrays
            ones_arr[ones_arr == i] = j  # i becomes 0, j becomes 1
            zeros_arr[zeros_arr == j] = i
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h.astype(float), best_c5
```
