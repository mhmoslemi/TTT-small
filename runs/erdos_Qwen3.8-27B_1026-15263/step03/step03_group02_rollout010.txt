```python
def projected_subgradient(h, target_sum, n_shrink=10, step=0.01, iters=100000):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        # Find shifts near the max
        thresh = m - n_shrink * (2.0/n)  # shifts within n_shrink*dx of max
        active = np.where(corr >= thresh)[0]
        
        # Gradient of sum of active correlations
        grad = np.zeros(n)
        for m_idx in active:
            k = m_idx - n + 1
            if k >= 0:
                valid = np.arange(n - k)
                grad[valid] += (1.0 - h[valid])
                grad[valid - k] += (-h[valid])
            else:
                valid = np.arange(-k, n)
                grad[valid] += (1.0 - h[valid])
                grad[valid + k] += (-h[valid])
        
        # Project: subtract mean to preserve sum
        grad -= np.mean(grad)
        
        # Clip to box
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            step *= 0.9
```

Actually, I realize I should think more carefully about the projected subgradient / active set approach. The key idea is:

At the optimum, the active set of shifts (those achieving the max) should be such that:
- All active correlations are equal (to the max value C5/dx)
- The gradient of the active correlations, projected onto the feasible set, should have a component in the direction that the optimum allows

By focusing on the active set, we can do a much more targeted optimization.

Here's a refined approach:
1. Compute correlations, find the max.
2. Identify active shifts: those with corr > max - epsilon (epsilon small, like 0.5 correlation units = ~0.004 in C5).
3. The active set defines a "reduced" problem: minimize the maximum of just the active correlations.
4. For the active set, construct the gradient and do a Newton-like step.
5. Re-identify the active set (it changes as h changes).

This is essentially the method of "active set" for min-max problems.

```python
def active_set_optimize(h, target_sum, max_iter=5000, tol=1e-8):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        
        # Active set: shifts with corr >= m - 1.0 (within 1 correlation unit of max)
        active_thresh = m - 1.0
        active = np.where(corr >= active_thresh)[0]
        
        # Compute gradient of the sum of active correlations
        grad = np.zeros(n)
        for m_idx in active:
            k = m_idx - n + 1
            if k >= 0:
                valid = np.arange(n - k)
                grad[valid] += (1.0 - h[valid])
                grad[valid - k] += (-h[valid])
            elif k < 0:
                valid = np.arange(-k, n)
                grad[valid] += (1.0 - h[valid])
                grad[valid + k] += (-h[valid])
        
        # We want to decrease the active correlations.
        # But changing h changes ALL correlations.
        # Use a line search: take a small step in -grad direction.
        
        # Project grad to satisfy sum constraint
        grad_centered = grad - np.mean(grad)
        
        # Determine step size via line search
        best_step = 0.01
        best_c5 = c5
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = h - step * grad_centered
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_step = step
                best_h = h_try
        
        if best_step == 0.01 and best_c5 >= c5 - 1e-12:
            # No improvement, try smaller
            step = 0.0001
            h_try = h - step * grad_centered
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5:
                h = h_try
                c5 = c5_try
            else:
                break
        else:
            h = best_h
            c5 = best_c5
    
    return h, c5
```
