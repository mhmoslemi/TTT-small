    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    n = len(h)

    # s=0 corresponds to k=n-1
    # We want s != 0, so exclude k=n-1
    return ac  # full array, we'll handle which index is s=0

    For the maximin problem: max_h min_{s!=0} Q_s(h)
    Q_s(h) = sum_{valid i} h[i]*h[i+s]
    
    Returns: (min_value, subgradient_vector)

    ac = np.correlate(h, h, mode='full')

    min_val = np.min(ac)  # We'll just use all lags for simplicity (s=0 is always n/2 which is large)

    # But to be safe, let's exclude it.
    ac_no0 = ac.copy()
    ac_no0[n-1] = np.inf
    min_val = np.min(ac_no0)
    
    active = np.where(ac_no0 <= min_val + tol)[0]
    
    # Subgradient of min is in convex hull of gradients of active functions
    # Q_s(h) = sum_i h[i]*h[i+s]
    # dQ_s/dh[j] = h[j-s] + h[j+s] (with valid index handling)
    # This is 2*(A_s @ h)[j] where A_s is the shift matrix
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s == 0:
            continue
        if s > 0:
            # h[j+s] valid for j in [0, n-1-s]
            grad[:n-s] += h[s:]
            # h[j-s] valid for j in [s, n-1]  
            grad[s:] += h[:n-s]
        else:
            ss = -s
            # h[j-s] = h[j+ss] valid for j in [0, n-1-ss]
            grad[:n-ss] += h[ss:]
            # h[j+s] = h[j-ss] valid for j in [ss, n-1]
            grad[ss:] += h[:n-ss]
    
    grad /= len(active)
    return min_val, grad

- t=0, t=1, or
- an intersection point of two of the lines

- At t=0 if the function is decreasing throughout

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_max_c5(h, tol=1e-6):
    """
    Compute the (average) subgradient of C5(h) = max_k corr[k]*dx.
    Returns (c5_value, subgradient).
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    max_val = np.max(c5_vals)
    
    # Active set
    active = np.where(c5_vals >= max_val - tol)[0]
    
    # Subgradient: average of d(corr[k]*dx)/dh for active k
    # d(corr[k])/dh[j] = (1-h[j+s]) * [0<=j+s<n] - h[j-s] * [0<=j-s<n]
    # where s = k - (n-1)
    n1 = n - 1
    grad = np.zeros(n)
    
    for k in active:
        s = k - n1
        if s > 0:
            # j+s valid: j in [0, n-1-s]
            grad[:n-s] += (1.0 - h[s:n])
            # j-s valid: j in [s, n-1]
            grad[s:] -= h[:n-s]
        elif s < 0:
            ss = -s
            # j+s = j-ss valid: j in [ss, n-1]
            grad[ss:] += (1.0 - h[:n-ss])
            # j-s = j+ss valid: j in [0, n-1-ss]
            grad[:n-ss] -= h[ss:]
        else:
            # s=0: s=0 means j+0=j always valid, j-0=j always valid
            grad[:] += (1.0 - h)
            grad[:] -= h
            # So grad[:] = 1 - 2*h
    
    grad *= dx / len(active)
    return max_val, grad

def subgradient_bures(h0, max_steps, lr_init, time_fn):

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_max_c5(h, tol=1e-8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    max_val = float(np.max(c5_vals))
    n1 = n - 1
    active = np.where(c5_vals >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - n1
        if s > 0:
            grad[:n - s] += (1.0 - h[s:n])
            grad[s:] -= h[:n - s]
        elif s < 0:
            t = -s
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
        else:
            grad[:] += (1.0 - 2.0 * h)
    grad *= dx / len(active)
    return max_val, grad

        c5_new = compute_c5(h_new)
        if c5_new < c5_val - 1e-14:
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 0.3)

def coordinate_descent_step(h, n_sweeps=20, time_fn=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        if time_fn and time_fn() < 15:
            break
        improved = False
        for j in range(n):
            if time_fn and time_fn() < 5:
                break
            t_old = h[j]
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = t_old
            n1 = n - 1
            slopes = np.zeros(2 * n - 1)
            k_min = n1 - j
            k_max = n1 + (n - 1 - j)
            slopes[k_min:k_max + 1] += (1.0 - h[j:j + n - j])
            k_min2 = j + 1
            k_max2 = n1 + j
            slopes[k_min2:k_max2 + 1] -= h[n - 1::-1]
            base = corr0 * dx
            candidates = [0.0, 1.0]
            g0 = float(np.max(base))
            g1 = float(np.max(base + slopes))
            k0 = int(np.argmax(base))
            k1 = int(np.argmax(base + slopes))
            if k0 != k1 and abs(slopes[k0] - slopes[k1]) > 1e-15:
                t_int = (base[k1] - base[k0]) / (slopes[k0] - slopes[k1])
                if 0.0 < t_int < 1.0:
                    candidates.append(t_int)
            best_t = t_old
            best_val = float(np.max(base + t_old * slopes))
            for tc in candidates:

            best_c5 = c5_cur

    n = len(h)

    coeffs, _, _, _ = np.linalg.lstsq(C, h - 0.5, rcond=None)
    h_sm = 0.5 + C @ coeffs
    return h_sm
