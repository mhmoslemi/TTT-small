import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_cd_pass(h, target_sum, n_sweeps=1, order=None):
    n = len(h)
    if order is None:
        order = np.arange(n)
    h_work = h.copy()
    m_vals = np.arange(2 * n - 1)
    for sweep in range(n_sweeps):
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        for j in order:
            delta_lo = -h_work[j]
            delta_hi = 1.0 - h_work[j]
            if delta_hi - delta_lo < 1e-14:
                continue
            idx_pos = j + m_vals - n + 1
            idx_neg = j - m_vals + n - 1
            a = np.zeros(2 * n - 1)
            mp = (idx_pos >= 0) & (idx_pos < n)
            mn = (idx_neg >= 0) & (idx_neg < n)

            a[mn] += (1.0 - h_work[idx_neg[mn]])
            pre_pos = a > 1e-14
            pre_neg = a < -1e-14
            pre_zero = ~pre_pos & ~pre_neg
            lo_T = 0.0
            hi_T = float(np.max(corr))
            if np.any(pre_zero) and np.max(corr[pre_zero]) >= hi_T - 1e-14:
                continue
            best_delta = 0.0
            for _ in range(50):

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k, t_start, budget_s):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        if time.time() - t_start > budget_s - 60:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m_val in top_idx:
            jj1 = j_arr + m_val - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - m_val + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        lr = lr_init / (1.0 + 0.05 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

            continue

        if best_n != n:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_work = np.interp(x_new, x_old, best_h)
        else:
            h_work = best_h.copy()
        h_work = project_to_sum(h_work, ts)
        try_update(h_work, n)

        if _tl(t_start, budget_s) > 300:
            h_sub, _ = topk_subgradient(h_work, n, ts, 2000, 0.04, 15, t_start, budget_s)
            try_update(h_sub, n)
            h_work = h_sub
            h_cd1 = exact_cd_pass(h_work, ts, n_sweeps=2)
            try_update(h_cd1, n)
            h_work = h_cd1

        n_cd_sweeps = max(4, min(40, int((_tl(t_start, budget_s) - 80) / 25)))
        for pass_i in range(n_cd_sweeps):
            if _tl(t_start, budget_s) < 60:
                break
            if pass_i % 3 == 0:
                order = np.arange(n)
            elif pass_i % 3 == 1:
                order = np.arange(n - 1, -1, -1)
            else:
                order = rng.permutation(n)
            h_cd = exact_cd_pass(h_work, ts, n_sweeps=1, order=order)
            try_update(h_cd, n)
            if compute_c5(h_cd) - compute_c5(h_work) > -1e-10:
                h_work = h_cd
            else:
                h_work = h_cd

        if _tl(t_start, budget_s) > 100:
            for pert_round in range(4):
                if _tl(t_start, budget_s) < 50:
                    break
                h_pert = h_work.copy()
                n_pert = max(5, n // 20)
                idx_pert = rng.choice(n, n_pert, replace=False)
                h_pert[idx_pert] += rng.uniform(-0.03, 0.03, n_pert)
                h_pert = project_to_sum(h_pert, ts)
                h_pert = exact_cd_pass(h_pert, ts, n_sweeps=2)
                try_update(h_pert, n)
                if compute_c5(h_pert) < compute_c5(h_work):
                    h_work = h_pert

        if _tl(t_start, budget_s) > 60:
            x = np.linspace(0, 2, n, endpoint=False)
            for freq in [3, 5, 7, 11]:
                if _tl(t_start, budget_s) < 40:
                    break
                for amp in [0.02, 0.04, 0.06]:
                    h_pert2 = h_work + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
                    h_pert2 = project_to_sum(h_pert2, ts)
                    h_pert2 = exact_cd_pass(h_pert2, ts, n_sweeps=1)
                    try_update(h_pert2, n)

    if best_h is not None and _tl(t_start, budget_s) > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        n_final_passes = max(2, min(8, int((_tl(t_start, budget_s) - 15) / 20)))
        for i in range(n_final_passes):
            if _tl(t_start, budget_s) < 15:
                break
            h_f = exact_cd_pass(h_f, ts_f, n_sweeps=1, order=np.arange(n_f))
            try_update(h_f, n_f)
            h_f = exact_cd_pass(h_f, ts_f, n_sweeps=1, order=np.arange(n_f - 1, -1, -1))
            try_update(h_f, n_f)
            h_f = exact_cd_pass(h_f, ts_f, n_sweeps=1, order=rng.permutation(n_f))
            try_update(h_f, n_f)

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
