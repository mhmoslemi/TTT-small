    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:

        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def f(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_gradient_descent(h0, n, target_sum, alpha, lr, maxiter, momentum=0.9):
    h = h0.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(maxiter):
        obj, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        # Project to [0,1] and sum constraint
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    fi = j_idx - m_idx + n - 1
    si = j_idx + m_idx - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    return np.sum(cf - cs, axis=1)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(x):
        return smooth_max_val(x, alpha)
    def g(x):
        return smooth_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, penalty, maxiter=2000):
    def f(x):
        sm = smooth_max_val(x, alpha)
        s = np.sum(x) - target_sum
        return sm + penalty * s * s
    def g(x):
        gr = smooth_max_grad(x, alpha)
        s = np.sum(x) - target_sum
        return gr + 2 * penalty * s * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa(h_start, n, t_end, rng):
    target_sum = n // 2
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.012
    min_T = 0.002
    no_improve = 0
    max_no_improve = 30000
    while time.time() < t_end:
        move_type = rng.randint(3)
        if move_type == 0:
            # Swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif move_type == 1:
            # Reverse a segment
            a = rng.randint(n)
            b = a + rng.randint(3, min(30, n - a))
            if b >= n:
                continue
            h[a:b+1] = h[a:b+1][::-1]
        else:
            # Move a 1 to a random 0 position (or vice versa)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (rng.uniform() < np.exp(-max(0, delta) / T)):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Undo the move
            if move_type == 0:
                h[i], h[j] = h[j], h[i]
            elif move_type == 1:
                h[a:b+1] = h[a:b+1][::-1]
            else:
                h[i], h[j] = h[j], h[i]
        
        if no_improve > max_no_improve:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.025)
            no_improve = 0
        
        T = max(T * 0.999998, min_T)
    
    return best_h, best_c5

def legendre_init(n, p, rng):
    """Create a binary sequence based on Legendre sequence modulo p."""
    # Compute Legendre symbols
    s = np.zeros(p, dtype=float)
    for j in range(1, p):
        s[j] = 1.0 if pow(j, (p - 1) // 2, p) == 1 else 0.0
    # s[0] = 0 by default
    # Embed into array of length n
    h = np.zeros(n, dtype=float)
    for i in range(n):
        h[i] = s[i % p]
    # Adjust to have exactly n//2 ones
    target = n // 2
    current = int(np.sum(h))
    if current > target:
        excess = current - target
        ones_idx = np.where(h == 1.0)[0]
        rng.shuffle(ones_idx)
        h[ones_idx[:excess]] = 0.0
    elif current < target:
        deficit = target - current
        zeros_idx = np.where(h == 0.0)[0]
        rng.shuffle(zeros_idx)
        h[zeros_idx[:deficit]] = 1.0
    return h

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0

        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    fi = j_idx - m_idx + n - 1
    si = j_idx + m_idx - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_val_grad(x, alpha)[0]

        return smooth_max_val_grad(x, alpha)[1]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:

    def f(x):
        sm = smooth_max_val_grad(x, alpha)[0]
        s = np.sum(x) - target_sum
        return sm + penalty * s * s
    def g(x):
        _, gr = smooth_max_val_grad(x, alpha)
        s = np.sum(x) - target_sum
        return gr + 2.0 * penalty * s * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa(h_start, n, t_end, rng, use_reversal=True):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.014
    min_T = 0.002
    no_improve = 0
    max_no_improve = 25000
    while time.time() < t_end:
        r = rng.rand()
        if r < 0.5:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            undo_type = 0
        elif r < 0.75 and use_reversal:
            a = rng.randint(n)
            length = rng.randint(3, min(25, n - a))
            b = a + length
            if b >= n:
                continue
            h[a:b] = h[a:b][::-1]
            undo_type = 1
        else:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            undo_type = 0

        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        accept = delta < 0 or (rng.uniform() < np.exp(-max(0.0, delta) / T))
        if accept:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if undo_type == 0:
                h[i], h[j] = h[j], h[i]
            elif undo_type == 1:
                h[a:b] = h[a:b][::-1]
            else:
                h[i], h[j] = h[j], h[i]

        if no_improve > max_no_improve:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.03)
            no_improve = 0
        T = max(T * 0.999996, min_T)
    return best_h, best_c5

def legendre_init(n, p, rng):
    s = np.zeros(p, dtype=float)
    for j in range(1, p):
        s[j] = 1.0 if pow(j, (p - 1) // 2, p) == 1 else 0.0
    h = np.zeros(n, dtype=float)
    for i in range(n):
        h[i] = s[i % p]
    target = n // 2
    current = int(np.sum(h))
    if current > target:
        excess = current - target
        ones_idx = np.where(h == 1.0)[0]
        perm = rng.permutation(ones_idx)
        h[perm[:excess]] = 0.0
    elif current < target:
        deficit = target - current
        zeros_idx = np.where(h == 0.0)[0]
        perm = rng.permutation(zeros_idx)
        h[perm[:deficit]] = 1.0
    return h

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    candidates = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            h_init = np.array(initial_h_values, dtype=float)
            n_init = len(h_init)
            h_init = project_to_sum(h_init, n_init / 2.0)
            candidates.append((compute_c5(h_init), h_init, n_init))
        except Exception:
            pass

    primes_to_try = [239, 241, 227, 233]
    for p in primes_to_try:
        try:
            h_leg = legendre_init(240, p, rng)
            candidates.append((compute_c5(h_leg), h_leg, 240))
        except Exception:
            pass

    for _ in range(8):
        h_rand = (rng.rand(240) < 0.5).astype(float)
        h_rand = make_binary(h_rand, 240, 120, rng)
        candidates.append((compute_c5(h_rand), h_rand, 240))

    candidates.sort(key=lambda x: x[0])
    best_c5, best_h, best_n = candidates[0]

    n_main = 240
    ts_main = n_main / 2.0

    h_bin = make_binary(best_h, n_main, 120, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.35 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    cont_end = t_start + 0.55 * budget_s
    h_work = best_h.copy()
    for alpha in [30, 100, 500, 2000]:
        if time.time() > cont_end:
            break
        h_opt = slsqp_optimize(h_work, n_main, ts_main, alpha, maxiter=1000)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_work = h_opt.copy()
        else:
            h_work = h_opt.copy()

    if time.time() < cont_end:
        for pen in [100.0, 1000.0]:
            if time.time() > cont_end:
                break
            h_lb = lbfgsb_optimize(best_h, n_main, ts_main, 5000, pen, maxiter=2000)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()

    sa2_end = t_start + 0.72 * budget_s
    if remaining() > 120:
        h_bin2 = make_binary(best_h, n_main, 120, rng)
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, sa2_end, rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    polish_end = t_start + 0.82 * budget_s
    if remaining() > 80:
        for alpha in [5000, 20000, 80000]:
            if time.time() > polish_end:
                break
            h_fin = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=1500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    for trial in range(6):
        if time.time() > t_start + 0.88 * budget_s:
            break
        noise = 0.12 * (1.0 - trial / 6.0) + 0.03
        h_shake = best_h + rng.uniform(-noise, noise, n_main)
        h_shake = project_to_sum(h_shake, ts_main)
        alpha_shake = [500, 2000, 10000][min(trial // 2, 2)]
        h_opt2 = slsqp_optimize(h_shake, n_main, ts_main, alpha_shake, maxiter=1200)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    for n_try in [480]:
        if remaining() < 150:
            break
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_bin_hi = make_binary(h_interp, n_try, n_try // 2, rng)
        hi_sa_end = min(t_start + budget_s - 40, time.time() + 80)
        h_hi, c5_hi = binary_sa(h_bin_hi, n_try, hi_sa_end, rng)
        if time.time() < t_start + budget_s - 30:
            h_hi = slsqp_optimize(h_hi, n_try, ts_try, 500, maxiter=1000)
            c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_try

    final_alpha = max(50000, int(200000 / max(best_c5, 0.1)))
    if remaining() > 20:
        h_final = slsqp_optimize(best_h, best_n, best_n / 2.0, final_alpha, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
