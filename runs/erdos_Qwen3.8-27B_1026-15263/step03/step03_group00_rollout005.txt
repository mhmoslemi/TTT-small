Let me think about the projected gradient more carefully:

The objective is F(h) = smooth_max(corr(h)) where corr(h) = relate(h, 1-h, full)

Gradient: ∂F/∂h[i] = ∑_k w_k * ∂corr_k/∂h[i]

where w_k is the softmax weight over correlations.

For the constraint sum(h) = n/2:
- Projected gradient: h_new = h - lr * grad
- Then project onto {0≤h≤1, sum(h)=n/2}

The projection onto {0≤h≤1, sum(h)=s} can be done by:
1. Clip to [0,1]
2. Adjust by adding a constant to all free variables to meet the sum
3. Iterate until convergence (or use a water-filling approach)

Actually, the projection onto the intersection of the hypercube and a hyperplane is:
- Find λ such that ∑clip(h_i - lr*grad_i - λ, 0, 1) = target_sum
- This is a 1D root-finding problem (bisection on λ)

This is much faster than SLSQP!

```python
def project_onto_box_sum(v, target_sum):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(v)
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    # g(lambda) = sum(clip(v_i - lambda, 0, 1)) is decreasing in lambda
    lo = v.min() - 1
    hi = v.max() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)
```

This is O(n) per projection, much faster than the iterative project_to_sum.

Actually, let me reconsider the gradient computation. The correlation at lag k (where k ranges from -(n-1) to n-1, indexed as 0 to 2n-2) is:
corr[m] = sum_i h[i] * (1-h[i+m-(n-1)]) for valid i

The gradient w.r.t. h[j]:
For lag m (correlation index), the terms involving h[j] are:
- h[j] appears in h[j] * (1-h[j+l]) where l = m-(n-1), contributing (1-h[j+l]) if j+l < n
- h[j] appears in h[i] * (1-h[j]) where i = j-l, contributing -h[i] if i >= 0

So grad[j] = sum_m w_m * [(1-h[j+l_m]) * 1[j+l_m < n] - h[j-l_m] * 1[j-l_m >= 0]]
where l_m = m - (n-1)

The previous code does this correctly with the index matrices. Let me keep that.

Now, for the Adam optimizer:
```
m_t = beta1 * m_{t-1} + (1-beta1) * g_t
v_t = beta2 * v_{t-1} + (1-beta2) * g_t²
m_hat = m_t / (1 - beta1^t)
v_hat = v_t / (1 - beta2^t)
h_new = h - lr * m_hat / (sqrt(v_hat) + eps)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient computation
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def project_box_sum(v, target_sum):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target_sum} using bisection on lambda."""
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def adam_projected_grad(h0, target_sum, alpha, lr, n_iters, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        obj, grad = smooth_max_and_grad(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        # Gradient for the active lag
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def multi_lag_coordinate(h, target_sum, max_iters=5000, n_lags=3):
    """Coordinate descent that considers top n_lags correlations simultaneously."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        # Get top n_lags correlation indices
        top_idx = np.argsort(corr)[::-1][:n_lags]
        # Weighted gradient (equal weights for top lags)
        g = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            if lag == 0:
                g[:] += 1.0 - 2.0 * h
            elif lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                k = -lag
                g[:n - k] += (1.0 - h[k:])
                g[k:] -= h[:n - k]
        g /= n_lags
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = 0.1 * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)
```

```python
def project_box_sum(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-14) & (result < 1.0 - 1e-14)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def adam_projected_grad(h0, target_sum, alpha, lr, n_iters, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        obj, grad = smooth_max_and_grad(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:

        h = h_new
    return best_h, best_c5

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.12):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target_sum)
            stale += 1
            if stale > 200:

        else:

    return h, c5

            stale += 1
            if stale > 200:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    t_deadline = t_start + budget_s - 3.0

    def remaining():
        return t_deadline - time.time()

    def ok():
        return remaining() > 1.0

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: Adam with annealed alpha at parent resolution
    alphas = [30, 100, 500, 2000, 8000]
    for alpha in alphas:
        if not ok():
            break
        lr = 0.02 / max(1.0, alpha / 100.0)
        n_iter = 4000 if alpha <= 500 else 2500
        h_opt, c5_opt = adam_projected_grad(best_h, ts_parent, alpha, lr, n_iter)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Coordinate descent
    if ok():
        h_cd, c5_cd = coordinate_descent(best_h, ts_parent, max_iters=40000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Multi-lag coordinate descent
    if ok():
        h_ml, c5_ml = multi_lag_coordinate(best_h, ts_parent, max_iters=15000, n_lags=3)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()
    if ok():
        h_ml5, c5_ml5 = multi_lag_coordinate(best_h, ts_parent, max_iters=15000, n_lags=5)
        if c5_ml5 < best_c5:
            best_c5 = c5_ml5
            best_h = h_ml5.copy()

    # Phase 4: Resolution sweep
    res_list = [n_parent]
    for nr in [480, 600, 720, 900, 1080]:
        if nr != n_parent and nr % 2 == 0:
            res_list.append(nr)
    res_list.sort()

    for n_tr in res_list:
        if not ok() or remaining() < 60:
            break
        if n_tr == n_parent:
            continue
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_box_sum(h_interp, ts_tr)
        for alpha in [100, 800]:
            if not ok():
                break
            lr = 0.015 / max(1.0, alpha / 100.0)
            h_opt, c5_opt = adam_projected_grad(h_interp, ts_tr, alpha, lr, 3000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt
        if ok():
            h_cd, c5_cd = coordinate_descent(h_interp, ts_tr, max_iters=25000, step_frac=0.08)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr

    # Phase 5: Diversity via noise perturbation
    n_best = best_n
    ts_best = n_best / 2.0
    n_trials = 12
    for trial in range(n_trials):
        if not ok() or remaining() < 40:
            break
        noise_level = 0.015 + 0.035 * (1.0 - trial / float(n_trials))
        h_shake = best_h + rng.normal(0, noise_level, n_best)
        h_shake = project_box_sum(h_shake, ts_best)
        h_cd, c5_cd = coordinate_descent(h_shake, ts_best, max_iters=8000, step_frac=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if ok():
            h_opt, c5_opt = adam_projected_grad(h_cd, ts_best, 1000, 0.01, 1500)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Final high-precision refinement
    if ok():
        h_final, c5_final = coordinate_descent(best_h, ts_best, max_iters=50000, step_frac=0.04)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if ok():
        h_ml_final, c5_ml_final = multi_lag_coordinate(best_h, ts_best, max_iters=20000, n_lags=4, step_frac=0.03)
        if c5_ml_final < best_c5:
            best_c5 = c5_ml_final
            best_h = h_ml_final.copy()

    if ok():
        h_opt, c5_opt = adam_projected_grad(best_h, ts_best, 20000, 0.003, 3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, ts_best)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
```
