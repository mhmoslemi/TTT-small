def lbgfsb_optimize(h0, target_sum, alpha, maxiter=2000):
    n = len(h0)
    penalty_weight = 1000.0
    
    def obj_and_grad(x):
        h = np.clip(x, 0, 1)
        so, sg = smooth_obj_and_grad(h, alpha)
        penalty = penalty_weight * (np.sum(x) - target_sum) ** 2
        pen_grad = 2 * penalty_weight * (np.sum(x) - target_sum) * np.ones(n)
        return so + penalty, sg + pen_grad
    
    res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                   jac=True, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_res = np.clip(res.x, 0, 1)
    h_res = project_to_sum(h_res, target_sum)
    return h_res

                        hi_r = mid

A better approach: since we only have one equality constraint, we can use the method of Lagrange multipliers implicitly. After L-BFGS-B converges, we project to the feasible set. The projection to {h: 0≤h≤1, sum(h)=n/2} is well-defined and can be computed efficiently.

Actually, let me just use a moderate penalty (mu=100) and then project at the end. The L-BFGS-B will naturally drive the sum constraint violation to be small.

Let me also add a **subgradient step** approach as an alternative to pair descent. The idea is:
1. Compute corr and find the index k* that achieves the max
2. Compute the subgradient: grad = d(corr[k*])/dh
3. Take a step: h_new = h - step * grad
4. Project to feasible set

This is simpler than pair descent and might escape some local optima differently.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    s_mat = k_idx - (n - 1)
    idx1 = j_idx + s_mat
    valid1 = (idx1 >= 0) & (idx1 < n)
    h_at_1 = np.take(h, np.clip(idx1, 0, n - 1))
    term1 = valid1 * (1.0 - h_at_1) * w[None, :]
    idx2 = j_idx - s_mat
    valid2 = (idx2 >= 0) & (idx2 < n)
    h_at_2 = np.take(h, np.clip(idx2, 0, n - 1))
    term2 = -valid2 * h_at_2 * w[None, :]
    grad = np.sum(term1 + term2, axis=1)
    return obj, grad

def lbgfsb_optimize(h0, target_sum, alpha, maxiter=2000):
    n = len(h0)
    mu = 200.0

    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        so, sg = smooth_obj_and_grad(h, alpha)
        penalty = mu * (np.sum(x) - target_sum) ** 2
        pen_grad = 2.0 * mu * (np.sum(x) - target_sum) * np.ones(n)
        return so + penalty, sg + pen_grad

    try:
        res = minimize(obj_and_grad, h0.copy(), method='L-BFGS-B',
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        h_res = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_res = h0.copy()
    h_res = project_to_sum(h_res, target_sum)
    return h_res

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    s_arr = np.arange(2 * n - 1) - nm1
    idx1 = j + s_arr
    valid1 = (idx1 >= 0) & (idx1 < n)
    B[valid1] += (1.0 - h[np.clip(idx1, 0, n - 1)[valid1]])
    idx2 = j - s_arr
    valid2 = (idx2 >= 0) & (idx2 < n)
    B[valid2] -= h[np.clip(idx2, 0, n - 1)[valid2]]
    return B

def solve_1d_min(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=8):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n]
        for _ in range(max(0, n_cand - 4)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_min(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def subgradient_step(h, corr, step_size, target_sum):
    n = len(h)
    k_star = int(np.argmax(corr))
    s_star = k_star - (n - 1)
    grad = np.zeros(n)
    j_arr = np.arange(n)
    idx1 = j_arr + s_star
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[np.clip(idx1, 0, n - 1)[valid1]])
    idx2 = j_arr - s_star
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[np.clip(idx2, 0, n - 1)[valid2]]
    h_new = h - step_size * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: L-BFGS-B with alpha annealing (40% budget)
    t_end1 = t_start + 0.42 * budget_s
    h_work = best_h.copy()
    alphas = [30, 60, 100, 200, 400, 800, 1500]
    for alpha in alphas:
        if time.time() >= t_end1:
            break
        elapsed_in_phase = time.time() - t_start + (t_end1 - t_start) * 0
        remaining = t_end1 - time.time()
        maxiter = min(3000, max(200, int(remaining / 0.04)))
        h_work = lbgfsb_optimize(h_work, best_n / 2.0, alpha, maxiter=maxiter)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 2: Pair coordinate descent with restarts (28% budget)
    t_end2 = t_start + 0.70 * budget_s
    h_work = best_h.copy()
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall_count = 0
    while time.time() < t_end2:
        h_work, corr_w, _ = pair_descent_sweep(h_work, corr_w, rng, n_cand=10)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count >= 150:
            noise_amp = min(0.08, 0.02 + 0.01 * (stall_count // 150))
            h_work = project_to_sum(h_work + rng.uniform(-noise_amp, noise_amp, best_n), best_n / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0
        elif stall_count >= 60 and stall_count % 20 == 0:
            step_size = 0.01 / best_n
            h_work = subgradient_step(h_work, corr_w, step_size, best_n / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    # Phase 3: L-BFGS-B again on best found (12% budget)
    t_end3 = t_start + 0.82 * budget_s
    h_work = best_h.copy()
    for alpha in [200, 500, 1000, 2000]:
        if time.time() >= t_end3:
            break
        remaining = t_end3 - time.time()
        maxiter = min(2000, max(100, int(remaining / 0.05)))
        h_work = lbgfsb_optimize(h_work, best_n / 2.0, alpha, maxiter=maxiter)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 4: Multi-resolution (10% budget)
    t_end4 = t_start + 0.92 * budget_s
    resolutions = [320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end4:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_st = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        for alpha in [100, 500]:
            if time.time() >= t_end4:
                break
            remaining = t_end4 - time.time()
            maxiter = min(1500, max(100, int(remaining / 0.08)))
            h_st = lbgfsb_optimize(h_st, ts_tr, alpha, maxiter=maxiter)
        corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
        max_sweeps = max(3, int((t_end4 - time.time()) / 4.0))
        for _ in range(min(max_sweeps, 25)):
            if time.time() >= t_end4:
                break
            h_st, corr_st, _ = pair_descent_sweep(h_st, corr_st, rng, n_cand=6)
        c5_st = compute_c5(h_st)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            best_n = n_tr

    # Phase 5: Final pair descent polish (8% budget)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    t_end5 = t_start + budget_s - 3
    while time.time() < t_end5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
