def compute_c5(h):

def project_to_sum(h, target_sum):

    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha

        else:

def corr_and_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_v = np.exp(alpha * (corr - max_c))
    w = exp_v / np.sum(exp_v)
    obj = max_c + np.log(np.sum(exp_v)) / alpha
    return corr, w, obj

def grad_from_corr_w(h, corr, w, alpha):
    n = len(h)
    lags = np.arange(2*n-1) - (n-1)
    m_idx = np.arange(n)[:, None]
    l_idx = lags[None, :]
    p1 = m_idx + l_idx
    p2 = m_idx - l_idx
    v1 = (p1 >= 0) & (p1 < n)
    v2 = (p2 >= 0) & (p2 < n)
    p1c = np.clip(p1, 0, n-1)
    p2c = np.clip(p2, 0, n-1)
    term1 = v1.astype(float) * (1.0 - h[p1c])
    term2 = v2.astype(float) * h[p2c]
    grad = np.sum((term1 - term2) * w[None, :], axis=1)
    return grad

def slsqp_opt(h0, ts, alpha, maxiter=500):
    n = len(h0)
    def f(x):
        c, w, obj = corr_and_weights(x, alpha)
        return obj
    def g(x):
        c, w, _ = corr_and_weights(x, alpha)
        return grad_from_corr_w(x, c, w, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})

When h[i_src] decreases by delta and h[i_dst] increases by delta, how does the correlate change?

corr[k] = sum_j h[j] * (1-h[j+k-(n-1)]) for valid j

The change in corr[k] from changing h[i_src] by -delta:
- For terms where j = i_src: contribution changes by -delta * (1-h[i_src+k-(n-1)])
- For terms where j+k-(n-1) = i_src, i.e., j = i_src-k+(n-1): contribution changes by h[j] * delta (because (1-h[j+k-(n-1)]) changes by +delta)

So Δcorr[k] = -delta*(1-h[i_src+k-(n-1)])*[valid] + delta*h[i_src-k+(n-1)]*[valid]

Similarly for h[i_dst] increasing by delta:
Δcorr[k] += delta*(1-h[i_dst+k-(n-1)])*[valid] - delta*h[i_dst-k+(n-1)]*[valid]

This is O(1) per k, so O(n) total. Much faster than recomputing the full correlate!
