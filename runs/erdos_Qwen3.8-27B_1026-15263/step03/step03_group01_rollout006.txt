import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_smooth_max_grad(h, top_k, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(top_k, len(corr))
    part = np.argpartition(corr, -k)[-k:]
    top_vals = corr[part]
    m = np.max(top_vals)
    shifted = alpha * (top_vals - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    smax = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    grad = np.zeros(n)
    for ii in range(len(part)):
        wi = w[ii]
        offset = part[ii] - n + 1
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - offset:hi2 - offset]
    return smax, grad

def fista_optimize(h0, ts, top_k, n_iter, alpha_schedule, lr_init=0.02):
    h = h0.astype(float).copy()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_iter):
        phase = i * len(alpha_schedule) // n_iter
        alpha = alpha_schedule[min(phase, len(alpha_schedule) - 1)]
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / (t_new + 1.0) * (h - h_prev)
        y = project_to_sum(y, ts)
        t = t_new
        smax, grad = topk_smooth_max_grad(y, top_k, alpha)
        gnorm = np.sum(grad * grad)
        if gnorm < 1e-30:
            break
        for bt in range(40):
            h_trial = project_to_sum(y - lr * grad, ts)
            smax_trial, _ = topk_smooth_max_grad(h_trial, top_k, alpha)
            if smax_trial <= smax - 1e-10 * lr * gnorm:
                break
            lr *= 0.5
            if lr < 1e-16:
                h_trial = y.copy()
                break
        h_new = project_to_sum(y - lr * grad, ts)
        h_prev = h
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if i > 100 and i % 100 == 0:
            lr = min(lr * 1.3, 0.5)
    return best_h, best_c5

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    ts = n_new / 2.0
    h_up = project_to_sum(h_up, ts)
    return h_up

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):

    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n // 2.0)

def stochastic_cd(h, ts, rng, n_steps, delta_lo, delta_hi, temp_init=0.2):
    n = len(h)
    h = h.astype(float).copy()
    c5_cur = compute_c5(h)
    temp = temp_init
    decay = 0.9995
    for step in range(n_steps):
        ones_idx = np.where(h > 0.03)[0]
        zeros_idx = np.where(h < 0.97)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        if h[i] < delta:
            delta = h[i] * 0.4
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.4
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / max(temp, 1e-7))
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-7, temp * decay)
    return h, c5_cur

def slsqp_opt(h0, ts, alpha, maxiter=1000):
    n = len(h0)
    top_k = min(50, 2 * n - 1)

    def obj_func(x):
        return topk_smooth_max_grad(x, top_k, alpha)[0]

    def obj_grad(x):
        return topk_smooth_max_grad(x, top_k, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_func, h0.copy(), method='SLSQP', jac=obj_grad,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    # Phase 2: Upsample and optimize at higher resolutions
    for n_tr in [720, 960, 1200]:
        if time_left() < 120:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        h_up = upsample_h(best_h, best_n, n_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr
        else:
            best_n = n_tr

        alpha_sched_tr = [20.0, 60.0, 200.0, 600.0, 2000.0, 6000.0, 20000.0]
        n_iter_tr = 2500
        h_opt, c5_opt = fista_optimize(h_up, ts_tr, 40, n_iter_tr, alpha_sched_tr, lr_init=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_tr

        if time_left() > 60:
            h_opt2, c5_opt2 = fista_optimize(best_h, ts_tr, 25, 1500,
                                             [1000.0, 5000.0, 15000.0, 50000.0], lr_init=0.008)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n_tr

    # Phase 3: Frequency perturbations and re-optimization
    n_b = best_n
    ts_b = n_b / 2.0
    for trial in range(30):
        if time_left() < 40:
            break
        amp = 0.06 * (1.0 - trial / 40.0) + 0.01
        nf = max(3, n_b // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        alpha_sched_sh = [100.0, 500.0, 2000.0, 8000.0]
        h_ot, c5_ot = fista_optimize(h_sh, ts_b, 30, 800, alpha_sched_sh, lr_init=0.015)
        if c5_ot < best_c5:
            best_c5 = c5_ot
            best_h = h_ot.copy()

    # Phase 4: Diverse restarts
    n_e = best_n
    ts_e = n_e / 2.0
    for trial in range(15):
        if time_left() < 40:
            break
        starts = make_diverse_starts(n_e, ts_e, rng, best_h, best_n, num_starts=10)
        for h_s in starts[:5]:
            if time_left() < 20:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            alpha_sched_s = [50.0, 300.0, 1500.0, 5000.0]
            h_o, c5_o = fista_optimize(h_s, ts_e, 30, 600, alpha_sched_s, lr_init=0.02)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Phase 5: Stochastic coordinate descent
    if time_left() > 120:
        n_cd = best_n
        ts_cd = n_cd / 2.0
        h_cd = best_h.copy()
        c5_cd = compute_c5(h_cd)
        h_cd, c5_cd = stochastic_cd(h_cd, ts_cd, rng, 60000, 0.005, 0.25, temp_init=0.15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() > 50:
            h_pol = slsqp_opt(best_h, ts_cd, 5000, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Second stochastic round with smaller steps
    if time_left() > 100:
        n_cd2 = best_n
        ts_cd2 = n_cd2 / 2.0
        h_cd2 = best_h.copy()
        h_cd2, c5_cd2 = stochastic_cd(h_cd2, ts_cd2, rng, 40000, 0.003, 0.08, temp_init=0.05)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        if time_left() > 40:
            h_pol2 = slsqp_opt(best_h, ts_cd2, 10000, maxiter=2000)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    # Phase 7: FISTA fine-tune with very high alpha
    if time_left() > 80:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f, c5_f = fista_optimize(best_h, ts_f, 15, 2000,
                                   [5000.0, 20000.0, 50000.0, 100000.0, 200000.0], lr_init=0.003)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: SLSQP final polish
    if time_left() > 50:
        n_sp = best_n
        ts_sp = n_sp / 2.0
        for alpha in [5000, 20000, 50000]:
            if time_left() < 25:
                break
            h_sp = slsqp_opt(best_h, ts_sp, alpha, maxiter=1500)
            c5_sp = compute_c5(h_sp)
            if c5_sp < best_c5:
                best_c5 = c5_sp
                best_h = h_sp.copy()

    # Phase 9: Tiny stochastic refinements
    if time_left() > 40:
        n_t = best_n
        ts_t = n_t / 2.0
        h_t, c5_t = stochastic_cd(best_h, ts_t, rng, 30000, 0.002, 0.05, temp_init=0.02)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 10: Final FISTA squeeze
    if time_left() > 30:
        n_fs = best_n
        ts_fs = n_fs / 2.0
        h_fs, c5_fs = fista_optimize(best_h, ts_fs, 10, 1500,
                                     [50000.0, 100000.0, 200000.0, 500000.0], lr_init=0.001)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
