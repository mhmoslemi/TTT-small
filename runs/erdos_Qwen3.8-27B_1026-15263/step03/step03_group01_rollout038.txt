from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def max_lag_gradient(h, corr):
    """Compute gradient of the max correlation element with respect to h."""
    n = len(h)
    m_star = np.argmax(corr)
    lag = m_star - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        k = lag
        # corr = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
        # d/dh[j] = (1-h[j+k]) * 1[j<=n-1-k] + (-h[j-k]) * 1[j>=k]
        forward_mask = np.arange(n) <= (n - 1 - k)
        backward_mask = np.arange(n) >= k
        grad[forward_mask] += (1.0 - h[k:min(n, n-k)])  # h[j+k] for j in forward_mask
        # Actually let me be more careful
        # For j in [0, n-1-k]: contribution is (1 - h[j+k])
        # For j in [k, n-1]: contribution is -h[j-k]
        j_idx = np.arange(n)
        fmask = j_idx <= (n - 1 - k)
        bmask = j_idx >= k
        grad[fmask] += (1.0 - h[j_idx[fmask] + k])
        grad[bmask] += (-h[j_idx[bmask] - k])
    else:
        k = -lag
        # corr = sum_{i=0}^{n-1-k} h[i+k](1-h[i])
        # d/dh[j] = (1-h[j-k]) * 1[j>=k] + (-h[j+k]) * 1[j<=n-1-k]
        j_idx = np.arange(n)

        grad[fmask] += (1.0 - h[j_idx[fmask] - k])
        grad[bmask] += (-h[j_idx[bmask] + k])
    return grad, m_star

def projected_subgradient(h0, n, ts, n_iter=5000, lr_init=0.01):
    """Projected subgradient method on the exact max correlation."""
    h = h0.copy()
    lr = lr_init
    obj_prev = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    stall = 0
    best_h = h.copy()
    best_obj = obj_prev
    
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        obj = float(np.max(corr))
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        
        grad, _ = max_lag_gradient(h, corr)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        obj_new = float(np.max(corr_new))
        
        if obj_new < obj:
            h = h_new
            # Adaptive step size (BB-like)
            if i > 0:
                # Simple: increase lr slightly if making progress
                lr = min(lr * 1.02, 0.1)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 100 or lr < 1e-10:
                break
    
    return best_h, best_obj

1. Computing all corr values (one np.correlate call: O(n²) but fast in C)
2. Computing the weights w_k (softmax)
3. For each lag k with significant weight, add w_k * grad_k to the total gradient

But step 3 is O(n²) in the worst case (all lags have significant weight). However, for large alpha, only a few lags have significant weight, so it's fast.

                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-6, 0.01 * np.exp(-step / 8000.0))

    return best_h, best_c5

def make_diverse_starts(n, ts, rng, best_h, best_n):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(project_to_sum(np.full(n, 0.5), ts))
    for f in [1, 2, 3, 5, 7]:
        for amp in [0.15, 0.3, 0.42]:
            s = 0.5 + amp * np.sin(np.pi * f * x + rng.uniform(0, 2 * np.pi))
            s2 = 0.5 + 0.5 * amp * np.sin(np.pi * 2 * f * x + rng.uniform(0, 2 * np.pi))
            s = 0.5 * s + 0.5 * s2
            starts.append(project_to_sum(s, ts))
    for nb in [2, 3, 4, 6, 8]:
        blk = n // nb
        h_b = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_b[i:min(i + blk, n)] = 1.0
        h_smooth = np.convolve(h_b, np.ones(min(5, blk)) / min(5, blk), mode='same')
        starts.append(project_to_sum(h_smooth, ts))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_o = np.linspace(0, 2, best_n, endpoint=False)
            h_up = np.interp(x, x_o, best_h, period=2.0)
            starts.append(project_to_sum(h_up, ts))
    for _ in range(6):
        h_mix = np.full(n, 0.5)
        nf = rng.randint(2, 7)
        freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
        for f in freqs:
            h_mix += rng.uniform(0.05, 0.3) * np.sin(np.pi * f * x + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(h_mix, ts))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    global alpha_sub
    alpha_sub = 5.0
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    target_n = 960
    if best_n < target_n:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h, period=2.0)
        best_h = project_to_sum(best_h, target_n / 2.0)
        best_n = target_n
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    mats = build_index_mats(n)

    # Phase 1: Continuation in alpha with SLSQP
    alphas = [50, 200, 800, 3000]
    for alpha in alphas:
        if time_left() < 40:
            break
        h_cur = best_h.copy()
        for _ in range(2):
            if time_left() < 25:
                break
            h_opt = slsqp_opt(h_cur, n, ts, alpha, mats, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cur = h_opt

    # Phase 2: Projected subgradient on exact top-k lags
    if time_left() > 120:
        h_sub = best_h.copy()
        n_iters = min(5000, int(time_left() * 8))
        h_sub, _ = projected_subgrad(h_sub, n, ts, n_iter=n_iters, lr_init=0.003, top_k=5)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 3: SLSQP polish after subgradient
    if time_left() > 60:
        for alpha in [3000, 10000]:
            if time_left() < 30:
                break
            h_pol = slsqp_opt(best_h, n, ts, alpha, mats, maxiter=800)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Multi-start with diverse initializations
    if time_left() > 150:
        starts = make_diverse_starts(n, ts, rng, best_h, best_n)
        for h_s in starts:
            if time_left() < 30:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for alpha in [200, 1500]:
                if time_left() < 20:
                    break
                h_o = slsqp_opt(h_s, n, ts, alpha, mats, maxiter=400)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_s = h_o

    # Phase 5: Stochastic coordinate descent targeting max lag
    if time_left() > 100:
        n_steps = min(50000, int(time_left() * 3))
        h_cd, c5_cd = stochastic_cd_exact(best_h, n, ts, rng, n_steps=n_steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Polish after stochastic
    if time_left() > 50:
        for alpha in [5000, 20000]:
            if time_left() < 25:
                break
            h_f = slsqp_opt(best_h, n, ts, alpha, mats, maxiter=1000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 7: Another round of subgradient with different params
    if time_left() > 80:
        alpha_sub = 2.0
        h_sub2 = best_h.copy()
        n_iters = min(4000, int(time_left() * 6))
        h_sub2, _ = projected_subgrad(h_sub2, n, ts, n_iter=n_iters, lr_init=0.002, top_k=10)
        c5_sub2 = compute_c5(h_sub2)
        if c5_sub2 < best_c5:
            best_c5 = c5_sub2
            best_h = h_sub2.copy()

    # Phase 8: Final stochastic refinement
    if time_left() > 50:
        n_steps = min(30000, int(time_left() * 2))
        h_t, c5_t = stochastic_cd_exact(best_h, n, ts, rng, n_steps=n_steps)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 9: Final SLSQP polish
    if time_left() > 30:
        h_fin = slsqp_opt(best_h, n, ts, 20000, mats, maxiter=1500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Try higher resolution if time permits
    if time_left() > 200:
        n2 = 1200
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts2)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n2
        mats2 = build_index_mats(n2)
        for alpha in [200, 1000, 5000]:
            if time_left() < 40:
                break
            h_h = slsqp_opt(h_up, n2, ts2, alpha, mats2, maxiter=500)
            c5_h = compute_c5(h_h)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n2
            h_up = h_h

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_n / 2.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
