- output[n]: lag = 1. h[i+1]*w[i] for i=0..n-2. ✓

So: output[n-1+k] = Σ_{i=0}^{n-1-k} h[i+k] * w[i] for k = 0, ..., n-1

This is: second_term[k] = Σ_{j=0}^{n-1-k} w[j] * h[k+j]

And: np.correlate(h, w[:n], mode='full')[n-1:2*n-1] gives me second_terms[0:n]

Great, so my formula is correct.

Now let me also verify the first_term:
first_term[k] = Σ_{j=0}^{k} w[j] * (1-h[k-j])

This is the standard linear convolution (w * (1-h))[k] = Σ_{j=0}^{k} w[j] * (1-h)[k-j]

So: np.convolve(w[:n], 1-h, mode='full')[:n] gives first_terms[0:n]. ✓

So the gradient is:
grad[k] = first_term[k] - second_term[k] = (w * (1-h))[k] - (w ⊛ h)[k]

Both computed in O(n log n) using FFT.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    one_minus_h = 1.0 - h
    first_terms = np.convolve(w[:n], one_minus_h, mode='full')[:n]
    second_terms = np.correlate(h, w[:n], mode='full')[n-1:]
    grad = first_terms - second_terms
    return sm_val, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        return eval_and_grad(ht, alpha)[0]
    def grad_fn(x):
        ht = np.clip(x, 0.0, 1.0)
        return eval_and_grad(ht, alpha)[1]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_ls(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-6:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-6})
                if res.fun < c5_cur - 1e-12:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def momentum_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    v = np.zeros(n)
    t_val = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    for it in range(n_iters):
        sm, g = eval_and_grad(h, alpha)
        v = 0.9 * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        t_val = 1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val) / 2.0
        beta = (t_val - 2.0) / (t_val - 1.0)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                lr = min(lr * 1.02, 0.5)
        else:
            h = h + beta * v
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_cur = compute_c5(h)
            lr *= 0.8
    return h_best, c5_best

    x_new = np.linspace(0, 2, n_new, endpoint=False)

    return h_interp
