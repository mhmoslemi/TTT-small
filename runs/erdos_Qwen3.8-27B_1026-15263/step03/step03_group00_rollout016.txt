def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):

def corr_gradient_single(h, lag):

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):

             'jac': lambda x: np.ones(n)}]

        return smooth_max_and_grad(x, alpha)[0]

        return smooth_max_and_grad(x, alpha)[1]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    raw_w = raw_w * 10  # Sharpen
    raw_w -= np.max(raw_w)
    w = np.exp(raw_w)
    w /= np.sum(w)
    g = np.zeros(n)
    for i in range(K):
        lag = top_k[i] - (n - 1)
        g += w[i] * corr_gradient_single(h, lag)
    return g, m_val

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def gradient_step_anneal(h0, target_sum, rng, max_iters, T_init, cooling, patience=300):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    stale = 0
    
    for it in range(max_iters):
        g, m_val = active_gradients(h, K=5)
        # Gradient-guided perturbation
        noise_scale = 0.01 + 0.03 * (T / T_init)
        grad_scale = 0.005 * (1.0 + T / T_init)
        h_new = h - grad_scale * g + rng.randn(n) * noise_scale
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        
        if delta < 0:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stale = 0
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            stale += 1
        else:
            stale += 1
        
        T = max(1e-8, T * cooling)
        if stale > patience:
            # Restart: perturb and increase temperature
            h = project_to_sum(h + rng.randn(n) * 0.05, target_sum)
            T = T_init * 0.5
            stale = 0
    
    return best_h, best_c5

def pair_swap_anneal(h0, target_sum, rng, max_iters, T_init, cooling, patience=500):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    stale = 0
    
    for it in range(max_iters):
        # Pair swap: move mass from i to j
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amt = rng.uniform(0.01, 0.1) * min(h[i], 1.0 - h[j])
        if amt < 1e-8:
            continue
        # Try both directions
        h_new = h.copy()
        if rng.random() < 0.5:
            h_new[i] -= amt
            h_new[j] += amt
        else:
            h_new[i] += amt
            h_new[j] -= amt
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        
        if delta < 0:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stale = 0
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            stale += 1
        else:
            stale += 1
        
        T = max(1e-8, T * cooling)
        if stale > patience:
            h = project_to_sum(h + rng.randn(n) * 0.03, target_sum)
            T = T_init * 0.3
            stale = 0
    
    return best_h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:

        if k == 0:

        else:
            g[:n - k] += (1.0 - h[k:])
