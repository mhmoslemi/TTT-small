    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = c_max + np.log(s) / alpha
    
    g = np.zeros(n)
    for j in range(len(top_idx)):
        wl = w[j]
        if wl < 1e-14:
            continue
        ci = int(top_idx[j])
        l = ci - (n - 1)
        lo1 = max(0, l)
        hi1 = min(n, n + l)
        if lo1 < hi1:
            g[lo1:hi1] += wl * (1.0 - h[lo1 - l:hi1 - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def slsqp_smooth(h0, n, target_sum, alpha, top_k, maxiter):
    def obj(x):
        v, _ = smooth_max_focal(np.clip(x, 0.0, 1.0), alpha, top_k)
        return v
    def grad(x):
        _, g = smooth_max_focal(np.clip(x, 0.0, 1.0), alpha, top_k)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, target_sum)
    except Exception:
        return project_box_sum(h0, target_sum)

def proj_grad_bb(h0, n, target_sum, alpha, top_k, n_iters):
    h = h0.copy()
    _, g_prev = smooth_max_focal(h, alpha, top_k)
    c5_prev = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_prev
    lr = 0.1
    s_prev = None
    y_prev = None
    
    for it in range(n_iters):
        _, g = smooth_max_focal(h, alpha, top_k)
        
        # Barzilai-Borwein step
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            ss = np.dot(s_prev, s_prev)
            if ss > 1e-14 and abs(sy) > 1e-14:
                lr_bb = ss / abs(sy)
                lr = min(max(lr_bb, 1e-4), 5.0)
        
        h_new = project_box_sum(h - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new < c5_prev:
            s_prev = h_new - h
            # Recompute gradient at new point for y
            _, g_new = smooth_max_focal(h_new, alpha, top_k)
            y_prev = g_new - g
            h = h_new
            c5_prev = c5_new
            g = g_new
        else:
            lr *= 0.7
            h = h_new  # still move to new point for BB
            s_prev = h_new - h
            _, g_new = smooth_max_focal(h_new, alpha, top_k)
            y_prev = g_new - g
            c5_prev = c5_new
            g = g_new
    
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            for d in steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_box_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_val = h_new[i]
                    h = h_new
            if best_c5 < c5_cur - 1e-15:
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def coord_descent(h0, n, target_sum, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            for d in steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_box_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    h = h_new
            if best_c5 < c5_cur - 1e-15:
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def interp_resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
