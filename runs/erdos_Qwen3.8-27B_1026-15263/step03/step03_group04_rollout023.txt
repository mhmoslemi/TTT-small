= ∫₀² [1/4 - ε/2 f(x+k) + ε/2 f(x) - ε² f(x)f(x+k)] dx
= 1/2 - ε/2 ∫f(x+k)dx + ε/2 ∫f(x)dx - ε² ∫f(x)f(x+k)dx

Now, ∫₀² f(x)dx = 0. But ∫₀² f(x+k)dx is NOT necessarily 0 unless f is periodic with period dividing k or we're doing a circular integral.

In the problem, h is defined on [0,2] and the correlation uses the "full" mode, which means for shift k, we only integrate over the overlapping region [0, 2-k] (for k>0) or [k, 2] (for k<0... actually in the discrete case with "full" mode, it's different).

Let me stick with the discrete formulation to avoid confusion.

In the discrete case:
corr[k] = ∑_{i=0}^{n-k'-1} h[i] * (1-h[i+k']) for k' > 0 (right shift)
corr[k] = ∑_{i=k'}^{n-1} h[i] * (1-h[i-k']) for k' < 0 (left shift)

where the exact indexing depends on how np.correlate works.

Given the complexity, let me implement a hybrid approach:
1. **Cutting-plane LP** for the main optimization (this is the novel part)
2. **Gradient polishing** for final refinement

The cutting-plane approach:
- We want to minimize t = max_k corr_k(h) * dx
- Bisection on t: for a given t, check feasibility
- Feasibility check: solve LP with cutting planes

For the LP:
Variables: h[0], ..., h[n-1]
Constraints:
  - 0 ≤ h[i] ≤ 1
  - ∑h[i] = n/2
  - For each active shift k: ∇_h(corr_k) · h ≤ rhs_k (linearized constraint)

The linearization at point h₀ for shift k:
corr_k(h) ≈ corr_k(h₀) + ∇_h(corr_k)|_{h₀} · (h - h₀)
We want corr_k(h) ≤ t/dx = t*n/2
So: ∇_h(corr_k)|_{h₀} · h ≤ t*n/2 - corr_k(h₀) + ∇_h(corr_k)|_{h₀} · h₀

The gradient ∇_h(corr_k)|_{h₀}:
corr_k = ∑_i h[i](1-h[i+k]) (for the relevant range of i)
∂corr_k/∂h[j] = (1-h₀[j+k]) for the first sum (when j appears as h[i])

So ∂corr_k/∂h[j] = (1-h₀[j+k]) - h₀[j-k] (with appropriate 0 for out-of-bounds)

The method should work like this:
1. Start with no cuts (or a few initial ones)
2. Solve LP → get h_new
3. Check actual C₅ at h_new
4. If C₅ ≤ t_target → success
5. Otherwise, add cuts for violated constraints
6. Go to step 2

The key question is: will this converge? For non-convex problems, it might cycle or fail to converge. But in practice, for this type of problem, it often works well because the feasible region is "almost" convex (the quadratic constraints are only mildly non-convex).

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_grad_shift(h, shift):
    n = len(h)
    grad = np.ones(n)
    if shift > 0:
        if shift < n:
            grad[:n-shift] -= h[shift:]
        if shift > 0:
            grad[shift:] -= h[:n-shift]
    elif shift < 0:
        s = -shift
        if s < n:
            grad[:n-s] -= h[s:]
            grad[s:] -= h[:n-s]
    else:
        grad -= 2.0 * h
    return grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        grad_k = compute_grad_shift(h, shift)
        grad += wi * grad_k
    return obj, grad

def optimize_gradient(h0, n, target_sum, alphas, maxiter_per_alpha, time_limit):
    h = h0.copy()
    t_start = time.time()
    for a in alphas:
        if time.time() - t_start > time_limit:
            break
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, a)
            s_dev = np.sum(x) - target_sum
            pen = 1000.0
            return obj + 0.5 * pen * s_dev * s_dev
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, a)
            s_dev = np.sum(x) - target_sum
            return grad + 1000.0 * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15, 'gtol': 1e-8})
            h = project_to_sum(res.x, target_sum)
        except Exception:
            pass
    return h

def cutting_plane_optimize(h_init, n, target_sum, t_target, max_iterations=1000, max_cuts=2000, time_limit=None, t_start=None):
    """Minimize C5 by iteratively adding cutting planes."""
    if t_start is None:
        t_start = time.time()
    dx = 2.0 / n
    t_corr = t_target / dx
    h = h_init.copy()
    cuts_A = []  # list of gradient vectors
    cuts_b = []  # list of rhs values
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iterations):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        # Solve LP
        try:
            if cuts_A:
                A_ub = np.array(cuts_A)
                b_ub = np.array(cuts_b)
                res = linprog(c=np.zeros(n), A_ub=A_ub, b_ub=b_ub,
                             A_eq=np.ones((1, n)), b_eq=np.array([target_sum]),
                             bounds=[(0.0, 1.0)] * n, method='highs')
            else:
                res = linprog(c=np.zeros(n),
                             A_eq=np.ones((1, n)), b_eq=np.array([target_sum]),
                             bounds=[(0.0, 1.0)] * n, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            # LP infeasible - relax t_target
            break
        
        h_new = res.x
        corr = compute_corr(h_new)
        c5_val = float(np.max(corr) * dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        if c5_val <= t_corr + 1e-10:

def exact_max_polish(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(5, len(corr))]
        best_improve = 0
        best_h = h
        for k_max in top_k:

                h_new = project_to_sum(h_new, target_sum)
```

**Paired Coordinate Descent:**
- Pick a pair (j, j')
- Set h[j] = h₀ + t, h[j'] = h₀' - t (preserving the sum)
- Minimize C₅ as a function of t over t ∈ [h₀', h₀] (to keep both in [0,1])

For a fixed pair (j, j'):

So C₅(t) = max_k (f_k(h) + t * g_k) where g_k = ∂f_k/∂h[j] - ∂f_k/∂h[j']

This is a 1D convex piecewise-linear function. We can find its minimum efficiently.

The minimum of max_k (c_k + g_k * t) over t ∈ [t_min, t_max]:
- If all g_k ≥ 0: minimum at t_min
- If all g_k ≤ 0: minimum at t_max
- Otherwise: the minimum is at a point where two of the lines "c_k + g_k * t" cross and form the upper envelope minimum

Finding the exact minimum of a 1D max-of-lines problem:
- Sort the lines by slope
- The minimum of the upper envelope occurs either at a boundary or at a crossing point
- We can binary search on the value: for a candidate value v, check if c_k + g_k * t ≤ v for all k → this gives a feasible interval for t

Actually, there's a simpler approach: the minimum of max_k (c_k + g_k * t) is the smallest v such that the system c_k + g_k * t ≤ v has a solution for t ∈ [t_min, t_max].

For each k: c_k + g_k * t ≤ v
→ if g_k > 0: t ≤ (v - c_k) / g_k
→ if g_k < 0: t ≥ (v - c_k) / g_k
→ if g_k = 0: need c_k ≤ v

So the feasible interval for t is:
[max(t_min, max_{g_k<0} (v-c_k)/g_k)] ≤ t ≤ [min(t_max, min_{g_k>0} (v-c_k)/g_k)]

And we also need c_k ≤ v for all g_k = 0.

Binary search on v to find the minimum!

This is efficient: O(1) per evaluation (just compute max of lower bounds and min of upper bounds), and we need O(log(precision)) evaluations.

```python
def min_max_lines(c, g, t_min, t_max):
    """Find min of max_k (c[k] + g[k]*t) over t in [t_min, t_max].
    Returns (t_opt, val_opt)."""
    # Binary search on v
    # v_min = max(c) at t=0... actually v_min = min at the optimal t
    # Lower bound for v: min_t max_k(c_k + g_k t) >= max_k c_k min(1, ...) 
    # Let's just use v_low = max(min(c + g*t_min, c + g*t_max)) and v_high = max(max(c + g*t_min, c + g*t_max))
    
    v_low = np.max(np.minimum(c + g * t_min, c + g * t_max))
    v_high = np.max(np.maximum(c + g * t_min, c + g * t_max))
    
    # Actually, v_low should be the minimum possible value
    # And v_high should be the maximum possible value
    # v_low = min over t of max_k(c_k + g_k t) -- this is what we want to find
    # v_high = max over t of max_k(c_k + g_k t) = max(max_k(c_k + g_k t_min), max_k(c_k + g_k t_max))
    
    # For binary search, we need:
    # v_low: some value below the answer (or the answer itself)
    # v_high: some value above the answer
    
    # v_high = max(c + g*t_min, c+g*t_max) element-wise, then max
    vals_at_bounds = np.maximum(c + g * t_min, c + g * t_max)
    v_high = np.max(vals_at_bounds)
    
    # v_low: use the value at t=0 clamped to [t_min, t_max]
    t_0 = np.clip(0.0, t_min, t_max)
    v_low = np.max(c + g * t_0)
    # But this might be above the true minimum. Let's use a lower bound:
    # v >= max_k min(c_k + g_k*t_min, c_k + g_k*t_max)
    # For each k, the minimum of c_k + g_k*t over t in [t_min, t_max] is:
    #   if g_k > 0: c_k + g_k*t_min
    #   if g_k < 0: c_k + g_k*t_max
    #   if g_k = 0: c_k
    min_per_k = np.where(g > 0, c + g*t_min, np.where(g < 0, c + g*t_max, c))
    v_low = np.max(min_per_k)  # This is a valid lower bound
    
    # Binary search
    for _ in range(200):
        v_mid = (v_low + v_high) / 2
        # Check feasibility: exists t in [t_min, t_max] such that c_k + g_k*t <= v_mid for all k
        # For g_k > 0: t <= (v_mid - c_k) / g_k
        # For g_k < 0: t >= (v_mid - c_k) / g_k
        # For g_k = 0: need c_k <= v_mid
        
        pos = g > 1e-15
        neg = g < -1e-15
        zero = ~(pos | neg)
        
        if np.any(zero) and np.any(c[zero] > v_mid):
            v_low = v_mid
            continue
        
        t_ub = t_max
        if np.any(pos):
            t_ub = min(t_ub, np.min((v_mid - c[pos]) / g[pos]))
        
        t_lb = t_min
        if np.any(neg):
            t_lb = max(t_lb, np.max((v_mid - c[neg]) / g[neg]))
        
        if t_lb <= t_ub + 1e-15:
            # Feasible: v_mid is achievable
            v_high = v_mid
        else:
            v_low = v_mid
    
    v_opt = v_high
    # Find optimal t
    pos = g > 1e-15
    neg = g < -1e-15
    zero = ~(pos | neg)
    
    t_ub = t_max
    if np.any(pos):
        t_ub = min(t_ub, np.min((v_opt - c[pos]) / g[pos]))
    t_lb = t_min
    if np.any(neg):
        t_lb = max(t_lb, np.max((v_opt - c[neg]) / g[neg]))
    t_opt = (t_lb + t_ub) / 2
    t_opt = np.clip(t_opt, t_min, t_max)
    
    return t_opt, v_opt
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_grad_shift(h, shift):
    n = len(h)
    grad = np.ones(n)
    if shift > 0 and shift < n:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    elif shift < 0 and -shift < n:
        s = -shift
        grad[:n - s] -= h[s:]
        grad[s:] -= h[:n - s]
    elif shift == 0:
        grad -= 2.0 * h
    return grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def min_max_lines(c, g, t_min, t_max):
    min_per_k = np.where(g > 0, c + g * t_min, np.where(g < 0, c + g * t_max, c.copy()))
    v_low = float(np.max(min_per_k))
    vals_at_bounds = np.maximum(c + g * t_min, c + g * t_max)
    v_high = float(np.max(vals_at_bounds))
    if v_high - v_low < 1e-15:
        return (t_min + t_max) * 0.5, v_high
    pos_mask = g > 1e-15
    neg_mask = g < -1e-15
    zero_mask = ~(pos_mask | neg_mask)
    for _ in range(80):
        v_mid = (v_low + v_high) * 0.5
        if np.any(zero_mask) and np.any(c[zero_mask] > v_mid):
            v_low = v_mid
            continue
        t_ub = t_max
        if np.any(pos_mask):
            t_ub = min(t_ub, float(np.min((v_mid - c[pos_mask]) / g[pos_mask])))
        t_lb = t_min
        if np.any(neg_mask):
            t_lb = max(t_lb, float(np.max((v_mid - c[neg_mask]) / g[neg_mask])))
        if t_lb <= t_ub + 1e-14:
            v_high = v_mid
        else:
            v_low = v_mid
    v_opt = v_high
    t_ub = t_max
    if np.any(pos_mask):
        t_ub = min(t_ub, float(np.min((v_opt - c[pos_mask]) / g[pos_mask])))
    t_lb = t_min
    if np.any(neg_mask):
        t_lb = max(t_lb, float(np.max((v_opt - c[neg_mask]) / g[neg_mask])))
    t_opt = (t_lb + t_ub) * 0.5
    t_opt = float(np.clip(t_opt, t_min, t_max))
    return t_opt, v_opt

def paired_coord_descent(h, n, target_sum, max_rounds=80, n_pairs=200, seed=0, time_limit_s=None, t_start=None):
    if t_start is None:
        t_start = time.time()
    dx = 2.0 / n
    h = h.copy()
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(seed)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for round_ in range(max_rounds):
        if time_limit_s is not None and time.time() - t_start > time_limit_s:
            break
        corr = compute_corr(h)
        best_improve = 0.0
        best_h_new = None
        best_c5_new = c5_cur
        for _ in range(n_pairs):
            j = rng.randint(n)
            jp = rng.randint(n)
            if j == jp:
                continue
            gj = np.ones(2 * n - 1)
            mj = (j + shifts >= 0) & (j + shifts < n)
            if np.any(mj):
                gj[mj] -= h[j + shifts[mj]]
            mj2 = (j - shifts >= 0) & (j - shifts < n)
            if np.any(mj2):
                gj[mj2] -= h[j - shifts[mj2]]
            gjp = np.ones(2 * n - 1)
            mj = (jp + shifts >= 0) & (jp + shifts < n)
            if np.any(mj):
                gjp[mj] -= h[jp + shifts[mj]]
            mj2 = (jp - shifts >= 0) & (jp - shifts < n)
            if np.any(mj2):
                gjp[mj2] -= h[jp - shifts[mj2]]
            g = gj - gjp
            c = corr
            t_min = max(-h[j], h[jp] - 1.0)
            t_max = min(1.0 - h[j], h[jp])
            if t_max <= t_min + 1e-14:
                continue
            t_opt, v_opt = min_max_lines(c, g, t_min, t_max)
            c5_cand = v_opt * dx
            if c5_cand < c5_cur - best_improve - 1e-15:
                h_new = h.copy()
                h_new[j] += t_opt
                h_new[jp] -= t_opt
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-15:
                    best_improve = c5_cur - c5_new
                    best_h_new = h_new
                    best_c5_new = c5_new
        if best_h_new is not None:
            h = best_h_new
            c5_cur = best_c5_new
        else:
            break
    return h, c5_cur

def structured_coord_descent(h, n, target_sum, max_rounds=30, time_limit_s=None, t_start=None):
    if t_start is None:
        t_start = time.time()
    dx = 2.0 / n
    h = h.copy()
    c5_cur = compute_c5(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    offsets = list(range(1, min(n, 30)))
    for round_ in range(max_rounds):
        if time_limit_s is not None and time.time() - t_start > time_limit_s:
            break
        corr = compute_corr(h)
        best_improve = 0.0
        best_h_new = None
        best_c5_new = c5_cur
        for offset in offsets:
            for j in range(0, n, 1):
                jp = (j + offset) % n
                if j == jp:
                    continue
                gj = np.ones(2 * n - 1)
                mj = (j + shifts >= 0) & (j + shifts < n)
                if np.any(mj):
                    gj[mj] -= h[j + shifts[mj]]
                mj2 = (j - shifts >= 0) & (j - shifts < n)
                if np.any(mj2):
                    gj[mj2] -= h[j - shifts[mj2]]
                gjp = np.ones(2 * n - 1)
                mj = (jp + shifts >= 0) & (jp + shifts < n)
                if np.any(mj):
                    gjp[mj] -= h[jp + shifts[mj]]
                mj2 = (jp - shifts >= 0) & (jp - shifts < n)
                if np.any(mj2):
                    gjp[mj2] -= h[jp - shifts[mj2]]
                g = gj - gjp
                c = corr
                t_min = max(-h[j], h[jp] - 1.0)
                t_max = min(1.0 - h[j], h[jp])
                if t_max <= t_min + 1e-14:
                    continue
                t_opt, v_opt = min_max_lines(c, g, t_min, t_max)
                c5_cand = v_opt * dx
                if c5_cand < c5_cur - best_improve - 1e-15:
                    h_new = h.copy()
                    h_new[j] += t_opt
                    h_new[jp] -= t_opt
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - best_improve - 1e-15:
                        best_improve = c5_cur - c5_new
                        best_h_new = h_new
                        best_c5_new = c5_new
            offsets = [o for o in offsets if o > 1] + [o * 2 for o in offsets if o * 2 < n]
        if best_h_new is not None:
            h = best_h_new
            c5_cur = best_c5_new
        else:
            break
    return h, c5_cur

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        grad += wi * compute_grad_shift(h, shift)
    return obj, grad

def optimize_gradient(h0, n, target_sum, alphas, maxiter_per_alpha, time_limit, t_start):
    h = h0.copy()
    for a in alphas:
        if time.time() - t_start > time_limit:
            break
        pen = 5000.0
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, a)
            s_dev = np.sum(x) - target_sum
            return obj + 0.5 * pen * s_dev * s_dev
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, a)
            s_dev = np.sum(x) - target_sum
            return grad + pen * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 50})
            h = project_to_sum(res.x, target_sum)
        except Exception:
            pass
    return h

def cutting_plane_optimize(h_init, n, target_sum, t_target, max_iterations=800, max_cuts=1500, time_limit_s=None, t_start=None):
    if t_start is None:
        t_start = time.time()
    dx = 2.0 / n
    t_corr = t_target / dx
    h = h_init.copy()
    cuts_A = []
    cuts_b = []
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iterations):
        if time_limit_s is not None and time.time() - t_start > time_limit_s:
            break
        try:
            if cuts_A:
                A_ub = np.array(cuts_A)
                b_ub = np.array(cuts_b)
                res = linprog(c=np.zeros(n), A_ub=A_ub, b_ub=b_ub,
                             A_eq=np.ones((1, n)), b_eq=np.array([target_sum]),
                             bounds=[(0.0, 1.0)] * n, method='highs',
                             options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
            else:
                res = linprog(c=np.zeros(n),
                             A_eq=np.ones((1, n)), b_eq=np.array([target_sum]),
                             bounds=[(0.0, 1.0)] * n, method='highs')
        except Exception:
            break
        if res is None or res.status != 0:
            break
        h_new = res.x
        corr = compute_corr(h_new)
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        if c5_val <= t_target + 1e-12:
            return best_h, best_c5
        violations = corr - t_corr
        n_violated = int(np.sum(violations > 1e-8))
        if n_violated == 0:
            break
        n_add = min(30, n_violated)
        top_idx = np.argsort(violations)[::-1][:n_add]
        for k_idx in top_idx:
            if violations[k_idx] <= 1e-10:
                continue
            shift = k_idx - (n - 1)
            grad = compute_grad_shift(h_new, shift)
            rhs = t_corr - corr[k_idx] + grad.dot(h_new) + 1e-12
            cuts_A.append(grad)
            cuts_b.append(rhs)
        if len(cuts_A) > max_cuts:
            viol_matrix = np.array([np.dot(cuts_A[i], h_new) - cuts_b[i] for i in range(len(cuts_A))])
            keep_mask = viol_matrix > -0.01 * t_corr
            keep_idx = np.where(keep_mask)[0]
            if len(keep_idx) > max_cuts:
                keep_idx = keep_idx[np.argsort(viol_matrix[keep_idx])[::-1][:max_cuts]]
            cuts_A = [cuts_A[i] for i in keep_idx]
            cuts_b = [cuts_b[i] for i in keep_idx]
        h = h_new
    return best_h, best_c5

def exact_max_polish(h, n, target_sum, max_rounds=300):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(8, len(corr))]
        best_improve = 0.0
        best_h = h
        best_c5_new = c5_cur
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = compute_grad_shift(h, shift)
            for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-15:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is h:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur
```
