  sum_i h[i] - sum_i h[i]*h[i+k]

  f(h) = sum_i h[i] - sum_i h[i]*h[i+k]

  - h[i] for 0<i<n-k appears in: h[i-k]*h[i] and h[i]*h[i+k] → contributes -(h[i-k]+h[i+k]) to gradient[i]
  - h[n-k] to h[n-1] appear in: h[i-k]*h[i] → contributes -h[i-k] to gradient[i]

  min f(h) = sum h[i] - sum_i h[i]*h[i+k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_k(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    g[k:] += (1.0 - h[:n - k])
    g[:n - k] -= h[k:]
    return g

def topk_gradient_descent(h, n, target_sum, t_end, K_list=(3, 5, 8, 12), base_step=0.005):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = base_step
    stall = 0
    stall_limit = 2000
    
    while time.time() < t_end:
        corr = compute_correlations(h)
        c5_curr = float(np.max(corr))
        
        if c5_curr < best_c5 - 1e-12:
            best_c5 = c5_curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_limit:
            step *= 0.8
            stall = 0
            if step < 0.0001:
                break
        
        n_corr = len(corr)
        center = n - 1
        lags = np.arange(n_corr)
        abs_lag = np.abs(lags - center)
        
        for K in K_list:
            if time.time() > t_end:
                break
            top_idx = np.argsort(corr)[-K:]
            top_idx = [t for t in top_idx if abs_lag[t] > 0]
            if not top_idx:

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -400.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=2000):
    def f(hv):
        return smooth_max_and_grad(hv, alpha)[0]
    def g(hv):
        return smooth_max_and_grad(hv, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0

        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

        if delta < -1e-12 or (delta > -1e-12 and delta / T + np.log(rng.uniform()) < 0):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            if move_type < 0.85:
                h[i2], h[j2] = h[j2], h[i2]
            ones_tmp = list(np.where(h == 1.0)[0])
            zeros_tmp = list(np.where(h == 0.0)[0])
            if move_type >= 0.6:
                ones = ones_tmp[:target_sum] if len(ones_tmp) <= target_sum else ones_tmp
                zeros = zeros_tmp[:n - target_sum] if len(zeros_tmp) <= n - target_sum else zeros_tmp
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.015)
            no_improve = 0
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
        
        T = max(T * 0.999999, min_T)
    
    return best_h, best_c5

def block_move_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = float(np.max(compute_correlations(h)))
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.005
    
    while time.time() < t_end:
        block_len = rng.randint(2, 6)
        start = rng.randint(0, n - block_len + 1)
        block = h[start:start + block_len].copy()
        
        if block.sum() == 0 or block.sum() == block_len:
            continue
        
        new_start = rng.randint(0, n - block_len + 1)
        if abs(start - new_start) < block_len:
            continue
        
        h_old = h[new_start:new_start + block_len].copy()
        h[start:start + block_len] = h_old
        h[new_start:new_start + block_len] = block
        
        c5_new = float(np.max(compute_correlations(h)))
        delta = c5_new - c5_curr
        
        if delta < -1e-12 or (delta > -1e-12 and delta / T + np.log(rng.uniform()) < 0):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[start:start + block_len] = block
            h[new_start:new_start + block_len] = h_old
        
        T = max(T * 0.99999, 0.0005)
    
    return best_h, best_c5

def continuous_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.002
    n_perturb = max(3, n // 40)
    
    while time.time() < t_end:
        idx = rng.choice(n, size=n_perturb, replace=False)
        noise_scale = 0.1 * (T / 0.002)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-noise_scale, noise_scale, n_perturb)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_curr
        if delta < -1e-12 or (delta / T + np.log(rng.uniform())) < 0:
            h = h_new
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        T = max(T * 0.999995, 0.0003)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return budget_s - elapsed()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n_main = best_n
    ts_main = n_main / 2.0
    h_work = best_h.copy()
    
    # Phase 1: Top-K gradient descent (main new contribution)
    t_phase1_end = t_start + 0.30 * budget_s
    h_pd, c5_pd = topk_gradient_descent(h_work, n_main, ts_main, t_phase1_end, 
                                         K_list=(2, 4, 6), base_step=0.008)
    if c5_pd < best_c5:
        best_c5 = c5_pd
        best_h = h_pd.copy()
    h_work = best_h.copy()
    
    # Phase 2: SLSQP polishing with adaptive alpha
    if remaining() > budget_s * 0.15:
        alphas = [30, 100, 400, 2000, 10000]
        for alpha in alphas:
            if remaining() < budget_s * 0.12:
                break
            h_opt = slsqp_optimize(h_work, n_main, ts_main, alpha, maxiter=3000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_work = h_opt.copy()
    
    # Phase 3: More top-K gradient descent from improved point
    if remaining() > budget_s * 0.20:
        t_phase3_end = t_start + 0.65 * budget_s
        h_pd2, c5_pd2 = topk_gradient_descent(h_work, n_main, ts_main, t_phase3_end,
                                               K_list=(2, 3, 5, 10), base_step=0.004)
        if c5_pd2 < best_c5:
            best_c5 = c5_pd2
            best_h = h_pd2.copy()
            h_work = h_pd2.copy()
    
    # Phase 4: Binary SA from current best
    if remaining() > budget_s * 0.25:
        h_bin = make_binary(h_work, n_main, int(ts_main), rng)
        t_sa_end = t_start + 0.78 * budget_s
        h_sa, c5_sa = binary_sa_improved(h_bin, n_main, int(ts_main), t_sa_end, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 5: Block move SA
    if remaining() > budget_s * 0.18:
        h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
        t_block_end = t_start + 0.85 * budget_s
        h_bl, c5_bl = block_move_sa(h_bin2, n_main, int(ts_main), t_block_end, rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
    
    # Phase 6: Continuous SA for fine-tuning
    if remaining() > budget_s * 0.12:
        t_cont_end = t_start + 0.90 * budget_s
        h_cs, c5_cs = continuous_sa(best_h, n_main, ts_main, t_cont_end, rng)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()
    
    # Phase 7: Final SLSQP polish
    if remaining() > 30:
        for alpha in [5000, 20000, 50000]:
            if remaining() < 20:
                break
            h_fin = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=5000)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
    
    # Phase 8: Try different n values
    if remaining() > 100:
        for n_try in [200, 280, 320, 360]:
            if remaining() < 80:
                break
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            h_interp = interpolated_h(best_h, best_n, n_try)
            h_interp = project_to_sum(h_interp, ts_try)
            
            h_try = h_interp
            for alpha in [50, 500, 5000]:
                if remaining() < 40:
                    break
                h_try = slsqp_optimize(h_try, n_try, ts_try, alpha, maxiter=1500)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
            
            if remaining() > 60:
                h_bin_try = make_binary(best_h, best_n, int(ts_try), rng)
                h_sa_try, c5_sa_try = binary_sa_improved(h_bin_try, n_try, int(ts_try),
                                                          t_start + remaining() - 20, rng)
                if c5_sa_try < best_c5:
                    best_c5 = c5_sa_try
                    best_h = h_sa_try.copy()
                    best_n = n_try
    
    # Final projection and verification
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
