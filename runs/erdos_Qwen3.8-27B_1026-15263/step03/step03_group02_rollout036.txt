    cs_idx = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - x[fc]) * w[None, :]
    cs = valid_s * x[cs_idx] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_max_lag_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    s = k_star - n + 1
    grad = np.zeros(n)
    if s == 0:
        grad[:] = h - (1.0 - h)
    elif s > 0:
        valid = n - s
        grad[:valid] += (1.0 - h[s:valid + s])
        grad[s:] -= h[:valid]
    else:
        s_abs = -s
        valid = n - s_abs
        grad[s_abs:] += (1.0 - h[:valid])
        grad[:valid] -= h[s_abs:s_abs + valid]
    return k_star, grad, corr

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_bounds(n):
    return [(0.0, 1.0)] * n

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng, t_init=0.015, t_min=0.002):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = t_init
    no_improve = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / T) + np.log(rng.uniform()) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, 0.025)
            no_improve = 0
        T = max(T * 0.999995, t_min)
    return best_h, best_c5

def slsqp_run(h0, n, target_sum, alpha, maxiter=600):
    global _g_alpha, _g_target_sum, _g_n
    _g_alpha = alpha
    _g_target_sum = target_sum
    _g_n = n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    bounds = make_bounds(n)
    try:
        res = minimize(_smax_obj, h0.copy(), method='SLSQP', jac=_smax_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_method(h, n, target_sum, t_end, rng, step_scale=0.00
1):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_scale
    for it in range(2000000):
        if time.time() > t_end:
            break
        k_star, grad, corr = get_max_lag_grad(h)
        h_new = h - step * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.01)
        elif c5_new < best_c5:
            h = h_new
            step = step * 0.95
        else:
            step = step * 0.9
            if step < 1e-6:
                h = best_h.copy()
                step = 0.0005
    return best_h, best_c5

def coord_descent(h, n, target_sum, n_sweeps=3, n_grid=200):
    h = h.copy()
    ts = target_sum
    for sweep in range(n_sweeps):
        order = np.random.RandomState(42 + sweep).permutation(n)
        for j in order:
            A = np.zeros(2 * n - 1)
            D = np.zeros(2 * n - 1)
            corr_curr = np.correlate(h, 1.0 - h, mode="full")
            for m in range(2 * n - 1):
                s_val = m - n + 1
                coeff1 = 0.0
                coeff2 = 0.0
                if 0 <= j + s_val < n:
                    coeff1 = (1.0 - h[j + s_val])
                if 0 <= j - s_val < n:
                    coeff2 = -h[j - s_val]
                D[m] = coeff1 + coeff2
            curr_contrib = h[j] * D
            A = corr_curr - curr_contrib
            t_vals = np.linspace(0.0, 1.0, n_grid)
            obj_vals = np.max(A[None, :] + t_vals[:, None] * D[None, :], axis=1)
            best_t = t_vals[np.argmin(obj_vals)]
            h[j] = best_t
        h = project_to_sum(h, ts)
    return h

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_ext = np.concatenate([h_src[-1:]], h_src, h_src[:1])
    x_ext = np.concatenate([np.array([x_src[0] - 2.0/n_src]), x_src, np.array([x_src[-1] + 2.0/n_src])])
    return np.interp(x_dst, x_ext, h_ext)

def shake_and_opt(h, n, ts, noise_mag, alpha, maxiter, t_end):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    h_shake = h + rng.uniform(-noise_mag, noise_mag, n)
    h_shake = project_to_sum(h_shake, ts)
    h_opt = slsqp_run(h_shake, n, ts, alpha, maxiter=maxiter)
    c5_opt = compute_c5(h_opt)
    if time.time() > t_end:
        return h, compute_c5(h)
    return h_opt, c5_opt

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.18 * budget_s
    if remaining() > 100:
        h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), sa_end1, rng, t_init=0.018, t_min=0.003)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)

    for alpha in [30, 100, 400, 1500, 5000]:
        if elapsed() > 0.45 * budget_s:
            break
        h_pol = slsqp_run(best_h, n_main, ts_main, alpha, maxiter=1000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    n_shakes = 25
    shake_start = elapsed()
    shake_budget = 0.25 * budget_s
    for trial in range(n_shakes):
        if elapsed() > 0.72 * budget_s:
            break
        noise = 0.12 * (1.0 - trial / float(n_shakes)) + 0.03
        h_shake = best_h + rng.uniform(-noise, noise, n_main)
        h_shake = project_to_sum(h_shake, ts_main)
        alpha_try = [50, 200, 800, 3000, 10000][min(trial % 5, 4)]
        h_opt = slsqp_run(h_shake, n_main, ts_main, alpha_try, maxiter=800)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if remaining() > 80:
        h_cd = coord_descent(best_h, n_main, ts_main, n_sweeps=5, n_grid=300)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining() > 60:
        h_sub = subgradient_method(best_h, n_main, ts_main, t_start + 0.85 * budget_s, rng)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    if remaining() > 50:
        for alpha_fin in [8000, 20000, 50000]:
            if remaining() < 20:
                break
            h_fin = slsqp_run(best_h, n_main, ts_main, alpha_fin, maxiter=1500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h

    if remaining() > 60:
        n_try = 256
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        c5_interp = compute_c5(h_interp)
        h_try = h_interp.copy()
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_try.copy()
            best_n = n_try
        for alpha in [50, 300, 1500]:
            if remaining() < 30:
                break
            h_try = slsqp_run(h_try, n_try, ts_try, alpha, maxiter=600)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
