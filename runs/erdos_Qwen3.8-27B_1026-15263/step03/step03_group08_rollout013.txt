import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_value_and_grad(h, alpha, top_k=None):
    corr = compute_corr(h)
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        corr_active = corr[idx]
        m = np.max(corr_active)
        shifted = alpha * (corr_active - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s) / alpha
        # gradient with respect to h
        n = len(h)
        g = np.zeros(n)
        for j, k in enumerate(idx):
            wl = w[j]
            if wl < 1e-12:
                continue
            l = k - (n - 1)
            # corr[k] = sum_i h[i] * (1 - h[i+l]) for valid i
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                # d/dh[i] of h[i]*(1-h[i+l]) = (1-h[i+l]) for the first term
                # and -h[i] for the second term (when i+l is the index)
                g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
                # For the (1-h[i+l]) part: d/dh[j] where j = i+l
                lo2 = max(0, -l)
                hi2 = min(n, n - l)
                if lo2 < hi2:
                    g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
        return sm_val, g
    else:
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s) / alpha
        n = len(h)
        g = np.zeros(n)
        for l in range(-(n-1), n):
            wl = w[l + n - 1]
            if wl < 1e-12:
                continue
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
                lo2 = max(0, -l)
                hi2 = min(n, n - l)
                if lo2 < hi2:
                    g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
        return sm_val, g

def slsqp_with_topk(h0, n, target_sum, alpha, top_k, maxiter):
    # Use SLSQP with soft-max of top_k correlations as objective
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        v, _ = soft_max_value_and_grad(h, alpha, top_k)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        _, g = soft_max_value_and_grad(h, alpha, top_k)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_descent(h0, n, target_sum, alpha, top_k, n_iters, lr_init, momentum=0.9):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        _, g = soft_max_value_and_grad(h, alpha, top_k)
        # Project gradient onto the constraint space (remove mean component)
        g = g - np.mean(g)  # project to sum-preserving direction
        v = momentum * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.9
            v *= 0.5
        h = h_new
    return best_h, best_c5

def line_search_gradient(h0, n, target_sum, alpha, top_k, n_iters, lr_init=0.1):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        _, g = soft_max_value_and_grad(h, alpha, top_k)
        g = g - np.mean(g)
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = np.clip(h - step * g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-10 * step * np.linalg.norm(g):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                lr = step * 1.5
                break
            step *= 0.5
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c5

def block_descent(h0, n, target_sum, alpha, top_k, n_blocks, block_size, n_iters_per_block):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

            h_test = h.copy()
            h_test[idx] = np.clip(x, 0.0, 1.0)

def generate_mechanical_word(n, slope, intercept=0.0):
    """Generate a mechanical word (Beatty sequence) of length n with given slope."""
    x = np.arange(n)
    h = np.floor((x + 1) * slope + intercept) - np.floor(x * slope + intercept)
    return h.astype(float)

def generate_sturmian(n, alpha):
    """Generate a Sturmian word."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.floor((i+1)*alpha) - np.floor(i*alpha)
    return h

        for top_k in [10, 20, 50]:
            if time_left() < 20:
                break
            n_iters = min(3000, int(time_left() * 2))
            h_new, c5_new = nesterov_active(h_cur, n, ts, top_k, n_iters, 0.03)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new

    # Phase 2: Try higher resolutions
    for n_try in [720, 960, 1200]:
        if time_left() < 80:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        if abs(ts_try - round(ts_try)) > 1e-9:
            continue
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        c5_interp = compute_c5(h_interp)
        h_try = h_interp

        for top_k in [50, 20, 10]:
            if time_left() < 25:
                break
            for alpha in [500, 2000, 5000]:
                if time_left() < 12:
                    break
                h_new = slsqp_toplevel(h_try, n_try, ts_try, alpha, top_k, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                h_try = h_new

        if time_left() > 40:
            n_iter = min(2000, int(time_left() * 1.5))
            h_new, c5_new = nesterov_active(h_try, n_try, ts_try, 10, n_iter, 0.02)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try

    # Phase 3: Try lower resolution (coarser can sometimes help)
    if time_left() > 80:
        for n_try in [360, 320]:
            if time_left() < 40:
                break
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            h_interp = interpolate_h(best_h, best_n, n_try)
            h_interp = project_to_sum(h_interp, ts_try)
            h_try = h_interp
            for top_k in [20, 50]:
                if time_left() < 15:
                    break
                for alpha in [500, 2000]:
                    if time_left() < 10:
                        break
                    h_new = slsqp_toplevel(h_try, n_try, ts_try, alpha, top_k, 300)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_try
                    h_try = h_new

    # Phase 4: Diverse starting points at best resolution
    n = best_n
    ts = n / 2.0
    if time_left() > 100:
        starts = []
        for _ in range(3):
            h_rand = (rng.rand(n) > 0.5).astype(float)
            h_rand = project_to_sum(h_rand, ts)
            starts.append(h_rand)
        for slope in [0.48, 0.49, 0.5, 0.51, 0.52]:
            h_mech = generate_mechanical(n, slope)
            h_mech = project_to_sum(h_mech, ts)
            starts.append(h_mech)
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [2, 3, 4, 5, 6, 8]:
            for amp in [0.35, 0.45]:
                for phase in [0.0, np.pi / 2, np.pi]:
                    hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                    hs = np.clip(hs, 0, 1)
                    hs = project_to_sum(hs, ts)
                    starts.append(hs)
        for delta in [0.1, 0.15, 0.25]:
            h_pert = best_h + rng.uniform(-delta, delta, n)
            h_pert = project_to_sum(h_pert, ts)
            starts.append(h_pert)

        scored = [(compute_c5(s), s) for s in starts]
        scored.sort(key=lambda t: t[0])
        for _, h_start in scored[:10]:
            if time_left() < 25:
                break
            h_s = h_start.copy()
            for top_k in [20, 50]:
                if time_left() < 15:
                    break
                h_new = slsqp_toplevel(h_s, n, ts, 1000.0, top_k, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_s = h_new

    # Phase 5: Final aggressive polish
    n = best_n
    ts = n / 2.0
    best_h = project_to_sum(best_h, ts)

    if time_left() > 50:
        for top_k in [5, 10, 15, 20]:
            if time_left() < 15:
                break
            for alpha in [2000, 5000, 10000, 20000]:
                if time_left() < 8:
                    break
                h_pol = slsqp_toplevel(best_h, n, ts, alpha, top_k, 600)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    if time_left() > 30:
        n_iter = min(3000, int(time_left() * 2))
        h_new, c5_new = nesterov_active(best_h, n, ts, 5, n_iter, 0.01, momentum=0.95)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 25:
        h_thr, c5_thr = threshold_and_reopt(best_h, n, ts, 0.05, 300)
        if c5_thr < best_c5:
            best_c5 = c5_thr
            best_h = h_thr.copy()
        h_thr2, c5_thr2 = threshold_and_reopt(best_h, n, ts, 0.1, 300)
        if c5_thr2 < best_c5:
            best_c5 = c5_thr2
            best_h = h_thr2.copy()

    if time_left() > 15:
        base_steps = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
        max_rnd = max(5, min(20, int(time_left() // 2)))
        h_cd, c5_cd = coordinate_descent(best_h, n, ts, max_rnd, base_steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
