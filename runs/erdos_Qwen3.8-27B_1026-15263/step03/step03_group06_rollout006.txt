            if i == j:
                continue
            if abs(i - j) > pair_dist_max and abs(i - j) > n - pair_dist_max:
                continue
            
            def f2d(params):
                h_tmp = h.copy()
                h_tmp[i] = params[0]
                h_tmp[j] = params[1]
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            
            x0 = [h[i], h[j]]
            bounds = [(0.0, 1.0), (0.0, 1.0)]
            try:
                res = minimize(f2d, x0, method='Nelder-Mead', bounds=bounds,
                             options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
                if res.fun < c5_cur - 1e-12:
                    h[i] = res.x[0]
                    h[j] = res.x[1]
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def simulated_annealing(h0, n, target_sum, n_iters, t_init, t_min, rng):
    """Simulated annealing on the h vector."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for it in range(n_iters):
        t = t_init * (t_min / t_init) ** (it / n_iters)
        # Propose a random perturbation
        h_new = h.copy()
        n_pert = rng.randint(1, max(2, n // 20))
        idx = rng.choice(n, n_pert, replace=False)
        h_new[idx] += rng.normal(0, 0.05, n_pert)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(t, 1e-8)):
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
    
    return h_best, c5_best
