def project_to_sum(h, target_sum):

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            jj1 = j_arr + m - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - m + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        lr = lr_init / (1.0 + 0.05 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d_scale in [-0.5, -0.25, -0.1, 0.1, 0.25, 0.5]:
                d = d_scale * base_delta
                nv = hi + d
                if nv < -1e-15 or nv > 1.0 + 1e-15:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_dft_matrix(n, K):
    B = np.zeros((n, 2 * K))
    i = np.arange(n)
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(2.0 * np.pi * k * i / n)
        B[:, 2 * (k - 1) + 1] = np.sin(2.0 * np.pi * k * i / n)
    return B

def softmax_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w = exp_vals / s
    obj = m + np.log(s / len(corr)) / alpha
    return obj, w

def gradient_from_weights(h, w):
    n = len(h)
    v = 1.0 - h
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    V = np.fft.rfft(v, N_fft)
    conv_result = np.fft.irfft(W * V, N_fft)
    A = conv_result[n - 1:2 * n - 1]
    corr_wh = np.correlate(w, h, mode='full')
    B = corr_wh[n - 1::-1][:n]
    return A - B

def compute_obj_grad(h, alpha):
    corr = np.correlate(h, 
1.0 - h, mode='full')
    obj, w = softmax_weights(corr, alpha)
    g = gradient_from_weights(h, w)
    return obj, g

def fourier_optimize(n, K, alpha, n_iters, lr_init, rng, h_init=None):
    B = build_dft_matrix(n, K)
    n_params = 2 * K
    if h_init is not None:
        residual = h_init - 0.5
        params, _, _, _ = np.linalg.lstsq(B, residual, rcond=None)
    else:
        params = rng.uniform(-0.4, 0.4, n_params)
    m_vec = np.zeros(n_params)
    v_vec = np.zeros(n_params)
    t = 0
    best_obj = float('inf')
    best_params = params.copy()
    h = 0.5 + B @ params
    h = np.clip(h, 0.0, 1.0)
    best_obj = compute_c5(h)
    for it in range(n_iters):
        t += 1
        h = 0.5 + B @ params
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        _, w = softmax_weights(corr, alpha)
        g_h = gradient_from_weights(h, w)
        g_params = B.T @ g_h
        m_vec = 0.9 * m_vec + 0.1 * g_params
        v_vec = 0.999 * v_vec + 0.001 * g_params * g_params
        m_hat = m_vec / (1.0 - 0.9 ** t)
        v_hat = v_vec / (1.0 - 0.999 ** t)
        lr = lr_init / (1.0 + 0.01 * it)
        params = params - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = 0.5 + B @ params
        h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_obj:
            best_obj = c5_new
            best_params = params.copy()
    h_final = 0.5 + B @ best_params
    h_final = np.clip(h_final, 0.0, 1.0)
    return h_final

def fourier_grid_alternating(h_init, n, target_sum, alpha, n_rounds, K, n_adam_per, rng):
    h = h_init.copy()
    B = build_dft_matrix(n, K)
    for r in range(n_rounds):
        residual = h - 0.5
        params, _, _, _ = np.linalg.lstsq(B, residual, rcond=None)
        m_vec = np.zeros(2 * K)
        v_vec = np.zeros(2 * K)
        t = 0
        n_fo = min(n_adam_per, 500)
        for _ in range(n_fo):
            t += 1
            h = 0.5 + B @ params
            h = np.clip(h, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            _, w = softmax_weights(corr, alpha)
            g_h = gradient_from_weights(h, w)
            g_params = B.T @ g_h
            m_vec = 0.9 * m_vec + 0.1 * g_params
            v_vec = 0.999 * v_vec + 0.001 * g_params * g_params
            m_hat = m_vec / (1.0 - 0.9 ** t)
            v_hat = v_vec / (1.0 - 0.999 ** t)
            params = params - 0.03 * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = 0.5 + B @ params
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        h = adam_step_grid(h, target_sum, alpha, n_adam_per, 0.04, rng)
    return h

def adam_step_grid(h, target_sum, alpha, n_iters, lr_init, rng=None):
    n = len(h)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_iters):
        t += 1
        _, g = compute_obj_grad(h, alpha)
        m_vec = 0.9 * m_vec + 0.1 * g
        v_vec = 0.999 * v_vec + 0.001 * g * g
        m_hat = m_vec / (1.0 - 0.9 ** t)
        v_hat = v_vec / (1.0 - 0.999 ** t)
        lr = lr_init / (1.0 + 0.005 * t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            jj1 = j_arr + m - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - m + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        lr = lr_init / (1.0 + 0.04 * it)
        h = h - lr * g / float(top_k)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    scales = np.array([-0.5, -0.25, -0.1, -0.05, 0.05, 0.1, 0.25, 0.5])
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d_scale in scales:
                d = d_scale * base_delta
                nv = hi + d
                if nv < -1e-15 or nv > 1.0 + 1e-15:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
