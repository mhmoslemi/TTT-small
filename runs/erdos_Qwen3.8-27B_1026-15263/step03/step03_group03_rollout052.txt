            fd = fc
            c = b - (b - a) / gr
            fc = np.max(A + D * c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = np.max(A + D * d)
    return (a + b) * 0.5

def compute_gradient(h, w, n):
    nm1 = n - 1
    g = np.zeros(n)
    for j in range(n):
        k_lo = max(0, nm1 - j)
        k_hi = min(2 * n - 2, nm1 - j + n - 1)
        if k_lo <= k_hi:
            idx = np.arange(k_lo, k_hi + 1)
            g[j] += np.dot(w[idx], 1.0 - h[j + idx - nm1])
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + nm1)
        if k_lo2 <= k_hi2:
            idx2 = np.arange(k_lo2, k_hi2 + 1)
            g[j] += np.dot(w[idx2], -h[j + nm1 - idx2])
    return g

def pair_descent_sweep(h, corr, rng, n_cand=15):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_candidates = [(j + 1) % n, (j - 1) % n, (j + 2) % n, (j - 2) % n,
                        (j + 3) % n, (j - 3) % n, (j + n // 4) % n, (j + n // 2) % n,
                        (j + 3 * n // 4) % n, (j + n // 3) % n, (j + 2 * n // 3) % n,
                        (j + n // 5) % n, (j + 2 * n // 5) % n, (j + 3 * n // 5) % n,
                        (j + 4 * n // 5) % n, (j + n // 6) % n, (j + 5 * n // 6) % n]
        for k in k_candidates:
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        for _ in range(max(0, n_cand - len(k_candidates))):
            k = rng.randint(n)
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def nesterov_subgrad_phase(h0, n, t_end, rng):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = np.zeros(n)
    momentum = 0.85
    alpha = 30.0
    dx = 2.0 / n
    best_local = np.max(corr) * dx
    best_h = h.copy()
    step = 0.02

    while time.time() < t_end:
        m_val = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m_val))
        w = exp_vals / np.sum(exp_vals)
        g = compute_gradient(h, w, n) * dx
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            g = g / g_norm * 0.5
        v = momentum * v - step * g
        h_new = h + v
        h_new = project_to_sum(h_new, n / 2.0)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx
        if c5_new < best_local - 1e-14:
            best_local = c5_new
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            v = v * 0.5
        else:
            step *= 0.95
            v = v * 0.5
            if step < 0.001:
                step = 0.02
                v = np.zeros(n)
                h = project_to_sum(h + rng.uniform(-0.02, 0.02, n), n / 2.0)
                corr = np.correlate(h, 1.0 - h, mode="full")
        alpha = min(alpha * 1.002, 150.0)
    return best_h, best_local

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Intensive pair descent (35% budget)
    t_end1 = t_start + 0.35 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0

    while time.time() < t_end1:
        h_work, corr_w, imp = pair_descent_sweep(h_work, corr_w, rng, n_cand=18)
        c5_w = np.max(corr_w) * (2.0 / n_w)
        if c5_w < best_c5 - 1e-13:
            best_c5 = c5_w
            best_h = h_work.copy()
            stall = 0
        else:
            stall += 1
        if stall > 0 and stall % 200 == 0:
            noise_scale = 0.02 + 0.02 * min(stall // 200, 3)
            h_work = project_to_sum(h_work + rng.uniform(-noise_scale, noise_scale, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
            if time.time() > t_start + 0.30 * budget_s:
                break

    # Phase 2: Nesterov subgradient (20% budget)
    t_end2 = t_start + 0.55 * budget_s
    h_sg = best_h.copy()
    n_sg = len(h_sg)
    h_nest, c5_nest = nesterov_subgrad_phase(h_sg, n_sg, t_end2, rng)
    if c5_nest < best_c5:
        best_c5 = c5_nest
        best_h = h_nest.copy()
    # Polish the subgradient result
    h_nest, corr_nest, _ = pair_descent_sweep(h_nest, np.correlate(h_nest, 1.0 - h_nest, mode="full"), rng, n_cand=20)
    c5_nest2 = np.max(corr_nest) * (2.0 / n_sg)
    if c5_nest2 < best_c5:
        best_c5 = c5_nest2
        best_h = h_nest.copy()

    # Phase 3: Multi-resolution exploration (25% budget)
    t_end3 = t_start + 0.80 * budget_s
    resolutions = [320, 360, 420, 480]
    for n_tr in resolutions:
        if time.time() >= t_end3:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        corr_tr = np.correlate(h_interp, 1.0 - h_interp, mode="full")
        max_sweeps = 40
        for _ in range(max_sweeps):
            if time.time() >= t_end3:
                break
            h_interp, corr_tr, _ = pair_descent_sweep(h_interp, corr_tr, rng, n_cand=12)
        c5_tr = np.max(corr_tr) * (2.0 / n_tr)
        if c5_tr < best_c5 - 1e-13:
            best_c5 = c5_tr
            best_h = h_interp.copy()
            best_n = n_tr
        # Also try random binary-like at this resolution
        h_rand = (rng.uniform(0, 1, n_tr) > 0.5).astype(float)
        h_rand = project_to_sum(h_rand, ts_tr)
        corr_rand = np.correlate(h_rand, 1.0 - h_rand, mode="full")
        for _ in range(25):
            if time.time() >= t_end3:
                break
            h_rand, corr_rand, _ = pair_descent_sweep(h_rand, corr_rand, rng, n_cand=10)
        c5_rand = np.max(corr_rand) * (2.0 / n_tr)
        if c5_rand < best_c5 - 1e-13:
            best_c5 = c5_rand
            best_h = h_rand.copy()
            best_n = n_tr

    # Phase 4: Final intensive polish (20% budget)
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    fin_stall = 0

    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=22)
        c5_fin = np.max(corr_fin) * (2.0 / n_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            fin_stall = 0
        else:
            fin_stall += 1
        if fin_stall > 150:
            h_fin = project_to_sum(h_fin + rng.uniform(-0.015, 0.015, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            fin_stall = 0

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
