        if abs(diff) < 1e-13:
            break
        mask = (h > 0) & (h < 1.0)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, target_ones):
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target_ones]] = 1.0
    return hb

def smooth_max_h(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(s)
    sw = np.sum(e)
    obj = m + np.log(sw) / alpha
    w = e / sw
    A = np.correlate(w, h, mode="full")[:n]
    B = np.correlate(w[::-1], h, mode="full")[:n]
    grad = 1.0 - A - B
    return obj, grad

def slsqp(h0, n, target_sum, alpha, maxiter=2000):
    def f(x):
        return smooth_max_h(x, alpha)[0]
    def g(x):
        return smooth_max_h(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa(h, n, target_ones, t_end, rng):
    c5, corr = get_c5_corr(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.008
    no_imp = 0
    while time.time() < t_end:
        # Targeted move
        if rng.random() < 0.6:
            k_star = np.argmax(corr)
            lag = k_star - (n - 1)
            if lag >= 0:
                vr = n - lag
                pc = np.where((h[:vr] == 1) & (h[lag:] == 0))[0]
                qc = np.where((h[:vr] == 0) & (h[lag:] == 1))[0]
            else:
                vs = -lag
                vr = n + lag
                pc = np.where((h[vs:] == 1) & (h[:vs+vr] == 0))[0] + vs
                qc = np.where((h[vs:] == 0) & (h[:vs+vr] == 1))[0] + vs
            if len(pc) > 0 and len(qc) > 0:
                p = pc[rng.randint(len(pc))]
                q = qc[rng.randint(len(qc))]
            else:
                p, q = rng.randint(n), rng.randint(n)
                if h[p] == h[q]:
                    continue
        else:
            p, q = rng.randint(n), rng.randint(n)
            if h[p] == h[q]:
                continue
        
        h[p], h[q] = h[q], h[p]
        nc5, ncorr = get_c5_corr(h)
        d = nc5 - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5, corr = nc5, ncorr
            if c5 < best_c5 - 1e-15:
                best_c5, best_h, no_imp = c5, h.copy(), 0
            else:
                no_imp += 1
        else:
            h[p], h[q] = h[q], h[p]
            no_imp += 1
        if no_imp > 8000:
            h[:] = best_h
            c5 = best_c5
            corr = np.correlate(h, 1.0-h, mode="full")
            T = min(T * 2.0, 0.03)
            no_imp = 0
        T = max(T * 0.99995, 0.002)
    return best_h, best_c5
