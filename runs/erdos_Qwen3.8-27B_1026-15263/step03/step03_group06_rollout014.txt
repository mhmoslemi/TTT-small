from scipy.optimize import minimize, minimize_scalar
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def correlate_fft(a, b):
    n = len(a)
    m = len(b)
    size = 2 * max(n, m)
    fa = fft(a, size)
    fb = fft(b, size)
    result = ifft(fa * np.conj(fb)).real[:n+m-1]

def active_set_smooth_max(h, alpha, eps=0.01, K_max=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    # Active set: lags where corr is within eps of max, or top K
    active = np.where(corr > cmax - eps)[0]
    if len(active) > K_max:
        active = np.argsort(corr)[-K_max:]
    # Smooth max over active set
    shifted = alpha * (corr[active] - cmax)
    exp_vals = np.exp(np.clip(shifted, -700, 700))
    sm = cmax + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha
    # Gradient
    w = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    for idx in active:
        wi = w[idx]
        # d(corr[idx] * dx)/d(h[j]) for each j
        # corr[idx] = sum_j h[j + (idx - n + 1)] * (1-h)[j] for valid j
        lag = idx - (n - 1)
        if lag >= 0:
            # corr = sum_{j=0}^{n-1-lag} h[j+lag] * (1-h[j])
            for j in range(n - lag):
                g[j + lag] += wi * dx * (1 - h[j])
                g[j] -= wi * dx * h[j + lag]
        else:
            lag = -lag
            # corr = sum_{j=lag}^{n-1} h[j] * (1-h[j-lag])
            for j in range(lag, n):
                g[j] += wi * dx * (1 - h[j - lag])
                g[j - lag] -= wi * dx * h[j]
    return sm, g
