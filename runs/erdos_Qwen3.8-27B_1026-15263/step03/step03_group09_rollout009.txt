import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        n_free = np.sum(mask)
        if n_free > 0:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2.0 * n - 1)) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        # First sum: sum_k w[k]*(1-h[j-k+n-1]) for valid k
        k_min = max(0, j - (n - 1))
        k_max = min(2 * n - 2, j + n - 1)
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            idx = j - ks + (n - 1)
            g[j] += np.dot(w[ks], 1.0 - h[idx])
        
        # Second sum: sum_k w[k]*h[j+k-n+1] for valid k
        k_min2 = max(0, n - 1 - j)
        k_max2 = min(2 * n - 2, 2 * n - 2 - j)
        if k_min2 <= k_max2:
            ks2 = np.arange(k_min2, k_max2 + 1)
            idx2 = j + ks2 - (n - 1)
            g[j] -= np.dot(w[ks2], h[idx2])
    
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_val_and_grad(h, alpha)
        return val
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_and_grad(h, alpha)
        return g
    
    def c_fun(x):
        return np.sum(x) - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_start, budget):
    h = h0.copy()
    v = np.zeros(n)
    m = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        if time.time() - t_start > budget - 60:
            break
        
        hm = h - m * v
        hm = np.clip(hm, 0.0, 1.0)
        _, g = smooth_max_val_and_grad(hm, alpha)
        
        v = 0.9 * v + g
        h_new = h - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        m = (1.0 + np.sqrt(1.0 + 4.0 * m * m)) / 2.0
        lr = lr_init / np.sqrt(it + 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, deltas_list, t_start, budget):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - t_start > budget - 30:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() - t_start > budget - 30:
                break
            hi = h[i]
            for d in deltas_list:
                for sign in [1.0, -1.0]:
                    nv = np.clip(hi + sign * d, 0.0, 1.0)
                    if abs(nv - hi) < 1e-15:
                        continue
                    h_new = h.copy()
                    h_new[i] = nv
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        break
                else:
                    continue
                break
        if not improved:
            break
    
    return h, c5_cur

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    return budget - (time.time() - t_start)
