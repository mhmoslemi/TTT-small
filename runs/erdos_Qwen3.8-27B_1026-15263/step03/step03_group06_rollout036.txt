def subgradient_descent(h0, n, target_sum, n_iters, step_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for it in range(n_iters):
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_iter = np.max(corr) * (2.0/n)
        
        # Find active lags (within tolerance of max)
        max_val = np.max(corr)
        tol = 0.01 * max_val
        active = np.where(corr >= max_val - tol)[0]
        
        # Compute subgradient (uniform weights over active set)
        n_active = len(active)
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            # g_L[j] = (1-h[j-L])*1[j-L valid] - h[j+L]*1[j+L valid]
            if lag >= 0:
                # h[j-lag] valid for j in [lag, n-1]
                g[lag:] += (1.0 - h[:n-lag])
                # h[j+lag] valid for j in [0, n-1-lag]
                g[:n-lag] -= h[lag:]
            else:
                L = -lag
                # h[j+L] valid for j in [0, n-1-L]
                g[:n-L] += (1.0 - h[L:])
                # h[j-L] valid for j in [L, n-1]
                g[L:] -= h[:n-L]
        g /= n_active
        g *= (2.0/n)  # dx factor
        
        # Step
        step = step_init / (1.0 + 0.01 * it)
        h_new = h - step * g
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Reduce step and retry
            step *= 0.5
            h_new = h - step * g
            h_new = project_to_sum(h_new, target_sum)

                c5_cur = c5_new

    corr = np.correlate(h, 1.0 - h, mode='full')

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, n, tol_frac=0.1):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = tol_frac * max_val
    active = np.where(corr >= max_val - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        if lag >= 0:
            size = n - lag
            if size > 0:
                g[lag:] += (1.0 - h[:size])
                g[:size] -= h[lag:]
        else:
            M = -lag
            size = n - M
            if size > 0:
                g[:size] += (1.0 - h[M:])
                g[M:] -= h[:size]
    g /= len(active)

    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    momentum = np.zeros(n)
    mu = 0.5

        h_new = h + momentum
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

                no_improve = 0
            elif no_improve > 100:

                no_improve = 0

        for i in active_coords:
            if time.time() > t_limit:
                break
            lo = max(0.0, h[i] - 0.5)
            hi = min(1.0, h[i] + 0.5)
            if hi - lo < 1e-5:
                continue
            try:
                res = minimize_scalar(_f1d, bounds=(lo, hi), method='bounded',
                                     args=(h, i, target_sum),
                                     options={'xatol': 1e-4})
                if res.fun < c5_cur - 1e-13:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        
        if not improved:
            break
    
    return h, c5_cur
