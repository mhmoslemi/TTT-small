    for j in range(n):

        k_end = min(2*n-2, 2*n-2-j)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha, target_sum, lambda_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s / len(exp_vals)) / alpha) * (2.0 / n)
    
    # Gradient of smooth max w.r.t. h
    g = np.zeros(n)
    for j in range(n):
        # First factor: h[j] appears as h[m] in corr[k] for k in [n-1-j, 2n-2-j]
        k_start = max(0, n - 1 - j)
        k_end = min(2 * n - 2, 2 * n - 2 - j)
        if k_start <= k_end:
            ks = np.arange(k_start, k_end + 1)
            h_idx = j + ks - n + 1
            h_idx = np.clip(h_idx, 0, n - 1)
            g[j] += np.dot(w[ks], 1.0 - h[h_idx])
        
        # Second factor: h[j] appears as h[m+k-n+1] in corr[k] for k in [j, j+n-1]
        k_start2 = j  # since j >= 0
        k_end2 = min(2 * n - 2, j + n - 1)
        if k_start2 <= k_end2:
            ks2 = np.arange(k_start2, k_end2 + 1)
            h_idx2 = j - ks2 + n - 1
            h_idx2 = np.clip(h_idx2, 0, n - 1)
            g[j] -= np.dot(w[ks2], h[h_idx2])
    
    g *= (2.0 / n)  # dx factor
    
    # Add sum penalty gradient
    sum_diff = np.sum(h) - target_sum
    g += 2.0 * lambda_sum * sum_diff * np.ones(n)
    
    val = sm_val + lambda_sum * sum_diff ** 2
    return val, g

    """Compute gradient of dot(weights, corr) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for idx in range(2*n - 1):
        w = weights[idx]
        if w == 0:
            continue
        if idx >= n - 1:
            # First factor: j in [idx-n+1, n-1]
            j_start = idx - n + 1
            j_end = n - 1
            for j in range(j_start, j_end + 1):

```python
def coord_descent_fast(h, n, target_sum, n_rounds, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_cur = np.max(corr) * dx
    
    for rnd in range(n_rounds):
        improved = False
        for j in rng.permutation(n):
            # Precompute gradient coefficients for this j
            g = np.zeros(2*n-1)
            # First factor: idx in [n-1-j, 2n-2-j]
            idx_start1 = max(0, n-1-j)
            idx_end1 = min(2*n-2, 2*n-2-j)
            for idx in range(idx_start1, idx_end1+1):
                g[idx] += (1.0 - h[j + n - 1 - idx])
            # Second factor: idx in [j, j+n-1]
            idx_start2 = max(0, j)
            idx_end2 = min(2*n-2, j+n-1)
            for idx in range(idx_start2, idx_end2+1):
                g[idx] -= h[j - n + 1 + idx]
            
            # But for idx = n-1, we have the quadratic correction
            # g[n-1] currently includes both first and second factor contributions
            # The actual change for idx=n-1 is: Δ*(1-2*old_val) + correction
            # Actually, the linear approximation gives g[n-1]*Δ
            # The exact change is: new*(1-new) - old*(1-old) = Δ*(1-new-old)
            # Linear approx: g[n-1]*Δ where g[n-1] = (1-2*old_val) [from first] + (-old_val)... 
            # Wait, for idx=n-1: first factor contribution = 1-h[j+n-1-(n-1)] = 1-h[j]
            # second factor contribution = -h[j-n+1+(n-1)] = -h[j]
            # Total g[n-1] = 1-h[j] - h[j] = 1-2*h[j] ✓ (matches d/dh[j] of h[j](1-h[j]))
            # The quadratic correction: exact = Δ*(1-new-old), linear = Δ*(1-2*old)
            # Difference: Δ*(1-new-old) - Δ*(1-2*old) = Δ*(old-new) = -Δ²
            # So new_corr[n-1] = old_corr[n-1] + g[n-1]*Δ - Δ²
            
            # Evaluate at several Δ values
            old_val = h[j]
            deltas = np.linspace(-old_val, 1.0-old_val, 20)
            best_c = c5_cur
            best_d = 0.0
            for d in deltas:
                new_corr = corr.copy()
                new_corr += d * g
                new_corr[n-1] -= d*d  # quadratic correction
                new_c = np.max(new_corr) * dx
                if new_c < best_c - 1e-12:
                    best_c = new_c
                    best_d = d
            if best_d != 0.0:
                h[j] = old_val + best_d
                h[j] = np.clip(h[j], 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_cur = np.max(corr) * dx
                # Project to sum
                h = project_to_sum(h, n/2.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_cur = np.max(corr) * dx
                improved = True
        if not improved:
            break
    return h, c5_cur
```

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
