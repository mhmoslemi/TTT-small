def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, top_m=5, sharpness=50.0):
    corr = compute_corr(h)
    n = len(h)
    max_val = np.max(corr)
    # Soft weights over top lags
    weights = np.exp(sharpness * (corr - max_val))
    weights /= np.sum(weights)
    g = np.zeros(n)
    # Only compute gradient for lags with significant weight
    threshold = np.max(weights) * 0.01
    for m in range(len(corr)):
        if weights[m] > threshold:
            lag = m - (n - 1)
            g += weights[m] * corr_gradient(h, lag)
    return g, corr

    g = np.zeros(n)
    for m in range(len(corr)):
        if w[m] > 1e-10:
            lag = m - (n - 1)
            g += w[m] * corr_gradient(h, lag)

    can_inc = h < 1.0 - 1e-12
    can_dec = h > 1e-12
    if not np.any(can_inc) or not np.any(can_dec):
        return h, c5_val, False

    g_inc = np.where(can_inc, g, np.inf)
    g_dec = np.where(can_dec, g, -np.inf)
    i_inc = int(np.argmin(g_inc))
    i_dec = int(np.argmax(g_dec))

    if i_inc == i_dec:
        return h, c5_val, False
    if g[i_inc] >= g[i_dec] - 1e-15:
        return h, c5_val, False

    h_work = h.copy()
    max_move = min(1.0 - h_work[i_inc], h_work[i_dec])

    lo, hi = 0.0, max_move
    best_step = 0.0
    best_c5 = c5_val
    best_h = None

    for _ in range(max_bisect):
        mid = (lo + hi) / 2.0
        if mid < 1e-15:
            break
        h_try = h_work.copy()
        h_try[i_inc] += mid
        h_try[i_dec] -= mid
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-16:
            best_c5 = c5_try
            best_step = mid
            best_h = h_try
            lo = mid
        else:
            hi = mid

    if best_step > 0:
        return best_h, best_c5, True
    return h, c5_val, False

def coord_descent_bisect(h, target_sum, max_iters=20000, patience=300, sharpness=30.0):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0

    for it in range(max_iters):
        h, c5_new, improved = cd_step_bisect(h, target_sum, sharpness=sharpness)
        if improved and c5_new < c5 - 1e-16:

            if stale > patience:
                break
    return h, c5

def subgradient_optimize(h, target_sum, max_iters=20000, init_alpha=0.005, sharpness=40.0):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    alpha = init_alpha
    v = np.zeros(n)
    stale = 0
    best_h = h.copy()
    best_c5 = c5

    for it in range(max_iters):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        w = np.exp(sharpness * (corr - max_corr))
        w /= np.sum(w)
        g = np.zeros(n)
        for m in range(len(corr)):
            if w[m] > 1e-10:
                lag = m - (n - 1)
                g += w[m] * corr_gradient(h, lag)

        v = 0.85 * v - alpha * g
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stale = 0
        else:
            alpha *= 0.92
            v *= 0.5
            stale += 1

        if stale > 4000:
            break
        if alpha < 1e-13:
            break

    return best_h, best_c5
