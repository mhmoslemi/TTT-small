def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 10)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def projected_gradient_step(h, grad, alpha, target_sum, step_size):
    h_new = h - step_size * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new

Let me think about the L-BFGS-B approach more carefully. The objective is:
f(h) = (1/α) ln(sum_k exp(α * corr_k))

where corr_k = sum_i h[i] * (1-h)[i+k-(n-1)]

The gradient is:
∂f/∂h[j] = sum_k w_k * (∂corr_k/∂h[j])

where w_k = exp(α * corr_k) / sum_l exp(α * corr_l)

And ∂corr_k/∂h[j] = (1-h[j+k-(n-1)]) - h[j-k+(n-1)] (with appropriate boundary handling)

For L-BFGS-B, we can handle the box constraints [0,1] natively. The sum constraint we handle by projecting the L-BFGS-B iterate.

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

def compute_B(h, j, n):

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Numerical stability

    w = exp_vals / np.sum(exp_vals)
    obj = m_val + (max_shifted + np.log(np.sum(exp_vals))) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        h = np.clip(v, 0, 1)
        return smooth_max_obj_grad(h, alpha)[0]

        h = np.clip(v, 0, 1)
        return smooth_max_obj_grad(h, alpha)[1]
    
    bounds = [(0.0, 1.0)] * n

    try:

    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, deadline):
    def objective(v):
        if time.time() > deadline:
            return 1e10
        h_clip = np.clip(v, 0, 1)
        return smooth_max_obj_grad(h_clip, alpha)[0]

    def gradient(v):
        h_clip = np.clip(v, 0, 1)
        return smooth_max_obj_grad(h_clip, alpha)[1]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objective, h0.copy(), method='L-BFGS-B', jac=gradient,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 30, 'maxfun': maxiter * 5})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def momentum_pg(h0, n, target_sum, alpha, n_iter, base_lr, deadline, beta=0.85, decay=0.998):
    h = h0.copy()
    v = np.zeros(n)
    lr = base_lr
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        _, grad = smooth_max_obj_grad(h, alpha)
        v = beta * v + (1.0 - beta) * grad
        h = h - lr * v
        h = project_to_sum(h, target_sum)
        lr *= decay
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Quick pair descent at original resolution (10%)
    t_end1 = t_start + 0.10 * budget_s
    h1 = best_h.copy()
    n1 = best_n
    corr1 = np.correlate(h1, 1.0 - h1, mode="full")
    while time.time() < t_end1:
        h1, corr1, _ = pair_descent_sweep(h1, corr1, rng, n_cand=10)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1

    # Phase 2: Interpolate to n=480, pair descent (25%)
    t_end2 = t_start + 0.35 * budget_s
    n2 = 480
    ts2 = n2 / 2.0
    x_old = np.linspace(0, 2, best_n, endpoint=False)
    x_new = np.linspace(0, 2, n2, endpoint=False)
    h2 = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts2)
    corr2 = np.correlate(h2, 1.0 - h2, mode="full")
    while time.time() < t_end2:
        h2, corr2, _ = pair_descent_sweep(h2, corr2, rng, n_cand=8)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

    # Phase 3: L-BFGS-B + momentum with smooth max (30%)
    t_end3 = t_start + 0.65 * budget_s
    n3 = best_n
    ts3 = n3 / 2.0
    h3 = best_h.copy()
    alpha_schedule = [40, 25, 15, 10, 8, 6, 5]
    for alpha in alpha_schedule:
        if time.time() >= t_end3:
            break
        h3 = lbfgsb_optimize(h3, n3, ts3, alpha, maxiter=150, deadline=t_end3)
        h3 = project_to_sum(h3, ts3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
        # Short pair descent between optimizations
        t_sub = min(t_end3, time.time() + 10)
        corr3 = np.correlate(h3, 1.0 - h3, mode="full")
        for _ in range(4):
            if time.time() >= t_sub:
                break
            h3, corr3, _ = pair_descent_sweep(h3, corr3, rng, n_cand=8)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
        # Momentum gradient as complement
        if time.time() < t_end3:
            h3 = momentum_pg(h3, n3, ts3, alpha, n_iter=50, base_lr=0.02, deadline=t_end3)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

    # Phase 4: Intensive pair descent with restarts (20%)
    t_end4 = t_start + 0.85 * budget_s
    h4 = best_h.copy()
    n4 = best_n
    corr4 = np.correlate(h4, 1.0 - h4, mode="full")
    stall = 0
    while time.time() < t_end4:
        h4, corr4, imp = pair_descent_sweep(h4, corr4, rng, n_cand=12)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5 - 1e-11:
            best_c5 = c5_4
            best_h = h4.copy()
            best_n = n4
            stall = 0
        else:
            stall += 1
        if stall == 80:
            noise = 0.04
            h4 = project_to_sum(h4 + rng.uniform(-noise, noise, n4), n4 / 2.0)
            corr4 = np.correlate(h4, 1.0 - h4, mode="full")
            stall = 0
        elif stall == 200:
            noise = 0.08
            h4 = project_to_sum(h4 + rng.uniform(-noise, noise, n4), n4 / 2.0)
            corr4 = np.correlate(h4, 1.0 - h4, mode="full")
            stall = 0
            # Also try L-BFGS-B from perturbed point
            h4 = lbfgsb_optimize(h4, n4, n4 / 2.0, 15, maxiter=80, deadline=t_end4)
            corr4 = np.correlate(h4, 1.0 - h4, mode="full")

    # Phase 5: Try higher resolutions (15%)
    remaining = time_left()
    if remaining > 20:
        resolutions_to_try = [600, 720, 840]
        per_res_time = max(8, remaining * 0.3)
        for n_try in resolutions_to_try:
            if time_left() < 8:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-6:
                continue
            x_cur = np.linspace(0, 2, best_n, endpoint=False)
            x_try = np.linspace(0, 2, n_try, endpoint=False)
            h_try = project_to_sum(np.interp(x_try, x_cur, best_h, period=2.0), ts_try)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            t_sub_end = min(t_start + budget_s - 3, time.time() + per_res_time * 0.5)
            while time.time() < t_sub_end:
                h_try, corr_try, _ = pair_descent_sweep(h_try, corr_try, rng, n_cand=6)
            # Quick L-BFGS-B polish
            if time_left() > 10:
                h_try = lbfgsb_optimize(h_try, n_try, ts_try, 12, maxiter=60, deadline=t_start + budget_s - 3)
                h_try = project_to_sum(h_try, ts_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            # Final pair descent at this resolution
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            while time.time() < t_start + budget_s - 3:
                h_try, corr_try, _ = pair_descent_sweep(h_try, corr_try, rng, n_cand=8)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Final feasibility check
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
