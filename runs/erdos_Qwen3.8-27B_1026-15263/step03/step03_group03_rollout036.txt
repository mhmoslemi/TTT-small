def exact_project(h, target):
    lo = -max(float(h.max()), 0.0) - 1.0
    hi = -min(float(h.min()), 1.0) + 2.0

        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max(obj_h, alpha):
    n = len(obj_h)
    corr = np.correlate(obj_h, 1.0 - obj_h, mode="full")
    m_val = np.max(corr)
    shifted = np.minimum(alpha * (corr - m_val), 0.0)
    exp_vals = np.exp(shifted)
    logsum = np.log(np.sum(exp_vals))
    obj = m_val + logsum / alpha
    w = exp_vals / np.sum(exp_vals)
    return obj, w, corr

def smooth_max_grad(obj_h, alpha):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def exact_project(h, target):
    lo = -max(float(np.max(h)), 0.0) - 1.0
    hi = -min(float(np.min(h)), 1.0) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.minimum(alpha * (corr - m_val), 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = m_val + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + (n - 1)
    second_idx = j_idx + m_idx - (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad


def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B


def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5


def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (j + n // 4) % n, (j + 3 * n // 4) % n]
        for _ in range(max(0, n_cand - 7)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved


def projected_grad_optimize(h0, target, alpha, n_steps, step0, momentum, time_limit=None, t_start=None):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    step = step0
    for i in range(n_steps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        obj, grad = smooth_max_obj_grad(h, alpha)
        vel = momentum * vel - step * grad
        h_new = exact_project(h + vel, target)
        obj_new = smooth_max_obj_grad(h_new, alpha)[0]
        if obj_new < obj:
            h = h_new
        else:
            step *= 0.5
            vel = momentum * vel
            h = exact_project(h + vel, target)
            if step < 1e-8:
                step = step0
                vel = np.zeros(n)
    return h
