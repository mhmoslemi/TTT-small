        if imp:
            stall = 0
        else:
            stall += 1
            if stall % 60 == 0:
                h_st = project_to_sum(h_st + rng.uniform(-0.08, 0.08, n_try), ts)
                corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
                B_st = compute_B_matrix(h_st)
    c5_st = compute_c5(h_st)
    if c5_st < target_c5:
        return h_st, c5_st
    return None, None

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Intensive pair descent with precomputed B (50% budget)
    t_end1 = t_start + 0.50 * budget_s
    h_w = best_h.copy()
    n_w = len(h_w)
    corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
    B_w = compute_B_matrix(h_w)
    stall_count = 0

    while time.time() < t_end1:
        h_w, corr_w, B_w, imp = pair_descent_sweep(h_w, corr_w, B_w, rng, n_cand=12)
        c5_w = np.max(corr_w) * (2.0 / n_w)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_w.copy()
            best_n = n_w
            stall_count = 0
        else:
            if not imp:
                stall_count += 1
        if stall_count >= 500:
            noise = 0.06
            h_w = project_to_sum(h_w + rng.normal(0, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            B_w = compute_B_matrix(h_w)
            stall_count = 0
        elif stall_count >= 200 and stall_count % 100 == 0:
            obj, grad, _ = smooth_max_grad(h_w, 100.0)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                h_w = project_to_sum(h_w - 0.02 * grad / gnorm * 0.5, n_w / 2.0)
            corr_w = np.correlate(h_w, 1.0 - h_w, mode="full")
            B_w = compute_B_matrix(h_w)

    # Phase 2: Gradient-based optimization (20% budget)
    t_end2 = t_start + 0.70 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    alphas = [30.0, 100.0, 300.0, 50.0, 150.0]
    h_grad, c5_grad = gradient_polish(best_h, n_f, ts_f, alphas, 800, rng, t_end2)
    if c5_grad < best_c5:
        best_c5 = c5_grad
        best_h = h_grad.copy()
        best_n = n_f

    # Phase 3: Multi-resolution exploration (20% budget)
    t_end3 = t_start + 0.90 * budget_s
    resolutions = []
    for rn in [160, 200, 240, 280, 320, 360, 400, 440, 480]:
        if rn != best_n and rn % 2 == 0:
            resolutions.append(rn)
    resolutions.sort(key=lambda x: abs(x - best_n))

    for n_tr in resolutions[:5]:
        if time.time() >= t_end3:
            break
        h_tr, c5_tr = try_resolution(best_h, best_n, n_tr, rng, t_end3, best_c5)
        if h_tr is not None and c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr

    # Phase 4: Final pair descent polish (10% budget)
    t_end4 = t_start + budget_s - 5.0
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    B_fin = compute_B_matrix(h_fin)
    stall_fin = 0
    while time.time() < t_end4:
        h_fin, corr_fin, B_fin, imp = pair_descent_sweep(h_fin, corr_fin, B_fin, rng, n_cand=14)
        c5_fin = np.max(corr_fin) * (2.0 / n_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            if not imp:
                stall_fin += 1
            if stall_fin >= 150:
                h_fin = project_to_sum(h_fin + rng.normal(0, 0.02, n_fin), n_fin / 2.0)
                corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
                B_fin = compute_B_matrix(h_fin)
                stall_fin = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
