from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)

    # params = [a_1, b_1, a_2, b_2, ..., a_K, b_K]

    return h

def fourier_obj_and_grad(params, K, n, alpha):

    h = fourier_h(params, K, x)
    # Penalty for violating [0,1]

    viol_low = np.maximum(0, -h)  # h < 0
    viol_high = np.maximum(0, h - 1)  # h > 1

    # Smooth max objective

The problem min max_m c(m) s.t. 0≤h≤1, sum(h)=n/2

This can be written as:
min t
s.t. c(m) ≤ t for all m
     0 ≤ h ≤ 1
     sum(h) = n/2

The constraint c(m) ≤ t is:
Σ_j h[j](1-h[j+m]) ≤ t/dx for valid j

This is a non-linear constraint (bilinear in h). So it's not directly a convex problem.

However, if we fix one part... Actually, c(m) = Σ_j h[j] - Σ_j h[j]h[j+m]. The first term is fixed (= n/2). So c(m) = n/2 - Σ_j h[j]h[j+m].

We want to minimize max_m (n/2 - Σ_j h[j]h[j+m]) which is equivalent to minimizing max_m (-Σ_j h[j]h[j+m]) or maximizing min_m Σ_j h[j]h[j+m].

So we want to MAXIMIZE the minimum autocorrelation!

max_h min_m A(m) where A(m) = Σ_j h[j]h[j+m] (with appropriate ranges)
s.t. 0 ≤ h ≤ 1, sum(h) = n/2

This is a max-min problem. The dual would be:
min_{p∈Δ} max_h Σ_m p[m] A(m) s.t. constraints
= min_{p∈Δ} max_h Σ_m p[m] Σ_j h[j]h[j+m]

The inner problem is: max_h h' Q(p) h s.t. 0≤h≤1, 1'h = n/2
where Q(p) is a matrix determined by p.

This is a quadratic program (QP) which is non-convex (maximizing a quadratic over a polytope). But it can be approximated.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kmax1 = min(j, n - 1)
        if kmax1 >= 0:
            ks = np.arange(kmax1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kmax2 = n - j
        if kmax2 > 0:
            ks2 = np.arange(kmax2)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def fourier_to_grid(params, K, n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[2*(k-1)] * np.cos(k * np.pi * x)
        h += params[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h

def fourier_obj(params, K, n, alpha, pen_weight):
    h = fourier_to_grid(params, K, n)
    h_clipped = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    sm_val, _ = smooth_max_and_grad(h_clipped, alpha)
    viol = np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h-1)**2)
    return sm_val * dx + pen_weight * viol * dx

def fourier_grad(params, K, n, alpha, pen_weight):
    x = np.linspace(0, 2, n, endpoint=False)
    h = fourier_to_grid(params, K, n)
    h_clipped = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    _, g_h = smooth_max_and_grad(h_clipped, alpha)
    g_pen = np.where(h < 0, -2.0 * h, 0.0) + np.where(h > 1, 2.0 * (h - 1), 0.0)
    g_total = g_h + pen_weight * g_pen
    g_params = np.zeros(2 * K)
    for k in range(1, K + 1):
        g_params[2*(k-1)] = np.dot(g_total, np.cos(k * np.pi * x)) * dx
        g_params[2*(k-1)+1] = np.dot(g_total, np.sin(k * np.pi * x)) * dx
    return g_params

def grid_to_fourier(h, K, n):
    x = np.linspace(0, 2, n, endpoint=False)
    h_centered = h - 0.5
    params = np.zeros(2 * K)
    for k in range(1, K + 1):
        params[2*(k-1)] = 2.0 * np.sum(h_centered * np.cos(k * np.pi * x)) / n
        params[2*(k-1)+1] = 2.0 * np.sum(h_centered * np.sin(k * np.pi * x)) / n
    return params

def adam_with_restarts(h0, n, target_sum, t_start, budget_s, max_alpha=1000.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 30.0
    t = 0
    total_iters = 0
    while _remaining(t_start, budget_s) > 15:
        n_phase = max(200, min(800, int(_remaining(t_start, budget_s) * 0.3)))
        lr = 0.06 / (1.0 + 0.05 * alpha)
        m_vec = np.zeros(n)
        v_vec = np.zeros(n)
        for i in range(n_phase):
            if _remaining(t_start, budget_s) < 15:
                break
            t += 1
            total_iters += 1
            _, g = smooth_max_and_grad(h, alpha)
            m_vec = beta1 * m_vec + (1.0 - beta1) * g
            v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
            m_hat = m_vec / (1.0 - beta1 ** t)
            v_hat = v_vec / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            h = h_new
            if c5_new < best_c5:

                best_h = h_new.copy()

def nesterov_adam(h0, n, target_sum, t_start, budget_s, alpha=100.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    v_vec = np.zeros(n)
    v2_vec = np.zeros(n)
    m2_vec = np.zeros(n)
    t = 0

    while _remaining(t_start, budget_s) > 20:

        for i in range(n_steps):

            v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
            m_hat = m_vec / (1.0 - beta1 ** t)
            v_hat = v_vec / (1.0 - beta2 ** t)
            h_nesterov = h - beta1 * (h - (h + lr * m_hat / (np.sqrt(v_hat) + eps)))

def subgradient_topk(h0, n, target_sum, t_start, budget_s, top_k=20, lr_init=0.04):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    j_arr = np.arange(n)
    it = 0
    while _remaining(t_start, budget_s) > 15:
        n_steps = max(100, min(1000, int(_remaining(t_start, budget_s) * 0.5)))
        for i in range(n_steps):
            if _remaining(t_start, budget_s) < 15:
                break
            it += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            top_idx = np.argsort(corr)[-top_k:]
            g = np.zeros(n)
            for m in top_idx:
                jj1 = j_arr + m - n + 1
                mask1 = (jj1 >= 0) & (jj1 < n)
                g[mask1] += (1.0 - h[jj1[mask1]])
                jj2 = j_arr - m + n - 1
                mask2 = (jj2 >= 0) & (jj2 < n)
                g[mask2] -= h[jj2[mask2]]
            lr = lr_init / (1.0 + 0.02 * it)
            h_new = h - lr * g / float(top_k)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-4.0, -3.0, -2.0, -1.0, 1.0, 2.0, 3.0, 4.0]) * base_delta
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-15 or nv > 1.0 + 1e-15:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def local_search_pairs(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    pairs_step = 0.003
    for _ in range(max_rounds):
        improved = False
        for _ in range(n * 2):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for di in [-pairs_step, pairs_step]:
                for dj in [-pairs_step, pairs_step]:
                    nv_i = h[i] + di
                    nv_j = h[j] + dj
                    if nv_i < 0 or nv_i > 1 or nv_j < 0 or nv_j > 1:
                        continue
                    h_new = h.copy()
                    h_new[i] = nv_i
                    h_new[j] = nv_j
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
            if improved:
                break
        if not improved:
            pairs_step *= 0.5
            if pairs_step < 1e-5:
                break
    return h, c5_cur

                    params0_hi = grid_to_fourier(h_interp, K_hi, n_hi)
                    try:
                        res_hi = minimize(fourier_obj, params0_hi, jac=fourier_grad,
                                        args=(K_hi, n_hi, 100.0, 100.0),
                                        method='L-BFGS-B',
                                        options={'maxiter': 200, 'ftol': 1e-14})
                        h_fh = np.clip(fourier_to_grid(res_hi.x, K_hi, n_hi), 0.0, 1.0)
                        h_fh = project_to_sum(h_fh, ts_hi)
                        try_update(h_fh, n_hi)
                    except Exception:
                        pass

            # Adam at higher res
            if _remaining(t_start, budget_s) > 60:
                h_hi_best = best_h.copy() if best_n != n_hi else h_interp.copy()
                if best_n == n_hi:
                    h_hi_best = h_interp.copy()
                else:
                    x_o2 = np.linspace(0, 2, best_n, endpoint=False)
                    x_n2 = np.linspace(0, 2, n_hi, endpoint=False)
                    h_hi_best = np.interp(x_n2, x_o2, best_h)
                    h_hi_best = project_to_sum(h_hi_best, ts_hi)
                h_ai, _ = adam_with_restarts(h_hi_best, n_hi, ts_hi, t_start, budget_s, max_alpha=500.0)
                try_update(h_ai, n_hi)

                if _remaining(t_start, budget_s) > 30:
                    h_cdi, _ = coord_descent(h_ai, n_hi, ts_hi, 20, rng, 0.002)
                    try_update(h_cdi, n_hi)

    # Phase 8: Final polishing on best solution
    if best_h is not None and _remaining(t_start, budget_s) > 30:
        n_fin = len(best_h)
        ts_fin = n_fin / 2.0
        h_fin = best_h.copy()

        if _remaining(t_start, budget_s) > 20:
            h_fin, _ = coord_descent(h_fin, n_fin, ts_fin, 30, rng, 0.0003)
            try_update(h_fin, n_fin)

        if _remaining(t_start, budget_s) > 15:
            h_fin = best_h.copy()
            h_fin, _ = coord_descent(h_fin, n_fin, ts_fin, 30, rng, 0.0001)
            try_update(h_fin, n_fin)

        if _remaining(t_start, budget_s) > 15:
            h_fin = best_h.copy()
            h_fin, _ = local_search_pairs(h_fin, n_fin, ts_fin, 10, rng)
            try_update(h_fin, n_fin)

    best_h = np.clip(best_h, 0.0, 1.0)
    ts_final = len(best_h) / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
