def corr_gradient_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def dual_ascent(h, target_sum, max_outer=200, eta=0.05, lr=0.15, time_budget=300):
    n = len(h)
    h = h.copy()
    h = project_to_sum(h, target_sum)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    p = np.ones(2 * n - 1) / (2.0 * n - 1)

    for outer in range(max_outer):
        if time.time() - t0 > time_budget:
            break
        corr = compute_corr(h)
        shifted = lr * (corr - np.max(corr))
        shifted -= np.max(shifted)
        p_new = np.exp(shifted)
        p_new = p_new / np.sum(p_new)
        p = 0.3 * p + 0.7 * p_new

        wg = np.zeros(n)
        active = np.where(p > 1e-6)[0]
        for lag_idx in active:
            lag = int(lag_idx) - (n - 1)
            wg += p[lag_idx] * corr_gradient_at_lag(h, lag)

        h_new = h - eta * wg
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            eta = max(eta * 0.9, 0.005)
    return best_h, best_c5

def binary_local_search(h, target_sum, max_iters=30000, time_budget=200):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    h_bin = project_to_sum(h_bin, target_sum)
    c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5
    t0 = time.time()

    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = compute_corr(h_bin)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_at_lag(h_bin, lag)

        best_gain = -np.inf
        best_o = -1
        best_z = -1
        sample_size = min(len(ones), 100)
        sample_zeros = min(len(zeros), 100)
        o_sample = ones[np.random.RandomState(it).choice(len(ones), sample_size, replace=False)]
        z_sample = zeros[np.random.RandomState(it + 999).choice(len(zeros), sample_zeros, replace=False)]

        for oi in o_sample:
            for zi in z_sample:
                gain = g[zi] - g[oi]
                if gain > best_gain:
                    best_gain = gain
                    best_o = int(oi)
                    best_z = int(zi)

        if best_gain < 0:
            h_bin[best_o] = 0.0
            h_bin[best_z] = 1.0
            c5_new = compute_c5(h_bin)
            if c5_new < c5:
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_bin.copy()
                    ones = np.where(h_bin > 0.5)[0]
                    zeros = np.where(h_bin < 0.5)[0]
            else:
                h_bin[best_o] = 1.0
                h_bin[best_z] = 0.0

    return best_h, best_c5

def coord_descent_fast(h, target_sum, max_iters=10000, step_frac=0.12, time_budget=200):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stale = 0
    t0 = time.time()

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_at_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-13:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stale = 0
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 150:
                break
    return best_h, best_c5

def anneal_blocks(h, target_sum, rng, max_iters=15000, time_budget=200):
    n = len(h)
    block_sizes = [1, 2, 3, 5, 8, 13, 21, 34, 55]
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = 0.008
    cooling = 0.9995
    t0 = time.time()

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        T = max(0.0005, T * cooling)
        bs = int(rng.choice(block_sizes))
        if bs >= n:
            continue
        start = int(rng.randint(0, n - bs))
        old_vals = h[start:start + bs].copy()
        h[start:start + bs] = 1.0 - old_vals
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[start:start + bs] = old_vals
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_curr = best_n
    ts = n_curr / 2.0

    if time_left() > 250:
        h_da, c5_da = dual_ascent(best_h, ts, max_outer=300, eta=0.04, lr=0.2, time_budget=200)
        if c5_da < best_c5:
            best_c5 = c5_da
            best_h = h_da.copy()

    if time_left() > 200:
        h_cd, c5_cd = coord_descent_fast(best_h, ts, max_iters=15000, step_frac=0.1, time_budget=150)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 200:
        h_da2, c5_da2 = dual_ascent(best_h, ts, max_outer=300, eta=0.02, lr=0.1, time_budget=150)
        if c5_da2 < best_c5:
            best_c5 = c5_da2
            best_h = h_da2.copy()

    if time_left() > 150:
        h_bl, c5_bl = binary_local_search(best_h, ts, max_iters=20000, time_budget=120)
        h_cd_b, c5_cd_b = coord_descent_fast(h_bl, ts, max_iters=8000, step_frac=0.15, time_budget=60)
        if c5_cd_b < best_c5:
            best_c5 = c5_cd_b
            best_h = h_cd_b.copy()

    if time_left() > 200:
        h_an, c5_an = anneal_blocks(best_h, ts, rng, max_iters=20000, time_budget=150)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an.copy()
        h_cd_a, c5_cd_a = coord_descent_fast(h_an, ts, max_iters=8000, step_frac=0.1, time_budget=80)
        if c5_cd_a < best_c5:
            best_c5 = c5_cd_a
            best_h = h_cd_a.copy()

    for n_tr in [420, 480, 600, 720, 840]:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        h_cd_r, c5_cd_r = coord_descent_fast(h_interp, ts_tr, max_iters=8000, step_frac=0.1, time_budget=60)
        if c5_cd_r < best_c5:
            best_c5 = c5_cd_r
            best_h = h_cd_r.copy()
            best_n = n_tr
            n_curr = n_tr
            ts = ts_tr
        h_da_r, c5_da_r = dual_ascent(h_cd_r, ts_tr, max_outer=150, eta=0.03, lr=0.15, time_budget=40)
        if c5_da_r < best_c5:
            best_c5 = c5_da_r
            best_h = h_da_r.copy()
            best_n = n_tr
            n_curr = n_tr
            ts = ts_tr

    n_starts = 12
    for trial in range(n_starts):
        if time_left() < 80:
            break
        noise = 0.015 + 0.04 * (1.0 - trial / float(n_starts))
        h_shake = best_h + rng.uniform(-noise, noise, best_n)
        h_shake = project_to_sum(h_shake, best_n / 2.0)
        h_cd_s, c5_cd_s = coord_descent_fast(h_shake, best_n / 2.0, max_iters=6000, step_frac=0.1, time_budget=30)
        if c5_cd_s < best_c5:
            best_c5 = c5_cd_s
            best_h = h_cd_s.copy()
        if time_left() > 40:
            h_da_s, c5_da_s = dual_ascent(h_cd_s, best_n / 2.0, max_outer=80, eta=0.02, lr=0.1, time_budget=20)
            if c5_da_s < best_c5:
                best_c5 = c5_da_s
                best_h = h_da_s.copy()

    if time_left() > 100:
        h_bl2, c5_bl2 = binary_local_search(best_h, best_n / 2.0, max_iters=30000, time_budget=80)
        h_cd_f, c5_cd_f = coord_descent_fast(h_bl2, best_n / 2.0, max_iters=10000, step_frac=0.08, time_budget=80)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time_left() > 60:
        h_da_f, c5_da_f = dual_ascent(best_h, best_n / 2.0, max_outer=200, eta=0.01, lr=0.08, time_budget=45)
        if c5_da_f < best_c5:
            best_c5 = c5_da_f
            best_h = h_da_f.copy()
        h_cd_final, c5_final = coord_descent_fast(best_h, best_n / 2.0, max_iters=20000, step_frac=0.04, time_budget=40)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cd_final.copy()

    if time_left() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        h_bin_final = project_to_sum(h_bin_final, best_n / 2.0)
        h_cd_bin, c5_bin = coord_descent_fast(h_bin_final, best_n / 2.0, max_iters=15000, step_frac=0.1, time_budget=25)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_cd_bin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
