            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fourier_to_h(coeffs, n):
    """Convert Fourier coefficients to h values.
    coeffs = [a_1, b_1, a_2, b_2, ..., a_K, b_K]
    h(x) = 1/2 + sum_k a_k * cos(k*pi*x/2 * 2) + b_k * sin(...)
    Wait, for period 2: h(x) = 1/2 + sum_k a_k * cos(k*pi*x) + b_k * sin(k*pi*x)
    """
    K = len(coeffs) // 2
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(K):
        a = coeffs[2*k]
        b = coeffs[2*k+1]
        h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, K):
    n = len(h)
    x = np.linspace(0, 2, n, endpoint=False)
    coeffs = []
    for k in range(1, K+1):
        a = (2.0/n) * np.sum(h * np.cos(k*np.pi*x))  # approximate
        b = (2.0/n) * np.sum(h * np.sin(k*np.pi*x))
        coeffs.extend([a, b])
    return np.array(coeffs)

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    # Compute gradient efficiently
    nonzero = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for j in nonzero:
        shift = j - (n - 1)
        wi = w[j]
        if shift >= 0:
            idx1 = np.arange(shift, n)
            grad[idx1] += wi * (1.0 - h[idx1 - shift])
            idx2 = np.arange(0, n - shift)
            grad[idx2] -= wi * h[idx2 + shift]
        else:
            s = -shift
            idx1 = np.arange(0, n - s)
            grad[idx1] += wi * (1.0 - h[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= wi * h[idx2 - s]
    return obj, grad

def proj_grad_smooth_max(h0, n, ts, alpha, n_iter=10000, lr_init=0.05, lr_min=1e-10):
    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_val(h, alpha)
    stall = 0
    for i in range(n_iter):
        obj, grad = smooth_max_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        obj_new = smooth_max_val(h_new, alpha)
        if obj_new < best_obj - 1e-16:
            h = h_new
            best_obj = obj_new
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.8
            stall += 1
            if stall > 50 or lr < lr_min:
                break
    return h, best_obj

def slsqp_smooth_max(h0, n, ts, alpha, maxiter=500):
    def fun(x):
        return smooth_max_val(x, alpha)
    def grad(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        result = minimize(fun, h0, method='SLSQP', jac=grad, bounds=bnds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_to_sum(result.x, ts)
    except:
        return h0

def top_lag_opt(h0, n, ts, M=20, n_outer=30, n_inner=5000, lr_init=0.02):
    """Iteratively optimize the top M lags."""
    h = h0.copy()
    for outer in range(n_outer):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Select top M lags with soft weighting
        sorted_idx = np.argsort(corr)[::-1]
        top_M = sorted_idx[:M]
        # Use exponential weights
        w = np.zeros(2*n-1)
        w[top_M] = np.exp(corr[top_M] - corr[top_M[0]])
        w[top_M] /= np.sum(w[top_M])
        
        # Projected gradient on the weighted sum
        h = top_lag_proj_grad(h, n, ts, w, n_inner, lr_init)
    return h

def top_lag_proj_grad(h, n, ts, w, n_inner, lr):
    h = h.copy()
    best_obj = np.sum(w * np.correlate(h, 1.0 - h, mode="full"))
    stall = 0
    for i in range(n_inner):
        # Compute gradient
        nonzero = np.where(w > 1e-12)[0]
        grad = np.zeros(n)
        for j in nonzero:
            shift = j - (n - 1)
            wi = w[j]
            if shift >= 0:
                idx1 = np.arange(shift, n)
                grad[idx1] += wi * (1.0 - h[idx1 - shift])
                idx2 = np.arange(0, n - shift)
                grad[idx2] -= wi * h[idx2 + shift]
            else:
                s = -shift
                idx1 = np.arange(0, n - s)
                grad[idx1] += wi * (1.0 - h[idx1 + s])
                idx2 = np.arange(s, n)
                grad[idx2] -= wi * h[idx2 - s]
        
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        obj_new = np.sum(w * np.correlate(h_new, 1.0 - h_new, mode="full"))
        if obj_new < best_obj:
            h = h_new
            best_obj = obj_new
            lr = min(lr * 1.01, 0.3)
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 100 or lr < 1e-10:
                break
    return h

def stochastic_cd(h, n, ts, rng, n_steps=30000):
    h = h.copy()
    c5_cur = compute_c5(h)
    for step in range(n_steps):
        # Pick a random pair and do a mass transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Determine max delta
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-6:
            continue
        delta = rng.uniform(0.005, min(0.2, max_d))
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new <= c5_cur:
            c5_cur = c5_new
        else:
            # Revert
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
    return h, c5_cur
