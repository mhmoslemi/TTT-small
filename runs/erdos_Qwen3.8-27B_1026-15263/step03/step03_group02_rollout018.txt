def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    # Project h to satisfy sum constraint while staying in [0,1]
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)

def make_binary(h, n, target_sum, rng):
    """Convert to binary, maintaining exact sum."""
    h_bin = (h > 0.5).astype(float)

    cs = valid_s * h[sc] * w[None, :]

    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad
