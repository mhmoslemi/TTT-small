import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_gradient(h, target_corr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        excess = corr[k_idx] - target_corr
        if excess <= 0:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            g[l:n] += excess * (1.0 - h[0:n - l])
            g[0:n - l] -= excess * h[l:n]
        else:
            m = -l
            g[0:n - m] += excess * (1.0 - h[m:n])
            g[m:n] -= excess * h[0:n - m]
    return corr, g

def adaptive_penalty_descent(h0, target_sum, C_init, n_iters, lr_init):
    n = len(h)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    C_cur = C_init
    lr = lr_init
    dx = 2.0 / n
    patience = 0
    for it in range(n_iters):
        target_corr = C_cur / dx
        corr, g = penalty_gradient(h, target_corr)
        max_corr = np.max(corr)
        if max_corr <= target_corr + 1e-10:
            C_cur *= 0.995
            target_corr = C_cur / dx
            continue
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        step = lr * g / g_norm
        h_new = np.clip(h - step, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        if c5_new <= c5_new and c5_new < compute_c5(h):
            h = h_new
            lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.85
            h = np.clip(h - lr * g / (g_norm + 1e-15), 0.0, 1.0)
            h = project_to_sum(h, target_sum)
        if it % 50 == 0 and patience > 20:
            C_cur = max(0.35, C_cur * 0.99)
            patience = 0
    return best_h, best_c5

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            g[l:n] += wk * (1.0 - h[0:n - l])
            g[0:n - l] -= wk * h[l:n]
        else:
            m2 = -l
            g[0:n - m2] += wk * (1.0 - h[m2:n])
            g[m2:n] -= wk * h[0:n - m2]
    return sm_val, g

def slsqp_refine(h0, target_sum, alpha, maxiter):
    n = len(h)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_obj_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_obj_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def level_top_corrs(h0, target_sum, n_levels, n_iters):
    n = len(h)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-n_levels:]
        top_vals = corr[top_k]
        target_level = top_vals.mean()
        g = np.zeros(n)
        for k_idx in top_k:
            val = corr[k_idx]
            sign = 1.0 if val > target_level else -1.0
            weight = abs(val - target_level) + 0.1
            l = k_idx - (n - 1)
            if l >= 0:
                g[l:n] += sign * weight * (1.0 - h[0:n - l])
                g[0:n - l] -= sign * weight * h[l:n]
            else:
                m = -l
                g[0:n - m] += sign * weight * (1.0 - h[m:n])
                g[m:n] -= sign * weight * h[0:n - m]
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        h_new = np.clip(h - lr * g / g_norm, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h):
            h = h_new
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9
    return best_h, best_c5

def fine_coordinate(h0, target_sum, n_rounds, step_sizes):
    n = len(h)
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(0, n, 1):
            best_local = c5_cur
            best_val = h[i]
            for d in step_sizes:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                h[i] = v
                h = project_to_sum(h, target_sum)
                c5_t = compute_c5(h)
                if c5_t < best_local:
                    best_local = c5_t
                    best_val = v
            h[i] = best_val
            h = project_to_sum(h, target_sum)
            if best_local < c5_cur - 1e-15:
                c5_cur = best_local
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n = best_n
    ts = n / 2.0
    h = best_h.copy()
    C_cur = compute_c5(h)
    # Phase 1: Quick penalty descent
    if time_left() > 60:
        n_iters = min(2000, int(time_left() * 3))
        h, c5 = adaptive_penalty_descent(h, ts, C_cur * 0.999, n_iters, 0.1)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Phase 2: Level top correlations
    if time_left() > 50:
        for n_lev in [3, 5, 8, 10]:
            if time_left() < 20:
                break
            n_it = min(1000, int(time_left() * 2))
            h, c5 = level_top_corrs(best_h, ts, n_lev, n_it)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    # Phase 3: SLSQP with smooth max, multiple alphas
    if time_left() > 40:
        for alpha in [100.0, 300.0, 800.0, 2000.0, 5000.0]:
            if time_left() < 15:
                break
            mit = max(50, min(400, int(time_left() * 0.5)))
            h_opt = slsqp_refine(best_h, ts, alpha, mit)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    # Phase 4: More penalty descent with tighter target
    if time_left() > 40:
        C_tight = max(0.35, best_c5 - 0.001)
        n_iters = min(3000, int(time_left() * 3))
        h, c5 = adaptive_penalty_descent(best_h, ts, C_tight, n_iters, 0.05)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Phase 5: Multi-start at same resolution
    if time_left() > 100:
        x = np.linspace(0, 2, n, endpoint=False)
        starts = []
        for delta in [0.03, 0.06, 0.1, 0.15]:
            h_p = best_h + rng.uniform(-delta, delta, n)
            h_p = project_to_sum(h_p, ts)
            starts.append(h_p)
        for freq in [1, 2, 3, 5, 8]:
            for amp in [0.15, 0.25, 0.35]:
                hs = best_h + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + rng.uniform(0, 2*np.pi))
                hs = project_to_sum(hs, ts)
                starts.append(hs)
        n_mixed = len(starts)
        per_start_time = max(15, (time_left() - 20) / max(1, n_mixed))
        for h_s in starts:
            if time_left() < 25:
                break
            c5_s = compute_c5(h_s)
            h_work = h_s.copy()
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_work.copy()
            # Quick penalty
            n_it = min(800, int(per_start_time * 2))
            C_t = max(0.35, c5_s * 0.998)
            h_r, c5_r = adaptive_penalty_descent(h_work, ts, C_t, n_it, 0.08)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            # Quick SLSQP
            if time_left() > 30:
                h_o = slsqp_refine(h_r, ts, 500.0, min(200, int(per_start_time)))
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
    # Phase 6: Try higher resolution n=720

            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_to_sum(h_up, ts2)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.01:
                h_h = h_up.copy()
                # Penalty at higher res

                    n_it = min(2000, int(time_left() * 2))
                    h_h, c5_h = adaptive_penalty_descent(h_h, ts2, c5_up * 0.999, n_it, 0.08)
                    if c5_h < best_c5:
                        best_c5 = c5_h
                        best_h = h_h.copy()
                        best_n = n2
                # SLSQP at higher res
                if time_left() > 50:
                    h_o = slsqp_refine(h_h, ts2, 500.0, min(300, int(time_left() * 0.6)))
                    c5_o = compute_c5(h_o)
                    if c5_o < best_c5:
                        best_c5 = c5_o
                        best_h = h_o.copy()
                        best_n = n2
    # Phase 7: Final SLSQP polish
    if time_left() > 30:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        for alpha in [1000.0, 3000.0]:
            if time_left() < 12:
                break
            h_p = slsqp_refine(best_h, ts_f, alpha, min(500, int(time_left() * 0.8)))
            c5_p = compute_c5(h_p)

                best_h = h_p.copy()
    # Phase 8: Final penalty + coordinate
    if time_left() > 25:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        C_t = max(0.34, best_c5 - 0.0005)
        n_it = min(2000, int(time_left() * 2))
        h_p, c5_p = adaptive_penalty_descent(best_h, ts_f, C_t, n_it, 0.03)

    if time_left() > 15:
