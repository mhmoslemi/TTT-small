import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    s = corr - m_val
    max_s = np.max(s)
    exp_s = np.exp(alpha * (s - max_s))
    return (m_val + max_s + np.log(np.sum(exp_s)) / alpha) * dx

    n = len(h)
    h = h.copy()
    lr = lr_init
    best_h = h.copy()
    best_obj = smooth_max_obj(h, alpha)
    for it in range(max_iter):
        if time_left_fn() < 3:

        corr = np.correlate(h, 1.0 - h, mode="full")

        v = 1.0 - 2.0 * h

        h_new = project_box_sum(h - lr * grad, target_sum)
        obj_new = smooth_max_obj(h_new, alpha)
        if obj_new < best_obj:

def binary_sa(h_bin, n, rng, time_left_fn, T_init=0.004, cooling=0.99992):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T_init
    T_min = 1e-8
    n_half = n // 2
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    no_improve = 0
    while time_left_fn() > 12:
        i = ones_list[rng.randint(n_half)]
        j = zeros_list[rng.randint(n_half)]
        h[i] = 0.0
        h[j] = 1.0
        ones_list[ones_list.index(i)] = j
        zeros_list[zeros_list.index(j)] = i
        c5_new = compute_c5(h)

                no_improve = 0

                no_improve += 1

            h[j] = 0.0
            h[i] = 1.0
