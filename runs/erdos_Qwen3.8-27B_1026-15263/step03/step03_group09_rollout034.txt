import time

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    _ts = target_sum
    def obj(x):
        val, _ = smooth_max_and_grad(x, alpha)
        return val
    def grad(x):
        _, g = smooth_max_and_grad(x, alpha)
        return g
    def cfun(x):
        return np.sum(x) - _ts
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for _ in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if t % 100 == 0:
            lr *= 0.9
    return best_h, best_c5

def subgradient_search(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            # For shift m, the contribution to the gradient
            # ∂corr[m]/∂h[j] = (1-h[j+m-n+1]) - h[j-m+n-1] (with validity)
            # First part: (1-h[j+m-n+1]) for j in [max(0, n-1-m), min(n-1, 2n-2-m)]
            lo1 = max(0, n - 1 - m)
            hi1 = min(n - 1, 2 * n - 2 - m)
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1 + 1)
                g[idx] += (1.0 - h[idx + m - (n - 1)])
            # Second part: -h[j-m+n-1] for j in [max(0, m-n+1), min(n-1, m)]
            lo2 = max(0, m - n + 1)
            hi2 = min(n - 1, m)
            if lo2 <= hi2:
                idx2 = np.arange(lo2, hi2 + 1)
                g[idx2] -= h[idx2 - m + (n - 1)]
        
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = h - lr * g / top_k
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    j_arr = np.arange(n)
    for r in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_h = h.copy()
            for d in [-2.0, -1.0, 0.5, 1.0, 2.0] * base_delta:
                nv = h[i] + d
                if nv < -1e-14 or nv > 1.0 + 1e-14:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val:
                    best_val = c5_new
                    best_h = h_new
            if best_val < c5_cur - 1e-15:
                h = best_h
                c5_cur = best_val
                improved = True
        if not improved:
            break
