```python
    # => grad_s^T h - C/dx <= grad_s^T h^k - A_s(h^k)
    m = 2 * n - 1  # number of shifts
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for m_idx in range(m):
        # shift s = m_idx - (n-1)
        s = m_idx - (n - 1)
        # Compute A_s(h) and gradient
        # h[i] for i in [0, n-1], h[i+s] for i+s in [0, n-1]
        i_start = max(0, -s)
        i_end = min(n, n - s) if s >= 0 else n
        # Actually: valid i are those with 0<=i<n and 0<=i+s<n
        # If s >= 0: i in [0, n-1-s]
        # If s < 0: i in [-s, n-1]
        
        grad = np.zeros(n)
        A_val = 0.0
        
        for i in range(max(0, -s), min(n, n - s)):
            # term: h[i] * (1 - h[i+s])
            A_val += h[i] * (1 - h[i+s])
            # gradient w.r.t. h[i]: (1-h[i+s])
            grad[i] += (1 - h[i+s])
            # gradient w.r.t. h[i+s]: -h[i]
            grad[i+s] += (-h[i])
        
        # Constraint: grad^T h - C/dx <= grad^T h - A_val
        A_ub[m_idx, :n] = grad
        A_ub[m_idx, n] = -1.0 / dx
        b_ub[m_idx] = grad @ h - A_val
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n] / dx, C_current
    return h, C_current, C_current
```

Let me redo: The linearization of A_s at h^k is:
A_s(h) ≈ A_s(h^k) + ∇A_s(h^k)^T (h - h^k)

Constraint: A_s(h) ≤ C/dx
Approx: A_s(h^k) + ∇A_s(h^k)^T (h - h^k) ≤ C/dx
⇒ ∇A_s(h^k)^T h ≤ C/dx + ∇A_s(h^k)^T h^k - A_s(h^k)
⇒ ∇A_s(h^k)^T h - C/dx ≤ ∇A_s(h^k)^T h^k - A_s(h^k)

The RHS at h = h^k is: ∇A_s(h^k)^T h^k - C_current/dx - A_s(h^k) + A_s(h^k) ... 

Hmm, let me just plug in h = h^k on the LHS:
∇A_s(h^k)^T h^k - C^k/dx

And the RHS is:
∇A_s(h^k)^T h^k - A_s(h^k)

So the constraint at h = h^k is:
∇A_s(h^k)^T h^k - C^k/dx ≤ ∇A_s(h^k)^T h^k - A_s(h^k)
⇒ C^k/dx ≥ A_s(h^k)

But C^k = max_s A_s(h^k) * dx, so C^k/dx = max_s A_s(h^k) ≥ A_s(h^k) for all s. ✓

The constraint is satisfied at h^k with equality only for the active shifts (those where A_s = max). Good.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    # Same as before
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    s = j - nm1  # shift for variable j in the first position
    # Actually, let me use the same formulation as before
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(32):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
            seen.add(k)
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0: h[j] = 0.0
            elif h[j] > 1: h[j] = 1.0
            if h[best_k] < 0: h[best_k] = 0.0
            elif h[best_k] > 1: h[best_k] = 1.0
            improved = True
    return h, corr, improved

def slp_iteration(h, n, dx):
    """One SLP iteration using LP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    C_current = np.max(corr)
    
    nm1 = n - 1
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for s in range(-nm1, nm1 + 1):
        m_idx = s + nm1
        grad = np.zeros(n)
        A_val = 0.0
        i_start = max(0, -s)
        i_end = min(n, n - s)
        for i in range(i_start, i_end):
            A_val += h[i] * (1.0 - h[i + s])
            grad[i] += (1.0 - h[i + s])
            grad[i + s] += (-h[i])
        A_ub[m_idx, :n] = grad
        A_ub[m_idx, n] = -1.0 / dx
        b_ub[m_idx] = grad @ h - A_val
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[n]) / dx
    except Exception:
        pass
    return h.copy(), C_current
```
