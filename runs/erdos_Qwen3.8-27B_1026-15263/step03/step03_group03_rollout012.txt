def compute_single_B(h, j, n):

    m_end = min(2 * n - 2, j)

def solve_1d_minimax(A, D, lo, hi):
    """Find delta in [lo, hi] that minimizes max(A + D*delta)."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # First, coarse grid
    n_grid = 20
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        lo_r, hi_r = lo, t_grid[1]
    elif best_idx == n_grid - 1:
        lo_r, hi_r = t_grid[-2], hi
    else:
        lo_r = t_grid[best_idx - 1]
        hi_r = t_grid[best_idx + 1]
    # Bisection
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def smooth_gradient_step(h, alpha, step_size):
    """One step of smoothed gradient descent on the max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_single_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    m_start = max(0, j - nm1)
    m_end = min(2 * n - 2, j)
    if m_start <= m_end:
        idx = np.arange(m_start, m_end + 1)
        partner = idx - j + nm1
        B[m_start:m_end + 1] += (1.0 - h[partner])
    m_start2 = max(0, nm1 - j)
    m_end2 = min(2 * n - 2, 2 * nm1 - j)
    if m_start2 <= m_end2:
        idx = np.arange(m_start2, m_end2 + 1)
        partner = idx - nm1 + j
        B[m_start2:m_end2 + 1] += -h[partner]
    return B

def solve_1d_minimax(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    n_grid = 24
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        lo_r, hi_r = lo, t_grid[1]
    elif best_idx == n_grid - 1:
        lo_r, hi_r = t_grid[-2], hi
    else:
        lo_r = t_grid[best_idx - 1]
        hi_r = t_grid[best_idx + 1]
    for _ in range(44):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent(h, corr, rng, n_cand=25):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    max_val = np.max(corr)
    for j in order:
        B_j = compute_single_B(h, j, n)
        best_val = max_val
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_single_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_minimax(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0.0:
                h[j] = 0.0
            elif h[j] > 1.0:
                h[j] = 1.0
            if h[best_k] < 0.0:
                h[best_k] = 0.0
            elif h[best_k] > 1.0:
                h[best_k] = 1.0
            improved = True
            max_val = np.max(corr)
    return h, corr, improved

def smooth_grad_descent(h, n, alpha, n_steps, base_step):
    target_sum = n / 2.0
    h_cur = h.copy()
    best_local = compute_c5(h_cur)
    step = base_step
    for _ in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        m_val = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m_val))
        w = exp_vals / np.sum(exp_vals)
        nm1 = n - 1
        grad = np.zeros(n)
        for j in range(n):
            m_lo = max(0, j - nm1)
            m_hi = min(2 * n - 2, j)
            if m_lo <= m_hi:
                idx = np.arange(m_lo, m_hi + 1)
                partner = idx - j + nm1
                grad[j] += np.dot(w[m_lo:m_hi + 1], 1.0 - h_cur[partner])
            m_lo2 = max(0, nm1 - j)
            m_hi2 = min(2 * n - 2, 2 * nm1 - j)
            if m_lo2 <= m_hi2:
                idx = np.arange(m_lo2, m_hi2 + 1)
                partner = idx - nm1 + j
                grad[j] += np.dot(w[m_lo2:m_hi2 + 1], -h_cur[partner])
        h_new = h_cur - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            h_cur = h_new
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.6
            if step < 1e-7:
                break
    return h_cur, best_local

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Intensive pair descent at original resolution
    t_end1 = t_start + 0.45 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall_count = 0
    while time.time() < t_end1:
        h_work, corr_w, imp = pair_descent(h_work, corr_w, rng, n_cand=30)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 0 and stall_count % 40 == 0:

            h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    # Phase 2: Smooth gradient descent
    t_end2 = t_start + 0.60 * budget_s
    for alpha in [30, 100, 300, 800]:
        if time.time() >= t_end2:
            break
        h_g, c5_g = smooth_grad_descent(best_h, best_n, alpha, n_steps=200, base_step=0.004)
        c5_g = compute_c5(h_g)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 3: Higher resolution exploration
    t_end3 = t_start + 0.85 * budget_s
    for n_hi in [360, 480]:
        if time.time() >= t_end3:
            break
        ts = n_hi / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h, period=2.0)
        h_hi = project_to_sum(h_hi, ts)
        # Also try random init
        h_rand = project_to_sum(rng.uniform(0, 1, n_hi), ts)
        for h_init_arr in [h_hi, h_rand]:
            if time.time() >= t_end3:
                break
            h_cur = h_init_arr.copy()
            corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            stall = 0
            while time.time() < t_end3:

                c5_cur = compute_c5(h_cur)

                    best_h = h_cur.copy()
                    best_n = n_hi
                    stall = 0
                else:
                    stall += 1
                if stall > 60:
                    h_cur = project_to_sum(h_cur + rng.uniform(-0.03, 0.03, n_hi), ts)
                    corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode="full")

                if stall > 150:
                    break

    # Phase 4: Final polish
    t_end4 = t_start + budget_s - 5
    if time.time() < t_end4:
        h_fin = best_h.copy()
        n_fin = len(h_fin)
        corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
        while time.time() < t_end4:
            h_fin, corr_fin, imp = pair_descent(h_fin, corr_fin, rng, n_cand=20)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
