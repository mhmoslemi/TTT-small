```python
def subgradient_at_argmax(h, n):
    """Compute subgradient at the argmax lag only."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    p = k_star - (n-1)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    if p >= 0:
        grad[:n-p] += one_minus_h[p:]
        grad[p:] -= h[:n-p]
    else:
        q = -p
        grad[q:] += one_minus_h[:n-q]
        grad[:n-q] -= h[q:]
    return grad * (2.0 / n), k_star
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def find_active_lags(corr, top_frac=0.97, max_lags=10):
    m = np.max(corr)
    active = np.where(corr >= top_frac * m)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    elif len(active) > max_lags:
        vals = corr[active]
        top_idx = np.argsort(vals)[::-1][:max_lags]
        active = active[top_idx]
    return active

def smooth_max_grad(h, active, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    subset = corr[active]
    m = np.max(subset)
    shifted = alpha * (subset - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(active)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = active[i] - (n - 1)
        if k >= 0:
            grad[:n - k] += wi * one_minus_h[k:]
            grad[k:] -= wi * h[:n - k]
        else:
            k2 = -k
            grad[k2:] += wi * one_minus_h[:n - k2]
            grad[:n - k2] -= wi * h[k2:]
    return grad, corr

def psm_optimize(h0, t_end, alpha_init=100.0, max_lags=8, top_frac=0.96):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_best = float(np.max(corr) * dx)
    h_best = h.copy()
    active = find_active_lags(corr, top_frac, max_lags)
    lr = 1.5
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    step_count = 0
    alpha = alpha_init
    while time.time() < t_end:
        alpha = alpha_init + 50.0 * step_count
        grad, corr = smooth_max_grad(h, active, alpha)
        grad_c5 = dx * grad
        best_c5_trial = c5_best
        best_h_trial = None
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            v = 0.9 * v - lr * factor
            h_new = h - v * grad_c5
            h_new = project_to_box_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5_trial:
                best_c5_trial = c5_new
                best_h_trial = h_new.copy()
                best_v = v.copy()
        if best_h_trial is not None:
            h = best_h_trial
            v = best_v
            c5_best = best_c5_trial
            h_best = h.copy()
            stall = 0
            step_count += 1
            lr = min(lr * 1.01, 5.0)
        else:
            v = np.zeros(n)
            lr *= 0.6
            stall += 1
            if stall > 80 or lr < 1e-8:
                break
        if step_count % 40 == 0 and step_count > 0:
            active = find_active_lags(corr, top_frac, max_lags)
    return h_best, c5_best

def psm_no_momentum(h0, t_end, alpha_init=200.0, max_lags=6, top_frac=0.97):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_best = float(np.max(corr) * dx)
    h_best = h.copy()
    active = find_active_lags(corr, top_frac, max_lags)
    lr = 2.0
    stall = 0
    step_count = 0
    alpha = alpha_init
    while time.time() < t_end:
        grad, corr = smooth_max_grad(h, active, alpha)
        grad_c5 = dx * grad
        best_c5_trial = c5_best
        best_h_trial = None
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03]:
            h_new = h - lr * factor * grad_c5
            h_new = project_to_box_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5_trial:
                best_c5_trial = c5_new
                best_h_trial = h_new.copy()
        if best_h_trial is not None:
            h = best_h_trial
            c5_best = best_c5_trial
            h_best = h.copy()
            stall = 0
            step_count += 1
            lr = min(lr * 1.015, 8.0)
        else:
            lr *= 0.65
            stall += 1
            if stall > 100 or lr < 1e-8:
                break
        if step_count % 30 == 0 and step_count > 0:
            active = find_active_lags(corr, top_frac, max_lags)
        alpha = alpha_init + 30.0 * step_count
    return h_best, c5_best

def binary_push_optimize(h0, t_end, n_push=0.03):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    c5_best = compute_c5(h)
    h_best = h.copy()
    strength = n_push
    for it in range(200):
        if time.time() > t_end:
            break
        h_push = h + strength * (2.0 * h - 1.0)
        h_push = np.clip(h_push, 0.0, 1.0)
        h_push = project_to_box_sum(h_push, target_sum)
        c5_push = compute_c5(h_push)
        if c5_push < c5_best - 1e-12:
            h = h_push
            c5_best = c5_push
            h_best = h.copy()
            strength *= 0.85
        else:
            strength *= 0.5
            if strength < 1e-5:
                break
    return h_best, c5_best

def local_search_coords(h0, t_end, n_deltas=7):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    c5_best = compute_c5(h)
    h_best = h.copy()
    if n_deltas <= 2:
        deltas = np.array([-0.01, 0.0])
    elif n_deltas <= 4:
        deltas = np.array([-0.02, -0.005, 0.0, 0.005, 0.02])
    else:
        deltas = np.array([-0.03, -0.01, -0.003, 0.0, 0.003, 0.01, 0.03])
    n_deltas_actual = len(deltas)
    rounds = 0
    while time.time() < t_end and rounds < 50:
        improved = False
        for j in range(n):
            if time.time() > t_end:
                break
            best_c5_j = c5_best
            best_delta = 0.0
            for d in deltas:
                if d == 0.0:
                    continue
                nv = h[j] + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h_trial = h.copy()
                h_trial[j] = nv
                h_trial = project_to_box_sum(h_trial, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5_j - 1e-13:
                    best_c5_j = c5_trial
                    best_delta = d
            if best_delta != 0.0:
                h[j] = h[j] + best_delta
                h = project_to_box_sum(h, target_sum)
                c5_best = best_c5_j
                h_best = h.copy()
                improved = True
        rounds += 1
        if not improved:
            break
    return h_best, c5_best

def pairwise_local(h0, t_end, rng, n_pairs=500):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    c5_best = compute_c5(h)
    h_best = h.copy()
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for swap_frac in [0.1, 0.3, 0.5, 0.7]:
            transfer = swap_frac * (h[i] - h[j])
            if abs(transfer) < 1e-10:
                continue
            h_trial = h.copy()
            h_trial[i] = h[i] - transfer
            h_trial[j] = h[j] + transfer
            if h_trial[i] < -1e-10 or h_trial[j] < -1e-10 or h_trial[i] > 1.0 + 1e-10 or h_trial[j] > 1.0 + 1e-10:
                continue
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_to_box_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                h = h_trial
                c5_best = c5_trial
                h_best = h.copy()
                break
    return h_best, c5_best

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
```
