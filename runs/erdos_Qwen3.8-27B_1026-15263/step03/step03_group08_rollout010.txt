    """Smooth max of correlations + penalty for sum constraint."""

    corr = compute_corr(h)
    m = np.max(corr)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, penalty, target_sum):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m + np.log(s) / alpha + penalty * (np.sum(h) - target_sum) ** 2
    
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-12:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    
    g += 2.0 * penalty * (np.sum(h) - target_sum)
    return obj, g

def lbfgs_pass(h0, n, target_sum, alpha, penalty, maxiter=500):
    def fun(x):
        v, _ = smooth_max_and_grad(x, alpha, penalty, target_sum)
        return v
    def grad(x):
        _, g = smooth_max_and_grad(x, alpha, penalty, target_sum)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def nesterov_topk(h0, n, target_sum, C_target, n_iters, top_k=10):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    mu = 0.7
    
    for it in range(n_iters):
        h_lk = np.clip(h + mu * v, 0.0, 1.0)
        h_lk = project_to_sum(h_lk, target_sum)
        
        corr = compute_corr(h_lk)
        top_idx = np.argsort(corr)[-top_k:]
        
        g = np.zeros(n)
        for idx in top_idx:
            l = idx - (n - 1)
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += (1.0 - h_lk[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= h_lk[lo2 + l:hi2 + l]
        
        # Project gradient
        g -= np.mean(g)
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        
        v = mu * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        cur_c5 = compute_c5(h)
        if c5_new <= cur_c5 + 1e-8:
            h = h_new
            if c5_new < best_c5 - 1e-7:
                lr = min(lr * 1.03, 0.15)
        else:
            lr *= 0.8
            v *= 0.5
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)

                if c5_new < c5_cur - 1e-12:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

            ambiguous = np.where((h > 0.1) & (h < 0.9))[0]
    return h, c5_cur

def targeted_pair_swap(h0, n, target_sum, C_target, rng, n_swaps=500):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    C_corr_target = C_target / dx
    offending = np.where(corr > C_corr_target)[0]
    for _ in range(n_swaps):
        if len(offending) == 0:
            break
        k = offending[rng.randint(len(offending))]
        l = k - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo >= hi and lo2 >= hi2:
            continue
        idx1_pool = list(range(lo, hi)) + list(range(lo2, hi2))
        idx1 = idx1_pool[rng.randint(len(idx1_pool))]
        idx_pool = list(range(n))
        idx2 = idx_pool[rng.randint(n)]
        if idx1 == idx2:
            continue
        h_new = h.copy()
        delta = 0.02 * rng.choice([-1, 1])
        h_new[idx1] = np.clip(h_new[idx1] + delta, 0.0, 1.0)
        h_new[idx2] = np.clip(h_new[idx2] - delta, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-10:
            h = h_new
            c5_cur = c5_new
            corr = compute_corr(h)
            C_corr_target = c5_cur / dx
            offending = np.where(corr > C_corr_target)[0]
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
    else:
        best_n = 480
        x = np.linspace(0, 2, best_n, endpoint=False)
        best_h = 0.5 + 0.35 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)

    best_n = len(best_h)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    n = best_n
    ts = n / 2.0

    alphas = [30, 100, 300, 800, 2000, 5000, 10000, 20000]
    for alpha in alphas:
        if time_left() < 35:
            break
        penalty = 500.0
        maxiter = min(250, max(40, int(time_left() * 0.25)))
        h_opt = lbfgs_pass(best_h, n, ts, alpha, penalty, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 80:
        for top_k in [5, 8, 12, 16]:
            if time_left() < 50:
                break
            n_iters = min(2500, max(500, int(time_left() * 1.2)))
            h_mom, c5_mom = nesterov_topk(best_h, n, ts, best_c5, n_iters, top_k, 0.04)
            if c5_mom < best_c5:
                best_c5 = c5_mom
                best_h = h_mom.copy()

    if time_left() > 60:
        for alpha in [5000, 15000, 30000, 50000]:
            if time_left() < 30:
                break
            h_opt = lbfgs_pass(best_h, n, ts, alpha, 1000.0, 150)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if time_left() > 50:
        h_sw, c5_sw = targeted_pair_swap(best_h, n, ts, best_c5, rng, 400)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    if time_left() > 40:
        h_bin, c5_bin = binarize_push(best_h, n, ts, rng, 300)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    if time_left() > 30:
        steps = [0.015, 0.008, 0.004, 0.002, 0.001, -0.015, -0.008, -0.004, -0.002, -0.001]
        max_rnd = min(40, max(5, int(time_left() // 5)))
        h_cd, c5_cd = coordinate_descent(best_h, n, ts, max_rnd, steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 40:
        h_mom2, c5_mom2 = nesterov_topk(best_h, n, ts, best_c5, 2000, 6, 0.02)
        if c5_mom2 < best_c5:
            best_c5 = c5_mom2
            best_h = h_mom2.copy()

    if time_left() > 30:
        alpha_final = 50000.0
        h_opt = lbfgs_pass(best_h, n, ts, alpha_final, 2000.0, 100)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 25:
        steps_fine = [0.005, 0.002, 0.001, 0.0005, -0.005, -0.002, -0.001, -0.0005]
        max_rnd_f = min(60, max(10, int(time_left() // 3)))
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, ts, max_rnd_f, steps_fine)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    if time_left() > 20:
        h_mom3, c5_mom3 = nesterov_topk(best_h, n, ts, best_c5, 1500, 8, 0.01)
        if c5_mom3 < best_c5:
            best_c5 = c5_mom3
            best_h = h_mom3.copy()

    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
