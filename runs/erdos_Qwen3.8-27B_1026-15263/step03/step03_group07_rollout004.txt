from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_top_lags(h, top_k=15):

    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:][::-1]

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_active(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = corr[active_idx]
    m = ac.max()
    shifted = np.clip(alpha * (ac - m), -500.0, 500.0)
    ev = np.exp(shifted)
    s = ev.sum()
    w = ev / s
    val = m + np.log(s / len(ev)) / alpha
    return val, w, corr

def grad_smooth_max_active(h, w, active_idx):
    n = len(h)
    g = np.zeros(n)
    for i, m_idx in enumerate(active_idx):
        wi = w[i]
        if wi < 1e-13:
            continue
        k = m_idx - (n - 1)

            nr = n - k

        else:

            nr = n - k2
            if nr > 0:
                g[k2:] += wi * (1.0 - h[:nr])
                g[:nr] -= wi * h[k2:]

def optimize_lbgfs(h0, n, target_sum, alpha, active_idx, maxiter, tol):

        h = np.clip(x, 0, 1)
        val, w, _ = smooth_max_active(h, active_idx, alpha)
        g = grad_smooth_max_active(h, w, active_idx)
        # Add penalty for sum constraint
        sum_diff = np.sum(h) - target_sum
        val += 1e6 * sum_diff**2
        g += 2e6 * sum_diff * np.ones(n)
        return val, g
    
    res = minimize(obj_and_grad, h0.copy(), jac=True, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-10})
    h_out = np.clip(res.x, 0, 1)
    h_out = project_to_sum(h_out, target_sum)
    return h_out

    h = h0.copy()

    for rnd in range(max_rounds):

        order = rng.permutation(n)

                c5_cur = best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_top_lags(h, top_k=15):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:][::-1]
    return idx, corr[idx]

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_val_and_weights(h, active_idx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = corr[active_idx]
    m = ac.max()
    shifted = np.clip(alpha * (ac - m), -500.0, 500.0)
    ev = np.exp(shifted)
    s = ev.sum()
    w = ev / s
    val = m + np.log(s / len(ev)) / alpha
    return val, w

def grad_sm_active(h, w, active_idx):
    n = len(h)
    g = np.zeros(n)
    for i in range(len(active_idx)):
        wi = w[i]
        if wi < 1e-13:
            continue
        k = int(active_idx[i]) - (n - 1)
        if k >= 0:
            g[k:] += wi * (1.0 - h[:n - k])
            g[:n - k] -= wi * h[k:]
        else:
            k2 = -k
            g[:n - k2] += wi * (1.0 - h[k2:])
            g[k2:] -= wi * h[:n - k2]
    return g

def slsqp_active(h0, n, target_sum, alpha, active_idx, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = sm_val_and_weights(h, active_idx, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, w = sm_val_and_weights(h, active_idx, alpha)
        return grad_sm_active(h, w, active_idx)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbgfs_active(h0, n, target_sum, alpha, active_idx, maxiter):
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        v, w = sm_val_and_weights(h, active_idx, alpha)
        g = grad_sm_active(h, w, active_idx)
        sd = np.sum(h) - target_sum
        v += 1e8 * sd * sd
        g += 2e8 * sd
        return v, g
    try:
        res = minimize(obj_grad, h0.copy(), jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_descent_binary(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_c5 = c5_cur
            best_v = h[i]
            cur_v = h[i]
            for frac in [0.2, 0.5, 0.8, 1.0]:
                for tgt in [0.0, 1.0]:
                    nv = cur_v + (tgt - cur_v) * frac
                    nv = max(0.0, min(1.0, nv))
                    if abs(nv - cur_v) < 1e-10:
                        continue
                    h_new = h.copy()
                    h_new[i] = nv
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-15:
                        best_c5 = c5_new
                        best_v = nv
            if best_c5 < c5_cur - 1e-15:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def pair_swap(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(max(50, n // 3)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = (h[i] - h[j]) * 0.5
            if abs(d) < 1e-8:
                continue
            h_new = h.copy()
            h_new[i] = max(0.0, min(1.0, h[i] - d))
            h_new[j] = max(0.0, min(1.0, h[j] + d))
            sd = np.sum(h_new) - target_sum
            if abs(sd) > 1e-12:
                continue
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                improved = True
        if not improved:
            break
    return h, c5_cur

def perturb_random(h0, n, target_sum, rng, sigma):

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

            if tl() < 60:
                break
            sigma = 0.1 / (1 + trial * 0.5)
            h_pert = perturb_random(h_cur, n, ts, rng, sigma)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.005:
                continue
            for alpha in [200, 800, 3000]:
                if tl() < 30:
                    break
                tk = 12
                active_idx, _ = get_top_lags(h_pert, tk)
                h_o = slsqp_active(h_pert, n, ts, alpha, active_idx, 300)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_pert = h_o
            if tl() < 40:
                break

    # Phase 6: Higher resolution
    if tl() > 250:
        for n_hi in [n * 2, n * 2 + 120]:
            if tl() < 150:
                break
            if n_hi % 2 != 0:
                n_hi += 1
            if n_hi > 2000:
                continue
            ts_hi = n_hi / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = project_to_sum(h_hi, ts_hi)
            for alpha in [200, 500, 1500, 4000]:
                if tl() < 40:
                    break
                tk = min(20, n_hi // 50)
                active_idx, _ = get_top_lags(h_hi, tk)
                h_hi_opt = slsqp_active(h_hi, n_hi, ts_hi, alpha, active_idx, 350)
                c5_hi = compute_c5(h_hi_opt)
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = h_hi_opt.copy()
                    best_n = n_hi
                h_hi = h_hi_opt
            if tl() > 60:
                h_cdh, c5_cdh = coord_descent_binary(h_hi, n_hi, ts_hi, 30, rng)
                if c5_cdh < best_c5:
                    best_c5 = c5_cdh
                    best_h = h_cdh.copy()
                    best_n = n_hi

    # Phase 7: Final polish
    if best_h is not None and tl() > 40:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        for alpha in [5000, 15000]:
            if tl() < 20:
                break
            tk = 8
            active_idx, _ = get_top_lags(h_f, tk)
            h_pol = slsqp_active(h_f, n_f, ts_f, alpha, active_idx, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f
            h_f = h_pol
        if tl() > 30:
            h_cdf, c5_cdf = coord_descent_binary(h_f, n_f, ts_f, 40, rng)
            if c5_cdf < best_c5:
                best_c5 = c5_cdf
                best_h = h_cdf.copy()
                best_n = n_f
            h_f = h_cdf
        if tl() > 20:
            h_psf, c5_psf = pair_swap(h_f, n_f, ts_f, 20, rng)
            if c5_psf < best_c5:
                best_c5 = c5_psf
                best_h = h_psf.copy()
                best_n = n_f

    # Final cleanup
    if best_h is not None:
        n_fin = len(best_h)
        ts_fin = n_fin / 2.0
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, ts_fin)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_fin)
