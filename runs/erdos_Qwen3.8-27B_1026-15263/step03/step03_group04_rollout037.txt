def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):

            break

            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Clip to avoid overflow
    exp_v = np.exp(np.clip(alpha * (corr - m), -500, 0))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def active_set_grad(h, top_m=2):
    """Compute gradient using only the top-m active shifts (exact max subgradient)."""
    n = len(h)
    corr = compute_corr(h)
    # Get the top m shifts
    top_indices = np.argsort(corr)[::-1][:top_m]
    grad = np.zeros(n)

        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    grad /= len(top_indices)
    return grad

    Hybrid optimizer: uses smooth max for the objective but can use
    active set gradient for direction when alpha is high.

    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        return obj
    
    def jac(x):
        _, g = smooth_max_obj_and_grad(x, alpha)
        return g
    
    def fun_pen(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * 1e6 * s_dev * s_dev
    
    def jac_pen(x):
        _, g = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return g + 1e6 * s_dev * np.ones(n)

        pass
    
    return project_to_sum(h, target_sum)

    c5_cur = compute_c5(h)

        # Use top 3 shifts
        top_k = np.argsort(corr)[::-1][:3]
        best_improve = 0

        best_c5_new = c5_cur

            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:

            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]

            for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:

                    best_c5_new = c5_new
        
        if best_h_new is h:
            break
        h = best_h_new
        c5_cur = best_c5_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(np.clip(alpha * (corr - m), -500, 0))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-12
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def lbfgs_smooth(h0, n, target_sum, alpha, maxiter=300):
    """L-BFGS-B with penalty for sum constraint."""
    h = h0.copy()
    pen = 1e6
    
    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * pen * s_dev * s_dev
    
    def jac(x):
        _, g = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return g + pen * s_dev * np.ones(n)
    
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h = res.x
    except Exception:
        pass
    return project_to_sum(h, target_sum)

def slsqp_smooth(h0, n, target_sum, alpha, maxiter=300):
    """SLSQP with equality constraint (no penalty needed)."""
    h = h0.copy()
    
    def fun(x):
        return smooth_max_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return smooth_max_obj_and_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

def exact_max_polish(h, n, target_sum, max_rounds=500):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:4]
        best_improve = 0
        best_h_new = h
        best_c5_new = c5_cur
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-15:
                    best_improve = c5_cur - c5_new
                    best_h_new = h_new.copy()
                    best_c5_new = c5_new
        if best_h_new is h:
            break
        h = best_h_new
        c5_cur = best_c5_new
    return h, c5_cur

    h = h0.copy()
    v = np.zeros(n)
    lr = lr0
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    stall = 0

        _, grad = smooth_max_obj_and_grad(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        v = beta * v + (1 - beta) * (-grad / gn)
        h_new = h + lr * v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:

            stall = 0
        else:
            stall += 1

        c5_cur = c5_new
        if stall > 100:
            v = np.zeros(n)
            lr *= 0.5
            stall = 0
            if lr < 1e-7:
                break
    return best_h, best_c5
