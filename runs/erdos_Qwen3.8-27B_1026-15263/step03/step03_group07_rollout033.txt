```python
def compute_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s) / alpha) * dx  # Note: dx factor!
    
    # Gradient computation (vectorized)
    # w has length 2n-1, w[m] corresponds to lag k = m - (n-1)
    g = np.zeros(n)
    
    for j in range(n):
        # Positive lags k >= 0: correlate index n-1+k
        # Contribution: w[n-1+k] * [(1-h[j+k]) if j+k<n - h[j-k] if j>=k]
        # Negative lags k < 0, let k2=-k: correlate index n-1-k2
        # Contribution: w[n-1-k2] * [(1-h[j-k2]) if j>=k2 - h[j+k2] if j+k2<n]

        if k_max_pos >= 0:
            ks = np.arange(0, k_max_pos + 1)
            g[j] += np.dot(w[n-1+ks], 1.0 - h[j+ks])
        
        k_max_neg = j  # max k such that j-k >= 0, i.e., k <= j
        if k_max_neg >= 0:
            ks = np.arange(0, k_max_neg + 1)
            g[j] -= np.dot(w[n-1+ks], h[j-ks])
```

```python
def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s) / alpha) * dx
    
    # Gradient (vectorized loop)
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    w_pos = w[n-1:]  # wp[k] = w[n-1+k], k=0..n-1
    w_neg = w[:n-1][::-1]  # wn[k] = w[n-1-k], k=1..n-1 -> wn[k-1] = w[n-1-k]

        if j + 1 > 0:
            g[j] -= np.dot(w_pos[:j+1], h[j::-1] if j > 0 else h[:1])
```

The main challenge is that SLSQP with 719 variables and 1 equality constraint + 719 bound constraints is expensive per iteration. Each iteration requires evaluating the objective and gradient, plus solving a QP subproblem.

An alternative is to use the **augmented Lagrangian** method:
- Minimize L(h, λ) = smooth_max(h) + λ*(sum(h) - n/2) + (μ/2)*(sum(h) - n/2)²
- subject to 0 ≤ h ≤ 1
- Update λ = λ + μ*(sum(h) - n/2) after each subproblem
- This can be solved with L-BFGS-B (which handles bounds efficiently)

This might be faster than SLSQP because L-BFGS-B is very efficient for bound-constrained problems.

```python
def augmented_lagrangian_optimize(h0, n, target_sum, alpha, max_outer=50, max_inner=200):
    mu = 1.0
    lam = 0.0
    h = h0.copy()
    
    for outer in range(max_outer):
        def objective(x):
            h_clipped = np.clip(x, 0.0, 1.0)
            sm, _ = smooth_max_obj_and_grad_raw(h_clipped, alpha)
            constraint = np.sum(h_clipped) - target_sum
            return sm + lam * constraint + 0.5 * mu * constraint**2
        
        def gradient(x):
            h_clipped = np.clip(x, 0.0, 1.0)
            _, g = smooth_max_obj_and_grad_raw(h_clipped, alpha)
            constraint = np.sum(h_clipped) - target_sum
            return g + (lam + mu * constraint) * np.ones(n)
        
        res = minimize(objective, h, jac=gradient, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': max_inner, 'ftol': 1e-12, 'gtol': 1e-8})
        h = np.clip(res.x, 0.0, 1.0)
```

```python
        if abs(diff) < 1e-14:
```

```python
            h[mask] += diff / np.sum(mask)

def smooth_max_eval(h, alpha, active_idx=None):

    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_idx is not None:
        vals = corr[active_idx]
    else:
        vals = corr

    exp_vals = np.exp(shifted)

    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, w, corr, m

def smooth_max_grad(h, alpha, active_idx=None):
    n = len(h)
    dx = 2.0 / n
    sm_val, w_local, corr, m = smooth_max_eval(h, alpha, active_idx)
    if active_idx is not None:
        w_full = np.zeros(2 * n - 1)
        w_full[active_idx] = w_local
    else:
        w_full = w_local
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    wp = w_full[n - 1:]
    wn_arr = w_full[:n - 1][::-1]
    for j in range(n):
        t1 = n - j
        if t1 > 0:
            g[j] += np.dot(wp[:t1], one_minus_h[j:j + t1])
        if j >= 0:
            g[j] -= np.dot(wp[:j + 1], h[j::-1] if j > 0 else h[:1])
        if j > 0:
            g[j] += np.dot(wn_arr[:j], one_minus_h[j - 1::-1])
        t4 = n - 1 - j
        if t4 > 0:
            g[j] -= np.dot(wn_arr[:t4], h[j + 1:j + 1 + t4])
    return sm_val * dx, g * dx

def lbfgs_augmented(h0, n, target_sum, alpha, max_outer=40, max_inner=300, active_idx=None):
    mu = 1.0
    lam = 0.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for outer in range(max_outer):
        def obj(x):
            hc = np.clip(x, 0.0, 1.0)
            sv, _ = smooth_max_grad(hc, alpha, active_idx)
            c = np.sum(hc) - target_sum
            return sv + lam * c + 0.5 * mu * c * c
        def grad(x):
            hc = np.clip(x, 0.0, 1.0)
            _, g = smooth_max_grad(hc, alpha, active_idx)
            c = np.sum(hc) - target_sum
            return g + (lam + mu * c) * np.ones(n)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_inner, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            break
        c = np.sum(h_new) - target_sum
        if abs(c) < 1e-10:
            h = h_new
        else:
            h = project_to_sum(h_new, target_sum)
        cv = compute_c5(h)
        if cv < best_val:
            best_val = cv
            best_h = h.copy()
        lam = lam + mu * c
        if abs(lam) > 50.0:
            mu *= 2.0
            lam = np.clip(lam, -50.0, 50.0)
        if abs(c) < 1e-12:
            break
    return best_h, best_val

def slsqp_smooth(h0, n, target_sum, alpha, maxiter=200, active_idx=None):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        sv, _ = smooth_max_grad(hc, alpha, active_idx)
        return sv
    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(hc, alpha, active_idx)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def get_active_lags(h, top_k=8):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    active = np.where(corr >= 0.95 * m)[0]
    if len(active) > top_k:
        idx = np.argsort(corr[active])[::-1][:top_k]
        active = active[idx]
    return sorted(active.tolist())

def fourier_perturb(h, n, mode, freq, amp, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    if mode == 'sin':

    else:
        pert = amp * np.cos(2.0 * np.pi * freq * x / 2.0)
    pert = pert + rng.randn(n) * amp * 0.1
    h_new = h + pert
    return project_to_sum(h_new, n / 2.0)

def coordinate_refine(h0, n, target_sum, max_rounds=50, rng=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if rng is None:
        rng = np.random.RandomState(0)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_d = 0.0
            best_v = c5_cur
            for d in [-0.003, -0.001, 0.001, 0.003]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)

                h = project_to_sum(h, target_sum)
                c5_cur = best_v
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_push_refine(h0, n, target_sum, n_iter=50):
    h = h0.copy()

    strength = 0.05

        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f
        h_f = h_pol
        if time_left() > 30:
            h_cd, c5_cd = coordinate_refine(h_f, n_f, ts_f, max_rounds=40, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_f
            h_f = h_cd
        print("Phase8: c5={:.6f}".format(compute_c5(h_f)))

    # Final: ensure feasibility
    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
```
