import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(h, k_idx):
    """Gradient of correlation at index k_idx (0-indexed in full mode)."""
    n = len(h)
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)

        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = compute_all_corr(h)
    m = np.max(corr)

def corr_gradient(h, k_idx):
    n = len(h)
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
