    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)

    s = np.sum(exp_vals)

        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v * (2.0 / n)

        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g * (2.0 / n)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

        lo_bound = max(-h[i], -(1.0 - h[j]), -search_range)
        if hi_bound - lo_bound < 1e-7:
            continue
        
        hi_i = h[i]
        hi_j = h[j]

        a = lo_bound
        b = hi_bound
        c = b - gr * (b - a)

        h[j] = hi_j - c
        corr_c = np.correlate(h, 1.0 - h, mode='full')

                d = c
                fd = fc

                h[i] = hi_i + c
                h[j] = hi_j - c
                corr_c = np.correlate(h, 1.0 - h, mode='full')
                fc = np.max(corr_c)
            else:

                c = d
                fc = fd
                d = a + gr * (b - a)
                h[i] = hi_i + d
                h[j] = hi_j - d
                corr_d = np.correlate(h, 1.0 - h, mode='full')
                fd = np.max(corr_d)
        
        t_opt = (a + b) / 2.0
        h[i] = hi_i + t_opt
        h[j] = hi_j - t_opt
        c5_new = compute_c5(h)
        
        if c5_new < c5_cur - 1e-14:
            h = project_to_sum(h, target_sum)
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] = hi_i
            h[j] = hi_j
    
    return best_h, best_c5
