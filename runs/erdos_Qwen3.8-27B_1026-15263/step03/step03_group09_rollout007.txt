        # (1-h)[j+shift] valid when 0<=j+shift<n => -shift <= j < n-shift
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            g[lo1:hi1] += wval * one_h[lo1 + shift:hi1 + shift]
        # -h[j-shift] valid when 0<=j-shift<n => shift <= j < n+shift
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            g[lo2:hi2] -= wval * h[lo2 - shift:hi2 - shift]
    
    return sm_val, g

def projected_gradient_descent(h0, n, target_sum, alpha, n_iters, lr_init, lr_decay=0.995):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iters):
        _, g = smooth_max_and_grad(h, alpha)
        momentum = beta * momentum + (1 - beta) * g
        h_new = h - lr * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        lr *= lr_decay
        if lr < 1e-6:
            break
    
    return best_h, best_c5

def perturb_and_optimize(h0, n, target_sum, rng, n_iters, rotate_iter):

    best_c5 = compute_c5(h)
    best_h = h.copy()

    freqs = np.arange(1, n_frequencies + 1)

        noise = np.zeros(n)
        for f in freqs:

            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) * 2.0 / n
            noise += amp * np.sin(np.pi * f * x + phase)
            noise += amp * rng.uniform(-1, 1) * np.cos(np.pi * f * x + phase)

        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_to_sum(h_pert, target_sum)
        
        # Optimize from perturbed point
        h_opt, c5_opt = projected_gradient_descent(h_pert, n, target_sum, 200, n_iters, 0.05)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        
        h = best_h
    
    return best_h, best_c5

def coord_descent_fine(h0, n, target_sum, max_rounds, rng):

    h = h0.copy()

            candidates = candidates[(candidates >= 0) & (candidates <= 1)]

            best_nv = hi

                h_new = h.copy()
                h_new[i] = nv

                h[i] = best_nv
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

        n_f = state['n']
        ts_f = n_f / 2.0
        h_f, _ = coord_descent_fine(state['h'], n_f, ts_f,
                                     max(5, min(80, int(_remaining(t_start, budget_s) / 3))),
                                     rng,
                                     np.array([-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]))
        _try_update(state, h_f, n_f)
    if state['h'] is not None and _remaining(t_start, budget_s) > 10:
        h_f2 = slsqp_optimize(state['h'], state['n'], state['n'] / 2.0, 4000,
                               min(200, int(_remaining(t_start, budget_s) * 0.8)))
        _try_update(state, h_f2, state['n'])

    best_h = state['h']
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
