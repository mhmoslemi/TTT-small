import time

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)

    g = np.zeros(n)

        wj = w[L + n - 1]

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(ht, alpha)
        return v

    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(ht, alpha)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_descent_ls(h0, n, target_sum, n_rounds, rng):

    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.5)
            hi = min(1.0, h[i] + 0.5)
            if hi - lo < 1e-8:
                continue

                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)

        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h_cur = h_pg

    n2 = n1 * 2
    if time.time() - t_start < budget_s * 0.72 and n2 <= 1440:
        h_interp = upsample(best_h, n2)
        ts2 = n2 / 2.0
        h_cur2 = h_interp
        c5_interp = compute_c5(h_cur2)

        alphas2 = [100.0, 200.0, 400.0, 800.0, 1600.0, 3200.0]
        for alpha in alphas2:
            if time.time() - t_start > budget_s * 0.72:
                break
            maxiter = max(80, min(400, int((budget_s * 0.72 - (time.time() - t_start)) * 3)))
            h_opt2 = slsqp_step(h_cur2, n2, ts2, alpha, maxiter)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2
            h_cur2 = h_opt2

        if time.time() - t_start < budget_s * 0.78:
            n_cd3 = min(12, max(4, int((budget_s * 0.78 - (time.time() - t_start)) // 10)))
            h_cd3, c5_cd3 = coord_descent_ls(h_cur2, n2, ts2, n_cd3, rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n2
                h_cur2 = h_cd3

        if time.time() - t_start < budget_s * 0.82:
            deltas2 = [-0.008, -0.004, -0.002, -0.001, 0.001, 0.002, 0.004, 0.008]
            h_cd4, c5_cd4 = coord_descent_fixed(h_cur2, n2, ts2, 25, rng, deltas2)
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()
                best_n = n2
                h_cur2 = h_cd4

        if time.time() - t_start < budget_s * 0.85:
            h_pc, c5_pc = pair_coord_descent(h_cur2, n2, ts2, 15, rng, 0.03)
            if c5_pc < best_c5:
                best_c5 = c5_pc
                best_h = h_pc.copy()
                best_n = n2

    n3 = n1 * 2
    if time.time() - t_start < budget_s * 0.60 and n3 <= 1440:
        for pert_round in range(4):
            if time.time() - t_start > budget_s * 0.60:
                break
            pert_amt = 0.04 / (pert_round + 1)
            h_pert = best_h + rng.normal(0, pert_amt, size=best_n)
            h_pert = project_to_sum(h_pert, best_n / 2.0)
            for alpha in [400.0, 1000.0, 2000.0]:
                if time.time() - t_start > budget_s * 0.60:
                    break
                h_opt = slsqp_step(h_pert, best_n, best_n / 2.0, alpha, 200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_pert = h_opt
            if time.time() - t_start > budget_s * 0.60:
                break
            h_cd, c5_cd = coord_descent_fixed(h_pert, best_n, best_n / 2.0, 10, rng,
                                              [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005])
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [2000.0, 5000.0, 10000.0]:
        if time.time() - t_start > budget_s * 0.95:
            break
        h_pol = slsqp_step(best_h, n_f, ts_f, alpha, 400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if time.time() - t_start < budget_s * 0.98:
        h_cd_final, c5_final = coord_descent_ls(best_h, n_f, ts_f, 8, rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cd_final

    if time.time() - t_start < budget_s * 0.995:
        deltas_final = [-0.003, -0.001, -0.0005, 0.0005, 0.001, 0.003]
        h_cd_f2, c5_f2 = coord_descent_fixed(best_h, n_f, ts_f, 30, rng, deltas_final)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_cd_f2

    if time.time() - t_start < budget_s * 0.995:
        h_pc_f, c5_pc_f = pair_coord_descent(best_h, n_f, ts_f, 20, rng, 0.02)
        if c5_pc_f < best_c5:
            best_c5 = c5_pc_f
            best_h = h_pc_f

    final_n = len(best_h)
    best_h = project_to_sum(best_h, final_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
