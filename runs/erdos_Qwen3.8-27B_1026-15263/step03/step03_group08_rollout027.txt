import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-13:
            break
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        free = (proj > 1e-14) & (proj < 1.0 - 1e-14)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def get_active_lags(h, m=50):
    """Get the top-m lags by correlation value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-m:]
    # Convert to lag values
    n = len(h)
    lags = idx - (n - 1)
    return lags, np.max(corr)

def smooth_max_and_grad(h, lags, alpha):
    """Compute smooth max over given lags and its gradient."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation values for active lags
    corr_vals = np.zeros(len(lags))
    for idx, l in enumerate(lags):
        if l >= 0:
            corr_vals[idx] = np.dot(h[l:], 1.0 - h[:n-l])
        else:
            k = -l
            corr_vals[idx] = np.dot(h[:n-k], 1.0 - h[k:])
    
    m_val = np.max(corr_vals)
    shifted = alpha * (corr_vals - m_val)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m_val + np.log(s) / alpha) * dx
    
    # Gradient
    g = np.zeros(n)
    for idx, l in enumerate(lags):
        wl = w[idx]
        if wl < 1e-14:
            continue
        if l >= 0:
            g[l:] += wl * (1.0 - h[:n-l])
            g[:n-l] -= wl * h[l:]
        else:
            k = -l
            g[:n-k] += wl * (1.0 - h[k:])
            g[k:] -= wl * h[:n-k]
    g *= dx
    
    return sm_val, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-14:
            break
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def get_active_lags(h, m=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-m:]
    lags = idx - (n - 1)
    return lags

def smooth_max_and_grad(h, lags, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_vals = np.zeros(len(lags))
    for idx, l in enumerate(lags):
        if l >= 0:
            corr_vals[idx] = np.dot(h[l:], 1.0 - h[:n - l])
        else:
            k = -l
            corr_vals[idx] = np.dot(h[:n - k], 1.0 - h[k:])
    m_val = np.max(corr_vals)
    shifted = alpha * (corr_vals - m_val)
    shifted = np.clip(shifted, -400.0, 400.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m_val + np.log(s) / alpha) * dx
    g = np.zeros(n)
    for idx, l in enumerate(lags):
        wl = w[idx]
        if wl < 1e-14:
            continue
        if l >= 0:
            g[l:] += wl * (1.0 - h[:n - l])
            g[:n - l] -= wl * h[l:]
        else:
            k = -l
            g[:n - k] += wl * (1.0 - h[k:])
            g[k:] -= wl * h[:n - k]
    g *= dx
    return sm_val, g

    if best_h is None:
        n_init = 480
        x = np.linspace(0, 2, n_init, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2.0 * np.pi * x)
        best_h = project_box_sum(best_h, n_init / 2.0)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    target_sum = n / 2.0

    # Phase 1: Augmented L-BFGS-B with increasing alpha
    alphas = [50.0, 100.0, 300.0, 800.0, 2000.0, 5000.0, 10000.0, 20000.0]
    for alpha in alphas:
        if time_left() < 25:
            break
        h_new = augmented_lbfgs(best_h, n, target_sum, alpha, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Try perturbations and re-optimize
    if time_left() > 120:
        perturb_deltas = [0.03, 0.06, 0.1, 0.15]
        x_grid = np.linspace(0, 2, n, endpoint=False)
        for delta in perturb_deltas:
            if time_left() < 30:
                break
            h_pert = best_h + rng.uniform(-delta, delta, n)
            h_pert = project_box_sum(h_pert, target_sum)
            # Also try structured perturbations
            for freq in [2, 3, 5]:
                h_struct = best_h + delta * 0.5 * np.sin(2.0 * np.pi * freq * x_grid)
                h_struct = project_box_sum(h_struct, target_sum)
                for h_try in [h_pert, h_struct]:
                    if time_left() < 20:
                        break
                    h_opt = augmented_lbfgs(h_try, n, target_sum, 2000.0, maxiter=150)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt
                        h_pert = best_h + rng.uniform(-delta, delta, n)
                        h_pert = project_box_sum(h_pert, target_sum)

    # Phase 3: Nesterov momentum refinement
    if time_left() > 60:
        n_iters = min(3000, int(time_left() * 4))
        h_nest, c5_nest = nesterov_projected(best_h, n, target_sum, n_iters, 0.05, 5000.0, m=40)
        if c5_nest < best_c5:
            best_c5 = c5_nest
            best_h = h_nest

    # Phase 4: High-alpha polish
    if time_left() > 40:
        for alpha in [20000.0, 50000.0]:
            if time_left() < 20:
                break
            h_pol = augmented_lbfgs(best_h, n, target_sum, alpha, maxiter=150)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 5: Coordinate descent polishing
    if time_left() > 25:
        steps = [0.005, 0.002, 0.001, 0.0005, -0.005, -0.002, -0.001, -0.0005]
        max_rnd = max(5, min(30, int(time_left() // 2)))
        h_cd, c5_cd = coord_descent(best_h, n, target_sum, max_rnd, steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 6: Try higher resolution
    if time_left() > 150:
        for n2 in [600, 720, 960]:
            if time_left() < 80:
                break
            if n2 % 2 != 0:
                continue
            target2 = n2 / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, best_h)
            h2 = project_box_sum(h2, target2)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5 + 0.003:
                for alpha in [500.0, 2000.0, 5000.0, 10000.0, 20000.0]:
                    if time_left() < 30:
                        break
                    h2 = augmented_lbfgs(h2, n2, target2, alpha, maxiter=120)
                    c5_2 = compute_c5(h2)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        n = n2
                        target_sum = target2
                if time_left() > 40:
                    n_iters2 = min(2000, int(time_left() * 2))
                    h2, c5_2 = nesterov_projected(best_h, n, target_sum, n_iters2, 0.03, 5000.0, m=40)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2

    # Final
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
