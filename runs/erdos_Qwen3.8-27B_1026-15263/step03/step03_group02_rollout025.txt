import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target_ones, rng):

    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        perm = rng.permutation(len(ones_idx))[:excess]
        h_bin[ones_idx[perm]] = 0.0
    elif n_ones < target_ones:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        perm = rng.permutation(len(zeros_idx))[:deficit]
        h_bin[zeros_idx[perm]] = 1.0

def sa_binary(h_start, n, target_ones, t_end, rng):

    corr = np.correlate(h, 1.0 - h, mode="full")

        j = rng.randint(n)
        if h[i] == h[j]:

def subgradient_descent(h0, n, target_sum, t_end, rng):
    """Subgradient method for the max of correlations"""
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.01
    patience = 0
    
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        mask_max = corr >= max_val - 1e-10
        k_indices = np.where(mask_max)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for k_idx in k_indices:
            # corr[k_idx] = sum_i h[i] * (1-h[i+k]) where k = k_idx - n + 1
            k = k_idx - n + 1
            # d/dh[i] of sum_j h[j]*(1-h[j+k]) at the max lag
            # = sum over j where j and j+k are in [0,n):
            #   if i == j: (1-h[j+k])
            #   if i == j+k: -h[j]
            # Combined: for each valid j, grad[j] += (1-h[j+k]), grad[j+k] -= h[j]
            if k >= 0:
                valid_j = np.arange(n - k)
                grad[valid_j] += (1.0 - h[valid_j + k])
                grad[valid_j + k] -= h[valid_j]
            else:
                valid_j = np.arange(-k, n)
                grad[valid_j] += (1.0 - h[valid_j + k])
                grad[valid_j + k] -= h[valid_j]
        
        grad /= len(k_indices)
        
        # Normalized step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience % 10 == 0:
                lr *= 0.95
            if patience > 500:
                h = best_h.copy()
                patience = 0
                lr = max(lr * 0.8, 0.001)
        
        # Adaptive learning rate
        if patience % 50 == 0 and patience > 0:
            lr = max(lr * 0.9, 0.0005)
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    """SLSQP with smooth max objective"""
    def objective_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -400.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val + np.log(np.sum(exp_vals)) / alpha
        
        # Gradient
        n_corr = 2 * n - 1
        grad = np.zeros(n)
        for i in range(n):
            # d/dx[i] of corr[k] for each k
            # corr[k] = sum_j x[j]*(1-x[j+k])
            # d/dx[i] = sum over valid (j, j+k) pairs involving i
            #   if i==j: (1-x[j+k])
            #   if i==j+k: -x[j]
            # We need sum_k w[k] * d(corr[k])/dx[i]
            # 
            # For which k is i involved?
            # k ranges from 0 to 2n-2, corresponding to lag = k - n + 1
            # i is involved if 0 <= i < n and 0 <= i+lag < n (as the "first" index)
            #   or 0 <= i-lag < n and 0 <= i < n (as the "second" index)
            
            # As first index (j=i): lag must satisfy 0 <= i+lag < n, i.e., -i <= lag < n-i
            # As second index (j+k=i, j=i-lag): 0 <= i-lag < n and 0 <= lag... 
            #   lag can be positive or negative
            
            # Let's enumerate lags:
            # lag = k - n + 1, k = 0..2n-2, so lag = 1-n..n-1
            # i as first: 0 <= i+lag < n => -i <= lag <= n-1-i
            # i as second: 0 <= i-lag < n => i-n+1 <= lag <= i
            
            lags_first = np.arange(max(-i, 1-n), min(n-1-i, n-1) + 1)
            lags_second = np.arange(max(i-n+1, 1-n), min(i, n-1) + 1)
            
            for lag in lags_first:
                k_idx = lag + n - 1
                j2 = i + lag
                if 0 <= j2 < n:
                    grad[i] += w[k_idx] * (1.0 - x[j2])
            
            for lag in lags_second:
                k_idx = lag + n - 1
                j1 = i - lag
                if 0 <= j1 < n:
                    grad[i] += w[k_idx] * (-x[j1])
            
            # Wait, this is O(n) per i, making the total O(n²) per gradient evaluation.
            # For n=240, that's 57600 operations per gradient, which is fine.
        
        return obj, grad
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    def constraint_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective_and_grad, h0.copy(), method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
