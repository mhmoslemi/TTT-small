        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

        else:

        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0

    m_idx = np.arange(2 * n - 1)[None, :]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target_ones, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_ones:
        return h_bin
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def _smooth_obj(x):
    alpha = _alpha_global
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

def _smooth_grad(x):
    alpha = _alpha_global
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]

    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - x[fc]) * w[None, :]
    cs = valid_s * x[sc] * w[None, :]
    return np.sum(cf - cs, axis=1)

    _alpha_global = alpha

                       bounds=bounds, constraints=cons,
