def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n-1)
    sc = np.clip(second_idx, 0, n-1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def projected_grad_descent(h0, n, target, t_start, t_end, rng):

    h = project_to_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = 50.0
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    y = h.copy()
    
    while time.time() < t_end:
        y = y + momentum * (y - h)
        obj, grad = smooth_max_obj_grad(y, alpha)
        h_new = np.clip(y - lr * grad, 0, 1)
        h_new = project_to_sum(h_new, target)
        v = momentum * v + (y - h_new)
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        elapsed = time.time() - t_start
        total = t_end - t_start
        progress = min(1.0, elapsed / max(total, 1e-6))
        lr = 0.005 * (1 - 0.5 * progress)
        alpha = 50.0 + 500.0 * progress
    
    return best_h, best_c5
