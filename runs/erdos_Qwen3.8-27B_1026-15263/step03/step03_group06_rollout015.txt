    """One subgradient step on the exact max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    
    # Subgradient of R[k_star]
    k = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        m1 = j - k
        if 0 <= m1 < n:
            g += (1.0 - h[m1])
        m2 = j + k
        if 0 <= m2 < n:

    """
    If alpha > 0: use log-sum-exp on top-k values
    If alpha == 0: use exact top-k average
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if alpha > 0:
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -300.0, 300.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        val = m + np.log(s) / alpha
    else:
        # Exact top-k average
        topk_indices = np.argsort(corr)[-k:]
        w = np.zeros(len(corr))
        w[topk_indices] = 1.0 / k
        val = np.mean(corr[topk_indices])
    
    # Gradient (same computation as before)
    grad = np.zeros(n)

            js2 = np.arange(-k_shift, n)
            grad[js2] -= w[i] * h[js2 + k_shift]

    idx = np.argsort(corr)[-k_top:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k_top
    val = np.mean(corr[idx])
    grad = np.zeros(n)
    for i in idx:
        wi = w[i]
        k = i - (n - 1)
        if k >= 0:
            js = np.arange(k, n)
            grad[js] += wi * (1.0 - h[js - k])
            js2 = np.arange(0, n - k)
            grad[js2] -= wi * h[js2 + k]
        else:
            js = np.arange(0, n + k)
            grad[js] += wi * (1.0 - h[js - k])
            js2 = np.arange(-k, n)
            grad[js2] -= wi * h[js2 + k]
    return val, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol=1e-14):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        ht = project_to_sum(ht, target_sum)
        v, _ = smooth_max_obj_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        ht = project_to_sum(ht, target_sum)
        _, g = smooth_max_obj_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def topk_slsqp(h0, n, target_sum, k_top, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        ht = project_to_sum(ht, target_sum)
        v, _ = topk_avg_obj_grad(ht, k_top)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        ht = project_to_sum(ht, target_sum)
        _, g = topk_avg_obj_grad(ht, k_top)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def subgradient_momentum(h0, n, target_sum, n_iters, lr_init, momentum):
    h = h0.copy()
    h_prev = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    lr = lr_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k_shift = k_star - (n - 1)
        grad = np.zeros(n)
        if k_shift >= 0:
            js = np.arange(k_shift, n)
            grad[js] += (1.0 - h[js - k_shift])
            js2 = np.arange(0, n - k_shift)
            grad[js2] -= h[js2 + k_shift]
        else:
            js = np.arange(0, n + k_shift)
            grad[js] += (1.0 - h[js - k_shift])
            js2 = np.arange(-k_shift, n)
            grad[js2] -= h[js2 + k_shift]
        h_new = h - lr * grad + momentum * (h - h_prev)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.85
            h_new = h - lr * grad + momentum * (h - h_prev)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
        h_prev = h
        h = h_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
    return h_best, c5_best

def subgradient_multi(h0, n, target_sum, n_iters, lr_init, momentum, n_focus=3):
    h = h0.copy()
    h_prev = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    lr = lr_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_focus:]
        grad = np.zeros(n)
        for k_star in top_idx:
            k_shift = k_star - (n - 1)
            if k_shift >= 0:
                js = np.arange(k_shift, n)
                grad[js] += (1.0 - h[js - k_shift])
                js2 = np.arange(0, n - k_shift)
                grad[js2] -= h[js2 + k_shift]
            else:
                js = np.arange(0, n + k_shift)
                grad[js] += (1.0 - h[js - k_shift])
                js2 = np.arange(-k_shift, n)
                grad[js2] -= h[js2 + k_shift]
        grad /= n_focus
        h_new = h - lr * grad + momentum * (h - h_prev)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.8
            h_new = h - lr * grad + momentum * (h - h_prev)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
        h_prev = h
        h = h_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
    return h_best, c5_best

def coord_descent_ls(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-6:
                continue
            h_orig = h[i]
            h[i] = 0.5 * (lo + hi)
            proj1 = project_to_sum(h.copy(), target_sum)
            h_tmp = h.copy()
            def f1d(v):
                h_tmp[i] = v
                h_p = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_p)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-5, 'maxiter': 50})
                if res.fun < c5_cur - 1e-12:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                h[i] = h_orig
        if not improved:
            break
    return h, c5_cur

def coord_descent_fixed(h0, n, target_sum, n_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0
    h_cur = best_h.copy()

    if time_left() > 300:
        for k_top in [20, 10, 5, 3, 2]:
            if time_left() < 60:
                break
            mi = max(50, min(300, int(time_left() * 0.15)))
            h_opt = topk_slsqp(h_cur, n1, ts1, k_top, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cur = h_opt

    if time_left() > 200:
        for alpha in [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]:
            if time_left() < 50:
                break
            mi = max(50, min(400, int(time_left() * 0.12)))
            h_opt = slsqp_optimize(h_cur, n1, ts1, alpha, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cur = h_opt

    if time_left() > 150:
        n_iters = min(3000, int(time_left() * 3))
        h_sg, c5_sg = subgradient_multi(h_cur, n1, ts1, n_iters, 0.05, 0.85, n_focus=3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h_cur = h_sg

    if time_left() > 100:
        n_iters2 = min(2000, int(time_left() * 2))
        h_sg2, c5_sg2 = subgradient_momentum(h_cur, n1, ts1, n_iters2, 0.03, 0.9)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h_cur = h_sg2

    if time_left() > 80:
        n_cd = min(15, max(3, int(time_left() // 15)))
        h_cd, c5_cd = coord_descent_ls(h_cur, n1, ts1, n_cd, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_cur = h_cd

    if time_left() > 50:
        deltas = [-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01]
        h_cd2, c5_cd2 = coord_descent_fixed(h_cur, n1, ts1, 20, rng, deltas)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h_cur = h_cd2

    if time_left() > 40:
        for alpha in [1000.0, 2000.0, 4000.0, 8000.0]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, n1, ts1, alpha, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h_cur = h_pol

    n2 = n1 * 2
    if time_left() > 250 and n2 <= 1440:
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = project_to_sum(h_interp, ts2)
        h_cur2 = h_interp

        for k_top in [10, 5, 3]:
            if time_left() < 60:
                break
            mi = max(30, min(200, int(time_left() * 0.1)))
            h_opt2 = topk_slsqp(h_cur2, n2, ts2, k_top, mi)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2
            h_cur2 = h_opt2

        for alpha in [50.0, 100.0, 200.0, 500.0, 1000.0]:
            if time_left() < 50:
                break
            mi = max(30, min(250, int(time_left() * 0.1)))
            h_opt2 = slsqp_optimize(h_cur2, n2, ts2, alpha, mi)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2
            h_cur2 = h_opt2

        if time_left() > 80:
            n_iters3 = min(2000, int(time_left() * 2))
            h_sg3, c5_sg3 = subgradient_multi(h_cur2, n2, ts2, n_iters3, 0.03, 0.85, n_focus=3)
            if c5_sg3 < best_c5:
                best_c5 = c5_sg3
                best_h = h_sg3.copy()
                best_n = n2
                h_cur2 = h_sg3

        if time_left() > 50:
            n_cd2 = min(8, max(2, int(time_left() // 20)))
            h_cd3, c5_cd3 = coord_descent_ls(h_cur2, n2, ts2, n_cd2, rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n2
                h_cur2 = h_cd3

        if time_left() > 30:
            h_cd4, c5_cd4 = coord_descent_fixed(h_cur2, n2, ts2, 15, rng,
                                                [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005])
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()
                best_n = n2

    n_f = best_n
    ts_f = n_f / 2.0
    n_starts = max(3, int(time_left() // 40))
    for start_i in range(n_starts):
        if time_left() < 30:
            break
        pert_scale = 0.04 / (start_i + 1)
        h_pert = best_h + rng.normal(0, pert_scale, size=n_f)
        h_pert = project_to_sum(h_pert, ts_f)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        h_work = h_pert if c5_pert < best_c5 else best_h.copy()
        for alpha in [100.0, 500.0, 2000.0]:
            if time_left() < 25:
                break
            h_opt = slsqp_optimize(h_work, n_f, ts_f, alpha, 150)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_f
                h_work = h_opt
            else:
                h_work = h_opt
        if time_left() > 25:
            h_cd, c5_cd = coord_descent_fixed(h_work, n_f, ts_f, 10, rng,
                                              [-0.003, -0.001, 0.001, 0.003])
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    if best_h is not None and time_left() > 20:
        n_p = len(best_h)
        ts_p = n_p / 2.0
        for alpha in [5000.0, 10000.0, 20000.0]:
            if time_left() < 15:
                break
            h_pol = slsqp_optimize(best_h, n_p, ts_p, alpha, 500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        if time_left() > 15:
            h_cd, c5_cd = coord_descent_ls(best_h, n_p, ts_p, 5, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
        if time_left() > 10:
            h_cd2, c5_cd2 = coord_descent_fixed(best_h, n_p, ts_p, 10, rng,
                                                [-0.001, -0.0005, 0.0005, 0.001])
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - ts_final) > 0.01:
        best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
