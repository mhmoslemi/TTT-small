            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_topk(h, top_k=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        indices = np.argsort(corr)[-top_k:]
        sub = corr[indices]
    else:
        indices = np.arange(len(corr))
        sub = corr
    return sub, indices

def smooth_max_val(sub, alpha):
    m = np.max(sub)
    shifted = alpha * (sub - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    return val, w

def grad_from_weights(h, w, indices, n):
    g = np.zeros(n)
    for idx in range(len(indices)):
        l = int(indices[idx]) - (n - 1)
        wl = w[idx]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return g

    n = len(x)
    sub, indices = corr_topk(x, top_k)
    val, w = smooth_max_val(sub, alpha)
    return val - lam * np.sum(x)

def grad_fn(x, alpha, top_k, lam):

    sub, indices = corr_topk(x, top_k)
    val, w = smooth_max_val(sub, alpha)
    g = grad_from_weights(x, w, indices, n)
    g -= lam
    return g

def lagrangian_solve(h0, n, alpha, top_k, maxiter=300):

    lo = -10.0
    hi = 10.0

        h_sol = res.x
        s = np.sum(h_sol)
        if abs(s - target_sum) < 1e-8:
            break
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return project_to_sum(h_sol, target_sum)

def project_grad_step(h, grad, target_sum, lr):

def sort_starts(starts):
    c5_vals = [compute_c5(h) for h in starts]
    paired = list(zip(c5_vals, starts))
    paired.sort(key=lambda x: x[0])
    return [p[1] for p in paired], [p[0] for p in paired]

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    elapsed = time.time() - t0
    remaining = budget_s - elapsed

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            if best_n % 2 != 0:
                best_h = interp_resample(best_h, best_n, best_n + 1)
                best_n += 1
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        x = np.linspace(0.0, 2.0, best_n, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)

    starts = [best_h.copy()]

    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10, 12]:
        for amp in [0.03, 0.08, 0.12]:
            hs = best_h + amp * np.sin(2.0 * np.pi * freq * x1 / 2.0)
            hs = project_to_sum(hs, ts1)
            starts.append(hs)

    for delta in [0.01, 0.02, 0.04]:
        hp = best_h + rng.uniform(-delta, delta, n1)
        hp = project_to_sum(hp, ts1)
        starts.append(hp)

    starts, _ = sort_starts(starts)
    starts = starts[:6]

    alpha_sched = [50, 200, 1000, 5000, 20000]
    topk_sched = [20, 40, 60, 100, 200]

    for si in range(len(starts)):
        if remaining < 80:
            break
        remaining = budget_s - (time.time() - t0)
        h_cur = starts[si].copy()
        c5_cur = compute_c5(h_cur)

        for ai in range(len(alpha_sched)):
            if remaining < 50:
                break
            remaining = budget_s - (time.time() - t0)
            alpha = alpha_sched[ai]
            tk = topk_sched[ai]
            mi = max(80, min(300, int(remaining * 0.3)))
            h_opt = lagrangian_optimize(h_cur, n1, alpha, tk, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n1
            h_cur = h_opt

        if remaining > 60:
            remaining = budget_s - (time.time() - t0)
            ni = min(1500, int(remaining * 2))
            h_pg, c5_pg = projected_grad_opt(h_cur, n1, 50000, 200, ni, 0.005)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n1

    remaining = budget_s - (time.time() - t0)

    for n2_factor in [1.5, 2.0]:
        if remaining < 100:
            break
        n2 = int(best_n * n2_factor)
        if n2 % 2 != 0:
            n2 += 1
        ts2 = n2 / 2.0

        h_interp = interp_resample(best_h, best_n, n2)
        h_interp = project_to_sum(h_interp, ts2)
        c5_interp = compute_c5(h_interp)

        alpha2_sched = [200, 1000, 5000, 20000]
        topk2 = [40, 80, 150, 300]

        for ai in range(len(alpha2_sched)):
            if remaining < 40:
                break
            remaining = budget_s - (time.time() - t0)
            alpha = alpha2_sched[ai]
            tk = min(topk2[ai], 2 * n2 - 1)
            mi = max(80, min(250, int(remaining * 0.35)))
            h_opt = lagrangian_optimize(h_interp, n2, alpha, tk, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n2
            h_interp = h_opt

        if remaining > 50:
            remaining = budget_s - (time.time() - t0)
            ni = min(1000, int(remaining * 2))
            h_pg, c5_pg = projected_grad_opt(h_interp, n2, 50000, min(300, 2*n2-1), ni, 0.003)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n2

    remaining = budget_s - (time.time() - t0)

    if remaining > 40:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha in [50000, 200000, 500000]:
            if remaining < 20:
                break
            remaining = budget_s - (time.time() - t0)
            tk = 2 * n_f - 1
            h_opt = lagrangian_optimize(best_h, n_f, alpha, tk, 400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        remaining = budget_s - (time.time() - t0)
        if remaining > 25:
            ni = min(3000, int(remaining * 3))
            h_pg, c5_pg = projected_grad_opt(best_h, n_f, 500000, 2 * n_f - 1, ni, 0.002)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            steps = [0.005, 0.002, 0.001, 0.0005, -0.005, -0.002, -0.001, -0.0005]
            mr = max(3, min(20, int(remaining // 3)))
            h_cd, c5_cd = coord_descent(best_h, n_f, ts_f, mr, steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        for delta in [0.005, 0.01]:
            if remaining < 10:
                break
            remaining = budget_s - (time.time() - t0)
            hp = best_h + rng.uniform(-delta, delta, len(best_h))
            hp = project_to_sum(hp, len(best_h) / 2.0)
            h_opt = lagrangian_optimize(hp, len(best_h), 100000, 2*len(best_h)-1, 300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
