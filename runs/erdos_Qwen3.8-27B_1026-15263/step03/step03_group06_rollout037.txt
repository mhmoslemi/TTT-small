    lo, hi = -max(np.max(h), 1.0) - 1, max(np.min(h), 0.0) + 1

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h_final = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_final)
    if abs(diff) > 1e-12:
        mask = (h_final > 1e-12) & (h_final < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h_final[mask] += diff / cnt
            h_final = np.clip(h_final, 0.0, 1.0)
    return h_final

def soft_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m_val + np.log(s) / alpha
    
    # Gradient via FFT
    # grad[j] = sum_{i=j}^{n-1+j} w[i] - (h*w)[n-1+j] - (h*w_rev)[n-1+j]
    w_cum = np.zeros(2 * n)
    w_cum[1:] = np.cumsum(w)
    j_idx = np.arange(n)
    part_a = w_cum[n + j_idx] - w_cum[j_idx]
    
    hw = fftconvolve(h, w, mode='full')
    w_rev = w[::-1]
    hw_rev = fftconvolve(h, w_rev, mode='full')
    
    part_b = hw[n - 1 + j_idx]
    part_c = hw_rev[n - 1 + j_idx]
    
    grad = part_a - part_b - part_c
    return sm_val, grad

    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    for it in range(n_iters):
        t += 1
        sm_val, g = soft_max_and_grad(h, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)

        h_new = project_to_sum(h_raw, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

        h_raw = y - lr * g
        h_new = project_to_sum(h_raw, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        if it % 100 == 0:
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                lr = min(lr * 1.05, 0.5)
            else:
                lr *= 0.9
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def slsqp_fft(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = soft_max_and_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = soft_max_and_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_refine(h0, n, target_sum, n_rounds, rng, time_limit):
    t_limit = time.time() + time_limit
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() > t_limit:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() > t_limit:
                break
            hi_idx = min(1.0, h[i] + 0.3)
            lo_idx = max(0.0, h[i] - 0.3)
            if hi_idx - lo_idx < 1e-8:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo_idx, hi_idx), method='bounded',
                                     options={'xatol': 1e-6})
                if res.fun < c5_cur - 1e-13:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def fourier_init(n, rng, n_freqs=20):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_freqs + 1):
        coeff = rng.uniform(-0.3, 0.3) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += coeff * np.cos(k * np.pi * x + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    return h

def bimodal_init(n, rng, n_bands=3):
    h = np.zeros(n)
    for _ in range(n_bands):
        center = rng.uniform(0, n)
        width = rng.uniform(n // 8, n // 3)
        for i in range(n):
            d = min(abs(i - center), n - abs(i - center))
            h[i] += np.exp(-0.5 * (d / width) ** 2)
    h = h / np.max(h)
    h = project_to_sum(h, n / 2.0)
    return h
