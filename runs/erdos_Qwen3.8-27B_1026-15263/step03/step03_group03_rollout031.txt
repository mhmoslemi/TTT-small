                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    nm1 = n - 1
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx + m_idx - nm1
    second_idx = j_idx - m_idx + nm1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_obj(h, alpha)
    def g(h):
        return smooth_max_grad(h, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_left():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Pair descent with adaptive noise (30% budget)
    t_end1 = t_start + 0.30 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0

    while time.time() < t_end1:
        h_work, corr_w, imp = pair_descent_pass(h_work, corr_w, rng, n_cand=10)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-10:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall >= 250:
            noise = min(0.10, 0.04 + 0.01 * (stall // 250))
            h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
        elif stall >= 80 and stall % 40 == 0:
            h_work = project_to_sum(h_work + rng.uniform(-0.03, 0.03, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    # Phase 2: SLSQP continuation at base resolution (30% budget)
    t_end2 = t_start + 0.60 * budget_s
    n_base = best_n
    h_cont = best_h.copy()
    alphas_cont = [15, 50, 150, 500, 1500]
    for alpha in alphas_cont:
        if time.time() >= t_end2 - 10:
            break
        h_new = slsqp_smooth(h_cont, n_base, n_base / 2.0, alpha, maxiter=600)
        # Interleave with pair descent
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        for _ in range(15):
            if time.time() >= t_end2 - 5:
                break
            h_new, corr_new, _ = pair_descent_pass(h_new, corr_new, rng, n_cand=10)
        h_new = project_to_sum(h_new, n_base / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_base
        h_cont = h_new.copy()

    # Phase 3: Multi-resolution exploration (25% budget)
    t_end3 = t_start + 0.85 * budget_s
    resolutions = []
    for delta in [0, 40, 80, 160, 240, -40]:
        r = best_n + delta
        if r >= 80 and r % 2 == 0 and r not in resolutions:
            resolutions.append(r)
    resolutions.sort()

    for n_tr in resolutions:
        if time.time() >= t_end3 - 20:
            break
        ts = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_tr = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts)

        # SLSQP at this resolution
        h_tr = slsqp_smooth(h_tr, n_tr, ts, 200, maxiter=500)
        corr_tr = np.correlate(h_tr, 1.0 - h_tr, mode="full")
        for _ in range(20):
            if time.time() >= t_end3 - 10:
                break
            h_tr, corr_tr, _ = pair_descent_pass(h_tr, corr_tr, rng, n_cand=10)
        h_tr = project_to_sum(h_tr, ts)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr

        # Random restart at this resolution
        if time_left() > 60:
            x_alt = np.linspace(0, 2, n_tr, endpoint=False)
            h_rand = project_to_sum(rng.uniform(0.1, 0.9, n_tr), ts)
            h_rand = slsqp_smooth(h_rand, n_tr, ts, 100, maxiter=300)
            corr_r = np.correlate(h_rand, 1.0 - h_rand, mode="full")
            for _ in range(10):
                if time.time() >= t_end3 - 5:
                    break
                h_rand, corr_r, _ = pair_descent_pass(h_rand, corr_r, rng, n_cand=8)
            h_rand = project_to_sum(h_rand, ts)
            c5_r = compute_c5(h_rand)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_rand.copy()
                best_n = n_tr

    # Phase 4: Final intensive pair descent (10% budget)
    t_end4 = t_start + budget_s - 5
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall4 = 0
    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent_pass(h_fin, corr_fin, rng, n_cand=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall4 = 0
        else:
            stall4 += 1
            if stall4 >= 120:
                h_fin = project_to_sum(h_fin + rng.uniform(-0.012, 0.012, n_fin), n_fin / 2.0)
                corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
                stall4 = 0

    # Final polish with SLSQP if time remains
    if time_left() > 15:
        n_f = best_n
        h_pol = slsqp_smooth(best_h, n_f, n_f / 2.0, 800, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
