```python
import numpy as np
from scipy.optimize import minimize
import time

_G_TARGET = 0.0
_G_ALPHA = 1.0
_G_ACTIVE = None
_G_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, active_idx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = vals.max()
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_v = np.exp(shifted)
    return m + np.log(exp_v.sum() / len(exp_v)) / alpha

def smooth_max_grad(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = vals.max()
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    
    g = np.zeros(n)
    for pos in range(len(active_idx)):
        wm = w[pos]
        if wm < 1e-14:
            continue
        k = active_idx[pos] - (n - 1)
        if k >= 0:
            g[:n-k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n-k]
        else:
            k2 = -k
            g[k2:] += wm * (1.0 - h[:n-k2])
            g[:n-k2] -= wm * h[k2:]
    return g

def _obj(x):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_val(h, _G_ACTIVE, _G_ALPHA)

def _grad(x):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_grad(h, _G_ACTIVE, _G_ALPHA)

def _con_fun(x):
    return np.sum(x) - _G_TARGET

def _con_jac(x):
    return np.ones(_G_N)

def slsqp_active(h0, n, target_sum, active_idx, alpha, maxiter=300):
    global _G_TARGET, _G_ALPHA, _G_ACTIVE, _G_N
    _G_TARGET = target_sum
    _G_ALPHA = alpha
    _G_ACTIVE = active_idx
    _G_N = n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def get_active(corr, frac=0.99):
    mx = np.max(corr)
    return np.where(corr >= frac * mx)[0]

def binary_push(h, n, target_sum, strength, n_iter):
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(n_iter):
        h_new = h + strength * (2.0 * h - 1.0)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            strength *= 0.92
        else:
            strength *= 0.7
            if strength < 1e-6:
                break
    return h, c5

def coord_descent(h, n, target_sum, max_rounds, rng):
    h = h.copy()
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_c = c5
            best_d = 0.0
            for d in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c = compute_c5(h_new)
                if c < best_c - 1e-15:
                    best_c = c
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def swap_optimize(h, n, target_sum, max_rounds, rng):
    h = h.copy()
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(3 * n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try small transfer from i to j
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                amt = (h[i] - h[j]) * frac
                if abs(amt) < 1e-10:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] - amt, 0.0, 1.0)
                h_new[j] = np.clip(h_new[j] + amt, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c = compute_c5(h_new)
                if c < c5 - 1e-15:
                    h = h_new
                    c5 = c
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_TARGET = 0.0
_G_ALPHA = 1.0
_G_ACTIVE = None
_G_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, active_idx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = vals.max()
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_v = np.exp(shifted)
    return m + np.log(exp_v.sum() / len(exp_v)) / alpha

def smooth_max_grad(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = vals.max()
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    g = np.zeros(n)
    for pos in range(len(active_idx)):
        wm = w[pos]
        if wm < 1e-14:
            continue
        k = active_idx[pos] - (n - 1)
        if k >= 0:
            g[:n-k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n-k]
        else:
            k2 = -k
            g[k2:] += wm * (1.0 - h[:n-k2])
            g[:n-k2] -= wm * h[k2:]
    return g

    _G_ACTIVE = active_idx
    _G_N = n
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:

def get_active(corr, frac=0.99):
    mx = np.max(corr)

def coord_descent(h, n, target_sum, max_rounds, rng):
```
