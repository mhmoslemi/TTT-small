    """Solve LP to find h given g = 1-h, minimizing max_k sum_i h[i]*g[i+k-(n-1)]"""
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Build matrix A: A[k, i] = g[i + k - (n-1)] if valid, else 0
    # Size: (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i + k - (n-1) must be in [0, n-1]
        # So i >= n-1-k and i <= n-1-k+n-1 = 2n-2-k
        # And i in [0, n-1]
        i_start = max(0, n-1-k)
        i_end = min(n-1, 2*n-2-k)
        if i_start <= i_end:
            j_start = i_start + k - (n-1)
            A[k, i_start:i_end+1] = g[j_start:j_start+(i_end-i_start+1)]
    
    # LP: minimize t
    # subject to: A @ h <= t, i.e., A @ h - t <= 0
    # -h <= 0, h <= 1, sum(h) = n/2
    # Variables: [h (n), t]
    
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # A @ h - t <= 0  =>  [A, -1] @ [h, t] <= 0
    # -h <= 0  =>  [-I, 0] @ [h, t] <= 0
    # h <= 1  =>  [I, 0] @ [h, t] <= 1
    
    A_ub = np.zeros((2*n-1 + 2*n, n+1))
    b_ub = np.zeros(2*n-1 + 2*n)

    A_ub[:2*n-1, :n] = A
    A_ub[:2*n-1, n] = -1.0
    b_ub[:2*n-1] = 0.0

    A_ub[2*n-1:2*n-1+n, :n] = -np.eye(n)
    b_ub[2*n-1:2*n-1+n] = 0.0
    
    # Next n rows: h <= 1
    A_ub[2*n-1+n:2*n-1+2*n, :n] = np.eye(n)
    b_ub[2*n-1+n:2*n-1+2*n] = 1.0

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0

    bounds = [(0, 1)] * n + [(None, None)]

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(32):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

    improved = False
    for j in order:

        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0: h[j] = 0.0
            elif h[j] > 1: h[j] = 1.0
            if h[best_k] < 0: h[best_k] = 0.0
            elif h[best_k] > 1: h[best_k] = 1.0
            improved = True
    return h, corr, improved

    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

        j_end = min(n - 1, k)
