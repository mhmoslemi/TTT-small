def smooth_max_and_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def block_sa(h_start, n, target_sum, t_end, rng, max_block=None):
    if max_block is None:
        max_block = max(2, n // 30)
    h = h_start.copy()
    c5_curr = float(np.max(compute_corr(h)))
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.012
    min_T = 0.0015
    no_improve = 0
    n_ones_target = target_sum
    
    while time.time() < t_end:
        r = rng.random()
        if r < 0.25:
            # Block swap
            bl = rng.randint(2, max_block + 1)
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            if len(ones_idx) < bl or len(zeros_idx) < bl:
                continue
            # Try to find contiguous blocks
            i0 = ones_idx[rng.randint(len(ones_idx))]
            # Extend to find contiguous block
            start = i0
            while start > 0 and start - bl >= 0 and h[start-1] == 1.0:
                start -= 1
            end = i0
            while end + bl < n and h[end+1] == 1.0:
                end += 1
            block_avail = end - start + 1
            if block_avail < bl:
                continue
            # Similarly for zeros
            j0 = zeros_idx[rng.randint(len(zeros_idx))]
            zstart = j0
            while zstart > 0 and h[zstart-1] == 0.0:
                zstart -= 1
            zend = j0
            while zend + bl < n and h[zend+1] == 0.0:
                zend += 1
            zblock_avail = zend - zstart + 1
            if zblock_avail < bl:
                continue
            # Pick random sub-blocks
            si = start + rng.randint(0, block_avail - bl + 1)
            sj = zstart + rng.randint(0, zblock_avail - bl + 1)
            h[si:si+bl], h[sj:sj+bl] = h[sj:sj+bl].copy(), h[si:si+bl].copy()
            c5_new = float(np.max(compute_corr(h)))
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[si:si+bl], h[sj:sj+bl] = h[si:si+bl].copy(), h[sj:sj+bl].copy()
        elif r < 0.75:
            # Point swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = float(np.max(compute_corr(h)))
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            # Directed: swap involving worst lag position
            corr = compute_corr(h)
            worst_m = int(np.argmax(corr))
            k = worst_m - (n - 1)
            if k != 0:
                abs_k = abs(k)
                if abs_k < n:
                    if k > 0:
                        # h[i]=1, h[i+k]=0
                        mask_i = h[:n-abs_k] == 1.0
                        mask_j = h[abs_k:] == 0.0
                        both = mask_i & mask_j  # wait, these are different lengths... 
                        # Actually h[:n-abs_k] has length n-abs_k, h[abs_k:] has length n-abs_k
                        both = (h[:n-abs_k] == 1.0) & (h[abs_k:] == 0.0)
                    else:
                        # h[i]=0, h[i+abs_k]=1 (for the negative lag direction)
                        # corr at negative lag k: pairs (i, i+|k|) with h[i]=0, h[i+|k|]=1
                        both = (h[:n-abs_k] == 0.0) & (h[abs_k:] == 1.0)
                    if np.any(both):
                        candidates = np.where(both)[0]
                        i = candidates[rng.randint(len(candidates))]
                        j = rng.randint(n)
                        if h[i] != h[j]:
                            h[i], h[j] = h[j], h[i]
                            c5_new = float(np.max(compute_corr(h)))
                            delta = c5_new - c5_curr
                            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                                c5_curr = c5_new
                                if c5_new < best_c5 - 1e-12:
                                    best_c5 = c5_new
                                    best_h = h.copy()
                                    no_improve = 0
                                else:
                                    no_improve += 1
                            else:
                                h[i], h[j] = h[j], h[i]
        
        if no_improve > 25000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.02)
            no_improve = 0
        T = max(T * 0.999999, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)
