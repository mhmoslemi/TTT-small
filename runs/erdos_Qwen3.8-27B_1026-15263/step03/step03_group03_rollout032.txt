    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_B(h, n):
    nm1 = n - 1
    j = np.arange(n)[:, None]
    m = np.arange(2 * n - 1)[None, :]
    idx1 = j + nm1 - m
    idx2 = j + m - nm1
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    B = valid1.astype(float) * (1.0 - h[idx1c]) - valid2.astype(float) * h[idx2c]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(36):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep_fast(h, corr, rng, n_cand=15):
    n = len(h)
    all_B = compute_all_B(h, n)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = all_B[j]
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (j + n // 4) % n, (j + 3 * n // 4) % n]
        n_extra = max(0, n_cand - len(k_set))
        if n_extra > 0:
            extra = rng.randint(n, size=n_extra)
            k_set = k_set + list(extra)
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            D = B_j - all_B[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

_smooth_alpha = 100.0
_smooth_target = 120.0

def _smo_obj(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = _smooth_alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    return m_val + log_sum / _smooth_alpha

def _smo_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = _smooth_alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    nm1 = n - 1
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + nm1
    second_idx = j_idx + m_idx - nm1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f.astype(float) * (1.0 - x[fc]) * w[None, :]
    cs = valid_s.astype(float) * x[sc] * w[None, :]
    return np.sum(cf - cs, axis=1)

def _cons_fun(x):
    return np.sum(x) - _smooth_target

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_local(h0, n, target_sum, alpha, maxiter=500):
    global _smooth_alpha, _smooth_target
    _smooth_alpha = alpha
    _smooth_target = target_sum
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smo_obj, h0.copy(), method='SLSQP', jac=_smo_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Quick pair descent from initial (12% budget)
    t_end1 = t_start + 0.12 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    while time.time() < t_end1:
        h_work, corr_w, _ = pair_descent_sweep_fast(h_work, corr_w, rng, n_cand=15)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w

    # Phase 2: Multi-resolution exploration (43% budget)
    t_end2 = t_start + 0.55 * budget_s
    resolutions = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300, 320, 360, 400, 480, 560, 640]

    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue

        starts = []
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        starts.append(h_interp)
        starts.append(project_to_sum(rng.uniform(0, 1, n_tr), ts_tr))

        h_blk = np.zeros(n_tr)
        blk = max(1, n_tr // 8)
        for i in range(0, n_tr, 2 * blk):
            h_blk[i:min(i + blk, n_tr)] = 1.0
        starts.append(project_to_sum(h_blk, ts_tr))

        period = max(2, n_tr // 5)
        h_per = np.zeros(n_tr)
        h_per[0:max(1, period // 2)] = 1.0
        h_per = np.tile(h_per, n_tr // period + 1)[:n_tr]
        starts.append(project_to_sum(h_per, ts_tr))

        for h_st in starts:
            if time.time() >= t_end2:
                break
            h_st = h_st.copy()
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            n_sweeps = min(15, max(3, int((t_end2 - time.time()) / 15.0)))
            for _ in range(n_sweeps):
                if time.time() >= t_end2:
                    break
                h_st, corr_st, _ = pair_descent_sweep_fast(h_st, corr_st, rng, n_cand=12)
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 3: Deep optimization (30% budget)
    t_end3 = t_start + 0.85 * budget_s
    h_deep = best_h.copy()
    n_d = len(h_deep)
    ts_d = n_d / 2.0
    corr_d = np.correlate(h_deep, 1.0 - h_deep, mode="full")
    stall = 0

    while time.time() < t_end3:
        h_deep, corr_d, _ = pair_descent_sweep_fast(h_deep, corr_d, rng, n_cand=22)
        c5_d = compute_c5(h_deep)
        if c5_d < best_c5 - 1e-12:
            best_c5 = c5_d
            best_h = h_deep.copy()
            best_n = n_d
            stall = 0
        else:
            stall += 1
        if stall > 0 and stall % 40 == 0:
            noise = min(0.12, 0.015 + 0.005 * (stall // 80))
            h_deep = project_to_sum(h_deep + rng.uniform(-noise, noise, n_d), ts_d)
            corr_d = np.correlate(h_deep, 1.0 - h_deep, mode="full")
            if stall > 400:
                stall = 0

    # SLSQP polish
    if time.time() < t_end3 - 30:
        for alpha in [30, 80, 200]:
            if time.time() >= t_end3:
                break
            h_pol = slsqp_local(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Final aggressive polish (15% budget)
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    t_end4 = t_start + budget_s - 3

    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent_sweep_fast(h_fin, corr_fin, rng, n_cand=28)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
