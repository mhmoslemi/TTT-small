```python
def compute_subgrad_max(h, n):
    """Compute subgradient of max_k corr_k at h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Active set: lags where corr is close to max
    tol = 1e-10
    active = (corr >= max_val - tol)
    if not np.any(active):
        active = (corr == max_val)
    
    # For each active lag, compute subgradient and average
    n_act = np.sum(active)
    sg = np.zeros(n)
    for m in np.where(active)[0]:
        s = m - (n - 1)  # shift
        # subgrad[j] = [0<=j-s<n]*(1-h[j-s]) - [0<=j+s<n]*h[j+s]
        # First term: j ranges over [max(0,s), min(n, n+s))
        # Second term: j ranges over [max(0,-s), min(n, n-s))
        j1_lo = max(0, s)
        j1_hi = min(n, n + s)
        if j1_lo < j1_hi:
            sg[j1_lo:j1_hi] += (1.0 - h[j1_lo - s:j1_hi - s])
        j2_lo = max(0, -s)
        j2_hi = min(n, n - s)
        if j2_lo < j2_hi:
            sg[j2_lo:j2_hi] -= h[j2_lo + s:j2_hi + s]
    sg /= n_act
    return sg, max_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Numerical stability
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_obj_and_grad(h, alpha)[0]
    def g(h):
        return smooth_obj_and_grad(h, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    def f(h):
        obj, _ = smooth_obj_and_grad(h, alpha)
        return obj + penalty * (np.sum(h) - target_sum) ** 2
    def g(h):
        _, grad = smooth_obj_and_grad(h, alpha)
        grad = grad + 2 * penalty * (np.sum(h) - target_sum) * np.ones(n)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def subgradient_step(h, step_size, use_smooth=True, alpha=100):
    n = len(h)
    if use_smooth:
        _, sg = smooth_obj_and_grad(h, alpha)
    else:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = (corr >= max_val - 1e-12)
        n_act = max(1, np.sum(active))
        sg = np.zeros(n)
        for m in np.where(active)[0]:
            s = m - (n - 1)
            j1_lo = max(0, s)
            j1_hi = min(n, n + s)
            if j1_lo < j1_hi:
                sg[j1_lo:j1_hi] += (1.0 - h[j1_lo - s:j1_hi - s])
            j2_lo = max(0, -s)
            j2_hi = min(n, n - s)
            if j2_lo < j2_hi:
                sg[j2_lo:j2_hi] -= h[j2_lo + s:j2_hi + s]
        sg /= n_act
```

def compute_c5(h):

        else:
            h[np.argmin(np.abs(h - 0.5))] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    fi = j_idx - m_idx + n - 1
    si = j_idx + m_idx - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target, alpha, maxiter):
    def f(x): return smooth_obj_grad(x, alpha)[0]
    def g(x): return smooth_obj_grad(x, alpha)[1]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=[(0.0, 1.0)]*n,
                       constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target,'jac':lambda x: np.ones(n)}],
                       options={'maxiter':maxiter, 'ftol':1e-15})
        return project_to_sum(res.x, target)
    except:
        return project_to_sum(h0.copy(), target)

def lbfgs_local(h0, n, target, alpha, maxiter, pen):
    def f(x):
        o,_ = smooth_obj_grad(x, alpha)
        return o + pen*(np.sum(x)-target)**2
    def g(x):
        _,gr = smooth_obj_grad(x, alpha)
        return gr + 2*pen*(np.sum(x)-target)*np.ones(n)
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter':maxiter, 'ftol':1e-15})
        return project_to_sum(res.x, target)
    except:
        return project_to_sum(h0.copy(), target)

def compute_B(j, h, n):
    nm1 = n-1
    B = np.zeros(2*n-1)
    lo1 = max(0, nm1-j)
    hi1 = min(2*n-2, 2*n-2-j)
    if lo1 <= hi1:
        mr = np.arange(lo1, hi1+1)
        B[mr] += 1.0 - h[j - nm1 + mr]
    lo2 = max(0, j)
    hi2 = min(2*n-2, j+nm1)
    if lo2 <= hi2:
        mr2 = np.arange(lo2, hi2+1)
        B[mr2] -= h[j + nm1 - mr2]
    return B

def pair_sweep(h, corr, rng, ncand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(j, h, n)
        best_val = float(np.max(corr))
        bd, bD, bk = 0.0, None, -1
        ks = list({(j+1)%n, (j+n-1)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n,
                   (j+max(1,n//5))%n, (j-max(1,n//5)+n)%n})
        while len(ks) < ncand:
            ks.append(rng.randint(n))
        seen = set()
        for k in ks:
            if k==j or k in seen: continue
            seen.add(k)
            B_k = compute_B(k, h, n)
            D = B_j - B_k
            dlo = max(-h[j], h[k]-1.0)
            dhi = min(1.0-h[j], h[k])
            if dhi-dlo < 1e-13: continue
            tg = np.linspace(dlo, dhi, 16)
            vs = np.max(corr[None,:] + D[None,:]*tg[:,None], axis=1)
            bi = int(np.argmin(vs))
            if bi==0: dopt=dlo
            elif bi==len(tg)-1: dopt=dhi
            else:
                lr,hr = tg[bi-1], tg[bi+1]
                for _ in range(28):
                    mid=(lr+hr)*0.5
                    if np.max(corr+D*mid)<=np.max(corr+D*lr): lr=mid
                    else: hr=mid
                dopt=(lr+hr)*0.5
            vo = np.max(corr + D*dopt)
            if vo < best_val - 1e-14:
                best_val, bd, bD, bk = vo, dopt, D, k
        if bk>=0 and abs(bd)>1e-13:
            corr += bd*bD
            h[j] = np.clip(h[j]+bd, 0.0, 1.0)
            h[bk] = np.clip(h[bk]-bd, 0.0, 1.0)
            improved = True
    return h, corr, improved

def subgrad_step(h, step, alpha):
    _, sg = smooth_obj_grad(h, alpha)
    return project_to_sum(h - step*sg, len(h)/2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, float('inf'), 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n/2.0)
            best_c5 = compute_c5(best_h)
        except: pass
    
    if best_h is None:
        best_n, best_h = 240, np.full(240, 0.5)
        best_c5 = compute_c5(best_h)
    
    def tl(): return budget_s - (time.time()-t0)
    
    # Phase 1: Quick pair descent (8% time)
    t1 = t0 + 0.08*budget_s
    hw = best_h.copy()
    cw = np.correlate(hw, 1.0-hw, mode="full")
    while time.time() < t1:
        hw, cw, _ = pair_sweep(hw, cw, rng, 8)
        c = compute_c5(hw)
        if c < best_c5: best_c5, best_h = c, hw.copy()
    
    # Phase 2: Main gradient optimization (55% time)
    t2 = t0 + 0.63*budget_s
    alphas = [30, 50, 80, 120, 180, 280, 400, 600]
    hcur, ncur = best_h.copy(), best_n
    
    for al in alphas:
        if time.time() >= t2: break
        # L-BFGS-B pass (fast)
        hnew = lbfgs_local(hcur, ncur, ncur/2.0, al, 2000, 50.0)
        c = compute_c5(hnew)
        if c < best_c5: best_c5, best_h, hcur = c, hnew.copy(), hnew.copy()
        else: hcur = hnew.copy()
        
        # SLSQP pass (more accurate)
        if time.time() < t2:
            hnew2 = slsqp_local(hcur, ncur, ncur/2.0, al, 600)
            c2 = compute_c5(hnew2)
            if c2 < best_c5: best_c5, best_h, hcur = c2, hnew2.copy(), hnew2.copy()
            else: hcur = hnew2.copy()
        
        # Pair descent burst
        cw2 = np.correlate(hcur, 1.0-hcur, mode="full")
