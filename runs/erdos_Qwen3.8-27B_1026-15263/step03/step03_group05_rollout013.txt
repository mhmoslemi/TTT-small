    old_block = h[start:start+length].copy()
    h[start:start+length] = 1.0 - old_block
    return project_box_sum(h, target_sum)

def local_search_sa(h, n, rng, time_fn, t0=1.0, t1=0.01, max_iter=None):
    """Simulated annealing with mixed moves."""
    ts = n / 2.0
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    if max_iter is None:
        max_iter = 100000
    
    for it in range(max_iter):
        if time_fn() < 10:
            break
        
        # Temperature schedule (geometric cooling)
        T = t0 * (t1 / t0) ** (it / max_iter)
        
        # Choose move type
        move_type = rng.random()
        
        if move_type < 0.4:
            # Point perturbation
            i = rng.randint(n)
            delta = rng.uniform(-0.5, 0.5)
            new_h = current_h.copy()
            new_h[i] += delta
            new_h = project_box_sum(new_h, ts)
        elif move_type < 0.6:
            # Block flip
            block_len = rng.choice([1, 2, 3, 5, 8, 13, 21])
            block_len = min(block_len, n // 4)
            start = rng.randint(0, n - block_len + 1)
            new_h = current_h.copy()
            new_h[start:start+block_len] = 1.0 - new_h[start:start+block_len]
            new_h = project_box_sum(new_h, ts)
        elif move_type < 0.8:
            # Pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.1, 0.5)
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_h = project_box_sum(new_h, ts)
        else:
            # Random restart of a random block
            block_len = rng.randint(n//10, n//3)
            start = rng.randint(0, n - block_len + 1)
            new_h = current_h.copy()
            new_h[start:start+block_len] = rng.uniform(0, 1, block_len)
            new_h = project_box_sum(new_h, ts)
        
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5
        
        # Acceptance criterion
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

def gradient_momentum(h, n, alpha, lr, momentum, max_iter, time_fn):
    """Gradient descent with momentum."""
    ts = n / 2.0
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

        if time_fn() < 10:
            break

        v = momentum * v - lr * grad
        h_new = project_box_sum(h_cur + v, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.95
    
    return best_h, best_c5

            for x in range(mod):

            for r in residues:

            if base.sum() < mod/2:

                for i in range(int(mod/2 - base.sum())):

        h = np.zeros(n)

        h = project_box_sum(h, ts)

def project_box_sum(y, s):

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
