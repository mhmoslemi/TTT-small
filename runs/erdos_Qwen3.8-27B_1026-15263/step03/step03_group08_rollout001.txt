def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

# More efficient gradient descent with backtracking
def proj_grad_descent(h0, n, target_sum, alpha, n_iters, lr_init=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_and_grad(h, alpha)
        # Project gradient to constraint manifold
        g = g - np.mean(g)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        # Backtracking line search
        g_normed = g / norm_g
        step = lr
        for _ in range(20):
            h_new = np.clip(h - step * g_normed * norm_g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 + 1e-12:
                break
            step *= 0.7
        else:
            h_new = h
            c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
        h = h_new
    return best_h, best_c5

def active_set_descent(h0, n, target_sum, n_iters, lr_init=0.03, n_top=12):
    dx = 2.0 / n
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        g = np.zeros(n)
        for k in top_idx:
            l = k - (n - 1)
            excess = corr[k]
            # Gradient of corr_k with respect to h
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2 + l:hi2 + l]
        # Normalize and project
        g = g - np.mean(g)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        h_new = np.clip(h - lr * g / norm_g * norm_g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.9
        h = h_new
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    all_steps = []
    for s in base_steps:
        for mult in [1.0, 0.5, 0.25]:
            all_steps.append(s * mult)
    for rnd in range(max_rounds):
        improved = False
        for i in range(0, n, 1):
            best_c5_local = c5_cur
            best_delta = 0.0
            for d in all_steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5_local - 1e-14:
                    best_c5_local = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5_local
                improved = True
        if not improved:
            break
    return h, c5_cur

def project_to_sum(h, target_sum):

def smooth_max_and_grad(h, alpha):

        h_cd, c5_cd = coordinate_descent(best_h, best_n, best_n / 2.0, max_rnd, base_steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 18:
        h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, 80000, 400)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if time_left() > 10:
        n_it = min(1500, int(time_left() * 2))
        h_wf3, c5_wf3 = water_filling_descent(best_h, best_n, best_n / 2.0, n_it, 0.008, 8)
        if c5_wf3 < best_c5:
            best_c5 = c5_wf3
            best_h = h_wf3.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
