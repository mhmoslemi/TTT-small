                best_delta = delta_opt
                best_D = D
                best_k = k

            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            adj = diff / np.sum(mask)
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
    return h

def make_binary(h, n):
    """Convert h to binary, maintaining sum = n/2."""
    target = n / 2.0
    current_sum = np.sum(h)
    # Sort by value, set top n/2 to 1, rest to 0
    # But we need exactly n/2 ones
    if abs(current_sum - target) < 1:
        # Close to binary, just threshold
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones == int(target):
            return h_bin
        # Fix by adjusting
        if n_ones > int(target):
            # Need to flip some 1s to 0s (start with smallest values that are 1)
            ones_idx = np.where(h_bin == 1)[0]
            ones_idx_sorted = ones_idx[np.argsort(h[ones_idx])]
            for idx in ones_idx_sorted:
                if n_ones == int(target):
                    break
                h_bin[idx] = 0
                n_ones -= 1
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            zeros_idx_sorted = zeros_idx[np.argsort(-h[zeros_idx])]
            for idx in zeros_idx_sorted:
                if n_ones == int(target):
                    break
                h_bin[idx] = 1
                n_ones += 1
        return h_bin
    else:
        # Not close to binary, use threshold approach
        h_bin = np.zeros(n)
        # Pick n/2 largest values
        idx = np.argsort(h)[::-1][:int(target)]
        h_bin[idx] = 1
        return h_bin

def compute_B_all(h, n):
    """Compute B_all[j, m] = partial corr[m] / partial h[j] for all j, m."""
    j_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2 * n - 1)[None, :]  # (1, 2n-1)
    s = m_idx - (n - 1)  # (1, 2n-1)
    
    pos1 = j_idx - s  # (n, 2n-1)
    pos2 = j_idx + s  # (n, 2n-1)
    
    valid1 = (pos1 >= 0) & (pos1 < n)
    valid2 = (pos2 >= 0) & (pos2 < n)
    
    pos1_c = np.clip(pos1, 0, n - 1)
    pos2_c = np.clip(pos2, 0, n - 1)
    
    B_all = valid1.astype(float) * (1.0 - h[pos1_c]) + valid2.astype(float) * (-h[pos2_c])
    return B_all

def binary_swap_descent(h, corr, n, max_sweeps=100, time_limit=None, t_start=None):
    """Perform binary swap descent. Returns (h, corr, improved_count)."""
    improved_count = 0
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > t_start + time_limit:
            break
        
        B_all = compute_B_all(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Vectorized: find best swap
        # D[j_idx, k_idx] = B_all[zeros[k_idx]] - B_all[ones[j_idx]]
        D = B_all[zeros][None, :, :] - B_all[ones][:, None, :]
        new_maxes = np.max(corr[None, None, :] + D, axis=2)
        
        flat_idx = np.argmin(new_maxes)
        j_idx, k_idx = np.unravel_index(flat_idx, new_maxes.shape)
        min_val = new_maxes[j_idx, k_idx]
        cur_max = np.max(corr)
        
        if min_val >= cur_max - 1e-10:
            break  # No improvement possible
        
        j = ones[j_idx]
        k = zeros[k_idx]
        h[j] = 1.0
        h[k] = 0.0
        corr = corr + D[j_idx, k_idx, :]  # This is B_all[k] - B_all[j]
        improved_count += 1
    
    return h, corr, improved_count

def pair_descent_sweep(h, corr, rng, n, n_cand=10):
    """Continuous pair descent."""
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B_all_single(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        cands = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n]
        for _ in range(n_cand - 3):
            cands.append(rng.randint(n))
        seen = set([j])
        for k in cands:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B_all_single(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            # Binary search for optimal delta
            lo, hi = delta_lo, delta_hi
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if np.max(corr + D * mid) <= np.max(corr + D * lo):
                    lo = mid
                else:
                    hi = mid
            delta_opt = (lo + hi) * 0.5
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def compute_B_all_single(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B
