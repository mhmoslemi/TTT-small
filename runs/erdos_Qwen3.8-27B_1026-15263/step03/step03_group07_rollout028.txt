2. **Key observations about the structure**:

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_iters + 1):
        sm_val, g = fast_sm_and_grad(h, alpha)
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * g * g
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        step = lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_fast(h0, n, target_sum, alpha, maxiter=200):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = fast_sm_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = fast_sm_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def active_lag_sm_and_grad(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w_active = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    w_full = np.zeros(2 * n - 1)
    w_full[active_idx] = w_active
    A = np.empty((n, n))
    for j in range(n):
        A[j, :] = w_full[n - 1 - j: 2 * n - 1 - j]
    g = A.dot(1.0 - h) - A.T.dot(h)
    return sm_val, g

def slsqp_active_fast(h0, n, target_sum, active_idx, alpha, maxiter=200):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = active_lag_sm_and_grad(h, active_idx, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = active_lag_sm_and_grad(h, active_idx, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def get_active_lags(h, threshold=0.99):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= threshold * max_corr)[0]
    return active

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def subgradient_refine(h0, n, target_sum, n_iters, lr_init=0.02):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        active = np.where(corr >= 0.98 * max_c)[0]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        A = np.empty((n, n))
        for j in range(n):
            A[j, :] = w[n - 1 - j: 2 * n - 1 - j]
        g = A.dot(1.0 - h) - A.T.dot(h)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return h_best, c5_cur

def periodic_start(n, period, duty=0.5):
    h = np.zeros(n)
    fill = max(1, int(period * duty))
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return h

def smooth_periodic_start(n, period, width=0.3):
    x = np.arange(n) / (n / float(period))
    phase = x % 1.0
    h = np.zeros(n)
    for i in range(n):
        p = phase[i]
        if p < 0.5 - width / 2 or p > 0.5 + width / 2:
            h[i] = 0.0
        elif p < 0.5:
            h[i] = (p - (0.5 - width / 2)) / (width / 2)
        else:
            h[i] = 1.0 - (p - 0.5) / (width / 2)
    h = np.clip(h, 0.0, 1.0)
    return h

def blurb_adjustment(h0, n, target_sum, n_trials=50, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_trials):
        idx = rng.randint(n - 2)
        h_new = h.copy()
        h_new[idx:idx + 2] = np.clip(h_new[idx:idx + 2] + 0.05 * rng.randn(2), 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    for alpha in [30.0, 80.0, 200.0, 500.0]:
        if time_left() < 30:
            break
        n_it = min(3000, max(500, int(time_left() * 3)))
        h_adam, c5_adam = adam_optimize(h_cur, n, ts, alpha, n_it, lr=0.008)
        if c5_adam < best_c5:
            best_c5 = c5_adam
            best_h = h_adam.copy()
            best_n = n
        h_cur = h_adam

    for alpha in [1000.0, 2000.0, 5000.0, 10000.0]:
        if time_left() < 25:
            break
        n_it = min(4000, max(800, int(time_left() * 4)))
        h_adam, c5_adam = adam_optimize(h_cur, n, ts, alpha, n_it, lr=0.003)
        if c5_adam < best_c5:
            best_c5 = c5_adam
            best_h = h_adam.copy()
            best_n = n
        h_cur = h_adam

    if time_left() > 60:
        active = get_active_lags(h_cur, 0.98)
        if len(active) > 0 and len(active) < 2 * n - 1:
            for alpha_a in [2000.0, 5000.0, 10000.0]:
                if time_left() < 20:
                    break
                h_act = slsqp_active_fast(h_cur, n, ts, active, alpha_a, 300)
                c5_act = compute_c5(h_act)
                if c5_act < best_c5:
                    best_c5 = c5_act
                    best_h = h_act.copy()
                    best_n = n
                h_cur = h_act
                active = get_active_lags(h_cur, 0.97)

    if time_left() > 80:
        h_slsqp = slsqp_fast(h_cur, n, ts, 5000.0, 300)
        c5_slsqp = compute_c5(h_slsqp)
        if c5_slsqp < best_c5:
            best_c5 = c5_slsqp
            best_h = h_slsqp.copy()
            best_n = n
        h_cur = h_slsqp

    if time_left() > 40:
        n_sg = min(5000, int(time_left() * 3))
        h_sg, c5_sg = subgradient_refine(h_cur, n, ts, n_sg, lr_init=0.01)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n
        h_cur = h_sg

    if time_left() > 30:
        h_blur, c5_blur = blurb_adjustment(h_cur, n, ts, 200, rng)
        if c5_blur < best_c5:
            best_c5 = c5_blur
            best_h = h_blur.copy()
            best_n = n
        h_cur = h_blur

    if time_left() > 200:
        n_hi = n * 2
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_cur)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

        for alpha in [500.0, 2000.0, 8000.0]:
            if time_left() < 30:
                break
            n_it = min(3000, max(300, int(time_left() * 2)))
            h_hi_adam, c5_hi_adam = adam_optimize(h_hi, n_hi, ts_hi, alpha, n_it, lr=0.005)
            if c5_hi_adam < best_c5:
                best_c5 = c5_hi_adam
                best_h = h_hi_adam.copy()
                best_n = n_hi
            h_hi = h_hi_adam

        if time_left() > 60:
            active_hi = get_active_lags(h_hi, 0.98)
            if len(active_hi) > 0 and len(active_hi) < 2 * n_hi - 1:
                h_hi_act = slsqp_active_fast(h_hi, n_hi, ts_hi, active_hi, 5000.0, 200)
                c5_hi_act = compute_c5(h_hi_act)
                if c5_hi_act < best_c5:
                    best_c5 = c5_hi_act
                    best_h = h_hi_act.copy()
                    best_n = n_hi
                h_hi = h_hi_act

    if time_left() > 100:
        for period in [3, 4, 5, 6]:
            if time_left() < 20:
                break
            h_per = smooth_periodic_start(n, period, 0.4)
            h_per = project_to_sum(h_per, ts)
            c5_per = compute_c5(h_per)
            if c5_per < best_c5 + 0.05:
                for alpha in [200.0, 1000.0]:
                    if time_left() < 15:
                        break
                    h_per_adam, c5_per_adam = adam_optimize(h_per, n, ts, alpha, 2000, lr=0.005)
                    if c5_per_adam < best_c5:
                        best_c5 = c5_per_adam
                        best_h = h_per_adam.copy()
                        best_n = n
                    h_per = h_per_adam

    if time_left() > 15:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        h_cd, c5_cd = coordinate_descent(h_f, n_f, ts_f, 40, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_f

    if time_left() > 10:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        n_it_final = min(5000, int(time_left() * 5))
        h_final_adam, c5_final = adam_optimize(h_f, n_f, ts_f, 8000.0, n_it_final, lr=0.001)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_adam.copy()
            best_n = n_f

    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)
    else:
        n_final = 720
        best_h = np.full(n_final, 0.5)
        best_c5 = 0.5

    return (best_h, best_c5, n_final)
