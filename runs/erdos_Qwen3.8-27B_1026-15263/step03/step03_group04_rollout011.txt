import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    """Gradient of corr[lag] with respect to h."""
    n = len(h)
    shift = lag - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-12
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        grad += wi * corr_gradient(h, k)
    return obj, grad

def exact_subgrad_step(h, target_sum, lr, top_m=3):
    """One step of subgradient descent with projection."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Get top-m lags
    top_idx = np.argsort(corr)[::-1][:top_m]
    # Average gradient of top-m
    grad = np.zeros(n)
    for k in top_idx:
        grad += corr_gradient(h, k)
    grad /= top_m
    # Step
    h_new = h - lr * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def subgrad_optimize(h0, n, target_sum, max_iters=5000, lr_init=0.01, lr_end=0.0001):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(max_iters):
        frac = it / max_iters
        lr = lr_init * (lr_end / lr_init) ** frac
        h_new = exact_subgrad_step(h, target_sum, lr)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def optimize_lbfgs(h0, n, target_sum, alpha, penalties, maxiter_per_level):
    h = h0.copy()
    for pen in penalties:
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return obj + 0.5 * pen * s_dev * s_dev
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return grad + pen * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_level, 'ftol': 1e-15, 'gtol': 1e-8})
            h = res.x
        except Exception:
            pass
    return project_to_sum(h, target_sum)

def exact_polish(h, n, target_sum, max_rounds=200, top_m=5):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        m = np.max(corr)
        top_k = np.argsort(corr)[::-1][:top_m]
        best_h = h
        best_c5 = c5_cur
        for k_max in top_k:
            grad = corr_gradient(h, k_max)
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        if best_h is h:
            break
        h = best_h
        c5_cur = best_c5
    return h, c5_cur

def binary_hill_climb(h_bin, n, target_sum, rng, max_rounds=200):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try all swaps in a limited random sample
        n_trials = min(5000, len(ones) * len(zeros))
        best_c5 = c5_cur
        best_pair = None
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_test = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if c5_test < best_c5 - 1e-10:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5_cur = best_c5
            improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_optimize(n, target_sum, t0, budget):
    """Optimize using Fourier parameterization."""
    x = np.linspace(0, 2, n, endpoint=False)
    n_fourier = min(n // 2, 40)
    best_c5 = np.inf
    best_h = None
    rng = np.random.RandomState(123)
    
    for n_coeffs in [10, 20, 30, 40]:
        if time.time() - t0 > budget:
            break
        # Random initialization of Fourier coefficients
        for trial in range(3):
            if time.time() - t0 > budget:
                break
            a = rng.randn(n_coeffs + 1) * 0.1
            a[0] = 0.0  # DC component fixed by constraint
            
            def coeff_to_h(coeffs):
                hh = np.zeros(n)
                for m in range(len(coeffs)):
                    if m == 0:
                        hh += coeffs[m]
                    else:
                        hh += coeffs[m] * np.cos(2 * np.pi * m * x / 2.0)
                hh += 0.5
                return hh
            
            h_init = coeff_to_h(a)
            h_init = project_to_sum(h_init, target_sum)
            
            def h_to_coeffs(hh):
                coeffs = np.zeros(len(a))
                for m in range(len(coeffs)):
                    if m == 0:
                        coeffs[m] = np.mean(hh) - 0.5
                    else:
                        coeffs[m] = np.mean(hh * np.cos(2 * np.pi * m * x / 2.0))
                return coeffs
            
            def fun(coeffs):
                hh = coeff_to_h(coeffs)
                hh = project_to_sum(hh, target_sum)
                return compute_c5(hh)
            
            res = minimize(fun, a, method='Nelder-Mead', 
                          options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            h_opt = coeff_to_h(res.x)
            h_opt = project_to_sum(h_opt, target_sum)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:

                best_h = h_opt.copy()
