def fft_correlate(a, b):
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b[::-1], N)
    c = np.fft.irfft(A * B, N)
    return c[:n + m - 1]

def project_to_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
