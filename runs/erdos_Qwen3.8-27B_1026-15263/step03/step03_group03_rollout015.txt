    #   sum_i δi = 0

    A_eq = np.zeros((1, n+1))

        return delta
    return None

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_all(h):
    n = len(h)
    B = np.zeros((n, 2*n-1))
    for m in range(2*n-1):
        l = m - n + 1
        idx = np.arange(n)
        idx1 = idx - l
        valid1 = (idx1 >= 0) & (idx1 < n)
        c1 = np.zeros(n)
        c1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = idx + l
        valid2 = (idx2 >= 0) & (idx2 < n)
        c2 = np.zeros(n)
        c2[valid2] = -h[idx2[valid2]]
        B[:, m] = c1 + c2
    return B

def lp_direction(h, corr, B):

    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    B_all = compute_B_all(h)
    grad = B_all.T @ w
    return obj, grad

def slsqp_polish(h0, n, target_sum, alpha, maxiter):
    def f(hv):
        return smooth_max_and_grad(hv, alpha)[0]
    def g(hv):
        return smooth_max_and_grad(hv, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lp_descent(h, time_limit, alpha_init=0.3, stagnation_limit=30):
    n = len(h)
    corr = compute_corr(h)
    current_max = np.max(corr)
    stagnation = 0
    
    while time.time() < time_limit:
        B = compute_B_all(h)
        delta, lp_val = solve_lp(h, corr, B)
        if delta is None:
            break
        
        if lp_val >= current_max - 1e-10:
            stagnation += 1
            if stagnation >= stagnation_limit:
                break
            # Try a small random perturbation
            noise = 0.01 * np.random.uniform(-1, 1, n)
            h = project_to_sum(h + noise, n / 2.0)
            corr = compute_corr(h)
            current_max = np.max(corr)
            continue
        
        # Line search
        improved = False
        for ls in range(5):
            alpha = alpha_init * (0.5 ** ls)
            if alpha < 1e-8:
                break

            h_try = project_to_sum(h_try, n / 2.0)

            if np.max(corr_try) < current_max - 1e-12:
                h = h_try
                corr = corr_try

                improved = True
                break
        
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= stagnation_limit:
                break
    
    return h, current_max

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j+1) % n, (j-1+n) % n, (j+n//2) % n,
                (j+n//3) % n, (j+2*n//3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    B = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        l = m - n + 1
        idx = np.arange(n)
        idx1 = idx - l
        valid1 = (idx1 >= 0) & (idx1 < n)
        c1 = np.zeros(n)
        if np.any(valid1):
            c1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = idx + l
        valid2 = (idx2 >= 0) & (idx2 < n)
        c2 = np.zeros(n)
        if np.any(valid2):
            c2[valid2] = -h[idx2[valid2]]
        B[:, m] = c1 + c2

    best_in_call = current_max
    
    while time.time() < time_limit:
        B = compute_B_all(h)
        delta = solve_lp(h, corr, B)
        
        if delta is None:
            stagnation += 1
        else:
            # Line search
            improved = False
            for ls in range(5):
                alpha = alpha_init * (0.5 ** ls)
                if alpha < 1e-8:
                    break
                h_try = np.clip(h + alpha * delta, 0.0, 1.0)
                h_try = project_to_sum(h_try, target)
                corr_try = compute_corr(h_try)
                if np.max(corr_try) < current_max - 1e-12:
                    h = h_try
                    corr = corr_try
                    current_max = np.max(corr)
                    if current_max < best_in_call:
                        best_in_call = current_max
                    improved = True
                    break
            if improved:
                stagnation = 0
            else:
                stagnation += 1
        
        if stagnation >= stagnation_limit:
            # Add noise to escape
            if rng is not None:
                noise_scale = min(0.05, 0.01 * stagnation / stagnation_limit)
                h = project_to_sum(h + noise_scale * rng.uniform(-1, 1, n), target)
                corr = compute_corr(h)
                current_max = np.max(corr)
                stagnation = 0
            else:
                break
    
    return h, current_max

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    B_all = compute_B_all(h)
    grad = B_all.T @ w
    return m_val, grad
