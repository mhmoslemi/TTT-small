def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, beta1=0.9, beta2=0.999):

    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    eps = 1e-8
    patience = 0
    for _ in range(n_iters):
        t += 1
        g = _fast_gradient(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g

    v_vec = np.zeros(n)
    v_prev = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    eps = 1e-8
    for _ in range(n_iters):
        t += 1
        v_t = beta1 * v_prev + (1.0 - beta1) * h
        g = _fast_gradient(v_t, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h = v_t - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        v_prev = h - beta1 * v_prev / (1.0 - beta1)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _g_target_sum, _g_n_pts, _g_alpha
    _g_target_sum = target_sum
    _g_n_pts = n
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_slsqp_obj, h0.copy(), jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            l = m - (n - 1)
            if l >= 0:
                if l < n:
                    g[l:n] += (1.0 - h[l:n])
                if n - l > 0:
                    g[0:n - l] -= h[0:n - l]
            else:
                lp = -l
                if lp < n:
                    g[0:n - lp] += (1.0 - h[0:n - lp])
                    g[lp:n] -= h[lp:n]
        lr = lr_init / (1.0 + 0.01 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        scale = max(base_delta * (0.5 ** (r // 5)), base_delta * 0.01)
        deltas = np.array([-3.0, -2.0, -1.0, 1.0, 2.0, 3.0]) * scale
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                nv = hi + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                c5_cur = best_val
                improved = True
            else:
                h[i] = hi
                h = project_to_sum(h, target_sum)
        if not improved:
            break
    return h, c5_cur

def binary_refine(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if h[i] < 0.3 or h[i] > 0.7:
                target_val = 1.0 if h[i] > 0.5 else 0.0
                h[i] = target_val
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[i] = 0.5
                    h = project_to_sum(h, target_sum)
        if not improved:
            break
    return h, c5_cur

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def _interp_to_n(h_src, n_src, n_dst, target_sum_dst):
    if n_src == n_dst:
        return project_to_sum(h_src.copy(), target_sum_dst)
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src, period=2)
    return project_to_sum(h_new, target_sum_dst)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts0 = best_n / 2.0
            best_h = project_to_sum(best_h, ts0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_cand

    n_targets = [360, 720]

    for n in n_targets:
        if _tl(t_start, budget_s) < 80:
            break
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue

        starts = []
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(_interp_to_n(best_h, best_n, n, ts))

        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
            for amp in [0.25, 0.4, 0.5, 0.6]:
                for phase in [0.0, 0.5, 1.0]:
                    hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase * np.pi)
                    starts.append(project_to_sum(hs, ts))

        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_to_sum(hr, ts))

        h_sym = np.zeros(n)
        h_sym[:n//4] = 1.0
        h_sym[3*n//4:] = 1.0
        starts.append(project_to_sum(h_sym, ts))

        n_starts = min(len(starts), 12)
        if n_starts > 12:
            starts = starts[:12]

        h_cur = None
        for si, h_start in enumerate(starts):
            if _tl(t_start, budget_s) < 40:
                break
            h_start = project_to_sum(h_start, ts)
            try_update(h_start, n)
            h_cur = h_start.copy()

            alphas = [20.0, 80.0, 300.0, 1000.0]
            for alpha in alphas:
                if _tl(t_start, budget_s) < 20:
                    break
                n_adam = min(3000, max(500, int(_tl(t_start, budget_s) * 0.8)))
                h_a, c5_a = adam_optimize(h_cur, n, ts, alpha, n_adam, 0.05)
                if c5_a < compute_c5(h_cur):
                    h_cur = h_a
                    try_update(h_cur, n)

            if _tl(t_start, budget_s) < 20:
                break

            maxiter = max(50, min(500, int(_tl(t_start, budget_s) * 0.5)))
            h_s = slsqp_optimize(h_cur, n, ts, 500.0, maxiter)
            try_update(h_s, n)
            h_cur = h_s

            if _tl(t_start, budget_s) < 30:
                break
            n_sub = min(3000, max(500, int(_tl(t_start, budget_s) * 0.5)))
            h_sg, _ = topk_subgradient(h_cur, n, ts, n_sub, 0.04, 8)
            try_update(h_sg, n)
            if compute_c5(h_sg) < compute_c5(h_cur):
                h_cur = h_sg

            if _tl(t_start, budget_s) < 20:
                break
            n_cd = max(10, min(60, int(_tl(t_start, budget_s) / 3.0)))
            h_cd, _ = coord_descent(h_cur, n, ts, n_cd, rng, 0.005)
            try_update(h_cd, n)
            h_cur = h_cd

    if best_h is not None and _tl(t_start, budget_s) > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [500.0, 2000.0, 5000.0]:
            if _tl(t_start, budget_s) < 15:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 600)
            try_update(h_pol, n_f)
            best_h = h_pol

        if _tl(t_start, budget_s) > 20:
            n_adam_f = min(5000, max(1000, int(_tl(t_start, budget_s) * 1.5)))
            h_af, _ = adam_optimize(best_h, n_f, ts_f, 2000.0, n_adam_f, 0.02)
            try_update(h_af, n_f)
            best_h = h_af

        if _tl(t_start, budget_s) > 20:
            n_sub_f = min(5000, max(1000, int(_tl(t_start, budget_s) * 1.0)))
            h_sg, _ = topk_subgradient(best_h, n_f, ts_f, n_sub_f, 0.02, 15)
            try_update(h_sg, n_f)
            best_h = h_sg

        if _tl(t_start, budget_s) > 20:
            n_cd_f = max(20, min(150, int(_tl(t_start, budget_s) / 2.0)))
            h_cd, _ = coord_descent(best_h, n_f, ts_f, n_cd_f, rng, 0.002)
            try_update(h_cd, n_f)
            best_h = h_cd

        if _tl(t_start, budget_s) > 15:
            h_cd2, _ = coord_descent(best_h, n_f, ts_f, 80, rng, 0.0005)
            try_update(h_cd2, n_f)
            best_h = h_cd2

        if _tl(t_start, budget_s) > 10:
            h_br, _ = binary_refine(best_h, n_f, ts_f, 20, rng)
            try_update(h_br, n_f)

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
