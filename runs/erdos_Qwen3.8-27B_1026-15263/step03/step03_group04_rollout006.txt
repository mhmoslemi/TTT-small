def gradient_at_lag(h, k_full):
    """Gradient of C(k_full) w.r.t. h, where k_full is the index in the full correlation."""
    n = len(h)
    shift = k_full - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def subgradient_descent(h0, n, target_sum, max_rounds=5000, lr0=0.01, decay=0.999):

        m = corr[top_idx[0]]

        weights /= weights.sum()

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

        return smooth_max_obj_and_grad(x, alpha)[0]

        return smooth_max_obj_and_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - target_sum
    def cjac(x):
        return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

def optimize_lbfgs_penalty(h0, n, target_sum, alpha, penalty, maxiter=300):

    excess = h.sum() - target_sum
    if excess > 0:
        ones = np.where(h == 1.0)[0]
        for i in range(int(round(excess))):
            h[rng.choice(ones)] = 0.0
    elif excess < 0:
        zeros = np.where(h == 0.0)[0]
        for i in range(int(round(-excess))):
            h[rng.choice(zeros)] = 1.0
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        n_trials = 300
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h_test = h.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            c5_test = compute_c5(h_test)
            if c5_test < c5_cur - 1e-13:
                h = h_test
                c5_cur = c5_test
                improved = True
                break
        if not improved:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            for _ in range(n_trials):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_test = h.copy()
                h_test[i] = 0.0
                h_test[j] = 1.0
                c5_test = compute_c5(h_test)
                if c5_test < c5_cur - 1e-13:
                    h = h_test
                    c5_cur = c5_test
                    improved = True
                    break
            if not improved:
                break
    return h, c5_cur

def make_random_binary(n, target_sum, rng):
    h = np.zeros(n)
    h[:int(round(target_sum))] = 1.0
    rng.shuffle(h)
    return h

def make_sinusoidal(n, freq, phase, amp, target_sum):
    x = np.linspace(0, 2, n, endpoint=False)
    s = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_to_sum(s, target_sum)

def make_blocks(n, nblks, target_sum):
    blk = n // nblks
    h = np.zeros(n)
    for i in range(0, n, 2 * blk):
        h[i:min(i + blk, n)] = 1.0
    return project_to_sum(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    ts = best_n / 2.0

    # Phase 1: SLSQP continuation at current resolution
    for a in [2000, 10000, 50000]:
        if remaining() < 60:
            break
        h_opt = slsqp_optimize(best_h, best_n, ts, a, maxiter=600)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: Subgradient on exact max
    if remaining() > 150:
        h_sg, c5_sg = subgradient_exact(best_h, best_n, ts, max_rounds=8000, lr0=0.006, decay=0.9998)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: Fine exact polish
    if remaining() > 60:
        h_pol, c5_pol = exact_polish(best_h, best_n, ts, max_rounds=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Binary push + combinatorial search
    if remaining() > 200:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_to_sum(h_bin, ts)
        h_bls, c5_bls = binary_push_and_search(h_bin, best_n, ts, rng, max_rounds=400)
        if c5_bls < best_c5:
            best_c5 = c5_bls
            best_h = h_bls.copy()
        # SLSQP from binary
        if remaining() > 60:
            h_opt = slsqp_optimize(best_h, best_n, ts, 10000, maxiter=500)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        # Final polish
        if remaining() > 40:
            h_pol, c5_pol = exact_polish(best_h, best_n, ts, max_rounds=500)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 5: Higher resolution n=480
    if remaining() > 350:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)

        h_sg2, c5_sg2 = subgradient_exact(h_interp, n2, ts2, max_rounds=5000, lr0=0.004, decay=0.9998)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2
            best_n = n2
            ts = ts2
            h_interp = h_sg2

        if remaining() > 150:
            h_opt = slsqp_optimize(best_h, best_n, ts, 5000, maxiter=500)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        if remaining() > 80:
            h_pol, c5_pol = exact_polish(best_h, best_n, ts, max_rounds=500)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 6: Multi-start exploration at best_n
    if remaining() > 200:
        n_s = best_n
        ts_s = n_s / 2.0
        starts = []
        for _ in range(4):
            starts.append(make_random_binary(n_s, ts_s, rng))
        for freq in [2, 3, 4, 6, 8]:
            for ph in [0.0, 0.7, 1.4, 2.1, 3.0]:
                starts.append(make_sinusoidal(n_s, freq, ph, 0.45, ts_s))
        for nb in [4, 6, 8, 10, 12, 16, 20]:
            starts.append(make_blocks(n_s, nb, ts_s))

        for h_s in starts:
            if remaining() < 40:
                break
            c5_s = compute_c5(h_s)
            current = h_s if c5_s < best_c5 else best_h.copy()
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_sg_s, c5_sg_s = subgradient_exact(h_s, n_s, ts_s, max_rounds=2000, lr0=0.005, decay=0.999)
            if c5_sg_s < best_c5:
                best_c5 = c5_sg_s
                best_h = h_sg_s
            if remaining() < 20:
                break

    # Phase 7: Final tight optimization
    if remaining() > 80:
        n_f = best_n
        ts_f = n_f / 2.0
        # L-BFGS-B with high penalty
        for a in [20000, 80000]:
            if remaining() < 40:
                break
            h_opt = lbfgs_penalty_optimize(best_h, n_f, ts_f, a, 500000, maxiter=600)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        # SLSQP final
        if remaining() > 40:
            h_opt = slsqp_optimize(best_h, n_f, ts_f, 50000, maxiter=800)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        # Exact polish
        if remaining() > 30:
            h_pol, c5_pol = exact_polish(best_h, n_f, ts_f, max_rounds=600)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
