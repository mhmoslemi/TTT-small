def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def f(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_grad(h, alpha)[1]
    def f_con(x):
        return np.sum(x) - target_sum
    def g_con(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': f_con, 'jac': g_con}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa_advanced(h_start, n, target_sum, t_end, rng, init_T=0.015, min_T=0.002):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = init_T
    no_improve = 0
    
    while time.time() < t_end:
        move_type = rng.random()
        if move_type < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or (delta / max(T, 1e-10)) + np.log(rng.uniform()) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
        elif move_type < 0.80:
            # Block swap
            block_len = min(rng.randint(2, max(3, n // 30)), n // 5)
            i = rng.randint(n - block_len)
            j = rng.randint(n - block_len)
            if abs(i - j) < block_len:
                continue
            block_i = h[i:i+block_len].copy()
            block_j = h[j:j+block_len].copy()
            if np.array_equal(block_i, block_j):
                continue
            h[i:i+block_len] = block_j
            h[j:j+block_len] = block_i
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or (delta / max(T, 1e-10)) + np.log(rng.uniform()) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i:i+block_len] = block_i
                h[j:j+block_len] = block_j
                no_improve += 1
        else:
            # Reversal
            i = rng.randint(n - 2)
            j = rng.randint(i + 2, n)
            h[i:j+1] = h[i:j+1][::-1]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or (delta / max(T, 1e-10)) + np.log(rng.uniform()) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i:j+1] = h[i:j+1][::-1]
                no_improve += 1
        
        if no_improve > 40000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.025)
            no_improve = 0
        T = max(T * 0.99999, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def f(hv):
        return smooth_max_obj_grad(hv, alpha)[0]
    def g(hv):
        return smooth_max_obj_grad(hv, alpha)[1]
    def f_con(x):
        return np.sum(x) - target_sum
    def g_con(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': f_con, 'jac': g_con}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

                    no_improve += 1
            else:
                h[i:i + blen] = bi
                h[j:j + blen] = bj
                no_improve += 1
        elif r < 0.90:
            i = rng.randint(n - 3)
            j = rng.randint(i + 3, n)
            h[i:j + 1] = h[i:j + 1][::-1]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or (delta / max(T, 1e-10)) + np.log(max(rng.uniform(), 1e-30)) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i:j + 1] = h[i:j + 1][::-1]
                no_improve += 1
        else:
            shift = rng.randint(1, max(2, n // 20))
            h = np.roll(h, shift)
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or (delta / max(T, 1e-10)) + np.log(max(rng.uniform(), 1e-30)) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h = np.roll(h, -shift)
                no_improve += 1
        if no_improve > 35000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, 0.03)
            no_improve = 0
        T = max(T * 0.99998, min_T)
    return best_h, best_c5

def thue_morse_binary(n, rng):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        t[i] = c & 1
    n_ones = int(np.sum(t))
    target = n // 2
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(t == 1.0)[0]
        rng.shuffle(ones_idx)
        t[ones_idx[:excess]] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(t == 0.0)[0]
        rng.shuffle(zeros_idx)
        t[zeros_idx[:deficit]] = 1.0
    return t

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 240
        best_h = thue_morse_binary(best_n, rng)
        best_c5 = compute_c5(best_h)
    n_main = best_n
    ts_main = n_main / 2.0
    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sa_end1 = t_start + 0.35 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), sa_end1, rng, init_T=0.020, min_T=0.003)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    if time.time() < t_start + 0.55 * budget_s:
        for alpha in [50, 300, 1500]:
            if time.time() > t_start + 0.55 * budget_s:
                break
            h_pol = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=600)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    if time.time() < t_start + 0.72 * budget_s:
        h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
        sa_end2 = t_start + 0.72 * budget_s
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, int(ts_main), sa_end2, rng, init_T=0.012, min_T=0.002)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        if time.time() < t_start + 0.78 * budget_s:
            h_pol2 = slsqp_optimize(best_h, n_main, ts_main, 500, maxiter=800)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
    for n_try in [200, 280, 320]:
        if time.time() > t_start + 0.90 * budget_s:
            break
        ts_try = n_try / 2.0
        if best_n == n_try:
            h_try0 = best_h.copy()
        else:
            h_try0 = interpolated_h(best_h, best_n, n_try)
            h_try0 = project_to_sum(h_try0, ts_try)
        h_try_bin = make_binary(h_try0, n_try, int(ts_try), rng)
        h_tm = thue_morse_binary(n_try, rng)
        h_try_bin = h_try_bin.copy()
        c5_try_bin = compute_c5(h_try_bin)
        c5_tm = compute_c5(h_tm)
        if c5_tm < c5_try_bin:
            h_try_bin = h_tm.copy()
            c5_try_bin = c5_tm
        sa_end_n = min(time.time() + 35, t_start + 0.90 * budget_s)
        h_try_sa, c5_try = binary_sa(h_try_bin, n_try, int(ts_try), sa_end_n, rng, init_T=0.020, min_T=0.003)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try_sa.copy()
            best_n = n_try
        if time.time() > t_start + 0.93 * budget_s:
            break
        h_try_cont = slsqp_optimize(h_try_sa, n_try, ts_try, 300, maxiter=500)
        c5_try_cont = compute_c5(h_try_cont)
        if c5_try_cont < best_c5:
            best_c5 = c5_try_cont
            best_h = h_try_cont.copy()
            best_n = n_try
    if time.time() < t_start + 0.97 * budget_s:
        for alpha in [2000, 8000]:
            if time.time() > t_start + 0.97 * budget_s:
                break
            h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=600)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
