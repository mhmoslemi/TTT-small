```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):
    h_bin = np.zeros(n)
    # Sort by value, set top target_ones to 1
    order = np.argsort(-h)
    h_bin[order[:target_ones]] = 1.0
    return h_bin

def sa_binary(h_start, n, target_ones, t_end, rng, use_blocks=False):
    h = h_start.copy()
    corr = compute_corr(h)
    c5_curr = float(np.max(corr) * (2.0 / n))
    best_c5 = c5_curr
    best_h = h.copy()
    
    dx = 2.0 / n
    T = 0.01
    min_T = 0.001
    no_improve = 0
    total_steps = 0
    
    while time.time() < t_end:
        # Choose a move
        if use_blocks and rng.random() < 0.3:
            # Block move: swap a block of 1s with a block of 0s

                j = rng.randint(n - block_len + 1)

                if np.all(h[i:i+block_len] == h[i]) and np.all(h[j:j+block_len] == h[j]) and h[i] != h[j]:
                    h[i:i+block_len], h[j:j+block_len] = h[j:j+block_len].copy(), h[i:i+block_len].copy()
                    new_corr = compute_corr(h)
                    c5_new = float(np.max(new_corr) * dx)

                    if delta < 0 or np.log(rng.uniform()) < delta / T:

                            no_improve = 0

                    found = True

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    n = len(h0)
    
    def smooth_max_val(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        return m_val + np.log(np.sum(exp_vals)) / alpha
```

```python
def single_sa(h_start, n, target_ones, t_end, rng, T_init=0.006, T_min=0.0003):
    h = h_start.copy()
    dx = 2.0 / n
    curr_corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = float(np.max(curr_corr) * dx)
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        c5_new = float(np.max(new_corr) * dx)
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < delta / T:
            curr_corr = new_corr
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        
        if no_improve > 30000:
            h = best_h.copy()
            curr_corr = np.correlate(h, 1.0 - h, mode="full")
            c5_curr = best_c5
            T = min(T * 2.5, 0.012)
            no_improve = 0
        T = max(T * 0.999998, T_min)
    
    return best_h, best_c5

def slsqp_optimize(h0, target_sum, alpha, maxiter=600):
    n = len(h0)
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
        ev = np.exp(shifted)
        return m + np.log(ev.sum()) / alpha
    
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
        ev = np.exp(shifted)
        w = ev / ev.sum()
        
        # Compute gradient using the formula derived earlier
        cw = np.cumsum(w)
        total_w = cw[-1]
        g = np.zeros(n)
        for j in range(n):
            # sum_{m>=j} w[m]
            sum1 = total_w if j == 0 else total_w - cw[j-1]
            # sum_{m=n-1-j}^{2n-2-j} w[m] (clamped to valid range)
            lo = n - 1 - j
            hi = min(2 * n - 2 - j, 2 * n - 2)
            if lo <= hi:
                sum2 = cw[hi] if lo == 0 else cw[hi] - cw[lo-1]
            else:
                sum2 = 0.0
            g[j] = (1.0 - h[j]) * sum1 - h[j] * sum2
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
```

```python
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

def projected_gradient(h0, target_sum, alpha, max_iter=2000, lr=0.001):
    h = h0.copy()

    for it in range(max_iter):
        obj, g = smooth_max_obj_and_grad(h, alpha)
        # Projected gradient step
        h_new = h - lr * g

        if obj_new < obj:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-8:
```

```python
            term2 = -w[lo2:hi2+1].dot(h[idx_lo:idx_hi+1])
        else:
            term2 = 0.0
        
        g[j] = term1 + term2
    return g
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones):
    h_bin = np.zeros(n)
    order = np.argsort(-h)
    h_bin[order[:target_ones]] = 1.0
    return h_bin

def smooth_max_cw(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    ev = np.exp(shifted)
    w = ev / ev.sum()
    obj = m + np.log(ev.sum()) / alpha
    return obj, w

def compute_grad_from_w(h, w):
    N = len(h)
    g = np.zeros(N)
    for j in range(N):
        lo1 = j
        hi1 = min(2*N-2, j+N-1)
        if lo1 <= hi1:
            term1 = (1.0 - h[j]) * w[lo1:hi1+1].sum()
        else:

        lo2 = max(0, N-1-j)
        hi2 = min(2*N-2, N-1+j)
        if lo2 <= hi2:

            idx_hi = j + hi2 - N + 1
            term2 = -w[lo2:hi2+1].dot(h[idx_lo:idx_hi+1])
        else:
            term2 = 0.0
        g[j] = term1 + term2
    return g

    h = h0.copy()

    lr = 0.01 / alpha

        g = compute_grad_from_w(h, w)

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        obj_new, _ = smooth_max_cw(h_new, alpha)
        if obj_new < obj:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
