def project_to_sum(h, target_sum):
    h = h.astype(float).copy()

    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term_a = valid_s * (1.0 - h[sc]) * w[None, :]
    term_b = valid_f * h[fc] * w[None, :]
    grad = np.sum(term_a - term_b, axis=1)
    return obj, grad

def pgd_optimize(h0, n, target_sum, alpha_schedule, max_iter, t_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_obj = smooth_max_and_grad(h, alpha_schedule[0])[0]
    lr = 0.5 / np.max(np.abs(smooth_max_and_grad(h, alpha_schedule[0])[1])) + 1e-8
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        obj, grad = smooth_max_and_grad(h, alpha_schedule[it % len(alpha_schedule)])
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
        # Project gradient to null space of sum constraint
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-10:
            break
        # Adaptive step size
        step = lr / (it + 1) ** 0.5
        h_new = h - step * grad_proj
        h_new = project_to_sum(h_new, target_sum)
        obj_new = compute_c5(h_new)
        if obj_new < compute_c5(h) * (1 - 1e-10):
            h = h_new
            lr = min(lr * 1.2, 1.0)
        else:
            lr = max(lr * 0.7, 1e-6)
    return best_h, best_obj

def make_binary(h, n, target_sum):
    h_bin = np.zeros(n)
    top_idx = np.argsort(-h)[:int(target_sum)]
    h_bin[top_idx] = 1.0
    # Ensure exact sum
    n_ones = int(np.sum(h_bin))
    if n_ones > int(target_sum):
        vals = h[top_idx]
        worst = np.argsort(vals)[:n_ones - int(target_sum)]
        h_bin[top_idx[worst]] = 0.0
    return h_bin

def binary_sa(h_start, n, target_sum_int, t_deadline, rng, block_sizes=(1,2,3,5,8)):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    min_T = 0.002
    no_improve = 0
    max_no_improve = 50000
    
    while time.time() < t_deadline:
        # Choose move type
        if rng.random() < 0.7:
            # Swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            # Block move: take a block and move it
            sz = block_sizes[rng.randint(len(block_sizes))]
            if sz >= n // 2:
                sz = n // 4
            start = rng.randint(0, n - sz)
            block = h[start:start+sz].copy()
            ones_in_block = int(np.sum(block))
            if ones_in_block == 0 or ones_in_block == sz:
                continue
            dest = rng.randint(0, n - sz)
            if dest == start:
                continue
            h[start:start+sz] = h[dest:dest+sz]
            h[dest:dest+sz] = block
            # Fix sum
            total = int(np.sum(h))
            if total != target_sum_int:
                # Revert and do a simple swap
                h[dest:dest+sz] = h[start:start+sz]
                h[start:start+sz] = block
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
                    c5_new = compute_c5(h)
                    delta = c5_new - c5_curr
                    if delta < 0 or np.log(rng.uniform()) < delta / T:
                        c5_curr = c5_new
                        if c5_new < best_c5 - 1e-14:
                            best_c5 = c5_new
                            best_h = h.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        h[i], h[j] = h[j], h[i]
                else:
                    continue
            else:
                c5_new = compute_c5(h)
                delta = c5_new - c5_curr
