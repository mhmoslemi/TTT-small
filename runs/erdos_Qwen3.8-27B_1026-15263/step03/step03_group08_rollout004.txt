                if c_t < best_c - 1e-15:
                    best_c = c_t
                    best_v = np.clip(nv, 0.0, 1.0)
            if best_v != h[i]:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def perturbation_search(h0, n, target_sum, rng, n_trials, scales, slsqp_alpha, maxiter):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for t in range(n_trials):
        scale = scales[t % len(scales)]
        h_p = h0 + rng.normal(0, scale, n)
        h_p = project_to_sum(h_p, target_sum)
        h_opt = slsqp_solve(h_p, n, target_sum, slsqp_alpha, maxiter)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def tl():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: Nesterov + SLSQP at base resolution
    if tl() > 120:
        h_ns, c5_ns = nesterov_smooth_max(best_h, n, ts, 300.0,
                                          min(1000, int(tl() * 0.8)), 0.03)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns.copy()

    for alpha in [100, 300, 800, 2000, 5000, 10000]:
        if tl() < 35:
            break
        mi = max(40, min(400, int(tl() * 0.4)))
        h_o = slsqp_solve(best_h, n, ts, alpha, mi)
        c_o = compute_c5(h_o)
        if c_o < best_c5:
            best_c5 = c_o
            best_h = h_o.copy()
        best_h = project_to_sum(best_h, ts)

    # Phase 2: Penalty method bisection
    if tl() > 100:
        C_lo = max(best_c5 - 0.005, 0.2)
        C_hi = best_c5 + 0.001
        for _ in range(4):
            if tl() < 30:
                break
            C_mid = (C_lo + C_hi) / 2.0
            h_pen = penalty_slsqp(best_h, n, ts, C_mid, 200)
            c_pen = compute_c5(h_pen)
            if c_pen <= C_mid + 0.0002:
                C_hi = C_mid
                if c_pen < best_c5:
                    best_c5 = c_pen
                    best_h = h_pen.copy()
            else:
                C_lo = C_mid
        # Final push at C_lo
        if tl() > 25:
            h_pen = penalty_slsqp(best_h, n, ts, C_lo, 300)
            c_pen = compute_c5(h_pen)
            if c_pen < best_c5:
                best_c5 = c_pen
                best_h = h_pen.copy()

    # Phase 3: Multi-resolution
    candidates = []
    for dn in [-10, -5, 5, 10, 15, 20, 30]:
        nt = best_n + dn
        if nt < 100 or nt % 2 != 0:
            continue
        tst = nt / 2.0
        if abs(tst - round(tst)) > 1e-9:
            continue
        candidates.append((nt, tst))

    for nt, tst in candidates:
        if tl() < 80:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, nt, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, tst)
        for alpha in [500, 2000]:
            if tl() < 25:
                break
            mi = max(30, min(250, int(tl() * 0.3)))
            h_o = slsqp_solve(h_interp, nt, tst, alpha, mi)
            c_o = compute_c5(h_o)
            if c_o < best_c5:
                best_c5 = c_o
                best_h = h_o.copy()
                best_n = nt
            h_interp = project_to_sum(h_o, tst)

    # Phase 4: Basin hopping with tiny perturbations
    if tl() > 120:
        scales = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02]
        n_trials = min(80, int(tl() / 4))
        h_bh, c5_bh = perturbation_search(best_h, best_n, best_n / 2.0,
                                           rng, n_trials, scales, 2000, 150)
        if c5_bh < best_c5:
            best_c5 = c5_bh
            best_h = h_bh.copy()

    # Phase 5: Cross-resolution mixing
    if tl() > 80:
        for _ in range(min(10, int(tl() // 8))):
            nt = best_n + rng.choice([-8, -4, 4, 8])
            if nt < 100 or nt % 2 != 0:
                continue
            tst = nt / 2.0
            if abs(tst - round(tst)) > 1e-9:
                continue
            h_cross = multi_res_crossover(best_h, best_n, best_h, best_n, nt, rng)
            h_cross = project_to_sum(h_cross, tst)
            for alpha in [1000, 3000]:
                if tl() < 20:
                    break
                h_o = slsqp_solve(h_cross, nt, tst, alpha, 150)
                c_o = compute_c5(h_o)
                if c_o < best_c5:
                    best_c5 = c_o
                    best_h = h_o.copy()
                    best_n = nt
                h_cross = project_to_sum(h_o, tst)

    # Phase 6: Final polish
    n_f = best_n
    ts_f = n_f / 2.0
    if tl() > 30:
        h_f = slsqp_solve(best_h, n_f, ts_f, 8000, 600)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    if tl() > 25:
        h_f = slsqp_solve(best_h, n_f, ts_f, 15000, 400)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    if tl() > 20:
        steps_cd = [0.002, 0.001, -0.002, -0.001, 0.0005, -0.0005]
        h_cd, c_cd = coord_descent(best_h, n_f, ts_f,
                                    min(200, int(tl() * 0.5)), steps_cd)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    if tl() > 15:
        h_ns2, c5_ns2 = nesterov_smooth_max(best_h, n_f, ts_f, 5000.0,
                                             min(500, int(tl() * 1.5)), 0.01)
        if c5_ns2 < best_c5:
            best_c5 = c5_ns2
            best_h = h_ns2.copy()

    # Final constraint check
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
