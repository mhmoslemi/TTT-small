def grad_component(h, j, n):
    """Compute the vector of gradient contributions of position j to all shifts.
    Returns a vector of length 2n-1 where index k corresponds to shift k-(n-1).
    g_j(s) = (1-h[j+s]) if 0<=j+s<n, minus h[j-s] if 0<=j-s<n
    """
    g = np.zeros(2*n - 1)
    # For shift s = k - (n-1):
    # Contribution from h[j] as first factor: (1-h[j+s]) if 0 <= j+s < n
    #   j+s = j + k - (n-1), need 0 <= j+k-(n-1) < n
    #   => n-1-j <= k <= n-1-j+n-1 = 2n-2-j
    #   => k in [n-1-j, 2n-2-j]
    #   contribution: 1 - h[j + k - (n-1)]
    
    # Contribution from h[j] as second factor (via h[i+s]=h[j], so i=j-s):
    #   -h[j-s] if 0 <= j-s < n
    #   j-s = j - k + (n-1), need 0 <= j-k+(n-1) < n
    #   => j+(n-1)-n < k <= j+(n-1)
    #   => j-1 < k <= j+n-1
    #   => k in [j, j+n-1] (since k is integer)
    #   contribution: -h[j - k + (n-1)]
    
    # First contribution: k in [n-1-j, 2n-2-j]
    k_start = n - 1 - j
    k_end = 2 * n - 2 - j  # inclusive
    if k_start <= k_end and k_start >= 0 and k_end < 2*n-1:
        idx = np.arange(k_start, k_end + 1)
        h_idx = j + idx - (n - 1)  # = j+s for each s
        g[idx] += (1.0 - h[h_idx])
    
    # Second contribution: k in [j, j+n-1]
    k_start2 = j
    k_end2 = j + n - 1
    if k_start2 >= 0 and k_end2 < 2*n-1 and k_start2 <= k_end2:
        idx2 = np.arange(k_start2, min(k_end2 + 1, 2*n-1))
        h_idx2 = j - idx2 + (n - 1)  # = j-s for each s
        g[idx2] -= h[h_idx2]
    
    return g
