def smooth_max_grad_sparse(h, alpha, n_top=50):
    """Compute gradient of smooth max using only top-n_top lags."""

    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)

    top_idx = np.argsort(w)[-n_top:]
    top_w = w[top_idx]

            grad[k:] -= wm * h[:n - k]
    
    return obj, grad

        if time_left() > 45:
            h_cd, c5_cd = multi_lag_coord_descent(h_interp, ts_tr, max_iters=8000, top_k=3, step_frac=0.1)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr

    # Phase 5: Random restarts with local optimization
    n_best = best_n
    ts_best = n_best / 2.0
    for trial in range(10):
        if time_left() < 45:
            break
        noise = 0.015 + 0.025 * (1.0 - trial / 10.0)
        h_shake = best_h + rng.uniform(-noise, noise, n_best)
        h_shake = project_to_sum(h_shake, ts_best)
        h_opt = proj_grad_nesterov(h_shake, ts_best, 200, n_iters=600, lr=0.05, momentum=0.8)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time_left() > 35:
            h_cd, c5_cd = coordinate_descent(h_opt, ts_best, max_iters=4000, step_frac=0.1)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 6: High-alpha polishing
    for alpha in [3000, 8000, 20000]:
        if time_left() < 25:
            break
        h_pol = proj_grad_nesterov(best_h, ts_best, alpha, n_iters=1500, lr=0.03, momentum=0.85)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Final coordinate descent
    if time_left() > 25:
        h_final, c5_final = multi_lag_coord_descent(best_h, ts_best, max_iters=15000, top_k=5, step_frac=0.06)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if time_left() > 15:
        h_final2, c5_final2 = coordinate_descent(best_h, ts_best, max_iters=10000, step_frac=0.04)
        if c5_final2 < best_c5:
            best_c5 = c5_final2
            best_h = h_final2.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
