        grad += w[k] * g_k
    return grad

def nesterov_pgd(h0, target_sum, alpha_sm, n_steps, time_limit, k_top=None):
    n = len(h0)
    h_curr = h0.copy()
    h_prev = h0.copy()
    t0 = time.time()
    beta_nest = 1.0
    step_size = 0.02
    
    f_curr = smooth_max_val(h_curr, alpha_sm)
    
    for k in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        beta_nest = (1.0 + np.sqrt(1.0 + 4.0 * beta_nest * beta_nest)) / 2.0
        momentum_coeff = (beta_nest - 1.0) / beta_nest
        y = h_curr + momentum_coeff * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        if k_top is not None and k_top < 2 * n - 1:
            grad = topk_subgradient(y, k_top)
        else:
            grad = full_smooth_max_grad(y, alpha_sm)
            grad -= np.mean(grad)
        
        h_new = y - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        f_new = smooth_max_val(h_new, alpha_sm)
        
        if f_new < f_curr - 1e-15:
            h_prev = h_curr
            h_curr = h_new
            f_curr = f_new
            step_size = min(step_size * 1.02, 0.1)
        else:
            step_size *= 0.7
            if step_size < 1e-10:
                break
    
    return h_curr, compute_c5(h_curr)

def lbfgs_optimize(h0, target_sum, alpha, maxiter, time_limit):
    n = len(h0)
    t0 = time.time()
    lam = 1e6
    
    def obj_func(x):
        return smooth_max_val(x, alpha) + lam * (np.sum(x) - target_sum) ** 2
    
    def grad_func(x):
        g = full_smooth_max_grad(x, alpha)
        g += 2.0 * lam * (np.sum(x) - target_sum)
        return g
    
    h_work = h0.copy()
    try:
        iterations_run = [0]
        max_iter_allowed = maxiter
        while iterations_run[0] < max_iter_allowed:
            if time.time() - t0 > time_limit:
                break
            chunk = min(200, max_iter_allowed - iterations_run[0])
            res = minimize(obj_func, h_work, jac=grad_func, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': chunk, 'ftol': 1e-18, 'gtol': 1e-14})
            h_work = res.x
            iterations_run[0] += chunk
    except Exception:
        pass
    h_work = project_to_sum(h_work, target_sum)
    return h_work

def coordinate_descent(h, target_sum, max_iters, step_frac, time_limit=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    sf = step_frac
    stale = 0
    
    for it in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = lag_gradient(h, lag)
        
        can_inc = h < 1.0 - 1e-12
        can_dec = h > 1e-12
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-13:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = sf * max_move
        if move < 1e-13:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            sf *= 0.9
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
            sf = min(sf * 1.01, step_frac)
    
    return h, c5

def fourier_perturb(h, rng, n_modes, amplitude):
    n = len(h)
    x = np.arange(n)
    perturbation = np.zeros(n)
    for m in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        perturbation += a * np.sin(2.0 * np.pi * m * x / n) + b * np.cos(2.0 * np.pi * m * x / n)
    return h + perturbation

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def multi_start_basin(h_best, ts, n, rng, n_trials, time_per_trial, time_limit_total):
    t0 = time.time()
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    
    for trial in range(n_trials):
        if time.time() - t0 > time_limit_total:
            break
        remaining = time_limit_total - (time.time() - t0)
        if remaining < time_per_trial:
            break
        
        decay = 1.0 - float(trial) / float(n_trials)
        amp = 0.02 + 0.06 * decay
        n_modes = 2 + int(4 * decay)
        
        h_pert = fourier_perturb(best_h, rng, n_modes, amp)
        h_pert = project_to_sum(h_pert, ts)
        c5_pert = compute_c5(h_pert)
        
        sub_time = min(15, remaining - 10)
        if sub_time > 5:
            h_opt = lbfgs_optimize(h_pert, ts, 2000, 400, sub_time)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        if time.time() - t0 < time_limit_total - 10:
            sub_time2 = min(10, time_limit_total - (time.time() - t0) - 5)
            if sub_time2 > 3:
                h_cd, c5_cd = coordinate_descent(h_opt, ts, 3000, 0.1, sub_time2)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    ts = n / 2.0
    
    # Phase 1: Nesterov PGD with low alpha (broad smoothing)
    if time_left() > 60:
        h1, c5_1 = nesterov_pgd(best_h, ts, 80, 5000, min(40, time_left() - 30), k_top=15)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    if time_left() > 40:
        h1b, c5_1b = nesterov_pgd(best_h, ts, 300, 3000, min(30, time_left() - 20), k_top=8)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()
    
    # Phase 2: L-BFGS-B with medium alpha
    for alpha in [500, 2000, 8000]:
        if time_left() < 30:
            break
        sub_time = min(40, time_left() - 15)
        if sub_time < 5:
            break
        h_opt = lbfgs_optimize(best_h, ts, alpha, 600, sub_time)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Coordinate descent refinement
    if time_left() > 50:
        h_cd, c5_cd = coordinate_descent(best_h, ts, 15000, 0.12, min(50, time_left() - 30))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: Basin hopping
    if time_left() > 120:
        h_b, c5_b = multi_start_basin(best_h, ts, n, rng, 25, 18, min(200, time_left() - 60))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Phase 5: Higher resolution
    for n_tr in [600, 720]:
        if time_left() < 100:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
            n = n_tr
            ts = ts_tr
        
        if time_left() > 30:
            h_opt = lbfgs_optimize(h_interp, n, ts, 1500, min(35, time_left() - 20))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                h_interp = h_opt
        
        if time_left() > 20:
            h_cd2, c5_cd2 = coordinate_descent(h_interp, ts, 8000, 0.1, min(20, time_left() - 15))
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n
    
    # Phase 6: Nesterov PGD with high alpha (fine smoothing)
    n = best_n
    ts = n / 2.0
    if time_left() > 40:
        h_ng, c5_ng = nesterov_pgd(best_h, ts, 5000, 5000, min(35, time_left() - 20), k_top=5)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()
    
    # Phase 7: Final L-BFGS-B polish
    if time_left() > 20:
        h_pol = lbfgs_optimize(best_h, ts, 30000, 1000, min(20, time_left() - 10))
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 8: Final coordinate descent
    if time_left() > 15:
        h_final, c5_final = coordinate_descent(best_h, ts, 20000, 0.05, min(15, time_left() - 5))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n)
