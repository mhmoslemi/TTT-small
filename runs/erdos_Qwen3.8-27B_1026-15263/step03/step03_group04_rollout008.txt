import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(h, k):
    """Gradient of g_k(h) = sum_i h[i]*(1-h[i+k]) w.r.t. h"""
    n = len(h)
    shift = k - (n - 1)  # convert to numpy.correlate convention
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def project_onto_simplex(h, target_sum):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target_sum}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Adjust only the interior points
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)

def optimize_lbfgs_smooth(h0, n, target_sum, alpha, maxiter=500):

        s = np.sum(exp_v)
        obj = m + np.log(s) / alpha
        s_dev = np.sum(x) - target_sum

            grad[:n-s] -= wp * h[s:]
        else:
            grad[:n+s] -= wp * h[-s:]
            grad[-s:] -= wp * h[:n+s]
    return grad

def smooth_max_val(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exp_v)) / alpha

def frank_wolfe(h0, target, max_iter=2000, alpha=200):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        grad = smooth_max_grad(h, alpha)
        # Solve: min grad·y s.t. sum(y)=target, 0≤y≤1
        # Solution: set y_i=1 for smallest grad values, y_i=0 for largest, one fractional
        idx = np.argsort(grad)
        y = np.zeros(n)
        rem = target
        for i in idx:
            if rem >= 1.0 - 1e-10:
                y[i] = 1.0
                rem -= 1.0
            else:
                y[i] = max(0, rem)
                rem = 0.0
                break
        # Step size: standard FW is 2/(t+3), or we can use backtracking
        step = 2.0 / (it + 3)
        h_new = (1.0 - step) * h + step * y
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def lbfgs_optimize(h0, target, alpha, maxiter=500):
    n = len(h0)
    def fun(x):
        return smooth_max_val(x, alpha) + 500.0 * (np.sum(x) - target)**2
    def jac(x):
        g = smooth_max_grad(x, alpha)
        g += 1000.0 * (np.sum(x) - target) * np.ones(n)
        return g
    try:
        res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter': maxiter, 'ftol':1e-15, 'gtol':1e-8})
        return project_sum(res.x, target)
    except:
        return h0.copy()

def binary_sa(h0, target, rng, max_rounds=500, temp_init=0.004):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    # Fix sum
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            zeros = np.where(h < 0.5)[0]
            if len(zeros) == 0: break
            h[rng.choice(zeros)] = 1.0
        else:
            ones = np.where(h > 0.5)[0]
            if len(ones) == 0: break
            h[rng.choice(ones)] = 0.0
    if abs(h.sum() - target) > 1e-10:
        return h0.copy(), compute_c5(h0)
    
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    temp = temp_init
    
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        n_trials = min(1000, len(ones) * len(zeros))
        best_delta = 0
        best_move = None
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            c5_t = compute_c5(h_test)
            delta = c5_t - c5_cur
            if delta < -best_delta:
                best_delta = -delta
                best_move = (i, j)
            elif temp > 1e-10 and delta > 0:
                if rng.random() < np.exp(-delta / temp):
                    best_move = (i, j)
                    best_delta = -delta
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            improved = True
        
        if not improved:
            temp *= 0.9
            if temp < 1e-10:
                break
    
    return best_h, best_c5

def triple_swap_search(h, target, rng, max_rounds=100):
    """Try swapping 1s and 0s in triples (2 ones for 2 zeros or 1 for 1)"""
    n = len(h)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_trials = min(500, len(ones)*len(zeros))
        for _ in range(n_trials):
            i1 = rng.choice(ones)
            i2 = rng.choice(ones)
            if i1 == i2: continue
            j1 = rng.choice(zeros)
            j2 = rng.choice(zeros)
            if j1 == j2: continue
            h_test = h.copy()
            h_test[i1] = 0.0
            h_test[i2] = 0.0
            h_test[j1] = 1.0
            h_test[j2] = 1.0
            c5_t = compute_c5(h_test)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_test.copy()
                h = h_test.copy()
                c5_cur = c5_t
                improved = True
        if not improved:
            break
    return best_h, best_c5

def diverse_starts(n, target, rng, base_h):
    starts = [base_h.copy()]
    x = np.linspace(0, 2, n, endpoint=False)
    # Sine-based
    for f in range(1, 12):
        for a in [0.3, 0.4, 0.5, 0.6]:
            for ph in [0, 1, 2, 3]:
                s = 0.5 + a * np.sin(2*np.pi*f*x/2.0 + ph*np.pi/4)
                starts.append(project_sum(s, target))
    # Block-based
    for nb in range(2, 15):
        blk = n // nb
        for off in range(min(blk, nb)):
            h = np.zeros(n)
            for i in range(0, n, 2*blk):
                h[(i+off) % n : (i+off+blk) % n] = 1.0
            # Handle wrap-around
            h2 = np.zeros(n)
            for i in range(0, n, 2*blk):
                start = (i+off) % n
                end = start + blk
                if end <= n:
                    h2[start:end] = 1.0
                else:
                    h2[start:] = 1.0
                    h2[:end-n] = 1.0
            starts.append(project_sum(h2, target))
    # Random
    for _ in range(5):
        starts.append(project_sum(rng.uniform(0,1,n), target))
    return starts
