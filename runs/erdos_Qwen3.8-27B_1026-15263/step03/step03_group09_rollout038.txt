        else:

        h = np.clip(h, 0.0, 1.0)

def fourier_to_grid(coeffs, n):
    """Convert Fourier coefficients to grid values.
    coeffs = [a0, a1, b1, a2, b2, ..., aK, bK]
    h(x) = a0/2 + sum_{k=1}^{K} [a_k cos(pi*k*x) + b_k sin(pi*k*x)]
    On grid: x_i = 2*i/n, so pi*k*x_i = pi*k*2*i/n = 2*pi*k*i/n
    """
    x = np.arange(n)
    K = (len(coeffs) - 1) // 2
    h = np.full(n, coeffs[0] / 2.0)
    for k in range(1, K + 1):
        a_k = coeffs[2 * k - 1]
        b_k = coeffs[2 * k]
        angle = 2.0 * np.pi * k * x / n
        h += a_k * np.cos(angle) + b_k * np.sin(angle)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')

            lo = mid

def fourier_init(rng, n, K):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = rng.uniform(0.03, 0.3) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        angle = 2.0 * np.pi * k * x / n
        h += amp * np.cos(phase) * np.cos(angle) + amp * np.sin(phase) * np.sin(angle)
    return project_simplex_box(np.clip(h, 0, 1), n / 2.0)

def coord_descent_adaptive(h0, n, target_sum, max_rounds, rng, t_start, deadline, base_delta):

            break

        idx_order = rng.permutation(n)
        for i in idx_order:

                return h, c5_cur

            best_h_new = None

                c5_cur = best_val
