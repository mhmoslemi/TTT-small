    h = h.copy()
    target_sum = n / 2.0
    assert abs(h.sum() - target_sum) < 1e-10
    
    for step in range(max_steps):
        c = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        
        # Find bad pairs for this lag
        shift = n - 1
        bad_pairs = []
        for i in range(n):
            j = i + max_idx - shift
            if 0 <= j < n and h[i] > 0.5 and h[j] < 0.5:
                bad_pairs.append((i, j))
        
        if not bad_pairs:
            break
        
        best_improvement = 0
        best_pair = None
        c_max = max_val
        
        for (i, j) in bad_pairs:
            # Compute new correlation
            h[i] = 0
            h[j] = 1
            c_new = np.correlate(h, 1-h, mode="full")
            new_max = c_new.max()
            h[i] = 1
            h[j] = 0

        h[j] = 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def thue_morse_binary(n, rng):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng.shuffle(ones)
    rng.shuffle(zeros)
    target = n // 2
    curr = len(ones)
    if curr > target:
        for i in range(curr - target):
            h[ones[i]] = 0
    elif curr < target:
        for i in range(target - curr):
            h[zeros[i]] = 1
    return h

def van_der_corput_binary(n, rng):
    h = np.zeros(n)
    for i in range(n):
        denom = 1
        num = i
        val = 0.0
        while num > 0:
            denom *= 2
            val += (num % 2) / denom
            num //= 2
        h[i] = 1.0 if val < 0.5 else 0.0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng.shuffle(ones)

    return h

def systematic_binary_search(h, n, max_steps=2000, time_fn=None):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    curr = len(ones)
    if curr > target:
        for i in range(curr - target):
            h[ones[i]] = 0
    elif curr < target:
        for i in range(target - curr):
            h[zeros[i]] = 1
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    for step in range(max_steps):
        if time_fn and time_fn() < 15:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(c))
        max_val = c[max_idx]
        shift = n - 1
        bad_pairs = []
        for i in range(n):
            j = i + max_idx - shift
            if 0 <= j < n:
                if h[i] > 0.5 and h[j] < 0.5:
                    bad_pairs.append((i, j))
        if not bad_pairs:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        best_imp = 0.0
        best_pair = None
        for (i, j) in bad_pairs:
            h[i] = 0.0
            h[j] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode="full")
            new_max = c_new.max()
            h[i] = 1.0
            h[j] = 0.0
            imp = max_val - new_max
            if imp > best_imp:
                best_imp = imp
                best_pair = (i, j)
        if best_pair is not None and best_imp > 0:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 100:
                break
    return best_h, best_c5

def smooth_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    s = np.exp(alpha * shifted - alpha * np.max(shifted))
    smax = np.sum(s)
    obj = (m + np.log(smax) / alpha) * dx
    return obj

def smooth_max_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    s = np.exp(alpha * shifted - alpha * np.max(shifted))
    w = s / np.sum(s)
    obj = (m + np.log(np.sum(s)) / alpha) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))

def find_lambda(V, target_sum):
    n = len(V)
    lo = V.min() - 15.0
    hi = V.max() + 15.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(sigmoid(V - mid))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def V_to_h(V, n):
    lam = find_lambda(V, n / 2.0)
    return sigmoid(V - lam)

def sigmoid_obj(V, n, alpha):
    h = V_to_h(V, n)
    return smooth_max_c5(h, alpha)

def sigmoid_grad(V, n, alpha):
    h = V_to_h(V, n)
    obj, grad_h = smooth_max_c5_grad(h, alpha)
    dw = h * (1.0 - h)
    grad_V = grad_h * dw
    return obj, grad_V

def projected_grad_descent(h0, n, alpha, max_iter, lr, time_fn):
    h = h0.copy()
    best_obj = smooth_max_c5(h, alpha)
    best_h = h.copy()
    lr_cur = lr
    for it in range(max_iter):
        if time_fn and time_fn() < 10:
            break
        obj, grad = smooth_max_c5_grad(h, alpha)
        h_new = h - lr_cur * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = smooth_max_c5(h_new, alpha)
        if obj_new < obj:
            lr_cur = min(lr_cur * 1.05, 1.0)
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h_new.copy()
        else:
            lr_cur *= 0.5
            h_new = h - lr_cur * grad
            h_new = project_box_sum(h_new, n / 2.0)
            obj_new = smooth_max_c5(h_new, alpha)
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_obj

def block_move_annealing(h, n, max_steps=3000, T_init=0.03, T_final=0.0005, time_fn=None):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    curr = len(ones)
    if curr > target:
        for i in range(curr - target):
            h[ones[i]] = 0
    elif curr < target:
        for i in range(target - curr):
            h[zeros[i]] = 1
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    block_sizes = [1, 2, 3]
    for step in range(max_steps):
        if time_fn and time_fn() < 15:
            break
        progress = step / max_steps
        T = T_init * (T_final / T_init) ** progress
        bs = block_sizes[np.random.randint(len(block_sizes))]
        i = np.random.randint(0, max(1, n - bs + 1))
        j = np.random.randint(0, max(1, n - bs + 1))
        block_i = np.arange(i, i + bs)
        block_j = np.arange(j, j + bs)
        valid_i = (h[block_i] > 0.5)
        valid_j = (h[block_j] < 0.5)
        n_flip_i = np.sum(valid_i)
        n_flip_j = np.sum(valid_j)
        n_flip = min(n_flip_i, n_flip_j)
        if n_flip == 0:
            continue
        block_i = block_i[valid_i][:n_flip]
        block_j = block_j[valid_j][:n_flip]
        h[block_i] = 0.0
        h[block_j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[block_i] = 1.0
            h[block_j] = 0.0
    return best_h, best_c5

            if c5_8 < best_c5:
                best_c5 = c5_8
                best_h = h8.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
