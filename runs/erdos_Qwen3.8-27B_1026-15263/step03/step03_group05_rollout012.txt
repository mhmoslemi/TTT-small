def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def get_worst_shifts(corr, n, threshold_frac=1.0):
    # Get shifts near the maximum
    max_val = np.max(corr)
    thresh = max_val * threshold_frac
    worst = np.where(corr >= thresh)[0]
    # Convert to shift indices (0 to n-1)
    # corr has length 2n-1, index k corresponds to shift k-(n-1)
    # shifts go from -(n-1) to (n-1)
    return worst - (n-1), max_val

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_step(h, corr, n, top_k=5, lr=0.01):
    # Find top-k worst shifts
    max_val = np.max(corr)
    thresh = max_val * 0.95  # shifts within 5% of max
    worst_idx = np.where(corr >= thresh)[0]
    if len(worst_idx) == 0:
        worst_idx = np.array([np.argmax(corr)])
    # Limit to top_k
    if len(worst_idx) > top_k:
        order = np.argsort(corr[worst_idx])[::-1][:top_k]
        worst_idx = worst_idx[order]
    
    dx = 2.0 / n
    sg = np.zeros(n)
    for k_shift in worst_idx:
        # k_shift is index in corr array (0 to 2n-2)
        # actual shift = k_shift - (n-1)
        k = k_shift - (n - 1)
        # sg[j] += (1 - h[(j+k)%n]) - h[(j-k)%n]
        sg += (1.0 - np.roll(h, -k)) - np.roll(h, k)
    
    sg *= dx / len(worst_idx)
    return sg

def coordinate_descent_worst(h, n, max_rounds, time_fn):
    """For the worst shift, do coordinate descent to minimize it."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for round_ in range(max_rounds):

        # h[j]*(1 - h[(j+k)%n] - h[(j-k)%n]) + h[(j-k)%n]
        # To minimize corr[k], we want h[j] = 0 if coeff > 0, h[j] = 1 if coeff < 0
        h_k_plus = np.roll(h, -k)  # h[(j+k) % n]
        h_k_minus = np.roll(h, k)   # h[(j-k) % n]

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def anneal_improve(h, n, rng, max_iter, T_init, T_end, time_fn):
    """Simulated annealing with point and block moves."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    T = T_init
    
    for it in range(max_iter):
        if time_fn() < 5:
            break
        # Linear cooldown
        T = T_init * (T_end / T_init) ** (it / max_iter)
        
        # Random move
        move_type = rng.randint(3)
        h_trial = h.copy()
        
        if move_type == 0:
            # Point flip (change one value)
            i = rng.randint(n)
            h_trial[i] = 1.0 - h_trial[i]
        elif move_type == 1:
            # Pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        else:
            # Block shift
            i = rng.randint(n)
            length = rng.randint(1, max(2, n // 16))
            j = rng.randint(n)
            block = h_trial[np.arange(i, i + length) % n].copy()
            h_trial[np.arange(i, i + length) % n] = h_trial[np.arange(j, j + length) % n]
            h_trial[np.arange(j, j + length) % n] = block
        
        h_trial = project_box_sum(h_trial, n / 2.0)
        c5_trial = compute_c5(h_trial)
        
        # Acceptance
        delta = c5_trial - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_trial
            c5_cur = c5_trial
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def subgradient_at_worst(h, corr, n, top_k=3, thresh_frac=0.95):
    max_val = np.max(corr)
    thresh = max_val * thresh_frac
    worst_idx = np.where(corr >= thresh)[0]
    if len(worst_idx) == 0:
        worst_idx = np.array([np.argmax(corr)])
    if len(worst_idx) > top_k:
        order = np.argsort(corr[worst_idx])[::-1][:top_k]
        worst_idx = worst_idx[order]
    dx = 2.0 / n
    sg = np.zeros(n)
    for k_idx in worst_idx:
        k = k_idx - (n - 1)
        sg += (1.0 - np.roll(h, -k)) - np.roll(h, k)
    sg *= dx / len(worst_idx)
    return sg

def nesterov_subgradient_optimize(h0, n, max_iter, lr_init, time_fn, top_k=3, momentum=0.9):
    h = h0.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if time_fn() < 5:
            break
        h_y = h + momentum * v
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = get_corr_scaled(h_y)
        sg = subgradient_at_worst(h_y, corr, n, top_k=top_k)
        v = momentum * v - lr * sg
        h_new = h + v
        h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > compute_c5(h) + 1e-8:
            lr = max(lr * 0.5, 1e-6)
            h_new = h - lr * sg
            h_new = project_box_sum(h_new, n / 2.0)
            c5_new = compute_c5(h_new)
            v = np.zeros(n)
        else:
            lr = min(lr * 1.02, 0.1)
        h = h_new
    return best_h, best_c5

def frank_wolfe_optimize(h0, n, max_iter, time_fn, binary_search_iters=20):

def lbfgs_smooth_optimize(h0, n, alpha, max_iter, time_fn):
    def obj_and_grad(c):
        h = np.clip(c, 0.0, 1.0)
        h = project_box_sum(h, n / 2.0)
        o, g = smooth_max_obj_and_grad(h, alpha)
        return o, g
    res = minimize(
        lambda c: obj_and_grad(c)[0],

    )
    h_opt = project_box_sum(res.x, n / 2.0)
    return h_opt, compute_c5(h_opt)
