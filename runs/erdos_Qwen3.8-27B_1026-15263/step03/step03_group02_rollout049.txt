def slsqp_obj_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo > i_hi:
            continue
        i_r = np.arange(i_lo, i_hi + 1)
        j_r = i_r + lag
        grad[j_r] += w[k] * (1.0 - h[i_r])
        grad[i_r] -= w[k] * h[j_r]
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def obj_func(x):
        return slsqp_obj_grad(x, alpha)[0]
    def grad_func(x):
        return slsqp_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa(h_start, n, target_sum, t_end, rng, max_no_improve=25000, cool_rate=0.9998):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.018
    no_improve = 0
    min_T = 0.0015
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    while time.time() < t_end:
        move_type = rng.rand()
        if move_type < 0.55:
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h[i], h[j] = h[j], h[i]
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        elif move_type < 0.80:
            blen = rng.randint(2, 5)
            i = rng.randint(0, n - blen - 1)
            j = rng.randint(0, n - blen - 1)
            if abs(i - j) < blen:
                continue
            bi = h[i:i + blen].astype(int)
            bj = h[j:j + blen].astype(int)
            if bi.sum() != bj.sum():
                continue
            h[i:i + blen], h[j:j + blen] = h[j:j + blen], h[i:i + blen]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i:i + blen], h[j:j + blen] = h[i:i + blen].copy(), h[j:j + blen].copy()
        else:
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            k1 = (i + 1) % n
            k2 = (j - 1) % n
            h[i], h[j] = h[j], h[i]
            if h[k1] != h[k2]:
                h[k1], h[k2] = h[k2], h[k1]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
                if h[k1] != h[k2]:
                    h[k1], h[k2] = h[k2], h[k1]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if no_improve > max_no_improve:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.025)
            no_improve = 0
        T = max(T * cool_rate, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return budget_s - elapsed()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 240
        pos = rng.choice(best_n, best_n // 2, replace=False)
        best_h = np.zeros(best_n)
        best_h[pos] = 1.0
        best_c5 = compute_c5(best_h)
    
    n_main = best_n
    ts_main = n_main // 2
    
    h_bin = make_binary(best_h, ts_main)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    sa_end1 = t_start + 0.40 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, ts_main, sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    
    h_bin2 = make_binary(best_h, ts_main)
    
    for alpha in [80, 300, 1200, 6000]:
        if elapsed() > 0.55 * budget_s:
            break
        h_pol = slsqp_optimize(best_h, n_main, float(ts_main), alpha, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_bin2 = make_binary(h_pol, ts_main)
        c5_b2 = compute_c5(h_bin2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2.copy()
    
    sa_end2 = t_start + 0.70 * budget_s
    h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, ts_main, sa_end2, rng)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()
    
    h_bin3 = make_binary(best_h, ts_main)
    
    for trial in range(6):
        if elapsed() > 0.82 * budget_s:
            break
        noise = 0.10 * (1.0 - trial / 6.0) + 0.03
        h_shake = best_h + rng.uniform(-noise, noise, n_main)
        h_shake = project_to_sum(h_shake, float(ts_main))
        alpha_pick = [200, 800, 4000][min(trial // 2, 2)]
        h_opt = slsqp_optimize(h_shake, n_main, float(ts_main), alpha_pick, maxiter=1200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_bin_sh = make_binary(h_opt, ts_main)
        c5_sh = compute_c5(h_bin_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_bin_sh.copy()
    
    for n_try in [200, 280, 320, 360]:
        if remaining() < 90:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try // 2
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, float(ts_try))
        for alpha in [50, 300, 1500]:
            if remaining() < 40:
                break
            h_try = slsqp_optimize(h_interp, n_try, float(ts_try), alpha, maxiter=600)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            h_bin_try = make_binary(h_try, ts_try)
            c5_bt = compute_c5(h_bin_try)
            if c5_bt < best_c5:
                best_c5 = c5_bt
                best_h = h_bin_try.copy()
                best_n = n_try
            h_interp = h_try
    
    if remaining() > 50:
        h_final_bin = make_binary(best_h, best_n // 2)
        sa_end3 = t_start + 0.94 * budget_s
        h_sa3, c5_sa3 = binary_sa(h_final_bin, best_n, best_n // 2, sa_end3, rng,
                                   max_no_improve=15000, cool_rate=0.9997)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()
    
    if remaining() > 30:
        h_fin = slsqp_optimize(best_h, best_n, float(best_n // 2), 20000, maxiter=1500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
