def smooth_max_val_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:

        lo = max(0, l)
        hi = min(n, n + l)

        hi2 = min(n, n - l)

def l_bfgsb_smooth_max(h0, n, target_sum, alpha, maxiter=500):

        v, _ = smooth_max_val_grad(x, alpha)
        return v
    
    def grad(x):
        _, g = smooth_max_val_grad(x, alpha)
        return g
    
    try:

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h + t, 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def lbfbs_optimize(h0, n, target_sum, alpha, maxiter=300):
    def obj(x):
        h = np.clip(x, 0, 1)
        v, _ = smooth_max_val_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0, 1)
        _, g = smooth_max_val_grad(h, alpha)
        return g
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14, 'maxfun': maxiter * n})

    v_hat = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    for it in range(max_iters):
        val, g = smooth_max_val_grad(h, alpha)
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)
        m_hat_hat = m_hat / (1.0 - beta1 ** (it + 1))
        v_hat_hat = v_hat / (1.0 - beta2 ** (it + 1))

Option 3 is the best. If I change h[i] by delta, how does the correlation change?

corr[l] = sum_j h[j] * (1-h[j+l])

If I change h[i] from h[i] to h[i]+d:
- For the term where i is the first factor: corr[l] changes by d * (1 - h[i+l]) (for valid i+l)
- For the term where i is the second factor: corr[l] changes by h[i-l] * (-d) = -d * h[i-l] (for valid i-l)

So the change in corr is:
Δcorr[l] = d * (1 - h[i+l]) - d * h[i-l] (with appropriate bounds)

This is O(1) per lag, so O(n) total to update the full correlation. Then finding the max is O(n). Total per coordinate step: O(n).

With 8 steps and n coordinates: 8 * n * O(n) = O(n²) per round. For n=960, that's about 7M per round. With 20 rounds, that's 140M. That should be doable in a few seconds.

Hmm, this is getting complicated. The issue is that the correlation update formula needs to be correct. Let me think more carefully.

The full correlation in numpy is:
corr[k] = sum_i a[i] * b[i + k - (n-1)]  (where valid indices are used)

For a = h, b = 1-h:
corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])

Let me define l = k - (n-1), so l ranges from -(n-1) to n-1:
corr[n-1+l] = sum_i h[i] * (1 - h[i+l])  (sum over valid i)

If I change h[i0] from h[i0] to h[i0] + d:
- Terms where i = i0: h[i0] → h[i0]+d, so the term becomes (h[i0]+d) * (1-h[i0+l])
  Change: d * (1 - h[i0+l])
- Terms where i+l = i0, i.e., i = i0-l: the term is h[i0-l] * (1-h[i0])
  New: h[i0-l] * (1-h[i0]-d)
  Change: -d * h[i0-l]

So the total change in corr[n-1+l] is:
Δ = d * (1 - h[i0+l]) [if i0+l is valid] + (-d * h[i0-l]) [if i0-l is valid]

Wait, I need to be more careful. The correlation is:
corr[n-1+l] = sum_{i: 0<=i<n and 0<=i+l<n} h[i] * (1-h[i+l])

When h[i0] changes to h[i0]+d:
1. If i0 is a valid first index (always, since 0<=i0<n):
   - For each l such that 0 <= i0+l < n: the term h[i0]*(1-h[i0+l]) changes
   - Change: d * (1 - h[i0+l])
   
2. If i0 is a valid second index, i.e., there exists i such that i+l = i0 and 0<=i<n:
   - i = i0-l, need 0 <= i0-l < n
   - The term h[i0-l]*(1-h[i0]) changes
   - Change: h[i0-l] * (1 - (h[i0]+d)) - h[i0-l]*(1-h[i0]) = -d * h[i0-l]

So for each lag l:
Δcorr[l] = d*(1-h[i0+l]) * [0<=i0+l<n] - d*h[i0-l] * [0<=i0-l<n]

But wait, this is the change BEFORE projection. After projection, all coordinates might change. So the incremental update is only valid if we don't project (i.e., if the sum constraint is already satisfied and remains satisfied).

In the coordinate descent, if I change h[i0] by d, the sum changes by d. To maintain the sum constraint, I'd need to adjust another coordinate. This makes the incremental update more complex.

Simpler approach: don't maintain the sum constraint during coordinate descent. Just minimize the max correlation with box constraints, then project at the end. The projection might slightly increase C₅, but it's a small effect.

OR: use a "paired" update: change h[i0] by +d and h[j0] by -d for some j0. This preserves the sum. But choosing j0 is non-trivial.

For simplicity, let me just do:

def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0,

1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h + t, 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def _obj_smooth(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_val_grad(h, alpha)
    return v

def _grad_smooth(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_val_grad(h, alpha)
    return g

def lbfbs_optimize(h0, n, target_sum, alpha, maxiter):
    def _obj(x):
        return _obj_smooth(x, alpha)

    def _grad(x):
        return _grad_smooth(x, alpha)

    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-8,
                                'ftol': 1e-14, 'maxfun': maxiter * 200})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)

    v_hat = np.zeros(n)

    best_c5 = compute_c5(h)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    for it in range(max_iters):
        val, g = smooth_max_val_grad(h, alpha)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        m_hat_hat = m_hat / (1.0 - beta1 ** (it + 1))

    if time_left() > 80:
        n_ms = best_n
        ts_ms = n_ms / 2.0
        for trial in range(8):
            if time_left() < 20:
                break
            delta = 0.04 * (trial + 1)
            h_p = best_h + rng.uniform(-delta, delta, n_ms)
            h_p = project_to_sum(h_p, ts_ms)
            c5_p = compute_c5(h_p)
            if c5_p > best_c5 + 0.02:
                continue
            h_cur = h_p
            ni = min(800, max(100, int(time_left() * 0.4)))
            h_a, c5_a = adam_optimize(h_cur, n_ms, ts_ms, 500.0, ni, 0.3)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
            h_cur = h_a
            if time_left() > 15:
                ni2 = min(2000, max(200, int(time_left() * 0.4)))
                h_sg, c5_sg = subgrad_topk(h_cur, n_ms, ts_ms, ni2, 0.4, 3)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()

    if time_left() > 30:
        nf = best_n
        tsf = nf / 2.0
        for alpha in [3000.0, 8000.0]:
            if time_left() < 15:
                break
            h_pol = lbfbs_optimize(best_h, nf, tsf, alpha, 80)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time_left() > 20:
        nf = best_n
        tsf = nf / 2.0
        steps_f = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
        h_cf, c5_cf = coord_descent_fast(best_h, nf, 20, 300, steps_f, 42)
        h_cf = project_to_sum(h_cf, tsf)
        h_cf = np.clip(h_cf, 0.0, 1.0)
        c5_cf = compute_c5(h_cf)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf.copy()

    if time_left() > 15:
        nf = best_n
        tsf = nf / 2.0
        ni = min(2000, max(100, int(time_left() * 1.5)))
        h_a, c5_a = adam_optimize(best_h, nf, tsf, 5000.0, ni, 0.1)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    nf = len(best_h)
    tsf = nf / 2.0
    best_h = project_to_sum(best_h, tsf)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, nf)
