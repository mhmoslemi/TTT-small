        h2 += amp * np.sin(np.pi * freq * x)
    elif mode == 1:
        h2 += amp * np.cos(np.pi * 2.0 * freq * x)
    elif mode == 2:
        h2 += amp * (np.sin(np.pi * freq * x) + 0.5 * np.sin(np.pi * 2.0 * freq * x))
    else:
        h2 += rng.uniform(-amp, amp, n)
    return project_to_sum(np.clip(h2, 0.0, 1.0), n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_base = len(best_h)
    ts_base = n_base / 2.0

    # Phase 1: Quick SLSQP polish on initial
    for alpha in [80, 400, 2000]:
        if time_left() < 25:
            break
        h_new = slsqp_optimize(best_h, n_base, ts_base, alpha, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: DE search round 1 (exploratory, larger mutations)
    if time_left() > 180:
        budget_de = min(time_left() - 80, 200)
        h_de, c5_de = de_run(best_h, n_base, ts_base, rng,
                             pop_size=14, gen_limit=45, F=0.9, CR=0.6,
                             t_start=t_start, time_budget=budget_de)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
        # Polish what we got
        if time_left() > 60:
            h_pol = slsqp_optimize(best_h, n_base, ts_base, 1000, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 3: DE search round 2 (exploitation, smaller mutations)
    if time_left() > 120:
        budget_de2 = min(time_left() - 60, 120)
        h_de2, c5_de2 = de_run(best_h, n_base, ts_base, rng,
                               pop_size=10, gen_limit=35, F=0.4, CR=0.8,
                               t_start=t_start, time_budget=budget_de2)
        if c5_de2 < best_c5:
            best_c5 = c5_de2
            best_h = h_de2.copy()
        if time_left() > 40:
            h_pol2 = slsqp_optimize(best_h, n_base, ts_base, 3000, maxiter=600)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    # Phase 4: Multi-scale coordinate descent
    if time_left() > 80:
        h_cd, c5_cd = multi_scale_cd(best_h, ts_base, t_start, time_budget=70)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Structured perturbations + local search
    if time_left() > 100:
        for mode in range(4):
            if time_left() < 40:
                break
            freqs = [2.0, 3.0, 4.0, 5.0, 7.0]
            for freq in freqs:
                if time_left() < 30:
                    break
                h_pert = structured_perturb(best_h, n_base, rng, mode=mode, freq=freq, amp=0.04)
                h_cd_p, c5_cd_p = coordinate_descent(h_pert, ts_base, max_iters=2500, step_frac=0.08)
                if c5_cd_p < best_c5:
                    best_c5 = c5_cd_p
                    best_h = h_cd_p.copy()
                if time_left() < 25:
                    break
                h_sl_p = slsqp_optimize(h_cd_p, n_base, ts_base, 2000, maxiter=400)
                c5_sl_p = compute_c5(h_sl_p)
                if c5_sl_p < best_c5:
                    best_c5 = c5_sl_p
                    best_h = h_sl_p.copy()

    # Phase 6: Higher resolution refinement
    for n_tr in [600, 720, 900]:
        if time_left() < 140:
            break
        ts_tr = n_tr / 2.0
        h_interp = interpolate_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
        for alpha in [150, 800, 4000]:
            if time_left() < 50:
                break
            h_opt = slsqp_optimize(h_interp, n_tr, ts_tr, alpha, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt
        if time_left() > 50:
            h_cd_h, c5_cd_h = multi_scale_cd(h_interp, ts_tr, t_start, time_budget=40)
            if c5_cd_h < best_c5:
                best_c5 = c5_cd_h
                best_h = h_cd_h.copy()
                best_n = n_tr
        # Quick DE at higher resolution if time permits
        if time_left() > 100 and n_tr <= 720:
            h_de_h, c5_de_h = de_run(h_interp, n_tr, ts_tr, rng,
                                     pop_size=8, gen_limit=20, F=0.5, CR=0.7,
                                     t_start=t_start, time_budget=min(time_left() - 50, 80))
            if c5_de_h < best_c5:
                best_c5 = c5_de_h
                best_h = h_de_h.copy()
                best_n = n_tr
            if time_left() > 30:
                h_pol_h = slsqp_optimize(best_h, best_n, best_n / 2.0, 5000, maxiter=500)
                c5_pol_h = compute_c5(h_pol_h)
                if c5_pol_h < best_c5:
                    best_c5 = c5_pol_h
                    best_h = h_pol_h.copy()

    # Phase 7: Final intensive refinement
    n_fin = best_n
    ts_fin = n_fin / 2.0
    if time_left() > 60:
        h_final, c5_final = multi_scale_cd(best_h, ts_fin, t_start, time_budget=50)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if time_left() > 30:
        h_pol_fin = slsqp_optimize(best_h, n_fin, ts_fin, 50000, maxiter=1500)
        c5_pol_fin = compute_c5(h_pol_fin)
        if c5_pol_fin < best_c5:
            best_c5 = c5_pol_fin
            best_h = h_pol_fin.copy()

    if time_left() > 20:
        h_cd_fin, c5_cd_fin = coordinate_descent(best_h, ts_fin, max_iters=30000, step_frac=0.002)
        if c5_cd_fin < best_c5:
            best_c5 = c5_cd_fin
            best_h = h_cd_fin.copy()

    if time_left() > 10:
        h_cd_ult, c5_cd_ult = coordinate_descent(best_h, ts_fin, max_iters=20000, step_frac=0.001)
        if c5_cd_ult < best_c5:
            best_c5 = c5_cd_ult
            best_h = h_cd_ult.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
