def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    obj = m_val + np.log(np.sum(ev)) / alpha
    
    j = np.arange(n)
    k = np.arange(n_corr)

    v2 = (idx2 >= 0) & (idx2 < n)
    i2c = np.clip(idx2, 0, n - 1)
    t2 = v2 * h[i2c]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.copy().astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    ev = np.exp(s)
    return m + np.log(np.sum(ev)) / alpha


def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    ev = np.exp(s)
    w = ev / np.sum(ev)
    obj = m + np.log(np.sum(ev)) / alpha
    j = np.arange(n)
    kk = np.arange(nc)
    idx1 = j[:, None] + kk[None, :] - (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    t1 = v1 * (1.0 - h[np.clip(idx1, 0, n - 1)])
    idx2 = j[:, None] - kk[None, :] + (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    t2 = v2 * h[np.clip(idx2, 0, n - 1)]
    grad = (t1 - t2).dot(w)
    return obj, grad

    val, _ = smooth_obj_grad(x, alpha)
    return val + pen * 0.5 * (np.sum(x) - ts) ** 2

    _, gr = smooth_obj_grad(x, alpha)

        return project_to_sum(np.clip(res.x, 0.0, 1.0), ts)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.copy().astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    ev = np.exp(s)
    w = ev / np.sum(ev)
    obj = m + np.log(np.sum(ev)) / alpha
    j = np.arange(n)
    kk = np.arange(nc)
    idx1 = j[:, None] + kk[None, :] - (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    t1 = v1 * (1.0 - h[np.clip(idx1, 0, n - 1)])
    idx2 = j[:, None] - kk[None, :] + (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    t2 = v2 * h[np.clip(idx2, 0, n - 1)]
    grad = (t1 - t2).dot(w)
    return obj, grad


def _pen_fun(x, alpha, ts, pen):
    val, _ = smooth_obj_grad(x, alpha)
    return val + pen * 0.5 * (np.sum(x) - ts) ** 2


def _pen_jac(x, alpha, ts, pen):
    _, gr = smooth_obj_grad(x, alpha)
    return gr + pen * (np.sum(x) - ts)


def grad_optimize(h0, n, ts, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    x0 = project_to_sum(h0.copy(), ts)
    try:
        res = minimize(_pen_fun, x0, method='L-BFGS-B', jac=_pen_jac,
                       args=(alpha, ts, pen), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-8, 'maxls': 20})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), ts)
    except Exception:
        return project_to_sum(x0, ts)

    h_bin = np.zeros(n)
    order = np.argsort(-h)
    h_bin[order[:target_ones]] = 1.0
    return h_bin
