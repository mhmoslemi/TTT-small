def smooth_max_and_grad(h, alpha):

    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_refine(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0

def sa_diverse(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.014
    min_T = 0.0015
    stall = 0
    max_stall = 40000
    accept_n = 0
    total_n = 0
    block_sizes = [1, 1, 1, 2, 3, 5]

    while time.time() < t_end:
        if total_n > 800:
            rate = accept_n / total_n
            if rate < 0.08:
                T = min(T * 1.15, 0.025)
            elif rate > 0.65:
                T = max(T * 0.88, min_T)
            accept_n = 0
            total_n = 0

        r = rng.random()
        if r < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif r < 0.85:
            bs = block_sizes[rng.randint(len(block_sizes))]
            if bs >= n // 4:
                bs = 1
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                delta = c5_new - c5_curr
                total_n += 1
                if delta < 0 or np.log(rng.uniform()) < -delta / max(T, 1e-10):
                    c5_curr = c5_new
                    accept_n += 1
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                else:
                    h[i], h[j] = h[j], h[i]
                T = max(T * 0.999998, min_T)
                stall += 1
                continue
            i = rng.randint(n - bs)
            j = rng.randint(n - bs)
            if abs(i - j) <= bs:
                continue
            sa = h[i:i+bs].sum()
            sb = h[j:j+bs].sum()
            if abs(sa - sb) > 1e-10:
                continue
            tmp = h[i:i+bs].copy()
            h[i:i+bs] = h[j:j+bs]
            h[j:j+bs] = tmp
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if len(set([i, j, k])) < 3:
                continue
            a, b, c = h[i], h[j], h[k]
            if len(set([a, b, c])) < 2:
                continue
            if a == b and b == c:
                continue
            h[i], h[j], h[k] = h[k], h[i], h[j]

        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        total_n += 1
        if delta < 0 or np.log(rng.uniform()) < -delta / max(T, 1e-10):
            c5_curr = c5_new
            accept_n += 1
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
        else:
            if r < 0.55:
                h[i], h[j] = h[j], h[i]
            elif r < 0.85:
                tmp = h[i:i+bs].copy()
                h[i:i+bs] = h[j:j+bs]
                h[j:j+bs] = tmp
            else:
                h[i], h[j], h[k] = h[j], h[k], h[i]

        T = max(T * 0.999997, min_T)
        stall += 1

        if stall > max_stall:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.8, 0.028)
            stall = 0

    return best_h, best_c5

def continuous_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.008
    min_T = 0.001
    stall = 0
    max_stall = 30000
    step_size = 0.15

    while time.time() < t_end:
        num_changes = rng.randint(1, 4)
        indices = rng.randint(0, n, num_changes)
        old_vals = h[indices].copy()
        new_vals = old_vals + rng.uniform(-step_size, step_size, num_changes)
        new_vals = np.clip(new_vals, 0.0, 1.0)
        diff = target_sum - (np.sum(h) - np.sum(old_vals) + np.sum(new_vals))
        new_vals += diff / num_changes
        new_vals = np.clip(new_vals, 0.0, 1.0)
        h[indices] = new_vals
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < -delta / max(T, 1e-10):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h = project_to_sum(h, target_sum)
            stall += 1

        T = max(T * 0.999995, min_T)
        step_size = max(step_size * 0.99999, 0.03)

        if stall > max_stall:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.015)
            step_size = min(step_size * 1.5, 0.2)
            stall = 0

    return best_h, best_c5

def thue_morse_binary(n, rng):
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    target = n // 2
    if tm.sum() != target:
        if tm.sum() > target:
            excess = int(tm.sum() - target)
            ones = np.where(tm == 1)[0]
            to_flip = rng.choice(ones, size=excess, replace=False)
            tm[to_flip] = 0
        else:
            deficit = int(target - tm.sum())
            zeros = np.where(tm == 0)[0]
            to_flip = rng.choice(zeros, size=deficit, replace=False)
            tm[to_flip] = 1
    return tm.astype(float)

def golden_spacing_binary(n, rng):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    m = n // 2
    positions = set()
    for i in range(m * 2):
        pos = int((i * phi) % 1.0 * (n + 1))
        pos = min(pos, n - 1)
        if pos not in positions:
            positions.add(pos)
            if len(positions) == m:
                break
    if len(positions) < m:
        available = [i for i in range(n) if i not in positions]
        extra = rng.choice(available, size=m - len(positions), replace=False)
        positions.update(extra.tolist())
    h[list(positions)] = 1.0
    return h

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = thue_morse_binary(best_n, rng)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    candidates = []
    h_parent = project_to_sum(best_h, ts_main)
    candidates.append(h_parent)
    candidates.append(thue_morse_binary(n_main, rng))
    candidates.append(golden_spacing_binary(n_main, rng))
    h_rand = rng.randint(0, 2, n_main).astype(float)

    h_shuffled = h_parent.copy()
    rng.shuffle(h_shuffled)
