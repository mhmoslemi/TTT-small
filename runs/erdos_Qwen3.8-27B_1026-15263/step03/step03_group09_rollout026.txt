import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad_fast(h, alpha, weight_thresh=1e-7):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    w = exp_v / s
    sm_val = m + np.log(s / len(corr)) / alpha
    
    g = np.zeros(n)
    active = np.where(w > weight_thresh)[0]
    
    for idx in active:
        p = idx - n + 1
        wm = w[idx]
        if p >= 0:
            g[:n - p] += wm * (1.0 - h[p:])
            g[p:] -= wm * h[:n - p]
        else:
            pp = -p
            g[pp:] += wm * (1.0 - h[:n - pp])
            g[:n - pp] -= wm * h[pp:]
    
    return sm_val, g

def nesterov_subgrad(h0, n, target_sum, alpha, n_iters, lr, top_k, t_start, budget_s):
    h = h0.copy()
    x = h0.copy()  # Nesterov auxiliary variable
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 0.9
    
    for it in range(n_iters):
        if budget_s - (time.time() - t_start) < 10:
            break
        
        # Nesterov step
        b_new = (beta * (it + 1.0)) / (it + 1.0 + beta)  # or use (1+sqrt(1+4*it*it))/(2*it)
        # Actually standard: t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
        # beta_k = (t_k - 1) / t_{k+1}
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        if it > 0:
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1))) / 2.0
            beta_k = (t_prev - 1.0) / t_k
        else:
            beta_k = 0.0
        
        x = h + beta_k * (h - x)
        
        # Gradient at x
        x_clipped = np.clip(x, 0.0, 1.0)
        # Use top-k subgradient
        corr = np.correlate(x_clipped, 1.0 - x_clipped, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        
        g = np.zeros(n)
        for m in top_idx:
            p = m - n + 1
            if p >= 0:
                g[:n - p] += (1.0 - x_clipped[p:])
                g[p:] -= x_clipped[:n - p]
            else:
                pp = -p
                g[pp:] += (1.0 - x_clipped[:n - pp])
                g[:n - pp] -= x_clipped[pp:]
        g /= top_k
        
        # Step
        lr_eff = lr / (1.0 + 0.1 * it)
        h_new = h - lr_eff * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    n_pts = n
    t_sum = target_sum
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad_fast(h, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad_fast(h, alpha)
        return g
    
    def c_fun(x):
        return np.sum(x) - t_sum
    
    def c_jac(x):
        return np.ones(n_pts)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_fast(h0, n, target_sum, max_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5(h)
    n_corr = 2 * n - 1
    
    for r in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                nv = hi + d
                if nv < -1e-14 or nv > 1.0 + 1e-14:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def thue_morse_pattern(n):
    # Generate Thue-Morse-like pattern
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0

def make_initial_h(n, method='uniform', rng=None):

        return np.full(n, 0.5)
    elif method == 'random':

                                                np.array([-1.0, 1.0]) * 0.0002)
            if c5_cf2 < best_c5:
                best_c5 = c5_cf2
                best_h = h_cf2.copy()
                best_n = n_f

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
