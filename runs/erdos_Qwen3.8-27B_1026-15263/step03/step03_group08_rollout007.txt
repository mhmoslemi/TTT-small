import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_h(h, alpha, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(w)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    for idx in top_idx:
        wl = w[idx]
        if wl < 1e-13:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            g += wl * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += wl * (1.0 - h[lag:])

        else:
            p = -lag
            g[p:] += wl * (1.0 - h[:n - p])
            g[:n - p] -= wl * h[p:]
    return sm_val, g

def slsqp_refine(h0, n, target_sum, alpha, maxiter, top_k=10):
    def obj_fn(x):
        val, _ = smooth_max_h(x, alpha, top_k)
        return val
    def grad_fn(x):
        _, g = smooth_max_h(x, alpha, top_k)
        return g
    def constr_fun(x):
        return np.sum(x) - target_sum
    def constr_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    bounds = [(0.0, 1.0)] * n

        g = np.zeros(n)
        for idx in top_idx:
            excess = corr[idx] - C_threshold
            if excess <= 0:
                continue
            wl = 1.0
            lag = idx - (n - 1)
            if lag == 0:
                g += wl * (1.0 - 2.0 * h)
            elif lag > 0:
                g[:n - lag] += wl * (1.0 - h[lag:])
                g[lag:] -= wl * h[:n - lag]
            else:
                p = -lag
                g[p:] += wl * (1.0 - h[:n - p])
                g[:n - p] -= wl * h[p:]
        g *= 2.0 * excess if np.any(excess > 0) else 0.0
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-16:
            break
        h_new = np.clip(h - lr * g / (norm_g + 1e-10), 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.9
        h = h_new
    return best_h, best_c5

def fine_coord_descent(h0, n, target_sum, n_rounds, step):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(0, n, 1):
            for d in [step, -step, step * 0.5, -step * 0.5]:
                h_temp = h.copy()
                h_temp[i] = np.clip(h_temp[i] + d, 0.0, 1.0)
                h_temp = project_to_sum(h_temp, target_sum)
                c5_temp = compute_c5(h_temp)
                if c5_temp < c5_cur - 1e-15:
                    h = h_temp
                    c5_cur = c5_temp
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    N = best_n
    ts = N / 2.0

    starts = [best_h.copy()]

    if time_left() > 120:
        K = 20
        for pop in [40, 60]:
            if time_left() < 30:
                break
            mi = max(20, min(80, int(time_left() // 3)))
            h_de, c5_de = de_fourier_search(N, K, ts, pop, mi, seed + len(starts))
            if h_de is not None:
                starts.append(h_de)
                if c5_de < best_c5:
                    best_c5 = c5_de
                    best_h = h_de.copy()
        K2 = 30
        if time_left() > 60:
            h_de2, c5_de2 = de_fourier_search(N, K2, ts, 40, 40, seed + 100)
            if h_de2 is not None:
                starts.append(h_de2)
                if c5_de2 < best_c5:
                    best_c5 = c5_de2
                    best_h = h_de2.copy()

    if N <= 600:
        N2 = N + 24
        if N2 % 2 != 0:
            N2 += 1
        ts2 = N2 / 2.0
        if abs(ts2 - round(ts2)) < 1e-9 and time_left() > 200:
            x_old = np.linspace(0, 2, N, endpoint=False)
            x_new = np.linspace(0, 2, N2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_sum(h_interp, ts2)
            starts.append(h_interp)

    for si, h_start in enumerate(starts[:6]):
        if time_left() < 60:
            break
        n_curr = len(h_start)
        ts_curr = n_curr / 2.0
        h_cur = h_start.copy()

        alpha_schedule = [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
        for alpha in alpha_schedule:
            if time_left() < 30:
                break
            maxiter = max(50, min(400, int(time_left() * 0.4)))
            if alpha >= 1000:
                top_k_use = 5
            elif alpha >= 200:
                top_k_use = 10
            else:
                top_k_use = 20
            h_opt = slsqp_refine(h_cur, n_curr, ts_curr, alpha, maxiter, top_k_use)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_curr
            h_cur = h_opt

        if time_left() > 60:
            n_mom = min(3000, int(time_left() * 1.5))
            h_mom, c5_mom = momentum_refine(h_cur, n_curr, ts_curr, n_mom, 0.05, 500.0, top_k=5)
            if c5_mom < best_c5:
                best_c5 = c5_mom
                best_h = h_mom.copy()
                best_n = n_curr
            h_cur = h_mom

        if time_left() > 40:
            C_pen = max(best_c5 * 0.999, 0.1)
            n_as = min(2000, int(time_left() * 1.5))
            h_as, c5_as = active_set_grad_step(h_cur, n_curr, ts_curr, n_as, 0.02, C_pen / (2.0 / n_curr))
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                best_n = n_curr
            h_cur = h_as

    if best_h is not None:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha in [2000.0, 5000.0, 10000.0]:
            if time_left() < 15:
                break
            h_pol = slsqp_refine(best_h, n_f, ts_f, alpha, 500, top_k=3)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 30:
            n_mom2 = min(4000, int(time_left() * 1.5))
            h_m2, c5_m2 = momentum_refine(best_h, n_f, ts_f, n_mom2, 0.01, 1000.0, top_k=3)
            if c5_m2 < best_c5:
                best_c5 = c5_m2
                best_h = h_m2.copy()

        if time_left() > 25:
            h_cd, c5_cd = fine_coord_descent(best_h, n_f, ts_f, 40, 0.005)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 15:
            h_pol2 = slsqp_refine(best_h, n_f, ts_f, 10000.0, 300, top_k=3)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

        if time_left() > 10:
            h_cd2, c5_cd2 = fine_coord_descent(best_h, n_f, ts_f, 60, 0.002)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
