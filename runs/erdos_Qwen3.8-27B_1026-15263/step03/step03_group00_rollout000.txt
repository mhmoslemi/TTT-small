    x = np.linspace(0, 2.0, n, endpoint=False)
    n_freqs = rng.randint(4, 9)
    for _ in range(n_freqs):
        f = rng.randint(2, n // 4 + 1)
        a = rng.uniform(0.08, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        h += a * np.sin(2.0 * np.pi * f * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = random_structural(best_n, rng)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    for lr in [0.003, 0.006, 0.012]:
        for mom in [0.6, 0.75, 0.88]:
            if time_left() < 30:
                break
            n_it = min(3000, max(500, int(time_left() * 2)))
            h_new, c5_new = momentum_subgrad(best_h, ts_parent, n_it, lr, mom, 0.001, rng)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    for top_k in [1, 3, 5, 8]:
        for alpha in [80, 400, 2000, 10000]:
            if time_left() < 25:
                break
            h_new = slsqp_smooth_max(best_h, n_parent, ts_parent, alpha, top_k, maxiter=350)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                best_h = slsqp_smooth_max(best_h, n_parent, ts_parent, max(alpha, 1000), 1, maxiter=200)
                c5_chk = compute_c5(best_h)
                if c5_chk < best_c5:
                    best_c5 = c5_chk

    target_ns = sorted(set([n_parent, 720, 900, 1080]))
    for n_tr in target_ns:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
        else:
            best_n = n_tr

        for lr in [0.003, 0.007, 0.012]:
            if time_left() < 40:
                break
            h_sub, c5_sub = momentum_subgrad(h_interp, ts_tr, 1500, lr, 0.8, 0.001, rng)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                best_n = n_tr
            h_interp = h_sub

        for alpha in [150, 800, 4000]:
            if time_left() < 30:
                break
            h_opt = slsqp_smooth_max(h_interp, n_tr, ts_tr, alpha, 3, maxiter=350)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt

        if time_left() > 30:
            h_opt2 = slsqp_smooth_max(h_interp, n_tr, ts_tr, 20000, 1, maxiter=500)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n_tr

    n_best = best_n
    ts_best = n_best / 2.0

    for trial in range(10):
        if time_left() < 50:
            break
        noise = 0.015 + 0.035 * (1.0 - trial / float(10))
        h_shake = best_h + rng.uniform(-noise, noise, n_best)
        h_shake = np.clip(h_shake, 0.0, 1.0)
        h_shake = project_to_sum(h_shake, ts_best)
        h_sub, c5_sub = momentum_subgrad(h_shake, ts_best, 1200, 0.006, 0.82, 0.001, rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
        if time_left() < 30:
            break
        h_opt = slsqp_smooth_max(h_sub, n_best, ts_best, 3000, 2, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 60:
        h_alt = random_structural(n_best, rng)
        h_alt, c5_alt = momentum_subgrad(h_alt, ts_best, 2000, 0.008, 0.8, 0.001, rng)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    if time_left() > 40:
        h_sub, c5_sub = momentum_subgrad(best_h, ts_best, 5000, 0.002, 0.9, 0.002, rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    for alpha in [5000, 25000, 100000]:
        if time_left() < 20:
            break
        h_pol = slsqp_smooth_max(best_h, n_best, ts_best, alpha, 1, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 20:
        h_pol2 = slsqp_smooth_max(best_h, n_best, ts_best, 500, 5, maxiter=600)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_best)
