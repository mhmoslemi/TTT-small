def smooth_max_grad_fast(h, W):
    """Compute gradient of smooth max using FFT-based convolution.
    W: weight array of length 2n-1 (same indexing as np.correlate output)"""
    n = len(h)
    first = np.convolve(W, 1.0 - h, mode='full')[:n]
    second = np.correlate(h, W, mode='full')[:n]
    return first - second

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_constraint(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_w(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj, w, corr

def fast_gradient(h, W):
    n = len(h)
    first = np.convolve(W, 1.0 - h, mode='full')[:n]
    second = np.correlate(h, W, mode='full')[:n]
    return first - second

def nesterov_smooth_max(h0, target_sum, alpha, n_iter, step_size, momentum_beta=None):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = x.copy()
    best_c5 = compute_c5(x)
    
    for it in range(n_iter):
        obj, W, corr = smooth_max_obj_and_w(y, alpha)
        g = fast_gradient(y, W)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if momentum_beta is not None:
            beta = momentum_beta
        else:
            beta = (t - 1.0) / t_new
        y = x + beta * (x - y)
        
        x_new = y - step_size * g
        x_new = project_to_constraint(x_new, target_sum)
        x = x_new
        t = t_new
        
        if it % 500 == 0:
            c5_check = compute_c5(x)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_x = x.copy()
    
    c5_final = compute_c5(x)
    if c5_final < best_c5:
        best_x = x.copy()
        best_c5 = c5_final
    return best_x, best_c5

def coordinate_descent_fast(h, target_sum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        # Compute gradient at this lag
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_constraint(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_constraint(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

The key question is: what alpha values to use and how to schedule them?

For a smooth max with parameter alpha:
- Small alpha (e.g., 10-50): very smooth, easy to optimize, but far from the true max
- Large alpha (e.g., 1000+): close to the true max, but harder to optimize (nearly non-smooth)

A good schedule: start with small alpha, gradually increase. At each alpha, do enough iterations to converge, then increase.

For the step size: with Nesterov acceleration, the optimal step size for a function with Lipschitz gradient L is 1/L. For the smooth max, L ≈ alpha (roughly). But since we're also projecting, we can be a bit more aggressive.

Actually, I realize the Lipschitz constant of the smooth max gradient might not be alpha. Let me think...

The smooth max is: f(x) = (1/alpha) * log(sum_k exp(alpha * c_k(x)))

∇f(x) = sum_k (exp(alpha * c_k) / sum_j exp(alpha * c_j)) * ∇c_k(x) = sum_k W[k] * ∇c_k(x)

∇²f(x) is a weighted average of the Hessians of c_k, plus additional terms from the W[k] depending on x.

The Hessian of c_k has bounded eigenvalues (at most about 2 in magnitude). The additional terms from the W[k] dependence add at most alpha * (variance of c_k) to the Hessian.

So L ≈ 2 + alpha * (spread of c_k values). If the spread is small (near the optimum), L ≈ 2. If the spread is large, L can be larger.

For safety, let me use step_size = 0.5 / (1 + alpha * 0.01) or something. Actually, let me just use a fixed step size that works well in practice.

Let me think about what step size to use. The gradient entries are bounded by 1 (since each ∂c_k/∂h_j is at most 1, and weights sum to 1). So the gradient norm is at most sqrt(n). A step of 0.1 would move h by at most 0.1*sqrt(n) ≈ 2.3 in norm, which is fine since h is in [0,1]^n.

But for convergence, we want the step to be small enough that the objective decreases. With Nesterov acceleration, a step of 1/L is optimal. If L ≈ 2 (near the optimum), then step ≈ 0.5.

Let me use step = 0.3 initially and decrease over time.
