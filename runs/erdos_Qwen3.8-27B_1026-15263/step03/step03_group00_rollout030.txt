def gradient_lag(h, lag):
    """Compute gradient of f(lag) = sum_j h[j]*(1-h[j+lag]) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # f = sum_{j=0}^{n-1-lag} h[j] * (1-h[j+lag])
            # df/dh[j] for j < n-lag: (1-h[j+lag])
            # df/dh[j] for j >= lag: -h[j-lag]
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        k = -lag
        # f = sum_{j=k}^{n-1} h[j] * (1-h[j-k])
        # df/dh[j] for j >= k: (1-h[j-k])
        # df/dh[j] for j < n-k: -h[j+k]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g

def compute_c5(h):

def project_to_sum(h, target_sum):

def weighted_gradient(h, w):

def lp_direction(c, n, target_sum):

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    def smooth_obj(x):
        corr = compute_corr(x)
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(np.clip(shifted, -500, 500))
        return m_val + np.log(np.sum(exp_vals)) / alpha

    def smooth_grad(x):
        corr = compute_corr(x)
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(np.clip(shifted, -500, 500))
        w = exp_vals / np.sum(exp_vals)
        return weighted_gradient(x, w)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0.copy(), method='SLSQP', jac=smooth_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
