def min_max_lines(d, e, lo=0.0, hi=1.0):
    """Minimize max_i (d[i]*v + e[i]) for v in [lo, hi].
    Returns (v_opt, obj_opt)."""
    n = len(d)
    # Sort by slope
    order = np.argsort(d)
    d_sorted = d[order]
    e_sorted = e[order]
    
    # Build upper envelope using deque
    # Each entry: (slope, intercept, x_left) where x_left is where this line becomes the max
    from collections import deque
    envelope = deque()  # stores (slope, intercept, x_start)
    
    for i in range(n):
        m, b = d_sorted[i], e_sorted[i]
        # Remove lines from the back that are never on the upper envelope
        while envelope:
            m_prev, b_prev, x_prev = envelope[-1]
            if m == m_prev:
                # Parallel: keep the one with higher intercept
                if b > b_prev:
                    envelope.pop()
                else:
                    break
            else:
                # Intersection of new line with previous
                x_int = (b_prev - b) / (m - m_prev)
                if x_int <= x_prev + 1e-12:
                    envelope.pop()
                else:
                    break
        if envelope:
            m_prev, b_prev, x_prev = envelope[-1]
            if m == m_prev:
                x_new = x_prev
            else:
                x_new = (b_prev - b) / (m - m_prev)
        else:
            x_new = -np.inf
        envelope.append((m, b, x_new))

    # The upper envelope is convex, so the minimum is where slope changes from negative to positive
    # or at a boundary
    
    # The envelope is a sequence of lines, each active on [x_start, x_next_start)
    # Find the point where the envelope achieves its minimum in [lo, hi]
    
    best_v = lo
    best_obj = float('inf')
    
    # Check boundaries
    for v_check in [lo, hi]:
        obj = max(d * v_check + e)
        if obj < best_obj:
            best_obj = obj
            best_v = v_check
    
    # Check vertices of the envelope in [lo, hi]
    env_list = list(envelope)
    for idx in range(len(env_list)):
        m, b, x_start = env_list[idx]
        if idx + 1 < len(env_list):
            x_end = env_list[idx+1][2]
        else:
            x_end = np.inf

            v_check = x_start
            obj = max(d * v_check + e)

                best_v = v_check

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def find_optimal_v(d, e):
    """Minimize max_k (d[k]*v + e[k]) for v in [0,1]."""
    n = len(d)
    # Sort by slope
    order = np.argsort(d)
    d_s = d[order]
    e_s = e[order]
    
    # Build upper envelope
    env_m = []
    env_b = []
    env_x = []
    
    for i in range(n):
        m, b = d_s[i], e_s[i]
        while len(env_m) > 0:
            m_prev = env_m[-1]
            b_prev = env_b[-1]
            x_prev = env_x[-1]
            if abs(m - m_prev) < 1e-14:
                if b <= b_prev:
                    break
                else:
                    env_m.pop()
                    env_b.pop()
                    env_x.pop()
            else:
                x_int = (b_prev - b) / (m - m_prev)
                if x_int <= x_prev + 1e-12:
                    env_m.pop()
                    env_b.pop()
                    env_x.pop()
                else:
                    break
        if len(env_m) > 0:
            if abs(m - env_m[-1]) < 1e-14:
                x_new = env_x[-1]
            else:
                x_new = (env_b[-1] - b) / (m - env_m[-1])
        else:
            x_new = -np.inf
        env_m.append(m)
        env_b.append(b)
        env_x.append(x_new)
    
    # Find minimum of upper envelope on [0,1]
    best_v = 0.0
    best_obj = np.max(d * 0.0 + e)  # = max(e)
    
    obj_1 = np.max(d * 1.0 + e)  # = max(d + e) = max(d) + max(e) ... no
    # Actually max_k (d[k] + e[k])
    if obj_1 < best_obj:
        best_obj = obj_1
        best_v = 1.0
    
    # Check interior vertices
    for idx in range(len(env_m)):
        x_v = env_x[idx]
        if 1e-12 < x_v < 1.0 - 1e-12:
            obj_v = np.max(d * x_v + e)
            if obj_v < best_obj:
                best_obj = obj_v
                best_v = x_v
    
    return best_v, best_obj

def coordinate_descent_pass(h, corr, n, dx, i_start=0):
    """One pass of coordinate descent. Updates h and corr in place."""
    for i in range(i_start, n):
        # Compute d_k for all lags
        n_lags = 2 * n - 1
        k_range = np.arange(-(n-1), n)
        j1 = i - k_range  # for term h[i-k]
        j2 = i + k_range  # for term h[i+k]
        
        d = np.zeros(n_lags)
        mask1 = (j1 >= 0) & (j1 < n)
        mask2 = (j2 >= 0) & (j2 < n)
        d[mask1] += (1.0 - h[j1[mask1]])
        d[mask2] -= h[j2[mask2]]
        
        e = corr - h[i] * d
        v_opt, _ = find_optimal_v(d, e)
        
        if abs(v_opt - h[i]) > 1e-12:
            corr += (v_opt - h[i]) * d
            h[i] = v_opt

        # Change: h[i] += delta, h[j] -= delta
        delta = rng.uniform(-0.3, 0.3)
        new_i = h[i] + delta
        new_j = h[j] - delta
        
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        
        # Compute change in corr
        j1_i = i - k_range
        j2_i = i + k_range
        mask1_i = (j1_i >= 0) & (j1_i < n)
        mask2_i = (j2_i >= 0) & (j2_i < n)
        d_i = np.zeros(n_lags)
        d_i[mask1_i] = 1.0 - h[j1_i[mask1_i]]
        d_i[mask2_i] -= h[j2_i[mask2_i]]
        
        j1_j = j - k_range
        j2_j = j + k_range
        mask1_j = (j1_j >= 0) & (j1_j < n)
        mask2_j = (j2_j >= 0) & (j2_j < n)
        d_j = np.zeros(n_lags)
        d_j[mask1_j] = 1.0 - h[j1_j[mask1_j]]
        d_j[mask2_j] -= h[j2_j[mask2_j]]
        
        # New corr = corr + delta * d_i - delta * d_j
        new_corr_vals = corr + delta * (d_i - d_j)
        new_c5 = np.max(new_corr_vals) * dx
        delta_c5 = new_c5 - c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            corr += delta * (d_i - d_j)
            h[i] = new_i
            h[j] = new_j
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        T *= cool_rate
    
    return best_h, best_c5
