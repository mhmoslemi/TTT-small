def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:

def project_sum_box(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    # h* = clip(x + lambda, 0, 1), find lambda so sum = target_sum
    lo = -x.max() - 1.0

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    dx = 2.0 / n
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + (n - 1)
    second_idx = j_idx + m_idx - (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1) * dx
    return obj, grad

def compute_B(h, j, n):

    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)

    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo, vals[0]
    if best_idx == len(t_grid) - 1:
        return hi, vals[-1]
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(36):
        mid = (lo_r + hi_r) * 0.5
        val_mid = np.max(A + D * mid)
        if val_mid <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5, np.max(A + D * (lo_r + hi_r) * 0.5)

def solve_2d_lp(corr, D1, D2, h_j, h_k1, h_k2):
    """Solve the 3-point move LP.
    Variables: [d1, d2, t]
    Minimize: t
    Subject to:
        corr[m] + d1*D1[m] + d2*D2[m] <= t  for all m
        -h_j <= d1 <= 1 - h_j
        -h_k1 <= d2 <= 1 - h_k1
        -(1-h_k2) <= d1+d2 <= h_k2
    """
    n_corr = len(corr)
    # A_ub: shape (n_constraints, 3)
    # b_ub: shape (n_constraints,)
    
    # Correlation constraints: D1[m]*d1 + D2[m]*d2 - t <= -corr[m]
    A_corr = np.column_stack([D1, D2, -np.ones(n_corr)])
    b_corr = -corr
    
    # Box constraints:
    # d1 <= 1-h_j, -d1 <= h_j
    # d2 <= 1-h_k1, -d2 <= h_k1
    # d1+d2 <= h_k2, -(d1+d2) <= 1-h_k2
    A_box = np.array([
        [1, 0, 0],
        [-1, 0, 0],
        [0, 1, 0],
        [0, -1, 0],
        [1, 1, 0],
        [-1, -1, 0]
    ])
    b_box = np.array([1-h_j, h_j, 1-h_k1, h_k1, h_k2, 1-h_k2])
    
    A_ub = np.vstack([A_corr, A_box])
    b_ub = np.concatenate([b_corr, b_box])
    
    c = np.array([0.0, 0.0, 1.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if res.success:
            return res.fun, res.x[0], res.x[1]
    except Exception:
        pass
    return np.max(corr), 0.0, 0.0

def pair_descent_sweep(h, corr, rng, n, n_cand=10, use_3point=False):
    """Enhanced coordinate descent with optional 3-point moves."""
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_d2 = 0.0
        best_D = None
        best_k = -1
        best_k2 = -1
        is_3point = False
        
        # Build candidate set
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n]
        for _ in range(max(0, n_cand - 4)):
            k_set.append(rng.randint(n))
        
        # Try 2-point moves
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt, val_opt = solve_1d(corr, D, delta_lo, delta_hi)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_d2 = 0.0
                best_D = D
                best_k = k
                best_k2 = -1
                is_3point = False
        
        # Try 3-point moves (if enabled)
        if use_3point:
            for _ in range(3):
                k1 = rng.randint(n)
                k2 = rng.randint(n)
                if k1 == j or k2 == j or k1 == k2:
                    continue
                B_k1 = compute_B(h, k1, n)
                B_k2 = compute_B(h, k2, n)
                D1 = B_j - B_k2
                D2 = B_k1 - B_k2
                # Solve 2D LP
                val_opt, d1, d2 = solve_2d_lp(corr, D1, D2, h[j], h[k1], h[k2])
                if val_opt < best_val - 1e-13:
                    best_val = val_opt
                    best_delta = d1
                    best_d2 = d2
                    best_D = None
                    best_k = k1
                    best_k2 = k2
                    is_3point = True
        
        # Apply best move
        if best_k >= 0 and (abs(best_delta) > 1e-13 or abs(best_d2) > 1e-13):
            if not is_3point:
                corr += best_delta * best_D
                h[j] += best_delta
                h[best_k] -= best_delta
            else:
                # Recompute correlation (safer for 3-point moves)
                h[j] += best_delta
                h[best_k] += best_d2
                h[best_k2] -= (best_delta + best_d2)
                h[j] = np.clip(h[j], 0, 1)
                h[best_k] = np.clip(h[best_k], 0, 1)
                h[best_k2] = np.clip(h[best_k2], 0, 1)
                # Recompute corr
                corr = np.correlate(h, 1.0 - h, mode="full")
            improved = True
    
    return h, corr, improved

            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + (max_shifted + np.log(np.sum(exp_vals))) / alpha
    dx = 2.0 / n
    nm1 = n - 1
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx + m_idx - nm1
    second_idx = j_idx - m_idx + nm1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1) * dx
    return obj, grad

def fista_smooth_max(h0, target_sum, alpha, max_iter, lr_init=2.0):
    n = len(h0)
    h = h0.copy()
    h_prev = h.copy()
    t_k = 1.0
    lr = lr_init
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h + (t_k - 1.0) / t_new * (h - h_prev)
        obj_y, g_y = smooth_max_and_grad(y, alpha)
        g_norm_sq = np.sum(g_y * g_y)
        if g_norm_sq < 1e-24:
            break
        for _ in range(60):
            h_cand = y - lr * g_y
            h_proj = project_sum_box(h_cand, target_sum)
            obj_cand, _ = smooth_max_and_grad(h_proj, alpha)
            if obj_cand <= obj_y + lr * 0.5 * g_norm_sq + 1e-15:
                break
            lr *= 0.5
        h_prev = h
        h = h_proj
        t_k = t_new
        if it % 50 == 49:
            c5_val = compute_c5(h)
            if c5_val < best_val:
                best_val = c5_val
                best_h = h.copy()
            if obj_cand < obj_y - 1e-10:
                lr = min(lr * 1.05, 500.0)
    c5_final = compute_c5(h)
    if c5_final < best_val:
        best_val = c5_final
        best_h = h.copy()
    return best_h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5, np.max(corr)
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo, vals[0]
    if best_idx == len(t_grid) - 1:
        return hi, vals[-1]
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    val_lo = np.max(corr + D * lo_r)
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        val_mid = np.max(corr + D * mid)
        if val_mid <= val_lo:
            lo_r = mid
            val_lo = val_mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5, np.max(corr + D * (lo_r + hi_r) * 0.5)

def solve_3point_grid(corr, D1, D2, d1_lo, d1_hi, d2_lo, d2_hi):
    n1, n2 = 12, 12
    d1_grid = np.linspace(d1_lo, d1_hi, n1)
    d2_grid = np.linspace(d2_lo, d2_hi, n2)
    D1_3d = D1[None, None, :] + d1_grid[:, None, None] * D1[None, None, :] * 0
    mesh1, mesh2 = np.meshgrid(d1_grid, d2_grid, indexing='ij')
    flat_d1 = mesh1.ravel()
    flat_d2 = mesh2.ravel()
    vals = np.max(corr[None, :] + D1[None, :] * flat_d1[:, None] + D2[None, :] * flat_d2[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    best_d1 = flat_d1[best_idx]
    best_d2 = flat_d2[best_idx]
    best_val = vals[best_idx]
    for _ in range(3):
        margin1 = (d1_hi - d1_lo) / n1 * 2
        margin2 = (d2_hi - d2_lo) / n2 * 2
        d1_lo_r = max(d1_lo, best_d1 - margin1)
        d1_hi_r = min(d1_hi, best_d1 + margin1)
        d2_lo_r = max(d2_lo, best_d2 - margin2)
        d2_hi_r = min(d2_hi, best_d2 + margin2)
        nf = 16
        d1_f = np.linspace(d1_lo_r, d1_hi_r, nf)
        d2_f = np.linspace(d2_lo_r, d2_hi_r, nf)
        mf1, mf2 = np.meshgrid(d1_f, d2_f, indexing='ij')
        ff1, ff2 = mf1.ravel(), mf2.ravel()
        vals_f = np.max(corr[None, :] + D1[None, :] * ff1[:, None] + D2[None, :] * ff2[:, None], axis=1)
        bi = int(np.argmin(vals_f))
        best_d1 = ff1[bi]
        best_d2 = ff2[bi]
        best_val = vals_f[bi]
    return best_val, best_d1, best_d2

def coord_sweep(h, corr, rng, n, n_cand=10, use_3point=False):
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_d2 = 0.0
        best_D = None
        best_k = -1
        best_k2 = -1
        is_3 = False
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt, val_opt = solve_1d(corr, D, delta_lo, delta_hi)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_d2 = 0.0
                best_D = D
                best_k = k
                best_k2 = -1
                is_3 = False
        if use_3point:
            for _ in range(4):
                k1 = rng.randint(n)
                k2 = rng.randint(n)
                if k1 == j or k2 == j or k1 == k2:
                    continue
                B_k1 = compute_B(h, k1, n)
                B_k2 = compute_B(h, k2, n)
                D1 = B_j - B_k2
                D2 = B_k1 - B_k2
                d1_lo = max(-h[j], -(1.0 - h[k2]))
                d1_hi = min(1.0 - h[j], h[k2])
                d2_lo = max(-h[k1], -(1.0 - h[k2]) - d1_lo)
                d2_hi = min(1.0 - h[k1], h[k2] - d1_lo)
                d2_lo = max(d2_lo, -(1.0 - h[k2]) - d1_hi)
                d2_hi = min(d2_hi, h[k2] - d1_hi)
                if d1_hi - d1_lo < 1e-13 or d2_hi - d2_lo < 1e-13:
                    continue
                val_opt, d1_opt, d2_opt = solve_3point_grid(corr, D1, D2, d1_lo, d1_hi, d2_lo, d2_hi)
                new_j = h[j] + d1_opt
                new_k1 = h[k1] + d2_opt
                new_k2 = h[k2] - d1_opt - d2_opt
                if new_j < -1e-10 or new_j > 1 + 1e-10 or new_k1 < -1e-10 or new_k1 > 1 + 1e-10 or new_k2 < -1e-10 or new_k2 > 1 + 1e-10:
                    continue
                if val_opt < best_val - 1e-13:
                    best_val = val_opt
                    best_delta = d1_opt
                    best_d2 = d2_opt
                    best_D = None
                    best_k = k1
                    best_k2 = k2
                    is_3 = True
        if best_k >= 0 and (abs(best_delta) > 1e-13 or abs(best_d2) > 1e-13):
            if not is_3:
                corr += best_delta * best_D
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            else:
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                h[best_k] = np.clip(h[best_k] + best_d2, 0.0, 1.0)
                h[best_k2] = np.clip(h[best_k2] - best_delta - best_d2, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode="full")
            improved = True
    return h, corr, improved

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: FISTA with smooth max (20% budget)
    t_end1 = t_start + 0.20 * budget_s
    n1 = len(best_h)
    ts1 = n1 / 2.0
    for alpha in [80, 200, 600, 2000, 5000]:
        if time.time() >= t_end1 - 5:
            break
        remaining = t_end1 - time.time()
        max_iter = max(30, int(remaining / 0.3))
        h_f = fista_smooth_max(best_h, ts1, alpha, max_iter, lr_init=2.0)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n1

    # Phase 2: Coordinate descent with 3-point moves (45% budget)
    t_end2 = t_start + 0.65 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall_count = 0
    sweep_count = 0
    while time.time() < t_end2:
        use_3 = (sweep_count % 4 >= 1)
        h_work, corr_w, imp = coord_sweep(h_work, corr_w, rng, n_w, n_cand=12, use_3point=use_3)
        c5_w = compute_c5(h_work)
        sweep_count += 1
        if c5_w < best_c5 - 1e-12:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall_count = 0
        else:
            stall_count += 1
        if stall_count >= 150:
            noise_scale = 0.02 + 0.01 * min(stall_count // 300, 5)
            h_work = project_to_sum(h_work + rng.uniform(-noise_scale, noise_scale, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0

    # Phase 3: Multi-resolution (25% budget)
    t_end3 = t_start + 0.90 * budget_s
    resolutions = [320, 400, 480, 360, 560, 440]
    for n_tr in resolutions:
        if time.time() >= t_end3 - 10:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_sum_box(h_up, ts_tr)
        corr_up = np.correlate(h_up, 1.0 - h_up, mode="full")
        max_sweeps = int((t_end3 - time.time()) / 2.5)
        max_sweeps = max(3, min(25, max_sweeps))
        for i in range(max_sweeps):
            if time.time() >= t_end3 - 5:
                break
            use_3 = (i % 3 >= 1)
            h_up, corr_up, _ = coord_sweep(h_up, corr_up, rng, n_tr, n_cand=8, use_3point=use_3)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr
        if time.time() < t_end3 - 15:
            h_rand = rng.uniform(0, 1, n_tr)
            h_rand = project_sum_box(h_rand, ts_tr)
            corr_rand = np.correlate(h_rand, 1.0 - h_rand, mode="full")
            ms2 = int((t_end3 - time.time()) / 2.5)
            ms2 = max(2, min(15, ms2))
            for i in range(ms2):
                if time.time() >= t_end3 - 5:
                    break
                h_rand, corr_rand, _ = coord_sweep(h_rand, corr_rand, rng, n_tr, n_cand=8)
            c5_rand = compute_c5(h_rand)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand.copy()
                best_n = n_tr

    # Phase 4: FISTA polish + final descent (10% budget)
    t_end4 = t_start + budget_s - 5
    n_fin = best_n
    ts_fin = n_fin / 2.0
    if time.time() < t_end4 - 20:
        for alpha in [300, 1000, 3000]:
            if time.time() >= t_end4 - 5:
                break
            h_pol = fista_smooth_max(best_h, ts_fin, alpha, max_iter=150, lr_init=1.0)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_fin
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_end4:
        h_fin, corr_fin, _ = coord_sweep(h_fin, corr_fin, rng, n_fin, n_cand=15, use_3point=True)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
