def vec_smooth_max_grad(h, alpha):

    best_c5 = compute_c5(h0)
    for p in range(n_perturbs):
        scale = 0.02 * (p + 1)
        h_pert = h0 + scale * rng.randn(n)
        h_pert = project_to_sum(h_pert, ts)
        h_opt = slsqp_run(h_pert, n, ts, alpha, maxiter // 2)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best = h_opt.copy()
            best_c5 = c5_opt
    return best, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    if time_left() < 20:
        return (best_h, best_c5, len(best_h))

    alphas_main = [15.0, 40.0, 100.0, 250.0, 500.0, 1000.0]
    for i_alpha, alpha in enumerate(alphas_main):
        if time_left() < 40:
            break
        maxit = 300
        h_opt = slsqp_run(h_cur, n, ts, alpha, maxit)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        h_cur = h_opt

    if time_left() < 30:
        return (best_h, best_c5, len(best_h))

    h_act, c5_act = activeset_loop(h_cur, n, ts, 25, 0.95)
    if c5_act < best_c5:
        best_c5 = c5_act
        best_h = h_act.copy()
        best_n = n
    h_cur = h_act

    if time_left() < 30:
        return (best_h, best_c5, len(best_h))

    n_sub = min(3000, int(time_left() * 3))
    h_sub, c5_sub = subgradient_fast(h_cur, n, ts, n_sub, 0.02, 500.0)
    if c5_sub < best_c5:
        best_c5 = c5_sub
        best_h = h_sub.copy()
        best_n = n
    h_cur = h_sub

    if time_left() < 50:
        return (best_h, best_c5, len(best_h))

    h_pert, c5_pert = perturb_and_optimize(h_cur, n, ts, 200.0, 4, rng, 100)
    if c5_pert < best_c5:
        best_c5 = c5_pert
        best_h = h_pert.copy()
        best_n = n
    h_cur = h_pert

    n_resolutions = [900, 1080]
    for n_new in n_resolutions:
        if n_new % 2 != 0:
            continue
        if time_left() < 150:
            break
        ts_new = n_new / 2.0
        h_interp = interpolate_h(h_cur, n, n_new)
        h_hi = project_to_sum(h_interp, ts_new)
        c5_interp = compute_c5(h_hi)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_hi.copy()
            best_n = n_new

        alphas_hi = [50.0, 200.0, 500.0, 1000.0]
        for alpha in alphas_hi:
            if time_left() < 50:
                break
            maxit = 250
            h_hi_opt = slsqp_run(h_hi, n_new, ts_new, alpha, maxit)
            c5_hi = compute_c5(h_hi_opt)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi_opt.copy()
                best_n = n_new
            h_hi = h_hi_opt

        if time_left() < 40:
            break

        h_act_hi, c5_act_hi = activeset_loop(h_hi, n_new, ts_new, 20, 0.95)
        if c5_act_hi < best_c5:
            best_c5 = c5_act_hi
            best_h = h_act_hi.copy()
            best_n = n_new
        h_hi = h_act_hi

        if time_left() < 30:
            break
        n_sub_hi = min(2000, int(time_left() * 2))
        h_sub_hi, c5_sub_hi = subgradient_fast(h_hi, n_new, ts_new, n_sub_hi, 0.02, 500.0)
        if c5_sub_hi < best_c5:
            best_c5 = c5_sub_hi
            best_h = h_sub_hi.copy()
            best_n = n_new
        h_hi = h_sub_hi
        n = n_new

    n_f = best_n
    ts_f = n_f / 2.0
    h_f = best_h.copy()

    alphas_polish = [2000.0, 5000.0, 10000.0]
    for alpha in alphas_polish:
        if time_left() < 35:
            break
        h_pol = slsqp_run(h_f, n_f, ts_f, alpha, 250)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    if time_left() < 30:
        pass
    else:
        h_act_f, c5_act_f = activeset_loop(h_f, n_f, ts_f, 30, 0.97)
        if c5_act_f < best_c5:
            best_c5 = c5_act_f
            best_h = h_act_f.copy()
            best_n = n_f
        h_f = h_act_f

    if time_left() < 20:
        pass
    else:
        n_final_sub = min(2000, int(time_left() * 2))
        h_fin_sub, c5_fin_sub = subgradient_fast(h_f, n_f, ts_f, n_final_sub, 0.01, 2000.0)
        if c5_fin_sub < best_c5:
            best_c5 = c5_fin_sub
            best_h = h_fin_sub.copy()

        h_f = h_fin_sub

    if time_left() > 15:
        h_f2, c5_f2 = perturb_and_optimize(h_f, n_f, ts_f, 500.0, 3, rng, 80)
        if c5_f2 < best_c5:
            best_c5 = c5_f2

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
