from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:n] += (1.0 - h[0:n-lag])
            g[0:n-lag] -= h[lag:n]
    else:

        g[0:n-k] += (1.0 - h[k:n])
        g[k:n] -= h[0:n-k]
    return g

def topk_smooth_obj_and_grad(h, K=5, alpha=100.0):
    n = len(h)
    corr = compute_corr(h)
    # Get top K values
    top_k_indices = np.argpartition(corr, -K)[-K:]
    top_k_vals = corr[top_k_indices]
    m_val = np.max(top_k_vals)
    # Softmax over top K
    shifted = alpha * (top_k_vals - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: weighted sum of gradients for top K lags
    grad = np.zeros(n)
    for idx, wi in zip(top_k_indices, w):
        lag = idx - (n - 1)
        grad += wi * corr_gradient_at_lag(h, lag)
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, K, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        return topk_smooth_obj_and_grad(x, K, alpha)[0]
    
    def gradient(x):
        return topk_smooth_obj_and_grad(x, K, alpha)[1]
    
    try:
        res = minimize(objective, h0.copy(), method='SLSQP', jac=gradient,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent_v2(h, target_sum, max_iters=5000, use_line_search=True):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_at_lag(h, lag)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        
        if use_line_search:
            # Try a few step sizes
            best_step = 0
            best_c5 = c5
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                move = frac * max_move
                if move < 1e-12:
                    continue
                h[i_inc] += move
                h[i_dec] -= move
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_try = compute_c5(h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_step = move
                h[i_inc] -= move
                h[i_dec] += move
            if best_step > 0:
                h[i_inc] += best_step
                h[i_dec] -= best_step
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5 = best_c5
                stale = 0
            else:
                stale += 1
                if stale > 30:
                    break
        else:
            move = 0.15 * max_move
            if move < 1e-10:
                break
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)

def structured_annealing(h0, target_sum, rng, budget_s, t_start):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 0.05  # initial temperature
    cooling = 0.9995
    
    while time.time() - t_start < budget_s - 30:
        T *= cooling
        if T < 1e-5:
            T = 1e-5
        
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_at_lag(h, lag)
        
        # Targeted move: bias toward reducing the worst lag
        if rng.random() < 0.7:
            # Move in gradient direction
            can_inc = h < 1.0 - 1e-10
            can_dec = h > 1e-10
            candidates_inc = np.where(can_inc)[0]
            candidates_dec = np.where(can_dec)[0]
            if len(candidates_inc) == 0 or len(candidates_dec) == 0:
                break
            i_inc = int(rng.choice(candidates_inc))
            i_dec = int(rng.choice(candidates_dec))
            if g[i_inc] >= g[i_dec]:
                continue
            max_move = min(1.0 - h[i_inc], h[i_dec])
            move = (0.02 + 0.05 * T) * max_move
            if move < 1e-12:
                continue
            h[i_inc] += move
            h[i_dec] -= move
        else:
            # Random perturbation
            noise_scale = 0.01 + 0.03 * T
            idx = rng.randint(n)
            if h[idx] > 0.01:
                h[idx] -= noise_scale * h[idx]
            elif h[idx] < 0.99:
                h[idx] += noise_scale * (1.0 - h[idx])
            else:
                continue
        
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h = best_h.copy()  # revert? No, keep current for SA
    
    return best_h, best_c5

def interp_to_n(h, n_new):

    x_old = np.linspace(0, 2.0, n_old, endpoint=False)

    h_periodic = np.concatenate([h, h])  # for periodic interpolation
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
        else:
            h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def topk_smooth(h, K, alpha):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    m = top_vals.max()
    a_vals = alpha * (top_vals - m)
    a_max = a_vals.max()
    e_vals = np.exp(a_vals - a_max)
    log_sum = a_max + np.log(e_vals.sum())
    obj = m + log_sum / alpha
    w = e_vals / e_vals.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        grad += wi * lag_gradient(h, lag)
    return obj, grad
