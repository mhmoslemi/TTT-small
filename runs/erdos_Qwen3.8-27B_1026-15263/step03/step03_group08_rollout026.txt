def project_to_sum(h, target_sum):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, top_k=15, margin=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    sorted_idx = np.argsort(corr)[::-1]
    top_val = corr[sorted_idx[0]]
    active = []
    for idx in sorted_idx:
        if corr[idx] >= top_val - margin:
            active.append(idx)
        if len(active) >= top_k:
            break
    g = np.zeros(n)
    for k_idx in active:
        l = k_idx - (n - 1)
        i_lo = max(0, -l)
        i_hi = min(n, n - l)
        lo1 = max(0, i_lo + l)
        hi1 = min(n, i_hi + l)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1 - l:hi1 - l])
        if i_lo < i_hi:
            g[i_lo:i_hi] -= h[i_lo + l:i_hi + l]
    return g * dx

def projected_subgradient(h0, n, target_sum, max_iters, top_k=15, margin=0.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.01
    prev_h = None
    prev_g = None
    for it in range(max_iters):
        g = active_gradient(h, top_k=top_k, margin=margin)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        if prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            yy = np.dot(y, y)
            if sy > 1e-10 and ss > 1e-10:
                step = sy / ss
            elif yy > 1e-10 and ss > 1e-10:
                step = ss / yy
            step = np.clip(step, 1e-7, 5.0)
        prev_h = h.copy()
        prev_g = g.copy()
        alpha = step
        c5_cur = compute_c5(h)
        for _ in range(25):
            h_new = h - alpha * g
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_cur - 1e-14:
                break
            alpha *= 0.5
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def smooth_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return (m + np.log(s) / alpha) * dx

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    g = np.zeros(n)
    for k_idx in range(len(corr)):
        wl = w[k_idx]
        if wl < 1e-10:
            continue
        l = k_idx - (n - 1)
        i_lo = max(0, -l)
        i_hi = min(n, n - l)
        lo1 = max(0, i_lo + l)
        hi1 = min(n, i_hi + l)
        if lo1 < hi1:
            g[lo1:hi1] += wl * (1.0 - h[lo1 - l:hi1 - l])
        if i_lo < i_hi:
            g[i_lo:i_hi] -= wl * h[i_lo + l:i_hi + l]
    return g * dx

def slsqp_smooth_max(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_c5(h, alpha)

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_grad(h, alpha)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def generate_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    if best_h is not None:
        if n == best_n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            h_interp = np.interp(x, x_old, best_h)
            starts.append(project_to_sum(h_interp, target_sum))
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10]:
        for amp in [0.2, 0.35, 0.5]:
            for phase in [0.0, np.pi / 2, np.pi, 3 * np.pi / 2]:
                hs = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
                hs = project_to_sum(hs, target_sum)
                if compute_c5(hs) < 0.44:
                    starts.append(hs)
    if best_h is not None:
        for delta in [0.03, 0.06, 0.1, 0.15]:
            h_pert = best_h + rng.uniform(-delta, delta, best_n)
            h_pert = project_to_sum(h_pert, best_n / 2.0)
            if n != best_n:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                h_pert = np.interp(x, x_old, h_pert)
            starts.append(project_to_sum(h_pert, target_sum))
    for n_blocks in [2, 3, 4, 5, 6, 8, 10, 12, 16]:
        block_size = n // n_blocks
        remainder = n % n_blocks
        values = []
        for i in range(n_blocks):

def optimize_single(h_start, n, target_sum, time_budget):
