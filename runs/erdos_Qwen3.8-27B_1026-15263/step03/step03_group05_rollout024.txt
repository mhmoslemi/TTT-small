        if it % 200 == 199:
            corr_check = np.correlate(h, 1.0 - h, mode="full")
            if len(np.unique(np.round(corr_check[k_worst - 3:k_worst + 4], 6))) < 2:
                lr *= 0.5
    return best_h, best_c5

def multi_shift_subgradient(h, n, time_fn, max_iter=3000, lr_init=0.03, top_k=5):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        if time_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        sorted_idx = np.argsort(corr)
        k_worst = sorted_idx[-top_k:]
        weights = np.zeros(len(corr))
        for rank, k in enumerate(k_worst):
            weights[k] = (top_k - rank) / top_k
        grad = np.zeros(n)
        for k in k_worst:
            grad += weights[k] * gradient_at_shift(h, int(k))
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        h_new = h - lr * grad / grad_norm * np.max(np.abs(grad))
        h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_old = compute_c5(h)
        if c5_new > c5_old:
            lr *= 0.8
        else:
            lr = min(lr * 1.03, 0.15)
        h = h_new
    return best_h, best_c5

def smooth_max_descent(h, n, time_fn, max_iter=3000, lr_init=0.05, top_k=5):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        if time_fn() < 10:
            break
        obj, grad, _ = smooth_max_grad(h, alpha=50, top_k=top_k)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_old = compute_c5(h)
        if c5_new > c5_old:
            lr *= 0.6
        else:
            lr = min(lr * 1.05, 0.3)
        h = h_new
    return best_h, best_c5

def simulated_anneal_binary(h, n, rng, time_fn, max_iters=50000):
    h = h.copy().round().clip(0, 1)
    ts = n / 2.0
    if abs(h.sum() - ts) > 1e-6:
        h = project_box_sum(h, ts).round().clip(0, 1)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.005
    T_min = 0.0001
    cool_rate = (T_min / T) ** (1.0 / max(1, max_iters))
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(max_iters):
        if time_fn() < 15:
            break
        T *= cool_rate
        if len(ones) == 0 or len(zeros) == 0:
            break
        move_type = rng.rand()
        if move_type < 0.6:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = rng.choice(ones)
                i2 = rng.choice(ones)
                while i2 == i1 and len(ones) > 1:
                    i2 = rng.choice(ones)
                j1 = rng.choice(zeros)
                j2 = rng.choice(zeros)
                while j2 == j1 and len(zeros) > 1:
                    j2 = rng.choice(zeros)
                if i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2:
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            if move_type < 0.6:
                h[j] = 1.0
                h[i] = 0.0
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_base = best_n
    ts = n_base / 2.0

    # Phase 1: Binary multi-flip optimization at base resolution
    if time_left() > 200:
        h_bin = (best_h > 0.5).astype(float)
        if abs(h_bin.sum() - ts) > 1e-6:
            excess = h_bin.sum() - ts
            if excess > 0:
                idx = np.where(h_bin > 0.5)[0]
                rng.shuffle(idx)
                for i in range(int(np.ceil(excess))):
                    if i < len(idx):
                        h_bin[idx[i]] = 0.0
            else:
                idx = np.where(h_bin < 0.5)[0]
                rng.shuffle(idx)
                for i in range(int(np.ceil(-excess))):
                    if i < len(idx):
                        h_bin[idx[i]] = 1.0
            if abs(h_bin.sum() - ts) > 1e-6:
                h_bin = project_box_sum(h_bin, ts)
        # 2-flip optimization
        h_2f, c5_2f = binary_multi_flip(h_bin, n_base, rng, max_rounds=200, time_fn=time_left, max_points=2)
        if c5_2f < best_c5:
            best_c5 = c5_2f
            best_h = h_2f.copy()
        # 1-flip optimization
        h_1f, c5_1f = binary_multi_flip(best_h, n_base, rng, max_rounds=150, time_fn=time_left, max_points=1)
        if c5_1f < best_c5:
            best_c5 = c5_1f
            best_h = h_1f.copy()

    # Phase 2: Subgradient refinement
    if time_left() > 200:
        h_cont = best_h.copy()
        h_sg, c5_sg = subgradient_descent(h_cont, n_base, time_left, max_iter=4000, lr_init=0.08, top_k=3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        h_ms, c5_ms = multi_shift_subgradient(best_h, n_base, time_left, max_iter=3000, lr_init=0.04, top_k=5)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 3: Smooth max descent with various top_k
    if time_left() > 150:
        for tk in [3, 5, 10]:
            if time_left() < 30:
                break
            h_sm, c5_sm = smooth_max_descent(best_h, n_base, time_left, max_iter=2000, lr_init=0.05, top_k=tk)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 4: Binary + SA at base resolution
    if time_left() > 200:
        h_sa = (best_h > 0.5).astype(float)
        if abs(h_sa.sum() - ts) > 1e-6:
            h_sa = project_box_sum(h_sa, ts)
            h_sa = (h_sa > 0.5).astype(float)
            if abs(h_sa.sum() - ts) > 1e-6:
                h_sa = project_box_sum(h_sa, ts)
        h_res, c5_res = simulated_anneal_binary(h_sa, n_base, rng, time_left, max_iters=30000)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
        # Polish the SA result with subgradient
        if c5_res <= best_c5 + 0.01 and time_left() > 30:
            h_pol, c5_pol = subgradient_descent(h_res, n_base, time_left, max_iter=2000, lr_init=0.05, top_k=3)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Higher resolution (480)
    if time_left() > 300:
        n_up = 480
        if n_up % 2 != 0:
            n_up += 1
        ts_up = n_up / 2.0
        x_old = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old, best_h, period=2.0)
        h_up = project_box_sum(h_up, ts_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
        # Binary at higher res
        if time_left() > 150:
            h_ub = (h_up > 0.5).astype(float)
            if abs(h_ub.sum() - ts_up) > 1e-6:
                h_ub = project_box_sum(h_ub, ts_up)
                h_ub = (h_ub > 0.5).astype(float)
                if abs(h_ub.sum() - ts_up) > 1e-6:
                    h_ub = project_box_sum(h_ub, ts_up)
            h_ub2, c5_ub2 = binary_multi_flip(h_ub, n_up, rng, max_rounds=100, time_fn=time_left, max_points=2)
            if c5_ub2 < best_c5:
                best_c5 = c5_ub2
                best_h = h_ub2.copy()
                best_n = n_up
        # Subgradient at higher res
        if time_left() > 100:
            h_sg2, c5_sg2 = subgradient_descent(best_h, best_n, time_left, max_iter=5000, lr_init=0.04, top_k=3)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2.copy()
            h_ms2, c5_ms2 = multi_shift_subgradient(best_h, best_n, time_left, max_iter=3000, lr_init=0.03, top_k=5)
            if c5_ms2 < best_c5:
                best_c5 = c5_ms2
                best_h = h_ms2.copy()

    # Phase 6: Shake + re-optimize
    if time_left() > 150:
        n_sh = best_n
        for trial in range(15):
            if time_left() < 30:
                break
            noise_scale = 0.15 * (1.0 - trial / 15.0) + 0.02
            h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n_sh)
            h_shake = project_box_sum(h_shake, n_sh / 2.0)
            for method in ['sg', 'ms']:
                if time_left() < 20:
                    break
                if method == 'sg':
                    h_opt, c5_opt = subgradient_descent(h_shake, n_sh, time_left, max_iter=2000, lr_init=0.05, top_k=3)
                else:
                    h_opt, c5_opt = multi_shift_subgradient(h_shake, n_sh, time_left, max_iter=1500, lr_init=0.03, top_k=5)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_shake = h_opt

    # Phase 7: Final fine-tuning with very small steps
    if time_left() > 30:
        n_fin = best_n
        for _ in range(3):
            if time_left() < 15:
                break
            h_fin, c5_fin = subgradient_descent(best_h, n_fin, time_left, max_iter=5000, lr_init=0.01, top_k=3)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        # Binary final polish
        if time_left() > 20:
            h_bf = (best_h > 0.5).astype(float)
            if abs(h_bf.sum() - n_fin / 2.0) <= 1e-6:
                h_bf2, c5_bf2 = binary_multi_flip(h_bf, n_fin, rng, max_rounds=50, time_fn=time_left, max_points=1)
                if c5_bf2 < best_c5:
                    best_c5 = c5_bf2
                    best_h = h_bf2.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
