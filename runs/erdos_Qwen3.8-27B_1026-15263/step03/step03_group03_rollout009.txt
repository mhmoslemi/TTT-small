    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    # Gradient of corr[k] w.r.t. h[j]
    # corr[k] = sum_i h[i]*(1-h[i+k-(n-1)])
    # d corr[k]/d h[j] = (1-h[j+k-(n-1)])*[valid] - h[j-k+(n-1)]*[valid]
    nm1 = n - 1
    grad = np.zeros(n)
    j_idx = np.arange(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        m = k - nm1
        # First term: (1-h[j+m]) for j+m in [0,n-1]
        j_min = max(0, -m)
        j_max = min(n, n - m)
        if j_min < j_max:
            second_idx = j_min + m
            grad[j_min:j_max] += w[k] * (1.0 - h[second_idx:second_idx + (j_max - j_min)])
        # Second term: -h[j-m] for j-m in [0,n-1]
        j_min2 = max(0, m)
        j_max2 = min(n, n + m)
        if j_min2 < j_max2:
            first_idx = j_min2 - m
            grad[j_min2:j_max2] -= w[k] * h[first_idx:first_idx + (j_max2 - j_min2)]
    return grad

One more improvement idea: in the pair descent, instead of trying fixed offsets (j+1, j-1, j+n/2, etc.), I can also try positions that are correlated with the "active" lags. If the maximum correlation is at lag m, then positions j and j+m are "in conflict" (both want to be low when shifted by m). So I should prioritize moves between positions that are separated by the active lag.

Let me implement this: compute the top 3 active lags, and for each j, add j+m (mod n) as a candidate k for each active lag m.

def pair_descent_sweep_v2(h, corr, rng, n_cand=12, sa_temp=0.0):
    n = len(h)
    # Find active lags
    top_lags = np.argsort(corr)[-5:]  # top 5 correlation lags
    top_lags = top_lags[np.argsort(corr[top_lags])[::-1]]  # sort by value desc
    
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Build candidate set
        k_set = set()
        k_set.add((j + 1) % n)
        k_set.add((j - 1 + n) % n)
        k_set.add((j + n // 2) % n)
        # Add active-lag-related positions
        for lag in top_lags:
            m = lag - (n - 1)  # actual shift
            k_set.add((j + m) % n)
            k_set.add((j - m) % n)
        # Fill with random
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.add(rng.randint(n))
        k_list = [k for k in k_set if k != j]
        
        seen = set()
        for k in k_list:
            if k in seen:

            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])

                    break
