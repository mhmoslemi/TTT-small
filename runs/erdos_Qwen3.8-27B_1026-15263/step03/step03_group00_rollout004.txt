def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient via weighted sum of individual lag gradients
    n = len(h)
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx, w_val in enumerate(w):
        if w_val > 1e-10:
            grad += w_val * corr_gradient(h, lags[idx])
    return obj, grad

def nesterov_subgrad(h0, target_sum, alpha, max_iters, verbose=False):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    h_prev = h.copy()
    t = 1.0
    step = 1.0  # initial step size
    
    for k in range(max_iters):
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0, 1)
        
        if alpha < 1e6:  # smooth version
            corr = compute_corr(y)
            m_val = np.max(corr)
            shifted = alpha * (corr - m_val) - np.max(alpha * (corr - m_val))
            exp_vals = np.exp(shifted)
            w = exp_vals / np.sum(exp_vals)
            # Compute gradient
            lags = np.arange(-(n-1), n)
            grad = np.zeros(n)
            for idx in range(len(lags)):
                if w[idx] > 1e-8:
                    grad += w[idx] * corr_gradient(y, lags[idx])
        else:  # true subgradient
            corr = compute_corr(y)
            k_star = int(np.argmax(corr))
            lag = k_star - (n - 1)
            grad = corr_gradient(y, lag)
        
        # Step
        x_new = y - step * grad
        x_new = project_to_sum(x_new, target_sum)
        
        h_prev = h
        h = x_new
        t = t_new
        step = 1.0 / np.sqrt(1.0 + k * 0.1)  # decreasing step
        
    return project_to_sum(h, target_sum)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def top_lag_gradient(h, n_top=5):
    n = len(h)
    corr = compute_corr(h)
    idx = np.argsort(corr)[::-1][:n_top]
    lags = idx - (n - 1)
    g = np.zeros(n)
    for lag in lags:
        g += corr_gradient(h, int(lag))
    g /= float(n_top)
    return g

def smooth_max_slsqp(h0, target_sum, alpha, maxiter=300):
    n = len(h0)
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        ev = np.exp(shifted)
        return m + np.log(np.sum(ev)) / alpha
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        ev = np.exp(shifted)
        w = ev / np.sum(ev)
        lags = np.arange(-(n-1), n)
        g = np.zeros(n)
        for idx in range(len(lags)):
            if w[idx] > 1e-8:
                g += w[idx] * corr_gradient(x, lags[idx])
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
