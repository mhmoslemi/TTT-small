def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute among free variables
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            room_up = np.sum(1.0 - h[mask])
            room_down = np.sum(h[mask])
            if diff > 0:
                adj = min(diff, room_up)
            else:
                adj = max(diff, -room_down)
            if abs(adj) > 1e-15:
                h[mask] += adj / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
                continue
        # If no free variables, adjust all
        if diff > 0:

        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h
