import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad(h, alpha, active_idx=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_idx is None:
        active_idx = np.arange(len(corr))
    work_corr = corr[active_idx]
    n_act = len(work_corr)
    m = np.max(work_corr)
    shifted = alpha * (work_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s / n_act) / alpha) * dx
    g = np.zeros(n)
    for pos in range(n_act):
        if w[pos] < 1e-14:
            continue
        L = int(active_idx[pos]) - (n - 1)
        wp = w[pos]
        if L > 0:
            g[L:] += wp * (1.0 - h[:n - L])
            g[:n - L] -= wp * h[L:]
        elif L < 0:
            Lp = -L
            g[:n - Lp] += wp * (1.0 - h[Lp:])
            g[Lp:] -= wp * h[:n - Lp]
        else:
            g += wp * (1.0 - 2.0 * h)
    return sm_val, g * dx


def get_active_lags(h, top_k=30, frac=0.97):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    threshold = frac * max_corr
    active = np.where(corr >= threshold)[0]
    top_idx = np.argsort(corr)[-top_k:][::-1]
    active = np.unique(np.concatenate([active, top_idx]))
    return active


def slsqp_solve(h0, n, target_sum, alpha, maxiter, active_idx=None):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad(h, alpha, active_idx)
        return v

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha, active_idx)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)


def lbfgs_solve(h0, n, target_sum, alpha, maxiter, active_idx=None, mu=50.0):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad(h, alpha, active_idx)
        penalty = mu * (np.sum(h) - target_sum) ** 2
        return v + penalty

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha, active_idx)
        g += 2.0 * mu * (np.sum(h) - target_sum) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)


def multi_alpha_optimize(h0, n, target_sum, alphas, maxiter_base, time_limit_fn, active_mode=True):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for alpha in alphas:
        if time_limit_fn() < 30:
            break
        if active_mode:
            active = get_active_lags(h, top_k=30, frac=0.96)
        else:
            active = None
        maxiter = min(maxiter_base, max(50, int((1300 - (time.time() - multi_alpha_optimize.t0)) * 0.3)))
        maxiter = max(50, min(maxiter_base, maxiter))
        h_new = slsqp_solve(h, n, target_sum, alpha, maxiter, active)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
        else:
            h_new2 = lbfgs_solve(h, n, target_sum, alpha, maxiter, active, mu=200.0)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < c5_cur:
                h = h_new2
                c5_cur = c5_new2
    return h, c5_cur


def perturb_h(h, n, target_sum, rng, pert_type='noise'):
    if pert_type == 'noise':
        h_pert = h + 0.02 * rng.randn(n)
    elif pert_type == 'binary_push':
        shift = 0.03 * (2.0 * h - 1.0)
        h_pert = h + shift
    elif pert_type == 'frequency':
        k = rng.choice([3, 5, 7, 11])
        freq = k / n
        h_pert = h + 0.01 * np.cos(2.0 * np.pi * freq * np.arange(n))
    else:
        h_pert = h + 0.015 * rng.randn(n)
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_to_sum(h_pert, target_sum)
    return h_pert


def coordinate_refine(h, n, target_sum, max_rounds, rng, deltas=None):
    h = h.copy()
    c5_cur = compute_c5(h)
    if deltas is None:
        deltas = [-0.003, -0.001, 0.001, 0.003, -0.0005, 0.0005]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_test = h.copy()
                h_test[i] = np.clip(h_test[i] + d, 0.0, 1.0)
                h_test = project_to_sum(h_test, target_sum)
                c5_test = compute_c5(h_test)
                if c5_test < best_val - 1e-15:
                    best_val = c5_test
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur


def subgradient_refine(h, n, target_sum, n_iters, lr_init, alpha, time_limit_fn):
    h = h.copy()
    c5_cur = compute_c5(h)
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        if time_limit_fn() < 10:
            break
        _, g = smooth_max_grad(h, alpha)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h = h_new
            lr = min(lr * 1.01, 0.5)
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 50:
                break
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            hv = np.array(initial_h_values, dtype=float)
            best_n = len(hv)
            ts = best_n / 2.0
            best_h = project_to_sum(hv, ts)

        except Exception:
            best_h = None

    if best_h is None:

        best_h = np.full(best_n, 0.5)

    n = best_n

    alphas_stage1 = [50.0, 150.0, 500.0, 1500.0]

    alphas_stage2 = [5000.0, 15000.0, 40000.0]
    if time_left() > 120:
        h_opt2, c5_opt2 = multi_alpha_optimize(best_h, n, ts, alphas_stage2, 250, time_left)
        if c5_opt2 < best_c5:

    if time_left() > 80:
        active = get_active_lags(best_h, top_k=15, frac=0.95)
        for alpha_a in [10000.0, 50000.0]:

                break
            h_act = slsqp_solve(best_h, n, ts, alpha_a, 300, active)
            c5_act = compute_c5(h_act)
            if c5_act < best_c5:

        best_h = np.clip(best_h, 0.0, 1.0)
