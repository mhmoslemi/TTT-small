The pair swap idea: since the sum constraint is ∑h = n/2, we can preserve the constraint by adding δ to h[i] and subtracting δ from h[j]. This is a 2D search that's more efficient than 1D coordinate descent.

Let me also think about the structure. What does a good solution look like? The integral ∫₀² h(x)(1-h(x+k)) dx measures the "overlap" of h with its complement shifted by k. To minimize the max over k, we want h to be as "uniform" as possible in its overlap properties.

For a truly uniform h = 0.5 everywhere:
∫₀.₅(1-0.₅) dx = 0.5 * 0.5 * 2 = 0.5

So the uniform solution gives C₅ = 0.5. We need to do much better, getting down to 0.38.

The optimal solution likely has a non-trivial structure. Let me think about what that structure might be...

For the Erdős problem, the overlap is related to the autocorrelation. If h is 1 on a set A and 0 on [0,2]\A, then h(x)(1-h(x+k)) = 1 when x ∈ A and x+k ∉ A (for x where both are defined). This is related to the difference set A - A.

For a step function taking values in [0,1], the optimal is somewhere between the all-0.5 case and a binary structure.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, k_top, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_top:]
    m = max(corr[top_idx])
    
    if k_top == 1:
        g = np.zeros(n)
        k = top_idx[0]
        l = k - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + l:hi2 + l]
        return m, g, corr
    
    shifted = alpha * (corr[top_idx] - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    
    g = np.zeros(n)
    for ii, k in enumerate(top_idx):
        wl = w[ii]
        if wl < 1e-15:
            continue
        l = k - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g, corr

def slsqp_opt(h0, n, ts, alpha, k_top, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _, _ = topk_grad(h, k_top, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g, _ = topk_grad(h, k_top, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, ts)
    except:
        return project_to_sum(h0, ts)

def penalty_descent(h0, n, ts, C_target, n_iters, lr_init, top_k):
    dx = 2.0 / n
    C_corr = C_target / dx
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            excess = corr[k] - C_corr
            if excess <= 0:
                continue
            l = k - (n - 1)
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += 2.0 * excess * (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= 2.0 * excess * h[lo2 + l:hi2 + l]
        
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-15:
            break
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h):
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
    return best_h, best_c5

def pair_perturb_search(h0, n, ts, n_trials, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for t in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.03, 0.03)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, ts, alpha, k_top, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k_top:]
        m = max(corr[top_idx])
        if k_top == 1:
            return m
        shifted = alpha * (corr[top_idx] - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        return m + np.log(s) / alpha
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        nz = len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k_top:]
        m = max(corr[top_idx])
        g = np.zeros(nz)
        if k_top == 1:
            k = top_idx[0]
            l = k - (nz - 1)
            lo = max(0, l)
            hi = min(nz, nz + l)
            if lo < hi:
                g[lo:hi] += (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(nz, nz - l)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2 + l:hi2 + l]
        else:
            shifted = alpha * (corr[top_idx] - m)
            shifted = np.clip(shifted, -500.0, 500.0)
            exp_vals = np.exp(shifted)
            s = exp_vals.sum()
            w = exp_vals / s
            for ii, k in enumerate(top_idx):
                wl = w[ii]
                if wl < 1e-15:
                    continue
                l = k - (nz - 1)
                lo = max(0, l)
                hi = min(nz, nz + l)
                if lo < hi:
                    g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
                lo2 = max(0, -l)
                hi2 = min(nz, nz - l)
                if lo2 < hi2:
                    g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def penalty_descent(h0, n, ts, C_target, n_iters, lr_init, top_k):
    dx = 2.0 / n
    C_corr = C_target / dx
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            excess = corr[k] - C_corr
            if excess <= 0:
                continue
            l = k - (n - 1)
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += 2.0 * excess * (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= 2.0 * excess * h[lo2 + l:hi2 + l]
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-15:
            break
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h):
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
    return best_h, best_c5

def guided_pair_search(h0, n, ts, n_trials, rng, accept_rate=0.02):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for t in range(n_trials):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        l = k_max - (n - 1)
        if l == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            delta = rng.uniform(-0.015, 0.015)
        else:
            lo_i = max(0, -l)
            hi_i = min(n, n - l)
            if hi_i - lo_i < 2:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(n - 1)) % n
                delta = rng.uniform(-0.015, 0.015)
            else:
                idxs = np.arange(lo_i, hi_i)
                h_i = h[idxs]
                h_j = h[idxs + l]
                bad = h_i * (1.0 - h_j)
                num_top = min(15, len(bad))
                top_bad = np.argsort(bad)[-num_top:]
                pick = top_bad[rng.randint(num_top)]
                i = int(idxs[pick])
                j = i + l
                if h_i[pick] > 0.3 and (1.0 - h_j[pick]) > 0.3:
                    if h_i[pick] > h_j[pick]:
                        delta = rng.uniform(0.005, 0.02)
                    else:
                        delta = rng.uniform(-0.02, -0.005)
                else:
                    delta = rng.uniform(-0.015, 0.015)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
