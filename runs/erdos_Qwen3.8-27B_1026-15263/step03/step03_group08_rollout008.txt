import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    # More efficient projection
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Distribute to elements not at bounds
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            # All at bounds, distribute to all
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lags(h, lags):
    """Compute gradient of sum of correlations at given lags."""
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        l = k - (n - 1)  # actual shift
        # corr[k] = sum_i h[i] * (1 - h[i+l]) for valid i
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            # d/dh[i] of h[i]*(1-h[i+l]) = (1-h[i+l])
            # d/dh[i+l] of h[i]*(1-h[i+l]) = -h[i]
            g[lo:hi] += (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + l:hi2 + l]
    return g

def projected_gradient_step(h, g, target_sum, lr):
    """Take a gradient step with projection."""
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def active_set_ascent(h, target_sum, n_iters, K, lr_init):
    """Active-set gradient descent minimizing max correlation."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    h_cur = h.copy()
    
    for it in range(n_iters):
        corr = compute_corr(h_cur)
        C5_cur = corr.max() * dx
        
        # Get top K active lags
        top_idx = np.argsort(corr)[-K:]
        
        # Gradient from top K lags
        g = grad_at_lags(h_cur, top_idx)
        
        # Project gradient to sum=0 subspace
        g_mean = g.mean()
        g_proj = g - g_mean
        
        # Backtracking line search
        h_trial = projected_gradient_step(h_cur, g_proj, target_sum, lr)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < C5_cur:
            h_cur = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_cur.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-8:
                lr = lr_init * 0.1
    return best_h, best_c5

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - np.sum(h)

            break

        if nmask > 0:
            h[mask] += diff / nmask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    """Gradient of correlation at shift l."""
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        # h[i] * (1-h[i+l]) for i in [0, n-l)

            g[i + l] -= h[i]

        for i in range(ll, n):
            g[i - ll] += (1.0 - h[i])
            g[i] -= h[i - ll]
    return g

    g = np.zeros(len(h))
    for k in lags:
        l = k - (len(h) - 1)
        g += grad_lag(h, l)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nmask = np.count_nonzero(mask)
        if nmask > 0:
            h[mask] += diff / nmask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_multi(h, lags):
    n = len(h)
    g = np.zeros(n)
    for l in lags:
        if l == 0:
            g[:] = 1.0 - 2.0 * h
        elif l > 0:
            g[:n - l] += (1.0 - h[l:])
            g[l:] -= h[:n - l]
        else:
            mp = -l
            g[mp:] += (1.0 - h[:n - mp])
            g[:n - mp] -= h[mp:]
    return g

def active_descent(h0, target_sum, K, max_iters, lr_init, seed=0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    active_idx = np.argsort(corr)[-K:]
    lr = lr_init
    stagnant = 0
    rng = np.random.RandomState(seed)

    for it in range(max_iters):
        if it % 50 == 0 or it == 0:
            active_idx = np.argsort(corr)[-K:]
        shifts = active_idx - (n - 1)
        g = grad_multi(h, shifts)
        g -= g.mean()

        accepted = False
        for _ in range(25):
            h_try = h - lr * g
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, target_sum)
            corr_try = compute_corr(h_try)
            c5_try = float(corr_try.max() * dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            if c5_try < corr.max() * dx * 1.0 - 1e-9:
                h = h_try
                corr = corr_try
                lr = min(lr * 1.03, 1.0)
                stagnant = 0
                accepted = True
                break
            else:
                lr *= 0.75
        if not accepted:
            lr = max(lr_init * 0.3, 1e-9)
            stagnant += 1
            if stagnant > 80:
                h = h + rng.uniform(-0.005, 0.005, n)
                h = project_to_sum(h, target_sum)
                corr = compute_corr(h)
                active_idx = np.argsort(corr)[-K:]
                stagnant = 0

    return best_h, best_c5

def coordinate_refine(h0, target_sum, max_rounds, step_sizes):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur

    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            h_cur = h
            c5_i = c5_cur
            for d in step_sizes:
                h_try = h_cur.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < c5_i - 1e-15:
                    h_cur = h_try
                    c5_i = c5_try
            if c5_i < c5_cur - 1e-15:
                h = h_cur
                c5_cur = c5_i
                improved = True
        if not improved:
            break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def make_templates(n, ts, rng):
    templates = []
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 8):
        for amp in [0.15, 0.25, 0.35, 0.45]:
            for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0, np.pi]:
                h_t = 0.5 + amp * np.sin(np.pi * freq * x + phase)
                h_t = project_to_sum(np.clip(h_t, 0, 1), ts)
                templates.append(h_t)
    for freq1 in [1, 2, 3]:
        for freq2 in [2, 4, 5]:
            h_t = 0.5 + 0.2 * np.sin(np.pi * freq1 * x) + 0.15 * np.sin(np.pi * freq2 * x)
            h_t = project_to_sum(np.clip(h_t, 0, 1), ts)
            templates.append(h_t)
    return templates

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
