sum_k w_k * h[(j-k)%n] = (w * h)_j (same thing, circular convolution is symmetric in this sense)

Wait: (w * h)_j = sum_k w[k] * h[(j-k)%n] (standard circular convolution)
And sum_k w[k] * h[(j+k)%n] = sum_k w[k] * h[((j+2k)-k)%n]... 

Actually, sum_k w[k] * h[(j+k)%n] = (w_rev * h)_j where w_rev[k] = w[-k mod n].

If w is symmetric (w[k] = w[n-k]), then both sums are equal to (w*h)_j.

In our case, w comes from the softmax of O(k), and O(k) = O(n-k) (by symmetry: O(k) = sum h[i](1-h[i+k]) and O(n-k) = sum h[i](1-h[i+k-n]) = sum h[i](1-h[i-k]) = sum h[j-k](1-h[j]) = sum h[j](1-h[j+k]) (by reindexing) = O(k)).

Wait, is O(k) = O(n-k)? Let's check:
O(n-k) = sum_i h[i] * (1 - h[(i+n-k)%n]) = sum_i h[i] * (1 - h[(i-k)%n])
Let j = i-k: = sum_j h[j+k] * (1 - h[j]) = sum_j (1-h[j]) * h[j+k] = sum_j h[j+k] * (1-h[j])

And O(k) = sum_i h[i] * (1-h[i+k]). Let j = i+k: = sum_j h[j-k] * (1-h[j]) = sum_j (1-h[j]) * h[j-k]

So O(n-k) = sum_j h[j+k]*(1-h[j]) and O(k) = sum_j h[j-k]*(1-h[j]).

These are NOT the same in general. O(k) = O(n-k) iff sum_j h[j+k](1-h[j]) = sum_j h[j-k](1-h[j]), which is not generally true.

However, the "full" correlation IS symmetric: corr_full[k] = corr_full[2n-2-k] (for the full correlation of length 2n-1).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = (m_val + max_shift + np.log(total) / alpha) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def periodic_overlap_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    O = np.zeros(n)
    for k in range(n):
        O[k] = np.sum(h * h1[np.roll(np.arange(n), -k)])
    O_k = O.copy()
    m_val = np.max(O_k)
    shifted = O_k - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = (m_val + max_shift + np.log(total) / alpha) * dx
    grad = np.zeros(n)
    for k in range(n):
        if k == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        else:
            grad += w[k] * (1.0 - h - np.roll(h, k) - np.roll(h, -k) + 2.0 * h)
    grad *= dx
    return obj, grad

def projected_gradient_descent(h0, alpha, max_iter, lr_init, time_fn, checkpoint=None):
    n = len(h0)
    h = h0.copy()
    lr = lr_init
    best_obj = compute_c5(h)
    best_h = h.copy()
    patience = 0
    for it in range(max_iter):
        if time_fn() < 8:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
        else:
            lr = min(lr * 1.05, 0.3)
        h = h_new
        if patience > 200 and it > 100:
            break
    if checkpoint and compute_c5(best_h) < best_obj:
        best_obj = compute_c5(best_h)
    return best_h, best_obj

def fw_iteration(h, alpha, gamma):
    n = len(h)
    obj, grad = smooth_max_obj_and_grad(h, alpha)
    idx = np.argpartition(grad, n // 2)[:n // 2]
    s = np.zeros(n)
    s[idx] = 1.0
    h_new = (1.0 - gamma) * h + gamma * s
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_box_sum(h_new, n / 2.0)
    return h_new

def fw_then_pg(h0, alpha, n_fw, n_pg, lr_init, time_fn):
    n = len(h0)
    h = h0.copy()
    for t in range(n_fw):
        if time_fn() < 10:
            break
        gamma = 2.0 / (t + 2)
        h = fw_iteration(h, alpha, gamma)
    h, _ = projected_gradient_descent(h, alpha, n_pg, lr_init, time_fn)
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new

def fourier_coeffs(h, n_harm):
    n = len(h)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    C = np.zeros((n, 2 * n_harm))
    for m in range(1, n_harm + 1):
        C[:, 2 * (m - 1)] = np.cos(np.pi * m * x)
        C[:, 2 * (m - 1) + 1] = np.sin(np.pi * m * x)
    target = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(C, target, rcond=None)
    return coeffs

def fourier_to_grid(coeffs, n, n_harm):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, n_harm + 1):
        a = coeffs[2 * (m - 1)]
        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return h

def fourier_obj(c, n, n_harm, alpha):
    h = fourier_to_grid(c, n, n_harm)
    h = project_box_sum(h, n / 2.0)
    obj, _ = smooth_max_obj_and_grad(h, alpha)
    return obj

def targeted_lag_optimize(h, n_lags, alpha, n_iter, lr, time_fn):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    for it in range(n_iter):
        if time_fn() < 10:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        top_k = np.argsort(corr)[-n_lags:]
        target = np.zeros(len(corr))
        for k in top_k:
            target[k] = corr[k]
        m_val = np.max(target[target > 0]) if np.any(target > 0) else 0
        for k in top_k:
            target[k] = 1.0
        shifted = target - m_val
        max_s = np.max(shifted)
        ev = np.exp(alpha * (shifted - max_s))
        w = ev / np.sum(ev)
        obj, grad = smooth_max_obj_and_grad(h_curr, alpha)
        h_new = h_curr - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        if compute_c5(h_new) < compute_c5(h_curr):
            h_curr = h_new
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.7
    return h_curr

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
