import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_to_box_sum(y, target_sum):
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = compute_corr(h) * dx
    m_val = np.max(corr_raw)
    shifted = alpha * (corr_raw - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha

    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            kp = -k
            grad[kp:] += wk * (1.0 - h[:n - kp])
            grad[:n - kp] -= wk * h[kp:]
    return obj, grad

def projected_gradient_smooth(h, target_sum, alpha_schedule, n_iters, lr0=0.01):
    n = len(h)
    h = h.copy()
    best_obj = smooth_max_obj_grad(h, alpha_schedule[0])[0]
    best_h = h.copy()
    prev_grad = None
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        alpha = alpha_schedule[0] * (alpha_schedule[-1] / alpha_schedule[0]) ** frac
        obj, grad = smooth_max_obj_grad(h, alpha)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        lr = lr0 / (1.0 + 0.01 * it)
        if prev_grad is not None:
            gy = np.dot(grad - prev_grad, h - prev_h)
            yy = np.dot(grad - prev_grad, grad - prev_grad)
            if yy > 1e-16:
                bb_lr = gy / yy
                if 0 < bb_lr < 10 * lr:
                    lr = 0.5 * (lr + bb_lr)
        prev_grad = grad.copy()
        prev_h = h.copy()
        h_new = project_to_box_sum(h - lr * grad, target_sum)
        new_obj = smooth_max_obj_grad(h_new, alpha)[0] if it % 10 == 0 else obj + 1e-9
        if new_obj <= obj + 1e-10:
            h = h_new
        else:
            lr *= 0.5
            h = project_to_box_sum(h - lr * grad, target_sum)
    return h, smooth_max_obj_grad(h, alpha_schedule[-1])[0]

def corr_gradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

def coordinate_descent(h, target_sum, max_iters=5000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 80:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def subgradient_minmax(h, target_sum, n_iters=3000, lr0=0.005):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(n_iters):
        corr = compute_corr(h) * (2.0 / n)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single(h, lag) * (2.0 / n)
        lr = lr0 / (1.0 + 0.005 * it)
        h_new = project_to_box_sum(h - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            lr *= 0.5
            h_new = project_to_box_sum(h - lr * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def fourier_perturb(h, rng, n_freqs=5, amp=0.1):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    perturbation = np.zeros(n)
    for j in range(n_freqs):
        freq = rng.randint(2, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation += amp * np.cos(np.pi * freq * x + phase)
    return h + perturbation

def slsqp_local(h0, n, target_sum, alpha, maxiter=300):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:

            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:

        best_h = np.full(best_n, 0.5)

    ts_parent = best_n / 2.0

    alpha_schedule_pg = np.array([50.0, 200.0, 1000.0, 5000.0, 20000.0])

    for alpha in [50, 200, 1000, 5000]:

            break
        h_new = slsqp_local(best_h, best_n, ts_parent, alpha, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 100:
