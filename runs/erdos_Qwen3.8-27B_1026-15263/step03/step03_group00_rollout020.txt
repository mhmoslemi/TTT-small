```python
def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    # Only use top 30 lags for efficiency
    top_k = min(30, len(corr))
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[top_idx]
    m_top = np.max(corr_top)
    shifted = alpha * (corr_top - m_top)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        if w[j] < 1e-10:
            continue
        lag = idx - (n - 1)
        grad += w[j] * corr_gradient(h, lag)
    # Scale to match the full objective gradient
    # The full smooth max gradient would use all lags, but we approximate
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def fine_coordinate_descent(h, target_sum, max_iters=50000, step_frac=0.05):
    """Very fine coordinate descent for final polish."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)

        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))

        if g[i_inc] >= g[i_dec] - 1e-13:

        move = step_frac * max_move
        if move < 1e-14:

        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
```

```python
        #     When h[a] goes from 1 to 0, (1-h[a]) goes from 0 to 1.
        #     So the change is h[a-lag]*(1-0) - h[a-lag]*(1-1) = h[a-lag] - 0 = h[a-lag]
        #     Wait no: the OLD contribution was h[a-lag]*(1-1) = 0, NEW is h[a-lag]*(1-0) = h[a-lag]
        #     So change = +h[a-lag]
        
        # Let me just compute it directly:
        # old_corr[m] = sum_i h[i]*(1-h[i+lag_m])
        # After flip: h_new[a] = 0, h_new[b] = 1
        
        # For efficiency, let me compute the delta:
        deltas = np.zeros(len(affected_m), dtype=float)
        for j, m in enumerate(affected_m):
            lag = m - (n - 1)
            delta = 0.0
            # Contribution from h[a] as first factor
            if 0 <= a + lag < n:
                delta += -(1.0 - h[a + lag])  # h[a] goes 1→0, so -(1-h[a+lag])
            # Contribution from h[a] as second factor (1-h[a])
            ia = a - lag
            if 0 <= ia < n:
                delta += h[ia]  # (1-h[a]) goes 0→1, so +h[ia]
            # Contribution from h[b] as first factor
            if 0 <= b + lag < n:
                delta += (1.0 - h[b + lag])  # h[b] goes 0→1, so +(1-h[b+lag])
            # Contribution from h[b] as second factor (1-h[b])
            ib = b - lag
            if 0 <= ib < n:
                delta += -h[ib]  # (1-h[b]) goes 1→0, so -h[ib]
            deltas[j] = delta
        
        new_corr_affected = corr[affected_m] + deltas
        new_max = max(np.max(new_corr_affected), 
                      np.max(corr[np.setdiff1d(np.arange(2*n-1), affected_m, assume_unique=False)]) if len(affected_m) < 2*n-1 else 0)
        
        # Actually this is getting too slow with the setdiff. Let me just track the max differently.
        
        # Simpler approach: just compute the full new correlation for the affected entries
        # and track the global max
        old_max = c5_cur
        # The new max is either:
        # 1. The max of the unaffected entries (which is at most old_max, since the old max might be in affected)
        # 2. The max of the new affected entries
        
        # To be safe, let's just compute: if old_max was at an unaffected position, it stays.
        # Otherwise, we need to find the new max among unaffected + new affected.
        
        # For simplicity, let's just compute the max of all:
        # new_corr = corr.copy(); new_corr[affected_m] = new_corr_affected
        # new_max = np.max(new_corr)
        # But this is O(n) per iteration which is fine.
        
        new_corr = corr.copy()
        new_corr[affected_m] = new_corr_affected
        new_max = float(np.max(new_corr))
        
        delta_c5 = new_max - old_max
        if delta_c5 < 0 or (T > 1e-8 and rng.random() < np.exp(-delta_c5 / T)):
            h = h.copy()
            h[a] = 0.0
            h[b] = 1.0
            corr = new_corr
            c5_cur = new_max
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        # else: don't accept
    
    return best_h, best_c5
```

Hmm, this is O(n) per iteration due to the `corr.copy()` and `np.setdiff1d`. For n=540, that's about 1080 operations per iteration (for the copy). With 100000 iterations, that's about 108 million operations. Should be fine.

But wait, the inner loop over `affected_m` could be large. Let me vectorize it.

Actually, let me reconsider the whole approach. The affected_m set has at most 4*(n) entries (roughly 2n from each flip). For n=540, that's about 2160 entries. The loop over them in Python would be slow. Let me vectorize.

Actually, I think a cleaner approach is:

For flip at position a (1→0):
- The correlation change is a vector of length 2n-1
- At position m (lag = m-(n-1)):
  - If 0 ≤ a+lag < n: change -= (1-h[a+lag])
  - If 0 ≤ a-lag < n: change += h[a-lag]

For flip at position b (0→1):
- At position m (lag = m-(n-1)):
  - If 0 ≤ b+lag < n: change += (1-h[b+lag])
  - If 0 ≤ b-lag < n: change -= h[b-lag]

I can compute this vectorized:

```python
def compute_flip_delta(h, a, b, n):
    """Compute the change in correlation vector when h[a]:1→0 and h[b]:0→1."""
    lag = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    delta = np.zeros(2*n-1)
    
    # Flip a: 1→0
    # Term 1: a+lag valid
    valid1 = (a + lag >= 0) & (a + lag < n)
    delta[valid1] -= (1.0 - h[a + lag[valid1]])
    # Term 2: a-lag valid
    valid2 = (a - lag >= 0) & (a - lag < n)
    delta[valid2] += h[a - lag[valid2]]
    
    # Flip b: 0→1
    # Term 1: b+lag valid
    valid3 = (b + lag >= 0) & (b + lag < n)
    delta[valid3] += (1.0 - h[b + lag[valid3]])
    # Term 2: b-lag valid
    valid4 = (b - lag >= 0) & (b - lag < n)
    delta[valid4] -= h[b - lag[valid4]]
    
    return delta
```

This is O(n) per call but vectorized, so it should be fast.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    top_k = min(30, len(corr))
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[top_idx]
    m_top = np.max(corr_top)
    shifted = alpha * (corr_top - m_top)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        if w[j] < 1e-10:
            continue
        lag = idx - (n - 1)
        grad += w[j] * corr_gradient(h, lag)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_flip_delta(h, a, b, n):
    lag = np.arange(2 * n - 1) - (n - 1)
    delta = np.zeros(2 * n - 1)
    valid1 = (a + lag >= 0) & (a + lag < n)
    if np.any(valid1):
        delta[valid1] -= (1.0 - h[a + lag[valid1]])
    valid2 = (a - lag >= 0) & (a - lag < n)
    if np.any(valid2):
        delta[valid2] += h[a - lag[valid2]]
    valid3 = (b + lag >= 0) & (b + lag < n)
    if np.any(valid3):
        delta[valid3] += (1.0 - h[b + lag[valid3]])
    valid4 = (b - lag >= 0) & (b - lag < n)
    if np.any(valid4):
        delta[valid4] -= h[b - lag[valid4]]
    return delta

def binary_simulated_annealing(h_init, target_sum, max_iters=100000, T_init=0.3, T_final=0.005, rng=None):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    h = project_to_sum(h, target_sum)
    corr = compute_corr(h)
    c5_cur = float(np.max(corr))
    best_c5 = c5_cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(1, max_iters))
    t_start = time.time()
    for it in range(max_iters):
        if it % 5000 == 0:
            if time.time() - t_start > 120:
                break
        T *= cooling
        if T < 1e-8:
            T = 1e-8
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        if a == b:
            continue
        delta = compute_flip_delta(h, a, b, n)
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        delta_c5 = new_max - c5_cur
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h[a] = 0.0
            h[b] = 1.0
            corr = new_corr
            c5_cur = new_max
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    return best_h, best_c5

def multi_lag_subgradient(h, target_sum, max_iters=5000, step_init=0.1, momentum=0.8):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    step = step_init
    c5 = compute_c5(h)
    patience = 0
    t_start = time.time()
    for it in range(max_iters):
        if it % 1000 == 0 and time.time() - t_start > 60:
            break
        corr = compute_corr(h)
        m_val = np.max(corr)
        alpha_w = 30.0
        weights = np.exp(alpha_w * (corr - m_val))
        total_w = np.sum(weights)
        if total_w < 1e-30:
            idx_max = int(np.argmax(corr))
            g = corr_gradient(h, idx_max - (n - 1))
        else:
            weights /= total_w
            g = np.zeros(n)
            for i in range(len(corr)):
                if weights[i] < 1e-8:
                    continue
                lag = i - (n - 1)
                g += weights[i] * corr_gradient(h, lag)
        v = momentum * v - step * g
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            patience = 0
        else:
            patience += 1
            step *= 0.97
            v *= 0.5
            if patience > 300 or step < 1e-12:
                break
    return h, c5

def coordinate_descent(h, target_sum, max_iters=20000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    t_start = time.time()
    for it in range(max_iters):
        if it % 2000 == 0 and time.time() - t_start > 60:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 200:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def fine_cd(h, target_sum, max_iters=30000, step_frac=0.05):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    t_start = time.time()
    for it in range(max_iters):
        if it % 2000 == 0 and time.time() - t_start > 45:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-13:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-14:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-15:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 500:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5
```
