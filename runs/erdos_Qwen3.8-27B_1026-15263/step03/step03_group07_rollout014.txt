```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum} using bisection."""
    h = h.astype(float)
    lo = h.min() - target_sum - 1
    hi = h.max() + target_sum + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max_grad_fft(h, alpha, active_idx=None):
    """Compute smooth max value and gradient using FFT-based convolution."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if active_idx is not None and len(active_idx) < len(corr):
        corr_sub = corr[active_idx]
        m = np.max(corr_sub)
        shifted = np.clip(alpha * (corr_sub - m), -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = np.zeros(2 * n - 1)
        w[active_idx] = exp_vals / s
        sm_val = m + np.log(s / len(exp_vals)) / alpha
    else:
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s / (2 * n - 1)) / alpha
    
    # Gradient via FFT
    S1 = np.convolve(1.0 - h, w, mode='full')[:n]
    S2 = np.correlate(h, w, mode='full')[:n]
    g = (S1 - S2) / alpha
    return sm_val, g

def get_active_lags(h, frac=0.99):
    """Get indices of lags near the maximum."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    threshold = frac * np.max(corr)
    active = np.where(corr >= threshold)[0]
    return active

def slsqp_fast(h0, n, target_sum, alpha, maxiter, active_idx=None):
    """SLSQP optimization with fast FFT gradient."""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad_fft(h, alpha, active_idx)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_fft(h, alpha, active_idx)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def pgd_fast(h0, n, target_sum, alpha, n_iters, lr_init, active_idx=None):
    """Projected gradient descent with fast FFT gradient."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        sm_val, g = smooth_max_grad_fft(h, alpha, active_idx)
        # adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        step = lr / (g_norm + 1e-10)
        h_new = project_feasible(h - step * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.1, 2.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 50 or lr < 1e-10:
                break
    return best_h, best_c5

def coordinate_descent_fast(h0, n, target_sum, alpha, max_rounds, rng, active_idx=None):
    """Coordinate descent using direct C5 evaluation."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try a few values around current h[i]
            best_val = c5_cur
            best_hi = h[i]
            # Compute sensitivity of corr to h[i]
            # corr[k] = sum_j h[j]*(1-h[j-s_k]) where s_k = k-(n-1)
            # dcorr[k]/dh[i] = (1-h[i-s_k])*1[0<=i-s_k<n] - h[i+s_k]*1[0<=i+s_k<n]
            s = np.arange(-(n-1), n)  # shifts
            # For shift s, corr at index (s+n-1) changes by:
            # (1-h[i-s])*1[0<=i-s<n] - h[i+s]*1[0<=i+s<n]
            # Build sensitivity vector
            sens = np.zeros(2*n-1)
            # Positive shifts: s > 0, i-s is valid when 0<=i-s<n
            for s_val in range(-(n-1), n):
                k = s_val + n - 1  # index in corr array
                ds = 0.0
                if 0 <= i - s_val < n:
                    ds += (1.0 - h[i - s_val])
                if 0 <= i + s_val < n:
                    ds -= h[i + s_val]
                sens[k] = ds
            
            # Now try a few deltas
            for d in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                h_new = h.copy()
                h_new[i] = new_val
                h_new = project_feasible(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_hi = new_val
            if best_hi != h[i]:
                h[i] = best_hi
                h = project_feasible(h, target_sum)
                c5_cur = min(c5_cur, best_val)
                improved = True
        if not improved:
            break
    return h, c5_cur
```

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_gs = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float)
    lo = h.min() - target_sum - 1.0
    hi = h.max() + target_sum + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max_grad_fft(h, alpha, active_idx=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    
    if active_idx is not None and len(active_idx) < nc:
        corr_sub = corr[active_idx]
        m = np.max(corr_sub)
        shifted = np.clip(alpha * (corr_sub - m), -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = np.zeros(nc)
        w[active_idx] = exp_vals / s
        sm_val = m + np.log(s / len(exp_vals)) / alpha
    else:
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s / nc) / alpha
    
    S1 = np.convolve(1.0 - h, w, mode='full')[:n]
    S2 = np.correlate(h, w, mode='full')[:n]
    g = (S1 - S2) / alpha
    return sm_val, g

def get_active_lags(h, frac=0.995):
    corr = np.correlate(h, 1.0 - h, mode='full')
    threshold = frac * np.max(corr)
    return np.where(corr >= threshold)[0]

def _slsqp_obj(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_grad_fft(h, _gs['alpha'], _gs.get('active_idx'))
    return v

def _slsqp_grad(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_grad_fft(h, _gs['alpha'], _gs.get('active_idx'))
    return g

def _con_fun(x):
    return np.sum(x) - _gs['target_sum']

def _con_jac(x):
    return np.ones(len(x))

def run_slsqp(h0, n, target_sum, alpha, maxiter, active_idx=None):
    _gs['alpha'] = alpha
    _gs['target_sum'] = target_sum
    _gs['active_idx'] = active_idx
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h0.copy(), jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def pgd_optimize(h0, n, target_sum, alpha, n_iters, lr, active_idx=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(n_iters):
        sm_val, g = smooth_max_grad_fft(h, alpha, active_idx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        h_new = project_feasible(h - lr * g / (g_norm + 1e-10), target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.05, 1.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 100 or lr < 1e-10:
                break
    return best
