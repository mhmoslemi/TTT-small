from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_gradient(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    center = n - 1
    for idx_pos in range(len(active_idx)):
        wm = w[idx_pos]
        if wm < 1e-10:
            continue
        k = active_idx[idx_pos] - center
        if k == 0:
            g += wm * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += wm * (1.0 - h[:n-k])
            g[:n-k] -= wm * h[k:]
        else:
            k2 = -k
            g[:n-k2] += wm * (1.0 - h[k2:])
            g[k2:] -= wm * h[:n-k2]
    return sm_val, g

def get_top_lags(h, top_m=30):

def slsqp_active(h0, n, target_sum, active_idx, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = fast_gradient(h, active_idx, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = fast_gradient(h, active_idx, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def proj_grad_descent(h0, n, target_sum, active_idx, alpha, n_iters, lr_init):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = fast_gradient(h, active_idx, alpha)
        g_proj = g - np.mean(g)
        h_new = np.clip(h - lr * g_proj, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

            h = h_new
            lr = min(lr * 1.03, 0.3)

            lr *= 0.6

                break
    return best_h, best_c5

def coord_descent_fine(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)

    return h_dst

    return h
