def project_box_sum(h, target_sum):

    lo = -2.0
    hi = 2.0

        else:

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    m_lo = max(0, nm1 - j)
    m_hi = min(2 * nm1, 2 * nm1 - j)
    if m_lo <= m_hi:
        idx = j - nm1 + np.arange(m_lo, m_hi + 1)
        B[m_lo:m_hi + 1] += (1.0 - h[idx])
    m_lo2 = max(0, j)
    m_hi2 = min(2 * nm1, j + nm1)
    if m_lo2 <= m_hi2:
        idx2 = j + nm1 - np.arange(m_lo2, m_hi2 + 1)
        B[m_lo2:m_hi2 + 1] += -h[idx2]
    return B

def solve_1d_pair(corr, D, lo, hi):

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = m_val + log_sum / alpha
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        # First term: sum_m w[m] * (1 - h[j + m - nm1]) for valid
        m_lo = max(0, nm1 - j)
        m_hi = min(2 * nm1, 2 * nm1 - j)
        if m_lo <= m_hi:
            idx = j + np.arange(m_lo, m_hi + 1) - nm1
            grad[j] += np.dot(w[m_lo:m_hi + 1], 1.0 - h[idx])
        # Second term: -sum_m w[m] * h[j - m + nm1] for valid
        m_lo2 = max(0, j - nm1)
        m_hi2 = min(2 * nm1, j)
        if m_lo2 <= m_hi2:
            idx2 = j - np.arange(m_lo2, m_hi2 + 1) + nm1
            grad[j] -= np.dot(w[m_lo2:m_hi2 + 1], h[idx2])
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(90):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    m_lo = max(0, nm1 - j)
    m_hi = min(2 * nm1, 2 * nm1 - j)
    if m_lo <= m_hi:
        idx = j - nm1 + np.arange(m_lo, m_hi + 1)
        B[m_lo:m_hi + 1] += (1.0 - h[idx])
    m_lo2 = max(0, j)
    m_hi2 = min(2 * nm1, j + nm1)
    if m_lo2 <= m_hi2:
        idx2 = j + nm1 - np.arange(m_lo2, m_hi2 + 1)
        B[m_lo2:m_hi2 + 1] += -h[idx2]
    return B

def solve_1d_pair(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 5) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_pair(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = m_val + log_sum / alpha
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        m_lo = max(0, nm1 - j)
        m_hi = min(2 * nm1, 2 * nm1 - j)
        if m_lo <= m_hi:
            idx = j + np.arange(m_lo, m_hi + 1) - nm1
            grad[j] += np.dot(w[m_lo:m_hi + 1], 1.0 - h[idx])
        m_lo2 = max(0, j - nm1)
        m_hi2 = min(2 * nm1, j)
        if m_lo2 <= m_hi2:
            idx2 = j - np.arange(m_lo2, m_hi2 + 1) + nm1
            grad[j] -= np.dot(w[m_lo2:m_hi2 + 1], h[idx2])
    return obj, grad

    h_prev = h.copy()
    t_k = 1.0
    step = 2.0 / (alpha + 2.0)
    best_val = float('inf')
    best_h = h0.copy()

    for it in range(n_iters):
        if it % pair_every == pair_every - 1:
            corr = np.correlate(h, 1.0 - h, mode="full")
            h, corr, _ = pair_descent_sweep(h, corr, rng, n_cand=8)
            h_prev = h.copy()
            t_k = 1.0
            c5_check = compute_c5(h)
            if c5_check < best_val:
                best_val = c5_check
                best_h = h.copy()
            continue

        obj, grad = smooth_obj_and_grad(h, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) * 0.5
        y = h + (t_k - 1.0) / t_new * (h - h_prev)
        h_new = y - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, target_sum)
        h_prev = h
        h = h_new
        t_k = t_new

    c5_final = compute_c5(h)
    if c5_final < best_val:
        best_val = c5_final
        best_h = h.copy()
    return best_h

        corr = np.correlate(h_f, 1.0 - h_f, mode="full")
        for _ in range(8):
            if time.time() >= t_end4:
                break
            h_f, corr, _ = pair_descent_sweep(h_f, corr, rng, n_cand=12)
        c5 = compute_c5(h_f)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()
            best_n = n_f

    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall = 0
    restart_count = 0
    while time.time() < t_start + budget_s - 3:
        h_fin, corr_fin, imp = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=14)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall = 0
        else:
            stall += 1
        if stall >= 150:
            restart_count += 1
            if restart_count % 3 == 0:
                h_fin = fourier_perturb(best_h, n_fin, rng, 0.04)
            elif restart_count % 3 == 1:
                h_fin = best_h + rng.uniform(-0.025, 0.025, n_fin)
            else:
                h_fin = fourier_perturb(best_h, n_fin, rng, 0.08)
            h_fin = project_box_sum(h_fin, n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall = 0

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
