import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def smooth_max_grad_fft(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    shifted = np.clip(alpha * (c - max_c), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = max_c + np.log(s / len(c)) / alpha
    
    w_arr = w
    w_flip = w_arr[::-1]
    
    conv1 = _fft_convolve(w_arr, h)
    conv2 = _fft_convolve(w_flip, h)
    
    cumsum = np.concatenate([[0.0], np.cumsum(w_arr)])
    lo = np.arange(n)
    hi = np.minimum(np.arange(n) + n - 1, 2*n-2)
    window_sum = cumsum[hi+1] - cumsum[lo]
    
    g = window_sum - conv1[n-1:2*n-1] - conv2[n-1:2*n-1]
    
    return sm_val, g

def adam_optimize_fft(h0, n, target_sum, alpha_schedule, n_iters, lr_init, rng):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for it in range(n_iters):
        alpha = alpha_schedule[min(it, len(alpha_schedule)-1)]
        t += 1
        _, g = smooth_max_grad_fft(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        lr *= 0.9995
    
    return best_h, best_c5

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)
