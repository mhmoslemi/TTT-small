import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, target_sum, step_size, n_active=1):
    """Take one subgradient step for the max problem"""
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Find lags where c(k) is close to max
    threshold = max_val - 0.01 * max_val
    active_lags = np.where(corr >= threshold)[0]
    
    # Average gradient over active lags
    n = len(h)
    g = np.zeros(n)
    for lag_idx in active_lags[:n_active]:  # limit number of active lags
        lag = lag_idx - (n - 1)
        if lag == 0:
            g[:] += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    
    g /= len(active_lags[:n_active])
    
    # Remove mean to satisfy sum constraint
    g -= np.mean(g)
    
    # Step
    h_new = h - step_size * g
    h_new = project_feasible(h_new, target_sum)
    return h_new

def subgradient_descent(h, target_sum, n_iters=5000, initial_step=0.05, decay=0.999):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = initial_step

    best_c5 = c5

            best_h = h_new.copy()
            best_c5 = c5_new
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            step *= 0.9
        step *= decay
        if step < 1e-8:

    return best_h, best_c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad


def penalized_obj_grad(x, alpha, target_sum, mu):
    obj, grad = smooth_max_and_grad(x, alpha)
    s = np.sum(x) - target_sum
    penalty = mu * s * s
    pen_grad = 2.0 * mu * s * np.ones_like(x)
    return obj + penalty, grad + pen_grad


def active_subgradient(h, target_sum, n_iters, base_step, n_active, decay):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step = base_step
    for it in range(n_iters):
        corr = compute_corr(h)
        max_val = float(corr[int(np.argmax(corr))])
        threshold = max_val - 0.015 * max_val
        active = np.where(corr >= threshold)[0]
        active = active[:n_active]
        g = np.zeros(n)
        for lag_idx in active:
            lag = int(lag_idx) - (n - 1)
            g += lag_gradient(h, lag)
        g /= len(active)
        g -= np.mean(g)
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            step = min(step * 1.04, 0.06)
        else:
            step *= 0.82
        if step < 1e-12:
            break
    return best_h, best_c5


def coord_desc(h, target_sum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = lag_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-11:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)


def lbfgs_opt(h0, target_sum, alpha, maxiter, mu=100.0):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        def fpen(x):
            return penalized_obj_grad(x, alpha, target_sum, mu)[0]
        def gpen(x):
            return penalized_obj_grad(x, alpha, target_sum, mu)[1]
        res = minimize(fpen, h0.copy(), method='L-BFGS-B', jac=gpen,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)


def slsqp_opt(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def cons_fun(x):
        return np.sum(x) - target_sum
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        def fobj(x):
            return smooth_max_and_grad(x, alpha)[0]
        def gobj(x):
            return smooth_max_and_grad(x, alpha)[1]
        res = minimize(fobj, h0.copy(), method='SLSQP', jac=gobj,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)


def two_step_optimize(h, target_sum, remaining):
    n = len(h)
    c5_cur = compute_c5(h)
    if remaining() > 30:
        h_sg, c5_sg = active_subgradient(h, target_sum, n_iters=2000, base_step=0.025, n_active=5, decay=0.9996)
        c5_cur = min(c5_cur, c5_sg)
        h = h_sg if c5_sg < c5_cur else h
    if remaining() > 30:
        h_lb = lbfgs_opt(h, target_sum, 300, maxiter=800, mu=50.0)
        c5_lb = compute_c5(h_lb)
        if c5_lb < c5_cur:
            h = h_lb
            c5_cur = c5_lb
    if remaining() > 25:
        h_cd, c5_cd = coord_desc(h, target_sum, max_iters=4000, step_frac=0.1)
        if c5_cd < c5_cur:
            h = h_cd
            c5_cur = c5_cd
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    remaining = lambda: budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts_cur = n_cur / 2.0
    h_cur = best_h.copy()

    if remaining() > 80:
        h_sg, c5_sg = active_subgradient(h_cur, ts_cur, n_iters=5000, base_step=0.035, n_active=8, decay=0.9997)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h_cur = h_sg

    for alpha in [40, 150, 600, 2500, 10000]:
        if remaining() < 35:
            break
        h_lb = lbfgs_opt(h_cur, ts_cur, alpha, maxiter=1200, mu=80.0)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()
            h_cur = h_lb

    if remaining() > 40:
        h_cd, c5_cd = coord_desc(h_cur, ts_cur, max_iters=10000, step_frac=0.12)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_cur = h_cd

    resolutions_to_try = []
    for nr in [180, 216, 240, 270, 300, 324, 360, 400, 450, 500, 540, 600, 720]:
        if nr % 2 == 0 and nr != n_cur:
            resolutions_to_try.append(nr)
    resolutions_to_try.sort(key=lambda x: abs(x - n_cur))

    for n_tr in resolutions_to_try[:5]:
        if remaining() < 90:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_feasible(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        h_work = h_interp.copy()
        c5_work = c5_interp
        if remaining() > 50:
            h_work, c5_work = two_step_optimize(h_work, ts_tr, remaining)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_tr
            h_cur = h_work.copy()
            n_cur = n_tr
            ts_cur = ts_tr

    n_cur = best_n
    ts_cur = n_cur / 2.0
    h_cur = best_h.copy()

    for alpha in [300, 1500, 7000, 30000, 100000]:
        if remaining() < 25:
            break
        h_s = slsqp_opt(h_cur, ts_cur, alpha, maxiter=800)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            h_cur = h_s

    n_restarts = 10
    for trial in range(n_restarts):
        if remaining() < 55:
            break
        noise = 0.04 + 0.06 * (1.0 - trial / float(n_restarts))
        h_shake = h_cur + rng.uniform(-noise, noise, n_cur)
        h_shake = project_feasible(h_shake, ts_cur)
        h_r, c5_r = active_subgradient(h_shake, ts_cur, n_iters=1200, base_step=0.02, n_active=4, decay=0.9995)
        if remaining() > 30:
            h_r2 = lbfgs_opt(h_r, ts_cur, 500, maxiter=600, mu=60.0)
            c5_r2 = compute_c5(h_r2)
            if c5_r2 < c5_r:
                h_r = h_r2
                c5_r = c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            h_cur = h_r
            n_cur = len(h_cur)
            ts_cur = n_cur / 2.0

    if remaining() > 30:
        h_cd2, c5_cd2 = coord_desc(h_cur, ts_cur, max_iters=12000, step_frac=0.07)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h_cur = h_cd2

    for alpha in [50000, 200000]:
        if remaining() < 20:
            break
        h_pol = slsqp_opt(h_cur, ts_cur, alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_cur = h_pol

    if remaining() > 20:
        h_cd3, c5_cd3 = coord_desc(h_cur, ts_cur, max_iters=20000, step_frac=0.04)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
