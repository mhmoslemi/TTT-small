    for it in range(maxiter):
        momentum = (t - 1.0) / (t + 1.0)
        y = x + momentum * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g, corr = topk_gradient(y, topk=3)
        lr = init_lr / (1.0 + 0.005 * it)
        x_new = y - lr * g
        x_new = np.clip(x_new, 0.0, 1.0)
        x_new = project_to_sum(x_new, target_sum)
        val = compute_c5(x_new)
        if val < best_val:
            best_val = val
            best_x = x_new.copy()
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_prev = x
        x = x_new
    return best_x, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    def check_time():
        return time.time() - t_start

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: SLSQP with multiple alphas
    for alpha in [50, 300, 1500, 5000]:
        if time_left() < 30:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Nesterov accelerated subgradient
    if time_left() > 100:
        h_nes, c5_nes = nesterov_subgrad(best_h, ts_parent, maxiter=5000, init_lr=0.08)
        if c5_nes < best_c5:
            best_c5 = c5_nes
            best_h = h_nes.copy()

    # Phase 3: Improved coordinate descent with line search
    step_fracs = [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8]
    if time_left() > 60:
        h_cd, c5_cd = coord_descent_improved(best_h, ts_parent, max_iters=8000, topk=3, step_fracs=step_fracs)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Swap-based local search
    if time_left() > 60:
        h_sw, c5_sw = swap_local_search(best_h, ts_parent, max_iters=5000, rng=rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 5: Block shift search
    if time_left() > 40:
        h_bl, c5_bl = block_shift_search(best_h, ts_parent, max_iters=3000, rng=rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
        # Follow up with coord descent
        if time_left() > 60:
            h_cd2, c5_cd2 = coord_descent_improved(best_h, ts_parent, max_iters=5000, topk=3, step_fracs=step_fracs)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 6: Multi-resolution
    resolutions = []
    for nr in [n_parent + 60, n_parent + 120, n_parent + 240, n_parent + 360]:
        if nr % 2 == 0:
            resolutions.append(nr)

    for n_tr in resolutions:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
        for alpha in [100, 500, 2000]:
            if time_left() < 40:
                break
            h_opt = slsqp_local(h_interp, n_tr, ts_tr, alpha, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt
        if time_left() > 50:
            h_cd3, c5_cd3 = coord_descent_improved(h_interp, ts_tr, max_iters=4000, topk=3, step_fracs=step_fracs)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n_tr

    # Go back to n_parent if it was better
    if best_n != n_parent and time_left() > 80:
        h_back = interp_to_n(best_h, n_parent)
        h_back = project_to_sum(h_back, ts_parent)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5:
            best_c5 = c5_back
            best_h = h_back.copy()
            best_n = n_parent

    # Phase 7: Structured restarts
    n_best = best_n
    ts_best = n_best / 2.0

    n_restarts = 10
    for trial in range(n_restarts):
        if time_left() < 50:
            break
        h_pert = structured_perturb(best_h, rng)
        c5_pert = compute_c5(h_pert)
        h_cd, c5_cd = coord_descent_improved(h_pert, ts_best, max_iters=4000, topk=3, step_fracs=step_fracs)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() < 40:
            break
        h_sw, c5_sw = swap_local_search(best_h, ts_best, max_iters=2000, rng=rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 8: Random restarts with Gaussian noise
    for trial in range(8):
        if time_left() < 40:
            break
        noise_level = 0.02 + 0.05 * (1.0 - trial / float(8))
        h_shake = best_h + rng.uniform(-noise_level, noise_level, n_best)
        h_shake = project_to_sum(h_shake, ts_best)
        h_cd, c5_cd = coord_descent_improved(h_shake, ts_best, max_iters=3000, topk=3, step_fracs=step_fracs)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() < 30:
            break
        h_opt = slsqp_local(h_cd, n_best, ts_best, 1000, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final SLSQP polish with high alpha
    for alpha in [3000, 10000, 30000]:
        if time_left() < 25:
            break
        h_pol = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 10: Very fine coordinate descent
    if time_left() > 30:
        fine_fracs = [0.01, 0.02, 0.04, 0.08, 0.12]
        h_final, c5_final = coord_descent_improved(best_h, ts_best, max_iters=12000, topk=5, step_fracs=fine_fracs)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 11: One more swap pass
    if time_left() > 20:
        h_sw2, c5_sw2 = swap_local_search(best_h, ts_best, max_iters=3000, rng=rng)
        if c5_sw2 < best_c5:
            best_c5 = c5_sw2
            best_h = h_sw2.copy()

    # Final: ensure feasibility
    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
