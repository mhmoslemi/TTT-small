def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-10:

        else:

def adam_smooth_minimax(h0, n, target_sum, alpha, n_iters, lr, t_start, budget_s, time_limit):
    """Adam optimizer for smooth minimax with FFT acceleration."""
    h = h0.copy().astype(float)
    N = 1 << (2 * n - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        t += 1
        
        corr, _ = get_corr_fft(h, N)
        shifted = alpha * corr * (2.0/n)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted - np.max(shifted))
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient
        H = np.fft.rfft(h, N)
        G = np.fft.rfft(1.0 - h, N)
        W = np.fft.rfft(w, N)
        grad = np.fft.irfft(W * np.conj(G), N)[:n]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def coord_descent_fast(h0, n, target_sum, n_rounds, rng, eval_time_limit):
    """Coordinate descent with FFT evaluation."""
    h = h0.copy().astype(float)
    N = 1 << (2 * n - 1).bit_length()

    for rnd in range(n_rounds):

            break

        for i in idx_order:
            # Binary search for best value at coordinate i
            best_c5 = c5_cur
            best_val = h[i]
            # Try a few values around current
            for delta in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:

                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

            i_start_g = k  # i-k >= 0
            i_end_g = n  # i < n
            i_start_h = 0  # i+k >= 0 always for k>=0, i>=0
            i_end_h = n - k  # i+k < n
        else:
            i_start_g = 0  # i-k >= 0, with k<0: i >= -k > 0... wait
            # k < 0: i - k = i + |k| >= 0 always. i + |k| < n → i < n - |k|
            i_start_g = 0
            i_end_g = n + k  # i < n+k for i-k < n (since k<0, n+k < n)
            # Wait: i - k < n → i < n + k. Since k < 0, n + k < n. So i in [0, n+k-1]
            i_start_h = -k  # i+k >= 0 → i >= -k
            i_end_h = n  # i+k < n always since i < n and k < 0
        
        # g[i-k] part: i in [i_start_g, i_end_g)
        if i_end_g > i_start_g:
            idx_g = np.arange(i_start_g, i_end_g)
            grad[i_start_g:i_end_g] += w_k * g[idx_g - k]
