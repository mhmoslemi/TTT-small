    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def pgd(h0, alpha, lr, iters, tf):
    n = len(h0)
    h = h0.copy()
    bo = lse_obj_grad(h, alpha)[0]
    bh = h.copy()
    for _ in range(iters):
        if tf() < 3:
            break
        o, g = lse_obj_grad(h, alpha)
        h2 = project_box_sum(h - lr * g, n / 2.0)
        o2, _ = lse_obj_grad(h2, alpha)
        if o2 < bo:
            bo = o2
            bh = h2.copy()
        if o2 > o:
            lr *= 0.5
            h2 = project_box_sum(h - lr * g, n / 2.0)
        else:
            lr = min(lr * 1.08, 0.5)
        h = h2
    return bh

def swap_climb(h0, n, tf, rng, iters, max_eps, accept_uphill=False, T_init=0.002):
    best = h0.copy()
    bc = compute_c5(h0)
    h = h0.copy()
    c = bc
    for it in range(iters):
        if tf() < 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = rng.uniform(0.003, max_eps)
        if h[i] < eps or h[j] > 1.0 - eps:
            continue
        h[i] -= eps
        h[j] += eps
        c2 = compute_c5(h)
        delta = c2 - c
        accept = False
        if delta < -1e-15:
            accept = True
        elif accept_uphill:
            T = T_init * (1.0 - it / max(1, iters)) + 0.0001
            if delta < 0.01 and rng.rand() < np.exp(-delta / T):
                accept = True
        if accept:
            c = c2
            if c2 < bc:
                bc = c2
                best = h.copy()
        else:
            h[i] += eps
            h[j] -= eps
    return best, bc

def multi_swap_climb(h0, n, tf, rng, iters, max_eps):
    best = h0.copy()
    bc = compute_c5(h0)
    h = h0.copy()
    c = bc
    for it in range(iters):
        if tf() < 3:
            break
        k = rng.randint(2, 5)
        idx = rng.choice(n, k, replace=False)
        deltas = rng.uniform(0.005, max_eps, k)
        sgn = rng.choice([1, -1], k)
        for m in range(k):
            ii = idx[m]
            d = sgn[m] * deltas[m]
            if d > 0 and h[ii] > 1.0 - d:
                d = 0.0
            elif d < 0 and h[ii] < -d:
                d = 0.0
            h[ii] += d
        c2 = compute_c5(h)
        if c2 < c - 1e-15:
            c = c2
            if c2 < bc:
                bc = c2
                best = h.copy()
        else:
            for m in range(k):
                ii = idx[m]
                d = sgn[m] * deltas[m]
                if d > 0 and h[ii] > 1.0 - d:
                    d = 0.0
                elif d < 0 and h[ii] < -d:
                    d = 0.0
                h[ii] -= d
    return best, bc

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def tf():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1e9
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: LSE gradient descent
    for alpha in [10, 30, 80, 200, 500, 1500, 5000, 15000]:
        if tf() < 12:
            break
        lr = min(0.06, 3.0 / max(alpha, 30))
        h_new = pgd(best_h, alpha, lr, max(300, int(3000 / max(alpha / 10, 1))), tf)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Extensive swap climbing (direct C5)
    if tf() > 40:
        h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 100000, 0.25)
        if c5_sc < best_c5:
            best_c5 = c5_sc
            best_h = h_sc.copy()

    # Phase 3: LSE fine polish
    if tf() > 30:
        for alpha in [5000, 20000, 80000]:
            if tf() < 10:
                break
            lr = min(0.008, 2.0 / max(alpha, 1000))
            h_new = pgd(best_h, alpha, lr, 3000, tf)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Fine swap climbing
    if tf() > 40:
        h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 150000, 0.08)
        if c5_sc < best_c5:
            best_c5 = c5_sc
            best_h = h_sc.copy()

    # Phase 5: Simulated annealing (accept uphill moves)
    if tf() > 60:
        h_sa, c5_sa = swap_climb(best_h, n, tf, rng, 80000, 0.3, accept_uphill=True, T_init=0.003)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish after SA
        if tf() > 20:
            for alpha in [5000, 30000]:
                if tf() < 8:
                    break
                h_new = pgd(best_h, alpha, 0.005, 2000, tf)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            if tf() > 15:
                h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 80000, 0.06)
                if c5_sc < best_c5:
                    best_c5 = c5_sc
                    best_h = h_sc.copy()

    # Phase 6: Multi-coordinate swap
    if tf() > 50:
        h_mc, c5_mc = multi_swap_climb(best_h, n, tf, rng, 50000, 0.15)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()
        if tf() > 20:
            h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 60000, 0.05)
            if c5_sc < best_c5:
                best_c5 = c5_sc
                best_h = h_sc.copy()

    # Phase 7: Upsample to higher resolution
    if tf() > 100:
        n_up = min(480, n * 2)
        if n_up % 2 != 0:
            n_up += 1
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_box_sum(h_up, n_up / 2.0)
        for alpha in [50, 200, 1000, 5000, 20000]:
            if tf() < 12:
                break
            lr = min(0.03, 2.0 / max(alpha, 30))
            h_opt = pgd(h_up, alpha, lr, max(200, int(2000 / max(alpha / 50, 1))), tf)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n_up
                n = n_up
            h_up = h_opt
        if tf() > 40:
            h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 80000, 0.15)
            if c5_sc < best_c5:
                best_c5 = c5_sc
                best_h = h_sc.copy()

    # Phase 8: Multi-start
    if tf() > 120:
        n_ms = n
        for trial in range(12):
            if tf() < 30:
                break
            sc = 0.05 * (1.0 - trial / 12.0) + 0.008
            h_sh = best_h + rng.uniform(-sc, sc, n_ms)
            h_sh = project_box_sum(h_sh, n_ms / 2.0)
            for alpha in [200, 2000, 10000]:
                if tf() < 10:
                    break
                lr = min(0.02, 2.0 / max(alpha, 50))
                h_opt = pgd(h_sh, alpha, lr, 1500, tf)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                h_sh = h_opt
            if tf() > 15:
                h_sc, c5_sc = swap_climb(h_sh, n_ms, tf, rng, 60000, 0.1)
                if c5_sc < best_c5:
                    best_c5 = c5_sc
                    best_h = h_sc.copy()

    # Phase 9: Final very high alpha
    if tf() > 20:
        for alpha in [50000, 200000, 500000]:
            if tf() < 8:
                break
            lr = 2.0 / alpha
            h_new = pgd(best_h, alpha, lr, 3000, tf)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 10: Ultimate fine swap
    if tf() > 20:
        h_sc, c5_sc = swap_climb(best_h, n, tf, rng, 100000, 0.03)
        if c5_sc < best_c5:
            best_c5 = c5_sc
            best_h = h_sc.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
