    momentum = 0.9

    for it in range(max_iter):
        if time_fn() < 3:
            break
        y = h + momentum * v_mom
        obj_y, grad_y = smooth_max_obj_and_grad(y, alpha)
        v_mom = momentum * v_mom - lr * grad_y
        h_new = h + v_mom
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        obj_exact = smooth_max_obj_and_grad(h, alpha)[0]
        if obj_new > obj_exact + 1e-8:
            lr *= 0.6
            v_mom *= 0.5
        else:
            lr = min(lr * 1.02, 0.2)
        h = h_new
        if stale > 300:
            break
    return best_h, best_obj


def sa_polish(h0, rng, time_fn, t_init, t_final, n_iter):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        if time_fn() < 3:
            break
        T = t_init * (t_final / t_init) ** (it / max(1, n_iter))
        r = rng.rand()
        if r < 0.3:
            i = rng.randint(n)
            delta = rng.uniform(-0.5, 0.5)
            h_new = h.copy()
            h_new[i] += delta
            h_new = project_box_sum(h_new, n / 2.0)
        elif r < 0.6:
            blk = rng.randint(2, max(3, n // 15))
            start = rng.randint(0, n - blk)
            delta = rng.uniform(-0.3, 0.3)
            h_new = h.copy()
            h_new[start:start + blk] += delta
            h_new = project_box_sum(h_new, n / 2.0)
        elif r < 0.8:
            i, j = rng.choice(n, size=2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            h_new = project_box_sum(h_new, n / 2.0)
        else:
            h_new = h + rng.normal(0, 0.1, n)
            h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        d = c5_new - c5_cur
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-12)):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5


def critical_lag_moves(h, rng, time_fn, n_rounds=500):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        if time_fn() < 4:
            break
        c5_cur, amax_idx, corr_arr = compute_c5_argmax(h)
        lag = amax_idx - (n - 1)
        improved = False
        n_trials = 80
        for _ in range(n_trials):
            i = rng.randint(n)
            j1 = (i + lag) % n
            j2 = (i - lag) % n
            h_new = h.copy()
            r = rng.rand()
            if r < 0.5:
                h_new[i] = h[j1]
            else:
                h_new[i] = h[j2]
            h_new = project_box_sum(h_new, n / 2.0)
            c5_t = compute_c5(h_new)
            if c5_t < c5_cur - 1e-12:
                h = h_new
                c5_cur = c5_t
                improved = True
                break
        if not improved:
            for _ in range(50):
                i = rng.randint(n)
                h_new = h.copy()
                h_new[i] = 1.0 - h[i]
                h_new = project_box_sum(h_new, n / 2.0)
                c5_t = compute_c5(h_new)
                if c5_t < c5_cur - 1e-12:
                    h = h_new
                    c5_cur = c5_t
                    improved = True
                    break
            if not improved:
                break
    return h, c5_cur


def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_box_sum(h_new, n_new / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at initial resolution with Nesterov
    cur_n = best_n
    cur_h = best_h.copy()
    for alpha in [5, 20, 80, 300, 1000, 4000]:
        if time_left() < 25:
            break
        h_new, _ = nesterov_pg(cur_h, alpha, 4000, min(0.08, 1.0 / (alpha + 10)), time_left)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            cur_h = h_new.copy()

    # Phase 2: Multi-resolution upsampling
    for n_target in [360, 480, 600, 720]:
        if n_target <= cur_n:
            continue
        if time_left() < 80:
            break
        if n_target % 2 != 0:
            n_target += 1
        cur_h = upsample_h(cur_h, cur_n, n_target)
        cur_n = n_target
        for alpha in [10, 50, 200, 800, 3000]:
            if time_left() < 20:
                break
            h_new, _ = nesterov_pg(cur_h, alpha, 3000, min(0.05, 0.8 / (alpha + 10)), time_left)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                cur_h = h_new.copy()

    # Phase 3: Simulated annealing
    if time_left() > 150:
        for trial in range(4):
            if time_left() < 40:
                break
            t_init = 0.4 / (trial + 1)
            h_sa, c5_sa = sa_polish(cur_h, rng, time_left, t_init, 0.0003, 4000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                cur_h = h_sa.copy()

    # Phase 4: Critical lag targeting
    if time_left() > 80:
        h_cl, c5_cl = critical_lag_moves(cur_h, rng, time_left, n_rounds=400)
        if c5_cl < best_c5:
            best_c5 = c5_cl
            best_h = h_cl.copy()
            cur_h = h_cl.copy()

    # Phase 5: Hard max subgradient polish
    if time_left() > 60:
        n = len(cur_h)
        h = cur_h.copy()
        c5_cur = compute_c5(h)
        lr = 0.02
        for it in range(10000):
            if time_left() < 10:
                break
            obj, grad = hard_max_subgrad(h)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            lr_eff = lr / max(gnorm, 0.01)
            h_new = h - lr_eff * grad
            h_new = project_box_sum(h_new, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                cur_h = h_new.copy()
                c5_cur = c5_new
                lr = min(lr * 1.05, 0.1)
            else:
                lr *= 0.7
        if best_c5 < c5_cur:
            cur_h = best_h.copy()

    # Phase 6: Shake and re-optimize
    if time_left() > 100:
        for trial in range(20):
            if time_left() < 30:
                break
            noise_scale = 0.12 * (1.0 - trial / 20.0) + 0.005
            h_shake = cur_h + rng.uniform(-noise_scale, noise_scale, len(cur_h))
            h_shake = project_box_sum(h_shake, len(cur_h) / 2.0)
            for alpha in [50, 300, 2000]:
                if time_left() < 15:
                    break
                h_opt, _ = nesterov_pg(h_shake, alpha, 2000, 0.03, time_left)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    cur_h = h_opt.copy()
                h_shake = h_opt

    # Phase 7: SA on best
    if time_left() > 40:
        h_sa2, c5_sa2 = sa_polish(best_h, rng, time_left, 0.08, 0.0001, 3000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 8: Final Nesterov with very high alpha
    if time_left() > 20:
        for alpha in [5000, 20000]:
            if time_left() < 10:
                break
            h_fin, _ = nesterov_pg(best_h, alpha, 3000, 0.005, time_left)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
