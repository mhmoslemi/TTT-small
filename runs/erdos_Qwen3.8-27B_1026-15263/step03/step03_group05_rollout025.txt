```python
def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    if alpha < 1e6:
        exp_vals = np.exp(alpha * (shifted - max_shift))
        w = exp_vals / np.sum(exp_vals)
    else:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n) if alpha < 1e6 else m_val * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def pgd_solve(h0, n, alpha, max_iter, lr, time_fn):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > compute_c5(h) + 1e-10:

def simulated_annealing(h0, n, rng, t_start, budget_s, n_iter=50000):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ts = n / 2.0
    T0 = 0.01
    Tend = 0.0001
    decay = (Tend / T0) ** (1.0 / max(1, n_iter))
    
    for it in range(n_iter):
        if budget_s - (time.time() - t_start) < 5:
            break
        T = T0 * (decay ** it)
        
        # Choose a random pair: one near 1, one near 0
        # To maintain sum constraint, swap h[i] and h[j]
        # Or: set h[i] = 1-h[i] and h[j] = 1-h[j] (if binary)
        # For continuous: h[i] += delta, h[j] -= delta
        
        # Method 1: Swap pair (maintains sum exactly)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_new = h.copy()
        h_new[i] = 1.0 - h[i]
        h_new[j] = 1.0 - h[j]
        # This maintains sum exactly!
        
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
```

```python
def compute_gradient(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            idx_above = i + (k - (n-1))  # for h[i+m] term
            idx_below = i - (k - (n-1))  # for h[i-m] term
            if 0 <= idx_below < n:
                s += w[k] * (1 - h[idx_below])
            if 0 <= idx_above < n:
                s -= w[k] * h[idx_above]
        grad[i] = s * dx
    return grad
```

```python
    idx2 = i_idx + m_vals  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.take(h, np.clip(idx2, 0, n-1))
    h2[~valid2] = 0.0
    term2 = (w_vals * h2).sum(axis=1)
    
    S_w = w.sum()
    grad = (S_w - term1 - term2) * dx
    return grad
```
