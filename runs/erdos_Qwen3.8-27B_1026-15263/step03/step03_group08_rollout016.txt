    c5_cur = compute_c5(h)
    steps_all = list(base_steps)
    for rnd in range(max_rounds):

def subgradient_step(h, target_sum, lr):
    """Take a subgradient step in the direction that reduces the max."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr > max_val - 0.01 * max_val)[0]  # near-active lags
    
    # Subgradient: average gradient over active lags
    g = np.zeros(n)
    for idx in active:
        l = idx - (n - 1)
        if l >= 0:
            if l < n:
                g[l:] += (1.0 - h[:n-l])
            if n - 1 - l >= 0:
                g[:n-l] -= h[l:]
        else:
            q = -l
            if n - 1 - q >= 0:
                g[:n-q] += (1.0 - h[q:])
            if q < n:
                g[q:] -= h[:n-q]
    g /= len(active)
    
    h_new = np.clip(h - lr * g, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, l):
    """Compute gradient of correlation at lag l."""
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            g[l:] += (1.0 - h[:n-l])
        if n - 1 - l >= 0:
            g[:n-l] -= h[l:]
    else:
        q = -l
        if n - 1 - q >= 0:
            g[:n-q] += (1.0 - h[q:])
        if q < n:
            g[q:] -= h[:n-q]
    return g

def active_set_grad(h, k_lags, alpha):
    """Compute smooth-max gradient over top-k lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_lags:]
    m = np.max(corr[top_idx])
    shifted = alpha * (corr[top_idx] - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    
    g = np.zeros(n)
    for j, idx in enumerate(top_idx):
        wl = w[j]
        if wl < 1e-14:
            continue
        l = idx - (n - 1)
        gk = compute_grad_lag(h, l)
        g += wl * gk
    return g, corr

def slp_step(h, target_sum, top_k):
    """One SLP iteration: linearize top-k lags and solve LP."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    
    for j, k in enumerate(top_idx):
        l = k - (n - 1)
        gk = compute_grad_lag(h, l)
        A_ub[j, :n] = gk
        A_ub[j, n] = -1.0
        b_ub[j] = np.dot(gk, h) - corr[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds_lprog = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lprog, method='highs')
    
    if res.success:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        return h_new
    return h.copy()

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, l):
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            g[l:] += (1.0 - h[:n-l])
        if n - 1 - l >= 0:
            g[:n-l] -= h[l:]
    else:
        q = -l
        if n - 1 - q >= 0:
            g[:n-q] += (1.0 - h[q:])
        if q < n:
            g[q:] -= h[:n-q]
    return g

        l = idx - (n - 1)
        gk = compute_grad_lag(h, l)
        A_ub[j, :n] = gk
        A_ub[j, n] = -1.0
        b_ub[j] = np.dot(gk, h) - corr[idx]

    return h.copy()

        wl = w[j]
        if wl < 1e-14:
            continue
        l = idx - (n - 1)
        g += wl * compute_grad_lag(h, l)

        except Exception:
            pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    stall_count = 0
    stall_threshold = 50

    # Phase 1: SLP iterations (top-k=50 first, then all lags)
    phase1_end = t_start + budget_s * 0.35
    use_all = False
    while time.time() < phase1_end:
        if use_all:
            h_slp = slp_step_all(h_cur, ts)
        else:
            h_slp = slp_step(h_cur, ts, top_k=50)

        c5_slp = compute_c5(h_slp)
        c5_cur = compute_c5(h_cur)

        if c5_slp < c5_cur - 1e-15:
            h_cur = h_slp
            stall_count = 0
        else:
            damp = 0.5
            h_half = project_to_sum(damp * h_cur + (1 - damp) * h_slp, ts)
            c5_half = compute_c5(h_half)
            if c5_half < c5_cur - 1e-15:
                h_cur = h_half
                stall_count = 0
            else:
                h_q = project_to_sum(0.75 * h_cur + 0.25 * h_slp, ts)
                c5_q = compute_c5(h_q)
                if c5_q < c5_cur - 1e-15:
                    h_cur = h_q
                    stall_count = 0
                else:
                    stall_count += 1

        c5_check = compute_c5(h_cur)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_cur.copy()
            best_n = n

        if stall_count > stall_threshold:
            if not use_all:
                use_all = True
                stall_count = 0
            else:
                break

    # Phase 2: Active-set gradient descent
    phase2_end = t_start + budget_s * 0.60
    lr = 0.02
    while time.time() < phase2_end:
        g, corr = active_set_gradient(h_cur, 8, 3000.0)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        g_normed = g / norm_g

        accepted = False
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            h_new = project_to_sum(h_cur - scale * g_normed, ts)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_cur) - 1e-16:
                h_cur = h_new
                lr = min(lr * 1.1, 0.1)
                accepted = True
                c5_check = compute_c5(h_cur)
                if c5_check < best_c5:
                    best_c5 = c5_check
                    best_h = h_cur.copy()
                    best_n = n
                break
        if not accepted:
            lr *= 0.7
            if lr < 1e-6:
                break

    # Phase 3: SLP with all lags again
    phase3_end = t_start + budget_s * 0.75
    h_slp = h_cur.copy()
    while time.time() < phase3_end:
        h_try = slp_step_all(h_slp, ts)
        c5_try = compute_c5(h_try)
        c5_base = compute_c5(h_slp)
        if c5_try < c5_base - 1e-15:
            h_slp = h_try
            c5_check = compute_c5(h_slp)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_slp.copy()
                best_n = n
        else:
            h_d = project_to_sum(0.7 * h_slp + 0.3 * h_try, ts)
            c5_d = compute_c5(h_d)
            if c5_d < c5_base - 1e-15:
                h_slp = h_d
                c5_check = compute_c5(h_slp)
                if c5_check < best_c5:
                    best_c5 = c5_check
                    best_h = h_slp.copy()
                    best_n = n
            else:
                break

    # Phase 4: Coordinate descent polish
    phase4_end = t_start + budget_s * 0.92
    steps_med = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
    h_cd = best_h.copy()
    c5_cd = compute_c5(h_cd)
    max_rnd = max(20, int((budget_s * 0.15) / (n * 8 * 2e-6)))
    for rnd in range(max_rnd):
        if time.time() > phase4_end:
            break
        improved = False
        for i in range(n):
            best_local = c5_cd
            best_local_h = h_cd
            for d in steps_med:
                h_new = h_cd.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_local - 1e-15:
                    best_local = c5_new
                    best_local_h = h_new
            if best_local < c5_cd - 1e-15:
                h_cd = best_local_h
                c5_cd = best_local
                improved = True
        if not improved:
            break
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        best_n = n

    # Phase 5: Final fine coordinate descent + SLP
    phase5_end = t_start + budget_s * 0.97
    steps_fine = [0.008, 0.004, 0.002, 0.001, -0.008, -0.004, -0.002, -0.001]
    h_f = best_h.copy()
    c5_f = compute_c5(h_f)
    for rnd in range(80):
        if time.time() > phase5_end:
            break
        improved = False
        for i in range(n):
            best_local = c5_f
            best_local_h = h_f
            for d in steps_fine:
                h_new = h_f.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_local - 1e-15:
                    best_local = c5_new
                    best_local_h = h_new
            if best_local < c5_f - 1e-15:
                h_f = best_local_h
                c5_f = best_local
                improved = True
        if not improved:
            break
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = n

    # Final SLP squeeze
    if time.time() < t_start + budget_s * 0.99:
        for _ in range(10):
            if time.time() > t_start + budget_s * 0.99:
                break
            h_try = slp_step_all(best_h, ts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n

    # Finalize
    best_h = project_to_sum(best_h, ts)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
