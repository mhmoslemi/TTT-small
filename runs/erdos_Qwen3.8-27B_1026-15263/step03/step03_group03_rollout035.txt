This matches! So grad = np.correlate(w, 1-2*h, mode="full")[:n].

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    m = np.max(a)
    return (m + np.log(np.sum(np.exp(a - m)))) / alpha

def smooth_max_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    m = np.max(a)
    exp_a = np.exp(a - m)
    w = exp_a / np.sum(exp_a)
    v = 1.0 - 2.0 * h

    return grad

    def obj(h):
        return smooth_max_val(h, alpha)

        return smooth_max_grad(h, alpha)

        return np.sum(h) - target_sum
    
    def cons_jac(h):
        return np.ones(n)

                    best_delta = delta_opt
                    best_B = D
                    best_k = k

                h[j] += best_delta
                h[best_k] -= best_delta
                h[j] = np.clip(h[j], 0.0, 1.0)
                h[best_k] = np.clip(h[best_k], 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    m = np.max(a)
    return (m + np.log(np.sum(np.exp(a - m)))) / alpha

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    m = np.max(a)
    exp_a = np.exp(a - m)
    w = exp_a / np.sum(exp_a)
    v = 1.0 - 2.0 * h
    return np.correlate(w, v, mode="full")[:len(h)]

def slsqp_step(h0, alpha, maxiter):
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        return smooth_max_val(h, alpha)
    
    def grad(h):
        return smooth_max_grad(h, alpha)
    
    def cfun(h):
        return np.sum(h) - target
    
    def cjac(h):
        return np.ones(n)
    
    res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_to_sum(res.x, target)

def lbfgsb_step(h0, alpha, maxiter):
    n = len(h0)
    target = n / 2.0
    lam = 50.0
    
    def obj(h):
        return smooth_max_val(h, alpha) + lam * (np.sum(h) - target) ** 2
    
    def grad(h):
        return smooth_max_grad(h, alpha) + 2.0 * lam * (np.sum(h) - target)
    
    res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_to_sum(res.x, target)

    rng.shuffle(order)
    v = 1.0 - 2.0 * h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * corr
    m = np.max(a)
    return (m + np.log(np.sum(np.exp(a - m)))) / alpha

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")

    m = np.max(a)
