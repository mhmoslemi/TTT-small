def active_set_descend(h0, n, target, n_iters, lr_init, top_k):
    """Gradient descent focusing on top K lags"""
    dx = 2.0 / n
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        
        g = np.zeros(n)
        for idx in top_idx:
            l = idx - (n - 1)
            if l >= 0:
                g[l:] += (1.0 - h[:n-l])
                g[:n-l] -= h[l:]
            else:
                m_ = -l
                g[:n-m_] += (1.0 - h[m_:])
                g[m_:] -= h[:n-m_]
        
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-15:
            break
        
        h_new = np.clip(h - lr * g / norm_g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 5.0)
        else:
            lr *= 0.9
        
        h = h_new
    
    return best_h, best_c5

def coordinate_descent(h0, n, target, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            for d in steps:
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                h_new = h.copy()
                h_new[i] = new_val
                h_new = project_to_sum(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_val = new_val
            if best_val != h[i]:
                h[i] = best_val
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    
    return h, c5_cur

def binarize_and_reoptimize(h0, n, target, threshold):
    """Push values toward 0 or 1, then re-optimize"""
    h = h0.copy()
    h[h < threshold] = 0.0
    h[h >= threshold] = 1.0
    h = project_to_sum(h, target)
    return h

def uupsample_piecewise(h, n_old, n_new):
    """Upsample using piecewise constant (nearest) interpolation"""
    ratio = n_new / n_old
    idx = np.floor(np.arange(n_new) / ratio).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return h[idx]

def lin_upsample(h, n_old, n_new):
    """Upsample using linear interpolation"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 2.0/n_new
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, n, target, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0, target)

            h_up = lin_upsample(best_h, best_n, n_new)
        h_up = project_to_sum(h_up, n_new / 2.0)
        candidates.append((h_up, n_new))
        if n_new == 2 * best_n:
            h_shift = np.empty(n_new)
            h_shift[0::2] = best_h
            h_shift[1::2] = np.roll(best_h, -1)
            h_shift = project_to_sum(h_shift, n_new / 2.0)
            candidates.append((h_shift, n_new))
    for delta in [0.02, 0.04, 0.08]:
        h_pert = best_h + rng.uniform(-delta, delta, best_n)
        h_pert = project_to_sum(h_pert, best_n / 2.0)
        candidates.append((h_pert, best_n))
    for n_s in [480, 720]:
        x = np.linspace(0, 2, n_s, endpoint=False)
        for freq in [2, 3, 4, 5, 7, 9]:
            for amp in [0.25, 0.4, 0.55]:
                for phase in [0.0, 0.5 * np.pi, np.pi]:
                    hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                    hs = project_to_sum(hs, n_s / 2.0)
                    c5_s = compute_c5(hs)
                    if c5_s < best_c5 + 0.03:
                        candidates.append((hs, n_s))
    scored = []
    for h_c, n_c in candidates:
        c5_c = compute_c5(h_c)
        scored.append((c5_c, h_c, n_c))
    scored.sort(key=lambda x: x[0])
    max_cands = 12
    if len(scored) > max_cands:
        seen_n = set()
        filtered = []
        for s in scored:
            if s[2] not in seen_n:
                filtered.append(s)
                seen_n.add(s[2])
            elif len(filtered) < max_cands:
                filtered.append(s)
        scored = filtered[:max_cands]
    time_per_cand = (budget_s * 0.6) / max(len(scored), 1)
    for c5_init, h_c, n_c in scored:
        if time_left() < 60:
            break
        target = n_c / 2.0
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_c.copy()
            best_n = n_c
        alloc = min(time_per_cand, time_left() - 50)
        if alloc < 20:
            break
        h_opt, c5_opt = optimize_candidate(h_c, n_c, target, alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_c
    if best_h is not None and time_left() > 30:
        n_f = best_n
        target_f = n_f / 2.0
        for alpha in [10000.0, 20000.0, 50000.0]:
            if time_left() < 25:
                break
            maxiter = max(100, min(600, int(time_left() * 0.5)))
            h_pol = slsqp_optimize(best_h, n_f, target_f, alpha, maxiter)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        if time_left() > 30:
            for top_k in [5, 10, 15]:
                if time_left() < 20:
                    break
                n_it = min(5000, int(time_left() * 2))
                h_as, c5_as = active_set_descend(best_h, n_f, target_f, n_it, 0.3, top_k)
                if c5_as < best_c5:
                    best_c5 = c5_as
                    best_h = h_as.copy()
        if time_left() > 25:
            for alpha in [200.0, 1000.0, 5000.0]:
                if time_left() < 18:
                    break
                n_it = min(2000, int(time_left() * 2))
                h_sm, c5_sm = smooth_max_descend(best_h, n_f, target_f, alpha, n_it, 0.05)
                if c5_sm < best_c5:
                    best_c5 = c5_sm
                    best_h = h_sm.copy()
        if time_left() > 20:
            steps_fine = [0.03, 0.015, 0.008, 0.004, 0.002, -0.03, -0.015, -0.008, -0.004, -0.002]
            max_rnd = max(10, min(30, int(time_left() // 2)))
            h_cd, c5_cd = coordinate_descent(best_h, n_f, target_f, max_rnd, steps_fine)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
        if time_left() > 15:
            for alpha in [30000.0, 80000.0]:
                if time_left() < 12:
                    break
                h_fin = slsqp_optimize(best_h, n_f, target_f, alpha, 400)
                c5_fin = compute_c5(h_fin)
                if c5_fin < best_c5:
                    best_c5 = c5_fin
                    best_h = h_fin.copy()
        if time_left() > 12:
            n_it_final = min(3000, int(time_left() * 2))
            h_af, c5_af = active_set_descend(best_h, n_f, target_f, n_it_final, 0.2, 8)
            if c5_af < best_c5:
                best_c5 = c5_af
                best_h = h_af.copy()
        if time_left() > 10:
            steps_very_fine = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]
            h_cv, c5_cv = coordinate_descent(best_h, n_f, target_f, 20, steps_very_fine)
            if c5_cv < best_c5:
                best_c5 = c5_cv
                best_h = h_cv.copy()
    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
