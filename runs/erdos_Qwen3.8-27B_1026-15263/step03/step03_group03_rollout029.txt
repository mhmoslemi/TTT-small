def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, alpha):
    """Compute the softmax-smoothed subgradient of C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    p = exps / np.sum(exps)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                g[:n - lag] += pk * (1.0 - h[lag:])
                g[lag:] -= pk * h[:n - lag]
        else:
            d = -lag
            if d < n:
                g[d:] += pk * (1.0 - h[:n - d])
                g[:n - d] -= pk * h[d:]
    return g, c5_from_corr(corr, n)

def c5_from_corr(corr, n):
    return float(np.max(corr) * 2.0 / n)

    t_start = time.time()
    h = h0.copy()
    v = np.zeros(n)

    momentum = 0.9
    alpha = 30.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0

        # Adaptive alpha (increase over time for sharper max)

def compute_B(h, j, n):
    """Compute the change in correlation when we increase h[j] by delta and decrease h[k] by delta."""
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    # Effect of increasing h[j] by 1 (and correspondingly the 1-h changes)
    # h[j] increases by delta:
    #   In corr: h contributes positively where it's the first argument
    #   1-h contributes negatively where 1-h is the second argument
    # 
    # corr[m] = sum_i h[i] * (1-h)[i + m - nm1]
    # If h[j] increases by delta:
    #   corr[m] changes by delta * (1-h)[j + m - nm1] for valid indices
    #   = delta * (1 - h[j + m - nm1])
    #   valid when 0 <= j + m - nm1 < n
    #   i.e., nm1 - j <= m <= nm1 - j + n - 1
    k_start = max(0, nm1 - j)
    k_end = min(2*nm1, nm1 - j + n - 1)
    if k_start <= k_end:
        idx = k_start - (nm1 - j)  # h index
        B[k_start:k_end+1] += (1.0 - h[idx:idx + (k_end - k_start + 1)])
    
    # If h[j] increases by delta, the (1-h) at position j decreases by delta:
    #   corr[m] changes by -delta * h[i] where i + m - nm1 = j
    #   i.e., i = j + nm1 - m
    #   valid when 0 <= j + nm1 - m < n
    #   i.e., j + nm1 - n + 1 <= m <= j + nm1
    k_start2 = max(0, j + nm1 - n + 1)
    k_end2 = min(2*nm1, j + nm1)
    if k_start2 <= k_end2:
        idx2 = j + nm1 - k_end2  # h index at k_end2
        # m goes from k_start2 to k_end2, i = j + nm1 - m goes from (j+nm1-k_start2) to (j+nm1-k_end2)
        m_range = np.arange(k_start2, k_end2 + 1)
        i_range = j + nm1 - m_range
        B[k_start2:k_end2+1] += -h[i_range]
    
    return B

def solve_1d(A, D, lo, hi):
    """Minimize max(A + D*t) over t in [lo, hi]."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Coarse grid
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def project_to_sum(h, target_sum):

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)

    k_end2 = min(2 * n - 2, j + nm1)

        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B
