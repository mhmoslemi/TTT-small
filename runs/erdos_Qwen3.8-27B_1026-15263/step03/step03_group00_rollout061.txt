def shift_h(h, shift_frac):
    """Cyclically shift h by shift_frac * (1/n) of the period"""
    n = len(h)
    shift = shift_frac / n
    x = np.arange(n) / n
    x_shifted = (x - shift) % 1.0
    # Interpolate
    idx = np.floor(x_shifted * n).astype(int)
    frac = x_shifted * n - idx
    h_shifted = (1 - frac) * h[idx % n] + frac * h[(idx + 1) % n]
    return h_shifted

        m = len(active_lags)

        # And we want to update h[frac_indices] so that this holds

        # The actual values of h are determined by the sum constraint and bounds.
        # At a KKT point, the values are what they are - we can't change them
        # without changing which entries are fractional.
        
        # So the KKT check is: is the current h consistent with KKT?
        # If not, we need to move in a direction that reduces the objective.

        # But at a KKT point, this should be zero.

        try:

            rhs = -A[:, m-1]

import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0, 1)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1 - 1e-10)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0, 1)
    return h

def corr_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        g[lag:] += 1.0 - h[:n-lag]
        g[:n-lag] -= h[lag:]
    else:
        k = -lag
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    return g

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    c = compute_corr(h)
    mv = c.max()
    s = alpha * (c - mv)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    obj = mv + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] > 1e-8:
            g += w[i] * corr_grad(h, i - (n-1))
    return obj, g

def slsqp(h0, n, ts, alpha, mx=800):
    def f(x): return smooth_max_obj_grad(x, alpha)[0]
    def j(x): return smooth_max_obj_grad(x, alpha)[1]
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=j,
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-ts,'jac':lambda x: np.ones(n)}],
                     options={'maxiter':mx,'ftol':1e-14})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0.copy(), ts)

def interp(h, nn):
    return np.interp(np.linspace(0,2,nn,endpoint=False),
                     np.linspace(0,2,len(h),endpoint=False), h, period=2.0)

    """Natural gradient (mirror descent) local search"""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    eta = eta0
    for it in range(maxit):
        corr = compute_corr(h)
        idx = corr.argmax()
        lag = idx - (n-1)
        g = corr_grad(h, lag)
        hs = np.clip(h, 1e-8, 1-1e-8)
        F = hs * (1 - hs)
        h_new = h - eta * F * g
        h_new = project_to_sum(h_new, ts)
        c5n = compute_c5(h_new)
        if c5n < c5 - 1e-15:
            h = h_new
            c5 = c5n
        else:
            eta *= 0.9
            if eta < 1e-8:
                break
            # Try smaller step
            h_try = h - eta * F * g
            h_try = project_to_sum(h_try, ts)
            if compute_c5(h_try) < c5:

def pair_cd(h, ts, maxit=20000, sf=0.15):

    n = len(h)
    h = h.copy()

        lag = corr.argmax() - (n-1)
        g = corr_grad(h, lag)
        ci = h < 1 - 1e-10
        cd = h > 1e-10
        if not ci.any() or not cd.any():
            break
        gi = np.where(ci, g, np.inf)
        gd = np.where(cd, g, -np.inf)
        ii = gi.argmin()
        idd = gd.argmax()
        if ii == idd or g[ii] >= g[idd] - 1e-12:
            break
        mv = min(1-h[ii], h[idd]) * sf
        if mv < 1e-10:
            break
        h[ii] += mv; h[idd] -= mv
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, ts)
        c5n = compute_c5(h)
        if c5n >= c5 - 1e-13:
            h[ii] -= mv; h[idd] += mv
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, ts)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5n
            stale = 0
    return h, c5

def shift_h(h, frac):

    idx = (xs * n).astype(int)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    return g

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    c = compute_corr(h)
    mv = c.max()
    s = alpha * (c - mv)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    obj = mv + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] > 1e-8:
            g += w[i] * corr_grad(h, i - (n - 1))
    return obj, g
