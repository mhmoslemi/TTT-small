import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(mask)
        if n_free > 0:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, top_k=None):
    n = len(h)
    corr = get_corr(h)
    if top_k is not None and top_k < len(corr):
        # Focus on top_k correlations
        top_idx = np.argsort(corr)[-top_k:]
        c_sub = corr[top_idx]
        m = np.max(c_sub)
        shifted = alpha * (c_sub - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w_sub = exp_vals / s
        sm_val = m + np.log(s) / alpha
        # Build weight vector (sparse)
        w = np.zeros(len(corr))
        w[top_idx] = w_sub
    else:
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s) / alpha
    
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        if l >= 0:
            lo = l
            hi = n
            if lo < hi:
                g[lo:hi] += wl * (1.0 - h[lo-l:hi-l])
            lo2 = 0
            hi2 = n - l
            if lo2 < hi2:
                g[lo2:hi2] -= wl * h[lo2+l:hi2+l]
        else:
            m2 = -l
            lo = 0
            hi = n - m2
            if lo < hi:
                g[lo:hi] += wl * (1.0 - h[lo+m2:hi+m2])
            lo2 = m2
            hi2 = n
            if lo2 < hi2:
                g[lo2:hi2] -= wl * h[lo2-m2:hi2-m2]
    return sm_val, g

def adam_optimize(h0, n, target_sum, alpha, top_k, n_iters, lr, beta1, beta2, eps):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    m_lambda = 0.0
    v_lambda = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(1, n_iters + 1):
        sm_val, g = smooth_max_and_grad(h, alpha, top_k)
        # Add constraint gradient (Lagrangian for sum constraint)
        m_lambda = beta1 * m_lambda + (1 - beta1) * g
        v_lambda = beta2 * v_lambda + (1 - beta2) * g**2
        m_hat = m_lambda / (1 - beta1**t)
        v_hat = v_lambda / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            # Binary search + grid for optimal h[i]
            for d in base_steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_val = h_new[i]
            if best_c5 < c5_cur - 1e-15:
                h = h.copy()
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = compute_c5(h)
                improved = True
        if not improved:
            break
