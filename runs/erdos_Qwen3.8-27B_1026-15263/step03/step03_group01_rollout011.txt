import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m_val + (max_sh + np.log(np.sum(exp_v))) / alpha) * dx
    
    # Gradient: only sum over significant w
    significant = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for k in significant:
        s = int(k) - (n - 1)
        wk = w[k]
        # First term: (1-h[i-s]) for valid i-s
        i_lo = max(0, s)
        i_hi = min(n - 1, s + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += wk * (1.0 - h[i_lo - s : i_hi - s + 1])
        # Second term: -h[i+s] for valid i+s
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= wk * h[i_lo2 + s : i_hi2 + s + 1]
    grad *= dx
    return obj, grad

def nesterov_pg(h0, n, ts, alpha, n_iter=5000, lr=0.01):
    h = h0.copy()
    y = h.copy()
    beta = 0.9
    obj_prev, _ = smooth_max_grad(y, alpha)
    best_h = h.copy()
    best_obj = obj_prev
    stall = 0
    for i in range(n_iter):
        if stall > 500:
            break
        _, grad = smooth_max_grad(y, alpha)
        # Nesterov step
        h_new = project_to_sum(y - lr * grad, ts)
        # Update y
        beta_new = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        y = h_new + (beta_new - 1) / (beta + 1) * (h_new - h)
        # Project y
        y = np.clip(y, 0, 1)
        # Evaluate
        obj_new = compute_c5(h_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            lr *= 0.99
            if lr < 1e-10:
                break
        h = h_new
    return best_h, compute_c5(best_h)

def targeted_2swap(h, n, ts, rng, n_steps=30000):
    """Targeted 2-point swaps that try to reduce the max correlation lag."""
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    for step in range(n_steps):
        # Find the max lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_lag = np.argmax(corr)
        s = max_lag - (n - 1)
        
        # Try to reduce correlation at this lag
        # corr at lag s: sum_i h[i]*(1-h[i-s]) for valid
        # To reduce: increase h[i-s] where h[i] is high, or decrease h[i] where h[i-s] is low
        if s >= 0:
            i_range = np.arange(s, n)
            vals_i = h[i_range]
            vals_is = h[i_range - s]
        else:
            i_range = np.arange(0, n + s)
            vals_i = h[i_range]
            vals_is = h[i_range - s]
        
        # Find pairs where transferring mass would help
        # h[i] high and h[i-s] low → increase h[i-s], decrease h[i]
        product = vals_i * (1 - vals_is)
        # Find the pair with largest product
        if len(product) == 0:
            break
        top = np.argmax(product)
        i_pos = i_range[top]
        is_pos = i_range[top] - s
        if i_pos == is_pos:
            # Try random pair
            i_pos = rng.randint(n)
            is_pos = rng.randint(n)
            if i_pos == is_pos:
                continue
        
        delta = min(h[i_pos], (1 - h[is_pos])) * 0.3
        if delta < 1e-5:
            continue
        
        h[i_pos] -= delta
        h[is_pos] += delta
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, ts)
        
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i_pos] += delta
            h[is_pos] -= delta
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, ts)
    
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):

    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load initial
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)

            best_c5 = compute_c5(best_h)
    except Exception:
        pass
    
    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Upsample to higher resolution
    target_n = 600
    if best_n < target_n and time_left() > 600:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, target_n / 2.0)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = target_n
    
    # Phase 2: Nesterov PG with increasing alpha
    n_cur = best_n
    ts_cur = n_cur / 2.0
    for alpha in [50, 100, 200, 500, 1000, 2000, 5000]:
        if time_left() < 60:
            break
        h_pg, c5_pg = nesterov_pg(best_h, n_cur, ts_cur, alpha, n_iter=3000, lr=0.005)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Phase 3: Targeted 2-swap
    if time_left() > 120:
        h_ts, c5_ts = targeted_2swap(best_h, n_cur, ts_cur, rng, n_steps=50000)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m_val + (max_sh + np.log(np.sum(exp_v))) / alpha) * dx
    significant = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for k in significant:
        s = int(k) - (n - 1)
        wk = w[k]
        i_lo = max(0, s)
        i_hi = min(n - 1, s + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - s:i_hi - s + 1])
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 + s:i_hi2 + s + 1]
    grad *= dx
    return obj, grad

def nesterov_pg(h0, n, ts, alpha, n_iter=4000, lr_init=0.008):
    h = h0.copy()
    y = h0.copy()
    lr = lr_init
    obj_cur = compute_c5(h)
    best_h = h.copy()
    best_obj = obj_cur
    stall = 0
    momentum = 0.85
    for i in range(n_iter):
        if stall > 800:
            break
        _, grad = smooth_max_grad(y, alpha)
        h_new = project_to_sum(y - lr * grad, ts)
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, ts)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_h = h_new.copy()
            obj_cur = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.1)
        else:
            stall += 1
            lr *= 0.95
            momentum = 0.85
            if lr < 1e-12:
                break
        h = h_new
    return best_h, compute_c5(best_h)

def targeted_lag_reduction(h, n, ts, rng, n_steps=40000):
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_lags = np.argsort(corr)[::-1][:3]
        for tl in top_lags:
            s = int(tl) - (n - 1)
            if s >= 0:
                i_range = np.arange(s, n)
            else:
                i_range = np.arange(0, n + s)
            if len(i_range) == 0:
                continue
            vals_i = h[i_range]
            vals_js = h[i_range - s]
            products = vals_i * (1.0 - vals_js)
            top_idx = np.argmax(products)
            i_pos = i_range[top_idx]
            j_pos = i_range[top_idx] - s
            if i_pos == j_pos:
                i_pos = rng.randint(n)
                j_pos = (i_pos + s) % n
                if i_pos == j_pos:
                    continue
            delta = min(h[i_pos], 1.0 - h[j_pos]) * 0.25
            if delta < 1e-5:
                continue
            h[i_pos] -= delta
            h[j_pos] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
            else:
                h[i_pos] += delta
                h[j_pos] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
            break
    return h, c5_cur

def binary_local_search(h, n, ts, rng, n_flips=50000):

    n_ones = int(np.sum(h_bin > 0.5))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h_bin > 0.5)[0]
        extra = n_ones - target_ones
        to_flip = rng.choice(ones, size=extra, replace=False)
        h_bin[to_flip] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin < 0.5)[0]
        need = target_ones - n_ones
        to_flip = rng.choice(zeros, size=need, replace=False)
        h_bin[to_flip] = 1.0
    c5_cur = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5_cur
    for _ in range(n_flips):
        if rng.random() < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h_bin[i] < 0.5 and h_bin[j] > 0.5:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c5_new = compute_c5(h_bin)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_bin.copy()
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            elif h_bin[i] < 0.5 and h_bin[j] < 0.5:
                ones_arr = np.where(h_bin > 0.5)[0]
                if len(ones_arr) == 0:
                    continue
                k = rng.choice(ones_arr)
                h_bin[i] = 1.0
                h_bin[j] = 1.0
                h_bin[k] = 0.0
                c5_new = compute_c5(h_bin)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_bin.copy()
                else:

    return best_h, best_c5
