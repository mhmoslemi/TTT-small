```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s) / alpha - np.log(len(corr)) / alpha

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return exp_vals / s

def grad_smooth_max(h, w):
    """Compute gradient of smooth max given weights w on correlations."""
    n = len(h)
    g = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if abs(wm) < 1e-15:
            continue
        # First term: (1-h[i+m-n+1]) for valid i
        offset1 = m - nm1
        i_start1 = max(0, -offset1)
        i_end1 = min(n, n - offset1)
        if i_start1 < i_end1:
            g[i_start1:i_end1] += wm * (1.0 - h[i_start1 + offset1:i_end1 + offset1])
        # Second term: -h[i-m+n-1] for valid i
        offset2 = -m + nm1
        i_start2 = max(0, -offset2)
        i_end2 = min(n, n - offset2)
        if i_start2 < i_end2:
            g[i_start2:i_end2] += wm * (-h[i_start2 + offset2:i_end2 + offset2])
    return g

def adam_smooth(h0, n, target_sum, alpha, n_iters, lr_init, decay):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    
    for t in range(1, n_iters + 1):
        corr = get_corr(h)
        w = smooth_max_weights(corr, alpha)
        g = grad_smooth_max(h, w)
        
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = lr_init * decay ** t
    
    return best_h, best_c5

def topk_adam(h0, n, target_sum, n_iters, lr_init, top_k, decay):
    """Adam-like optimizer using top-k subgradient."""
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    j_arr = np.arange(n)
    
    for t in range(1, n_iters + 1):
        corr = get_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        nm1 = n - 1
        for m in top_idx:
            offset1 = m - nm1
            i_s1 = max(0, -offset1)
            i_e1 = min(n, n - offset1)
            if i_s1 < i_e1:
                g[i_s1:i_e1] += (1.0 - h[i_s1 + offset1:i_e1 + offset1])
            offset2 = -m + nm1
            i_s2 = max(0, -offset2)
            i_e2 = min(n, n - offset2)
            if i_s2 < i_e2:
                g[i_s2:i_e2] += -h[i_s2 + offset2:i_e2 + offset2]
        g /= top_k
        
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = lr_init * decay ** t
    
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, base_delta, rng, time_limit=None, t_start=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-4, -3, -2, -1, 1, 2, 3, 4]) * base_delta
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hv = h[i]
            for d in deltas:
                nv = hv + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(hv + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_start(n, rng, target_sum):
    """Generate a random Fourier-based starting point."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    n_harm = 8
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * k * x + phase)
    h += 0.5
    return project_to_sum(np.clip(h, 0, 1), target_sum)

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Main optimization loop over resolutions
    resolutions = [720, 900, 360]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return exp_vals / s

def grad_smooth_max(h, w):
    n = len(h)
    g = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        offset1 = m - nm1
        i_s1 = max(0, -offset1)
        i_e1 = min(n, n - offset1)
        if i_s1 < i_e1:
            g[i_s1:i_end1] += wm * (1.0 - h[i_s1 + offset1:i_e1 + offset1])
        offset2 = nm1 - m
        i_s2 = max(0, -offset2)
        i_e2 = min(n, n - offset2)
        if i_s2 < i_e2:
            g[i_s2:i_e2] += wm * (-h[i_s2 + offset2:i_e2 + offset2])
    return g

def topk_grad(h, top_k):
    n = len(h)
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    nm1 = n - 1
    for m in top_idx:
        offset1 = m - nm1
        i_s1 = max(0, -offset1)
        i_e1 = min(n, n - offset1)
        if i_s1 < i_e1:
            g[i_s1:i_e1] += (1.0 - h[i_s1 + offset1:i_e1 + offset1])
        offset2 = nm1 - m
        i_s2 = max(0, -offset2)
        i_e2 = min(n, n - offset2)
        if i_s2 < i_e2:
            g[i_s2:i_e2] += -h[i_s2 + offset2:i_e2 + offset2]
    return g / top_k

def adam_smooth(h0, n, target_sum, alpha, n_iters, lr_init, decay, t_start, budget):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, n_iters + 1):
        if budget - (time.time() - t_start) < 10:
            break
        corr = get_corr(h)
        w = smooth_max_weights(corr, alpha)
        g = grad_smooth_max(h, w)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        lr = lr_init * (decay ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def topk_adam(h0, n, target_sum, n_iters, lr_init, top_k, decay, t_start, budget):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, n_iters + 1):
        if budget - (time.time() - t_start) < 10:
            break
        g = topk_grad(h, top_k)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        lr = lr_init * (decay ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent_bisect(h0, n, target_sum, max_rounds, rng, t_start, budget, min_time=5):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if budget - (time.time() - t_start) < min_time:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if budget - (time.time() - t_start) < min_time:
                break
            hv = h[i]
            best_val = c5_cur
            best_nv = hv
            lo = 0.0
            hi = 1.0
            if hv > 0.01:
                lo = max(0.0, hv - 0.3)
            if hv < 0.99:
                hi = min(1.0, hv + 0.3)
            for _ in range(12):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                h_new1 = h.copy()
                h_new1[i] = mid1
                h_new1 = project_to_sum(h_new1, target_sum)
                c1 = compute_c5(h_new1)
                h_new2 = h.copy()
                h_new2[i] = mid2
                h_new2 = project_to_sum(h_new2, target_sum)
                c2 = compute_c5(h_new2)
                if c1 < c2:
                    hi = mid2
                else:
                    lo = mid1
            candidate = (lo + hi) / 2.0
            h_new = h.copy()
            h_new[i] = candidate
            h_new = project_to_sum(h_new, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_val - 1e-14:
                h = h_new
                c5_cur = c_new
                best_nv = candidate
                improved = True
                h[i] = np.clip(best_nv, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
        if not improved:
            break
    return h, c5_cur

def coord_descent_fixed(h0, n, target_sum, max_rounds, base_delta, rng, t_start, budget, min_time=5):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-4, -3, -2, -1, 1, 2, 3, 4]) * base_delta
    for rnd in range(max_rounds):
        if budget - (time.time() - t_start) < min_time:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if budget - (time.time() - t_start) < min_time:
                break
            best_val = c5_cur
            best_delta = 0.0
            hv = h[i]
            for d in deltas:
                nv = hv + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(hv + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_start(n, rng, target_sum):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, 12):
        amp = rng.uniform(0.02, 0.25) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * k * x + phase)
    h += 0.5
    return project_to_sum(np.clip(h, 0, 1), target_sum)

def sinusoidal_start(n, freq, amp, phase, target_sum):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_to_sum(np.clip(h, 0, 1), target_sum)

def multiharmonic_start(n, rng, target_sum):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    freqs = rng.choice([2, 3, 4, 5, 6, 7, 8], size=4, replace=False)
    for f in freqs:
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * f * x + phase)
    h += 0.5
    return project_to_sum(np.clip(h, 0, 1), target_sum)
```
