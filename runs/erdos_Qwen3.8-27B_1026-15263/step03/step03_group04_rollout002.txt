**Fourier approach**: The correlation c[k] can be related to the Fourier transform. If H is the DFT of h, then the autocorrelation of h is the IFFT of |H|^2. But we need the cross-correlation of h with (1-h), which is:
cross_corr = IFFT(conj(H) * (1-H_dft)) ... hmm, this gets complicated because 1-h in the time domain is not simply related to H.

Actually, the DFT of (1-h) is (1-DFT(1)) - DFT(h) = 0 - H (for the non-DC component) + (n - sum(h)) for DC.
Wait: DFT(1-h)[m] = DFT(1)[m] - DFT(h)[m] = (sum(h) if m==0 else 0) ... no.
DFT of the constant vector 1 is [n, 0, 0, ..., 0] (the DC bin is n, all others 0).
DFT of (1-h) = [n, 0, ..., 0] - H = [n - H[0], -H[1], -H[2], ..., -H[n-1]]

So cross-DFT[m] = conj(H[m]) * DFT(1-h)[m]
For m=0: conj(H[0]) * (n - H[0])
For m>0: conj(H[m]) * (-H[m]) = -|H[m]|^2

So the cross-correlation c[k] = IFFT(cross-DFT)[k]
= (1/n) * (conj(H[0])*(n-H[0]) + sum_{m=1}^{n-1} (-|H[m]|^2) * exp(2πimk/n))

This is interesting! The cross-correlation is determined by:
- A DC term: |H[0]|^2 * something
- The negative of the auto-spectrum at non-DC frequencies

To minimize max c[k], we want to:
1. Make the auto-spectrum |H[m]|^2 as flat as possible (to make the IFFT small in max norm)
2. Handle the DC term

This is similar to designing a "flat spectrum" sequence, which is what Barker sequences and related constructions try to do.

For a binary sequence, the flat spectrum property is hard to achieve exactly, but we can get close.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exp_v = np.exp(np.clip(shifted * alpha, -1000, 0))
    s = np.sum(exp_v) + 1e-300
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-9
    active = np.where(w > thresh)[0]
    for k in active:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_lbfgsb(h0, n, target_sum, alpha_steps, pen_steps, maxiter_per):
    h = h0.copy()
    for a in alpha_steps:
        for pen in pen_steps:
            def fun(x):
                obj, grad = smooth_max_grad(x, a)
                s_dev = np.sum(x) - target_sum
                reg = 0.05 * np.sum(x * x)  # small L2 to discourage all-0.5
                return obj + 0.5 * pen * s_dev * s_dev + reg
            def jac(x):
                _, grad = smooth_max_grad(x, a)
                s_dev = np.sum(x) - target_sum
                return grad + pen * s_dev + 0.1 * x
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': maxiter_per, 'ftol': 1e-15, 'gtol': 1e-8})
            h = res.x
        h = project_to_sum(h, target_sum)
    return h

def subgradient_polish(h, n, target_sum, max_rounds, t0, time_limit):
    h = h.copy()
    c5 = compute_c5(h)
    for r in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corr = get_corr(h)
        top_k = np.argsort(corr)[::-1][:5]
        best_improve = 0
        best_h = None
        for k_idx in top_k:
            shift = k_idx - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05]:
                h_new = project_to_sum(h - lr * grad, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-14:
                    if c5 - c5_new > best_improve:
                        best_improve = c5 - c5_new
                        best_h = h_new
        if best_h is None:
            break
        h = best_h
        c5 = compute_c5(h)
    return h, c5

def binary_sa_search(h_init, n, target_sum, rng, t0, time_limit, max_rounds=2000):
    h_bin = (h_init > 0.5).astype(float)
    # Fix sum
    attempts = 0

        if h_bin.sum() > target_sum:

            zeros = np.append(zeros, idx)

            idx = rng.choice(zeros)

            zeros = np.where(h_bin == 0)[0]

        attempts += 1

        return h_bin, compute_c5(h_bin)
    
    c5 = compute_c5(h_bin)

            c5 = c5_test

                best_c5 = c5
                best_h = h_bin.copy()

    return best_h, best_c5
