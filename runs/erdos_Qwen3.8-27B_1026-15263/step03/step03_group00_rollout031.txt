                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = h.copy()

        return project_to_sum(h_new, target_sum)

def coordinate_descent(h, target_sum, max_iters=5000, step_frac=0.1):
    """Classic coordinate descent on the max lag"""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n-lag])
                g[:n-lag] -= h[lag:]
        else:
            k = -lag
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[:n-k] += (1.0 - h[k:])
                g[k:] -= h[:n-k]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def nelder_mead_free(h, target_sum, maxiter=200):
    """Nelder-Mead on the free subspace"""
    n = len(h)
    free_mask = (h > 0.01) & (h < 0.99)
    n_free = np.sum(free_mask)
    if n_free < 2 or n_free > 30:
        return h  # NM is only good in low dimensions
    
    free_idx = np.where(free_mask)[0]
    fixed_sum = np.sum(h[~free_mask])
    free_target = target_sum - fixed_sum
    x0 = h[free_idx].copy()
    
    def nm_obj(x_free):
        x_full = h.copy()
        x_full[free_idx] = x_free
        x_full = np.clip(x_full, 0, 1)
        # Project to maintain sum
        diff = target_sum - np.sum(x_full)
        if abs(diff) > 1e-10:
            x_full[free_idx] += diff / n_free
            x_full[free_idx] = np.clip(x_full[free_idx], 0, 1)
        corr = np.correlate(x_full, 1.0 - x_full, mode="full")
        return np.max(corr)
    
    try:
        res = minimize(nm_obj, x0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        h_new = h.copy()
        h_new[free_idx] = np.clip(res.x, 0, 1)
        return project_to_sum(h_new, target_sum)
    except Exception:
        return h

def smooth_max_weights(corr, alpha):
    m_val = np.max(corr)

    shifted -= np.max(shifted)

        obj = m_val + np.log(norm) / alpha
        g = np.zeros(n)
        for j, ti in enumerate(top_idx):
            wm = w_top[j]
            if wm < 1e-10:
                continue
            k = ti - (n - 1)
            if k >= 0:
                if k == 0:
                    g += wm * (1.0 - 2.0 * x)
                else:
                    g[:n - k] += wm * (1.0 - x[k:])
                    g[k:] -= wm * x[:n - k]
            else:
                kk = -k
                g[:n - kk] += wm * (1.0 - x[kk:])
                g[kk:] -= wm * x[:n - kk]
        return obj, g

    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_and_grad, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts_cur = n_cur / 2.0

    alphas_schedule = [30, 100, 400, 1500, 5000, 20000]
    for alpha in alphas_schedule:
        if time_left() < 80:
            break
        h_new = slsqp_full(best_h, ts_cur, alpha, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    for _ in range(4):
        if time_left() < 60:
            break
        h_sub = slsqp_subspace(best_h, ts_cur, alpha=8000, maxiter=500, tol=0.005)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    if time_left() > 50:
        h_bal = top_lag_balance(best_h, ts_cur, alpha=3000, n_top=3, maxiter=400)
        c5_bal = compute_c5(h_bal)
        if c5_bal < best_c5:
            best_c5 = c5_bal
            best_h = h_bal.copy()

    resolutions_to_try = []
    for nr in [480, 540, 600, 660, 720, 800]:
        if nr % 2 == 0 and nr != n_cur:
            resolutions_to_try.append(nr)
    resolutions_to_try.sort()

    for n_try in resolutions_to_try:
        if time_left() < 150:
            break
        ts_try = n_try / 2.0
        h_interp = interp_periodic(best_h, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            n_cur = n_try
            ts_cur = ts_try
        for alpha in [200, 1000, 5000]:
            if time_left() < 50:
                break
            h_opt = slsqp_full(h_interp, ts_try, alpha, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n_cur = n_try
                ts_cur = ts_try
            h_interp = h_opt
        if time_left() > 40:
            h_sub = slsqp_subspace(h_interp, ts_try, alpha=5000, maxiter=400, tol=0.005)
            c5_sub = compute_c5(h_sub)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                n_cur = n_try
                ts_cur = ts_try
        if time_left() > 30:
            nm_h = nm_free_subspace(h_interp, ts_try, maxiter=200)
            c5_nm = compute_c5(nm_h)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = nm_h.copy()
                n_cur = n_try
                ts_cur = ts_try

    if time_left() > 60:
        h_cd, c5_cd = coord_desc_adaptive(best_h, ts_cur, max_iters=10000, base_step=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    n_restarts = 6
    for trial in range(n_restarts):
        if time_left() < 70:
            break
        noise = 0.015 + 0.035 * (1.0 - trial / float(n_restarts))
        h_shake = best_h + rng.uniform(-noise, noise, n_cur)
        h_shake = project_to_sum(h_shake, ts_cur)
        h_opt = slsqp_full(h_shake, ts_cur, alpha=3000, maxiter=250)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time_left() > 40:
            h_sub = slsqp_subspace(h_opt, ts_cur, alpha=8000, maxiter=300, tol=0.005)
            c5_sub = compute_c5(h_sub)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
        if time_left() > 30:
            h_cd, c5_cd = coord_desc_adaptive(best_h, ts_cur, max_iters=5000, base_step=0.06)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    for alpha in [30000, 80000]:
        if time_left() < 35:
            break
        h_pol = slsqp_full(best_h, ts_cur, alpha, maxiter=600)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_sub = slsqp_subspace(best_h, ts_cur, alpha=alpha, maxiter=400, tol=0.003)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    if time_left() > 40:
        h_bal = top_lag_balance(best_h, ts_cur, alpha=10000, n_top=4, maxiter=500)
        c5_bal = compute_c5(h_bal)
        if c5_bal < best_c5:
            best_c5 = c5_bal
            best_h = h_bal.copy()

    if time_left() > 30:
        nm_h = nm_free_subspace(best_h, ts_cur, maxiter=200)
        c5_nm = compute_c5(nm_h)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = nm_h.copy()

    if time_left() > 25:
        h_cd, c5_cd = coord_desc_adaptive(best_h, ts_cur, max_iters=20000, base_step=0.03)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 20:
        h_pol = slsqp_full(best_h, ts_cur, alpha=100000, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = project_to_sum(best_h, n_cur / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_cur)
