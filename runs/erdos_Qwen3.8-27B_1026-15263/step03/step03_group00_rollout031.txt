import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    expv = np.exp(shifted)
    w = expv / np.sum(expv)
    obj = m_val + np.log(np.sum(expv)) / alpha
    # Gradient
    grad = np.zeros(n)
    # Skip negligible weights
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    for k in active:
        m = k - (n - 1)
        wm = w[k]
        if m >= 0:
            grad[m:] += wm * (1.0 - h[:n - m])
            grad[:n - m] -= wm * h[m:]
        else:
            kk = -m
            grad[:n - kk] += wm * (1.0 - h[kk:])
            grad[kk:] -= wm * h[:n - kk]
    return obj, grad

def nesterov_pg(h0, target_sum, alpha, n_iters, step_size):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    for k in range(n_iters):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        obj, grad = smooth_obj_grad(y, alpha)
        x_new = project_to_sum(y - step_size * grad, target_sum)
        y = x_new + beta * (x_new - x)
        y = np.clip(y, 0.0, 1.0)  # keep y in bounds loosely
        x = x_new
        t = t_new
    return x

def coordinate_descent_polish(h, target_sum, max_iters=5000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            stale += 1
            if stale > 100:
                break
            continue
        stale = 0
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 200:
                break
        else:
            c5 = c5_new
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_obj_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - target_sum
    def cjac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_at_topk(h, K=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    topk = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for k in topk:
        m = k - (n - 1)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            kk = -m
            g[:n - kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n - kk]
    g /= K
    return g, np.max(corr)
