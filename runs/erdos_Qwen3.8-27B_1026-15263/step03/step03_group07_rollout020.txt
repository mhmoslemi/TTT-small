def fft_gradient(h, n, alpha):
    """Compute gradient of smooth max using FFT."""
    # Compute correlations
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Compute softmax weights
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # w is indexed by lag: w[k + n - 1] = weight for lag k
    # w length is 2n-1
    
    # For A[j] = sum_k w[k] * h[j+k]:
    # Use convolution: A = ifft(fft(w_rev) * fft(h))
    # where w_rev[m] = w[2*(n-1)-m] (reverse of w)
    
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    
    h_padded = np.zeros(N_fft)
    h_padded[:n] = h
    
    w_padded = np.zeros(N_fft)
    w_padded[:2*n-1] = w
    w_rev_padded = np.zeros(N_fft)
    w_rev_padded[:2*n-1] = w[::-1]
    
    # B[j] = sum_k w[k] * h[j-k] = convolve(w, h)[j + n - 1]
    B_full = np.fft.ifft(np.fft.fft(w_padded) * np.fft.fft(h_padded)).real
    B = B_full[n-1:n-1+n]  # indices n-1 to 2n-2
    
    # A[j] = sum_k w[k] * h[j+k] = convolve(w_rev, h)[j + n - 1]
    A_full = np.fft.ifft(np.fft.fft(w_rev_padded) * np.fft.fft(h_padded)).real

def project_onto_constraint(x, target_sum):
