def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 3: Aggressive polishing (key improvement)
    if remaining() > 100:
        h_pol, c5_pol = aggressive_polish(best_h, n1, ts1,
                                           max_rounds=3000, top_m=20, rel_thresh=0.003)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Try higher resolution
    if remaining() > 350:
        for n2_factor in [2, 3]:
            n2 = n1 * n2_factor
            if n2 > 720:
                continue
            ts2 = n2 / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = project_to_sum(h_interp, ts2)

            for a in [50, 200, 1000, 5000, 20000]:
                if remaining() < 80:
                    break
                h_opt = optimize_penalty(h_interp, n2, ts2, a,
                                         penalties=[100, 1000, 10000, 100000],
                                         maxiter_per_level=100)
                c5 = compute_c5(h_opt)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n2
                    h_interp = h_opt.copy()

            if remaining() > 100:
                h_pol2, c5_pol2 = aggressive_polish(best_h, best_n, best_n / 2.0,
                                                     max_rounds=1500, top_m=15, rel_thresh=0.002)
                if c5_pol2 < best_c5:
                    best_c5 = c5_pol2
                    best_h = h_pol2.copy()
            if remaining() < 350:
                break

    # Phase 5: Try lower resolutions for different topologies
    if remaining() > 200:
        for n_try in [80, 120, 160, 200]:
            if remaining() < 60:
                break
            ts_try = n_try / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_down = np.interp(x_new, x_old, best_h, period=2.0)
            h_down = project_to_sum(h_down, ts_try)
            for a in [100, 500, 2000, 10000]:
                if remaining() < 30:
                    break
                h_opt = optimize_penalty(h_down, n_try, ts_try, a,
                                         penalties=[100, 1000, 10000],
                                         maxiter_per_level=120)
                c5 = compute_c5(h_opt)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n_try
                    h_down = h_opt.copy()
            if remaining() > 30:
                h_pol3, c5_pol3 = aggressive_polish(h_down, n_try, ts_try,
                                                     max_rounds=500, top_m=10, rel_thresh=0.002)
                if c5_pol3 < best_c5:
                    best_c5 = c5_pol3
                    best_h = h_pol3.copy()
                    best_n = n_try

    # Phase 6: DE-based coarse search for alternative topologies
    if remaining() > 200:
        n_coarse = 48
        ts_coarse = n_coarse / 2.0
        t_de_start = time.time()
        h_de = de_coarse_search(n_coarse, ts_coarse, rng, time_limit=120)
        t_de_end = time.time()
        # Upsample and refine
        n_up = best_n
        ts_up = n_up / 2.0
        x_c = np.linspace(0, 2, n_coarse, endpoint=False)
        x_u = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_u, x_c, h_de, period=2.0)
        h_up = project_to_sum(h_up, ts_up)
        for a in [50, 500, 5000]:
            if remaining() < 40:
                break
            h_opt = optimize_penalty(h_up, n_up, ts_up, a,
                                     penalties=[1000, 10000, 100000],
                                     maxiter_per_level=100)
            c5 = compute_c5(h_opt)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h_opt.copy()
                h_up = h_opt.copy()
        if remaining() > 60:
            h_pol4, c5_pol4 = aggressive_polish(h_up, n_up, ts_up,
                                                 max_rounds=1000, top_m=12, rel_thresh=0.002)
            if c5_pol4 < best_c5:
                best_c5 = c5_pol4
                best_h = h_pol4.copy()

    # Phase 7: Multi-start at best resolution
    if remaining() > 150:
        n_best = best_n
        ts_best = n_best / 2.0
        starts = make_starts(n_best, ts_best, rng)
        for h_s in starts[:8]:
            if remaining() < 40:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for a in [100, 1000, 5000]:
                if remaining() < 25:
                    break
                h_opt = optimize_penalty(h_s, n_best, ts_best, a,
                                         penalties=[100, 1000, 10000],
                                         maxiter_per_level=80)
                c5 = compute_c5(h_opt)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    h_s = h_opt.copy()

    # Phase 8: Binary flip search
    if remaining() > 100:
        h_bf, c5_bf = binary_search_flip(best_h, best_n, best_n / 2.0, rng, max_rounds=150)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        if remaining() > 40:
            h_pol5, c5_pol5 = aggressive_polish(best_h, best_n, best_n / 2.0,
                                                 max_rounds=500, top_m=10, rel_thresh=0.002)
            if c5_pol5 < best_c5:
                best_c5 = c5_pol5
                best_h = h_pol5.copy()

    # Phase 9: Final aggressive polish + SLSQP
    if remaining() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        h_pol_f, c5_f = aggressive_polish(best_h, n_f, ts_f,
                                           max_rounds=2000, top_m=25, rel_thresh=0.001)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_pol_f.copy()
        if remaining() > 30:
            for a in [10000, 50000]:
                if remaining() < 20:
                    break
                h_opt = optimize_penalty(best_h, n_f, ts_f, a,
                                         penalties=[10000, 100000],
                                         maxiter_per_level=200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h_opt.copy()
        if remaining() > 20:
            h_pol_g, c5_g = aggressive_polish(best_h, n_f, ts_f,
                                               max_rounds=1000, top_m=30, rel_thresh=0.0005)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_pol_g.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
