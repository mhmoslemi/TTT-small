def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

            c5_curr = best_c5
            T = min(T * 1.5, 0.018)
            no_improve = 0

        T = max(T * 0.999998, min_T)

    return best_h, best_c5


def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)


def steepest_descent_discrete(h, n, target_sum_int, max_sweeps=50):
    h = h.copy().astype(float)
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    ones_idx = np.where(h == 1.0)[0].tolist()
    zeros_idx = np.where(h == 0.0)[0].tolist()

    for sweep in range(max_sweeps):
        improved = False
        for i in ones_idx:
            for j in zeros_idx:
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < c5_curr - 1e-14:
                    c5_curr = c5_new
                    improved = True
                    if c5_curr < best_c5:
                        best_c5 = c5_curr
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = int(n_main / 2.0)

    h_bin = make_binary(best_h, n_main, ts_main, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    t_phase1_end = t_start + 0.40 * budget_s
    h_sa, c5_sa = guided_sa(h_bin, n_main, ts_main, t_phase1_end, rng, use_guided=True)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    t_phase2_end = t_start + 0.50 * budget_s
    if remaining() > 60:
        h_sa2, c5_sa2 = random_sa(h_bin, n_main, ts_main, t_phase2_end, rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    t_phase3_end = t_start + 0.65 * budget_s
    if remaining() > 80:
        h_rd, c5_rd = steepest_descent_discrete(h_bin, n_main, ts_main, max_sweeps=30)
        if c5_rd < best_c5:
            best_c5 = c5_rd
            best_h = h_rd.copy()
        h_guided2, c5_g2 = guided_sa(h_rd, n_main, ts_main, t_phase3_end, rng, use_guided=True)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    h_bin_current = make_binary(best_h, n_main, ts_main, rng)

    alphas = [20, 80, 300, 1200, 5000]
    t_slsqp_end = t_start + 0.80 * budget_s
    for alpha in alphas:
        if elapsed() > t_slsqp_end - 30:
            break
        h_pol = slsqp_optimize(h_bin_current, n_main, float(ts_main), alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_bin_current = make_binary(h_pol, n_main, ts_main, rng)

    t_multi_end = t_start + 0.92 * budget_s
    trial_n_values = [200, 280, 320]
    for n_try in trial_n_values:
        if elapsed() > t_multi_end:
            break
        if n_try % 2 != 0:
            continue
        if n_try == n_main:
            continue
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = h_interp
        for alpha in [30, 200, 1500]:
            if elapsed() > t_multi_end - 20:
                break
            h_try = slsqp_optimize(h_try, n_try, ts_try, alpha, maxiter=1000)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    if remaining() > 50:
        h_shake = best_h + rng.uniform(-0.05, 0.05, best_n)
        h_shake = project_to_sum(h_shake, best_n / 2.0)
        h_fin = slsqp_optimize(h_shake, best_n, best_n / 2.0, 3000, maxiter=2000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if remaining() > 30:
        h_fin2 = slsqp_optimize(best_h, best_n, best_n / 2.0, 10000, maxiter=2500)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
