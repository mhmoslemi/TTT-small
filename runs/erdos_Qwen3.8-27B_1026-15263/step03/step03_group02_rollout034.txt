import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    s = np.sum(np.exp(shifted))
    return m_val + np.log(s) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    j = np.arange(n)
    k = np.arange(n_corr)
    idx1 = j[:, None] - k[None, :] + n - 1
    idx2 = j[:, None] + k[None, :] - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    dcorr = (1.0 - h[idx1c]) * valid1 - h[idx2c] * valid2
    grad = dcorr @ w
    return m_val + np.log(np.sum(exp_vals)) / alpha, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000, ftol=1e-14):
    def f(x):
        return smooth_max_obj(x, alpha)
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):

    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:

        ones_idx = np.where(h_bin == 1.0)[0]

        order = np.argsort(-h[zeros_idx])[:deficit]

def interpolated_h(h_src, n_src, n_dst):

def perturb_smooth(h, n, target_sum, rng, style):

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    alphas = [8, 20, 50, 100, 300, 800, 2000, 5000, 12000, 30000]
    alpha_idx = 0

    for alpha in alphas:
        if elapsed() > 0.35 * budget_s:
            break
        h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    sa_end1 = t_start + 0.45 * budget_s
    if remaining() > 50:
        h_sa, c5_sa = continuous_sa(best_h, n_main, ts_main, sa_end1, rng, t0=0.006, t_min=0.001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    for alpha in [500, 2000, 8000, 20000, 50000]:
        if elapsed() > 0.55 * budget_s:
            break
        h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=800)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    n_perturbs = 12
    perturb_budget_per = min(remaining() / (n_perturbs + 2), 60.0)

    for trial in range(n_perturbs):
        if remaining() < 30:
            break
        style = trial % 4
        h_pert = perturb_smooth(best_h, n_main, ts_main, rng, style)
        alpha_use = [200, 1000, 5000, 15000][min(trial // 3, 3)]
        t_pert_end = time.time() + perturb_budget_per
        h_opt = slsqp_optimize(h_pert, n_main, ts_main, alpha_use, maxiter=500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    sa_end2 = time.time() + min(remaining() * 0.3, 120)
    if remaining() > 80:
        h_bin = make_binary(best_h, n_main, int(ts_main), rng)
        h_sa2, c5_sa2 = binary_sa(h_bin, n_main, int(ts_main), sa_end2, rng)
        c5_sa2 = compute_c5(h_sa2)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        h_from_bin = project_to_sum(h_sa2.astype(float) * 0.3 + best_h.astype(float) * 0.7, ts_main)
        for alpha in [1000, 5000]:
            if elapsed() > 0.85 * budget_s:
                break
            h_opt = slsqp_optimize(h_from_bin, n_main, ts_main, alpha, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    for n_try in [180, 200, 280, 320, 360, 480]:
        if remaining() < 60:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = h_interp
        for alpha in [50, 300, 1500]:
            if remaining() < 30:
                break
            h_try = slsqp_optimize(h_try, n_try, ts_try, alpha, maxiter=500)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    for alpha in [50000, 100000, 200000]:
        if elapsed() > 0.97 * budget_s:
            break
        h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=1500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if remaining() > 20:
        h_final = best_h.copy()
        c5_final = compute_c5(h_final)
        for _ in range(5):
            h_shake = h_final + rng.normal(0, 0.01, best_n)
            h_shake = project_to_sum(h_shake, best_n / 2.0)
            h_fin2 = slsqp_optimize(h_shake, best_n, best_n / 2.0, 50000, maxiter=500)
            c5_fin2 = compute_c5(h_fin2)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()
                h_final = h_fin2.copy()
                c5_final = c5_fin2

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
