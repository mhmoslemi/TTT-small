```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            count = np.sum(mask)
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve_fixed(g, n, top_k=None):
    """Solve LP: min t s.t. corr(h, g)[s]*dx <= t for all s, sum h = n/2, 0<=h<=1"""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    if top_k is not None:
        corr_approx = np.correlate((1-g), g, mode='full')  # approximate
        idx = np.argsort(corr_approx)[::-1][:top_k]
        shifts = [int(k - (n-1)) for k in idx]
    else:
        shifts = list(range(-(n-1), n))
    
    for s in shifts:
        if s >= 0:
            c = g[:n-s] @ h_var[s:]
        else:
            c = g[-s:] @ h_var[:n+s]
        constraints.append(c * dx <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return None
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1)
    return None

def alternating_lp(h0, n, max_iters, t_deadline, use_all_shifts=True):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        
        g = 1.0 - h
        h_new = lp_solve_fixed(g, n, top_k=None if use_all_shifts else 50)
        if h_new is None:
            break
        h_new = project_to_sum(h_new, n/2.0)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def subgradient_polish(h, n, max_iters, t_deadline, lr_init=0.1):
    """Polish using subgradient of max over active shifts."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    ts = n / 2.0
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr) * 2.0/n)
        
        # Find active shifts (top few)
        threshold = c5_cur - 0.01
        active = np.where(corr * 2.0/n >= threshold)[0]
        
        # Compute subgradient (average of gradients of active shifts)
        grad = np.zeros(n)
        for k in active:
            s = int(k - (n-1))
            if s >= 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] -= h[s:]
            else:
                grad[:-s] += (1.0 - h[-s:])
                grad[-s:] -= h[:n+s]
        
        grad /= max(len(active), 1)
        
        # Take step
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.8
        
        if lr < 1e-6:
            break
    
    return best_h, best_c5

def frank_wolfe_polish(h, n, max_iters, t_deadline):
    """Frank-Wolfe (conditional gradient) on the max objective."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ts = n / 2.0
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr) * 2.0/n)
        
        # Find the worst shift
        k_max = int(np.argmax(corr))
        s = k_max - (n-1)
        
        # Compute gradient at worst shift
        grad = np.zeros(n)
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            grad[:-s] += (1.0 - h[-s:])
            grad[-s:] -= h[:n+s]
        
        # Linear minimization: min_v v·grad s.t. sum v = n/2, 0<=v<=1
        # The optimal v is: put 1 where grad is most negative, 0 where most positive
        order = np.argsort(grad)
        v = np.zeros(n)
        v[order[:int(ts)]] = 1.0
        # Actually need exactly n/2 ones (assuming n is even)
        n_ones = int(ts)
        v[order[:n_ones]] = 1.0
        
        # FW step size
        step = 2.0 / (it + 2)
        h_new = (1 - step) * h + step * v
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            count = np.sum(mask)
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_step(g, n):
    """Solve: min t s.t. g[:n-|s|] @ h[|s|:] * dx <= t for all s, sum h = n/2, 0<=h<=1"""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    for s in range(-(n-1), n):
        abs_s = abs(s)
        if abs_s == 0:
            c = g @ h_var
        else:
            c = g[:n-abs_s] @ h_var[abs_s:]
        constraints.append(c * dx <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None
    
    if h_var.value is not None:
        return np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
    return None

def alternating_lp_optimize(h0, n, max_iters, t_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        g = 1.0 - h
        h_new = lp_step(g, n)
        if h_new is None:
            break
        h_new = project_to_sum(h_new, n / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def grad_max_corr(h, n, top_k=5):
    """Compute gradient of max of top_k correlation values."""
    corr = compute_corr(h)
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        s = int(k - (n - 1))
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            sa = -s
            grad[:n-sa] += (1.0 - h[sa:])
            grad[sa:] -= h[:n-sa]
    grad /= float(len(idx))
    return grad

def subgradient_optimize(h0, n, max_iters, t_deadline, top_k=5):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ts = n / 2.0
    lr = 0.05
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        grad = grad_max_corr(h, n, top_k)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr *= 0.999
    
    return best_h, best_c5

def frank_wolfe(h0, n, max_iters, t_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ts = n / 2.0
    n_ones = int(ts)
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        
        grad = np.zeros(n)
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            sa = -s
            grad[:n-sa] += (1.0 - h[sa:])
            grad[sa:] -= h[:n-sa]
        
        # Linear min: min v·grad s.t. sum v = n_ones, 0<=v<=1
        order = np.argsort(grad)
        v = np.zeros(n)
        v[order[:n_ones]] = 1.0
        
        step = 2.0 / (it + 2.0)
        h_new = (1.0 - step) * h + step * v
        h_new = project_to_sum(h_new, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            count = np.sum(mask)
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_step(g, n):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1

    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = [cp.GLPK, cp.CBC, cp.CPLEX, cp.OSQP]
    for solver in solvers:

            prob.solve(solver=solver, verbose=False)
```
