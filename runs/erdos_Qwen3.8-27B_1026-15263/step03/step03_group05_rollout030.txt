import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def projected_gradient_descent(h0, n, alpha, max_iter, lr_init, time_fn):
    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        else:
            lr = min(lr * 1.1, 0.5)
        h = h_new
    return best_h, best_obj

            h[j] += 0.01

                    h[i] -= d
