def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha, extra_lag=None, extra_weight=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    if extra_lag is not None and extra_weight > 0:
        obj += extra_weight * corr[extra_lag]
        w = w.copy()
        obj = 0.0
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    if extra_lag is not None and extra_weight > 0:
        lag_j = extra_lag
        fi2 = np.arange(n) - lag_j + n - 1
        si2 = np.arange(n) + lag_j - n + 1
        vf2 = (fi2 >= 0) & (fi2 < n)

        fc2 = np.clip(fi2, 0, n - 1)
        sc2 = np.clip(si2, 0, n - 1)

    corr = np.correlate(h, 1.0 - h, mode="full")
    mean_c = np.mean(corr)
    obj = np.sum((corr - mean_c) ** 2) / (2.0 * len(corr))
    dcorr_dh = np.zeros((n, 2 * n - 1))
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    dcorr_dh = vf * (1.0 - h[fc]) - vs * h[sc]
    grad = dcorr_dh.T @ (corr - mean_c) / len(corr)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800, extra_lag=None, extra_weight=0.0):
    def f(x):
        return smooth_max_obj_grad(x, alpha, extra_lag, extra_weight)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha, extra_lag, extra_weight)[1]
    def cfun(x):
        return np.sum(x) - ts
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def variance_opt(h0, n, ts, maxiter=500):
    def f(x):
        return variance_obj_grad(x)[0]
    def g(x):
        return variance_obj_grad(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def fourier_perturb(h, rng, n_modes, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_modes, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def stochastic_cd(h, n, ts, rng, n_steps=40000, delta_range=(0.005, 0.2)):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.2
    for step in range(n_steps):
        ones_idx = np.where(h > 0.03)[0]
        zeros_idx = np.where(h < 0.97)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp) if temp > 1e-8 else 0.0
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-6, 0.2 * np.exp(-step / 4000.0))
    return h, c5_cur

def local_search_h(h, n, ts, rng, n_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(n_rounds):
        best_delta = 0
        best_c5 = c5_cur
        best_pair = None
        for _ in range(30):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] < 0.01 or h[j] > 0.99:
                continue
            for delta in [0.05, 0.1, 0.15, 0.2]:
                if h[i] < delta or 1.0 - h[j] < delta:
                    continue
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5_t = compute_c5(h)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_delta = delta
                    best_pair = (i, j, delta)
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        if best_delta > 0:
            i, j, delta = best_pair
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def make_anti_periodic(h, n):
    h_ap = h.copy()
    half = n // 2
    h_ap[:half] = 0.5 * (h[:half] + (1.0 - h[half:2*half]))
    h_ap[half:2*half] = 1.0 - h_ap[:half]
    h_ap[2*half:] = h[2*half:]
    return project_to_sum(h_ap, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 480
        best_h = np.linspace(0, 2, best_n, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(np.pi * best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 1: Upsample to n=960 and initialize
    n1 = 960
    ts1 = n1 / 2.0
    h1 = upsample_h(best_h, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1

    # Phase 2: Variance flattening at n=960
    if time_left() > 60:
        h_v = variance_opt(h1, n1, ts1, maxiter=400)
        c5_v = compute_c5(h_v)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_v.copy()
            best_n = n1
            h1 = h_v
        else:
            h_v2 = variance_opt(h1, n1, ts1, maxiter=200)
            c5_v2 = compute_c5(h_v2)
            if c5_v2 < best_c5:
                best_c5 = c5_v2
                best_h = h_v2.copy()
                best_n = n1
                h1 = h_v2

    # Phase 3: Continuation with smooth max at n=960
    alphas = [5, 15, 50, 150, 500, 1500, 5000, 10000]
    for i_a, alpha in enumerate(alphas):
        if time_left() < 40:
            break
        maxiter_a = 600 if i_a < 4 else 1000
        h_opt = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=maxiter_a)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Worst-lag targeted optimization
    for trial in range(8):
        if time_left() < 30:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        worst_lag = int(np.argmax(corr))
        h_wl = slsqp_opt(best_h, best_n, best_n / 2.0, 2000, maxiter=800,
                         extra_lag=worst_lag, extra_weight=0.1)
        c5_wl = compute_c5(h_wl)
        if c5_wl < best_c5:
            best_c5 = c5_wl
            best_h = h_wl.copy()
        else:
            h_wl2 = slsqp_opt(best_h, best_n, best_n / 2.0, 3000, maxiter=600,
                              extra_lag=worst_lag, extra_weight=0.05)
            c5_wl2 = compute_c5(h_wl2)
            if c5_wl2 < best_c5:
                best_c5 = c5_wl2
                best_h = h_wl2.copy()

    # Phase 5: Fourier-space perturbations and re-optimization
    for trial in range(20):
        if time_left() < 35:
            break
        n_modes = max(5, best_n // 30)
        amp = 0.06 * (1.0 - trial / 25.0) + 0.01
        h_f = fourier_perturb(best_h, rng, n_modes, amp)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        for alpha in [200, 1000]:
            if time_left() < 20:
                break
            h_ot = slsqp_opt(h_f, best_n, best_n / 2.0, alpha, maxiter=400)
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            h_f = h_ot

    # Phase 6: Anti-periodic symmetry search
    if time_left() > 60:
        h_ap = make_anti_periodic(best_h, best_n)
        c5_ap = compute_c5(h_ap)
        for alpha in [100, 500, 2000]:
            if time_left() < 20:
                break
            h_ap = slsqp_opt(h_ap, best_n, best_n / 2.0, alpha, maxiter=500)
            c5_ap = compute_c5(h_ap)
            # Enforce anti-periodicity
            half = best_n // 2
            h_ap[:half] = 0.5 * (h_ap[:half] + (1.0 - h_ap[half:2*half]))
            h_ap[half:2*half] = 1.0 - h_ap[:half]
            h_ap = project_to_sum(h_ap, best_n / 2.0)
            c5_ap = compute_c5(h_ap)
            if c5_ap < best_c5:
                best_c5 = c5_ap
                best_h = h_ap.copy()

    # Phase 7: Try n=1200 if time permits
    if time_left() > 150:
        n2 = 1200
        ts2 = n2 / 2.0
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        for alpha in [50, 200, 800, 3000]:
            if time_left() < 40:
                break
            h2 = slsqp_opt(h2, n2, ts2, alpha, maxiter=600)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 8: Stochastic coordinate descent
    if time_left() > 80:
        h_cd = best_h.copy()
        n_cd = len(h_cd)
        c5_cd = compute_c5(h_cd)
        h_cd, c5_cd = stochastic_cd(h_cd, n_cd, n_cd / 2.0, rng,
                                    n_steps=50000, delta_range=(0.005, 0.15))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: High-alpha polish
    if time_left() > 50:
        for alpha in [5000, 15000, 50000]:
            if time_left() < 30:
                break
            h_p = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=1200)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 10: Local search
    if time_left() > 40:
        h_ls, c5_ls = local_search_h(best_h, best_n, best_n / 2.0, rng, n_rounds=100)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            if time_left() > 25:
                h_p = slsqp_opt(best_h, best_n, best_n / 2.0, 10000, maxiter=800)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Phase 11: Second round of perturbation + optimization
    if time_left() > 60:
        for trial in range(10):
            if time_left() < 25:
                break
            h_t = fourier_perturb(best_h, rng, max(8, best_n // 20), 0.015)
            h_t = slsqp_opt(h_t, best_n, best_n / 2.0, 3000, maxiter=500)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 12: Tiny stochastic refinements
    if time_left() > 40:
        h_t, c5_t = stochastic_cd(best_h, best_n, best_n / 2.0, rng,
                                  n_steps=30000, delta_range=(0.003, 0.08))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final: one more polish
    if time_left() > 20:
        h_f = slsqp_opt(best_h, best_n, best_n / 2.0, 20000, maxiter=1000)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
