def eval_swap(corr, h, i, j, n):
    m_arr = np.arange(2 * n - 1)
    
    # Delta from h[i]: 1 -> 0
    idx_i = i + m_arr - n + 1
    mask_i = (idx_i >= 0) & (idx_i < n)
    delta_i = np.zeros(2 * n - 1)
    delta_i[mask_i] = 2.0 * h[idx_i[mask_i]] - 1.0
    
    # Delta from h[j]: 0 -> 1
    idx_j = j + m_arr - n + 1
    mask_j = (idx_j >= 0) & (idx_j < n)
    delta_j = np.zeros(2 * n - 1)
    delta_j[mask_j] = 1.0 - 2.0 * h[idx_j[mask_j]]
    
    new_corr = corr + delta_i + delta_j
    
    # Correction for interaction
    m_A = j - i + n - 1
    if 0 <= m_A < 2 * n - 1:
        new_corr[m_A] += 1.0  # correction for case A
    m_B = i - j + n - 1
    if 0 <= m_B < 2 * n - 1:
        new_corr[m_B] -= 1.0  # correction for case B
    
    return float(np.max(new_corr)), new_corr
