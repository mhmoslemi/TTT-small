import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    return (m_val + max_shift + np.log(np.sum(np.exp(alpha * (shifted - max_shift)))) / alpha) * dx

def fw_direction(grad, n):
    """Frank-Wolfe direction: argmin grad·y s.t. 0≤y≤1, sum(y)=n/2"""
    indices = np.argsort(grad)
    y_star = np.zeros(n)
    y_star[indices[:n // 2]] = 1.0
    return y_star

def frank_wolfe_opt(h0, n, alpha, max_iter, time_fn):
    h = h0.copy()
    best_obj = smooth_max_obj(h, alpha)
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        y_star = fw_direction(grad, n)
        # Backtracking line search
        t = 1.0
        for _ in range(20):
            h_try = (1 - t) * h + t * y_star
            obj_try = smooth_max_obj(h_try, alpha)
            if obj_try <= obj - 1e-4 * t * np.abs(grad).max():
                break
            t *= 0.5
        h = (1 - t) * h + t * y_star
        obj_h = smooth_max_obj(h, alpha)
        if obj_h < best_obj:
            best_obj = obj_h
            best_h = h.copy()
    return best_h, best_obj

def binary_local_search(h, n, rng, max_rounds, time_fn):
    """2-flip local search on binary vector"""
    h = h.copy().astype(float)
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time_fn() < 5:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        # Try all pairs if feasible, otherwise random sampling
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 50000:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c5_t = compute_c5(h)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_pair = (i, j)
                    h[i] = 1.0
                    h[j] = 0.0
        else:
            n_trials = 10000
            for _ in range(n_trials):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = best_c5
    return h, c5_cur

def ga_binary(rng, n, pop_size, n_generations, time_fn, initial_pop=None):
    """Genetic algorithm on binary vectors with exactly n/2 ones"""
    ts = n // 2
    # Initialize population
    pop = []
    if initial_pop is not None:
        pop.append(initial_pop.copy())
    while len(pop) < pop_size:
        h = np.zeros(n)
        idx = rng.choice(n, size=ts, replace=False)
        h[idx] = 1.0
        pop.append(h)
    
    # Evaluate
    fitness = [compute_c5(p) for p in pop]
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    for gen in range(n_generations):
        if time_fn() < 10:
            break
        # Sort by fitness
        order = np.argsort(fitness)
        pop = [pop[i] for i in order]
        fitness = [fitness[i] for i in order]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = pop[0].copy()
        
        new_pop = [pop[0].copy()]  # Elitism
        
        while len(new_pop) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament = rng.choice(len(pop), size=5, replace=False)
                parents.append(pop[tournament[np.argmin([fitness[t] for t in tournament])]])
            
            # One-point crossover
            if rng.random() < 0.7:
                # Crossover that maintains sum constraint
                i1, i2 = parents
                cross_point = rng.randint(1, n - 1)
                child = np.zeros(n)
                # Take first part from i1, second from i2, then adjust
                child[:cross_point] = i1[:cross_point]
                child[cross_point:] = i2[cross_point:]
                # Adjust to have exactly ts ones
                curr = int(child.sum())
                if curr > ts:
                    excess = curr - ts
                    ones_idx = np.where(child > 0.5)[0]
                    rng.shuffle(ones_idx)
                    for k in range(excess):
                        child[ones_idx[k]] = 0.0
                elif curr < ts:
                    deficit = ts - curr
                    zeros_idx = np.where(child < 0.5)[0]
                    rng.shuffle(zeros_idx)
                    for k in range(deficit):
                        child[zeros_idx[k]] = 1.0
            else:
                child = parents[0].copy()
            
            # Mutation: flip pairs
            n_mut = rng.randint(1, max(2, n // 20) + 1)
            for _ in range(n_mut):
                ones_idx = np.where(child > 0.5)[0]
                zeros_idx = np.where(child < 0.5)[0]
                if len(ones_idx) > 0 and len(zeros_idx) > 0:
                    i = rng.choice(ones_idx)
                    j = rng.choice(zeros_idx)
                    child[i] = 0.0
                    child[j] = 1.0
            
            new_pop.append(child)
        
        pop = new_pop
        fitness = [compute_c5(p) for p in pop]
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - (m_val + np.max(shifted))
    return (m_val + np.max(corr - m_val) + np.log(np.sum(np.exp(alpha * (shifted)))) / alpha) * dx

def fw_direction(grad, n):
    indices = np.argsort(grad)
    y_star = np.zeros(n)
    y_star[indices[:n // 2]] = 1.0
    return y_star

def frank_wolfe_opt(h0, n, alpha, max_iter, time_fn):
    h = h0.copy()
    best_obj = smooth_max_obj(h, alpha)
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        y_star = fw_direction(grad, n)
        if np.allclose(h, y_star):
            break
        t = 1.0
        best_t = 0.0
        best_obj_t = obj
        for _ in range(12):
            h_try = (1.0 - t) * h + t * y_star
            obj_try = smooth_max_obj(h_try, alpha)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
            if obj_try <= obj - 1e-10:
                break
            t *= 0.5
        if best_t > 0:
            h = (1.0 - best_t) * h + best_t * y_star
            obj_h = smooth_max_obj(h, alpha)
            if obj_h < best_obj:
                best_obj = obj_h
                best_h = h.copy()
    return best_h, best_obj

def binary_local_search_exhaustive(h, n, time_fn, max_rounds=200):
    h = h.copy().astype(float)
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time_fn() < 5:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5_cur = best_c5
    return h, c5_cur

def ga_binary(rng, n, pop_size, n_generations, time_fn, initial_pop=None):
    ts = n // 2
    pop = []
    if initial_pop is not None:
        pop.append(initial_pop.copy().astype(float))
    while len(pop) < pop_size:

    best_c5 = fitness[best_idx]
    for gen in range(n_generations):

            break

        pop = [pop[i] for i in order]
        fitness = fitness[order]
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = pop[0].copy()
        new_pop = [pop[0].copy()]
        while len(new_pop) < pop_size:

                curr = int(child.sum())
                if curr > ts:
                    excess = curr - ts
                    ones_idx = np.where(child > 0.5)[0]
                    rng.shuffle(ones_idx)
                    for k in range(min(excess, len(ones_idx))):
                        child[ones_idx[k]] = 0.0
                elif curr < ts:

                    j = rng.choice(zeros_idx)
                    child[i] = 0.0
                    child[j] = 1.0
            new_pop.append(child)
        pop = new_pop
        fitness = np.array([compute_c5(p) for p in pop])
    return best_h, best_c5

def make_binary(h, n, rng):
    ts = n // 2
    h_bin = (h > 0.5).astype(float)
    curr = int(h_bin.sum())
    if curr > ts:
        ones = np.where(h_bin > 0.5)[0]
        rng.shuffle(ones)
        for i in range(curr - ts):
            if i < len(ones):
                h_bin[ones[i]] = 0.0
    elif curr < ts:
        zeros_idx = np.where(h_bin < 0.5)[0]
        rng.shuffle(zeros_idx)
        for i in range(ts - curr):
            if i < len(zeros_idx):
                h_bin[zeros_idx[i]] = 1.0
    return h_bin
