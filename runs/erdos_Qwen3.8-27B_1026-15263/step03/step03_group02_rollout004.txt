def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / len(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    S = np.exp(shifted).sum()
    return m + np.log(S) / alpha

def smooth_max_grad_fft(h, alpha):
    n = len(h)
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    S = exp_vals.sum()
    w = exp_vals / S
    obj = m + np.log(S) / alpha
    
    # FFT-based gradient
    F_len = 1
    conv_len = 3 * n - 2
    while F_len < conv_len:
        F_len *= 2
    
    f = 1.0 - h
    g = w
    conv1 = np.fft.irfft(np.fft.rfft(f, F_len) * np.fft.rfft(g, F_len), F_len)
    T1 = conv1[n-1:2*n-1]
    
    h_arr = h
    g_rev = w[::-1]
    conv2 = np.fft.irfft(np.fft.rfft(h_arr, F_len) * np.fft.rfft(g_rev, F_len), F_len)
    T2 = conv2[n-1:2*n-1]
    
    grad = T1 - T2
    return obj, grad

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter=500, time_limit=120):
    t0 = time.time()
    h = h0.copy()
    
    def objective(x):
        return smooth_max_obj(x, alpha)
    
    def gradient(x):
        return smooth_max_grad_fft(x, alpha)[1]
    
    # Use L-BFGS-B with penalty for the sum constraint
    penalty = 100.0
    def penalized_obj(x):
        base = smooth_max_obj(x, alpha)
        penalty_term = penalty * (np.sum(x) - target_sum)**2
        return base + penalty_term
    
    def penalized_grad(x):
        base_grad = smooth_max_grad_fft(x, alpha)[1]
        pen_grad = 2 * penalty * (np.sum(x) - target_sum) * np.ones(n)
        return base_grad + pen_grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized_obj, h.copy(), method='L-BFGS-B', jac=penalized_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        result = project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        result = project_to_sum(h.copy(), target_sum)
    
    return result

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng, verbose=False):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.01
    no_improve = 0
    min_T = 0.002
    best_T = T
    iters = 0
    
    while time.time() < t_end:
        iters += 1
        move_type = rng.randint(3)
        
        if move_type == 0:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif move_type == 1:
            # Block swap
            L = rng.randint(2, min(15, n//4))
            i = rng.randint(0, n - L)
            j = rng.randint(0, n - L)
            if abs(i - j) < L:
                continue
            h[i:i+L], h[j:j+L] = h[j:j+L].copy(), h[i:i+L].copy()
            # Ensure still binary with correct sum
            if h[i:i+L].sum() != h[j:j+L].sum() + h[i:i+L].sum():
                pass # block swap preserves sum
        else:
            # Flip one bit (compensate by flipping another)
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / max(T, 1e-10)) + np.log(rng.uniform()) < 0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                best_T = T
            else:
                no_improve += 1
        else:
            # Revert
            if move_type == 0:
                h[i], h[j] = h[j], h[i]
            elif move_type == 1:
                h[i:i+L], h[j:j+L] = h[j:j+L].copy(), h[i:i+L].copy()
            else:
                h[i] = 1.0 - h[i]
                h[j] = 1.0 - h[j]
        
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.02)
            no_improve = 0
        
        T = max(T * 0.99999, min_T)
    
    return best_h, best_c5
