        sa_end2 = t_start + 0.62 * budget_s
        h_sa2, c5_sa2 = sa_search(h_bin2, n_main, target_ones, sa_end2, rng, t_init=0.010)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        if remaining() > 120:
            h_grad2, c5_grad2 = grad_refine(best_h, n_main, target_ones, [500, 3000, 10000], maxiter_each=500)
            if c5_grad2 < best_c5:
                best_c5 = c5_grad2
                best_h = h_grad2.copy()

    for trial in range(4):
        if remaining() < 90:
            break
        noise_scale = 0.09 * (1.0 - trial / 4.0) + 0.02
        h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n_main)
        h_shake = project_to_sum(h_shake, target_ones)
        h_b = make_binary(h_shake, n_main, target_ones, rng)
        t_end = min(t_start + 0.80 * budget_s, time.time() + max(20, remaining() // 5))
        h_r, c5_r = sa_search(h_b, n_main, target_ones, t_end, rng, t_init=0.011)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        if remaining() > 60:
            h_gr, c5_gr = grad_refine(best_h, n_main, target_ones, [200, 1500], maxiter_each=300)
            if c5_gr < best_c5:
                best_c5 = c5_gr
                best_h = h_gr.copy()

    if remaining() > 50:
        h_fin, c5_fin = grad_refine(best_h, n_main, target_ones, [5000, 20000, 50000], maxiter_each=600)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    for n_try in [n_main + 12, n_main + 24, n_main + 36, n_main + 48]:
        if remaining() < 90:
            break
        if n_try % 2 != 0:
            continue
        t_ones = n_try // 2
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, t_ones)
        h_try = make_binary(h_interp, n_try, t_ones, rng)
        t_end = min(time.time() + max(20, remaining() // 4), t_start + 0.96 * budget_s)
        h_try2, c5_try = sa_search(h_try, n_try, t_ones, t_end, rng, t_init=0.010)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try2.copy()
            best_n = n_try
        if remaining() > 50:
            h_ig, c5_ig = grad_refine(h_interp, n_try, t_ones, [100, 800], maxiter_each=250)
            if c5_ig < best_c5:
                best_c5 = c5_ig
                best_h = h_ig.copy()
                best_n = n_try

    if remaining() > 25:
        h_last, c5_last = grad_refine(best_h, best_n, best_n // 2, [8000, 30000], maxiter_each=400)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
