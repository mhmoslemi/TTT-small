    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def accelerated_pg(h0, n, ts, alpha, n_iter=2000, lr_init=0.02, momentum=0.85):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    stall = 0
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = project_to_sum(h + v, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-15:
            h = h_new
            obj_prev = obj_new
            stall = 0
        else:
            lr *= 0.5
            v *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    return h

def informed_stochastic(h, n, ts, rng, n_steps=50000):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    temp = 0.04
    stagnant = 0
    for step in range(n_steps):
        corr = compute_corr(h)
        m_val = np.max(corr)
        c5_cur = m_val * 2.0 / n
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:]
        worst_idx = top_idx[-1]
        shift = worst_idx - (n - 1)
        if shift >= 0:
            contribs = h[:n-shift] * (1.0 - h[shift:])
            if np.max(contribs) < 1e-8:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = min(rng.uniform(0.02, 0.1), h[i] * 0.8, (1.0 - h[j]) * 0.8)
                if d < 1e-5:
                    continue
                h[i] -= d
                h[j] += d
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stagnant = 0
                else:
                    if rng.random() < np.exp(-(c5_new - best_c5) / temp):
                        h[np.argmin(c5_new - best_c5) % n]  # noop
                    else:
                        i2, j2 = i, j
                        h[i2] += d
                        h[j2] -= d
                        h = np.clip(h, 0.0, 1.0)
                        h = project_to_sum(h, ts)
                stagnant += 1
                temp = max(1e-7, temp * 0.999)
                continue
            i_max = np.argmax(contribs)
            i_dec = i_max
            i_inc = i_max + shift
        else:
            s = -shift
            contribs = h[s:] * (1.0 - h[:n-s])
            if np.max(contribs) < 1e-8:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = min(rng.uniform(0.02, 0.1), h[i] * 0.8, (1.0 - h[j]) * 0.8)
                if d < 1e-5:
                    continue
                h[i] -= d
                h[j] += d
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stagnant = 0
                else:
                    if rng.random() >= np.exp(-(c5_new - best_c5) / temp):
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        h = project_to_sum(h, ts)
                stagnant += 1
                temp = max(1e-7, temp * 0.999)
                continue
            i_max = np.argmax(contribs)
            i_dec = s + i_max
            i_inc = i_max
        d_base = rng.uniform(0.03, 0.15)
        d = min(d_base, h[i_dec] * 0.6, (1.0 - h[i_inc]) * 0.6)
        if d < 1e-5:
            continue
        h_try = h.copy()
        h_try[i_dec] -= d
        h_try[i_inc] += d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_to_sum(h_try, ts)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
            best_h = h_try.copy()
            stagnant = 0
            temp = max(1e-7, temp * 0.997)
        else:
            acc = np.exp(-(c5_try - best_c5) / temp)
            if rng.random() < acc:
                h = h_try
            stagnant += 1
            temp = max(1e-7, temp * 0.9995)
        if stagnant > 3000:
            h = best_h + rng.normal(0, 0.05, n)
            h = project_to_sum(h, ts)
            stagnant = 0
            temp = 0.04
    return best_h, best_c5

def coord_descent_polish(h, n, ts, n_rounds=5, deltas=None):
    if deltas is None:
        deltas = [0.001, 0.005, 0.01, 0.02, 0.05]
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                if h[i] + d > 1.0 or h[i] - d < 0.0:
                    continue
                h_try = h.copy()
                h_try[i] += d
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_to_sum(h_try, ts)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
                else:
                    h_try2 = h.copy()
                    h_try2[i] -= d
                    h_try2 = np.clip(h_try2, 0.0, 1.0)
                    h_try2 = project_to_sum(h_try2, ts)
                    c5_try2 = compute_c5(h_try2)
                    if c5_try2 < best_c5 - 1e-10:
                        h = h_try2
                        best_c5 = c5_try2
                        best_h = h_try2.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0

    # Phase 1: SLSQP continuation
    alphas1 = [20, 50, 100, 300, 1000, 3000, 10000]
    for alpha in alphas1:
        if time_left() < 40:
            break
        h_new = slsqp_opt(best_h, n1, ts1, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Accelerated projected gradient
    if time_left() > 60:
        h_pg = accelerated_pg(best_h, n1, ts1, 2000, n_iter=3000, lr_init=0.015, momentum=0.9)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Higher resolution attempts
    for n_tr in [512, 576]:
        if time_left() < 120:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts_tr)
        for alpha in [100, 500, 2000, 5000]:
            if time_left() < 50:
                break
            h_opt = slsqp_opt(h_up, n_tr, ts_tr, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_up = h_opt
        if time_left() > 40:
            h_pg2 = accelerated_pg(h_up, n_tr, ts_tr, 3000, n_iter=2000, lr_init=0.01, momentum=0.9)
            c5_pg2 = compute_c5(h_pg2)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
                best_n = n_tr

    # Phase 4: Informed stochastic descent (main new component)
    if time_left() > 200:
        n_cd = best_n
        ts_cd = n_cd / 2.0
        h_cd = best_h.copy()
        n_steps = min(100000, int(time_left() * 2))
        h_cd, c5_cd = informed_stochastic(h_cd, n_cd, ts_cd, rng, n_steps=n_steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: SLSQP polish after stochastic
    if time_left() > 60:
        n_p = best_n
        ts_p = n_p / 2.0
        for alpha in [2000, 10000, 30000]:
            if time_left() < 30:
                break
            h_pol = slsqp_opt(best_h, n_p, ts_p, alpha, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Coordinate descent polish
    if time_left() > 30:
        n_cd2 = best_n
        ts_cd2 = n_cd2 / 2.0
        h_cd2, c5_cd2 = coord_descent_polish(best_h, n_cd2, ts_cd2, n_rounds=3,
                                              deltas=[0.001, 0.003, 0.01, 0.03])
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 7: Second informed stochastic round
    if time_left() > 120:
        n_s2 = best_n
        ts_s2 = n_s2 / 2.0
        rng2 = np.random.RandomState(seed + 12345)
        h_s2, c5_s2 = informed_stochastic(best_h, n_s2, ts_s2, rng2, n_steps=min(50000, int(time_left())))
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    # Phase 8: Final SLSQP + coord descent
    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = slsqp_opt(best_h, n_f, ts_f, 50000, maxiter=2000)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if time_left() > 20:
        n_f2 = best_n
        ts_f2 = n_f2 / 2.0
        h_f2, c5_f2 = coord_descent_polish(best_h, n_f2, ts_f2, n_rounds=4,
                                            deltas=[0.0005, 0.001, 0.003, 0.01])
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Phase 9: Frequency perturbation + re-optimize (few trials)
    if time_left() > 80:
        n_b = best_n
        ts_b = n_b / 2.0
        for trial in range(20):
            if time_left() < 30:
                break
            amp = 0.03 * (1.0 - trial / 25.0) + 0.005
            h_sh = freq_perturb(best_h, rng, max(3, n_b // 30), amp)
            c5_sh = compute_c5(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh.copy()
            h_ot = slsqp_opt(h_sh, n_b, ts_b, 1000, maxiter=400)
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
