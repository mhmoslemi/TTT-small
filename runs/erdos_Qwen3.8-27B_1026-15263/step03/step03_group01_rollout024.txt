            if c5_new < best_c5:
                best_c5 = c5_new
                h_best = h.copy()
        else:
            accept = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-5, 0.15 * np.exp(-step / (n_steps * 0.4)))
    return h_best, best_c5

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    nj_arr = np.arange(n)[:, None]
    nm_arr = np.arange(2 * n - 1)[None, :]
    fi_arr = nj_arr - nm_arr + n - 1
    si_arr = nj_arr + nm_arr - n + 1
    vf_arr = (fi_arr >= 0) & (fi_arr < n)
    vs_arr = (si_arr >= 0) & (si_arr < n)
    fc_arr = np.clip(fi_arr, 0, n - 1)
    sc_arr = np.clip(si_arr, 0, n - 1)

    def f(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        max_sh = np.max(shifted)
        exp_v = np.exp(shifted - max_sh)
        return m_val + (max_sh + np.log(np.sum(exp_v))) / alpha

    def g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        max_sh = np.max(shifted)
        exp_v = np.exp(shifted - max_sh)
        w = exp_v / np.sum(exp_v)
        grad = np.sum(vf_arr * (1.0 - x[fc_arr]) * w[None, :] - vs_arr * x[sc_arr] * w[None, :], axis=1)
        return grad

    def c_fun(x):
        return np.sum(x) - ts

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def make_diverse_starts(n, ts, rng, best_h, best_n, n_starts=8):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for f in [2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.4)
        s = 0.5 + amp * np.sin(2.0 * np.pi * f * x / 2.0 + phase)
        starts.append(project_to_sum(s, ts))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            h_up = upsample_h(best_h, best_n, n)
            starts.append(h_up)
    for _ in range(3):
        h_mix = 0.5 * np.ones(n)
        nf = rng.randint(3, 8)
        freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
        for f in freqs:
            ph = rng.uniform(0, 2 * np.pi)
            am = rng.uniform(0.05, 0.2)
            h_mix += am * np.sin(2.0 * np.pi * f * x / 2.0 + ph)
        starts.append(project_to_sum(h_mix, ts))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts_init = best_n / 2.0
            best_h = project_to_sum(best_h, ts_init)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Upsample to 960 and optimize
    n_target = 960
    ts_target = n_target / 2.0
    h_work = upsample_h(best_h, best_n, n_target)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
        best_n = n_target

    # Phase 2: Adam optimization with alpha continuation at n=960
    alphas = [20, 80, 300, 1000, 4000, 15000, 50000]
    n_adam_per_alpha = [30000, 30000, 25000, 20000, 15000, 10000, 8000]
    for idx_a, alpha in enumerate(alphas):
        if time_left() < 60:
            break
        n_iter_here = n_adam_per_alpha[idx_a]
        h_work = adam_projected(h_work, n_target, ts_target, alpha, n_iter_here, lr_init=0.03)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_target

    # Phase 3: Subgradient steps targeting worst lag
    if time_left() > 100:
        h_sub = subgrad_steps(h_work, n_target, ts_target, n_iter=20000, lr=0.05)
        if h_sub[1] < best_c5:
            best_c5 = h_sub[1]
            best_h = h_sub[0].copy()
            best_n = n_target
        h_work = h_sub[0]

    # Phase 4: SLSQP polish at high alpha
    if time_left() > 80:
        for alpha in [5000, 20000]:
            if time_left() < 40:
                break
            h_sls = slsqp_opt(h_work, n_target, ts_target, alpha, maxiter=1000)
            c5_sls = compute_c5(h_sls)
            if c5_sls < best_c5:
                best_c5 = c5_sls
                best_h = h_sls.copy()
                best_n = n_target
            h_work = h_sls

    # Phase 5: Stochastic coordinate descent on exact objective (large deltas)
    if time_left() > 120:
        h_cd, c5_cd = stochastic_cd_exact(h_work, n_target, ts_target, rng,
                                          n_steps=80000, delta_lo=0.02, delta_hi=0.3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_target
        h_work = h_cd

    # Phase 6: More Adam at very high alpha after stochastic perturbation
    if time_left() > 100:
        for alpha in [10000, 50000]:
            if time_left() < 40:
                break
            h_work = adam_projected(h_work, n_target, ts_target, alpha, 15000, lr_init=0.02)
            c5_work = compute_c5(h_work)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
                best_n = n_target

    # Phase 7: Stochastic CD with small deltas (finishing)
    if time_left() > 80:
        h_cd2, c5_cd2 = stochastic_cd_exact(h_work, n_target, ts_target, rng,
                                            n_steps=60000, delta_lo=0.005, delta_hi=0.12)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n_target
        h_work = h_cd2

    # Phase 8: Subgradient fine-tuning
    if time_left() > 60:
        h_sub2, c5_sub2 = subgrad_steps(h_work, n_target, ts_target, n_iter=15000, lr=0.02)
        if c5_sub2 < best_c5:
            best_c5 = c5_sub2
            best_h = h_sub2[0].copy()
            best_n = n_target
        h_work = h_sub2[0]

    # Phase 9: Frequency-space perturbations + re-optimize
    if time_left() > 80:
        for trial in range(15):
            if time_left() < 30:
                break
            amp = 0.04 * (1.0 - trial / 20.0) + 0.01
            nf_harm = max(5, n_target // 30)
            h_pert = freq_perturb(h_work, rng, nf_harm, amp)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n_target
            h_work = adam_projected(h_pert, n_target, ts_target, 10000, 8000, lr_init=0.02)
            c5_work = compute_c5(h_work)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
                best_n = n_target

    # Phase 10: Diverse restarts at 960
    if time_left() > 100:
        starts = make_diverse_starts(n_target, ts_target, rng, best_h, best_n, n_starts=6)
        for h_s in starts:
            if time_left() < 25:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_target
            h_s = adam_projected(h_s, n_target, ts_target, 500, 10000, lr_init=0.04)
            c5_s2 = compute_c5(h_s)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s.copy()
                best_n = n_target
            if time_left() > 20:
                h_s_final, c5_final = stochastic_cd_exact(h_s, n_target, ts_target, rng,
                                                          n_steps=20000, delta_lo=0.01, delta_hi=0.15)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_s_final.copy()
                    best_n = n_target

    # Phase 11: Final Adam polish
    if time_left() > 50:
        h_work = adam_projected(best_h, best_n, best_n / 2.0, 50000, 15000, lr_init=0.01)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_target

    # Phase 12: Final tiny stochastic refinement
    if time_left() > 40:
        h_t, c5_t = stochastic_cd_exact(best_h, best_n, best_n / 2.0, rng,
                                        n_steps=30000, delta_lo=0.003, delta_hi=0.08)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_target

    # Phase 13: SLSQP final polish
    if time_left() > 30:
        h_f = slsqp_opt(best_h, best_n, best_n / 2.0, 50000, maxiter=1500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_target

    # Final validation and constraint enforcement
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
