        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    threshold = max(1e-12, 1e-10 * np.max(w))
    active = np.where(w > threshold)[0]
    sh = np.zeros(3 * n)
    sh[n:2 * n] = h
    grad = np.zeros(n)
    for idx in active:
        s_shift = idx - (n - 1)
        h_plus = sh[n + s_shift:n + s_shift + n]
        h_minus = sh[n - s_shift:n - s_shift + n]
        grad += w[idx] * (1.0 - h_plus - h_minus)
    grad *= dx
    return sm_val, grad

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_start, budget_s, reserve):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    stalled = 0
    for it in range(n_iters):
        if time.time() - t_start > budget_s - reserve:
            break
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled % 100 == 0:
                lr *= 0.9
        h = h_new
    return best_h, best_c5

def topk_subgrad(h0, n, target_sum, n_iters, lr_init, top_k, t_start, budget_s, reserve):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() - t_start > budget_s - reserve:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        w_top = np.zeros(2 * n - 1)
        w_top[top_idx] = 1.0 / float(top_k)
        sh = np.zeros(3 * n)
        sh[n:2 * n] = h
        g = np.zeros(n)
        for s_idx in range(2 * n - 1):
            wk = w_top[s_idx]
            if wk < 1e-14:
                continue
            s_shift = s_idx - (n - 1)
            h_plus = sh[n + s_shift:n + s_shift + n]
            h_minus = sh[n - s_shift:n - s_shift + n]
            g += wk * (1.0 - h_plus - h_minus)
        lr = lr_init / (1.0 + 0.015 * it)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta, t_start, budget_s, reserve):
    h = h0.copy()
    c5_cur = compute_c5(h)
    scale = base_delta
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - reserve:
            break
        improved = False
        deltas = np.array([-3.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0]) * scale
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            scale *= 0.5
            if scale < 1e-6:
                break
    return h, c5_cur

def slsqp_refine(h0, n, target_sum, alpha, maxiter, t_start, budget_s, reserve):
    n_local = n
    ts_local = target_sum
    al_local = alpha
    def _c_fun(x):
        return np.sum(x) - ts_local
    def _c_jac(x):
        return np.ones(n_local)
    def _obj(x):
        hh = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(hh, al_local)
        return v
    def _grad(x):
        hh = np.clip(x, 0.0, 1.0)
        _, gg = smooth_max_and_grad(hh, al_local)
        return gg
    bounds = [(0.0, 1.0)] * n_local
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts0 = best_n / 2.0
            best_h = project_to_sum(best_h, ts0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    alphas_phase1 = [30.0, 80.0, 200.0, 500.0]
    for alpha in alphas_phase1:
        if time.time() - t_start > budget_s - 350:
            break
        n_iters = min(4000, max(500, int(60 / 0.015)))
        h_new, c5_new = adam_optimize(best_h, n, ts, alpha, n_iters, 0.12, t_start, budget_s, 350)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    for alpha in [300.0, 800.0, 2000.0]:
        if time.time() - t_start > budget_s - 200:
            break
        h_slsqp = slsqp_refine(best_h, n, ts, alpha, 300, t_start, budget_s, 200)
        c5_s = compute_c5(h_slsqp)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_slsqp

    if time.time() - t_start < budget_s - 150:
        h_new, c5_new = topk_subgrad(best_h, n, ts, 3000, 0.06, 15, t_start, budget_s, 150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() - t_start < budget_s - 100:
        h_new, c5_new = adam_optimize(best_h, n, ts, 1000.0, 2000, 0.05, t_start, budget_s, 100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    h_new, c5_new = coord_descent(best_h, n, ts, 80, rng, 0.01, t_start, budget_s, 60)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new

    if time.time() - t_start < budget_s - 40:
        h_new, c5_new = coord_descent(best_h, n, ts, 120, rng, 0.004, t_start, budget_s, 40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() - t_start < budget_s - 25:
        h_new, c5_new = coord_descent(best_h, n, ts, 200, rng, 0.001, t_start, budget_s, 25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() - t_start < budget_s - 15:
        h_new, c5_new = coord_descent(best_h, n, ts, 400, rng, 0.0003, t_start, budget_s, 15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() - t_start < budget_s - 8:
        h_new, c5_new = coord_descent(best_h, n, ts, 600, rng, 0.0001, t_start, budget_s, 8)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() - t_start < budget_s - 5:
        h_new, c5_new = coord_descent(best_h, n, ts, 800, rng, 0.00005, t_start, budget_s, 5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = project_to_sum(best_h, ts)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
