def weighted_overlap(h, lags, weights):
    """Compute sum_k w_k * overlap[k]"""
    total = 0.0
    for idx in range(len(lags)):
        k = abs(lags[idx])
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        if k == 0:
            total += w * np.sum(h * (1-h))
        else:
            total += w * np.sum(h[k:] * (1-h[:len(h)-k]))
    return total

def weighted_overlap_grad(h, lags, weights):
    """Gradient of weighted overlap"""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(lags)):
        k = abs(lags[idx])
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        if k == 0:
            grad += w * (1 - 2*h)
        else:
            # ∂overlap[k]/∂h[j] = 1[j≥k] - (B_k h)[j]
            # = 1[j≥k] - h[j-k]*1[j≥k] - h[j+k]*1[j≤n-1-k]
            grad[k:] += w * (1 - h[:n-k])  # 1[j≥k] - h[j-k] for j≥k
            grad[:n-k] += w * (-h[k:])  # -h[j+k] for j≤n-1-k
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / float(len(h))
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_overlap(h, lags, weights):
    total = 0.0
    for idx in range(len(lags)):
        k = abs(lags[idx])
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        n = len(h)
        if k == 0:
            total += w * float(np.sum(h * (1.0 - h)))
        else:

    return total

def weighted_overlap_grad(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)

        w = weights[idx]

            grad += w * (1.0 - 2.0 * h)
        else:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] += w * (-h[k:])
    return grad

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = weighted_overlap(h, lags, weights)
    velocity = np.zeros(n)
    alpha = 0.5

    for it in range(max_iters):

        velocity = momentum * velocity - alpha * grad
        h_trial = h + velocity
        h_trial = np.clip(h_trial, 0.0, 1.0)
