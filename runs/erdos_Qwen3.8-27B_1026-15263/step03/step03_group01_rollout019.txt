def weighted_grad(h, w_pos, w_neg):
    """
    w_pos[τ] = weight for positive lag τ (τ = 0, 1, ..., n-1)
    w_neg[m] = weight for negative lag -m (m = 1, 2, ..., n-1)
    Returns the gradient sum.
    """
    n = len(h)
    grad = np.zeros(n)
    
    # Positive lags τ = 0, ..., n-1
    # Contribution: (1-h[j-τ]) for j >= τ, minus h[j+τ] for j <= n-1-τ
    for τ in range(n):
        wt = w_pos[τ]
        if wt < 1e-15:
            continue
        if τ == 0:
            grad += wt * (1.0 - 2.0 * h)
        else:
            # (1-h[j-τ]) for j in [τ, n-1]
            grad[τ:] += wt * (1.0 - h[:n-τ])
            # -h[j+τ] for j in [0, n-1-τ]
            grad[:n-τ] -= wt * h[τ:]
    
    # Negative lags -m, m = 1, ..., n-1
    # Contribution: (1-h[j+m]) for j in [0, n-1-m], minus h[j-m] for j in [m, n-1]
    for m in range(1, n):
        wt = w_neg[m]
        if wt < 1e-15:

        # (1-h[j+m]) for j in [0, n-1-m]
        grad[:n-m] += wt * (1.0 - h[m:])
        # -h[j-m] for j in [m, n-1]
        grad[m:] -= wt * h[:n-m]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_lag_grad(h, w):
    """Compute sum_k w[k] * grad of corr_k w.r.t. h.
    w is an array of length 2n-1 (the weights for each lag in the full correlation).
    """
    n = len(h)
    n_full = 2 * n - 1
    grad = np.zeros(n)
    center = n - 1
    
    for k in range(n_full):
        wt = w[k]
        if wt < 1e-15:
            continue
        L = k - center  # lag offset
        if L >= 0:
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            else:
                grad[L:] += wt * (1.0 - h[:n - L])
                grad[:n - L] -= wt * h[L:]
        else:
            m = -L
            grad[:n - m] += wt * (1.0 - h[m:])
            grad[m:] -= wt * h[:n - m]
    
    return grad

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    grad = weighted_lag_grad(h, w)
    return obj, grad

def topk_grad(h, top_k=10):
    """Compute the average gradient of the top-k correlation values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find top-k indices
    idx = np.argpartition(corr, -top_k)[-top_k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / len(idx)
    obj = np.mean(corr[idx])
    grad = weighted_lag_grad(h, w)
    return obj, grad, idx

def proj_grad_step(h, ts, n, alpha, n_iter=1000, lr_init=0.01, lr_end=1e-6):
    """Projected gradient with smooth max."""
    h = h.copy()
    for i in range(n_iter):
        t = i / max(n_iter - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** t
        _, grad = smooth_max_obj_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        # Check if improvement
        obj_new = smooth_max_obj_grad(h_new, alpha)[0]
        obj_old = smooth_max_obj_grad(h, alpha)[0]
        if obj_new > obj_old + 1e-15:
            # Step was too large, reduce
            lr *= 0.5
            h_new = project_to_sum(h - lr * grad, ts)
        h = h_new
    return h

def slsqp_smooth(h0, n, ts, alpha, maxiter=2000):
    """SLSQP optimization with smooth max."""
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def fprime(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=fprime,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def topk_opt(h0, n, ts, top_k_init=20, n_iter=2000, lr_init=0.005):
    """Optimize using top-k subgradient with decreasing k."""
    h = h0.copy()
    for i in range(n_iter):
        # Decrease top_k over time
        frac = i / max(n_iter - 1, 1)
        top_k = max(1, int(top_k_init * (1.0 - frac * 0.9)))
        # Compute top-k gradient
        _, grad, _ = topk_grad(h, top_k)
        # Adaptive step size
        lr = lr_init / (1.0 + i * 0.001)
        h_new = project_to_sum(h - lr * grad, ts)
        # Accept if C5 improves
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_old:
            h = h_new
        else:
            # Try smaller step
            h_new2 = project_to_sum(h - lr * 0.1 * grad, ts)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 <= c5_old:
                h = h_new2
    return h

def make_structured_perturb(h, rng, n, mode):

    x = np.linspace(0, 2, n, endpoint=False)
    h_new = h.copy()
    if mode == 0:
        # Adding a single frequency
        f = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        h_new += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    elif mode == 1:
        # Adding two frequencies
        f1 = rng.randint(1, n // 4)
        f2 = rng.randint(1, n // 4)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.03, 0.15)
        a2 = rng.uniform(0.03, 0.15)
        h_new += a1 * np.sin(2 * np.pi * f1 * x / 2.0 + p1)
        h_new += a2 * np.sin(2 * np.pi * f2 * x / 2.0 + p2)
    elif mode == 2:
        # Frequency shift (modulate)
        f_shift = rng.randint(1, 5)
        h_new *= np.cos(np.pi * f_shift * x / 2.0) ** 2
        h_new += (1 - np.cos(np.pi * f_shift * x / 2.0) ** 2) * 0.5
    elif mode == 3:
        # Local bump
        center = rng.uniform(0, 2)
        width = rng.uniform(0.1, 0.5)
        amp = rng.uniform(-0.3, 0.3)
        bump = amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        h_new += bump
    elif mode == 4:
        # Block flip (swap a block's values with its complement)
        blk = n // rng.randint(4, 12)
        start = rng.randint(0, n - blk)
        h_new[start:start+blk] = 1.0 - h_new[start:start+blk]
    else:
        # Small random noise
        h_new += rng.uniform(-0.05, 0.05, n)
    return project_to_sum(h_new, n / 2.0)
