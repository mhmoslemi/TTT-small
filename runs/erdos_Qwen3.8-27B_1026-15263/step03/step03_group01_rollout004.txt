        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_m:]
        # Gradient of c_k for each active k
        # c_k = sum_i h[i+k](1-h[i])
        # For the full output, index p corresponds to lag k = p - (n-1)
        grad = np.zeros(n)
        for p in top_idx:
            k = p - (n - 1)  # lag
            # ∂c_k/∂h[j] = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)
            nj = np.arange(n)
            v1 = (nj - k >= 0) & (nj - k < n)
            v2 = (nj + k >= 0) & (nj + k < n)
            g1 = np.where(v1, 1.0 - h[nj - k], 0.0)
            g2 = np.where(v2, h[nj + k], 0.0)
            grad += g1 - g2
        grad /= top_m
        # Projected gradient step
        h_new = project_to_sum(h - lr * grad, ts)

        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, best_c5
