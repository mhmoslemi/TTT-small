def ipm_opt(h0, n, ts, alpha, maxiter=3000):

    Uses the barrier method: minimize smooth_max + barrier for [0,1] constraints.
    The sum constraint is handled by parameterization.
    """
    # Parameterize: h[i] = exp(z[i]) / (1 + exp(z[i]))  (sigmoid)
    # Then sum(h) = ts is one constraint on z
    # This is complex; let's stick with projection
    
    h = h0.copy()
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(maxiter):
        obj = smooth_max_obj(h, alpha)
        grad = smooth_max_grad(h, alpha)
        # Project gradient onto the constraint hyperplane
        grad -= np.mean(grad) * np.ones(n)  # project to sum-preserving subspace
        # Add barrier for [0,1]
        mu = 0.1
        barrier_grad = mu / (h + 1e-8) - mu / (1.0 - h + 1e-8)

def block_sa(h, n, ts, rng, n_steps=30000, block_sizes=(2,4,8,16)):
    """Simulated annealing with block moves."""
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.02
    best_local = c5_cur
    best_h_local = h.copy()
    
    for step in range(n_steps):
        bs = block_sizes[rng.randint(len(block_sizes))]
        i = rng.randint(0, n - bs + 1)
        block = h[i:i+bs].copy()
        # Perturb the block
        delta = 0.05 * rng.randn(bs)
        block += delta
        block = np.clip(block, 0, 1)
        # Preserve sum
        old_sum = np.sum(h[i:i+bs])
        new_sum = np.sum(block)
        if new_sum > 0:
            block = block * old_sum / new_sum
            block = np.clip(block, 0, 1)
        h[i:i+bs] = block
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, ts)
        
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_local:
                best_local = c5_new
                best_h_local = h.copy()
        else:
            # Accept with probability
            if temp > 1e-6 and rng.random() < np.exp(-(c5_new - c5_cur) / temp):
                c5_cur = c5_new
            else:
                # Revert
                h[i:i+bs] = block - delta
                h = np.clip(h, 0, 1)
                h = project_to_sum(h, ts)
        
        temp = max(1e-7, 0.02 * np.exp(-step / (n_steps / 5.0)))
    
    return best_h_local, best_local

def structured_perturb(h, rng, n):

        # Phase shift

        h_new = np.interp(x - shift, x, h, period=2.0)
    elif choice == 1:
        # Stretch/compress
        factor = rng.uniform(0.9, 1.1)
        x_new = x * factor
        h_new = np.interp(x_new % 2.0, x, h, period=2.0)
        # Hmm, this doesn't preserve structure well. Let's do:
        x_new = np.linspace(0, 2*factor, n, endpoint=False) % 2.0
        x_new = np.sort(x_new)
        h_new = np.interp(x_new, x, h, period=2.0)
    elif choice == 2:
        # Frequency mix
        H = np.fft.rfft(h)
        for k in range(1, min(20, n//2)):
            if rng.random() < 0.3:
                H[k] *= rng.uniform(0.7, 1.3)
        h_new = np.fft.irfft(H, n=n)
    elif choice == 3:
        # Local smooth perturbation
        kernel = np.exp(-0.5 * (np.arange(n) / rng.uniform(5, 30))**2)
        h_new = h + rng.uniform(0.03, 0.1) * kernel * rng.randn()
    else:
        # Mirror + blend
        h_mirror = h[::-1]
        h_new = rng.uniform(0.3, 0.7) * h + rng.uniform(0.3, 0.7) * h_mirror
    
    return project_to_sum(h_new, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_grad_c5(h):
    """If there's a unique max lag, return its exact gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    
    # corr[lag] = sum_i h[i+lag]*(1-h[i]) for valid i
    # d/dh[j] = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]

    if lag >= 0:
        for i in range(n - lag):
            j = i + lag
            grad[j] += (1.0 - h[i])
            grad[i] -= h[j]
    else:
        for i in range(n + lag):
            j = i
            grad[i] += -(h[i - lag])  # h[j+lag] = h[i+lag] where lag<0... 
            # Wait, let me redo

            #   from h[i+lag] term: j = i+lag, so i = j-lag, coeff = (1-h[i]) = (1-h[j-lag])
            #   from (1-h[i]) term: j = i, coeff = -h[i+lag] = -h[j+lag]

        if 0 <= j - lag < n:
            grad[j] += (1.0 - h[j - lag])
        # Term 2: -h[j+lag] if 0 <= j+lag < n
        if 0 <= j + lag < n:
            grad[j] -= h[j + lag]
    return grad, k_max

def smooth_c5(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exps)) / alpha

def smooth_c5_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / np.sum(exps)
    
    # grad[j] = sum_k w[k] * [(1-h[j-lag_k])*valid - h[j+lag_k]*valid]
    # where lag_k = k - (n-1)
    # Vectorized:
    nj = np.arange(n)[:, None]
    k_idx = np.arange(2*n-1)[None, :]
    lags = k_idx - (n-1)
    
    fi = nj - lags  # index j - lag = j - k + (n-1)
    si = nj + lags  # index j + lag = j + k - (n-1)
    
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    
    grad = np.sum(vf * (1.0 - h[fc]) * w[None,:] - vs * h[sc] * w[None,:], axis=1)
    return grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return smooth_c5(x, alpha)
    def g(x):
        return smooth_c5_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_nesterov(h0, n, ts, alpha, n_iter=5000, lr_init=0.02):
    """Nesterov accelerated projected gradient descent."""
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    t = 1.0
    obj_prev = smooth_c5(h, alpha)
    
    for it in range(n_iter):
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        
        h_y = h + beta * v
        grad = smooth_c5_grad(h_y, alpha)
        # Project gradient to preserve sum
        grad -= np.mean(grad)
        
        v_new = (h - lr * grad) - h
        h_new = project_to_sum(h_y + v_new, ts)  # = project_to_sum(h - lr*grad, ts)
        
        obj_new = smooth_c5(h_new, alpha)
        if obj_new < obj_prev:
            h = h_new
            v = h - h_y  # velocity
            obj_prev = obj_new
            t = t_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h

        old_block = h[i:i+bs].copy()
        
        # Perturb
        amp = rng.uniform(0.02, 0.15)

        h = project_to_sum(h, ts)
        
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new

        else:

                c5_cur = c5_new
            else:
                h[i:i+bs] = old_block
                h = project_to_sum(h, ts)
        
        stall += 1
        if stall > 5000:
            temp *= 0.5
            stall = 0

def pair_transfer(h, n, ts, rng, n_steps=20000, delta_range=(0.005, 0.2)):
    """Pair mass transfer moves (similar to previous stochastic_cd but refined)."""
    h = h.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    temp = 0.01
    
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if rng.random() < 0.5:
            # Move from j to i
            if h[j] < delta:
                delta = h[j] * 0.5
            if 1.0 - h[i] < delta:
                delta = (1.0 - h[i]) * 0.5
            if delta < 1e-5:
                continue
            h[j] -= delta
            h[i] += delta
        else:
            # Move from i to j
            if h[i] < delta:
                delta = h[i] * 0.5
            if 1.0 - h[j] < delta:
                delta = (1.0 - h[j]) * 0.5
            if delta < 1e-5:
                continue
            h[i] -= delta
            h[j] += delta
        
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            if temp > 1e-8:
                prob = np.exp(-(c5_new - c5_cur) / temp)
                if rng.random() < prob:
                    c5_cur = c5_new
                else:
                    # Revert
                    h = best_h.copy() if False else h  # revert manually
                    # Actually we need to undo the move
                    if rng.random() < 0.5:
                        # was move j->i
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    h = np.clip(h, 0, 1)
                    h = project_to_sum(h, ts)
        
        temp = max(1e-8, 0.01 * np.exp(-step / max(1, n_steps/10.0)))
    
    return best_h, best_c5
