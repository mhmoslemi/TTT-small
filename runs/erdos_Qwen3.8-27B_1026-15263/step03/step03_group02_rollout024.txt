import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary_from_frac(h, n, target_sum, rng):
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:target_sum]] = 1.0
    return result

def make_binary_multi(h, n, target_sum, rng):
    best_h = None
    best_c5 = float('inf')
    order = np.argsort(-h)
    for _ in range(3):
        candidate = np.zeros(n)
        candidate[order[:target_sum]] = 1.0
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
        for _ in range(20):
            ones = np.where(candidate == 1.0)[0]
            zeros = np.where(candidate == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidate[i], candidate[j] = candidate[j], candidate[i]
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
    return best_h

def sa_random(h, n, target_sum, t_end, rng):
    h = h.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.01
    T_min = 0.002
    no_improve = 0
    while time.time() < t_end:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / T + np.log(rng.uniform()) < 0.0):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = 0.015
            no_improve = 0
        T = max(T * 0.999997, T_min)
    return best_h, best_c5

def compute_soft_grad(h, corr, alpha):
    n = len(h)
    n1 = 2 * n - 1
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf, axis=1) - np.sum(cs, axis=1)
    return grad

def sa_grad_guided(h, n, target_sum, t_end, rng):
    h = h.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.006
    T_min = 0.001
    no_improve = 0
    grad_refresh = 80
    step_count = 0
    alpha = 80.0
    corr = compute_corr(h)
    grad = compute_soft_grad(h, corr, alpha)
    while time.time() < t_end:
        step_count += 1
        if step_count % grad_refresh == 0:
            corr = compute_corr(h)
            alpha = min(alpha * 1.05, 500.0)
            grad = compute_soft_grad(h, corr, alpha)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ones_scores = grad[ones]
        zeros_scores = -grad[zeros]
        if rng.random() < 0.6:
            n_top = min(10, len(ones), len(zeros))
            top_o = np.argsort(-ones_scores)[:n_top]
            top_z = np.argsort(-zeros_scores)[:n_top]
            i = ones[top_o[rng.randint(n_top)]]
            j = zeros[top_z[rng.randint(n_top)]]
        elif rng.random() < 0.3:
            i = ones[np.argmax(ones_scores)]
            j = zeros[np.argmax(zeros_scores)]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / T + np.log(rng.uniform()) < 0.0):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve > 40000:
            h = best_h.copy()
            c5_curr = best_c5
            T = 0.008
            no_improve = 0
            corr = compute_corr(h)
            alpha = 80.0
            grad = compute_soft_grad(h, corr, alpha)
        T = max(T * 0.999998, T_min)
    return best_h, best_c5

def slsqp_smooth_max(h, n, target_sum, alpha, maxiter):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f_obj(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        corr = compute_corr(h_vec)
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        return m_val + np.log(np.sum(exp_vals)) / alpha
    def f_grad(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        corr = compute_corr(h_vec)
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        n1 = 2 * n - 1
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(n1)[None, :]
        first_idx = j_idx - m_idx + n - 1
        second_idx = j_idx + m_idx - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        cf = valid_f * (1.0 - h_vec[fc]) * w[None, :]
        cs = valid_s * h_vec[sc] * w[None, :]
        grad = np.sum(cf, axis=1) - np.sum(cs, axis=1)
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_obj, h0.copy(), method='SLSQP', jac=f_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def local_search_binary(h, n, target_sum, max_rounds=50):
    h = h.copy()
    c5_curr = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        corr = compute_corr(h)
        alpha = 100.0
        grad = compute_soft_grad(h, corr, alpha)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        ones_sorted = ones[np.argsort(-grad[ones])]
        zeros_sorted = zeros[np.argsort(grad[zeros])]
        found = False
        for a in range(min(15, len(ones_sorted))):
            if found:
                break
            for b in range(min(15, len(zeros_sorted))):
                i = ones_sorted[a]
                j = zeros_sorted[b]
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < c5_curr - 1e-12:
                    c5_curr = c5_new
                    improved = True
                    found = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, c5_curr

    def elapsed():
