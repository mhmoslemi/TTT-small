The terms involving h[i] are:
- k=i: h[i]·(1-h[i-lag_j]) → changes by δ·(1-h[i-lag_j])
- k-lag_j=i, i.e., k=i+lag_j: h[i+lag_j]·(1-h[i]) → changes by -h[i+lag_j]·δ

So Δcorr[j] = δ·[(1-h[i-lag_j]) - h[i+lag_j]] (with appropriate validity checks)

This means I can update the correlation in O(n) per coordinate change (one per lag), rather than O(n²) for a full recompute.

from scipy.optimize import minimize
import time

_g_target_sum = 0.0
_g_n_pts = 0
_g_alpha = 1.0

def _c_fun(x):
    return np.sum(x) - _g_target_sum

def _c_jac(x):
    return np.ones(_g_n_pts)

def _obj_slsqp(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _g_alpha)
    return v

def _grad_slsqp(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _g_alpha)
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_active = mask.sum()
        if n_active > 0:
            h[mask] += diff / n_active
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)

    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for j in active:
        wj = w[j]
        lag = j - (n - 1)
        k1_lo = max(0, lag)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):

    _g_target_sum = target_sum
    _g_n_pts = n
    _g_alpha = alpha

    try:

                       bounds=bounds, constraints=constraints,

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):

    v_vec = np.zeros(n)

    best_h = h.copy()
    lr = lr_init

    beta2 = 0.999

    for _ in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_cand

    for n in [360, 480]:
        if _remaining(t_start, budget_s) < 150:
            break
        ts = n / 2.0

        if best_n == n and best_h is not None:
            starts = [best_h.copy()]
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts = [project_to_sum(h_interp, ts)]

        starts.append(np.full(n, 0.5))
        for _ in range(3):
            starts.append(fourier_init(n, 8, rng))
        for freq in [2, 3, 4, 5]:
            for amp in [0.3, 0.45]:
                starts.append(sine_init(n, freq, 0.0, amp, rng))

        for h_start in starts:
            if _remaining(t_start, budget_s) < 80:
                break
            h_start = project_to_sum(h_start, ts)
            try_update(h_start, n)
            h_cur = h_start.copy()

            alphas = [10.0, 30.0, 80.0, 200.0, 500.0, 1500.0]
            for alpha in alphas:
                if _remaining(t_start, budget_s) < 40:
                    break
                maxiter = max(50, min(400, int(_remaining(t_start, budget_s) * 0.4)))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                try_update(h_opt, n)
                h_cur = h_opt

            if _remaining(t_start, budget_s) > 60:
                n_adam = min(2000, max(400, int(_remaining(t_start, budget_s) * 1.2)))
                h_a, _ = adam_optimize(h_cur, n, ts, 300.0, n_adam, 0.06)
                try_update(h_a, n)
                h_cur = h_a

            if _remaining(t_start, budget_s) > 45:
                n_sub = min(4000, max(800, int(_remaining(t_start, budget_s) * 1.5)))
                h_sg, _ = subgradient_optimize(h_cur, n, ts, n_sub, 0.04, 5)
                try_update(h_sg, n)
                if _remaining(t_start, budget_s) > 30:
                    h_sg2, _ = subgradient_optimize(h_sg, n, ts, n_sub // 2, 0.02, 15)
                    try_update(h_sg2, n)
                    h_cur = h_sg2
                else:
                    h_cur = h_sg

        if _remaining(t_start, budget_s) < 30:
            break

        if best_n == n:
            h_cd, _ = coord_descent(best_h, n, ts, 60, rng, 0.005, 60.0)
            try_update(h_cd, n)
            if _remaining(t_start, budget_s) > 20:
                h_cd2, _ = coord_descent(best_h, n, ts, 80, rng, 0.001, 40.0)
                try_update(h_cd2, n)

    if best_h is not None and _remaining(t_start, budget_s) > 15:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [2000.0, 5000.0]:
            if _remaining(t_start, budget_s) < 10:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 300)
            try_update(h_pol, n_f)

        if _remaining(t_start, budget_s) > 20:
            h_cd, _ = coord_descent(best_h, n_f, ts_f, 150, rng, 0.0005, 45.0)
            try_update(h_cd, n_f)

        if _remaining(t_start, budget_s) > 10:
            h_cd2, _ = coord_descent(best_h, n_f, ts_f, 100, rng, 0.0001, 30.0)
            try_update(h_cd2, n_f)

        if _remaining(t_start, budget_s) > 8:
            h_cd3, _ = coord_descent(best_h, n_f, ts_f, 50, rng, 0.00003, 15.0)
            try_update(h_cd3, n_f)

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
