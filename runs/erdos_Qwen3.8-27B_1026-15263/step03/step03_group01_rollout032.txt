def lbfgs_opt(h0, n, ts, alpha, maxiter=2000):
    h = h0.copy()
    
    def f(x):
        x_clip = np.clip(x, 0.0, 1.0)
        return smooth_max_and_grad(x_clip, alpha)[0]
    
    def g(x):
        x_clip = np.clip(x, 0.0, 1.0)
        _, grad = smooth_max_and_grad(x_clip, alpha)
        return grad
    
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxiter * 2})
        result = project_to_sum(r.x, ts)
        return result
    except Exception:
        return project_to_sum(h, ts)

def projected_gradient(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_and_grad(h, alpha)
    best_h = h.copy()
    best_obj = obj_prev
    stall = 0
    
    for i in range(n_iter):
        _, grad = smooth_max_and_grad(h, alpha)
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        obj_new, _ = smooth_max_and_grad(h_new, alpha)
        
        if obj_new < obj_prev:
            h = h_new
            obj_prev = obj_new
            stall = 0
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_h = h.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    
    return best_h

def block_cd(h, n, ts, rng, block_sizes=(4, 8, 16), n_steps=20000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for step in range(n_steps):
        bs = rng.choice(block_sizes)
        start = rng.randint(0, n - bs)
        block = h[start:start+bs].copy()
        
        # Random perturbation of the block
        perturbation = rng.uniform(-0.3, 0.3, size=bs)
        block_new = np.clip(block + perturbation, 0.0, 1.0)
        
        # Project the block to maintain the sum
        # We need sum(block_new) = sum(block) to maintain total sum
        target_block_sum = np.sum(block)
        diff = target_block_sum - np.sum(block_new)
        mask = (block_new > 1e-12) & (block_new < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            block_new[mask] += diff / np.sum(mask)
            block_new = np.clip(block_new, 0.0, 1.0)
            # Fix any remaining difference
            for _ in range(50):
                d = target_block_sum - np.sum(block_new)
                if abs(d) < 1e-12:
                    break
                block_new[mask] += d / max(1, np.sum(mask))
                block_new = np.clip(block_new, 0.0, 1.0)
        
        h[start:start+bs] = block_new
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif c5_new < c5_cur:
            c5_cur = c5_new
        else:
            # Revert
            h[start:start+bs] = block
    
    return best_h

def make_fur_start(n, ts, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    # Start with a few frequency components
    h = np.zeros(n)
    n_components = rng.randint(3, 8)
    freqs = np.sort(rng.choice(np.arange(1, n//4 + 1), size=n_components, replace=False))
    for f in freqs:
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2.0 * np.pi * f * x / 2.0 + phase)
    h += 0.5
    return project_to_sum(h, ts)

def make_block_start(n, ts, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(4, 12)
    block_size = n // n_blocks
    for i in range(0, n, block_size):
        h[i:min(i+block_size, n)] = rng.uniform(0.0, 1.0)
    return project_to_sum(h, ts)

def make_grad_start(n, ts, rng, best_h, best_n):
    if best_h is not None and best_n > 0:
        if best_n == n:
            h = best_h + rng.normal(0, 0.1, n)
        else:
            x = np.linspace(0, 2, n, endpoint=False)
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            h = np.interp(x, x_old, best_h) + rng.normal(0, 0.05, n)
    else:
        h = np.full(n, 0.5) + rng.normal(0, 0.1, n)
    return project_to_sum(h, ts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(mask)
        if n_free > 0:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def lbfgs_opt(h0, n, ts, alpha, maxiter=2000):
    def f(x):
        xc = np.clip(x, 0.0, 1.0)
        return smooth_max_and_grad(xc, alpha)[0]
    
    def g(x):
        xc = np.clip(x, 0.0, 1.0)
        _, grad = smooth_max_and_grad(xc, alpha)
        return grad
    
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0, method='L-BFGS-B', jac=g, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-11})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_and_grad(h, alpha)
    best_h = h.copy()
    best_obj = obj_prev
    stall = 0
    for i in range(n_iter):
        _, grad = smooth_max_and_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_and_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.05, 0.3)
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
        else:
            lr *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    return best_h

def block_moves(h, n, ts, rng, n_steps=20000, block_range=(2, 12)):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        bs = rng.randint(block_range[0], block_range[1] + 1)
        start = rng.randint(0, max(1, n - bs))
        block = h[start:start+bs].copy()
        noise = rng.uniform(-0.2, 0.2, bs)
        block_new = np.clip(block + noise, 0.0, 1.0)
        # Preserve block sum
        target = np.sum(block)
        for _ in range(100):
            d = target - np.sum(block_new)
            if abs(d) < 1e-12:
                break
            mask = (block_new > 1e-10) & (block_new < 1.0 - 1e-10)
            if np.sum(mask) > 0:
                block_new[mask] += d / np.sum(mask)
                block_new = np.clip(block_new, 0.0, 1.0)
            else:
                break
        h[start:start+bs] = block_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            c5_cur = c5_new
        else:
            h[start:start+bs] = block
    return best_h
