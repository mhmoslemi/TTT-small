            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)

    s = np.sum(exp_vals)

        lo = max(0, l)
        hi = min(n, n + l)

        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha, K=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if K is not None and K < len(corr):
        top_idx = np.argsort(corr)[-K:]
    else:
        top_idx = np.arange(len(corr))
    sub_corr = corr[top_idx]

    shifted = np.clip(shifted, -700.0, 700.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_blocks):
            start = rng.randint(0, n - block_size + 1)
            direction = rng.randint(0, 1)  # 0 = spread out, 1 = spread in
            # Create a perturbation pattern
            if direction == 0:
                # Make block more binary
                center = np.mean(h[start:start+block_size])
                delta = 0.1 * rng.random()
                pert = np.zeros(block_size)
                pert[0] = -delta
                pert[-1] = delta
                if block_size > 2:
                    pert[1:-1] = 0
            else:

                pert = rng.uniform(-0.05, 0.05, block_size)
                pert -= np.sum(pert) / block_size  # keep sum local

                best_h = h_new.copy()
                h = h_new.copy()
                improved = True

    return best_h, best_c5

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        return project_to_sum(h0, target_sum)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, K=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if K is not None and K < len(corr):
        top_idx = np.argsort(corr)[-K:]
    else:
        top_idx = np.arange(len(corr))
    sub_corr = corr[top_idx]
    m = np.max(sub_corr)
    shifted = alpha * (sub_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        wl = w[idx]
        if wl < 1e-15:
            continue
        l = top_idx[idx] - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
