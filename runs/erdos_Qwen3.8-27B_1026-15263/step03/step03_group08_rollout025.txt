import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1000.0
_g_n = 480
_g_target = 240.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nm = np.sum(mask)
        if nm > 0:
            h[mask] += diff / nm
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def _obj_f(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _g_alpha)
    diff = np.sum(h) - _g_target
    return v + 5000.0 * diff * diff

def _grad_f(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _g_alpha)
    diff = np.sum(h) - _g_target
    return g + 10000.0 * diff * np.ones(_g_n)

def _sum_eq(x):
    return np.sum(x) - _g_target

def _sum_jac(x):
    return np.ones(_g_n)

def lbfgs_run(h0, n, target_sum, alpha, maxiter):
    global _g_alpha, _g_n, _g_target
    _g_alpha = alpha
    _g_n = n
    _g_target = target_sum
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_f, h0.copy(), jac=_grad_f, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slsqp_run(h0, n, target_sum, alpha, maxiter):
    global _g_alpha, _g_n, _g_target
    _g_alpha = alpha
    _g_n = n
    _g_target = target_sum

    def _o(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, _g_alpha)
        return v

    def _g(x):
        h = np.clip(x, 0.0, 1.0)
        _, gv = smooth_max_and_grad(h, _g_alpha)
        return gv

    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_o, h0.copy(), jac=_g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_local_search(h0, n, target_sum, max_swaps, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    ones = np.where(h > 0.99)[0]
    zeros = np.where(h < 0.01)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c5_cur
    swaps_done = 0
    while swaps_done < max_swaps:
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                if h[i] < 0.99 or h[j] > 0.01:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-14:
                    c5_cur = c5_new
                    improved = True
                    swaps_done += 1
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                if swaps_done >= max_swaps:
                    break
            if swaps_done >= max_swaps:
                break
        if not improved:
            break
    return h, c5_cur

def snap_and_optimize(h0, n, target_sum, threshold, alpha, maxiter, time_left_fn):
    h = h0.copy()
    for i in range(n):
        if h[i] > 1.0 - threshold:
            h[i] = 1.0
        elif h[i] < threshold:
            h[i] = 0.0
    h = project_to_sum(h, target_sum)
    c5_snap = compute_c5(h)
    if time_left_fn() < 30:
        return h, c5_snap
    h_opt = lbfgs_run(h, n, target_sum, alpha, maxiter)
    c5_opt = compute_c5(h_opt)
    return h_opt, min(c5_snap, c5_opt)

def coordinate_descent_fine(h0, n, target_sum, max_rounds, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            for d in step_sizes:
                nv = h[i] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    best_val = nv
            if best_c5 < c5_cur:
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def perturb_h(h, delta, rng):
    hp = h + rng.uniform(-delta, delta, len(h))
    return project_to_sum(hp, len(h) / 2.0)

def sine_starts(n, ts):
    x = np.linspace(0, 2, n, endpoint=False)

def binary_starts(n, ts, rng, k):

        idx = rng.choice(n, size=n // 2, replace=False)
        hb = np.zeros(n)
