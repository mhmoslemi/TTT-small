def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def corr_gradient_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
    else:
        kk = -k
        if kk == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[kk:] += (1.0 - h[:n-kk])
            g[:n-kk] -= h[kk:]
    return g

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

            best_c5 = c5_final
            best_h = h_cd_final.copy()

    if time_left() > 30:
        h_pen_final, c5_pen_final = penalty_refinement(best_h, ts_best, max_iters=3000)
        if c5_pen_final < best_c5:
            best_c5 = c5_pen_final
            best_h = h_pen_final.copy()
        h_cd2, c5_cd2 = coordinate_descent_multi(best_h, ts_best, max_iters=15000, step_frac=0.03, top_k=5)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
