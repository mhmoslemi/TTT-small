from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    """Compute gradient of f_lag with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # f_lag = sum_{t=0}^{n-1-lag} h[t+lag] * (1-h[t])
        # d/dh[i] = (1-h[i-lag]) for i >= lag, minus h[i+lag] for i < n-lag
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        j = -lag
        # f_{-j} = sum_{t=0}^{n-1-j} h[t] * (1-h[t+j])
        # d/dh[i] = (1-h[i+j]) for i < n-j, minus h[i-j] for i >= j
        g[:n-j] += (1.0 - h[j:])
        g[j:] -= h[:n-j]
    return g

def subgradient_step(h, target_sum, alpha, top_m=5, max_iter=5000):
    """Active-set subgradient method with adaptive step size."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = 0.1
    for it in range(max_iter):
        corr = compute_corr(h)
        m = np.max(corr)
        # Find top M lags
        top_idx = np.argsort(corr)[-top_m:]
        # Compute weighted gradient (weights proportional to closeness to max)
        weights = np.exp(20 * (corr[top_idx] - m) / max(m, 1e-10))
        weights /= np.sum(weights)
        g = np.zeros(n)
        for w, idx in zip(weights, top_idx):

            g += w * lag_gradient(h, lag)
        # Projected gradient step

        h_new = project_to_sum(h_new, target_sum)

        if c5_new < c5:

            c5 = c5_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h, c5

def coordinate_descent_v2(h, target_sum, max_iters=10000, step_frac=0.15):
    """Improved coordinate descent with pair moves and top-lag focus."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m = np.max(corr)
        top_idx = np.argsort(corr)[-3:]
        # Compute combined gradient
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += lag_gradient(h, lag)
        g /= len(top_idx)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        # Find best pair
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))

        if g[i_inc] >= g[i_dec] - 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        j = -lag
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    return g

def subgradient_optimize(h, target_sum, n_iters, top_m, base_step, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.9
    step = base_step
    for it in range(n_iters):
        corr = compute_corr(h)
        m_val = np.max(corr)
        top_idx = np.argsort(corr)[-top_m:]
        weights = np.exp(15.0 * (corr[top_idx] - m_val) / max(m_val, 1e-10))
        weights /= np.sum(weights)
        g = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx) - (n - 1)
            g += w * lag_gradient(h, lag)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - step * momentum
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            step = min(step * 1.01, 1.0)
        else:
            step *= 0.85
            if step < 1e-8:
                break
    return h, c5

def coord_descent(h, target_sum, max_iters, step_frac, top_m=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    sf = step_frac
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_val = np.max(corr)
        top_idx = np.argsort(corr)[-top_m:]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += lag_gradient(h, lag)
        g /= top_m
        can_inc = h < 1.0 - 1e-11
        can_dec = h > 1e-11
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e18)
        g_dec = np.where(can_dec, g, -1e18)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-13:
            if sf < 0.005:
                break
            sf *= 0.5
            continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = sf * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
            sf = min(sf * 1.02, 0.3)
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 80:
                sf *= 0.5
                stale = 0
                if sf < 0.003:
                    break
    return h, c5

def swap_search(h, target_sum, max_rounds, n_swaps_per_round, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for rd in range(max_rounds):
        improved = False
        for _ in range(n_swaps_per_round):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or abs(h[i] - h[j]) < 1e-10:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-13:
                c5 = c5_new
                improved = True
            else:
                h[i], h[j] = old_i, old_j
        if not improved:
            stale += 1
            if stale > 15:
                break
        else:
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def slsqp_refine(h, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x):
        corr = compute_corr(x)
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        return m_val + np.log(np.sum(exp_vals)) / alpha

    def smooth_grad(x):
        corr = compute_corr(x)
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        n_local = len(x)
        g = np.zeros(n_local)
        for k in range(2 * n_local - 1):
            lag = k - (n_local - 1)
            g += w[k] * lag_gradient(x, lag)
        return g

    try:
        res = minimize(smooth_obj, h.copy(), method='SLSQP', jac=smooth_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h.copy(), target_sum)

                best_h = h_interp.copy()
                best_n = n_try
                n_cur = n_try
                ts = ts_try

    if time_left() > 30:
        h_final, c5_final = coord_descent(best_h, ts, max_iters=15000,
                                          step_frac=0.04, top_m=3)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if time_left() > 20:
        h_sub2, c5_sub2 = subgradient_optimize(best_h, ts, 3000, top_m=3,
                                               base_step=0.01, rng=rng)
        if c5_sub2 < best_c5:
            best_c5 = c5_sub2
            best_h = h_sub2.copy()

    if time_left() > 15:
        h_cd2, c5_cd2 = coord_descent(best_h, ts, max_iters=10000,
                                      step_frac=0.02, top_m=2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    return (best_h, best_c5, best_n)
