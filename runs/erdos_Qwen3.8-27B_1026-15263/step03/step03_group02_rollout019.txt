def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_obj(h, alpha, target_sum)
    def g(h):
        return smooth_max_grad(h, alpha)
    def cfun(h):
        return np.sum(h) - target_sum
    def cjac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        result = project_to_sum(res.x, target_sum)
        return result
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

OK so the convention is: out[k] = sum_m h[m] * (1-h[m-k]) for valid m.

The "lag" k (actual shift) goes from -(n-1) to n-1, stored at index k+(n-1).

For the max to be small, we want for all k: sum_m h[m] * (1-h[m-k]) to be small.

This means: for all shifts k, the number of positions where h[m]=1 and h[m-k]=0 should be small.

Equivalently: for all k, h should be "similar" to h shifted by k (where 1s overlap with 1s and 0s overlap with 0s).

This is exactly the autocorrelation condition! We want h to have low "cross-correlation" with its complement at all shifts.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    T = 4.0
    min_T = 0.5
    no_improve = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_corr))
        delta = new_max - c5_curr
        if delta < 0 or np.log(rng.uniform()) < -delta / T:
            c5_curr = new_max
            corr = new_corr
            if c5_curr < best_c5 - 1e-12:
                best_c5 = c5_curr
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
        if no_improve > 60000:
            h = best_h.copy()
            c5_curr = best_c5
            corr = np.correlate(h, 1.0 - h, mode="full")
            T = min(T * 2.5, 10.0)
            no_improve = 0
        T = max(T * 0.999997, min_T)
    return best_h, best_c5

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    m = k_idx - (n - 1)
    jm = j_idx - m
    valid1 = (jm >= 0) & (jm < n)
    jm_clipped = np.clip(jm, 0, n - 1)
    term1 = valid1 * (1.0 - h[jm_clipped])
    jp = j_idx + m
    valid2 = (jp >= 0) & (jp < n)
    jp_clipped = np.clip(jp, 0, n - 1)
    term2 = valid2 * h[jp_clipped]
    grad = np.sum(w[None, :] * (term1 - term2), axis=1)
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_obj(h, alpha)
    def g(h):
        return smooth_max_grad(h, alpha)
    def cfun(h):
        return np.sum(h) - target_sum
    def cjac(h):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def local_continuous_descent(h, n, target_sum, max_iter=200, step=0.02):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    n_corr = 2 * n - 1
    for it in range(max_iter):
        improved = False
        max_k = int(np.argmax(corr))
        for i in range(n):
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
                actual_delta = h_new[i] - h[i]
                if abs(actual_delta) < 1e-10:
                    continue
                j = (i + 1) % n
                attempts = 0
                while attempts < 20:
                    if h_new[j] + actual_delta >= 0.0 and h_new[j] + actual_delta <= 1.0:
                        h_new[j] += actual_delta
                        if abs(np.sum(h_new) - target_sum) < 1e-10:
                            break
                    j = (j + 1) % n
                    attempts += 1
                else:
                    h_proj = project_to_sum(h_new, target_sum)
                    if np.allclose(h_proj, h_new, atol=1e-6):
                        h_new = h_proj
                    else:
                        continue
                new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
                new_max = float(np.max(new_corr))
                if new_max < best_c5 / dx - 1e-10:
                    h = h_new.copy()
                    corr = new_corr
                    best_c5 = float(np.max(new_corr) * dx)
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:

            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n

    h_bin = make_binary(best_h, n_main, int(ts_main), rng)

        best_h = h_bin.copy()

    # Phase 1: Binary SA - main run
    sa_end1 = t_start + 0.30 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h_current = h_sa.copy()

    # Phase 2: Multi-start binary SA with perturbations
    for restart in range(4):
        if elapsed() > 0.45 * budget_s:
            break
        n_flip = rng.randint(1, 8)
        h_pert = h_current.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_flip):
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = ones[rng.randint(len(ones))]
            zi = zeros[rng.randint(len(zeros))]
            h_pert[oi], h_pert[zi] = h_pert[zi], h_pert[oi]
        sa_end_r = min(t_start + 0.45 * budget_s, t_start + elapsed() + 0.03 * budget_s)
