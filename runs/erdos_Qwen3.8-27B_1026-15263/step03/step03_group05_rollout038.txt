    g_lo = np.max(A)
    g_hi = np.max(A + B)

    best_val = min(g_lo, g_hi)

    B_sorted = B[order]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):

        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid

            #   contribution(t) = t*(1-h[i-s])
            #   difference = (t - old_t)*(1-h[i-s])
            #   coeff of t: (1-h[i-s])

            #   contribution(t) = t*(1-t)
            #   difference = t*(1-t) - old_t*(1-old_t)
            #   = t - t² - old_t + old_t²
            #   This is quadratic!

            # For each t in t_grid, compute max_j corr[j](t)
            # corr[j](t) = corr[j](old_t) + delta_j(t)

            # For efficiency, precompute:
            # linear_coeff[j] = coefficient of (t - old_t) in corr[j](t) - corr[j](old_t)
            # quadratic_term[j] = coefficient of (t-t² - old_t + old_t²) [only for j=n-1]
            
            # Actually, for j=n-1: corr[n-1](t) = corr[n-1](old_t) + (t - t² - old_t + old_t²)
            # For all other j: corr[j](t) = corr[j](old_t) + linear_coeff[j] * (t - old_t)
            
            linear_coeff = np.zeros(2 * n - 1)
            for j in range(2 * n - 1):
                s_val = s_arr[j]
                if s_val == 0:
                    continue  # handle separately
                c1 = cond1[j]
                c2 = cond2[j]
                if c1 and c2:
                    linear_coeff[j] = 1.0 - h[i + s_val] - h[i - s_val]
                elif c1:
                    linear_coeff[j] = -h[i + s_val]
                elif c2:
                    linear_coeff[j] = 1.0 - h[i - s_val]
                # else: 0 (h[i] doesn't contribute to this lag)
            
            # Now evaluate at each t:
            # corr[j](t) = corr[j] + linear_coeff[j]*(t - old_t) for j ≠ n-1
            # corr[n-1](t) = corr[n-1] + (t - t² - old_t + old_t²)
            
            # Vectorized: for each t, compute all corr[j](t) and take max
            # This is O(n_t * n) per coordinate
            
            for t in t_grid:
                if t == old_t:
                    continue
                # Linear part
                vals = corr + linear_coeff * (t - old_t)
                # Quadratic part for j = n-1
                vals[n-1] = corr[n-1] + (t - t*t - old_t + old_t*old_t)
                obj = np.max(vals)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            if best_t != old_t:
                h[i] = best_t
                h = project_box_sum(h, ts)
                # Update corr
                corr = np.correlate(h, 1.0 - h, mode="full")
                improved = True
                new_c5 = float(np.max(corr) * (2.0/n))
                if new_c5 < c5_cur:
                    c5_cur = new_c5
    
    h = project_box_sum(h, ts)
    c5_cur = compute_c5(h)
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def projected_gradient_descent(h0, n, alpha, max_iter, lr_init, time_fn):
    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        else:
            lr = min(lr * 1.1, 0.5)
        h = h_new
    return best_h, best_obj

def coord_descent(h_in, n, max_passes=100, n_t=100, time_fn=None):
    h = h_in.copy()
    ts = n / 2.0
    t_grid = np.linspace(0.0, 1.0, n_t)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_cur = float(np.max(corr) * (2.0 / n))
    s_arr = np.arange(2 * n - 1) - (n - 1)
    
    for p in range(max_passes):
        if time_fn is not None and time_fn() < 15:
            break
        any_change = False
        for i in range(n):
            old_t = h[i]
            
            # Compute linear coefficients
            sv = s_arr
            idx1 = i + sv
            idx2 = i - sv
            c1 = (idx1 >= 0) & (idx1 < n)
            c2 = (idx2 >= 0) & (idx2 < n)
            both = c1 & c2
            only1 = c1 & ~c2
            only2 = c2 & ~c1
            
            linear_coeff = np.zeros(2 * n - 1)
            if np.any(both):
                linear_coeff[both] = 1.0 - h[idx1[both]] - h[idx2[both]]
            if np.any(only1):
                linear_coeff[only1] = -h[idx1[only1]]
            if np.any(only2):
                linear_coeff[only2] = 1.0 - h[idx2[only2]]
            
            # Evaluate at all t_grid points
            dt = t_grid - old_t
            vals = corr[np.newaxis, :] + linear_coeff[np.newaxis, :] * dt[:, np.newaxis]
            quad = t_grid - t_grid * t_grid - old_t + old_t * old_t
            vals[:, n - 1] = corr[n - 1] + quad
            obj = np.max(vals, axis=1)

            best_t = t_grid[best_idx]

    c5_cur = compute_c5(h)
    return h, c5_cur

def simulated_annealing(h_in, n, n_iter=5000, T_init=0.05, T_final=0.001, rng=None, time_fn=None):
    h = h_in.copy()
    ts = n / 2.0
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time_fn is not None and time_fn() < 10:
            break
        T = T_init * (T_final / T_init) ** (it / n_iter)
        
        # Random move: change one or two coordinates
        if rng.rand() < 0.5:
            # Single coordinate change
            i = rng.randint(n)
            old = h[i]
            h[i] = rng.uniform(0, 1)
            h = project_box_sum(h, ts)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(corr) * dx)
            delta = c5_new - c5_cur
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i] = old
                h = project_box_sum(h, ts)
        else:
            # Pair swap
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
            h = project_box_sum(h, ts)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(corr) * dx)
            delta = c5_new - c5_cur
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = old_i, old_j
                h = project_box_sum(h, ts)
    
    return best_h, best_c5

        return budget_s - (time.time() - t_start)

    best_c5 = float('inf')

            best_c5 = compute_c5(best_h)
        except Exception:
            pass

        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    for target_n in [best_n, 480]:
        if time_left() < 200:
            break
        if target_n == best_n:
            h_work = best_h.copy()
        else:

            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, target_n, endpoint=False)
            h_work = np.interp(x_new, x_old, best_h, period=2.0)
            h_work = project_box_sum(h_work, target_n / 2.0)

            best_h = h_cd.copy()
            best_n = n_w

        if time_left() > 60:
            for alpha in [100, 500, 2000]:
                if time_left() < 30:
                    break
                h_pg, _ = projected_gradient_descent(best_h, best_n, alpha, 2000, 0.05, time_left)
                c5_pg = compute_c5(h_pg)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
        
        # Coordinate descent again after gradient
        if time_left() > 60:
            h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_passes=100, n_t=100, time_fn=time_left)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
    
    # Phase 2: Simulated annealing + coordinate descent restarts
    if time_left() > 100:
        for trial in range(10):
            if time_left() < 60:
                break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def projected_gradient_descent(h0, n, alpha, max_iter, lr_init, time_fn):
    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        else:
            lr = min(lr * 1.1, 0.5)
        h = h_new
    return best_h, best_obj

def coord_descent(h_in, n, max_passes=100, n_t=100, time_fn=None):
    h = h_in.copy()
    ts = n / 2.0
    t_grid = np.linspace(0.0, 1.0, n_t)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_cur = float(np.max(corr) * (2.0 / n))
    s_arr = np.arange(2 * n - 1) - (n - 1)
    
    for p in range(max_passes):
        if time_fn is not None and time_fn() < 15:
            break
        any_change = False
        for i in range(n):
            old_t = h[i]
            sv = s_arr
            idx1 = i + sv
            idx2 = i - sv
            c1 = (idx1 >= 0) & (idx1 < n)
            c2 = (idx2 >= 0) & (idx2 < n)
            both = c1 & c2
            only1 = c1 & ~c2
            only2 = c2 & ~c1
            linear_coeff = np.zeros(2 * n - 1)
            if np.any(both):
                linear_coeff[both] = 1.0 - h[idx1[both]] - h[idx2[both]]
            if np.any(only1):
                linear_coeff[only1] = -h[idx1[only1]]
            if np.any(only2):
                linear_coeff[only2] = 1.0 - h[idx2[only2]]
            dt = t_grid - old_t
            vals = corr[np.newaxis, :] + linear_coeff[np.newaxis, :] * dt[:, np.newaxis]
            quad = t_grid - t_grid * t_grid - old_t + old_t * old_t
            vals[:, n - 1] = corr[n - 1] + quad
            obj = np.max(vals, axis=1)
            best_idx = np.argmin(obj)
            best_t = t_grid[best_idx]
            if abs(best_t - old_t) > 1e-10:
                h[i] = best_t
                h = project_box_sum(h, ts)
                corr = np.correlate(h, 1.0 - h, mode="full")
                any_change = True
                new_c5 = float(np.max(corr) * (2.0 / n))
                if new_c5 < c5_cur:
                    c5_cur = new_c5
        if not any_change and p > 2:
            break
    h = project_box_sum(h, ts)
    c5_cur = compute_c5(h)
    return h, c5_cur

def simulated_annealing(h_in, n, n_iter=3000, T_init=0.04, T_final=0.001, rng=None, time_fn=None):

    best_c5 = c5_cur
    dx = 2.0 / n
    for it in range(n_iter):
        if time_fn is not None and time_fn() < 10:
            break
        T = T_init * (T_final / T_init) ** (it / float(n_iter))
        if rng.rand() < 0.4:
            i = rng.randint(n)
            old = h[i]
            h[i] = rng.uniform(0, 1)
            h = project_box_sum(h, ts)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(corr) * dx)
            delta = c5_new - c5_cur
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i] = old
                h = project_box_sum(h, ts)
                corr = np.correlate(h, 1.0 - h, mode="full")
        elif rng.rand() < 0.8:
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
            h = project_box_sum(h, ts)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(corr) * dx)
            delta = c5_new - c5_cur
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = old_i, old_j
                h = project_box_sum(h, ts)
                corr = np.correlate(h, 1.0 - h, mode="full")
        else:
            i = rng.randint(n)
            j = (i + rng.randint(1, n // 4 + 1)) % n
            old_i, old_j = h[i], h[j]
            delta_move = rng.uniform(0.1, 0.5)
            h[i] = np.clip(h[i] + delta_move, 0, 1)
            h[j] = np.clip(h[j] - delta_move, 0, 1)
            h = project_box_sum(h, ts)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(corr) * dx)
            delta = c5_new - c5_cur
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = old_i, old_j
                h = project_box_sum(h, ts)
                corr = np.correlate(h, 1.0 - h, mode="full")
    return best_h, best_c5
