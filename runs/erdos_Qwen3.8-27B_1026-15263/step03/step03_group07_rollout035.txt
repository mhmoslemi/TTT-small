def active_lag_gradient(h, n, top_m=5):
    """Compute gradient using only the top-m most active lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for m_idx in top_indices:
        k = m_idx - (n - 1)
        # d(corr[m_idx])/d(h[j]) for each j
        if k >= 0:
            for j in range(n):
                d = 0.0
                if j <= n - 1 - k:
                    d += (1.0 - h[j + k])
                if k <= j:
                    d -= h[j - k]
                g[j] += d
        else:
            p = -k
            for j in range(n):
                d = 0.0
                if j >= p:
                    d += (1.0 - h[j - p])
                if j <= n - 1 - p:
                    d -= h[j + p]
                g[j] += d
    return g / top_m, corr

def active_lag_gradient_vec(h, n, top_m=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for m_idx in top_indices:
        k = m_idx - (n - 1)
        if k >= 0:
            # First term: (1-h[j+k]) for j=0..n-1-k
            if n - 1 - k >= 0:
                g[0:n-k] += (1.0 - h[k:n])
            # Second term: -h[j-k] for j=k..n-1
            if k < n:
                g[k:n] -= h[0:n-k]
        else:
            p = -k
            # First term: (1-h[j-p]) for j=p..n-1
            g[p:n] += (1.0 - h[0:n-p])
            # Second term: -h[j+p] for j=0..n-1-p
            g[0:n-p] -= h[p:n]
    return g / top_m, corr

    s = np.sum(exp_vals)
    w = exp_vals / s
    return w

def smooth_max_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = get_smooth_max_weights(corr, alpha)
    
    w_pos = w[n-1:]
    w_neg = w[n-1::-1]
    w0 = w[n-1]
    omh = 1.0 - h
    
    S1 = np.correlate(omh, w_pos, 'full')[0:n]
    S2 = -np.convolve(w_pos, h, 'full')[0:n]
    S3 = np.convolve(w_neg, omh, 'full')[0:n] - w0 * omh
    S4 = -np.correlate(h, w_neg, 'full')[0:n] + w0 * h
    
    return S1 + S2 + S3 + S4

def active_lag_gradient(h, top_m=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for m_idx in top_indices:
        k = m_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                g[0:n-k] += (1.0 - h[k:n])
            if k < n and n - k > 0:
                g[k:n] -= h[0:n-k]
        else:
            p = -k
            if n - p > 0:
                g[p:n] += (1.0 - h[0:n-p])
                g[0:n-p] -= h[p:n]
    return g / top_m

def projected_gradient_smooth(h0, n, target_sum, alpha, max_iters, lr_init):
    h = h0.copy()
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    lr = lr_init
    y = h0.copy()
    mu = 0.0
    
    for it in range(max_iters):
        g = smooth_max_gradient(y, alpha)
        # Nesterov step
        h_cand = y - lr * g
        h_cand = project_to_sum(h_cand, target_sum)
        c5_cand = compute_c5(h_cand)
        
        # Update Nesterov momentum
        mu_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
        y = h_cand + ((mu - 1.0) / mu_new) * (h_cand - h)
        y = project_to_sum(y, target_sum)
        mu = mu_new
        
        if c5_cand < c5_best:
            c5_best = c5_cand
            h_best = h_cand.copy()
            h = h_cand
            lr = min(lr * 1.01, 0.5)
        else:
            h = h_cand  # still move forward for momentum
            lr *= 0.8
            if lr < 1e-8:
                lr = lr_init * 0.1
    
    return h_best, c5_best

def projected_gradient_active(h0, n, target_sum, top_m, max_iters, lr_init):
    h = h0.copy()
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    lr = lr_init
    
    for it in range(max_iters):
        g = active_lag_gradient(h, top_m)
        h_cand = h - lr * g
        h_cand = project_to_sum(h_cand, target_sum)
        c5_cand = compute_c5(h_cand)
        
        if c5_cand < c5_best:
            c5_best = c5_cand
            h_best = h_cand.copy()
            h = h_cand
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return h_best, c5_best

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_size=0.01):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            # Try positive and negative deltas
            for d in [-step_size, -step_size/2, -step_size/5, step_size/5, step_size/2, step_size]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

def binary_push(h0, n, target_sum, strength, n_iter):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for it in range(n_iter):
        # Push values toward 0 or 1
        h_push = h + strength * np.sign(h - 0.5)
        h_push = project_to_sum(np.clip(h_push, 0.0, 1.0), target_sum)
        c5_new = compute_c5(h_push)
        if c5_new < c5_cur:
            h = h_push
            c5_cur = c5_new
            strength *= 0.9
        else:
            strength *= 0.7
            if strength < 1e-5:
                break
    return h, c5_cur
