    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    """Compute smooth max and gradient w.r.t. h."""
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    return obj, w

def smooth_max_h_obj_grad(h, alpha):
    """Objective and gradient for h with smooth max of corr * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m + (max_sh + np.log(np.sum(exp_v))) / alpha) * dx
    
    # Gradient: d/dh_i [sum_k w_k * corr_k] * dx
    # corr_k = sum_i h[i] * (1-h[i+k-(n-1)])
    # d(corr_k)/dh_i = (1-h[i+k-(n-1)]) - h[i-(k-(n-1))]  (for valid indices)
    # Let me index differently.
    # full[k] = sum_{i} h[i] * (1-h[i+k-(n-1)])
    # Let l = k - (n-1) be the lag.
    # full[k] = sum_{i: 0<=i<n, 0<=i+l<n} h[i-l] * (1-h[i])
    # Wait, I'm confusing myself. Let me use the definition directly.
    
    # np.correlate(a, v, "full")[k] = sum_i a[i+k] * v[i]
    # where i ranges over valid values
    # So: corr[k] = sum_i h[i+k] * (1-h[i])
    # d(corr[k])/dh[j] = (1-h[j-k]) * [j+k valid] + (-h[j+k]) * [j-k valid]
    # Wait: corr[k] = sum_i h[i+k] * (1-h[i])
    # Let j = i+k, so i = j-k
    # corr[k] = sum_j h[j] * (1-h[j-k])  (for valid j, j-k)
    # d(corr[k])/dh[j] = (1-h[j-k]) - h[j+k]  (for valid j-k and j+k)
    
    # Actually, let me be more careful.
    # corr[k] = sum_{i} h[i+k] * (1-h[i]) where i+k < n and i >= 0
    # Let j = i+k, so i = j-k
    # corr[k] = sum_{j} h[j] * (1-h[j-k]) where 0 <= j-k < n and i = j-k >= 0
    # = sum_{j=k}^{n-1} h[j] * (1-h[j-k])  (for k >= 0)
    # But in full mode, k ranges from 0 to 2n-2.
    
    # Let me use a direct indexing approach.
    # For the k-th element of the full correlation:
    # corr_full[k] = sum_{i=max(0, k-n+1)}^{min(n-1, k)} h[i] * (1-h[i + k - (n-1)])
    # Let me define: for pair (i, j) where j = i + k - (n-1), i.e., k = n-1+j-i
    # corr_full[k] = sum_{i,j: j = i+k-(n-1), 0<=i<n, 0<=j<n} h[i] * (1-h[j])
    
    # d(corr_full[k])/dh[m] = (1-h[m]) * [m is used as i in corr_k] - h[m] * [m is used as j in corr_k]
    # m is used as i when: 0 <= m < n and 0 <= m + k - (n-1) < n  =>  n-1-m <= k <= n-1+m
    # m is used as j when: 0 <= m < n and 0 <= m + (n-1) - k < n  =>  m+n-1-k >= 0 and < n  =>  m <= k-(n-1) and m >= k-2n+2+1... 
    # This is getting complicated. Let me just use the vectorized approach.
    
    n_arr = np.arange(n)
    # For each k (0 to 2n-2), the pairs (i,j) with j = i + k - (n-1)
    # i ranges from max(0, n-1-k) to min(n-1, k)
    # Let's construct the gradient directly.
    
    # corr_full[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # Let l = k - (n-1) (the lag)
    # corr_full[k] = sum_i h[i] * (1-h[i+l]) for valid i (0<=i<n, 0<=i+l<n)
    
    # d/dh[m] corr_full[k]:
    # = (1-h[m+l]) * [0<=m<n and 0<=m+l<n]  -- m appears as the first index
    #   + (-h[m-l]) * [0<=m<n and 0<=m-l<n]  -- m appears as the second index (i+l=m, so i=m-l)
    #   = (1-h[m+l]) * [valid] - h[m-l] * [valid]
    
    # Total gradient: dh[m] obj = dx * sum_k w[k] * [(1-h[m+l_k]) - h[m-l_k]]
    # where l_k = k - (n-1)
    
    # Since corr_full[k] = sum_i h[i](1-h[i+l_k]), and this is symmetric in the sense that
    # corr_full[k] with l_k = l is the same as corr_full[2(n-1)-k] with l_{comp} = -l
    # (just with the roles of h swapped... no, actually they're different)
    
    # Let me just compute it directly for correctness.
    # For each m (0 to n-1):
    #   grad[m] = dx * sum_{k: m+l_k in [0,n)} w[k] * (1-h[m+l_k])
    #            + dx * sum_{k: m-l_k in [0,n)} w[k] * (-h[m-l_k])
    
    # First term: for each m, sum over k where l_k = k-(n-1) satisfies 0 <= m + k - (n-1) < n
    # i.e., (n-1-m) <= k <= (n-1-m+n-1) = (2n-2-m)
    # k ranges from max(0, n-1-m) to min(2n-2, 2n-2-m)
    
    # Second term: for each m, sum over k where m - (k-(n-1)) >= 0, i.e., k <= m+n-1
    # and m - (k-(n-1)) < n, i.e., k > m - (n-1) - ... 
    # m - k + n - 1 >= 0 => k <= m + n - 1
    # m - k + n - 1 < n => k > m - 1... wait
    # Let j = m - l_k = m - k + n - 1
    # Need 0 <= j < n => 0 <= m - k + n - 1 < n => m+1-n <= k <= m+n-1
    # k ranges from max(0, m+1-n) to min(2n-2, m+n-1)
    
    # This is complex. Let me just use the matrix approach for correctness (it's O(n²) which is fine for n≤1000).
    
    lags = np.arange(2*n-1) - (n-1)  # -（n-1）, ..., 0, ..., n-1
    # For each m, and each lag l:
    #   m contributes to corr at lag l as first index if 0 <= m+l < n: contrib = 1-h[m+l]
    #   m contributes to corr at lag l as second index if 0 <= m-l < n: contrib = -h[m-l]
    
    # Let's vectorize:
    # For first index contribution:
    #   valid1[m, l_idx] = (0 <= m + lags[l_idx] < n)
    #   contrib1[m, l_idx] = valid1[m, l_idx] * (1 - h[m + lags[l_idx]])
    
    # For second index contribution:
    #   valid2[m, l_idx] = (0 <= m - lags[l_idx] < n)
    #   contrib2[m, l_idx] = valid2[m, l_idx] * (-h[m - lags[l_idx]])
    
    m_idx = np.arange(n)[:, None]
    l_arr = lags[None, :]
    
    # First index: m + l valid
    fi = m_idx + l_arr
    vf = (fi >= 0) & (fi < n)
    fc = np.clip(fi, 0, n-1)
    
    # Second index: m - l valid
    si = m_idx - l_arr
    vs = (si >= 0) & (si < n)
    sc = np.clip(si, 0, n-1)
    
    grad = dx * (vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :]).sum(axis=1)
    
    return obj, grad

def active_set_obj_grad(h, active_lags, alpha):
    """Smooth max over only the active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    active_corr = corr[active_lags]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m + (max_sh + np.log(np.sum(exp_v))) / alpha) * dx
    
    # Reconstruct full weight vector (zeros for inactive)
    w_full = np.zeros(2*n-1)
    w_full[active_lags] = w
    
    n_arr = np.arange(n)
    lags = np.arange(2*n-1) - (n-1)
    m_idx = np.arange(n)[:, None]
    l_arr = lags[None, :]
    fi = m_idx + l_arr
    vf = (fi >= 0) & (fi < n)
    fc = np.clip(fi, 0, n-1)
    si = m_idx - l_arr
    vs = (si >= 0) & (si < n)
    sc = np.clip(si, 0, n-1)
    
    grad = dx * (vf * (1.0 - h[fc]) * w_full[None, :] - vs * h[sc] * w_full[None, :]).sum(axis=1)
    
    return obj, grad

def slsqp_opt(h0, n, target_sum, alpha, active_lags=None, maxiter=1000):
    if active_lags is not None:
        def f(x): return active_set_obj_grad(x, active_lags, alpha)[0]
        def g(x): return active_set_obj_grad(x, active_lags, alpha)[1]
    else:
        def f(x): return smooth_max_h_obj_grad(x, alpha)[0]
        def g(x): return smooth_max_h_obj_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_to_sum(r.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
