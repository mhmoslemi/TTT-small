```python
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)

        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-8)[0]

        grad = np.zeros(n)

            k = m - (n-1)

            elif k > 0:
                g = np.zeros(n)
                g[k:] += (1.0 - h[:n-k])
                g[:n-k] -= h[k:]
            else:
                kk = -k
                g[:n-kk] += (1.0 - h[kk:])
                g[kk:] -= h[:n-kk]
            grad += g
        grad /= len(active)
        # In the interior, grad should be constant (= lambda)
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if not np.any(free):
            break
        # Project gradient to zero mean on free variables
        grad[free] -= np.mean(grad[free])
        # Take a step
        step = 0.01
        h_new = h + step * (-grad)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return h, c5
```

A better approach: compute the quadratic coefficients analytically for each active lag, then minimize the max of quadratics in 1D.

For a fixed lag k, f_k(h + t*(e_i - e_j)):
The change in f_k is:
- At index i: h[i] → h[i]+t
- At index j: h[j] → h[j]-t

The change in f_k = Δf_k(t) is quadratic in t (since f_k is quadratic in h).

Specifically:
f_k(h) = sum_valid h[m](1-h[m+k])

Changing h[i] by +t and h[j] by -t:
Δf_k = (terms involving h[i] or h[i]+k or h[i]-k with +t) + (terms involving h[j] or h[j]+k or h[j]-k with -t) + (cross terms if i and j are within distance k of each other)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wi = w[m_idx]
        if wi < 1e-14:
            continue
        k = m_idx - (n - 1)
        grad += wi * lag_gradient(h, k)
    return obj, grad

def slsqp_opt(h0, target_sum, alpha, maxiter=800):
    n = len(h0)
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_desc_line_search(h, target_sum, max_iters=20000):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = lag_gradient(h, lag)
        can_inc = h < 1.0 - 1e-12
        can_dec = h > 1e-12
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e18)
        g_dec = np.where(can_dec, g, -1e18)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-14:
            break
        t_lo = max(-h[i_inc], -0.999)
        t_hi = min(1.0 - h[i_inc], h[i_dec])
        if t_hi <= t_lo:
            break
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = t_lo, t_hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h_c = h.copy()
        h_c[i_inc] += c
        h_c[i_dec] -= c
        f_c = compute_c5(h_c)
        h_d = h.copy()
        h_d[i_inc] += d
        h_d[i_dec] -= d
        f_d = compute_c5(h_d)
        for _ in range(40):
            if f_c < f_d:
                b = d
                d = c
                f_d = f_c
                c = b - gr * (b - a)
                h_c = h.copy()
                h_c[i_inc] += c
                h_c[i_dec] -= c
                f_c = compute_c5(h_c)
            else:
                a = c
                c = d
                f_c = f_d
                d = a + gr * (b - a)
                h_d = h.copy()
                h_d[i_inc] += d
                h_d[i_dec] -= d
                f_d = compute_c5(h_d)
        t_opt = (a + b) / 2.0
        if abs(t_opt) < 1e-14:
            break
        h[i_inc] += t_opt
        h[i_dec] -= t_opt
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            stale = 0
        else:
            stale += 1
            if stale > 30:
                break
    return h, c5

def kkt_gradient_method(h, target_sum, max_iters=5000, base_lr=0.02):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    lr = base_lr
    for it in range(max_iters):
        corr = compute_corr(h)
        max_val = np.max(corr)
        tol = 1e-8
        active = np.where(corr >= max_val - tol)[0]
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            grad += lag_gradient(h, k)
        grad /= len(active)
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            grad[free] -= np.mean(grad[free])
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if lr < 0.1:
                lr *= 1.05
        else:
            lr *= 0.6
            if lr < 1e-14:
                break
    return h, c5

def fourier_perturb(h, n, target_sum, modes, amplitude, rng):
    phi = np.arange(n)
    pert = np.zeros(n)
    for m in modes:
        freq = m * 2 * np.pi / n
        pert += amplitude * (rng.randn() * np.cos(phi * freq) + rng.randn() * np.sin(phi * freq))
    return project_to_sum(h + pert, target_sum)

def interp_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def adam_subgrad(h, target_sum, max_iters=10000, lr=0.01, beta1=0.9, beta2=0.999):
    n = len(h)
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    c5 = compute_c5(h)
    for t in range(1, max_iters + 1):
        corr = compute_corr(h)
        m_val = np.max(corr)
        active = np.where(corr >= m_val - 1e-8)[0]
        g = np.zeros(n)
        for m_idx in active:
            k = m_idx - (n - 1)
            g += lag_gradient(h, k)
        g /= len(active)
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            g[free] -= np.mean(g[free])
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            h = h  # don't update
    return h, c5
```
