def project_feasible(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target_sum - np.sum(h_proj)
    free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
    if np.any(free):
        h_proj[free] += diff / np.sum(free)
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def topk_subgradient_step(h, n, target_sum, k, step_size):
    """One step of top-k subgradient method."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(corr)[-k:]
    # Compute weighted subgradient
    g = np.zeros(n)
    for idx in top_idx:
        l = idx - (n - 1)  # actual shift
        # Gradient of corr[idx] with respect to h
        # corr[idx] = sum_i h[i] * (1-h[i+l]) for valid i

    g /= k
    h_new = project_feasible(h - step_size * g, target_sum)

    sorted_idx = np.argsort(g)

    h_lp[sorted_idx[:int(target_sum)]] = 1.0

    frac = target_sum - int(target_sum)
    if frac > 0:
