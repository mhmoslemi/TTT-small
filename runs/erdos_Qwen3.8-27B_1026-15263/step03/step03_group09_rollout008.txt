        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        t += 1
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1.0-h_clip, mode='full')
        w = smooth_max_weights(corr, alpha)
        g = compute_gradient_fast(h_clip, w)
        m_vec = 0.9*m_vec + 0.1*g
        v_vec = 0.999*v_vec + 0.001*g*g
        m_hat = m_vec / (1 - 0.9**t)
        v_hat = v_vec / (1 - 0.999**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = lr_init / (1 + 0.01*it)
    return best_h, best_c5
