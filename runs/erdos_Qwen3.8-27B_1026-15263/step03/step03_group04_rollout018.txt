import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradient(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    return grad

def topk_grad(h, k=10):
    corr = compute_corr(h)
    n = len(h)
    top_idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for ks in top_idx:
        shift = ks - (n - 1)
        grad += shift_gradient(h, shift)
    return grad / k, corr

def smooth_max_grad(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-8
    sig = np.where(w > thresh)[0]
    for ks in sig:
        wi = w[ks]
        shift = ks - (n - 1)
        grad += wi * shift_gradient(h, shift)
    return obj, grad

                   momentum=0.9, max_iter=50000):

    v = np.zeros(n)
    y = h.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break

        grad, corr = topk_grad(y, k)
        v = momentum * v + grad
        y = h - lr * v
        y = project_to_sum(y, target_sum)

            y = h - lr * v
            y = project_to_sum(y, target_sum)
            c5_new = compute_c5(y)

            best_h = y.copy()
        
        h = y
        # Gradually increase lr or keep adaptive

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradient(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    return grad

def topk_grad(h, k=10):
    corr = compute_corr(h)
    n = len(h)
    top_idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for ks in top_idx:
        shift = ks - (n - 1)
        grad += shift_gradient(h, shift)
    return grad / k, corr

def smooth_max_grad(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-8
    sig = np.where(w > thresh)[0]
    for ks in sig:
        wi = w[ks]
        shift = ks - (n - 1)
        grad += wi * shift_gradient(h, shift)
    return obj, grad

def nesterov_optimize(h0, n, target_sum, t0, time_limit, k=10, lr_init=0.005,
                      momentum=0.85, max_iter=100000):

    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    max_stall = 2000

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad, corr = topk_grad(h, k)
        v = momentum * v + grad
        h_new = h - lr * v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new <= compute_c5(h):
            h = h_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall % 500 == 0:
                lr = min(lr * 1.02, lr_init * 3)
        else:
            lr *= 0.5
            stall += 1
        if stall > max_stall:
            v = np.zeros(n)
            lr = lr_init
            stall = 0
            h = best_h.copy()
    return best_h, best_c5

        else:

        return smooth_max_grad(x, alpha)[0]

                       bounds=[(0.0, 1.0)] * n,

        ts = best_n / 2.0
        h_final, c5_final = directed_flip_polish(best_h, best_n, ts, t0,
                                                  t0 + min(30, remaining()), rng, max_rounds=2000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
