import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def subgradient_exact(h, target_sum, max_steps, c_init=1.0, decay=0.999):
    """Subgradient descent on the exact max with Polyak-like step size."""
    n = len(h)
    dx = 2.0 / n
    h = project_to_sum(h.copy(), target_sum)
    c5_best = compute_c5(h)
    c = c_init
    h_prev = h.copy()
    
    for step in range(max_steps):
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        max_c5 = max_val * dx
        
        # Compute subgradient for the maximizing lag
        shift = max_idx - (n - 1)
        grad = np.zeros(n)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
        
        # Also add constraint gradient (keep sum constant)
        # Project gradient to the tangent space of the constraint
        grad_mean = np.mean(grad)
        grad_tan = grad - grad_mean
        
        # Step
        h_new = h - c * grad_tan
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_best:
            c5_best = c5_new
            h = h_new
            # Can increase step slightly
            c = min(c * 1.05, 0.5)
        else:
            h = h_prev * 0.5 + h_new * 0.5  # Partial step
            h = project_to_sum(h, target_sum)
            c *= decay
            h = project_to_sum(h - 0.5 * c * grad_tan, target_sum)
        
        h_prev = h.copy()
    
    return h, c5_best

    n = len(h)

def optimize_smooth(h0, n, target_sum, alpha, maxiter=300):
    """L-BFGS-B optimization with smooth max objective and penalty on sum."""
    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 1e6 * s_dev * s_dev
    
    def jac(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + 2e6 * s_dev * np.ones(n)
    
    try:
        res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0.copy(), target_sum)

                    best_c5 = c5_sg2
                    best_h = h_sg2
                    h_ref = h_sg2
                h_bin = (h_ref > 0.5).astype(float)
                n_target = int(round(ts1))
                n_ones = int(h_bin.sum())
                if n_ones > n_target:
                    ones = np.where(h_bin == 1)[0]
                    rng.shuffle(ones)
                    for idx in ones[:n_ones - n_target]:
                        h_bin[idx] = 0.0
                elif n_ones < n_target:
                    zeros = np.where(h_bin == 0)[0]
                    rng.shuffle(zeros)
                    for idx in zeros[:n_target - n_ones]:
                        h_bin[idx] = 1.0
    
    if remaining() > 50:
        h_cd, c5_cd = coord_descent(best_h, best_n, ts1, max_rounds=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    if remaining() > 200:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        c5_int = compute_c5(h_interp)
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_interp
            best_n = n2
            ts2 = best_n / 2.0
        else:
            ts2 = n2 / 2.0
        
        for a in [20, 200, 2000, 20000]:
            if remaining() < 40:
                break
            h_opt = optimize_smooth(h_interp, n2, ts2, a, maxiter=250)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
                h_interp = h_opt
            h_sg3, c5_sg3 = sg_subgradient(h_opt, n2, ts2, max_steps=400, top_lags=8)
            if c5_sg3 < best_c5:
                best_c5 = c5_sg3
                best_h = h_sg3
                best_n = n2
                h_interp = h_sg3
        
        if remaining() > 40:
            h_ls2, c5_ls2 = line_search_subgradient(best_h, best_n, best_n / 2.0, max_steps=8000, top_lags=10)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2
    
    if remaining() > 150:
        n_try = best_n
        ts_try = n_try / 2.0
        starts = make_self_comp_starts(n_try, ts_try, rng, n_starts=6)
        for h_s in starts:
            if remaining() < 25:
                break
            c5_s = compute_c5(h_s)
            h_cur = h_s
            for a in [50, 500, 5000]:
                if remaining() < 15:
                    break
                h_opt = optimize_smooth(h_cur, n_try, ts_try, a, maxiter=150)
                c5_o = compute_c5(h_opt)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_opt
                    best_n = n_try
                    h_cur = h_opt
            h_sgl, c5_sgl = line_search_subgradient(h_cur, n_try, ts_try, max_steps=5000, top_lags=8)
            if c5_sgl < best_c5:
                best_c5 = c5_sgl
                best_h = h_sgl
                best_n = n_try
    
    for rnd in range(3):
        if remaining() < 25:
            break
        n_f = best_n
        ts_f = n_f / 2.0
        h_sg4, c5_sg4 = sg_subgradient(best_h, n_f, ts_f, max_steps=500, top_lags=10, c_init=0.05, momentum=0.9)
        if c5_sg4 < best_c5:
            best_c5 = c5_sg4
            best_h = h_sg4
        h_ls3, c5_ls3 = line_search_subgradient(best_h, n_f, ts_f, max_steps=8000, top_lags=12)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3
        h_bin2, c5_bin2 = systematic_binary_search(best_h, n_f, ts_f, rng, max_rounds=100)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            h_ref2 = optimize_smooth(h_bin2, n_f, ts_f, 10000, maxiter=300)
            c5_r2 = compute_c5(h_ref2)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_ref2
        h_cd2, c5_cd2 = coord_descent(best_h, n_f, ts_f, max_rounds=40)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2
    
    if remaining() > 20:
        h_final = optimize_smooth(best_h, best_n, best_n / 2.0, 100000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
