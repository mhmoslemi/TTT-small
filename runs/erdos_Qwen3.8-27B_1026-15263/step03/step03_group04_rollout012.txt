def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    h = np.asarray(h, dtype=float)
    n = len(h)
    
    # Quick check
    if np.sum(h) == target_sum and np.all(h >= -1e-15) and np.all(h <= 1+1e-15):
        return np.clip(h, 0, 1)
    
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo = h.min() - 1.0
    hi = h.max() + 1.0

    n = len(h)

def frank_wolfe_step(h, grad, target_sum):
    """Frank-Wolfe step: find s in vertices of {0≤s≤1, sum=target_sum} minimizing grad·s"""
    # The vertex that minimizes grad·s is: s[i]=1 for the target_sum indices with smallest grad[i]
    n = len(h)
    order = np.argsort(grad)
    s = np.zeros(n)

    # Handle fractional part
    frac = target_sum - int(target_sum)
    if frac > 1e-10:
        s[order[int(target_sum)]] = frac
    return s

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if abs(h.sum() - target_sum) < 1e-14 and h.min() >= -1e-14 and h.max() <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    mx = np.max(shifted)
    exp_v = np.exp(shifted - mx)
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def exact_subgrad(h, top_k=8):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    m = corr[top_idx[0]]
    grad = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    grad /= float(top_k)
    return m, grad

    k = int(target_sum)
    s[order[:k]] = 1.0
    frac = target_sum - k
    if frac > 1e-10:
        s[order[k]] = frac
    return s

def optimize_smooth(h0, n, target_sum, alpha, maxiter=400):
    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        sv = x.sum() - target_sum
        return obj + 1e4 * sv * sv
    def jac(x):
        _, g = smooth_max_obj_and_grad(x, alpha)
        sv = x.sum() - target_sum
        return g + 2e4 * sv * np.ones(n)
    try:

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: Frank-Wolfe
    if remaining() > 200:
        h_fw = fw_methods(best_h, ts1, max_iter=4000, top_k=12)
        c5 = compute_c5(h_fw)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fw

    # Phase 3: Momentum projected gradient
    if remaining() > 150:
        h_pg = momentum_proj_grad(best_h, ts1, max_iter=3000, lr0=0.02, beta=0.9)
        c5 = compute_c5(h_pg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pg

    # Phase 4: Subgradient polish (exact max, adaptive step)
    if remaining() > 100:
        h_sp = subgrad_polish(best_h, ts1, max_rounds=500, n_lr=25)
        c5 = compute_c5(h_sp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sp

    # Phase 5: Frank-Wolfe again after polish
    if remaining() > 150:
        h_fw2 = fw_methods(best_h, ts1, max_iter=5000, top_k=15)
        c5 = compute_c5(h_fw2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fw2

    # Phase 6: Binary simulated annealing
    if remaining() > 200:
        h_sa, c5_sa = binary_sa(best_h, n1, ts1, n_iter=40000, t_max=0.04, t_min=0.0005, rng=rng)
        if c5_sa < best_c5:
            # Refine the binary solution
            h_ref = optimize_smooth(h_sa, n1, ts1, 500, maxiter=500)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
            else:
                best_c5 = c5_sa
                best_h = h_sa
        h_sp2 = subgrad_polish(best_h, ts1, max_rounds=300, n_lr=20)
        c5 = compute_c5(h_sp2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sp2

    # Phase 7: Higher resolution (n=480)
    if remaining() > 400:
        n2 = n1 * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        for a in [30, 100, 500, 2000, 10000, 50000]:
            if remaining() < 80:
                break
            h_opt = optimize_smooth(h_interp, n2, ts2, a, maxiter=300)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
                h_interp = h_opt
        if remaining() > 100:
            h_fw3 = fw_methods(best_h, best_n / 2.0, max_iter=3000, top_k=12)
            c5 = compute_c5(h_fw3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fw3
        if remaining() > 80:
            h_sp3 = subgrad_polish(best_h, best_n / 2.0, max_rounds=300, n_lr=20)
            c5 = compute_c5(h_sp3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sp3

    # Phase 8: Multi-start at best resolution
    if remaining() > 200:
        n_b = best_n
        ts_b = n_b / 2.0
        starts = make_diverse_starts(n_b, ts_b, rng, best_h, best_n)
        for h_s in starts[:15]:
            if remaining() < 40:
                break
            c5_s = compute_c5(h_s)
            h_cur = h_s
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
            for a in [100, 1000, 10000]:
                if remaining() < 25:
                    break
                h_opt = optimize_smooth(h_cur, n_b, ts_b, a, maxiter=200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    h_cur = h_opt

    # Phase 9: Pair transfer local search
    if remaining() > 100:
        h_pt, c5_pt = pair_transfer_search(best_h, best_n, best_n / 2.0, rng, n_trials=5000)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 10: Final polish
    if remaining() > 60:
        h_final = subgrad_polish(best_h, best_n / 2.0, max_rounds=400, n_lr=30)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final
    if remaining() > 40:
        h_fw_final = fw_methods(best_h, best_n / 2.0, max_iter=3000, top_k=20)
        c5_f = compute_c5(h_fw_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fw_final
    if remaining() > 30:
        h_sm = optimize_smooth(best_h, best_n, best_n / 2.0, 200000, maxiter=500)
        c5_f = compute_c5(h_sm)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_sm

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
