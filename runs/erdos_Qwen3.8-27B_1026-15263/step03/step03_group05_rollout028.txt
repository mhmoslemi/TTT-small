import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_top_k_obj_and_grad(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    if K >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
    
    top_vals = corr[top_idx]
    m_val = np.max(top_vals)
    shifted = top_vals - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)
    for t, idx in enumerate(top_idx):
        wt = w[t] * dx
        i_low = max(0, n - 1 - idx)
        i_high = min(n - 1, n - 1 + idx)
        i_range = np.arange(i_low, i_high + 1)
        j2 = i_range + idx - n + 1
        grad[i_range] += wt * (1.0 - h[j2])
        grad[j2] += wt * (-h[i_range])
    
    return obj, grad

def projected_gradient_topk(h, n, alpha, K, max_iter, lr_init, time_fn):
    h = h.copy()
    lr = lr_init
    best_obj = smooth_max_top_k_obj_and_grad(h, alpha, K)[0]
    best_h = h.copy()
    target_sum = n / 2.0
    
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_top_k_obj_and_grad(h, alpha, K)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        obj_new = smooth_max_top_k_obj_and_grad(h_new, alpha, K)[0]
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, target_sum)
            obj_new = smooth_max_top_k_obj_and_grad(h_new, alpha, K)[0]
        else:
            lr = min(lr * 1.05, 1.0)
        
        h = h_new

def correlation_repair(h, n, K=5, max_rounds=100):
    """Try to reduce the top K lags by equalizing h values at correlated positions."""
    h = h.copy()
    target_sum = n / 2.0
    c5_cur = compute_c5(h)
    
    for _ in range(max_rounds):
        corr = np.correlate(h, 1-h, mode="full")
        # Get top K lags
        top_k = np.argpartition(corr, -K)[-K:]
        best_lag = top_k[np.argmax(corr[top_k])]
        
        # For this lag, find the positions that contribute most
        k = best_lag
        i_low = max(0, n - 1 - k)
        i_high = min(n - 1, n - 1 + k)
        i_range = np.arange(i_low, i_high + 1)
        j_range = i_range + k - n + 1
        
        # Contribution of each pair: h[i] * (1-h[j])
        contrib = h[i_range] * (1 - h[j_range])
        
        # Find the pair with the largest contribution
        max_idx = np.argmax(contrib)
        i_star = i_range[max_idx]
        j_star = j_range[max_idx]
        
        # Try to equalize: move h[i_star] down and h[j_star] up (or vice versa)
        # h[i_star] is high, h[j_star] is low (since h[i]*(1-h[j]) is large)
        # So we want to decrease h[i_star] and increase h[j_star]
        delta = min(h[i_star] - h[j_star], 0.1) * 0.5
        if delta > 1e-6:
            h[i_star] -= delta
            h[j_star] += delta
            # This preserves sum
            h = np.clip(h, 0, 1)
            
            c5_new = compute_c5(h)
            if c5_new > c5_cur:
                h[i_star] += delta
                h[j_star] -= delta
            else:
                c5_cur = c5_new
    
    return h, c5_cur

def project_box_sum(y, s):

    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0

def smooth_max_topk_obj_grad(h, alpha, K):

                if time_left() < 15:
                    break
                h_new, _ = pgd_topk(best_h, n, alpha, K, 3000, 0.1, time_left)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 2: Correlation repair
    if time_left() > 150:
        h_rep, c5_rep = correlation_repair(best_h, n, rng, 20000, time_left)
        if c5_rep < best_c5:
            best_c5 = c5_rep
            best_h = h_rep.copy()

    # Phase 3: Simulated annealing
    if time_left() > 200:
        h_sa, c5_sa = simulated_annealing(best_h, n, rng, time_left, 20000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Gradient polish after SA
    if time_left() > 100:
        for alpha in [500, 2000, 8000]:
            if time_left() < 15:
                break
            for K in [5, 15]:
                if time_left() < 10:
                    break
                h_new, _ = pgd_topk(best_h, n, alpha, K, 3000, 0.05, time_left)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 5: Fourier-space L-BFGS
    if time_left() > 120:
        for n_f in [n, min(480, n * 2)]:
            if time_left() < 60:
                break
            if n_f % 2 != 0:
                n_f += 1
            ts_f = n_f / 2.0
            if n_f != n:
                x_old = np.linspace(0, 2, n, endpoint=False)
                x_new = np.linspace(0, 2, n_f, endpoint=False)
                h_init = np.interp(x_new, x_old, best_h, period=2.0)
                h_init = project_box_sum(h_init, ts_f)
            else:
                h_init = best_h.copy()
            n_harm = min(40, max(12, n_f // 6))
            coeffs = grid_to_fourier(h_init, n_f, n_harm)
            for K in [15, 40]:
                if time_left() < 30:
                    break
                try:
                    res = minimize(
                        fourier_obj, coeffs.copy(),
                        args=(n_f, n_harm, K),
                        method='L-BFGS-B',
                        options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-12}
                    )
                    h_opt = fourier_to_grid(res.x, n_f, n_harm)
                    h_opt = project_box_sum(h_opt, ts_f)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n_f
                        n = n_f
                        coeffs = grid_to_fourier(h_opt, n_f, n_harm)
                    else:
                        coeffs = grid_to_fourier(h_init, n_f, n_harm)
                except Exception:
                    pass
            if n_f != n:
                h_topk, _ = pgd_topk(best_h, n, 2000, 15, 2000, 0.05, time_left)
                c5_topk = compute_c5(h_topk)
                if c5_topk < best_c5:
                    best_c5 = c5_topk
                    best_h = h_topk.copy()

    # Phase 6: Second SA round with different moves
    if time_left() > 120:
        h_sa2, c5_sa2 = simulated_annealing(best_h, n, rng, time_left, 15000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            if time_left() > 30:
                h_pol, _ = pgd_topk(best_h, n, 5000, 10, 2000, 0.03, time_left)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 7: Multi-start exploration
    if time_left() > 80:
        for trial in range(15):
            if time_left() < 25:
                break
            ns = 0.15 * (1.0 - trial / 15.0) + 0.01
            h_pert = best_h + rng.normal(0, ns, n)
            h_pert = project_box_sum(h_pert, n / 2.0)
            for alpha in [200, 2000]:
                if time_left() < 15:
                    break
                h_new, _ = pgd_topk(h_pert, n, alpha, 15, 1500, 0.1, time_left)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_pert = h_new

    # Phase 8: Final correlation repair + gradient
    if time_left() > 40:
        h_rep2, c5_rep2 = correlation_repair(best_h, n, rng, 10000, time_left)
        if c5_rep2 < best_c5:
            best_c5 = c5_rep2
            best_h = h_rep2.copy()
        if time_left() > 20:
            h_fin, _ = pgd_topk(best_h, n, 10000, 5, 5000, 0.02, time_left)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
