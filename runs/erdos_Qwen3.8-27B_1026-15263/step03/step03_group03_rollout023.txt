def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Compute gradient using the subgradient formula
    g = np.zeros(n)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    # For correlation index k, lag d = k - (n-1)
    d_idx = m_idx - (n - 1)
    
    # First term: (1-h[j-d]) * 1[0<=j-d<n]
    # j-d >= 0 -> j >= d; j-d < n -> j < n+d
    valid_f = (j_idx >= d_idx) & (j_idx < n + d_idx) & (j_idx >= 0) & (j_idx < n)
    fc_idx = np.clip(j_idx - d_idx, 0, n-1)
    cf = valid_f * (1.0 - h[fc_idx]) * w[None, :]
    
    # Second term: -h[j+d] * 1[0<=j+d<n]
    valid_s = (j_idx + d_idx >= 0) & (j_idx + d_idx < n) & (j_idx >= 0) & (j_idx < n)
    sc_idx = np.clip(j_idx + d_idx, 0, n-1)
    cs = valid_s * h[sc_idx] * w[None, :]
    
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def compute_subgradient(h, top_m=5, alpha=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:top_m]
    top_vals = corr[top_idx]
    weights = np.exp(alpha * (top_vals - top_vals[0]))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for m_idx in range(top_m):
        k = top_idx[m_idx]
        d = k - (n - 1)
        w = weights[m_idx]
        
        # First term: (1-h[j-d]) for valid j
        if d >= 0:
            # j from d to n-1, j-d from 0 to n-1-d
            g[d:] += w * (1.0 - h[:n-d])
        else:
            # j from 0 to n-1+d, j-d from -d to n-1
            g[:n+d] += w * (1.0 - h[-d:])
        
        # Second term: -h[j+d] for valid j
        if d >= 0:
            # j from 0 to n-1-d, j+d from d to n-1
            g[:n-d] -= w * h[d:]
        else:
            # j from -d to n-1, j+d from 0 to n-1+d
            g[-d:] -= w * h[:n+d]
    
    return g, corr

The problem says: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

In the discrete setting with dx = 2/n:

where corr_full = np.correlate(h, 1-h, mode="full")

The "full" correlation has 2n-1 values. The index k in the correlation corresponds to different shifts. Specifically, corr_full[k] for k = 0, 1, ..., 2n-2.

The shift corresponding to index k in the full correlation is (k - (n-1)) * dx in terms of the continuous variable. So the shifts range from -(n-1)*dx to (n-1)*dx, i.e., from -(n-1)*2/n to (n-1)*2/n.

For large n, this is approximately [-2, 2]. The problem asks for the max over all shifts k, which in the discrete setting is the max over all 2n-1 correlation values.

So the evaluation is correct: C₅ = max(corr_full) * dx.

Now, one subtlety: the "full" correlation at index k represents the shift (k - (n-1)), but the actual integral ∫₀² h(x)(1-h(x+k)) dx only makes sense for certain values of k (specifically, when the overlap of [0,2] and [-k, 2-k] is non-trivial). For the discrete correlation, all 2n-1 values are computed, but some of them correspond to shifts where the overlap is small (near the edges).

This is fine - the max will typically be achieved at a shift where the overlap is significant.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def compute_subgradient(h, top_m=5, alpha=10.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:top_m]
    top_vals = corr[top_idx]
    weights = np.exp(alpha * (top_vals - top_vals[0]))
    weights /= weights.sum()
    g = np.zeros(n)
    for m_i in range(top_m):
        k = top_idx[m_i]
        d = k - (n - 1)
        w = weights[m_i]
        if d >= 0:
            g[d:] += w * (1.0 - h[:n - d])
            g[:n - d] -= w * h[d:]
        else:
            g[:n + d] += w * (1.0 - h[-d:])
            g[-d:] -= w * h[:n + d]
    return g, corr

def fw_linear_oracle(g, n):
    idx = np.argpartition(g, n // 2)[:n // 2]
    y = np.zeros(n)
    y[idx] = 1.0
    return y

def accelerated_fw(h0, n, target_sum, max_iter, time_limit, top_m=5, alpha=10.0):
    h = h0.copy()
    x = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for t in range(1, max_iter + 1):
        if time.time() > time_limit:
            break
        gamma = 2.0 / (t + 2.0)
        g, corr = compute_subgradient(x, top_m, alpha)
        y = fw_linear_oracle(g, n)
        x = (1.0 - gamma) * x + gamma * y
        if t % 50 == 0:
            x = project_to_sum(x, target_sum)
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5

            stall = 0
        else:

        if stall > 200:
            x = project_to_sum(x + 0.05 * np.random.RandomState(t).uniform(-1, 1, n), target_sum)
            stall = 0
    return best_h, best_c5

def pair_descent_sweep(h, corr, rng, n_cand=12, adaptive=False):
    n = len(h)
    corr_max_val = np.max(corr)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = corr_max_val
        best_delta = 0.0
        best_D = None
        best_k = -1
        if adaptive and np.max(corr) > 0:
            max_idx = np.argmax(corr)
            max_lag = max_idx - (n - 1)
            nearby = [(j + dd) % n for dd in [-3, -2, -1, 1, 2, 3]]
            lag_c = [(j + max_lag) % n, (j - max_lag) % n, (j + 2 * max_lag) % n, (j - 2 * max_lag) % n]
            rand_c = [rng.randint(n) for _ in range(max(0, n_cand - 10))]
            candidates = nearby + lag_c + rand_c
        else:
            candidates = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
            candidates += [rng.randint(n) for _ in range(max(0, n_cand - 5))]
        seen = set([j])
        for k in candidates:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    g = np.zeros(n)
    m_idx = np.arange(2 * n - 1)
    d_arr = m_idx - (n - 1)
    w_exp = w
    for m_i in range(len(m_idx)):
        d = d_arr[m_i]
        wt = w_exp[m_i]
        if wt < 1e-10:
            continue
        if d >= 0:
            g[d:] += wt * (1.0 - h[:n - d])
            g[:n - d] -= wt * h[d:]
        else:
            g[:n + d] += wt * (1.0 - h[-d:])
            g[-d:] -= wt * h[:n + d]
    return obj, g

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
