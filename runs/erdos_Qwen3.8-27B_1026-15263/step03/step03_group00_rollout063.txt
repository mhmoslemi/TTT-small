import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def topk_grad(h, K=5, decay=0.7):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    # Sort in descending order
    order = np.argsort(top_vals)[::-1]
    top_idx = top_idx[order]
    
    weights = np.array([decay ** i for i in range(K)])
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for w_val, idx in zip(weights, top_idx):
        if w_val < 1e-10:
            continue
        lag = int(idx) - (n - 1)
        g += w_val * corr_gradient(h, lag)
    return g

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Only compute gradient for significant lags
    g = np.zeros(n)
    significant = w > 1e-8
    for idx in np.where(significant)[0]:
        wi = w[idx]
        lag = int(idx) - (n - 1)
        g += wi * corr_gradient(h, lag)
    return g

def slsqp_optimize(h0, target_sum, alpha, maxiter=1000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_obj(x, alpha)
    
    def g(x):
        return smooth_max_grad(x, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    fracs = [0.5, 0.33, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    for it in range(max_iters):
        g = topk_grad(h, K, decay)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        h_save = h.copy()
        best_c5 = c5
        best_frac = 0.0
        for frac in fracs:
            delta = frac * max_move
            if delta < 1e-13:
                break
            h = h_save.copy()
            h[i_inc] += delta
            h[i_dec] -= delta
            h = project_to_sum(h, target_sum)
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_frac = frac
            else:
                break
        if best_frac > 0.0:
            h[i_inc] += best_frac * max_move
            h[i_dec] -= best_frac * max_move
            h = project_to_sum(h, target_sum)
            c5 = best_c5
            stale = 0
        else:
            stale += 1
            if stale > 200:
                break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def subgradient_polish(h, target_sum, max_iters, alpha0):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for t in range(max_iters):
        corr = compute_corr(h)
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = -corr_gradient(h, lag)
        lr = alpha0 / (1.0 + 0.005 * float(t))
        h_new = h + lr * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts_cur = n_cur / 2.0

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP refinement with increasing sharpness
    for alpha in [50, 300, 1500, 6000, 20000]:
        if time_left() < 60:
            break
        h_new = slsqp_step(best_h, ts_cur, alpha, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Coordinate descent with line search, top-K=5
    if time_left() > 100:
        max_it = min(100000, int(time_left() * 20))
        h_cd, c5_cd = coord_desc_ls(best_h, ts_cur, max_it, K=5, decay=0.7)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Subgradient polish
    if time_left() > 60:
        h_sg, c5_sg = subgradient_polish(best_h, ts_cur, max_iters=5000, alpha0=0.8)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        if time_left() > 40:
            h_cd2, c5_cd2 = coord_desc_ls(best_h, ts_cur, 20000, K=3, decay=0.8)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 4: Resolution increase
    for n_tr in [720, 900, 1080, 1260]:
        if time_left() < 150:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_cur = compute_c5(h_interp)
        for alpha in [200, 1500, 8000]:
            if time_left() < 60:
                break
            h_opt = slsqp_step(h_interp, ts_tr, alpha, maxiter=500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_cur:
                h_interp = h_opt
                c5_cur = c5_opt
        if time_left() > 80:
            h_cd3, c5_cd3 = coord_desc_ls(h_interp, ts_tr, min(30000, int(time_left() * 10)), K=5, decay=0.7)
            if c5_cd3 < c5_cur:
                h_interp = h_cd3
                c5_cur = c5_cd3
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_interp.copy()
            best_n = n_tr
            n_cur = n_tr
            ts_cur = ts_tr

    # Phase 5: Diverse restarts with structured perturbations
    n_restart = max(8, min(35, int(time_left() / 40)))
    for r in range(n_restart):
        if time_left() < 35:
            break
        style = r % 4
        if style == 0:
            noise = 0.015 + 0.05 * (1.0 - r / float(n_restart))
            h_shake = best_h + rng.uniform(-noise, noise, best_n)
        elif style == 1:
            h_shake = best_h.copy()
            blen = rng.randint(4, max(5, best_n // 8))
            b1 = rng.randint(0, max(1, best_n - blen))
            b2 = rng.randint(0, max(1, best_n - blen))
            if abs(b1 - b2) > blen:
                tmp = h_shake[b1:b1+blen].copy()
                h_shake[b1:b1+blen] = h_shake[b2:b2+blen]
                h_shake[b2:b2+blen] = tmp
            noise_sm = 0.01
            h_shake += rng.uniform(-noise_sm, noise_sm, best_n)
        elif style == 2:
            h_shake = best_h.copy()
            blen = rng.randint(4, max(5, best_n // 5))
            b1 = rng.randint(0, max(1, best_n - blen))
            h_shake[b1:b1+blen] = 1.0 - h_shake[b1:b1+blen]
        else:
            h_shake = best_h.copy()
            shift = rng.randint(1, best_n // 4)
            h_shake = np.roll(h_shake, shift)
            mask = rng.random(best_n) < 0.1
            h_shake[mask] = 1.0 - h_shake[mask]

        h_shake = project_to_sum(h_shake, ts_cur)
        h_cd4, c5_cd4 = coord_desc_ls(h_shake, ts_cur, max_iters=6000, K=3, decay=0.8)
        if c5_cd4 < best_c5:
            best_c5 = c5_cd4
            best_h = h_cd4.copy()
        if time_left() > 40:
            h_opt4 = slsqp_step(h_cd4, ts_cur, 2000, maxiter=400)
            c5_opt4 = compute_c5(h_opt4)
            if c5_opt4 < best_c5:
                best_c5 = c5_opt4
                best_h = h_opt4.copy()

    # Phase 6: Fine polishing with very small steps
    if time_left() > 80:
        h_fin, c5_fin = coord_desc_ls(best_h, ts_cur, max_iters=150000, K=3, decay=0.9)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Final SLSQP with very high alpha
    for alpha in [50000, 200000]:
        if time_left() < 30:
            break
        h_pol = slsqp_step(best_h, ts_cur, alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final coordinate descent sweep
    if time_left() > 30:
        h_last, c5_last = coord_desc_ls(best_h, ts_cur, max_iters=50000, K=2, decay=1.0)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
