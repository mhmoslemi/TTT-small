import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, threshold=0.995):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    active = np.where(corr >= threshold * max_corr)[0]
    # Convert to lag indices: lag = idx - (n-1)
    lags = active - (n - 1)
    return lags

def fast_smooth_max_and_grad(h, active_lags, alpha):
    n = len(h)
    if len(active_lags) == 0:
        return 0.0, np.zeros(n)
    
    # Compute correlation values at active lags
    active_vals = np.zeros(len(active_lags))
    for idx, k in enumerate(active_lags):
        k = int(k)
        if k == 0:
            active_vals[idx] = np.sum(h * (1.0 - h))
        elif k > 0:
            active_vals[idx] = np.sum(h[k:] * (1.0 - h[:-k]))
        else:
            k2 = -k
            active_vals[idx] = np.sum(h[:-k2] * (1.0 - h[k2:]))
    
    # Softmax
    m = np.max(active_vals)
    shifted = alpha * (active_vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Gradient
    g = np.zeros(n)
    for idx, k in enumerate(active_lags):
        if w[idx] < 1e-15:
            continue
        wi = w[idx]
        k = int(k)
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                g[k:] += wi * (1.0 - h[:-k])
                g[:n-k] -= wi * h[k:]
        else:
            k2 = -k
            if k2 < n:
                g[:n-k2] += wi * (1.0 - h[k2:])
                g[k2:] -= wi * h[:-k2]
    
    return sm_val, g

def adam_optimize(h0, target_sum, n, alphas, max_total_iters, time_limit):
    """Adam optimizer with active set updates"""
    t_start = time.time()
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    
    for alpha in alphas:
        iters_for_alpha = max(100, max_total_iters // len(alphas))
        active_update_freq = 50
        
        for it in range(iters_for_alpha):
            if time.time() - t_start > time_limit:
                break
            
            # Update active set periodically
            if it % active_update_freq == 0:
                active_lags = get_active_lags(h, threshold=0.995)
            
            # Compute gradient
            _, g = fast_smooth_max_and_grad(h, active_lags, alpha)
            
            # Adam update
            m_hat = beta1 * m_hat + (1 - beta1) * g
            v_hat = beta2 * v_hat + (1 - beta2) * g**2
            t = it + 1
            m_hat_unbiased = m_hat / (1 - beta1**t)
            v_hat_unbiased = v_hat / (1 - beta2**t)
            
            h_new = h - lr * m_hat_unbiased / (np.sqrt(v_hat_unbiased) + eps)
            
            # Project to feasible set
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            
            # Check improvement

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, threshold=0.995):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    if max_corr <= 0:
        return np.array([0])
    active = np.where(corr >= threshold * max_corr)[0]
    lags = active - (n - 1)
    return lags

def fast_smooth_max_and_grad(h, active_lags, alpha):
    n = len(h)
    if len(active_lags) == 0:
        return 0.0, np.zeros(n)
    active_vals = np.zeros(len(active_lags))
    for idx in range(len(active_lags)):
        k = int(active_lags[idx])
        if k == 0:
            active_vals[idx] = np.sum(h * (1.0 - h))
        elif k > 0 and k < n:
            active_vals[idx] = np.sum(h[k:] * (1.0 - h[:-k]))
        elif k < 0 and -k < n:
            k2 = -k
            active_vals[idx] = np.sum(h[:-k2] * (1.0 - h[k2:]))
        else:
            active_vals[idx] = 0.0
    m = np.max(active_vals)
    shifted = alpha * (active_vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for idx in range(len(active_lags)):
        if w[idx] < 1e-15:
            continue
        wi = w[idx]
        k = int(active_lags[idx])
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0 and k < n:
            g[k:] += wi * (1.0 - h[:-k])
            g[:n - k] -= wi * h[k:]
        elif k < 0 and -k < n:
            k2 = -k
            g[:n - k2] += wi * (1.0 - h[k2:])
            g[k2:] -= wi * h[:-k2]
    return sm_val, g

def adam_anneal(h0, target_sum, alphas, iters_per_alpha, lr_init, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    global_iter = 0
    for alpha in alphas:
        active_lags = get_active_lags(h, threshold=0.995)
        for it in range(iters_per_alpha):
            if time.time() - t_start > time_limit:
                return best_h, best_c5
            if it % 30 == 0:
                active_lags = get_active_lags(h, threshold=0.995)
            _, g = fast_smooth_max_and_grad(h, active_lags, alpha)
            global_iter += 1
            m_hat = beta1 * m_hat + (1.0 - beta1) * g
            v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
            m_u = m_hat / (1.0 - beta1 ** global_iter)
            v_u = v_hat / (1.0 - beta2 ** global_iter)
            step = m_u / (np.sqrt(v_u) + eps)
            h_new = h - lr * step
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            h = h_new
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h.copy()
        lr *= 0.85
    return best_h, best_c5

def subgradient_anneal(h0, target_sum, alphas, iters_per_alpha, lr_init, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for alpha in alphas:
        h_mom = h.copy()
        for it in range(iters_per_alpha):
            if time.time() - t_start > time_limit:
                return best_h, best_c5
            active_lags = get_active_lags(h, threshold=0.995)
            _, g = fast_smooth_max_and_grad(h, active_lags, alpha)
            h_mom = 0.9 * h_mom + 0.1 * h
            h_new = h_mom - lr * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h_new.copy()
            h = h_new
        lr *= 0.9
    return best_h, best_c5

def coordinate_descent(h0, target_sum, n_rounds, rng, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    base_deltas = np.array([0.001, 0.002, 0.004, 0.008, 0.016])
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            c5_trial = c5_cur
            best_delta = 0.0
            for d in base_deltas:
                for sign in [1, -1]:
                    dd = sign * d
                    if h[i] + dd < 0.0 or h[i] + dd > 1.0:
                        continue
                    h_new = h.copy()
                    h_new[i] = h[i] + dd
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_trial - 1e-14:
                        c5_trial = c5_new
                        best_delta = dd
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = c5_trial
                improved = True
        if not improved:
            break
    return h, c5_cur

def pairwise_push(h0, target_sum, n_rounds, rng, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for _ in range(n // 2):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = h[i] - h[j]
            if abs(d) < 1e-6:
                continue
            h_new = h.copy()
            h_new[i] = np.clip(h[i] - d * 0.3, 0.0, 1.0)
            h_new[j] = np.clip(h[j] + d * 0.3, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h = h_new
                c5_cur = c5_new
                improved = True
    return h, c5_cur

def interpolate_between(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def elapsed():
        return time.time() - t_start
    def remaining():
        return budget_s - elapsed()

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(720, 0.5)

    best_n = len(best_h)
    ts = best_n / 2.0

    best_c5 = compute_c5(best_h)
    h_work = best_h.copy()

    h_work, c5_w = adam_anneal(h_work, ts, alphas_p1, 400, 0.005, budget_s * 0.25, t_start)
    if c5_w < best_c5:
        best_c5 = c5_w
        best_h = h_work.copy()

    # Phase 2: Subgradient with momentum

    h_work, c5_w = subgradient_anneal(h_work, ts, alphas_p2, 300, 0.003, budget_s * 0.40, t_start)
    if c5_w < best_c5:
        best_c5 = c5_w
        best_h = h_work.copy()
    else:
        h_work = best_h.copy()

    # Phase 3: Coordinate descent polish
    h_cd, c5_cd = coordinate_descent(h_work, ts, 20, rng, budget_s * 0.55, t_start)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
    h_work = best_h.copy()

    # Phase 4: Try higher resolution
    for n_hi in [1080, 1440]:
        if remaining() < 120:
            break
        ts_hi = n_hi / 2.0
        h_hi = interpolate_between(best_h, n_hi)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        alphas_hi = [100.0, 300.0, 800.0, 2000.0]
        h_hi, c5_hi = adam_anneal(h_hi, ts_hi, alphas_hi, 300, 0.004, budget_s * 0.85, t_start)
        if c5_hi < best_c5:

            best_h = h_hi
            best_n = n_hi
        alphas_hi2 = [2000.0, 5000.0, 10000.0]

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi
            best_n = n_hi

    # Phase 5: Final coordinate descent at best resolution
    ts_final = len(best_h) / 2.0
    h_cd2, c5_cd2 = coordinate_descent(best_h, ts_final, 40, rng, budget_s * 0.95, t_start)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2

    # Phase 6: Pairwise push
    h_pw, c5_pw = pairwise_push(best_h, len(best_h) / 2.0, 100, rng, budget_s * 0.97, t_start)

    # Final: one more coordinate descent round

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
