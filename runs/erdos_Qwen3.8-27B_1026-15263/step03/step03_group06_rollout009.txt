def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    e = np.exp(shifted)
    s = e.sum()
    w = e / s
    sm = m + np.log(s / len(corr)) / alpha
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for lag in range(1, n):
        wp = w[n-1+lag]
        wn = w[n-1-lag]
        if wp > 1e-18:
            g[lag:] += wp * (1.0 - h[:n-lag])
            g[:n-lag] += wp * (-h[lag:])
        if wn > 1e-18:
            g[:n-lag] += wn * (1.0 - h[lag:])
            g[lag:] += wn * (-h[:n-lag])
    return sm, g

def active_set_obj_grad(h, active_indices):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    am = np.max(corr[active_indices])
    active_mask = corr[active_indices] > am - 1e-10
    if np.sum(active_mask) == 0:
        val = am
        g = np.zeros(n)
        return val, g
    w = np.zeros(len(corr))
    w[active_indices[active_mask]] = 1.0 / np.sum(active_mask)
    val = np.dot(w, corr)
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for lag in range(1, n):
        wp = w[n-1+lag]
        wn = w[n-1-lag]
        if wp > 0:
            g[lag:] += wp * (1.0 - h[:n-lag])
            g[:n-lag] += wp * (-h[lag:])
        if wn > 0:
            g[:n-lag] += wn * (1.0 - h[lag:])
            g[lag:] += wn * (-h[:n-lag])
    return val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol=1e-15):
    def obj(x):
        v, _ = smooth_max_obj_grad(x, alpha)
        return v
    def grad(x):
        _, g = smooth_max_obj_grad(x, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slsqp_active_set(h0, n, target_sum, top_k, maxiter):
    corr = compute_corr(h0)
    m = np.max(corr)
    active = np.where(corr > m - (m+1)*1e-6)[0]
    if len(active) > top_k:
        top_idx = np.argsort(corr)[-top_k:]
        active = top_idx
    def obj(x):
        v, _ = active_set_obj_grad(x, active)
        return v
    def grad(x):
        _, g = active_set_obj_grad(x, active)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_tight(h0, n, target_sum, n_rounds, rng, tol=1e-8):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 1.0)
            hi = min(1.0, h[i] + 1.0)
            h_save = h[i]
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': tol})
                if res.fun < c5_cur - 1e-14:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                h[i] = h_save
        if not improved:
            break
    return h, c5_cur

def lbtg_step(h0, n, target_sum, alpha, n_iters, lr):
    h = h0.copy()
    m = np.zeros(n)
    vm = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        val, g = smooth_max_obj_grad(h, alpha)
        m = 0.9 * m + 0.1 * g
        gm = 0.9 * vm + 0.1 * val
        g_use = m - vm * (gm - val)
        h_new = h - lr * g_use
        h_new = h_new - (np.sum(h_new) - target_sum) / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
        h = h_new
    return best_h, best_c5

def fourier_init(n, n_mode, rng, mean_shift=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5 + mean_shift)
    for j in range(1, n_mode + 1):
        freq = j * np.pi
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(freq * x) + b * np.sin(freq * x)
    target_sum = n / 2.0
    h = project_to_sum(h, target_sum)
    return h
