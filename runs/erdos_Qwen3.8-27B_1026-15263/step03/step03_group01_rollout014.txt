So the first term is the sum of h over a suffix (for k>0) or prefix (for k<0).

The second term is the "autocorrelation" at lag k (over the valid domain).

So C₅ = max_k [S_k - A_k] * dx where S_k is the sum of h over the valid domain, and A_k is the autocorrelation.

To minimize C₅, we want to minimize max_k (S_k - A_k).

Note that S_k for k=0 is sum(h) = N/2. For k>0, S_k = sum(h[k:]) which is ≤ N/2. For k<0, S_k = sum(h[:N-|k|]) which is also ≤ N/2.

To minimize this, we want ∑h[i]² to be as large as possible, which means h should be as "peaky" as possible (close to binary).

For k ≠ 0, the trade-off is different. We want A_k to be as large as possible relative to S_k.

    lr = lr_init
    stuck = 0

        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top-K indices
        if topK > 1:
            top_idx = np.argpartition(-corr, topK)[:topK]
        else:

        grad = np.zeros(N)

            if stuck > 100:

                stuck = 0
        
        h = h_new
        if it % 1000 == 0:
            lr *= 0.95
    
    return best_h, best_c5

def block_perturb(h, N, target, rng, n_blocks=8):
    h_new = h.copy()
    blk = N // n_blocks
    for _ in range(n_blocks // 2):
        i = rng.randint(0, n_blocks // 2)
        j = rng.randint(n_blocks // 2, n_blocks)
        a = i * blk
        b = j * blk
        swap_amt = rng.uniform(0.02, 0.1) * (h[a:a+blk].sum() - h[b:b+blk].sum())
        if 0 < swap_amt < h[a:a+blk].min() * 0.5:
            h_new[a:a+blk] -= swap_amt / blk
            h_new[b:b+blk] += swap_amt / blk
    return project_to_sum(np.clip(h_new, 0, 1), target)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n // 2 * 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

    h_up = np.interp(x_new, x_old, h, period=2.0)
    return project_to_sum(h_up, n_new / 2.0)
