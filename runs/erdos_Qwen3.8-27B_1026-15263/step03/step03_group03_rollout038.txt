def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = m_val + log_sum / alpha
    
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Compute gradient using the B-vector approach
    # grad[j] = sum_m weights[m] * B_j[m]
    # We'll compute this efficiently
    nm1 = n - 1
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-15:
            continue
        # B_j[m] = (1 - h[j+m-n+1]) - h[j-m+n-1] (with validity)
        # First term: valid when n-1-m <= j <= 2n-2-m, intersected with [0, n-1]
        j_lo1 = max(0, nm1 - m)
        j_hi1 = min(nm1, 2 * nm1 - m)
        if j_lo1 <= j_hi1:
            p = j_lo1 + m - nm1
            grad[j_lo1:j_hi1+1] += wm * (1.0 - h[p:j_hi1 + m - nm1 + 1])
        
        # Second term: valid when m-n+1 <= j <= m, intersected with [0, n-1]
        j_lo2 = max(0, m - nm1)
        j_hi2 = min(nm1, m)
        if j_lo2 <= j_hi2:
            q = j_lo2 - m + nm1
            grad[j_lo2:j_hi2+1] -= wm * h[q:j_hi2 - m + nm1 + 1]
    
    return obj, grad

def projected_gradient(h0, target_sum, alpha, max_iter, eta0):
    """Projected gradient descent on smooth max."""
    n = len(h0)
    h = h0.copy()
    eta = eta0
    best_obj = smooth_max_and_grad(h, alpha)[0]
    best_h = h.copy()
    
    for it in range(max_iter):
        obj, grad = smooth_max_and_grad(h, alpha)
        
        # Backtracking line search
        for _ in range(30):
            h_new = h - eta * grad
            h_new = project_to_sum(h_new, target_sum)
            obj_new = smooth_max_and_grad(h_new, alpha)[0]
            if obj_new < obj:
                h = h_new
                break
            eta *= 0.5
        else:
            break
        
        if smooth_max_and_grad(h, alpha)[0] < best_obj:
            best_obj = smooth_max_and_grad(h, alpha)[0]
            best_h = h.copy()
        
        eta = min(eta * 1.1, 0.1)
    
    return best_h

The gradient is: sum_k w_k * grad(corr_k) where w_k are softmax weights.
The Hessian involves: (1/alpha) * sum_k w_k * Hessian(corr_k) + sum_k grad(w_k) * grad(corr_k)^T

The Hessian of corr_k with respect to h is... corr_k is bilinear in h, so its Hessian is bounded. Specifically, corr_k = sum_i h[i]*(1-h[i+k-(n-1)]), so the Hessian has entries that are either 0 or -1 (for the quadratic terms). So ||Hessian(corr_k)|| <= 2 (spectral norm, roughly).

import numpy as np
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_mask = np.sum(mask)
        if n_mask > 0:
            h[mask] += diff / n_mask
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * nm1, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * nm1, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        f_lo = np.max(A + D * lo_r)
        f_mid = np.max(A + D * mid)
        if f_mid <= f_lo:
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j+1) % n, (j-1+n) % n, (j+n//2) % n, (j+n//3) % n, (j+2*n//3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0: h[j] = 0.0
            elif h[j] > 1: h[j] = 1.0
            if h[best_k] < 0: h[best_k] = 0.0
            elif h[best_k] > 1: h[best_k] = 1.0
            improved = True
    return h, corr, improved

def smooth_max_grad(h, alpha):
    """Compute smooth max objective and gradient using FFT acceleration."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = m_val + log_sum / alpha
    wt = np.exp(shifted - log_sum)
    
    # Gradient via FFT
    wt_rev = wt[::-1]
    one_minus_h = 1.0 - h
    S1_full = fftconvolve(one_minus_h, wt_rev, mode='full')
    S2_full = fftconvolve(h, wt_rev, mode='full')
    S1 = S1_full[n-1 : 2*n-1]
    S2 = S2_full[2*n-2 : 3*n-2][::-1]
    grad = S1 - S2
    return obj, grad

def projected_gradient_opt(h0, target_sum, alpha, time_limit, eta0=None):
    """Projected gradient descent on smooth max with time limit."""
    n = len(h0)
    if eta0 is None:
        eta0 = 1.0 / alpha
    h = h0.copy()
    obj, grad = smooth_max_grad(h, alpha)
    best_obj = obj
    best_h = h.copy()
    eta = eta0
    t_start = time.time()
    
    while time.time() - t_start < time_limit:
        h_new = h - eta * grad
        h_new = project_to_sum(h_new, target_sum)
        obj_new, grad_new = smooth_max_grad(h_new, alpha)
        
        if obj_new < obj:
            h = h_new
            obj = obj_new
            grad = grad_new
            eta = min(eta * 1.05, 1.0)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
        else:
            eta *= 0.5
            if eta < 1e-10:
                break
    
    return best_h, best_obj

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:

            best_h = project_to_sum(best_h, best_n / 2.0)

        except Exception:
            pass

        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    def time_left():
        return budget_s - (time.time() - t_start)

        c5_w = compute_c5(h_work)

            best_n = n_w
