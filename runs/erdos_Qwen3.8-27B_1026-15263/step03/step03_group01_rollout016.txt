def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n

def subgradient_worst_lag(h, n):
    """Compute subgradient of the max overlap w.r.t. h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_idx = np.argmax(corr)
    m = m_idx - (n - 1)  # actual lag
    # grad[j] = (1-h[j+m]) * 1[0<=j+m<n] - h[j-m] * 1[j-m>=0]
    grad = np.zeros(n)

        valid = np.arange(0, n - m)
        grad[valid] = 1.0 - h[valid + m]
        # j-m >= 0 means j >= m
        valid2 = np.arange(m, n)

        # j - m = j + m_abs >= 0 always

        grad[valid2] -= h[valid2 - m]

    # = sum_k w[k] for valid - sum_k w[k]*h[j+k]
    
    nj = np.arange(n)[:, None]

    si = nj + nm - n + 1

    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_active = np.sum(mask)
        if n_active > 0:
            h[mask] += diff / n_active
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, m, n, dx):
    grad = np.zeros(n)
    if m >= 0:
        if n - m > 0:
            valid1 = np.arange(0, n - m)
            grad[valid1] = 1.0 - h[valid1 + m]
        if m < n:
            valid2 = np.arange(m, n)
            grad[valid2] -= h[valid2 - m]
    else:
        p = -m
        if p < n:
            valid1 = np.arange(p, n)
            grad[valid1] = 1.0 - h[valid1 - p]
        if n - p > 0:
            valid2 = np.arange(0, n - p)
            grad[valid2] -= h[valid2 + p]
    return grad * dx

def averaged_subgradient(h, n, eps_frac=0.005):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    threshold = m_val - eps_frac * max(1.0, m_val)
    active_idx = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for k_idx in active_idx:
        m = k_idx - (n - 1)
        grad += lag_gradient(h, m, n, dx)
    grad /= len(active_idx)
    return grad, m_val * dx

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def subgrad_momentum(h0, n, ts, n_iter=30000, lr_init=0.005, mu=0.85, eps_frac=0.005):
    h = h0.copy()
    lr = lr_init
    momentum = np.zeros(n)
    best_c5 = compute_c5(h)
    stall = 0
    for i in range(n_iter):
        grad, c5_cur = averaged_subgradient(h, n, eps_frac)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
        momentum = mu * momentum - lr * grad
        h_new = project_to_sum(h + momentum, ts)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_cur + lr * 0.01:
            h = h_new
            if c5_new < best_c5:
                lr = min(lr * 1.02, 0.05)
            else:
                lr = max(lr * 0.99, 1e-7)
        else:
            lr *= 0.5
            momentum *= 0.5
            if lr < 1e-10:
                break
    return h, best_c5

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def targeted_perturb(h, rng, n, wt_lag, amp):
    dx = 2.0 / n
    grad = lag_gradient(h, wt_lag, n, dx)
    norm = np.linalg.norm(grad)

    direction = -grad / norm
    noise = rng.randn(n) * 0.3
    direction = direction + noise / (norm + 1e-10)
    h_new = h + amp * direction
    return project_to_sum(h_new, n / 2.0)

            best_c5 = c5_sg
            best_h = h_sg.copy()
        h_sh = h_sg

        if time_left() > 30:
            h_opt = slsqp_opt(h_sh, n_b, ts_b, 1000, maxiter=500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if time_left() > 60:
        h_topk, c5_topk = subgrad_momentum(best_h, best_n, ts_b, n_iter=25000, lr_init=0.002, mu=0.9, eps_frac=0.02)
        if c5_topk < best_c5:
            best_c5 = c5_topk
            best_h = h_topk.copy()

    if time_left() > 40:
        for alpha in [5000, 15000, 50000, 100000]:
            if time_left() < 25:
                break
            h_f = slsqp_opt(best_h, best_n, ts_b, alpha, maxiter=2000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    if time_left() > 30:
        h_sg_final, c5_final = subgrad_momentum(best_h, best_n, ts_b, n_iter=20000, lr_init=0.001, mu=0.92, eps_frac=0.002)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_sg_final.copy()

    if time_left() > 25:
        h_f2 = slsqp_opt(best_h, best_n, ts_b, 80000, maxiter=3000)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
