    n = len(h)
    corr = compute_corr_full(h)
    if top_k is not None and top_k < len(corr):
        # Only use top_k largest values
        top_indices = np.argsort(corr)[-top_k:]
        corr_use = np.zeros_like(corr)
        corr_use[top_indices] = corr[top_indices]
    else:
        corr_use = corr
    
    m = np.max(corr_use)

    shifted -= np.max(shifted)

def spectral_shake(h, rng, strength, n_modes):

    n = len(h)

    c5 = compute_c5(h)

        corr = compute_corr_full(h)
        worst_idx = int(np.argmax(corr))
        lag = worst_idx - (n - 1)

        js = max(0, -lag)

        je2 = min(n-1, n-1+lag)
        if js2 <= je2:

        # We want to decrease corr[worst_idx], so we want to move in direction -g

        # Change in corr ≈ -g[i] * delta + g[j] * delta = (g[j] - g[i]) * delta

        i_best = int(np.argmax(g_inc))

            h = np.clip(h, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode="full")

        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
        else:
            h[i_best] -= move
            h[j_best] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 80:
                break
    
    return h, c5
