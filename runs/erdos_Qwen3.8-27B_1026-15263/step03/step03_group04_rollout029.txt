def run_lbfgs(h0, n, target_sum, alpha, penalty, maxiter):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    args = (alpha, target_sum, penalty)
    try:
        res = minimize(_obj_fn, h, method='L-BFGS-B', jac=_jac_fn,
                       args=args, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        h = res.x
    except Exception:
        pass
    return project_to_sum(h, target_sum)

def exact_max_polish(h, n, target_sum, max_rounds=500):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    for r in range(max_rounds):
        corr = compute_corr(h)
        m = np.max(corr)
        thresh = m * 0.99
        active_k = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for k in active_k:
            shift = k - (n - 1)
            grad += gradient_at_shift(h, shift)
        grad /= len(active_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        best_c5_local = c5_cur
        best_h_local = h
        for lr in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]:
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5_local - 1e-14:
                best_c5_local = c5_new
                best_h_local = h_new
        if best_h_local is h:
            for lr in [0.01, 0.02, 0.005, 0.001]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5_local - 1e-14:
                    best_c5_local = c5_new
                    best_h_local = h_new
            if best_h_local is h:
                break
        h = best_h_local
        c5_cur = best_c5_local
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def make_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(h_bin))
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])][:diff]
        h_bin[zeros_sorted] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])][:-diff]
        h_bin[ones_sorted] = 0.0
    return h_bin

def binary_sa(h_bin, n, target_sum, t_start, t_end, rounds, rng):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for r in range(rounds):
        temp = t_start * (t_end / t_start) ** (float(r) / float(rounds))
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        accept = c5_new < c5_cur
        if not accept and temp > 1e-8:
            delta = c5_new - c5_cur
            if delta < 2.0 * temp:
                accept = rng.random() < np.exp(-delta / temp)
        if accept:
            c5_cur = c5_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def binary_greedy_polish(h_bin, n, target_sum, max_rounds=200):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        worst_k = np.argmax(corr)
        worst_shift = worst_k - (n - 1)
        grad = gradient_at_shift(h, worst_shift)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ones_scored = ones[np.argsort(-grad[ones])][:30]
        zeros_scored = zeros[np.argsort(grad[zeros])][:30]
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for ii in range(len(ones_scored)):
            i = ones_scored[ii]
            for jj in range(len(zeros_scored)):
                j = zeros_scored[jj]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < c5_cur - best_improve:
                    best_improve = c5_cur - c5_new
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5_cur = c5_cur - best_improve
        else:
            break
    return h, c5_cur

def continuous_to_binary_refine(h_cont, n, target_sum, rng, elapsed_check, budget_s, t0):
    h_bin = make_binary(h_cont, n)
    h_bin, c5_bin = binary_greedy_polish(h_bin, n, target_sum, max_rounds=100)
    h_sa, c5_sa = binary_sa(h_bin, n, target_sum, 0.05, 0.001, 500, rng)
    if c5_sa < c5_bin:
        h_bin = h_sa
        c5_bin = c5_sa
    h_bin, c5_bin2 = binary_greedy_polish(h_bin, n, target_sum, max_rounds=100)
    if c5_bin2 < c5_bin:
        c5_bin = c5_bin2
    return h_bin, c5_bin

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0

    # Phase 1: Continuation at current resolution
    alphas_p1 = [20, 100, 500, 2000, 5000, 10000, 20000, 50000]
    for a in alphas_p1:
        if budget_s - (time.time() - t0) < 80:
            break
        h_opt = run_lbfgs(best_h, n1, ts1, a, 1e6, maxiter=200)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: Exact max polish
    if budget_s - (time.time() - t0) > 60:
        h_pol, c5_pol = exact_max_polish(best_h, n1, ts1, max_rounds=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 3: Binary refinement cycle 1
    if budget_s - (time.time() - t0) > 150:
        h_bin, c5_bin = continuous_to_binary_refine(best_h, n1, ts1, rng, budget_s, t0)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish from binary
        for a in [500, 5000, 20000]:
            if budget_s - (time.time() - t0) < 40:
                break
            h_opt = run_lbfgs(best_h, n1, ts1, a, 1e6, maxiter=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        h_pol, c5_pol = exact_max_polish(best_h, n1, ts1, max_rounds=300)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Binary refinement cycle 2 (from updated continuous)
    if budget_s - (time.time() - t0) > 150:
        h_bin2, c5_bin2 = continuous_to_binary_refine(best_h, n1, ts1, rng, budget_s, t0)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        for a in [2000, 10000, 50000]:
            if budget_s - (time.time() - t0) < 30:
                break
            h_opt = run_lbfgs(best_h, n1, ts1, a, 1e7, maxiter=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        h_pol, c5_pol = exact_max_polish(best_h, n1, ts1, max_rounds=300)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 5: Diverse binary starts
    if budget_s - (time.time() - t0) > 250:
        x = np.linspace(0, 2, n1, endpoint=False)
        diverse_starts = []
        for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
            for amp in [0.3, 0.4, 0.45]:
                for ph in [0.0, np.pi / 3, 2.0 * np.pi / 3.0, np.pi]:
                    s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + ph)
                    diverse_starts.append(project_to_sum(s, ts1))
        for nblks in [3, 4, 5, 6, 8, 10, 12]:
            blk = n1 // nblks
            hb = np.zeros(n1)
            for i in range(0, n1, 2 * blk):
                hb[i:min(i + blk, n1)] = 1.0
            diverse_starts.append(project_to_sum(hb, ts1))

        n_diverse = len(diverse_starts)
        max_trials = min(n_diverse, max(1, int((budget_s - (time.time() - t0)) // 60)))
        for di in range(max_trials):
            if budget_s - (time.time() - t0) < 50:
                break
            h_s = diverse_starts[di]
            h_s_bin = make_binary(h_s, n1)
            h_s_bin, c5_s = binary_greedy_polish(h_s_bin, n1, ts1, max_rounds=50)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s_bin.copy()

    # Phase 6: Higher resolution
    if budget_s - (time.time() - t0) > 300:
        n2 = n1 * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        for a in [100, 1000, 5000, 20000]:
            if budget_s - (time.time() - t0) < 60:
                break
            h_opt = run_lbfgs(h_interp, n2, ts2, a, 1e6, maxiter=150)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
                h_interp = h_opt
        if budget_s - (time.time() - t0) > 60:
            h_pol, c5_pol = exact_max_polish(best_h, best_n, best_n / 2.0, max_rounds=300)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 7: Final extreme polish
    if budget_s - (time.time() - t0) > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        h_pol, c5_pol = exact_max_polish(best_h, n_f, ts_f, max_rounds=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        for a in [20000, 50000, 100000]:
            if budget_s - (time.time() - t0) < 20:
                break
            h_opt = run_lbfgs(best_h, n_f, ts_f, a, 1e8, maxiter=300)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        h_pol2, c5_pol2 = exact_max_polish(best_h, n_f, ts_f, max_rounds=300)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
