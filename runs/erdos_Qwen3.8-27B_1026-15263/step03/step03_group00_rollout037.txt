def smooth_max_obj_and_grad(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    n = len(h)
    grad = np.zeros(n)
    # For each lag m, the gradient contribution is w[m] * d(corr[m])/dh
    # corr[m] = sum_i h[i+m'] * (1-h[i]) where indices are appropriate
    # d(corr[m])/dh[j] = (1-h[j]) if j appears in the first factor

Let me think about how to handle the sum constraint with L-BFGS-B. The cleanest way is to parameterize:
h[n-1] = target_sum - sum(h[0:n-1])

Then the bounds on h[n-1] give: 0 ≤ target_sum - sum(h[0:n-1]) ≤ 1
Which means: target_sum - 1 ≤ sum(h[0:n-1]) ≤ target_sum

This is a linear inequality constraint on the sum, which L-BFGS-B can't handle directly.

Alternative: Use a penalty term for the sum constraint:
L(h) = smooth_max(corr(h)) + λ * (sum(h) - target_sum)²

With large λ, this approximately enforces the constraint. Then use L-BFGS-B with bounds [0,1].

λ = (sum(∇smooth_max) - 0) / n ... no, that's not right.

Actually, the KKT condition for the sum constraint is:
∇f(h) + λ * 1 + μ = 0 (where μ is for the bound constraints)

For the active set (coordinates not at bounds): ∇f(h)_j + λ = 0 for all j in active set
So λ = -mean(∇f(h)[active_set])

Then the effective gradient is: ∇f(h) - mean(∇f(h)[active_set]) * 1[active_set]

This is the "projected gradient" step. Let me implement this.

2. Project the gradient onto the tangent space of the constraint (subtract the mean of the gradient on the active set)

    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    """Compute gradient as weighted sum of top-k correlation gradients."""
    n = len(h)
    grad = np.zeros(n)
    for lag in k_lags:
        if lag >= 0:
            if lag == 0:
                grad[:] += (1.0 - 2.0 * h)
            else:
                grad[lag:] += (1.0 - h[:n-lag])
                grad[:n-lag] -= h[lag:]
        else:
            k = -lag
            if k == 0:
                grad[:] += (1.0 - 2.0 * h)
            else:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
    return grad

def smooth_max_fast(h, alpha, top_k=20):
    """Compute smooth max and gradient using only top-k correlations."""
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Find top-k lags
    n = len(h)
    top_indices = np.argsort(w)[-top_k:]
    top_indices = top_indices[np.argsort(w[top_indices])[::-1]]
    
    grad = np.zeros(n)
    for idx in top_indices:
        wi = w[idx]
        if wi < 1e-12:
            break
        lag = idx - (n - 1)
        if lag >= 0:

                grad[:] += wi * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wi * (1.0 - h[:n-lag])
                grad[:n-lag] -= wi * h[lag:]
        else:
            k = -lag

                grad[:] += wi * (1.0 - 2.0 * h)
            else:
                grad[:n-k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:n-k]
    return obj, grad

def projected_gradient_step(h, grad, target_sum, step_size):
    """Take a projected gradient step."""
    n = len(h)
    # Project gradient onto tangent space (sum constraint)
    active = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(active) > 0:
        grad_projected = grad.copy()
        grad_projected[active] -= np.mean(grad[active])
    else:
        grad_projected = grad
    
    h_new = h - step_size * grad_projected
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def accelerated_gradient(h0, target_sum, alpha, n_iter, top_k=20):
    """Nesterov accelerated projected gradient method."""
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    
    for t in range(n_iter):
        obj, grad = smooth_max_fast(y, alpha, top_k)
        tt = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / 2.0 if t > 0 else 1.0
        # Actually, standard Nesterov: t_t = (1 + sqrt(1+4*t^2))/2
        t_t = (1.0 + np.sqrt(1.0 + 4.0 * (t+1)**2)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * t**2)) / 2.0 if t > 0 else 1.0
        
        step = 2.0 / (t_t + 1.0)
        y_new = projected_gradient_step(y, grad, target_sum, step)
        
        beta = (t_prev - 1.0) / t_t
        h_new = (1.0 - beta) * y_new + beta * h
        
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_obj:
            best_obj = c5
            best_h = h_new.copy()
        
        h = h_new
        y = (1.0 + 1.0/t_t) * h - (1.0/t_t) * h  # This is just h
        # Actually, Nesterov: y_{t+1} = h_{t+1} + (t_t/(t_t+1))(h_{t+1} - h_t)
        y = h_new + (t_t / (t_t + 1.0)) * (h_new - h0)  # This is wrong
        
        h = h_new
    
    return best_h, best_obj

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Resolution escalation
    for n_new in [600, 720, 900, 1080]:
        if time_left() < 120:
            break
        target_new = n_new / 2.0
        h_interp = interp_to_n(best_h, n_new)
        h_interp = project_to_sum(h_interp, target_new)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_new
        for alpha in [200, 2000, 10000]:
            if time_left() < 50:
                break
            h_opt = slsqp_optimize(h_interp, n_new, target_new, alpha, maxiter=400, top_k=40)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_new
            h_interp = h_opt
        if time_left() > 50:
            h_cd, c5_cd = multi_lag_coord_desc(h_interp, target_new, alpha=500, max_iters=20000, top_k=5, step_frac=0.04)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_new

    # Phase 5: Perturbation restarts at best resolution
    target = best_n / 2.0
    n_trials = 8
    for trial in range(n_trials):
        if time_left() < 60:
            break
        noise_level = 0.02 + 0.03 * (1.0 - trial / float(n_trials))
        h_shake = best_h + rng.normal(0, noise_level, best_n)
        h_shake = project_to_sum(h_shake, target)
        h_sym = enforce_symmetry(h_shake)
        h_sym = project_to_sum(h_sym, target)
        h_opt = slsqp_optimize(h_sym, best_n, target, 5000, maxiter=400, top_k=40)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time_left() < 60:
            break
        h_cd, c5_cd = multi_lag_coord_desc(h_opt, target, alpha=500, max_iters=15000, top_k=5, step_frac=0.04)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Block coordinate descent
    if time_left() > 80:
        h_blk, c5_blk = block_coord_desc(best_h, target, alpha=1000, block_size=12, max_iters=5000, top_k=5)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk.copy()

    # Phase 7: Final high-alpha SLSQP polish
    for alpha in [50000, 200000, 1000000]:
        if time_left() < 30:
            break
        h_pol = slsqp_optimize(best_h, best_n, target, alpha, maxiter=1000, top_k=60)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Final coordinate descent with annealing
    if time_left() > 30:
        h_ann, c5_ann = coord_desc_anneal(best_h, target, alpha=2000, max_iters=30000, top_k=7, step_frac=0.03, t_anneal=0.5)
        if c5_ann < best_c5:
            best_c5 = c5_ann
            best_h = h_ann.copy()

    # Phase 9: Ultra-fine coordinate descent
    if time_left() > 20:
        h_fin, c5_fin = multi_lag_coord_desc(best_h, target, alpha=5000, max_iters=50000, top_k=10, step_frac=0.015)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
