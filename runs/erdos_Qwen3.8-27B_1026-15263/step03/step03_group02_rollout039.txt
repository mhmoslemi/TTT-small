import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    """For binary h, returns the max corr value (integer)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(np.max(corr))

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def parallel_tempering(h_init, n, target_sum, t_end, rng, n_chains=8):
    """Run parallel tempering with single-swap moves."""
    ts = int(target_sum)
    
    # Initialize chains
    chains = [h_init.copy() for _ in range(n_chains)]
    # Diversify chains
    for c in range(1, n_chains):
        for _ in range(500 * c):
            i = rng.randint(n)
            if chains[c][i] > 0.5:
                zeros = np.where(chains[c] < 0.5)[0]
                if len(zeros) > 0:
                    j = zeros[rng.randint(len(zeros))]
                    chains[c][i], chains[c][j] = chains[c][j], chains[c][i]
    
    # Temperatures (geometric spacing)
    T_max = 0.05
    T_min = 0.002
    temps = np.geomspace(T_max, T_min, n_chains)
    
    best_h = None
    best_mc = int(np.max(np.correlate(h_init, 1-h_init, mode="full")))
    best_c5 = best_mc * 2.0 / n
    
    # Compute initial max_corr for each chain

    for c in range(n_chains):
        mc = int(np.max(np.correlate(chains[c], 1-chains[c], mode="full")))
        max_corrs.append(mc)
        if mc < best_mc:
            best_mc = mc
            best_h = chains[c].copy()
            best_c5 = mc * 2.0 / n
    
    swap_interval = 50  # swap between neighboring chains every 50 iterations
    
    while time.time() < t_end:
        for c in range(n_chains):
            i = rng.randint(n)
            j = rng.randint(n)
            if chains[c][i] == chains[c][j]:
                continue
            chains[c][i], chains[c][j] = chains[c][j], chains[c][i]
            new_mc = int(np.max(np.correlate(chains[c], 1-chains[c], mode="full")))
            delta = new_mc - max_corrs[c]
            
            if delta <= 0 or rng.uniform() < np.exp(-delta / temps[c]):
                max_corrs[c] = new_mc
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_h = chains[c].copy()
                    best_c5 = mc * 2.0 / n
            else:

            c = rng.randint(n_chains - 1)

    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]

    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(np.max(corr))

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h_cont, n, target_ones, rng):
    h_bin = (h_cont > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_ones:
        return h_bin
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h_cont[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h_cont[zeros_idx])[:deficit]

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad
