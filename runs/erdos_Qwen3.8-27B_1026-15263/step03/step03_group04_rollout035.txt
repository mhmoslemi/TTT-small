```python
    for it in range(max_iters):
        corr = compute_corr(h)
        m = np.max(corr)
        if m * (2.0/n) < c5_cur - 1e-10:
            c5_cur = m * (2.0/n)
        
        # Active set: top m correlations
        idx = np.argsort(corr)[::-1][:top_m]
        
        # Compute gradient as weighted sum of active gradients
        alpha = 100.0  # Sharpen the weights
        w = np.exp(alpha * (corr[idx] - corr[idx[0]]))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for mi, k in enumerate(idx):
            wi = w[mi]
            shift = k - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
        
        # Polyak step size: (f(x) - f*) / ||grad||^2
        # We don't know f*, so use (f(x) - target) / ||grad||^2
        f_cur = m  # in correlation units (not yet multiplied by dx)
        step = (f_cur - target_C) / (np.dot(grad, grad) + 1e-10)
        step = min(step, 0.1)  # Cap the step
        
        h_new = h - step * grad
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            # Slowly decrease target
            target_C = m * (2.0/n) - 1e-4
        else:
            # Reduce step and try again
            h_try = h - step * 0.1 * grad
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h = h_try
                c5_cur = c5_try
            else:
                target_C = max(target_C - 1e-5, 0)
    
    return h, c5_cur

def binary_delta_swap(h, corr, i, j, n):
    """Compute the change in all correlations when h[i]:0->1 and h[j]:1->0"""
    # For each lag L (shift = k - (n-1)), compute delta
    # corr[k] = sum_{valid i} h[i] * (1 - h[i+L])
    # where L = k - (n-1), valid means 0<=i<n and 0<=i+L<n
    
    delta = np.zeros(2*n - 1)
    
    for L in range(-(n-1), n):
        # Effect of h[i] going 0->1:
        # - i contributes (1-h[i+L]) if i and i+L are valid
        # - (i-L) contributes -h[i-L] if i-L and i are valid (i.e., i-L valid and (i-L)+L=i valid)
        
        # i as first index: valid if 0<=i<n and 0<=i+L<n
        v1 = (0 <= i < n) and (0 <= i+L < n)
        # i-L as first index, i as second: valid if 0<=i-L<n and 0<=i<n (i.e., (i-L)+L=i valid)
        v2 = (0 <= i-L < n) and (0 <= i < n)
        
        # j as first index: valid if 0<=j<n and 0<=j+L<n
        v3 = (0 <= j < n) and (0 <= j+L < n)
        # j-L as first index, j as second: valid if 0<=j-L<n and 0<=j<n
        v4 = (0 <= j-L < n) and (0 <= j < n)
        
        d = 0.0
        if v1:
            d += (1.0 - h[i+L])  # h[i] goes from 0 to 1
        if v2:
            d -= h[i-L]  # (1-h[i]) goes from 1 to 0
        if v3:
            d -= (1.0 - h[j+L])  # h[j] goes from 1 to 0
        if v4:
            d += h[j-L]  # (1-h[j]) goes from 0 to 1
        
        delta[L + (n-1)] = d
    
    return delta

def binary_local_search(h, n, target_sum, rng, max_rounds=2000):
    """Local search on binary representation with fast delta evaluation"""
    h = h.copy()
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Fix sum
    while abs(h.sum() - target_sum) > 1e-10:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1.0
    
    corr = compute_corr(h)
    c5_cur = np.max(corr) * (2.0/n)
    
    for r in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        n_trials = min(200, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(zeros)  # 0 -> 1
            j = rng.choice(ones)   # 1 -> 0
            
            if i == j:
                continue
            
            delta = binary_delta_swap(h, corr, i, j, n)
            new_corr = corr + delta
            new_max = np.max(new_corr)
            
            if new_max < np.max(corr) - 1e-10:
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                c5_cur = new_max * (2.0/n)
                improved = True
                break
        
        if not improved:
            break
    
    return h, c5_cur
```

Actually, I can vectorize the delta computation. Let me think...

For a given swap (i: 0→1, j: 1→0), the delta at lag L is:
delta[L] = (1-h[i+L])*v1 - h[i-L]*v2 - (1-h[j+L])*v3 + h[j-L]*v4

where v1, v2, v3, v4 are validity indicators.

This can be vectorized: for all L at once.

```python
def binary_delta_fast(h, i, j, n):
    L = np.arange(-(n-1), n)  # All lags
    i_plus_L = i + L
    i_minus_L = i - L
    j_plus_L = j + L
    j_minus_L = j - L
    
    v1 = (i_plus_L >= 0) & (i_plus_L < n)
    v2 = (i_minus_L >= 0) & (i_minus_L < n)
    v3 = (j_plus_L >= 0) & (j_plus_L < n)
    v4 = (j_minus_L >= 0) & (j_minus_L < n)
    
    delta = np.zeros(2*n-1)
    delta[v1] += (1.0 - h[i_plus_L[v1]])
    delta[v2] -= h[i_minus_L[v2]]
    delta[v3] -= (1.0 - h[j_plus_L[v3]])
    delta[v4] += h[j_minus_L[v4]]
    
    return delta
```

This is vectorized and should be fast. Each call is O(n) but with good constant factor due to vectorization.

With this, 50 trials per round * 1000 rounds = 50,000 calls * O(480) ≈ 24 million operations. Should be a few seconds.

Let me also think about whether I should try a different approach entirely. 

**Alternative: Formulate as a sequence of QP subproblems**

At each iteration:
1. Compute current max correlation C_cur and active set A = {k : corr[k] > C_cur - epsilon}
2. Solve: min C s.t. corr[k] ≤ C for k in A, sum(h) = n/2, 0 ≤ h ≤ 1
3. This is a QP (quadratic objective and constraints due to h*h terms)

But solving QPs with quadratic constraints is expensive. The smooth max approach is essentially a surrogate for this.

**Another alternative: Use the structure of the problem**

The correlation at lag L is:
corr[L] = sum_i h[i] - sum_i h[i]h[i+L] (for valid i)

Let S = sum_i h[i] = n/2 (fixed).
Then corr[L] = n/2 - sum_i h[i]h[i+L]

So minimizing max_L corr[L] is equivalent to minimizing max_L (n/2 - sum_i h[i]h[i+L])
= n/2 - min_L sum_i h[i]h[i+L]

Wait, that's not right. We want to minimize max_L corr[L] = max_L (n/2 - sum_i h[i]h[i+L])
= n/2 - min_L sum_i h[i]h[i+L]

So we want to **maximize** min_L sum_i h[i]h[i+L]!

This is the minimum of the autocorrelation (at various lags). We want to make the autocorrelation as uniformly high as possible.

This is an interesting reformulation. The autocorrelation R(L) = sum_i h[i]h[i+L] (for valid i).

For a binary h with n/2 ones:
R(L) = number of pairs (i, i+L) where both are 1.

We want to maximize min_L R(L).

This is equivalent to: find a subset A of {0,...,n-1} of size n/2 such that the minimum number of "L-returns" (pairs in A at distance L) is maximized.

This is related to combinatorial design and difference sets!

A (v, k, λ) difference set has the property that every non-zero element is represented exactly λ times as a difference of two elements in the set. This means the autocorrelation is constant for all non-zero lags.

If such a set exists with v=2n, k=n, then we'd achieve the perfect solution where all non-zero lag correlations are equal.

For v=2n, k=n: we need λ = k(k-1)/(v-1) = n(n-1)/(2n-1). This is integer only for specific values of n.

For n=7: λ = 7*6/13 = 42/13 — not integer.
For n=3: λ = 3*2/5 = 6/5 — not integer.
For n=13: λ = 13*12/25 = 156/25 — not integer.

Hmm, so perfect difference sets don't exist for most sizes. But we can get close.

Actually, for the boundary issue (correlations at the edges have fewer terms), this is more complicated. The "valid" range for lag L has min(n, n-|L|) terms. So R(0) has n terms, R(1) has n-1 terms, etc.

This makes the problem more complex than a pure difference set problem.

**Practical approach**: Let me focus on the optimization and try to squeeze out the last bit of improvement.

Key insight from the reformulation: we want to maximize min_L R(L) where R(L) = sum_{valid i} h[i]h[i+L].

For the continuous problem, this is: maximize min_{L∈[0,2]} ∫ h(x)h(x+L) dx (with appropriate domain restrictions).

This is a "max-min" problem, which is dual to the original "min-max" problem.

Let me try a different optimization approach: instead of minimizing max corr, try to **equalize** the correlations. If all correlations are equal, then the max is minimized (given the constraints).

**Equalization approach**:
1. Compute all correlations
2. Find the mean correlation
3. For lags above mean, try to reduce them; for lags below mean, try to increase them
4. This is like a "balancing" problem

But this is essentially what the smooth max does when alpha is small (it approximates the mean). As alpha increases, it focuses more on the maximum.

I think the best strategy is:
1. Use a very fine continuation (many alpha values from low to high)
2. At each step, use L-BFGS-B with many iterations
3. At the end, use a subgradient method with very small steps
4. Try multiple resolutions

Let me also add: **momentum-based subgradient**

Instead of vanilla subgradient, use Nesterov's accelerated gradient or Adam-like momentum:
- Maintain a velocity vector v
- v = beta * v + (1-beta) * grad
- h = h - lr * v

This can help escape shallow local minima.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = int(np.sum(mask))
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def lbfgs_optimize(h, n, target_sum, alpha, maxiter=500):
    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 1e5 * s_dev * s_dev
    
    def jac(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + 2e5 * s_dev * np.ones(n)
    
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_to_sum(res.x, target_sum)
    except:
        return h

def subgradient_with_momentum(h, n, target_sum, max_iters=5000, top_m=5, momentum=0.9):
    h = h.copy()
    c5_cur = compute_c5(h)
    v = np.zeros(n)
    base_lr = 0.05
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m = np.max(corr)
        
        # Top-m weighted gradient
        idx = np.argsort(corr)[::-1][:top_m]
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[idx] - corr[idx[0]]))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for mi, k in enumerate(idx):
            wi = w[mi]
            shift = k - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        lr = base_lr / (1 + it * 0.01)
        
        h_new = h - lr * v
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
        else:
            # Try smaller step
            h_try = h - lr * 0.1 * v
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h = h_try
                c5_cur = c5_try
            else:
                v *= 0.5  # Damp momentum
    
    return h, c5_cur

def binary_delta_fast(h, i, j, n):
    L = np.arange(-(n-1), n)
    iL = i + L
    i_mL = i - L
    jL = j + L
    j_mL = j - L
    
    v1 = (iL >= 0) & (iL < n)
    v2 = (i_mL >= 0) & (i_mL < n)
    v3 = (jL >= 0) & (jL < n)

    if np.any(v3):

def binary_search(h_start, n, target_sum, rng, max_rounds=2000, trials_per_round=100):

    # Fix sum
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros)] = 1.0
        else:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones)] = 0.0
    
    corr = compute_corr(h)
    c5_cur = np.max(corr) * (2.0 / n)
    
    for r in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for _ in range(trials_per_round):
            i = rng.choice(zeros)
            j = rng.choice(ones)
            if i == j:
                continue
            delta = binary_delta_fast(h, int(i), int(j), n)
            new_corr = corr + delta
            new_max = np.max(new_corr)
            if new_max < np.max(corr) - 1e-12:
                h[int(i)] = 1.0
                h[int(j)] = 0.0
                corr = new_corr
                c5_cur = new_max * (2.0 / n)
                improved = True
                break
        
        if not improved:
            break
    
    return h, c5_cur
```
