def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):

def slsqp_opt(h0, n, ts, func, maxiter=1000, ftol=1e-15):

        ]
        bnds = [(0.0, 1.0)] * n
        r = minimize(func, h0.copy(), method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        if r.success or r.fun < float('inf'):
            return project_to_sum(r.x, ts)
        return project_to_sum(h0, ts)
    except Exception:
        return project_to_sum(h0, ts)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

                h = project_to_sum(h, ts)

import numpy as np
from scipy.optimize import minimize
import time

_params = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s = np.sum(exp_v)
    return m_val + (max_sh + np.log(s)) / alpha

def smooth_max_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    nl = 2 * n - 1
    nj = np.arange(n)[:, None]
    lag_arr = np.arange(nl)[None, :] - (n - 1)
    j_minus_L = nj - lag_arr
    valid1 = (j_minus_L >= 0) & (j_minus_L < n)
    j_plus_L = nj + lag_arr
    valid2 = (j_plus_L >= 0) & (j_plus_L < n)
    jmc = np.clip(j_minus_L, 0, n - 1)
    jpc = np.clip(j_plus_L, 0, n - 1)
    grad = np.sum(valid1 * (1.0 - x[jmc]) * w[None, :] - valid2 * x[jpc] * w[None, :], axis=1)
    return grad

def _s_smooth_obj(x):
    return smooth_max_obj(x, _params['alpha'])

def _s_smooth_grad(x):
    return smooth_max_grad(x, _params['alpha'])

def _sum_con_fun(x):
    return np.sum(x) - _params['ts']

def _sum_con_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, n, ts, alpha, maxiter=1500):
    _params['alpha'] = alpha
    _params['ts'] = ts
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_s_smooth_obj, h0.copy(), method='SLSQP', jac=_s_smooth_grad,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13})
        if np.all(np.isfinite(r.x)):
            return project_to_sum(r.x, ts)
    except Exception:
        pass
    return project_to_sum(h0.copy(), ts)

def active_lag_obj(x, top_m, alpha_lm):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    top_idx = np.argsort(corr)[-top_m:]
    top_corr = corr[top_idx]
    shifted = alpha_lm * (top_corr - top_corr.max())
    exp_v = np.exp(shifted - shifted.max())
    w = exp_v / exp_v.sum()
    return float(np.dot(w, top_corr))

def active_lag_grad(x, top_m, alpha_lm):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    top_idx = np.argsort(corr)[-top_m:]
    top_corr = corr[top_idx]
    shifted = alpha_lm * (top_corr - top_corr.max())
    exp_v = np.exp(shifted - shifted.max())
    w_top = exp_v / exp_v.sum()
    grad = np.zeros(n)
    for m in range(top_m):
        k_idx = top_idx[m]
        L = k_idx - (n - 1)
        wt = w_top[m]
        j = np.arange(n)
        j_mL = j - L
        v1 = (j_mL >= 0) & (j_mL < n)
        j_pL = j + L
        v2 = (j_pL >= 0) & (j_pL < n)
        jmc = np.clip(j_mL, 0, n - 1)
        jpc = np.clip(j_pL, 0, n - 1)
        grad += wt * (v1 * (1.0 - x[jmc]) - v2 * x[jpc])
    return grad

def slsqp_active(h0, n, ts, top_m, alpha_lm, maxiter=1000):
    _params['ts'] = ts
    _params['top_m'] = top_m
    _params['alpha_lm'] = alpha_lm
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    bnds = [(0.0, 1.0)] * n

    def _act_obj(x):
        return active_lag_obj(x, top_m, alpha_lm)

    def _act_grad(x):
        return active_lag_grad(x, top_m, alpha_lm)

    try:
        r = minimize(_act_obj, h0.copy(), method='SLSQP', jac=_act_grad,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        if np.all(np.isfinite(r.x)):
            return project_to_sum(r.x, ts)
    except Exception:
        pass
    return project_to_sum(h0.copy(), ts)

def proj_grad_descent(h0, n, ts, alpha, n_iter=5000, lr_init=0.03):
    h = h0.copy()
    lr = lr_init
    obj_prev = smooth_max_obj(h, alpha)
    stall = 0
    for i in range(n_iter):
        grad = smooth_max_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new = smooth_max_obj(h_new, alpha)
        if obj_new < obj_prev - 1e-18:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 150 or lr < 1e-14:
                break
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def block_swap_move(h, rng, max_block_frac=0.15):
    n = len(h)
    max_bs = max(5, int(n * max_block_frac))
    bs = rng.randint(5, min(max_bs, n // 3) + 1)
    i = rng.randint(0, n - 2 * bs + 1)
    j = i + bs + rng.randint(1, n - 2 * bs - bs + 1)
    if j + bs > n:
        j = n - bs
    if abs(i - j) <= bs:
        return h
    h_new = h.copy()
    tmp = h_new[i:i + bs].copy()
    h_new[i:i + bs] = h_new[j:j + bs]
    h_new[j:j + bs] = tmp
    return h_new

def window_perturb(h, rng, window_range=(20, 100)):
    n = len(h)
    w = rng.randint(window_range[0], min(window_range[1], n // 2) + 1)
    start = rng.randint(0, n - w)
    h_new = h.copy()
    h_new[start:start + w] += rng.normal(0, 0.1, w)
    return project_to_sum(h_new, np.sum(h.new) if False else np.sum(h))

def stochastic_refine(h, n, ts, rng, n_steps, delta_max, temp_init):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = temp_init
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] < h[j]:
            i, j = j, i
        max_d = h[i] - h[j]
        if max_d < 1e-5:
            continue
        delta = rng.uniform(0.003, min(delta_max, max_d))
        new_i = h[i] - delta
        new_j = h[j] + delta
        if new_i < 0 or new_j > 1:
            delta = min(delta, h[i] * 0.5, (1.0 - h[j]) * 0.5)
            if delta < 1e-6:
                continue
            new_i = h[i] - delta
            new_j = h[j] + delta
        saved_i = h[i]
        saved_j = h[j]
        h[i] = new_i
        h[j] = new_j
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            if np.exp(-(c5_new - c5_cur) / max(temp, 1e-15)) < rng.random():
                h[i] = saved_i
                h[j] = saved_j
                h = project_to_sum(h, ts)
        temp = max(1e-7, temp_init * np.exp(-step / (n_steps * 0.8)))
    return h, c5_cur

def greedy_coord_descent(h, n, ts, n_sweeps=5, relax=0.3):
    h = h.copy()
    c5_cur = compute_c5(h)
    for sweep in range(n_sweeps):
        idx = np.arange(n)
        rng_local = np.random.RandomState(sweep * 1000 + 42)
        rng_local.shuffle(idx)
        for pos in range(n):
            i = idx[pos]
            h_old_i = h[i]
            best_t = h_old_i
            best_c5 = c5_cur
            for t in np.linspace(max(0, h_old_i - relax), min(1, h_old_i + relax), 11):
                if abs(t - h_old_i) < 1e-10:
                    continue
                h[i] = t
                h = project_to_sum(h, ts)
                c5_try = compute_c5(h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_t = t
                h[i] = h_old_i
            h[i] = best_t
            h = project_to_sum(h, ts)
            c5_cur = compute_c5(h)
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    c5_480 = best_c5
    h_480 = best_h.copy()

    # Phase 1: Upsample to n=960
    n_target = 960
    ts_target = n_target / 2.0
    h_cur = upsample_h(best_h, n_target)
    h_cur = project_to_sum(h_cur, ts_target)
    n_cur = n_target
    ts_cur = ts_target

    # Phase 2: Multi-alpha SLSQP at n=960
    alphas = [30, 80, 200, 500, 1500, 4000, 8000]
    for alpha in alphas:
        if time_left() < 50:
            break
        h_opt = slsqp_smooth(h_cur, n_cur, ts_cur, alpha, maxiter=2000)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_cur
        h_cur = h_opt

    # Phase 3: Active-lag optimization
    for trial in range(12):
        if time_left() < 35:
            break
        top_m = rng.randint(3, 9)
        alpha_lm = rng.choice([30, 60, 100, 200])
        h_act = slsqp_active(h_cur, n_cur, ts_cur, top_m, alpha_lm, maxiter=1200)
        c5_act = compute_c5(h_act)
        if c5_act < best_c5:
            best_c5 = c5_act
            best_h = h_act.copy()
            best_n = n_cur
        h_cur = h_act

    # Phase 4: Projected gradient
    if time_left() > 60:
        h_pg = proj_grad_descent(h_cur, n_cur, ts_cur, alpha=500, n_iter=8000, lr_init=0.02)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n_cur
        h_cur = h_pg

    # Phase 5: Block swap exploration
    for trial in range(25):
        if time_left() < 25:
            break
        h_sw = block_swap_move(h_cur, rng)
        h_sw = project_to_sum(h_sw, ts_cur)
        h_opt = slsqp_smooth(h_sw, n_cur, ts_cur, 200, maxiter=800)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_cur
            h_cur = h_opt

    # Phase 6: Heavy stochastic refinement
    if time_left() > 120:
        n_steps = min(100000, int((time_left() - 100) / 0.001))
        n_steps = max(n_steps, 40000)
        h_st, c5_st = stochastic_refine(h_cur, n_cur, ts_cur, rng, n_steps, 0.25, 0.15)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            best_n = n_cur
            h_cur = h_st
        # Polish after stochastic
        if time_left() > 40:
            h_pol = slsqp_smooth(h_cur, n_cur, ts_cur, 3000, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_cur
                h_cur = h_pol

    # Phase 7: Fine stochastic with small deltas
    if time_left() > 80:
        h_st2, c5_st2 = stochastic_refine(h_cur, n_cur, ts_cur, rng, 50000, 0.08, 0.05)
        if c5_st2 < best_c5:
            best_c5 = c5_st2
            best_h = h_st2.copy()
            best_n = n_cur
            h_cur = h_st2

    # Phase 8: Greedy coordinate descent
    if time_left() > 80:
        h_gc, c5_gc = greedy_coord_descent(h_cur, n_cur, ts_cur, n_sweeps=3, relax=0.2)
        if c5_gc < best_c5:
            best_c5 = c5_gc
            best_h = h_gc.copy()
            best_n = n_cur
            h_cur = h_gc

    # Phase 9: Final SLSQP polish with very high alpha
    if time_left() > 30:
        for alpha in [5000, 15000]:
            if time_left() < 20:
                break
            h_f = slsqp_smooth(h_cur, n_cur, ts_cur, alpha, maxiter=3000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                best_n = n_cur
                h_cur = h_f

    # Phase 10: Try n=1440 if time allows
    if time_left() > 200:
        n_hh = 1440
        ts_hh = n_hh / 2.0
        h_hh = upsample_h(best_h, n_hh)
        h_hh = project_to_sum(h_hh, ts_hh)
        for alpha in [100, 500, 2000, 5000]:
            if time_left() < 30:
                break
            h_opt = slsqp_smooth(h_hh, n_hh, ts_hh, alpha, maxiter=1500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hh
                h_cur = h_opt
            else:
                h_cur = h_opt
        if time_left() > 60:
            h_st3, c5_st3 = stochastic_refine(h_cur, best_n, best_n / 2.0, rng, 30000, 0.1, 0.1)
            if c5_st3 < best_c5:
                best_c5 = c5_st3
                best_h = h_st3.copy()
                best_n = len(h_st3)

    # Phase 11: One more round of active-lag + polish
    if time_left() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        for trial in range(5):
            if time_left() < 30:
                break
            top_m = rng.randint(2, 6)
            alpha_lm = rng.choice([50, 150, 300])
            h_al = slsqp_active(best_h, n_f, ts_f, top_m, alpha_lm, maxiter=1500)
            c5_al = compute_c5(h_al)
            if c5_al < best_c5:
                best_c5 = c5_al
                best_h = h_al.copy()
                best_n = n_f

    # Phase 12: Final micro-stochastic
    if time_left() > 40:
        h_ms, c5_ms = stochastic_refine(best_h, best_n, best_n / 2.0, rng, 30000, 0.05, 0.03)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            best_n = len(h_ms)
        if time_left() > 20:
            h_fp = slsqp_smooth(best_h, best_n, best_n / 2.0, 10000, maxiter=2000)
            c5_fp = compute_c5(h_fp)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()
                best_n = len(h_fp)

    # If n=960 didn't beat n=480, fall back
    if best_c5 >= c5_480 and best_n != 480:
        best_h = h_480.copy()
        best_c5 = c5_480
        best_n = 480
        # Try to improve at 480 with any remaining time
        if time_left() > 60:
            h_480_imp = slsqp_smooth(best_h, 480, 240.0, 5000, maxiter=3000)
            c5_480_imp = compute_c5(h_480_imp)
            if c5_480_imp < best_c5:
                best_c5 = c5_480_imp
                best_h = h_480_imp

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    target = len(best_h) / 2.0
    for _ in range(1000):
        diff = target - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
