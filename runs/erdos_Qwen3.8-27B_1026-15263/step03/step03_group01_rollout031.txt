import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = m_val + (max_sh + np.log(sum_exp)) / alpha
    
    # FFT-based gradient
    fft_size = 1
    while fft_size < 3 * n - 1:
        fft_size *= 2
    
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    
    F1 = np.fft.rfft(one_minus_h, fft_size)
    F2 = np.fft.rfft(h_rev, fft_size)
    Fw = np.fft.rfft(w, fft_size)
    
    conv1 = np.fft.irfft(F1 * Fw, fft_size)[:n]
    conv2 = np.fft.irfft(F2 * Fw, fft_size)[:n][::-1]
    
    grad = conv1 - conv2
    return obj, grad

def slsqp_fft(h0, ts, alpha, maxiter=500):
    n = len(h0)
    def f(x):
        return smooth_max_fft(x, alpha)[0]
    def g(x):
        return smooth_max_fft(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)
