```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, target_sum):
    """Project v onto {0 <= h <= 1, sum(h) = target_sum}"""
    # h_i = clip(v_i - lambda, 0, 1), find lambda by bisection
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any floating point issues
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_max(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr, np.max(corr)

def smooth_max_obj(h, alpha, top_m=None):
    """Compute smooth max of correlation * dx, optionally on top_m entries"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_m is not None and top_m < len(corr):
        idx = np.argsort(corr)[::-1][:top_m]
        vals = corr[idx]
    else:
        vals = corr
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    obj = (m + (np.max(shifted) + np.log(np.sum(e))) / alpha) * dx
    return obj, vals, m

def grad_smooth_max(h, alpha, top_m=None):
    """Gradient of smooth max objective with respect to h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_len = len(corr)
    if top_m is not None and top_m < c_len:
        idx = np.argsort(corr)[::-1][:top_m]
        vals = corr[idx]
    else:
        idx = np.arange(c_len)
        vals = corr
    
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    w = e / np.sum(e)
    
    # Gradient computation
    # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])
    # d(corr[k])/dh[j] = (1 - h[j+k-(n-1)]) * 1[0 <= j+k-(n-1) < n]
    #                    - h[j-k+(n-1)] * 1[0 <= j-k+(n-1) < n]
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii] - (n - 1)  # actual shift
        wi = w[ii] * dx
        # First term: (1 - h[j+k]) for j where 0 <= j+k < n
        jp = np.arange(max(0, -k), min(n, n - k))
        if len(jp) > 0:
            grad[jp] += wi * (1.0 - h[jp + k])
        # Second term: -h[j-k] for j where 0 <= j-k < n
        jm = np.arange(max(0, k), min(n, n + k))
        if len(jm) > 0:
            grad[jm] -= wi * h[jm - k]
    
    return grad

def fista_opt(h0, n, ts, alpha, top_m, n_iter=5000, lr_init=0.01):
    """FISTA (accelerated projected gradient) optimization"""
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    obj_prev, _, _ = smooth_max_obj(h, alpha, top_m)
    lr = lr_init
    nesterov_t = 1.0
    
    for i in range(n_iter):
        grad = grad_smooth_max(y, alpha, top_m)
        # Gradient step
        z = y - lr * grad
        # Project
        h_new = project_box_sum(z, ts)
        # Evaluate
        obj_new, _, _ = smooth_max_obj(h_new, alpha, top_m)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)  # Keep y in feasible region roughly
        h = h_new
        t_k = t_new
        
        if obj_new < obj_prev:
            obj_prev = obj_new
            # Don't change lr (or increase slightly)
        else:
            lr *= 0.8
            # Revert
            # (in FISTA we can't easily revert, so just reduce lr)
    
    return h, obj_prev

def compute_grad(h, alpha, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_len = len(corr)  # 2n-1
    
    if top_m < c_len:
        idx = np.argsort(corr)[::-1][:top_m]
    else:
        idx = np.arange(c_len)
    
    vals = corr[idx]
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    w = e / np.sum(e) * dx  # include dx for the objective scaling
    
    grad = np.zeros(n)
    n1 = n - 1
    for ii in range(len(idx)):
        s = idx[ii] - n1  # shift
        wi = w[ii]
        if wi < 1e-15:
            continue
        # Term 1: (1-h[j-s]) for j where 0 <= j-s < n
        if s >= 0:
            j_start, j_end = s, n
        else:
            j_start, j_end = 0, n + s
        if j_end > j_start:
            grad[j_start:j_end] += wi * (1.0 - h[j_start-s:j_end-s])
        # Term 2: -h[j+s] for j where 0 <= j+s < n
        if s >= 0:
            j_start2, j_end2 = 0, n - s
        else:
            j_start2, j_end2 = -s, n
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= wi * h[j_start2+s:j_end2+s]
    
    return grad
```

This should be efficient. For each of the top_m shifts, we do O(n) work, so total is O(n * top_m).

Actually, I realize I should also try a different approach: **direct optimization of the top-M active constraints using a barrier/penalty method**.

The idea: Let A be the set of top-M shifts. We want:
minimize max_{k in A} corr[k] * dx
subject to 0 ≤ h ≤ 1, sum(h) = n/2

We can reformulate as:
minimize t
subject to corr[k] * dx ≤ t for k in A
           0 ≤ h ≤ 1
           sum(h) = n/2

This is a non-convex QCQP (quadratically constrained quadratic program) because corr[k] is quadratic in h.

For the non-convex part, we can use:
- Alternating direction method
- Penalty method: minimize sum_{k in A} max(0, corr[k]*dx - t)² + t
- Or: minimize t + μ * sum_{k in A} max(0, corr[k]*dx - t)²

The gradient with respect to h is similar to before, and the gradient with respect to t is simple.

But since t is just 1D, we can eliminate it:
minimize max(0, max_{k in A} corr[k]*dx) + μ * sum_{k in A} max(0, corr[k]*dx - C_target)²

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, target_sum):
    h = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha, top_m):
    c_len = len(corr)
    if top_m < c_len:
        idx = np.argpartition(corr, -top_m)[-top_m:]
        vals = corr[idx]
    else:
        vals = corr
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    return m + (np.max(shifted) + np.log(np.sum(e))) / alpha

def compute_obj_grad(h, alpha, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_len = len(corr)
    
    if top_m < c_len:
        idx = np.argpartition(corr, -top_m)[-top_m:]
        vals = corr[idx]
    else:
        idx = np.arange(c_len)
        vals = corr
    
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    w = e / np.sum(e) * dx
    
    obj = (m + (np.max(shifted) + np.log(np.sum(e))) / alpha) * dx
    
    n1 = n - 1
    grad = np.zeros(n)
    for ii in range(len(idx)):
        s = idx[ii] - n1
        wi = w[ii]
        if wi < 1e-16:
            continue
        # (1-h[j-s]) for 0 <= j-s < n
        js_lo = max(0, s)
        js_hi = min(n, n + s) if s < 0 else n
        # Wait: 0 <= j-s < n means s <= j < s+n
        # j ranges from max(0, s) to min(n-1, s+n-1)
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * (1.0 - h[j_lo - s:j_hi - s])
        
        # -h[j+s] for 0 <= j+s < n
        # 0 <= j+s < n means -s <= j < n-s
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wi * h[j_lo2 + s:j_hi2 + s]
    
    return obj, grad

def fista(h0, n, ts, alpha, top_m, n_iters, lr_init=0.02, time_fn=None, time_limit=None):
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    lr = lr_init
    obj, _ = compute_obj_grad(h, alpha, top_m)
    
    for i in range(n_iters):
        if time_fn is not None and time_fn() > time_limit:
            break
        grad = compute_obj_grad(y, alpha, top_m)[1]
        z = y - lr * grad
        h_new = project_box_sum(z, ts)
        obj_new, _ = compute_obj_grad(h_new, alpha, top_m)
        
        if obj_new <= obj:
            obj = obj_new
        else:
            lr *= 0.75
            if lr < 1e-10:
                break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_k = t_new
    
    return h

def lbfgs_opt(h0, n, ts, alpha, top_m, maxiter=500, time_fn=None, time_limit=None):
    def obj_fn(x):
        return compute_obj_grad(x, alpha, top_m)[0]
    
    def grad_fn(x):
        return compute_obj_grad(x, alpha, top_m)[1]
    
    def penalty(x):
        d = np.sum(x) - ts
        return 1e4 * d * d
    
    def full_obj(x):
        o, g = compute_obj_grad(x, alpha, top_m)
        d = np.sum(x) - ts
        return o + 1e4 * d * d
    
    def full_grad(x):
        o, g = compute_obj_grad(x, alpha, top_m)
        d = np.sum(x) - ts
        g = g + 2e4 * d * np.ones(n)
        return g
    
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(full_obj, h0.copy(), jac=full_grad, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_box_sum(res.x, ts)
    except:
        return project_box_sum(h0, ts)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, target_sum):
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj_grad(h, alpha, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_len = len(corr)
    if top_m < c_len:
        idx = np.argpartition(corr, c_len - top_m)[c_len - top_m:]
    else:
        idx = np.arange(c_len)
    vals = corr[idx]
    m = np.max(vals)
    shifted = alpha * (vals - m)
    e = np.exp(shifted - np.max(shifted))
    w = e / np.sum(e)
    obj = (m + (np.max(shifted) + np.log(np.sum(e))) / alpha) * dx
    n1 = n - 1
    grad = np.zeros(n)
    for ii in range(len(idx)):
        s = idx[ii] - n1
        wi = w[ii] * dx
        if wi < 1e-16:
            continue
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * (1.0 - h[j_lo - s:j_hi - s])
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wi * h[j_lo2 + s:j_hi2 + s]
    return obj, grad

def fista_opt(h0, n, ts, alpha, top_m, n_iters, time_fn=None, time_limit=None, lr_init=None):
    if lr_init is None:
        lr_init = min(0.5, 30.0 / max(alpha, 1))
    lr = lr_init
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0

    for i in range(n_iters):
        if time_fn is not None and time_fn() > time_limit:
            break

        z = y - lr * grad
        h_new = project_box_sum(z, ts)

        if obj_new <= obj + 1e-15:
            obj = obj_new
