def project_to_sum(h, target_sum):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target_sum, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def two_opt(h, target_sum, t_end, rng, max_rounds=100000):
    """Steepest descent via pair swaps."""
    n = len(h)
    h = h.astype(float).copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    round_num = 0
    while time.time() < t_end and round_num < max_rounds:
        round_num += 1
        improved = False
        ones_list = list(ones)
        zeros_list = list(zeros)
        # Shuffle for diversity
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        
        for i in ones_list:
            if h[i] != 1.0:
                continue
            for j in zeros_list:
                if h[j] != 0.0:
                    continue
                # Tentative swap
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < c5_curr - 1e-14:
                    c5_curr = c5_new
                    ones.remove(i)
                    ones.add(j)
                    zeros.remove(j)
                    zeros.add(i)
                    improved = True
                    if c5_curr < best_c5 - 1e-14:
                        best_c5 = c5_curr
                        best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            break
    
    return best_h, best_c5
