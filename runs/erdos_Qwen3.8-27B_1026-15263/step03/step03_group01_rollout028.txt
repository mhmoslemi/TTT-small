        else:

def exact_max_obj_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    obj = corr[k_max]
    lag = k_max - n + 1
    
    # d(corr[k_max])/d(h[j]) = (1-h[j+lag]) * I[valid] - h[j-lag] * I[valid]
    idx1 = np.arange(n) + lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = np.arange(n) - lag
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    grad = np.zeros(n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    grad[valid2] -= h[idx2[valid2]]
    return obj, grad

Now, the issue with using exact max is that the gradient only points in the direction of reducing the single worst lag. This can cause oscillation. A better approach is to use a "top-K" max: consider the K largest correlation values and average their gradients.

```python
def topk_max_obj_grad(h, K=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    obj = corr[top_k[0]]  # the maximum
    # Weights: softmax over top K
    shifted = corr[top_k] - obj
    w = np.exp(shifted * 50.0) / np.sum(np.exp(shifted * 50.0))
    
    grad = np.zeros(n)
    for k_idx, w_k in zip(top_k, w):
        lag = k_idx - n + 1
        idx1 = np.arange(n) + lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = np.arange(n) - lag
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid1] += w_k * (1.0 - h[idx1[valid1]])
        grad[valid2] -= w_k * h[idx2[valid2]]
    return obj, grad
```

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj_grad(h, K, alpha_k=50.0):
    """Compute objective and gradient using top-K lag approximation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    obj = corr[top_k[0]]
    shifted = corr[top_k] - obj
    ex = np.exp(shifted * alpha_k)
    w = ex / np.sum(ex)
    
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(K):
        k = top_k[idx]
        lag = k - n + 1
        wk = w[idx]
        idx1 = j + lag
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - lag
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[v1] += wk * (1.0 - h[idx1[v1]])
        grad[v2] -= wk * h[idx2[v2]]
    return obj, grad

def smooth_max_obj_grad(h, alpha):
    """Full smooth max (log-sum-exp) objective and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    lag = nm - n + 1
    idx1 = nj + lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = nj - lag
    valid2 = (idx2 >= 0) & (idx2 < n)
    fc = np.clip(idx1, 0, n - 1)
    sc = np.clip(idx2, 0, n - 1)
    term1 = np.sum(valid1 * (1.0 - h[fc]) * w[None, :], axis=1)
    term2 = np.sum(valid2 * h[sc] * w[None, :], axis=1)
    grad = term1 - term2
    return obj, grad

def adam_opt(h0, n, ts, n_iter, K, alpha_k, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with projection for top-K objective."""
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_obj = float('inf')
    best_h = h.copy()
    stall = 0
    for t in range(1, n_iter + 1):
        obj, grad = topk_obj_grad(h, K, alpha_k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = project_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), ts)
        obj_new, _ = topk_obj_grad(h_new, K, alpha_k)
        if obj_new < obj:
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            lr *= 0.8
            stall += 1
            if stall > 200 or lr < 1e-12:
                break
    return best_h, best_obj

def mom_grad_opt(h0, n, ts, n_iter, K, alpha_k, lr=0.05, momentum=0.9):
    """Momentum gradient descent with projection."""
    h = h0.copy()
    vel = np.zeros(n)
    best_obj = float('inf')
    best_h = h.copy()
    for t in range(n_iter):
        obj, grad = topk_obj_grad(h, K, alpha_k)
        vel = momentum * vel - lr * grad
        h_new = project_to_sum(h + vel, ts)
        obj_new, _ = topk_obj_grad(h_new, K, alpha_k)
        if obj_new < obj:
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
        else:
            vel = 0.5 * vel
            h = project_to_sum(h - 0.5 * lr * grad, ts)
    return best_h, best_obj

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    """SLSQP with smooth max."""
    def f(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[0]
    def g(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def stochastic_cd(h, n, ts, rng, n_steps, delta_range, temperature):
    """Stochastic coordinate descent (mass transfer)."""
    h = h.copy()
    c5_cur = compute_c5(h)
    for step in range(n_steps):
        ones_idx = np.where(h > 0.02)[0]
        zeros_idx = np.where(h < 0.98)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.8
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.8
        if delta < 1e-5:
            continue
        old_i, old_j = h[i], h[j]
        h[i] -= delta
        h[j] += delta
        h[i] = max(0.0, h[i])
        h[j] = min(1.0, h[j])
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - c5_cur) / temperature) if temperature > 0 else 0.0
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] = old_i
                h[j] = old_j
    return h, c5_cur

def make_structured_starts(n, ts, rng, base_h=None, base_n=0):

    starts = []

    # Uniform

    # Low frequency sinusoids

            starts.append(project_to_sum(s, ts))
    
    # Multi-frequency

        s = np.full(n, 0.5)

            s += rng.uniform(0.05, 0.2) * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2*np.pi))
        starts.append(project_to_sum(s, ts))
    
    # Block patterns (periodic on-off)
    for period in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        for on_frac in [0.3, 0.4, 0.5]:
            on_len = max(1, int(period * on_frac))
            h_b = np.zeros(n)
            for start in range(0, n, period):
                h_b[start:min(start+on_len, n)] = 1.0
            starts.append(project_to_sum(h_b, ts))
    
    # Upsampled base
    if base_h is not None and base_n > 0:
        if base_n == n:
            starts.append(base_h.copy())
        else:
            x_o = np.linspace(0, 2, base_n, endpoint=False)
            h_up = np.interp(x, x_o, base_h, period=2.0)
            starts.append(project_to_sum(h_up, ts))
    
    # Random
    starts.append(project_to_sum(rng.uniform(0, 1, n), ts))
    
    return starts

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj_grad(h, K, alpha_k=50.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    obj = corr[top_k[0]]
    shifted = corr[top_k] - obj
    ex = np.exp(shifted * alpha_k)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(K):
        lag = top_k[idx] - n + 1
        wk = w[idx]
        if wk < 1e-10:
            continue
        idx1 = j + lag
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - lag
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[v1] += wk * (1.0 - h[idx1[v1]])
        grad[v2] -= wk * h[idx2[v2]]
    return obj, grad

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    lag = nm - n + 1
    idx1 = nj + lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = nj - lag
    valid2 = (idx2 >= 0) & (idx2 < n)
    fc = np.clip(idx1, 0, n - 1)
    sc = np.clip(idx2, 0, n - 1)
    term1 = np.sum(valid1 * (1.0 - h[fc]) * w[None, :], axis=1)
    term2 = np.sum(valid2 * h[sc] * w[None, :], axis=1)
    grad = term1 - term2
    return obj, grad

def adam_opt(h0, n, ts, n_iter, K, alpha_k, lr=0.02, beta1=0.9, beta2=0.999):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    stall = 0
    best_obj = float('inf')
    best_h = h.copy()
    for t in range(1, n_iter + 1):
        obj, grad = topk_obj_grad(h, K, alpha_k)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = project_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), ts)
        obj_new, _ = topk_obj_grad(h_new, K, alpha_k)
        if obj_new < obj:
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            lr *= 0.8
            stall += 1
            if stall > 300 or lr < 1e-12:
                break
    return best_h, best_obj

def mom_grad_opt(h0, n, ts, n_iter, K, alpha_k, lr=0.03, momentum=0.85):
    h = h0.copy()
    vel = np.zeros(n)
    stall = 0
    for t in range(n_iter):
        obj, grad = topk_obj_grad(h, K, alpha_k)
        vel = momentum * vel - lr * grad
        h_new = project_to_sum(h + vel, ts)
        obj_new, _ = topk_obj_grad(h_new, K, alpha_k)
        if obj_new < obj:
            h = h_new
            stall = 0
        else:
            lr *= 0.8
            vel *= 0.5
            h_new = project_to_sum(h - lr * grad, ts)
            obj_new, _ = topk_obj_grad(h_new, K, alpha_k)
            if obj_new < obj:
                h = h_new
            stall += 1
            if stall > 300 or lr < 1e-12:
                break
    return h

def slsqp_opt(h0, n, ts, alpha, maxiter=400):
    def f(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[0]
    def g(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def stochastic_cd(h, n, ts, rng, n_steps, d_lo, d_hi, temp0, cool):
    h = h.copy()
    c5_cur = compute_c5(h)
    for step in range(n_steps):
        temp = temp0 * np.exp(-step / cool)
        hi_mask = h > 0.01
        lo_mask = h < 0.99
        n_hi = np.sum(hi_mask)
        n_lo = np.sum(lo_mask)
        if n_hi == 0 or n_lo == 0:
            break
        hi_idx = np.where(hi_mask)[0]
        lo_idx = np.where(lo_mask)[0]
        i = hi_idx[rng.randint(n_hi)]
        j = lo_idx[rng.randint(n_lo)]
        if i == j:
            continue
        delta = rng.uniform(d_lo, d_hi)
        if h[i] < delta:
            delta = h[i] * 0.9
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.9
        if delta < 1e-5:
            continue
        old_i = h[i]
        old_j = h[j]
        h[i] = old_i - delta
        h[j] = old_j + delta
        c5_new = compute_c5(h)
        if c5_new >= c5_cur:
            accept = np.exp(-(c5_new - c5_cur) / max(temp, 1e-8))
            if rng.random() > accept:
                h[i] = old_i
                h[j] = old_j
        else:
            c5_cur = c5_new
    return h, c5_cur

def make_starts(n, ts, rng, base_h=None, base_n=0):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for f in range(1, 13):
        for amp in [0.25, 0.35, 0.45]:
            ph = rng.uniform(0, 2 * np.pi)
            s = 0.5 + amp * np.sin(2.0 * np.pi * f * x / 2.0 + ph)
            starts.append(project_to_sum(s, ts))
    for _ in range(6):
        nf = rng.randint(2, 7)
        freqs = rng.choice(np.arange(1, 25), size=nf, replace=False)
        s = np.full(n, 0.5)
        for f in freqs:
            s += rng.uniform(0.05, 0.18) * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(s, ts))
    for period in [2, 3, 4, 5, 6, 7, 8, 10, 12, 15]:
        on = max(1, int(period * 0.45))
        h_b = np.zeros(n)
        for st in range(0, n, period):
            h_b[st:min(st + on, n)] = 1.0
        starts.append(project_to_sum(h_b, ts))
    if base_h is not None and base_n > 0:
        if base_n == n:
            starts.append(base_h.copy())
        elif base_n < n:
            x_o = np.linspace(0, 2, base_n, endpoint=False)
            h_up = np.interp(x, x_o, base_h)
            starts.append(project_to_sum(h_up, ts))
    starts.append(project_to_sum(rng.uniform(0, 1, n), ts))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    target_n = 960
    if best_n < target_n:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_to_sum(h_up, target_n / 2.0)
        c5_up = compute_c5(h_up)
        best_h = h_up
        best_n = target_n
        best_c5 = c5_up

    ts = best_n / 2.0

    # Phase A: Adam continuation
    phases_a = [(100, 10, 0.03, 1200), (50, 30, 0.025, 1200), (20, 80, 0.02, 1500),
                (10, 150, 0.015, 1500), (5, 300, 0.01, 2000), (3, 500, 0.008, 2000)]
    for K, ak, lr, nit in phases_a:
        if tl() < 50:
            break
        h_opt, _ = adam_opt(best_h, best_n, ts, nit, K, ak, lr=lr)
        c5_new = compute_c5(h_opt)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()

    # Phase B: SLSQP polish
    for alpha in [200, 1000, 5000]:
        if tl() < 35:
            break
        h_s = slsqp_opt(best_h, best_n, ts, alpha, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase C: Momentum gradient
    if tl() > 80:
        h_m = mom_grad_opt(best_h, best_n, ts, 3000, 5, 200, lr=0.02, momentum=0.9)
        c5_m = compute_c5(h_m)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase D: Multi-start exploration
    n_trials = 25
    for trial in range(n_trials):
        if tl() < 35:
            break
        starts = make_starts(best_n, ts, rng, best_h, best_n)
        n_s = min(5, len(starts))
        idxs = rng.choice(len(starts), size=n_s, replace=False)
        for ii in idxs:
            if tl() < 20:
                break
            h_s = starts[ii]
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_o, _ = adam_opt(h_s, best_n, ts, 800, 15, 100, lr=0.015)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Phase E: Stochastic CD (large steps)
    if tl() > 70:
        h_cd, c5_cd = stochastic_cd(best_h, best_n, ts, rng, 40000, 0.01, 0.35, 0.004, 6000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase F: Adam polish
    if tl() > 50:
        h_f, _ = adam_opt(best_h, best_n, ts, 3000, 3, 400, lr=0.004)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase G: SLSQP final
    if tl() > 35:
        h_p = slsqp_opt(best_h, best_n, ts, 8000, maxiter=500)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase H: Stochastic CD (small steps)
    if tl() > 40:
        h_t, c5_t = stochastic_cd(best_h, best_n, ts, rng, 30000, 0.003, 0.1, 0.001, 5000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase I: Final tiny Adam
    if tl() > 30:
        h_z, _ = adam_opt(best_h, best_n, ts, 2000, 2, 600, lr=0.003)
        c5_z = compute_c5(h_z)
        if c5_z < best_c5:
            best_c5 = c5_z
            best_h = h_z.copy()

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
