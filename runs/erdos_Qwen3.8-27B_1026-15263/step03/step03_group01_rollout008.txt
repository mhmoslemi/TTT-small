def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                lo1 = k
                grad[lo1:] += (1.0 - h[:n-k])
            hi2 = n - 1 - k
            if hi2 >= 0:
                grad[:hi2+1] -= h[k:]
        else:
            m = -k
            hi1 = n - 1 - m
            if hi1 >= 0:
                grad[:hi1+1] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad, float(np.max(corr))

def adam_topk(h0, n, ts, n_iter, k_start, k_end, lr=0.02, warmup=100):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_iter + 1):
        frac = t / n_iter
        k_cur = int(k_start - (k_start - k_end) * frac)
        k_cur = max(k_cur, 3)
        lr_cur = lr / (1 + t * 1e-5)
        if t <= warmup:
            lr_cur *= t / warmup
        grad, _ = topk_grad(h, k_cur)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_sum(h - step, ts)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wi * (1.0 - h[:n-k])
            hi2 = n - 1 - k
            if hi2 >= 0:
                grad[:hi2+1] -= wi * h[k:]
        else:
            m = -k
            hi1 = n - 1 - m
            if hi1 >= 0:
                grad[:hi1+1] += wi * (1.0 - h[m:])
            grad[m:] -= wi * h[:n-m]
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f(x):
        return smooth_max_grad(x, alpha)[0]
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def interp_up(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h_old, period=2.0)
    return h_up

def freq_perturb(h, rng, n_harm, amp):

    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += (1.0 - h[:n-k])
            hi2 = n - 1 - k
            if hi2 >= 0:
                grad[:hi2+1] -= h[k:]
        else:
            m = -k
            hi1 = n - 1 - m
            if hi1 >= 0:
                grad[:hi1+1] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    sw = np.sum(exp_v)
    w = exp_v / sw
    obj = m_val + (max_sh + np.log(sw)) / alpha
    grad = np.zeros(n)

def slsqp_opt(h0, n, ts, alpha, maxiter=500):

        return smooth_max_grad(x, alpha)[0]
    def g(x):
        return smooth_max_grad(x, alpha)[1]

        return project_to_sum(r.x, ts)
