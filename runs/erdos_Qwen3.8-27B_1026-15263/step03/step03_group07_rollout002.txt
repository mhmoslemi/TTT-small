import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -600.0, 600.0)
    ev = np.exp(shifted)
    s = ev.sum()
    w = ev / s
    val = m + np.log(s / len(ev)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kp = min(j, n - 1)
        ks = np.arange(0, kp + 1)
        g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kn = n - 1 - j
        if kn >= 0:
            ks2 = np.arange(0, kn + 1)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        if kn >= 1:
            ks3 = np.arange(1, kn + 1)
            g[j] += np.dot(w[n - 1 - ks3], 1.0 - h[j + ks3])
        if j >= 1:
            ks4 = np.arange(1, j + 1)
            g[j] -= np.dot(w[n - 1 - ks4], h[j - ks4])
    return val, g

def smax_active_and_grad(h, active_idx, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = corr[active_idx]

    val = m + np.log(s / len(ev)) / alpha

                    g[j] += wv * (1.0 - h[j + k2])

                    g[j] -= wv * h[j - k2]

    if k >= 0:
        valid = n - k
        return float(np.dot(h[k:k+valid], 1.0 - h[0:valid]))
    else:
        k2 = -k
        valid = n - k2
        return float(np.dot(h[0:valid], 1.0 - h[k2:k2+valid]))

def single_lag_grad(h, idx, n):
    k = idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        for j in range(k, n):

def slsqp_single_lag(h0, n, ts, lag_idx, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return single_lag_val(h, lag_idx, n)
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        return single_lag_grad(h, lag_idx, n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def frank_wolfe_step(h, n, ts, alpha, step_size):
    _, g = smax_and_grad(h, alpha)
    g_adj = g - g.mean()
    order = np.argsort(g_adj)
    s = np.zeros(n)
    s[order[:int(ts)]] = 1.0
    frac = ts - int(ts)
    if frac > 1e-10 and int(ts) < n:
        s[order[int(ts)]] = frac
    h_new = h + step_size * (s - h)
    h_new = project_to_sum(h_new, ts)
    return h_new, g

def coord_descent(h0, n, ts, max_rounds, rng, adaptive=True):
    h = h0.copy()
    c5_cur = compute_c5(h)
    base_step = 0.01
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-16:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5_cur = best_val
                improved = True
        if adaptive and improved:
            base_step *= 0.9
        elif not improved:
            break
    return h, c5_cur

def binary_push_refine(h0, n, ts, n_iter, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for it in range(n_iter):
        strength = 0.02 * (1.0 + 0.5 * np.sin(it * 0.3))
        h_pulled = h + strength * (2.0 * h - 1.0)

        h_pulled = project_to_sum(h_pulled, ts)
        c5_new = compute_c5(h_pulled)
        if c5_new < c5_cur - 1e-15:
            h = h_pulled
            c5_cur = c5_new
        else:
            idx = rng.choice(n, size=min(20, n), replace=False)
            h_pert = h.copy()
            for i in idx:
                d = rng.uniform(-0.03, 0.03)

            c5_p = compute_c5(h_pert)
            if c5_p < c5_cur:
                h = h_pert
                c5_cur = c5_p
    return h, c5_cur

def projected_grad_descent(h0, n, ts, alpha, n_iters, lr0):

    h_best = h.copy()

def interpolate_h(h_old, n_old, n_new, ts_new):

            if len(active) > 0 and len(active) < n:
                for alpha_a in [500.0, 2000.0, 5000.0]:
                    if tleft() < 30:
                        break
                    h_act = slsqp_active(h_cur, n, ts, active, alpha_a, 300)
                    c5_act = compute_c5(h_act)
                    if c5_act < best_c5:
                        best_c5 = c5_act
                        best_h = h_act.copy()
                        best_n = n
                    h_cur = h_act
                    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                    m_val = np.max(corr)
                    active = np.where(corr >= thresh * m_val)[0]
                    if len(active) == 0:
                        break

    # Phase 4: Frank-Wolfe + projected gradient
    if tleft() > 80:
        n_fw = min(500, int(tleft() * 3))
        for alpha_fw in [200.0, 500.0]:
            if tleft() < 20:
                break
            t_fw = 0
            h_fw = h_cur.copy()
            c5_fw = compute_c5(h_fw)
            for t in range(n_fw):
                step = 2.0 / (t + 2.0) if t > 5 else 0.05
                h_fw, _ = frank_wolfe_step(h_fw, n, ts, alpha_fw, step)
                if t % 50 == 0:
                    c5_check = compute_c5(h_fw)
                    if c5_check < best_c5:
                        best_c5 = c5_check
                        best_h = h_fw.copy()
                        best_n = n
                    if tleft() < 20:
                        break
            c5_fw = compute_c5(h_fw)
            if c5_fw < compute_c5(h_cur):
                h_cur = h_fw

    # Phase 5: Projected gradient descent
    if tleft() > 60:
        n_pg = min(3000, int(tleft() * 2))
        h_pg, c5_pg = projected_grad_descent(h_cur, n, ts, 500.0, n_pg, 0.03)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n
        h_cur = h_pg

    # Phase 6: Coordinate descent
    if tleft() > 50:
        n_cd = max(5, min(50, int(tleft() // 10)))
        h_cd, c5_cd = coord_descent(h_cur, n, ts, n_cd, rng, adaptive=True)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        h_cur = h_cd

    # Phase 7: Binary push
    if tleft() > 30:
        h_bp, c5_bp = binary_push_refine(h_cur, n, ts, 40, rng)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
            best_n = n
        h_cur = h_bp

    # Phase 8: Higher resolution
    if tleft() > 200 and n <= 900:
        n_hi = n + (60 - n % 60) if n % 60 != 0 else n + 60
        if n_hi % 2 != 0:
            n_hi += 1
        ts_hi = n_hi / 2.0
        h_hi = interpolate_h(h_cur, n, n_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

        for alpha_hi in [50.0, 200.0, 800.0]:
            if tleft() < 40:
                break
            h_hi_opt = slsqp_smooth(h_hi, n_hi, ts_hi, alpha_hi,
                                    max(50, min(300, int(tleft() * 0.5))))
            c5_hi_opt = compute_c5(h_hi_opt)
            if c5_hi_opt < best_c5:
                best_c5 = c5_hi_opt
                best_h = h_hi_opt.copy()
                best_n = n_hi
            h_hi = h_hi_opt

        if tleft() > 40:
            corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode='full')
            m_hi = np.max(corr_hi)
            active_hi = np.where(corr_hi >= 0.999 * m_hi)[0]
            if 0 < len(active_hi) < n_hi:
                h_hi_act = slsqp_active(h_hi, n_hi, ts_hi, active_hi, 1000.0, 200)
                c5_hi_act = compute_c5(h_hi_act)
                if c5_hi_act < best_c5:
                    best_c5 = c5_hi_act
                    best_h = h_hi_act.copy()
                    best_n = n_hi
                h_hi = h_hi_act

        if tleft() > 30:
            h_hi_cd, c5_hi_cd = coord_descent(h_hi, n_hi, ts_hi, 10, rng)
            if c5_hi_cd < best_c5:
                best_c5 = c5_hi_cd
                best_h = h_hi_cd.copy()
                best_n = n_hi

    # Phase 9: Final polishing on best solution
    if tleft() > 40 and best_h is not None:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_f = best_h.copy()

        for alpha_f in [2000.0, 5000.0]:
            if tleft() < 25:
                break
            h_pol = slsqp_smooth(h_f, n_f, ts_f, alpha_f,
                                 max(50, min(400, int(tleft() * 0.7))))
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f
            h_f = h_pol

        if tleft() > 30:
            corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
            m_f = np.max(corr_f)
            active_f = np.where(corr_f >= 0.9995 * m_f)[0]
            if 0 < len(active_f) < n_f:
                h_act_f = slsqp_active(h_f, n_f, ts_f, active_f, 3000.0, 300)
                c5_act_f = compute_c5(h_act_f)
                if c5_act_f < best_c5:
                    best_c5 = c5_act_f
                    best_h = h_act_f.copy()
                    best_n = n_f
                h_f = h_act_f

        if tleft() > 30:
            n_cd_f = max(5, min(60, int(tleft() // 5)))
            h_cd_f, c5_cd_f = coord_descent(h_f, n_f, ts_f, n_cd_f, rng)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = h_cd_f.copy()
                best_n = n_f
            h_f = h_cd_f

        if tleft() > 20:
            h_bp_f, c5_bp_f = binary_push_refine(h_f, n_f, ts_f, 20, rng)
            if c5_bp_f < best_c5:
                best_c5 = c5_bp_f
                best_h = h_bp_f.copy()
                best_n = n_f

        if tleft() > 15:
            n_pg_f = min(2000, int(tleft() * 1.5))
            h_pg_f, c5_pg_f = projected_grad_descent(h_f, n_f, ts_f, 2000.0, n_pg_f, 0.01)
            if c5_pg_f < best_c5:
                best_c5 = c5_pg_f
                best_h = h_pg_f.copy()
                best_n = n_f

    # Final projection and evaluation
    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)
        return (best_h, best_c5, n_final)

    return (best_h, best_c5, best_n)
