```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    # Skip weights that are too small
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if k < n:
                g[k:] += wm * (1.0 - h[:n-k])
                g[:n-k] -= wm * h[k:]
        else:
            m2 = -k
            if m2 < n:
                g[:n-m2] += wm * (1.0 - h[m2:])
                g[m2:] -= wm * h[:n-m2]
    
    return sm_val, g

def lbfgsb_smooth_max(h0, alpha, maxiter=500):
    n = len(h)
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_value_and_grad(h, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_value_and_grad(h, alpha)
        return g
    
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return np.clip(res.x, 0.0, 1.0)

def projected_gradient(h0, alpha, n_iters, lr_init):
    n = len(h)
    ts = n / 2.0
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    lr = lr_init
    momentum = 0.9
    
    v = np.zeros(n)
    for it in range(n_iters):
        val, g = smooth_max_value_and_grad(h, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            h = h_new
        else:
            lr *= 0.8
            v *= 0.5
            if lr < 1e-8:
                break
    return h_best, c5_best
```

Let me also think about **what alpha values to use**. The smooth max with parameter alpha approximates the max as:
smooth_max = max + (1/alpha) * log(sum(exp(alpha*(x_i - max))))/n_corr

The error is O(log(n_corr)/alpha). For n_corr = 1439 (2n-1), log(1439) ≈ 7.3. So for alpha=1000, the error is about 0.007. This means the smooth max overestimates the true max by about 0.007.

To get a good final answer, I want alpha large enough that this error is small. Alpha = 5000 gives error ≈ 0.0015.

But for optimization, smaller alpha is better (smoother landscape). So the strategy is:
- Optimize with small alpha (good convergence)
- Switch to large alpha (accurate)
- The final C₅ should be computed directly (not from the smooth max)

One more thought: **the problem has a specific structure that might allow for a better initialization**. 

The Erdős problem is about finding a subset of [0,1] (or [0,2]) with measure 1/2 such that the maximum overlap between the set and its complement at any shift is minimized. This is related to "well-distributed" sets.

A good initialization might be based on a **low-discrepancy sequence** or a **balanced sequence** (like a Thue-Morse sequence).

Let me try a Thue-Morse based initialization:
- Generate the Thue-Morse sequence of length n
- Use it as the initial h (0s and 1s)
- Then smooth it slightly

Actually, the Thue-Morse sequence has low autocorrelation, which is exactly what we want! Let me try this.

The Thue-Morse sequence: t(0)=0, t(1)=1, t(2n)=t(n), t(2n+1)=1-t(n).
It has the property that in any block of length 2^k, the number of 0s equals the number of 1s, and the autocorrelation is low.

Let me generate it and use as initialization.

```python
def thue_morse(n):
    t = np.zeros(n, dtype=float)
    t[0] = 0.0
    for i in range(1, n):
        # t[i] = popcount(i) mod 2
        t[i] = bin(i).count('1') % 2
    return t
```

```python
def thue_morse_fast(n):

        # Use the recurrence: t[i] = t[i >> 1] XOR (i & 1)
        # Or: t[i] = popcount(i) % 2

```

Actually, the most efficient way in numpy:

```python
def thue_morse_numpy(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
```

Wait, that's not right either. The Thue-Morse sequence satisfies t[i] = t[i//2] XOR (i % 2) only for the recursive definition. Let me just use the popcount definition:

```python
def thue_morse_numpy(n):
    # t[i] = number of 1-bits in i, mod 2
    idx = np.arange(n)

    t = np.zeros(n)
```

This is O(n log n) which is fine for n=720.

The Thue-Morse sequence alternates in a balanced way, so its autocorrelation should be relatively flat. This makes it a good starting point for our optimization.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if k < n:
                g[k:] += wm * (1.0 - h[:n-k])
                g[:n-k] -= wm * h[k:]
        else:
            m2 = -k
            if m2 < n:
                g[:n-m2] += wm * (1.0 - h[m2:])
                g[m2:] -= wm * h[:n-m2]
    
    return sm_val, g

def generate_thue_morse(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') & 1
    return t.astype(float)

def make_initialization(n, method, rng):
    if method == 'thue_morse':
        h = generate_thue_morse(n)
    elif method == 'random':
        h = rng.rand(n)
    elif method == 'symmetric_random':
        half = n // 2
        h_half = rng.rand(half)
        h = np.concatenate([h_half, h_half[::-1]])
        if n % 2 == 1:
            h = np.concatenate([h, [0.5]])
            h = h[:n]
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.01, 0.99)
    h = project_to_sum(h, n / 2.0)
    return h

def optimize_lbfgsb(h0, alpha, mu, maxiter):
    n = len(h)
    ts = n / 2.0
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_value_and_grad(h, alpha)
        penalty = 0.5 * mu * (np.sum(h) - ts) ** 2
        return val + penalty
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, gv = smooth_max_value_and_grad(h, alpha)
        pen_grad = mu * (np.sum(h) - ts) * np.ones(n)
        return gv + pen_grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_sum(h_opt, ts)
        return h_opt
    except Exception:
        return h0

def optimize_active_lags(h0, active_idx, alpha, maxiter):
    n = len(h0)
    ts = n / 2.0
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        ac = corr[active_idx]
        m = np.max(ac)
        shifted = alpha * (ac - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        ev = np.exp(shifted)
        s = np.sum(ev)
        w = ev / s
        sm_val = m + np.log(s / len(ev)) / alpha
        
        # Gradient
        g = np.zeros(n)
        for idx_pos, m_idx in enumerate(active_idx):
            wm = w[idx_pos]
            if wm < 1e-12:
                continue
            k = m_idx - (n - 1)
            if k >= 0:
                if k < n:
                    g[k:] += wm * (1.0 - h[:n-k])
                    g[:n-k] -= wm * h[k:]
            else:
                m2 = -k
                if m2 < n:
                    g[:n-m2] += wm * (1.0 - h[m2:])
                    g[m2:] -= wm * h[:n-m2]
        
        penalty = 0.5 * 100.0 * (np.sum(h) - ts) ** 2
        return sm_val + penalty
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        ac = corr[active_idx]
        m = np.max(ac)
        shifted = alpha * (ac - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        ev = np.exp(shifted)
        s = np.sum(ev)
        w = ev / s
        
        g = np.zeros(n)
        for idx_pos, m_idx in enumerate(active_idx):
            wm = w[idx_pos]
            if wm < 1e-12:
                continue
            k = m_idx - (n - 1)
            if k >= 0:
                if k < n:
                    g[k:] += wm * (1.0 - h[:n-k])
                    g[:n-k] -= wm * h[k:]
            else:
                m2 = -k
                if m2 < n:
                    g[:n-m2] += wm * (1.0 - h[m2:])
                    g[m2:] -= wm * h[:n-m2]
        
        pen_grad = 100.0 * (np.sum(h) - ts) * np.ones(n)
        return g + pen_grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_sum(h_opt, ts)
        return h_opt
    except Exception:
        return h0

def coordinate_descent(h0, max_rounds, rng, deltas=None):
    n = len(h0)
    ts = n / 2.0
    h = h0.copy()
    c5_cur = compute_c5(h)
    if deltas is None:
        deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_val = c5_cur
            best_h = h.copy()
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-12:
                    best_val = c5_new
                    best_h = h_new
            if best_val < c5_cur - 1e-12:
                h = best_h
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def local_perturbation(h0, n_trials, rng, base_scale):
    n = len(h0)
    ts = n / 2.0
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    
    for _ in range(n_trials):
        # Random perturbation
        mask = rng.rand(n) < 0.3  # perturb 30% of coordinates
        h_new = h0.copy()
        h_new[mask] += rng.uniform(-base_scale, base_scale, mask.sum())
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
            h0 = h_new  # continue from here
            base_scale *= 0.95
    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
