```python
def project_box_sum(v, target_sum):
    """Project v onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}"""
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    # clip(v - lambda, 0, 1) = max(0, min(1, v - lambda))
    # This is a monotone decreasing function of lambda
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_fft(h, w):
    n = len(h)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    W_pad = np.zeros(N_fft)
    W_pad[:2 * n - 1] = w
    Wf = w[::-1]
    Wf_pad = np.zeros(N_fft)
    Wf_pad[:2 * n - 1] = Wf
    fft_h = np.fft.fft(h_pad)
    conv = np.fft.ifft(np.fft.fft(W_pad) * fft_h).real
    Wf_conv = np.fft.ifft(np.fft.fft(Wf_pad) * fft_h).real
    offset = n - 1
    return 1.0 - conv[offset:offset + n] - Wf_conv[offset:offset + n]

def smooth_max_value_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = compute_grad_fft(h, w)
    return sm_val, g, w, corr

def mgsd_optimize(h0, n, target_sum, alpha, n_iters, lr_init, momentum):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    for it in range(n_iters):
        sm_val, g, w, corr = smooth_max_value_grad(h, alpha)
        v = momentum * v - lr * g
        h_new = project_box_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= compute_c5(h):
            lr = min(lr * 1.04, 1.0)
        else:
            lr *= 0.7
            v *= 0.3
        h = h_new
    return best_h, best_c5

def topk_refine(h0, n, target_sum, k, n_iters, lr_init, momentum):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    for it in range(n_iters):
        corr = get_corr(h)
        top_idx = np.argsort(corr)[-k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / k
        h_pad = np.zeros(N_fft)
        h_pad[:n] = h
        W_pad = np.zeros(N_fft)
        W_pad[:2 * n - 1] = w
        Wf = w[::-1]
        Wf_pad = np.zeros(N_fft)
        Wf_pad[:2 * n - 1] = Wf
        fft_h = np.fft.fft(h_pad)
        conv = np.fft.ifft(np.fft.fft(W_pad) * fft_h).real
        Wf_conv = np.fft.ifft(np.fft.fft(Wf_pad) * fft_h).real
        offset = n - 1
        g = 1.0 - conv[offset:offset + n] - Wf_conv[offset:offset + n]
        v = momentum * v - lr * g
        h_new = project_box_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.8
            v *= 0.4
        h = h_new
    return best_h, best_c5

def coord_refine(h0, n, target_sum, n_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    steps_all = list(steps) + [-s for s in steps]
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_val = h[i]
            for d in steps_all:
                nv = h[i] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                h = project_box_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_val = h[i]
            if best_c5 < c5_cur - 1e-15:
                h[i] = best_val
                h = project_box_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
            else:
                h[i] = h[i]
        h = project_box_sum(h, target_sum)
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_src, n_src, n_dst):
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_box_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    resolutions = sorted(set([best_n, 720, 960]))

    for n in resolutions:
        if time_left() < 120:
            break
        if n % 2 != 0:
            n += 1
        ts = n / 2.0

        starts = []

        if best_n == n:
            starts.append(best_h.copy())
        else:
            h_interp = interpolate_h(best_h, best_n, n)
            starts.append(project_box_sum(h_interp, ts))

        for delta in [0.02, 0.05, 0.1, 0.2]:
            h_pert = best_h + rng.uniform(-delta, delta, best_n)
            h_pert = project_box_sum(h_pert, best_n / 2.0)
            if n != best_n:
                h_pert = interpolate_h(h_pert, best_n, n)
            starts.append(project_box_sum(h_pert, ts))

        x = np.linspace(0, 2, n, endpoint=False)
        for freqs in [(1, 2), (2, 3), (1, 3), (3, 5), (2, 5), (1, 4)]:
            for amps in [(0.15, 0.1), (0.25, 0.12), (0.3, 0.15)]:
                for _ in range(2):
                    ph1 = rng.uniform(0, 2 * np.pi)
                    ph2 = rng.uniform(0, 2 * np.pi)
                    hs = 0.5 + amps[0] * np.sin(2 * np.pi * freqs[0] * x / 2.0 + ph1) + \
                               amps[1] * np.sin(2 * np.pi * freqs[1] * x / 2.0 + ph2)
                    hs = project_box_sum(hs, ts)
                    if compute_c5(hs) < best_c5 + 0.03:
                        starts.append(hs)

        for h_start in starts:
            if time_left() < 70:
                break
            h_start = project_box_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()

            for alpha, lr_i, mom, n_it_base in [(50.0, 0.3, 0.8, 300),
                                                  (200.0, 0.2, 0.85, 400),
                                                  (800.0, 0.15, 0.85, 500),
                                                  (3000.0, 0.1, 0.8, 600),
                                                  (8000.0, 0.08, 0.8, 600)]:
                if time_left() < 30:
                    break
                n_it = min(n_it_base, max(50, int(time_left() * 0.4)))
                h_cur, c5_cur = mgsd_optimize(h_cur, n, ts, alpha, n_it, lr_i, mom)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n

            if time_left() > 40:
                for k in [5, 10, 20]:
                    if time_left() < 25:
                        break
                    n_it = min(400, max(100, int(time_left() * 0.3)))
                    h_cur, c5_cur = topk_refine(h_cur, n, ts, k, n_it, 0.08, 0.8)
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h_cur.copy()
                        best_n = n

    if best_h is not None and time_left() > 50:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha, lr_i in [(5000.0, 0.06), (10000.0, 0.05), (20000.0, 0.04)]:
            if time_left() < 25:
                break
            n_it = max(200, int(time_left() * 0.5))
            h_f, c5_f = mgsd_optimize(best_h, n_f, ts_f, alpha, n_it, lr_i, 0.8)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        if time_left() > 30:
            for k in [3, 5, 8, 12]:
                if time_left() < 20:
                    break
                n_it = min(500, max(100, int(time_left() * 0.4)))
                h_f, c5_f = topk_refine(best_h, n_f, ts_f, k, n_it, 0.05, 0.75)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()

        if time_left() > 25:
            base_steps = [0.01, 0.005, 0.002, 0.02]
            max_rnd = max(5, min(20, int(time_left() // 3)))
            h_cd, c5_cd = coord_refine(best_h, n_f, ts_f, max_rnd, base_steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_box_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
```
