import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    H1 = np.fft.rfft(1.0 - h, N)
    corr_circ = np.fft.irfft(H * np.conj(H1), N)
    full_corr = np.concatenate([corr_circ[N - n + 1:], corr_circ[:n]])
    return full_corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fft(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            cnt = np.sum(mask)
            if abs(diff) <= cnt * 0.49:
                h[mask] += diff / cnt
            else:
                h += diff / len(h)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad_fft(h, alpha, lambda_bin=0.0):
    n = len(h)
    corr = compute_corr_fft(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha

    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    H = np.fft.rfft(h, N)

    fft_wh = np.fft.irfft(W * H, N)
    fft_hw = np.fft.irfft(H * np.conj(W), N)

    term1 = fft_wh[n - 1:2 * n - 1]
    idx2 = np.arange(n)
    idx2_mod = (idx2 - n + 1 + N) % N
    term2 = fft_hw[idx2_mod]

    S = np.sum(w)
    g_sm = S - term1 - term2

    if lambda_bin > 0.0:
        g_sm = g_sm - 8.0 * lambda_bin * h
        sm_val = sm_val - 4.0 * lambda_bin * np.sum(h * h)

    return sm_val, g_sm

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init, lambda_bin=0.0):
    h = h0.copy()
    y = h0.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best

    for it in range(n_iters):
        _, g = smooth_max_grad_fft(y, alpha, lambda_bin)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new > c5_cur * 1.02 + 1e-6:
            lr *= 0.5
            if lr < 1e-6:
                lr = lr_init * 0.1
                t_val = 1.0
                y = h.copy()
            continue

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = c5_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.92

    return h_best, c5_best

def slsqp_step(h0, n, target_sum, alpha, maxiter, lambda_bin=0.0):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad_fft(ht, alpha, lambda_bin)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_fft(ht, alpha, lambda_bin)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_fixed(h0, n, target_sum, n_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def simulated_annealing(h0, n, target_sum, alpha, n_iters, T_init, T_min, rng, lambda_bin=0.0):
    h_cur = h0.copy()
    c5_cur = compute_c5(h0)
    h_best = h0.copy()
    c5_best = c5_cur
    y = h0.copy()
    t_val = 1.0
    lr = 0.1

    for it in range(n_iters):
        T = T_init * (T_min / T_init) ** (it / max(1, n_iters))
        _, g = smooth_max_grad_fft(y, alpha, lambda_bin)
        noise = rng.normal(0, T, size=n)
        h_new = np.clip(y - lr * g + noise, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < c5_cur:
            accept = True
        else:
            delta = c5_new - c5_cur
            accept = rng.random() < np.exp(-delta / max(T, 1e-10))

        if accept:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            mo = (t_val - 1.0) / t_new
            y = h_new + mo * (h_new - h_cur)
            y = np.clip(y, 0.0, 1.0)
            h_cur = h_new
            t_val = t_new
            c5_cur = c5_new
            lr = min(lr * 1.02, 0.4)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        else:
            lr *= 0.95
            t_val = max(1.0, t_val * 0.9)
            y = np.clip(h_cur + 0.1 * (h_cur - y), 0.0, 1.0)

    return h_best, c5_best

def binary_snap(h, target_sum):
    n = len(h)
    h_snap = h.copy()
    threshold = 0.45
    mask0 = h_snap < threshold
    mask1 = h_snap > 1.0 - threshold
    h_snap[mask0] = 0.0
    h_snap[mask1] = 1.0
    free = ~(mask0 | mask1)
    if np.any(free):
        needed = target_sum - np.sum(h_snap)
        cnt = np.sum(free)
        if abs(needed) <= cnt * 0.5:
            h_snap[free] += needed / cnt
        h_snap = np.clip(h_snap, 0.0, 1.0)
    h_snap = project_to_sum(h_snap, target_sum)
    return h_snap

def generate_fourier_init(n, rng):
    n_freqs = 20
    freqs = rng.randint(1, n // 2 + 1, size=n_freqs)
    amps = rng.uniform(0.1, 0.3, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(np.pi * f * x + p)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    return h

def generate_binary_pattern(n, rng, strategy='random'):
    h = np.zeros(n)
    n_ones = n // 2
    if strategy == 'random':
        ones = rng.choice(n, size=n_ones, replace=False)
        h[ones] = 1.0
    elif strategy == 'golden':
        golden = (1 + np.sqrt(5)) / 2
        positions = np.sort((np.arange(n_ones) * golden) % 1 * n).astype(int)
        h[positions] = 1.0
    elif strategy == 'interleaved':
        h[0::2] = 1.0
    elif strategy == 'block3':
        positions = []
        for i in range(n_ones // 3 + 1):
            base = i * 3
            for j in range(min(2, n - base)):
                positions.append(base + j)
        positions = positions[:n_ones]
        h[positions] = 1.0
    return project_to_sum(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_n % 2 != 0:
        best_n += 1
        if best_h is not None:
            best_h = np.resize(best_h, best_n)
            best_h = project_to_sum(best_h, best_n / 2.0)
        else:
            best_h = np.full(best_n, 0.5)
    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0
    h_cur = best_h.copy()

    # Phase 1: Nesterov with adaptive alpha
    alphas_p1 = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    for alpha in alphas_p1:
        if time_left() < 80:
            break
        n_iters = max(500, min(3000, int(time_left() * 1.5)))
        h_opt, c5_opt = nesterov_optimize(h_cur, n1, ts1, alpha, n_iters, 0.15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    # Phase 2: SLSQP refinement
    if time_left() > 120:
        for alpha in [200.0, 500.0, 1000.0]:
            if time_left() < 60:
                break
            maxiter = max(50, min(400, int(time_left() * 0.7)))
            h_opt = slsqp_step(h_cur, n1, ts1, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cur = h_opt

    # Phase 3: Simulated annealing
    if time_left() > 150:
        T_init = 0.05
        T_min = 0.001
        n_sa = min(2000, int(time_left() * 1.2))
        h_sa, c5_sa = simulated_annealing(h_cur, n1, ts1, 300.0, n_sa, T_init, T_min, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_cur = h_sa

    # Phase 4: Binary pressure optimization
    if time_left() > 120:
        for lam in [0.1, 0.3, 1.0, 3.0]:
            if time_left() < 60:
                break
            n_iters = max(300, min(1500, int(time_left() * 1.0)))
            h_bin, c5_bin = nesterov_optimize(h_cur, n1, ts1, 500.0, n_iters, 0.1, lambda_bin=lam)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            h_cur = h_bin

    # Phase 5: Binary snap and re-optimize
    if time_left() > 60:
        h_snap = binary_snap(best_h, ts1)
        c5_snap = compute_c5(h_snap)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()

        h_work = best_h.copy()
        for alpha in [300.0, 800.0, 2000.0]:
            if time_left() < 30:
                break
            h_opt, c5_opt = nesterov_optimize(h_work, n1, ts1, alpha, 500, 0.08)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_work = h_opt
            h_snap2 = binary_snap(h_work, ts1)
            c5_snap2 = compute_c5(h_snap2)
            if c5_snap2 < best_c5:
                best_c5 = c5_snap2
                best_h = h_snap2.copy()
                h_work = h_snap2

    # Phase 6: Coordinate descent polish
    if time_left() > 60:
        deltas = [-0.03, -0.015, -0.008, -0.004, -0.002, 0.002, 0.004, 0.008, 0.015, 0.03]
        h_cd, c5_cd = coord_descent_fixed(best_h, n1, ts1, 20, rng, deltas)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Diverse restarts
    if time_left() > 200:
        strategies = ['golden', 'random', 'block3']
        for strat in strategies:
            if time_left() < 80:
                break
            h_init = generate_binary_pattern(n1, rng, strat)
            h_r, c5_r = nesterov_optimize(h_init, n1, ts1, 100.0, 800, 0.15)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                h_cur = h_r
            else:
                h_cur = h_r

            if time_left() > 80:
                h_r2, c5_r2 = nesterov_optimize(h_r, n1, ts1, 400.0, 600, 0.08)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 8: Fourier init restart
    if time_left() > 150:
        for _ in range(3):
            if time_left() < 60:
                break
            h_f = generate_fourier_init(n1, rng)
            h_fo, c5_fo = nesterov_optimize(h_f, n1, ts1, 50.0, 1000, 0.2)
            if c5_fo < best_c5:
                best_c5 = c5_fo
                best_h = h_fo.copy()
            if time_left() > 60:
                h_fo2, c5_fo2 = nesterov_optimize(h_fo, n1, ts1, 500.0, 800, 0.08)
                if c5_fo2 < best_c5:
                    best_c5 = c5_fo2
                    best_h = h_fo2.copy()

    # Phase 9: Final high-alpha polish
    if time_left() > 40:
        for alpha in [1000.0, 3000.0, 8000.0]:
            if time_left() < 20:
                break
            h_pol, c5_pol = nesterov_optimize(best_h, n1, ts1, alpha, 400, 0.05)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        h_snap_final = binary_snap(best_h, ts1)
        c5_snap_final = compute_c5(h_snap_final)
        if c5_snap_final < best_c5:
            best_c5 = c5_snap_final
            best_h = h_snap_final.copy()

        if time_left() > 20:
            h_pol2, c5_pol2 = nesterov_optimize(best_h, n1, ts1, 5000.0, 300, 0.03)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    # Final processing
    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
