import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Euclidean projection onto {x: sum(x)=target, 0<=x<=1}"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def soft_max_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_c = np.exp(shifted)
    s = np.sum(exp_c)
    w = exp_c / s
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-8
    active = np.where(w > threshold)[0]
    
    for k in active:
        l = k - (n - 1)
        wk = w[k]
        if l >= 0:
            # First term: grad[0:n-l] += wk * (1-h[l:n])
            if n - l > 0:
                grad[0:n-l] += wk * (1.0 - h[l:n])
            # Second term: grad[l:n] -= wk * h[0:n-l]
            if n - l > 0:
                grad[l:n] -= wk * h[0:n-l]
        else:
            lp = -l
            # First term: grad[lp:n] += wk * (1-h[0:n-lp])
            if n - lp > 0:
                grad[lp:n] += wk * (1.0 - h[0:n-lp])
            # Second term: grad[0:n-lp] -= wk * h[lp:n]
            if n - lp > 0:
                grad[0:n-lp] -= wk * h[lp:n]
    
    return grad

def nesterov_run(h_init, n, target_sum, alpha, n_iters, lr_init, time_limit):
    t_start = time.time()
    h = h_init.copy()
    y = h.copy()
    t_val = 1.0
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        g = soft_max_grad(y, alpha)
        h_new = y - lr * g
        h_new = project_h(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mu = (t_val - 1.0) / t_new
        y = h_new + mu * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            stall += 1
            if stall > 100:
                lr *= 0.7
                stall = 0

def coord_descent(h_init, n, target_sum, n_rounds, rng, time_limit):
    t_start = time.time()
    h = h_init.copy()
    c5 = compute_c5(h)
    
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            # Try small perturbations
            for d in [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-12:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    
    return h, c5

def soft_max_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_c = np.exp(shifted)
    s = np.sum(exp_c)
    w = exp_c / s
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    
    for k in active:
        l = k - (n - 1)
        wk = w[k]
        if l >= 0:
            # First term: (1-h[i+l]) for i+l in [0,n) => i in [0, n-1-l]
            if l < n:
                grad[0:n-l] += wk * (1.0 - h[l:n])
            # Second term: -h[i-l] for i-l in [0,n) => i in [l, n-1]
            if l < n:
                grad[l:n] -= wk * h[0:n-l]
        else:
            lp = -l  # positive
            # First term: (1-h[i+l]) = (1-h[i-lp]) for i-lp in [0,n) => i in [lp, n-1]
            if lp < n:
                grad[lp:n] += wk * (1.0 - h[0:n-lp])
            # Second term: -h[i-l] = -h[i+lp] for i+lp in [0,n) => i in [0, n-1-lp]
            if lp < n:
                grad[0:n-lp] -= wk * h[lp:n]
    
    return grad
