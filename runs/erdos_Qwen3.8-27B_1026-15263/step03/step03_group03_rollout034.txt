def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):

        else:

    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Structured candidates

                 (j + n // 3) % n, (j + 2 * n // 3) % n]
        # Random candidates
        for _ in range(n_cand - 5):
            k_set.append(rng.randint(n))
        
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:

                best_delta = delta_opt
                best_D = D
                best_k = k

            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

            break

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * nm1 - j, 2 * n - 2)

    return B

        h_t = project_to_sum(interp_h(best_h, best_n, target_n), ts_t)
        omh_t = 1.0 - h_t
        corr_t = np.correlate(h_t, omh_t, mode="full")
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = target_n

        stall = 0
        res_end = time.time() + per_res_budget
        while time.time() < res_end:
            if stall < 50:
                nc = 12
            elif stall < 150:
                nc = 18
            else:
                nc = 25
            h_t, corr_t, omh_t = pair_sweep(h_t, corr_t, omh_t, rng, n_cand=nc)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-10:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = target_n
                stall = 0
            else:
                stall += 1

            if stall == 80:
                h_t = project_to_sum(h_t + perturb_low_freq(h_t, target_n, rng, 0.04), ts_t)
                omh_t = 1.0 - h_t
                corr_t = np.correlate(h_t, omh_t, mode="full")
                stall = 0
            elif stall == 200:
                h_t = project_to_sum(h_t + perturb_low_freq(h_t, target_n, rng, 0.08), ts_t)
                omh_t = 1.0 - h_t
                corr_t = np.correlate(h_t, omh_t, mode="full")
                stall = 0
            elif stall == 400:
                h_t = project_to_sum(h_t + perturb_low_freq(h_t, target_n, rng, 0.12), ts_t)
                omh_t = 1.0 - h_t
                corr_t = np.correlate(h_t, omh_t, mode="full")
                stall = 0
            elif stall == 650:
                h_t = project_to_sum(h_t + rng.uniform(-0.15, 0.15, target_n), ts_t)
                omh_t = 1.0 - h_t
                corr_t = np.correlate(h_t, omh_t, mode="full")
                stall = 0
            elif stall >= 900:
                h_t = project_to_sum(interp_h(best_h, best_n, target_n), ts_t)
                omh_t = 1.0 - h_t
                corr_t = np.correlate(h_t, omh_t, mode="full")
                stall = 0

    # Phase 3: Final polish (30% budget - 5s safety)
    h_fin = best_h.copy()
    n_fin = best_n
    omh_fin = 1.0 - h_fin
    corr_fin = np.correlate(h_fin, omh_fin, mode="full")

    while elapsed() < budget_s - 5:
        h_fin, corr_fin, omh_fin = pair_sweep(h_fin, corr_fin, omh_fin, rng, n_cand=25)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
