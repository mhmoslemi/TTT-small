import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k1_start = max(0, j)
    k1_end = min(2 * n - 2, j + nm1)
    if k1_start <= k1_end:
        k1 = np.arange(k1_start, k1_end + 1)
        B[k1] += (1.0 - h[j + nm1 - k1])
    k2_start = max(0, nm1 - j)
    k2_end = min(2 * n - 2, 2 * nm1 - j)
    if k2_start <= k2_end:
        k2 = np.arange(k2_start, k2_end + 1)
        B[k2] += -h[j + k2 - nm1]
    return B

def solve_1d(A, D, lo, hi):
    # Minimize max(A + D*t) over t in [lo, hi]
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Coarse grid
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    # Golden section or ternary search
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1

                           (j + n // 3) % n, (j + 2 * n // 3) % n]))
        for _ in range(max(0, n_cand - len(k_list))):
            k_list.append(rng.randint(n))
        for k in k_list:
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - k_idx + n - 1
    second_idx = j_idx + k_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    grad = np.sum(w[None, :] * (valid_f * (1.0 - h[fc]) - valid_s * h[sc]), axis=1)
    return obj, grad, corr

    h_new = h - lr * grad
    h_new = project_to_sum(h_new, target_sum)
    obj_new, _, _ = smooth_max_grad(h_new, alpha)
    return h_new, obj, obj_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k1_start = max(0, j)
    k1_end = min(2 * n - 2, j + nm1)
    if k1_start <= k1_end:
        k1 = np.arange(k1_start, k1_end + 1)
        B[k1] += (1.0 - h[j + nm1 - k1])
    k2_start = max(0, nm1 - j)
    k2_end = min(2 * n - 2, 2 * nm1 - j)
    if k2_start <= k2_end:
        k2 = np.arange(k2_start, k2_end + 1)
        B[k2] += -h[j + k2 - nm1]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (j + n // 4) % n, (j + 3 * n // 4) % n]
        for _ in range(max(0, n_cand - 7)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - k_idx + n - 1
    second_idx = j_idx + k_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    grad = np.sum(w[None, :] * (valid_f * (1.0 - h[fc]) - valid_s * h[sc]), axis=1)
    return obj, grad

def projected_gradient_optimize(h0, target_sum, alphas, max_iter_per_alpha, t_check):
    n = len(h0)
    h_cur = h0.copy()
    best_obj = float('inf')
    best_h_pg = h0.copy()
    for alpha in alphas:
        lr = 0.5 / alpha
        obj_cur, _ = smooth_max_grad(h_cur, alpha)
        for _ in range(max_iter_per_alpha):
            if time.time() >= t_check:
                return best_h_pg
            _, grad_cur = smooth_max_grad(h_cur, alpha)
            grad_norm = np.max(np.abs(grad_cur))
            if grad_norm < 1e-12:
                break
            h_try = project_to_sum(h_cur - lr * grad_cur, target_sum)
            obj_try, _ = smooth_max_grad(h_try, alpha)
            if obj_try < obj_cur - 1e-14:
                h_cur = h_try
                obj_cur = obj_try
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_h_pg = h_cur.copy()
            else:
                lr *= 0.6
            if lr < 1e-10:
                break
    return best_h_pg

def structured_restart(h_curr, n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    rtype = rng.randint(4)
    if rtype == 0:
        freq = rng.randint(3, 20)
        phase = rng.uniform(0, 2 * np.pi)
        pattern = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
    elif rtype == 1:
        period = rng.choice([4, 6, 8, 12, 16])
        pattern = (np.sin(np.pi * n / period * x) > 0).astype(float)
    elif rtype == 2:
        nblocks = rng.randint(3, 12)
        pattern = np.zeros(n)
        per_block = n // nblocks
        for i in range(nblocks):
            if i % 2 == 0:
                pattern[i * per_block:(i + 1) * per_block] = 1.0
    else:
        freq1 = rng.randint(2, 10)
        freq2 = rng.randint(10, 30)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        pattern = 0.5 + 0.3 * np.sin(np.pi * freq1 * x + p1) + 0.2 * np.sin(np.pi * freq2 * x + p2)
    pattern = project_to_sum(pattern, n / 2.0)
    mix = 0.5 + 0.3 * rng.uniform()
    h_new = project_to_sum(mix * h_curr + (1 - mix) * pattern, n / 2.0)
    return h_new
