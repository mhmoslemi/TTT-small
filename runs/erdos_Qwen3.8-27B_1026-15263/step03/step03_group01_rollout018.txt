def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        diff = target_sum - np.sum(h)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    m = 2 * n - 1
    nj = np.arange(n)[:, None]
    nm = np.arange(m)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=1000):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_descend(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        # Backtracking line search
        step = lr
        for _ in range(30):
            h_new = project_to_sum(h - step * grad, ts)
            obj_new, _ = smooth_max_obj_grad(h_new, alpha)
            if obj_new < obj_prev:
                break
            step *= 0.5
        else:
            break
        if step < 1e-15:
            break
        h = h_new
        obj_prev = obj_new
        lr = min(step * 1.5, 0.5)
    return h

def binarize_push(h, n, ts, n_rounds=100, threshold=0.3):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        # Try to push the most extreme values further
        h_sorted = np.sort(h)
        # Values closest to 0 (from above) and closest to 1 (from below)
        low_mask = h < threshold
        high_mask = h > 1 - threshold
        n_low = np.sum(low_mask)
        n_high = np.sum(high_mask)
        if n_low == 0 or n_high == 0:

        # Push: move mass from low to high

        h[low_mask] -= delta

        c5_new = compute_c5(h)
        if c5_new > c5_cur:

    return h, c5_cur
