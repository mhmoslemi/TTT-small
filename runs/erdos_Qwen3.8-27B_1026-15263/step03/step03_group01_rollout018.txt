import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    corr = get_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + np.log(np.sum(exp_v)) / alpha
    n = len(h)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    idx_pos = nj + nm - n + 1
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    idx_pos_c = np.clip(idx_pos, 0, n - 1)
    idx_neg = nj + n - 1 - nm
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    idx_neg_c = np.clip(idx_neg, 0, n - 1)
    grad = np.sum(w[None, :] * (valid_neg * (1.0 - h[idx_neg_c]) - valid_pos * h[idx_pos_c]), axis=1)
    return obj, grad

def nesterov_pg(h0, n, ts, alpha, max_iter=5000, lr_init=0.1):
    h = h0.copy().astype(float)
    y = h.copy()
    t_k = 1.0
    obj_prev, _ = smooth_max_obj_grad(y, alpha)
    best_h = h.copy()
    best_obj = obj_prev
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        _, grad = smooth_max_obj_grad(y, alpha)
        x_new = project_to_sum(y - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(x_new, alpha)
        if obj_new > obj_prev + 1e-12:
            lr *= 0.5
            if lr < 1e-14:
                break
            continue
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + (t_k - 1.0) / (t_new + 1.0) * (x_new - y)
        y = project_to_sum(y, ts)
        h = x_new
        obj_prev = obj_new
        t_k = t_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h.copy()
        if obj_new < best_obj + 1e-14:
            stall += 1
        else:
            stall = 0
        if stall > 200:
            break
    return best_h

def binary_local_search(h_bin, n, max_iter=20000):
    h = h_bin.copy()
    ts = n / 2.0
    c5_cur = compute_c5(h)
    x = np.arange(n)
    for it in range(max_iter):
        ones_idx = x[h > 0.5]
        zeros_idx = x[h < 0.5]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.randint(len(ones_idx))
        j = np.random.randint(len(zeros_idx))
        a = ones_idx[i]
        b = zeros_idx[j]
        h[a], h[b] = h[b], h[a]
        c5_new = compute_c5(h)
        if c5_new > c5_cur:
            h[a], h[b] = h[b], h[a]
        else:
            c5_cur = c5_new
    return h, c5_cur

def lag_aware_perturb(h, rng, n, ts, n_steps=30000, delta_max=0.3):
    h = h.copy().astype(float)
    c5_cur = compute_c5(h)
    corr = get_corr(h)
    worst_lag = np.argmax(corr)
    temp = 0.1

            corr = get_corr(h)
            worst_lag = np.argmax(corr)
        i = rng.randint(n)
        j = rng.randint(n)

            continue
        delta = rng.uniform(0.01, delta_max)
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            if rng.random() < np.exp(-(c5_new - c5_cur) / max(1e-6, temp)):
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-7, 0.1 * np.exp(-step / 10000.0))
    return h, c5_cur

def slsqp_refine(h0, n, ts, alpha, maxiter=2000):
    def f(x):
        return smooth_max_obj_grad(project_to_sum(x, ts), alpha)[0]
    def g(x):
        return smooth_max_obj_grad(project_to_sum(x, ts), alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)
