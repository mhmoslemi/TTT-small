def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, time_limit=None, t_start=None):

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    stall_count = 0
    for i in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 100 == 0:
                lr *= 0.9
        h = h_new
    return best_h, best_c5

                raise ValueError("time out")
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v

        return g
    
    try:

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_opt = project_to_sum(res.x, target_sum)

def fourier_perturb(h0, n, K, amplitude, rng):
    """Add a random Fourier perturbation to h0."""
    x = np.linspace(0, 2, n, endpoint=False)
    coeffs = rng.uniform(-amplitude, amplitude, 2 * K)
    perturbation = np.zeros(n)
    for k in range(K):
        perturbation += coeffs[2*k] * np.cos(k * np.pi * x)
        perturbation += coeffs[2*k+1] * np.sin(k * np.pi * x)
    h_new = h0 + perturbation
    return project_to_sum(h_new, n // 2)

def fine_coord_descent(h0, n, target_sum, max_rounds, rng, steps_per_dim, delta):
    """Very fine coordinate descent."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            best_val = c5_cur
            best_i = -1
            for s in range(-steps_per_dim, steps_per_dim + 1):
                d = s * delta
                nv = hi + d
                if nv < 0 or nv > 1:
                    continue
                h_new = h.copy()
                h_new[i] = nv
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-16:
                    best_val = c5_new
                    best_i = s
            if best_i >= 0:
                h[i] = hi + best_i * delta
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w_m = w[m_idx]
        if w_m < 1e-12:
            continue
        offset = m_idx - (n - 1)
        i_start1 = max(0, -offset)
        i_end1 = min(n, n - offset)
        if i_end1 > i_start1:
            g[i_start1:i_end1] += w_m * (1.0 - h[i_start1 + offset:i_end1 + offset])
        i_start2 = max(0, offset)
        i_end2 = min(n, n + offset)
        if i_end2 > i_start2:
            g[i_start2:i_end2] -= w_m * h[i_start2 - offset:i_end2 - offset]
    return sm_val, g

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, time_limit=None, t_start=None):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    stall_count = 0
    for _ in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 200 == 0:
                lr *= 0.85
        h = h_new
    return best_h, best_c5

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter, time_limit=None, t_start=None):
    def obj_fn(x):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                raise ValueError("timeout")
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    try:
        res = minimize(obj_fn, h0.copy(), jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

        perturbation += coeffs[2 * k + 1] * np.sin(k * np.pi * x)
    h_new = h0 + perturbation

        h_cd, _ = fine_coord_descent(h_cur, n, ts, 100, rng, 0.001, 21)
        try_update(h_cd, n)
        h_cur = h_cd

    if _tl(t_start, budget_s) > 60:
        for amp in [0.03, 0.05, 0.08, 0.12]:
            if _tl(t_start, budget_s) < 40:
                break
            for trial in range(3):
                h_pert = fourier_perturb(h_cur, n, 8, amp, rng)
                try_update(h_pert, n)
                h_opt, _ = adam_optimize(h_pert, n, ts, 300.0, 1500, 0.04,
                                         time_limit=_tl(t_start, budget_s) - 15, t_start=t_start)
                try_update(h_opt, n)
                h_cur = h_opt
                if _tl(t_start, budget_s) < 25:
                    break

    if _tl(t_start, budget_s) > 120:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        if abs(ts2 - round(ts2)) < 1e-9:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_to_sum(h_up, ts2)
            try_update(h_up, n2)
            h_cur2 = h_up.copy()

            for alpha2 in [150.0, 400.0, 800.0]:
                if _tl(t_start, budget_s) < 40:
                    break
                h_a2, _ = adam_optimize(h_cur2, n2, ts2, alpha2, 2000, 0.04,
                                        time_limit=_tl(t_start, budget_s) - 10, t_start=t_start)
                try_update(h_a2, n2)
                h_cur2 = h_a2

            if _tl(t_start, budget_s) > 50:
                h_lb2 = lbfgs_optimize(h_cur2, n2, ts2, 600.0, 300,
                                       time_limit=_tl(t_start, budget_s) - 10, t_start=t_start)
                try_update(h_lb2, n2)
                h_cur2 = h_lb2

            if _tl(t_start, budget_s) > 30:
                h_cd2, _ = fine_coord_descent(h_cur2, n2, ts2, 30, rng, 0.005, 9)
                try_update(h_cd2, n2)
                h_cur2 = h_cd2

            if _tl(t_start, budget_s) > 25:
                h_cd2, _ = fine_coord_descent(h_cur2, n2, ts2, 50, rng, 0.002, 13)
                try_update(h_cd2, n2)
                h_cur2 = h_cd2

            if _tl(t_start, budget_s) > 60:
                for amp in [0.02, 0.04, 0.06]:
                    if _tl(t_start, budget_s) < 35:
                        break
                    for trial in range(3):
                        h_pert2 = fourier_perturb(h_cur2, n2, 10, amp, rng)
                        try_update(h_pert2, n2)
                        h_opt2, _ = adam_optimize(h_pert2, n2, ts2, 500.0, 1200, 0.02,
                                                  time_limit=_tl(t_start, budget_s) - 12, t_start=t_start)
                        try_update(h_opt2, n2)
                        h_cur2 = h_opt2
                        if _tl(t_start, budget_s) < 20:
                            break

    n_f = best_n
    ts_f = n_f / 2.0
    h_f = best_h.copy()

    if _tl(t_start, budget_s) > 30:
        h_f, _ = adam_optimize(h_f, n_f, ts_f, 800.0, 2000, 0.015,
                               time_limit=_tl(t_start, budget_s) - 10, t_start=t_start)
        try_update(h_f, n_f)

    if _tl(t_start, budget_s) > 20:
        h_f, _ = fine_coord_descent(h_f, n_f, ts_f, 50, rng, 0.003, 15)
        try_update(h_f, n_f)

    if _tl(t_start, budget_s) > 15:
        h_f, _ = fine_coord_descent(h_f, n_f, ts_f, 80, rng, 0.001, 21)
        try_update(h_f, n_f)

    if _tl(t_start, budget_s) > 10:
        h_f, _ = fine_coord_descent(h_f, n_f, ts_f, 100, rng, 0.0005, 31)
        try_update(h_f, n_f)

    if _tl(t_start, budget_s) > 8:
        h_f, _ = fine_coord_descent(h_f, n_f, ts_f, 100, rng, 0.0002, 41)
        try_update(h_f, n_f)

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
