    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)

            break

        else:

        h = np.clip(h, 0.0, 1.0)

    n = len(h)

Then O_k(t) = O_k(v) + (t-v) * c_k = c_k * t + (O_k(v) - c_k * v)

So O_k as a function of t = h[i] is: O_k(t) = c_k * t + d_k where d_k = O_k(v) - c_k * v.

The max of these linear functions is convex and piecewise linear in t. Its minimum can be found by:
1. Finding all breakpoints: t where c_k * t + d_k = c_l * t + d_l, i.e., t = (d_l - d_k) / (c_k - c_l)
2. Evaluating at each valid breakpoint (in [0,1]) and at boundaries

Or more efficiently: since the function is convex, I can use binary search on the slope.

The slope at point t is: the average of c_k over all k that are active (i.e., O_k(t) = max).
- If all active c_k > 0, the function is increasing, so we should decrease t.
- If all active c_k < 0, the function is decreasing, so we should increase t.
- If there are active c_k both positive and negative (or zero), we're at a minimum.

Algorithm:
1. Compute c_k and d_k for all k
2. Current t = h[i]
3. Evaluate O_k(t) for all k, find max M and active set A
4. If min_{k∈A} c_k ≤ 0 ≤ max_{k∈A} c_k, stop (we're at minimum)
5. If all active c_k > 0: we need to decrease t. Find the smallest t' < t where a new constraint becomes active or an active one drops out. Move to t'.
6. If all active c_k < 0: we need to increase t. Similar.
7. Repeat until we find the minimum or hit a boundary.

Step 5/6: The next breakpoint in the direction of movement is the minimum (or maximum) of the breakpoints t = (d_l - d_k)/(c_k - c_l) over all pairs (k,l) where k is active and l is not (for decreasing t) or k is not active and l is active (for increasing t).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr[n-1:]
    m = np.max(O)
    shifted = alpha * dx * (O - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / n) / (alpha * dx)
    conv_A2_full = np.correlate(h, w, mode='full')
    conv_A2 = conv_A2_full[:n]
    conv_B_full = np.convolve(w, h, mode='full')
    conv_B = conv_B_full[:n]
    pw = np.zeros(n + 1)
    pw[1:] = np.cumsum(w)
    pw_rev = pw[1:][::-1]
    grad = dx * (pw_rev - conv_A2 - conv_B)
    return sm_val, grad

def nesterov_optimize(h0, n, target_sum, alpha_init, alpha_final, n_iters, lr_init):
    h = h0.copy()
    h_prev = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    decay = (alpha_final / alpha_init) ** (1.0 / max(n_iters, 1))
    alpha = alpha_init
    for it in range(n_iters):
        alpha = alpha_init * (decay ** it)
        t += 1
        y = h + (t - 1) / (t + 2) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target_sum)
        _, g = fast_smooth_max_and_grad(y, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_prev = h.copy()
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 100 == 0:
            lr = max(lr_init * 0.95 ** (it // 10), lr_init * 0.3)
    return best_h, best_c5

def fixed_lr_subgradient(h0, n, target_sum, alpha_init, alpha_final, n_iters, lr_init):
    h = h0.copy()
    h_prev = h0.copy()
    t = 0
    best_c5 = compute_c5(h)

def coord_descent_ternary(h0, n, target_sum, max_rounds, rng, n_term=35):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = 0.0
            hi = 1.0
            def obj_func(t):
                h_new = h.copy()
                h_new[i] = t
                h_new = project_to_sum(h_new, target_sum)
                return compute_c5(h_new)
            f_lo = obj_func(lo)
            f_hi = obj_func(hi)
            for _ in range(n_term):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = obj_func(m1)
                f2 = obj_func(m2)
                if f1 <= f2:
                    hi = m2
                    f_hi = f2
                else:
                    lo = m1
                    f_lo = f1
            t_opt = (lo + hi) / 2.0
            h_new = h.copy()
            h_new[i] = t_opt
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h = h_new
                c5_cur = c5_new
                improved = True
        if not improved:
            break
    return h, c5_cur

def topk_exact_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    momentum = 0.85
    v_mom = np.zeros(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            jj1 = j_arr + (m - n + 1)
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - (m - n + 1)
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        v_mom = momentum * v_mom + (1.0 - momentum) * g
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = h - lr * v_mom / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
