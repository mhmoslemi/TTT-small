def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, lags, weights):

    n = len(h)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        g += w * corr_gradient(h, lag)
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]

    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    """Round h to binary while maintaining sum constraint."""
    n = len(h)
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:int(target_sum)]] = 1.0
    return h_bin

def subgradient_step(h, target_sum, lags_weights, step_size):
    """Take a projected subgradient step."""
    n = len(h)
    g = np.zeros(n)

    g_centered = g - np.mean(g)

    h_new = project_to_sum(h_new, target_sum)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, lags, weights):
    n = len(h)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w > 1e-8:
            g += w * corr_gradient(h, lag)
    return g

def line_search_pair(h, i_inc, i_dec, max_move, n_divs=6):

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]).astype(float) * w[None, :]
    cs = valid_s * h[sc].astype(float) * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_max_and_grad(x, alpha)[0], h0.copy(),
                       method='SLSQP', jac=lambda x: smooth_max_and_grad(x, alpha)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def block_perturb(h, rng, n_blocks=5, strength=0.08):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        size = max(1, n // (2 * n_blocks))
        start = rng.randint(0, n - size)
        h_new[start:start + size] += rng.uniform(-strength, strength)
    return h_new

def filtered_noise_perturb(h, rng, base_amp=0.04):
    n = len(h)
    noise = rng.uniform(-base_amp, base_amp, n)
    sigma = rng.uniform(2.0, 15.0)
    noise = gaussian_filter1d(noise, sigma=sigma)
    return h + noise

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: SLSQP polish at parent resolution
    for alpha in [80, 400, 2000]:
        if time_left() < 25:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Coordinate descent with line search
    if time_left() > 100:
        h_cd, c5_cd = coord_desc_lsearch(best_h, ts_parent, max_iters=12000, n_lags=3, n_divs=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Fast coordinate descent for additional refinement
    if time_left() > 80:
        h_f, c5_f = coord_desc_fast(best_h, ts_parent, max_iters=30000, step_frac=0.08)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 4: Multi-scale resolution increase
    target_resolutions = [720, 900, 1080, 1260]
    for n_tr in target_resolutions:
        if time_left() < 150:
            break
        if n_tr % 2 != 0:
            continue
        if n_tr <= best_n:
            continue
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

        # SLSQP at new resolution (only if not too large)
        if n_tr <= 900:
            for alpha in [100, 600, 3000]:
                if time_left() < 50:
                    break
                h_opt = slsqp_local(h_interp, n_tr, ts_tr, alpha, maxiter=400)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                h_interp = h_opt

        # Coordinate descent with line search at new resolution
        if time_left() > 60:
            h_cd2, c5_cd2 = coord_desc_lsearch(h_interp, ts_tr, max_iters=6000, n_lags=3, n_divs=5)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_tr

        # Fast coordinate descent
        if time_left() > 40:
            h_fast, c5_fast = coord_desc_fast(best_h, ts_tr, max_iters=20000, step_frac=0.07)
            if c5_fast < best_c5:
                best_c5 = c5_fast
                best_h = h_fast.copy()

    n_best = best_n
    ts_best = n_best / 2.0

    # Phase 5: Random restarts with various strategies
    n_trials = 18
    for trial in range(n_trials):
        if time_left() < 50:
            break
        frac = trial / float(n_trials)
        if trial % 4 == 0:
            noise = 0.02 + 0.05 * (1.0 - frac)
            h_shake = best_h + rng.uniform(-noise, noise, n_best)
        elif trial % 4 == 1:
            h_shake = block_perturb(best_h, rng, n_blocks=4 + trial % 4, strength=0.06)
        elif trial % 4 == 2:
            h_shake = filtered_noise_perturb(best_h, rng, base_amp=0.03 + 0.03 * (1.0 - frac))
        else:
            # Cyclical perturbation
            h_shake = best_h.copy()
            freq = rng.randint(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.04 * (1.0 - frac)
            x = np.linspace(0, 2.0, n_best, endpoint=False)
            h_shake += amp * np.sin(np.pi * freq * x + phase)

        h_shake = project_to_sum(h_shake, ts_best)

        # Quick SLSQP
        h_opt = slsqp_local(h_shake, n_best, ts_best, 300, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Coordinate descent refinement
        if time_left() > 30:
            h_cd3, c5_cd3 = coord_desc_lsearch(h_opt, ts_best, max_iters=4000, n_lags=3, n_divs=5)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()

        # Fast coordinate descent
        if time_left() > 25:
            h_fast2, c5_fast2 = coord_desc_fast(best_h, ts_best, max_iters=10000, step_frac=0.06)
            if c5_fast2 < best_c5:
                best_c5 = c5_fast2
                best_h = h_fast2.copy()

    # Phase 6: High-resolution attempt from best solution
    if time_left() > 180 and best_n < 1440:
        n_hi = min(1440, best_n + 360)
        if n_hi % 2 != 0:
            n_hi += 1
        ts_hi = n_hi / 2.0
        h_hi = interp_to_n(best_h, n_hi)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
        # Coordinate descent at high res
        if time_left() > 60:
            h_cd4, c5_cd4 = coord_desc_lsearch(h_hi, ts_hi, max_iters=5000, n_lags=4, n_divs=5)
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()
                best_n = n_hi
        if time_left() > 40:
            h_fast3, c5_fast3 = coord_desc_fast(best_h, n_hi, max_iters=15000, step_frac=0.06)
            if c5_fast3 < best_c5:
                best_c5 = c5_fast3
                best_h = h_fast3.copy()

    # Phase 7: Final intensive polish
    n_best = best_n
    ts_best = n_best / 2.0

    if time_left() > 50:
        h_pol, c5_pol = coord_desc_lsearch(best_h, ts_best, max_iters=15000, n_lags=5, n_divs=7)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 30:
        for alpha in [8000, 30000]:
            if time_left() < 20:
                break
            h_s = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1000)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    if time_left() > 20:
        h_fin, c5_fin = coord_desc_fast(best_h, ts_best, max_iters=50000, step_frac=0.04)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if time_left() > 15:
        h_fin2, c5_fin2 = coord_desc_lsearch(best_h, ts_best, max_iters=10000, n_lags=6, n_divs=8)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
