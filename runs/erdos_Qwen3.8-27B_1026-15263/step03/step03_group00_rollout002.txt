        lags = [t - (n - 1) for t in top_idx]
        vals = np.array([corr[t] for t in top_idx])
        spread = vals[0] - vals[-1]
        if spread < 1e-12:
            weights = np.ones(top_k) / top_k
        else:
            weights = np.exp(-50.0 * (vals - vals[0]) / spread)
            weights /= weights.sum()
        g = np.zeros(n)
        for lag, w in zip(lags, weights):
            g += w * corr_gradient(h, lag)
        g_centered = g - np.mean(g)
        h_new = h - alpha * g_centered
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            alpha = min(alpha * 1.05, 0.3)
        else:
            alpha *= 0.7
        if alpha < 1e-7:
            break
    return best_h, best_c5

def structured_perturb(h, rng, n, noise_scale):
    h_p = h.copy()
    for mode in range(4):
        if mode == 0:
            phase = rng.uniform(0, 2 * np.pi)
            freq = rng.randint(2, 10)
            perturb = noise_scale * np.sin(freq * np.pi * np.arange(n) / n + phase)
            h_p = h + perturb
        elif mode == 1:
            idx = rng.choice(n, size=n // 10, replace=False)
            h_p[idx] += rng.uniform(-noise_scale, noise_scale, len(idx))
        elif mode == 2:
            shift = rng.randint(1, n // 20)
            h_p = np.roll(h, shift)
            h_p = project_to_sum(h_p, n / 2.0)
            h_p += noise_scale * 0.5 * rng.randn(n)
        else:
            h_p = h + noise_scale * rng.randn(n)
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_to_sum(h_p, n / 2.0)
        return h_p
    return project_to_sum(h, n / 2.0)

    t_start = time.time()
