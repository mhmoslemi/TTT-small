    t0 = time.time()

        corr = np.correlate(x, 1.0-x, mode="full")
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        exp_v = np.exp(shifted)

        penalty = 1e6 * (np.sum(x) - target_sum) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
    return g

def gradient_smooth_max(h, w):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        lag = idx - (n - 1)
        grad += w[idx] * corr_gradient(h, lag)
    return grad

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    exp_v = np.exp(shifted)
    return m + np.log(exp_v.sum()) / alpha, exp_v / exp_v.sum()

def project_onto_simplex_interval(h, target_sum):
    h = np.clip(h.copy().astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_smooth_max(h0, n, target_sum, alpha, maxiter):
    def func(x):
        val, _ = smooth_max_obj(x, alpha)
        penalty = 1e5 * (x.sum() - target_sum) ** 2
        return val + penalty
    
    def grad(x):
        _, w = smooth_max_obj(x, alpha)
        g = gradient_smooth_max(x, w)
        g += 2e5 * (x.sum() - target_sum) / n * np.ones(n)
        return g
    
    try:
        res = minimize(func, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_onto_simplex_interval(res.x, target_sum)
    except Exception:
        return project_onto_simplex_interval(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h, period=2.0)

def coordinate_descent(h, target_sum, max_iters=20000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best = c5
    stale = 0
    for _ in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = np.where(h < 1.0 - 1e-9)[0]
        can_dec = np.where(h > 1e-9)[0]
        if len(can_inc) == 0 or len(can_dec) == 0:
            break
        g_inc = np.full(n, np.inf)
        g_inc[can_inc] = g[can_inc]
        g_dec = np.full(n, -np.inf)
        g_dec[can_dec] = g[can_dec]
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        old_iinc, old_idec = h[i_inc], h[i_dec]
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_simplex_interval(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best - 1e-14:
            best = c5_new
            stale = 0
        else:

            h[i_dec] = old_idec
            h = project_onto_simplex_interval(h, target_sum)
            stale += 1
            if stale > 100:
                break
    return h, best

But I can linearize: C_k(h) = sum_i h[i](1-h[i+k]) ≈ C_k(h₀) + g_k(h₀)ᵀ(h - h₀)

     sum(h) = n/2
     0 ≤ h ≤ 1

This is a linear program! I can solve it with scipy.linprog or even just use scipy.optimize.linprog.

Then iterate: solve LP → get new h → compute actual C_k values → update active set → repeat.

def bundle_method(h0, n, target_sum, max_rounds=100):
    h = h0.copy()
    active_lags = []
    t_cur = compute_c5(h)
    
    for round_idx in range(max_rounds):
        corr = compute_corr(h)
        # Find active lags (within tolerance of max)
        max_val = np.max(corr)
        tol = 1e-4
        active_mask = corr >= max_val - tol
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            active_lags = [np.argmax(corr)]
        if len(active_lags) > 50:
            # Keep only top 50
            top_ids = np.argsort(corr)[::-1][:50]
            active_lags = top_ids
        
        # Solve LP: minimize t
        # s.t. for each active lag k: C_k(h_old) + g_k(h_old)·(h - h_old) ≤ t
        #      sum(h) = target_sum
        #      0 ≤ h ≤ 1
        
        # Variables: h[0], ..., h[n-1], t (total n+1 variables)
        # Objective: minimize t → c = [0, ..., 0, 1]
        
        # Constraints:
        # For each active lag: g_kᵀ h - t ≤ C_k(h_old) - g_k·h_old
        # sum(h) = target_sum
        # 0 ≤ h ≤ 1, t free
        
        m = len(active_lags)
        # c vector
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # A_ub, b_ub: g_kᵀ h - t ≤ C_k(h_old) - g_k·h_old
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, idx in enumerate(active_lags):
            lag = idx - (n - 1)
            g_k = corr_gradient(h, lag)
            c_k = corr[idx]  # C_k(h)
            A_ub[i, :n] = g_k
            A_ub[i, n] = -1.0
            b_ub[i] = c_k - g_k @ h
        
        # A_eq, b_eq: sum(h) = target_sum
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        A_eq[0, n] = 0.0
        b_eq = np.array([target_sum])
        
        # Bounds: 0 ≤ h ≤ 1, t free (use large bound)
        big_M = 10.0
        bounds = [(0.0, 1.0)] * n + [(-big_M, big_M)]
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_simplex_interval(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < t_cur:
            h = h_new
            t_cur = c5_new
        else:
            # Step back toward h_old
            h = 0.5 * (h + h_new)
            h = project_onto_simplex_interval(h, target_sum)
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_weights(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    return m + np.log(exp_v.sum()) / alpha, w, corr

def grad_smooth_max(h, w):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-13:
            continue
        lag = idx - (n - 1)
        grad += w[idx] * corr_gradient(h, lag)
    return grad

def lbfgs_smooth(h0, n, ts, alpha, maxiter):
    def f(x):
        val, _, _ = smooth_max_val_weights(x, alpha)
        pen = 1e4 * (x.sum() - ts) ** 2
        return val + pen
    def g(x):
        _, w, _ = smooth_max_val_weights(x, alpha)
        grad = grad_smooth_max(x, w)
        grad += 2e4 * (x.sum() - ts) * np.ones(n) / n
        return grad
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 50})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0.copy(), ts)

def bundle_lp(h, n, ts, max_rounds, max_active):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for rnd in range(max_rounds):
        corr = compute_corr(h_cur)
        max_c = np.max(corr)
        active = np.where(corr >= max_c - 1e-6 * max_c)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > max_active:
            top_idx = np.argsort(corr)[::-1][:max_active]
            active = top_idx
        m = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, idx in enumerate(active):
            lag = idx - (n - 1)
            g_k = corr_gradient(h_cur, lag)
            c_k = corr[idx]
            A_ub[i, :n] = g_k
            A_ub[i, n] = -1.0
            b_ub[i] = c_k - g_k.dot(h_cur)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([ts])
        bounds = [(0.0, 1.0)] * n + [(-10.0, 10.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 5000})
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = project_to_sum(h_lp, ts)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_cur - 1e-12:
            h_new = 0.7 * h_cur + 0.3 * h_lp
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new
            else:
                h_cur = h_lp
                c5_cur = c5_lp
        else:
            h_cur = 0.5 * h_cur + 0.5 * h_lp
            h_cur = np.clip(h_cur, 0.0, 1.0)
            h_cur = project_to_sum(h_cur, ts)
            c5_cur = compute_c5(h_cur)
            if c5_cur >= c5_cur + 1e-10:
                break
    return h_cur, compute_c5(h_cur)

def coord_desc(h, ts, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for _ in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = np.where(h < 1.0 - 1e-9)[0]
        can_dec = np.where(h > 1e-9)[0]
        if len(can_inc) == 0 or len(can_dec) == 0:
            break
        g_inc = np.full(n, np.inf)
        g_inc[can_inc] = g[can_inc]
        g_dec = np.full(n, -np.inf)
        g_dec[can_dec] = g[can_dec]
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_m = min(1.0 - h[i_inc], h[i_dec])
        mv = step_frac * max_m
        if mv < 1e-11:
            break
        oi, od = h[i_inc], h[i_dec]
        h[i_inc] += mv
        h[i_dec] -= mv
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
        else:
            h[i_inc] = oi
            h[i_dec] = od
            h = project_to_sum(h, ts)
            stale += 1
            if stale > 80:
                break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n_cur = best_n
    ts = n_cur / 2.0
    for alpha in [40, 120, 400, 1200, 4000, 12000]:
        if rem() < 50:
            break
        h_try = lbfgs_smooth(best_h, n_cur, ts, alpha, 400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
        best_h = h_try
        n_cur = len(best_h)
        ts = n_cur / 2.0
    if rem() > 80:
        best_h, c5_b = bundle_lp(best_h, n_cur, ts, max_rounds=40, max_active=30)
        if c5_b < best_c5:
            best_c5 = c5_b
    if rem() > 40:
        best_h, c5_cd = coord_desc(best_h, ts, 20000, 0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
    for n_new in [720, 900, 1080]:
        if rem() < 100:
            break
        ts_new = n_new / 2.0
        hi = interp_to_n(best_h, n_new)
        hi = project_to_sum(hi, ts_new)
        for alpha in [300, 1500, 6000]:
            if rem() < 40:
                break
            hi = lbfgs_smooth(hi, n_new, ts_new, alpha, 300)
        if rem() > 60:
            hi, c5_b = bundle_lp(hi, n_new, ts_new, 30, 30)
        if rem() > 30:
            hi, c5_cd = coord_desc(hi, ts_new, 15000, 0.08)
        c5_hi = compute_c5(hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = hi.copy()
            n_cur = n_new
            ts = ts_new
    n_restarts = 10
    for r in range(n_restarts):
        if rem() < 50:
            break
        noise = 0.015 + 0.03 * (1.0 - r / float(n_restarts))
        hs = best_h + rng.uniform(-noise, noise, n_cur)
        hs = project_to_sum(hs, ts)
        for alpha in [800, 3000]:
            if rem() < 25:

            hs = lbfgs_smooth(hs, n_cur, ts, alpha, 250)
        if rem() > 40:

        c5_hs = compute_c5(hs)
        if c5_hs < best_c5:
            best_c5 = c5_hs
            best_h = hs.copy()
    if rem() > 30:
        if rem() > 80:
            best_h, c5_cd2 = coord_desc(best_h, ts, 40000, 0.05)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
        if rem() > 25:
            for alpha in [10000, 50000]:
                if rem() < 15:
                    break
                hp = lbfgs_smooth(best_h, n_cur, ts, alpha, 600)

    best_h = project_to_sum(best_h, n_cur / 2.0)

    return (best_h, best_c5, n_cur)
