```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single_lag(h, lag):
    """Gradient of corr[lag + n - 1] with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, lags, weights):
    """Weighted sum of gradients for multiple lags"""
    g = np.zeros(len(h))
    for lag, w in zip(lags, weights):
        g += w * corr_gradient_single_lag(h, lag)
    return g

def smooth_max_obj_grad(h, alpha, top_k=5):
    """Smooth max of top-k correlation values with gradient"""
    n = len(h)
    corr = compute_corr(h)
    # Use log-sum-exp over all lags weighted by alpha
    # But for efficiency, focus on top lags
    top_indices = np.argsort(corr)[-top_k:]
    shifted = alpha * (corr[top_indices] - corr[top_indices].max())
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    obj = corr[top_indices].max() + np.log(exp_vals.sum()) / alpha

    g = np.zeros(n)
    for idx, wi in zip(top_indices, w):
        lag = idx - (n - 1)
        g += wi * corr_gradient_single_lag(h, lag)
    return obj, g

def slsqp_optimize(h0, target_sum, alpha, maxiter, top_k=5):

        return smooth_max_obj_grad(x, alpha, top_k)[0]

        return smooth_max_obj_grad(x, alpha, top_k)[1]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_gs_target = 0.0
_gs_alpha = 100.0
_gs_topk = 5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_topk(h, alpha, topk):
    corr = compute_corr(h)
    n = len(h)
    idx = np.argsort(corr, kind='stable')
    top_idx = idx[-topk:]
    vals = corr[top_idx]
    vmax = vals.max()
    shifted = alpha * (vals - vmax)
    e = np.exp(shifted)
    w = e / e.sum()
    obj = vmax + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for ti, wi in zip(top_idx, w):
        if wi > 1e-10:
            lag = ti - (n - 1)
            g += wi * lag_gradient(h, lag)
    return obj, g

def _gs_obj(x):
    return smooth_topk(x, _gs_alpha, _gs_topk)[0]

def _gs_grad(x):
    return smooth_topk(x, _gs_alpha, _gs_topk)[1]

def _gs_con(x):
    return np.sum(x) - _gs_target

def _gs_con_jac(x):
    return np.ones(len(x))

def slsqp_run(h0, target_sum, alpha, maxiter, topk):
    global _gs_target, _gs_alpha, _gs_topk
    _gs_target = target_sum
    _gs_alpha = alpha
    _gs_topk = topk
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _gs_con, 'jac': _gs_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_gs_obj, h0.copy(), method='SLSQP', jac=_gs_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        if time.time() - t_start > time_budget:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_lags:]
        lags = top_idx - (n - 1)
        vals = corr[top_idx]
        vmargin = vals - vals.min() + 1e-8
        w = vmargin / vmargin.sum()
        g = np.zeros(n)
        for lag, wi in zip(lags, w):
            g += wi * lag_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        frac = max(0.15, 1.0 - 0.7 * it / float(max_iters))
        max_move = min(1.0 - h[i_inc], h[i_dec])

        h[i_inc] += move

        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1

    c5_cur = best_c5
    T = t_init
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_gs_target = 0.0
_gs_alpha = 100.0
_gs_topk = 5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, tgt):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += 1.0 - h[:n-lag]
            g[:n-lag] -= h[lag:]
    else:
        k = -lag
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    return g

def topk_smooth(h, alpha, topk):
    c = compute_corr(h)
    n = len(h)
    ti = np.argsort(c, kind='stable')[-topk:]
    v = c[ti]
    vm = v.max()
    s = alpha * (v - vm)
    e = np.exp(s)
    w = e / e.sum()
    obj = vm + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for i, wi in zip(ti, w):
        if wi > 1e-15:
            g += wi * lag_grad(h, i - (n-1))
    return obj, g

def _o(x):
    return topk_smooth(x, _gs_alpha, _gs_topk)[0]

def _g(x):
    return topk_smooth(x, _gs_alpha, _gs_topk)[1]

def _c(x):
    return float(x.sum() - _gs_target)

def _cj(x):
    return np.ones(len(x))

def slsqp(h0, tgt, alpha, mi, topk):
    global _gs_target, _gs_alpha, _gs_topk
    _gs_target = tgt
    _gs_alpha = alpha
    _gs_topk = topk
    n = len(h0)
    cons = [{'type':'eq', 'fun':_c, 'jac':_cj}]
    bnds = [(0.0, 1.0)]*n
    try:
        r = minimize(_o, h0.copy(), method='SLSQP', jac=_g,
                     bounds=bnds, constraints=cons,
                     options={'maxiter':mi, 'ftol':1e-14})
        return project_to_sum(r.x, tgt)
    except Exception:
        return project_to_sum(h0.copy(), tgt)

def cdm(h, tgt, mi, tl_, sf, dl):
    # multi-lag coordinate descent
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    st = 0
    for it in range(mi):
        if time.time() > dl:
            break
        c = compute_corr(h)
        tidx = np.argsort(c)[-tl_:]
        lags = tidx - (n-1)
        vals = c[tidx]
        w = (vals - vals.min() + 1e-8)
        w /= w.sum()
        g = np.zeros(n)
        for lg, wi in zip(lags, w):
            g += wi * lag_grad(h, lg)
        ci = h < 1.0 - 1e-10
        cd_ = h > 1e-10
        if not ci.any() or not cd_.any():
            break
        gi = np.where(ci, g, np.inf)
        gd = np.where(cd_, g, -np.inf)
        ii = int(np.argmin(gi))
        id_ = int(np.argmax(gd))
        if ii == id_:
            break
        if g[ii] >= g[id_] - 1e-12:
            break
        fr = max(0.15, 1.0 - 0.7*it/float(mi))
        mv = sf * fr * min(1.0-h[ii], h[id_])
        if mv < 1e-11:
            break
        h[ii] += mv
        h[id_] -= mv
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, tgt)
        c5n = compute_c5(h)
        if c5n >= c5 - 1e-14:
            h[ii] -= mv
            h[id_] += mv
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, tgt)
            st += 1
            if st > 300:
                break
        else:
            c5 = c5n
            st = 0
    return h, c5

def anneal(h0, tgt, rng, dl, ti, tm):
    n = len(h0)
    h = h0.copy()

    bc5 = compute_c5(bh)
    cc5 = bc5
    T = ti
    ns = int((dl - time.time()) * 400)

            lg = m - (n-1)
            g = lag_grad(h, lg)
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                g /= gn
            mk = rng.random(n) < 0.4
            sc = 0.012 * rng.uniform(0.5, 1.0)
            hp[mk] -= sc * g[mk]
        elif r < 0.55:
            bs = int(rng.randint(5, 21))
            i0 = int(rng.randint(0, n - bs))
            ix = np.arange(i0, i0+bs)
            ta = np.arange(bs, dtype=float)
            dp = ta - (bs-1)/2.0
            dm = float(dp @ dp) + 1e-10
            am = rng.uniform(-0.04, 0.04)
            hp[ix] += am * dp / dm * bs
        elif r < 0.95:
            b1 = int(rng.randint(4, 14))
            b2 = int(rng.randint(4, 14))
            i1 = int(rng.randint(0, n-b1))
            i2 = int(rng.randint(0, n-b2))
            if abs(i1-i2) < b1+b2:
                i2 = (i1+b1+int(rng.randint(0, max(1, n-b2)))) % n
            x1 = np.arange(i1, min(i1+b1, n))
            x2 = np.arange(i2, min(i2+b2, n))
            if len(x1) and len(x2):
                sw = rng.uniform(-0.015, 0.015)
                hp[x1] += sw
                hp[x2] -= sw * len(x1) / len(x2)
        else:
            # big jump
            ns2 = n // 4
            ix = rng.choice(n, ns2, replace=False)

            rng.shuffle(v)

        hp = np.clip(hp, 0, 1)

            h = hp
            cc5 = pc5
            if pc5 < bc5:
                bc5 = pc5
                bh = h.copy()
        T *= cool
    return bh, bc5
```

```python
from scipy.optimize import minimize
import time

_gs_target = 0.0
_gs_alpha = 100.0
_gs_topk = 5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, tgt):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    return g

def topk_smooth(h, alpha, topk):
    c = compute_corr(h)
    n = len(h)
    ti = np.argsort(c, kind='stable')[-topk:]
    v = c[ti]
    vm = v.max()
    s = alpha * (v - vm)
    e = np.exp(s)
    w = e / e.sum()
    obj = vm + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for i, wi in zip(ti, w):
        if wi > 1e-15:
            g += wi * lag_grad(h, i - (n - 1))
    return obj, g

def _o(x):
    return topk_smooth(x, _gs_alpha, _gs_topk)[0]

def _g(x):
    return topk_smooth(x, _gs_alpha, _gs_topk)[1]

def _c(x):
    return float(x.sum() - _gs_target)

def _cj(x):
    return np.ones(len(x))

    global _gs_target, _gs_alpha, _gs_topk

    _gs_alpha = alpha
    _gs_topk = topk

                     bounds=bnds, constraints=cons,

        hs = bh + rng.normal(0, nz, n)
        hs = project_to_sum(hs, ts)
        dl_cdm = time.time() + min(rem * 0.2, 22)
        hc, c5c = cdm(hs, ts, 3000, 3, 0.12, dl_cdm)
        if c5c < bc5:
            bc5, bh = c5c, hc
        rem = tleft(t0, budget_s)
        if rem < 20:
            break
        ho = slsqp(hc, ts, 5000, 400, 5)
        c5o = compute_c5(ho)
        if c5o < bc5:
            bc5, bh = c5o, ho

    rem = tleft(t0, budget_s)
    if rem > 25:
        dl = time.time() + min(rem * 0.15, 22)
        hs2, c5s = sgrad(bh, ts, dl)
        if c5s < bc5:
            bc5, bh = c5s, hs2

    rem = tleft(t0, budget_s)
    if rem > 35:
        dl = time.time() + min(rem * 0.45, 70)
        hf, c5f = cdm(bh, ts, 20000, 3, 0.04, dl)
        if c5f < bc5:
            bc5, bh = c5f, hf

    rem = tleft(t0, budget_s)
    if rem > 18:
        ho = slsqp(bh, ts, 100000, 1000, 5)
        c5o = compute_c5(ho)
        if c5o < bc5:
            bc5, bh = c5o, ho

    rem = tleft(t0, budget_s)
    if rem > 12:
        dl = time.time() + min(rem * 0.6, 25)
        hf, c5f = cdm(bh, ts, 10000, 3, 0.02, dl)
        if c5f < bc5:
            bc5, bh = c5f, hf

    rem = tleft(t0, budget_s)
    while rem > 30:
        dl = time.time() + min(rem * 0.25, 50)
        hc, c5c = cdm(bh, ts, 5000, 3, 0.05, dl)
        if c5c < bc5:
            bc5, bh = c5c, hc
        rem = tleft(t0, budget_s)
        if rem < 20:
            break
        ho = slsqp(bh, ts, 20000, 800, 7)
        c5o = compute_c5(ho)
        if c5o < bc5:
            bc5, bh = c5o, ho
        rem = tleft(t0, budget_s)

    bh = project_to_sum(bh, ts)
    bh = np.clip(bh, 0, 1)
    bc5 = compute_c5(bh)
    return (bh, bc5, n)
```
