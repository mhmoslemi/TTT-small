from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    h_rev = h[::-1]
    term1 = np.convolve(w, 1.0 - h, mode='full')[n-1:2*n-1]
    term2 = np.convolve(w, h_rev, mode='full')[:n]
    grad = term1 - term2
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.01
    no_improve = 0
    min_T = 0.002
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / T) + np.log(rng.uniform()) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:

                no_improve = 0

            c5_curr = best_c5
            T = min(T * 2.0, 0.015)
            no_improve = 0
        T = max(T * 0.999998, min_T)
    return best_h, best_c5

    h = h0.copy()
    t = 2.0 / len(h)
    lr = 1.0 / (alpha * 2.0)  # initial step size (Lipschitz constant is about alpha)
    # Actually the Lipschitz constant of the gradient of smooth_max is about alpha
    # (since the Hessian of the log-sum-exp is bounded by alpha * I)
    # So we need lr < 1/L where L ≈ alpha
    h_prev = h.copy()
    grad_prev = smooth_max_and_grad(h, alpha)[1]
    
    while time.time() < t_end:
        # Barzilai-Borwein step size
        s = h - h_prev
        y = smooth_max_and_grad(h, alpha)[1] - grad_prev
        ss = np.dot(s, s)
        sy = np.dot(s, y)
        if ss > 1e-15 and abs(sy) > 1e-15:
            lr_bb = ss / sy if sy > 0 else ss / (2*alpha)
            lr = max(1e-8, min(lr_bb, 1.0/(2*alpha)))
        
        h_new = h - lr * smooth_max_and_grad(h, alpha)[1]
        h_new = project_to_sum(h_new, len(h)/2.0)
        
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            h_prev = h
            h = h_new
            grad_prev = smooth_max_and_grad(h, alpha)[1]
        else:
            # Backtrack
            lr *= 0.5
            # Don't update h
    
    return h

from scipy.optimize import minimize

_current_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    h_rev = h[::-1]
    term1 = np.convolve(w, 1.0 - h, mode='full')[n-1:2*n-1]
    term2 = np.convolve(w, h_rev, mode='full')[:n]
    grad = term1 - term2
    return obj, grad

def obj_func(x):
    return smooth_max_and_grad(x, _current_alpha)[0]

def obj_grad(x):
    return smooth_max_and_grad(x, _current_alpha)[1]

    return np.sum(x) - len(x) / 2.0
