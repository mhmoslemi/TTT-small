import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_max(h):
    """Return just the max of the correlation (unscaled)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target_sum:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def best_ls(h, n, t_end, max_passes=100):
    """Best-improvement local search on binary sequence."""
    h = h.copy()
    cur = corr_max(h)
    best = cur
    best_h = h.copy()
    
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                v = corr_max(h)
                h[i] = 1.0
                h[j] = 0.0
                d = v - cur
                if d < best_delta - 1e-10:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        cur = corr_max(h)
        if cur < best:
            best = cur
            best_h = h.copy()
    return best_h, best

def first_ls(h, n, t_end, max_passes=200):

                    cur = v
                    if cur < best:
                        best = cur
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            break
    return best_h, best
