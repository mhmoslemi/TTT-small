import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:n]
        else:
            l = -k
            g[:n - l] += (1.0 - h[l:n])
            g[l:] -= h[:n - l]
    return g / K, corr

def nesterov_subgrad(h0, n, target_sum, n_iters, K_init, K_max, lr_init, mom):

        g_look, _ = subgrad_topk(h_look, K_cur)

        h_new = project_to_sum(np.clip(h + v, 0.0, 1.0), target_sum)

def coord_descent(h0, n, target_sum, max_rounds, steps, seed=42):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    rng = np.random.RandomState(seed)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_local = c5_cur
            best_local_h = h.copy()
            for d in steps:
                h_new = h.copy()
                val = h[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_new[i] = val
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_local - 1e-14:
                    best_local = c5_new
                    best_local_h = h_new
            if best_local < c5_cur - 1e-14:
                h = best_local_h
                c5_cur = best_local
                improved = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def smooth_max_opt(h0, n, target_sum, alpha, maxiter, seed=0):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
        return m + np.log(np.sum(np.exp(shifted))) / alpha

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            wl = w[k_idx]
            if wl < 1e-12:
                continue
            k = k_idx - (n - 1)
            if k >= 0:
                g[k:] += wl * (1.0 - h[:n - k])
                g[:n - k] -= wl * h[k:n]
            else:
                l = -k
                g[:n - l] += wl * (1.0 - h[l:n])
                g[l:] -= wl * h[:n - l]
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    ts = n / 2.0

    # Phase 1: Nesterov subgradient with varying K
    h_cur = best_h.copy()
    for K_pair in [(3, 5), (5, 8), (8, 15), (3, 3), (5, 10)]:
        if time_left() < 120:
            break
        n_it = min(8000, int(time_left() * 0.15))
        if n_it < 500:
            n_it = 500
        h_opt, c5_opt = nesterov_subgrad(h_cur, n, ts, n_it, K_pair[0], K_pair[1], 0.15, 0.85)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    # Phase 2: Top-3 directed method
    if time_left() > 100:
        n_it = min(6000, int(time_left() * 0.12))
        h_opt, c5_opt = top3_directed(h_cur, n, ts, n_it, 0.12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt if c5_opt < compute_c5(h_cur) else h_cur

    # Phase 3: Multi-start exploration
    if time_left() > 100:
        for pert in [0.01, 0.02, 0.04, 0.06]:
            if time_left() < 60:
                break
            h_pert = best_h + rng.uniform(-pert, pert, n)
            h_pert = project_to_sum(h_pert, ts)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 + 0.02:
                n_it = min(4000, int(time_left() * 0.08))
                if n_it < 300:
                    n_it = 300
                h_opt, c5_opt = nesterov_subgrad(h_pert, n, ts, n_it, 3, 8, 0.1, 0.85)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 4: Coordinate descent - medium steps
    if time_left() > 60:
        steps_med = [0.03, 0.015, 0.008, -0.03, -0.015, -0.008]
        max_rnd = max(3, min(12, int(time_left() // 15)))
        h_opt, c5_opt = coord_descent(best_h, n, ts, max_rnd, steps_med, seed=1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Pair optimization
    if time_left() > 50:
        n_pairs = min(5000, int(time_left() * 0.3))
        h_opt, c5_opt = pair_optimize(best_h, n, ts, n_pairs, seed=42)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Coordinate descent - fine steps
    if time_left() > 40:
        steps_fine = [0.008, 0.004, 0.002, 0.001, -0.008, -0.004, -0.002, -0.001]
        max_rnd = max(3, min(10, int(time_left() // 10)))
        h_opt, c5_opt = coord_descent(best_h, n, ts, max_rnd, steps_fine, seed=2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Smooth max SLSQP polish
    if time_left() > 30:
        for alpha in [500.0, 1000.0, 2000.0]:
            if time_left() < 15:
                break
            h_opt = smooth_max_opt(best_h, n, ts, alpha, 400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 8: Ultra-fine coordinate descent
    if time_left() > 20:
        steps_uf = [0.002, 0.001, 0.0005, 0.0002, -0.002, -0.001, -0.0005, -0.0002]
        max_rnd = max(2, min(6, int(time_left() // 8)))
        h_opt, c5_opt = coord_descent(best_h, n, ts, max_rnd, steps_uf, seed=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final subgradient polish
    if time_left() > 15:
        n_it = min(3000, int(time_left() * 0.5))
        h_opt, c5_opt = top3_directed(best_h, n, ts, max(n_it, 500), 0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = project_to_sum(best_h, ts)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
