import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_with_val(h, i, t, n):
    """Compute C5 with h[i] set to t, and slack (h[n-1]) adjusted to maintain sum."""
    slack = n - 1
    h_new = h.copy()
    h_new[i] = t
    h_new[slack] = h[slack] + h[i] - t  # maintain sum
    if h_new[slack] < -1e-10 or h_new[slack] > 1.0 + 1e-10:
        return float('inf')
    h_new[slack] = np.clip(h_new[slack], 0.0, 1.0)
    corr = np.correlate(h_new, 1.0 - h_new, mode="full")
    return float(np.max(corr) * (2.0 / n))

def coord_line_search(h, n, target_sum, iters_per_coord=40):
    """Coordinate descent with exact line search using the slack variable trick."""
    slack = n - 1
    c5_cur = compute_c5(h)
    
    for i in range(slack):  # update coordinates 0..n-2
        # Determine valid range for h[i]
        h_slack = h[slack]
        h_i = h[i]
        t_min = max(0.0, h_slack + h_i - 1.0)
        t_max = min(1.0, h_slack + h_i)
        
        if t_min >= t_max:
            continue
        
        # Ternary search
        lo, hi = t_min, t_max
        for _ in range(iters_per_coord):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            c1 = c5_with_val(h, i, m1, n)
            c2 = c5_with_val(h, i, m2, n)
            if c1 < c2:
                hi = m2
            else:
                lo = m1
        
        t_best = (lo + hi) / 2.0
        c_best = c5_with_val(h, i, t_best, n)
        
        if c_best < c5_cur - 1e-15:
            h[i] = t_best
            h[slack] = h_slack + h_i - t_best
            h[slack] = np.clip(h[slack], 0.0, 1.0)
            c5_cur = c_best
    
    return h, c5_cur

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Fast gradient computation
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for i in range(n):
        v1 = (shifts <= i) & (shifts > i - n)
        if np.any(v1):
            j1 = i - shifts[v1]
            grad[i] += np.dot(w[v1], 1.0 - h[j1])
        v2 = (shifts >= -i) & (shifts < n - i)
        if np.any(v2):
            j2 = i + shifts[v2]
            grad[i] -= np.dot(w[v2], h[j2])
    
    return sm_val, grad

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for _ in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.995
    
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            shift = m - (n - 1)
            jj1 = j_arr - shift
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr + shift
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        
        lr = lr_init / (1.0 + 0.05 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_with_val(h, i, t, n):
    slack = n - 1
    h_new = h.copy()
    h_new[i] = t
    h_new[slack] = h[slack] + h[i] - t
    if h_new[slack] < -1e-10 or h_new[slack] > 1.0 + 1e-10:
        return float('inf')
    h_new[slack] = np.clip(h_new[slack], 0.0, 1.0)
    corr = np.correlate(h_new, 1.0 - h_new, mode="full")
    return float(np.max(corr) * (2.0 / n))

def coord_line_search(h, n, target_sum, iters_per_coord=50, max_passes=10):
    slack = n - 1
    c5_cur = compute_c5(h)
    
    for p in range(max_passes):
        improved = False
        for i in range(slack):
            h_slack = h[slack]
            t_min = max(0.0, h_slack + h[i] - 1.0)
            t_max = min(1.0, h_slack + h[i])
            
            if t_min >= t_max - 1e-14:
                continue
            
            lo, hi = t_min, t_max
            for _ in range(iters_per_coord):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                c1 = c5_with_val(h, i, m1, n)
                c2 = c5_with_val(h, i, m2, n)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            
            t_best = (lo + hi) / 2.0
            c_best = c5_with_val(h, i, t_best, n)
            
            if c_best < c5_cur - 1e-15:
                h[i] = t_best
                h[slack] = h_slack + h[i] - t_best
                h[slack] = np.clip(h[slack], 0.0, 1.0)
                c5_cur = c_best
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for i in range(n):
        v1 = (shifts > i - n) & (shifts <= i)
        if np.any(v1):
            j1 = i - shifts[v1]
            term_B = np.dot(w[v1], h[j1])
        else:
            term_B = 0.0
        v2 = (shifts >= -i) & (shifts < n - i)
        if np.any(v2):
            j2 = i + shifts[v2]
            term_A = np.dot(w[v2], 1.0 - h[j2])
        else:
            term_A = 0.0
        grad[i] = term_A - term_B
    
    return sm_val, grad

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for _ in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.997
    
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            shift = m - (n - 1)
            jj1 = j_arr - shift
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr + shift
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        
        lr = lr_init / (1.0 + 0.03 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)
