    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):

def project_to_sum(h, target_sum):

        if abs(diff) < 1e-14:
            break

        else:

def compute_grad_lag(h, k):
    """Gradient of corr at physical shift k (can be negative, zero, or positive)."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:

        if l < n:
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
    return g

def subgradient_step(h, n_active=5, momentum=0.9, step_size=0.1):
    """One subgradient step with momentum. Returns (new_h, gradient, active_indices)."""
    n = len(h)

    # Get top n_active indices

    g = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        g += compute_grad_lag(h, k)
    g /= n_active
    
    return g, top_idx

def fw_direction(h, g, target_sum):
    """Compute the Frank-Wolfe direction: vertex in direction -g."""
    n = len(h)
    # Sort by g value, take the target_sum smallest
    sorted_idx = np.argsort(g)
    v = np.zeros(n)
    v[sorted_idx[:int(target_sum)]] = 1.0
    return v

    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

Now let me think about the FW direction more carefully. The feasible set is:
S = {v : 0 <= v_i <= 1 for all i, sum(v) = n/2}

The vertex of S in direction -g (i.e., minimizing g·v over S) is:
- Sort g in ascending order: g_{(1)} <= g_{(2)} <= ... <= g_{(n)}
- Set v_{(i)} = 1 for i = 1, ..., n/2 (the n/2 smallest g values)
- Set v_{(i)} = 0 for i = n/2+1, ..., n

This gives sum(v) = n/2 exactly (assuming n is even, which it is).

The FW gap is g·(h - v) = sum h_i * g_i - sum v_i * g_i.
If this gap is small, we're close to a FW stationary point (though not necessarily the global optimum for non-convex problems).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        l = abs(k)
        if l < n:
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
    return g

def get_active_gradients(h, n_active=5):
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Get top n_active
    top_idx = np.argsort(corr)[-n_active:]
    g = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        g += compute_grad_lag(h, k)
    g /= n_active
    return g, corr, max_val

def fw_vertex(g, target_sum):
    n = len(g)
    int_sum = int(round(target_sum))
    sorted_idx = np.argsort(g)
    v = np.zeros(n)
    v[sorted_idx[:int_sum]] = 1.0
    return v

def fw_optimize(h0, target_sum, max_iters=20000, time_limit=120, n_active=5):

    h = h0.copy()
    best_h = h.copy()

    t_start = time.time()

        v = fw_vertex(g, target_sum)

            break

            c5_try = compute_c5(h_try)

def slsqp_smooth_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    g = np.zeros(n)
    for m_idx in range(len(corr)):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        g += w[m_idx] * compute_grad_lag(h, k)
    return obj, g

def slsqp_polish(h0, target_sum, alpha=500, maxiter=500):
    n = len(h0)
    # Use a simpler constraint formulation
    # We'll use the equality constraint sum(h) = target_sum and bounds [0,1]
    try:
        result = minimize(
            lambda x: slsqp_smooth_max(x, alpha)[0],
            h0.copy(),
            jac=lambda x: slsqp_smooth_max(x, alpha)[1],
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_to_sum(result.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_desc(h0, target_sum, max_iters=5000, time_limit=60):
    n = len(h0)

    best_h = h.copy()

        i_dec = int(np.argmax(np.where(can_dec, g, -np.inf)))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec]:

            move = frac * max_move
            if move < 1e-12:

            h[i_dec] -= move
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                break
            else:
                h[i_inc] -= move
                h[i_dec] += move
                h = np.clip(h, 0, 1)
                h = project_to_sum(h, target_sum)
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    # Periodic interpolation
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    result = np.interp(x_new, x_ext, h_ext)
    return result

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        l = abs(k)
        if l < n:
            g[l:] += (1.0 - h[:n - l])
            g[:n - l] -= h[l:]
    return g

def get_subgradient(h, n_active=5):
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    k_act = min(n_active, n + len(corr) - 1)
    top_idx = np.argsort(corr)[-k_act:]
    g = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        g += compute_grad_lag(h, k)
    g /= k_act
    return g, corr, max_val

def fw_vertex(g, target_sum):

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    result = np.interp(x_new, x_ext, h_ext)
    return result

def slsqp_obj(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

def slsqp_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    for m_idx in range(len(corr)):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        g += w[m_idx] * compute_grad_lag(x, k)
    return g

def slsqp_eq_fun(x):
    return np.sum(x) - x[0] * 0.0

def slsqp_polish(h0, target_sum, alpha=1000, maxiter=500):
    n = len(h0)
    try:
        res = minimize(
            slsqp_obj, h0.copy(), args=(alpha,),
            jac=slsqp_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

            best_c5 = c5_r
            best_h = h_r.copy()

    if budget_s - (time.time() - t_start) > 25:
        h_r, c5_r = fw_optimize(best_h, ts_best, max_iters=50000, time_limit=18, n_active=5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    if budget_s - (time.time() - t_start) > 15:
        h_cd, c5_cd = coord_desc(best_h, ts_best, max_iters=30000, time_limit=12)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if budget_s - (time.time() - t_start) > 10:
        h_slq = slsqp_polish(best_h, ts_best, alpha=2000, maxiter=600)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq

    if budget_s - (time.time() - t_start) > 5:
        h_r, c5_r = fw_optimize(best_h, ts_best, max_iters=40000, time_limit=4, n_active=3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
