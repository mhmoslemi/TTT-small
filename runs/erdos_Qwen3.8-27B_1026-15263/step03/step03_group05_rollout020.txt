def compute_c5(h):

    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    if n == 0:
        return y
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(100):

        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

    obj = (m + log_sum / alpha) * dx

    # Gradient: use the same approach as before (works well enough)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    if n == 0:
        return y
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    log_sum = np.log(np.sum(np.exp(alpha * shifted)))
    return (m + log_sum / alpha) * dx

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return grad

def nesterov_pgd(h0, n, alpha, max_iter, time_fn, lr_init=None):
