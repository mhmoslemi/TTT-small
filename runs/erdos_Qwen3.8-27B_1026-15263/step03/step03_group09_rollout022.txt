import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.full(n, np.sum(w))
    for m_lag in range(2 * n - 1):
        wl = w[m_lag]
        if wl < 1e-15:
            continue
        k = m_lag - n + 1
        js = max(0, k)
        je = min(n, k + n)
        if js < je:
            g[js:je] -= wl * h[js - k:je - k]
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            g[js2:je2] -= wl * h[js2 + k:je2 + k]
    return sm_val, g

def fourier_eval(coeffs, n, x):
    h = np.full(n, 0.5)
    K = len(coeffs) // 2
    for k in range(1, K + 1):
        h += coeffs[2 * (k - 1)] * np.cos(k * np.pi * x)
        h += coeffs[2 * (k - 1) + 1] * np.sin(k * np.pi * x)
    return h

def fourier_obj(coeffs, n, x, alpha, pw):
    h = fourier_eval(coeffs, n, x)
    sm_val, g = smooth_max_grad(h, alpha)
    vu = np.maximum(h - 1.0, 0.0)
    vl = np.maximum(-h, 0.0)
    penalty = pw * (np.sum(vu ** 2) + np.sum(vl ** 2))
    g = g + pw * (2.0 * vu - 2.0 * vl)
    K = len(coeffs) // 2
    gc = np.zeros(2 * K)
    for k in range(1, K + 1):
        gc[2 * (k - 1)] = np.sum(g * np.cos(k * np.pi * x))
        gc[2 * (k - 1) + 1] = np.sum(g * np.sin(k * np.pi * x))
    return sm_val + penalty, gc

def fourier_optimize(n, x, K, alpha, pw, maxiter, t_start, budget):
    x0 = np.zeros(2 * K)
    res = minimize(lambda c: fourier_obj(c, n, x, alpha, pw)[0],
                   x0,
                   jac=lambda c: fourier_obj(c, n, x, alpha, pw)[1],
                   method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

def adam_grid(h0, n, target_sum, alpha, n_iters, lr_init, t_start, budget):
    h = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    for _ in range(n_iters):
        if time.time() - t_start > budget - 5:
            break
        t += 1
        _, g = smooth_max_grad(h, alpha)
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        mh = m_v / (1 - b1 ** t)
        vh = v_v / (1 - b2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, lr_init * 2)
        else:
            lr *= 0.95
    return best_h, best_c5

def subgrad_grid(h0, n, target_sum, n_iters, lr_init, top_k, t_start, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    lr = lr_init
    for it in range(n_iters):
        if time.time() - t_start > budget - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            k = m - n + 1
            js = max(0, k)
            je = min(n, k + n)
            if js < je:
                g[js:je] += (1.0 - h[js - k:je - k])
            js2 = max(0, -k)
            je2 = min(n, n - k)
            if js2 < je2:
                g[js2:je2] -= h[js2 + k:je2 + k]
        g /= float(top_k)
        # Backtracking line search
        step = lr
        best_local_c5 = best_c5
        for _ in range(20):

            h_new = np.clip(h_new, 0.0, 1.0)

            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-12:

            step *= 0.5
        h = h_new

            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(step * 1.5, lr_init)
        else:
            lr = max(step * 0.5, lr_init * 0.01)
    return best_h, best_c5

def coord_descent_gs(h0, n, target_sum, max_rounds, rng, t_start, budget):
    """Coordinate descent with golden section search along each coordinate."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    target = target_sum
    for rnd in range(max_rounds):
        if time.time() - t_start > budget - 5:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Golden section search for h[i] in [0, 1]
            # We fix all other coordinates and optimize h[i]
            # The constraint sum(h) = target means we need to adjust
            # Actually, let's just try a few values around the current one
            
            # Try values: current ± delta for several deltas
            candidates = []
            base = h[i]
            for frac in [0.01, 0.03, 0.1, 0.3]:
                candidates.append(base + frac)
                candidates.append(base - frac)
            candidates.append(base)
            
            best_val = c5_cur
            best_hi = base
            for cand in candidates:
                if cand < 0.0 or cand > 1.0:
                    continue
                h_new = h.copy()
                h_new[i] = cand
                h_new = project_to_sum(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_val:
                    best_val = c5_new
                    best_hi = cand
            if best_hi != base and best_val < c5_cur - 1e-14:
                h[i] = best_hi
                h = project_to_sum(h, target)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def perturb(h, n, target_sum, rng, strength):
    """Random perturbation."""
    h_new = h.copy()
    h_new += rng.uniform(-strength, strength, n)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.full(n, np.sum(w))
    for m_lag in range(2 * n - 1):
        wl = w[m_lag]
        if wl < 1e-15:
            continue
        k = m_lag - n + 1
        js = max(0, k)
        je = min(n, k + n)
        if js < je:
            g[js:je] -= wl * h[js - k:je - k]
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            g[js2:je2] -= wl * h[js2 + k:je2 + k]
    return sm_val, g

def fourier_eval(coeffs, n, x):
    h = np.full(n, 0.5)
    K = len(coeffs) // 2
    for k in range(1, K + 1):
        h += coeffs[2 * (k - 1)] * np.cos(k * np.pi * x)
        h += coeffs[2 * (k - 1) + 1] * np.sin(k * np.pi * x)
    return h

def _fourier_impl(coeffs, n, x, alpha, pw):
    h = fourier_eval(coeffs, n, x)
    sm_val, g = smooth_max_grad(h, alpha)
    vu = np.maximum(h - 1.0, 0.0)
    vl = np.maximum(-h, 0.0)
    penalty = pw * (np.sum(vu ** 2) + np.sum(vl ** 2))
    g_total = g + pw * (2.0 * vu - 2.0 * vl)
    K = len(coeffs) // 2
    gc = np.zeros(2 * K)
    for k in range(1, K + 1):
        gc[2 * (k - 1)] = np.sum(g_total * np.cos(k * np.pi * x))
        gc[2 * (k - 1) + 1] = np.sum(g_total * np.sin(k * np.pi * x))
    return sm_val + penalty, gc

def fourier_fun(coeffs, n, x, alpha, pw):
    return _fourier_impl(coeffs, n, x, alpha, pw)[0]

def fourier_jac(coeffs, n, x, alpha, pw):
    return _fourier_impl(coeffs, n, x, alpha, pw)[1]

def fourier_optimize(n, x, K, alpha, pw, maxiter):
    x0 = np.zeros(2 * K)
    args = (n, x, alpha, pw)
    try:
        res = minimize(fourier_fun, x0, args=args, jac=fourier_jac,
                       method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return res.x
    except Exception:

def subgrad_grid(h0, n, target_sum, n_iters, lr_init, top_k, t_start, budget):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        if time.time() - t_start > budget - 5.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            k = m - n + 1
            js = max(0, k)

            je2 = min(n, n - k)

                candidates.append(base + frac)

            best_val = c5_cur

                h_new[i] = cand
                h_new = project_to_sum(h_new, target_sum)

                    best_hi = cand

                h[i] = best_hi
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
