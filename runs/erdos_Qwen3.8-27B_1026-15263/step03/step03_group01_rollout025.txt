def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s = np.sum(exp_v)

    w = exp_v / s

    j = np.arange(2*n-1)[None, :] - (n-1)
    mj = m_idx + j
    valid1 = (mj >= 0) & (mj < n)
    hj1 = np.where(valid1, h[np.clip(mj, 0, n-1)], 0.0)
    mj2 = m_idx - j
    valid2 = (mj2 >= 0) & (mj2 < n)
    hj2 = np.where(valid2, h[np.clip(mj2, 0, n-1)], 0.0)
