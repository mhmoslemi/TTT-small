def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_and_grad(h, alpha)[0]

        return smooth_max_and_grad(h, alpha)[1]

                       bounds=bounds, constraints=cons,

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_and_grad(h, alpha)[0]
    def _g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_f, h0.copy(), method='SLSQP', jac=_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa_enhanced(h_start, n, target_sum, t_end, rng, max_seg=25, kick_size=8):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.012
    min_T = 0.0015
    no_improve = 0
    kick_cooldown = 0

    while time.time() < t_end:
        r = rng.random()
        if r < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif r < 0.80:
            seg_len = rng.randint(4, max_seg + 1)
            i = rng.randint(n - seg_len + 1)
            h[i:i+seg_len] = h[i:i+seg_len][::-1]
        else:
            seg_len = rng.randint(2, 6)
            i = rng.randint(n - seg_len + 1)
            h[i:i+seg_len] = h[i:i+seg_len][::-1]

        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / T) + np.log(rng.uniform() + 1e-300) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                kick_cooldown = 0
            else:
                no_improve += 1
        else:
            if r < 0.55:
                h[i], h[j] = h[j], h[i]
            elif r < 0.80:
                h[i:i+seg_len] = h[i:i+seg_len][::-1]
            else:
                h[i:i+seg_len] = h[i:i+seg_len][::-1]
            no_improve += 1

        kick_cooldown += 1
        if no_improve > 60000 and kick_cooldown > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            n_kick = min(kick_size, n // 4)
            ones_pos = np.where(h == 1.0)[0]
            zeros_pos = np.where(h == 0.0)[0]
            kick_ones = rng.choice(ones_pos, size=n_kick, replace=False)
            kick_zeros = rng.choice(zeros_pos, size=n_kick, replace=False)
            h[kick_ones] = 0.0
            h[kick_zeros] = 1.0
            T = min(T * 3.0, 0.035)
            no_improve = 0
            kick_cooldown = 0

        T = max(T * 0.999997, min_T)

    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def power_sum_optimize(h0, n, target_sum, p, maxiter=600):
    dx = 2.0 / n
    def _f(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        corr = np.maximum(corr, 1e-12)
        return np.sum(corr ** p) ** (1.0 / p)
    def _g(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        corr = np.maximum(corr, 1e-12)
        obj = np.sum(corr ** p) ** (1.0 / p)
        weights = corr ** (p - 1)
        w_sum = np.sum(corr ** p)
        coeff = (w_sum ** (1.0 / p - 1.0)) * weights
        n_local = n
        2n1 = 2 * n_local - 1
        j_idx = np.arange(n_local)[:, None]
        m_idx = np.arange(2n1)[None, :]
        first_idx = j_idx - m_idx + n_local - 1
        second_idx = j_idx + m_idx - n_local + 1
        valid_f = (first_idx >= 0) & (first_idx < n_local)
        valid_s = (second_idx >= 0) & (second_idx < n_local)
        fc = np.clip(first_idx, 0, n_local - 1)
        sc = np.clip(second_idx, 0, n_local - 1)
        cf = valid_f * (1.0 - h[fc]) * coeff[None, :]
        cs = valid_s * h[sc] * coeff[None, :]
        grad = coeff * (np.sum(cf, axis=1) - np.sum(cs, axis=1))
        return grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_f, h0.copy(), method='L-BFGS-B', jac=_g,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-6})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    # Phase 1: Continuous optimization to refine starting point
    for alpha in [30, 100, 500]:
        if remaining() < 60:
            break
        h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Power sum optimization as alternative
    if remaining() > 200:
        h_ps = power_sum_optimize(best_h, n_main, ts_main, p=30, maxiter=800)
        c5_ps = compute_c5(h_ps)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 2: Binary SA (primary search, ~30% budget)
    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.30 * budget_s
    h_sa, c5_sa = binary_sa_enhanced(h_bin, n_main, int(ts_main), sa_end1, rng, max_seg=30)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Continuous optimization from best binary
    for alpha in [50, 300, 1500]:
        if remaining() < 80:
            break
        h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Power sum from current best
    if remaining() > 150:
        h_ps2 = power_sum_optimize(best_h, n_main, ts_main, p=50, maxiter=1000)
        c5_ps2 = compute_c5(h_ps2)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2.copy()

    # Phase 4: Second binary SA round
    h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
    sa_end2 = t_start + 0.58 * budget_s
    if remaining() > 120:
        h_sa2, c5_sa2 = binary_sa_enhanced(h_bin2, n_main, int(ts_main), sa_end2, rng, max_seg=35)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Try higher resolutions
    for n_try in [280, 320, 360]:
        if remaining() < 150:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)

        h_opt = slsqp_optimize(h_interp, n_try, ts_try, 200, maxiter=400)
        c5_opt = compute_c5(h_opt)

        h_bin_try = make_binary(h_opt, n_try, int(ts_try), rng)
        c5_bt = compute_c5(h_bin_try)

        sa_end_try = t_start + 0.75 * budget_s
        if time.time() < sa_end_try:
            h_sa_try, c5_sa_try = binary_sa_enhanced(h_bin_try, n_try, int(ts_try), sa_end_try, rng, max_seg=25)
            c5_bt = min(c5_bt, c5_sa_try)
            if c5_sa_try <= compute_c5(h_bin_try):
                h_bin_try = h_sa_try

        h_fin = slsqp_optimize(h_bin_try, n_try, ts_try, 500, maxiter=500)
        c5_fin = compute_c5(h_fin)

        best_candidate = h_fin if c5_fin <= c5_bt else h_bin_try
        best_candidate_c5 = min(c5_fin, c5_bt)

        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = best_candidate.copy()
            best_n = n_try

    # Phase 6: Final polish on best solution
    if remaining() > 50:
        for alpha in [3000, 15000]:
            if remaining() < 25:
                break
            h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=1500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 7: One more SA round if time permits
    if remaining() > 80:
        h_bin_final = make_binary(best_h, best_n, int(best_n / 2.0), rng)
        h_sa_final, c5_sa_final = binary_sa_enhanced(h_bin_final, best_n, int(best_n / 2.0),
                                                      t_start + 0.97 * budget_s, rng, max_seg=30)
        if c5_sa_final < best_c5:
            best_c5 = c5_sa_final
            best_h = h_sa_final.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
