def compute_corr(h):

def project_to_sum(h, target_sum):

def continuous_refine(h_bin, n, target_sum, t_end, rng):
    """Refine a binary pattern by allowing some positions to take continuous values."""
    h = h_bin.copy()
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))
    
    # Identify critical positions: those that contribute to the top lags
    top_lags = np.argsort(corr)[-10:]
    
    # Find positions involved in the top lags
    critical = set()
    for k in top_lags:
        d = k - (n - 1)
        if d >= 0:
            for i in range(n - d):
                if h[i+d] * (1-h[i]) > 0:
                    critical.add(i)
                    critical.add(i+d)
        else:
            for i in range(-d, n):
                if h[i] * (1-h[i+d]) > 0:
                    critical.add(i)
                    critical.add(i+d)
    
    # Free up a subset of critical positions (the ones with the most "work" to do)
    # Strategy: alternate between free positions
    n_free = min(len(critical), 30)
    free_positions = list(rng.choice(sorted(critical), size=n_free, replace=False))
    
    # Optimize: minimize t subject to corr_k ≤ t for all k, sum(h) = target_sum, 0 ≤ h ≤ 1
    # Fixed positions stay binary, free positions vary
    
    fixed_mask = np.ones(n, dtype=bool)
    fixed_mask[free_positions] = False
    h_fixed = h.copy()
    
    def objective(x):
        h = h_fixed.copy()
        h[free_positions] = np.clip(x, 0, 1)
        # Fix sum constraint
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            # Adjust free positions
            h[free_positions] += diff / n_free
            h = np.clip(h, 0, 1)
        corr = compute_corr(h)
        return np.max(corr)
    
    def gradient(x):
        h = h_fixed.copy()
        h[free_positions] = np.clip(x, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            h[free_positions] += diff / n_free
            h = np.clip(h, 0, 1)
        # Compute gradient of max(corr) w.r.t. free positions
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        d = k_max - (n - 1)
        grad = np.zeros(n_free)
        for idx, pos in enumerate(free_positions):
            # d(corr[k_max])/d(h[pos])
            # corr[k_max] = sum_m h[m+d]*(1-h[m]) (for d>=0)
            # or sum_m h[m]*(1-h[m+d]) (for d<0)
            if d >= 0:
                # pos appears as h[m] when m=pos: contributes -h[pos+d] if pos < n-d
                # pos appears as h[m+d] when m=pos-d: contributes (1-h[pos-d]) if 0 <= pos-d <= n-1-d
                g = 0.0
                if pos < n - d:
                    g -= h[pos+d]
                if 0 <= pos-d <= n-1-d:
                    g += (1-h[pos-d])
            else:
                dd = -d
                # corr[n-1-d] = sum_m h[m]*(1-h[m-d]) for m from d to n-1
                # pos appears as h[m] when m=pos: contributes (1-h[pos-d]) if d <= pos < n
                # pos appears as h[m-d] when m-d=pos, m=pos+d: contributes -h[pos+d] if d <= pos+d < n
                g = 0.0
                if dd <= pos < n:
                    g += (1-h[pos+dd])
                if dd <= pos+dd < n:
                    g -= h[pos+dd]
            grad[idx] = g
        return grad
    
    x0 = h[free_positions].copy()
    # Use SLSQP
    try:
        res = minimize(objective, x0, method='SLSQP', jac=gradient,
                       bounds=[(0,1)]*n_free, maxiter=500, ftol=1e-15)
        h_new = h_fixed.copy()
        h_new[free_positions] = np.clip(res.x, 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_curr:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5_curr

The key insight is: if the current max is at lag k*, we should focus on reducing the correlation at k*. This is a "bottleneck" approach.

For a binary sequence, the correlation at lag k* (let's say k* > 0 for simplicity) is:
corr[k*] = number of i in [0, n-1-k*] where h[i+k*]=1 and h[i]=0

To reduce this, we want to either:
- Change some h[i] from 0 to 1 where h[i+k*]=1 (eliminating a "10" pair at lag k*)
- Change some h[i+k*] from 1 to 0 where h[i]=0 (same thing, different perspective)
- Do a swap: find i with h[i]=0, h[i+k*]=1 and j with h[j]=1, h[j+k*]=0, then swap h[i] and h[j]

The third option is the most promising because it reduces corr[k*] by 2 (removes two "bad" pairs) without changing the number of ones.

So the algorithm should be:
1. Find k* = argmax(corr)
2. Find all "bad" pairs at lag k*: B = {(i, i+k*) : h[i]=0, h[i+k*]=1}
3. Find all "good" pairs at lag k*: G = {(j, j+k*) : h[j]=1, h[j+k*]=0}
4. For each b in B and g in G, check if swapping h[b[0]] and h[g[0]] (or h[b[1]] and h[g[1]]) reduces the max correlation
5. Make the best swap found

This is a "2-exchange" move specifically designed to attack the bottleneck.

```python
def block_move(h, n, rng):
    """Generate a random block move: take a contiguous block and rotate it."""
    L = rng.randint(3, 15)  # block length
    start = rng.randint(0, n - L + 1)
    shift = rng.randint(1, L)  # rotation amount
    
    h_new = h.copy()
    block = h[start:start+L]
    h_new[start:start+L] = np.concatenate([block[shift:], block[:shift]])
    return h_new
```

But block moves are expensive to evaluate (we need to recompute the full correlation), and they might be too "large" to make gradual progress. Let me use them sparingly, only when stuck.

OK let me now write the final, complete solution. I'll combine:
1. Informed binary local search (targeted swaps + random swaps + block moves when stuck)
2. Continuous refinement (SLSQP on a subset of free variables)
3. Multi-start with different seeds
4. Try different n values

Let me also think about the continuous refinement more carefully.

For the continuous refinement, the key idea is:
- Start with a binary pattern
- Allow some positions to vary continuously
- Minimize the max correlation

The optimal continuous values can be found by solving:
minimize t
subject to: for all k, sum_i h[i+k]*(1-h[i]) ≤ t * n/2 (scaling by dx)
           sum(h) = n/2
           0 ≤ h ≤ 1

But this is non-convex (bilinear constraints). However, if we fix which positions are "high" (near 1) and which are "low" (near 0), and only allow a few "middle" positions to vary, the problem becomes approximately convex.

A simpler approach: use subgradient descent on the max function.

The subgradient of max(f_1, ..., f_m) at point x is:
- If only f_k is maximum: ∇f_k
- If multiple are maximum: convex hull of {∇f_j : j in argmax}

So we can use a subgradient method:
1. Compute corr = correlate(h, 1-h)
2. Find k* = argmax(corr)
3. Compute ∇f_{k*}(h) (gradient of the k*-th correlation w.r.t. h)
4. Take a step: h ← h - α * ∇f_{k*}(h)
5. Project to feasible set (sum = n/2, 0 ≤ h ≤ 1)
6. Repeat

This is the "max-margin" idea: always attack the worst constraint.

The gradient of f_k(h) = sum_i h[i+k]*(1-h[i]) w.r.t. h[pos]:
- If pos appears as h[i] (i=pos): contribution is -h[i+k] = -h[pos+k] (if valid)
- If pos appears as h[i+k] (i+k=pos, i=pos-k): contribution is (1-h[i]) = (1-h[pos-k]) (if valid)

So ∂f_k/∂h[pos] = (1-h[pos-k]) * 1[0 ≤ pos-k ≤ n-1-k] - h[pos+k] * 1[0 ≤ pos ≤ n-1-k] (for k ≥ 0)

This is computable in O(1) per position, so the full gradient is O(n).

Let me implement this:

```python
def subgradient_refine(h, n, target_sum, t_end, rng):
    """Subgradient method focused on the maximum lag."""
    h = h.copy().astype(float)
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    
    alpha = 0.1  # step size
    min_alpha = 0.001
    
    while time.time() < t_end:
        k_max = int(np.argmax(corr))
        d = k_max - (n - 1)
        
        # Compute gradient of f_{k_max}
        grad = np.zeros(n)
        if d >= 0:
            for i in range(n - d):
                # h[i+d]*(1-h[i])
                # d/dh[i]: -h[i+d]
                grad[i] -= h[i+d]
                # d/dh[i+d]: (1-h[i])
                grad[i+d] += (1-h[i])
        else:
            dd = -d
            for i in range(dd, n):
                # h[i]*(1-h[i-dd])
                # d/dh[i]: (1-h[i-dd])
                grad[i] += (1-h[i-dd])
                # d/dh[i-dd]: -h[i]
                grad[i-dd] -= h[i]
        
        # Take step
        h_new = h - alpha * grad
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        corr = compute_corr(h)
        
        # Anneal step size
        alpha = max(alpha * 0.999, min_alpha)
    
    return best_h, best_c5
```

Wait, but this subgradient method converges slowly and might not give the best results. Let me think of a better approach.

Actually, I think the most effective approach is:
1. Use the binary local search to find the best binary pattern
2. Then use a more sophisticated continuous optimization to refine it

For the continuous optimization, I'll use a "smooth max" approach with SLSQP, but with a better initialization (starting from the binary pattern).

Let me also consider: what if we use a **different n**? 

For n=240, we need max(corr) ≤ 45.696 for C5 < 0.3808. Since continuous values allow non-integer correlations, this is achievable even with a binary pattern that has max(corr) = 46, as long as the continuous refinement can shave off about 0.3.

The previous program achieved C5 = 0.380898 from a binary pattern with max(corr) = 46. That means the continuous refinement reduced it from 46/120 = 0.38333 to 0.380898, a reduction of about 0.0024 in C5, or 0.29 in corr terms.

To get C5 < 0.3808, we need a reduction of 46 - 0.3808*120 = 46 - 45.696 = 0.304 in corr. So we need just a tiny bit more reduction.

This suggests that:
1. With a slightly better binary pattern (even just finding a different one with the same max(corr)=46 but better "structure"), we might be able to get more continuous refinement
2. Or with a better continuous optimization algorithm, we can extract more from the same binary pattern

Let me focus on both:
- Better binary search (to find patterns where the max lag is more "concentrated" and easier to refine)
- Better continuous optimization (to extract more value)

For the continuous optimization, one key improvement: instead of using a single smooth parameter α, use **multiple α values** and take the best result. Different α values correspond to different "smoothness" levels:
- Small α: only the max lag matters → good for concentrated maxima
- Large α: all lags contribute → good for distributed maxima

Let me write the final solution:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(600):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_local_search(h, n, target_sum, t_end, rng):
    h = h.copy()
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    stall = 0
    max_stall = 3000
    
    while time.time() < t_end:
        k_max = int(np.argmax(corr))
        d = k_max - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        
        # Targeted swaps
        if d > 0:
            vz = zeros[zeros + d < n]
            bz = vz[np.array([h[i+d]==1.0 for i in vz])]
            vo = ones[ones + d < n]
            go = vo[np.array([h[i+d]==0.0 for i in vo])]
        elif d < 0:
            dd2 = -d
            vz = zeros[zeros + dd2 < n]
            bz = vz[np.array([h[i+dd2]==1.0 for i in vz])]
            vo = ones[ones + dd2 < n]
            go = vo[np.array([h[i+dd2]==0.0 for i in vo])]
        else:
            bz = zeros
            go = ones
        
        n_targeted = min(100, max(1, len(bz) * len(go) // 10))
        for _ in range(n_targeted):
            if len(bz) == 0 or len(go) == 0:
                break
            i = bz[rng.randint(len(bz))]
            j = go[rng.randint(len(go))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            nc = compute_corr(h)
            c5n = float(np.max(nc))
            if c5n < c5_curr - 1e-15:
                c5_curr = c5n; corr = nc
                if c5n < best_c5 - 1e-12:
                    best_c5 = c5n; best_h = h.copy()
                stall = 0; improved = True; break
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            # Random swaps
            for _ in range(30):
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                if i == j: continue
                h[i], h[j] = h[j], h[i]
                nc = compute_corr(h)
                c5n = float(np.max(nc))
                if c5n < c5_curr - 1e-15:
                    c5_curr = c5n; corr = nc
                    if c5n < best_c5 - 1e-12:
                        best_c5 = c5n; best_h = h.copy()
                    stall = 0; improved = True; break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            stall += 1
            if stall > max_stall:
                # Perturb
                np2 = rng.randint(3, 12)
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                for _ in range(np2):
                    i = zeros[rng.randint(len(zeros))]
                    j = ones[rng.randint(len(ones))]
                    if i != j:
                        h[i], h[j] = h[j], h[i]
                corr = compute_corr(h)
                c5_curr = float(np.max(corr))
                stall = 0
                max_stall = min(max_stall + 2000, 30000)
    
    return best_h, best_c5

def smooth_max_alpha(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mval = np.max(corr)
    shifted = np.clip(alpha * (corr - mval), -500.0, 0.0)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    obj = mval + np.log(np.sum(ev)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    fi = j_idx - m_idx + n - 1
    si = j_idx + m_idx - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad
