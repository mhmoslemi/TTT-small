def compute_c5(h):

    dx = 2.0 / n

def project_to_sum(h, target_sum):

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_objective(h, alpha, target_sum):
    return smooth_max_and_grad(h, alpha)[0]

def slsqp_gradient(h, alpha, target_sum):
    return smooth_max_and_grad(h, alpha)[1]

def slsqp_sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

    return np.ones_like(h)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_objective, h0.copy(), args=(alpha, target_sum),
                       method='SLSQP', jac=lambda x, a, ts: slsqp_gradient(x, a, ts),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_interp.copy()
                best_n = n_try

            h_bin_try = make_binary(h_interp, n_try, half_try, rng)
            c5_bin_try = compute_c5(h_bin_try)
            sa_end_try = min(t_start + elapsed() + 30, t_start + budget_s - 40)
            if time.time() < sa_end_try:
                h_sa_try, c5_sa_try = binary_sa_diverse(h_bin_try, n_try, half_try, sa_end_try, rng, T_init=0.01)
                if c5_sa_try < best_c5:
                    best_c5 = c5_sa_try
                    best_h = h_sa_try.copy()
                    best_n = n_try
                    h_try = h_sa_try.copy()

            for alpha in [40, 300, 1500]:
                if remaining() < 30:
                    break
                h_opt_try = slsqp_optimize(h_try, n_try, ts_try, alpha, maxiter=800)
                c5_opt_try = compute_c5(h_opt_try)
                if c5_opt_try < best_c5:
                    best_c5 = c5_opt_try
                    best_h = h_opt_try.copy()
                    best_n = n_try
                    h_try = h_opt_try.copy()

    if remaining() > 25:
        noise = 0.05
        h_shake = best_h + rng.uniform(-noise, noise, best_n)
        h_shake = project_to_sum(h_shake, best_n / 2.0)
        h_fin = slsqp_optimize(h_shake, best_n, best_n / 2.0, 10000, maxiter=1500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
