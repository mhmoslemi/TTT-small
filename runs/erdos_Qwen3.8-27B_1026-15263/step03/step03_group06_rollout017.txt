def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum} using bisection on mu."""
    h = np.asarray(h, dtype=float)
    if abs(np.sum(h) - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    
    # We need sum(clip(h - mu, 0, 1)) = target_sum
    # clip(h - mu, 0, 1) is decreasing in mu
    # When mu is very negative, clip gives 1 for all, sum = n
    # When mu is very positive, clip gives 0 for all, sum = 0
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_to_sum(h, target_sum):

    if abs(np.sum(h) - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 2.0
    hi = h.max() + 2.0

        mid = (lo + hi) / 2.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(free):

            result = np.clip(result, 0.0, 1.0)

    k = j_full - (n - 1)
    grad = np.zeros(n)

    if i_lo2 <= i_hi2:
        grad[i_lo2:i_hi2+1] -= h[i_lo2 + k : i_hi2 + k + 1]
    return grad

            h[i] = res.x
            h = project_to_sum(h, target_sum)
            best_val = res.fun
    except Exception:
        pass

    top_idx = np.argsort(corr)[-top_m:]
    c_top = corr[top_idx]
    c_max = np.max(c_top)
    shifted = alpha * (c_top - c_max)
    shifted = np.clip(shifted, -400.0, 400.0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    sm_val = c_max + np.log(np.sum(exps) / len(exps)) / alpha
    
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi > 1e-10:
            k = idx - (n - 1)
            i_lo = max(0, k)
            i_hi = min(n - 1, n - 1 + k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += wi * (1.0 - h[i_lo - k:i_hi - k + 1])
            i_lo2 = max(0, -k)
            i_hi2 = min(n - 1, n - 1 - k)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2+1] -= wi * h[i_lo2 + k:i_hi2 + k + 1]
    return sm_val, grad

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree > 0:
            result[free] += diff / nfree
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_gradient(h, corr, alpha, top_m):
    n = len(h)
    top_idx = np.argsort(corr)[-top_m:]
    c_top = corr[top_idx]
    c_max = np.max(c_top)
    shifted = alpha * (c_top - c_max)
    shifted = np.clip(shifted, -400.0, 400.0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    sm_val = c_max + np.log(np.sum(exps) / len(exps)) / alpha
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi > 1e-12:
            k = idx - (n - 1)
            i_lo = max(0, k)
            i_hi = min(n - 1, n - 1 + k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += wi * (1.0 - h[i_lo - k:i_hi - k + 1])
            i_lo2 = max(0, -k)
            i_hi2 = min(n - 1, n - 1 - k)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] -= wi * h[i_lo2 + k:i_hi2 + k + 1]
    return sm_val, grad

def coord_sweep(h, n, target_sum, rng, n_rounds, time_limit_fn):
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time_limit_fn() < 30:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            h_save = h[i]
            best_x = h_save
            best_val = c5_cur
            evals = [0.0, 1.0, h_save]
            delta = 0.02
            for d in [-0.06, -0.03, -0.01, 0.01, 0.03, 0.06]:
                v = h_save + d
                if 0.0 <= v <= 1.0:
                    evals.append(v)
            for v in evals:
                h[i] = v
                h_p = project_to_sum(h, target_sum)
                val = compute_c5(h_p)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x = v
            if best_x != h_save:
                h[i] = best_x
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
            else:
                h[i] = h_save
        if not improved:
            break
    return h, c5_cur

def fine_coord_sweep(h, n, target_sum, n_rounds, time_limit_fn):
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(12345)
    for rnd in range(n_rounds):
        if time_limit_fn() < 20:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            h_save = h[i]
            lo = max(0.0, h_save - 0.05)
            hi = min(1.0, h_save + 0.05)
            if hi - lo < 1e-8:
                continue
            def f1d(x):
                h[i] = x
                h_p = project_to_sum(h, target_sum)
                c = compute_c5(h_p)
                h[i] = h_save
                return c
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 50})
                if res.fun < c5_cur - 1e-14:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
            h[i] = h_save
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    target_sum = n / 2.0
    h = project_to_sum(h, target_sum)

    grad_buf = np.zeros(n)

    alpha = 30.0
    top_m = 15
    lr = 0.02
    stall = 0

        c5_cur = float(np.max(corr) * (2.0 / n))

            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0

            lr *= 0.5

            stall = 0
            if alpha > 50000 or lr < 1e-5:

        sm_val, grad = smooth_max_gradient(h, corr, alpha, top_m)

        # Gradient step with backtracking
        step_done = False
        for attempt in range(15):
            h_try = project_to_sum(h - lr * grad, target_sum)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            c5_try = float(np.max(corr_try) * (2.0 / n))
            if c5_try < c5_cur - 1e-13:
                h = h_try
                step_done = True
                break
            lr *= 0.5

        if not step_done:
            # Try a pure negative correlation step (reduce the max lag directly)
            k_max = np.argmax(corr)
            grad_single = np.zeros(n)
            k = k_max - (n - 1)
            i_lo = max(0, k)
            i_hi = min(n - 1, n - 1 + k)
            if i_lo <= i_hi:
                grad_single[i_lo:i_hi + 1] += (1.0 - h[i_lo - k:i_hi - k + 1])
            i_lo2 = max(0, -k)
            i_hi2 = min(n - 1, n - 1 - k)

            for scale in [0.005, 0.002, 0.001]:

                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                c5_try = float(np.max(corr_try) * (2.0 / n))
                if c5_try < c5_cur - 1e-14:

                    step_done = True
                    break

    best_c5 = compute_c5(best_h)
