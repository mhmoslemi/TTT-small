    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / alpha, 0.0)
    exp_v = np.exp(alpha * shifted)
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
