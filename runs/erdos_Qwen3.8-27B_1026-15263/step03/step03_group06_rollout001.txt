```python
        # h[j] appears as h[i] when i=j: contributes (1-h[j+k_shift]) if j+k_shift valid
        # h[j] appears as h[i+k_shift] when i=j-k_shift: contributes -h[j-k_shift] if j-k_shift valid
        for j in range(n):
            contrib = 0.0
            if 0 <= j + k_shift < n:
                contrib += (1 - h[j + k_shift])
            if 0 <= j - k_shift < n:
                contrib -= h[j - k_shift]
            g[j] += contrib
    g *= (2.0/n) / n_critical  # dx / n_critical
    return obj, g
```

Wait, this is O(n * n_critical) per gradient evaluation, which for n=720 and n_critical=10 is 7200 operations. Much faster than O(n²).

But actually, I can vectorize this:

For a given k_shift:
- ∂c/∂h[j] = (1-h[j+k_shift])·1[0≤j+k_shift<n] - h[j-k_shift]·1[0≤j-k_shift<n]

The first term: for j = 0 to n-1-k_shift (if k_shift ≥ 0), it's (1-h[j+k_shift]).
The second term: for j = max(0,k_shift) to n-1, it's -h[j-k_shift].

So:
g_k = zeros(n)
if k_shift >= 0:
    g_k[:n-k_shift] += (1 - h[k_shift:])
    g_k[k_shift:] -= h[:n-k_shift]
else:
    g_k[-k_shift:] += (1 - h[:n+k_shift])
    g_k[:n+k_shift] -= h[-k_shift:]

This is O(n) per critical shift! Total O(n * n_critical).

This is a major speedup. Let me build the optimization around this.

Another thought: maybe the optimal solution has a specific pattern. Let me think about what pattern minimizes the max of the correlation.

If h is a "smooth" function (not a step function), the correlation at small shifts will be large (since h(x) ≈ h(x+ε) for small ε). To reduce this, h should have sharp transitions. But sharp transitions increase the correlation at specific shifts (the width of the transition region).

There's a fundamental tension: 
- Sharp transitions → low correlation at small shifts, but high at specific shifts
- Smooth transitions → moderate correlation everywhere

The optimum is somewhere in between, likely with a few sharp-ish transitions and some smoothing.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj_grad(h, n_top):
    """Compute objective and gradient using top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_top:]
    obj = np.mean(corr[top_idx]) * dx
    
    g = np.zeros(n)
    for ki in top_idx:
        k_shift = ki - (n - 1)
        if k_shift >= 0:
            g[:n-k_shift] += (1.0 - h[k_shift:])
            g[k_shift:] -= h[:n-k_shift]
        else:
            ks = -k_shift
            g[ks:] += (1.0 - h[:n-ks])
            g[:n-ks] -= h[ks:]
    g *= dx / n_top
    return obj, g, top_idx

def projected_grad_descent(h0, n, target_sum, n_top, n_iters, lr_init):
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        obj, g, top_idx = topk_obj_grad(h, n_top)
        # Project gradient: subtract mean to maintain sum constraint
        g_proj = g - np.mean(g)
        h_new = h - lr * g_proj
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
        h = h_new
    return h_best, c5_best

def coord_descent_ls(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-6:
                continue
            # Use a few trial points
            best_val = c5_cur
            best_v = h[i]
            for v in np.linspace(lo, hi, 7):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_new = compute_c5(h_tmp)
                if c5_new < best_val:
                    best_val = c5_new
                    best_v = v
            if best_v != h[i]:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    n_corr = 2*n - 1
    
    def obj(x):

        corr = np.correlate(ht, 1.0 - ht, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        den = 1.0 + s  # ignore zeros
        return (m + np.log(s) / alpha) * (2.0/n)
    
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        corr = np.correlate(ht, 1.0 - ht, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        
        g = np.zeros(n)
        for j in range(n):
            # h[j] appears in corr[i] for various i
            # corr[i] = sum_l h[l] * (1-h[l+i-(n-1)])
            # h[j] appears when l=j (as h) or l+i-(n-1)=j (as 1-h)

            #   contributes (1-h[j+i-(n-1)]) * w[i]

            #   contributes -h[j+i-(n-1)] * w[i]

            k = j

            # i range: 0 <= j+i-(n-1) < n → n-1-j <= i <= 2n-2-j, clipped to [0, 2n-2]
            i_lo1 = max(0, n-1-j)
            i_hi1 = min(2n-2, 2n-2-j)
            if i_hi1 >= i_lo1:
                i_range = np.arange(i_lo1, i_hi1+1)
                idx = j + i_range - (n-1)
                g[j] += np.dot(w[i_range], 1.0 - h[idx])
            
            # h[j] as the "second" factor: corr[i] contains h[j+i-(n-1)]*(1-h[j]) for valid j+i-(n-1)
            # Wait, corr[i] = sum_l h[l]*(1-h[l+i-(n-1)])
            # h[j] appears as 1-h[...] when l+i-(n-1) = j, i.e., i = j-l+n-1
            # For fixed j: l = j - i + n - 1, need 0 <= l < n → n-1-j <= i <= n-1-j+n-1
            # contributes -h[l] * w[i] = -h[j-i+n-1] * w[i]
            i_lo2 = max(0, n-1-j)
            i_hi2 = min(2n-2, 2n-2-j)
            if i_hi2 >= i_lo2:
                i_range = np.arange(i_lo2, i_hi2+1)
                l = j - i_range + (n-1)
                # Need 0 <= l < n
                valid = (l >= 0) & (l < n)
                if np.any(valid):
                    g[j] -= np.dot(w[i_range[valid]], h[l[valid]])
        
        g *= 2.0 / n
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
```

```python
    n = len(h)

    lo, hi = -1.0, 1.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)
```

```python
            best_h = h_cd.copy()
        h_cur = h_cd

    if time_left() > 120:
        n2 = n1 // 2
        ts2 = n2 / 2.0
        h_low = interpolate_h(best_h, n1, n2)
        h_low = project_box_sum(h_low, ts2)
        for n_top in [5, 10, 20]:
            if time_left() < 30:
                break
            n_iters = min(5000, int(time_left() * 12))
            h_pg2, c5_pg2 = projected_grad_descent(h_low, n2, ts2, n_top, n_iters, 0.15)
            h_up = interpolate_h(h_pg2, n2, n1)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
            h_best_up = h_pg2

        if time_left() > 40:
            h_pg3, c5_pg3 = projected_grad_descent(h_best_up, n2, ts2, 40, 3000, 0.1)
            h_up3 = interpolate_h(h_pg3, n2, n1)
            c5_up3 = compute_c5(h_up3)
            if c5_up3 < best_c5:
                best_c5 = c5_up3
                best_h = h_up3.copy()

    if time_left() > 200:
        for pert_round in range(8):
            if time_left() < 40:
                break
            amplitude = 0.08 / (pert_round + 1)
            h_pert = fourier_perturb(best_h, n1, rng, amplitude)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
            h_work = h_pert
            for n_top in [10, 30]:
                if time_left() < 25:
                    break
                h_w, c5_w = projected_grad_descent(h_work, n1, ts1, n_top, 2000, 0.1)
                if c5_w < best_c5:
                    best_c5 = c5_w
                    best_h = h_w.copy()
                h_work = h_w
            if time_left() > 30:
                h_cd2, c5_cd2 = coord_descent_fast(h_work, n1, ts1, 4, rng)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()

    if time_left() > 150:
        for n3 in [n1 * 2]:
            if n3 > 2000 or time_left() < 80:
                continue
            ts3 = n3 / 2.0
            h_fin = interpolate_h(best_h, n1, n3)
            h_fin = project_box_sum(h_fin, ts3)
            for n_top in [5, 10, 20]:
                if time_left() < 30:
                    break
                h_f, c5_f = projected_grad_descent(h_fin, n3, ts3, n_top, 3000, 0.08)
                c5_f_true = compute_c5(f) if False else compute_c5(h_f)
                if c5_f_true < best_c5:
                    best_c5 = c5_f_true
                    best_h = h_f.copy()
                    best_n = n3

    if best_h is not None and time_left() > 30:
        n_p = len(best_h)
        ts_p = n_p / 2.0
        h_pol = slsqp_step(best_h, n_p, ts_p, 500.0, 300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

        if time_left() > 20:
            h_pol2 = slsqp_step(best_h, n_p, ts_p, 2000.0, 200)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2

    if best_h is not None and time_left() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_cd3, c5_cd3 = coord_descent_fast(best_h, n_f, ts_f, 5, rng)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3

    if best_h is not None and time_left() > 10:
        h_fin2 = np.random.RandomState(seed).normal(0, 0.005, size=len(best_h))
        h_fin2 = best_h + h_fin2
        h_fin2 = project_box_sum(h_fin2, len(best_h) / 2.0)
        h_fin2 = slsqp_step(h_fin2, len(best_h), len(best_h) / 2.0, 1000.0, 200)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2

    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_box_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
```
