One more thought: the "smooth max" approach in the parent uses log-sum-exp, which is a common technique. An alternative is to use the "power mean":
M_p(corr) = (sum(corr^p) / len(corr))^(1/p)

As p→∞, this approaches the max. The gradient is:

Actually, I think the key issue is that log-sum-exp with large alpha is numerically unstable (the exp can overflow). The power mean Avoids this if we normalize:
M_p = (sum((corr - m + eps)^p) / len)^(1/p) + m - eps

where m = max(corr) and eps is a small offset.

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def proj_sum(h, target):
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_at_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        g[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return g

def subgrad_descent(h, target, n_steps, rng, top_k=5, momentum=0.9, anneal=False):
    n = len(h)
    v = np.zeros(n)
    c5 = compute_c5(h)
    step = 0.008
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_steps):
        corr = compute_corr(h)
        m = np.max(corr)
        top = np.argsort(corr)[::-1][:top_k]
        w = np.exp(-(corr[top] - m) * 15.0)
        w = w / w.sum()
        g = np.zeros(n)
        for i, k in enumerate(top):
            s = k - (n - 1)
            g += w[i] * grad_at_shift(h, s)
        v = momentum * v - step * g
        h_new = proj_sum(h + v, target)
        c5_new = compute_c5(h_new)
        if anneal and it % 100 == 99 and it > 0:
            temp = 0.001 * (1.0 - it / n_steps)
            if c5_new > c5 and rng.random() < np.exp(-(c5_new - c5) / temp):
                h = h_new
                c5 = c5_new
                continue
        if c5_new < c5 - 1e-13:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            step = min(step * 1.015, 0.04)
        else:
            step = max(step * 0.85, 0.0003)
    return best_h, best_c5

def binary_1swap(h, n, target, rng, n_iters=3000):
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(n_iters):
        i = rng.randint(n)
        if h[i] > 0.5:
            cands = np.where(h < 0.5)[0]
            if len(cands) == 0:
                continue
            j = rng.choice(cands)
            h[i], h[j] = 0.0, 1.0
            c5_new = compute_c5(h)
            if c5_new >= c5:
                h[i], h[j] = 1.0, 0.0
            else:
                c5 = c5_new
        else:
            cands = np.where(h > 0.5)[0]
            if len(cands) == 0:
                continue
            j = rng.choice(cands)
            h[i], h[j] = 1.0, 0.0
            c5_new = compute_c5(h)
            if c5_new >= c5:
                h[i], h[j] = 0.0, 1.0
            else:
                c5 = c5_new
    return h, c5

def binary_2swap(h, n, target, rng, n_iters=2000):
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(n_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = rng.choice(ones)
        i2 = rng.choice(ones)
        if i1 == i2:
            continue
        j1 = rng.choice(zeros)
        j2 = rng.choice(zeros)
        if j1 == j2:
            continue
        oj1, oj2, oi1, oi2 = h[j1], h[j2], h[i1], h[i2]
        h[i1], h[i2] = 0.0, 0.0
        h[j1], h[j2] = 1.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-12:
            c5 = c5_new
        else:
            h[i1], h[i2], h[j1], h[j2] = oi1, oi2, oj1, oj2
    return h, c5

def binary_to_fixed_sum(h_bin, target):
    h_bin = h_bin.copy()
    diff = int(round(target - h_bin.sum()))
    if diff > 0:
        z_idx = np.where(h_bin == 0)[0]
        for i in range(min(diff, len(z_idx))):
            h_bin[z_idx[i]] = 1.0
    elif diff < 0:
        o_idx = np.where(h_bin == 1)[0]
        for i in range(min(-diff, len(o_idx))):
            h_bin[o_idx[i]] = 0.0
    return h_bin

def smooth_optimize(h0, target, alpha, maxiter=300):
    n = len(h0)
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        return m + np.log(np.sum(exp_v)) / alpha + 0.5 * 1e6 * (np.sum(x) - target) ** 2
    def jac(x):
        corr = compute_corr(x)
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        w = exp_v / np.sum(exp_v)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-10:
                s = k - (n - 1)
                g += w[k] * grad_at_shift(x, s)
        g += 1e6 * (np.sum(x) - target) * np.ones(n)
        return g
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return proj_sum(res.x, target)
    except Exception:
        return proj_sum(h0, target)

def exact_polish(h, target, max_rounds=500):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top = np.argsort(corr)[::-1][:4]
        g = np.zeros(n)
        for k in top:
            s = k - (n - 1)
            g += grad_at_shift(h, s)
        improved = False
        for lr in [0.0003, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            h_new = proj_sum(h - lr * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                improved = True
                break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)

    best_n = len(best_h)
    target = best_n / 2.0
    best_h = proj_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient with momentum at parent resolution
    if remaining() > 120:
        h, c5 = subgrad_descent(best_h, target, 4000, rng, top_k=5, momentum=0.9, anneal=False)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 2: Binary 1-swap local search
    if remaining() > 100:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = binary_to_fixed_sum(h_bin, target)
        h_b, c5_b = binary_1swap(h_bin, best_n, target, rng, n_iters=8000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 3: Subgradient from binary result
    if remaining() > 100:
        h, c5 = subgrad_descent(best_h, target, 3000, rng, top_k=4, momentum=0.85, anneal=False)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 4: Binary 2-swap
    if remaining() > 80:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = binary_to_fixed_sum(h_bin, target)
        h_b, c5_b = binary_2swap(h_bin, best_n, target, rng, n_iters=5000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 5: Smooth optimization with continuation
    if remaining() > 120:
        for a in [500, 2000, 8000, 20000]:
            if remaining() < 30:
                break
            h_opt = smooth_optimize(best_h, target, a, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 6: Exact polish
    if remaining() > 60:
        h_pol, c5_pol = exact_polish(best_h, target, max_rounds=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 7: Perturbation and re-optimize (escape local optima)
    if remaining() > 150:
        for n_pert in [3, 5, 8, 12]:
            if remaining() < 50:
                break
            h_pert = best_h.copy()
            idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[idx] = rng.uniform(0, 1, n_pert)
            h_pert = proj_sum(h_pert, target)
            h_opt, c5_opt = subgrad_descent(h_pert, target, 2000, rng, top_k=5, momentum=0.9, anneal=True)
            h_bin = np.where(h_opt > 0.5, 1.0, 0.0)
            h_bin = binary_to_fixed_sum(h_bin, target)
            h_b, c5_b = binary_1swap(h_bin, best_n, target, rng, n_iters=2000)
            h_fin, c5_fin = exact_polish(h_b, target, max_rounds=200)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin

    # Phase 8: Try higher resolution
    if remaining() > 200:
        n2 = best_n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = proj_sum(h_up, target2)
        h_up, c5_up = subgrad_descent(h_up, target2, 2000, rng, top_k=5, momentum=0.9, anneal=False)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
            best_n = n2
            target = target2
            if remaining() > 60:
                h_pol, c5_pol = exact_polish(best_h, target, max_rounds=300)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 9: Final binary polish
    if remaining() > 50:
        target_cur = best_n / 2.0
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = binary_to_fixed_sum(h_bin, target_cur)
        h_b, c5_b = binary_1swap(h_bin, best_n, target_cur, rng, n_iters=5000)
        h_b, c5_b = binary_2swap(h_b, best_n, target_cur, rng, n_iters=2000)
        h_fin, c5_fin = exact_polish(h_b, target_cur, max_rounds=300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 10: Final subgradient polish with very small steps
    if remaining() > 30:
        target_cur = best_n / 2.0
        for _ in range(3000):
            if remaining() < 10:
                break
            corr = compute_corr(best_h)
            m = np.max(corr)
            top = np.argsort(corr)[::-1][:3]
            g = np.zeros(best_n)
            for k in top:
                s = k - (best_n - 1)
                g += grad_at_shift(best_h, s)
            improved = False
            for lr in [0.0002, 0.0005, 0.001, 0.002]:
                h_new = proj_sum(best_h - lr * g, target_cur)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
            if not improved:
                break

    best_h = proj_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
