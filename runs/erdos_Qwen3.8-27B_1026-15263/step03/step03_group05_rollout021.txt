def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    delta = corr - m_val
    exp_vals = np.exp(alpha * delta)
    sum_exp = np.sum(exp_vals)
    log_sum = np.log(sum_exp)
    obj = (m_val + log_sum / alpha) * (2.0 / n)
    w = exp_vals / sum_exp
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

            best_h = h3.copy()

    # Phase 4: Very high alpha Adam
    if time_left() > 150:
        h4, _ = adam_pg(best_h, n1, [3000, 10000, 30000], 2000, 0.01, 0.9, 0.999, 1e-8, time_left)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Critical lag focused descent
    if time_left() > 100:
        h5, c5_5 = critical_lag_descent(best_h, n1, rng, max_rounds=30, time_fn=time_left)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Swap-based local search
    if time_left() > 80:
        h6, c5_6 = local_search_swaps(best_h, n1, rng, max_rounds=50, time_fn=time_left)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Shake and re-optimize (multiple attempts)
    if time_left() > 200:
        for trial in range(20):
            if time_left() < 40:
                break
            noise_scale = 0.1 * (1.0 - trial / 20.0) + 0.02
            h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n1)
            h_shake = project_box_sum(h_shake, n1 / 2.0)
            h_sh, _ = adam_pg(h_shake, n1, [50, 500, 3000], 1500, 0.02, 0.9, 0.999, 1e-8, time_left)
            c5_sh = compute_c5(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh.copy()
            if time_left() > 30:
                h_ls, c5_ls = local_search_exact(h_sh, n1, rng, max_rounds=20, time_fn=time_left)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Phase 8: Interleaved Adam + local search
    if time_left() > 150:
        for cycle in range(5):
            if time_left() < 30:
                break
            h_c, _ = adam_pg(best_h, n1, [5000, 20000], 1500, 0.005, 0.9, 0.999, 1e-8, time_left)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
            if time_left() > 20:
                h_ls2, c5_ls2 = local_search_exact(best_h, n1, rng, max_rounds=20, time_fn=time_left)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()

    # Phase 9: Try different n (upsample to 360 or 480)
    if time_left() > 120:
        for n_try in [360, 480]:
            if time_left() < 60:
                break
            if n_try <= best_n:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h, period=2.0)
            h_up = project_box_sum(h_up, n_try / 2.0)
            h_up2, _ = adam_pg(h_up, n_try, [20, 200, 1000, 5000], 1500, 0.03, 0.9, 0.999, 1e-8, time_left)
            c5_up = compute_c5(h_up2)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up2.copy()
                best_n = n_try
            if time_left() > 30:
                h_up3, c5_up3 = local_search_exact(h_up2, n_try, rng, max_rounds=20, time_fn=time_left)
                if c5_up3 < best_c5:
                    best_c5 = c5_up3
                    best_h = h_up3.copy()
                    best_n = n_try

    # Phase 10: Final ultra-fine Adam
    if time_left() > 30:
        h_fin, _ = adam_pg(best_h, best_n, [10000, 50000, 100000], 3000, 0.003, 0.9, 0.999, 1e-8, time_left)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 11: Final local search
    if time_left() > 15:
        h_final_ls, c5_final = local_search_exact(best_h, best_n, rng, max_rounds=30, time_fn=time_left)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_ls.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix sum exactly
    target_sum = best_n / 2.0
    current_sum = best_h.sum()
    diff = target_sum - current_sum
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(best_h)
        else:
            idx = np.argsort(-best_h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            else:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
