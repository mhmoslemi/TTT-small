    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_momentum(h0, target_sum, lr=0.1, momentum=0.9, max_iter=5000, time_limit=60):
    """Projected subgradient with heavy-ball momentum on the exact max."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr(h)
        m_idx = int(np.argmax(corr))
        lag = m_idx - (n - 1)
        grad = corr_grad_lag(h, lag)
        
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Revert and reduce step size
            lr *= 0.7
            v = momentum * v
    
    return best_h, best_c5

def binary_round(h, target_sum):
    """Round h to binary, maintaining sum constraint."""
    n = len(h)
    h_copy = h.copy()
    # Sort by value descending, take top target_sum as 1s
    order = np.argsort(h_copy)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:int(target_sum)]] = 1.0
    return h_bin

def binary_local_search(h_bin, target_sum, max_iters=50000, time_limit=60):
    """Local search on binary solutions: flip pairs (1→0, 0→1)."""
    n = len(h_bin)
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = time.time()
    rng = np.random.RandomState(123)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random pair
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try flip
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            # Update sets
            idx_in_ones = np.where(ones == i)[0]
            if len(idx_in_ones) > 0:
                ones = np.delete(ones, idx_in_ones[0])
            if j not in zeros:
                zeros = np.append(zeros, j)
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_grad_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(n_lags):
        if w[idx] < 1e-10:
            continue
        lag = idx - (n - 1)
        g = corr_grad_lag(h, lag)
        grad += w[idx] * g
    gnorm = np.linalg.norm(grad)
    if gnorm > 0:
        grad *= dx_factor(n)
    return m_val + np.log(np.sum(exp_vals)) / alpha, grad

def dx_factor(n):
    return 2.0 / n

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_momentum(h0, target_sum, lr, momentum, max_iter, time_limit):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = time.time()
    stale_count = 0
    current_lr = lr

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr(h)
        m_idx = int(np.argmax(corr))
        lag = m_idx - (n - 1)
        grad = corr_grad_lag(h, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        v = momentum * v - current_lr * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stale_count = 0
            current_lr = min(lr * 1.05, 2.0)
        else:
            current_lr *= 0.75
            stale_count += 1
            if stale_count > 200:
                break
    return best_h, best_c5

def binary_round(h, target_sum):
    n = len(h)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:int(round(target_sum))]] = 1.0
    return h_bin

def binary_local_search(h_bin, target_sum, max_iters, time_limit, seed=0):
    n = len(h_bin)
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = time.time()
    rng = np.random.RandomState(seed)
    stale = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)

        if lag == 0:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if lag > 0:
            good_ones = ones[(ones - lag >= 0)]
            good_ones = good_ones[np.array([h[p - lag] == 0 for p in good_ones])]
            good_zeros = zeros[(zeros + lag < n)]
            good_zeros = good_zeros[np.array([h[q + lag] == 0 for q in good_zeros])]
        else:
            k = -lag
            good_ones = ones[(ones + k < n)]
            good_ones = good_ones[np.array([h[p + k] == 0 for p in good_ones])]
            good_zeros = zeros[(zeros - k >= 0)]
            good_zeros = good_zeros[np.array([h[q - k] == 0 for q in good_zeros])]

        if len(good_ones) > 0 and len(good_zeros) > 0:
            p = good_ones[rng.randint(len(good_ones))]
            q = good_zeros[rng.randint(len(good_zeros))]
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]

        h[p] = 0.0
        h[q] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            stale = 0
        else:
            h[p] = 1.0
            h[q] = 0.0
            stale += 1
            if stale > 300:
                break
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def coordinate_descent(h, target_sum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_grad_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 150:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def perturb_and_optimize(h, target_sum, rng, noise_level, n, budget_remain):
    h_pert = h + rng.uniform(-noise_level, noise_level, n)

            c5_cd = c5_opt
    return h_cd, c5_cd
