def exact_max_polish(h0, n, target_sum, max_rounds, time_limit):

        if time.time() - t_start > time_limit:

        top_k = np.argsort(corr)[::-1][:min(5, len(corr))]
        best_improve = 0
        best_h = h

            grad = corr_grad_at_k(h, k_max, n)
            for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
