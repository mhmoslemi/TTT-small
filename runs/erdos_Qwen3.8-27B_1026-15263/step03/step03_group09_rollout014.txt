Let me also think about the "2-opt" for binary sequences. Given a binary h with n/2 ones:
- Pick a position i where h[i] = 1 and a position j where h[j] = 0
- Swap them: h[i] = 0, h[j] = 1
- Check if C5 improves

This is O(n² * n) = O(n³) if done naively (n² pairs, each requiring O(n) to evaluate). But we can be smarter:
- Only try swaps that might improve the worst lag
- Pre-compute the effect of a swap on each lag

For a swap at positions i (1→0) and j (0→1):
The change in corr[k] is:
Δcorr[k] = -h_old[i] * (1-h_old[i+k-n+1]) + ... 

This gets complicated. Let me just do a simpler version: try random swaps and keep improvements.

Actually, for the continuous case, the "swap" becomes: decrease h[i] by ε and increase h[j] by ε (maintaining the sum constraint). This is like a stochastic gradient step.

Let me simplify my approach:

1. Start from initial_h_values
2. Run Adam with smooth max (vectorized gradient) for many iterations
3. Run coordinate descent for fine-tuning
4. Try a few alternative starting points (binary sequences, sine waves)
5. Return the best solution

Time allocation:
- 70% for Adam optimization
- 20% for coordinate descent
- 10% for alternative starts

import numpy as np
from scipy.optimize import minimize
import time

def fast_correlate(h, v):
    n = len(h)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    V = np.fft.rfft(v[::-1], size)
    return np.fft.irfft(H * V, size)[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = fast_correlate(h, one_minus_h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    cc1 = np.correlate(w, one_minus_h, 'full')
    cc2 = np.correlate(w, h, 'full')
    g = cc1[2 * n - 2:3 * n - 2] - cc2[n - 1:2 * n - 1][::-1]
    return sm_val, g

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_start=None, budget_s=None):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    patience = 0
    for it in range(n_iters):
        if t_start is not None and budget_s is not None:
            if budget_s - (time.time() - t_start) < 10:
                break
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1
            if patience > 200:
                lr *= 0.9
                patience = 0
            elif patience > 500:
                lr *= 0.8
                patience = 0
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k, t_start=None, budget_s=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        if t_start is not None and budget_s is not None:
            if budget_s - (time.time() - t_start) < 5:
                break
        corr = fast_correlate(h, 1.0 - h)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m_idx in top_idx:
            k = m_idx
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                g[i] += (1.0 - h[n - 1 - k + i])
                j2 = i - n + 1 + k
                if 0 <= j2 < n:
                    g[i] -= h[j2]
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def topk_subgradient_fast(h0, n, target_sum, n_iters, lr_init, top_k, t_start=None, budget_s=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if t_start is not None and budget_s is not None:
            if budget_s - (time.time() - t_start) < 5:
                break
        corr = fast_correlate(h, 1.0 - h)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        one_minus_h = 1.0 - h
        for m_idx in top_idx:
            k = m_idx
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                idx_i = np.arange(i_lo, i_hi + 1)
                g[idx_i] += one_minus_h[n - 1 - k + idx_i]
                j2 = idx_i - n + 1 + k
                valid = (j2 >= 0) & (j2 < n)
                if np.any(valid):
                    g[idx_i[valid]] -= h[j2[valid]]
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta, t_start=None, budget_s=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-4.0, -3.0, -2.0, -1.0, 1.0, 2.0, 3.0, 4.0]) * base_delta
    for r in range(max_rounds):
        if t_start is not None and budget_s is not None:
            if budget_s - (time.time() - t_start) < 5:
                break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-15 or nv > 1.0 + 1e-15:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_local_search(h0, n, target_sum, max_iters, rng, t_start=None, budget_s=None):
    h = h0.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, compute_c5(h)
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    for it in range(max_iters):
        if t_start is not None and budget_s is not None:
            if budget_s - (time.time() - t_start) < 5:
                break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def generate_thue_morse(n_target):
    n_power = 1
    while n_power < n_target:
        n_power *= 2
    seq = np.zeros(n_power, dtype=np.int32)
    for i in range(1, n_power):
        seq[i] = seq[i >> 1] ^ (i & 1)
    if n_power == n_target:
        return seq
    result = np.zeros(n_target, dtype=np.int32)
    result[:min(n_power, n_target)] = seq[:min(n_power, n_target)]
    remaining = n_target - min(n_power, n_target)
    ones_so_far = np.sum(result[:min(n_power, n_target)])
    total_ones_needed = n_target // 2
    if remaining > 0:
        if ones_so_far < total_ones_needed:
            result[min(n_power, n_target):] = 1
            result[min(n_power, n_target):] = (np.arange(remaining) % 2)
            result[min(n_power, n_target):] = 1 - result[min(n_power, n_target):]
            n_ones_appended = np.sum(result[min(n_power, n_target):])
            need = total_ones_needed - ones_so_far
            if n_ones_appended < need:
                result[min(n_power, n_target):need - n_ones_appended + remaining - n_ones_appended] = 1
            elif n_ones_appended > need:
                result[min(n_power, n_target):n_ones_appended - need + remaining - n_ones_appended] = 0
        else:
            result[min(n_power, n_target):] = 0
    return result

def generate_random_binary(n, target_ones, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target_ones]] = 1.0
    return h

def generate_sine_binary(n, freq, phase, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    vals = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    threshold = np.sort(vals)[n // 2]
    h = (vals >= threshold).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        excess = np.where(h == 1.0)[0]
        to_flip = excess[-(n_ones - n // 2):]
        h[to_flip] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h == 0.0)[0]
        to_flip = deficit[:n // 2 - n_ones]
        h[to_flip] = 1.0
    return h

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1300, **kwargs):

            best_n = n_cand

        ts = n / 2.0

            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(project_to_sum(h_interp, ts))
        else:
            starts.append(np.full(n, 0.5))

        h_tm = generate_thue_morse(n)
        starts.append(project_to_sum(h_tm.astype(float), ts))

        for _ in range(3):
            h_rb = generate_random_binary(n, n // 2, rng)
            starts.append(project_to_sum(h_rb, ts))

        x = np.linspace(0, 2, n, endpoint=False)

                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
                starts.append(project_to_sum(hs, ts))

        h_cur = None

            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:

                best_h = h_start.copy()
                best_n = n

        if h_cur is not None and _tl(t_start, budget_s) > 30:

                lr_a = 0.05 if alpha < 100 else 0.02
