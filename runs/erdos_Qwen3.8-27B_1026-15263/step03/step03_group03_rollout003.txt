import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_idx = np.arange(2 * n - 1)
    idx1 = j + k_idx - nm1
    valid1 = (idx1 >= 0) & (idx1 < n)
    B[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = j - k_idx + nm1
    valid2 = (idx2 >= 0) & (idx2 < n)
    B[valid2] += -h[idx2[valid2]]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=20, temp=0.0):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Candidate selection: neighbors, structured, random
        k_set = [(j + 1) % n, (j - 1 + n) % n]
        for frac in [1/4, 1/3, 1/2, 2/3, 3/4]:
            k_set.append((int(j + frac * n)) % n)
        for _ in range(n_cand - 7):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if temp > 0 and rng.random() < 0.05:
                # Occasionally accept slightly worse moves (simulated annealing)
                if val_opt < best_val + temp:
                    if val_opt < best_val - 1e-13:
                        best_val = val_opt
                        best_delta = delta_opt
                        best_D = D
                        best_k = k
            else:
                if val_opt < best_val - 1e-13:
                    best_val = val_opt
                    best_delta = delta_opt
                    best_D = D
                    best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def top_lag_guided_descent(h, corr, rng, n_top=5, n_iters=100):
    """Use top contributing lags to guide pair selection."""
    n = len(h)
    target_sum = n / 2.0
    for _ in range(n_iters):
        top_lags = np.argsort(corr)[-n_top:]
        best_improved = False
        for lag in top_lags:
            k = lag
            nm1 = n - 1
            # Find which (i, i+k-nm1) pairs contribute most to corr[k]
            if k < nm1:
                i_start = 0
                i_end = n - 1 - (nm1 - k)
                idx_shift = k - nm1  # negative
            else:
                i_start = k - nm1
                i_end = n - 1
                idx_shift = k - nm1  # positive or zero
            
            # For this lag, contribution of position i is h[i] * (1-h[i+idx_shift])
            valid_i = np.arange(max(0, nm1-k), min(n, n-1+k+1))
            partners = valid_i + (k - nm1)
            valid = (partners >= 0) & (partners < n)
            valid_i = valid_i[valid]
            partners = partners[valid]
            
            contrib = h[valid_i] * (1.0 - h[partners])
            # Try to reduce contribution by moving h from valid_i to somewhere that doesn't affect this lag
            if len(valid_i) == 0:
                continue
            j = valid_i[np.argmax(contrib)]
            B_j = compute_B(h, j, n)
            # Try moving to a random position
            for _ in range(5):
                k2 = rng.randint(n)
                if k2 == j:
                    continue
                B_k2 = compute_B(h, k2, n)
                D = B_j - B_k2
                delta_lo = max(-h[j], h[k2] - 1.0)
                delta_hi = min(1.0 - h[j], h[k2])
                if delta_hi - delta_lo < 1e-13:

                    h[k2] = np.clip(h[k2] - delta_opt, 0.0, 1.0)

                    break

    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_idx = np.arange(2 * n - 1)
    idx1 = j + k_idx - nm1
    valid1 = (idx1 >= 0) & (idx1 < n)
    B[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = j - k_idx + nm1
    valid2 = (idx2 >= 0) & (idx2 < n)
    B[valid2] += -h[idx2[valid2]]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

        for k in k_cands:
            if k == j:
                continue

            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)

            if val_opt < best_val - 1e-14:
                best_val = val_opt

    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def top_lag_guided(h, corr, rng, n_top=6, n_trials=200):
    n = len(h)
    ts = n / 2.0
    for _ in range(n_trials):
        top_lags = np.argsort(corr)[-n_top:]
        any_improved = False
        for lag in top_lags:
            nm1 = n - 1
            if lag <= nm1:
                i_lo = 0
                i_hi = n - 1 - (nm1 - lag)
                shift = lag - nm1
            else:
                i_lo = lag - nm1
                i_hi = n - 1
                shift = lag - nm1
            valid_i = np.arange(max(0, i_lo), min(n, i_hi + 1))
            partners = valid_i + shift
            valid = (partners >= 0) & (partners < n)
            vi = valid_i[valid]
            vp = partners[valid]
            if len(vi) == 0:
                continue
            contrib = h[vi] * (1.0 - h[vp])
            for j in vi[np.argsort(-contrib)][:3]:
                B_j = compute_B(h, j, n)
                for _ in range(4):
                    k2 = rng.randint(n)
                    if k2 == j:
                        continue
                    B_k2 = compute_B(h, k2, n)
                    D = B_j - B_k2
                    delta_lo = max(-h[j], h[k2] - 1.0)
                    delta_hi = min(1.0 - h[j], h[k2])
                    if delta_hi - delta_lo < 1e-13:
                        continue
                    delta_opt = solve_1d(corr, D, delta_lo, delta_hi)

                    if val_opt < np.max(corr) - 1e-13:

                        h[k2] = np.clip(h[k2] - delta_opt, 0.0, 1.0)
                        any_improved = True
                        break
        if not any_improved:
            break
    return h
