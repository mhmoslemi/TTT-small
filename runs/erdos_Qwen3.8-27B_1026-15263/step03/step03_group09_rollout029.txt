import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gradient(h, alpha):
    """Compute gradient of smooth max of correlation using FFT."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # g[j] = conv(1-h, w, full)[j+n-1] - conv(w, h_rev, full)[2n-2-j]
    h_rev = h[::-1]
    conv1 = np.convolve(1.0 - h, w, mode='full')
    conv2 = np.convolve(w, h_rev, mode='full')
    
    g = np.zeros(n)
    g[:] = conv1[n-1:2*n-1]  # indices n-1 to 2n-2, which is j+n-1 for j=0..n-1
    g[:] -= conv2[:n][::-1]  # conv2[2n-2-j] for j=0..n-1 is conv2[n-1], conv2[n-2], ..., conv2[0]
    
    return g

def adam_optimize(h0, n, target_sum, n_iters, lr_init, alpha, time_limit=None, t_start=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        t += 1
        g = smooth_max_gradient(h, alpha)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            lr *= 0.98
    
    return best_h, best_c5

def fourier_to_h(coeffs, n):
    K = len(coeffs) // 2
    j = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        angle = 2.0 * np.pi * k * j / n
        h += coeffs[2*(k-1)] * np.cos(angle)
        h += coeffs[2*(k-1)+1] * np.sin(angle)
    return h

def fourier_optimize(n, K, n_runs, rng, time_per_run):
    target = n / 2.0
    best_h = None

Note that ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - R_h(k) (in the periodic case, or approximately for the non-periodic case with appropriate bounds).

If h is a "spread out" function (like a Gaussian), its autocorrelation is also a Gaussian, centered at k=0 with width proportional to h's width. The maximum overlap would be at k near 0 (where autocorrelation is high) or at k far from 0 (where the overlap region is small but h might be large).

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)

def smooth_max_gradient_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    w = exp_vals / np.sum(exp_vals)
