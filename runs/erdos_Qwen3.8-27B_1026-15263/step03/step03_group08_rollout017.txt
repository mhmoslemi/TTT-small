import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def top_k_gradient(h, n, top_k):
    """Compute gradient of the sum of top-k correlations (subgradient of max)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        l = k - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + l:hi2 + l]
    return g / top_k

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.01, b1=0.9, b2=0.999, eps=1e-8):
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        t += 1
        _, grad = smooth_max_and_grad(h, alpha)
        # Project gradient to satisfy equality constraint
        grad = grad - np.mean(grad)
        
        m_hat = b1 * m_hat + (1 - b1) * grad
        v_hat = b2 * v_hat + (1 - b2) * grad ** 2
        m_hat_corr = m_hat / (1 - b1 ** t)
        v_hat_corr = v_hat / (1 - b2 ** t)
        
        h_new = h - lr_init * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def backtracking_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.1):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        _, grad = smooth_max_and_grad(h, alpha)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        c5_cur = compute_c5(h)
        step = lr_init / max(gnorm, 1e-6)
        
        accepted = False
        for _ in range(15):
            h_new = np.clip(h - step * grad, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-10 * max(1, abs(c5_cur)):
                accepted = True
                break
            step *= 0.5
        
        if accepted:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            break
    
    return best_h, best_c5

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.05):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(n_iters):
        y = h + 0.9 * v
        y = np.clip(y, 0.0, 1.0)
        _, grad = smooth_max_and_grad(y, alpha)
        grad = grad - np.mean(grad)
        
        v_new = 0.9 * v - lr * grad / max(np.linalg.norm(grad), 1e-8)
        h_new = np.clip(h + v_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.95
            v_new *= 0.5
        
        v = v_new
        h = h_new
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):

        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v

        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=constraints,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def topk_subgrad(h, n, top_k, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    w = rng.dirichlet(np.ones(top_k))
    g = np.zeros(n)
    for j, k in enumerate(top_idx):
        wl = w[j]
        l = k - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return g

def slsqp_refine(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.01):
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for it in range(n_iters):
        t += 1
        _, grad = smooth_max_and_grad(h, alpha)
        grad = grad - np.mean(grad)
        m_hat = b1 * m_hat + (1 - b1) * grad
        v_hat = b2 * v_hat + (1 - b2) * (grad ** 2)
        m_u = m_hat / (1 - b1 ** t)
        v_u = v_hat / (1 - b2 ** t)
        h_new = h - lr_init * m_u / (np.sqrt(v_u) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def fista_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.5):
    h = h0.copy()
    z = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iters):
        if t > 1:
            t1 = (1 + np.sqrt(1 + 4 * (t - 1) ** 2)) / 2.0
            z = h + ((t - 1) / t1) * (h - z)
            z = np.clip(z, 0.0, 1.0)
        else:
            z = h.copy()
        _, grad = smooth_max_and_grad(z, alpha)
        grad = grad - np.mean(grad)
        step = lr_init / (t + 1)
        h_new = np.clip(z - step * grad, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def backtracking_descent(h0, n, target_sum, alpha, n_iters, lr_init=0.1):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        _, grad = smooth_max_and_grad(h, alpha)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        c5_cur = compute_c5(h)
        step = lr_init / max(gnorm ** 2, 1e-8)
        accepted = False
        for _ in range(14):
            h_new = np.clip(h - step * grad, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_cur - 1e-13:
                accepted = True
                break
            step *= 0.4
        if accepted:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            break
    return best_h, best_c5

def subgrad_descent(h0, n, target_sum, n_iters, lr_init, top_k, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        g = topk_subgrad(h, n, top_k, rng)
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        step = lr / (1 + 0.0005 * it)
        h_new = np.clip(h - step * g / gnorm, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.005, 0.5)
        else:
            lr *= 0.97
        h = h_new
    return best_h, best_c5

def fine_coordinate(h0, n, target_sum, max_rounds=80):
    h = h0.copy()
    c5_cur = compute_c5(h)
    steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-4:]
        active = set()
        for k in top_idx:
            l = k - (n - 1)
            active.update(range(max(0, l), min(n, n + l)))
            active.update(range(max(0, -l), min(n, n - l)))
        active_list = list(active)
        np.random.shuffle(active_list)
        for i in active_list[:100]:
            for d in steps:
                nv = h[i] + d
                if nv < 0 or nv > 1:
                    continue
                h_new = h.copy()
                h_new[i] = nv
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur
