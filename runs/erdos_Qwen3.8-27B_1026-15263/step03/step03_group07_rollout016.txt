import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_active_lags(corr, threshold=0.99):
    max_corr = np.max(corr)
    if max_corr <= 0:
        return np.array([], dtype=int)
    return np.where(corr >= threshold * max_corr)[0]

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad_lag(h, active_idx, alpha, penalty, target_sum):
    """Objective and gradient for L-BFGS-B with penalty method."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    active_corr = corr[active_idx]
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w_active = exp_vals / s
    
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for pos, m_idx in enumerate(active_idx):
        wm = w_active[pos]
        if wm < 1e-14:
            continue
        k = m_idx - (n - 1)
        kk = abs(k)
        if kk > 0:
            g[kk:] += wm * (1.0 - h[:n-kk])
            g[:n-kk] -= wm * h[kk:]
        else:
            g += wm * (1.0 - 2.0 * h)
    
    # Add penalty gradient
    sum_diff = np.sum(h) - target_sum
    f = sm_val + penalty * sum_diff * sum_diff
    g = g + 2.0 * penalty * sum_diff
    
    return f, g

def optimize_lbfgs(h0, target_sum, alpha, penalty, active_idx, maxiter):
    n = len(h0)
    # Pad active_idx if empty
    if len(active_idx) == 0:
        corr = np.correlate(h0, 1.0 - h0, mode='full')
        active_idx = get_active_lags(corr, 0.95)
    
    # Convert to params that L-BFGS-B can handle
    def func_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        f, g = obj_and_grad_lag(h, active_idx, alpha, penalty, target_sum)
        # When x is at boundary, zero out gradient for that component
        g[x <= 0.0] = 0.0
        g[x >= 1.0] = 0.0
        return f, g
    
    bounds = [(0.0, 1.0)] * n
    x0 = h0.copy()
    
    res = fmin_l_bfgs_b(func_and_grad, x0, fprime=None, 
                         bounds=bounds, maxiter=maxiter, 
                         pgtol=1e-10, factr=1e12)
    h_opt = np.clip(res[0], 0.0, 1.0)
    h_opt = project_to_sum(h_opt, target_sum)
    return h_opt

def coordinate_descent(h0, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    n = len(h)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01]
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_c5 = c5_cur
            best_h_i = h[i]
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h_i = h_new[i]
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

                best_n = n

def projected_gradient_step(h, active_idx, alpha, lr, target_sum):
    n = len(h)
    sm_val, g = smooth_max_grad(h, active_idx, alpha)
    free = (h > 1e-8) & (h < 1.0 - 1e-8)
    if np.any(free):
        lam = np.mean(g[free])
    else:
        lam = 0.0
    h_new = h - lr * (g - lam)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def projected_gradient_descent(h0, active_idx, alpha, n_iters, lr_init, target_sum):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        h_new = projected_gradient_step(h, active_idx, alpha, lr, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
    return best_h, c5_cur

def coordinate_descent_random(h0, target_sum, n_rounds, n_subset, rng, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    n = len(h)
    for rnd in range(n_rounds):
        idx_order = rng.choice(n, size=min(n_subset, n), replace=False)
        for i in idx_order:
            best_c5 = c5_cur
            best_val = h[i]
            for d in step_sizes:
                trial = h[i] + d
                if trial < 0.0 or trial > 1.0:
                    continue
                h[i] = trial
                h = project_to_sum(h, target_sum)
                c5_trial = compute_c5(h)
                if c5_trial < best_c5 - 1e-14:
                    best_c5 = c5_trial
                    best_val = trial
                h[i] = best_val
            if best_val != h[i]:
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
    return h, c5_cur

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    resolutions = [best_n]
    if best_n <= 720:
        resolutions.extend([900, 1080, 1440])
    elif best_n <= 900:
        resolutions.extend([1080, 1440])
    elif best_n <= 1080:
        resolutions.extend([1440])

    alpha_schedule = [20, 80, 300, 1000, 3000, 10000]
    penalty_schedule = [1e-4, 1e-3, 1e-2, 0.1, 1.0, 10.0]

    for n in resolutions:
        if time_left() < 100:
            break

        ts = n / 2.0

        if n != best_n:
            h_cur = interpolate_h(best_h, n)
            h_cur = project_to_sum(h_cur, ts)
            c5_interp = compute_c5(h_cur)
            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_cur.copy()
                best_n = n
        else:
            h_cur = best_h.copy()

        # L-BFGS-B optimization with increasing alpha
        for stage, alpha in enumerate(alpha_schedule):
            if time_left() < 40:
                break
            corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            thresh = 0.995 if alpha > 1000 else 0.99
            active = get_active_lags(corr_cur, thresh)
            if len(active) == 0:
                active = get_active_lags(corr_cur, 0.90)
            penalty = penalty_schedule[min(stage, len(penalty_schedule) - 1)]
            maxiter = min(300, max(50, int(time_left() * 0.15)))
            h_new = run_lbfgs(h_cur, active, alpha, penalty, ts, maxiter)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
            h_cur = h_new

        # Projected gradient descent
        if time_left() > 50:
            corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            active = get_active_lags(corr_cur, 0.99)
            if len(active) == 0:
                active = get_active_lags(corr_cur, 0.90)
            n_pg = min(200, max(30, int(time_left() * 0.1)))
            h_pg, c5_pg = projected_gradient_descent(h_cur, active, 2000.0, n_pg, 0.1, ts)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n
            h_cur = h_pg

        # Coordinate descent with random subsets
        if time_left() > 40:
            n_rounds = min(25, max(5, int(time_left() * 0.02)))
            n_subset = min(n, max(50, n // 8))
            steps = [0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01]
            h_cd, c5_cd = coordinate_descent_random(h_cur, ts, n_rounds, n_subset, rng, steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n
            h_cur = h_cd

        # Second L-BFGS pass with very high alpha
        if time_left() > 35:
            corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            active = get_active_lags(corr_cur, 0.998)
            if len(active) == 0:
                active = get_active_lags(corr_cur, 0.99)
            maxiter = min(200, max(30, int(time_left() * 0.08)))
            h_new = run_lbfgs(h_cur, active, 50000.0, 10.0, ts, maxiter)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
            h_cur = h_new

    # Final polishing at best resolution
    if best_h is not None and time_left() > 30:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_f = best_h.copy()

        # Very fine coordinate descent
        if time_left() > 20:
            steps_fine = [0.003, 0.001, 0.0005, 0.0002, -0.0002, -0.0005, -0.001, -0.003]
            n_rounds = min(40, max(5, int(time_left() * 0.05)))
            n_subset = min(n_f, max(80, n_f // 5))
            h_cd, c5_cd = coordinate_descent_random(h_f, ts_f, n_rounds, n_subset, rng, steps_fine)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_f
            h_f = h_cd

        # Final L-BFGS with extreme alpha
        if time_left() > 25:
            corr_cur = np.correlate(h_f, 1.0 - h_f, mode='full')
            active = get_active_lags(corr_cur, 0.999)
            if len(active) == 0:
                active = get_active_lags(corr_cur, 0.995)
            maxiter = min(300, max(50, int(time_left() * 0.3)))
            h_new = run_lbfgs(h_f, active, 100000.0, 50.0, ts_f, maxiter)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_f
            h_f = h_new

        # Projected gradient final
        if time_left() > 15:
            corr_cur = np.correlate(h_f, 1.0 - h_f, mode='full')
            active = get_active_lags(corr_cur, 0.999)
            if len(active) == 0:
                active = get_active_lags(corr_cur, 0.995)
            n_pg_f = min(100, max(10, int(time_left() * 0.5)))
            h_pg, c5_pg = projected_gradient_descent(h_f, active, 50000.0, n_pg_f, 0.05, ts_f)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n_f

    # Ensure validity
    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, ts_final)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
