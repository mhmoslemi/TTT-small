    h_new = h.copy()

        # Move a block of 1s
        target = 1.0
    else:
        target = 0.0

    runs = []

        while j < n and abs(h[j] - target) < 0.5:
            j += 1
        if j - i >= 2:
            runs.append((i, j))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * dx
    return obj, grad

def gradient_polish(h, n, alphas, max_iter_each, lr_base, time_fn):
    h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        if time_fn() < 10:
            break
        obj, grad = smooth_max_obj(h, alpha)
        lr = lr_base / np.sqrt(alpha)
        for it in range(max_iter_each):
            if time_fn() < 5:
                break
            h_new = h - lr * grad
            h_new = project_box_sum(h_new, n / 2.0)
            obj_new, grad_new = smooth_max_obj(h_new, alpha)
            if obj_new > obj:
                lr *= 0.7
                h_new = project_box_sum(h - lr * grad, n / 2.0)
                obj_new, grad_new = smooth_max_obj(h_new, alpha)
            else:
                lr = min(lr * 1.05, 0.5 / (alpha * 0.01))
            h = h_new
            grad = grad_new
            obj = obj_new
        c5_check = compute_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
    return h, best_c5

        C_mat[:, 2*(m-1)] = np.cos(np.pi * m * x)
        C_mat[:, 2*(m-1)+1] = np.sin(np.pi * m * x)
    coeffs, _, _, _ = np.linalg.lstsq(C_mat, target, rcond=None)
    
    for alpha in [50, 200, 800, 3000]:
        if time_fn() < 15:
            break
        try:
            res = minimize(
                fourier_obj_wrap, coeffs.copy(),
                args=(n, n_harm, alpha),
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            x_grid = np.linspace(0.0, 2.0, n, endpoint=False)
            h_opt = np.full(n, 0.5)
            for m in range(1, n_harm + 1):
                a = res.x[2*(m-1)]
                b = res.x[2*(m-1)+1]
                h_opt += a * np.cos(np.pi*m*x_grid) + b * np.sin(np.pi*m*x_grid)
            h_opt = project_box_sum(h_opt, n/2.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                target = best_h - 0.5
                coeffs, _, _, _ = np.linalg.lstsq(C_mat, target, rcond=None)
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    
    if time_left() < 20:
        best_h = project_box_sum(best_h, n/2.0)
        return (best_h, compute_c5(best_h), n)
    
    phase1_time = min(50, time_left() * 0.08)
    phase2_time = min(700, time_left() * 0.55)
    phase3_time = min(350, time_left() * 0.25)
    
    end_phase1 = t_start + phase1_time
    def t1():
        return min(time_left(), end_phase1 - time.time())
    
    alphas_pg = [20, 80, 300, 1000]
    for alpha in alphas_pg:
        if t1() < 8:
            break
        obj, grad = smooth_max_obj(best_h, alpha)
        lr = 0.3 / np.sqrt(alpha)
        for it in range(2000):
            if t1() < 3:
                break
            h_new = best_h - lr * grad
            h_new = project_box_sum(h_new, n / 2.0)
            obj_new, grad_new = smooth_max_obj(h_new, alpha)
            if obj_new > obj:
                lr *= 0.6
                h_new = project_box_sum(best_h - lr * grad, n / 2.0)
                obj_new, grad_new = smooth_max_obj(h_new, alpha)
            else:
                lr = min(lr * 1.03, 0.2)
            best_h = h_new
            grad = grad_new
            obj = obj_new
        c5_chk = compute_c5(best_h)
        if c5_chk < best_c5:
            best_c5 = c5_chk
    
    end_phase2 = t_start + phase1_time + phase2_time
    def t2():
        return min(time_left(), end_phase2 - time.time())
    
    best_h, best_c5 = simulated_annealing(best_h, n, t2, rng, phase2_time * 0.7)
    
    if time_left() > 100:
        best_h, best_c5 = iterated_local_search(best_h, n, t2, rng, 5)
    
    if time_left() < 30:
        best_h = project_box_sum(best_h, n/2.0)
        best_c5 = compute_c5(best_h)
        return (best_h, best_c5, n)
    
    end_phase3 = min(t_start + budget_s - 30, t_start + phase1_time + phase2_time + phase3_time)
    def t3():
        return min(time_left(), end_phase3 - time.time())
    
    best_h, best_c5 = fourier_refine(best_h, n, t3, rng)
    
    if time_left() > 80:
        n_up = min(480, n * 2)
        if n_up % 2 != 0:
            n_up += 1
        ts_up = n_up / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_up = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old, best_h, period=2.0)
        h_up = project_box_sum(h_up, ts_up)
        
        c5_up = compute_c5(h_up)
        if c5_up <= best_c5 + 0.001:
            n_up_work = n_up
            for alpha in [20, 100, 500, 2000]:
                if t3() < 15:
                    break
                obj, grad = smooth_max_obj(h_up, alpha)
                lr = 0.2 / np.sqrt(alpha)
                for it in range(1500):
                    if t3() < 5:
                        break
                    h_new = h_up - lr * grad
                    h_new = project_box_sum(h_new, ts_up)
                    obj_new, grad_new = smooth_max_obj(h_new, alpha)
                    if obj_new > obj:
                        lr *= 0.6
                        h_new = project_box_sum(h_up - lr * grad, ts_up)
                        obj_new, grad_new = smooth_max_obj(h_new, alpha)
                    else:
                        lr = min(lr * 1.02, 0.15)
                    h_up = h_new
                    grad = grad_new
                    obj = obj_new
                c5_chk = compute_c5(h_up)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h_up.copy()
                    best_n = n_up_work
            
            if time_left() > 40:
                best_h, best_c5 = simulated_annealing(best_h, best_n, t3, rng, 60)
    
    if time_left() > 20:
        best_h, _ = gradient_polish(best_h, best_n, [1000, 5000, 20000], 2000, 0.05, time_left)
    
    if time_left() > 15:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        for n_final in [240, 480]:
            if n_final == best_n:
                continue
            x_fn = np.linspace(0, 2, n_final, endpoint=False)
            h_fn = np.interp(x_fn, x_old, best_h, period=2.0)
            h_fn = project_box_sum(h_fn, n_final / 2.0)
            c5_fn = compute_c5(h_fn)
            if c5_fn < best_c5:
                best_c5 = c5_fn
                best_h = h_fn.copy()
                best_n = n_final
    
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
