    h = h0.copy()
    for a in alphas:
        for pen in penalties:
            def fun(x):
                obj, grad = smooth_max_obj_and_grad(x, a)
                s_dev = np.sum(x) - target_sum
                return obj + 0.5 * pen * s_dev * s_dev
            def jac(x):

                s_dev = np.sum(x) - target_sum
