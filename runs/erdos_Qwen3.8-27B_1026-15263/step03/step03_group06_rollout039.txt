import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad_fast(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    shifted = np.clip(alpha * (c - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    
    N = 1
    while N < 4 * n:
        N *= 2
    
    W = np.fft.rfft(w, N)
    H = np.fft.rfft(h, N)
    H_rev = np.fft.rfft(h[::-1], N)
    
    conv1 = np.fft.irfft(W * H, N)
    conv2 = np.fft.irfft(W * H_rev, N)
    
    term1 = conv1[n-1:2*n-1]
    term2 = conv2[2*n-2:n-2:-1]
    
    grad = 1.0 - term1 - term2
    return sm_val, grad

def opt_projected_grad(h0, target_sum, alpha, n_iters, lr_init, rng):
    n = len(h0)
    h = h0.copy()
    lr = lr_init
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for it in range(n_iters):
        sm_val, grad = smooth_max_and_grad_fast(h, alpha)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-5:
                h_pert = project_to_sum(h + rng.normal(0, 0.01, n), target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_cur:
                    h = h_pert
                    c5_cur = c5_pert
                lr = 0.05
    return h_best, c5_best

def fft_correlate(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    one_minus_h = 1.0 - h
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    O = np.fft.rfft(one_minus_h, N)
    # Cross-correlation: ifft(conj(fft(a)) * fft(b)) or ifft(fft(a) * conj(fft(b)))
    # np.correlate(a, b, 'full')[k] = sum_n a[n+k-(nb-1)] * b[n]
    # This is the same as the cross-correlation.
    # Using FFT: corr = ifft(fft(a, N) * conj(fft(b, N)))
    # But we need to handle the indexing correctly.
    
    # Let me use: np.correlate(a, b) = convolve(a, b[::-1]) reversed
    # Or: use the formula directly
    # corr_full[k] = sum_i a[i] * b[i+k-na+1] for k=0,...,na+nb-2
    # where na=len(a), nb=len(b)
    # If na=nb=n: corr_full[k] = sum_i a[i] * b[i+k-n+1]
    
    # Alternative: corr_full = ifft(fft(a, N) * conj(fft(b, N))) with proper indexing
    # The circular cross-correlation is: (a ⊛ b)[k] = sum_n a[n+k] * conj(b[n])
    # For linear correlation with zero-padding to N >= 2n-1:
    # (a ⊛ b)[k] = sum_n a[n+k] * b[n] for k = 0, ..., N-1
    # The first n values are the positive lags (0 to n-1)
    # The last n-1 values (N-n+1 to N-1) are the negative lags (-(n-1) to -1)
    
    result = np.fft.irfft(H * np.conj(O), N)
    # result[0] = sum a[i]*b[i] (lag 0)
    # result[1] = sum a[i+1]*b[i] (lag 1) = sum b[i]*a[i+1]
    # result[N-1] = sum a[i-1]*b[i] (lag -1)
    # ...
    
    # np.correlate(h, 1-h, 'full') order: lags -(n-1), ..., 0, ..., n-1
    # = result[N-(n-1):] + result[:n]
    # = result[N-n+1:] + result[:n]
    return np.concatenate([result[N-n+1:], result[:n]])

def fft_correlate(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT."""
    n1, n2 = len(a), len(b)
    n = max(n1, n2)
    N = 1
    while N < 2 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(A * np.conj(B), N)
    return np.concatenate([result[N-n+1:], result[:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_correlate_full(h):
    """Compute np.correlate(h, 1-h, mode='full') via FFT."""
    n = len(h)
    b = 1.0 - h
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(H * np.conj(B), N)
    return np.concatenate([result[N-n+1:], result[:n]])

    n = len(h)
    corr = fft_correlate_full(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s

    # Compute gradient via FFT
    N = 1
    while N < 4 * n:
        N *= 2
    
    W = np.fft.rfft(w, N)
    H = np.fft.rfft(h, N)
    H_rev = np.fft.rfft(h[::-1], N)
    
    conv1 = np.fft.irfft(W * H, N)
    conv2 = np.fft.irfft(W * H_rev, N)
    
    term1 = conv1[n-1:2*n-1]
    term2 = conv2[2*n-2:n-2:-1]
    
    grad = 1.0 - term1 - term2
    return sm_val, grad

        sm_val, grad = smooth_max_and_grad_fast(h, alpha)
        h_try = np.clip(h - lr * grad, 0.0, 1.0)
        h_try = project_to_sum(h_try, target_sum)
        sm_try, _ = smooth_max_and_grad_fast(h_try, alpha)
        
        if sm_try < sm_val:
            h = h_try
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-7:
                # Random restart from current point
                h_pert = project_to_sum(h + rng.normal(0, 0.05, n), target_sum)
                if compute_c5(h_pert) < c5_cur:
                    h = h_pert
                    c5_cur = compute_c5(h)
                lr = 0.05
        
        if (it + 1) % check_every == 0:

    c5_final = compute_c5(h)
    if c5_final < c5_cur:
        c5_cur = c5_final
    
    return h, c5_cur

def slsqp_fast(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        v, _ = smooth_max_and_grad_fast(x, alpha)
        return v
    
    def grad(x):
        _, g = smooth_max_and_grad_fast(x, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

        h_snap = project_to_sum(h_snap, target_sum)

    return h

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_correlate_full(h):
    n = len(h)
    b = 1.0 - h
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(H * B.conj(), N)
    return np.concatenate([result[N - n + 1:], result[:n]])

def smooth_max_and_grad_fast(h, alpha):
    n = len(h)
    corr = fft_correlate_full(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    N = 1
    while N < 4 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    H = np.fft.rfft(h, N)
    H_rev = np.fft.rfft(h[::-1], N)
    conv1 = np.fft.irfft(W * H, N)
    conv2 = np.fft.irfft(W * H_rev, N)
    term1 = conv1[n - 1:2 * n - 1]
    term2 = conv2[2 * n - 2:n - 2:-1]
    grad = 1.0 - term1 - term2
    return sm_val, grad

def pgd_optimize(h0, target_sum, alpha, n_iters, lr_init, rng, check_every=100):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    lr = lr_init
    stall_count = 0
    for it in range(n_iters):
        sm_val, grad = smooth_max_and_grad_fast(h, alpha)
        h_try = np.clip(h - lr * grad, 0.0, 1.0)
        h_try = project_to_sum(h_try, target_sum)
        sm_try, _ = smooth_max_and_grad_fast(h_try, alpha)
        if sm_try < sm_val:
            h = h_try
            lr = min(lr * 1.01, 0.5)
            stall_count = 0
        else:
            lr *= 0.5
            stall_count += 1
            if stall_count > 50:
                h_pert = project_to_sum(h + rng.normal(0, 0.03, n), target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_cur:
                    h = h_pert
                    c5_cur = c5_pert
                lr = lr_init
                stall_count = 0
        if (it + 1) % check_every == 0:
            c5_check = compute_c5(h)
            if c5_check < c5_cur:
                c5_cur = c5_check
    c5_final = compute_c5(h)
    c5_cur = min(c5_cur, c5_final)
    return h, c5_cur

def slsqp_fast(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        v, _ = smooth_max_and_grad_fast(np.clip(x, 0.0, 1.0), alpha)
        return v
    def grad(x):
        _, g = smooth_max_and_grad_fast(np.clip(x, 0.0, 1.0), alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_fast(h0, n, target_sum, n_rounds, rng, n_tries=5):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_v = h[i]
            offsets = rng.uniform(0, 1, n_tries)
            trial_vals = np.clip(h[i] + (offsets - 0.5) * 0.2, 0.0, 1.0)
            for tv in trial_vals:
                h_tmp = h.copy()
                h_tmp[i] = tv
                h_tmp = project_to_sum(h_tmp, target_sum)
                c5_tmp = compute_c5(h_tmp)
                if c5_tmp < best_val - 1e-12:
                    best_val = c5_tmp
                    best_v = tv
            if best_v != h[i]:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
