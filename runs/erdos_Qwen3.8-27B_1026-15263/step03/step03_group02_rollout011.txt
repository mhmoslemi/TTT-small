    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -50, 0)
    exp_v = np.exp(shifted)

def project_to_sum(h, target_sum):

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0

        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0

def soft_max_obj(h, alpha):

def soft_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -50, 0)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for mshift in range(1, n):

            js = np.arange(mshift, n)
            grad[js] -= wp * h[js - mshift]
        wn = w[n - 1 - mshift]
        if wn > 1e-14:
            jf = np.arange(mshift, n)
            grad[jf] += wn * (1.0 - h[jf - mshift])
            js = np.arange(n - mshift)
            grad[js] -= wn * h[js + mshift]
    return grad

def lbfgs_optimize(h0, n, target_sum, alpha, penalty, maxiter=300):
    def f(x):
        obj, _ = _obj_grad_helper(x, alpha)
        penalty_val = penalty * (np.sum(x) - target_sum) ** 2
        return obj + penalty_val

    def g(x):
        _, grad = _obj_grad_helper(x, alpha)
        grad += 2.0 * penalty * (np.sum(x) - target_sum)
        return grad

    _obj_grad_helper._n = n
    try:
        res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def _obj_grad_helper(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -50, 0)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    obj = m_val + np.log(s) / alpha
    w = exp_v / s
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for mshift in range(1, n):
        wp = w[n - 1 + mshift]
        if wp > 1e-14:
            jf = np.arange(n - mshift)
            grad[jf] += wp * (1.0 - h[jf + mshift])
            js = np.arange(mshift, n)
            grad[js] -= wp * h[js - mshift]
        wn = w[n - 1 - mshift]
        if wn > 1e-14:
            jf = np.arange(mshift, n)
            grad[jf] += wn * (1.0 - h[jf - mshift])
            js = np.arange(n - mshift)
            grad[js] -= wn * h[js + mshift]
    return obj, grad

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def make_periodic_pattern(n, period, frac_ones, rng):
    h = np.zeros(n)
    k = max(1, int(period * frac_ones))
    for start in range(0, n, period):
        end = min(start + k, n)
        h[start:end] = 1.0
    return make_binary(h, n, n // 2, rng)

def golden_angle_pattern(n, rng):
    golden = (1 + np.sqrt(5)) / 2.0
    positions = np.floor(np.arange(n) * (1.0 / (golden - 1.0)) / 2.0).astype(int) % n
    h = np.zeros(n)
    h[positions[:n // 2]] = 1.0
    return make_binary(h, n, n // 2, rng)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    n_main = 240
    ts_main = n_main // 2

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
            n_main = best_n
            ts_main = best_n // 2
        except Exception:
            best_h = golden_angle_pattern(n_main, rng)
            best_c5 = compute_c5(best_h)

    if best_h is None or best_n != n_main:
        h_init = golden_angle_pattern(n_main, rng)
        best_h = h_init
        best_c5 = compute_c5(best_h)
        best_n = n_main

    h_bin = make_binary(best_h, n_main, ts_main, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.40 * budget_s
    h_sa, c5_sa = fast_sa(h_bin, n_main, ts_main, sa_end1, rng, use_targeted=True)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    best_h = h_sa.copy()

    alphas = [8, 20, 60, 200, 800, 3000]
    penalties = [50.0, 200.0, 800.0, 3000.0]
    for idx, alpha in enumerate(alphas):
        if remaining() < 40:
            break
        pen = penalties[min(idx // 2, len(penalties) - 1)]
        maxit = max(50, int(remaining() * 0.08))
        h_opt = lbfgs_optimize(best_h, n_main, float(ts_main), alpha, pen, maxiter=maxit)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_binary = make_binary(h_opt, n_main, ts_main, rng)
        c5_binary = compute_c5(h_binary)
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary.copy()

    sa_end2 = t_start + 0.72 * budget_s
    if remaining() > 30:
        h_sa2, c5_sa2 = fast_sa(best_h, n_main, ts_main, sa_end2, rng, use_targeted=True)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    ntries = [300, 360, 480]
    for n_try in ntries:
        if remaining() < 120:
            break
        ts_try = n_try // 2
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, float(ts_try))
        h_bin_try = make_binary(h_interp, n_try, ts_try, rng)
        c5_try = compute_c5(h_bin_try)
        if c5_try < best_c5 - 0.001:
            for alpha in [20, 100, 500]:
                if remaining() < 40:
                    break
                h_opt_try = lbfgs_optimize(h_bin_try, n_try, float(ts_try), alpha, 500.0, maxiter=200)
                c5_opt_try = compute_c5(h_opt_try)
                if c5_opt_try < best_c5:
                    best_c5 = c5_opt_try
                    best_h = h_opt_try.copy()
                    best_n = n_try
                h_bin_try = make_binary(h_opt_try, n_try, ts_try, rng)
                c5_try2 = compute_c5(h_bin_try)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_bin_try.copy()
                    best_n = n_try
            if remaining() > 60:
                h_sa3, c5_sa3 = fast_sa(best_h, best_n, best_n // 2, t_start + 0.92 * budget_s, rng, use_targeted=True)
                if c5_sa3 < best_c5:
                    best_c5 = c5_sa3
                    best_h = h_sa3.copy()

    for trial in range(6):
        if remaining() < 50:
            break
        noise = 0.12 * (1.0 - trial / 6.0) + 0.03
        h_shake = best_h + rng.uniform(-noise, noise, best_n)
        h_shake = project_to_sum(h_shake, float(best_n // 2))
        h_shake_bin = make_binary(h_shake, best_n, best_n // 2, rng)
        alpha_use = [100, 500, 2000][min(trial // 2, 2)]
        h_opt = lbfgs_optimize(h_shake_bin, best_n, float(best_n // 2), alpha_use, 1000.0, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_bin_final = make_binary(h_opt, best_n, best_n // 2, rng)
        c5_bf = compute_c5(h_bin_final)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bin_final.copy()

    if remaining() > 30:
        h_final_sa, c5_final = fast_sa(best_h, best_n, best_n // 2, t_start + 0.97 * budget_s, rng, use_targeted=True)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_sa.copy()

    best_h = project_to_sum(best_h, float(best_n // 2))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
