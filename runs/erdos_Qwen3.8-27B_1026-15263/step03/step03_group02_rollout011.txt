```python
def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = np.sum(exp_vals)
    w = exp_vals / w_sum
    obj = m_val + np.log(w_sum) / alpha
    
    # grad[p] = dx * sum_d w[n-1+d] * (I[d<=p<n+d] * (1-h[p-d]) - I[-d<=p<n-d] * h[p+d])
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[n-1+d] * dx
        if wd < 1e-15:
            continue
        # First term: d <= p < n+d, equivalently max(0, d) <= p <= min(n-1, n+d-1)
        # and contribution is (1-h[p-d])
        p_lo1 = max(0, d)
        p_hi1 = min(n-1, n+d-1)
        if p_lo1 <= p_hi1:
            p_range = np.arange(p_lo1, p_hi1+1)
            grad[p_range] += wd * (1.0 - h[p_range - d])
        # Second term: -d <= p < n-d, equivalently max(0, -d) <= p <= min(n-1, n-d-1)
        # and contribution is -h[p+d]
        p_lo2 = max(0, -d)
        p_hi2 = min(n-1, n-d-1)
        if p_lo2 <= p_hi2:
            p_range2 = np.arange(p_lo2, p_hi2+1)
            grad[p_range2] -= wd * h[p_range2 + d]
    
    return obj, grad
```
