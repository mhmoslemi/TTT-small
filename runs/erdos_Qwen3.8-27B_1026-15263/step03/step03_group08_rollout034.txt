import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad_smooth_max(h, alpha):
    """Compute smooth max value and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        lag = idx - (n - 1)
        wl = w[idx]
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            g[lo1:hi1] += wl * (1.0 - h[lo1 + lag:hi1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 - lag:hi2 - lag]
    return sm_val, g

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init=0.01):
    """Nesterov accelerated projected gradient for smooth max objective."""
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t = 1.0
    lr = lr_init
    
    for it in range(n_iters):
        val, g = corr_and_grad_smooth_max(y, alpha)
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        x_new = np.clip(y - lr * g, 0.0, 1.0)
        x_new = project_to_sum(x_new, target_sum)
        y = x_new + beta * (x_new - h)
        y = project_to_sum(y, target_sum)
        h = x_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive lr
        if c5 < best_c5 + 1e-8:  # making progress
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = corr_and_grad_smooth_max(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = corr_and_grad_smooth_max(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def penalty_descent(h0, n, target_sum, C_corr_target, n_iters, lr_init):
    """Penalty method focusing on exceeding correlations."""
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        excess = np.maximum(corr - C_corr_target, 0.0)
        # Weight by excess^2 for focus on worst
        total_excess = np.sum(excess)
        if total_excess < 1e-12:
            break
        # Gradient of sum(excess^2) / 2
        g = np.zeros(n)
        nz = np.where(excess > 1e-12)[0]
        for idx in nz:
            lag = idx - (n - 1)
            w = excess[idx]
            lo1 = max(0, -lag)
            hi1 = min(n, n - lag)
            if lo1 < hi1:
                g[lo1:hi1] += w * (1.0 - h[lo1 + lag:hi1 + lag])
            lo2 = max(0, lag)
            hi2 = min(n, n + lag)
            if lo2 < hi2:
                g[lo2:hi2] -= w * h[lo2 - lag:hi2 - lag]
        
        # Nesterov momentum
        v = 0.9 * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h):
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.85
            v *= 0.5
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_h = h.copy()
            for d in steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new
            if best_c5 < c5_cur - 1e-14:
                h = best_h
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

        h_pen, c5_pen = penalty_gradient(h_cur, n, ts, C_corr,
                                          min(4000, int(tl() * 3)), 0.05)
        if c5_pen < best_c5:
            best_c5 = c5_pen
            best_h = h_pen.copy()
        h_cur = h_pen

    # Phase 4: Try n=600
    if tl() > 180:
        n2 = 600
        ts2 = n2 / 2.0
        h_interp = interpolate_h(best_h, n2)
        h_interp = project_to_sum(h_interp, ts2)
        c5_interp = compute_c5(h_interp)
        h_cur2 = h_interp.copy()
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n2

        alphas_p2 = [50.0, 120.0, 300.0, 700.0, 1500.0]
        for alpha in alphas_p2:
            if tl() < 55:
                break
            n_iter2 = min(350, max(100, int(tl() * 0.12)))
            h_opt2, c5_opt2 = nesterov_optimize(h_cur2, n2, ts2, alpha, n_iter2, 0.04)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2
            h_cur2 = h_opt2

        if tl() > 50:
            h_opt2 = slsqp_optimize(h_cur2, n2, ts2, 8000.0, maxiter=250)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2

    # Phase 5: Try n=720 if time permits
    if tl() > 150:
        n3 = 720
        ts3 = n3 / 2.0
        h_interp3 = interpolate_h(best_h, n3)
        h_interp3 = project_to_sum(h_interp3, ts3)
        c5_3 = compute_c5(h_interp3)
        h_cur3 = h_interp3.copy()
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_interp3.copy()
            best_n = n3

        for alpha in [80.0, 200.0, 500.0, 1200.0]:
            if tl() < 45:
                break
            n_iter3 = min(250, max(80, int(tl() * 0.12)))
            h_opt3, c5_opt3 = nesterov_optimize(h_cur3, n3, ts3, alpha, n_iter3, 0.03)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
                best_n = n3
            h_cur3 = h_opt3

    # Phase 6: Coordinate descent polish
    if tl() > 35:
        n_cd = best_n
        base_steps = [0.015, 0.008, 0.004, 0.002, -0.015, -0.008, -0.004, -0.002]
        max_rnd = min(40, max(10, int(tl() // 4)))
        h_cd, c5_cd = coordinate_descent(best_h, n_cd, n_cd / 2.0, max_rnd, base_steps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final Nesterov with high alpha
    if tl() > 20:
        h_fin, c5_fin = nesterov_optimize(best_h, best_n, best_n / 2.0,
                                           alpha=8000.0, n_iters=600, lr_init=0.015)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: One more SLSQP if time allows
    if tl() > 30:
        h_fin2 = slsqp_optimize(best_h, best_n, best_n / 2.0, 20000.0, maxiter=200)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    # Final validation
    n_final = best_n
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
