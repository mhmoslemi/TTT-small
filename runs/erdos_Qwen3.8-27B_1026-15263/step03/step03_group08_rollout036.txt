    g = np.zeros(n)
    n_lags = 2 * n - 1
    for idx in range(n_lags):
        wl = w[idx]
        if wl < 1e-12:
            continue
        l = idx - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def slsqp_obj(x, alpha, target_sum):
    hx = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(hx, alpha)
    return v

def slsqp_grad(x, alpha, target_sum):
    hx = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(hx, alpha)
    return g

def slsqp_cfun(x, alpha, target_sum):
    return np.sum(x) - target_sum

def slsqp_cjac(x, alpha, target_sum):
    return np.ones(len(x))

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    constraints = [{'type': 'eq', 'fun': slsqp_cfun, 'jac': slsqp_cjac,
                   'args': (alpha, target_sum)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h0.copy(), jac=slsqp_grad, args=(alpha, target_sum),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def mw_gradient_descent(h0, n, target_sum, n_iters, lr_init, beta_start, beta_end, t_start, budget):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    n_lags = 2 * n - 1

    for it in range(n_iters):
        if time.time() - t_start > budget - 30:
            break
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        shifted = beta * corr - np.max(beta * corr)
        shifted = np.clip(shifted, -500.0, 500.0)
        w = np.exp(shifted)
        w /= np.sum(w)

        g = np.zeros(n)
        for idx in range(n_lags):
            wl = w[idx]
            if wl < 1e-10:
                continue
            l = idx - (n - 1)
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]

        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new > best_c5 + 1e-6:
            lr *= 0.9
        elif c5_new < best_c5 - 1e-9:
            lr = min(lr * 1.03, 1.5)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new

    return best_h, best_c5

def active_set_slsqp(h0, n, target_sum, n_rounds, gap, alpha_tail, t_start, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    n_lags = 2 * n - 1

    for rnd in range(n_rounds):
        if time.time() - t_start > budget - 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        active = np.where(corr >= m - gap)[0]

        if len(active) < 2:
            break

        def _ao_obj(x, _active=active, _alpha=alpha_tail, _ts=target_sum):
            hx = np.clip(x, 0.0, 1.0)
            c = np.correlate(hx, 1.0 - hx, mode='full')
            m2 = np.max(c[_active])
            sh = _alpha * (c[_active] - m2)
            sh = np.clip(sh, -500.0, 500.0)
            ev = np.exp(sh)
            s = np.sum(ev)
            return m2 + np.log(s) / _alpha

        def _ao_grad(x, _active=active, _alpha=alpha_tail, _ts=target_sum):
            hx = np.clip(x, 0.0, 1.0)
            c = np.correlate(hx, 1.0 - hx, mode='full')
            m2 = np.max(c[_active])
            sh = _alpha * (c[_active] - m2)
            sh = np.clip(sh, -500.0, 500.0)
            ev = np.exp(sh)
            s = np.sum(ev)
            w = ev / s
            g = np.zeros(n)
            for j in range(len(active)):
                wl = w[j]
                if wl < 1e-12:
                    continue
                idx = active[j]
                l = idx - (n - 1)
                lo = max(0, l)
                hi = min(n, n + l)
                if lo < hi:
                    g[lo:hi] += wl * (1.0 - hx[lo - l:hi - l])
                lo2 = max(0, -l)
                hi2 = min(n, n - l)
                if lo2 < hi2:
                    g[lo2:hi2] -= wl * hx[lo2 + l:hi2 + l]
            return g

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_ao_obj, h.copy(), jac=_ao_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': slsqp_cfun, 'jac': slsqp_cjac,
                                        'args': (alpha_tail, target_sum)}],
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_new = project_to_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new.copy()
        except Exception:
            pass

    return h, best_c5

def fine_coord_descent(h0, n, target_sum, max_rounds, delta_init, t_start, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    delta = delta_init

    for rnd in range(max_rounds):
        if time.time() - t_start > budget - 15:
            break
        improved = False
        for i in range(n):
            cur_val = h[i]
            best_i_val = cur_val
            best_c5_i = best_c5
            for sign in [1.0, -1.0]:
                for frac in [1.0, 0.5, 0.25, 0.1]:
                    d = delta * frac
                    val = cur_val + sign * d
                    if val < 0.0 or val > 1.0:
                        continue
                    h[i] = val
                    h = project_to_sum(h, target_sum)
                    c = compute_c5(h)
                    if c < best_c5_i - 1e-13:
                        best_c5_i = c
                        best_i_val = h[i]
                    h[i] = cur_val
            if best_i_val != cur_val:
                h[i] = best_i_val
                h = project_to_sum(h, target_sum)
                best_c5 = best_c5_i
                improved = True
        if not improved:
            break
        delta *= 0.6

    return h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def diverse_starts(h_base, n, rng, k):
    ts = n / 2.0
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    starts = [h_base.copy()]
    for i in range(k):
        delta = 0.03 + 0.1 * i
        h_p = h_base + rng.uniform(-delta, delta, n)
        h_p = project_to_sum(h_p, ts)
        starts.append(h_p)
    for freq in [1, 2, 3, 5]:
        for amp in [0.15, 0.3]:
            for phase in [0.0, np.pi / 2.0, np.pi]:
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                hs = project_to_sum(hs, ts)
                starts.append(hs)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    elapsed = lambda: time.time() - t_start

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals():
        try:
            iv = initial_h_values
            if iv is not None:
                best_h = np.array(iv, dtype=float)
                best_n = len(best_h)
                ts = best_n / 2.0
                best_h = project_to_sum(best_h, ts)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: SLSQP at current resolution with increasing alpha
    alpha_seq_1 = [300.0, 800.0, 2000.0, 5000.0]
    for alpha in alpha_seq_1:
        if elapsed() > budget_s - 200:
            break
        maxiter = max(80, min(1000, int((budget_s - 200 - elapsed()) * 0.5)))
        h_opt = slsqp_optimize(best_h, n, ts, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Active set SLSQP
    if elapsed() < budget_s - 120:
        for gap in [0.05, 0.02, 0.01, 0.005]:
            if elapsed() > budget_s - 80:
                break
            h_as, c5_as = active_set_slsqp(best_h, n, ts, 5, gap, 50.0, t_start, budget_s)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()

    # Phase 3: Try higher resolution
    n_high = min(960, n * 2)
    if n_high % 2 != 0:
        n_high += 1
    ts_high = n_high / 2.0

    if elapsed() < budget_s - 300:
        h_high = upsample(best_h, n_high)
        h_high = project_to_sum(h_high, ts_high)
        c5_high = compute_c5(h_high)

        if c5_high <= best_c5 + 0.01:
            for alpha in [500.0, 2000.0, 5000.0]:
                if elapsed() > budget_s - 100:
                    break
                maxiter = max(50, min(800, int((budget_s - 100 - elapsed()) * 0.3)))
                h_opt_h = slsqp_optimize(h_high, n_high, ts_high, alpha, maxiter)
                c5_opt_h = compute_c5(h_opt_h)
                if c5_opt_h < c5_high:
                    c5_high = c5_opt_h
                    h_high = h_opt_h.copy()

            if elapsed() < budget_s - 60:
                h_as_h, c5_as_h = active_set_slsqp(h_high, n_high, ts_high, 5, 0.01, 50.0, t_start, budget_s)
                if c5_as_h < c5_high:
                    c5_high = c5_as_h
                    h_high = h_as_h.copy()

            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 4: MW gradient descent at best resolution
    n_cur = len(best_h)
    ts_cur = n_cur / 2.0
    if elapsed() < budget_s - 80:
        n_iters = min(2000, int((budget_s - 80 - elapsed()) * 2))
        h_mw, c5_mw = mw_gradient_descent(best_h, n_cur, ts_cur, n_iters, 0.08, 100.0, 2000.0, t_start, budget_s)
        if c5_mw < best_c5:
            best_c5 = c5_mw
            best_h = h_mw.copy()

    # Phase 5: Final SLSQP polish with very high alpha
    for alpha in [5000.0, 15000.0, 30000.0]:
        if elapsed() > budget_s - 40:
            break
        maxiter = max(100, min(2000, int((budget_s - 40 - elapsed()) * 0.5)))
        h_opt = slsqp_optimize(best_h, n_cur, ts_cur, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Fine coordinate descent
    if elapsed() < budget_s - 20:
        max_rnd = max(3, min(15, int((budget_s - 20 - elapsed()) // 5)))
        h_cd, c5_cd = fine_coord_descent(best_h, n_cur, ts_cur, max_rnd, 0.015, t_start, budget_s)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final ultra-fine coordinate descent
    if elapsed() < budget_s - 10:
        h_cd2, c5_cd2 = fine_coord_descent(best_h, n_cur, ts_cur, 8, 0.004, t_start, budget_s)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Final validation
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
