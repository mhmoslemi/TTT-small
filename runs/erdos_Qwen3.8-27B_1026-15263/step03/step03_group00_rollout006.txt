def subgradient_optimize(h, target_sum, max_iters=10000):
    n = len(h)
    h = h.copy()
    step0 = 0.01  # initial step size
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m = np.max(corr)
        # Get all lags within 1% of max
        threshold = m * 0.99
        active_lags = np.where(corr >= threshold)[0]
        
        # Compute subgradient as average of active lag gradients
        grad = np.zeros(n)
        for lag_idx in active_lags:
            lag = lag_idx - (n-1)
            grad += corr_gradient(h, lag)
        grad /= len(active_lags)
        
        # Step size: t_k = c / sqrt(k) (Polyak step size)
        t = step0 / np.sqrt(it + 1)
        
        # Update
        h_new = h - t * grad
        h_new = project_to_sum(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, h, h])
    h_ext_x = np.concatenate([x_old, x_old + 2.0, x_old + 4.0])
    h_new = np.interp(x_new, h_ext_x, h_ext)
    return h_new

def anneal_perturb(h, target_sum, rng, max_iters, time_limit, temp_init, temp_final):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    stale = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        temp = temp_init * (temp_final / temp_init) ** (float(it) / max(1, max_iters))
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        i_inc = int(rng.randint(n))
        i_dec = int(rng.randint(n))
        if i_inc == i_dec:
            continue
        if h[i_inc] >= 1.0 - 1e-10 or h[i_dec] <= 1e-10:
            continue
        step = 0.02 * min(1.0 - h[i_inc], h[i_dec])
        if step < 1e-12:
            continue
        h_trial = h.copy()
        h_trial[i_inc] += step
        h_trial[i_dec] -= step
        h_trial = project_to_sum(h_trial, target_sum)
        c5_new = compute_c5(h_trial)
        delta = c5_new - compute_c5(h)
        if delta < 0 or rng.uniform() < np.exp(-delta / max(temp, 1e-10)):
            h = h_trial
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_trial.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1
        if stale > 500:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_remaining():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_curr = best_n
    ts = n_curr / 2.0

    # Phase 1: Subgradient optimization
    if time_remaining() > 200:
        h_sg, c5_sg = subgradient_opt(best_h, ts, max_iters=8000, step_init=0.03,
                                       top_m=3, time_limit=min(180, time_remaining() - 20))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            n_curr = len(best_h)
            ts = n_curr / 2.0

    # Phase 2: Projected gradient descent on smooth approximation
    if time_remaining() > 150:
        for alpha, lr, iters in [(500, 0.02, 3000), (2000, 0.005, 2000)]:
            if time_remaining() < 60:
                break
            h_pg, c5_pg = projected_grad_descent(best_h, ts, alpha, iters, lr,
                                                   time_limit=min(120, time_remaining() - 20))
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 3: Line-search coordinate descent
    if time_remaining() > 100:
        h_cd, c5_cd = coord_descent_ls(best_h, ts, max_iters=15000,
                                        time_limit=min(180, time_remaining() - 20))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Pair coordinate descent (finer)
    if time_remaining() > 80:
        h_pc, c5_pc = pair_coord_descent(best_h, ts, max_iters=10000,
                                          time_limit=min(100, time_remaining() - 20))
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Phase 5: Resolution escalation
    higher_ns = [720, 900]
    for n_hi in higher_ns:
        if time_remaining() < 200:
            break
        ts_hi = n_hi / 2.0
        h_interp = interp_to_n(best_h, n_hi)
        h_interp = project_to_sum(h_interp, ts_hi)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_hi
        if time_remaining() > 120:
            h_sg2, c5_sg2 = subgradient_opt(h_interp, ts_hi, max_iters=5000, step_init=0.02,
                                             top_m=3, time_limit=min(100, time_remaining() - 30))
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2.copy()
                best_n = n_hi
            h_interp = h_sg2
        if time_remaining() > 60:
            h_cd2, c5_cd2 = coord_descent_ls(h_interp, ts_hi, max_iters=8000,
                                              time_limit=min(50, time_remaining() - 20))
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_hi

    # Phase 6: More intensive local optimization at best resolution
    n_curr = best_n
    ts = n_curr / 2.0
    if time_remaining() > 100:
        h_sg3, c5_sg3 = subgradient_opt(best_h, ts, max_iters=10000, step_init=0.015,
                                         top_m=5, time_limit=min(90, time_remaining() - 20))
        if c5_sg3 < best_c5:
            best_c5 = c5_sg3
            best_h = h_sg3.copy()

    # Phase 7: Fine line-search coordinate descent
    if time_remaining() > 80:
        h_cd3, c5_cd3 = coord_descent_ls(best_h, ts, max_iters=20000,
                                          time_limit=min(80, time_remaining() - 20),
                                          step_fracs=[0.3, 0.1, 0.03, 0.01, 0.003])
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 8: Simulated annealing perturbation
    if time_remaining() > 80:
        h_an, c5_an = anneal_perturb(best_h, ts, rng, max_iters=30000,
                                      time_limit=min(70, time_remaining() - 20),
                                      temp_init=0.01, temp_final=0.001)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an.copy()
        if time_remaining() > 40:
            h_cd4, c5_cd4 = coord_descent_ls(best_h, ts, max_iters=10000,
                                              time_limit=min(35, time_remaining() - 15))
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()

    # Phase 9: Final fine-tuning
    n_curr = best_n
    ts = n_curr / 2.0
    if time_remaining() > 40:
        h_sg4, c5_sg4 = subgradient_opt(best_h, ts, max_iters=8000, step_init=0.008,
                                         top_m=5, time_limit=min(30, time_remaining() - 15))
        if c5_sg4 < best_c5:
            best_c5 = c5_sg4
            best_h = h_sg4.copy()

    if time_remaining() > 25:
        h_cd5, c5_cd5 = coord_descent_ls(best_h, ts, max_iters=15000,
                                          time_limit=min(20, time_remaining() - 10),
                                          step_fracs=[0.1, 0.03, 0.01, 0.003, 0.001])
        if c5_cd5 < best_c5:
            best_c5 = c5_cd5
            best_h = h_cd5.copy()

    # Final projection and evaluation
    best_n = len(best_h)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
