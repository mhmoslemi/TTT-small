                    g_total[:] += (1.0 - 2.0 * h)
                else:
                    g_total[lag:] += (1.0 - h[:n - lag])
                    g_total[:n - lag] -= h[lag:]
            else:
                k = -lag
                g_total[:n - k] += (1.0 - h[k:])
                g_total[k:] -= h[:n - k]
        g_total /= len(active)
        lr = lr_init * (1.0 + it / 500.0) ** (-0.7)
        h_new = h - lr * g_total
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    n = 540
    target_sum = n / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        n = len(best_h)
        target_sum = n / 2.0
        best_h = project_to_sum(best_h, target_sum)
        best_c5 = compute_c5(best_h)
    else:
        best_h = np.full(540, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: DE with diverse initialization
    pop_size = 25
    pop = np.zeros((pop_size, n))
    fitness = np.zeros(pop_size)

    pop[0] = best_h
    fitness[0] = best_c5

    h_tb = np.zeros(n)
    h_tb[:n // 4] = 1.0
    h_tb[n // 2:3 * n // 4] = 1.0
    pop[1] = project_to_sum(h_tb, target_sum)
    fitness[1] = compute_c5(pop[1])

    h_ob = np.zeros(n)
    h_ob[:n // 2] = 1.0
    pop[2] = project_to_sum(h_ob, target_sum)
    fitness[2] = compute_c5(pop[2])

    h_osc = 0.5 + 0.5 * np.sin(4.0 * np.pi * np.arange(n) / n)
    pop[3] = project_to_sum(np.clip(h_osc, 0, 1), target_sum)
    fitness[3] = compute_c5(pop[3])

    h_two = np.zeros(n)
    h_two[:n // 6] = 1.0
    h_two[n // 3:2 * n // 3] = 1.0
    h_two[5 * n // 6:] = 1.0
    pop[4] = project_to_sum(h_two, target_sum)
    fitness[4] = compute_c5(pop[4])

    for i in range(5, pop_size):
        noise = rng.uniform(-0.25, 0.25, n)
        pop[i] = project_to_sum(best_h + noise, target_sum)
        fitness[i] = compute_c5(pop[i])

    max_gen = 120
    n_stagnant = 0
    for gen in range(max_gen):
        if time_left() < 100:
            break
        F_gen = 0.85 - 0.35 * (gen / float(max_gen))
        CR_gen = min(0.6 + 0.3 * (gen / float(max_gen)), 0.9)
        prev_best = fitness.min()
        pop, fitness = de_step(pop, fitness, F_gen, CR_gen, rng, n, target_sum)
        cur_best_idx = int(np.argmin(fitness))
        if fitness[cur_best_idx] < best_c5 - 1e-14:
            best_c5 = fitness[cur_best_idx]
            best_h = pop[cur_best_idx].copy()
            n_stagnant = 0
        elif fitness[cur_best_idx] >= prev_best - 1e-12:
            n_stagnant += 1
        else:
            n_stagnant = 0

        if n_stagnant > 25 and time_left() > 200:
            bi = int(np.argmin(fitness))
            for i in range(pop_size):
                if i == bi:
                    continue
                noise = rng.uniform(-0.12, 0.12, n)
                pop[i] = project_to_sum(best_h + noise, target_sum)
                fitness[i] = compute_c5(pop[i])
            n_stagnant = 0

    # Phase 2: SLSQP refinement with increasing alpha
    for alpha in [30, 100, 400, 1500, 6000, 25000]:
        if time_left() < 35:
            break
        h_new = slsqp_smooth(best_h, n, target_sum, alpha, maxiter=1200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Coordinate descent
    if time_left() > 80:
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, max_iters=20000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Multi-lag subgradient
    if time_left() > 80:
        h_sg, c5_sg = multi_lag_subgradient(best_h, target_sum, max_iters=3000, lr_init=0.003, rng=rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Second DE round (tighter around best)
    if time_left() > 180:
        pop_size2 = 20
        pop2 = np.zeros((pop_size2, n))
        fit2 = np.zeros(pop_size2)
        pop2[0] = best_h
        fit2[0] = best_c5
        for i in range(1, pop_size2):
            noise = rng.uniform(-0.04, 0.04, n)
            pop2[i] = project_to_sum(best_h + noise, target_sum)
            fit2[i] = compute_c5(pop2[i])
        n_stag2 = 0
        for gen in range(100):
            if time_left() < 100:
                break
            pop2, fit2 = de_step(pop2, fit2, 0.4, 0.8, rng, n, target_sum)
            bi2 = int(np.argmin(fit2))
            if fit2[bi2] < best_c5 - 1e-14:
                best_c5 = fit2[bi2]
                best_h = pop2[bi2].copy()
                n_stag2 = 0
            else:
                n_stag2 += 1
            if n_stag2 > 35 and time_left() > 120:
                for i in range(pop_size2):
                    if i == bi2:
                        continue
                    noise = rng.uniform(-0.06, 0.06, n)
                    pop2[i] = project_to_sum(best_h + noise, target_sum)
                    fit2[i] = compute_c5(pop2[i])
                n_stag2 = 0

    # Phase 6: Final SLSQP polishing
    for alpha in [50000, 150000, 300000]:
        if time_left() < 25:
            break
        h_new = slsqp_smooth(best_h, n, target_sum, alpha, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Final coordinate descent
    if time_left() > 25:
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, max_iters=25000, step_frac=0.04)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: One more SLSQP at very high alpha
    if time_left() > 15:
        h_new = slsqp_smooth(best_h, n, target_sum, 500000, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_to_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
