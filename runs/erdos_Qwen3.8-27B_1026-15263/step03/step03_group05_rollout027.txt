import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def projected_gradient_descent(h0, n, alpha, max_iter, lr_init, time_fn):
    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        else:
            lr = min(lr * 1.1, 0.5)
        h = h_new
    return best_h, best_obj

def sa_with_targeted_moves(h0, n, rng, time_fn):
    """SA with a mix of random and targeted moves."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ts = n / 2.0
    
    T_start = 0.15
    T_end = 0.00001
    total_budget_iters = 300000
    it = 0
    stagnation = 0
    max_stagnation = 10000
    
    while time_fn() > 15:
        it += 1
        T = max(T_end, T_start * (T_end / T_start) ** (it / total_budget_iters))
        
        # Decide move type
        r = rng.random()
        prev_h = h.copy()
        
        if r < 0.25:
            # Single perturbation
            i = rng.randint(n)
            h[i] += rng.uniform(-0.5, 0.5)
        elif r < 0.45:

            L = rng.randint(2, 6)
            i = rng.randint(max(1, n - L))

        elif r < 0.6:

            j = rng.randint(n)

        elif r < 0.75:

            L = rng.randint(5, max(6, n // 15))

def coordinate_descent_c5(h0, n, time_fn, max_rounds=200):
    """Direct coordinate descent with exact C5 evaluation."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    ts = n / 2.0
    # Candidate values to try for each coordinate
    base_candidates = [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]
    
    for r in range(max_rounds):
        if time_fn() < 5:
            break
        improved = False
        # Random order for diversity
        order = np.arange(n)
        rng_cd = np.random.RandomState(r * 12345 + 7)
        rng_cd.shuffle(order)
        
        for i in order:
            if time_fn() < 5:
                break
            old_val = h[i]
            best_val = old_val
            best_c5 = c5_cur
            
            # Try a few candidate values
            for cv in base_candidates:
                if abs(cv - old_val) < 1e-10:
                    continue
                h[i] = cv
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_val = cv
            
            # Also try values near current
            for cv in [old_val + 0.1, old_val - 0.1, old_val + 0.2, old_val - 0.2]:
                cv = np.clip(cv, 0.0, 1.0)
                if abs(cv - old_val) < 1e-10 or abs(cv - best_val) < 1e-10:
                    continue
                h[i] = cv
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_val = cv
            
            h[i] = best_val
            if best_val != old_val:
                improved = True
                c5_cur = best_c5
        
        if not improved:
            break
    
    h = project_box_sum(h, ts)
    c5_cur = compute_c5(h)
    return h, c5_cur
