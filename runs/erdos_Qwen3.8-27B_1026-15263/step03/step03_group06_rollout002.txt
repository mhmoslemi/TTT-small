**Spectral/Fourier optimization**: Instead of optimizing h in the time domain, optimize the Fourier coefficients directly. The constraint 0≤h≤1 becomes a constraint in the time domain, but the objective (max correlation) might have a nice form in Fourier space.

The autocorrelation of h: R_h[k] = sum_i h[i]*h[i+k]

**Alternative: Min-max via bisection**

```python
    # correlate(h, 1-h, 'full')[k] = corr[(k-(n-1)) % N]
    corr_full = np.concatenate([corr[N-(n-1):], corr[:n-1]])
    return float(np.max(corr_full) * dx)

def compute_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2*n-1:
        N *= 2
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr_full = np.concatenate([corr[N-(n-1):], corr[:n-1]])
    return corr_full

def smooth_max_and_grad_fft(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Compute gradient via FFT
    # w has length 2n-1
    # conv1[j] = ((1-h) * w)[j+n-1]
    # conv2[j] = (h * w_rev)[n-1+j]
    w_rev = w[::-1]  # w_rev[i] = w[2n-2-i]
    
    N = 1
    while N < 2*n:
        N *= 2
    
    F1h = np.fft.rfft(1.0 - h, N)
    Fh = np.fft.rfft(h, N)
    Fw = np.fft.rfft(w, N)
    Fw_rev = np.fft.rfft(w_rev, N)
    
    conv1_full = np.fft.irfft(F1h * Fw, N)
    conv2_full = np.fft.irfft(Fh * Fw_rev, N)
    
    conv1 = conv1_full[n-1:2*n-1]  # j=0..n-1 → index j+n-1
    conv2 = conv2_full[n-1:2*n-1]  # j=0..n-1 → index n-1+j
    
    grad = conv1 - conv2
    return sm_val, grad
```

For a standard linear convolution: (a * b)[k] = sum_m a[m]*b[k-m]
With FFT: ifft(FFT(a, N) * FFT(b, N)) gives the circular convolution, which equals linear convolution when N >= len(a)+len(b)-1.

(1-h) has length n, w has length 2n-1.
Linear convolution has length n + (2n-1) - 1 = 3n-2.
Need N >= 3n-2.

Similarly for h (length n) and w_rev (length 2n-1): same.

So N should be >= 3n-2, not just 2n-1. Let me fix that.

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr_raw = np.fft.irfft(fa * np.conj(fb), N)
    corr_full = np.concatenate([corr_raw[N - (n-1):], corr_raw[:n-1]])
    return corr_full

def compute_c5_fft(h):
    corr = compute_corr_fft(h)
    n = len(h)
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def smooth_max_grad_fft(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    w_rev = w[::-1].copy()
    N = 1
    while N < 3 * n - 1:
        N *= 2
    F1h = np.fft.rfft(1.0 - h, N)
    Fh = np.fft.rfft(h, N)
    Fw = np.fft.rfft(w, N)
    Fw_rev = np.fft.rfft(w_rev, N)
    conv1_full = np.fft.irfft(F1h * Fw, N)
    conv2_full = np.fft.irfft(Fh * Fw_rev, N)
    conv1 = conv1_full[n-1:2*n-1]
    conv2 = conv2_full[n-1:2*n-1]
    grad = conv1 - conv2
    return sm_val, grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0 and abs(diff) <= 0.5 * cnt:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_proj(h0, n, target_sum, alpha, n_iters, t_end):
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    h_best = h.copy()
    c5_best = compute_c5_fft(h)
    lr = 0.01
    
    for it in range(n_iters):
        if it >= t_end:
            break
        _, g = smooth_max_grad_fft(y, alpha)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break
        g = g / norm_g * min(1.0, norm_g)  # normalize if too large
        
        # Determining step size with backtracking
        step = lr
        for _ in range(20):
            h_new = y - step * g
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5_fft(h_new)
            if c5_new < c5_best or c5_new < compute_c5_fft(h) * 1.01:
                break
            step *= 0.5
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = compute_c5_fft(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
    
    return h_best, c5_best

def coord_descent_fft(h0, n, target_sum, n_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5_fft(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5_fft(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_init(n, rng, n_modes):
    # Create a function with a few Fourier modes
    freqs = np.arange(1, n_modes + 1)
    phases = rng.uniform(0, 2*np.pi, n_modes)
    amps = rng.uniform(0.1, 0.3, n_modes)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + sum(amps[k] * np.cos(2*np.pi*freqs[k]*x/2 + phases[k]) for k in range(n_modes))
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n/2.0)
    return h
```
