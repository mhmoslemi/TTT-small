                n_it = min(800, int((time_limit_total - (time.time() - t_start)) * 1.5))
                n_it = max(50, n_it)
                h_opt, c5_opt = momentum_bts(h_cur, n, ts, n_it, alpha_a, alpha_a * 2.0, t_start, time_limit_total - 40)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt

            if time.time() - t_start < time_limit_total - 50:
                maxiter_sl = max(50, min(400, int((time_limit_total - (time.time() - t_start)) * 0.4)))
                h_opt2 = slsqp_optimize(h_cur, n, ts, 800.0, maxiter_sl)
                c5_opt2 = compute_c5(h_opt2)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n
                h_cur = h_opt2

            if time.time() - t_start < time_limit_total - 30:
                h_cd1, c5_cd1 = coarse_coord_descent(h_cur, n, ts, 15, t_start, time_limit_total - 30)
                if c5_cd1 < best_c5:
                    best_c5 = c5_cd1
                    best_h = h_cd1.copy()
                    best_n = n
                h_cur = h_cd1

    if best_h is not None and time.time() - t_start < time_limit_total - 30:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha_p in [2000.0, 5000.0, 10000.0]:
            if time.time() - t_start > time_limit_total - 20:
                break
            n_it_p = min(500, int((time_limit_total - (time.time() - t_start)) * 0.8))
            n_it_p = max(30, n_it_p)
            h_pol, c5_pol = momentum_bts(best_h, n_f, ts_f, n_it_p, alpha_p, alpha_p * 3.0, t_start, time_limit_total - 20)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time.time() - t_start < time_limit_total - 15:
            steps_fine = [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]
            h_cd2, c5_cd2 = fine_coord_descent(best_h, n_f, ts_f, 80, steps_fine, t_start, time_limit_total - 10)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

        if time.time() - t_start < time_limit_total - 8:
            steps_ultra = [0.003, 0.001, -0.003, -0.001]
            h_cd3, c5_cd3 = fine_coord_descent(best_h, n_f, ts_f, 120, steps_ultra, t_start, time_limit_total - 5)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
