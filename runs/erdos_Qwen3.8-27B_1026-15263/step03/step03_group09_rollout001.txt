        h_cd2, _ = coord_descent(h_start, ts, n_cd2, rng, 0.002, n_deltas=9)
        try_update(h_cd2, n_pts)
        h_start = h_cd2

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)

    if _tl(t_start, budget_s) > 30:
        h_cd_final, _ = coord_descent(best_h, ts_final, min(30, int(_tl(t_start, budget_s) * 0.3)), rng, 0.001, n_deltas=11)
        try_update(h_cd_final, n_final)

    if _tl(t_start, budget_s) > 20:
        h_fc_final, _ = fine_coord_descent(best_h, ts_final, min(20, int(_tl(t_start, budget_s) * 0.2)), rng, n_trials=60)
        try_update(h_fc_final, n_final)

    if _tl(t_start, budget_s) > 15:
        h_pol, _ = topk_subgrad_optimize(best_h, ts_final, 5000, 0.001, 2, 10, momentum=0.2, lr_decay=0.05)
        try_update(h_pol, n_final)

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
