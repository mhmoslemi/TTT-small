def project_bisect(h, target_sum):

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init, time_deadline):
    """Nesterov accelerated gradient with backtracking line search."""
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best
    
    for it in range(n_iters):
        if time.time() > time_deadline:
            break
        
        val, g = compute_grad_fast(y, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y = h + momentum * (h - y)
        y = np.clip(y, 0.0, 1.0)
        
        # Recompute gradient at y
        val, g = compute_grad_fast(y, alpha)
        
        # Backtracking line search
        lr_try = lr
        h_new = project_bisect(y - lr_try * g, target_sum)
        c5_new = compute_c5(h_new)
        
        # Simple backtracking
        retries = 0
        while c5_new > val + 0.5 * lr_try * np.dot(g, h_new - y) and retries < 20:
            lr_try *= 0.5
            h_new = project_bisect(y - lr_try * g, target_sum)
            c5_new = compute_c5(h_new)
            retries += 1
        
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                lr = min(lr_try * 1.1, 1.0)
            else:
                lr = lr_try
        else:
            lr = max(lr_try, 1e-6)
        
        t_val = t_new
    
    return h_best, c5_best

def coordinate_descent(h0, n, target_sum, n_rounds, rng, time_deadline):
    """Coordinate descent with line search on each coordinate."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(n_rounds):
        if time.time() > time_deadline:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() > time_deadline:
                break
            lo = max(0.0, h[i] - 0.1)
            hi = min(1.0, h[i] + 0.1)
            if hi - lo < 1e-8:
                continue
            
            h_orig_i = h[i]
            # Try a few values
            candidates = [lo, lo+0.25*(hi-lo), lo+0.5*(hi-lo), lo+0.75*(hi-lo), hi]
            if h_orig_i > lo and h_orig_i < hi:
                candidates.append(h_orig_i)
            
            best_c5 = c5_cur
            best_val = h_orig_i
            
            for v in candidates:
                if v < lo or v > hi:
                    continue
                h[i] = v
                h = project_bisect(h, target_sum)
                c5_test = compute_c5(h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_val = v
                h[i] = h_orig_i
                h = project_bisect(h, target_sum)
            
            if best_c5 < c5_cur - 1e-12:
                h[i] = best_val
                h = project_bisect(h, target_sum)
                c5_cur = best_c5
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bisect(h, target_sum):
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_val_weights(f, alpha):
    m = np.max(f)
    shifted = np.clip(alpha * (f - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    return val, w

def compute_grad_fast(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    val, w = smooth_max_val_weights(f, alpha)
    g = np.zeros(n)
    for s_val in range(-(n - 1), n):
        wm = w[s_val + n - 1]
        if wm < 1e-14:
            continue
        if s_val >= 0:
            g[s_val:] += wm * dx * (1.0 - h[:n - s_val])
            g[:n - s_val] -= wm * dx * h[s_val:]
        else:
            p = -s_val
            g[:n - p] += wm * dx * (1.0 - h[p:])
            g[p:] -= wm * dx * h[:n - p]
    return val, g

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init, deadline):
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best
    for it in range(n_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y = h + momentum * (h - y)
        y = np.clip(y, 0.0, 1.0)
        val, g = compute_grad_fast(y, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        lr_try = lr
        h_new = project_bisect(y - lr_try * g, target_sum)
        c5_new = compute_c5(h_new)
        retries = 0
        while c5_new > val + 0.5 * lr_try * np.dot(g, h_new - y) and retries < 15:
            lr_try *= 0.5
            h_new = project_bisect(y - lr_try * g, target_sum)
            c5_new = compute_c5(h_new)
            retries += 1
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                lr = min(lr_try * 1.05, 0.5)
            else:
                lr = lr_try
        else:
            lr = max(lr_try, 1e-7)
            if lr < 1e-7:
                break
        t_val = t_new
    return h_best, c5_best

def coordinate_descent(h0, n, target_sum, n_rounds, rng, deadline):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() > deadline:
                break
            lo = max(0.0, h[i] - 0.08)
            hi = min(1.0, h[i] + 0.08)
            if hi - lo < 1e-7:
                continue
            h_save = h.copy()
            best_c5 = c5_cur
            best_h = h_save.copy()
            fracs = [0.0, 0.2, 0.35, 0.5, 0.65, 0.8, 1.0]
            for fr in fracs:
                v = lo + fr * (hi - lo)
                h[i] = v
                h = project_bisect(h, target_sum)
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-13:
                    best_c5 = c5_test
                    best_h = h.copy()
            if best_c5 < c5_cur - 1e-13:
                h = best_h
                c5_cur = best_c5
                improved = True
            else:
                h = h_save
        if not improved:
            break
    return h, c5_cur

def subgradient_step(h, n, target_sum, step_size, deadline):
    c5_val, max_idx = compute_c5(h), 0
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    f = corr * dx
    max_lag = np.argmax(f)
    s = max_lag - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += dx * (1.0 - h[:n - s])
        g[:n - s] -= dx * h[s:]
    else:
        p = -s
        g[:n - p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n - p]
    h_new = project_bisect(h - step_size * g, target_sum)
    c5_new = compute_c5(h_new)
    if c5_new < c5_val:
        return h_new, c5_new
    h_new2 = project_bisect(h - 0.5 * step_size * g, target_sum)
    c5_new2 = compute_c5(h_new2)
    if c5_new2 < c5_val:
        return h_new2, c5_new2
    return h, c5_val

def fourier_init(n, rng, target_sum):
    n_freqs = rng.randint(5, 15)
    h = np.full(n, 0.5)
    for k in range(n_freqs):
        freq = rng.randint(1, max(2, n // 6))
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return project_bisect(h, target_sum)

def binary_init(n, rng, target_sum):
    n_ones = int(round(target_sum))
    if n_ones <= 0 or n_ones >= n:
        return np.full(n, 0.5)
    h = np.zeros(n)
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    return h

def cluster_init(n, rng, target_sum):
    n_clusters = rng.randint(3, 12)
    h = np.zeros(n)
    for c in range(n_clusters):
        center = rng.randint(0, n)
        width = rng.randint(max(2, n // 30), n // 6 + 1)
        start = max(0, center - width // 2)
        end = min(n, center + width // 2)
        h[start:end] = 1.0
    return project_bisect(h, target_sum)

def block_init(n, rng, target_sum):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        fill_len = rng.randint(1, n // 8 + 1)
        gap_len = rng.randint(1, n // 8 + 1)
        h[pos:min(pos + fill_len, n)] = 1.0
        pos += fill_len + gap_len
    return project_bisect(h, target_sum)
