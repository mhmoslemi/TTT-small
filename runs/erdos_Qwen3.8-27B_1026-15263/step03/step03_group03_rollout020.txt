from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    """Compute the vector B where B[m] = ∂corr[m]/∂h[j]"""
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    # First term: (1-h[j+m-nm1]) for valid m
    # j+m-nm1 is valid when 0 <= j+m-nm1 < n, i.e., nm1-j <= m < n+nm1-j
    k_start = max(0, nm1 - j)
    k_end = min(2*n-2, n + nm1 - j - 1)
    if k_start <= k_end:
        B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
    
    # Second term: -h[j-m+nm1] for valid m
    # j-m+nm1 is valid when 0 <= j-m+nm1 < n, i.e., j-nm1 <= m <= j+nm1
    k_start2 = max(0, j - nm1)
    k_end2 = min(2*n-2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
    return B

def solve_1d(A, D, lo, hi):
    """Solve min_t max_i (A[i] + D[i]*t) over t in [lo, hi]"""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=15, active_set=None):
    """Pair coordinate descent with active-set-aware candidate selection."""
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Build candidate set
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, 
                 (j + n // 3) % n, (j + 2*n // 3) % n,
                 (j + n // 4) % n, (j + 3*n // 4) % n]
        # Add random candidates
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        # If active_set provided, add nearby indices
        if active_set is not None:
            for ak in active_set[:5]:
                k_set.append((j + ak) % n)
                k_set.append((j - ak) % n)
        
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)

def lp_subgradient_step(h, corr, n, K=50):
    """LP-based subgradient step.
    Returns h_new which is the LP solution (or None if LP fails).
    """
    # Find top K correlation indices
    top_idx = np.argsort(corr)[-K:]
    
    # Compute gradient rows for these K indices
    nm1 = n - 1
    j_range = np.arange(n)
    G = np.zeros((K, n))
    for li, m in enumerate(top_idx):
        idx1 = j_range + m - nm1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_range - m + nm1
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid1):
            G[li, valid1] += (1.0 - h[idx1[valid1]])
        if np.any(valid2):
            G[li, valid2] -= h[idx2[valid2]]
    
    # Set up LP: minimize t
    # Variables: [h[0], ..., h[n-1], t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G[m,:] @ h - t <= G[m,:] @ h0 - corr[m]
    G_h0 = G @ h
    b_ub = G_h0 - corr[top_idx]
    A_ub = np.hstack([G, -np.ones((K, 1))])
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[j] <= 1, t free
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def simulated_annealing_step(h, t_temp, rng, n):

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, n / 2.0)
    return h_new

    for m in range(2*n-1):

        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_range - m + nm1
        valid2 = (idx2 >= 0) & (idx2 < n)

        if res.success:

        pass
    return None

def cutting_plane_method(h0, n, max_iter=1000, t_end=None):
    """Kelley cutting plane method for the max-of-bilinear objective."""
    h = h0.copy()
    corr = np.correlate(h, 1-h, mode="full")
    
    # Cuts: each cut is (G_row, b) where G_row @ h - t <= b
    cuts_G = [np.zeros(n)]
    cuts_b = [np.max(corr)]
    
    for it in range(max_iter):
        if t_end is not None and time.time() >= t_end:
            break
        
        # Solve LP with all cuts
        K = len(cuts_G)
        G_all = np.vstack(cuts_G)
        b_all = np.array(cuts_b)
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([G_all, -np.ones((K, 1))])
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_all, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if not res.success:
            break
        h_new = res.x[:n]
        
        # Evaluate true objective at h_new
        corr_new = np.correlate(h_new, 1-h_new, mode="full")
        corr_max_new = np.max(corr_new)
        
        # Find the worst shift at h_new
        m_worst = np.argmax(corr_new)
        
        # Check if we need a new cut
        # The cut for m_worst at h_new is:
        # G[m_worst, :] @ h - t <= G[m_worst, :] @ h_new - corr_new[m_worst]
        nm1 = n - 1
        j_range = np.arange(n)
        idx1 = j_range + m_worst - nm1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_range - m_worst + nm1
        valid2 = (idx2 >= 0) & (idx2 < n)
        G_new = np.zeros(n)
        if np.any(valid1):
            G_new[valid1] += (1.0 - h_new[idx1[valid1]])
        if np.any(valid2):
            G_new[valid2] -= h_new[idx2[valid2]]
        
        b_new = G_new @ h_new - corr_new[m_worst]
        
        # Check if this cut is redundant (already implied by existing cuts)
        # A simple check: if G_new @ h - t <= b_new is implied by existing cuts at h_new
        # This is complex; instead, just check if the cut is "active" enough
        # A simpler approach: add the cut if it's not too similar to existing ones
        
        # Add the cut (with limit on total cuts)
        if len(cuts_G) < 200:
            cuts_G.append(G_new)
            cuts_b.append(b_new)
        
        # Check for convergence
        optimality_gap = corr_max_new - res.x[-1]
        if optimality_gap < 1e-8:
            h = h_new
            break
        
        h = h_new
    
    corr = np.correlate(h, 1-h, mode="full")
    return h, compute_c5(h)

where each f_m is bilinear (neither convex nor concave). The standard cutting plane method doesn't directly apply.

However, I can still use it heuristically:

This is NOT a lower bound on f_m(h) in general (since f_m is neither convex nor concave).

So the cutting plane approach doesn't give valid lower bounds. It's just a heuristic.

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, n + nm1 - j - 1)
    if k_start <= k_end:
        B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
    k_start2 = max(0, j - nm1)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=15):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n//2) % n,
                 (j + n//3) % n, (j + 2*n//3) % n,
                 (j + n//4) % n, (j + 3*n//4) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def lp_step(h, corr, n, K=60):
    """Single LP-based step. Returns (h_new, improved) or (None, False)."""
    nm1 = n - 1
    # Get top K correlation indices
    top_idx = np.argsort(corr)[-K:]
    j_range = np.arange(n)
    
    G = np.zeros((K, n))
    for li, m in enumerate(top_idx):
        idx1 = j_range + m - nm1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_range - m + nm1
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid1):
            G[li, valid1] += (1.0 - h[idx1[valid1]])
        if np.any(valid2):
            G[li, valid2] -= h[idx2[valid2]]
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    G_h = G @ h
    b_ub = G_h - corr[top_idx]
    A_ub = np.hstack([G, -np.ones((K, 1))])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_lp = res.x[:n]
            return h_lp
    except Exception:
        pass
    return None
