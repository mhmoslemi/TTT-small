def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        # dc[k]/dh[j] = (1-h[j-k+nm1])*[valid] - h[j+k-nm1]*[valid]
        # grad[j] = sum_k w[k] * dx * dc[k]/dh[j]
        # First term: sum over k where 0 <= j-k+nm1 < n

        hi1 = min(2*n-2, j + nm1)

            grad[j] += np.sum(w[ks] * (1.0 - h[j - ks + nm1]))
        # Second term: sum over k where 0 <= j+k-nm1 < n

            ks = np.arange(lo2, hi2+1)
            grad[j] -= np.sum(w[ks] * h[j + ks - nm1])
        grad[j] *= dx

def pg_pnorm_step(h, corr, p, step_size, target_sum):

    n = len(h)

    c_scaled = corr * dx
    c_pos = np.maximum(c_scaled, 0.0)

    for j in range(n):
        lo1 = max(0, j + nm1 - n + 1)
        hi1 = min(2*n-2, j + nm1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1+1)
            grad[j] += np.sum(weights[ks] * (1.0 - h[j - ks + nm1]))
        lo2 = max(0, nm1 - j)
        hi2 = min(2*n-2, nm1 - j + n - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2+1)
            grad[j] -= np.sum(weights[ks] * h[j + ks - nm1])

    # Projected gradient step
    h_new = h - step_size * grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_B(h, n):
    nm1 = n - 1
    all_B = np.zeros((n, 2 * n - 1))
    for j in range(n):
        lo1 = max(0, j + nm1 - n + 1)
        hi1 = min(2 * n - 2, j + nm1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1)
            all_B[j, ks] += (1.0 - h[j - ks + nm1])
        lo2 = max(0, nm1 - j)
        hi2 = min(2 * n - 2, nm1 - j + n - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            all_B[j, ks] += -h[j + ks - nm1]
    return all_B

def solve_1d_fast(corr, D, lo, hi):

        return (lo + hi) * 0.5

def pair_descent_fast(h, corr, all_B, rng, n_cand=12):

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    s_exp = np.sum(exp_vals)
    w = exp_vals / s_exp
    obj = (m_val + np.log(s_exp) / alpha) * dx
    
    nm1 = n - 1
    grad = np.zeros(n)
    for j in range(n):
        lo1 = max(0, j + nm1 - n + 1)
        hi1 = min(2 * n - 2, j + nm1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1)
            grad[j] += np.sum(w[ks] * (1.0 - h[j - ks + nm1]))
        lo2 = max(0, nm1 - j)
        hi2 = min(2 * n - 2, nm1 - j + n - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            grad[j] -= np.sum(w[ks] * h[j + ks - nm1])
    grad *= dx
    return obj, grad

    n = len(h0)
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
