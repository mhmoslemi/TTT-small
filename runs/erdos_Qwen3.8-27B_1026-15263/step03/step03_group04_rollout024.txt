import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = int(np.sum(mask))
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_shift(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[s:n] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:n]
    else:
        p = -s
        if p < n:
            grad[:n-p] += (1.0 - h[p:n])
            grad[p:n] -= h[:n-p]
    return grad

def smooth_max_obj_grad(h, alpha, max_shifts=50):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Only keep top shifts to limit computation
    sorted_idx = np.argsort(corr)[::-1][:max_shifts]
    exp_v = np.zeros(len(corr))
    exp_v[sorted_idx] = np.exp(alpha * (corr[sorted_idx] - m))
    s_exp = np.sum(exp_v)
    if s_exp < 1e-300:
        exp_v[sorted_idx[0]] = 1.0
        s_exp = 1.0
    w = exp_v / s_exp
    obj = m + np.log(s_exp) / alpha
    grad = np.zeros(n)
    for k in sorted_idx:
        if w[k] > 1e-12:
            shift = k - (n - 1)
            grad += w[k] * corr_grad_shift(h, shift, n)
    return obj, grad

def exact_max_polish(h, n, target_sum, max_rounds=500, n_top=20):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        sorted_idx = np.argsort(corr)[::-1][:n_top]
        best_improve = 0.0
        best_h = None
        best_c5_new = c5_cur
        for k_max in sorted_idx:
            shift = k_max - (n - 1)
            grad = corr_grad_shift(h, shift, n)
            for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-16:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur

def subgradient_descent(h, n, target_sum, max_iter=2000, alpha_start=0.1, alpha_end=0.001):
    """Cycle through top shifts and take subgradient steps."""
    h = h.copy()
    c5_cur = compute_c5(h)
    for it in range(max_iter):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:10]
        # Average gradient over top shifts
        grad = np.zeros(n)
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad += corr_grad_shift(h, shift, n)
        grad /= len(top_k)
        # Adaptive step size
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iter)
        h_new = h - alpha * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
        else:
            alpha *= 0.5
            h_new = h - alpha * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
    return h, c5_cur

def lbfgs_smooth(h, n, target_sum, alpha, maxiter=500):
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        h_opt = project_to_sum(res.x, target_sum)
        return h_opt
    except Exception:
        return project_to_sum(h, target_sum)

def stochastic_refine(h, n, target_sum, rng, n_trials=200, noise_range=(0.05, 0.4)):
    h_best = h.copy()
    c5_best = compute_c5(h)
    for _ in range(n_trials):
        h_new = h_best.copy()
        sigma = rng.uniform(noise_range[0], noise_range[1])
        noise = rng.normal(0, sigma, n)
        for _ in range(2):
            noise = np.convolve(noise, np.ones(7)/7, mode='same')
        h_new += noise
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new.copy()
    return h_best, c5_best

def binary_search_refine(h, n, target_sum, rng, max_rounds=100):
    """Try to binary-ify the solution and optimize."""
    ts = target_sum
    n_ones = int(round(ts))
    h_bin = np.zeros(n)
    # Start from current h, threshold at 0.5
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n_ones ones
    diff = int(n_ones) - int(np.sum(h_bin))
    if diff > 0:
        # Need more ones: flip zeros closest to 0.5
        zero_idx = np.where(h_bin == 0)[0]
        zero_scores = h[zero_idx]  # higher is closer to 1
        flip_idx = zero_idx[np.argsort(zero_scores)][::-1][:diff]
        h_bin[flip_idx] = 1.0
    elif diff < 0:
        one_idx = np.where(h_bin == 1)[0]
        one_scores = h[one_idx]  # lower is closer to 0
        flip_idx = one_idx[np.argsort(one_scores)][:(-diff)]
        h_bin[flip_idx] = 0.0
    
    # Local search: swap pairs
    c5_cur = compute_c5(h_bin)
    for r in range(max_rounds):
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = compute_c5(h_bin)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, c5_cur

Actually, let me think about this more carefully. The objective is:
C5 = max_k sum_i h[i](1-h[i-s(k)]) * dx

where s(k) = k - (n-1) is the shift.

If we flip h[i] from 0 to 1, the change in the k-th term is:
Δ_k = (1-h[i-s(k)]) * dx - (h[i-s(k)] - h[i-s(k)])... 

Actually it's simpler: if h[i] goes from 0 to 1:
- For each shift s, the term at index i changes: h[i](1-h[i-s]) goes from 0*(1-h[i-s]) = 0 to 1*(1-h[i-s])
- Also, for each shift s, the term at index i+s changes: h[i+s](1-h[i]) goes from h[i+s]*(1-0) = h[i+s] to h[i+s]*(1-1) = 0

So flipping h[i] from 0 to 1:
Δ at shift s: +1*(1-h[i-s]) - h[i+s]*1 = 1 - h[i-s] - h[i+s]

And flipping h[i] from 1 to 0:
Δ at shift s: -1*(1-h[i-s]) + h[i+s]*1 = -(1-h[i-s]) + h[i+s] = h[i-s] + h[i+s] - 1

This is useful for the binary search.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = int(np.sum(mask))
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_shift(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[s:n] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:n]
    else:
        p = -s
        if p < n:
            grad[:n - p] += (1.0 - h[p:n])
            grad[p:n] -= h[:n - p]
    return grad

def smooth_max_obj_grad(h, alpha, max_shifts=50):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1][:max_shifts]
    exp_vals = np.exp(alpha * (corr[sorted_idx] - m))
    s_exp = np.sum(exp_vals)
    if s_exp < 1e-300:
        exp_vals[0] = 1.0
        s_exp = 1.0
    w = exp_vals / s_exp
    obj = m + np.log(s_exp) / alpha
    grad = np.zeros(n)
    for ii in range(len(sorted_idx)):
        if w[ii] > 1e-12:
            k = sorted_idx[ii]
            shift = k - (n - 1)
            grad += w[ii] * corr_grad_shift(h, shift, n)
    return obj, grad

def lbfgs_smooth(h, n, target_sum, alpha, maxiter=500):
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h.copy(), target_sum)

def exact_max_polish(h, n, target_sum, max_rounds=500, n_top=20, time_limit=None, t0=None):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        corr = compute_corr(h)
        sorted_idx = np.argsort(corr)[::-1][:n_top]
        best_improve = 0.0
        best_h = None
        best_c5_new = c5_cur
        for k_max in sorted_idx:
            shift = k_max - (n - 1)
            grad = corr_grad_shift(h, shift, n)
            for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-16:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur

def subgradient_descent(h, n, target_sum, max_iter=2000, time_limit=None, t0=None):
    h = h.copy()
    c5_cur = compute_c5(h)
    alpha_cur = 0.1
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:10]
        grad = np.zeros(n)
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad += corr_grad_shift(h, shift, n)
        grad /= len(top_k)
        h_new = h - alpha_cur * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            alpha_cur = max(0.001, alpha_cur * 0.98)
        else:
            alpha_cur *= 0.5
    return h, c5_cur

def stochastic_refine(h, n, target_sum, rng, n_trials=200, noise_range=(0.05, 0.35), time_limit=None, t0=None):
    h_best = h.copy()
    c5_best = compute_c5(h)
    for t in range(n_trials):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        sigma = rng.uniform(noise_range[0], noise_range[1])
        noise = rng.normal(0, sigma, n)
        for _ in range(2):
            noise = np.convolve(noise, np.ones(7) / 7, mode='same')
        h_new = h_best + noise
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new.copy()
    return h_best, c5_best

def binary_refine(h, n, target_sum, rng, max_rounds=150, time_limit=None, t0=None):
    ts = target_sum
    n_ones = int(round(ts))
    h_bin = (h > 0.5).astype(float)
    diff = n_ones - int(np.sum(h_bin))
    if diff > 0:
        zero_idx = np.where(h_bin == 0)[0]
        if len(zero_idx) >= diff:
            zero_scores = h[zero_idx]
            flip_idx = zero_idx[np.argsort(zero_scores)][::-1][:diff]
            h_bin[flip_idx] = 1.0
    elif diff < 0:
        one_idx = np.where(h_bin == 1)[0]
        if len(one_idx) >= (-diff):
            one_scores = h[one_idx]
            flip_idx = one_idx[np.argsort(one_scores)][:(-diff)]
            h_bin[flip_idx] = 0.0
    c5_cur = compute_c5(h_bin)
    for r in range(max_rounds):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(800, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = compute_c5(h_bin)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, c5_cur
