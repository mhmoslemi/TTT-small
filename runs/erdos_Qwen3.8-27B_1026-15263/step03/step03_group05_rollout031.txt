            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_pol = gradient_continuation(h_sa, time_left, alpha_list=[50, 300, 1000])
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 6: Shake and re-optimize (multiple rounds)
    for trial in range(15):
        if time_left() < 50:
            break
        noise_scale = 0.15 * (1.0 - trial / 15.0) + 0.03
        h_shake = best_h + rng.uniform(-noise_scale, noise_scale, best_n)
        h_shake = project_box_sum(h_shake, best_n / 2.0)
        h_opt = gradient_continuation(h_shake, time_left, alpha_list=[30, 200, 800])
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Also try binary version
        if time_left() > 30:
            h_b = (h_opt > 0.5).astype(float)
            h_b = project_box_sum(h_b, best_n / 2.0)
            h_bf, c5_bf = binary_local_search(h_b, best_n, rng, time_left, max_rounds=200)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()

    # Phase 7: High-resolution upsample
    if time_left() > 100:
        n_up = min(480, best_n * 2)
        if n_up % 2 != 0:
            n_up += 1
        h_up = upsample(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
        if time_left() > 50:
            h_up2 = gradient_continuation(h_up, time_left, alpha_list=[20, 100, 500])
            c5_up2 = compute_c5(h_up2)
            if c5_up2 < best_c5:
                best_c5 = c5_up2
                best_h = h_up2.copy()
                best_n = n_up

    # Phase 8: Final high-alpha polish
    for alpha in [2000, 8000, 30000]:
        if time_left() < 15:
            break
        h_fin, _ = gradient_descent_bt(best_h, alpha, time_left, max_iter=3000, lr_init=0.01)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
