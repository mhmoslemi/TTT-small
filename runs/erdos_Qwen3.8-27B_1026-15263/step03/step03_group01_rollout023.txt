import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_symmetric(h, n):
    """Project h onto the anti-symmetric subspace h[i+n/2] = 1-h[i]."""
    h_sym = np.empty(n)
    h_sym[:n//2] = (h[:n//2] + (1.0 - h[n//2:])) / 2.0
    h_sym[n//2:] = 1.0 - h_sym[:n//2]
    return h_sym

def smooth_max_obj_grad_sym(h_half, alpha, n):
    """Compute smooth max objective and gradient for symmetric case."""
    nh = n // 2
    h = np.empty(n)
    h[:nh] = h_half
    h[nh:] = 1.0 - h_half
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2*n-1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    grad_full = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    grad_half = grad_full[:nh] - grad_full[nh:]
    return obj, grad_half

def optimize_symmetric(h_half, n, alpha, maxiter=5000):
    """Optimize using L-BFGS-B on the symmetric subspace."""
    nh = n // 2
    bnds = [(0.0, 1.0)] * nh
    
    def f(x):
        return smooth_max_obj_grad_sym(x, alpha, n)[0]
    
    def g(x):
        return smooth_max_obj_grad_sym(x, alpha, n)[1]
    
    try:
        result = minimize(f, h_half.copy(), method='L-BFGS-B', jac=g,
                         bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return np.clip(result.x, 0.0, 1.0)
    except Exception:
        return h_half.copy()

**Projection approach for unconstrained case:**

3. Project h_new onto {h: sum(h) = n/2, 0 ≤ h ≤ 1}

The projection onto this set can be done as follows:

- This is equivalent to finding a threshold τ such that h*[i] = clip(h_new[i] - τ, 0, 1) and sum(h*) = n/2
- We can find τ by bisection (since sum(h*) is monotonically decreasing in τ)

This is O(n log n) per projection (for the bisection + sorting if needed, but actually bisection is O(n) per iteration and we need O(log(1/ε)) iterations).

def project_sum_box(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}."""
    h = np.clip(h, 0.0, 1.0).copy()
    # Bisection on tau: find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_symmetric(h, n):
    nh = n // 2
    h_sym = np.empty(n)
    h_sym[:nh] = (h[:nh] + (1.0 - h[nh:])) / 2.0
    h_sym[nh:] = 1.0 - h_sym[:nh]
    return h_sym

def build_grad_indices(n):
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    return fc, sc, vf, vs

def smooth_max_obj_grad_sym(h_half, alpha, n, idx=None):
    nh = n // 2
    h = np.empty(n)
    h[:nh] = h_half
    h[nh:] = 1.0 - h_half
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    if idx is None:
        fc, sc, vf, vs = build_grad_indices(n)
    else:
        fc, sc, vf, vs = idx
    grad_full = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    grad_half = grad_full[:nh] - grad_full[nh:]
    return obj, grad_half

def smooth_max_obj_grad_full(h, alpha, idx=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    if idx is None:
        fc, sc, vf, vs = build_grad_indices(n)
    else:
        fc, sc, vf, vs = idx
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def opt_sym_LBFGS(h_half, n, alpha, maxiter=3000, idx=None):
    nh = n // 2
    bnds = [(0.0, 1.0)] * nh

    def f(x):
        return smooth_max_obj_grad_sym(x, alpha, n, idx)[0]

    def g(x):
        return smooth_max_obj_grad_sym(x, alpha, n, idx)[1]

    try:
        result = minimize(f, h_half.copy(), method='L-BFGS-B', jac=g,
                         bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return np.clip(result.x, 0.0, 1.0)
    except Exception:
        return h_half.copy()

def project_sum_box(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-12:
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if np.any(mask):
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def projected_grad_sym(h_half, n, alpha, n_iter=5000, lr_init=0.005, idx=None):
    nh = n // 2
    h = h_half.copy()
    v = np.zeros(nh)
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad_sym(h, alpha, n, idx)
    best_h = h.copy()
    best_obj = obj_prev
    stall = 0
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad_sym(h, alpha, n, idx)
        v = 0.9 * v - lr * grad
        h_new = np.clip(h + v, 0.0, 1.0)
        obj_new, _ = smooth_max_obj_grad_sym(h_new, alpha, n, idx)
        if obj_new < obj_prev:
            h = h_new
            obj_prev = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.5
            v = np.zeros(nh)
            stall += 1
            if stall > 80 or lr < 1e-14:
                break
    return best_h

def freq_perturb_sym(h_half, n, rng, n_harm, amp):
    nh = n // 2
    h_full = np.empty(n)
    h_full[:nh] = h_half
    h_full[nh:] = 1.0 - h_half
    mean_h = np.mean(h_full)
    H = np.fft.rfft(h_full - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    h_sym = project_symmetric(h_new, n)
    return h_sym[:nh].copy()
