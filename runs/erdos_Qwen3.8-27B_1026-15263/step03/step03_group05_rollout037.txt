def single_point_delta(h, full_corr, i, delta):
    """Return the change in full_corr when h[i] changes by delta."""
    n = len(h)
    g = 1.0 - h
    n_minus_1 = n - 1
    
    delta_corr = np.zeros(2*n - 1)
    
    # Contribution from h[i] term: delta * g[i + k - (n-1)] for valid k
    # Valid when 0 <= i + k - (n-1) < n
    # => (n-1-i) <= k <= (n-1-i) + (n-1) = 2n-2-i
    k1_start = max(0, n_minus_1 - i)
    k1_end = min(2*n-2, n_minus_1 - i + n - 1)
    if k1_start <= k1_end:
        g_idx = i + np.arange(k1_start, k1_end + 1) - n_minus_1
        # g_idx should be in [0, n-1] by construction
        delta_corr[k1_start:k1_end+1] += delta * g[g_idx]
    
    # Contribution from g[i] term: -delta * h[i - k + (n-1)] for valid k
    # Valid when 0 <= i - k + (n-1) < n
    # => i - (n-1) <= k <= i + (n-1)
    k2_start = max(0, i - n_minus_1)
    k2_end = min(2*n-2, i + n_minus_1)
    if k2_start <= k2_end:
        h_idx = i - np.arange(k2_start, k2_end + 1) + n_minus_1
        delta_corr[k2_start:k2_end+1] -= delta * h[h_idx]
    
    # Special case: k = n-1 where both terms hit the same point
    # The above double-counts: at k=n-1, we added delta*g[i] and -delta*h[i]
    # But the correct change is delta*(1-2*h[i]) - delta^2
    # What we computed: delta*g[i] - delta*h[i] = delta*(1-h[i]) - delta*h[i] = delta*(1-2*h[i])
    # Correct: delta*(1-2*h[i]) - delta^2
    # So we need to subtract delta^2 at k=n-1
    if 0 <= n_minus_1 < 2*n - 1:
        delta_corr[n_minus_1] -= delta * delta
    
    return delta_corr
