def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_max(h, n):
    """Compute subgradient of max_k overlap(k) at h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    # k_star is in [0, 2n-2], corresponding to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    
    # Compute gradient of overlap(lag) w.r.t. h
    g = np.zeros(n)
    for i in range(n):
        # i as first factor: h[i]*(1-h[i+lag])
        j = i + lag
        if 0 <= j < n:
            g[i] += dx * (1.0 - h[j])
        # i as second factor: h[i-lag]*(1-h[i])
        j2 = i - lag
        if 0 <= j2 < n:
            g[i] -= dx * h[j2]
    
    # Also include other lags that are close to the max (for robustness)
    # Use top-5 lags with weighted average
    top_k = np.argsort(corr)[-5:]
    weights = np.exp(corr[top_k] / max(1e-10, np.max(corr) * 0.1))
    weights /= weights.sum()
    
    g_total = np.zeros(n)
    for w, k_idx in zip(weights, top_k):
        lag_i = k_idx - (n - 1)
        for i in range(n):
            j = i + lag_i
            if 0 <= j < n:
                g_total[i] += w * dx * (1.0 - h[j])
            j2 = i - lag_i
            if 0 <= j2 < n:
                g_total[i] -= w * dx * h[j2]
    
    return g_total, corr.max() * dx, k_star

    top_m = 10

    temp = corr[k_top]
    weights = np.exp((temp - temp.max()) / (temp.max() - temp.min() + 1e-10) * 5.0)

                c5_best = c5_new
                h_best = h_new.copy()
                alpha = alpha * 1.5
                if alpha > 2000.0:
                    alpha = 2000.0
        else:
            alpha = max(5.0, alpha * 0.95)
            lr = lr * 0.9
        it += 1
    return h_best, c5_best

def two_coord_descent(h0, n, target, n_rounds, n_pairs, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    for rnd in range(n_rounds):
        improved = False
        high_idx = np.where(h > 0.3)[0]
        low_idx = np.where(h < 0.7)[0]
        if len(high_idx) == 0 or len(low_idx) == 0:
            break
        for _ in range(n_pairs):
            i = int(rng.randint(len(high_idx)))
            j = int(rng.randint(len(low_idx)))
            idx_i = high_idx[i]
            idx_j = low_idx[j]
            if idx_i == idx_j:
                continue
            lo = -h[idx_i] + 1e-8
            hi = min(1.0 - h[idx_i], h[idx_j]) - 1e-8
            if hi - lo < 1e-7:
                continue
            def f2d(delta):
                h_tmp = h.copy()
                h_tmp[idx_i] += delta
                h_tmp[idx_j] -= delta
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f2d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-7, 'maxiter': 30})
                if res.fun < c5_cur - 1e-13:
                    h[idx_i] += res.x
                    h[idx_j] -= res.x
                    h = np.clip(h, 0.0, 1.0)
                    h = project_to_sum(h, target)
                    c5_cur = res.fun
                    improved = True
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h.copy()
            except Exception:
                pass
        if not improved:
            break
    return h_best, c5_best

def binary_swap_search(h0, n, target, n_rounds, rng):
    h_bin = np.round(h0).astype(float)
    excess = np.sum(h_bin) - target
    if excess > 0:
        idx1 = np.where(h_bin == 1)[0]
        rng.shuffle(idx1)
        h_bin[idx1[:int(np.ceil(excess))]] = 0
    elif excess < 0:
        idx0 = np.where(h_bin == 0)[0]
        rng.shuffle(idx0)
        h_bin[idx0[:int(np.ceil(-excess))]] = 1
    c5_cur = compute_c5(h_bin)
    c5_best = c5_cur
    h_best = h_bin.copy()
    for rnd in range(n_rounds):
        improved = False
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        n_attempts = min(200, len(ones_idx) * len(zeros_idx))
        for _ in range(n_attempts):
            i = int(rng.randint(len(ones_idx)))
            j = int(rng.randint(len(zeros_idx)))
            idx_i = ones_idx[i]
            idx_j = zeros_idx[j]
            h_bin[idx_i] = 0
            h_bin[idx_j] = 1
            c5_new = compute_c5(h_bin)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                improved = True
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_bin.copy()
                ones_idx = np.where(h_bin == 1)[0]
                zeros_idx = np.where(h_bin == 0)[0]
            else:
                h_bin[idx_i] = 1
                h_bin[idx_j] = 0
        if not improved:
            break
    return h_best, c5_best

def fine_coord_descent(h0, n, target, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    for rnd in range(n_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            lo = max(0.0, h[i] - 0.05)
            hi = min(1.0, h[i] + 0.05)
            if hi - lo < 1e-7:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-8, 'maxiter': 20})
                if res.fun < c5_cur - 1e-14:
                    h[i] = res.x
                    h = project_to_sum(h, target)
                    c5_cur = res.fun
                    improved = True
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h.copy()
            except Exception:
                pass
        if not improved:
            break
    return h_best, c5_best

def multi_start_optimize(h_init, n, target, n_starts, time_limit, t_start, rng):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for s in range(n_starts):
        if time.time() - t_start > time_limit * 0.8:
            break
        pert_scale = 0.03 / (s + 1)
        h_pert = h_init + rng.normal(0, pert_scale, size=n)
        h_pert = project_to_sum(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        h_work = h_pert if c5_pert < best_c5 else best_h.copy()
        phase_budget = time_limit * 0.7 / n_starts
        time_start = time.time()
        h_gd, c5_gd = grad_descent_phase(h_work, n, target, phase_budget * 0.5, time.time())
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
        h_cd, c5_cd = two_coord_descent(h_gd, n, target, 3, 30, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            n = len(best_h)
            target = n / 2.0
            best_h = project_to_sum(best_h, target)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        n = 720
        target = n / 2.0
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    target = n / 2.0
    h_work = best_h.copy()

    if budget_s > 100:
        h_gd, c5_gd = grad_descent_phase(h_work, n, target, 120, time.time())
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
        h_work = h_gd

    if budget_s > 200:
        h_cd, c5_cd = two_coord_descent(h_work, n, target, 8, 40, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        h_work = h_cd

    if budget_s > 300:
        h_bin, c5_bin = binary_swap_search(h_work, n, target, 30, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_work = h_bin

    if budget_s > 400:
        h_refine = project_to_sum(best_h + (best_h - 0.5) * 0.1, target)
        h_cd2, c5_cd2 = two_coord_descent(h_refine, n, target, 8, 50, rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        h_work = h_cd2

    if budget_s > 500:
        h_fine, c5_fine = fine_coord_descent(h_work, n, target, 5, rng)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
        h_work = h_fine

    if budget_s > 700:
        h_mult, c5_mult = multi_start_optimize(best_h, n, target, 3, 200, t_start, rng)
        if c5_mult < best_c5:
            best_c5 = c5_mult
            best_h = h_mult.copy()

    if budget_s > 900:
        h_cd3, c5_cd3 = two_coord_descent(best_h, n, target, 10, 60, rng)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    if budget_s > 1000:
        h_bin2, c5_bin2 = binary_swap_search(best_h, n, target, 50, rng)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    remaining = budget_s - (time.time() - t_start)
    if remaining > 30:
        h_fine2, c5_fine2 = fine_coord_descent(best_h, n, target, int(min(5, max(1, remaining / 10))), rng)
        if c5_fine2 < best_c5:
            best_c5 = c5_fine2
            best_h = h_fine2.copy()

    best_h = project_to_sum(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
