def project_to_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if lag >= 0:
            if lag == 0:
                g[:n] += w * (1.0 - 2.0 * h)
            else:
                g[lag:n] += w * (1.0 - h[:n-lag])
                g[:n-lag] -= w * h[lag:n]
        else:
            k = -lag
            g[:n-k] += w * (1.0 - h[k:n])
            g[k:n] -= w * h[:n-k]
    return g

    best_c5 = compute_c5(h)
    stale = 0

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g


def multi_lag_gradient(h, lags, weights):
    n = len(h)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        g += w * corr_gradient(h, lag)
    return g


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)


def round_to_01(h, target_count):

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f_and_grad(x, alpha=alpha):
        corr = compute_corr(x)
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val + np.log(np.sum(exp_vals)) / alpha
        lags = np.arange(len(corr)) - (n - 1)
        g = np.zeros(n)
        for lam, wv in zip(lags, w):
            g += wv * corr_gradient(x, lam)
        return obj, g

    def f(x):
        return f_and_grad(x)[0]

    def g(x):
        return f_and_grad(x)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def coordinate_descent(h, target_sum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e18)
        g_dec = np.where(can_dec, g, -1e18)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 80:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5
