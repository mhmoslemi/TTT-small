    #         They're adjacent: [0, n-1-k] ends at n-1-k, [n-k, n-1] starts at n-k = (n-1-k)+1.
    #         So together they cover [0, n-1] completely! No gap, no overlap.
    
    # OK so for each lag k > 0:
    #   For i in [0, n-1-k]: grad[i] += w[n-1+k] * (-h[i+k]) + w[n-1-k] * (1-h[i+k])
    #   For i in [n-k, n-1]: grad[i] += w[n-1+k] * (1-h[i-k]) + w[n-1-k] * (-h[i-k])
    
    # And for k = 0 (i = n-1):
    #   corr = sum_i h[i]*(1-h[i]), d/dh[i] = (1-h[i]) - h[i] = 1 - 2h[i]
    
    # Let me just compute this for the alpha values I'll use. For n=240, this is doable.
    
    # Actually, for the optimization, I can use a simpler approach: just compute the gradient
    # numerically or use the analytical formula with vectorization.
    
    # Let me use a clean vectorized approach:
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)
    # k = 0
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    # For each k > 0
    for k in range(1, n):
        wk_pos = w[n-1+k]  # weight for lag +k
        wk_neg = w[n-1-k]  # weight for lag -k
        if wk_pos < 1e-15 and wk_neg < 1e-15:
            continue
        # i in [0, n-1-k]
        idx1 = np.arange(n - k)
        grad[idx1] += (wk_neg * (1.0 - h[idx1 + k]) - wk_pos * h[idx1 + k])
        # i in [n-k, n-1]
        idx2 = np.arange(n - k, n)
        grad[idx2] += (wk_pos * (1.0 - h[idx2 - k]) - wk_neg * h[idx2 - k])
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800, time_limit=None, t_start=None):
    def f(h):
        return smooth_max_val_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_val_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        result = project_to_sum(res.x, target_sum)
        return result
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def block_swap_sa(h, n, target_ones, t_end, rng, max_block=4):
    """SA with block swap moves."""
    h = h.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    no_improve = 0
    min_T = 0.002
    
    while time.time() < t_end:
        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif r < 0.85:
            # Block swap: swap two contiguous blocks of same length
            blen = 1 + rng.randint(max_block)
            if blen >= n // 2:
                blen = 1
            i = rng.randint(n - blen)
            j = rng.randint(n - blen)
            if abs(i - j) < blen:
                continue
            # Check that the blocks have different composition (otherwise swap is useless)
            block_i = h[i:i+blen]
            block_j = h[j:j+blen]
            if np.array_equal(block_i, block_j):
                continue
            h[i:i+blen], h[j:j+blen] = block_j.copy(), block_i.copy()
        else:
            # Multi-point: move up to 3 ones to random zero positions
            num_moves = 1 + rng.randint(3)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < num_moves or len(zeros) < num_moves:
                continue
            chosen_ones = rng.choice(ones, num_moves, replace=False)
            chosen_zeros = rng.choice(zeros, num_moves, replace=False)
            h[chosen_ones] = 0.0
            h[chosen_zeros] = 1.0
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / T) + np.log(rng.uniform()) < 0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            if r < 0.6:
                h[i], h[j] = h[j], h[i]
            elif r < 0.85:
                h[i:i+blen], h[j:j+blen] = block_i.copy(), block_j.copy()
            else:
                h[chosen_ones] = 1.0
                h[chosen_zeros] = 0.0
        
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.025)
            no_improve = 0
        T = max(T * 0.999996, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target_ones]] = 1.0
    return h_bin

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk_pos = w[n - 1 + k]
        wk_neg = w[n - 1 - k]
        if wk_pos < 1e-12 and wk_neg < 1e-12:
            continue
        idx1 = np.arange(n - k)
        grad[idx1] += (wk_neg * (1.0 - h[idx1 + k]) - wk_pos * h[idx1 + k])
        idx2 = np.arange(n - k, n)
