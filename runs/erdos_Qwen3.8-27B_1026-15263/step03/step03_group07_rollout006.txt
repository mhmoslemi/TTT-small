Actually, let me reconsider. The previous solution got 0.380899 with a lot of SLSQP. The issue might be that SLSQP is too slow and we're not taking enough effective steps. Let me focus on:

1. Fast projected gradient (100x more steps per second than SLSQP)
2. Adaptive alpha (start low, increase)
3. Nesterov momentum
4. Periodic re-identification of active lags

Let me also think about what alpha to use. The smooth max with alpha converges to the true max as alpha→∞. But for very large alpha, the gradient is very peaked (only the top lag matters), which can cause oscillation. A good range is alpha = 50-2000.

Let me also add a **spectral method** as an alternative. The problem can be viewed as:
minimize max_k C(h, k) where C(h,k) = sum h[i]*(1-h[i+k])

If we fix the "shape" of h (e.g., as a sum of sinusoids), we can optimize the coefficients. But this is too restrictive.

Another idea: **use the fact that the optimal h should be "as far from 0.5 as possible" at each point** (to minimize the k=0 contribution), while maintaining good "anti-correlation" at other lags. This suggests a structure where h alternates between values near 0 and 1 in a pattern that minimizes the maximum overlap.

For a "block" pattern: h = [1,1,...,1, 0,0,...,0] (n/2 ones, n/2 zeros)
- k=0: overlap = 0
- k=n/2: overlap = n/2 (all the 1s map to the 0s)
- C₅ = (n/2)*(2/n) = 1.0. Terrible.

For a "checkerboard": h = [1,0,1,0,...]
- k=0: overlap = 0
- k=1: overlap = n/2 (each 1 is next to a 0)
- C₅ = 0.5. Better but not great.

For a "golden ratio" spacing of 1s: distribute n/2 ones so that they're as evenly spaced as possible. This minimizes the maximum autocorrelation. But for the "1-minus" correlation, it's different.

I think the best approach is just to do good numerical optimization. Let me write clean, efficient code.

One more thought: the problem size n=720 means we have 720 variables. SLSQP with 720 variables is actually quite slow (each iteration is O(n²) or worse due to the BFGS update). Projected gradient is O(n * |active|) per step, which for |active| = 20-50 is much faster.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(corr, top_k=40, frac=0.98):
    m = np.max(corr)
    thresh = frac * m
    active = np.where(corr >= thresh)[0]
    if len(active) > top_k:
        order = np.argsort(corr[active])[::-1][:top_k]
        active = active[order]
    return active

def grad_active(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = corr[active_idx]
    m = np.max(ac)
    sh = np.clip(alpha * (ac - m), -700.0, 700.0)
    ev = np.exp(sh)
    s = ev.sum()
    w = ev / s
    sm_val = m + np.log(s / len(ev)) / alpha
    g = np.zeros(n)
    omh = 1.0 - h
    for idx in range(len(active_idx)):
        wm = w[idx]
        if wm < 1e-15:
            continue
        k = active_idx[idx] - (n - 1)
        if k >= 0:
            g[k:] += wm * omh[:n-k]
            g[:n-k] -= wm * h[k:]
        else:
            k2 = -k
            g[:n-k2] += wm * omh[k2:]
            g[k2:] -= wm * h[:n-k2]
    return sm_val, g

def projected_grad_opt(h0, n, target_sum, alpha, n_iters, momentum, lr_init, recompute=25):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = get_active_lags(corr, top_k=50, frac=0.97)
    
    stall = 0
    for it in range(n_iters):
        g_val, g = grad_active(h, active, alpha)
        v = momentum * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            h = h_new  # still move but don't update best
            lr *= 0.8
            v = momentum * v * 0.5
            stall += 1
            if stall > 50 or lr < 1e-9:
                break
        
        if (it + 1) % recompute == 0:
            corr = np.correlate(h_best, 1.0 - h_best, mode='full')
            active = get_active_lags(corr, top_k=50, frac=0.97)
    
    return h_best, c5_cur

def slsqp_optim(h0, n, target_sum, alpha, maxiter, active_idx=None):
    n_corr = 2 * n - 1
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        if active_idx is not None:
            v, _ = grad_active(h, active_idx, alpha)
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            sh = np.clip(alpha * (corr - m), -700.0, 700.0)
            ev = np.exp(sh)
            s = ev.sum()
            v = m + np.log(s / n_corr) / alpha
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        if active_idx is not None:
            _, g = grad_active(h, active_idx, alpha)
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            sh = np.clip(alpha * (corr - m), -700.0, 700.0)
            ev = np.exp(sh)
            s = ev.sum()
            w = ev / s
            g = np.zeros(n)
            omh = 1.0 - h
            for m_idx in range(n_corr):
                wm = w[m_idx]
                if wm < 1e-15:
                    continue
                k = m_idx - (n - 1)
                if k >= 0:
                    g[k:] += wm * omh[:n-k]
                    g[:n-k] -= wm * h[k:]
                else:
                    k2 = -k
                    g[:n-k2] += wm * omh[k2:]
                    g[k2:] -= wm * h[:n-k2]
        return g
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_desc(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in [-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01]:
                h_new = h.copy()
                val = h_new[i] + d
                if val < 0 or val > 1:
                    continue
                h_new[i] = val
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-16:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def make_binary_pattern(n, rng):
    """Create a random binary pattern with n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0

def make_sin_pattern(n, freq, amp):
    """Create a sinusoidal pattern centered at 0.5."""

    h = 0.5 + amp * np.sin(x)
    h = np.clip(h, 0.0, 1.0)
    return h
