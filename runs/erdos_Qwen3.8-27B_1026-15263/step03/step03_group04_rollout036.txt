import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_at_k(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        hi1 = n - lag
        grad[:hi1] += (1.0 - h[lag:])
        grad[lag:] -= h[:hi1]
    else:
        d = -lag
        grad[d:] += (1.0 - h[:-d])
        grad[:-d] -= h[d:]
    return grad

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        grad += w[k] * grad_corr_at_k(h, k)
    return obj, grad

def l_bfgs_step(h, n, target_sum, alpha, maxiter):
    def fun(x):
        obj, _ = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 1e6 * s_dev * s_dev
    
    def jac(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + 2e6 * s_dev * np.ones(n)
    
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                                'maxfun': maxiter * 3})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h, target_sum)

def nesterov_subgradient(h0, target_sum, n_iter, gamma):
    """Nesterov accelerated subgradient method."""
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for t in range(1, n_iter + 1):
        # Subgradient at y
        corr = compute_corr(y)
        k_star = np.argmax(corr)
        g = grad_corr_at_k(y, k_star)
        
        # Step
        step = gamma / np.sqrt(t)
        z = y - step * g
        z = project_to_sum(z, target_sum)
        
        # Nesterov update
        alpha_t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y_new = z + (t / (t + alpha_t)) * (z - h)
        y = project_to_sum(y_new, target_sum)
        h = z
        
        # Track best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_coord_descent(h0, target_sum, n_sweeps, rng):

def pair_flip_search(h0, target_sum, rng, n_trials, temperature=0.0):
    """Simulated annealing with pair flips."""
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for i in range(n_trials):
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx1 = rng.choice(ones)
        idx0 = rng.choice(zeros)
        
        h_test = h.copy()
        h_test[idx1] = 0.0
        h_test[idx0] = 1.0
        h_test = project_to_sum(h_test, target_sum)
        c5_test = compute_c5(h_test)
        
        delta = c5_test - c5_cur
        if delta < 0 or (temperature > 0 and rng.random() < np.exp(-delta / temperature)):
            h = h_test
            c5_cur = c5_test
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    
    return best_h, best_c5

def block_swap_search(h0, target_sum, rng, n_trials):

    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)

    for r in range(max_rounds):
        corr = compute_corr(h)
        # Get top few k values

        h_cd2, c5_cd2 = binary_coord_descent(h_bin, ts, 40, rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

        # Try different thresholds for binarizing
        for thr in [0.3, 0.7]:
            if remaining() < 40:
                break
            h_t = (best_h > thr).astype(float)
            h_t = project_to_sum(h_t, ts)
            h_cd3, c5_cd3 = binary_coord_descent(h_t, ts, 30, rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3

    # Phase 5: Pair flip annealing + block swaps
    if remaining() > 150:
        h_b = (best_h > 0.5).astype(float)
        h_b = project_to_sum(h_b, ts)
        for temp in [0.5, 0.1, 0.01, 0.0]:
            if remaining() < 30:
                break
            h_new, c5_new = pair_flip_anneal(h_b, ts, rng, 3000, temp)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_b = (best_h > 0.5).astype(float)
            h_b = project_to_sum(h_b, ts)
        if remaining() > 40:
            h_new, c5_new = block_swap_search(best_h, ts, rng, 5000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 6: Multi-resolution
    if remaining() > 300:
        for n_try in [best_n * 2]:
            if remaining() < 150:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = project_to_sum(h_interp, ts_try)
            h_new, c5_new = smooth_max_continuation(
                h_interp, n_try, ts_try, [10, 100, 1000, 5000, 20000], 200, 120)
            c5_new = compute_c5(h_new)
            if remaining() > 40:
                h_new, c5_new = exact_polish(h_new, n_try, ts_try, 200)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_try
                ts = ts_try

        if remaining() > 100:
            for n_try in [best_n // 2, best_n // 3]:
                if remaining() < 60 or n_try < 40:
                    break
                ts_try = n_try / 2.0
                if abs(ts_try - round(ts_try)) > 1e-9:
                    continue
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n_try, endpoint=False)
                h_down = np.interp(x_new, x_old, best_h, period=2.0)
                h_down = project_to_sum(h_down, ts_try)
                c5_s = compute_c5(h_down)
                h_new, c5_new = smooth_max_continuation(
                    h_down, n_try, ts_try, [50, 500, 5000], 150, 50)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    best_n = n_try
                    ts = ts_try

    # Phase 7: Multi-start exploration
    if remaining() > 120:
        starts = make_starts(best_n, ts, rng)
        for h_s in starts[:25]:
            if remaining() < 25:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_new, c5_new = smooth_max_continuation(
                h_s, best_n, ts, [100, 1000, 10000], 100, 8)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 8: Final intensive polish
    if remaining() > 60:
        h_new, c5_new = exact_polish(best_h, best_n, best_n/2.0, 500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        if remaining() > 30:
            h_new = l_bfgs_optimize(best_h, best_n, best_n/2.0, 50000, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        h_new, c5_new = exact_polish(best_h, best_n, best_n/2.0, 300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 9: One last binary CD
    if remaining() > 30:
        h_cd, c5_cd = binary_coord_descent(best_h, best_n/2.0, 20, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
