from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for idx in range(len(w)):
        wl = w[idx]
        if wl < 1e-14:
            continue
        l = idx - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g, corr

def active_set_grad(h, n_active=20):
    """Gradient using only the top n_active correlation lags with equal weights."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Find lags within a small tolerance of the max
    threshold = m - 0.01 * m  # 1% tolerance
    active = np.where(corr >= threshold)[0]
    if len(active) > n_active:
        # Take the top n_active
        active = np.argsort(corr)[-n_active:]
    if len(active) == 0:
        active = [np.argmax(corr)]
    w = np.full(len(active), 1.0 / len(active))
    g = np.zeros(n)
    for idx, ai in enumerate(active):
        wl = w[idx]
        l = ai - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return g, m, active

def slsqp_optimize(h0, target_sum, alpha, maxiter=200):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _, _ = smooth_max_obj_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g, _ = smooth_max_obj_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def gradient_descent_with_backtrack(h0, target_sum, alpha, n_iters, lr_init=0.1):
    """Smooth-max gradient descent with backtracking line search."""
    n = len(h0)
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        val, g, corr = smooth_max_obj_and_grad(h, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        h_old_c5 = compute_c5(h)
        for bt in range(20):
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new < h_old_c5 * (1 - 1e-4 * step * g_norm):
                h = h_new
                lr = min(step * 1.5, 2.0)
                break
            step *= 0.5
        else:
            # Couldn't find a good step, reduce learning rate
            lr = step
            h = project_to_sum(h - lr * 0.1 * g, target_sum)
    
    return best_h, best_c5

def coordinate_descent(h0, target_sum, max_rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(len(h)):
            best_c5 = c5_cur
            best_val = h[i]
            h_new = h.copy()
            for d in steps:
                h_new[i] = best_val + d
                if h_new[i] < 0 or h_new[i] > 1:
                    continue
                h_new[i] = np.clip(h_new[i], 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_val = h_new[i]
            if best_c5 < c5_cur - 1e-15:
                h = h.copy()
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

**Key insight for improvement**: The previous solution's coordinate descent is probably too slow to be useful (it re-evaluates the full correlation for each coordinate change). Instead, I can use a **fast coordinate descent** that only updates the affected correlations.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for idx in range(len(w)):
        wl = w[idx]
        if wl < 1e-14:
            continue
        l = idx - (n - 1)
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def _obj_func(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_obj_grad(h, alpha)
    return v

def _grad_func(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_obj_grad(h, alpha)
    return g

def _cons_fun(x, ts):
    return np.sum(x) - ts

def _cons_jac(x, ts):
    return np.ones(len(x))

def slsqp_optimize(h0, ts, alpha, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (ts,)}]
    try:
        res = minimize(_obj_func, h0.copy(), jac=_grad_func, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0.copy(), ts)

def grad_descent_backtrack(h0, ts, alpha, n_iters, lr_init=0.1):
    n = len(h0)
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        val, g = smooth_max_obj_grad(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        step = lr
        c5_old = compute_c5(h)
        accepted = False
        for bt in range(30):
            h_try = h - step * g
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, ts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            if c5_try < c5_old:
                h = h_try
                lr = min(step * 2.0, 5.0)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr = step
            if step < 1e-10:
                break
            h = project_to_sum(h - step * g, ts)
    return best_h, best_c5

def coord_descent(h0, ts, max_rounds, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(len(h)):
            best_c5 = c5_cur
            best_val = h[i]
            for d in step_sizes:
                trial = h[i] + d
                if trial < -1e-10 or trial > 1.0 + 1e-10:
                    continue
                h[i] = np.clip(trial, 0.0, 1.0)
                h = project_to_sum(h, ts)
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_val = h[i]
            h[i] = best_val
            if best_val != h[i] or best_c5 < c5_cur:
                # recompute to be safe
                h = project_to_sum(h, ts)
                c5_cur = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, c5_cur
