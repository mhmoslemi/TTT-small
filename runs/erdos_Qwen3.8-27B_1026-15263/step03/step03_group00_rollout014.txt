import numpy as np
from scipy.optimize import minimize
import time

_slqp_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def _slsqp_f(x):
    n = len(x)
    alpha = _slqp_alpha
    corr = compute_corr(x)
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

def _slsqp_g(x):
    n = len(x)
    alpha = _slqp_alpha
    corr = compute_corr(x)
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    w = exp_vals / s
    g = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] > 1e-12:
            lag = int(idx) - (n - 1)
            g += w[idx] * grad_lag(x, lag)
    return g

def slsqp_refine(h, target_sum, alpha, maxiter):
    global _slqp_alpha
    _slqp_alpha = alpha
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_slsqp_f, h.copy(), method='SLSQP', jac=_slsqp_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h.copy(), target_sum)

def _multi_lag_grad(h, K):
    n = len(h)
    corr = compute_corr(h)
    top_k = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for idx in top_k:
        lag = int(idx) - (n - 1)
        g += grad_lag(h, lag)
    g /= float(K)
    return g

def coord_desc_ls(h, target_sum, max_iters, K, t_start, budget_s):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    no_improve = 0
    fracs = [0.15, 0.3, 0.5, 0.7, 0.9]
    for it in range(max_iters):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 30:
            break
        g = _multi_lag_grad(h, K)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-13:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        if max_move < 1e-12:
            break
        best_c5_val = c5
        best_frac = 0.0
        for frac in fracs:
            step = frac * max_move
            if step < 1e-14:
                continue
            h[i_inc] += step
            h[i_dec] -= step
            c5_test = compute_c5(h)
            if c5_test < best_c5_val - 1e-15:
                best_c5_val = c5_test
                best_frac = frac
            h[i_inc] -= step
            h[i_dec] += step
        if best_frac > 0:
            step = best_frac * max_move
            h[i_inc] += step
            h[i_dec] -= step
            c5 = best_c5_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
    return h, c5
