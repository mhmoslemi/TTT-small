import numpy as np
from scipy.optimize import minimize
import time

_g_target_sum = 0.0
_g_slsqp_n = 0
_g_slsqp_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_s(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def multi_lag_grad(h, n_top):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_top]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        g += corr_gradient_s(h, lag)
    return g

def multi_lag_descent(h_in, target_sum, max_iters, n_top, momentum_beta):
    n = len(h_in)
    h = h_in.copy()
    c5 = compute_c5(h)
    v = np.zeros(n)
    stale = 0
    step_scale = 1.0
    for it in range(max_iters):
        g = multi_lag_grad(h, n_top)
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        v = momentum_beta * v + (1.0 - momentum_beta) * (-g / g_norm)
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-14:
            break
        v_dir = v / v_norm
        found = False
        steps = [0.15 * step_scale, 0.1 * step_scale, 0.05 * step_scale,
                 0.02 * step_scale, 0.01 * step_scale, 0.005 * step_scale,
                 0.002 * step_scale, 0.001 * step_scale]
        for step in steps:
            if step < 1e-10:
                continue
            h_trial = h + step * v_dir
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_to_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5 - 1e-15:
                h = h_trial
                c5 = c5_trial
                stale = 0
                found = True
                step_scale = min(step_scale * 1.05, 1.5)
                break
        if not found:
            stale += 1
            step_scale = max(step_scale * 0.8, 0.05)
            if stale > 400:
                break
    return h, c5

def fine_coord_descent(h_in, target_sum, max_iters, step_frac):
    n = len(h_in)
    h = h_in.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_s(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e10)
        g_dec = np.where(can_dec, g, -1e10)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if g[i_inc] >= g[i_dec] - 1e-13:
            stale += 1
            if stale > 100:
                break
            continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            stale = 0
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 200:
                break
    return h, c5

def interp_to_n(h_in, n_new):
    n_old = len(h_in)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_in, period=2.0)

def _slq_obj_grad(x):
    n = _g_slsqp_n
    alpha = _g_slsqp_alpha
    x = np.clip(x, 0, 1)
    corr = compute_corr(x)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    g = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] > 1e-15:
            lag = int(idx) - (n - 1)
            g += w[idx] * corr_gradient_s(x, lag)
    return obj, g

def _slq_con_fun(x):
    return np.sum(x) - _g_target_sum

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    global _g_slsqp_n, _g_slsqp_alpha, _g_target_sum
    _g_slsqp_n = n
    _g_slsqp_alpha = alpha
    _g_target_sum = target_sum
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slq_obj_grad, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def fourier_to_h(coeffs, n):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    n_harm = len(coeffs) // 2
    for k in range(1, n_harm + 1):
        a = coeffs[2 * (k - 1)]
        b = coeffs[2 * (k - 1) + 1]
        h += a * np.cos(np.pi * k * x)
        h += b * np.sin(np.pi * k * x)
    return np.clip(h, 0, 1)

def h_to_fourier_coeffs(h, n_harm):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    coeffs = np.zeros(2 * n_harm)
    for k in range(1, n_harm + 1):
        coeffs[2 * (k - 1)] = (2.0 / n) * np.sum(h * np.cos(np.pi * k * x))
        coeffs[2 * (k - 1) + 1] = (2.0 / n) * np.sum(h * np.sin(np.pi * k * x))
    return coeffs

def fourier_optimize(h_in, n, n_harm, target_sum, max_iters, lr_init):
    coeffs = h_to_fourier_coeffs(h_in, n_harm)
    h = fourier_to_h(coeffs, n)
    h = project_to_sum(h, target_sum)
    c5 = compute_c5(h)
    d = 2 * n_harm
    lr = lr_init
    stale = 0
    for it in range(max_iters):
        grad = np.zeros(d)
        eps = 1e-4
        for j in range(d):
            c_p = coeffs.copy()
            c_p[j] += eps
            c_m = coeffs.copy()
            c_m[j] -= eps
            hp = project_to_sum(fourier_to_h(c_p, n), target_sum)
            hm = project_to_sum(fourier_to_h(c_m, n), target_sum)
            grad[j] = (compute_c5(hp) - compute_c5(hm)) / (2.0 * eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = lr / gnorm
        coeffs_new = coeffs - step * grad
        coeffs_new = np.clip(coeffs_new, -1.5, 1.5)
        h_new = project_to_sum(fourier_to_h(coeffs_new, n), target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            coeffs = coeffs_new
            h = h_new
            c5 = c5_new
            stale = 0
        else:
            stale += 1
            lr *= 0.85
            if stale > 60:
                break
    return coeffs, h, c5

def block_optimize(h_in, n, n_blocks, target_sum, max_iters, lr_init):
    block_size = n // n_blocks
    blocks = np.zeros(n_blocks)
    for k in range(n_blocks):
        s = k * block_size
        e = s + block_size if k < n_blocks - 1 else n
        blocks[k] = np.mean(h_in[s:e])
    blocks = np.clip(blocks, 0, 1)
    h = np.zeros(n)
    for k in range(n_blocks):
        s = k * block_size
        e = s + block_size if k < n_blocks - 1 else n
        h[s:e] = blocks[k]
    h = project_to_sum(h, target_sum)
    c5 = compute_c5(h)
    lr = lr_init
    stale = 0
    for it in range(max_iters):
        grad = np.zeros(n_blocks)
        eps = 1e-4
        for j in range(n_blocks):
            bp = blocks.copy()
            bp[j] = min(bp[j] + eps, 1.0)
            bm = blocks.copy()
            bm[j] = max(bm[j] - eps, 0.0)
            hp = np.zeros(n)
            hm = np.zeros(n)
            for k in range(n_blocks):
                s = k * block_size
                e = s + block_size if k < n_blocks - 1 else n
                hp[s:e] = bp[k]
                hm[s:e] = bm[k]
            hp = project_to_sum(hp, target_sum)
            hm = project_to_sum(hm, target_sum)
            grad[j] = (compute_c5(hp) - compute_c5(hm)) / (2.0 * eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = lr / gnorm
        blocks_new = np.clip(blocks - step * grad, 0, 1)
        h_new = np.zeros(n)
        for k in range(n_blocks):
            s = k * block_size
            e = s + block_size if k < n_blocks - 1 else n
            h_new[s:e] = blocks_new[k]
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            blocks = blocks_new
            h = h_new
            c5 = c5_new
            stale = 0
        else:
            stale += 1
            lr *= 0.85
            if stale > 50:
                break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():

    rng = np.random.RandomState(seed)

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
