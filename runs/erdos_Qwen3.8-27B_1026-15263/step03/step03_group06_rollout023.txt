import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(z, target):
    n = len(z)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(z - lam, 0.0, 1.0)

def subgradient_at_lag(h, m):
    n = len(h)
    shift = m - n + 1
    g = np.zeros(n)
    idx = np.arange(n)
    k1 = idx + shift
    mask1 = (k1 >= 0) & (k1 < n)
    g[mask1] += (1.0 - h[k1[mask1]])
    k2 = idx - shift
    mask2 = (k2 >= 0) & (k2 < n)
    g[mask2] -= h[k2[mask2]]
    return g

def fast_subgradient_opt(h0, n, target_sum, n_iters, alpha, top_k, lr_init, mu, time_limit):
    h = h0.copy().astype(float)
    h = project_box_sum(h, target_sum)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t_start = time.time()
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(corr.max())
        c5 = cmax * (2.0 / n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Sparse smooth max
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        
        w_raw = alpha * (corr[top_idx] - cmax)
        w_raw = np.clip(w_raw, -50.0, 50.0)
        w = np.exp(w_raw)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] > 1e-8:
                g += w[i] * subgradient_at_lag(h, int(top_idx[i]))
        
        # Momentum
        v = mu * v - lr * g
        h_new = project_box_sum(h + v, target_sum)
        
        # Adaptive lr
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        
        if c5_new > c5 * 1.01 + 0.0001:
            lr *= 0.7
            v *= 0.3
            # Don't accept
        else:
            if c5_new < best_c5 * 0.999:
                lr = min(lr * 1.05, 0.5)
            else:
                lr *= 0.92
            h = h_new
    
    return best_h, best_c5

        for _ in range(n // 2):
            if time.time() - t_start > time_limit:

            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            
            def f2d(params):
                h[i] = params[0]
                h[j] = params[1]
                h = project_box_sum(h, target_sum)
                return compute_c5(h)
            
            # Try a small neighborhood
            best_params = (hi, hj)
            best_val = best_c5
            for di in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                for dj in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                    h_test = h.copy()
                    h_test[i] = np.clip(hi + di, 0, 1)
                    h_test[j] = np.clip(hj + dj, 0, 1)
                    h_test = project_box_sum(h_test, target_sum)
                    c5_test = compute_c5(h_test)
                    if c5_test < best_val - 1e-12:
                        best_val = c5_test
                        best_params = (h_test[i], h_test[j])
            
            if best_val < best_c5 - 1e-12:
                h[:] = project_box_sum(h, target_sum)  # reset
                h[i] = best_params[0]
                h[j] = best_params[1]
                h = project_box_sum(h, target_sum)
                best_c5 = best_val
                improved = True
        
        if not improved:
            break
    
    return h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(z, target):
    n = len(z)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(z - lam, 0.0, 1.0)

def subgradient_at_lag(h, m):
    n = len(h)
    shift = m - n + 1
    g = np.zeros(n)
    idx = np.arange(n)
    k1 = idx + shift
    m1 = (k1 >= 0) & (k1 < n)
    g[m1] += 1.0 - h[k1[m1]]
    k2 = idx - shift
    m2 = (k2 >= 0) & (k2 < n)
    g[m2] -= h[k2[m2]]
    return g

def subgradient_optimize(h0, target_sum, n_iters, alpha, top_k, lr_init, mu, time_limit):
    n = len(h0)
    h = project_box_sum(h0.astype(float), target_sum)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    stagnation = 0
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr))
        
        if cmax < best_c5 * n / 2.0:
            best_c5 = cmax * 2.0 / n
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Sparse smooth max gradient
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        w_raw = np.clip(alpha * (corr[top_idx] - cmax), -50.0, 50.0)
        w = np.exp(w_raw)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w /= ws
        
        g = np.zeros(n)
        for i in range(top_k):
            if w[i] > 1e-8:
                g += w[i] * subgradient_at_lag(h, int(top_idx[i]))
        
        v = mu * v - lr * g
        h_new = project_box_sum(h + v, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new > best_c5 * 1.02 + 0.0002:
            lr *= 0.5
            v *= 0.5
        else:
            if c5_new < best_c5:
                lr = min(lr * 1.03, 0.3)
            else:
                lr *= 0.95
            h = h_new
        
        # Occasionally increase alpha for precision
        if it % 10000 == 0:
            alpha *= 1.5
    
    return best_h, best_c5

def coord_desc_line_search(h0, target_sum, n_rounds, rng, time_limit):
    n = len(h0)
    h = project_box_sum(h0.astype(float), target_sum)
    best_c5 = compute_c5(h)

            if time.time() - t0 > time_limit:

            lo = max(0.0, hi - 0.1)
            hi_bound = min(1.0, hi + 0.1)
            
            def f1d(v):
                h[i] = v
                h = project_box_sum(h, target_sum)
                c5 = compute_c5(h)
                h[i] = hi  # restore
                return c5

                                     options={'xatol': 1e-6, 'maxiter': 50})

                    h = project_box_sum(h, target_sum)
                    best_c5 = res.fun
                    improved = True
            except:
                pass
        if not improved:

    return g

def sg_optimize(h0, target_sum, n_iters, alpha, top_k, lr_init, mu, time_limit):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()

    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr))
        c5 = cmax * (2.0 / n)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        w_raw = np.clip(alpha * (corr[top_idx] - cmax), -50.0, 50.0)
        w = np.exp(w_raw)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / ws

        g = np.zeros(n)
        for i in range(top_k):
            if w[i] > 1e-8:
                g += w[i] * subgradient_at_lag(h, int(top_idx[i]))

        v = mu * v - lr * g
        h_new = project_box_sum(h + v, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new > best_c5 * 1.005 + 0.0001:
            lr *= 0.5
            v *= 0.5
        else:
            if c5_new < best_c5:
                lr = min(lr * 1.02, 0.1)
            else:
                lr *= 0.97
            h = h_new

    return best_h, best_c5

def coord_desc(h0, target_sum, n_rounds, rng, time_limit):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(h)
    t0 = time.time()

    for rnd in range(n_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() - t0 > time_limit:
                break
            hi_val = h[i]
            lo = max(0.0, hi_val - 0.05)
            hi_b = min(1.0, hi_val + 0.05)
            h_save = h.copy()
            best_f = best_c5
            best_new_h = None

            for v in np.linspace(lo, hi_b, 9):
                h[i] = v
                h_cand = project_box_sum(h, target_sum)
                c5 = compute_c5(h_cand)
                if c5 < best_f:
                    best_f = c5
                    best_new_h = h_cand.copy()
            h = h_save.copy()

            if best_new_h is not None:
                h = best_new_h
                best_c5 = best_f
                improved = True

        if not improved:
            break

    return h, best_c5

def coord_desc_fine(h0, target_sum, n_rounds, rng, time_limit):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(h)
    t0 = time.time()

    for rnd in range(n_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() - t0 > time_limit:
                break
            hi_val = h[i]
            lo = max(0.0, hi_val - 0.02)
            hi_b = min(1.0, hi_val + 0.02)

            def f1d(v):
                h[i] = v
                h_cand = project_box_sum(h, target_sum)
                val = compute_c5(h_cand)
                h[i] = hi_val
                return val

            try:
                res = minimize_scalar(f1d, bounds=(lo, hi_b), method='bounded',
                                     options={'xatol': 1e-7, 'maxiter': 30})
                if res.fun < best_c5 - 1e-14:
                    h[i] = res.x
                    h = project_box_sum(h, target_sum)
                    best_c5 = res.fun
                    improved = True
            except Exception:
                pass

        if not improved:
            break

    return h, best_c5

def perturbation_restart(h0, target_sum, rng, n_restarts, time_limit):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()

    for r in range(n_restarts):
        if time.time() - t0 > time_limit:
            break
        scale = 0.03 / (r + 1)
        h_pert = h0 + rng.normal(0, scale, size=n)
        h_pert = project_box_sum(h_pert, target_sum)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h0 = h_pert.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def tleft():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    best_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)

    if best_h is None:
        n = 720
        best_h = np.full(n, 0.5)
        x = np.linspace(0, 2, n, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x)

    n = len(best_h)
    target_sum = n / 2.0
    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast subgradient, moderate alpha
    time_p1 = min(350, tleft() * 0.3)
    h1, c5_1 = sg_optimize(best_h, target_sum, 1000000, 100.0, 20, 0.005, 0.9, time_p1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Higher alpha, more focused
    time_p2 = min(300, tleft() * 0.25)
    h2, c5_2 = sg_optimize(best_h, target_sum, 800000, 500.0, 10, 0.003, 0.85, time_p2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: Coarse coordinate descent
    time_p3 = min(150, tleft() * 0.2)
    h3, c5_3 = coord_desc(best_h, target_sum, 3, rng, time_p3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Phase 4: Very high alpha subgradient
    if tleft() > 100:
        time_p4 = min(200, tleft() * 0.3)
        h4, c5_4 = sg_optimize(best_h, target_sum, 600000, 2000.0, 8, 0.002, 0.85, time_p4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Perturbation to escape local optima
    if tleft() > 80:
        time_p5 = min(60, tleft() * 0.15)
        h5, c5_5 = perturbation_restart(best_h, target_sum, rng, 8, time_p5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5
        if tleft() > 60:
            time_p5b = min(80, tleft() * 0.3)
            h5b, c5_5b = sg_optimize(best_h, target_sum, 400000, 1000.0, 10, 0.003, 0.88, time_p5b)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b

    # Phase 6: Fine coordinate descent
    if tleft() > 40:
        time_p6 = min(100, tleft() * 0.4)
        h6, c5_6 = coord_desc_fine(best_h, target_sum, 3, rng, time_p6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: Final subgradient polish
    if tleft() > 30:
        time_p7 = min(60, tleft() - 10)
        h7, c5_7 = sg_optimize(best_h, target_sum, 300000, 5000.0, 5, 0.001, 0.8, time_p7)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    # Phase 8: Ultra-fine coordinate descent
    if tleft() > 20:
        time_p8 = max(10, tleft() - 5)
        h8, c5_8 = coord_desc_fine(best_h, target_sum, 2, rng, time_p8)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8

    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
