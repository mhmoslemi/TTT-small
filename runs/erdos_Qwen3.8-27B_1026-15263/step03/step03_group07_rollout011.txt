import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s / len(exp_vals)) / alpha, exp_vals / s

def smooth_max_value_active(corr, active_idx, alpha):
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s / len(exp_vals)) / alpha, exp_vals / s

def vectorized_gradient(h, weights, active_idx, N):
    """Compute gradient of smooth max w.r.t. h using active lag weights."""
    grad = np.zeros(N)
    for m, w in zip(active_idx, weights):
        if w < 1e-15:
            continue
        k = m - N + 1
        if k >= 0:
            if N - 1 - k >= 0:
                grad[:N-k] += w * (1.0 - h[k:N])
            if N - k > 0:
                grad[k:] -= w * h[:N-k]
        else:
            k2 = -k
            if N - k2 > 0:
                grad[k2:] += w * (1.0 - h[:N-k2])
            grad[:N-k2] -= w * h[k2:N]
    return grad

def slsqp_smooth_max(h0, N, target_sum, alpha, maxiter, active_idx=None):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h)
        if active_idx is not None and len(active_idx) < N:
            v, _ = smooth_max_value_active(corr, active_idx, alpha)
        else:
            v, _ = smooth_max_value(corr, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h)
        if active_idx is not None and len(active_idx) < N:
            _, w = smooth_max_value_active(corr, active_idx, alpha)
            g = vectorized_gradient(h, w, active_idx, N)
        else:
            _, w = smooth_max_value(corr, alpha)
            g = vectorized_gradient(h, w, np.arange(len(corr)), N)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(N)}]
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def projected_subgrad(h0, N, target_sum, n_iters, lr_init, alpha, active_threshold=0.95):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        active = np.where(corr >= active_threshold * max_corr)[0]
        _, w = smooth_max_value_active(corr, active, alpha)
        g = vectorized_gradient(h, w, active, N)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h_best, c5_cur

def coordinate_descent(h0, N, target_sum, max_rounds, rng, delta_sizes=None):

    c5_cur = compute_c5(h)
    if delta_sizes is None:
        delta_sizes = [0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01]
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(N):
            best_val = c5_cur
            best_d = 0.0
            for d in delta_sizes:
                h_new = h.copy()
                new_val = h[i] + d
                if 0 < new_val < 1:  # skip if already at boundary
                    h_new[i] = new_val
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_val - 1e-15:
                        best_val = c5_new
                        best_d = d

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_params(corr, alpha, active_idx=None):
    if active_idx is not None and len(active_idx) < len(corr):
        ac = corr[active_idx]
        m = np.max(ac)
        shifted = alpha * (ac - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        val = m + np.log(s / len(exp_vals)) / alpha
        w = exp_vals / s
        return val, w, ac
    else:
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        val = m + np.log(s / len(exp_vals)) / alpha
        w = exp_vals / s
        return val, w, corr


def vec_grad(h, weights, active_idx, N):
    grad = np.zeros(N)
    for m, w in zip(active_idx, weights):
        if w < 1e-15:
            continue
        k = m - N + 1
        if k >= 0:
            nk = N - k
            if nk > 0:
                grad[:nk] += w * (1.0 - h[k:N])
                grad[k:] -= w * h[:nk]
        else:
            k2 = -k
            nk = N - k2
            if nk > 0:
                grad[k2:] += w * (1.0 - h[:nk])
                grad[:nk] -= w * h[k2:N]
    return grad


def slsqp_opt(h0, N, ts, alpha, maxiter, active_idx=None):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h)
        v, _, _ = smooth_max_params(corr, alpha, active_idx)
        return v

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h)
        _, w, _ = smooth_max_params(corr, alpha, active_idx)
        if active_idx is not None:
            idx = active_idx
        else:
            idx = np.arange(len(corr))
        return vec_grad(h, w, idx, N)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(N)}]
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)


def subgrad_opt(h0, N, ts, n_iters, lr_init, alpha, thresh=0.95):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        corr = compute_corr(h)
        mx = np.max(corr)
        active = np.where(corr >= thresh * mx)[0]
        _, w, _ = smooth_max_params(corr, alpha, active)
        g = vec_grad(h, w, active, N)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.5)
            stall = 0
        else:
            lr *= 0.75
            stall += 1
            if stall > 200 or lr < 1e-8:
                break
    return h_best, c5_cur

    if fine:

    else:
        dvals = [0.02, 0.01, 0.005, 0.002, -0.002, -0.005, -0.01, -0.02]
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(N):
            best_v = c5_cur
            best_d = 0.0
            base = h[i]

                h[i] = nv
                h = project_to_sum(h, ts)

                c5_cur = best_v
                improved = True
            else:
                h[i] = base
                h = project_to_sum(h, ts)
        if not improved:
            break
    return h, c5_cur

        if tleft() > 15:
            h_pw, c5_pw = pairwise_refine(h_f, n_f, ts_f, min(800, int(tleft() * 4)), rng)
            if c5_pw < best_c5:
                best_c5 = c5_pw
                best_h = h_pw.copy()
                best_n = n_f

    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
