def adam_smooth(h0, n, target_sum, alphas, n_iters_per_alpha, lr_init, t_start, time_limit):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        m_vec = np.zeros(n)
        v_vec = np.zeros(n)
        t = 0
        for it in range(n_iters_per_alpha):
            if time.time() - t_start > time_limit:
                break
            t += 1
            _, g = smooth_max_grad(h, alpha)
            # Normalize gradient
            g_norm = np.linalg.norm(g)
            if g_norm > 0:
                g = g / g_norm
            m_vec = 0.9 * m_vec + 0.1 * g
            v_vec = 0.999 * v_vec + 0.001 * g * g
            m_hat = m_vec / (1 - 0.9**t)
            v_hat = v_vec / (1 - 0.999**t)
            # Cosine annealing learning rate
            lr = lr_init * 0.5 * (1 + np.cos(np.pi * it / n_iters_per_alpha))

def topk_subgrad(h0, n, target_sum, n_iters, lr_init, top_k, t_start, time_limit):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    momentum = np.zeros(n)
    beta_m = 0.9

        if time.time() - t_start > time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full')

        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            lag = m - (n - 1)
            jj1 = j_arr + lag
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - lag
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        g /= float(top_k)
        momentum = 0.9 * momentum + 0.1 * g
        lr = lr_init / (1.0 + 0.03 * it)
        h_new = h - lr * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        h = h_new
