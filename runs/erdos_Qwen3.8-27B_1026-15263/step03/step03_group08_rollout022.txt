import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum, max_iter=500):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        # Find elements not at bounds
        can_inc = h < 1.0 - 1e-12
        can_dec = h > 1e-12
        if diff > 0:
            mask = can_inc
            if np.sum(mask) == 0:
                break
            adj = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.9)
            h[mask] += adj
        else:
            mask = can_dec
            if np.sum(mask) == 0:
                break
            adj = max(diff / np.sum(mask), -(h[mask]).min() * 0.9)
            h[mask] += adj
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -800.0, 800.0)
    log_s = np.log(np.sum(np.exp(shifted)))
    return m + log_s / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -800.0, 800.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    g = np.zeros(n)
    for l in range(-(n-1), n):
        idx = l + n - 1
        wl = w[idx]
        if wl < 1e-15:
            continue
        if l >= 0:
            # d/dh[j] for j in [l, n-1]: +(1-h[j-l])
            # d/dh[j] for j in [0, n-1-l]: -h[j+l]
            if l == 0:
                g += wl * (1.0 - 2.0 * h)
            else:
                g[l:] += wl * (1.0 - h[:-l])
                g[:n-l] -= wl * h[l:]
        else:
            p = -l
            # d/dh[j] for j in [0, n-1-p]: +(1-h[j+p])
            # d/dh[j] for j in [p, n-1]: -h[j-p]
            g[:n-p] += wl * (1.0 - h[p:])
            g[p:] -= wl * h[:-p]
    return g

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, time_limit):
    t_start = time.time()
    def obj(x):
        if time.time() - t_start > time_limit:
            return smooth_max_val(x, alpha) + 100 * (np.sum(x) - target_sum)**2
        return smooth_max_val(x, alpha) + 100 * (np.sum(x) - target_sum)**2
    def grad(x):
        g = smooth_max_grad(x, alpha)
        g += 200 * (np.sum(x) - target_sum) / n
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def penalty_target_descent(h0, n, target_sum, C_target, n_iters, lr):
    dx = 2.0 / n
    C_corr = C_target / dx
    h = h0.copy()
    best_h = h.copy()

    corr_full = 2 * n - 1

        corr = np.correlate(h, 1.0 - h, mode='full')

        top_idx = np.argsort(corr)[-top_k:]

            excess = corr[k_idx] - C_corr
            if excess <= 0:

            l = k_idx - (n - 1)
            wl = excess / C_corr  # normalize
            if l >= 0:

                    g += 2.0 * wl * (1.0 - 2.0 * h)
                else:

                    g[:n-l] -= 2.0 * wl * h[l:]
            else:
                p = -l
                g[:n-p] += 2.0 * wl * (1.0 - h[p:])
                g[p:] -= 2.0 * wl * h[:-p]
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-16:
            break
        g_norm = g / norm_g
        # Backtracking line search

            best_h = h_new.copy()
        if c5_new < compute_c5(h):
            h = h_new
        else:
            # Try small random perturbation to escape

def coordinate_descent_refine(h0, n, target_sum, max_rounds, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        # Randomize order for diversity
        order = np.random.permutation(n)
        for i in order:
            best_c5 = c5_cur
            best_val = h[i]
            for d in step_sizes:
                if abs(d) < 1e-15:
                    continue
                h_new = h.copy()
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                h_new[i] = new_val
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    best_val = new_val
            if best_val != h[i]:
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum, max_iter=500):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            headroom = (1.0 - h[mask]).min()
            adj = min(diff / cnt, headroom * 0.99)
            h[mask] += adj
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            floorroom = h[mask].min()
            adj = max(diff / cnt, -floorroom * 0.99)
            h[mask] += adj
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -800.0, 800.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        idx = l + n - 1
        wl = w[idx]
        if wl < 1e-15:
            continue
        if l >= 0:
            if l == 0:
                g += wl * (1.0 - 2.0 * h)
            else:
                g[l:] += wl * (1.0 - h[:-l])
                g[:n - l] -= wl * h[l:]
        else:
            p = -l
            g[:n - p] += wl * (1.0 - h[p:])
            g[p:] -= wl * h[:-p]
    return g

def smooth_max_val(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -800.0, 800.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lbfgsb_run(h0, n, target_sum, alpha, maxiter, time_limit):
    t0 = time.time()
    pen_weight = 50.0

    def obj(x):
        if time.time() - t0 > time_limit:
            return smooth_max_val(x, alpha) + pen_weight * (np.sum(x) - target_sum) ** 2
        return smooth_max_val(x, alpha) + pen_weight * (np.sum(x) - target_sum) ** 2

    def grad(x):
        g = smooth_max_grad(x, alpha)
        g += 2.0 * pen_weight * (np.sum(x) - target_sum) / n
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 40, 'maxfun': maxiter * 5})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def targeted_descent(h0, n, target_sum, C_target, n_iters, lr_init):
    dx = 2.0 / n
    C_corr_target = C_target / dx
    h = np.array(h0, dtype=float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    rng_local = np.random.RandomState(123)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(25, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k_idx in top_idx:
            excess = corr[k_idx] - C_corr_target
            if excess <= 1e-12:
                continue
            wl = excess
            l = k_idx - (n - 1)
            if l >= 0:
                if l == 0:
                    g += wl * (1.0 - 2.0 * h)
                else:
                    g[l:] += wl * (1.0 - h[:-l])
                    g[:n - l] -= wl * h[l:]
            else:
                p = -l
                g[:n - p] += wl * (1.0 - h[p:])
                g[p:] -= wl * h[:-p]
        ng = np.linalg.norm(g)
        if ng < 1e-15:
            break
        g = g / ng
        c5_cur = compute_c5(h)
        step = lr
        accepted = False
        for _ in range(15):
            h_try = project_to_sum(h - step * g, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-16:
                h = h_try
                accepted = True
                break
            step *= 0.6
        if accepted:
            if step > lr:
                lr = min(step * 1.1, 1.0)
            else:
                lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                h = project_to_sum(h + 0.005 * rng_local.randn(n), target_sum)
                lr = 0.02
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, step_sizes, rng):
    h = np.array(h0, dtype=float)
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            best_c = c5_cur
            best_v = h[i]
            for d in step_sizes:
                nv = h[i] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h_tmp = h.copy()
                h_tmp[i] = nv
                h_tmp = project_to_sum(h_tmp, target_sum)
                c_tmp = compute_c5(h_tmp)
                if c_tmp < best_c - 1e-16:
                    best_c = c_tmp
                    best_v = nv
            if best_v != h[i]:
                h[i] = best_v
                h = project_to_sum(h, target_sum)
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def gen_starts(n, rng, parent_h, parent_n):
    starts = []
    ts = n / 2.0
    x = np.linspace(0, 2, n, endpoint=False)

    if parent_n == n and parent_h is not None:
        starts.append(parent_h.copy())
    elif parent_h is not None:
        x_old = np.linspace(0, 2, parent_n, endpoint=False)
        h_i = np.interp(x, x_old, parent_h)
        starts.append(project_to_sum(h_i, ts))

    for delta in [0.01, 0.03, 0.06, 0.1]:
        hp = parent_h + rng.uniform(-delta, delta, parent_n)
        hp = project_to_sum(hp, parent_n / 2.0)
        if n != parent_n:
            x_old = np.linspace(0, 2, parent_n, endpoint=False)
            hp = np.interp(x, x_old, hp)
        starts.append(project_to_sum(hp, ts))

    for freq in range(1, 7):
        for amp in [0.35, 0.5, 0.65]:
            for ph in [0.0, np.pi / 2, np.pi]:
                hs = 0.5 + amp * np.sin(np.pi * freq * x + ph)
                starts.append(project_to_sum(np.clip(hs, 0, 1), ts))

    for f1 in [1, 2, 3]:
        for f2 in [4, 5, 7]:
            hs = 0.5 + 0.25 * np.sin(np.pi * f1 * x) + 0.2 * np.sin(np.pi * f2 * x + 0.5)
            starts.append(project_to_sum(np.clip(hs, 0, 1), ts))

    for period in [4, 6, 8, 12]:
        pat = np.zeros(n)
        fill = period // 2
        for i in range(n):
            pat[i] = 1.0 if (i % period) < fill else 0.0
        s = np.sum(pat)
        if s > 0:
            pat = pat * (ts / s)
        starts.append(project_to_sum(np.clip(pat, 0, 1), ts))

    for _ in range(4):
        coeffs = rng.uniform(-0.3, 0.3, 5)
        hs = np.full(n, 0.5)
        for k in range(5):
            hs += coeffs[k] * np.sin(np.pi * (k + 1) * x + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(np.clip(hs, 0, 1), ts))

    starts.sort(key=lambda h: compute_c5(h))
    return starts[:10]
