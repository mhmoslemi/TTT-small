def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-13:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corrs(corr, alpha, center_bias=None):
    """Smooth max of correlation values with optional bias toward center shifts."""
    if center_bias is not None:
        n = len(corr)
        center = n // 2
        # Shift index from 0 to n-1 maps to shift -(n-1) to (n-1)
        shifts = np.arange(n) - center  # -center to center
        bias = center_bias * np.exp(-np.abs(shifts) / center)
        corr = corr - bias
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals)) + m_val
    obj = log_sum / alpha
    return obj

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target_ones:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_c = len(corr)
    center = n - 1

    term2 = valid2 * (-h[j2c])
    grad = np.sum(w[None, :] * (term1 + term2), axis=1)
    
    return obj, grad

def lbfgs_optimize(h0, target_sum, alphas, max_iter_each=300):

    for alpha in alphas:

            return smooth_max_obj_grad(h, alpha)[0]
        def fprime(h):
            return smooth_max_obj_grad(h, alpha)[1]
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, h.copy(), method='L-BFGS-B', jac=fprime,
                         bounds=bounds, options={'maxiter': max_iter_each, 'ftol': 1e-16})
            h = project_to_sum(res.x, target_sum)
        except Exception:
            pass
    return h

    term2 = valid2 * (-h[j2c])

    best_c5 = c5_curr

        _, grad = smooth_max_obj_grad(h, alpha)

        # Prefer flips that decrease the objective
        # grad[i] > 0 means increasing h[i] increases objective
        # So we want to decrease h[i] where grad[i] > 0 (flip 1->0)
        # And increase h[i] where grad[i] < 0 (flip 0->1)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score each candidate
        # For flipping h[i] from 1 to 0: change in obj ≈ -grad[i] * 1 (decrease h by 1)
        # For flipping h[j] from 0 to 1: change in obj ≈ +grad[j] * 1 (increase h by 1)
        # Total change ≈ grad[j] - grad[i]
        # We want this to be negative, so we want grad[j] < grad[i]
        
        # Use Gumbel sampling for diversity
        scores_ones = -grad[ones] + T * rng.gumbel(size=len(ones))
        scores_zeros = grad[zeros] + T * rng.gumbel(size=len(zeros))
        
        i_idx = ones[np.argmax(scores_ones)]
        j_idx = zeros[np.argmax(scores_zeros)]
        
        h[i_idx] = 0.0
        h[j_idx] = 1.0
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
