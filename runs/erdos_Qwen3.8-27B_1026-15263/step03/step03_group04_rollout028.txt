def subgradient_polish(h, n, target_sum, max_iters=5000):

    h = h.copy()
    dx = 2.0 / n
    c5_cur = compute_c5(h)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * dx
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)

        grad = np.zeros(n)

        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1+shift:hi1+shift])

def pair_coord_descent(h, n, target_sum, max_sweeps=100, rng=None):
    """
    Coordinate descent on pairs of variables, preserving the sum constraint.
    """
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    
    all_shifts = np.arange(-(n-1), n)  # shifts from -(n-1) to n-1
    
    for sweep in range(max_sweeps):
        improved = False
        # Randomize the order of coordinates
        order = rng.permutation(n) if rng is not None else np.arange(n)
        
        for i in order:
            # Try pairing with a few candidates
            best_t = h[i]
            best_c5 = c5_cur
            best_j = -1
            best_s = 0
            
            # Candidate partners: i+1, i-1, random, i+n/2
            candidates = [(i+1) % n, (i-1) % n, (i + n//2) % n]
            if rng is not None:
                candidates.append(rng.randint(n))
            
            for j in candidates:
                if j == i:
                    continue
                s = h[i] + h[j]
                
                # Compute alpha_i and alpha_j for all shifts
                # alpha_i[k] = grad of c_{shift_k} w.r.t. h[i]
                # = (1-h[i+shift])·I(0≤i+shift<n) - h[i-shift]·I(0≤i-shift<n)
                
                alpha_i = np.zeros(2*n-1)
                alpha_j = np.zeros(2*n-1)
                
                for idx, sh in enumerate(all_shifts):
                    # For coordinate i:
                    if 0 <= i + sh < n:
                        alpha_i[idx] += (1.0 - h[i + sh])
                    if 0 <= i - sh < n:
                        alpha_i[idx] -= h[i - sh]
                    
                    # For coordinate j:
                    if 0 <= j + sh < n:
                        alpha_j[idx] += (1.0 - h[j + sh])
                    if 0 <= j - sh < n:
                        alpha_j[idx] -= h[j - sh]
