```python
def subgradient_optimize(h0, n, target_sum, steps=50000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Compute subgradient of max(corr)
        max_val = np.max(corr)
        max_indices = np.where(corr >= max_val - 1e-3)[0]
        # Subgradient at each j: (1/h[j-idx+n-1]) - h[j+idx-n+1] averaged over max indices
        # Use smooth max for stability
        alpha = 5.0
        shifted = alpha * (corr - max_val)
        exp_vals = np.exp(np.clip(shifted, -500, 0))
        w = exp_vals / np.sum(exp_vals)
        # Gradient
        n_corr = 2 * n - 1
        j_arr = np.arange(n)[:, None]
        m_arr = np.arange(n_corr)[None, :]
        first_idx = j_arr - m_arr + n - 1
        second_idx = j_arr + m_arr - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n-1)
        sc = np.clip(second_idx, 0, n-1)
        grad = np.sum(valid_f * (1.0 - h[fc]) * w[None, :] - valid_s * h[sc] * w[None, :], axis=1)
        # Step
        h = h - lr * grad
        # Project to [0,1] and sum constraint
        h = np.clip(h, 0, 1)
        # Fix sum
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        lr *= 0.999
    return h
```

```python
    first_idx = j_arr - m_arr + n - 1
    second_idx = j_arr + m_arr - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n-1)
    sc = np.clip(second_idx, 0, n-1)
    return fc, sc, valid_f, valid_s

def compute_grad(h, alpha, fc, sc, valid_f, valid_s):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    max_val = np.max(corr)
    shifted = np.clip(alpha * (corr - max_val), -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = max_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.sum(valid_f * (1.0 - h[fc]) * w[None, :] - valid_s * h[sc] * w[None, :], axis=1)
    return obj, grad
```
