import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum, tol=1e-12, max_iter=1000):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < tol:
            break
        mask = (h > tol) & (h < 1.0 - tol)
        if np.any(mask):
            step = diff / np.sum(mask)
            h[mask] += step
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_max(h, alpha):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    ev = np.exp(shifted)
    s = np.sum(ev)
    return m + np.log(s / len(ev)) / alpha

def fast_smooth_max_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    sm_val = m + np.log(s / len(ev)) / alpha
    
    grad = np.zeros(n)
    w_pos = w[n-1:]
    w_neg = w[:n]
    omh = 1.0 - h
    
    for k in range(n):
        wk = w_pos[k]
        if wk < 1e-14:
            continue
        if k < n:
            grad[k:] += wk * omh[:n-k]
        grad[:n-k] -= wk * h[k:]
    
    for k in range(1, n):
        wk = w_neg[k]
        if wk < 1e-14:
            continue
        grad[:n-k] += wk * omh[k:]
        grad[k:] -= wk * h[:n-k]
    
    return sm_val, grad

def slsqp_smooth_max(h0, n, target_sum, alpha, maxiter=300, ftol=1e-14):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        return fast_smooth_max(hc, alpha)
    
    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        _, g = fast_smooth_max_grad(hc, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def active_lag_optimize(h0, n, target_sum, active_idx, alpha, maxiter=200):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = get_corr(hc)
        ac = corr[active_idx]
        m = np.max(ac)

def hinge_optimize(h0, n, target_sum, T, maxiter=200):

        hc = np.clip(x, 0.0, 1.0)
        corr = get_corr(hc)
        excess = np.maximum(corr - T, 0.0)
        return np.sum(excess**2)

        corr = get_corr(hc)
        excess = np.maximum(corr - T, 0.0)
        grad = np.zeros(n)
        omh = 1.0 - hc

            wk = 2.0 * excess[k_idx]
            if wk < 1e-14:
                continue
            k = k_idx - (n-1)

                if k < n:
                    grad[k:] += wk * omh[:n-k]

def coordinate_descent(h0, n, target_sum, max_rounds, rng, deltas=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if deltas is None:
        deltas = [-0.005, -0.002, -0.001, -0.0005, 0.0005, 0.001, 0.002, 0.005]
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def project_to_sum(h, target_sum, tol=1e-12, max_iter=1000):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < tol:
            break
        mask = (h > tol) & (h < 1.0 - tol)
        if np.any(mask):
            step = diff / np.sum(mask)
            h[mask] += step
        else:
            h += diff / len(h)

    shifted = np.clip(alpha * (corr - m), -500, 500)
    ev = np.exp(shifted)
    s = np.sum(ev)
    return m + np.log(s / len(ev)) / alpha

    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)

                h = project_to_sum(h, ts)
                c5_cur = best_val
                improved = True

    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: SLSQP smooth max with increasing alpha
    alphas_p1 = [5.0, 15.0, 50.0, 150.0, 500.0, 1500.0]
    for alpha in alphas_p1:
        if time_left() < 80:
            break
        maxiter = max(80, min(600, int(time_left() * 0.7)))
        h_opt = do_slsqp_smooth_max(best_h, n, ts, alpha, maxiter, 1e-15)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Very high alpha SLSQP
    for alpha in [5000.0, 15000.0, 50000.0]:
        if time_left() < 60:
            break
        maxiter = max(100, min(500, int(time_left() * 0.6)))
        h_opt = do_slsqp_smooth_max(best_h, n, ts, alpha, maxiter, 1e-15)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Active set refinement
    if time_left() > 120:
        corr = get_corr(best_h)
        max_corr = np.max(corr)
        for thresh_frac in [0.995, 0.99, 0.98]:
            if time_left() < 40:
                break
            threshold = thresh_frac * max_corr
            active = np.where(corr >= threshold)[0]
            if 0 < len(active) < n:
                for alpha_a in [1000.0, 5000.0, 20000.0]:
                    if time_left() < 35:
                        break
                    h_act = do_slsqp_active(best_h, n, ts, active, alpha_a, 250)
                    c5_act = compute_c5(h_act)
                    if c5_act < best_c5:
                        best_c5 = c5_act
                        best_h = h_act
                    corr = get_corr(best_h)
                    max_corr = np.max(corr)
                    threshold = thresh_frac * max_corr
                    active = np.where(corr >= threshold)[0]
                    if len(active) == 0 or len(active) >= n:
                        break

    # Phase 4: Hinge loss optimization
    if time_left() > 100:
        dx = 2.0 / n
        corr_max = np.max(get_corr(best_h))
        for T_frac in [1.0, 0.995, 0.99, 0.985, 0.98, 0.975, 0.97]:
            if time_left() < 30:
                break
            T_try = T_frac * corr_max
            h_h = do_slsqp_hinge(best_h, n, ts, T_try, 200)
            c5_h = compute_c5(h_h)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h
                corr_max = np.max(get_corr(best_h))

    # Phase 5: Coordinate descent
    if time_left() > 50:
        deltas_cd = [-0.003, -0.001, -0.0005, 0.0005, 0.001, 0.003]
        n_rounds = min(40, max(10, int((time_left() - 20) / 3)))
        h_cd, c5_cd = coord_descent(best_h, n, ts, n_rounds, rng, deltas_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 6: Projected subgradient
    if time_left() > 50:
        n_pg = min(1500, int((time_left() - 20) * 3))
        h_pg, c5_pg = projected_subgrad(best_h, n, ts, 5000.0, n_pg, 0.08)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 7: Binary push
    if time_left() > 30:
        h_bp, c5_bp = binary_push(best_h, n, ts, 50)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp

    # Phase 8: Random restart + re-optimize
    if time_left() > 150:
        for mag in [0.03, 0.015, 0.008]:
            if time_left() < 70:
                break
            h_pert = random_perturb(best_h, n, ts, rng, mag)
            for alpha_r in [500.0, 3000.0]:
                if time_left() < 40:
                    break
                h_opt = do_slsqp_smooth_max(h_pert, n, ts, alpha_r, 200, 1e-14)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 9: Try higher resolution
    if time_left() > 250 and n <= 800:
        n_hi = n * 2
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi
            n = n_hi
            ts = ts_hi

        for alpha in [100.0, 1000.0, 5000.0, 20000.0]:
            if time_left() < 50:
                break
            h_opt = do_slsqp_smooth_max(best_h, n, ts, alpha, 150, 1e-14)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

        if time_left() > 40:
            h_cd2, c5_cd2 = coord_descent(best_h, n, ts, 20, rng,
                                         [-0.002, -0.0005, 0.0005, 0.002])
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Phase 10: Final polishing at best resolution
    if time_left() > 40:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        for alpha in [20000.0, 80000.0]:
            if time_left() < 30:
                break
            h_pol = do_slsqp_smooth_max(best_h, n_f, ts_f, alpha, 300, 1e-15)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

        if time_left() > 30:
            corr_f = get_corr(best_h)
            max_f = np.max(corr_f)
            active_f = np.where(corr_f >= 0.995 * max_f)[0]
            if 0 < len(active_f) < n_f:
                h_af = do_slsqp_active(best_h, n_f, ts_f, active_f, 20000.0, 250)
                c5_af = compute_c5(h_af)
                if c5_af < best_c5:
                    best_c5 = c5_af
                    best_h = h_af

        if time_left() > 25:
            h_cd3, c5_cd3 = coord_descent(best_h, n_f, ts_f, 50, rng,
                                         [-0.001, -0.0003, 0.0003, 0.001])
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3

        if time_left() > 20:
            h_pg2, c5_pg2 = projected_subgrad(best_h, n_f, ts_f, 10000.0, 800, 0.03)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2

    # Final: project and compute
    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)
    else:
        n_final = 720
        best_h = np.full(n_final, 0.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
