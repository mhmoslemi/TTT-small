def compute_grad_smooth_max(h, w, n):
    """Compute gradient of smooth max of correlate(h, 1-h) w.r.t. h"""
    grad = np.zeros(n)
    # w has length 2n-1, indexed by k=0,...,2n-2
    # For each k, lag = k - (n-1)
    # d_ℓ[j] = (1-h[j-ℓ]) - h[j+ℓ] (OOB = 0)
    for k in range(len(w)):
        wk = w[k]
        if abs(wk) < 1e-12:
            continue
        lag = k - (n - 1)
        # (1-h[j-lag]) for valid j
        if lag >= 0:
            j_start, j_end = lag, n
            grad[j_start:j_end] += wk * (1.0 - h[:n-lag])
            # h[j+lag] for valid j
            j_start2, j_end2 = 0, n - lag
            grad[j_start2:j_end2] -= wk * h[lag:]
        else:
            m = -lag
            # (1-h[j+m]) for valid j: j=0 to n-1-m
            j_start, j_end = 0, n - m
            grad[j_start:j_end] += wk * (1.0 - h[m:])
            # h[j-m] for valid j: j=m to n-1
            j_start2, j_end2 = m, n
            grad[j_start2:j_end2] -= wk * h[:n-m]
    return grad * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    
    # Compute gradient directly
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wk * (1.0 - h[:n-lag])
                grad[:n-lag] -= wk * h[lag:]
        else:
            m = -lag
            if n - m > 0:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:n-m]
    grad *= (2.0 / n)
    return obj, grad

def momentum_gradient_descent(h0, alpha, momentum, lr, max_iter, time_budget_fn):
    n = len(h0)
    v = np.zeros(n)
    h = h0.copy()
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    
    for it in range(max_iter):
        if time_budget_fn() < 5:
            break
        obj, grad = smooth_max_and_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_obj:
            best_obj = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_obj

        # For each k, the change in c_k when h[j] changes by delta is delta * d_k
        # We want to find t (new value of h[j]) that minimizes max_k (c_k + (t-h[j])*d_k)
        
        hj = h[j]

        d = np.zeros(2*n-1)

        # The max of lines is convex, so the minimum is at a breakpoint or at the boundary

def min_max_lines(a, b, t_lo=0.0, t_hi=1.0):
    """Minimize max_k (a[k] + b[k]*t) for t in [t_lo, t_hi]"""
    # Binary search for the point where the subdifferential contains 0
    for _ in range(50):
        t_mid = 0.5 * (t_lo + t_hi)
        vals = a + b * t_mid
        active = np.where(vals == np.max(vals) + 1e-12)[0]
        if len(active) == 0:
            active = np.where(vals >= np.max(vals) - 1e-12)[0]
        if np.min(b[active]) <= 0 <= np.max(b[active]):
            return t_mid
        elif np.min(b[active]) > 0:
            t_hi = t_mid  # function is increasing, minimum is to the left
        else:
            t_lo = t_mid  # function is decreasing, minimum is to the right
    return 0.5 * (t_lo + t_hi)

        elif not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at initial resolution
    if time_left() > 60:
        n1 = best_n
        for alpha in [20, 100, 500, 2000]:
            if time_left() < 20:
                break
            h_step, c5_step = momentum_smooth_max(best_h, alpha, 0.9, 0.05, 2000, time_left)
            if c5_step < best_c5:
                best_c5 = c5_step
                best_h = h_step.copy()

    # Phase 2: Upsample to higher resolution
    n_high = 480
    if n_high % 2 != 0:
        n_high += 1
    x_low = np.linspace(0, 2, best_n, endpoint=False)
    x_high = np.linspace(0, 2, n_high, endpoint=False)
    h_high = np.interp(x_high, x_low, best_h, period=2.0)
    h_high = project_box_sum(h_high, n_high / 2.0)

    # Phase 3: Optimize at high resolution with smooth max
    h_work = h_high.copy()
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
        best_n = n_high

    if time_left() > 200:
        for alpha in [20, 100, 500, 2000, 8000]:
            if time_left() < 30:
                break
                break
            h_step, c5_step = momentum_smooth_max(h_work, alpha, 0.9, 0.03, 3000, time_left)
            if c5_step < best_c5:
                best_c5 = c5_step
                best_h = h_step.copy()
                best_n = n_high
                h_work = h_step.copy()

    # Phase 4: Coordinate descent polish at high resolution
    if time_left() > 60:
        h_cd = best_h.copy()
        h_cd, c5_cd = coordinate_descent(h_cd, max_passes=300, time_fn=time_left)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_high

    # Phase 5: Multi-start diversification
    if time_left() > 200:
        n_ms = best_n
        for trial in range(30):
            if time_left() < 40:
                break
            noise_scale = 0.15 * (1.0 - trial / 30.0) + 0.02
            h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n_ms)
            h_shake = project_box_sum(h_shake, n_ms / 2.0)
            h_opt, c5_opt = momentum_smooth_max(h_shake, 500, 0.9, 0.03, 1500, time_left)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_ms
            h_cd2, c5_cd2 = coordinate_descent(h_opt, max_passes=100, time_fn=time_left)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_ms

    # Phase 6: Try even higher resolution
    if time_left() > 150:
        n_vhigh = 720
        x_cur = np.linspace(0, 2, best_n, endpoint=False)
        x_vh = np.linspace(0, 2, n_vhigh, endpoint=False)
        h_vh = np.interp(x_vh, x_cur, best_h, period=2.0)
        h_vh = project_box_sum(h_vh, n_vhigh / 2.0)
        h_vh_opt, c5_vh = momentum_smooth_max(h_vh, 500, 0.9, 0.02, 2000, time_left)
        if c5_vh < best_c5:
            best_c5 = c5_vh
            best_h = h_vh_opt.copy()
            best_n = n_vh
        h_vh_cd, c5_vh_cd = coordinate_descent(h_vh_opt, max_passes=150, time_fn=time_left)
        if c5_vh_cd < best_c5:
            best_c5 = c5_vh_cd
            best_h = h_vh_cd.copy()
            best_n = n_vh

    # Phase 7: Final coordinate descent polish
    if time_left() > 20:
        h_final, c5_final = coordinate_descent(best_h, max_passes=500, time_fn=time_left)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            best_n = len(h_final)

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
