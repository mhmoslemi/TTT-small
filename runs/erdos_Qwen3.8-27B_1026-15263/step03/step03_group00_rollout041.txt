Range of τ: from min(h_i) - 1 to max(h_i) + 1 (or narrower).

```python
def project_bounded_simplex(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)
```

def logsumexp_obj_grad(h, alpha, top_k=20):
    """Compute log-sum-exp objective and gradient, using only top_k contributing lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Compute weights
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    obj = max_corr + np.log(np.sum(exp_vals)) / alpha
    
    # Get top contributing lags
    top_indices = np.argsort(w)[-top_k:]
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in top_indices:
        w_i = w[idx]
        if w_i < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w_i * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += w_i * (1.0 - h[lag:])
            grad[lag:] -= w_i * h[:n-lag]
        else:
            k = -lag
            grad[:n-k] += w_i * (1.0 - h[k:])
            grad[k:] -= w_i * h[:n-k]
    
    return obj, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded_simplex(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    obj = cmax + np.log(exp_v.sum()) / alpha
    if top_k < len(w):
        idx = np.argsort(w)[-top_k:]
    else:
        idx = np.arange(len(w))
    grad = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            continue
        lag = i - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        else:
            k = -lag
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
    return obj, grad

def projected_gradient_descent(h, target_sum, alphas, max_iter_per_alpha, step_init):
    n = len(h)
    best_h = h.copy()
    best_obj = None
    h_curr = h.copy()
    prev_g = None
    prev_h = None
    step_size = step_init
    
    for alpha in alphas:
        for it in range(max_iter_per_alpha):
            obj, g = lse_obj_grad(h_curr, alpha)
            if best_obj is None or obj < best_obj:
                best_obj = obj
                best_h = h_curr.copy()
            
            if prev_g is not None:
                s = h_curr - prev_h
                y = g - prev_g
                sy = s.dot(y)
                yy = y.dot(y)
                ss = s.dot(s)
                if yy > 1e-15:
                    bb1 = ss / sy if sy > 1e-15 else step_size
                    bb2 = sy / yy if sy > 1e-15 else step_size
                    step_size = 0.5 * (bb1 + bb2) if bb1 > 0 and bb2 > 0 else step_size
                    step_size = np.clip(step_size, 1e-6, 1.0)
            
            # Backtracking line search
            t = step_size
            for _ in range(20):
                h_trial = project_bounded_simplex(h_curr - t * g, target_sum)
                obj_trial, _ = lse_obj_grad(h_trial, alpha)
                if obj_trial < obj - 1e-12:
                    break
                t *= 0.5
            
            prev_h = h_curr.copy()
            prev_g = g.copy()
            h_curr = h_trial
        
        # Update step size for next alpha
        step_size = 1.0 / alpha
    
    return best_h

    stale = 0

        alpha_local = 50.0

        exp_v = np.exp(shifted)

        # Weighted gradient

        top_idx = np.argsort(w)[-top_k:]

            if wi < 1e-10:
                continue

                grad += wi * (1.0 - 2.0 * h)
            elif lag > 0:

                grad[k:] -= wi * h[:n-k]

                    h_t1 = h.copy(); h_t1[i_inc] += t1; h_t1[i_dec] -= t1
                    h_t1 = project_bounded_simplex(h_t1, target_sum)

def mirror_descent_step(h, target_sum, n_eta, n_inner, eta_start):
    """Mirror descent on the dual variable p."""
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    K = 2 * n - 1
    p = np.ones(K) / K
    eta = eta_start
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for outer in range(n_eta):
        eta_cur = eta_start / (1.0 + 0.1 * outer)
        
        # Inner: gradient ascent to maximize sum_k p_k * R(k)
        for inner in range(n_inner):
            # Compute weighted gradient
            corr_h = np.correlate(h_curr, h_curr, mode="full") * dx
            # R(k) = corr_h[m] where m corresponds to lag k
            # Weighted sum: sum_m p[m] * corr_h[m]
            # Gradient: for each j, sum_m p[m] * (h[j+lag_m] + h[j-lag_m]) * dx
            # This is expensive to compute directly; use the top p entries
            top_p = np.argsort(p)[-30:]
            grad = np.zeros(n)
            for m in top_p:
                pm = p[m]
                if pm < 1e-10:
                    continue
                lag = m - (n - 1)
                w = pm * dx
                if lag == 0:
                    grad += w * 2.0 * h_curr
                elif lag > 0:
                    grad[:n-lag] += w * h_curr[lag:]
                    grad[lag:] += w * h_curr[:n-lag]
                else:
                    k = -lag
                    grad[:n-k] += w * h_curr[k:]
                    grad[k:] += w * h_curr[:n-k]
            
            h_new = project_bounded_simplex(h_curr + 0.02 * grad, target_sum)
            h_curr = h_new
        
        # Compute R(k) for current h
        corr_h = np.correlate(h_curr, h_curr, mode="full") * dx
        c5_cur = 1.0 - np.min(corr_h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_curr.copy()
        
        # Update p: multiplicative weight on R values
        # We want to focus on lags with LOW R (since we want to maximize min R)
        # So: p_k ∝ exp(-eta * R_k) (focus on the minimum)
        # Or equivalently: p_k ∝ exp(eta * (1-R_k)) = exp(eta * (1-R_k))
        R = corr_h
        max_R = np.max(R)
        p = np.exp(eta_cur * (R - max_R))  # This puts more weight on low R
        # Actually we want to focus on the MINIMUM, so:
        # p_k should be larger for smaller R_k
        # p_k ∝ exp(-eta * R_k) → p_k ∝ exp(eta * (max_R - R_k))
        p = np.exp(eta_cur * (max_R - R))
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(K) / K
        else:
            p = p / p_sum
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded_simplex(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    obj = cmax + np.log(exp_v.sum()) / alpha
    if top_k < len(w):
        idx = np.argsort(w)[-top_k:]
    else:
        idx = np.arange(len(w))
    grad = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            continue
        lag = i - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        else:
            k = -lag
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
    return obj, grad

def pgd_step(h_prev, h_curr, g_prev, g_curr, step_init):
    s = h_curr - h_prev
    y = g_curr - g_prev
    sy = s.dot(y)
    yy = y.dot(y)
    ss = s.dot(s)
    if yy > 1e-18 and sy > 1e-18:
        bb1 = ss / sy
        bb2 = sy / yy
        step = 0.5 * (bb1 + bb2)
        step = min(max(step, 1e-8), 2.0)
    else:
        step = step_init
    return step

def projected_gradient_descent(h0, target_sum, alphas, max_iter, step_init):
    h_curr = h0.copy()
    best_h = h0.copy()
    best_obj = float('inf')
    prev_h = None
    prev_g = None
    step_size = step_init
    for alpha in alphas:
        step_size = max(1.0 / alpha, 1e-8)
        for it in range(max_iter):
            obj, g = lse_obj_grad(h_curr, alpha)
            if obj < best_obj:
                best_obj = obj
                best_h = h_curr.copy()
            if prev_g is not None:
                step_size = pgd_step(prev_h, h_curr, prev_g, g, step_size)
            t = step_size
            h_trial = h_curr
            obj_trial = obj
            for _ in range(25):
                h_trial = project_bounded_simplex(h_curr - t * g, target_sum)
                obj_trial, _ = lse_obj_grad(h_trial, alpha)
                if obj_trial < obj - 1e-14:
                    break
                t *= 0.5
            if obj_trial >= obj - 1e-14:
                h_trial = project_bounded_simplex(h_curr - t * g, target_sum)
                obj_trial, _ = lse_obj_grad(h_trial, alpha)
                if obj_trial >= obj - 1e-14:
                    break
            prev_h = h_curr.copy()
            prev_g = g.copy()
            h_curr = h_trial
    return best_h

def weighted_corr_gradient(h, corr, alpha_w, top_k=15):
    n = len(h)
    cmax = np.max(corr)
    shifted = alpha_w * (corr - cmax)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    if top_k < len(w):
        idx = np.argsort(w)[-top_k:]
    else:
        idx = np.arange(len(w))
    grad = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            continue
        lag = i - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        else:
            k = -lag
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
    return grad, w

def find_backtrack_step(h, i_inc, i_dec, t_max, target_sum, base_c5):
    t = t_max
    for _ in range(10):
        if t < 1e-12:
            return 0.0
        h_trial = h.copy()
        h_trial[i_inc] += t
        h_trial[i_dec] -= t
        h_trial = project_bounded_simplex(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < base_c5 - 1e-14:
            return t
        t *= 0.5
    return 0.0

def coord_desc_multi(h, target_sum, max_iters, rng, top_k=15, alpha_w=40.0):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        grad, w = weighted_corr_gradient(h, corr, alpha_w, top_k)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, grad, np.inf)
        g_dec = np.where(can_dec, grad, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if grad[i_inc] >= grad[i_dec] - 1e-12:
            break
        t_max = min(1.0 - h[i_inc], h[i_dec])
        if t_max < 1e-12:
            break
        t_opt = find_backtrack_step(h, i_inc, i_dec, t_max, target_sum, c5)
        if t_opt < 1e-12:
            stale += 1
            if stale > 50:
                break
            continue
        h[i_inc] += t_opt
        h[i_dec] -= t_opt
        h = np.clip(h, 0.0, 1.0)
        h = project_bounded_simplex(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
        else:
            h[i_inc] -= t_opt
            h[i_dec] += t_opt
            h = np.clip(h, 0.0, 1.0)
            h = project_bounded_simplex(h, target_sum)
            stale += 1
            if stale > 80:
                break
    return h, c5

def mirror_descent_dual(h0, target_sum, n_outer, n_inner, eta_start):
    n = len(h0)
    dx = 2.0 / n
    h_curr = h0.copy()
    K = 2 * n - 1
    p = np.ones(K) / K
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for outer in range(n_outer):
        eta_cur = eta_start * (0.95 ** outer)
        inner_step = 0.03 / (1.0 + 0.05 * outer)
        for inner in range(n_inner):
            corr_hh = np.correlate(h_curr, h_curr, mode="full")
            top_idx = np.argsort(p)[-40:]
            grad = np.zeros(n)
            for m in top_idx:
                pm = p[m]
                if pm < 1e-10:
                    continue
                lag = m - (n - 1)
                wv = pm * dx
                if lag == 0:
                    grad += wv * 2.0 * h_curr
                elif lag > 0:
                    grad[:n - lag] += wv * h_curr[lag:]
                    grad[lag:] += wv * h_curr[:n - lag]
                else:
                    k = -lag
                    grad[:n - k] += wv * h_curr[k:]
                    grad[k:] += wv * h_curr[:n - k]
            h_new = project_bounded_simplex(h_curr + inner_step * grad, target_sum)
            h_curr = h_new
        corr_hh = np.correlate(h_curr, h_curr, mode="full") * dx
        c5_cur = 1.0 - float(np.min(corr_hh))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_curr.copy()
        max_R = float(np.max(corr_hh))
        p = np.exp(eta_cur * (max_R - corr_hh))
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(K) / K
        else:
            p = p / p_sum
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h, period=2.0)
