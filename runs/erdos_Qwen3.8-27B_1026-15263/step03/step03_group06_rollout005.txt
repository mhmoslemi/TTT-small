import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(mask)
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha, top_k=None):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    b = 1.0 - h
    size = 2 * n
    
    corr = np.correlate(h, b, mode="full")
    
    if top_k is not None and top_k < len(corr):
        # Only focus on top_k correlations
        idx_top = np.argpartition(corr, -top_k)[-top_k:]
        corr_s = corr[idx_top]
        m = np.max(corr_s)
        shifted = alpha * (corr_s - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_s = np.exp(shifted)
        w_s = exp_s / np.sum(exp_s)
        sm_val = m + np.log(np.sum(exp_s)) / alpha
        # Build full weight vector (zero except at top_k positions)
        w = np.zeros(len(corr))
        w[idx_top] = w_s
    else:
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        sm_val = m + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient via FFT convolution
    W_fft = np.fft.rfft(w, size)
    b_fft = np.fft.rfft(b, size)
    h_fft = np.fft.rfft(h, size)
    w_rev = w[::-1]
    Wr_fft = np.fft.rfft(w_rev, size)
    
    conv_wb = np.fft.irfft(W_fft * b_fft, size)
    conv_whr = np.fft.irfft(Wr_fft * h_fft, size)
    
    g = conv_wb[n-1:2*n-1] - conv_whr[n-1:2*n-1]
    
    return sm_val, g

def adam_optimize(h0, n, target_sum, alphas, iters_per_alpha, lr_init=0.05, 
                  beta1=0.8, beta2=0.999, eps=1e-8, top_k=None, verbose=False):
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for a_idx, alpha in enumerate(alphas):
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        t = 0
        lr = lr_init / (1.0 + 0.1 * a_idx)
        
        for it in range(iters_per_alpha):
            t += 1
            _, g = smooth_max_grad(h, alpha, top_k=top_k)
            
            # Gradient projection: remove component in direction of all-ones (for sum constraint)
            g_mean = np.mean(g)
            g_proj = g - g_mean  # This ensures the step preserves sum approximately
            
            m_v = beta1 * m_v + (1 - beta1) * g_proj
            v_v = beta2 * v_v + (1 - beta2) * g_proj**2
            m_hat = m_v / (1 - beta1**t)
            v_hat = v_v / (1 - beta2**t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_to_sum(h_new, target_sum)
            
            c5_new = compute_c5(h_new)
            
            if c5_new > c5_best * 1.1 + 0.05:
                lr *= 0.5
                if lr < 1e-6:
                    break
                continue
            
            h = h_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
                lr = min(lr * 1.01, 0.2)
        
        if verbose:
            print(f"  alpha={alpha}, c5={c5_best:.6f}")
    
    return h_best, c5_best

def block_coord_descent(h0, n, target_sum, n_rounds, rng, block_size=2):
    """Coordinate descent on blocks of adjacent elements."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(n_rounds):
        improved = False
        indices = list(range(n))
        rng.shuffle(indices)
        
        for i in indices:
            i2 = (i + 1) % n if block_size > 1 else -1
            h_orig_i = h[i]
            h_orig_i2 = h[i2] if i2 >= 0 else 0
            
            # Try perturbations
            best_c5 = c5_cur
            best_val = None
            for scale in [0.02, 0.01, 0.005, 0.002]:
                for signs in [(1,1), (1,-1), (-1,1), (-1,-1)] if block_size > 1 else [(1,)]:
                    if block_size > 1:
                        sv = list(signs)
                    else:
                        sv = [signs[0]]
                    h_trial = h.copy()
                    h_trial[i] = np.clip(h_trial[i] + sv[0]*scale, 0, 1)
                    if block_size > 1:
                        h_trial[i2] = np.clip(h_trial[i2] + sv[1]*scale, 0, 1)
                    h_trial = project_to_sum(h_trial, target_sum)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_val = h_trial.copy()
            
            if best_val is not None:
                h = best_val
                c5_cur = best_c5
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

def single_coord_descent(h0, n, target_sum, n_rounds, rng):
    """Fast single-coordinate descent with golden section line search."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    gr = (1 + np.sqrt(5)) / 2
    
    for rnd in range(n_rounds):
        improved = False
        indices = rng.permutation(n)
        
        for i in indices:
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-4:
                continue
            
            # Bounded minimization
            a = lo
            b = hi
            c = b - (b - a) / gr
            d = a + (b - a) / gr
            
            h_c = h.copy(); h_c[i] = c; h_c = project_to_sum(h_c, target_sum)
            h_d = h.copy(); h_d[i] = d; h_d = project_to_sum(h_d, target_sum)
            fc = compute_c5(h_c)
            fd = compute_c5(h_d)
            
            for _ in range(15):
                if b - a < 1e-5:
                    break
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - (b - a) / gr
                    h_c = h.copy(); h_c[i] = c; h_c = project_to_sum(h_c, target_sum)
                    fc = compute_c5(h_c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + (b - a) / gr
                    h_d = h.copy(); h_d[i] = d; h_d = project_to_sum(h_d, target_sum)
                    fd = compute_c5(h_d)
            
            best_pt = (a + b) / 2
            h_best = h.copy(); h_best[i] = best_pt; h_best = project_to_sum(h_best, target_sum)
            c5_best = compute_c5(h_best)
            
            if c5_best < c5_cur - 1e-12:
                h = h_best
                c5_cur = c5_best
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

def fourier_init(n, K, rng, mean=0.5):
    """Generate initial h using Fourier modes."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, mean)
    for k in range(1, K+1):
        amp = rng.uniform(0.05, 0.3) / np.sqrt(k)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
    h = project_to_sum(np.clip(h, 0, 1), n/2.0)
    return h

def structured_init(n, rng):
    """Create structured initial patterns."""
    x = np.linspace(0, 2, n, endpoint=False)
    patterns = []
    
    # Sine wave

        for amp in [0.2, 0.3, 0.4, 0.45]:

            patterns.append(np.clip(h, 0, 1))
    
    # Block patterns
    for nblocks in [4, 6, 8, 10, 12]:
        block_w = 2.0 / nblocks
        h = np.zeros(n)
        for i in range(nblocks):
            if i % 2 == 0:
                start = int(i * n / nblocks)
                end = int((i+1) * n / nblocks)
                h[start:end] = 1.0 / (nblocks / 2)  # scale so integral = 1
        h = np.clip(h, 0, 1)
        patterns.append(h)
    
    # Random binary
    h = (rng.rand(n) < 0.5).astype(float)
    patterns.append(h)
    
    # Smoother random
    h = 0.5 + 0.3 * np.sin(np.pi * x * 3) + 0.2 * np.sin(np.pi * x * 7)
    patterns.append(np.clip(h, 0, 1))
    
    # Return the pattern with best C5
    best_h = None
    best_c5 = float('inf')
    for p in patterns:
        p_adj = project_to_sum(p, n/2.0)
        c5 = compute_c5(p_adj)
        if c5 < best_c5:
            best_c5 = c5
            best_h = p_adj
    return best_h, best_c5
