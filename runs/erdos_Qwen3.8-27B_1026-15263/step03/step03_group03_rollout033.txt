def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = corr - m
    exp_vals = np.exp(alpha * shifted)
    Z = exp_vals.sum()
    obj = m + np.log(Z) / alpha
    w = alpha * exp_vals / Z
    n = len(h)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2*n-1)[None, :]
    first_j = j_idx - k_idx + n - 1
    valid_f = (first_j >= 0) & (first_j < n)
    second_i = j_idx + k_idx - n + 1
    valid_s = (second_i >= 0) & (second_i < n)
    fc = np.clip(first_j, 0, n-1)
    sc = np.clip(second_i, 0, n-1)
    t1 = valid_f * (1.0 - h[fc]) * w[None, :]
    t2 = valid_s * h[sc] * w[None, :]
    grad = np.sum(t1 - t2, axis=1)
    return obj, grad

def lbfgsb_optimize(h0, target_sum, alpha, penalty, maxiter):
    n = len(h0)
    def obj(x):
        sm, _ = smooth_max_obj_grad(x, alpha)
        s = np.sum(x) - target_sum

                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return project_box_sum(res.x, target_sum)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        candidates = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n]
        for _ in range(max(0, n_cand - 5)):
            candidates.append(int(rng.randint(n)))
        seen = set()
        for k in candidates:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = min(max(h[j], 0.0), 1.0)
            h[best_k] = min(max(h[best_k], 0.0), 1.0)
            improved = True
    return h, corr, improved

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = h.astype(float)
    lo, hi = -3.0, 3.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = corr - m
    exp_vals = np.exp(alpha * shifted)
    Z = exp_vals.sum()
    obj = m + np.log(Z) / alpha
    w = alpha * exp_vals / Z
    n = len(h)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    first_j = j_idx - k_idx + n - 1
    valid_f = (first_j >= 0) & (first_j < n)
    second_i = j_idx + k_idx - n + 1
    valid_s = (second_i >= 0) & (second_i < n)
    fc = np.clip(first_j, 0, n - 1)
    sc = np.clip(second_i, 0, n - 1)
    t1 = valid_f * (1.0 - h[fc]) * w[None, :]
    t2 = valid_s * h[sc] * w[None, :]
    grad = np.sum(t1 - t2, axis=1)
    return obj, grad

def lbfgsb_smooth(h0, target_sum, alpha, penalty, maxiter, ftol):
    n = len(h0)
    def obj_func(x):
        sm, _ = smooth_max_obj_grad(x, alpha)
        s = np.sum(x) - target_sum
        return sm + penalty * s * s
    def grad_func(x):
        _, g = smooth_max_obj_grad(x, alpha)
        s = np.sum(x) - target_sum
        return g + 2.0 * penalty * s
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h0, method='L-BFGS-B', jac=grad_func,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-9})
    return project_box_sum(res.x, target_sum)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent(h, corr, rng, n_cand):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        candidates = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                      (j + n // 3) % n, (j + 2 * n // 3) % n,
                      (j + n // 4) % n, (j + 3 * n // 4) % n]
        for _ in range(max(0, n_cand - 7)):
            candidates.append(int(rng.randint(n)))
        seen = set()
        for k in candidates:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] = min(max(h[j] + best_delta, 0.0), 1.0)
            h[best_k] = min(max(h[best_k] - best_delta, 0.0), 1.0)
            improved = True
    return h, corr, improved

def multi_res_optimize(h_parent, n_parent, n_target, target_sum_target, alphas, maxiter_each, t_deadline):
    x_parent = np.linspace(0, 2, n_parent, endpoint=False)
    x_target = np.linspace(0, 2, n_target, endpoint=False)
    h_new = project_box_sum(np.interp(x_target, x_parent, h_parent, period=2.0), target_sum_target)
    for a in alphas:
        if time.time() > t_deadline - 5:
            break
        h_new = lbfgsb_smooth(h_new, target_sum_target, a, penalty=500.0, maxiter=maxiter_each, ftol=1e-15)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:

            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    ts = best_n / 2.0

    # Phase 1: L-BFGS-B with increasing alpha (45% budget)
    t_end1 = t_start + 0.45 * budget_s
    h_work = best_h.copy()
    alpha_schedule = [30, 50, 100, 200, 400, 800, 1500]
    for a in alpha_schedule:
        if time.time() >= t_end1 - 10:
            break
        mi = 500 if a <= 200 else 350
        h_new = lbfgsb_smooth(h_work, ts, a, penalty=1000.0, maxiter=mi, ftol=1e-16)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = best_n
        h_work = h_new

    # Phase 2: Pair coordinate descent with restarts (20% budget)
    t_end2 = t_start + 0.65 * budget_s
    h_cd = best_h.copy()
    corr_cd = np.correlate(h_cd, 1.0 - h_cd, mode="full")
    stall = 0
    noise_level = 0.02
    while time.time() < t_end2 - 5:
        h_cd, corr_cd, _ = pair_descent(h_cd, corr_cd, rng, n_cand=14)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = len(h_cd)
            stall = 0
            noise_level = 0.02
        else:
            stall += 1
        if stall % 200 == 0:
            noise_level = min(0.15, 0.02 + 0.02 * (stall // 200))
            h_cd = project_box_sum(h_cd + rng.uniform(-noise_level, noise_level, best_n), ts)
            corr_cd = np.correlate(h_cd, 1.0 - h_cd, mode="full")
        if stall >= 1500:
            break

    # Phase 3: Multi-resolution exploration (20% budget)
    t_end3 = t_start + 0.85 * budget_s
    x_best = np.linspace(0, 2, best_n, endpoint=False)
    res_list = [best_n * 2, best_n * 3 // 2, best_n + 80, best_n - 40, best_n * 5 // 3]
    res_list = [r for r in res_list if r % 2 == 0 and 60 <= r <= 800]
    res_list = sorted(set(res_list), key=lambda x: abs(x - best_n))
    for n_tr in res_list:
        if time.time() >= t_end3 - 15:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        h_tr = multi_res_optimize(best_h, best_n, n_tr, ts_tr,
                                  alphas=[100, 300, 800], maxiter_each=200,
                                  t_deadline=t_end3)
        # Quick pair descent polish
        corr_tr = np.correlate(h_tr, 1.0 - h_tr, mode="full")
        for _ in range(20):
            if time.time() >= t_end3 - 5:
                break
            h_tr, corr_tr, _ = pair_descent(h_tr, corr_tr, rng, n_cand=10)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr
            x_best = np.linspace(0, 2, best_n, endpoint=False)

    # Phase 4: Final high-alpha L-BFGS-B + coordinate descent (15% budget)
    t_end4 = t_start + budget_s - 5
    h_fin = best_h.copy()
    n_fin = best_n
    ts_fin = n_fin / 2.0

    if time.time() < t_end4 - 60:
        h_fin = lbfgsb_smooth(h_fin, ts_fin, 2000, penalty=2000.0, maxiter=300, ftol=1e-16)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
        h_fin = best_h.copy()

    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent(h_fin, corr_fin, rng, n_cand=16)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
