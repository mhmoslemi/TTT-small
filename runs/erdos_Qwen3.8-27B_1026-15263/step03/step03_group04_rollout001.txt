    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Use log-sum-exp for numerical stability
    a_max = np.max(shifted)
    s = np.sum(np.exp(shifted - a_max))
    obj = m + (a_max + np.log(s)) / alpha
    return obj, corr

def smooth_max_grad(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    a_max = np.max(shifted)
    exp_v = np.exp(shifted - a_max)
    s = np.sum(exp_v)
    w = exp_v / s
    grad = np.zeros(n)
    thresh = 1e-12
    for k in range(len(corr)):
        if w[k] < thresh:
            continue
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return grad

def optimize_lbfgsb(h0, n, target_sum, alpha, maxiter=300):
    def fun(x):
        return smooth_max_obj(x, alpha)[0] + 0.5 * 1e6 * (np.sum(x) - target_sum)**2
    def jac(x):
        g = smooth_max_grad(x, alpha)
        return g + 1e6 * (np.sum(x) - target_sum)
    try:

                      bounds=[(0.0, 1.0)] * n,

        if best_h is None:

        h = best_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k_idx, n):
    """Gradient of correlation at specific lag index k_idx."""
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def smooth_max(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    sh = alpha * (corr - m)
    s = np.sum(np.exp(sh - sh.max()))
    obj = m + (sh.max() + np.log(s)) / alpha
    exp_v = np.exp(sh - sh.max())
    w = exp_v / s
    grad = np.zeros(n)
    thresh = 1e-12
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_penalty(h0, n, target_sum, alpha, pen, maxiter=200):
    def fun(x):
        obj, _ = smooth_max(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * pen * s_dev * s_dev
    def jac(x):
        _, g = smooth_max(x, alpha)
        s_dev = np.sum(x) - target_sum
        return g + pen * s_dev * np.ones(n)
    try:
        res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def exact_max_polish(h, n, target_sum, max_rounds=500):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:5]
        best_improve = 0
        best_h = None
        for k_max in top_k:
            grad = lag_gradient(h, k_max, n)
            for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    if c5_cur - c5_new > best_improve:
                        best_improve = c5_cur - c5_new
                        best_h = h_new.copy()
        if best_h is None:
            break
        h = best_h
        c5_cur = compute_c5(h)
    return h, c5_cur

def binary_sa(h_init, n, target_sum, n_iters, t0, t1, rng):
    """Simulated annealing on binary vectors."""
    # Initialize
    h = h_init.copy()
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:int(round(target_sum))]] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    
    cool = (t1 / t0) ** (1.0 / max(n_iters, 1))
    t = t0
    for it in range(n_iters):
        t *= cool
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-18)):
            cur_c5 = new_c5
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin < 0.5)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c5

    """Try to improve binary vector by targeted swaps."""
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    corr = compute_corr(h)
    for r in range(n_rounds):
        top_lags = np.argsort(corr)[::-1][:8]
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Try to find a swap that reduces the max
        best_swap = None
        best_c5 = c5_cur
        n_trials = 200
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_swap = (i, j)

                h[i] = 1.0
                h[j] = 0.0
        if best_swap is not None:
            h[best_swap[0]] = 0.0
            h[best_swap[1]] = 1.0
            c5_cur = best_c5
            improved = True
        if not improved:
            break
    return h, c5_cur
