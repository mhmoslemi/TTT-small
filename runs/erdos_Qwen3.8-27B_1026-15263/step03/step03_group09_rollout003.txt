        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if t % 500 == 0:
            lr *= 0.7
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k, decay):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            jj1 = j_arr + m - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - m + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        lr = lr_init / (1.0 + decay * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def fine_coord_descent(h0, n, target_sum, n_rounds, rng, base_delta, shrink):
    h = h0.copy()
    c5_cur = compute_c5(h)
    delta = base_delta
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            best_val = c5_cur
            best_nv = hi
            for sign in [-1, 1]:
                for s in [1.0, 0.5, 0.25, 0.1]:
                    nv = hi + sign * delta * s
                    if nv < -1e-15 or nv > 1.0 + 1e-15:
                        continue
                    h[i] = nv
                    h = project_to_sum(h, target_sum)
                    c5_new = compute_c5(h)
                    if c5_new < best_val - 1e-15:
                        best_val = c5_new
                        best_nv = nv
            if best_nv != hi:
                h[i] = best_nv
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
            else:
                h[i] = hi
                h = project_to_sum(h, target_sum)
        if not improved:
            break
        delta *= shrink
    return h, c5_cur

def pair_swap_descent(h0, n, target_sum, n_iters, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([0.005, 0.01, 0.02, 0.002, 0.001])
    for _ in range(n_iters):
        improved = False
        a, b = rng.choice(n, 2, replace=False)
        if a == b:
            continue
        for d in deltas:
            for sa in [-1, 1]:
                na = h[a] + sa * d
                nb = h[b] - sa * d
                if na < -1e-15 or na > 1.0 + 1e-15:
                    continue
                if nb < -1e-15 or nb > 1.0 + 1e-15:
                    continue
                h[a] = na
                h[b] = nb
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[a] -= sa * d
                    h[b] += sa * d
                if improved:
                    break
            if improved:
                break
        if not improved:
            h[a] = h[a]
            h[b] = h[b]
    return h, c5_cur

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_sum(h_new, n_new / 2.0)

def make_fourier_starts(n, rng, n_starts):
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    for _ in range(n_starts):
        max_freq = rng.randint(3, 12)
        a = rng.uniform(-0.4, 0.4, max_freq + 1)
        b = rng.uniform(-0.4, 0.4, max_freq)
        h = np.full(n, 0.5)
        for k in range(1, max_freq):
            h += a[k] * np.cos(2.0 * np.pi * k * x / 2.0)
            h += b[k] * np.sin(2.0 * np.pi * k * x / 2.0)
        h = project_to_sum(h, n / 2.0)
        starts.append(h)
    return starts

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts0 = best_n / 2.0
            best_h = project_to_sum(best_h, ts0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_update(h_cand, n_cand):
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = n_cand

    n_list = [best_n, 720]
    if best_n != 360:
        n_list = [best_n]
    if best_n != 720:
        n_list.append(720)
    n_list = sorted(set(n_list))

    for n in n_list:
        if _tl(t_start, budget_s) < 120:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0

        if best_n == n and best_h is not None:
            h_cur = best_h.copy()
        else:
            if best_n >= 2:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h_interp = np.interp(x_new, x_old, best_h)
                h_cur = project_to_sum(h_interp, ts)
            else:
                h_cur = np.full(n, 0.5)
        c5_cur = compute_c5(h_cur)

        for alpha in [10.0, 50.0, 200.0, 500.0]:
            if _tl(t_start, budget_s) < 40:
                break
            maxiter = max(50, min(800, int(_tl(t_start, budget_s) * 0.8)))
            h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_cur:
                h_cur = h_opt
                c5_cur = c5_opt
        try_update(h_cur, n)

        if _tl(t_start, budget_s) > 60:
            n_adam = min(4000, max(800, int(_tl(t_start, budget_s) * 1.5)))
            h_a, c5_a = adam_optimize(h_cur, n, ts, 80.0, n_adam, 0.06)
            if c5_a < c5_cur:
                h_cur = h_a
                c5_cur = c5_a
        try_update(h_cur, n)

        if _tl(t_start, budget_s) > 50:
            n_sub = min(8000, max(2000, int(_tl(t_start, budget_s) * 2.0)))
            for tk in [5, 10, 20]:
                if _tl(t_start, budget_s) < 30:
                    break
                h_sg, c5_sg = topk_subgradient(h_cur, n, ts, n_sub // 3, 0.04, tk, 0.02)
                if c5_sg < c5_cur:
                    h_cur = h_sg
                    c5_cur = c5_sg
        try_update(h_cur, n)

        if _tl(t_start, budget_s) > 40:
            n_cd = max(50, min(400, int(_tl(t_start, budget_s) * 0.5)))
            h_cd, c5_cd = fine_coord_descent(h_cur, n, ts, n_cd, rng, 0.008, 0.95)
            if c5_cd < c5_cur:
                h_cur = h_cd
                c5_cur = c5_cd
        try_update(h_cur, n)

    if best_h is not None and _tl(t_start, budget_s) > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [300.0, 800.0, 2000.0]:
            if _tl(t_start, budget_s) < 20:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 1000)
            try_update(h_pol, n_f)
            best_h = h_pol

        if _tl(t_start, budget_s) > 40:
            n_adam_f = min(6000, max(1000, int(_tl(t_start, budget_s) * 2.0)))
            h_af, c5_af = adam_optimize(best_h, n_f, ts_f, 400.0, n_adam_f, 0.02)
            if c5_af < best_c5:
                best_h = h_af
                best_c5 = c5_af

        if _tl(t_start, budget_s) > 30:
            n_sub_f = min(12000, max(3000, int(_tl(t_start, budget_s) * 2.5)))
            for tk in [3, 5, 10]:
                if _tl(t_start, budget_s) < 15:
                    break
                h_sg, c5_sg = topk_subgradient(best_h, n_f, ts_f, n_sub_f // 3, 0.02, tk, 0.01)
                if c5_sg < best_c5:
                    best_h = h_sg
                    best_c5 = c5_sg

    if best_h is not None and _tl(t_start, budget_s) > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        remaining = _tl(t_start, budget_s)
        n_cd_f = max(200, min(2000, int(remaining * 2.0)))
        h_cd, c5_cd = fine_coord_descent(best_h, n_f, ts_f, n_cd_f, rng, 0.003, 0.93)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    if best_h is not None and _tl(t_start, budget_s) > 20:
        n_f = best_n
        ts_f = n_f / 2.0
        remaining = _tl(t_start, budget_s)
        n_ps = max(2000, min(20000, int(remaining * 3.0)))
        h_ps, c5_ps = pair_swap_descent(best_h, n_f, ts_f, n_ps, rng)
        if c5_ps < best_c5:
            best_h = h_ps
            best_c5 = c5_ps

    if best_h is not None and _tl(t_start, budget_s) > 10:
        n_f = best_n
        ts_f = n_f / 2.0
        n_cd2 = max(100, min(500, int(_tl(t_start, budget_s) * 1.5)))
        h_cd2, c5_cd2 = fine_coord_descent(best_h, n_f, ts_f, n_cd2, rng, 0.0008, 0.9)
        if c5_cd2 < best_c5:
            best_h = h_cd2
            best_c5 = c5_cd2

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
