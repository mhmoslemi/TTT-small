    """Compute gradients of all c_k with respect to h."""
    # grad[k + n - 1][j] = ∂c_k/∂h[j] for k = -(n-1), ..., n-1

    grad = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        idx = k + n - 1

        #             and as h[i] when i=j (valid if 0<=j<=n-1-k, i.e., k<=n-1-j)
        # If k < 0 (kp=-k): h[j] appears as h[i] when i=j (valid if kp<=j<=n-1, i.e., -k<=j)
        #                   and as h[i-kp] when i-kp=j, i=j+kp (valid if kp<=j+kp<=n-1, i.e., 0<=j<=n-1-kp)
        
        pass  # This is getting complicated, let me use a different approach
    
    # Alternative: compute gradient by the formula
    # d(c_k)/d(h[j]) for k >= 0:
    #   (1-h[j-k]) if j-k in [0, n-1-k] (i.e., j in [k, n-1])
    #   -h[j+k] if j+k in [0, n-1] (i.e., j in [0, n-1-k])
    # For k < 0 (kp = -k):
    #   (1-h[j-kp]) if j-kp in [0, n-1-kp] (i.e., j in [kp, n-1])
    #   -h[j+kp] if j+kp in [0, n-1] (i.e., j in [0, n-1-kp])
    
    for idx in range(2*n-1):
        k = idx - (n-1)
        w_val = w[idx]
        if w_val < 1e-15:
            continue
        if k >= 0:
            # (1-h[j-k]) for j in k..n-1
            g[k:] += w_val * (1 - h[:n-k])
            # -h[j+k] for j in 0..n-1-k
            g[:n-k] -= w_val * h[k:]
        else:
            kp = -k
            g[kp:] += w_val * (1 - h[:n-kp])
            g[:n-kp] -= w_val * h[kp:]
    
    return val, g

    lr = lr_init
    h_best = h.copy()
    c5_best = compute_c5(h)
    
    for it in range(n_iters):

            break
        val, g = smooth_max_obj_grad(y, alpha)

            continue

            h_best = h_new.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_sca_data(h, n):
    G = np.zeros((2 * n - 1, n))
    b = np.zeros(2 * n - 1)
    for k in range(n):
        idx = k + n - 1
        G[idx, k:] = 1.0 - h[:n - k]
        G[idx, :n - k] -= h[k:]
        b[idx] = np.dot(G[idx], h) - np.dot(h[k:], 1.0 - h[:n - k])

        idx = n - 1 - kp
        G[idx, kp:] = 1.0 - h[:n - kp]
        G[idx, :n - kp] -= h[kp:]
        b[idx] = np.dot(G[idx], h) - np.dot(h[kp:], 1.0 - h[:n - kp])
    return G, b
