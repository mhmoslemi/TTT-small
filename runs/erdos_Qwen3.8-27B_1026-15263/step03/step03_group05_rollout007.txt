def subgradient_step(h, top_k=5, lr=0.1):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

For binary h, a "swap" is: pick i with h[i]=1, pick j with h[j]=0, set h[i]=0, h[j]=1. This preserves the sum.

The change in c_m for a swap (i: 1→0, j: 0→1):
- For the term h[i]*(1-h[i+m-n+1]): changes from (1-h[i+m-n+1]) to 0, so c_m decreases by (1-h[i+m-n+1])
- For the term h[j]*(1-h[j+m-n+1]): changes from 0 to (1-h[j+m-n+1]), so c_m increases by (1-h[j+m-n+1])
- For the term h[i+m-n+1]*(1-h[i]): the (1-h[i]) changes from 0 to 1, so c_m increases by h[i+m-n+1]
- For the term h[j+m-n+1]*(1-h[j]): the (1-h[j]) changes from 1 to 0, so c_m decreases by h[j+m-n+1]

Net change in c_m: -(1-h[i+m-n+1]) + (1-h[j+m-n+1]) + h[i+m-n+1] - h[j+m-n+1]
= -(1-2h[i+m-n+1]) + (1-2h[j+m-n+1])

So Δc_m = (1-2h[j+m-n+1]) - (1-2h[i+m-n+1]) = 2*(h[i+m-n+1] - h[j+m-n+1])

This is valid when all indices are in range. For boundary effects, some terms don't change.

This means: for the swap to decrease c_m, we need h[i+m-n+1] < h[j+m-n+1], i.e., the position that was 1 (i) should have a 0 at its "corresponding" position, and the position that was 0 (j) should have a 1 at its "corresponding" position.

This is a useful structural insight. For the max c_m, we want to find swaps that decrease it.

If such pairs exist, the swap strictly decreases c_{k*}. Let's try all such pairs and pick the one that best reduces the overall max.

This is a very targeted local search. Let me implement it:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary_sum(h, n, rng):
    h = h.copy().round().astype(float)
    ts = int(n / 2)
    ones_count = int(h.sum())
    if ones_count > ts:
        idx = np.where(h > 0.5)[0]
        rng.shuffle(idx)
        for i in range(min(ones_count - ts, len(idx))):
            h[idx[i]] = 0.0
    elif ones_count < ts:
        idx = np.where(h < 0.5)[0]
        rng.shuffle(idx)
        for i in range(min(ts - ones_count, len(idx))):
            h[idx[i]] = 1.0
    return h

def directed_local_search(h, n, rng, time_fn, max_rounds=8000):
    h = fix_binary_sum(h, n, rng)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for rnd in range(max_rounds):
        if time_fn() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new_c5 = c5_cur
        best_i, best_j = -1, -1
        n_trials = min(150, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if nc < best_new_c5 - 1e-14:
                best_new_c5 = nc
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5_cur = best_new_c5
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def simulated_annealing(h, n, rng, time_fn, t_init=0.4, t_min=0.005, cooling=0.9998, max_steps=200000):
    h = fix_binary_sum(h, n, rng)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = t_init
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    
    for step in range(max_steps):
        if time_fn() < 5:
            break
        i = int(rng.choice(ones_idx))
        j = int(rng.choice(zeros_idx))
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
            c5_cur = c5_new
            ones_idx[ones_idx == i] = j
            zeros_idx[zeros_idx == j] = i
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def targeted_swap_search(h, n, rng, time_fn, max_rounds=5000):
    h = fix_binary_sum(h, n, rng)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for rnd in range(max_rounds):
        if time_fn() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        worst_k = int(np.argmax(corr))
        offset = worst_k - n + 1
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        good_i = []
        good_j = []
        for idx in ones:
            hi2 = idx + offset
            if 0 <= hi2 < n and h[hi2] < 0.5:
                good_i.append(idx)
        for idx in zeros:
            hj2 = idx + offset
            if 0 <= hj2 < n and h[hj2] > 0.5:
                good_j.append(idx)
        
        if len(good_i) == 0 or len(good_j) == 0:
            break
        
        best_new_c5 = c5_cur
        best_i, best_j = -1, -1
        n_trials2 = min(80, len(good_i) * len(good_j))
        for _ in range(n_trials2):
            ii = int(rng.choice(good_i))
            jj = int(rng.choice(good_j))
            h[ii] = 0.0
            h[jj] = 1.0
            nc = compute_c5(h)
            h[ii] = 1.0
            h[jj] = 0.0
            if nc < best_new_c5 - 1e-14:
                best_new_c5 = nc
                best_i, best_j = ii, jj
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5_cur = best_new_c5
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def resample_binary(h_old, n_old, n_new, rng):
    if n_new == n_old:
        return fix_binary_sum(h_old, n_new, rng)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    h_bin = (h_interp > 0.5).astype(float)
    return fix_binary_sum(h_bin, n_new, rng)

def multi_resolution_optimize(h_start, n_start, rng, time_fn):
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    
    for n_res in [n_start, min(480, n_start * 2), 360]:
        if time_fn() < 120:
            break
        if n_res % 2 != 0:
            n_res += 1
        if n_res == n_start and len(h_start) == n_res:
            h_cur = fix_binary_sum(h_start, n_res, rng)
        else:
            h_cur = resample_binary(best_h, len(best_h), n_res, rng)
        
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        
        h_t, c5_t = targeted_swap_search(h_cur, n_res, rng, time_fn, max_rounds=3000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
        
        h_d, c5_d = directed_local_search(h_t, n_res, rng, time_fn, max_rounds=5000)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d.copy()
        
        if time_fn() > 60:
            h_sa, c5_sa = simulated_annealing(h_d, n_res, rng, time_fn, t_init=0.3, t_min=0.003, cooling=0.9999, max_steps=50000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
            
            h_d2, c5_d2 = directed_local_search(h_sa, n_res, rng, time_fn, max_rounds=3000)
            if c5_d2 < best_c5:
                best_c5 = c5_d2
                best_h = h_d2.copy()
    
    return best_h, best_c5

def diversify_and_optimize(h_best, n, rng, time_fn, n_trials=8):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    x = np.linspace(0, 2, n, endpoint=False)
    
    for trial in range(n_trials):
        if time_fn() < 40:
            break
        if trial == 0:
            h_init = fix_binary_sum(h_best + rng.uniform(-0.4, 0.4, n), n, rng)
        elif trial == 1:
            h_init = fix_binary_sum(h_best * 0.5 + 0.5 * (np.sin(np.pi * 3 * x + rng.rand() * 6.28) > 0).astype(float), n, rng)
        elif trial == 2:

            h_init = fix_binary_sum(h_raw * 0.3 + h_best * 0.7, n, rng)
        elif trial == 3:
            h_flip = 1.0 - h_best
            h_init = fix_binary_sum(h_flip * 0.6 + h_best * 0.4, n, rng)
        elif trial == 4:
            idx = np.arange(n)
            shift = rng.randint(1, n // 4)
            h_shifted = np.roll(h_best, shift)
            h_init = fix_binary_sum(h_shifted * 0.7 + h_best * 0.3, n, rng)
        elif trial == 5:
            freq = rng.randint(2, 8)
            h_wave = (np.sin(2 * np.pi * freq * x / 2.0 + rng.rand() * 6.28) > 0).astype(float)
            h_init = fix_binary_sum(h_wave * 0.5 + h_best * 0.5, n, rng)
        elif trial == 6:

            h_init = fix_binary_sum(h_best * rng.uniform(0.3, 0.7) + (1.0 - h_best) * rng.uniform(0.3, 0.7), n, rng)
        
        h_d, c5_d = directed_local_search(h_init, n, rng, time_fn, max_rounds=3000)

                best_h = h_sa.copy()
            
            h_d2, c5_d2 = directed_local_search(h_sa, n, rng, time_fn, max_rounds=2000)
            if c5_d2 < best_c5:
                best_c5 = c5_d2
                best_h = h_d2.copy()
    
    return best_h, best_c5
