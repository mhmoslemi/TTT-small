    except Exception:
        return h0.copy()

def subgradient_step(h, step_size, n_active=5):
    n = len(h)
    target_sum = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    indices = np.argsort(corr)[::-1][:n_active]
    nm1 = n - 1
    grad = np.zeros(n)
    j_arr = np.arange(n)
    for k in indices:
        idx_t1 = j_arr + k - nm1
        idx_t2 = j_arr - k + nm1
        v1 = (idx_t1 >= 0) & (idx_t1 < n)
        v2 = (idx_t2 >= 0) & (idx_t2 < n)
        g = np.zeros(n)
        g[v1] += (1.0 - h[idx_t1[v1]])
        g[v2] += -h[idx_t2[v2]]
        grad += g
    grad /= len(indices)
    h_new = h - step_size * grad
    return project_feasible(h_new, target_sum)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed()

    # Phase 1: Smart pair descent (35% budget)
    t_end1 = t_start + 0.35 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    B_w = compute_all_B(h_work, n_w)
    stall = 0
    while elapsed() < t_end1:
        h_work, corr_w, imp = pair_descent_smart(h_work, corr_w, rng, B_w, n_cand=12, n_top_lags=4)
        B_w = compute_all_B(h_work, n_w)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-10:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall > 0 and stall % 60 == 0:
            noise_level = 0.03 + 0.02 * min(stall / 200.0, 1.0)
            h_work = project_to_sum(h_work + rng.uniform(-noise_level, noise_level, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            B_w = compute_all_B(h_work, n_w)
            stall = 0

    # Phase 2: SLSQP with continuation at multiple resolutions (35% budget)
    t_end2 = t_start + 0.70 * budget_s
    resolutions = [best_n, best_n - 40, best_n + 40, 200, 280, 320]
    resolutions = [r for r in resolutions if r >= 80 and r % 2 == 0]
    seen_res = set()
    unique_res = []
    for r in resolutions:
        if r not in seen_res:
            seen_res.add(r)
            unique_res.append(r)
    alphas = [5, 20, 50, 200, 1000, 5000]
    for n_tr in unique_res:
        if elapsed() >= t_end2:
            break
        ts = n_tr / 2.0
        if abs(ts - round(ts)) > 1e-6:
            continue
        h_start = interpolate_h(best_h, best_n, n_tr)
        h_start = project_to_sum(h_start, ts)
        for alpha in alphas:
            if elapsed() >= t_end2:
                break
            tl = min(30.0, time_left() * 0.3)
            if tl < 3.0:
                break
            h_res = slsqp_optimize(h_start, ts, alpha, max_iter=300, time_limit=tl)
            h_res = project_to_sum(h_res, ts)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_tr
            h_start = h_res.copy()

    # Phase 3: Subgradient exploration + pair descent (20% budget)
    t_end3 = t_start + 0.90 * budget_s
    h_sub = best_h.copy()
    n_sub = best_n
    ts_sub = n_sub / 2.0
    step_sz = 0.02
    while elapsed() < t_end3:
        h_sub = subgradient_step(h_sub, step_sz, n_active=5)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            best_n = n_sub
            step_sz = max(0.001, step_sz * 0.95)
        else:
            step_sz = max(0.001, step_sz * 0.98)
        if int((elapsed() - t_start)) % 3 == 0:
            corr_sub = np.correlate(h_sub, 1.0 - h_sub, mode="full")
            B_sub = compute_all_B(h_sub, n_sub)
            h_sub, corr_sub, _ = pair_descent_smart(h_sub, corr_sub, rng, B_sub, n_cand=8, n_top_lags=3)
            h_sub = project_to_sum(h_sub, ts_sub)
            c5_sub = compute_c5(h_sub)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                best_n = n_sub

    # Phase 4: Final intensive pair descent (15% budget)
    t_end4 = t_start + budget_s - 5
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    B_fin = compute_all_B(h_fin, n_fin)
    stall_fin = 0
    while elapsed() < t_end4:
        h_fin, corr_fin, imp = pair_descent_smart(h_fin, corr_fin, rng, B_fin, n_cand=15, n_top_lags=5)
        B_fin = compute_all_B(h_fin, n_fin)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-10:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            stall_fin += 1
        if stall_fin > 0 and stall_fin % 40 == 0:
            noise = 0.02 + 0.01 * min(stall_fin / 100.0, 1.0)
            h_fin = project_to_sum(h_fin + rng.uniform(-noise, noise, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            B_fin = compute_all_B(h_fin, n_fin)
            stall_fin = 0

    # Final SLSQP polish
    if elapsed() < t_end4 - 5:
        n_f = best_n
        h_pol = slsqp_optimize(best_h, n_f / 2.0, 5000, max_iter=200, time_limit=10.0)
        h_pol = project_to_sum(h_pol, n_f / 2.0)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Final pair descent
    if elapsed() < budget_s - 2:
        h_ld = best_h.copy()
        corr_ld = np.correlate(h_ld, 1.0 - h_ld, mode="full")
        B_ld = compute_all_B(h_ld, best_n)
        while elapsed() < budget_s - 1:
            h_ld, corr_ld, _ = pair_descent_smart(h_ld, corr_ld, rng, B_ld, n_cand=15, n_top_lags=5)
            B_ld = compute_all_B(h_ld, best_n)
            c5_ld = compute_c5(h_ld)
            if c5_ld < best_c5:
                best_c5 = c5_ld
                best_h = h_ld.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
