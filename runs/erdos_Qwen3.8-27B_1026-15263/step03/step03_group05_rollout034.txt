def coord_descent_sweep(h, order, n_bisect=50):
    n = len(h)
    n_lags = 2 * n - 1

    for i in order:
        old_h = h[i]

            scale = 0.03 + 0.3 * (1.0 - r)
            h_r = best_h + rng.normal(0, scale, n_r)
            h_r = project_box_sum(h_r, n_r / 2.0)
            n_rounds = max(20, int(100 * (1.0 - r)))
            for _ in range(n_rounds):
                if time_left() < 20:
                    break
                h_r = coord_descent_sweep(h_r, rng.permutation(n_r), n_bisect=40)
            h_r = project_box_sum(h_r, n_r / 2.0)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final high-precision coordinate descent
    if time_left() > 30:
        n_hp = best_n
        for sweep in range(2000):
            if time_left() < 10:
                break
            best_h = coord_descent_sweep(best_h, rng.permutation(n_hp), n_bisect=60)
            if sweep % 50 == 49:
                best_h = project_box_sum(best_h, n_hp / 2.0)
        best_h = project_box_sum(best_h, n_hp / 2.0)
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
