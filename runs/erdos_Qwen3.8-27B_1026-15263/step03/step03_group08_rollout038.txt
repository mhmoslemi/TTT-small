def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:

        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

        cnt = 0
        while c:
            cnt += c & 1
            c >>= 1
        tm[i] = cnt % 2

        for j in range(n):
            # Compute a[k]
            a = np.zeros(2*n - 1)
            valid1 = (j - lags >= 0) & (j - lags < n)
            a[valid1] += (1.0 - h[j - lags[valid1]])
            valid2 = (j + lags >= 0) & (j + lags < n)
            a[valid2] -= h[j + lags[valid2]]
            
            # Constant term: c[k] = corr[k] - a[k] * h[j]
            c_const = corr - a * h[j]
            
            # Find best x in [0,1] minimizing max_k (a[k]*x + c_const[k])
            xs = np.linspace(0.0, 1.0, n_samples)
            new_corrs = np.outer(xs, a) + c_const[np.newaxis, :]
            maxes = np.max(new_corrs, axis=1)
            idx = np.argmin(maxes)
            best_x = xs[idx]
            
            if best_x != h[j]:
                h[j] = best_x
                # Update corr: corr[k] = a[k] * best_x + c_const[k]
                corr = a * best_x + c_const
                total_improvement += 1
        
        # Project to satisfy sum constraint
        h = project_to_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h, np.max(corr)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

            excess = corr[kk] - corr[k_max]
            if excess > -0.1 * c5_cur:  # only add if close to max
                g += 0.5 * ga
