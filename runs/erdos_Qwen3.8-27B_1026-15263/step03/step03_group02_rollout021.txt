            n_mask = np.sum(mask)
            if n_mask == 0:

            h[mask] += diff / n_mask
        h = np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Compute gradient vectorized
    # corr[k] for k = 0..2n-2, lag s = k - (n-1)
    # For s >= 0: corr = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
    #   d/dh[i] = (1-h[i+s]) for i < n-s [as first factor]
    #             -h[i-s] for i >= s [as second factor, where j=i-s, h[j+s]=h[i], d/dh[i] of h[j]*(1-h[j+s]) = -h[j] = -h[i-s]]
    # For s < 0, let k = -s > 0: corr = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
    #   = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
    #   d/dh[i] = (1-h[i-k]) for i >= k [as first factor h[i]]
    #             -h[i+k] for i < n-k [as second factor, where the term is h[j+k]*(1-h[j]), 
    #               d/dh[i] when i = j: -h[j+k] = -h[i+k]]
    
    # Let me think of this differently.
    # Full correlation: corr_full[m] = sum_i a[i] * b[i + m - (n-1)]
    # where a = h, b = 1-h
    # So corr_full[m] = sum_i h[i] * (1 - h[i + m - n + 1])
    # Let s = m - n + 1 (the lag)
    # corr_full[m] = sum_i h[i] * (1 - h[i+s])
    # where i ranges over valid indices: 0 <= i < n and 0 <= i+s < n
    #   → max(0, -s) <= i < min(n, n-s)
    
    # d(corr_full[m])/d(h[p]) for a fixed p:
    # Term 1: h[p] appears as the first factor when i = p, valid if 0 <= p < n and 0 <= p+s < n
    #   → contributes (1 - h[p+s]) if valid
    # Term 2: h[p] appears as the second factor (inside 1-h) when i+s = p, i.e., i = p-s
    #   → valid if 0 <= p-s < n and 0 <= p < n
    #   → contributes -h[p-s] if valid
    #   (because d/dh[p] of (1-h[p]) = -1, and the other factor is h[p-s])
    
    # So grad[p] = sum_m w[m] * [(1-h[p+s_m]) * 1{valid1} + (-h[p-s_m]) * 1{valid2}]
    # where s_m = m - n + 1
    
    # This is still O(n * 2n) = O(n²) which is fine for n=240.
    
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)  # s values: -(n-1), ..., 0, ..., n-1
    for m in range(2*n-1):
        if w[m] < 1e-16:
            continue
        s = lags[m]
        wm = w[m]
        # Valid i range: max(0, -s) to min(n, n-s) - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        # First factor: h[i] for i in [i_lo, i_hi)
        # d/dh[p] contributes w[m] * (1-h[p+s]) if p in [i_lo, i_hi) and p+s in [0, n)
        # p in [i_lo, i_hi) means max(0,-s) <= p < min(n, n-s)
        # p+s in [0,n) means -s <= p < n-s
        # Combined: max(0,-s) <= p < min(n, n-s) AND -s <= p < n-s
        # = max(0,-s) <= p < min(n, n-s) (since max(0,-s) >= -s and min(n,n-s) <= n-s)
        # So first factor is valid for p in [max(0,-s), min(n, n-s))
        # Contribution: (1 - h[p+s])
        
        # Second factor: h[p] appears as h[i+s] where i = p-s
        # Valid if i in [i_lo, i_hi) → i_lo <= p-s < i_hi → i_lo+s <= p < i_hi+s
        # And p in [0, n)
        # Combined: max(0, i_lo+s) <= p < min(n, i_hi+s)
        # i_lo + s = max(0,-s) + s = max(s, 0)
        # i_hi + s = min(n, n-s) + s = min(n+s, n) = n (since s can be negative, min(n+s, n) = n if s>=0, n+s if s<0)
        # Wait, if s >= 0: i_hi = n-s, i_hi+s = n. So p < n.
        # If s < 0: i_hi = n, i_hi+s = n+s. So p < n+s.
        # So: p in [max(s,0), min(n, n+max(0,s)))... 
        # Let me just be more careful.
        # i_lo = max(0, -s), i_hi = min(n, n-s)
        # p-s in [i_lo, i_hi) → p in [i_lo+s, i_hi+s) = [max(0,-s)+s, min(n,n-s)+s)
        # max(0,-s)+s = max(s,0)
        # min(n,n-s)+s: if s>=0, n-s+s=n; if s<0, n+s
        # So p in [max(s,0), n) if s>=0, or p in [0, n+s) if s<0
        # Combined with p in [0,n):
        # If s>=0: p in [s, n)
        # If s<0: p in [0, n+s)
        # Contribution: -h[i] where i = p-s, so -h[p-s]
        
        if s >= 0:
            # First factor: p in [0, n-s), contribution (1-h[p+s])
            if n - s > 0:
                p1 = np.arange(n - s)
                grad[p1] += wm * (1.0 - h[p1 + s])
            # Second factor: p in [s, n), contribution -h[p-s]
            if n - s > 0:
                p2 = np.arange(s, n)
                grad[p2] += wm * (-h[p2 - s])
        else:
            k = -s  # k > 0
            # First factor: p in [k, n), contribution (1-h[p-k])
            if n - k > 0:
                p1 = np.arange(k, n)
                grad[p1] += wm * (1.0 - h[p1 - k])
            # Second factor: p in [0, n-k), contribution -h[p+k]
            if n - k > 0:
                p2 = np.arange(n - k)
                grad[p2] += wm * (-h[p2 + k])
    
    return obj, grad

        return smooth_max_obj_grad(x, alpha)[0]

        return smooth_max_obj_grad(x, alpha)[1]

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nm = np.sum(mask)
        if nm > 0:
            h[mask] += diff / nm
        else:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            nm = np.sum(mask)
            if nm == 0:
                break
            h[mask] += diff / nm
