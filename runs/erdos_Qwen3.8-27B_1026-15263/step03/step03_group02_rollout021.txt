import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def grad_descent_refine(h0, n, target_sum, alpha, n_iters, lr, t_budget):

            break
        obj, grad = smooth_max_and_grad(h, alpha)
        h = h - lr * grad
        h = project_to_sum(h, target_sum)

def qr_construction(n, rng):

    def is_prime(m):
        if m < 2:
            return False
        for i in range(2, int(m**0.5) + 1):
            if m % i == 0:
                return False
        return True
    
    p = n - 1
    while p > 2:
        if is_prime(p) and p % 4 == 3:
            break
        p -= 1
    
    if p < 7:
        # Fallback: random
        h = np.zeros(n)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        return h

    residues = set()
    for x in range(1, p):
        residues.add(x * x % p)
    
    # Create sequence
    h = np.zeros(n)
    for i in range(1, p):

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
        'jac': lambda x: np.ones(n)
    }]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def grad_descent_refine(h0, n, target_sum, alpha, n_iters, lr, t_budget):
    h = h0.copy()
    for i in range(n_iters):
        if time.time() > t_budget:
            break
        obj, grad = smooth_max_and_grad(h, alpha)
        h = h - lr * grad
        h = project_to_sum(h, target_sum)

    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})

        ts_try = n_try / 2.0
        h_interp = interpolate_up(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = make_binary(h_interp, n_try, int(ts_try), rng)
        sa_end3 = min(t_start + budget_s - 50, time.time() + 45)
        h_try_sa, c5_try_sa = binary_sa(h_try, n_try, int(ts_try), sa_end3, rng,
                                         block_sizes=[1, 1, 1, 2, 2, 3])
        if c5_try_sa < best_c5:
            best_c5 = c5_try_sa
            best_h = h_try_sa.copy()
            best_n = n_try
        elif remaining() > 40:
            h_ct = grad_descent_refine(h_try_sa, n_try, ts_try, 100,
                                        n_iters=800, lr=4.0/(100*n_try),
                                        t_budget=time.time() + 25)
            c5_ct = compute_c5(h_ct)
            if c5_ct < best_c5:
                best_c5 = c5_ct
                best_h = h_ct.copy()
                best_n = n_try

    if remaining() > 50:
        for alpha in [3000, 15000]:
            if remaining() < 25:
                break
            h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    if remaining() > 25:
        h_final_bin = make_binary(best_h, best_n, int(best_n / 2.0), rng)
        h_final, c5_final = binary_sa(h_final_bin, best_n, int(best_n / 2.0),
                                       t_start + budget_s - 3, rng,
                                       block_sizes=[1, 1, 1, 1, 2, 2, 3])
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
