    n_cur = best_n
    ts_cur = n_cur / 2.0

    phase1_time = min(240, budget_s * 0.18)
    if time_left() < phase1_time:
        phase1_time = max(60, time_left() * 0.5)

    t_p1_start = time.time()
    alphas_p1 = [30, 80, 200, 500, 1500, 4000]
    for alpha in alphas_p1:
        if time.time() - t_p1_start > phase1_time:
            break
        n_iter = max(800, int(200 * (time_left() / max(1, phase1_time))))
        h_new = pg_optimize(best_h, alpha, n_iter, ts_cur, lr_init=0.008)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time.time() - t_p1_start < phase1_time and time_left() > 100:
        h_pol = active_set_polish(best_h, n_iter=2000, ts=ts_cur)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    t_p2_start = time.time()
    p2_budget = min(300, budget_s * 0.22)
    n_high = 720
    ts_high = n_high / 2.0
    h_up = upsample_h(best_h, n_high)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
        best_n = n_high

    alphas_p2 = [50, 150, 500, 1500, 4000, 10000]
    for alpha in alphas_p2:
        if time.time() - t_p2_start > p2_budget:
            break
        n_iter = max(600, int(150 * (time_left() / max(1, p2_budget))))
        h_opt = pg_optimize(h_up, alpha, n_iter, ts_high, lr_init=0.006)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_high
        h_up = h_opt

    if time.time() - t_p2_start < p2_budget and time_left() > 80:
        h_pol2 = active_set_polish(h_up, n_iter=2000, ts=ts_high)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
            best_n = n_high

    t_p3_start = time.time()
    p3_budget = min(200, budget_s * 0.15)
    h_cd, c5_cd = coord_descent_exact(best_h, best_n / 2.0, n_steps=60000, rng=rng)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
    if time.time() - t_p3_start < p3_budget and time_left() > 60:
        h_post = pg_optimize(best_h, 3000, 1500, best_n / 2.0, lr_init=0.004)
        c5_post = compute_c5(h_post)
        if c5_post < best_c5:
            best_c5 = c5_post
            best_h = h_post.copy()

    t_p4_start = time.time()
    p4_budget = min(180, budget_s * 0.12)
    for restart in range(8):
        if time.time() - t_p4_start > p4_budget or time_left() < 80:
            break
        x = np.linspace(0, 2, best_n, endpoint=False)
        h_r = 0.5 + 0.3 * np.sin(2.0 * np.pi * (rng.randint(1, 5)) * x / 2.0 + rng.uniform(0, 6.28))
        h_r += 0.15 * np.sin(2.0 * np.pi * (rng.randint(3, 12)) * x / 2.0 + rng.uniform(0, 6.28))
        h_r = project_to_sum(h_r, best_n / 2.0)
        for alpha in [100, 800, 3000]:
            if time.time() - t_p4_start > p4_budget:
                break
            h_r = pg_optimize(h_r, alpha, 800, best_n / 2.0, lr_init=0.005)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    h_mixed = 0.6 * best_h + 0.4 * upsample_h(best_h, best_n)
    h_mixed = project_to_sum(h_mixed, best_n / 2.0)
    h_mixed = pg_optimize(h_mixed, 2000, 1000, best_n / 2.0, lr_init=0.004)
    c5_m = compute_c5(h_mixed)
    if c5_m < best_c5 and time_left() > 40:
        best_c5 = c5_m
        best_h = h_mixed.copy()

    t_p5_start = time.time()
    p5_budget = min(120, budget_s * 0.1)
    h_cd2, c5_cd2 = coord_descent_exact(best_h, best_n / 2.0, n_steps=40000, rng=rng)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()
    if time.time() - t_p5_start < p5_budget and time_left() > 50:
        h_fin = pg_optimize(best_h, 8000, 2000, best_n / 2.0, lr_init=0.003)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        h_as = active_set_polish(best_h, n_iter=3000, ts=best_n / 2.0)
        c5_as = compute_c5(h_as)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()

    if time_left() > 30:
        h_fin2 = proj_grad_final(best_h, best_n, time_left() - 30)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)

def proj_grad_final(h0, n, time_limit_s):
    t0 = time.time()
    ts = n / 2.0
    h = h0.copy()
    obj_prev, grad = smooth_max_and_grad(h, 5000)
    lr = 0.003
    stall = 0
    max_iter = 5000
    for i in range(max_iter):
        if time.time() - t0 > time_limit_s - 5:
            break
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new = smooth_max_and_grad(h_new, 5000)[0]
        steps = 0
        while obj_new >= obj_prev - 1e-15 and steps < 20:
            lr *= 0.4
            h_new = project_to_sum(h - lr * grad, ts)
            obj_new = smooth_max_and_grad(h_new, 5000)[0]
            steps += 1
        if steps < 20:
            h = h_new
            obj_prev = obj_new
            stall = 0
            if steps == 0:
                lr = min(lr * 1.03, 0.2)
        else:
            stall += 1
            if stall > 50:
                break
        _, grad = smooth_max_and_grad(h, 5000)
    return h
