    h = h.copy()
    dx = 2.0/n
    c5_cur = compute_c5(h)
    corr_cur = np.correlate(h, 1-h, mode="full")
    
    for round_idx in range(max_rounds):
        if time_fn() < 10:
            break
        
        # Find worst shift
        worst_idx = np.argmax(corr_cur)
        worst_shift = worst_idx - (n-1)  # actual lag
        
        # Find positions contributing to the worst shift
        # corr[wrong_idx] = sum h[i]*(1-h[i-shift]) or similar
        # Let's just try all flips that reduce the worst correlation
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improvement = 0
        best_flip = None
        
        # Try targeted flips
        # For the worst shift, try flipping positions that directly reduce it
        if worst_shift > 0:
            # corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # To reduce: flip h[i+k] from 1 to 0 where h[i]=0, or flip h[i] from 0 to 1 where h[i+k]=1
            candidates = []
            for i in range(n - worst_shift):
                if h[i + worst_shift] == 1 and h[i] == 0:
                    candidates.append((i + worst_shift, 0))  # flip this 1 to 0
                if h[i] == 0 and h[i + worst_shift] == 1:
                    candidates.append((i, 1))  # flip this 0 to 1
        
        # Also try random flips
        for _ in range(200):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            if i == j:
                continue
            h[i] = 0; h[j] = 1
            c5_new = np.max(np.correlate(h, 1-h, mode="full")) * dx
            if c5_new < c5_cur - 1e-12:
                best_improvement = c5_cur - c5_new
                best_flip = (i, j)
                corr_cur = np.correlate(h, 1-h, mode="full")
                c5_cur = c5_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break
            else:
                h[i] = 1; h[j] = 0
        
        if best_flip is None:
            # No improvement found, try harder or break
            if round_idx > 20:
                break

Actually, I recall that np.correlate(a, v) computes the cross-correlation:

For mode='full', tau ranges from -(Nv-1) to Na-1 where Na, Nv are the lengths.
If Na = Nv = n, tau ranges from -(n-1) to n-1.
The output is indexed from 0 to 2n-2, with output[k] = c[tau] where tau = k - (n-1).

Now, the gradient of output[k] with respect to h[j]:
output[k] = sum_n h[n + k - (n-1)] * (1-h[n])

Let tau = k - (n-1). Then output[k] = sum_n h[n+tau] * (1-h[n])

d(output[k])/dh[j] = sum_n [d(h[n+tau])/dh[j] * (1-h[n]) + h[n+tau] * d(1-h[n])/dh[j]]
= sum_n [δ_{n+tau, j} * (1-h[n]) + h[n+tau] * (-δ_{n,j})]
= (1-h[j-tau]) * [if 0≤j-tau≤n-1] - h[j+tau] * [if 0≤j+tau≤n-1]

Wait, let me be more careful:
Term 1: δ_{n+tau, j} means n = j-tau. Valid if 0 ≤ j-tau ≤ n-1.
  Contribution: (1-h[j-tau])

Term 2: -δ_{n,j} means n = j. Valid if 0 ≤ j ≤ n-1 (always true) and 0 ≤ j+tau ≤ n-1.
  Contribution: -h[j+tau]

So: d(output[k])/dh[j] = (1-h[j-tau]) * I(0≤j-tau≤n-1) - h[j+tau] * I(0≤j+tau≤n-1)

where tau = k - (n-1).

For the max function, the subgradient at h is:
grad[j] = d(output[k*])/dh[j] * dx

where k* is the index of the maximum.

This gives us a subgradient method:
h ← project(h - lr * grad)

where project is projection onto {h: 0≤h≤1, sum(h)=n/2}.

This is similar to what the previous code did with the smooth max, but using the exact subgradient. The advantage is that we don't need the smooth parameter alpha, but the disadvantage is that the subgradient is only valid at the achieving k*, which can change discretely.

A practical approach is **subgradient with averaging** (Mirzadeh et al.): maintain a running average of the iterates.

Let me also try **polyak step-size**:
lr_t = (C_target - f_t) / ||grad_t||^2

where C_target is a lower bound (or the current best value) and f_t is the current objective.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_of_max_corr(h):
    """Compute subgradient of max_k corr_k(h) at h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    tau = k_star - (n - 1)
    
    # d(output[k_star])/dh[j] = (1-h[j-tau])*I(0<=j-tau<=n-1) - h[j+tau]*I(0<=j+tau<=n-1)
    grad = np.zeros(n)
    
    # Term 1: (1-h[j-tau]) for j where 0 <= j-tau <= n-1
    # j-tau in [0, n-1] means j in [tau, tau+n-1] intersect [0, n-1]
    j_start1 = max(0, tau)
    j_end1 = min(n - 1, tau + n - 1)
    if j_start1 <= j_end1:
        idx1 = np.arange(j_start1, j_end1 + 1)
        grad[idx1] += (1.0 - h[idx1 - tau])
    
    # Term 2: -h[j+tau] for j where 0 <= j+tau <= n-1
    # j+tau in [0, n-1] means j in [-tau, n-1-tau] intersect [0, n-1]
    j_start2 = max(0, -tau)
    j_end2 = min(n - 1, n - 1 - tau)
    if j_start2 <= j_end2:
        idx2 = np.arange(j_start2, j_end2 + 1)
        grad[idx2] -= h[idx2 + tau]
    
    return grad, k_star, corr

def binary_local_search_fast(h, n, rng, time_fn, max_rounds=5000):
    """Fast binary local search using direct correlation computation."""
    h = h.astype(float).copy()
    # Round to binary
    h = (h > 0.5).astype(float)
    # Adjust sum
    curr = h.sum()
    target = n / 2.0
    if curr != target:
        if curr > target:
            excess = int(curr - target)
            ones = np.where(h == 1)[0]
            rng.shuffle(ones)
            h[ones[:excess]] = 0
        else:
            deficit = int(target - curr)
            zeros = np.where(h == 0)[0]
            rng.shuffle(zeros)
            h[zeros[:deficit]] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5_cur = np.max(corr) * dx
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stall = 0
    for rnd in range(max_rounds):
        if time_fn() < 15:
            break
        
        improved = False
        n_trials = 300
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1 - h, mode="full")
            new_c5 = np.max(new_corr)
            if new_c5 < np.max(corr):
                corr = new_corr
                c5_cur = np.max(corr) * dx
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                stall = 0
            else:
                h[i] = 1
                h[j] = 0

    return h, c5_cur

    """Simulated annealing for binary h."""
    h = h.astype(float).copy()

            t *= cooling ** 10000

            continue

                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0

def continuous_refine_subgrad(h, n, time_fn, max_iter=20000):
    """Subgradient method with Polyak step for continuous h."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    
    f_current = compute_c5(h)
    best_f = f_current
    best_h = h.copy()
    
    # Running average for ergodic subgradient method
    h_avg = h.copy()
    
    for it in range(max_iter):
        if time_fn() < 15:
            break
        
        grad, k_star, corr = subgradient_of_max_corr(h)
        grad *= dx  # Scale to match objective
        
        # Polyak step size
        # We need a lower bound; use 0.3 for now (we know C5 > 0.3)
        # Actually, use the current best as an upper bound estimate for the gap
        f_val = np.max(corr) * dx
        gap = f_val - 0.3  # crude lower bound
        grad_norm_sq = np.sum(grad ** 2)
        if grad_norm_sq < 1e-15:
            break
        lr = min(gap / grad_norm_sq, 0.5)
        
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        
        f_new = compute_c5(h_new)
        h_avg = (it * h_avg + h_new) / (it + 1)
        f_avg = compute_c5(h_avg)
        
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            f_current = f_new
        
        if f_avg < best_f:
            best_f = f_avg
            best_h = h_avg.copy()
        
        h = h_new
        f_current = f_new
    
    return best_h, best_f
