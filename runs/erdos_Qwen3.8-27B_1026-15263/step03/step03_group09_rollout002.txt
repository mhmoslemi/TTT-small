def project_to_sum(h, target_sum):

        if abs(diff) < 1e-15:
            break

            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_gradient(h, n, top_k=10):
    """Compute top-k subgradient, vectorized."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for m in top_idx:
        L = m - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            g[:n+L] += (1.0 - h[-L:])
            g[-L:] -= h[:n+L]
    g /= len(top_idx)
    return g, corr, top_idx

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for k_idx in range(len(corr)):
        L = k_idx - (n - 1)
        if w[k_idx] < 1e-10:

            g[L:] += w[k_idx] * (1.0 - h[:n-L])
            g[:n-L] -= w[k_idx] * h[L:]
        else:
            g[:n+L] += w[k_idx] * (1.0 - h[-L:])
            g[-L:] -= w[k_idx] * h[:n+L]

def pair_coord_descent(h, n, target_sum, max_rounds, rng, n_pairs_per_round):
    """Pair-based coordinate descent: swap mass between pairs."""
    h = h.copy()
    c5_cur = compute_c5(h)
    n_pts = n
    
    for round_idx in range(max_rounds):
        improved = False
        # Pick random pairs to try
        for _ in range(n_pairs_per_round):
            i = rng.randint(n_pts)
            j = rng.randint(n_pts)
            if i == j:
                continue
            
            # Try transferring mass from i to j (and vice versa)
            for sign in [1, -1]:
                for delta in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                    h_test = h.copy()
                    h_test[i] += sign * delta
                    h_test[j] -= sign * delta
                    if h_test[i] < -1e-10 or h_test[i] > 1 + 1e-10:
                        continue
                    if h_test[j] < -1e-10 or h_test[j] > 1 + 1e-10:
                        continue
                    h_test = np.clip(h_test, 0, 1)
                    c5_test = compute_c5(h_test)
                    if c5_test < c5_cur - 1e-15:
                        h = h_test
                        c5_cur = c5_test
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return h, c5_cur

    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
    else:

    if remaining_time(t_start, budget_s) > 15:
        h_f = best_h.copy()
        n_pr_f = min(50, int(remaining_time(t_start, budget_s) * 0.5))
        h_f, c5_f = pair_coord_descent(h_f, n_f, ts_f, n_pr_f, rng, 10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if remaining_time(t_start, budget_s) > 10:
        h_f = best_h.copy()
        h_bin = threshold_restart(h_f, n_f, ts_f)
        h_bin, c5_bin = optimize_momentum(h_bin, n_f, ts_f, 2000, 0.03, 10, 0.85)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    h_final = best_h.copy()
    n_final = len(h_final)
    ts_final = n_final / 2.0
    h_final = project_to_sum(h_final, ts_final)
    h_final = np.clip(h_final, 0.0, 1.0)
    h_final = project_to_sum(h_final, ts_final)
    c5_final = compute_c5(h_final)

    return (h_final, c5_final, n_final)
