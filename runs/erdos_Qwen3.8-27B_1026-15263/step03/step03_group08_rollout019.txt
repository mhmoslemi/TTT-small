    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)

    s = np.sum(exp_vals)

    sum_dev = np.sum(h) - n / 2.0

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, penalty_coeff, target_sum):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    sum_dev = np.sum(h) - target_sum
    sm_val += penalty_coeff * sum_dev * sum_dev
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    g += 2.0 * penalty_coeff * sum_dev
    return sm_val, g

def active_lag_obj_grad(h, top_k, penalty_coeff, target_sum):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    obj = np.sum(corr[top_idx])

    g = np.zeros(n)

        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + l:hi2 + l]
    g += 2.0 * penalty_coeff * sum_dev
    return obj, g

def lbfgsb_run(h0, n, alpha, penalty_coeff, target_sum, maxiter):
    def obj(x):
        v, _ = smooth_max_obj_grad(x, alpha, penalty_coeff, target_sum)
        return v
    def grad(x):
        _, g = smooth_max_obj_grad(x, alpha, penalty_coeff, target_sum)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)

        return h0.copy()
