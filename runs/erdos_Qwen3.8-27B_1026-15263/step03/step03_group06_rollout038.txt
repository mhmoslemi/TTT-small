def subgradient_step(h, n, target_sum, n_top=10, lr=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top n_top lags
    top_indices = np.argsort(corr)[-n_top:]
    top_indices = top_indices[np.argsort(corr[top_indices])]  # ascending
    weight = np.zeros(len(corr))
    max_val = corr[top_indices[-1]]
    for idx in top_indices:
        weight[idx] = np.exp((corr[idx] - max_val) * 10)  # softmax-like
    weight /= weight.sum()
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for w, k_idx in zip(weight, top_indices):
        if w < 1e-6:
            continue
        k = k_idx - (n - 1)  # actual lag
        # ∂c_k/∂h[j] = (1-h[j-k]) * 1[0 ≤ j-k < n] - h[j+k] * 1[0 ≤ j+k < n]
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += w * g
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new, grad

        top_idx = np.argpartition(corr, -n_top)[-n_top:]

            else:

                grad[ka:] -= w * y[:n_y-ka]

        # Also enforce sum constraint on y
        y = project_to_sum(y, target_sum)

        t_val = t_new

    corr = np.correlate(h, 1.0 - h, mode='full')
    if K < len(corr):
        top_idx = np.argpartition(corr, -K)[-K:]
    else:
        top_idx = np.arange(len(corr))
    top_corr = corr[top_idx]
    m = float(np.max(corr))
    shifted = alpha * (top_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    w = exp_vals / s
    sm_val = m + np.log(s / K) / alpha
    grad = np.zeros(n)
    center = n - 1
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        k = int(top_idx[i]) - center
        if k >= 0:
            n_valid = n - k
            grad[k:] += wi * (1.0 - h[:n_valid])
            grad[:n_valid] -= wi * h[k:]
        else:
            ka = -k
            n_valid = n - ka
            grad[:n_valid] += wi * (1.0 - h[ka:])
            grad[ka:] -= wi * h[:n_valid]
    return sm_val, grad

def nesterov_descent(h0, target_sum, n_iters, alpha, K, lr_init):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_val = 1.0
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    for it in range(n_iters):
        sm_val, g = topk_grad(y, alpha, K)
        lr = lr_init / t_val
        h_new = y - lr * g
        h_new = project_to_sum(h_new, target_sum)
        obj = compute_c5(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = project_to_sum(y, target_sum)
        h = h_new
        t_val = t_new
    return best_h, best_obj

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    K = min(2 * n, 200)
    def obj(x):
        v, _ = topk_grad(x, alpha, K)
        return v
    def grad(x):
        _, g = topk_grad(x, alpha, K)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent(h0, n, target_sum, n_rounds, rng, xatol=1e-6):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try to optimize h[i] via line search
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            if hi - lo < 1e-8:
                continue
            best_v = c5_cur
            best_x = h[i]
            # Use bounded minimization
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': xatol})
                if res.fun < best_v - 1e-12:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            n = len(best_h)
            ts = n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        n = 360
        ts = n / 2.0
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = len(best_h)
    ts1 = n1 / 2.0
    h_cur = best_h.copy()

    # Phase 1: Projected gradient descent with momentum (fast global-ish search)
    if time_left() > 180:
        n_iters = min(80000, int(time_left() * 15))
        h_pgd, c5_pgd = proj_grad_descent(h_cur, ts1, n_iters, alpha=15.0, K=40, lr_init=0.03)
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()
            h_cur = h_pgd

    if time_left() > 150:
        n_iters2 = min(50000, int(time_left() * 12))
        h_nm, c5_nm = nesterov_momentum(h_cur, ts1, n_iters2, alpha=30.0, K=60, lr_init=0.02, beta=0.7)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()
            h_cur = h_nm

    # Phase 2: SLSQP with increasing alpha
    alphas_p2 = [50.0, 150.0, 400.0, 1000.0, 3000.0]
    for alpha in alphas_p2:
        if time_left() < 50:
            break
        K = min(2 * n1, 300)
        maxiter = max(40, min(250, int(time_left() * 0.4)))
        h_opt = slsqp_topk(h_cur, n1, ts1, alpha, K, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    # Phase 3: Coordinate descent with line search
    if time_left() > 100:
        n_cd = min(15, max(3, int(time_left() // 12)))
        h_cd, c5_cd = coord_descent_ls(h_cur, n1, ts1, n_cd, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_cur = h_cd

    # Phase 4: Fixed-step coordinate descent
    if time_left() > 60:
        deltas = [-0.015, -0.008, -0.004, -0.002, 0.002, 0.004, 0.008, 0.015]
        h_cd2, c5_cd2 = coord_descent_fixed(h_cur, n1, ts1, 25, rng, deltas)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h_cur = h_cd2

    # Phase 5: Perturbation + re-optimization (multi-start)
    if time_left() > 180:
        for pert_round in range(6):
            if time_left() < 50:
                break
            sigma = 0.04 / (pert_round + 1)
            h_pert = best_h + rng.normal(0, sigma, size=n1)
            h_pert = project_to_sum(h_pert, ts1)
            c5_pert = compute_c5(h_pert)
            h_work = h_pert if c5_pert < best_c5 else best_h.copy()

            for alpha in [300.0, 1500.0]:
                if time_left() < 35:
                    break
                K = min(2 * n1, 250)
                h_opt = slsqp_topk(h_work, n1, ts1, alpha, K, 150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_work = h_opt
                else:
                    h_work = h_opt

            if time_left() > 35:
                h_cd, c5_cd = coord_descent_fixed(h_work, n1, ts1, 8, rng,
                                                   [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005])
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 6: Higher resolution
    n2 = n1 * 2
    if time_left() > 250 and n2 <= 1440:
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = project_to_sum(h_interp, ts2)
        h_cur2 = h_interp
        c5_cur2 = compute_c5(h_interp)

        for alpha in [80.0, 300.0, 1000.0, 3000.0]:
            if time_left() < 60:
                break
            K = min(2 * n2, 400)
            maxiter = max(30, min(200, int(time_left() * 0.35)))
            h_opt2 = slsqp_topk(h_cur2, n2, ts2, alpha, K, maxiter)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
            h_cur2 = h_opt2

        if time_left() > 80:
            n_cd2 = min(8, max(3, int(time_left() // 15)))
            h_cd2, c5_cd2 = coord_descent_ls(h_cur2, n2, ts2, n_cd2, rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                h_cur2 = h_cd2

        if time_left() > 50:
            h_cd3, c5_cd3 = coord_descent_fixed(h_cur2, n2, ts2, 15, rng,
                                                 [-0.008, -0.004, -0.002, 0.002, 0.004, 0.008])
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()

        if time_left() > 40:
            n_iters3 = min(30000, int(time_left() * 8))
            h_pgd2, c5_pgd2 = proj_grad_descent(h_cur2, ts2, n_iters3, alpha=50.0, K=80, lr_init=0.02)
            if c5_pgd2 < best_c5:
                best_c5 = c5_pgd2
                best_h = h_pgd2.copy()
                h_cur2 = h_pgd2

    # Phase 7: Final precise SLSQP with full gradient
    if best_h is not None and time_left() > 30:
        nf = len(best_h)
        tsf = nf / 2.0
        h_fin = best_h.copy()
        for alpha in [5000.0, 20000.0]:
            if time_left() < 20:
                break
            h_pol = slsqp_full(h_fin, nf, tsf, alpha, 200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
                h_fin = h_pol

    # Phase 8: Final coordinate descent polish
    if best_h is not None and time_left() > 25:
        nf = len(best_h)
        tsf = nf / 2.0
        h_cd, c5_cd = coord_descent_ls(best_h, nf, tsf, 5, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final validation
    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
