def compute_c5(h):

    dx = 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

            break

            idxs = rng.choice(n_pop, size=3, replace=False)
            return idxs[int(np.argmin([fitness[i] for i in idxs]))]

            p2 = tournament()

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:

    else:

            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)

    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)

    cs = valid_s * h[sc] * w[None, :]

    best_h = h.copy()
    best_c5 = c5_cur
    n_corr = 2 * n - 1
    for it in range(max_iters):
        t = t_start * (t_end / t_start) ** (it / max_iters)
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        if rng.random() < 0.15:
            m_star = int(rng.choice(n_corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        candidates_inc = np.where(can_inc)[0]
        candidates_dec = np.where(can_dec)[0]
        i_inc = int(rng.choice(candidates_inc))
        i_dec = int(rng.choice(candidates_dec))
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            continue
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    return best_h, best_c5

def genetic_algorithm(h0, target_sum, rng, n_pop=20, n_gen=150, t_budget=300):
    n = len(h0)
    t_start = time.time()
    pop = [h0.copy()]
    for i in range(1, n_pop):
        noise = rng.normal(0, 0.04 + 0.03 * (1.0 - i / float(n_pop)), n)
        h_mut = np.clip(h0 + noise, 0, 1)
        pop.append(project_to_sum(h_mut, target_sum))
    fitness = [compute_c5(h) for h in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    prev_best_a = best_c5
    stall_count = 0
    gen = 0
    while gen < n_gen:
        if time.time() - t_start > t_budget:
            break
        def tournament():
            idxs = rng.choice(n_pop, size=3, replace=False)
            return int(idxs[np.argmin([fitness[i] for i in idxs])])
        new_pop = [best_h.copy()]
        for i in range(1, n_pop):
            p1 = tournament()
            p2 = tournament()
            alpha = rng.uniform(0.3, 0.7)
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]
            child = np.clip(child, 0, 1)
            child = project_to_sum(child, target_sum)
            if rng.random() < 0.35:
                n_mut = max(5, n // 15)
                mask = rng.choice(n, size=n_mut, replace=False)
                child[mask] += rng.normal(0, 0.025, n_mut)
                child = np.clip(child, 0, 1)
                child = project_to_sum(child, target_sum)
            new_pop.append(child)
        fitness = [compute_c5(x) for x in new_pop]
        best_idx = int(np.argmin(fitness))
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = new_pop[best_idx].copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 20:
            mid = n_pop // 3
            pop[mid:2 * mid] = []
            for i in range(mid):
                noise = rng.normal(0, 0.06, n)
                h_m = np.clip(best_h + noise, 0, 1)
                pop.insert(mid + i, project_to_sum(h_m, target_sum))
            fitness = [compute_c5(x) for x in pop]
            stall_count = 0
        pop = new_pop
        gen += 1
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    for alpha in [50, 200, 1000, 5000]:
        if time_left() < 80:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    for alpha in [100, 500, 2000]:
        if time_left() < 80:
            break
        h_new = lbfgsb_local(best_h, n_parent, ts_parent, alpha, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 150:
        h_sa, c5_sa = simulated_annealing(best_h, ts_parent, rng,
                                          max_iters=40000, t_start=0.25, t_end=0.005,
                                          step_frac=0.1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_cd, c5_cd = coordinate_descent(best_h, ts_parent, max_iters=5000, step_frac=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 120:
        h_ml, c5_ml = multi_lag_coord_descent(best_h, ts_parent, max_iters=8000,
                                              step_frac=0.08, n_top_lags=5)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    for n_tr in [720, 900, 1080]:
        if time_left() < 200:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        for alpha in [100, 500, 2000]:
            if time_left() < 60:
                break
            h_opt = lbfgsb_local(h_interp, n_tr, ts_tr, alpha, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_interp:
                h_interp = h_opt
                c5_interp = c5_opt
        if time_left() > 60:
            h_cd, c5_cd = coordinate_descent(h_interp, ts_tr, max_iters=4000, step_frac=0.08)
            if c5_cd < c5_interp:
                h_interp = h_cd
                c5_interp = c5_cd
        if time_left() > 60:
            h_ml, c5_ml = multi_lag_coord_descent(h_interp, ts_tr, max_iters=4000,
                                                  step_frac=0.06, n_top_lags=5)
            if c5_ml < c5_interp:
                h_interp = h_ml
                c5_interp = c5_ml
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

    if time_left() > 300:
        n_ga = min(best_n, 360)
        ts_ga = n_ga / 2.0
        h_ga = interp_to_n(best_h, n_ga) if best_n != n_ga else best_h.copy()
        h_ga = project_to_sum(h_ga, ts_ga)
        c5_ga = compute_c5(h_ga)
        ga_budget = min(time_left() - 150, 400)
        if ga_budget > 60:
            h_ga_best, c5_ga_best = genetic_algorithm(h_ga, ts_ga, rng,
                                                       n_pop=20, n_gen=200,
                                                       t_budget=ga_budget)
            if best_n != n_ga:
                h_up = interp_to_n(h_ga_best, best_n)
                h_up = project_to_sum(h_up, best_n / 2.0)
            else:
                h_up = h_ga_best
            c5_up = compute_c5(h_up)
            h_cd, c5_cd = coordinate_descent(h_up, best_n / 2.0, max_iters=5000, step_frac=0.08)
            if c5_cd < c5_up:
                h_up = h_cd
                c5_up = c5_cd
            h_ml, c5_ml = multi_lag_coord_descent(h_up, best_n / 2.0, max_iters=3000,
                                                  step_frac=0.06, n_top_lags=5)
            if c5_ml < c5_up:
                h_up = h_ml
                c5_up = c5_ml
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    n_best = best_n
    ts_best = n_best / 2.0

    for alpha in [2000, 10000, 50000]:
        if time_left() < 50:
            break
        h_pol = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 80:
        h_cd, c5_cd = coordinate_descent(best_h, ts_best, max_iters=15000, step_frac=0.05)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 60:
        h_ml, c5_ml = multi_lag_coord_descent(best_h, ts_best, max_iters=10000,
                                              step_frac=0.04, n_top_lags=3)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    for alpha in [100000, 500000]:
        if time_left() < 30:
            break
        h_pol2 = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1500)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    if time_left() > 20:
        h_final, c5_final = coordinate_descent(best_h, ts_best, max_iters=8000, step_frac=0.03)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
