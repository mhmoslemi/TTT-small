import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = np.sum(mask)
        if n_free > 0:
            h = h.copy()
            h[mask] += diff / n_free
        else:
            h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_subgrad(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for m in idx:
        lag = abs(int(m) - (n - 1))
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    return g / float(max(k, 1))

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / float(len(exp_vals))) / alpha
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        lag = abs(int(idx) - (n - 1))
        g[lag:] += w[idx] * (1.0 - h[:n - lag])
        g[:n - lag] -= w[idx] * h[lag:]
    return sm_val, g

def adam_topk(h0, n, target_sum, n_iters, lr_init, k, time_limit, t_global_start):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    patience = 0
    for it in range(n_iters):
        if time.time() - t_global_start > time_limit:
            break
        t += 1
        g = topk_subgrad(h, k)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1
            if patience > 80:
                lr *= 0.93
                patience = 0
    return best_h, best_c5

def slsqp_smooth(h0, n, target_sum, alpha, maxiter, time_limit, t_global_start):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def obj_val(x):
        h = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_obj_and_grad(h, alpha)
        return val

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_obj_and_grad(h, alpha)
        return g

    def cfun(x):
        return np.sum(x) - target_sum

    def cjac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_val, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except Exception:
        pass
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, time_limit, t_global_start, rng):

            break

        mi = max(50, min(400, int(_rem(t_start, budget_s) * 0.5)))
        tl = min(50, _rem(t_start, budget_s) * 0.5)
        h_opt, c5_opt = slsqp_smooth(h_f, n_f, ts_f, 500.0, mi, tl, t_start)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_f
        h_f = h_opt

    if _rem(t_start, budget_s) > 20:
        tl = min(30, _rem(t_start, budget_s) * 0.5)
        h_cd, c5_cd = coord_descent(h_f, n_f, ts_f, 100, t_start + tl, t_start, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_f
        h_f = h_cd

    if _rem(t_start, budget_s) > 10:
        tl = min(15, _rem(t_start, budget_s) * 0.5)
        h_cd2, c5_cd2 = coord_descent(h_f, n_f, ts_f, 60, t_start + tl, t_start, rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n_f

    # Final
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
