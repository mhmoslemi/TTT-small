import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exact sum."""
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    while h_bin.sum() > target:
        # Find a 1 to flip to 0 that minimizes damage
        candidates = np.where(h_bin == 1)[0]
        # Prefer flipping positions where h was closest to 0.5
        idx = candidates[np.argmin(h[candidates])]
        h_bin[idx] = 0
    while h_bin.sum() < target:
        candidates = np.where(h_bin == 0)[0]
        idx = candidates[np.argmax(h[candidates])]
        h_bin[idx] = 1
    return h_bin

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_penalty(h0, n, target_sum, alpha, penalties, maxiter_per_level):
    h = h0.copy()
    for pen in penalties:
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return obj + 0.5 * pen * s_dev * s_dev
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return grad + pen * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_level, 'ftol': 1e-14, 'gtol': 1e-7})
            h = res.x
        except Exception:
            pass
    return project_to_sum(h, target_sum)

    T_min = 1e-9
    
    while time.time() - t0 < max_time:
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        p = ones[i]
        q = zeros[j]
        
        h[p] = 0.0
        h[q] = 1.0
        
        new_corr = compute_corr(h)
        new_obj = float(np.max(new_corr))
        
        if new_obj <= obj or rng.rand() < np.exp(-(new_obj - obj) / T):
            corr = new_corr
            obj = new_obj
            ones[i] = q
            zeros[j] = p
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
        else:
            # Revert
            h[p] = 1.0
            h[q] = 0.0
        
        T = max(T * cooling, T_min)
    
    return best_h, best_obj * dx

Let me also add one more idea: **directed local search** on the binary problem. Instead of pure random swaps, try:
1. For each position p with h[p]=1, compute the "cost" of moving it to each position q with h[q]=0
2. Pick the best (most improving) move
3. Repeat until no improvement

This is a steepest-descent method on the binary space. It's more expensive per iteration (O(n²) to evaluate all moves) but makes more informed moves.

For n=240, evaluating all n/2 * n/2 = 14400 possible moves takes 14400 * O(n) = O(3.4M) operations per round. At ~1μs per operation, that's ~3.4s per round. With a few rounds, this is feasible.

        # Actually, we want to flip the one that's least "certain", i.e., lowest h
