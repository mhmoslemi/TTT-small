import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * corr
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for delta in range(-(n - 1), n):
        wi = w[n - 1 + delta]
        if wi < 1e-14:
            continue
        c = wi * alpha
        if delta == 0:
            grad += c * (1.0 - 2.0 * h)
        elif delta > 0:
            nv = n - delta
            grad[:nv] += c * (1.0 - h[delta:])
            grad[delta:] -= c * h[:nv]
        else:
            k = -delta
            nv = n - k
            grad[k:] += c * (1.0 - h[:nv])
            grad[:nv] -= c * h[k:]
    return grad * dx

def lbfgsb_optimize(h0, target_sum, alpha, maxiter=500, penalty=1e5):
    n = len(h0)
    h = h0.copy()
    
    def obj(x):
        return smooth_max_obj(x, alpha) + penalty * (np.sum(x) - target_sum) ** 2
    
    def grad(x):
        g = smooth_max_grad(x, alpha)
        g += 2.0 * penalty * (np.sum(x) - target_sum)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        h = res.x
    except Exception:
        pass
    return project_to_sum(h, target_sum)

def exact_subgrad_step(h, target_sum, step_size=0.001):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    active = np.where(np.abs(corr - m_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for idx in active:
        delta = idx - (n - 1)
        if delta == 0:
            grad += (1.0 - 2.0 * h)
        elif delta > 0:
            nv = n - delta
            grad[:nv] += (1.0 - h[delta:])
            grad[delta:] -= h[:nv]
        else:
            k = -delta
            nv = n - k
            grad[k:] += (1.0 - h[:nv])
            grad[:nv] -= h[k:]
    
    grad = grad * dx / len(active)
    
    can_inc = h < 1.0 - 1e-10
    can_dec = h > 1e-10
    if not np.any(can_inc) or not np.any(can_dec):
        return h
    
    i_dec = int(np.argmax(np.where(can_dec, grad, -1e10)))
    i_inc = int(np.argmin(np.where(can_inc, grad, 1e10)))
    
    if i_inc == i_dec or grad[i_inc] >= grad[i_dec]:
        return h
    
    max_move = min(1.0 - h[i_inc], h[i_dec])
    move = min(step_size, max_move)
    if move < 1e-12:
        return h
    
    h_new = h.copy()
    h_new[i_inc] += move
    h_new[i_dec] -= move
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    
    if compute_c5(h_new) < compute_c5(h):
        return h_new
    return h

def coordinate_descent_fast(h, target_sum, max_iters=20000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        
        if lag == 0:
            g = 1.0 - 2.0 * h
        elif lag > 0:
            g = np.zeros(n)
            nv = n - lag
            g[:nv] += (1.0 - h[lag:])
            g[lag:] -= h[:nv]
        else:
            k = -lag
            g = np.zeros(n)
            nv = n - k
            g[k:] += (1.0 - h[:nv])
            g[:nv] -= h[k:]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, 1e10)
        g_dec = np.where(can_dec, g, -1e10)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-14:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h[-1:], h, h[:1]])
    x_ext = np.concatenate([[-x_old[0]], x_old, [x_old[-1] + 2.0/n_old]])
    x_new_wrapped = np.where(x_new > 1.0, x_new - 2.0, x_new)
    h_new = np.interp(x_new_wrapped, x_ext, h_ext)
    return h_new

def make_fuzzy_rect(n, center, half_width, trans_width, n_smooth=4):
    x = np.linspace(0, 2.0, n, endpoint=False)
    left = center - half_width
    right = center + half_width
    t = np.zeros(n)
    dl = (x - left) / trans_width
    dr = (right - x) / trans_width
    t = np.power(np.clip(dl, 0, 1), n_smooth) * np.power(np.clip(dr, 0, 1), n_smooth)
    return t
