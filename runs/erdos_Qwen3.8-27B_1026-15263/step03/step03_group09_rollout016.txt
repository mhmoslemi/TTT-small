def nesterov_adam(h0, n, target_sum, alpha_schedule):

    best_h = h.copy()

    # Nesterov momentum
    m_vec = np.zeros(n)  # momentum

Let me also think about the "binary pushing" step more carefully. The idea is:
- After smooth optimization, many values of h might be in (0, 0.2) or (0.8, 1) but not exactly at the boundaries.
- Pushing them to the boundaries (0 or 1) and re-projecting might reduce C5.
- This is because the function h(1-h) is concave, so the product h(x)(1-h(x+k)) is "more spread out" when h is in the interior.

Actually, I'm not sure that pushing to binary always helps. Let me think of a counterexample: if h[i] = 0.3 and its neighbors are all 0.7, pushing h[i] to 0 might increase the overlap at certain shifts.

A better approach: for each coordinate i, try moving it to 0 or to 1 (whichever is closer), re-project, and check if C5 improved. If yes, keep the move. This is a local search.

Let me implement this as a "binary local search":

```python
def binary_local_search(h0, n, target_sum, max_rounds):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for _ in range(max_rounds):
        improved = False
        # Try each coordinate in random order
        for i in np.random.permutation(n):
            # Try moving to 0
            if h[i] > 0.1:
                h_try = h.copy()
                h_try[i] = 0.0
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
            # Try moving to 1
            if h[i] < 0.9:
                h_try = h.copy()
                h_try[i] = 1.0
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
        if not improved:
            break
    
    return h, c5_cur
```

This is O(n) evaluations per round, and each evaluation is O(n²) (for the correlation). So each round is O(n³). For n=720, that's about 373M operations per round. That's too slow!

We need to be smarter. Instead of evaluating the full C5 after each coordinate move, we can use a first-order approximation or only check the affected correlations.

When we change h[i] by δ, only the correlations that involve h[i] change. Specifically, corr[m] changes for all m (since h[i] appears in corr[m] = Σ_j h[j](1-h[j+m-n+1]) for many j values). Actually, h[i] appears in corr[m] whenever i is a valid index in the sum, which is for all m (since for any m, there exists j such that the terms involve h[i]).

Wait, more precisely:
corr[m] = Σ_j h[j] * (1-h[j+m-n+1])

h[i] appears in this sum:
- As h[j] when j = i: contributes h[i] * (1-h[i+m-n+1])
- As h[j+m-n+1] when j+m-n+1 = i, i.e., j = i-m+n-1: contributes h[i-m+n-1] * (1-h[i])

So changing h[i] by δ affects corr[m] by:
δ * (1-h[i+m-n+1]) * 1_{valid} - δ * h[i-m+n-1] * 1_{valid}

for each m. This means ALL correlations change when we modify a single coordinate! So we can't avoid recomputing the full correlation.

However, we can compute the change in the MAX correlation more efficiently:
- Compute the old max: m_old = max(corr)
- Compute the new corr: corr_new = corr + δ * (change_vector)
- The new max is max(corr_new)

The change_vector has length 2n-1, so computing the new max is O(n). And computing the change_vector is also O(n) (it's just a shifted version of h and 1-h).

So for a single coordinate change:
- Compute change_vector: O(n)
- Compute new max: O(n)
Total: O(n) per coordinate evaluation.

For a full round of n coordinates: O(n²) per round. For n=720, that's 518K operations per round. Very fast!

Let me implement this:

```python
def binary_local_search_fast(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    for _ in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            for new_val in [0.0, 1.0]:
                if abs(h[i] - new_val) < 0.05:
                    continue
                delta = new_val - h[i]
                
                # Compute change in corr
                # corr[m] changes by: delta * (1-h[i+m-n+1]) * 1_{valid} - delta * h[i-m+n-1] * 1_{valid}
                change = np.zeros(2*n-1)
                for m in range(2*n-1):
                    # First part: i+m-n+1 valid?
                    idx1 = i + m - n + 1
                    if 0 <= idx1 < n:
                        change[m] += delta * (1.0 - h[idx1])
                    # Second part: i-m+n-1 valid?
                    idx2 = i - m + n - 1
                    if 0 <= idx2 < n:
                        change[m] -= delta * h[idx2]
                
                corr_new = corr + change
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5_cur - 1e-13:
                    # Accept the move
                    h[i] = new_val
                    h = project_to_sum(h, target_sum)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    c5_cur = compute_c5(h)
                    improved = True
                    break  # move to next coordinate
        
        if not improved:
            break
    
    return h, c5_cur
```

Wait, this is still O(n) per coordinate (the loop over m), so O(n²) per round. For n=720, that's 518K per round, which is fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        lo_b = max(0, n - 1 - j)
        hi_b = min(2 * n - 2, 2 * n - 2 - j)
        if lo_b <= hi_b:
            ms = np.arange(lo_b, hi_b + 1)
            idx = j + ms - n + 1
            g[j] += np.dot(w[ms], b[idx])
        
        lo_h = j
        hi_h = min(2 * n - 2, j + n - 1)
        if lo_h <= hi_h:
            ms = np.arange(lo_h, hi_h + 1)
            idx = j - ms + n - 1
            g[j] -= np.dot(w[ms], h[idx])
    
    return sm_val, g

def nesterov_adam(h0, n, target_sum, alpha_schedule, t_start, budget_s):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m_vec = np.zeros(n)
    beta = 0.9
    
    for alpha in alpha_schedule:
        base_lr = 0.03
        lr = base_lr
        stall = 0
        max_stall = 300
        
        while time.time() - t_start < budget_s - 30:
            _, g = smooth_max_and_grad(h, alpha)
            g = g / (np.max(np.abs(g)) + 1e-15)  # normalize gradient
            m_vec = beta * m_vec + (1 - beta) * g
            h_new = h - lr * m_vec
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                stall = 0
            else:
                h = h_new
                stall += 1
                if stall > max_stall:
                    lr *= 0.7
                    m_vec = 0.7 * m_vec
                    stall = 0
                    if lr < 1e-5:
                        break
        
        # Decay momentum between alpha levels
        m_vec *= 0.5
    
    return best_h, best_c5

def fast_binary_search(h0, n, target_sum, max_rounds, rng, t_start, budget_s):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m_arr = np.arange(2 * n - 1)
    
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - 20:
            break
        improved = False
        for i in rng.permutation(n):
            for new_val in [0.0, 1.0]:
                if abs(h[i] - new_val) < 0.1:
                    continue
                delta = new_val - h[i]
                
                idx1 = i + m_arr - n + 1
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = i - m_arr + n - 1
                valid2 = (idx2 >= 0) & (idx2 < n)
                
                change = np.zeros(2 * n - 1)
                change[valid1] += delta * (1.0 - h[idx1[valid1]])
                change[valid2] -= delta * h[idx2[valid2]]
                
                corr_new = corr + change
                c5_new = float(np.max(corr_new)) * dx
                
                if c5_new < c5_cur - 1e-13:
                    h[i] = new_val
                    h = project_to_sum(h, target_sum)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    c5_cur = float(np.max(corr)) * dx
                    improved = True
                    break
        
        if not improved:
            break
    
    return h, c5_cur

def coord_descent_fast(h0, n, target_sum, max_rounds, rng, t_start, budget_s, base_delta=0.01):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m_arr = np.arange(2 * n - 1)
    deltas = np.array([-3, -2, -1.5, -1, -0.5, 0.5, 1, 1.5, 2, 3]) * base_delta
    
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - 20:
            break
        improved = False
        for i in rng.permutation(n):
            best_c5 = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                nv = np.clip(nv, 0.0, 1.0)
                delta = nv - hi
                if abs(delta) < 1e-12:
                    continue
                
                idx1 = i + m_arr - n + 1
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = i - m_arr + n - 1
                valid2 = (idx2 >= 0) & (idx2 < n)
                
                change = np.zeros(2 * n - 1)
                change[valid1] += delta * (1.0 - h[idx1[valid1]])
                change[valid2] -= delta * h[idx2[valid2]]
                
                corr_new = corr + change
                c5_new = float(np.max(corr_new)) * dx
                
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_delta = d
            
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_cur = float(np.max(corr)) * dx
                improved = True
        
        if not improved:
            break
        # Reduce step size if no big improvements
        base_delta *= 0.8
    
    return h, c5_cur
