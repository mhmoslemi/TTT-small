import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    w = exp_vals / S
    obj = m_val + np.log(S) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def adam_optimize(h0, n, target_sum, alpha_schedule, n_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with smooth max objective."""
    h = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_obj = smooth_max_and_grad(h, alpha_schedule[0])[0]
    
    for t in range(n_steps):
        alpha = alpha_schedule[min(t, len(alpha_schedule)-1)]
        obj, grad = smooth_max_and_grad(h, alpha)
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_sum(h, target_sum)
    
    return best_h, best_obj

def binary_local_search(h_start, n, target_sum, t_end, rng, max_no_improve=5000):
    """Targeted local search for binary solutions."""
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    no_improve = 0
    T = 0.008
    
    while time.time() < t_end:
        # Random pair swap
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        
        old_i, old_j = h[i], h[j]
        h[i], h[j] = old_j, old_i
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        
        if delta < -1e-12:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
            no_improve = 0
        else:
            # Accept with probability exp(-delta/T)
            if delta < 0 or np.log(rng.uniform()) < -delta / T:
                c5_curr = c5_new
                no_improve += 1
            else:
                h[i], h[j] = old_i, old_j
                no_improve += 1
        
        if no_improve > max_no_improve:
            h = best_h.copy()
            c5_curr = best_c5
            T = max(T * 1.5, 0.002)
            no_improve = 0
        
        T = max(T * 0.99999, 0.001)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    w = exp_vals / S
    obj = m_val + np.log(S) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def adam_optimize(h0, n, target_sum, alpha_schedule, steps_total, lr_init=0.02, t_end=None):
    h = h0.copy().astype(float)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    step_count = 0
    
    alphas = alpha_schedule
    steps_per_alpha = max(1, steps_total // len(alphas))
    
    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if t_end is not None and time.time() > t_end:
                break
            if step_count >= steps_total:
                break
            
            obj, grad = smooth_max_and_grad(h, alpha)
            t += 1
            step_count += 1
            
            m_buf = beta1 * m_buf + (1 - beta1) * grad
            v_buf = beta2 * v_buf + (1 - beta2) * grad**2
            m_hat = m_buf / (1 - beta1**t)
            v_hat = v_buf / (1 - beta2**t)
            
            lr = lr_init / (1 + 0.0005 * t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_to_sum(h, target_sum)
            
            if step_count % 500 == 0:
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
        
        m_buf = np.zeros(n)
        v_buf = np.zeros(n)
        t = 0
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter=500):
    h = h0.copy().astype(float)
    
    def obj_func(x):
        return smooth_max_and_grad(x, alpha)[0]
    
    def grad_func(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    def con_fun(x):
        return np.sum(x) - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_func, h, method='L-BFGS-B', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        h_res = project_to_sum(res.x, target_sum)
        return h_res
    except Exception:
        return project_to_sum(h, target_sum)

def binary_sa(h_start, n, target_sum, t_end, rng, T_init=0.008, T_min=0.001):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < -1e-12:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
            no_improve = 0
        elif delta >= 0 and np.log(rng.uniform()) < -delta / T:
            c5_curr = c5_new
            no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.015)
            no_improve = 0
        T = max(T * 0.99999, T_min)
    
    return best_h, best_c5

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    w = exp_vals / S
    obj = m_val + np.log(S) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad
