import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_top_lags(h, m=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argsort(corr)[-m:]


def smooth_max_and_grad(h, top_idx, alpha):
    n = len(h)
    active_corr = np.zeros(len(top_idx))
    for pos, m_idx in enumerate(top_idx):
        k = m_idx - (n - 1)
        if k >= 0:
            active_corr[pos] = np.dot(h[:n-k], 1.0 - h[k:])
        else:
            k2 = -k
            active_corr[pos] = np.dot(h[k2:], 1.0 - h[:n-k2])
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for pos, m_idx in enumerate(top_idx):
        wm = w[pos]
        if wm < 1e-15:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                g[:n-k] += wm * (1.0 - h[k:])
                g[k:] -= wm * h[:n-k]
        else:
            k2 = -k
            if n - k2 > 0:
                g[k2:] += wm * (1.0 - h[:n-k2])
                g[:n-k2] -= wm * h[k2:]
    return sm_val, g


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


class SLSQPProblem:
    def __init__(self, top_idx, alpha, n, ts):
        self.top_idx = top_idx
        self.alpha = alpha
        self.n = n
        self.ts = ts

    def obj(self, x):
        h_c = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h_c, self.top_idx, self.alpha)
        return v

    def grad(self, x):
        h_c = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h_c, self.top_idx, self.alpha)
        return g

    def cfun(self, x):
        return np.sum(x) - self.ts

    def cjac(self, x):
        return np.ones(self.n)


def do_slsqp(h0, top_idx, alpha, n, ts, maxiter):
    prob = SLSQPProblem(top_idx, alpha, n, ts)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': prob.cfun, 'jac': prob.cjac}]
    try:
        res = minimize(prob.obj, h0.copy(), jac=prob.grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)


def coordinate_descent(h0, n, target_sum, max_rounds, step, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        idx = rng.choice(n, size=min(n, 300), replace=False)
        for i in idx:
            best_val = c5_cur
            best_d = 0.0
            for d in [-step, -step * 0.5, step * 0.5, step]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

            c5_new = compute_c5(h_new)

    return h, c5_cur
