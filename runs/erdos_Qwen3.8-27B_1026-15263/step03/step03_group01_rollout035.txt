        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    # Full overlap vector
    ov = compute_overlaps(h)
    # Shift to be indexed 0..2n-2, corresponding to lags -(n-1)..n-1

    top_ov = ov[top_idx]
    max_ov = np.max(top_ov)
    
    # Softmax weights
    alpha_w = 50.0
    w = np.exp(alpha_w * (top_ov - max_ov))
    w = w / np.sum(w)
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in top_idx:

        if k == 0:
            grad += wi * (1 - 2*h) * dx

In the "full" mode, the output has length 2n-1, and c_k is at index k + (n-1).

- k=1: c = sum_{n=0}^{n-2} h_{n+1} * (1-h_n)

These are NOT the same! c[k] ≠ c[-k] in general.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_subgradient(h, K, dx, alpha=50.0):
    """Compute subgradient of max overlap using top-K lags with softmax weighting."""
    n = len(h)
    ov = compute_overlaps(h)  # length 2n-1, values already multiplied by dx
    max_ov = np.max(ov)
    
    # Get top K indices
    if K >= len(ov):
        top_idx = np.arange(len(ov))
    else:
        # Use partition for efficiency
        part = np.argpartition(ov, -K)[-K:]
        top_idx = part
    
    top_ov = ov[top_idx]
    w = np.exp(alpha * (top_ov - max_ov))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(len(top_idx))
        w[np.argmax(top_ov)] = 1.0
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for idx_i, p in enumerate(top_idx):
        ell = p - (n - 1)  # lag
        wi = w[idx_i]
        if wi < 1e-10:
            continue
        if ell == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        elif ell > 0:
            # dc/dh[j] = (1-h[j-ell]) for j >= ell, minus h[j+ell] for j < n-ell
            if n - 1 - ell + 1 > 0:
                # (1-h[j-ell]) for j = ell, ..., n-1
                grad[ell:] += wi * (1.0 - h[:-ell]) * dx
                # -h[j+ell] for j = 0, ..., n-1-ell
                grad[:n-ell] -= wi * h[ell:] * dx
        else:
            m = -ell
            # (1-h[j-m]) for j >= m, minus h[j+m] for j < n-m
            grad[m:] += wi * (1.0 - h[:-m]) * dx
            grad[:n-m] -= wi * h[m:] * dx
    
    return max_ov, grad

def projected_grad_opt(h, n, ts, K, n_iter, alpha, lr_init=0.1):
    """Projected gradient descent with top-K subgradient."""
    dx = 2.0 / n
    h = h.copy()
    lr = lr_init
    best_h = h.copy()
    best_ov = compute_c5(h)
    stall = 0
    
    for it in range(n_iter):
        max_ov, grad = topk_subgradient(h, K, dx, alpha)
        
        # Backtracking line search
        for _ in range(50):
            h_new = project_to_sum(h - lr * grad, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_ov:
                h = h_new
                best_ov = c5_new
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 1.0)
                break
            else:
                lr *= 0.5
        else:
            stall += 1
            if stall > 200 or lr < 1e-14:
                break
    
    return best_h, best_ov

        ov = compute_overlaps(x)
        m_val = np.max(ov)
        shifted = alpha * (ov - m_val)
        max_sh = np.max(shifted)
        exp_v = np.exp(shifted - max_sh)
        w = exp_v / np.sum(exp_v)

                grad[:n_local-m] -= w[p] * x[m:] * dx
