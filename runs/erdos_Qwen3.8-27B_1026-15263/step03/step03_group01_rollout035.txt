```python
    h = np.empty(n)
    h[:n-1] = result.x
    h[n-1] = target_sum - np.sum(result.x)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, target_sum)
    
    return h
```

Wait, the issue with L-BFGS-B is that it only handles bound constraints, not the linear constraint sum(h) = target_sum. The reparameterization trick helps but the gradient computation becomes tricky when clipping is involved.

Let me think of a cleaner approach. Actually, the simplest and most effective approach might be:

1. Use the smooth max gradient (from previous code) but with L-BFGS-B instead of SLSQP
2. Handle the sum constraint via projection at each function evaluation
3. Use a much higher alpha for the smooth max (making it closer to the true max)

But actually, SLSQP with the equality constraint should be fine. The issue is convergence speed.

Let me try a different angle: **What if I use a penalty method?**

Penalty approach:
- Minimize: smooth_max(corr, alpha) + mu * (sum(h) - n/2)²
- Subject to: 0 ≤ h ≤ 1 (handled by L-BFGS-B bounds)
- Increase mu over iterations

This eliminates the need for the equality constraint in the solver!

```python
def penalty_lbfgs(h0, n, target_sum, alpha, mu, maxiter):
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Smooth max
        shifted = alpha * (corr - np.max(corr))
        exp_v = np.exp(shifted - np.max(shifted))
        w = exp_v / np.sum(exp_v)
        obj_corr = np.max(corr) + np.log(np.sum(exp_v)) / alpha
        
        # Penalty
        s = np.sum(h) - target_sum
        obj = obj_corr + mu * s * s
        
        # Gradient of smooth max (simplified)
        # d/dh[j] of corr[k] for the max component...
        # Actually, let me use the full smooth max gradient
        # This is O(n^2) which is expensive for n=480+
        
        # Simpler: just use the gradient of the max component + penalty gradient
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        
        grad = np.zeros(n)
        if i_min <= i_max:
            first_pos = np.arange(i_min + k, i_max + k + 1)
            second_pos = np.arange(i_min, i_max + 1)
            grad[first_pos] += (1.0 - h[second_pos])
            grad[second_pos] -= h[first_pos]
        
        # Penalty gradient
        grad += 2 * mu * s
        
        return obj, grad
    
    result = minimize(obj_grad, h0.copy(), method='L-BFGS-B', jac=True,
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    
    h = np.clip(result.x, 0.0, 1.0)
    h = project_to_sum(h, target_sum)
    return h
```

This is O(n) per gradient evaluation instead of O(n²) for the full smooth max gradient! This should be much faster.

But wait - using just the gradient of the max component is a subgradient, not the true gradient. This might cause issues with L-BFGS-B which assumes a smooth objective. However, in practice, near the optimum the max is usually unique, so it should work.

Let me also think about **what makes this problem hard**:
- The objective is max of many quadratic functions
- Near the optimum, multiple lags are close to the maximum
- This creates a "plateau" that gradient methods struggle with

**Key insight**: At the optimum, ideally all the "active" lags (those at the maximum) should have equal values, and their gradients should be in the positive cone of the active constraints. This is the KKT condition.

So a better approach might be:
1. Identify active lags (within ε of max)
2. Minimize the sum (or average) of the active lags
3. The gradient of the sum of active lags is the sum of individual gradients
4. This is smoother than the max and captures the KKT conditions better

```python
def multi_lag_gradient(h, n_active=3, tol=0.001):
    """Gradient of the sum of the top n_active correlation components."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    
    # Find active lags
    active = np.where(corr >= max_val - tol)[0]
    active = active[:n_active]
    
    grad = np.zeros(n)
    for k_star in active:
        k = k_star - (n - 1)
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:
            continue
        first_pos = np.arange(i_min + k, i_max + k + 1)
        second_pos = np.arange(i_min, i_max + 1)
        grad[first_pos] += (1.0 - h[second_pos])
        grad[second_pos] -= h[first_pos]
    
    return corr, grad
```

This is O(n_active * n) per evaluation, which is much better than O(n²) for the full smooth max.

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        n_mask = np.sum(mask)
        if n_mask > 0:
            h[mask] += diff / n_mask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_max_lag_info(h):
    """Get correlation, max index, and lag."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    return corr, k_star, k

def gradient_at_lag(h, k):
    """Gradient of c_k with respect to h."""
    n = len(h)
    i_min = max(0, -k)
    i_max = min(n - 1, n - 1 - k)
    grad = np.zeros(n)
    if i_min > i_max:
        return grad
    first_pos = np.arange(i_min + k, i_max + k + 1)
    second_pos = np.arange(i_min, i_max + 1)
    grad[first_pos] += (1.0 - h[second_pos])
    grad[second_pos] -= h[first_pos]
    return grad

def multi_lag_gradient(h, n_active=5, rel_tol=0.005):
    """Gradient of sum of top n_active correlation components."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Active lags: those within rel_tol * max_val of the max
    threshold = max_val - rel_tol * max_val
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    active = active[:n_active]
    
    grad = np.zeros(n)
    for k_star in active:
        k = int(k_star) - (n - 1)
        grad += gradient_at_lag(h, k)
    
    return corr, grad / len(active)

def nesterov_multi_lag(h0, n, target_sum, n_iter=5000, lr=0.001, momentum=0.9, n_active=5, rel_tol=0.005):
    """Nesterov accelerated gradient with multi-lag objective."""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        y = h + momentum * v
        y = project_to_sum(y, target_sum)
        
        corr, grad = multi_lag_gradient(y, n_active=n_active, rel_tol=rel_tol)
        
        v = momentum * v - lr * grad
        h_new = project_to_sum(h + v, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        # Decay learning rate
        lr *= 0.9995
    
    return best_h, best_c5

def penalty_lbfgs(h0, n, target_sum, mu, maxiter=2000, n_active=3, rel_tol=0.002):
    """L-BFGS-B with penalty for sum constraint."""
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr, grad_ml = multi_lag_gradient(h, n_active=n_active, rel_tol=rel_tol)
        s = np.sum(h) - target_sum
        obj = np.max(corr) + mu * s * s
        g = grad_ml + 2.0 * mu * s
        return obj, g
    
    res = minimize(obj_grad, h0.copy(), method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    
    h = np.clip(res.x, 0.0, 1.0)
    h = project_to_sum(h, target_sum)
    return h

def smooth_max_obj_grad(h, alpha):
    """Original smooth max objective and gradient (O(n^2))."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    # Build gradient: grad[j] = sum_k w[k] * d(corr[k])/d(h[j])
    # corr[k] = sum_i h[i+k-(n-1)] * (1-h[i]) for valid i
    # Let lag = k - (n-1)
    # d(corr[k])/d(h[j]) = (1-h[j-lag])*I[j-lag valid] - h[j+lag]*I[j valid]
    
    lags = np.arange(-(n-1), n)  # length 2n-1
    # For each lag l, the valid i range is [max(0,-l), min(n-1, n-1-l)]
    # h[j] appears as h[i+l] when i=j-l (valid if max(0,-l) <= j-l <= min(n-1, n-1-l))
    # h[j] appears as h[i] when i=j (valid if max(0,-l) <= j <= min(n-1, n-1-l))
    
    # Vectorized:
    nj = np.arange(n)[:, None]  # (n, 1)
    nl = lags[None, :]  # (1, 2n-1)
    
    # j appears as first factor (h[i+l]) when i = j-l is valid
    # i = j-l, valid if 0 <= j-l <= n-1 AND 0 <= i <= n-1-l (i.e., j-l <= n-1-l, i.e., j <= n-1)
    # And i >= 0 means j >= l
    # And i <= n-1-l means j-l <= n-1-l means j <= n-1
    # So: 0 <= j <= n-1 AND j >= l AND j-l >= 0 (same) AND j <= n-1
    # Simplified: max(0, l) <= j <= min(n-1, n-1+l) ... hmm this is getting confusing
    
    # Let me use a different approach. For each j and each lag l:
    # first factor: h[i+l] = h[j] when i = j-l
    #   valid if: 0 <= j-l <= n-1 (i in range) AND the term h[i+l]*h[i] is in the sum
    #   The sum for lag l is over i from max(0,-l) to min(n-1, n-1-l)
    #   So we need: max(0,-l) <= j-l <= min(n-1, n-1-l)
    #   Which means: max(0,-l)+l <= j <= min(n-1, n-1-l)+l
    #   = max(l, 0) <= j <= min(n-1+l, n-1)
    
    # second factor: h[i] = h[j] when i = j
    #   valid if: max(0,-l) <= j <= min(n-1, n-1-l)
    
    # This is complex to vectorize efficiently for n=480+. Let me use the loop approach for the active lags only.
    
    # Actually, for the full smooth max, I need all lags weighted by w. Let me use a different strategy:
    # Only compute contributions from lags where w[k] > threshold
    
    threshold = 1.0 / (n * 10)  # Only consider lags with significant weight
    active_k = np.where(w > threshold)[0]

    for k_idx in active_k:
        l = int(k_idx) - (n - 1)
        grad += w[k_idx] * gradient_at_lag(h, l)
    
    return obj, grad
```

```python
    h_new = np.interp(x_new, x_old, h, period=2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.any(mask):
        h[mask] += diff / np.sum(mask)
    else:
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_at_lag(h, k):
    n = len(h)
    i_min = max(0, -k)
    i_max = min(n - 1, n - 1 - k)
    grad = np.zeros(n)
    if i_min > i_max:
        return grad
    first_pos = np.arange(i_min + k, i_max + k + 1)
    second_pos = np.arange(i_min, i_max + 1)
    grad[first_pos] += (1.0 - h[second_pos])
    grad[second_pos] -= h[first_pos]
    return grad

def multi_lag_grad(h, n_active, rel_tol):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(np.max(corr))
    threshold = max_val * (1.0 - rel_tol)
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        order = np.argsort(-corr[active])
        active = active[order[:n_active]]
    grad = np.zeros(n)
    for k_star in active:
        k = int(k_star) - (n - 1)
        grad += gradient_at_lag(h, k)
    if len(active) > 1:
        grad /= float(len(active))
    return corr, max_val, grad

def nesterov_multilag(h0, target_sum, n_iter, lr, momentum, n_active, rel_tol, check_interval):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        y = project_to_sum(h + momentum * v, target_sum)
        corr, max_val, grad = multi_lag_grad(y, n_active, rel_tol)
        v = momentum * v - lr * grad
        h = project_to_sum(h + v, target_sum)
        if (it + 1) % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def lbfgs_penalty(h0, target_sum, mu, n_active, rel_tol, maxiter):
    n = len(h0)
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr, max_val, grad_ml = multi_lag_grad(h, n_active, rel_tol)
        s = np.sum(h) - target_sum
        obj = max_val + mu * s * s
        g = grad_ml + 2.0 * mu * s
        return obj, g
    res = minimize(obj_grad, h0.copy(), method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': maxiter * 3})
    h = np.clip(res.x, 0.0, 1.0)
    h = project_to_sum(h, target_sum)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    max_sh = float(np.max(shifted))
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    n_active_smooth = int(np.sum(w > 1.0 / (n * 5)))
    active_k = np.where(w > 1.0 / (n * 5))[0]
    grad = np.zeros(n)
    for k_idx in active_k:
        l = int(k_idx) - (n - 1)
        grad += w[k_idx] * gradient_at_lag(h, l)
    return obj, grad

def slsqp_smooth(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(r.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Polish at original resolution with fast gradient
    n1 = best_n
    ts1 = n1 / 2.0
    for params in [
        (3000, 0.002, 0.9, 5, 0.02, 50),
        (3000, 0.001, 0.9, 3, 0.01, 50),
        (2000, 0.0005, 0.85, 3, 0.005, 50),
    ]:
        if time_left() < 20:
            break
        n_it, lr, mom, n_act, r_tol, chk = params
        h_opt, c5_opt = nesterov_multilag(best_h, ts1, n_it, lr, mom, n_act, r_tol, chk)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # L-BFGS polish at original resolution
    for mu in [10.0, 100.0]:
        if time_left() < 15:
            break
        h_pol = lbfgs_penalty(best_h, ts1, mu, 3, 0.005, 400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 2: Upsample to 960
    if time_left() > 120:
        n2 = 960
        ts2 = n2 / 2.0
        h2 = upsample_h(best_h, n2)
        h2 = project_to_sum(h2, ts2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        # Multi-lag Nesterov at 960
        for params in [
            (4000, 0.002, 0.9, 5, 0.02, 100),
            (4000, 0.001, 0.9, 4, 0.01, 100),
            (3000, 0.0005, 0.88, 3, 0.005, 100),
            (2000, 0.0003, 0.85, 3, 0.003, 100),
        ]:
            if time_left() < 30:
                break
            n_it, lr, mom, n_act, r_tol, chk = params
            h_opt, c5_opt = nesterov_multilag(h2, ts2, n_it, lr, mom, n_act, r_tol, chk)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n2
                h2 = h_opt.copy()

        # L-BFGS polish at 960
        for mu in [10.0, 50.0, 200.0]:
            if time_left() < 20:
                break
            h_pol = lbfgs_penalty(h2, ts2, mu, 3, 0.003, 500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n2
                h2 = h_pol.copy()

        # SLSQP smooth max at high alpha
        if time_left() > 40:
            for alpha in [500.0, 2000.0]:
                if time_left() < 25:
                    break
                h_s = slsqp_smooth(h2, ts2, alpha, 500)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    best_n = n2
                    h2 = h_s.copy()

    # Phase 3: Try 1920 resolution
    if time_left() > 180:
        n3 = 1920
        ts3 = n3 / 2.0
        h3 = upsample_h(best_h, n3)
        h3 = project_to_sum(h3, ts3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3

        for params in [
            (3000, 0.001, 0.9, 5, 0.02, 100),
            (3000, 0.0005, 0.9, 3, 0.01, 100),
            (2000, 0.0003, 0.85, 3, 0.005, 100),
        ]:
            if time_left() < 30:
                break
            n_it, lr, mom, n_act, r_tol, chk = params
            h_opt, c5_opt = nesterov_multilag(h3, ts3, n_it, lr, mom, n_act, r_tol, chk)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n3
                h3 = h_opt.copy()

        for mu in [10.0, 100.0]:
            if time_left() < 20:
                break
            h_pol = lbfgs_penalty(h3, ts3, mu, 3, 0.003, 400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n3
                h3 = h_pol.copy()

    # Phase 4: Escape via frequency perturbation + re-optimize
    n_esc = best_n
    ts_esc = n_esc / 2.0
    for trial in range(20):
        if time_left() < 40:
            break
        amp = 0.04 * (1.0 - trial / 25.0) + 0.008
        nf = max(5, n_esc // 40)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()

        h_opt, c5_opt = nesterov_multilag(h_sh, ts_esc, 2000, 0.001, 0.9, 4, 0.01, 100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if time_left() > 50:
            h_pol = lbfgs_penalty(h_opt, ts_esc, 50.0, 3, 0.004, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Final fine-tuning
    n_f = best_n
    ts_f = n_f / 2.0
    if time_left() > 30:
        h_opt, c5_opt = nesterov_multilag(best_h, ts_f, 3000, 0.0003, 0.8, 3, 0.003, 50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 25:
        h_pol = lbfgs_penalty(best_h, ts_f, 200.0, 3, 0.002, 800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 20:
        h_sm = slsqp_smooth(best_h, ts_f, 5000.0, 800)
        c5_sm = compute_c5(h_sm)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
