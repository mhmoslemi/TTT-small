        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_lbf(h0, n, target_sum, alpha, penalty, maxiter):
    h = h0.copy()
    def fun(x):
        obj, _ = smooth_max_obj_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * penalty * s_dev * s_dev
    def jac(x):
        _, grad = smooth_max_obj_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + penalty * s_dev * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 100})
        h = res.x
    except Exception:
        pass
    return project_to_sum(h, target_sum)

def exact_polish(h, n, target_sum, max_rounds, n_active, step_range):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        m = np.max(corr)
        if m < 1e-15:
            break
        tol = 0.01 * m
        active_k = np.where(corr > m - tol)[0]
        if len(active_k) > n_active:
            idx_sorted = np.argsort(corr[active_k])[::-1][:n_active]
            active_k = active_k[idx_sorted]
        grad = np.zeros(n)
        for k in active_k:
            shift = k - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
        grad /= len(active_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        best_h = None
        best_c5 = c5_cur
        for lr in step_range:
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5
    return h, c5_cur

def slsqp_opt(h0, n, target_sum, alpha, maxiter):
    h = h0.copy()
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h

def binary_swap_polish(h, n, target_sum, rng, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.55)[0]
        zeros = np.where(h < 0.45)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(800, len(ones) * len(zeros))
        best_c5 = c5_cur
        best_pair = None
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            h_test = project_to_sum(h_test, target_sum)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-13:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            h = project_to_sum(h, target_sum)
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def rem():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    step_range_coarse = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    step_range_fine = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    # Phase 1: Continuation at parent resolution
    n1 = best_n
    ts1 = n1 / 2.0
    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 100000, 300000]
    for a in alphas_p1:
        if rem() < 90:
            break
        h_opt = optimize_lbf(best_h, n1, ts1, a, 100000, 300)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: SLSQP at high alpha
    if rem() > 100:
        for a in [10000, 50000, 200000]:
            if rem() < 50:
                break
            h_opt = slsqp_opt(best_h, n1, ts1, a, maxiter=300)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 3: Exact max polish - intensive
    if rem() > 60:
        h_pol, c5_pol = exact_polish(best_h, n1, ts1, 500, 12, step_range_coarse)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Second round of continuation from polished solution
    if rem() > 120:
        for a in [50000, 200000, 500000]:
            if rem() < 40:
                break
            h_opt = optimize_lbf(best_h, n1, ts1, a, 1000000, 400)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        if rem() > 40:
            h_pol, c5_pol = exact_polish(best_h, n1, ts1, 400, 15, step_range_fine)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 5: Kick and re-optimize (escape local minima)
    if rem() > 200:
        for kick_size in [0.1, 0.05, 0.02, 0.01]:
            if rem() < 60:
                break
            n_kicks = max(3, int(rem() / 40))
            for _ in range(n_kicks):
                if rem() < 30:
                    break
                noise = rng.normal(0, kick_size, n1)
                h_pert = project_to_sum(best_h + noise, ts1)
                h_opt = optimize_lbf(h_pert, n1, ts1, 20000, 100000, 200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
            if rem() > 30:
                h_pol, c5_pol = exact_polish(best_h, n1, ts1, 300, 12, step_range_coarse)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 6: Binary structure search
    if rem() > 100:
        h_bin = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin.sum() - ts1) > 1e-10 and attempts < 50000:
            if h_bin.sum() > ts1:
                idx = rng.choice(np.where(h_bin > 0)[0])
                h_bin[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin < 1)[0])
                h_bin[idx] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts1) < 1e-10:
            h_bf, c5_bf = binary_swap_polish(h_bin, n1, ts1, rng, max_rounds=300)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
            if rem() > 60:
                h_opt = optimize_lbf(h_bf, n1, ts1, 50000, 100000, 300)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                h_pol, c5_pol = exact_polish(best_h, n1, ts1, 300, 12, step_range_fine)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 7: Higher resolution
    if rem() > 250:
        n2 = n1 * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        for a in [500, 5000, 50000, 200000]:
            if rem() < 60:
                break
            h_opt = optimize_lbf(h_interp, n2, ts2, a, 100000, 200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
            h_interp = h_opt
        if rem() > 60:
            h_pol, c5_pol = exact_polish(best_h, best_n, best_n / 2.0, 300, 15, step_range_coarse)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 8: Try alternative resolutions
    if rem() > 200:
        for n_try in [120, 180, 300, 360]:
            if rem() < 80:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_down = np.interp(x_new, x_old, best_h, period=2.0)
            h_down = project_to_sum(h_down, ts_try)
            c5_s = compute_c5(h_down)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_down.copy()
                best_n = n_try
            for a in [2000, 20000, 100000]:
                if rem() < 30:
                    break
                h_opt = optimize_lbf(h_down, n_try, ts_try, a, 100000, 200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n_try
                h_down = h_opt
            if rem() > 30:
                h_pol, c5_pol = exact_polish(best_h, best_n, best_n / 2.0, 200, 12, step_range_fine)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 9: Final intensive polish
    if rem() > 50:
        n_f = best_n
        ts_f = n_f / 2.0
        h_pol, c5_pol = exact_polish(best_h, n_f, ts_f, 1000, 20, step_range_fine)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        if rem() > 30:
            h_opt = optimize_lbf(best_h, n_f, ts_f, 500000, 1000000, 500)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
            h_pol, c5_pol = exact_polish(best_h, n_f, ts_f, 500, 20, step_range_fine)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
