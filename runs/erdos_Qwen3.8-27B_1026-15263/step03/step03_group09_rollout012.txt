```python
def fft_gradient(h, w):
    n = len(h)
    N = 1  # Find appropriate FFT size
    while N < 3*n:
        N *= 2
    A = np.zeros(N)
    B = np.zeros(N)
    Wp = np.zeros(N)
    W_rev = np.zeros(N)
    A[:n] = 1.0 - h
    B[:n] = h
    Wp[:2*n-1] = w
    W_rev[:2*n-1] = w[::-1]
    
    t1 = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(Wp), N)
    t2 = np.fft.irfft(np.fft.rfft(B) * np.fft.rfft(W_rev), N)
    
    return t1[n-1:2*n-1] - t2[n-1:2*n-1]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, w

def fft_gradient(h, w):
    n = len(h)
    # FFT size >= 3n-2
    N = 1
    while N < 3*n - 1:
        N *= 2
    A = np.zeros(N)
    B = np.zeros(N)
    Wp = np.zeros(N)
    W_rev = np.zeros(N)
    A[:n] = 1.0 - h
    B[:n] = h
    Wp[:2*n-1] = w
    W_rev[:2*n-1] = w[::-1]
    
    t1 = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(Wp), N)
    t2 = np.fft.irfft(np.fft.rfft(B) * np.fft.rfft(W_rev), N)
    
    return t1[n-1:2*n-1] - t2[n-1:2*n-1]

def adam_with_fft(h0, n, target_sum, time_budget, alpha_start=10.0, alpha_end=500.0):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    t_start = time.time()
    
    while time.time() - t_start < time_budget:
        t += 1
        # Anneal alpha
        progress = min(1.0, t / (time_budget * 20))  # rough estimate of iterations
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm_val, w = soft_max_weights(corr, alpha)
        grad = fft_gradient(h, w)
        
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * grad * grad
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if t % 100 == 0:
            if c5_new > best_c5 * 1.01:
                lr *= 0.9
            elif c5_new < best_c5 * 0.99:
                lr = min(lr * 1.05, 0.5)
    
    return best_h, best_c5

def topk_subgradient_fast(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            tau = m - n + 1
            # Add (1-h[j-tau]) for valid j-tau
            jj1 = j_arr - tau
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            # Subtract h[j+tau] for valid j+tau
            jj2 = j_arr + tau
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def coord_descent_fast(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try a few values around current
            candidates = [h[i] + d * base_delta for d in [-2, -1, 1, 2]]
            best_val = c5_cur
            best_nv = h[i]
            for nv in candidates:
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_val:
                    best_val = c5_new
                    best_nv = nv
            if best_nv != h[i]:
                h[i] = best_nv
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts0 = best_n / 2.0
            best_h = project_to_sum(best_h, ts0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        if h_cand is None:
            return
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_cand
    
    # Phase 1: Optimize at current resolution
    n = best_n
    ts = n / 2.0
    start_h = best_h.copy()
    
    # FFT Adam with alpha annealing
    time_for_phase1 = min(200, _tl(t_start, budget_s) * 0.2)
    h_opt, c5_opt = adam_with_fft(start_h, n, ts, time_for_phase1, alpha_start=5.0, alpha_end=200.0)
    try_update(h_opt, n)
    
    # Top-k subgradient
    if _tl(t_start, budget_s) > 200:
        n_iters = min(5000, int(_tl(t_start, budget_s) * 0.15))
        h_sg, _ = topk_subgradient_fast(h_opt, n, ts, n_iters, 0.03, 20)
        try_update(h_sg, n)
    
    # Try higher resolutions
    for n_new in [420, 480, 540, 600, 720]:
        if _tl(t_start, budget_s) < 150:
            break
        if n_new % 2 != 0:
            continue
        ts_new = n_new / 2.0
        
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, ts_new)
        try_update(h_interp, n_new)
        
        # Optimize at new resolution
        time_budget_here = min(100, _tl(t_start, budget_s) * 0.15)
        h_opt2, _ = adam_with_fft(h_interp, n_new, ts_new, time_budget_here, alpha_start=5.0, alpha_end=300.0)
        try_update(h_opt2, n_new)
        
        if _tl(t_start, budget_s) > 80:
            n_iters2 = min(3000, int(_tl(t_start, budget_s) * 0.1))
            h_sg2, _ = topk_subgradient_fast(h_opt2, n_new, ts_new, n_iters2, 0.02, 30)
            try_update(h_sg2, n_new)
    
    # Phase 2: Final polishing at best resolution
    n_f = best_n
    ts_f = n_f / 2.0
    
    # Coordinate descent
    if _tl(t_start, budget_s) > 60:
        n_cd = max(30, min(100, int(_tl(t_start, budget_s) * 0.3)))
        h_cd, _ = coord_descent_fast(best_h, n_f, ts_f, n_cd, rng, 0.01)
        try_update(h_cd, n_f)
    
    # More subgradient with smaller steps
    if _tl(t_start, budget_s) > 40:
        n_sg = min(10000, int(_tl(t_start, budget_s) * 0.4))
        h_sg3, _ = topk_subgradient_fast(best_h, n_f, ts_f, n_sg, 0.01, 30)
        try_update(h_sg3, n_f)
    
    # Final coordinate descent with small steps
    if _tl(t_start, budget_s) > 30:
        n_cd2 = max(20, min(50, int(_tl(t_start, budget_s) * 0.5)))
        h_cd2, _ = coord_descent_fast(best_h, n_f, ts_f, n_cd2, rng, 0.003)
        try_update(h_cd2, n_f)
    
    # Even finer
    if _tl(t_start, budget_s) > 20:
        h_cd3, _ = coord_descent_fast(best_h, n_f, ts_f, 30, rng, 0.001)
        try_update(h_cd3, n_f)
    
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    return m + np.log(s / len(exp_vals)) / alpha, w

def fft_gradient(h, w):
    n = len(h)
    N = 1
    while N < 3 * n - 1:
        N *= 2
    A = np.zeros(N)
    B = np.zeros(N)
    Wp = np.zeros(N)
    W_rev = np.zeros(N)
    A[:n] = 1.0 - h
    B[:n] = h
    Wp[:2*n-1] = w
    W_rev[:2*n-1] = w[::-1]

    t2 = np.fft.irfft(np.fft.rfft(B) * np.fft.rfft(W_rev), N)
    return t1[n-1:2*n-1] - t2[n-1:2*n-1]

    beta2 = 0.999

        sm_val, w = soft_max_weights(corr, alpha)
        grad = fft_gradient(h, w)

        v_vec = beta2 * v_vec + (1.0 - beta2) * grad * grad
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)

def topk_subgrad(h0, n, target_sum, n_iters, lr_init, top_k):

                improved = True
            else:
```

        v_hat = v_vec / (1.0 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve > 300:
            lr *= 0.85
            no_improve = 0
    
    return best_h, best_c5

def topk_subgrad(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    no_improve = 0
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            tau = m - n + 1
            jj1 = j_arr - tau
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr + tau
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        
        lr = lr_init / (1.0 + 0.01 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
