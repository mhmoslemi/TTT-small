            else:
                grad[:n - lag] += wj * (1.0 - h[lag:])
                grad[lag:] += wj * (-h[:n - lag])
        else:
            k = -lag
            if k == 0:
                grad[:] += wj * (1.0 - 2.0 * h)
            else:
                grad[k:] += wj * (1.0 - h[:n - k])

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500, top_m=50):
    def f(x):
        return top_lags_gradient(x, alpha, top_m)[0]
    def g(x):
        return top_lags_gradient(x, alpha, top_m)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        
        if lag >= 0:
            if lag == 0:
                g = 1.0 - 2.0 * h
            else:
                g = np.zeros(n)
                g[:n - lag] = (1.0 - h[lag:])
                g[lag:] += -h[:n - lag]
        else:
            k = -lag
            g = np.zeros(n)
            g[k:] = (1.0 - h[:n - k])
            g[:n - k] += -h[k:]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

These are the same set of values but in reverse order! So g[lag:] += (1-h[:n-lag]) is NOT the same as adding (1-h[j+lag]) to g[j] for j=0,...,n-1-lag.

Unless... the code is computing the gradient of f_{-k} instead of f_k?

Let me check: f_{-k} = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])

df_{-k}/dh[j]:
- h[j] appears as h[i+k] when i=j-k (valid if j>=k and j-k<=n-1-k, i.e., k<=j<=n-1): contributes (1-h[j-k])
- h[j] appears as h[i] when i=j (valid if j<=n-1): contributes -h[j+k] * 1_{j+k<n}

Wait: f_{-k} = sum_{i=0}^{n-1-k} h[i+k](1-h[i])

df_{-k}/dh[j]:
- h[j] appears as h[i+k] where i = j-k, valid if 0 <= j-k <= n-1-k, i.e., k <= j <= n-1: contributes (1-h[j-k])
- h[j] appears as h[i] where i = j, valid if 0 <= j <= n-1-k: contributes -h[j+k]

So df_{-k}/dh[j] = (1-h[j-k]) * 1_{k<=j<=n-1} + (-h[j+k]) * 1_{0<=j<=n-1-k}

For j < k: only second term: -h[j+k]
For k <= j <= n-1-k: both: (1-h[j-k]) - h[j+k]
For j > n-1-k: only first: (1-h[j-k])

Now let's check the code for "lag" = -k (i.e., the else branch):
```
k = -lag  (so k > 0)
g[:n - k] += (1.0 - h[k:])   → g[j] += (1-h[j+k]) for j = 0, ..., n-1-k
g[k:] -= h[:n - k]           → g[j] -= h[j-k] for j = k, ..., n-1
```

So g[j] = (1-h[j+k]) * 1_{j<=n-1-k} + (-h[j-k]) * 1_{j>=k}

This matches df_k/dh[j] (for positive k), NOT df_{-k}/dh[j]!

And the if branch (lag >= 0):
```
g[lag:] += (1.0 - h[:n - lag])  → g[j] += (1-h[j-lag]) for j = lag, ..., n-1
g[:n - lag] -= h[lag:]           → g[j] -= h[j+lag] for j = 0, ..., n-1-lag
```

So g[j] = (1-h[j-lag]) * 1_{j>=lag} + (-h[j+lag]) * 1_{j<=n-1-lag}

This is: for j < lag: -h[j+lag]
For lag <= j <= n-1-lag: (1-h[j-lag]) - h[j+lag]
For j > n-1-lag: (1-h[j-lag])

This matches df_{-k}/dh[j] where k = lag! 

So the if branch (lag >= 0) computes the gradient of f_{-lag}, and the else branch (lag < 0, k=-lag>0) computes the gradient of f_k.

In the code, `lag = m_star - (n-1)` where m_star is the argmax of the correlate. Since c_k = f_{-k} (as I showed earlier), the argmax of c corresponds to the argmax of f at the negated lag.

If m_star corresponds to correlate "lag" L = m_star - (n-1), then c_L = f_{-L}. The gradient of c_L = f_{-L} is what we need.

If L >= 0: we need gradient of f_{-L}, which is the if branch. ✓
If L < 0: we need gradient of f_{-L} = f_{|L|}, which is the else branch with k = |L| = -L. ✓

Great, so the code is consistent. The "lag" variable in the code is the correlate's lag, and the gradient computed is indeed the gradient of the corresponding correlation value.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_grad(h, alpha, top_m=50):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    if top_m < len(corr):
        idx = np.argpartition(-corr, top_m)[:top_m]
    else:
        idx = np.arange(len(corr))
    corr_sel = corr[idx]
    m_sel = np.max(corr_sel)
    shifted = alpha * (corr_sel - m_sel)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_sel + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for j in range(len(idx)):
        lag = idx[j] - (n - 1)
        wj = w[j]
        if wj < 1e-10:
            continue
        g = lag_gradient(h, lag)
        grad += wj * g
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=400, top_m=50):
    def f(x):
        return smooth_max_grad(x, alpha, top_m)[0]
    def g(x):
        return smooth_max_grad(x, alpha, top_m)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = lag_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def block_coordinate_descent(h, target_sum, max_iters=5000, block_size=4, step_frac=0.08):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = lag_gradient(h, lag)
        can_inc = np.where(h < 1.0 - 1e-10)[0]
        can_dec = np.where(h > 1e-10)[0]
        if len(can_inc) < block_size or len(can_dec) < block_size:
            break
        g_inc = np.full(n, np.inf)
        g_inc[can_inc] = g[can_inc]
        inc_sel = np.argsort(g_inc)[:block_size]
        g_dec = np.full(n, -np.inf)
        g_dec[can_dec] = g[can_dec]
        dec_sel = np.argsort(g_dec)[-block_size:]
        if g[inc_sel[0]] >= g[dec_sel[0]] - 1e-12:
            break
        max_move = min(np.min(1.0 - h[inc_sel]), np.min(h[dec_sel]))
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[inc_sel] += move
        h[dec_sel] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
        else:
            h[inc_sel] -= move
            h[dec_sel] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 60:
                break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def random_fourier_start(n, rng, n_freq=8, amp_scale=0.15):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, n_freq + 1):
        a = rng.uniform(-amp_scale, amp_scale) / np.sqrt(m)
        phi = rng.uniform(0, 2.0 * np.pi)
        h += a * np.cos(np.pi * m * x + phi)
    h = np.clip(h, 0.0, 1.0)
    return h
