    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)

    grad = np.zeros(n)

        hi2 = min(n, n + shift)

        if hi2 > lo2:

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)

            h[mask] += diff / count
        else:

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_continuation(h0, n, target_sum, alphas, maxiter_per_level=200):
    h = h0.copy()
    for a in alphas:
        def fun(x, aa=a):
            obj, grad = smooth_max_obj_and_grad(x, aa)
            s_dev = np.sum(x) - target_sum
            return obj + 100.0 * s_dev * s_dev
        def jac(x, aa=a):
            _, grad = smooth_max_obj_and_grad(x, aa)
            s_dev = np.sum(x) - target_sum
            return grad + 200.0 * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_level, 'ftol': 1e-14, 'gtol': 1e-8})
            h = res.x
        except:
            pass
        h = project_to_sum(h, target_sum)
    return h

def exact_max_polish(h, n, target_sum, max_rounds=300):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(8, len(corr))]
        best_h = None
        best_c5 = c5_cur
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5
    return h, c5_cur

def binary_sa(h0, n, target_sum, rng, n_iter=100000, t_start=0.03, t_end=0.0003):
    h = (h0 > 0.5).astype(np.float64)
    target_ones = int(round(target_sum))
    
    while h.sum() < target_ones:
        zeros = np.where(h == 0)[0]
        if len(zeros) == 0: break
        h[zeros[rng.randint(len(zeros))]] = 1.0
    while h.sum() > target_ones:
        ones = np.where(h == 1)[0]
        if len(ones) == 0: break
        h[ones[rng.randint(len(ones))]] = 0.0
    
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    
    for i in range(n_iter):
        t = t_start * (t_end / t_start) ** (i / max(n_iter, 1))
        
        pi = rng.randint(n_ones)
        pj = rng.randint(n_zeros)
        i1 = ones_arr[pi]
        i0 = zeros_arr[pj]
        
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new = compute_c5(h)
        
        if c5_new < c5_cur or (c5_new > c5_cur and rng.random() < np.exp(-(c5_new - c5_cur) / t)):
            c5_cur = c5_new
            ones_arr[pi] = i0
            zeros_arr[pj] = i1
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    
    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SA at original resolution (multiple restarts)
    for restart in range(3):
        if remaining() < 60:
            break
        h_sa, c5_sa = binary_sa(best_h, best_n, best_n/2.0, rng,
                                n_iter=200000, t_start=0.04/(restart+1), t_end=0.0002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Phase 2: Continuous optimization
    if remaining() > 100:
        h_opt = optimize_continuation(best_h, best_n, best_n/2.0,
                                       alphas=[20, 50, 100, 200, 500, 1000, 2000, 5000, 10000])
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: Exact max polish
    if remaining() > 60:
        h_pol, c5_pol = exact_max_polish(best_h, best_n, best_n/2.0, max_rounds=300)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    # Phase 4: Try different grid sizes
    grid_sizes = sorted(set([best_n//2, best_n//3, best_n*2, 60, 80, 100, 120,
                              140, 160, 180, 200, 280, 320, 360, 400, 480]))
    for n_try in grid_sizes:
        if n_try < 20 or n_try > 600 or n_try == best_n:
            continue
        if remaining() < 120:
            break
        
        # Interpolate from current best
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h, period=2.0)
        h_try = project_to_sum(h_try, n_try/2.0)
        
        # SA at this resolution
        h_sa2, c5_sa2 = binary_sa(h_try, n_try, n_try/2.0, rng,
                                   n_iter=150000, t_start=0.03, t_end=0.0003)
        
        # Continuous optimization
        h_opt2 = optimize_continuation(h_sa2, n_try, n_try/2.0,
                                        alphas=[50, 200, 1000, 5000])
        c5_opt2 = compute_c5(h_opt2)
        
        # Polish
        h_pol2, c5_pol2 = exact_max_polish(h_opt2, n_try, n_try/2.0, max_rounds=100)
        
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2
            best_n = n_try
    
    # Phase 5: Final SA and polish on best
    if remaining() > 60:
        h_sa3, c5_sa3 = binary_sa(best_h, best_n, best_n/2.0, rng,
                                   n_iter=300000, t_start=0.02, t_end=0.0001)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3
    
    if remaining() > 40:
        h_pol3, c5_pol3 = exact_max_polish(best_h, best_n, best_n/2.0, max_rounds=300)
        if c5_pol3 < best_c5:
            best_c5 = c5_pol3
            best_h = h_pol3
    
    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
