Actually, for coordinate descent with the sum constraint, the cleanest approach is:
- Choose a pair (i, j)
- Transfer δ from i to j (or vice versa)
- Optimize δ optimally

This is what the stochastic CD was doing, but with random pairs and random δ. Instead, I'll do systematic pairs and exact δ optimization.

For a pair (i, j) and transfer δ (h[i] -= δ, h[j] += δ):
The objective is max_k [corr_k(δ) * dx] where corr_k depends on δ.

Since each corr_k is a quadratic in δ:
corr_k(δ) = corr_k(0) + a_k * δ + b_k * δ²

We want to minimize max_k corr_k(δ) * dx over δ ∈ [min(0, h[i]), min(1-h[i], h[j])].

This is a 1D problem: minimize max of quadratics over an interval.
We can solve it with bisection: find δ where the top two quadratics cross.

Actually this is complex. Let me just use golden section search for the 1D subproblem:

For a given pair (i,j), define:
f(δ) = compute_c5(h with h[i]-=δ, h[j]+=δ)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m_val = np.max(c)
    shifted = alpha * (c - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vs * (1.0 - h[sc]) * w[None, :] - vf * h[fc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=1000):
    def f_obj(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g_obj(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def c_eq(x):
        return np.sum(x) - ts
    def c_eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f_obj, h0.copy(), method='SLSQP', jac=g_obj, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_opt(h0, n, ts, alpha, n_iter=8000, lr_init=0.02):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    stall = 0
    best_obj = obj_prev
    best_h = h.copy()
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-15:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.03, 0.3)
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
        else:
            lr *= 0.5
            stall += 1
            if stall > 400 or lr < 1e-14:
                break
    return best_h

def pair_swap_cd(h, n, ts, rng, n_iters=50000, delta_range=(0.005, 0.25)):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.1
    for step in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] < 1e-6 or h[j] > 1.0 - 1e-6:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.8
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.8
        if delta < 1e-6:
            continue
        h[i] -= delta
        h[j] += delta
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = project_to_sum(h, ts)
        temp = max(1e-6, 0.1 * np.exp(-step / 8000.0))
    return h, c5_cur

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def nested_opt(h0, n, ts, alphas, n_iter_each, lr_init, time_fn, min_time=30):

        if time_fn() < min_time:

    return h
