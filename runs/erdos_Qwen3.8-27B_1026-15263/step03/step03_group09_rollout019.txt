import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_fast(h, alpha):
    """Compute gradient of smooth max using vectorized correlation/convolution."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    B = 1.0 - h
    T1 = np.correlate(w, B, mode='full')[0:n]
    conv = np.convolve(w, h, mode='full')
    T2 = conv[n-1 : 3*n-2]
    
    g = T1 - T2
    return sm_val, g

def adam_optimize_fast(h0, n, target_sum, alpha, n_iters, lr_init, t_start, budget_s):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    patience = 0
    max_patience = 200
    
    for it in range(n_iters):
        if time.time() - t_start > budget_s - 30:
            break
        t += 1
        _, g = get_grad_fast(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                lr *= 0.5
                patience = 0
        h = h_new
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * it)
    
    return best_h, best_c5

- The "least common multiple" structure suggests h should be periodic with period n/p for small p.

Actually, this is getting quite theoretical. Let me just focus on the practical optimization.

Given the reformulation C5 = 1 - 2*min(autocorr)/n, the problem is equivalent to:
Maximize min_j autocorr[j] subject to 0≤h≤1, sum h = n/2.

This is a maximin problem, which is still non-convex, but it gives us a different optimization perspective.

For the smooth approximation, instead of doing soft-max of the overlaps c[k], we can do soft-MIN of the autocorrelations autocorr[j]. The soft-min is: -soft_max(-autocorr).
