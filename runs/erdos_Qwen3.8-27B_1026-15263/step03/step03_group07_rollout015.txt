**Key improvement**: Vectorize the gradient computation using NumPy!

For a given set of active lags K = {k_1, k_2, ..., k_p} with weights w_1, ..., w_p:

This can be computed as:
For each active lag k_p with weight w_p:
  - Add w_p * (1-h) to gradient at positions j where j-k_p is valid: gradient[k_p : k_p+n] += w_p * (1 - h[:n-k_p])
  - Subtract w_p * h from gradient at positions j where j+k_p is valid: gradient[:n-k_p] -= w_p * h[k_p:]
  - For negative lags (k_p < 0), swap the roles

This is O(n * p) with small constants using NumPy sliced operations!

```python
def fast_gradient_active(h, active_lags, weights):
    """
    h: array of length n
    active_lags: array of lag values (k = m - (n-1))
    weights: array of weights for each active lag
    
    Returns gradient of soft-max of corr values at active lags.
    """
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k, w in zip(active_lags, weights):
        if w < 1e-15:
            continue
        if k >= 0:
            # d(corr[n-1+k])/d(h[j]) = (1-h[j-k]) for j>=k, -h[j+k] for j < n-k
            if k == 0:
                grad += w * (one_minus_h - h)
            else:
                # j ranges from k to n-1: contribution (1-h[j-k])
                grad[k:] += w * one_minus_h[:n-k]
                # j ranges from 0 to n-1-k: contribution -h[j+k]
                grad[:n-k] -= w * h[k:]
        else:
            k2 = -k
            # d(corr[n-1-k2])/d(h[j]) = -h[j-k2] for j>=k2, (1-h[j+k2]) for j < n-k2
            if k2 == 0:
                grad += w * (one_minus_h - h)
            else:
                # Lag -k2: corr[n-1-k2] = Σ_i h[i] * (1-h[i+k2])
                # d/d(h[j]): (1-h[j+k2]) for j < n-k2, -h[j-k2] for j >= k2
                grad[:n-k2] += w * one_minus_h[k2:]
                grad[k2:] -= w * h[:n-k2]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_active_lags(corr, n, threshold=0.999):
    """Get lags (as offsets from n-1) where corr is near maximum."""
    max_corr = np.max(corr)
    active = np.where(corr >= threshold * max_corr)[0]
    lags = active - (n - 1)
    return lags, corr[active]

def fast_smooth_max_grad(h, alpha, active_lags=None, weights=None):
    """
    Compute soft-max of correlation values and its gradient.
    If active_lags is None, use all lags.
    """
    n = len(h)
    corr = get_corr(h)
    
    if active_lags is not None:
        idx = active_lags + (n - 1)
        idx = idx[(idx >= 0) & (idx < 2*n-1)]
        sel_corr = corr[idx]
        sel_lags = active_lags[(active_lags + (n-1) >= 0) & (active_lags + (n-1) < 2*n-1)]
    else:
        sel_corr = corr
        sel_lags = np.arange(-(n-1), n)
    
    m = np.max(sel_corr)
    shifted = alpha * (sel_corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    if weights is not None:
        w = weights
    
    # Fast gradient computation
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k, wi in zip(sel_lags, w):
        if wi < 1e-15:
            continue
        k = int(k)
        if k == 0:
            grad += wi * (one_minus_h - h)
        elif k > 0:
            # corr[n-1+k] = Σ_i h[i+k]*(1-h[i])
            # d/dh[j]: (1-h[j-k]) for j>=k, -h[j+k] for j<n-k
            grad[k:] += wi * one_minus_h[:n-k]
            grad[:n-k] -= wi * h[k:]
        else:
            k2 = -k
            # corr[n-1-k2] = Σ_i h[i]*(1-h[i+k2])
            # d/dh[j]: (1-h[j+k2]) for j<n-k2, -h[j-k2] for j>=k2
            grad[:n-k2] += wi * one_minus_h[k2:]
            grad[k2:] -= wi * h[:n-k2]
    
    return sm_val, grad

def penalty_lbgfgs(h0, n, target_sum, alpha, lam, maxiter=500):
    """L-BFGS-B with penalty for sum constraint."""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = fast_smooth_max_grad(h, alpha)
        s = np.sum(h) - target_sum
        return v + lam * s * s
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = fast_smooth_max_grad(h, alpha)
        s = np.sum(h) - target_sum
        return g + 2 * lam * s * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    """Fast coordinate descent with binary search on step size."""
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]
    
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
```

```python
def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_max(h, alpha, active_lags=None):
    """Compute soft-max of correlation values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_lags is not None:
        idx = active_lags + (n - 1)
        valid = (idx >= 0) & (idx < 2*n - 1)
        idx = idx[valid]
        sel = corr[idx]
    else:
        sel = corr
    m = np.max(sel)
    shifted = np.clip(alpha * (sel - m), -500, 500)
    exp_v = np.exp(shifted)

    val = m + np.log(len(sel)) / alpha - np.log(np.sum(exp_v)) / alpha
    # Simpler: val = m + log(sum(exp)) / alpha - log(len) / alpha
    # Or: log-sum-exp trick

def grad_from_weights(h, lags, weights):
    """Compute gradient given lag indices and weights. Vectorized."""
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h

        if w < 1e-15:

        k = int(k)
        if k == 0:
            grad += w * (omh - h)
        elif k > 0:

                grad[:n-k] -= w * h[k:]
        else:
```

```python
    n = len(h)
```

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
