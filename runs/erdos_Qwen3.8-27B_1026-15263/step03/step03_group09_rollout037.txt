import numpy as np
from scipy.optimize import minimize, linprog
import time

# Helper functions: compute_c5, project_to_sum, smooth_max_and_grad, etc.

def lp_descent_step(h, n, target_sum):
    """Solve the LP linearization and return the LP solution."""
    # Build A matrix
    nt = 2*n - 1
    A = np.zeros((nt, n))
    b = np.zeros(nt)
    
    h_t = h
    c_t = np.correlate(h_t, 1.0-h_t, mode='full')
    
    for k in range(nt):
        b[k] = -c_t[k]
        for i in range(n):
            j = i + k - n + 1
            if 0 <= j < n:
                A[k, i] += (1.0 - h_t[j])
            j2 = i - k + n - 1
            if 0 <= j2 < n:
                A[k, i] += -h_t[j2]
    
    # Add the -t column
    A_full = np.hstack([A, -np.ones((nt, 1))])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality constraint: sum(h) = target_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, 2.0)]  # t can be up to 2
    
    res = linprog(c_obj, A_ub=A_full, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h.copy()

    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / nc) / alpha
    
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        js1 = max(0, n - 1 - k)
        je1 = min(n, 2 * n - 1 - k)
        if je1 > js1:
            hs = js1 + k - n + 1
            g[js1:je1] += wk * one_minus_h[hs:hs + (je1 - js1)]
        js2 = max(0, k - n + 1)
        je2 = min(n, k + 1)
        if je2 > js2:
            hs = js2 - k + n - 1
            g[js2:je2] += -wk * h[hs:hs + (je2 - js2)]
    return sm_val, g

def lp_step(h, n, target_sum):
    """One step of LP-based linearization."""
    nc = 2 * n - 1
    c_t = np.correlate(h, 1.0 - h, mode='full')
    
    A = np.zeros((nc, n))
    for k in range(nc):
        # Part 1: coefficient from first factor
        js1 = max(0, n - 1 - k)
        je1 = min(n, 2 * n - 1 - k)
        if je1 > js1:
            hs = js1 + k - n + 1
            A[k, js1:je1] += 1.0 - h[hs:hs + (je1 - js1)]
        # Part 2: coefficient from second factor
        js2 = max(0, k - n + 1)
        je2 = min(n, k + 1)
        if je2 > js2:
            hs = js2 - k + n - 1
            A[k, js2:je2] += -h[hs:hs + (je2 - js2)]
    
    # Full LP matrix: A_full @ [h; t] <= b
    A_full = np.hstack([A, -np.ones((nc, 1))])
    b_ub = -c_t.copy()
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    
    res = linprog(c_obj, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h.copy()
