        else:
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < -1e-15 or (delta / T + np.log(rng.uniform()) < 0.0):
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
        if step % 500 == 0 and step % 50000 == 0:
            h_fast = h.copy()
            c5_fast = c5_curr
            T_fast = T
            relevant = time.time() < t_end - 30
            if relevant:
                for _ in range(2000):
                    ri = rng.randint(n)
                    rj = rng.randint(n)
                    if h_fast[ri] == h_fast[rj]:
                        continue
                    h_fast[ri], h_fast[rj] = h_fast[rj], h_fast[ri]
                    c5_f = compute_c5(h_fast)
                    d = c5_f - c5_fast
                    if d < -1e-15 or (d / T_fast + np.log(rng.uniform()) < 0.0):
                        c5_fast = c5_f
                    else:
                        h_fast[ri], h_fast[rj] = h_fast[rj], h_fast[ri]
                    T_fast = max(T_fast * 0.999, 0.005)
                    if c5_f < best_c5 - 1e-14:
                        best_c5 = c5_f
                        best_h = h_fast.copy()
        T = max(T * 0.999997, 0.001)
        if no_improve > stall_limit:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.03)
            no_improve = 0
            n_swaps_kick = rng.randint(3, 8)
            h = kick(h, n, half, rng, n_swaps_kick)
            c5_curr = compute_c5(h)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n // 2)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    half = n // 2

    h_bin = binarize(best_h, half)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    t1_end = t_start + 45.0
    h_ls, c5_ls = local_search(h_bin, n, half, t1_end, rng)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    t2_end = t_start + 520.0
    h_sa, c5_sa = multi_sa(best_h, n, half, t2_end, rng, n_chains=1, T_base=0.025)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    t3_end = min(t_start + 555.0, t_start + budget_s - 400.0)
    if t3_end > time.time():
        h_ls2, c5_ls2 = local_search(best_h, n, half, t3_end, rng)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    n_restarts = 8
    for r in range(n_restarts):
        if time.time() > t_start + budget_s - 200.0:
            break
        n_kick = 4 + 2 * r
        h_k = kick(best_h, n, half, rng, n_kick)
        t_r_end = time.time() + 20.0 + 3.0 * (n_restarts - r)
        t_r_end = min(t_r_end, t_start + budget_s - 200.0 - r * 23.0)
        if t_r_end <= time.time():
            continue
        h_sr, c5_sr = sa_run(h_k, n, half, t_r_end, rng,
                             T_init=0.028 - 0.002 * r, T_min=0.001)
        if c5_sr < best_c5:
            best_c5 = c5_sr
            best_h = h_sr.copy()
        if time.time() < t_start + budget_s - 200.0:
            t_lsr = min(time.time() + 12.0, t_start + budget_s - 200.0)
            h_lsr, c5_lsr = local_search(h_sr, n, half, t_lsr, rng)
            if c5_lsr < best_c5:
                best_c5 = c5_lsr
                best_h = h_lsr.copy()

    if time.time() < t_start + budget_s - 100.0:
        t_fin_ls = min(time.time() + 30.0, t_start + budget_s - 100.0)
        h_fin, c5_fin = local_search(best_h, n, half, t_fin_ls, rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if time.time() < t_start + budget_s - 80.0:
        n2 = 480
        half2 = n2 // 2
        x_src = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_dst, x_src, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, half2)
        h_bin2 = binarize(h_interp, half2)
        c5_480 = compute_c5(h_bin2)
        if c5_480 < best_c5:
            best_c5 = c5_480
            best_h = h_bin2.copy()
            best_n = n2
        t_480_end = min(time.time() + 50.0, t_start + budget_s - 40.0)
        if t_480_end > time.time():
            h_480_ls, c5_480_ls = local_search(h_bin2, n2, half2, t_480_end, rng)
            if c5_480_ls < best_c5:
                best_c5 = c5_480_ls
                best_h = h_480_ls.copy()
                best_n = n2

    if time.time() < t_start + budget_s - 30.0:
        h_final_sa_end = t_start + budget_s - 20.0
        h_fsa, c5_fsa = sa_run(best_h, best_n, best_n // 2, h_final_sa_end, rng,
                               T_init=0.012, T_min=0.001)
        if c5_fsa < best_c5:
            best_c5 = c5_fsa
            best_h = h_fsa.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_n // 2)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
