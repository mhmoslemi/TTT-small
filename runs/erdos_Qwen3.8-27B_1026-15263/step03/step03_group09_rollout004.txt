    # Use np.take with mode='clip' to avoid out-of-bounds
    I2_clipped = np.clip(I2, 0, n - 1)
    I3_clipped = np.clip(I3, 0, n - 1)
    
    term1 = np.where(Mask2, one_minus_h[I2_clipped], 0.0)
    term2 = np.where(Mask3, h[I3_clipped], 0.0)
