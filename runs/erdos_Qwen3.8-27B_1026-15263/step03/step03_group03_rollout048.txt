def compute_B(h, j, n):

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad, corr

def projected_gradient_smooth(h0, n, target_sum, alpha, n_iter, lr_init, momentum):

    vel = np.zeros(n)
    lr = lr_init
    best_obj = np.inf
    best_h = h.copy()
    for it in range(n_iter):
        obj, grad, corr = smooth_max_obj_grad(h, alpha)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        # Momentum
        vel = momentum * vel - lr * grad
        h_new = h + vel
        # Project to [0,1] with sum constraint
        h_new = project_to_sum(h_new, target_sum)
        # Check if we improved
        obj_new, _, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj + 1e-12:
            h = h_new
        else:
            # Halve learning rate
            lr *= 0.5
            vel = 0.5 * vel
            h_new = h + vel
            h_new = project_to_sum(h_new, target_sum)
            h = h_new
        if it % 100 == 0:
            lr = lr_init * (0.95 ** (it // 100))
    return best_h

    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Structured + random candidates
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (j + n // 4) % n, (j + 3 * n // 4) % n,
                 (j + n // 5) % n, (j + 2 * n // 5) % n,
                 (j + 3 * n // 5) % n, (j + n // 6) % n, (j + 5 * n // 6) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-15:
                continue
            delta_opt = solve_1d_precise(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-15:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-15:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def swap_descent(h, corr, rng, n_trials=30):
    """Try swapping pairs of coordinates"""
    n = len(h)
    improved = False
    for _ in range(n_trials):
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        B_j = compute_B(h, j, n)
        B_k = compute_B(h, k, n)
        # Swap means delta = (h[k] - h[j]) / 2 applied to j, and negative to k
        # Actually swap: h[j] += d, h[k] -= d where d = (h[k]-h[j])/2
        d = (h[k] - h[j]) * 0.5
        if abs(d) < 1e-14:
            continue
        if h[j] + d < -1e-14 or h[j] + d > 1 + 1e-14:
            continue
        if h[k] - d < -1e-14 or h[k] - d > 1 + 1e-14:
            continue
        D = B_j - B_k
        new_corr = corr + d * D
        if np.max(new_corr) < np.max(corr) - 1e-14:
            corr = new_corr
            h[j], h[k] = h[k], h[j]
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]).astype(float) * w[None, :]
    cs = valid_s * h[sc].astype(float) * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def projected_gradient_smooth(h0, n, target_sum, alpha, n_iter, lr_init=0.005, momentum=0.9):
    h = h0.copy()
    vel = np.zeros(n)
    lr = lr_init
    for it in range(n_iter):
        obj, grad = smooth_max_obj_grad(h, alpha)
        vel = momentum * vel - lr * grad
        h_new = h + vel
        h_new = project_to_sum(h_new, target_sum)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new <= obj + 1e-14:
            h = h_new
        else:
            lr = max(lr * 0.5, 1e-8)
            vel *= 0.5
            h_new = h + vel
            h = project_to_sum(h_new, target_sum)
        if it % 50 == 0 and it > 0:
            lr = lr_init * (0.97 ** (it // 50))
    return h

def project_to_sum(h, target_sum):

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break

        else:
            h = h + diff / len(h)

            idx = np.argmax(1.0 - h)

def compute_B(h, j, n):

    B = np.zeros(2 * n - 1)

    if k_start2 <= k_end2:

    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall_count = 0

    while time.time() < t_end1:
        h_work, corr_w, _ = pair_descent_sweep(h_work, corr_w, rng, n_cand=20)
        h_work, corr_w, _ = swap_descent(h_work, corr_w, rng, n_trials=20)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-13:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall_count = 0
        else:
            stall_count += 1
        if stall_count == 40:
            h_work = partial_perturb(h_work, rng, n_w / 2.0, frac=0.12, noise=0.02)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0
        elif stall_count == 100:
            h_work = partial_perturb(h_work, rng, n_w / 2.0, frac=0.25, noise=0.04)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0
        elif stall_count == 200:
            h_work = project_to_sum(best_h + rng.uniform(-0.06, 0.06, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0
        elif stall_count >= 350:
            h_work = project_to_sum(best_h + rng.uniform(-0.1, 0.1, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0

    t_end2 = t_start + 0.68 * budget_s
    resolutions = [120, 160, 200, 240, 280, 320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        cand_starts = [project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)]
        cand_starts.append(project_to_sum(rng.uniform(0, 1, n_tr), ts_tr))
        for h_st in cand_starts:
            if time.time() >= t_end2:
                break
            h_st = project_to_sum(h_st, ts_tr)
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            sweeps = min(25, max(2, int((t_end2 - time.time()) / 5.0)))
            st_stall = 0
            for _ in range(sweeps):
                if time.time() >= t_end2:
                    break
                h_st, corr_st, _ = pair_descent_sweep(h_st, corr_st, rng, n_cand=14)
                c5_st = compute_c5(h_st)
                if c5_st < best_c5:
                    best_c5 = c5_st
                    best_h = h_st.copy()
                    best_n = n_tr
                    st_stall = 0
                else:
                    st_stall += 1
                if st_stall == 20:
                    h_st = partial_perturb(h_st, rng, ts_tr, frac=0.2, noise=0.03)
                    corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
                    st_stall = 0

    t_end3 = t_start + 0.82 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    h_pol = best_h.copy()
    for alpha in [80, 250, 600, 1500]:
        if time.time() >= t_end3:
            break
        n_iters = max(20, int((t_end3 - time.time()) / 5.0))
        h_pol = projected_gradient_smooth(h_pol, n_f, ts_f, alpha, n_iters)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f
        if time.time() < t_end3:
            h_pol2 = h_pol.copy()
            corr_pol = np.correlate(h_pol2, 1.0 - h_pol2, mode="full")
            n_sweeps2 = min(15, max(2, int((t_end3 - time.time()) / 4.0)))
            for _ in range(n_sweeps2):
                if time.time() >= t_end3:
                    break
                h_pol2, corr_pol, _ = pair_descent_sweep(h_pol2, corr_pol, rng, n_cand=16)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
                best_n = n_f

    t_end4 = t_start + budget_s - 4
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall_fin = 0
    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=20)
        h_fin, corr_fin, _ = swap_descent(h_fin, corr_fin, rng, n_trials=15)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-13:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            stall_fin += 1
        if stall_fin == 30:
            h_fin = partial_perturb(h_fin, rng, n_fin / 2.0, frac=0.1, noise=0.01)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall_fin = 0
        elif stall_fin == 80:
            h_fin = partial_perturb(h_fin, rng, n_fin / 2.0, frac=0.2, noise=0.03)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall_fin = 0
        elif stall_fin >= 160:
            h_fin = project_to_sum(best_h + rng.uniform(-0.04, 0.04, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall_fin = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
