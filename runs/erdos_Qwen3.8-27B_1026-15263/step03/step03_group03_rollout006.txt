def smooth_max_obj_and_grad(h, alpha, penalty_lambda):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_s = np.exp(shifted - np.max(shifted))
    log_sum_exp = np.log(exp_s.sum())
    obj_soft = m + log_sum_exp / alpha
    # Penalty for sum constraint
    sum_diff = np.sum(h) - target
    obj = obj_soft + penalty_lambda * sum_diff ** 2
    
    # Gradient
    w = exp_s / exp_s.sum()
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)

    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]).astype(float) * w[None, :]
    cs = valid_s * h[sc].astype(float) * w[None, :]
    grad = np.sum(cf - cs, axis=1) + 2 * penalty_lambda * sum_diff
    return obj, grad

    target = n / 2.0
    h0 = project_to_sum(h0, target)
    
    def fun(x):
        return smooth_max_obj_and_grad(x, alpha, penalty_lambda)[0]
    
    def jac(x):
        return smooth_max_obj_and_grad(x, alpha, penalty_lambda)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0.copy(), method='L-BFGS-B', jac=jac,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        h_opt = project_to_sum(res.x, target)
        return h_opt

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]

def solve_1d(A, D, lo, hi):

    best_idx = int(np.argmin(vals))
    if best_idx == 0:

    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]

        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D

            h[j] += best_delta

            if h[j] < 0: h[j] = 0.0

            if h[best_k] < 0: h[best_k] = 0.0
            elif h[best_k] > 1: h[best_k] = 1.0
            improved = True

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    return h


def smooth_max_obj_grad(h, alpha, pen_lambda):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_s = np.exp(shifted - np.max(shifted))
    exp_sum = exp_s.sum()
    obj_soft = m + np.log(exp_sum) / alpha
    sum_diff = np.sum(h) - target
    obj = obj_soft + pen_lambda * sum_diff * sum_diff
    w = exp_s / exp_sum
    nm1 = n - 1
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * nn, 1)[None, :] if False else np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + nm1
    second_idx = j_idx + m_idx - nm1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f.astype(float) * (1.0 - h[fc]) * w[None, :]
    cs = valid_s.astype(float) * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1) + 2.0 * pen_lambda * sum_diff
    return obj, grad


def lfbgs_step(h0, n, alpha, pen_lambda, maxiter):
    target = n / 2.0
    h0 = project_to_sum(h0, target)

    def fun(x):
        return smooth_max_obj_grad(x, alpha, pen_lambda)[0]

    def jac(x):
        return smooth_max_obj_grad(x, alpha, pen_lambda)[1]

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0.copy(), method='L-BFGS-B', jac=jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0, target)


def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B


def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5


def pair_descent_sweep(h, corr, rng, n_cand):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = list(((j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                      (j + n // 3) % n, (j + 2 * n // 3) % n))
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = min(1.0, max(0.0, h[j] + best_delta))
            h[best_k] = min(1.0, max(0.0, h[best_k] - best_delta))
            improved = True
    return h, corr, improved

    return np.interp(x_new, x_old, h, period=2.0)

            h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0
        elif stall_count >= 300:
            h_work = perturb_and_descend(h_work, n_w, rng, 0.08, 15, 10)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0

    # Phase 3: L-BFGS-B polish at best and adjacent resolutions (13%)
    t_end3 = t_start + 0.88 * budget_s
    polish_res = [best_n]
    for delta in [-40, 40, -80, 80]:
        np_try = best_n + delta
        if np_try >= 120 and np_try <= 640 and np_try % 2 == 0 and np_try not in polish_res:
            polish_res.append(np_try)

    for n_pol in polish_res:
        if time.time() >= t_end3 or (budget_s - (time.time() - t_start)) < 15:
            break
        ts_pol = n_pol / 2.0
        if n_pol == len(best_h):
            h_pol = best_h.copy()
        else:
            h_pol = project_to_sum(upsample(best_h, n_pol), ts_pol)
        for alpha in [200, 600, 2000]:
            if time.time() >= t_end3 or (budget_s - (time.time() - t_start)) < 10:
                break
            h_pol = lfbgs_step(h_pol, n_pol, alpha, pen_lambda, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_pol

    # Phase 4: Final pair descent polish (12%)
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    t_stop = t_start + budget_s - 3
    while time.time() < t_stop:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, 16)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
