For the Frank-Wolfe method:
- At each iteration j:
  1. Compute smooth max objective and gradient: (f, g)
  2. Frank-Wolfe direction: s* = argmin_{s∈[0,1]^n, sum(s)=n/2} <g, s>
     Since the feasible set is a hypersimplex (intersection of hypercube and hyperplane):
     s*_i = 1 for the n/2 indices with smallest g_i, 0 otherwise
  3. Line search: find α ∈ [0,1] minimizing f(h + α(s* - h))
     Project h + α(s*-h) to feasible set
     (Actually, for Frank-Wolfe, we typically do: h_new = (1-α)h + α*s*
     and this is automatically in the convex hull, hence feasible)
  4. Update: h ← (1-α)h + α*s*

The line search can be exact (try a few α values) or use the standard FW schedule α = 2/(t+2).

For the GA:
- Population of binary vectors with exactly n/2 ones
- Fitness: C5 value
- Crossover: uniform (maintain sum constraint by swapping)
- Mutation: flip a pair (one 0→1, one 1→0)
- Selection: tournament

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = shifted.max()
    exp_vals = np.exp(alpha * (shifted - max_shift))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    obj = (m_val + max_shift + np.log(sum_exp) / alpha) * dx
    
    # Gradient computation (vectorized)
    j_idx = np.arange(n)[:, None].astype(np.float64)
    m_idx = np.arange(2*n-1)[None, :].astype(np.float64)
    h_idx_A = j_idx + m_idx - (n-1)
    valid_A = (h_idx_A >= 0) & (h_idx_A <= n-1)
    term_A = np.where(valid_A, h[np.clip(h_idx_A.astype(int), 0, n-1)] * w[None, :], 0.0).sum(axis=1)
    
    h_idx_B = j_idx - m_idx + (n-1)
    valid_B = (h_idx_B >= 0) & (h_idx_B <= n-1)
    term_B = np.where(valid_B, h[np.clip(h_idx_B.astype(int), 0, n-1)] * w[None, :], 0.0).sum(axis=1)
    
    grad = dx * (np.sum(w) - term_A - term_B)
    return obj, grad

def frank_wolfe_step(h, g, n):
    """Find FW direction: s* = argmin <g, s> over s in [0,1]^n, sum(s) = n/2"""
    # Sort by gradient: pick n/2 smallest
    n_ones = n // 2
    sorted_idx = np.argsort(g)
    s_star = np.zeros(n)
    s_star[sorted_idx[:n_ones]] = 1.0
    return s_star

def fw_line_search(h, s_star, alpha_init, n, alpha_sm):
    """Exact line search for FW step."""
    best_alpha = 0.0
    best_obj = smooth_max_and_grad(h, alpha_sm)[0]
    for a in np.concatenate([np.linspace(0.0, 1.0, 20), [alpha_init]]):
        h_try = (1.0 - a) * h + a * s_star
        # This is automatically in [0,1]^n and sum = n/2
        obj_try = smooth_max_and_grad(h_try, alpha_sm)[0]
        if obj_try < best_obj:
            best_obj = obj_try
            best_alpha = a
    return best_alpha, best_obj

def ga_evolve(pop, fitnesses, rng, n, pop_size, mutation_rate=0.15):
    """One GA generation for binary vectors with exactly n/2 ones."""
    new_pop = []
    # Elitism: keep best 2
    order = np.argsort(fitnesses)
    for i in range(min(2, len(pop))):
        new_pop.append(pop[order[i]].copy())
    
    while len(new_pop) < pop_size:
        # Tournament selection
        p1 = pop[order[rng.randint(len(order))]]
        p2 = pop[order[rng.randint(len(order))]]
        parent = p1 if fitnesses[order[rng.randint(len(order))]] <= fitnesses[order[rng.randint(len(order))]] else p2
        
        # Actually, let me do proper tournament
        idxs = rng.choice(len(pop), size=4, replace=False)
        winner = idxs[np.argmin(fitnesses[idxs])]
        parent = pop[winner]
        
        # Mutation: flip pairs to maintain sum
        child = parent.copy()
        n_flips = max(1, int(n * mutation_rate * 0.5))
        ones = np.where(child > 0.5)[0]
        zeros = np.where(child < 0.5)[0]
        for _ in range(n_flips):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            if i == j:
                continue
            child[i] = 0.0
            child[j] = 1.0
            ones = np.where(child > 0.5)[0]
            zeros = np.where(child < 0.5)[0]
        new_pop.append(child)
    
    return new_pop

def binary_to_sum(h, n, rng):
    """Adjust binary vector to have exactly n/2 ones."""
    h = (h > 0.5).astype(float)
    curr = int(h.sum())
    target = n // 2
    if curr > target:
        excess = curr - target
        ones = np.where(h > 0.5)[0]
        rng.shuffle(ones)
        for i in range(excess):
            h[ones[i]] = 0.0
    elif curr < target:
        deficit = target - curr
        zeros = np.where(h < 0.5)[0]
        rng.shuffle(zeros)
        for i in range(deficit):
            h[zeros[i]] = 1.0
    return h
