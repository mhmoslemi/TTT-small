def solve_1d(A, D, lo, hi):
    """Minimize max(A + D*t) for t in [lo, hi]."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Grid search for initial bracket
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    # Bisection
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Candidate selection
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, 
                 (j + n // 3) % n, (j + 2 * n // 3) % n, (j + n // 4) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = max(0.0, min(1.0, h[j]))
            h[best_k] = max(0.0, min(1.0, h[best_k]))
            improved = True
    return h, corr, improved

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * nm1, 2 * nm1 - j)
    if k_start < k_end:
        B[k_start:k_end] += (1.0 - h[k_start - nm1 + j : k_end - nm1 + j])
    k_start2 = max(0, j)
    k_end2 = min(2 * nm1, j + nm1)
    if k_start2 < k_end2:
        B[k_start2:k_end2] += -h[j + nm1 - k_end2 : j + nm1 - k_start2 + 1][::-1]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):

    order = np.arange(n)
    rng.shuffle(order)

        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:

                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:

            h[j] = max(0.0, min(1.0, h[j] + best_delta))
            h[best_k] = max(0.0, min(1.0, h[best_k] - best_delta))
            improved = True
    return h, corr, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    nm1 = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    # Vectorized gradient
    j_idx = np.arange(n)[:, None]
    i_idx = np.arange(n)[None, :]
    k1 = i_idx + nm1 - j_idx
    k2 = j_idx + nm1 - i_idx
    valid1 = (k1 >= 0) & (k1 < 2 * nm1)
    valid2 = (k2 >= 0) & (k2 < 2 * nm1)
    k1c = np.clip(k1, 0, 2 * nm1 - 1)
    k2c = np.clip(k2, 0, 2 * nm1 - 1)
    term1 = valid1 * (1.0 - h)[None, :] * w[k1c]
    term2 = valid2 * h[None, :] * w[k2c]
    grad = term1.sum(axis=1) - term2.sum(axis=1)
    return obj, grad, corr

def gradient_refine(h, alpha, lr_scale, n_steps=50):
    """Do gradient descent on smooth max with projection."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        obj, grad, _ = smooth_max_grad(h, alpha)
        # Line search
        lr = lr_scale
        for _ in range(10):
            h_try = project_to_sum(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                break
            lr *= 0.5
        else:
            break
    return h, best_c5

def slsqp_optimize(h0, n, target, alpha, maxiter=200):
    def obj_fn(h):
        o, _, _ = smooth_max_grad(h, alpha)
        return o
    def grad_fn(h):
        _, g, _ = smooth_max_grad(h, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        result = project_to_sum(res.x, target)
        return result
    except Exception:
        return project_to_sum(h0.copy(), target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * nm1, 2 * nm1 - j)
    if k_start < k_end:
        B[k_start:k_end] += (1.0 - h[k_start - nm1 + j:k_end - nm1 + j])
    k_start2 = max(0, j)
    k_end2 = min(2 * nm1, j + nm1)
    if k_start2 < k_end2:
        B[k_start2:k_end2] += -h[j + nm1 - k_end2:j + nm1 - k_start2 + 1][::-1]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(44):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def smooth_max_grad(h, alpha):
    n = len(h)
    nm1 = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    j_idx = np.arange(n)[:, None]
    i_idx = np.arange(n)[None, :]
    k1 = i_idx + nm1 - j_idx
    k2 = j_idx + nm1 - i_idx
    valid1 = (k1 >= 0) & (k1 < 2 * nm1)
    valid2 = (k2 >= 0) & (k2 < 2 * nm1)
    k1c = np.clip(k1, 0, 2 * nm1 - 1)
    k2c = np.clip(k2, 0, 2 * nm1 - 1)
    term1 = valid1 * (1.0 - h)[None, :] * w[k1c]
    term2 = valid2 * h[None, :] * w[k2c]
    grad = term1.sum(axis=1) - term2.sum(axis=1)
    return obj, grad, corr

def pair_descent_sweep(h, corr, rng, n_cand=12, alpha=50):
    n = len(h)
    nm1 = n - 1
    m = np.max(corr)
    shifted = corr - m
    w = np.exp(alpha * shifted)
    w /= np.sum(w)
    j_idx = np.arange(n)[:, None]
    i_idx = np.arange(n)[None, :]
    k1m = i_idx + nm1 - j_idx
    k2m = j_idx + nm1 - i_idx
    valid1 = (k1m >= 0) & (k1m < 2 * nm1)
    valid2 = (k2m >= 0) & (k2m < 2 * nm1)
    k1c = np.clip(k1m, 0, 2 * nm1 - 1)
    k2c = np.clip(k2m, 0, 2 * nm1 - 1)
    term1 = valid1 * (1.0 - h)[None, :] * w[k1c]
    term2 = valid2 * h[None, :] * w[k2c]

def lbfgs_smooth(h0, alpha, target, penalty, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_max_grad(x, alpha)[0] + penalty * (np.sum(x) - target) ** 2,
            h0.copy(),
            method='L-BFGS-B',
            jac=lambda x: smooth_max_grad(x, alpha)[1] + 2 * penalty * (np.sum(x) - target) * np.ones(n),
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0.copy(), target)

def slsqp_smooth(h0, alpha, target, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_max_grad(x, alpha)[0],
            h0.copy(),
            method='SLSQP',
            jac=lambda x: smooth_max_grad(x, alpha)[1],
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0.copy(), target)
