import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def pair_swap_sa(h, n, rng, time_fn, T_init=0.005, max_steps=500000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for step in range(max_steps):
        if time_fn() < 5:
            break
        T = T_init * (0.9995 ** step)
        
        i, j = rng.choice(n, size=2, replace=False)
        if abs(h[i] - h[j]) < 1e-10:
            continue
        h[i], h[j] = h[j], h[i]
        c5_try = compute_c5(h)
        delta = c5_try - c5_cur
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

    h = h0.copy()
    lr = lr_init
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):

        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)

        else:

def upsample(h, n_new):
    n_old = len(h)

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)

        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)

def transfer_sa(h, n, rng, time_fn, delta_init=0.3, max_steps=500000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur

        if time_fn() < 5:
            break
        # Adaptive delta
        frac = step / max(step + 1, max_steps)
        delta = delta_init * (1 - 0.999 * frac)
        
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.uniform(-delta, delta)
        
        if h[i] + d < 0 or h[j] - d < 0 or h[i] + d > 1 or h[j] - d > 1:
            continue
        
        h[i] += d
        h[j] -= d
        c5_try = compute_c5(h)

            h[j] += d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

def pnorm_obj_and_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = np.sum(np.abs(corr) ** p) ** (1.0 / p) * (2.0 / n)
    sign_corr = np.sign(corr)
    grad_corr = abs(corr) ** (p - 1) * sign_corr / (np.sum(np.abs(corr) ** p) ** (1.0 - 1.0 / p)) if np.sum(np.abs(corr)) > 0 else corr
    weight = (abs(corr) ** (p - 1)) / (np.sum(np.abs(corr) ** p) ** (1.0 / p) * p) if np.max(np.abs(corr)) > 1e-15 else np.ones(len(corr)) / len(corr)
    v = 1.0 - 2.0 * h
    grad = np.correlate(weight, v, mode="valid") * (2.0 / n)
    return obj, grad

    lr = lr_init
    if use_pnorm:
        best_obj = pnorm_obj_and_grad(h, p)[0]
    else:
        best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    best_h = h.copy()
    for it in range(max_iter):
        if time_fn() < 5:
            break
        if use_pnorm:
            obj, grad = pnorm_obj_and_grad(h, p)
        else:
            obj, grad = smooth_max_obj_and_grad(h, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        if use_pnorm:
            obj_new = pnorm_obj_and_grad(h_new, p)[0]
        else:
            obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new > obj:
            lr *= 0.5
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            if use_pnorm:
                obj_new = pnorm_obj_and_grad(h_new, p)[0]
            else:
                obj_new = smooth_max_obj_and_grad(h_new, alpha)[0]
        else:
            lr = min(lr * 1.15, 0.5)
        h = h_new
    return best_h, best_obj

def transfer_sa(h, n, rng, time_fn, delta_init=0.4, max_steps=500000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(max_steps):
        if time_fn() < 3:
            break
        frac = step / max(step + 1, max_steps)
        delta = delta_init * (1.0 - 0.95 * frac) + 0.01
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-delta, delta)
        hi_new = h[i] + d
        hj_new = h[j] - d
        if hi_new < -1e-10 or hi_new > 1.0 + 1e-10 or hj_new < -1e-10 or hj_new > 1.0 + 1e-10:
            continue
        h[i] = hi_new
        h[j] = hj_new
        c5_try = compute_c5(h)
        T = 0.004 * (0.9999 ** step)
        delta_c5 = c5_try - c5_cur
        if delta_c5 < -1e-15 or (T > 1e-12 and rng.rand() < np.exp(-delta_c5 / T)):
            c5_cur = c5_try
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c5

def pair_swap_sa(h, n, rng, time_fn, max_steps=300000):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.003
    for step in range(max_steps):
        if time_fn() < 3:
            break
        T = 0.003 * (0.9998 ** step)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or abs(h[i] - h[j]) < 1e-10:
            continue
        h[i], h[j] = h[j], h[i]
        c5_try = compute_c5(h)
        delta = c5_try - c5_cur
        if delta < -1e-15 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            c5_cur = c5_try
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_box_sum(h_new, n_new / 2.0)

def fourier_to_grid(coeffs, n, n_harm):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, n_harm + 1):
        a = coeffs[2 * (m - 1)]
        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return h

def grid_to_fourier(h, n, n_harm):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    C = np.zeros((n, 2 * n_harm))
    for m in range(1, n_harm + 1):
        C[:, 2 * (m - 1)] = np.cos(np.pi * m * x)
        C[:, 2 * (m - 1) + 1] = np.sin(np.pi * m * x)
    target = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(C, target, rcond=None)
    return coeffs

def fourier_obj(c, n, n_harm, alpha):
    h = fourier_to_grid(c, n, n_harm)
    h = project_box_sum(h, n / 2.0)
    obj, _ = smooth_max_obj_and_grad(h, alpha)
    return obj

def alternating_optimize(h0, n, rng, time_fn, n_cycles=20):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for cycle in range(n_cycles):
        if time_fn() < 10:
            break
        for alpha in [50, 500, 5000]:
            if time_fn() < 5:
                break
            h_g, _ = projected_gradient_descent(h, n, alpha, 500, 0.03, time_fn)
            c5_g = compute_c5(h_g)
            if c5_g < c5_cur:
                c5_cur = c5_g
                h = h_g
        if time_fn() < 5:
            break
        h_s, c5_s = transfer_sa(h, n, rng, time_fn, delta_init=0.3, max_steps=50000)
        if c5_s < c5_cur:
            c5_cur = c5_s
            h = h_s
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n

    # Phase 1: Quick gradient polish at original resolution
    for alpha in [50, 500, 5000, 20000]:
        if time_left() < 15:
            break
        h_g, _ = projected_gradient_descent(best_h, n1, alpha, 1000, 0.03, time_left)
        c5_g = compute_c5(h_g)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Transfer SA at original resolution
    if time_left() > 40:
        h_sa, c5_sa = transfer_sa(best_h, n1, rng, time_left, delta_init=0.4, max_steps=200000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Upsample to 480 and optimize
    if time_left() > 100:
        n_up = 480
        h_480 = upsample(best_h, n_up)
        # Multiple gradient passes with increasing alpha
        for alpha in [20, 100, 500, 2000, 10000, 50000]:
            if time_left() < 20:
                break
            h_g, _ = projected_gradient_descent(h_480, n_up, alpha, 1500, 0.02, time_left)
            c5_g = compute_c5(h_g)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
                best_n = n_up
                h_480 = h_g.copy()
        # Transfer SA at 480
        if time_left() > 60:
            h_sa2, c5_sa2 = transfer_sa(h_480, n_up, rng, time_left, delta_init=0.3, max_steps=400000)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n_up
        # More gradient after SA
        if time_left() > 30:
            for alpha in [5000, 50000]:
                if time_left() < 15:
                    break
                h_g2, _ = projected_gradient_descent(best_h, best_n, alpha, 1000, 0.02, time_left)
                c5_g2 = compute_c5(h_g2)
                if c5_g2 < best_c5:
                    best_c5 = c5_g2
                    best_h = h_g2.copy()

    # Phase 4: Try p-norm optimization at current best resolution
    if time_left() > 80:
        n_p = best_n
        for p in [20, 50, 100]:
            if time_left() < 20:
                break
            h_p, _ = projected_gradient_descent(best_h, n_p, 0, 2000, 0.02, time_left, use_pnorm=True, p=p)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 5: Upsample to 720 if time permits
    if time_left() > 150:
        n_720 = 720
        h_720 = upsample(best_h, n_720)
        for alpha in [50, 500, 5000, 50000]:
            if time_left() < 25:
                break
            h_g, _ = projected_gradient_descent(h_720, n_720, alpha, 1500, 0.015, time_left)
            c5_g = compute_c5(h_g)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
                best_n = n_720
                h_720 = h_g.copy()
        if time_left() > 60:
            h_sa3, c5_sa3 = transfer_sa(h_720, n_720, rng, time_left, delta_init=0.2, max_steps=300000)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()
                best_n = n_720
        if time_left() > 20:
            h_g3, _ = projected_gradient_descent(best_h, best_n, 50000, 2000, 0.01, time_left)
            c5_g3 = compute_c5(h_g3)
            if c5_g3 < best_c5:
                best_c5 = c5_g3
                best_h = h_g3.copy()

    # Phase 6: Fourier optimization for structured solutions
    if time_left() > 80:
        n_f = best_n
        if n_f % 2 != 0:
            n_f += 1
            h_f = upsample(best_h, n_f)
        else:
            h_f = best_h.copy()
        n_harm = min(80, max(20, n_f // 4))
        coeffs = grid_to_fourier(h_f, n_f, n_harm)
        for alpha in [100, 1000, 10000]:
            if time_left() < 30:
                break
            try:
                res = minimize(
                    fourier_obj, coeffs,
                    args=(n_f, n_harm, alpha),
                    method='L-BFGS-B',
                    options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-14}
                )
                h_opt = fourier_to_grid(res.x, n_f, n_harm)
                h_opt = project_box_sum(h_opt, n_f / 2.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_f
                    coeffs = grid_to_fourier(h_opt, n_f, n_harm)
            except Exception:
                pass

    # Phase 7: Multi-start perturbation
    if time_left() > 50:
        n_sh = best_n
        for trial in range(8):
            if time_left() < 15:
                break
            noise = 0.08 * (1.0 - trial / 8.0) + 0.01
            h_shake = best_h + rng.uniform(-noise, noise, n_sh)
            h_shake = project_box_sum(h_shake, n_sh / 2.0)
            h_opt, _ = projected_gradient_descent(h_shake, n_sh, 5000, 800, 0.02, time_left)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_sh
            if time_left() > 20:
                h_sa4, c5_sa4 = transfer_sa(h_opt, n_sh, rng, time_left, delta_init=0.2, max_steps=30000)
                if c5_sa4 < best_c5:
                    best_c5 = c5_sa4
                    best_h = h_sa4.copy()

    # Phase 8: Final intensive SA + gradient loop
    if time_left() > 40:
        n_final = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        for cyc in range(10):
            if time_left() < 10:
                break
            h_g, _ = projected_gradient_descent(h_cur, n_final, 50000, 500, 0.01, time_left)
            c5_g = compute_c5(h_g)
            if c5_g < c5_cur:
                c5_cur = c5_g
                h_cur = h_g
            h_s, c5_s = transfer_sa(h_cur, n_final, rng, time_left, delta_init=0.15, max_steps=50000)
            if c5_s < c5_cur:
                c5_cur = c5_s
                h_cur = h_s
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
