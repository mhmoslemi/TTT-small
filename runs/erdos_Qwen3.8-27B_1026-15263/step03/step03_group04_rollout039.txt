import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, shift):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def multi_lag_subgrad(h, n, target_sum, t0, time_limit, max_iters=10000, top_T=12):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.04
    patience = 0
    max_patience = 800

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = compute_corr(h)
        idx = np.argsort(corr)[::-1][:top_T]
        vals = corr[idx]
        temp = max(0.3, vals[0] - vals[-1])
        exp_v = np.exp((vals - vals[0]) / temp)
        w = exp_v / np.sum(exp_v)

        total_grad = np.zeros(n)
        for i in range(len(idx)):
            shift = idx[i] - (n - 1)
            total_grad += w[i] * grad_at_lag(h, shift)

        momentum = beta * momentum + (1.0 - beta) * total_grad
        h_new = h - lr * momentum
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
            lr = min(lr * 1.01, 0.3)
        else:
            h = h_new
            patience += 1
            if patience % 200 == 0:
                lr *= 0.92
                if lr < 0.0005:
                    break
                if patience > max_patience:
                    break
    return best_h, best_c5

    best_h = h.copy()

                best_h = h_opt

    return best_h, best_c5

def smooth_max_optimize(h, n, target_sum, alpha, maxiter):
    def obj_and_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        s = np.sum(exp_v)
        w = exp_v / s
        obj = m + np.log(s) / alpha
        grad = np.zeros(n)
        thresh = 1e-10
        sig = np.where(w > thresh)[0]
        for k in sig:
            wi = w[k]
            shift = k - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
        return obj, grad

    def fun(x):
        obj, _ = _obj_grad(x, n, alpha)
        return obj

    def jac(x):
        _, g = _obj_grad(x, n, alpha)
        return g

    def _obj_grad(x, nn, alpha):
        corr = compute_corr(x)
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        s = np.sum(exp_v)
        w = exp_v / s
        obj = m + np.log(s) / alpha
        grad = np.zeros(nn)
        thresh = 1e-10
        sig = np.where(w > thresh)[0]
        for k in sig:
            wi = w[k]
            shift = k - (nn - 1)
            lo1 = max(0, -shift)
            hi1 = min(nn, nn - shift)
            lo2 = max(0, shift)
            hi2 = min(nn, nn + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += wi * (1.0 - x[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= wi * x[lo2 - shift:hi2 - shift]
        return obj, grad

    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        h_out = project_to_sum(res.x, target_sum)
    except Exception:
        h_out = h
    return h_out

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    ts = best_n / 2.0

    # Phase 1: Multi-lag subgradient with momentum (main workhorse)
    if remaining() > 120:
        h1, c5_1 = multi_lag_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 10, max_iters=8000, top_T=15)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Smooth max optimization at various alphas
    if remaining() > 80:
        for alpha in [500, 2000, 5000, 10000]:
            if remaining() < 30:
                break
            h2 = smooth_max_optimize(best_h, best_n, ts, alpha, maxiter=400)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2

    # Phase 3: More subgradient refinement
    if remaining() > 100:
        h3, c5_3 = multi_lag_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 10, max_iters=6000, top_T=20)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Annealed subgradient for exploration
    if remaining() > 80:
        h4, c5_4 = annealed_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 5, max_iters=3000, seed=seed+1)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Fourier perturbation search
    if remaining() > 60:
        h5, c5_5 = fourier_perturb_search(best_h, best_n, ts, t0, t0 + remaining() - 5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Phase 6: Block perturbation search
    if remaining() > 60:
        h6, c5_6 = block_perturb_search(best_h, best_n, ts, t0, t0 + remaining() - 5, seed=seed+2)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: Upsample to 2x resolution and refine
    if remaining() > 200:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        h_up = upsample(best_h, best_n, n2)
        h_up = project_to_sum(h_up, ts2)
        c5_up = compute_c5(h_up)

        h7, c5_7 = multi_lag_subgrad(h_up, n2, ts2, t0, t0 + remaining() - 30, max_iters=8000, top_T=15)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7
            best_n = n2
            ts = ts2

        if remaining() > 60:
            h7b = smooth_max_optimize(best_h, best_n, ts, 5000, maxiter=300)
            c5_7b = compute_c5(h7b)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b

        if remaining() > 40:
            h7c, c5_7c = multi_lag_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 5, max_iters=5000, top_T=20)
            if c5_7c < best_c5:
                best_c5 = c5_7c
                best_h = h7c

    # Phase 8: Try downsampled alternatives (different topology)
    if remaining() > 150:
        for n_try in [120, 160, 180]:
            if remaining() < 50:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            h_down = upsample(best_h, best_n, n_try)
            h_down = project_to_sum(h_down, ts_try)
            c5_down = compute_c5(h_down)
            if c5_down < best_c5:
                best_c5 = c5_down
                best_h = h_down
                best_n = n_try
                ts = ts_try
            h_d, c5_d = multi_lag_subgrad(h_down, n_try, ts_try, t0, t0 + remaining() - 10, max_iters=4000, top_T=12)
            if c5_d < best_c5:
                best_c5 = c5_d
                best_h = h_d
                best_n = n_try
                ts = ts_try

    # Phase 9: Final intensive polishing
    if remaining() > 50:
        h9, c5_9 = multi_lag_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 5, max_iters=10000, top_T=25)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9

    if remaining() > 30:
        for alpha in [2000, 5000, 10000, 20000]:
            if remaining() < 15:
                break
            h_f = smooth_max_optimize(best_h, best_n, ts, alpha, maxiter=300)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    if remaining() > 20:
        h9b, c5_9b = multi_lag_subgrad(best_h, best_n, ts, t0, t0 + remaining() - 2, max_iters=5000, top_T=20)
        if c5_9b < best_c5:
            best_c5 = c5_9b
            best_h = h9b

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
