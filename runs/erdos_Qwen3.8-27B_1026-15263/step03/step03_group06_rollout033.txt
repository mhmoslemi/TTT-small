            g[s:] -= wk * h[:n-s]
        else:
            d = -s
            g[d:] += wk * (1.0 - h[:n-d])
            g[:n-d] -= wk * h[d:]
    return g, m  # return gradient and max value

def projected_gradient_descent(h0, n, target_sum, n_iters, alpha, top_k, lr_init, momentum):

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    gamma = (lo + hi) / 2.0
    return np.clip(h - gamma, 0.0, 1.0)

def active_gradient(h, n, top_k, alpha):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    shifted = alpha * (top_vals - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w_top = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    for i in range(len(top_idx)):
        wk = w_top[i]
        if wk < 1e-10:
            continue
        k = int(top_idx[i])
        s = k - (n - 1)
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            d = -s
            g[d:] += wk * (1.0 - h[:n-d])
            g[:n-d] -= wk * h[d:]
    return g, m

def projected_gradient(h0, n, target_sum, n_iters, alpha, top_k, lr_init, momentum):
    h = h0.copy()
    v = np.zeros(n)
    h_best = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best
    lr = lr_init
    for it in range(n_iters):
        g, m_val = active_gradient(h, n, top_k, alpha)
        v = momentum * v - lr * g
        h_new = project_to_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_cur + 1e-5:
            lr *= 0.5
            v = momentum * v * 0.5
            h_new = project_to_sum(h + v, target_sum)
            c5_new = compute_c5(h_new)
        h = h_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 1.0 / (n * 0.5))
        else:
            lr *= 0.97
    return h_best, c5_best

def simulated_annealing(h0, n, target_sum, n_steps, t_init, t_final, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    log_t_init = np.log(t_init)
    log_t_final = np.log(t_final)
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        t_val = np.exp(log_t_init + frac * (log_t_final - log_t_init))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        corr_cur = compute_corr(h)
        h[i] = hi_new
        h[j] = hj_new
        corr_new = compute_corr(h)
        c5_new = float(np.max(corr_new) * 2.0 / n)
        c5_diff = c5_new - c5_cur
        accept = False
        if c5_diff < 0:
            accept = True
        else:
            if t_val > 1e-15:
                if rng.uniform() < np.exp(-c5_diff / t_val):
                    accept = True

            c5_cur = c5_new

                c5_best = c5_new
                h_best = h.copy()

    return h_best, c5_best

def fourier_init(n, n_modes, rng):
    h = np.full(n, 0.5)
    j = np.arange(n)
    for k in range(1, n_modes + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(2.0 * np.pi * k * j / n) + b * np.sin(2.0 * np.pi * k * j / n)
    return project_to_sum(h, n / 2.0)

def fourier_optimize(n, target_sum, n_modes, max_iter, rng):
    j = np.arange(n)
    basis = [np.ones(n) / np.sqrt(n)]
    for k in range(1, n_modes + 1):
        basis.append(np.cos(2.0 * np.pi * k * j / n) / np.sqrt(n))
        basis.append(np.sin(2.0 * np.pi * k * j / n) / sqrt(n + 1) if False else np.sin(2.0 * np.pi * k * j / n) / np.sqrt(n))
    n_basis = len(basis)
    B = np.column_stack(basis)
    
    def fourier_obj(coeff):
        h_raw = B @ coeff
        h_p = project_to_sum(h_raw, target_sum)
        return compute_c5(h_p)
    
    best_coeff = np.zeros(n_basis)
    best_coeff[0] = np.sqrt(n) * 0.5
    best_c5 = fourier_obj(best_coeff)

        coeff0 = best_coeff.copy()
        for k in range(1, n_basis):

                          options={'maxiter': max_iter, 'xatol': 1e-4, 'fatol': 1e-6})
            if res.fun < best_c5:
                best_c5 = res.fun
                best_coeff = res.x.copy()
        except Exception:
            pass

    h = h0.copy()

    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + d, 0.0, 1.0)

                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True

            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 9: Final gradient polish
    if time_left() > 20:
        n_f2 = len(best_h)
        ts_f2 = n_f2 / 2.0
        for alpha_val in [2000.0, 5000.0]:
            if time_left() < 15:
                break
            h_fin, c5_fin = projected_gradient(best_h, n_f2, ts_f2, 1000, alpha_val, 20, 0.1, 0.8)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
    
    # Final validation
    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
