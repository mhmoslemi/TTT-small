def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

                                    t0=0.008, t_min=0.001, cool=0.999997)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        h_work = h_sa2.copy()

    if remaining() > 80:
        h_gls2, c5_gls2 = greedy_local_search(h_work, n_main, n_trials=10000, rng=rng)
        if c5_gls2 < best_c5:
            best_c5 = c5_gls2
            best_h = h_gls2.copy()
            h_work = h_gls2.copy()

    if remaining() > 120:
        for alpha in [8000, 40000, 200000]:
            if remaining() < 50:
                break
            h_fin = slsqp_optimize(h_work, n_main, ts_main, alpha, maxiter=2500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                h_work = h_fin.copy()

    for n_try in [280, 320, 360]:
        if remaining() < 120:
            break
        ts_try = n_try // 2
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = make_binary(h_interp, n_try, ts_try, rng)
        for alpha in [30, 200, 1500]:
            if remaining() < 40:
                break
            h_try = slsqp_optimize(h_try, n_try, ts_try, alpha, maxiter=800)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    for trial in range(4):
        if elapsed() > budget_s - 80:
            break
        period = 6 + trial * 3
        h_periodic = make_periodic_start(n_main, period, rng)
        c5_p = compute_c5(h_periodic)
        if c5_p < best_c5 + 0.05:
            h_p_sa, c5_p_sa = sa_enhanced(h_periodic, n_main, ts_main,
                                          min(t_start + elapsed() + 60, t_start + budget_s - 30),
                                          rng, t0=0.012, t_min=0.002, cool=0.99999)
            if c5_p_sa < best_c5:
                best_c5 = c5_p_sa
                best_h = h_p_sa.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
