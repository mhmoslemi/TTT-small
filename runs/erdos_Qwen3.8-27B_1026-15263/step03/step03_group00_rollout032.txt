def corr_gradient_active(h, lag_idx):
    """Gradient of corr[lag_idx] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    lag = lag_idx - (n - 1)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:

        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]

        if abs(diff) < 1e-14:

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    v = np.zeros(n)  # momentum

    best_h = h.copy()

        # Subgradient step with momentum

            step = min(step * 1.1, 0.1)

And our quadratic term is -∑_{i<j} 2 * λ_{j-i} * h[i]h[j] (since for each pair (i,j) with i<j, the term -h[i]h[j] appears with coefficient -λ_{j-i} in both the i-sum and j-sum... actually no.

Let me be more careful. ∑_{i=0}^{n-1-k} h[i+k]h[i] counts each pair (i, i+k) once. So the total quadratic term is:
-∑_{k=1}^{n-1} λ_k ∑_{i=0}^{n-1-k} h[i+k]h[i]

This equals -∑_{1≤k≤n-1} λ_k ∑_{i=0}^{n-1-k} h[i+k]h[i] = -∑_{0≤i<j≤n-1} λ_{j-i} h[j]h[i]

In matrix form, if I define B[j,i] = λ_{j-i} for j > i (and 0 otherwise), then the quadratic term is -h^T B h = -∑_i ∑_j B[j,i] h[j]h[i] = -∑_{i<j} B[j,i] h[j]h[i] (since B[j,i] = 0 for j≤i).

For the QP, we want to write it as (1/2) h^T Q h + c^T h. So:
(1/2) h^T Q h = -∑_{i<j} λ_{j-i} h[j]h[i]
Q[j,i] + Q[i,j] = -2λ_{|i-j|} for i≠j
Q[i,i] = 0

Since Q is symmetric: Q[i,j] = -λ_{|i-j|} for i≠j, Q[i,i] = 0.

So the QP is:
minimize (1/2) h^T Q h + c^T h
subject to 0 ≤ h ≤ 1, sum(h) = n/2

where Q[i,j] = -λ_{|i-j|} (i≠j), Q[i,i] = 0, and c[i] = ∑_{k=0}^{i} λ_k (for i ≤ n-1).

Now, is Q negative definite? Q = -A where A[i,j] = λ_{|i-j|} for i≠j and A[i,i] = 0.
A is a symmetric matrix with zero diagonal and positive off-diagonal entries (assuming λ_k > 0). By Gershgorin, the eigenvalues of A are in the range [-ρ(A), ρ(A)] where ρ(A) is the spectral radius. Since A has positive off-diagonals, by Perron-Frobenius, the largest eigenvalue is positive. But the smallest eigenvalue could be negative or positive.

Actually, A is a "conference-type" matrix (symmetric, zero diagonal, constant off-diagonal by distance). For a circulant version, the eigenvalues would be ∑_{k≠0} λ_k ω^{k} for the k-th Fourier mode. For a Toeplitz (non-circulant) version, it's similar but not exactly the same.

In any case, Q might be indefinite, making the QP non-convex. But we can still solve it with a general-purpose QP solver or with projected gradient.

Actually, for the purpose of the alternating method, I don't need the QP to be convex. I just need to find a good approximate solution. I can use:

Let me use scipy.optimize.minimize with the 'L-BFGS-B' method, which handles bounds and is efficient for large problems.

Here's the refined algorithm:

1. Initialize λ uniformly over all lags: λ_k = 1/(n-1)
2. Iterate:
   a. Solve the QP: minimize (1/2)h^T Q h + c^T h s.t. 0≤h≤1, sum(h)=n/2
   b. Compute c_k for all k
   c. Update λ: put more weight on lags that are "tight" (close to max)
      λ_k ∝ exp(c_k / τ) for some temperature τ
   d. Decrease τ over time
3. Return the best h found

This is essentially a **cutting-plane / bundle method** for the min-max problem!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(h, lag_idx):
    n = len(h)
    g = np.zeros(n)
    lag = lag_idx - (n - 1)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def corr_gradient_multi(h, lag_indices, weights):
    n = len(h)
    g = np.zeros(n)
    for idx, w in zip(lag_indices, weights):
        g += w * corr_gradient(h, idx)
    return g

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_objective_grad(h, alpha, smooth_lags=0):
    n = len(h)
    corr = compute_corr(h)
    if smooth_lags > 0:
        top_k = np.argsort(corr)[-smooth_lags:]
        vals = corr[top_k]
        shifted = alpha * (vals - np.max(vals))
        exps = np.exp(shifted)
        w = exps / np.sum(exps)
        obj = np.max(corr) + np.log(np.sum(exps)) / alpha
        grad = np.zeros(n)
        for idx, wi in zip(top_k, w):
            grad += wi * corr_gradient(h, idx)
    else:
        active = int(np.argmax(corr))
        obj = corr[active]
        grad = corr_gradient(h, active)
    return obj, grad, corr

def lbfgs_multilag(h0, n, target_sum, n_lags, maxiter=500):
    def f_and_g(x):
        corr = compute_corr(x)
        top_k = np.argsort(corr)[-n_lags:]
        shift = 0.1 * (corr[top_k] - np.max(corr))
        exps = np.exp(shift)
        w = exps / np.sum(exps)
        obj = np.sum(w * corr[top_k])
        grad = np.zeros(n)
        for idx, wi in zip(top_k, w):
            grad += wi * corr_gradient(x, idx)
        return obj, grad

    def f(x):
        return f_and_g(x)[0]

    def g(x):
        return f_and_g(x)[1]

    try:
        res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def fine_coord_descent(h0, target_sum, max_iters=20000, top_lags=2):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    stale = 0
    step_base = 0.1
    for it in range(max_iters):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_lags:]
        vals = corr[top_idx]
        shift = 50.0 * (vals - np.max(vals))
        exps = np.exp(shift)
        w = exps / np.sum(exps)
        g = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            g += wi * corr_gradient(h, idx)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_base * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
            if it % 1000 == 0 and it > 0:
                step_base = min(step_base * 0.95, 0.01)
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            step_base *= 0.7
            if step_base < 1e-6:
                step_base = 0.05
            if stale > 200:
                break
    return h, c5

def fourier_perturb(h, rng, amplitude=0.02, n_modes=3):
    n = len(h)
    x = np.arange(n) / float(n)
    pert = np.zeros(n)
    for _ in range(n_modes):
        mode = rng.randint(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        coeff = rng.uniform(-1, 1) * amplitude
        pert += coeff * np.cos(2 * np.pi * mode * x + phase)
    h_new = h + pert
    return project_to_sum(h_new, n / 2.0)

def local_smooth_perturb(h, rng, amplitude=0.03, width=5):
    n = len(h)
    center = rng.randint(0, n)
    h_new = h.copy()
    for d in range(-width, width + 1):
        idx = (center + d) % n
        weight = np.exp(-0.5 * (d / width) ** 2)
        h_new[idx] += rng.normal(0, amplitude * weight)
    return project_to_sum(h_new, n / 2.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_periodic = np.concatenate([h, h])
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    x_new_ext = x_new
    result = np.interp(x_new_ext, x_ext, h_ext)
    return result

def mirror_perturb(h, rng, amplitude=0.01):
    n = len(h)
    h_new = h.copy()
    h_flipped = h[::-1].copy()
    for i in range(n):
        h_new[i] += amplitude * rng.normal(0, 1) * (h_flipped[i] - h[i])
    return project_to_sum(h_new, n / 2.0)
