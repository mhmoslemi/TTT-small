```python
    hi = 1.0 - np.min(y) + 1.0
```

```python
def smooth_max_and_grad_correct(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)  # actual lag
        for m in range(n):
            # (1-h[m-k]) if 0<=m-k<n
            idx1 = m - k
            if 0 <= idx1 < n:
                grad[m] += w[j] * (1.0 - h[idx1])
            # -h[m+k] if 0<=m+k<n
            idx2 = m + k
            if 0 <= idx2 < n:
                grad[m] -= w[j] * h[idx2]
    grad *= dx
    return obj, grad
```

This is O(n * 2n) = O(n²) per gradient evaluation. For n=240, that's about 115K operations, very fast. For n=720, about 1M operations, still fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        # First term: w[j] * (1-h[m-k]) for valid m
        m1_s = max(0, k)
        m1_e = min(n, k + n)
        if m1_s < m1_e:
            idx = np.arange(m1_s, m1_e) - k
            grad[m1_s:m1_e] += w[j] * (1.0 - h[idx])
        # Second term: -w[j] * h[m+k] for valid m
        m2_s = max(0, -k)
        m2_e = min(n, n - k)
        if m2_s < m2_e:
            idx = np.arange(m2_s, m2_e) + k
            grad[m2_s:m2_e] -= w[j] * h[idx]
    grad *= dx
    return obj, grad

def projected_grad_descent(h0, n, alpha, max_iter, lr_init, time_fn):
    h = h0.copy()
    ts = n / 2.0
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    total_w = np.sum(np.ones(2*n-1))  # just for reference
    
    for it in range(max_iter):
        if time_fn() < 5:
            break
        obj, grad = smooth_max_and_grad(h, alpha)
        h_new = project_box_sum(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.1, 0.5 / np.max(np.abs(grad)) * 10 if np.max(np.abs(grad)) > 0 else lr)
        else:
            lr *= 0.5
            if lr < 1e-10:
                lr = 1e-8
        h = h_new
    return best_h, best_c5

def block_swap_search(h, n, rng, time_fn, max_rounds=50):
    ts = n / 2.0
    h = h.copy()
    c5_cur = compute_c5(h)
    
    for round in range(max_rounds):
        if time_fn() < 10:
            break
        best_c5 = c5_cur
        best_move = None
        
        # Try block swaps: flip a block of length L
        for L in [2, 3, 4, 5, 8]:
            n_start = n - L + 1
            n_trials = min(200, n_start)
            for _ in range(n_trials):
                s = rng.randint(0, n_start)
                block = np.arange(s, s + L)
                h[block] = 1.0 - h[block]
                if abs(h.sum() - ts) > 1e-10:
                    # Rejection: sum changed
                    # For binary, flipping a block of L ones to zeros (or vice versa)
                    # changes sum by L (or -L). We need to also flip L points elsewhere.
                    # For non-binary, the sum change is 2*(sum of block) - L... 
                    # Actually for the block swap to preserve sum, we need a paired swap.
                    h[block] = 1.0 - h[block]  # undo
                    continue
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_move = (block, h.copy())
                h[block] = 1.0 - h[block]  # undo
        
        if best_move is None:
            # Try 2-point swaps
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            n_trials = min(300, len(ones) * len(zeros))
            for _ in range(n_trials):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0.0, 1.0
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_move = ((i, j), h.copy())
                h[i], h[j] = 1.0, 0.0
        
        if best_move is None:
