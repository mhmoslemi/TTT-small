        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    
    nm1 = n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    
    i2 = j_idx + k_idx - nm1
    i4 = j_idx - k_idx + nm1
    
    valid1 = (i2 >= 0) & (i2 < n)
    valid2 = (i4 >= 0) & (i4 < n)
    
    c2 = np.clip(i2, 0, n - 1)
    c4 = np.clip(i4, 0, n - 1)
    
    term1 = valid1 * (1.0 - h[c2]) * w[None, :]
    term2 = valid2 * (-h[c4]) * w[None, :]
    
    grad = np.sum(term1 + term2, axis=1)
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_max_obj(x, alpha)
    
    def g(x):
        return smooth_max_grad(x, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Generate candidate k values
        k_set = []
        for offset in [1, -1, n//2, n//3, 2*n//3, n//4, 3*n//4, n//5, 2*n//5]:
            k_set.append((j + offset) % n)
        while len(k_set) < n_cand:
            k_set.append(rng.randint(n))
        
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def subgradient_step(h, corr, lr):
    """Take a projected subgradient step on the max of corr."""
    n = len(h)
    nm1 = n - 1
    # Find the argmax of corr
    k_max = np.argmax(corr)
    
    # Compute subgradient for that specific k
    # corr[k] = sum_i h[i] * (1-h[i+k-nm1]) for valid i
    # d(corr[k])/d(h[m]) = (1-h[m+k-nm1]) if m+k-nm1 valid - h[m-k+nm1] if m-k+nm1 valid
    
    m = np.arange(n)
    i1 = m + k_max - nm1
    i2 = m - k_max + nm1
    
    valid1 = (i1 >= 0) & (i1 < n)
    valid2 = (i2 >= 0) & (i2 < n)
    
    subgrad = np.zeros(n)
    subgrad[valid1] += (1.0 - h[i1[valid1]])
    subgrad[valid2] += (-h[i2[valid2]])
    
    # Projected subgradient step
    h_new = h - lr * subgrad
    h_new = project_to_sum(h_new, n / 2.0)
    return h_new

            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved
