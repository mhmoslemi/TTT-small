def smooth_max_grad_efficient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        k = n - 1 + m
        wk = w[k]
        if m == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif m > 0:
            # corr[k] = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
            # d/dh[j] for j in [0, n-1-m]: (1-h[j+m])
            # d/dh[j] for j in [m, n-1]: -h[j-m]
            valid_j_first = slice(0, n-m)
            valid_j_second = slice(m, n)
            grad[valid_j_first] += wk * (1.0 - h[m:n])
            grad[valid_j_second] += wk * (-h[0:n-m])
        else:
            q = -m
            # corr[k] = sum_{j=q}^{n-1} h[j]*(1-h[j-q])
            # d/dh[j] for j in [q, n-1]: (1-h[j-q])
            # d/dh[j] for j in [0, n-1-q]: -h[j+q]
            grad[q:n] += wk * (1.0 - h[0:n-q])
            grad[0:n-q] += wk * (-h[q:n])
    
    return obj, grad

        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_f
        for alpha in [200, 1000, 5000]:
            if time_left() < 30:
                break
            h_opt, _ = adam_run(h_f, ts_f, n_f, alpha, max_iter=1500, lr=0.03)
            h_opt = project_to_sum(h_opt, ts_f)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_f
            h_f = h_opt

    # Phase 7: Diverse restarts
    n_best = best_n
    ts_best = n_best / 2.0
    n_trials = 25

    for trial in range(n_trials):
        if time_left() < 25:
            break
        if trial % 5 == 0:
            noise_lvl = 0.1 * (1.0 - trial / float(n_trials)) + 0.02
            h_shake = best_h + rng.uniform(-noise_lvl, noise_lvl, n_best)
        elif trial % 5 == 1:
            h_shake = 1.0 - best_h + rng.uniform(-0.08, 0.08, n_best)
        elif trial % 5 == 2:
            freq = rng.uniform(0.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.15)
            h_shake = best_h + amp * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
        elif trial % 5 == 3:
            n_blocks = rng.randint(6, 30)
            blk = n_best // n_blocks
            h_shake = np.zeros(n_best)
            for i in range(n_blocks):
                h_shake[i * blk:(i + 1) * blk] = rng.uniform(0.1, 0.9)
            h_shake = project_to_sum(h_shake, ts_best)
            h_shake = 0.7 * best_h + 0.3 * h_shake
        else:
            # Random smooth function
            n_modes = rng.randint(2, 8)
            h_shake = np.zeros(n_best)
            x = np.arange(n_best) / n_best
            for _ in range(n_modes):
                f = rng.uniform(1, 10)
                ph = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.02, 0.1)
                h_shake += amp * np.sin(2 * np.pi * f * x + ph)
            h_shake = 0.5 + h_shake

        h_shake = project_to_sum(h_shake, ts_best)
        for alpha in [300, 2000]:
            if time_left() < 15:
                break
            h_opt, _ = adam_run(h_shake, ts_best, n_best, alpha, max_iter=1200, lr=0.04)
            h_opt = project_to_sum(h_opt, ts_best)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 8: SLSQP final polish
    for alpha in [5000, 20000]:
        if time_left() < 25:
            break
        h_pol = slsqp_refine(best_h, ts_best, n_best, alpha, maxiter=300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 9: Subgradient polish (works on exact max, not smooth approx)
    if time_left() > 30:
        h_sg, c5_sg = subgrad_polish(best_h, ts_best, n_best, max_iter=3000, lr_init=0.003)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 10: Very fine Adam with high alpha
    if time_left() > 20:
        h_fin, _ = adam_run(best_h, ts_best, n_best, 50000, max_iter=3000, lr=0.003)
        h_fin = project_to_sum(h_fin, ts_best)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 11: Final tiny steps
    if time_left() > 10:
        h_last = best_h.copy()
        for it in range(2000):
            if time_left() < 3:
                break
            obj, grad = smooth_max_grad(h_last, 100000)
            step = 0.002 / (1.0 + it * 0.005)
            h_new = h_last - step * grad
            h_new = project_to_sum(h_new, ts_best)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_last) - 1e-15:
                h_last = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_last.copy()

    best_n = len(best_h)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
