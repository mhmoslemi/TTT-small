import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_bounded_sum(v, target_sum):
    n = len(v)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        projected = np.clip(v - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]

        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):

    rng.shuffle(order)

        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)

            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = max(0.0, min(1.0, h[j]))
            h[best_k] = max(0.0, min(1.0, h[best_k]))
            improved = True
    return h, corr, improved

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_sh = np.exp(shifted - max_sh)
    w = exp_sh / np.sum(exp_sh)
    log_sum = max_sh + np.log(np.sum(exp_sh))
    obj = m_val + log_sum / alpha
    j_idx = np.arange(n)[:, None]
    p_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - p_idx + n - 1
    second_idx = j_idx + p_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]).astype(float) * w[None, :]
    cs = valid_s * h[sc].astype(float) * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def projected_gradient_opt(h0, target_sum, alpha, max_iter, time_fn):
    n = len(h0)
    h = h0.copy()
    obj, grad = smooth_obj_grad(h, alpha)
    best_h = h.copy()
    best_obj = obj
    step = 1.0 / (alpha * n * 2.0)
    for t in range(max_iter):
        if time_fn() < 10:
            break
        h_new = project_bounded_sum(h - step * grad, target_sum)
        obj_new, grad_new = smooth_obj_grad(h_new, alpha)
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            grad = grad_new
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
            step = min(step * 1.05, 1.0 / (alpha * n * 0.5))
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h

def subgradient_opt(h0, target_sum, max_iter, time_fn, eps=0.01):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h) / (2.0 / n)
    step = 0.5 / n
    for t in range(max_iter):
        if time_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        if m_val < best_val:
            best_val = m_val
            best_h = h.copy()
        active = (corr >= m_val - eps)
        n_active = np.sum(active)
        if n_active == 0:
            active = (corr == m_val)
            n_active = np.sum(active)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        j_idx = np.arange(n)[:, None]
        p_idx = np.arange(2 * n - 1)[None, :]
        first_idx = j_idx - p_idx + n - 1
        second_idx = j_idx + p_idx - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        cf = valid_f * (1.0 - h[fc]).astype(float) * w[None, :]
        cs = valid_s * h[sc].astype(float) * w[None, :]
        grad = np.sum(cf - cs, axis=1)
        h_new = project_bounded_sum(h - step * grad, target_sum)
        h = h_new
        step *= 0.999
    return best_h

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def f(x):
        return smooth_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_obj_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - target_sum
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_bounded_sum(res.x, target_sum)
    except Exception:
        return project_bounded_sum(h0, target_sum)

def interp_h(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_src, h_src, period=2.0)
    return project_bounded_sum(h_new, n_new / 2.0)

def optimize_at_resolution(h_in, n_tr, rng, t_start, budget_s, frac_budget):
    t_end = t_start + frac_budget * budget_s
    target_sum = n_tr / 2.0
    h = h_in.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_local = compute_c5(h)
    best_h_local = h.copy()

    def elapsed_left():
        return budget_s * frac_budget - (time.time() - t_start)

    for _ in range(15):
        if time.time() >= t_end:
            break
        h, corr, _ = pair_descent_sweep(h, corr, rng, n_cand=12)
        c5 = compute_c5(h)
        if c5 < best_local:
            best_local = c5
            best_h_local = h.copy()

    alphas = [10, 30, 80, 200, 500]
    for alpha in alphas:
        if time.time() >= t_end:
            break
        h_pg = projected_gradient_opt(best_h_local, target_sum, alpha, max_iter=300, time_fn=lambda: t_end - time.time())
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_local:
            best_local = c5_pg
            best_h_local = h_pg.copy()
            corr = np.correlate(best_h_local, 1.0 - best_h_local, mode="full")
            for _ in range(5):
                if time.time() >= t_end:
                    break
                best_h_local, corr, _ = pair_descent_sweep(best_h_local, corr, rng, n_cand=10)
                c5_check = compute_c5(best_h_local)
                if c5_check < best_local:
                    best_local = c5_check
                    best_h_local = best_h_local.copy()

    if time.time() < t_end - 30:
        h_sg = subgradient_opt(best_h_local, target_sum, max_iter=500,
                               time_fn=lambda: t_end - time.time(), eps=0.005)
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_local:
            best_local = c5_sg
            best_h_local = h_sg.copy()

    corr = np.correlate(best_h_local, 1.0 - best_h_local, mode="full")
    for _ in range(10):
        if time.time() >= t_end:
            break
        best_h_local, corr, _ = pair_descent_sweep(best_h_local, corr, rng, n_cand=15)
        c5 = compute_c5(best_h_local)
        if c5 < best_local:
            best_local = c5
            best_h_local = best_h_local.copy()

    return best_h_local, best_local

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_bounded_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    t_end_p1 = t_start + 0.12 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    while time.time() < t_end_p1:
        h_work, corr_w, _ = pair_descent_sweep(h_work, corr_w, rng, n_cand=12)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w

    t_phase2 = 0.55
    t_end_p2 = t_start + (0.12 + t_phase2) * budget_s
    resolutions = [best_n]
    for offset in [80, 160, 240, 360]:
        res = best_n + offset
        if res <= 800 and res % 2 == 0:
            resolutions.append(res)
    resolutions = sorted(set(resolutions))

    time_per_res = t_phase2 * budget_s / len(resolutions) * 0.9

    for n_tr in resolutions:
        if time.time() >= t_end_p2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        h_start = interp_h(best_h, n_tr)
        t_res_start = time.time()
        h_opt, c5_opt = optimize_at_resolution(h_start, n_tr, rng, t_start, budget_s, t_phase2)
        frac_used = (time.time() - t_res_start) / (t_end_p2 - t_start + 1e-9)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_tr

    t_end_p3 = t_start + 0.92 * budget_s
    n_f = best_n
    for alpha in [30, 100, 300]:
        if time.time() >= t_end_p3:
            break
        h_pg = projected_gradient_opt(best_h, n_f / 2.0, alpha, max_iter=400,
                                       time_fn=lambda: t_end_p3 - time.time())
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if time.time() < t_end_p3 - 20:
        try:
            h_slq = slsqp_optimize(best_h, n_f / 2.0, 100, maxiter=800)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
        except Exception:
            pass

    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=15)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_bounded_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
