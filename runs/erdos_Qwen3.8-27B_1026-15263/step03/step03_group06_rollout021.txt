from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree > 0 and abs(diff) <= nfree * 0.5:
            h[free] += diff / nfree
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    sm_val = m + np.log(total) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-16:
            continue
        k = i - (n - 1)
        # Part 1: wi * (1-h[m-k]) for valid m
        m_start = max(0, k)
        m_end = min(n, k + n)
        if m_start < m_end:
            j_start = m_start - k
            j_end = j_start + (m_end - m_start)
            grad[m_start:m_end] += wi * one_minus_h[j_start:j_end]
        # Part 2: -wi * h[m+k] for valid m
        m_start2 = max(0, -k)
        m_end2 = min(n, n - k)
        if m_start2 < m_end2:
            j_start2 = m_start2 + k
            j_end2 = j_start2 + (m_end2 - m_start2)
            grad[m_start2:m_end2] -= wi * h[j_start2:j_end2]
    
    return sm_val, grad

def lbfgsb_optimize(h0, target_sum, alpha, maxiter, penalty_coeff):
    n = len(h0)
    def obj(x):
        v, _ = smooth_max_value_and_grad(x, alpha)
        sum_pen = penalty_coeff * (np.sum(x) - target_sum) ** 2
        return v + sum_pen
    def grad(x):
        _, g = smooth_max_value_and_grad(x, alpha)
        g = g + 2.0 * penalty_coeff * (np.sum(x) - target_sum) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_sum(h_opt, target_sum)
    return h_opt

def coord_descent_fine(h0, target_sum, n_rounds, rng, step_sizes):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            for ds in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + ds, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-14:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def coord_descent_ls(h0, target_sum, n_rounds, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = np.clip(v, 0.0, 1.0)
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-6})
                if res.fun < c5 - 1e-13:
                    h[i] = np.clip(res.x, 0.0, 1.0)
                    h = project_to_sum(h, target_sum)
                    c5 = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5

def pairwise_update(h0, target_sum, n_rounds, rng, n_pairs_per_round):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_pairs_per_round):
            i, j = rng.choice(n, size=2, replace=False)
            # Try to adjust h[i] and h[j] together
            # Keep sum the same: h[i] + h[j] = const
            s = h[i] + h[j]
            # h[i] = s/2 + d, h[j] = s/2 - d, d in [-(s/2), min(s/2, 1-s/2)]
            d_max = min(s/2, 1.0 - s/2)
            if d_max < 1e-6:

            lo = max(-s/2, -(1.0 - s/2))
            hi = min(s/2, 1.0 - s/2)
            if hi - lo < 1e-6:

            def f1d(d):
                h_tmp = h.copy()
                h_tmp[i] = s/2 + d
                h_tmp[j] = s/2 - d
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-6})
                if res.fun < c5 - 1e-13:
                    h[i] = s/2 + res.x
                    h[j] = s/2 - res.x
                    c5 = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5

**Idea: Use the structure of the problem.**

The correlation at lag k is:
c[k] = ∑_i h[i] * (1-h[i+k])

This can be rewritten as:
c[k] = ∑_i h[i] - ∑_i h[i]*h[i+k]
     = n/2 - ∑_i h[i]*h[i+k]  (using the sum constraint)

So minimizing max_k c[k] is equivalent to minimizing:
max_k (n/2 - autocorr_h[k])

where autocorr_h[k] = ∑_i h[i]*h[i+k]

Or equivalently, maximizing min_k autocorr_h[k].

This is a nice reformulation! We want to maximize the minimum autocorrelation (at all lags).

The autocorrelation of h is related to the power spectrum of h. By the Wiener-Khinchin theorem:
autocorr_h[k] = (1/n) ∑_f |H[f]|² * e^{2πifk/n}

where H is the DFT of h. So we want the power spectrum |H[f]|² to be "flat" (uniform across frequencies) to make the autocorrelation as flat as possible.

This suggests an approach based on **spectral optimization**:
1. Compute the DFT of h
2. Modify the power spectrum to be more flat
3. Inverse DFT to get new h
4. Project to [0,1] and sum constraint

This is like an **iterative magnitude replacement** (IMR) algorithm similar to those used in phase retrieval!

Let me think about this more carefully. If I could make the autocorrelation perfectly flat, then:
autocorr_h[k] = (n/2)²/n = n/4 for all k (by Parseval's theorem, the total power is ∑|H|² = n*∑h², and the DC component is (∑h)² = (n/2)²)

Wait, that's not quite right. Let me be more careful.

∑_k autocorr_h[k] = ∑_k ∑_i h[i]*h[i+k] = ∑_i h[i] * ∑_k h[i+k] = ∑_i h[i] * n/2 = n/2 * n/2 = n²/4

So the average autocorrelation is (n²/4)/(2n-1) ≈ n/4 for large n.

If the autocorrelation is perfectly flat, each value would be n²/4/(2n-1) ≈ n/4.

Then c[k] = n/2 - n/4 = n/4 for all k.
C₅ = (n/4) * 2/n = 1/2.

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):

        if abs(diff) < 1e-14:

    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    n_minus_1 = n - 1
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        k = i - n_minus_1
        m_start = max(0, k)
        m_end = min(n, k + n)
        if m_start < m_end:
            j_start = m_start - k
            j_end = j_start + (m_end - m_start)
            grad[m_start:m_end] += wi * one_minus_h[j_start:j_end]
        m_start2 = max(0, -k)
        m_end2 = min(n, n - k)
        if m_start2 < m_end2:
            j_start2 = m_start2 + k
            j_end2 = j_start2 + (m_end2 - m_start2)
            grad[m_start2:m_end2] -= wi * h[j_start2:j_end2]
    return sm_val, grad

def lbfgsb_with_lagrangian(h0, target_sum, alpha, n_outer, maxiter_inner, eta):
    n = len(h0)
    lam = 0.0
    h_cur = h0.copy()
    bounds = [(0.0, 1.0)] * n
    for it in range(n_outer):
        def obj(x):
            v, _ = smooth_max_value_and_grad(x, alpha)
            return v + lam * (np.sum(x) - target_sum)
        def grad(x):
            _, g = smooth_max_value_and_grad(x, alpha)
            return g + lam * np.ones(n)
        try:
            res = minimize(obj, h_cur.copy(), jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max(10, maxiter_inner), 'gtol': 1e-9})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            h_new = h_cur.copy()
        violation = np.sum(h_new) - target_sum
        lam += eta * violation
        lam = np.clip(lam, -50.0, 50.0)
        h_cur = h_new
    h_cur = project_to_sum(h_cur, target_sum)
    return h_cur

def lbfgsb_penalty(h0, target_sum, alpha, maxiter, pen_coeff):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smooth_max_value_and_grad(x, alpha)
        return v + pen_coeff * (np.sum(x) - target_sum) ** 2
    def grad(x):
        _, g = smooth_max_value_and_grad(x, alpha)
        return g + 2.0 * pen_coeff * (np.sum(x) - target_sum) * np.ones(n)
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-16})
        h_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_opt = h0.copy()
    h_opt = project_to_sum(h_opt, target_sum)
    return h_opt

def coord_descent_ls(h0, target_sum, n_rounds, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = np.clip(v, 0.0, 1.0)
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            lo = max(0.0, h[i] - 0.25)
            hi = min(1.0, h[i] + 0.25)
            if hi - lo < 1e-6:
                continue
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-7})
                if res.fun < c5 - 1e-13:
                    h[i] = np.clip(res.x, 0.0, 1.0)
                    h = project_to_sum(h, target_sum)
                    c5 = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5

def coord_descent_fixed(h0, target_sum, n_rounds, rng, deltas):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5 = best_val
                improved = True
        if not improved:
            break
    return h, c5

def pairwise_update(h0, target_sum, n_rounds, rng, n_pairs):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_pairs):
            i, j = rng.choice(n, size=2, replace=False)
            s = h[i] + h[j]
            d_max = min(s / 2.0, 1.0 - s / 2.0)
            if d_max < 1e-6:
                continue
            lo = -d_max
            hi = d_max
            def f1d(d):
                h_tmp = h.copy()
                h_tmp[i] = s / 2.0 + d
                h_tmp[j] = s / 2.0 - d
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-7})
                if res.fun < c5 - 1e-13:
                    h[i] = s / 2.0 + res.x
                    h[j] = s / 2.0 - res.x
                    c5 = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5

def block_move(h0, target_sum, n_blocks, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for _ in range(n_blocks):
        block_len = rng.randint(3, min(15, n // 3))
        start = rng.randint(0, n - block_len)
        end = start + block_len
        shift = rng.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[start:end] = np.clip(h_new[start:end] + shift, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
    return h, c5

def projected_gradient_bb(h0, target_sum, alpha, n_iters):
    n = len(h0)
    h = h0.copy()
    h_prev = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    gamma = 1.0
    for it in range(n_iters):
        sm_val, g = smooth_max_value_and_grad(h, alpha)
        g = g + 2.0 * 50.0 * (np.sum(h) - target_sum) * np.ones(n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        hh = h - h_prev
        hh_norm = np.linalg.norm(hh)
        gg = sm_val - compute_c5(projected_gradient_bb_dummy(h)) if False else 0.0
        if hh_norm < 1e-15:
            lr = 0.1 / (g_norm + 1e-15)
        else:
            numerator = np.dot(hh, g - (h_prev and (smooth_max_value_and_grad(h_prev, alpha)[1] + 2.0 * 50.0 * (np.sum(h_prev) - target_sum) * np.ones(n))))
            if abs(numerator) > 1e-15 and hh_norm > 1e-15:
                lr = np.dot(hh, hh) / max(abs(numerator), 1e-15)
                lr = max(1e-6, min(lr, 2.0 / (g_norm + 1e-15)))
            else:
                lr = 0.1 / (g_norm + 1e-15)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur * 1.05:
            h_prev = h
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        else:
            h_prev = h
    return h_best, c5_best

def projected_gradient_bb_dummy(h):
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_h = project_to_sum(best_h, len(best_h) / 2.0)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        n_init = 360
        best_h = np.full(n_init, 0.5)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    target_sum = n / 2.0
    h_work = best_h.copy()

    # Phase 1: L-BFGS-B with Lagrangian at increasing alpha
    alphas_p1 = [10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
    for alpha in alphas_p1:
        if time_left() < 80:
            break
        h_opt = lbfgsb_with_lagrangian(h_work, target_sum, alpha, 8, 40, 0.5)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_work = h_opt

    # Phase 2: Penalty L-BFGS-B at high alpha
    if time_left() > 100:
        for alpha in [5000.0, 10000.0, 20000.0]:
            if time_left() < 40:
                break
            h_opt = lbfgsb_penalty(h_work, target_sum, alpha, 80, 100.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_work = h_opt

    # Phase 3: Coordinate descent with line search
    if time_left() > 150:
        n_cd = min(12, max(4, int(time_left() // 25)))
        h_cd, c5_cd = coord_descent_ls(h_work, target_sum, n_cd, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_work = h_cd

    # Phase 4: Pairwise updates
    if time_left() > 100:
        h_pw, c5_pw = pairwise_update(h_work, target_sum, 15, rng, 40)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()
            h_work = h_pw

    # Phase 5: Fixed-step coordinate descent
    if time_left() > 80:
        deltas = [-0.02, -0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01, 0.02]
        h_fd, c5_fd = coord_descent_fixed(h_work, target_sum, 20, rng, deltas)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()
            h_work = h_fd

    # Phase 6: Block moves
    if time_left() > 60:
        h_bm, c5_bm = block_move(h_work, target_sum, 200, rng)
        if c5_bm < best_c5:
            best_c5 = c5_bm
            best_h = h_bm.copy()
            h_work = h_bm

    # Phase 7: Re-optimize after local search
    if time_left() > 80:
        h_opt = lbfgsb_penalty(h_work, target_sum, 10000.0, 100, 200.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_work = h_opt

    # Phase 8: Resolution increase
    n2 = n * 2
    if time_left() > 250 and n2 <= 2000:
        target_sum2 = n2 / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, target_sum2)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_low2 = compute_c5(h_interp)

        alphas_p2 = [200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
        for alpha in alphas_p2:
            if time_left() < 60:
                break
            h_opt2 = lbfgsb_with_lagrangian(h_interp, target_sum2, alpha, 8, 40, 0.5)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                n = n2
                target_sum = target_sum2
            h_interp = h_opt2

        if time_left() > 80:
            h_opt2 = lbfgsb_penalty(h_interp, target_sum2, 10000.0, 80, 200.0)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                n = n2
                target_sum = target_sum2
            h_interp = h_opt2

        if time_left() > 80:
            n_cd2 = min(8, max(3, int(time_left() // 30)))
            h_cd2, c5_cd2 = coord_descent_ls(h_interp, target_sum2, n_cd2, rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                n = n2
                target_sum = target_sum2
                h_interp = h_cd2

        if time_left() > 60:
            h_pw2, c5_pw2 = pairwise_update(h_interp, target_sum2, 10, rng, 30)
            if c5_pw2 < best_c5:
                best_c5 = c5_pw2
                best_h = h_pw2.copy()
                n = n2
                target_sum = target_sum2
                h_interp = h_pw2

        if time_left() > 50:
            deltas2 = [-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01]
            h_fd2, c5_fd2 = coord_descent_fixed(h_interp, target_sum2, 15, rng, deltas2)
            if c5_fd2 < best_c5:
                best_c5 = c5_fd2
                best_h = h_fd2.copy()
                n = n2
                target_sum = target_sum2

    # Phase 9: Perturbation and re-optimization
    n_p = len(best_h)
    ts_p = n_p / 2.0
    if time_left() > 120:
        for pert_round in range(4):
            if time_left() < 40:
                break
            amp = 0.04 / (pert_round + 1)
            h_pert = best_h + rng.normal(0, amp, size=n_p)
            h_pert = project_to_sum(h_pert, ts_p)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_to_sum(h_pert, ts_p)
            c5_pert = compute_c5(h_pert)

            h_start = h_pert if c5_pert < best_c5 else best_h.copy()
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

            for alpha in [1000.0, 5000.0, 15000.0]:
                if time_left() < 30:
                    break
                h_opt = lbfgsb_penalty(h_start, ts_p, alpha, 60, 200.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_start = h_opt

            if time_left() > 35:
                h_cd, c5_cd = coord_descent_fixed(h_start, ts_p, 10, rng,
                                                  [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01])
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

            if time_left() > 30:
                h_pw, c5_pw = pairwise_update(best_h, ts_p, 10, rng, 30)
                if c5_pw < best_c5:
                    best_c5 = c5_pw
                    best_h = h_pw.copy()

    # Phase 10: Final polish with very high alpha
    if best_h is not None and time_left() > 25:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_opt = lbfgsb_penalty(best_h, ts_f, 30000.0, 150, 500.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

        if time_left() > 15:
            h_cd, c5_cd = coord_descent_ls(best_h, ts_f, 3, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Final projection
    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
