```python
        else:

            g[k2:] -= h[:n-k2]

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
```

```python
    h = h.copy()
```

```python
def projected_grad_descent(h0, target_sum, max_iters, alpha, step_init):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step = step_init
    for it in range(max_iters):
        corr = np.correlate(h, 1.0-h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        grad = compute_gradient(h, w)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-14:
            break
        # Normalized gradient step
        h_new = h - step * grad / norm_g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            step *= 1.02  # increase step slightly
            step = min(step, 0.5)
        else:
            step *= 0.7
            if step < 1e-8:
                break
```

Let me also think about a completely different idea: **simulated annealing**.

The idea: instead of always taking a step that decreases C₅, occasionally accept a step that increases C₅ (to escape local optima). The acceptance probability is exp(-ΔC₅ / T) where T is a cooling temperature.

```python
def simulated_annealing(h0, target_sum, max_iters, T_init, T_final, alpha):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    rng = np.random.RandomState(42)
    for it in range(max_iters):
        T = T_init * (T_final/T_init) ** (it/max_iters)
        corr = np.correlate(h, 1.0-h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        grad = compute_gradient(h, w)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-14:
            break
        step = 0.01 * (1 + 0.5 * (1 - it/max_iters))
        h_new = h - step * grad / norm_g + 0.005 * rng.randn(n)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    val = m + np.log(np.sum(e)) / alpha
    return val * dx

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    w = e / np.sum(e)
    g = np.zeros(n)
    nc = len(corr)
    for k in range(nc):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        p = abs(lag)
        wk = w[k]
        if p == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
        else:
            g[:n-p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n-p]
    return g * dx

def slsqp_opt(h0, n, target_sum, alpha, maxiter, ftol):
    def f(x):
        return smooth_max_obj(x, alpha)
    def g(x):
        return grad_smooth_max(x, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def proj_grad_descent(h0, target_sum, max_iters, alpha, step_init=0.05):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step = step_init
    for it in range(max_iters):
        dx = 2.0/n
        corr = np.correlate(h, 1.0-h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        e = np.exp(shifted)
        w = e / np.sum(e)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-10:
                continue
            lag = k - (n-1)
            p = abs(lag)
            if p == 0:
                g += w[k] * (1.0 - 2.0*h)
            elif lag > 0:
                g[p:] += w[k] * (1.0 - h[:n-p])
                g[:n-p] -= w[k] * h[p:]
            else:
                g[:n-p] += w[k] * (1.0 - h[p:])
                g[p:] -= w[k] * h[:n-p]
        g *= dx
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        h_new = h - step * g / ng
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.6
            if step < 1e-8:
                break
    return best_h, best_c5

def coord_descent(h0, target_sum, max_iters=5000):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0-h, mode='full')
        m_star = int(np.argmax(corr))
        lag = m_star - (n-1)
        p = abs(lag)
        g = np.zeros(n)
        if p == 0:
            g = 1.0 - 2.0*h
        elif lag > 0:
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
        else:
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        # Line search
        lo, hi = 0.0, max_move
        for _ in range(30):
            mid = 0.5 * (lo + hi)
            h[i_inc] += mid
            h[i_dec] -= mid
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_mid = compute_c5(h)
            if c5_mid < c5 - 1e-12:
                lo = mid
            else:
                hi = mid
            if hi - lo < 1e-10:
                break
        h[i_inc] += lo
        h[i_dec] -= lo
        h = np.clip(h, 0.0, 1.0)

            h[i_dec] += lo
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    return (m + np.log(np.sum(e)) / alpha) * dx

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    w = e / np.sum(e)
    g = np.zeros(n)
    nc = len(corr)
    for k in range(nc):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        p = abs(lag)
        wk = w[k]
        if p == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:

            g[:n - p] -= wk * h[p:]
        else:

    def f(x):
        return smooth_max_val(x, alpha)

    def g(x):
        return smooth_max_grad(x, alpha)

    bounds = [(0.0, 1.0)] * n
    try:

                best_c5, best_h, best_n = c5_cd, h_cd, n2

    if rem() > 100:
        n3 = 2160
        ts3 = n3 / 2.0
        h3 = interp_to_n(best_h, n3)
        h3 = project_to_sum(h3, ts3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3, n3

        for alpha in [5000, 30000]:
            if rem() < 40:
                break
            h_opt = slsqp_opt(h3, n3, ts3, alpha, 500, 1e-16)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt, n3
            h3 = h_opt

        if rem() > 50:
            h_gd, c5_gd = proj_grad_descent(h3, ts3, 1000, 10000, 0.05)
            if c5_gd < best_c5:
                best_c5, best_h, best_n = c5_gd, h_gd, n3

    n_best = best_n
    ts_best = n_best / 2.0

    if rem() > 60:
        time_left_multi = min(rem() - 30, 200)
        h_ms, c5_ms = multi_start_opt(best_h, n_best, ts_best, rng, time_left_multi, [5000, 20000])
        if c5_ms < best_c5:
            best_c5, best_h = c5_ms, h_ms

    if rem() > 30:
        h_cd, c5_cd = coord_descent(best_h, ts_best, 8000, 0.06)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    if rem() > 25:
        h_pol = slsqp_opt(best_h, n_best, ts_best, 80000, 1200, 1e-17)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    if rem() > 15:
        h_cd2, c5_cd2 = coord_descent(best_h, ts_best, 5000, 0.04)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2

    if rem() > 10:
        h_pol2 = slsqp_opt(best_h, n_best, ts_best, 200000, 1500, 1e-17)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5, best_h = c5_pol2, h_pol2

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
```
