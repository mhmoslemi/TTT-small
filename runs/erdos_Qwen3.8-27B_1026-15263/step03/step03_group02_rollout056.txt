def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_objective_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha)
    # Gradient computation
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def fast_slsqp(h0, n, target_sum, alpha, maxiter=500, ftol=1e-15):
    def f(h):
        return smooth_max_objective_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_objective_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_opt(h0, n, target_sum, alpha, maxiter=500):
    def obj_fn(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def grad_fn(h):
        return smooth_max_obj_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

            if cands:
                a, b = cands[rng.randint(len(cands))]
                if k > 0:
                    ones = np.where(h == 1.0)[0]
                    valid = [x for x in ones if x != b]
                else:
                    zeros = np.where(h == 0.0)[0]
                    valid = [x for x in zeros if x != a]
                if valid:
                    c = valid[rng.randint(len(valid))]
                    h[a], h[c] = h[c], h[a]
                    c5_new = compute_c5(h)
                    delta = c5_new - c5_curr
                    if delta <= 0 or rng.uniform() < np.exp(-delta / T):
                        c5_curr = c5_new
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_h = h.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        h[a], h[c] = h[c], h[a]
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                delta = c5_new - c5_curr
                if delta <= 0 or rng.uniform() < np.exp(-delta / T):
                    c5_curr = c5_new
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[i], h[j] = h[j], h[i]
                    no_improve += 1
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta <= 0 or rng.uniform() < np.exp(-delta / T):
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1

        steps += 1
        if steps % restart_interval == 0 and no_improve > restart_interval // 3:
            h = best_h.copy()
            n_pert = max(3, n // 80)
            for _ in range(n_pert):
                a = rng.randint(n)
                b = rng.randint(n)
                if h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
            c5_curr = compute_c5(h)
            T = min(T_init, T * 1.8)
            no_improve = 0

        T = max(T * cool, T_min)

    return best_h, best_c5

def multi_swap_sa(h_start, n, target_sum, t_limit, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.018
    T_min = 0.001
    no_improve = 0

    while time.time() < t_limit:
        n_swaps = 1 + rng.randint(0, 4)
        changes = []
        used = set()
        for _ in range(n_swaps):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            if i in used or j in used:
                continue
            h[i], h[j] = h[j], h[i]
            changes.append((i, j))
            used.add(i)
            used.add(j)

        if not changes:
            continue

        c5_new = compute_c5(h)
        delta = c5_new - c5_curr

        if delta <= 0 or rng.uniform() < np.exp(-delta / T):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            for i, j in reversed(changes):
                h[i], h[j] = h[j], h[i]
            no_improve += 1

        if no_improve > 25000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.2, 0.04)
            no_improve = 0

        T = max(T * 0.99999, T_min)

    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = int(n_main / 2.0)

    h_bin = make_binary(best_h, n_main, ts_main, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Directed SA (40% of budget)
    sa_end1 = t_start + 0.40 * budget_s
    h1, c5_1 = directed_sa(h_bin, n_main, ts_main, sa_end1, rng,
                           T_init=0.012, T_min=0.0005, cool=0.9999985, restart_interval=15000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-swap SA (15%)
    h_bin2 = make_binary(best_h, n_main, ts_main, rng)
    sa_end2 = t_start + 0.55 * budget_s
    if remaining() > 60:
        h2, c5_2 = multi_swap_sa(h_bin2, n_main, ts_main, sa_end2, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Gradient polish (15%)
    h_grad = best_h.copy()
    grad_end = t_start + 0.70 * budget_s
    for alpha in [30, 100, 500, 2000]:
        if elapsed() > grad_end:
            break
        h_pol = slsqp_opt(h_grad, n_main, n_main / 2.0, alpha, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_grad = h_pol.copy()

    # Phase 4: Second directed SA (15%)
    h_bin3 = make_binary(best_h, n_main, ts_main, rng)
    sa_end3 = t_start + 0.85 * budget_s
    if remaining() > 80:
        h3, c5_3 = directed_sa(h_bin3, n_main, ts_main, sa_end3, rng,
                               T_init=0.008, T_min=0.0003, cool=0.999999, restart_interval=10000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 5: Try other n values
    for n_try in [200, 280, 320]:
        if remaining() < 50:
            break
        ts_try = n_try // 2
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, float(ts_try))
        h_try_bin = make_binary(h_interp, n_try, ts_try, rng)
        t_end = t_start + budget_s - max(remaining() * 0.3, 25)
        h_try, c5_try = directed_sa(h_try_bin, n_try, ts_try, t_end, rng,
                                    T_init=0.01, T_min=0.001, cool=0.999998, restart_interval=12000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 6: Final gradient polish
    if remaining() > 20:
        h_fin = slsqp_opt(best_h, best_n, best_n / 2.0, 5000, maxiter=800)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
