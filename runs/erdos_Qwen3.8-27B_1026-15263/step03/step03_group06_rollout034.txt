            if abs(adj) <= min(np.min(h[mask]), 1.0 - np.max(h[mask])):
                h[mask] += adj
            else:
                # Need to saturate some
                if adj > 0:
                    # Increase: add to those with most room
                    h[mask] += min(adj, 1.0 - np.max(h[mask]))
                else:
                    h[mask] += max(adj, -np.min(h[mask]))
        else:
            # All are at bounds, can't project (shouldn't happen if target is feasible)
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_corr_at_lag(h, k):
    """Compute gradient of C(k) = sum_{valid i} h[i]*(1-h[i+k]) w.r.t. h"""
    n = len(h)
    if k >= n or -k >= n:
        return np.zeros(n)
    ak = abs(k)
    grad = np.zeros(n)
    if k >= 0:
        # C(k) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # dC/dh[j] for j < n-k: (1-h[j+k]) + h[j]*(-1) if j+k is also in range
        # Actually: dC/dh[j] = sum over terms where h[j] appears
        # h[j] appears as h[i] when i=j (if j < n-k): contributes (1-h[j+k])
        # h[j] appears as h[i+k] when i=j-k (if k<=j<n): contributes -h[j-k]
        grad[:n-ak] += (1.0 - h[ak:])  # h[j] as the first factor, for j < n-ak
        grad[ak:] -= h[:n-ak]  # h[j] as the second factor (1-h[j-ak]), for j >= ak
    else:
        # k < 0, let m = -k > 0
        # C(k) = C(-m) by the structure: sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # dC/dh[j]: h[j] appears as h[i] when i=j (if j>=m): contributes (1-h[j-m])
        # h[j] appears as h[i-m] when i=j+m (if j+m<n): contributes -h[j+m]
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

def smooth_max_and_grad(h, alpha):
    """Compute smooth max of correlation values and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Gradient: d(sm_val)/dh[j] = sum_k w_k * d(C_k)/dh[j]
    # C_k (in the "full" correlation indexing) corresponds to lag k-(n-1)
    # For the "full" output of np.correlate(h, 1-h, mode='full'):
    # index n-1+k corresponds to lag k

            continue
        k = idx - (n-1)  # lag
        gk = gradient_of_corr_at_lag(h, k)
        g += w[idx] * gk
    return sm_val, g

The first bracket is constant. The second is linear in h_unknown. The third is quadratic.

    lr = lr_init

            h = h_new
            c5_cur = c5_new

                h_best = h.copy()

  Wait: ∂/∂h[j] of (1-h[j-k]) = -1, and it's multiplied by h[j-k]. So the contribution is -h[j-k].
  But we need j-k to be a valid index in [0, n-1-k], so j-k ≥ 0 (j ≥ k) and j-k ≤ n-1-k (j ≤ n-1, always true).

So: ∂C(k)/∂h[j] = (1-h[j+k]) * 1[j < n-k] - h[j-k] * 1[j ≥ k]

For lag = -k (k ≥ 1):
C(-k) = sum_{i=k}^{n-1} h[i] * (1-h[i-k])

∂C(-k)/∂h[j]:
- If j ≥ k: h[j] appears in the term i=j: contributes (1-h[j-k])
- If j < n-k: h[j] appears as h[i-k] where i = j+k: contributes -h[j+k]
  (j+k ≥ k since j ≥ 0, and j+k ≤ n-1 since j ≤ n-1-k)

So: ∂C(-k)/∂h[j] = (1-h[j-k]) * 1[j ≥ k] - h[j+k] * 1[j < n-k]

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s) / alpha - np.log(len(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)

    s = np.sum(exp_vals)
    w = exp_vals / s

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        return smooth_max_obj(x, alpha)
    def grad(x):
        return smooth_max_grad(x, alpha)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0.copy()
