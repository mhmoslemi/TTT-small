    # f(+inf) = -s < 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, s):
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)

def project_to_sum(h, target_sum):
    return project_box_sum(h, target_sum)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # For numerical stability, subtract the max (already done)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m in range(n_lags):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            # "a" term: h[i] for i >= k, paired with (1-h[i-k])
            if k < n:
                idx = np.arange(k, n)
                grad[idx] += wm * (1.0 - h[idx - k])
            # "b" term: -h[i] for i <= n-1-k, paired with h[i+k]
            if k < n:
                idx = np.arange(0, n - k)
                grad[idx] += wm * (-h[idx + k])
        else:
            kp = -k
            # "a" term: h[i] for i <= n-1-kp, paired with (1-h[i+kp])
            if kp < n:
                idx = np.arange(0, n - kp)
                grad[idx] += wm * (1.0 - h[idx + kp])
            # "b" term: -h[i] for i >= kp, paired with h[i-kp]
            if kp < n:
                idx = np.arange(kp, n)
                grad[idx] += wm * (-h[idx - kp])
    
    return obj, grad

def fista_smooth_max(h0, alpha, target_sum, max_iter, t_deadline=None, verbose=False):
    h = h0.copy()
    z = h0.copy()
    prev_h = h0.copy()
    t = 1.0
    L = 2.0 * alpha  # Initial Lipschitz estimate
    
    best_obj = compute_c5(h)
    best_h = h.copy()
    stall = 0
    
    for k in range(max_iter):
        if t_deadline is not None and k % 50 == 0 and time.time() > t_deadline:
            break
        
        obj_z, grad_z = smooth_max_obj_grad(z, alpha)
        
        # Gradient step with backtracking
        step = 1.0 / L
        y = z - step * grad_z
        y = project_box_sum(y, target_sum)
        
        obj_y, _ = smooth_max_obj_grad(y, alpha)
        
        backtracks = 0
        while obj_y > obj_z + np.dot(grad_z, y - z) + 0.5 * L * np.dot(y - z, y - z) + 1e-14 and backtracks < 15:
            L *= 2.0
            step = 1.0 / L
            y = z - step * grad_z
            y = project_box_sum(y, target_sum)
            obj_y, _ = smooth_max_obj_grad(y, alpha)
            backtracks += 1
        
        prev_h = h.copy()
        h = y
        
        # Track best
        c5_cur = np.max(np.correlate(h, 1.0 - h, mode="full"))  # without dx
        if c5_cur < best_obj / (2.0 / len(h)) - 1e-12:  # Compare raw max corr
            best_obj = c5_cur * (2.0 / len(h))
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        z = h + momentum * (h - prev_h)
        z = np.clip(z, 0.0, 1.0)
        # Re-project to maintain sum constraint approximately
        if abs(z.sum() - target_sum) > 1e-6:
            z = project_box_sum(z, target_sum)
        t = t_new
        
        # Decay L slightly to allow larger steps (helps with non-convexity)
        L = max(L * 0.999, 2.0 * alpha * 0.1)
    
    return best_h, best_obj

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(32):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10, n_max_sweeps=1):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0:
                h[j] = 0.0
            elif h[j] > 1:
                h[j] = 1.0
            if h[best_k] < 0:
                h[best_k] = 0.0
            elif h[best_k] > 1:
                h[best_k] = 1.0
            improved = True
    return h, corr, improved

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, s):
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                idx_a = np.arange(k, n)
                grad[idx_a] += wm * (1.0 - h[idx_a - k])
                idx_b = np.arange(0, n - k)
                grad[idx_b] += wm * (-h[idx_b + k])
        else:
            kp = -k
            if kp < n:
                idx_a = np.arange(0, n - kp)
                grad[idx_a] += wm * (1.0 - h[idx_a + kp])
                idx_b = np.arange(kp, n)
                grad[idx_b] += wm * (-h[idx_b - kp])
    
    return obj, grad

        y = z - step * grad_z
