            valid2 = slice(k, n)
            g[valid1] += wm * (1.0 - h[k:n])
            g[valid2] -= wm * h[:n-k]
        else:
            m_ = -k
            # c[-m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            # dc/dh[j] = (1-h[j-m]) for j in [m, n-1], -h[j+m] for j in [0, n-1-m]
            valid1 = slice(m_, n)
            valid2 = slice(0, n - m_)
            g[valid1] += wm * (1.0 - h[:n-m_])
            g[valid2] -= wm * h[m_:]
    
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, active_idx=None):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_value_and_grad(h, alpha, active_idx)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_value_and_grad(h, alpha, active_idx)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def grad_guided_coord_descent(h0, n, target_sum, max_rounds, rng, alpha=500.0):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        _, g = smooth_max_value_and_grad(h, alpha)
        # Sort coordinates by |gradient|
        grad_mag = np.abs(g)
        # Try the top coordinates
        n_try = min(n, max(50, n // 3))
        top_idx = np.argsort(grad_mag)[-n_try:]
        top_idx = rng.permutation(top_idx)
        
        improved = False
        for i in top_idx:
            step = 0.1 * grad_mag[i] / (np.max(grad_mag) + 1e-15)
            step = np.clip(step, 0.001, 0.05)
            # Try moving in gradient direction
            for sign in [-1, 1]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + sign * step, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            # Try smaller steps on random coordinates
            for _ in range(n // 5):
                i = rng.randint(n)
                for d in [0.001, -0.001, 0.0005, -0.0005]:
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        break
            if not improved:
                break
    return h, c5_cur

def active_lag_refinement(h0, n, target_sum, alpha, maxiter, threshold=0.99):
    h = h0.copy()
    for _ in range(5):
        c5_cur = compute_c5(h)
        corr = compute_correlation(h)
        max_corr = np.max(corr)
        active = np.where(corr >= threshold * max_corr)[0]
        if len(active) < 1:
            break
        h = slsqp_optimize(h, n, target_sum, alpha, maxiter, active_idx=active)
        c5_new = compute_c5(h)
        if c5_new >= c5_cur:
            break
    return h, compute_c5(h)

def perturb_and_optimize(h_base, n, target_sum, rng, n_perturb, alpha, maxiter):
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    for p in range(n_perturb):
        # Add small random perturbation
        pert = rng.randn(n) * 0.02
        h_pert = h_base + pert
        h_pert = project_to_sum(h_pert, target_sum)
        h_opt = slsqp_optimize(h_pert, n, target_sum, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5
