import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter=2000, penalty=100.0):
    """L-BFGS-B with penalty for sum constraint."""
    def f_pen(h):
        obj, grad = smooth_max_and_grad(h, alpha)
        penalty_term = 0.5 * penalty * (np.sum(h) - target_sum)**2
        grad_pen = grad + penalty * (np.sum(h) - target_sum) * np.ones(n)
        return obj + penalty_term, grad_pen
    
    # L-BFGS-B doesn't support returning gradient directly in the same way
    # Let's use a wrapper
    def f(h):
        return smooth_max_and_grad(h, alpha)[0] + 0.5 * penalty * (np.sum(h) - target_sum)**2
    
    def g(h):
        _, grad = smooth_max_and_grad(h, alpha)
        return grad + penalty * (np.sum(h) - target_sum) * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def make_binary(h, n, target_sum, rng):

def smooth_max_and_grad(h, alpha):

    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1

    valid_s = (second_idx >= 0) & (second_idx < n)

    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter=3000, penalty=500.0):
