    y = h.copy()  # Nesterov momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    lr = lr_init
    
    for it in range(max_iters):
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        y_new = y + beta * (y - h)
        
        # Compute correlations
        corr = np.correlate(y_new, 1.0 - y_new, mode='full')
        
        # Use weighted sum of top m lags
        # Find top m
        m = min(top_m, len(corr))
        top_idx = np.argsort(corr)[-m:]
        
        # Weights: softmax over top m
        vals = corr[top_idx]
        w = np.exp(alpha_smooth * (vals - np.max(vals)))
        w /= np.sum(w)

        g = np.zeros(n)
