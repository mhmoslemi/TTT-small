```python
def compute_grad_c5_smooth(h, w, dx):
    """Compute gradient of smooth max C5 objective.
    w: weights for each correlation lag (length 2n-1)
    """
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # For each lag k (0 to 2n-2), j = k-(n-1)
    # corr[k] = sum_i h[i]*(1-h[i+j]) for valid i
    # d(corr[k])/d(h[m]) = (1-h[m+j])*[valid m as i] - h[m-j]*[valid m-j as i]
    
    # Precompute valid ranges
    # For lag k, j = k-(n-1):
    #   i ranges from max(0,-j) to min(n-1, n-1-j)
    
    for k in range(2*n-1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        j = k - (n-1)
        i_min = max(0, -j)
        i_max = min(n-1, n-1-j)
        if i_max < i_min:
            continue
        # First term: for m in [i_min, i_max], contribution is (1-h[m+j])
        grad[i_min:i_max+1] += wk * one_minus_h[i_min+j:i_max+j+1]
        # Second term: for m in [i_min+j, i_max+j], contribution is -h[m-j]
        s2_start = i_min + j
        s2_end = i_max + j + 1
        if s2_start >= 0 and s2_end <= n:
            grad[s2_start:s2_end] -= wk * h[i_min:i_max+1]
    
    return grad
```

```python
def compute_smooth_max_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    size = 2 * n - 1
    
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    scaled = corr * dx
    
    # Smooth max
    m_val = np.max(scaled)
    shifted = alpha * (scaled - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + np.log(np.sum(exp_v)) / alpha
    
    # Gradient using FFT
    # A[k] = w[k] for k=0,...,2n-2
    A = w.copy()
    u = A[::-1]  # reversed
    
    # S[m] = P[2n-1-m] - P[n-1-m] where P is prefix sum of A
    P = np.zeros(2*n)
    P[1:] = np.cumsum(A)
    S = np.zeros(n)
    for m in range(n):
        S[m] = P[2*n-1-m] - P[n-1-m]
    # Vectorize:
    # S[m] = P[2n-1-m] - P[n-1-m] for m=0,...,n-1
    # 2n-1-m for m=0,...,n-1 is 2n-1, 2n-2, ..., n
    # n-1-m for m=0,...,n-1 is n-1, n-2, ..., 0
    S = P[n:2*n][::-1] - P[0:n][::-1]
    # Wait: P[2n-1-m] for m=0 is P[2n-1], for m=n-1 is P[n]
    # So S[m] = P[2n-1-m] - P[n-1-m]
    # S[0] = P[2n-1] - P[n-1]
    # S[n-1] = P[n] - P[0]
    
    idx1 = np.array([2*n-1-m for m in range(n)])  # 2n-1, 2n-2, ..., n
    idx2 = np.array([n-1-m for m in range(n)])  # n-1, n-2, ..., 0
    S = P[idx1] - P[idx2]
    
    # Cross-correlation of (A+u) and h
    Au = A + u
    # (Au R h)[k] = sum_i Au[i+k]*h[i] for k=0,...,2n-2
    # Using FFT: 
    N = 1  # next power of 2 >= 2n-1
    while N < 4*n:
        N *= 2
    FFT_Au = np.fft.rfft(Au, N)
    FFT_h = np.fft.rfft(h, N)
    corr_Au_h = np.fft.irfft(FFT_Au * np.conj(FFT_h), N)[:2*n-1]
    # corr_Au_h[k] = sum_i Au[i+k]*h[i] = (Au R h)[k]
    
    # T2[m] + T3[m] = (Au R h)[n-1-m]
    # So (T2+T3)[m] = corr_Au_h[n-1-m]
    T = corr_Au_h[n-1::-1]  # T[m] = corr_Au_h[n-1-m] for m=0,...,n-1
    
    grad = S - T
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m_val = np.max(scaled)
    shifted = alpha * (scaled - m_val)
    max_shifted = np.max(shifted)
    exp_v = np.exp(shifted - max_shifted)
    denom = np.sum(exp_v)
    return m_val + (max_shifted + np.log(denom)) / alpha

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    size = 2 * n - 1
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    
    m_val = np.max(scaled)
    shifted = alpha * (scaled - m_val)
    max_shifted = np.max(shifted)
    exp_v = np.exp(shifted - max_shifted)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_shifted + np.log(np.sum(exp_v))) / alpha
    
    # FFT-based gradient
    A = w  # length 2n-1
    u = A[::-1]
    Au = A + u
    
    # S[m] = P[2n-1-m] - P[n-1-m]
    P = np.zeros(2*n)
    P[1:] = np.cumsum(A)
    idx1 = 2*n-1 - np.arange(n)  # [2n-1, 2n-2, ..., n]
    idx2 = n-1 - np.arange(n)  # [n-1, n-2, ..., 0]
    S = P[idx1] - P[idx2]
    
    # D[k] = (h ⊛ Au)[k] via FFT
    N = 1
    while N < 4*n:
        N *= 2
    FFT_h = np.fft.rfft(h, N)
    FFT_Au = np.fft.rfft(Au, N)
    D = np.fft.irfft(np.conj(FFT_h) * FFT_Au, N)[:size]
    
    T = D[n-1::-1]  # T[m] = D[n-1-m]
    
    grad = S - T
    return obj, grad

def lbfgs_opt(h0, n, ts, alpha, maxiter=300):
    def f(x):
        return smooth_max_obj(x, alpha)
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    bnds = [(0.0, 1.0)] * n
    # Add equality constraint for sum
    # L-BFGS-B doesn't support equality constraints directly
    # We'll use a penalty or project after each step
    # Actually, let's use a penalty approach
    penalty = 1e4
    def f_pen(x):
        obj = smooth_max_obj(x, alpha)
        sum_viol = np.sum(x) - ts
        return obj + penalty * sum_viol**2
    def g_pen(x):
        _, grad = smooth_max_obj_grad(x, alpha)
        sum_viol = np.sum(x) - ts
        return grad + 2*penalty*sum_viol
    try:
        r = minimize(f_pen, h0.copy(), method='L-BFGS-B', jac=g_pen, 
                     bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)
```
