import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, top_k, temperature):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    weights = np.exp(temperature * (corr - m_val))
    total_w = np.sum(weights)
    if total_w < 1e-300:
        top_idx = np.argmax(corr)
        return corr_gradient(h, int(top_idx) - (n - 1))
    weights = weights / total_w
    top_indices = np.argsort(weights)[::-1][:top_k]
    g = np.zeros(n)
    for idx in top_indices:
        w = weights[idx]
        if w < 1e-10:
            break
        lag = int(idx) - (n - 1)
        g += w * corr_gradient(h, lag)
    return g

def coord_desc_linesearch(h, target_sum, max_iters, top_k, temperature, base_step):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        g = multi_lag_gradient(h, top_k, temperature)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        step = base_step * max_move
        if step < 1e-12:
            break
        best_step = step
        best_val = c5
        for attempt in range(6):
            trial = h.copy()
            trial[i_inc] += best_step
            trial[i_dec] -= best_step
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_sum(trial, target_sum)
            v = compute_c5(trial)
            if v < best_val:
                best_val = v
                h = trial
                c5 = v
                stale = 0
                break
            best_step *= 0.5
        else:
            stale += 1
            if stale > 200:
                break
    return h, c5

def fourier_mutation(h, n, target_sum, rng, n_modes, amplitude):
    x = np.arange(n)
    delta = np.zeros(n)
    for j in range(1, n_modes + 1):
        freq = j * 2 * np.pi / n
        a = rng.randn() * amplitude / np.sqrt(j)
        b = rng.randn() * amplitude / np.sqrt(j)
        delta += a * np.cos(freq * x) + b * np.sin(freq * x)
    h_new = h + delta
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def block_mutation(h, n, target_sum, rng, n_blocks, amplitude):
    h_new = h.copy()
    block_size = max(2, n // n_blocks)
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        block_len = rng.randint(max(2, block_size // 3), block_size + 1)
        block_len = min(block_len, n - start)
        shift = rng.uniform(-amplitude, amplitude)
        h_new[start:start + block_len] += shift
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def crossover_parents(h1, h2, rng, target_sum):
    t = rng.uniform(0.2, 0.8)
    child = t * h1 + (1.0 - t) * h2
    child = project_to_sum(child, target_sum)
    return child
