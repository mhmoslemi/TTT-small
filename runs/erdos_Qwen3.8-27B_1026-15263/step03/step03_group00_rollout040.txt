import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, target_sum, lr, momentum):
    """One step of subgradient descent with momentum for the max-k problem."""
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    # Find all active lags (within tolerance)
    tol = 1e-6 * max(1.0, m_val)
    active = np.where(corr >= m_val - tol)[0]
    # Average subgradient over active lags
    n_active = len(active)
    g = np.zeros(n)
    for lag_idx in active:
        lag = int(lag_idx) - (n - 1)
        if lag >= 0:
            if lag == 0:
                g[:] += (1.0 - 2.0 * h)
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    g /= n_active
    return g

    best_c5 = compute_c5(h0)
    
    for t in range(n_steps):
        lr = lr_init / (1.0 + t * 0.01)
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target_sum)
        
        g = subgradient_step(y, target_sum, lr, momentum)
        
        # Projected gradient step
        v_new = momentum * v - lr * g
        h_new = y + v_new
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        v = v_new
        h = h_new
    
    return best_h, best_c5

def binarize(h, target_sum):
    """Convert to binary, adjusting to get exactly target_sum ones."""
    n = len(h)
    # Sort by value, keep top target_sum
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    for i in range(int(round(target_sum))):
        h_bin[sorted_idx[i]] = 1.0
    return h_bin

def binary_local_search(h, target_sum, max_iters=20000, patience=500):
    """Local search on binary vectors using swaps."""
    n = len(h)
    h = binarize(h, target_sum)
    c5 = compute_c5(h)
    corr = compute_corr(h)
    best_lag = int(np.argmax(corr))
    best_lag_val = corr[best_lag]
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    no_improve = 0
    rng = np.random.RandomState(123)
    
    for it in range(max_iters):
        # Recompute if needed
        corr = compute_corr(h)
        m_val = np.max(corr)
        
        # Try to find a beneficial swap
        # Focus on the active lag
        active_lag = int(np.argmax(corr)) - (n - 1)
        
        # Get gradient for active lag
        g = np.zeros(n)
        if active_lag >= 0:
            if active_lag == 0:
                g[:] = (1.0 - 2.0 * h)
            else:
                g[active_lag:] += (1.0 - h[:n - active_lag])
                g[:n - active_lag] -= h[active_lag:]
        else:
            k = -active_lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        
        # We want to swap a 1 (at position i) with a 0 (at position j)
        # such that the change in the active lag cost is negative
        # Change in cost = -g[i] + g[j] (approximately, for the active lag)
        # So we want g[i] > g[j], i.e., move mass from high-gradient to low-gradient
        
        # Find best swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a limited number of random swaps
        best_swap_c5 = c5
        best_swap = None
        
        n_tries = min(20, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            c5_try = compute_c5(h)
            if c5_try < best_swap_c5:
                best_swap_c5 = c5_try
                best_swap = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            c5 = best_swap_c5
            no_improve = 0
        else:
            # Try gradient-guided swap
            g_ones = g[ones]
            g_zeros = g[zeros]
            # We want to move 1 to where gradient is low, from where gradient is high
            i = ones[int(np.argmax(g_ones))]
            j = zeros[int(np.argmin(g_zeros))]
            if i != j and g[i] > g[j]:
                h[i] = 0.0
                h[j] = 1.0
                c5_try = compute_c5(h)
                if c5_try <= c5 + 1e-10:  # Allow tiny increase
                    c5 = c5_try
                    no_improve = 0
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > patience:
            # Perturb
            n_flip = np.random.randint(5, 20)
            flip_ones = np.random.choice(ones, size=min(n_flip, len(ones)), replace=False)
            flip_zeros = np.random.choice(zeros, size=min(n_flip, len(zeros)), replace=False)
            n_flip = min(len(flip_ones), len(flip_zeros))
            h[flip_ones[:n_flip]] = 0.0
            h[flip_zeros[:n_flip]] = 1.0
            c5 = compute_c5(h)
            no_improve = 0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    
    return h, c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbfgs_local(h0, n, target_sum, alpha, maxiter):
    """L-BFGS-B optimization with smooth max."""
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    # Add penalty for sum constraint in the objective
    penalty = 1000.0
    def f_pen(x):
        base = smooth_max_and_grad(x, alpha)[0]
        sum_viol = np.sum(x) - target_sum
        return base + penalty * sum_viol * sum_viol
    
    def g_pen(x):
        _, grad = smooth_max_and_grad(x, alpha)
        sum_viol = np.sum(x) - target_sum
        return grad + 2 * penalty * sum_viol
    
    try:
        res = minimize(f_pen, h0.copy(), method='L-BFGS-B', jac=g_pen,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                               'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize(h, target_sum):
    n = len(h)
    k = int(round(target_sum))
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:k]] = 1.0
    return h_bin


def lags_of(k, p, i, j):
    n = len(k)
    n1 = n - 1
    a_from_first = k >= n1 - i
    a_from_second = k >= i + 1
    b_from_first = k >= n1 - j
    b_from_second = k >= j + 1
    valid = (a_from_first | a_from_second | b_from_first | b_from_second) & (k >= 0) & (k <= n1)
    res = np.zeros(len(k))
    res[valid & a_from_first] += 1
    res[valid & a_from_second] -= 1
    res[valid & b_from_first] -= 1
    res[valid & b_from_second] += 1
    res = res.astype(np.int64)
    indices = np.arange(2 * n - 1)
    mask_f = (indices >= 0) & (indices < n)
    mask_s = (indices >= 0) & (indices < n)
    fc = np.clip(indices, 0, n - 1)
    sc = np.clip(indices, 0, n - 1)
    res_f = res * np.where(mask_f, i - fc, 0)
    res_s = res * np.where(mask_s, j - sc, 0)
    return res_f, res_s


def corr_delta(h, i, j):
    n = len(h)
    k = np.arange(2 * n - 1)
    n1 = n - 1
    idx = k - n1
    abs_idx = np.abs(idx)
    result = np.zeros(2 * n - 1, dtype=np.float64)
    pos = idx >= 0
    neg = ~pos
    res_f, res_s = lags_of(k, n1, i, j)
    fc = np.clip(k, 0, n - 1)
    sc = np.clip(k + n1, 0, n - 1)
    mf = (k >= 0) & (k < n)
    ms = (k + n1 >= 0) & (k + n1 < n)
    result[pos] = -res_f[pos] * h[np.clip(k[pos], 0, n - 1)] + res_s[pos] * h[np.clip(k[pos] + n1, 0, n - 1)]
    kf = np.clip(k[neg], 0, n - 1)
    ks = np.clip(k[neg] + n1, 0, n - 1)
    result[neg] = -res_f[neg] * h[kf] + res_s[neg] * h[ks]

def binary_local_search(h, target_sum, max_iters=20000, patience=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    n = len(h)
    h = binarize(h, target_sum)
    c5 = compute_c5(h)
    corr = compute_corr(h)
    no_improve = 0
    best_c5 = c5
    best_h = h.copy()

    for it in range(max_iters):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        n_tries = 8
        best_swap_c5 = c5
        best_swap = None

        for _ in range(n_tries):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            delta = corr_delta(h, i, j)
            corr_try = corr + delta
            c5_try = float(np.max(corr_try)) / n
            if c5_try < best_swap_c5 - 1e-12:
                best_swap_c5 = c5_try
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + corr_delta(h, i, j)
            h[i] = 1.0
            h[j] = 0.0
            h[i] = 0.0
            h[j] = 1.0
            corr = compute_corr(h)
            c5 = best_swap_c5
            no_improve = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > patience:
                n_flip = rng.randint(5, min(20, len(ones), len(zeros)))
                fi = rng.choice(ones, size=n_flip, replace=False)
                fj = rng.choice(zeros, size=n_flip, replace=False)
                h[fi] = 0.0
                h[fj] = 1.0
                corr = compute_corr(h)
                c5 = float(np.max(corr)) / n
                no_improve = 0

    if best_c5 < c5:
        return best_h, best_c5
    return h, c5


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad


def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]

    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def lbfgs_local(h0, n, target_sum, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    penalty = 500.0

    def f_pen(x):
        base = smooth_max_and_grad(x, alpha)[0]
        sv = np.sum(x) - target_sum
        return base + penalty * sv * sv

    def g_pen(x):
        _, grad = smooth_max_and_grad(x, alpha)
        sv = np.sum(x) - target_sum
        return grad + 2.0 * penalty * sv

    try:
        res = minimize(f_pen, h0.copy(), method='L-BFGS-B', jac=g_pen,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                               'gtol': 1e-8, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]

        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5


def interp_to_n(h, n_new):

    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)


def nesterov_subgrad(h0, target_sum, n_steps, lr_init=0.05, momentum=0.9, decay=0.999):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init

    for t in range(n_steps):
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target_sum)

        corr = compute_corr(y)
        m_val = np.max(corr)
        tol = max(1e-8, 1e-6 * m_val)
        active = np.where(corr >= m_val - tol)[0]
        
        g = np.zeros(n)
        for lag_idx in active:
            lag = int(lag_idx) - (n - 1)
            if lag >= 0:
                if lag == 0:
                    g[:] += (1.0 - 2.0 * y)
                else:
                    g[lag:] += (1.0 - y[:n - lag])
                    g[:n - lag] -= y[lag:]
            else:
                k = -lag
                g[:n - k] += (1.0 - y[k:])
                g[k:] -= y[:n - k]
        g /= len(active)

        v = momentum * v - lr * g
        h_new = y + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        lr *= decay

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: Binary local search (fast with analytical swap delta)
    if time_left() > 100:
        h_bin = binarize(best_h, ts_parent)
        h_bl, c5_bl = binary_local_search(h_bin, ts_parent, max_iters=30000, patience=200, rng=np.random.RandomState(seed))
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 2: Subgradient with Nesterov momentum
    if time_left() > 80:
        h_sg, c5_sg = nesterov_subgrad(best_h, ts_parent, n_steps=3000, lr_init=0.08, momentum=0.85, decay=0.9995)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: SLSQP with smooth max
    for alpha in [50, 200, 1000, 5000]:
        if time_left() < 30:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: L-BFGS-B
    for alpha in [100, 500, 2000]:
        if time_left() < 30:
            break
        h_new = lbfgs_local(best_h, n_parent, ts_parent, alpha, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Coordinate descent
    if time_left() > 60:
        h_cd, c5_cd = coordinate_descent(best_h, ts_parent, max_iters=10000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Multi-resolution
    resolutions = [480, 540, 600, 720]
    for n_tr in resolutions:
        if time_left() < 120:
            break
        if n_tr % 2 != 0:
            n_tr += 1
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)

        # Binary search at this resolution
        h_bin_tr = binarize(h_interp, ts_tr)
        h_bl_tr, c5_bl_tr = binary_local_search(h_bin_tr, ts_tr, max_iters=15000, patience=150, rng=np.random.RandomState(seed + n_tr))
        if c5_bl_tr < c5_interp:
            c5_interp = c5_bl_tr
            h_interp = h_bl_tr

        # SLSQP polish
        for alpha in [100, 500]:
            if time_left() < 40:
                break
            h_opt = slsqp_local(h_interp, n_tr, ts_tr, alpha, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_interp:
                c5_interp = c5_opt
                h_interp = h_opt

        # Coordinate descent
        if time_left() > 40:
            h_cd_tr, c5_cd_tr = coordinate_descent(h_interp, ts_tr, max_iters=5000, step_frac=0.1)
            if c5_cd_tr < c5_interp:
                c5_interp = c5_cd_tr
                h_interp = h_cd_tr

        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

    # Phase 7: Multi-start perturbations
    n_best = best_n
    ts_best = n_best / 2.0

    for trial in range(25):
        if time_left() < 40:
            break
        strategy = trial % 5
        if strategy == 0:
            noise = 0.03 + 0.04 * (1.0 - trial / 25.0)
            h_shake = best_h + rng.uniform(-noise, noise, n_best)
        elif strategy == 1:
            h_shake = best_h.copy()
            n_flip = rng.randint(10, 40)
            perm = rng.permutation(n_best)
            ones_mask = h_shake > 0.5
            ones_pos = perm[ones_mask[perm]]
            zeros_pos = perm[~ones_mask[perm]]
            nf = min(n_flip, len(ones_pos), len(zeros_pos))
            h_shake[ones_pos[:nf]] = 0.0
            h_shake[zeros_pos[:nf]] = 1.0
        elif strategy == 2:
            freq = rng.randint(2, 12)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n_best, endpoint=False)
            pert = 0.06 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            h_shake = best_h + pert
        elif strategy == 3:
            h_shake = best_h.copy()
            start = rng.randint(0, max(1, n_best - 30))
            length = rng.randint(10, min(40, n_best - start))
            h_shake[start:start + length] = 1.0 - h_shake[start:start + length]
        else:
            h_shake = best_h.copy()
            n_blocks = rng.randint(3, 8)
            for _ in range(n_blocks):
                start = rng.randint(0, max(1, n_best - 15))
                length = rng.randint(5, min(15, n_best - start))
                h_shake[start:start + length] = 1.0 - h_shake[start:start + length]

        h_shake = project_to_sum(h_shake, ts_best)

        h_cd, c5_cd = coordinate_descent(h_shake, ts_best, max_iters=4000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        if time_left() < 40:
            break
        h_opt = slsqp_local(h_cd, n_best, ts_best, 500, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Quick binary polish
        if time_left() > 60:
            h_bin_p = binarize(h_opt, ts_best)
            h_bp, c5_bp = binary_local_search(h_bin_p, ts_best, max_iters=8000, patience=100, rng=np.random.RandomState(seed + trial * 7))
            if c5_bp < best_c5:
                best_c5 = c5_bp
                best_h = h_bp.copy()

    # Phase 8: Final polishing
    for alpha in [2000, 10000, 50000]:
        if time_left() < 25:
            break
        h_pol = slsqp_local(best_h, best_n, ts_best, alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 30:
        h_final_cd, c5_final = coordinate_descent(best_h, ts_best, max_iters=15000, step_frac=0.05)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_cd.copy()

    # Final binary search for last-mile improvement
    if time_left() > 30:
        h_bin_final = binarize(best_h, ts_best)
        h_bf, c5_bf = binary_local_search(h_bin_final, ts_best, max_iters=30000, patience=500, rng=np.random.RandomState(seed + 999))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final SLSQP polish after binary
    if time_left() > 20:
        for alpha in [20000, 100000]:
            h_p = slsqp_local(best_h, best_n, ts_best, alpha, maxiter=1000)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
