            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, threshold=0.99):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    active = np.where(corr >= threshold * max_corr)[0]
    return active, max_corr

def slp_optimize(h, n, target_sum, n_slp_iters=20, use_active=True, active_threshold=0.95):
    """Sequential Linear Programming"""
    h = h.copy()
    c5_cur = compute_c5(h)
    
    for it in range(n_slp_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        
        if use_active:
            active = np.where(corr >= active_threshold * max_corr)[0]
        else:
            active = np.arange(len(corr))
        
        if len(active) == 0:
            active = np.argmax(corr).__class__([np.argmax(corr)])
        
        # Build LP
        # Variables: h[0], ..., h[n-1], t
        # Minimize t
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for idx in active:
            k = idx - (n - 1)
            m = abs(k)
            # corr[idx] = sum_{i=0}^{n-1-m} h[i+m](1-h[i])
            # Linearize:
            grad = np.zeros(n)
            for i in range(n - m):
                # h[i+m] * (1-h[i])
                # d/dh[i+m]: (1-h[i])
                # d/dh[i]: -h[i+m]
                grad[i+m] += (1.0 - h[i])
                grad[i] -= h[i+m]
            
            # f_lin(h) = corr[idx] + grad · (h - h0) = grad · h + (corr[idx] - grad · h)
            const = corr[idx] - np.dot(grad, h)
            
            # Constraint: grad · h + const ≤ t
            # i.e., grad · h - t ≤ -const
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-const)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        # Equality constraint: sum(h) = target_sum
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[n]
            
            # Check actual c5
            c5_new = compute_c5(h_new)
            
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
            else:
                # Try a step
                for alpha in [0.5, 0.25, 0.1]:
                    h_try = h + alpha * (h_new - h)
                    h_try = project_to_sum(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur:
                        h = h_try
                        c5_cur = c5_try
                        break
    
    return h, c5_cur

from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, active_idx=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_idx is not None:
        active_corr = corr[active_idx]
    else:
        active_corr = corr
        active_idx = np.arange(len(corr))
    
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for pos, idx in enumerate(active_idx):
        wm = w[pos]
        if wm < 1e-12:
            continue
        k = idx - (n - 1)
        m_abs = abs(k)
        g[m_abs:] += wm * (1.0 - h[:n - m_abs])
        g[:n - m_abs] -= wm * h[m_abs:]
    
    return sm_val, g

def slsqp_smooth(h0, n, target_sum, alpha, active_idx, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha, active_idx)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha, active_idx)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slp_optimize(h, n, target_sum, n_iters=30, active_threshold=0.90):

    c5_cur = compute_c5(h)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        active = np.where(corr >= active_threshold * max_corr)[0]
        
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub_rows = []
        b_ub_vals = []
        
        for idx in active:
            k = idx - (n - 1)
            m_abs = abs(k)
            grad = np.zeros(n)
            grad[m_abs:] += (1.0 - h[:n - m_abs])
            grad[:n - m_abs] -= h[m_abs:]
            const = corr[idx] - np.dot(grad, h)
            
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(-const)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, max_corr + 1.0)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            c5_new = compute_c5(h_new)
            
            if c5_new < c5_cur - 1e-10:
                h = h_new
                c5_cur = c5_new
            else:
                # Try damped step
                improved = False
                for alpha_step in [0.5, 0.25, 0.1]:
                    h_try = h + alpha_step * (h_new - h)
                    h_try = project_to_sum(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-10:
                        h = h_try
                        c5_cur = c5_try
                        improved = True
                        break
                if not improved:
                    break
        else:
            break
    
    return h, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-12:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

Actually, the simplest fix is to set the lower bound on t to 0. Since the true C5 is positive, this is always valid.

Let me also set an upper bound on t to the current max correlation value, so the LP is bounded.

Here's the final code:

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_active(h, alpha, active_idx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for pos, idx in enumerate(active_idx):
        wm = w[pos]
        if wm < 1e-12:
            continue
        k = idx - (n - 1)
        m_abs = abs(k)
        g[m_abs:] += wm * (1.0 - h[:n - m_abs])
        g[:n - m_abs] -= wm * h[m_abs:]
    return sm_val, g


def slsqp_smooth(h0, n, target_sum, alpha, active_idx, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_active(h, alpha, active_idx)
        return v

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_active(h, alpha, active_idx)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)


def slp_optimize(h, n, target_sum, n_iters=30, active_threshold=0.90, max_lags=150):
    h = h.copy()
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        active = np.where(corr >= active_threshold * max_corr)[0]
        if len(active) > max_lags:
            active = active[np.argsort(corr[active])[-max_lags:]]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub_rows = []
        b_ub_vals = []
        for idx in active:
            k = idx - (n - 1)
            m_abs = abs(k)
            grad = np.zeros(n)
            grad[m_abs:] += (1.0 - h[:n - m_abs])
            grad[:n - m_abs] -= h[m_abs:]
            const = corr[idx] - np.dot(grad, h)
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(-const)

        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        bounds = [(0.0, 1.0)] * n + [(0.0, max_corr + 1.0)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success:
            break
        h_new = res.x[:n]
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
        else:
            improved = False
            for alpha_step in [0.5, 0.25, 0.1, 0.05]:
                h_try = h + alpha_step * (h_new - h)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
                    break
            if not improved:
                break
    return h, c5_cur


def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [-0.005, -0.002, -0.0005, 0.0005, 0.002, 0.005]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-12:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur


def projected_subgradient(h0, n, target_sum, n_iters, lr_init, alpha, active_idx):
    h = h0.copy()
    c5_cur = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_active(h, alpha, active_idx)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h = h_new
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    # Stage 1: SLP optimization
    if time_left() > 80:
        h_cur, c5_s = slp_optimize(h_cur, n, ts, n_iters=40, active_threshold=0.85, max_lags=120)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_cur.copy()

    if time_left() > 80:
        h_cur, c5_s = slp_optimize(h_cur, n, ts, n_iters=40, active_threshold=0.92, max_lags=100)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_cur.copy()

    # Stage 2: Smooth max + SLSQP
    alphas = [200.0, 800.0, 2000.0, 5000.0]
    for alpha in alphas:
        if time_left() < 80:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        max_corr = np.max(corr)
        active = np.where(corr >= 0.97 * max_corr)[0]
        if len(active) > 80:
            active = active[np.argsort(corr[active])[-80:]]
        maxiter = max(100, min(400, int(time_left() * 0.3)))
        h_opt = slsqp_smooth(h_cur, n, ts, alpha, active, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    # Stage 3: SLP with tighter threshold
    if time_left() > 80:
        h_cur, c5_s = slp_optimize(h_cur, n, ts, n_iters=30, active_threshold=0.96, max_lags=80)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_cur.copy()

    # Stage 4: Projected subgradient for fine-tuning
    if time_left() > 40:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        max_corr = np.max(corr)
        active = np.where(corr >= 0.99 * max_corr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        n_pg = min(3000, int(time_left() * 3))
        h_pg, c5_pg = projected_subgradient(h_cur, n, ts, n_pg, 0.03, 500.0, active)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
        h_cur = h_pg

    # Stage 5: Try higher resolution
    n_hi = n * 2
    if time_left() > 250 and n_hi <= 1440:
        ts_hi = n_hi / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_cur)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

        if time_left() > 120:
            h_hi, c5_hi_s = slp_optimize(h_hi, n_hi, ts_hi, n_iters=25, active_threshold=0.88, max_lags=150)
            if c5_hi_s < best_c5:
                best_c5 = c5_hi_s
                best_h = h_hi.copy()
                best_n = n_hi

        if time_left() > 100:
            corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode="full")
            max_corr_hi = np.max(corr_hi)
            active_hi = np.where(corr_hi >= 0.96 * max_corr_hi)[0]

            if c5_hi_opt < best_c5:

            h_hi = h_hi_opt

        if time_left() > 60:

    # Stage 6: Final polish at best resolution
    n_f = best_n

    if time_left() > 60:

        if c5_f_opt < best_c5:

    if time_left() > 40:

                                         step_sizes=[-0.003, -0.001, -0.0003, 0.0003, 0.001, 0.003])
