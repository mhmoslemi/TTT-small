**New idea: Coordinate descent on the binary representation**

For a binary h, the objective c_k = number of (i, i+k) pairs where h[i]=1, h[i+k]=0.
If we flip h[i] from 1 to 0 and h[j] from 0 to 1:

This allows very fast evaluation of the effect of a swap, enabling much faster local search.

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)

            count = np.sum(mask)

def grad_for_lag(h, k, n):

    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_test = compute_c5(h)
            if c5_test < best_c5 - 1e-14:
                best_c5 = c5_test
                best_swap = (i, j)
            else:
                h[i] = 1.0
                h[j] = 0.0
        if best_swap is not None:
            i, j = best_swap
            c5_cur = best_c5
            improved = True
        if not improved:
            break
    return h, c5_cur

def multi_restart_binary(h0, n, ts, rng, n_restarts=5, max_rounds=150):
    best_h = h0.copy()
    best_c5 = float('inf')
    for r in range(n_restarts):
        h_start = h0.copy()
        n_perturb = max(1, n // 20)
        flip_idx = rng.choice(n, size=n_perturb, replace=False)
        for idx in flip_idx:
            h_start[idx] = 1.0 - h_start[idx]
        h_start = project_to_sum(h_start, ts)
        h_bin, c5_bin = binary_local_search(h_start, n, ts, rng, max_rounds=max_rounds)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    return best_h, best_c5

def top_sum_obj_grad(h, k_top, alpha):
    n = len(h)
    corr = compute_corr_array(h)
    sorted_idx = np.argsort(corr)[::-1][:k_top]
    top_vals = corr[sorted_idx]
    m = np.max(top_vals)
    exp_v = np.exp(alpha * (top_vals - m))
    s = np.sum(exp_v)
    w = exp_v / s
    val = m + np.log(s) / alpha
    grad = np.zeros(n)
    for ii, k in enumerate(sorted_idx):
        grad += w[ii] * grad_for_lag(h, k, n)
    return val, grad

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    ts = best_n / 2.0

    # Phase 1: Continuation optimization at current resolution
    if remaining() > 200:
        h_opt = optimize_continuation(best_h, best_n, ts,
                                      alphas=[20, 100, 500, 2000, 10000],
                                      maxiter=150, top_k=4)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: Subgradient polish
    if remaining() > 150:
        h_pol, c5_pol = subgradient_polish(best_h, best_n, ts, max_rounds=300, top_k=5)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 3: Binary local search with restarts
    if remaining() > 300:
        h_bin, c5_bin = multi_restart_binary(best_h, best_n, ts, rng,
                                             n_restarts=5, max_rounds=200)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # Continuous refinement from best binary
        if remaining() > 150:
            h_ref = best_h.copy()
            for a in [100, 1000, 10000]:
                if remaining() < 40:
                    break
                h_opt = optimize_continuation(h_ref, best_n, ts, [a], maxiter=200, top_k=4)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    h_ref = h_opt
            if remaining() > 60:
                h_pol, c5_pol = subgradient_polish(best_h, best_n, ts, max_rounds=200, top_k=5)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 4: Diversified starts
    if remaining() > 300:
        n = best_n
        x = np.linspace(0, 2, n, endpoint=False)
        starts = [best_h.copy()]
        for freq in [1, 2, 3, 4, 5, 6, 8]:
            for amp in [0.35, 0.45]:
                for phase in [0, np.pi / 3, 2 * np.pi / 3, np.pi]:
                    s = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
                    starts.append(project_to_sum(s, ts))
        for nblks in [4, 6, 8, 10, 12, 16, 20]:
            blk = n // nblks
            if blk < 1:
                continue
            hb = np.zeros(n)
            for ii in range(0, n, 2 * blk):
                hb[ii:min(ii + blk, n)] = 1.0
            starts.append(project_to_sum(hb, ts))

        for h_s in starts:
            if remaining() < 50:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                continue
            h_opt = optimize_continuation(h_s, n, ts, [200, 5000], maxiter=100, top_k=4)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        # Binary search from diversified best
        if remaining() > 100:
            h_bin, c5_bin = multi_restart_binary(best_h, n, ts, rng,
                                                 n_restarts=3, max_rounds=150)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                if remaining() > 60:
                    h_pol, c5_pol = subgradient_polish(best_h, n, ts, max_rounds=200, top_k=5)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol

    # Phase 5: Try different resolutions
    if remaining() > 250:
        for n_try in [120, 80, 60, 160, 180, 320]:
            if remaining() < 80:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_down = np.interp(x_new, x_old, best_h, period=2.0)
            h_down = project_to_sum(h_down, ts_try)

            if remaining() > 60:
                h_opt = optimize_continuation(h_down, n_try, ts_try,
                                              [50, 500, 5000], maxiter=100, top_k=4)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n_try
                    ts = ts_try
                    h_down = h_opt

            if remaining() > 50:
                h_bin, c5_bin = binary_local_search(h_down, n_try, ts_try, rng, max_rounds=100)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()
                    best_n = n_try
                    ts = ts_try

    # Phase 6: Higher resolution
    if remaining() > 200:
        n2 = min(best_n * 2, 480)
        if n2 > best_n:
            ts2 = n2 / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = project_to_sum(h_interp, ts2)
            h_opt = optimize_continuation(h_interp, n2, ts2,
                                          [50, 500, 5000], maxiter=100, top_k=4)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
                ts = ts2
            if remaining() > 60:
                h_pol, c5_pol = subgradient_polish(best_h, best_n, ts, max_rounds=200, top_k=5)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 7: Aggressive final optimization
    if remaining() > 120:
        n_f = best_n
        ts_f = n_f / 2.0
        h_opt = optimize_continuation(best_h, n_f, ts_f,
                                      [5000, 20000, 50000], maxiter=200, top_k=3)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 8: Final subgradient polish
    if remaining() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        h_pol, c5_pol = subgradient_polish(best_h, n_f, ts_f, max_rounds=300, top_k=7)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 9: One more binary attempt
    if remaining() > 60:
        h_bin, c5_bin = binary_local_search(best_h, best_n, best_n / 2.0, rng, max_rounds=100)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            if remaining() > 40:
                h_pol, c5_pol = subgradient_polish(best_h, best_n, best_n / 2.0,
                                                   max_rounds=150, top_k=5)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
