def smooth_max_and_grad(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    # Gradient computation
    nj = np.arange(n)[:, None]
    nm = np.arange(2*n-1)[None, :]
    d = nm - (n-1)  # d[k] = k - (n-1)

    # j - d_k = nj - (nm - (n-1)) = nj - nm + (n-1)
    idx1 = nj - nm + (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    
    # term2[j] = sum_k w[k]*dx*h[j+d_k] where 0<=j+d_k<n
    # j + d_k = nj + (nm - (n-1)) = nj + nm - (n-1)
    idx2 = nj + nm - (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    grad = np.sum((h1 - h2) * (w * dx)[None, :], axis=1)
    return obj, grad

    def f(x):

Actually, a better approach: use the fact that for a fixed d, the operation is just a shifted vector. So:

grad[j] = sum_d w[d] * dx * [(1-h[j-d]) if 0≤j-d<n else 0 - h[j+d] if 0≤j+d<n else 0]

I can compute this as:
- A = sum_d w[d] * (1-h)[j-d] * 1[0≤j-d<n]  -- this is a "correlation" of w with (1-h)
- B = sum_d w[d] * h[j+d] * 1[0≤j+d<n]  -- this is a "cross-correlation"

Let me use FFT for this:
- A[j] = sum_d w[d] * (1-h)[j-d] where 0≤j-d<n
  Let a = 1-h (padded to length 2n), b = w (padded to length 2n)
  Then A is the circular cross-correlation at the right offsets...

Actually, this is getting complicated. Let me just use the direct approach but precompute the index arrays.

For a given n, the index arrays idx1 and idx2 are fixed. I can precompute them once and reuse.

```python
class GradComputer:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        nj = np.arange(n)[:, None]
        nm = np.arange(2*n-1)[None, :]
        self.idx1 = np.clip(nj - nm + (n-1), 0, n-1)
        self.valid1 = (nj - nm + (n-1) >= 0) & (nj - nm + (n-1) < n)
        self.idx2 = np.clip(nj + nm - (n-1), 0, n-1)
        self.valid2 = (nj + nm - (n-1) >= 0) & (nj + nm - (n-1) < n)
    
    def compute(self, h, alpha):
        corr = np.correlate(h, 1.0 - h, mode="full") * self.dx
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_sh = np.max(shifted)
        exp_v = np.exp(shifted - max_sh)
        w = exp_v / np.sum(exp_v)
        obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
        h1 = self.valid1 * (1.0 - h[self.idx1])
        h2 = self.valid2 * h[self.idx2]
        grad = np.sum((h1 - h2) * (w * self.dx)[None, :], axis=1)
        return obj, grad
```

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

class SmoothMaxGrader:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        nj = np.arange(n)[:, None]
        nm = np.arange(2 * n - 1)[None, :]
        raw1 = nj - nm + (n - 1)
        self.idx1 = np.clip(raw1, 0, n - 1)
        self.valid1 = (raw1 >= 0) & (raw1 < n)
        raw2 = nj + nm - (n - 1)
        self.idx2 = np.clip(raw2, 0, n - 1)
        self.valid2 = (raw2 >= 0) & (raw2 < n)
    
    def f_grad(self, h, alpha):
        corr = np.correlate(h, 1.0 - h, mode="full") * self.dx
        m = np.max(corr)
        shifted = alpha * (corr - m)
        max_sh = np.max(shifted)

        h2 = self.valid2 * h[self.idx2]

        return obj, grad

def slsqp_opt(h0, grader, alpha, maxiter=1000):
    n = grader.n
    ts = n / 2.0
    def f(x):
        return grader.f_grad(x, alpha)[0]
    def g(x):
        return grader.f_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_sum(h_new, n_new / 2.0)

def stochastic_cd(h, n, rng, n_steps=50000, delta_lo=0.01, delta_hi=0.3, temp_init=0.1):
    ts = n / 2.0
    c5_cur = compute_c5(h)
    temp = temp_init
    best_local = c5_cur
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-6:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_local:
                best_local = c5_new
        else:
            if rng.random() < np.exp(-(c5_new - c5_cur) / max(temp, 1e-8)):
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-8, temp_init * np.exp(-step / (n_steps / 3.0)))
    return h, c5_cur

    H = np.fft.rfft(h - mean_h)

    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

Alternative: Don't precompute the full matrix. Instead, compute the gradient using a loop over the unique lags:

For each unique lag d (from -(n-1) to n-1):
- The contribution to grad from lag d is: w[d] * dx * [(1-h[valid range]) - h[valid range]]

This is a loop of 2n-1 iterations, each O(n). Total: O(n²) but with low memory.

Actually, even better: I can compute the gradient using two "shifted sums":

grad[j] = sum_d w[d] * dx * (1-h[j-d]) * 1[0≤j-d<n] - sum_d w[d] * dx * h[j+d] * 1[0≤j+d<n]

First sum: For a fixed d, (1-h[j-d]) is just (1-h) shifted right by d. The validity condition 0≤j-d<n means j is in [d, d+n-1] (intersected with [0,n)).

So the first sum is:
A[j] = sum_{d: 0≤j-d<n} w[d] * (1-h)[j-d]
     = sum_{d: max(0,d)≤j-d<n} ... 

Let me think of it as a convolution. Define:
a = 1-h (length n, zero-padded to length 2n-1)
b = w (length 2n-1)

Then A = cross-correlation of a and b, but only the "valid" part.

Actually, I think the simplest memory-efficient approach is:

```python
def compute_grad(self, h, w):
    n = self.n
    dx = self.dx
    # For each lag d (indexed by k in 0..2n-2, d = k-(n-1)):
    # Contribution to grad[j] from lag k:
    #   dx*w[k] * (1-h[j-d_k]) if 0<=j-d_k<n
    #   -dx*w[k] * h[j+d_k] if 0<=j+d_k<n
    
    # Efficient computation using shifted arrays:
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if abs(w[k]) < 1e-15:
            continue
        # First term: (1-h)[j-d] valid where 0<=j-d<n, i.e., d<=j<d+n
        lo = max(0, d)
        hi = min(n, d + n)
        if lo < hi:
            # j ranges from lo to hi-1, and j-d ranges from lo-d to hi-1-d (which is 0 to hi-1-d-1+... )
            # (1-h)[j-d] for j in [lo, hi-1] is one_minus_h[lo-d : hi-1-d+1] = one_minus_h[lo-d : hi-d]
            grad[lo:hi] += dx * w[k] * one_minus_h[lo-d:hi-d]
        # Second term: -h[j+d] valid where 0<=j+d<n, i.e., -d<=j<n-d
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= dx * w[k] * h[lo2+d:hi2+d]
    return grad
```

This is O(n * 2n) = O(n²) but only uses O(n) extra memory. For n=1920, that's about 3839 iterations, each doing O(n) work. Total: ~7.3M operations. Should take about 10-20ms in numpy.

But the Python loop overhead for 3839 iterations might add up. Let me benchmark mentally: each iteration does a few numpy array operations on slices of length ~n. The overhead per iteration is maybe 1-2μs for the Python loop + ~5μs for the numpy ops. Total: ~7.6μs * 3839 ≈ 29ms. That's acceptable.

Actually, I can vectorize over the lags that have significant weight. In practice, for a large alpha, only a few lags have significant weight (the ones near the argmax). So I can skip lags with w[k] < threshold.

```python
def compute_grad_efficient(self, h, w, threshold=1e-10):
    n = self.n
    dx = self.dx
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    active = np.where(w > threshold)[0]
    for k in active:
        d = k - (n - 1)
        wdx = dx * w[k]
        lo = max(0, d)
        hi = min(n, d + n)
        if lo < hi:
            grad[lo:hi] += wdx * one_minus_h[lo-d:hi-d]
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wdx * h[lo2+d:hi2+d]
    return grad
```

Let me also add one more novel idea: **coordinate descent on the lags**

Instead of optimizing h directly, I can think of the problem as:
- Given h, the max is at some lag k*
- To reduce the max at k*, I should modify h to reduce corr[k*]
- The derivative tells me which modifications are most effective
- But I also need to ensure I don't increase other lags too much

This is essentially what the smooth max gradient already does, but I can be more aggressive: focus only on the top M lags and try to equalize them.

**Equalization by Newton step:**
If the top M lags are k₁, ..., k_M with values c₁, ..., c_M, I want to find a perturbation δh such that:
c_i(δh) ≈ C for all i (all equal)
where C = mean(c₁, ..., c_M)

This is a system of M equations in n unknowns (overdetermined if M < n, which is typical). The least-squares solution is:
δh = -J^T (JJ^T)^{-1} (c - C*1)

where J is the M×n Jacobian with J[i,j] = ∂corr[k_i]/∂h[j].

This is a Newton-like step that equalizes the active lags. I can solve this with numpy.

```python
def equalization_step(h, n, M=5, max_cmds=100, step_scale=0.1):
    """Try to equalize the top M correlation values."""
    ts = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_M = np.argsort(corr)[-M:]  # indices of top M lags
    C_target = np.mean(corr[top_M])
    
    for cmd in range(max_cmds):
        # Compute Jacobian for the active lags
        # J[i, j] = ∂corr[k_i]/∂h[j] * dx
        J = np.zeros((M, n))
        for i, k in enumerate(top_M):
            d_val = k - (n - 1)
            one_minus_h = 1.0 - h
            # ∂corr[k]/∂h[j] = (1-h[j-d])*1[0≤j-d<n] - h[j+d]*1[0≤j+d<n] (times dx)
            lo = max(0, d_val)
            hi = min(n, d_val + n)
            if lo < hi:
                J[i, lo:hi] += one_minus_h[lo-d_val:hi-d_val] * dx
            lo2 = max(0, -d_val)
            hi2 = min(n, n - d_val)
            if lo2 < hi2:
                J[i, lo2:hi2] -= h[lo2+d_val:hi2+d_val] * dx
        
        # Newton step: solve J δh = -(corr[top_M] - C_target)
        residual = C_target - corr[top_M]
        # Least squares
        try:
            delta_h, _, _, _ = np.linalg.lstsq(J, residual, rcond=None)
        except:
            break
        
        # Scale and apply
        scale = step_scale / (np.max(np.abs(delta_h)) + 1e-10)
        h_new = project_to_sum(h + scale * delta_h, ts)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        
        if c5_new < c5_old:
            h = h_new
            # Update top M
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            top_M = np.argsort(corr)[-M:]
            C_target = np.mean(corr[top_M])
            step_scale *= 0.9
        else:
            step_scale *= 0.5
            if step_scale < 1e-6:
                break
    
    return h, compute_c5(h)
```
