import numpy as np
import time

# Core functions (all top-level, no closures)

def _fft_corr(a, b):
    n = len(a)
    size = 2*n - 1
    fs = 1
    while fs < size:
        fs *= 2
    return np.fft.irfft(np.fft.rfft(a, fs) * np.fft.rfft(b[::-1], fs), fs)[:size]

def _c5(h):
    return float(np.max(_fft_corr(h, 1.0-h)) * 2.0/len(h))

def _proj(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-14: break
        m = (h > 1e-12) & (h < 1.0-1e-12)
        if m.any(): h[m] += d/m.sum()
        else: h += d/len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_topk(h, corr, k):
    n = len(h)
    idx = np.argsort(corr)[-k:]
    g = np.zeros(n)
    for m in idx:
        s = m - n + 1
        # (1-h)[i+s] for valid i
        lo1 = max(0, -s); hi1 = min(n, n-s)
        if lo1 < hi1: g[lo1:hi1] += 1.0 - h[lo1+s:hi1+s]
        # -h[i-s] for valid i
        lo2 = max(0, s); hi2 = min(n, n+s)
        if lo2 < hi2: g[lo2:hi2] -= h[lo2-s:hi2-s]
    return g / k

def _nesterov(h0, n, ts, iters, lr0, topk, mom):
    h = h0.copy(); v = np.zeros(n)
    bc = _c5(h); bh = h.copy(); lr = lr0
    for it in range(iters):
        corr = _fft_corr(h, 1.0-h)
        g = _grad_topk(h, corr, topk)
        v = mom*v - lr*g
        h = _proj(h + v, ts)
        c = _c5(h)
        if c < bc: bc = c; bh = h.copy()
        if (it+1) % 200 == 0: lr *= 0.99
    return bh, bc

def _coord(h0, n, ts, rounds, rng, step):
    h = h0.copy(); c0 = _c5(h)
    best = h.copy(); bc = c0
    for r in range(rounds):
        imp = False
        for i in rng.permutation(n):
            for d in (-step, step):
                old = h[i]
                h[i] = np.clip(old+d, 0, 1)
                h = _proj(h, ts)
                c = _c5(h)
                if c < bc - 1e-14:
                    bc = c; best = h.copy(); imp = True
                else:
                    h[i] = old
                    h = _proj(h, ts)
        if not imp: break
    return best, bc

def _run(h0, n, ts, budget_left, rng):

        results.append((c2, h2))

import numpy as np
import time


def _fft_corr(a, b):
    n = len(a)
    size = 2 * n - 1
    fs = 1
    while fs < size:
        fs *= 2
    return np.fft.irfft(np.fft.rfft(a, fs) * np.fft.rfft(b[::-1], fs), fs)[:size]


def _c5(h):
    n = len(h)
    corr = _fft_corr(h, 1.0 - h)
    return float(np.max(corr) * 2.0 / n)


def _proj(h, ts):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _grad_topk(h, corr, k):
    n = len(h)
    idx = np.argsort(corr)[-k:]
    g = np.zeros(n)
    for m in idx:
        s = m - n + 1
        lo1 = max(0, s)
        hi1 = min(n - 1, n - 1 + s)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += (1.0 - h[lo1 - s:hi1 + 1 - s])
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= h[lo2 + s:hi2 + 1 + s]
    return g / k


def _nesterov(h0, n, ts, iters, lr0, topk, mom, time_limit):
    h = h0.copy()
    v = np.zeros(n)
    bc = _c5(h)
    bh = h.copy()
    lr = lr0
    t_start = time.time()
    for it in range(iters):
        if time.time() - t_start > time_limit:
            break
        corr = _fft_corr(h, 1.0 - h)
        g = _grad_topk(h, corr, topk)
        v = mom * v - lr * g
        h = _proj(h + v, ts)
        c = _c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
        if (it + 1) % 500 == 0:
            lr *= 0.97
        elif (it + 1) % 2000 == 0:
            lr *= 0.90
            v *= 0.5
    return bh, bc


def _adam_topk(h0, n, ts, iters, lr0, topk, time_limit):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    bc = _c5(h)
    bh = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    t_start = time.time()
    for it in range(iters):
        if time.time() - t_start > time_limit:
            break
        t += 1
        corr = _fft_corr(h, 1.0 - h)
        g = _grad_topk(h, corr, topk)
        m_vec = b1 * m_vec + (1 - b1) * g
        v_vec = b2 * v_vec + (1 - b2) * g * g
        m_hat = m_vec / (1 - b1 ** t)
        v_hat = v_vec / (1 - b2 ** t)
        lr = lr0 / (1 + 0.001 * t)
        h = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), ts)
        c = _c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
    return bh, bc


def _coord(h0, n, ts, rounds, rng, step):
    h = h0.copy()
    bc = _c5(h)
    best = h.copy()
    for r in range(rounds):
        imp = False
        for i in rng.permutation(n):
            for d in (-step, step):
                old = h[i]
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h = _proj(h, ts)
                c = _c5(h)
                if c < bc - 1e-14:
                    bc = c
                    best = h.copy()
                    imp = True
                else:
                    h[i] = old
                    h = _proj(h, ts)
        if not imp:
            break
    return best, bc


def _binary_snap(h, n, ts):
    hb = (h > 0.5).astype(float)
    hb = _proj(hb, ts)
    return hb, _c5(hb)

        best_h, _ = _coord(best_h, best_n, ts, 30, rng, 0.002)
        best_c = _c5(best_h)

    best_h = _proj(best_h, best_n / 2.0)
    best_c = _c5(best_h)
    return (best_h, best_c, best_n)
