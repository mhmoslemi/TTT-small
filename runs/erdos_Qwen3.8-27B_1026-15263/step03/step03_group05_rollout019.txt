```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad
```

For Frank-Wolfe:

- Minimize over feasible set: s* = argmin {<g, s> : 0 ≤ s ≤ 1, sum(s) = n/2}
- The solution to this linear program is: sort components of g, set s_i = 1 for the smallest n/2 components, 0 for the rest (adjusting for ties).
- Update: h_new = (1-t)*h + t*s* where t is a step size (can use backtracking or 2/(t+2))

```python
def frank_wolfe_step(h, grad, n, target_sum):
    """Solve min <grad, s> s.t. 0 <= s <= 1, sum(s) = target_sum"""
    # The LP solution: sort by grad, take smallest target_sum components as 1
    order = np.argsort(grad)
    s_star = np.zeros(n)
    s_star[order[:int(target_sum)]] = 1.0
    return s_star

def frank_wolfe_optimize(h0, alpha, n_iter, time_fn):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    best_obj = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time_left() < 5:
            break
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        s_star = frank_wolfe_step(h, grad, n, target_sum)
        # Step size: use 2/(it+2) or backtracking
        t = 2.0 / (it + 2.0)
        h_new = (1.0 - t) * h + t * s_star
        # Project to ensure exact sum
        h_new = project_box_sum(h_new, target_sum)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_obj
```

Let me also think about a completely different approach: **direct optimization of the max using subgradient methods**.

The subgradient of f(h) = max_k g_k(h) at h is g_{k*}(h) where k* = argmax_k g_k(h). If there are multiple maximizers, the subdifferential is the convex hull of their gradients.

A subgradient method would be:
h_{t+1} = project(h_t - η_t * ∇g_{k*_t}(h_t))

where k*_t is (one of) the maximizing lags at iteration t.

This is simpler than the smooth max approach and might converge to the same solution. The advantage is that it doesn't require tuning alpha.

The disadvantage is that it only uses one lag at a time, which might be slow if there are many "active" lags.

A compromise: use the top-m lags and average their gradients. This is the "shrunken subgradient" or "bundle" approach.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = (m_val + max_shift + np.log(s) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad, corr, w

def projected_gradient_step(h, grad, lr, target_sum):
    h_new = h - lr * grad
    return project_box_sum(h_new, target_sum)
