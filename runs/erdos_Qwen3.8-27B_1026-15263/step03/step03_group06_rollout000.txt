    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask_lo = h < 1.0 - 1e-10
        mask_hi = h > 1e-10
        mask = mask_lo & mask_hi
        n_mask = np.sum(mask)
        if n_mask > 0:
            max_adj = np.minimum(1.0 - h[mask], h[mask]).max()
            if abs(diff) / n_mask < max_adj:
                h[mask] += diff / n_mask
            else:
                h[mask] += np.sign(diff) * 0.5 * max_adj
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    m = np.max(corr)

    shifted = np.clip(shifted, -700.0, 700.0)

    s = np.sum(exp_vals)

    h = h0.copy()

    v_v = np.zeros(n)
    t = 0
    h_best = h.copy()
    c5_best = compute_c5(h)
    stall_count = 0
    
    for it in range(n_iters):
        t += 1
        if t % check_interval != 0:
            sm_val, g = smooth_max_and_grad(h, alpha)
        else:
            sm_val, g = smooth_max_and_grad(h, alpha)

        v_v = beta2 * v_v + (1.0 - beta2) * (g * g)
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_mask = np.sum(mask)
        if n_mask > 0 and abs(diff) <= n_mask * 0.49:
            h[mask] += diff / n_mask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for offset in range(n):
        wk = w[n - 1 + offset]
        if wk < 1e-14:
            continue
        m_hi = n - 1 - offset
        g[0:m_hi+1] += wk * (1.0 - h[offset:n])
        g[offset:n] -= wk * h[0:m_hi+1]
    for pos in range(1, n):
        wk = w[n - 1 - pos]
        if wk < 1e-14:
            continue
        g[pos:n] += wk * (1.0 - h[0:n-pos])
        g[0:n-pos] -= wk * h[pos:n]
    return sm_val, g

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    h_best = h.copy()
    c5_best = compute_c5(h)
    stall = 0
    check_every = max(20, n // 30)
    for it in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_v = beta1 * m_v + (1.0 - beta1) * g
        v_v = beta2 * v_v + (1.0 - beta2) * (g * g)
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        if t % check_every == 0:
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    break
        h = h_new
    return h_best, c5_best

def coord_descent_fine(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01, 0.02])
    all_deltas = np.concatenate([deltas, -deltas])
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in all_deltas:
                bv = h[i] + d
                if bv < 0.0 or bv > 1.0:
                    continue
                h[i] = bv
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_val - 1e-13:
                    best_val = c5_new
                    best_d = d
                else:
                    h[i] = bv
                    h = project_to_sum(h, target_sum)
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

    elif idx == 2:

        return h2

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        ts = best_n / 2.0
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    ts = n / 2.0
    h_cur = best_h.copy()

    if time_left() > 120:
        for alpha in [30.0, 80.0, 150.0]:
            if time_left() < 40:
                break
            h_opt = slsqp_step(h_cur, n, ts, alpha, 300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cur = h_opt

    alpha_schedule = [50.0, 100.0, 200.0, 400.0, 800.0, 1600.0, 3200.0, 6400.0, 12800.0]
    for ai, alpha in enumerate(alpha_schedule):
        if time_left() < 50:
            break
        frac = time_left() / budget_s
        max_iters = int(8000 * frac * 0.4)
        lr = 0.03 if alpha < 100 else (0.015 if alpha < 500 else 0.008)
        h_adam, c5_adam = adam_optimize(h_cur, n, ts, alpha, max_iters, lr=lr)
        if c5_adam < best_c5:
            best_c5 = c5_adam
            best_h = h_adam.copy()
        h_cur = h_adam

    if time_left() > 150:
        for pi in range(12):
            if time_left() < 25:
                break
            h_pert = perturb_structured(best_h, n, rng, pi)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_to_sum(h_pert, ts)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
            if time_left() > 35:
                h_reopt, c5_reopt = adam_optimize(h_pert, n, ts, 500.0, 800, lr=0.01)
                c5_reopt = compute_c5(h_reopt)
                if c5_reopt < best_c5:
                    best_c5 = c5_reopt
                    best_h = h_reopt.copy()
                    h_cur = h_reopt

    if time_left() > 100:
        h2_opt, c5_2 = adam_optimize(best_h, n, ts, 2000.0, 2000, lr=0.005)
        c5_2 = compute_c5(h2_opt)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt.copy()

    if time_left() > 80:
        n_rounds = max(3, int(time_left() // 20))
        h_cd, c5_cd = coord_descent_fine(best_h, n, ts, n_rounds, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 40:
        for alpha in [3000.0, 8000.0, 15000.0]:
            if time_left() < 15:
                break
            h_pol = slsqp_step(best_h, n, ts, alpha, 400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    if time_left() > 20:
        h_cd2, c5_cd2 = coord_descent_fine(best_h, n, ts, 3, rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    best_h = project_to_sum(best_h, ts)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
