def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    """SLSQP with smooth max"""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_f(alpha, h, n)
        return -v  # minimize negative (i.e., maximize)
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_f(alpha, h, n)
        return -g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h0):
            return project_to_sum(res.x, target_sum)
    except Exception:
        pass
    return project_to_sum(h0, target_sum)

def anneal_perturb(h0, n, target_sum, T, n_pert):

    for _ in range(n_pert):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_corr(h):
    n = len(h)
    m = 1
    while m < 2 * n - 1:
        m *= 2
    H = np.fft.rfft(h, m)
    O = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * np.conj(O), m)[:2 * n - 1]
    return corr

def project_to_sum(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13 * n:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = n / 2.0 - corr
    m = np.max(ac)
    exponents = -alpha * (ac - m)
    exponents = np.clip(exponents, -700.0, 700.0)
    exp_vals = np.exp(exponents)
    s = np.sum(exp_vals)
    sm_val = m - np.log(s) / alpha
    w = exp_vals / s
    g = np.zeros(n)
    for l_idx in range(2 * n - 1):
        if w[l_idx] < 1e-13:
            continue
        l = l_idx - (n - 1)
        al = abs(l)
        if al == 0:
            g += w[l_idx] * 2.0 * h
        else:
            if l > 0:
                g[:n - l] += w[l_idx] * h[l:]
                g[l:] += w[l_idx] * h[:n - l]
            else:
                g[:n + l] += w[l_idx] * h[-l:]
                g[-l:] += w[l_idx] * h[:n + l]
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha, n)
        return -v
    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha, n)
        return -g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_res = project_to_sum(res.x, target_sum)
        if compute_c5(h_res) <= compute_c5(h0) + 1e-10:
            return h_res
    except Exception:
        pass
    return project_to_sum(h0, target_sum)

def subgradient_method(h0, n, target_sum, n_iters, lr_init, top_k=8, use_fft=True):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    v = np.zeros(n)
    mom = 0.85
    for it in range(n_iters):
        if use_fft:
            corr = fast_corr(h)
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
        ac = n / 2.0 - corr
        sorted_idx = np.argsort(ac)
        active = sorted_idx[:top_k]
        g = np.zeros(n)
        for l_idx in active:
            l = l_idx - (n - 1)
            al = abs(l)
            if al == 0:
                g += 2.0 * h
            else:
                if l > 0:
                    g[:n - l] += h[l:]
                    g[l:] += h[:n - l]
                else:
                    g[:n + l] += h[-l:]
                    g[-l:] += h[:n + l]
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        v = mom * v + lr * g
        vn = np.linalg.norm(v)
        if vn > 1.0:
            v = v / vn
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.005, 1.5)
        else:
            lr *= 0.94
            v *= 0.7
        h = h_new
    return best_h, best_c5

def nesterov_smooth(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        val_y, g_y = smooth_max_and_grad(y, alpha, n)
        v = 0.9 * v - lr * g_y
        vn = np.linalg.norm(v)
        if vn > 1.0:
            v = v / vn
        h_new = np.clip(y + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.008, 1.0)
        else:
            lr *= 0.92
            v *= 0.5
        beta = (1.0 + np.sqrt(1.0 + 4.0 * 0.81)) / 2.0
        y = h_new + ((beta - 1.0) / beta) * (h_new - y)
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target_sum)
        h = h_new
    return best_h, best_c5

def multi_start_optimize(h0, n, target_sum, rng, time_budget):
    t0 = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    perturbations = []
    for scale in [0.03, 0.05, 0.08, 0.12]:
        for _ in range(2):
            p = rng.normal(0, scale, n)
            perturbations.append(p)
    freq_starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for amp in [0.1, 0.2, 0.3]:
            for phase in [0.0, np.pi / 2, np.pi, 3 * np.pi / 2]:
                freq_starts.append(amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase))
    for pert in perturbations:
        if time.time() - t0 > time_budget * 0.4:
            break
        h_try = project_to_sum(h0 + pert, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-6:
            n_it = min(2000, 200 + int((time_budget - (time.time() - t0)) * 2))
            h_opt, c5_opt = subgradient_method(h_try, n, target_sum, n_it, 0.015, top_k=6)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    for fs in freq_starts[:6]:
        if time.time() - t0 > time_budget * 0.7:
            break
        h_try = project_to_sum(0.5 + fs, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 + 0.02:
            n_it = min(1500, 150 + int((time_budget - (time.time() - t0)) * 1.5))
            h_opt, c5_opt = subgradient_method(h_try, n, target_sum, n_it, 0.01, top_k=8)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
