from scipy.optimize import minimize
import time

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            # Need boundary adjustment
            if diff > 0:
                # Need to increase some values
                # Pick values closest to 1 and increase them
                order = np.argsort(-h)
                idx = order[0]
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                # Need to decrease some values
                order = np.argsort(h)
                idx = order[0]
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def sa_with_bias(h_start, n, target_sum, t_end, rng, verbose=False):
    """SA with lag-biased moves"""
    h = h_start.copy()
    c5_curr, corr_curr = compute_c5_and_corr(h)
    best_c5 = c5_curr
    best_h = h.copy()
    
    T = 0.01
    min_T = 0.001
    no_improve = 0
    max_no_improve = 50000
    improve_count = 0
    
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    
    while time.time() < t_end:
        # Choose move type
        move_type = rng.random()
        
        if move_type < 0.5:
            # Random swap
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]

            offset = peak_idx - (n - 1)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

    # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])
    # d corr[k] / d h[i] = (1 - h[i+k-(n-1)]) - h[i-(k-(n-1))]

    #                     - h[i + k - (n-1)] - h[i - (k-(n-1))] (second term, product rule)
    # Wait, that's not right either. Let me be very careful.

    # where s(i,k) = i + k - (n-1) and both i and s(i,k) must be in [0, n-1]
    
    # d corr[k] / d h[j] = (1 - h[s(j,k)]) * 1_{j valid as first index}
    #                    + sum over i where s(i,k)=j of h[i] * (-1)
    #                    = (1 - h[s(j,k)]) * 1_{j valid as first index}
    #                    - h[j - (k-(n-1))] * 1_{j valid as second index}

    # For each k, the contribution to grad[j] from w[k] is:
    # w[k] * [(1 - h[s(j,k)]) if j is valid first index for lag k
    #         - h[j - offset_k] if j is valid second index for lag k]
    # where offset_k = k - (n-1)
    
    gradient = np.zeros(n)
    for k in range(2 * n - 1):

            # d corr[k]/d h[j] += -h[i]
            gradient[i] += w[k] * (1.0 - h[j])
            gradient[j] += w[k] * (-h[i])

One key optimization: instead of computing the full correlation every time (O(n²)), I can compute the change in correlation when I make a swap.

When I swap h[i] (was 1) and h[j] (was 0):
- The new h has h[i]=0, h[j]=1
- The change in corr[k] for each k:
  - corr[k] = sum_l h[l] * (1 - h[l + d_k]) where d_k = k - (n-1)
  - The change comes from:
    - h[i] changed from 1 to 0: removes contribution (1 - h[i + d_k]) from corr[k]
    - h[j] changed from 0 to 1: adds contribution (1 - h[j + d_k]) to corr[k]
    - h[i] appears in (1 - h[...]) when l + d_k = i, i.e., l = i - d_k: changes (1-h[l]) from (1-1)=0 to (1-0)=1, so adds h[l] to corr[k]... wait this is getting confusing.

Let me be very precise. Let's define:
corr[k] = sum_l h[l] * (1 - h[l + d]) where d = k - (n-1), and the sum is over valid l where both l and l+d are in [0, n-1].

After swapping h[i] (1→0) and h[j] (0→1):
new_h[i] = 0, new_h[j] = 1, all others unchanged.

new_corr[k] = sum_l new_h[l] * (1 - new_h[l + d])

The difference new_corr[k] - corr[k] comes from terms where l or l+d equals i or j.

Case 1: l = i
- Old: h[i] * (1 - h[i+d]) = 1 * (1 - h[i+d])
- New: 0 * (1 - new_h[i+d]) = 0
- Change: -(1 - h[i+d]) ... but wait, h[i+d] might also have changed if i+d = j
  - If i+d ≠ j: change = -(1 - h[i+d])
  - If i+d = j: old was 1*(1-0) = 1, new was 0*(1-1) = 0, change = -1

Case 2: l + d = i, i.e., l = i - d
- Old: h[i-d] * (1 - h[i]) = h[i-d] * (1-1) = 0
- New: new_h[i-d] * (1 - new_h[i]) = new_h[i-d] * (1-0) = new_h[i-d]
  - If i-d ≠ j: change = h[i-d]
  - If i-d = j: change = 1 (since new_h[j] = 1)

Case 3: l = j
- Old: h[j] * (1 - h[j+d]) = 0 * (1 - h[j+d]) = 0
- New: 1 * (1 - new_h[j+d]) = 1 - new_h[j+d]
  - If j+d ≠ i: change = 1 - h[j+d]
  - If j+d = i: change = 1 - 0 = 1

Case 4: l + d = j, i.e., l = j - d
- Old: h[j-d] * (1 - h[j]) = h[j-d] * (1-0) = h[j-d]
- New: new_h[j-d] * (1 - new_h[j]) = new_h[j-d] * (1-1) = 0
  - If j-d ≠ i: change = -h[j-d]
  - If j-d = i: change = -1

These four cases might overlap (e.g., if i+d=j and i-d=j, which would mean d=0, but d=0 means k=n-1 which is the middle lag where corr=0 for binary).

For efficiency, I can precompute the "delta" for a swap in O(1) per lag... but actually I need to check all lags, and for each lag there are at most 4 affected terms. So the total change computation is O(n) per swap (going through all n+1 possible d values... well, 2n-1 lags).

Actually, for each lag k (with offset d = k-(n-1)), the change is determined by whether i and j are "visible" at that lag. The visibility conditions are:
- i is a valid first index: 0 ≤ i < n and 0 ≤ i+d < n
- i is a valid second index: 0 ≤ i-d < n and 0 ≤ i < n (i.e., 0 ≤ i-d < n)
- j is a valid first index: 0 ≤ j < n and 0 ≤ j+d < n
- j is a valid second index: 0 ≤ j-d < n

For each of these, we can compute the contribution in O(1). So the total change for all lags is O(n) (since there are 2n-1 lags, and for each we do O(1) work).

But O(n) per swap is the same as O(n²) for the full correlation (since there are n positions and n lags). So we don't gain much unless we can compute only the max change.

Actually, we can! We don't need to compute all the new corr values - we just need to know if the max decreased. We can:
1. Keep track of the current corr array
2. For a proposed swap, compute the delta for each lag in O(1)
3. Update the corr array in O(n) (add delta to each element)
4. Find the new max in O(n)

Total: O(n) per step, which is 2x479 ≈ 1000 operations vs. 240*479 ≈ 115k for the full correlation. That's a 100x speedup!
