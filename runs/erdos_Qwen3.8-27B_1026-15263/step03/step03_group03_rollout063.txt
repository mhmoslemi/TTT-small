This is a 2D minimax problem. The feasible region (dj, dk) is a polygon (intersection of [0,1] constraints).

For a 2D minimax of linear functions over a polygon:
- The optimum is at a vertex of the arrangement of the level sets
- Practically, we can use a grid search + local refinement

2. Do a grid search over the polygon (10x10 or so)
3. Refine the best with a few bisection steps in each direction

The feasible region constraints:
- h[j] + dj ∈ [0, 1] → dj ∈ [-h[j], 1-h[j]]
- h[k] + dk ∈ [0, 1] → dk ∈ [-h[k], 1-h[k]]
- h[l] - dj - dk ∈ [0, 1] → dj + dk ∈ [h[l]-1, h[l]]

So the feasible region is the intersection of:
- Rectangle: dj ∈ [dj_lo, dj_hi], dk ∈ [dk_lo, dk_hi]
- Strip: dj + dk ∈ [sl_lo, sl_hi] where sl_lo = h[l]-1, sl_hi = h[l]

This is a convex polygon (at most a hexagon). I can parameterize it and do a grid search.

For the grid search: I'll sample the (dj, dk) space on a coarse grid within the polygon, evaluate the objective at each point, and then refine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12, do_triples=False, n_triple=5):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    
    # Triple moves for a subset of indices
    if do_triples:
        triple_indices = order[:n_triple]
        for j in triple_indices:
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            k = (j + 1 + rng.randint(n)) % n
            l = (j + 1 + rng.randint(n)) % n
            if l == j or l == k:
                continue
            dj, dk = triple_move_optimize(h, corr, j, k, l, n)
            if abs(dj) > 1e-10 or abs(dk) > 1e-10:
                dl = -dj - dk
                h[j] = np.clip(h[j] + dj, 0, 1)
                h[k] = np.clip(h[k] + dk, 0, 1)
                h[l] = np.clip(h[l] + dl, 0, 1)
                # Recompute corr
                corr = np.correlate(h, 1.0 - h, mode="full")
                improved = True
    
    return h, corr, improved

    B_k = compute_B(h, k, n)
    B_l = compute_B(h, l, n)
    A1 = B_j - B_l
    A2 = B_k - B_l
    
    dj_lo = max(-h[j], h[l] - 1.0 - (1.0 - h[k]))
    dj_hi = min(1.0 - h[j], h[l] + h[k])
    dk_lo = max(-h[k], h[l] - 1.0 - (1.0 - h[j]))
    dk_hi = min(1.0 - h[k], h[l] + h[j])
    
    if dj_hi - dj_lo < 1e-10 or dk_hi - dk_lo < 1e-10:
        return 0.0, 0.0
    
    n_grid = 6
    dj_vals = np.linspace(dj_lo, dj_hi, n_grid)
    current_max = np.max(corr)
    best_val = current_max
    best_dj = 0.0
    best_dk = 0.0
    
    for dj in dj_vals:
        dk_lo_j = max(dk_lo, h[l] - 1.0 - dj)
        dk_hi_j = min(dk_hi, h[l] - dj)
        if dk_hi_j - dk_lo_j < 1e-10:
            continue
        dk_grid = np.linspace(dk_lo_j, dk_hi_j, n_grid)
        vals = np.max(corr[None, :] + dj * A1[None, :] + dk_grid[:, None] * A2[None, :], axis=1)
        min_idx = np.argmin(vals)

            best_dj = dj
            best_dk = dk_grid[min_idx]

    return 0.0, 0.0

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w_sum = np.sum(exp_vals)
    w = exp_vals / w_sum
    obj = m_val + np.log(w_sum) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        B_j = compute_B(h, j, n)
        grad[j] = np.dot(w, B_j)
    
    return obj, grad

def gradient_descent_phase(h0, n, target_sum, alpha, n_steps, lr_init, momentum, t_deadline):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for step in range(n_steps):
        if time.time() >= t_deadline:
            break
        obj, grad = smooth_max_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h = project_to_sum(h_new, target_sum)
        # Adaptive lr
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        else:
            lr *= 0.95
            v *= 0.5
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def compute_B_all(h, n):
    B_all = np.zeros((n, 2 * n - 1))
    for j in range(n):
        B_all[j] = compute_B(h, j, n)
    return B_all

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def solve_2d(corr, A1, A2, dj_lo, dj_hi, dk_lo, dk_hi, h_l):
    n_grid = 6
    current_max = np.max(corr)
    best_val = current_max
    best_dj = 0.0
    best_dk = 0.0
    dj_vals = np.linspace(dj_lo, dj_hi, n_grid)
    for dj in dj_vals:
        dk_lo_j = max(dk_lo, h_l - 1.0 - dj)
        dk_hi_j = min(dk_hi, h_l - dj)
        if dk_hi_j - dk_lo_j < 1e-12:
            continue
        dk_grid = np.linspace(dk_lo_j, dk_hi_j, n_grid)
        vals = np.max(corr[None, :] + dj * A1[None, :] + dk_grid[:, None] * A2[None, :], axis=1)
        min_idx = int(np.argmin(vals))
        if vals[min_idx] < best_val - 1e-12:
            best_val = vals[min_idx]
            best_dj = dj
            best_dk = dk_grid[min_idx]
    if best_val < current_max - 1e-12:
        return best_dj, best_dk
    return 0.0, 0.0

def pair_descent_sweep(h, corr, B_all, rng, n_cand=12, do_triples=False, n_triple=8):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False

    for idx in range(n):
        j = order[idx]
        B_j = B_all[j]
        current_max = np.max(corr)
        best_val = current_max
        best_delta = 0.0
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = B_all[k]
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_k = k

        if best_k >= 0 and abs(best_delta) > 1e-13:
            D = B_j - B_all[best_k]
            corr += best_delta * D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            B_all[j] = compute_B(h, j, n)
            B_all[best_k] = compute_B(h, best_k, n)
            improved = True

    if do_triples:
        triple_candidates = order[:n_triple]
        for j in triple_candidates:
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            k = (j + 1 + rng.randint(n - 1)) % n
            l_choices = [rng.randint(n) for _ in range(3)]
            for l in l_choices:
                if l == j or l == k or h[l] <= 1e-6 or h[l] >= 1.0 - 1e-6:
                    continue
                A1 = B_all[j] - B_all[l]
                A2 = B_all[k] - B_all[l]
                dj_lo = max(-h[j], h[l] - 1.0 - (1.0 - h[k]))
                dj_hi = min(1.0 - h[j], h[l] + h[k])
                dk_lo = max(-h[k], h[l] - 1.0 - (1.0 - h[j]))
                dk_hi = min(1.0 - h[k], h[l] + h[j])
                if dj_hi - dj_lo < 1e-10 or dk_hi - dk_lo < 1e-10:
                    continue
                dj, dk = solve_2d(corr, A1, A2, dj_lo, dj_hi, dk_lo, dk_hi, h[l])
                if abs(dj) > 1e-10 or abs(dk) > 1e-10:
                    dl = -dj - dk
                    h[j] = np.clip(h[j] + dj, 0.0, 1.0)
                    h[k] = np.clip(h[k] + dk, 0.0, 1.0)
                    h[l] = np.clip(h[l] + dl, 0.0, 1.0)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    B_all[j] = compute_B(h, j, n)
                    B_all[k] = compute_B(h, k, n)
                    B_all[l] = compute_B(h, l, n)
                    improved = True
                    break

    return h, corr, B_all, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w_sum = np.sum(exp_vals)
    w = exp_vals / w_sum
    obj = m_val + np.log(w_sum) / alpha
    grad = np.zeros(n)
    for j in range(n):
        B_j = compute_B(h, j, n)
        grad[j] = np.dot(w, B_j)
    return obj, grad

def gradient_descent_phase(h0, n, target_sum, alpha, max_steps, t_deadline):
    h = h0.copy()
    v = np.zeros(n)
    lr = 1.0 / (alpha * 3.0)
    momentum = 0.85
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stalled = 0

    for step in range(max_steps):
        if time.time() >= t_deadline:
            break
        obj, grad = smooth_max_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
            lr = min(lr * 1.05, 5.0 / (alpha * n))
        else:
            h = h_new
            stalled += 1
            lr *= 0.9
            v *= 0.3
            if stalled > 200:
                break

    return best_h, best_c5

def binary_sa_phase(h_init, n, target_sum, rng, t_deadline):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    h = project_to_sum(h, target_sum)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-6:
        order = np.argsort(np.abs(h - 0.5))
        for idx in range(abs(int(diff))):
            pos = order[idx]
            if diff > 0:
                h[pos] = 1.0
            else:
                h[pos] = 0.0
            h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()

    ones_idx = np.where(h == 1.0)[0].tolist()
    zeros_idx = np.where(h == 0.0)[0].tolist()

    t_sa = 0.5
    cool = 0.9995
    steps = 0
    while time.time() < t_deadline:
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        B_i = compute_B(h, i, n)
        B_j = compute_B(h, j, n)
        D = B_j - B_i
        new_max = np.max(corr + D)
        old_max = np.max(corr)
        delta = new_max - old_max
        if delta < 0 or (t_sa > 1e-8 and rng.random() < np.exp(-delta / t_sa)):
            corr += D
            h[i] = 0.0
            h[j] = 1.0
            ones_idx.remove(i)
            zeros_idx.remove(j)
            ones_idx.append(j)
            zeros_idx.append(i)
            c5_now = float(np.max(corr) * (2.0 / n))
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        t_sa *= cool
        steps += 1

    return best_h, best_c5
```
