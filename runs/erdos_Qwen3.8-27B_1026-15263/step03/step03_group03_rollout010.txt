    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    idx1 = j + np.arange(2*n-1) - nm1
    v1 = (idx1 >= 0) & (idx1 < n)
    B[v1] += 1.0 - h[idx1[v1]]
    idx2 = j - np.arange(2*n-1) + nm1
    v2 = (idx2 >= 0) & (idx2 < n)
    B[v2] += -h[idx2[v2]]
    return B

def solve_1d_minimax(corr, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    # Grid search to find approximate best
    n_grid = 20
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == n_grid - 1:
        return hi
    lo_r = t_grid[max(0, best_idx - 2)]
    hi_r = t_grid[min(n_grid - 1, best_idx + 2)]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    idx1 = j + np.arange(2 * n - 1) - nm1
    v1 = (idx1 >= 0) & (idx1 < n)
    B[v1] += 1.0 - h[idx1[v1]]
    idx2 = j - np.arange(2 * n - 1) + nm1
    v2 = (idx2 >= 0) & (idx2 < n)
    B[v2] += -h[idx2[v2]]
    return B

def solve_1d_minimax(corr, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    n_grid = 16
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == n_grid - 1:
        return hi
    lo_r = t_grid[max(0, best_idx - 2)]
    hi_r = t_grid[min(n_grid - 1, best_idx + 2)]
    for _ in range(36):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=20):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = float(np.max(corr))
        best_delta = 0.0
        best_D = None
        best_k = -1
        n_h = n // 2
        n_t = n // 3
        cands = [(j+1)%n, (j-1+n)%n, (j+n_t)%n, (j+2*n_t)%n,
                 (j+n_h)%n, (j+7)%n, (j-7+n)%n, (j+n_h//2)%n]
        while len(cands) < n_cand:
            cands.append(rng.randint(n))
        seen = set()
        for k in cands:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d_minimax(corr, D, delta_lo, delta_hi)
            val_opt = float(np.max(corr + D * delta_opt))
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def global_best_pair(h, corr, n, rng, n_trials=200):
    best_val = float(np.max(corr))
    best_j = -1
    best_k = -1
    best_delta = 0.0
    best_D = None
    for _ in range(n_trials):
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        B_j = compute_B(h, j, n)
        B_k = compute_B(h, k, n)
        D = B_j - B_k
        delta_lo = max(-h[j], h[k] - 1.0)
        delta_hi = min(1.0 - h[j], h[k])
        if delta_hi - delta_lo < 1e-14:
            continue
        delta_opt = solve_1d_minimax(corr, D, delta_lo, delta_hi)
        val_opt = float(np.max(corr + D * delta_opt))
        if val_opt < best_val - 1e-14:
            best_val = val_opt
            best_j = j
            best_k = k
            best_delta = delta_opt
            best_D = D
    return best_j, best_k, best_delta, best_D

def smooth_obj_grad(h, alpha):
    corr = compute_corr(h)
    m_val = float(np.max(corr))
    shifted = corr - m_val
    max_s = float(np.max(shifted))
    if max_s > 40:
        shifted = shifted - max_s
        m_val_adj = m_val + max_s
    else:
        m_val_adj = m_val
    exp_vals = np.exp(alpha * shifted)
    sum_exp = float(np.sum(exp_vals))
    obj = m_val_adj + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    n = len(h)
    nm1 = n - 1
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j + np.arange(2*n-1) - nm1
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - np.arange(2*n-1) + nm1
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[j] = np.sum(w[v1] * (1.0 - h[idx1[v1]])) - np.sum(w[v2] * h[idx2[v2]])
    return obj, grad

def projected_gradient_smooth(h0, target_sum, alpha, max_iter, t_deadline):
    n = len(h0)
    h_curr = h0.copy()
    s = 1.0 / (n + alpha)
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        obj_y, grad = smooth_obj_grad(h_curr, alpha)
        obj_prev = obj_y
        s_local = s
        for _ in range(20):
            x_new = h_curr - s_local * grad
            x_new = project_to_sum(np.clip(x_new, 0.0, 1.0), target_sum)
            obj_new, _ = smooth_obj_grad(x_new, alpha)
            if obj_new <= obj_prev - 0.1 * s_local * np.dot(grad, x_new - h_curr):
                break
            s_local *= 0.5
        s = s_local
        h_curr = x_new
    return project_to_sum(h_curr, target_sum)

def slsqp_smooth(h0, target_sum, alpha, maxiter, t_deadline):
    n = len(h0)
    def fun(x):
        return smooth_obj_grad(x, alpha)[0]
    def jac(x):
        return smooth_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': fun_sum, 'jac': jac_sum}]
    bounds = [(0.0, 1.0)] * n
    def fun_sum(x):
        return np.sum(x) - target_sum
    def jac_sum(x):
        return np.ones(n)
    try:
        if time.time() > t_deadline:
            return project_to_sum(h0, target_sum)
        res = minimize(fun, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def structured_perturb(h, rng, n, target_sum, mode=0):
    if mode == 0:
        return project_to_sum(h + rng.uniform(-0.08, 0.08, n), target_sum)
    elif mode == 1:
        freq = rng.randint(2, n // 4 + 1)
        x = np.arange(n)
        pert = 0.05 * np.sin(2.0 * np.pi * freq * x / n)
        return project_to_sum(h + pert, target_sum)
    elif mode == 2:
        blk = max(1, n // 20)
        pert = np.zeros(n)
        for i in range(0, n, 2 * blk):
            pert[i:min(i+blk, n)] = rng.uniform(-0.1, 0.1)
        return project_to_sum(h + pert, target_sum)
    else:
        idx = rng.choice(n, size=n // 5, replace=False)
        pert = np.zeros(n)
        pert[idx] = rng.uniform(-0.1, 0.1, len(idx))
        return project_to_sum(h + pert, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():

            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - (time.time() - t_start)

    # Phase 1: Intensive pair descent with global best-pair (50% budget)
    t_end1 = t_start + 0.50 * budget_s
    h_w = best_h.copy()
    n_w = len(h_w)
    ts_w = n_w / 2.0
    corr_w = compute_corr(h_w)
    stall = 0
    sweep_count = 0

    while tleft() > 0.48 * budget_s:
        h_w, corr_w, _ = pair_descent_sweep(h_w, corr_w, rng, n_cand=18)
        sweep_count += 1
        c5_i = float(np.max(corr_w) * (2.0 / n_w))
        if c5_i < best_c5 - 1e-12:
            best_c5 = c5_i
            best_h = h_w.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1

        if sweep_count % 100 == 0 and stall > 50:
            bj, bk, bd, bD = global_best_pair(h_w, corr_w, n_w, rng, n_trials=300)
            if bj >= 0 and abs(bd) > 1e-14:
                corr_w += bd * bD
                h_w[bj] = np.clip(h_w[bj] + bd, 0.0, 1.0)
                h_w[bk] = np.clip(h_w[bk] - bd, 0.0, 1.0)
                c5_g = float(np.max(corr_w) * (2.0 / n_w))
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_w.copy()
                    stall = 0

        if stall > 300:
            pmode = 0 if stall % 4 == 0 else (1 if stall % 4 == 1 else (2 if stall % 4 == 2 else 3))
            h_w = structured_perturb(h_w, rng, n_w, ts_w, pmode)
            corr_w = compute_corr(h_w)
            stall = 0
            if stall >= 100:
                noise = min(0.2, 0.05 + 0.02 * (stall // 300))
                h_w = project_to_sum(h_w + rng.uniform(-noise, noise, n_w), ts_w)
                corr_w = compute_corr(h_w)
                stall = 0

    # Phase 2: Smooth max optimization (20% budget)
    t_end2 = t_start + 0.70 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [30, 100, 400]:
        if tleft() < 0.28 * budget_s:
            break
        h_pg = projected_gradient_smooth(best_h, ts_f, alpha, max_iter=150,

        c5_pg = compute_c5(h_pg)

            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n_f

    for alpha in [100, 500]:

        h_sl = slsqp_smooth(best_h, ts_f, alpha, maxiter=400,

        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n_f

    # Phase 3: Multi-resolution (15% budget)
    t_end3 = t_start + 0.85 * budget_s
    resolutions = [120, 160, 200, 240, 280, 320, 400]
    for n_tr in resolutions:
        if tleft() < 0.15 * budget_s:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_tr = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        corr_tr = compute_corr(h_tr)
        max_sweeps = min(40, max(3, int(tleft() / 8.0)))
        sub_stall = 0
        for _ in range(max_sweeps):
            if tleft() < 0.15 * budget_s:

    # Phase 4: Final fine pair descent polish (15% budget)
