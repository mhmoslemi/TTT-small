def fourier_to_grid(coeffs, n, n_harm):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, n_harm + 1):
        a = coeffs[2 * (m - 1)]
        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return h

def grid_to_fourier(h, n, n_harm):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    C = np.zeros((n, 2 * n_harm))
    for m in range(1, n_harm + 1):
        C[:, 2 * (m - 1)] = np.cos(np.pi * m * x)
        C[:, 2 * (m - 1) + 1] = np.sin(np.pi * m * x)
    target = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(C, target, rcond=None)
    return coeffs

def fourier_obj_fn(c, n, n_harm, alpha):

def fourier_lbfgs(coeffs, n, n_harm, alpha, max_iter=400, time_fn=None):

        return fourier_obj_fn(c, n, n_harm, alpha)

        h_opt = project_box_sum(h_opt, n / 2.0)

            h_opt, _ = fourier_lbfgs(coeffs, best_n, n_h, alpha_f, max_iter=500, time_fn=time_left)
            if h_opt is not None:
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    coeffs = grid_to_fourier(best_h, best_n, n_h)

    # Phase 5: Multi-start perturbations
    for trial in range(4):
        if time_left() < 60:
            break
        noise_scale = 0.15 * (1.0 - trial / 4.0) + 0.03
        h_shake = best_h + rng.uniform(-noise_scale, noise_scale, best_n)
        h_shake = project_box_sum(h_shake, best_n / 2.0)
        for alpha_s in [50, 300, 2000, 10000]:
            if time_left() < 20:
                break
            h_opt, _ = pgd_momentum(h_shake, alpha_s, 2000, 0.05, time_left)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_shake = h_opt

    # Phase 6: Binary flip improvement (for semi-binary solutions)
    if time_left() > 40:
        h_bin = (best_h > 0.5).astype(float)
        curr = h_bin.sum()
        if curr != best_n / 2.0:
            if curr > best_n / 2.0:
                excess = int(curr - best_n / 2.0)
                idx = np.where(h_bin > 0.5)[0]
                rng.shuffle(idx)
                for i in range(min(excess, len(idx))):
                    h_bin[idx[i]] = 0.0
            else:
                deficit = int(best_n / 2.0 - curr)
                idx = np.where(h_bin < 0.5)[0]
                rng.shuffle(idx)
                for i in range(min(deficit, len(idx))):
                    h_bin[idx[i]] = 1.0
        h_bf, c5_bf = binary_flip_improve(h_bin, best_n, rng, max_rounds=60, time_fn=time_left)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # Smooth the binary solution
        if time_left() > 30:
            n_harm_b = min(30, best_n // 8)
            coeffs_b = grid_to_fourier(best_h, best_n, n_harm_b)
            for alpha_b in [20, 100, 500]:
                if time_left() < 15:
                    break
                h_sm, _ = fourier_lbfgs(coeffs_b, best_n, n_harm_b, alpha_b, max_iter=300, time_fn=time_left)
                if h_sm is not None:
                    c5_sm = compute_c5(h_sm)
                    if c5_sm < best_c5:
                        best_c5 = c5_sm
                        best_h = h_sm.copy()
                        coeffs_b = grid_to_fourier(best_h, best_n, n_harm_b)

    # Phase 7: Upscale to n=480
    if time_left() > 100:
        n_up = 480
        ts_up = n_up / 2.0
        x_old2 = np.linspace(0, 2, best_n, endpoint=False)
        x_up = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old2, best_h, period=2.0)
        h_up = project_box_sum(h_up, ts_up)
        for alpha_up in [50, 200, 1000, 5000, 15000]:
            if time_left() < 25:
                break
            h_opt, _ = pgd_momentum(h_up, alpha_up, 4000, 0.03, time_left)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_up

    # Phase 8: Final targeted + high-alpha polish
    if time_left() > 40:
        top_lags_fin = find_top_lags(best_h, 8)
        ew_fin = [(k, 0.3) for k in top_lags_fin]
        for alpha_fin in [5000, 20000, 50000]:
            if time_left() < 15:
                break
            h_opt, _ = pgd_momentum(best_h, alpha_fin, 5000, 0.01, time_left, extra_lags=ew_fin)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 9: Very fine final polish
    if time_left() > 20:
        h_opt, _ = pgd_momentum(best_h, 50000, 5000, 0.005, time_left)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
