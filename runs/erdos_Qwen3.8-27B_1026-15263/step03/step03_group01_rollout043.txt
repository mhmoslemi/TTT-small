    j_arr = np.arange(n)  # shape (n,)

    idx2 = j_arr[:, None] + L[None, :]  # shape (n, 2n-1)
    
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)

    h2 = h[np.clip(idx2, 0, n-1)]  # shape (n, 2n-1)

    term2 = mask2.astype(float) * w[None, :] * h2

def project_to_sum(h, target_sum):

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    L = np.arange(-(n - 1), n)

    mask2 = (idx2 >= 0) & (idx2 < n)

    h2 = h[np.clip(idx2, 0, n - 1)]

    term2 = mask2.astype(np.float64) * w[None, :] * h2
    grad = term1.sum(axis=1) - term2.sum(axis=1)
