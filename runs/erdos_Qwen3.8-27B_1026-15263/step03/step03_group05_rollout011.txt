Let me also think about the structure of the problem. The Erdős overlap problem: we want to minimize the maximum overlap between h and 1-h at any shift. This is related to the "fingerprint" of the set {x: h(x)=1}.

For a binary solution with exactly n/2 ones, the overlap at shift k is:
|{i: h[i]=1} ∩ {i: h[i+k]=0}| (counting)

We want this to be as small as possible for ALL k.

This is related to the concept of a "Salem-Zygmund" type set or a "Bose-Chowla" type construction, but for the continuous/discrete setting.

Actually, for the binary case with n/2 ones out of n positions, the minimum maximum overlap is related to the autocorrelation of the -1/+1 sequence. If we define s[i] = 2*h[i] - 1 ∈ {-1, +1}, then:
h[i]*(1-h[i+k]) = (1+s[i])/2 * (1-s[i+k])/2 = (1 - s[i] - s[i+k] + s[i]*s[i+k])/4

Sum over valid i:
= (number of valid i)/4 - sum s[i]/4 - sum s[i+k]/4 + sum s[i]*s[i+k]/4

For the sum constraint (exactly n/2 ones), sum s[i] = 0.
For the cross terms, if we're in the "bulk" (away from edges), sum s[i+k] ≈ sum s[i] = 0.

So approximately: overlap ≈ (n/4) + (autocorrelation of s at lag k)/4

To minimize the max overlap, we want to minimize the max autocorrelation of s.

This is exactly the problem of finding a binary sequence with minimal peak autocorrelation! Classic signal processing / combinatorics problem.

Known constructions:
- Barker sequences (lengths 2,3,4,5,7,11,13)
- m-sequences (from finite fields)
- Legendre sequences
-_quad-residue sequences

For longer sequences, random sequences have peak autocorrelation ≈ O(sqrt(n)), while optimal might be lower.

def compute_grads(h, n):
    m_vals = np.arange(-(n-1), n)  # 2n-1 values
    i = np.arange(n)  # n values
    # For each (m, i) pair, compute:
    # valid1 = (0 <= i < n) AND (0 <= i+m < n) → (i+m >= 0) AND (i+m < n)
    # valid2 = (0 <= i-m < n) [i is already in [0,n)]

    # m_vals has shape (2n-1,), i has shape (n,)
    # im = i[None, :] + m_vals[:, None] has shape (2n-1, n)
    im = i[None, :] + m_vals[:, None]
    im_neg = i[None, :] - m_vals[:, None]
    
    valid1 = (im >= 0) & (im < n)  # (2n-1, n)
    valid2 = (im_neg >= 0) & (im_neg < n)  # (2n-1, n)

    """Gradient of corr at given lag index (0 to 2n-2)."""
    m = lag - (n - 1)  # actual shift
    g = np.zeros(n)
    i = np.arange(n)
    im = i + m
    valid1 = (im >= 0) & (im < n)
    im_neg = i - m
    valid2 = (im_neg >= 0) & (im_neg < n)
    g[valid1] = 1.0 - h[im[valid1]]
    g[valid2] -= h[im_neg[valid2]]
    return g

def subgradient_descent(h, n, max_iter, time_fn):
    """Nesterov accelerated subgradient method."""
    x = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    gamma = 0.1  # step size
    
    for it in range(max_iter):
        if time_fn() < 10:
            break
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        c5_val = np.max(corr) * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = y.copy()
        
        # Find active lags (within 1% of max)
        max_val = np.max(corr)
        threshold = max_val - 0.01 * max_val
        active = np.where(corr >= threshold)[0]
        
        # Subgradient = average gradient over active lags
        g = np.zeros(n)
        for lag in active:
            g += grad_at_lag(y, n, lag)
        g /= len(active)
        
        # Nesterov update
        t_k = 2.0 / (it + 3)
        step = gamma / (it + 1)
        x_new = project_box_sum(y - step * g, n / 2.0)
        y = (1.0 - t_k) * x_new + t_k * x
        y = project_box_sum(y, n / 2.0)
        x = x_new
        # Hmm, Nesterov for subgradient is tricky with projections
        # Let me just do projected subgradient
        y = x_new
        x = x_new
    
    return best_h, best_c5

def slp_lp_step(h, n):
    """One SLP iteration using LP subproblem."""
    corr_cur = np.correlate(h, 1.0 - h, mode="full")
    grads = compute_grads_matrix(h, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    rhs = grads @ h - corr_cur
    lhs = grads @ h_var - t_var * cp.ones(2 * n - 1)
    constraints.append(lhs <= rhs)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in [cp.GLOP, cp.CBC, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            break
        except:
            continue
    
    if h_var.value is None:
        return None
    return np.clip(h_var.value, 0.0, 1.0)

def compute_grads_matrix(h, n):
    m_vals = np.arange(-(n-1), n)
    i = np.arange(n)
    im = i[None, :] + m_vals[:, None]
    im_neg = i[None, :] - m_vals[:, None]
    valid1 = (im >= 0) & (im < n)
    valid2 = (im_neg >= 0) & (im_neg < n)
    safe_im = np.clip(im, 0, n-1)
    safe_im_neg = np.clip(im_neg, 0, n-1)
    h_im = np.where(valid1, h[safe_im], 0.0)
    h_im_neg = np.where(valid2, h[safe_im_neg], 0.0)
    grads = valid1.astype(float) * (1.0 - h_im) - valid2.astype(float) * h_im_neg
    return grads

def binary_flip_improve(h, n, rng, max_rounds=200, time_fn=None):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        if time_fn and time_fn() < 5:
            break
        # Find current max correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_lag = np.argmax(corr)
        m = max_lag - (n - 1)
        
        # Find the "worst" pairs at this lag
        # We want to reduce corr at max_lag
        # corr[m] = sum_i h[i]*(1-h[i+m]) for valid i
        # To reduce: turn off a 1 where h[i+m]=0, or turn on a 0 where h[i+m]=1
        # But we need to maintain sum constraint (one flip 1->0, one flip 0->1)
        
        # Strategy: find i where h[i]=1 and h[i+m]=0 (contributes to overlap)
        # and j where h[j]=0 and h[j+m]=1 (contributes 0 to overlap but removing would help)
        # Actually, let's just do random pairs
        
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = 300
        for _t in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            old_i, old_j = h[i], h[j]
            h[i] = 0.0
            h[j] = 1.0
            c5_t = compute_c5(h)
            if c5_t < c5_cur - 1e-12:
                c5_cur = c5_t
                improved = True
            else:
                h[i] = old_i
                h[j] = old_j
        if not improved:
            break
    return h, c5_cur

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    if n == 0: return y
    lo = float(-np.max(y)) - 2.0
    hi = 1.0 - float(np.min(y)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s: lo = mid
        else: hi = mid
    return np.clip(y - 0.5*(lo+hi), 0.0, 1.0)

def compute_grads_matrix(h, n):
    m_vals = np.arange(-(n-1), n)
    i = np.arange(n)
    im = i[None,:] + m_vals[:,None]
    im_neg = i[None,:] - m_vals[:,None]
    v1 = (im >= 0) & (im < n)
    v2 = (im_neg >= 0) & (im_neg < n)
    sim = np.clip(im, 0, n-1)
    simn = np.clip(im_neg, 0, n-1)
    h_im = np.where(v1, h[sim], 0.0)
    h_imn = np.where(v2, h[simn], 0.0)
    return v1.astype(float)*(1.0 - h_im) - v2.astype(float)*h_imn

def slp_lp(h, n):
    corr = np.correlate(h, 1.0-h, mode="full")
    G = compute_grads_matrix(h, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n/2.0, G @ hv - tv <= G @ h - corr]
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.GLOP, cp.CBC, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if hv.value is not None: return np.clip(hv.value, 0.0, 1.0)
        except: continue
    return None

            else: h[i]=oi; h[j]=oj
        if not imp: break
