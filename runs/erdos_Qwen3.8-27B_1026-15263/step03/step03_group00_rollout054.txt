import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def weighted_subgrad_step(h, target_sum, top_k=10, alpha=50, step_size=0.01):
    n = len(h)
    corr = compute_corr(h)
    top_indices = np.argsort(corr)[-top_k:]
    max_c = corr[top_indices].max()

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def weighted_subgrad_step(h, top_k, alpha, step_size):
    n = len(h)
    corr = compute_corr(h)
    top_indices = np.argsort(corr)[-top_k:]
    max_c = corr[top_indices].max()
    exp_c = np.exp(alpha * (corr[top_indices] - max_c))
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        lag = idx - (n - 1)
        grad += w * corr_gradient(h, lag)
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum_global)
    return h_new

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:

        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:

        h[i_dec] -= move

def binary_simulated_annealing(h_bin, target_sum, rng, max_iters=30000, T_init=0.008, T_decay=0.9997):
    n = len(h_bin)
    h = h_bin.copy()
    c5 = compute_c5(h)
    T = T_init
    best_c5 = c5
    best_h = h.copy()
    for it in range(max_iters):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = int(rng.choice(ones))
        i0 = int(rng.choice(zeros))
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_bin_sa
        if delta < 0 or rng.random() < np.exp(-abs(delta) / T):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        T *= T_decay
        if T < 1e-6:
            T = 1e-6
    return best_h, best_c5

c5_bin_sa = 0.0

def binary_sa_safe(h_bin, target_sum, rng, max_iters=20000, T_init=0.006, T_decay=0.9998):
    n = len(h_bin)
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    T = T_init
    best_c5 = c5_cur
    best_h = h.copy()
    for it in range(max_iters):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = int(rng.choice(ones))
        i0 = int(rng.choice(zeros))
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-(abs(delta) + 1e-12) / T):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        T *= T_decay
        if T < 5e-6:
            T = 5e-6
    return best_h, best_c5

def binarize(h, n):
    target_ones = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones < target_ones:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:target_ones - n_ones]
        h_bin[zeros[order]] = 1.0
    elif n_ones > target_ones:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:n_ones - target_ones]
        h_bin[ones[order]] = 0.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: SLSQP smooth max with progressive alpha
    for alpha in [80, 400, 2000]:
        if time_left() < 25:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Subgradient descent
    if time_left() > 80:
        h_sg, c5_sg = subgrad_descent(best_h, ts_parent, max_iters=4000, alpha=35, top_k=12)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Multi-resolution refinement
    res_list = []
    for nr in [420, 480, 540, 600, 660, 720]:
        if nr > n_parent and nr % 2 == 0:
            res_list.append(nr)
    res_list.sort()

    for n_tr in res_list:
        if time_left() < 90:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

        for alpha in [150, 800, 4000]:
            if time_left() < 45:
                break
            h_opt = slsqp_local(h_interp, n_tr, ts_tr, alpha, maxiter=450)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt

        if time_left() > 40:
            h_sg2, c5_sg2 = subgrad_descent(h_interp, ts_tr, max_iters=2500, alpha=35, top_k=10)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2.copy()
                best_n = n_tr
            h_interp = h_sg2

    # Phase 4: Coordinate descent polishing
    n_best = best_n
    ts_best = n_best / 2.0
    if time_left() > 60:
        h_cd, c5_cd = coordinate_descent(best_h, ts_best, max_iters=15000, step_frac=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Binary optimization
    if time_left() > 100:
        h_bin = binarize(best_h, n_best)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        h_bg, c5_bg = binary_greedy(h_bin, max_iters=8000)
        if c5_bg < best_c5:
            best_c5 = c5_bg
            best_h = h_bg.copy()

        h_sa, c5_sa = binary_sa_safe(h_bg, ts_best, rng, max_iters=25000, T_init=0.005, T_decay=0.9998)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        h_bg2, c5_bg2 = binary_greedy(h_sa, max_iters=5000)
        if c5_bg2 < best_c5:
            best_c5 = c5_bg2
            best_h = h_bg2.copy()

    # Phase 6: Perturbation restarts
    n_trials = 8
    for trial in range(n_trials):
        if time_left() < 50:
            break
        noise_level = 0.04 + 0.03 * (1.0 - trial / float(n_trials))
        h_shake = best_h + rng.uniform(-noise_level, noise_level, n_best)
        h_shake = project_to_sum(h_shake, ts_best)
        h_cd_t, c5_t = coordinate_descent(h_shake, ts_best, max_iters=4000, step_frac=0.1)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_cd_t.copy()
        if time_left() < 40:
            break
        h_opt_t = slsqp_local(h_cd_t, n_best, ts_best, 2000, maxiter=400)
        c5_opt_t = compute_c5(h_opt_t)
        if c5_opt_t < best_c5:
            best_c5 = c5_opt_t
            best_h = h_opt_t.copy()

    # Phase 7: Final high-alpha SLSQP and coordinate descent
    if time_left() > 40:
        for alpha in [8000, 30000]:
            if time_left() < 25:
                break
            h_pol = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time_left() > 30:
        h_final, c5_final = coordinate_descent(best_h, ts_best, max_iters=25000, step_frac=0.04)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 8: Final binary check from best continuous
    if time_left() > 40:
        h_bin2 = binarize(best_h, n_best)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        h_bg3, c5_bg3 = binary_greedy(h_bin2, max_iters=6000)
        if c5_bg3 < best_c5:
            best_c5 = c5_bg3
            best_h = h_bg3.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
