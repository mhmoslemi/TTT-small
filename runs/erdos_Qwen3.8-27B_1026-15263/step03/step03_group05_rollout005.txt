import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * dx
    return obj, w, corr

def compute_grad_h(w, h, n):

    grad_h = np.zeros(n)

        b = coeffs[2 * (m - 1) + 1]

def grid_to_fourier(h, n, n_harm, x):
    C = np.zeros((n, 2 * n_harm))

        C[:, 2 * (m - 1) + 1] = np.sin(np.pi * m * x)
    target = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(C, target, rcond=None)
    return coeffs
