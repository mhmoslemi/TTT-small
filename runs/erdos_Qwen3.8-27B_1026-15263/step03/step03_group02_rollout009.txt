def make_binary_from_continuous(h, target_sum, rng):
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    # Find threshold: take top target_sum elements
    threshold = h_sorted[min(target_sum, n) - 1] if target_sum > 0 else 1.0
    h_bin = (h >= threshold).astype(float)
    # Fix count
    n_ones = int(np.sum(h_bin))
    if n_ones > target_sum:
        # Too many ones, remove the smallest ones
        ones_vals = h[np.where(h_bin == 1.0)[0]]
        remove_count = n_ones - target_sum
        remove_idx = np.argsort(ones_vals)[:remove_count]
        all_ones_idx = np.where(h_bin == 1.0)[0]
        h_bin[all_ones_idx[remove_idx]] = 0.0
    elif n_ones < target_sum:
        zeros_vals = h[np.where(h_bin == 0.0)[0]]
        add_count = target_sum - n_ones
        add_idx = np.argsort(zeros_vals)[::-1][:add_count]
        all_zeros_idx = np.where(h_bin == 0.0)[0]
        h_bin[all_zeros_idx[add_idx]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng, T_init=0.015, T_min=0.002, cool=0.999995):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    stall_counter = 0
    
    while time.time() < t_end:
        move_type = rng.rand()
        ok = False
        if move_type < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            ok = True
        elif move_type < 0.8:
            # Block swap
            block_len = rng.randint(1, 5)
            i = rng.randint(n - block_len)
            j = rng.randint(n - block_len)
            if i == j:
                continue
            # Find a block where h[i:i+block_len] != h[j:j+block_len]
            if not np.array_equal(h[i:i+block_len], h[j:j+block_len]):
                tmp = h[i:i+block_len].copy()
                h[i:i+block_len] = h[j:j+block_len]
                h[j:j+block_len] = tmp
                ok = True
            else:
                continue
        else:
            # Multi-bit: flip a 0 to 1 and a 1 to 0 (random positions)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            ok = True
        
        if not ok:
            continue
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform() + 1e-30) < delta / T:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            if move_type < 0.6:
                h[i], h[j] = h[j], h[i]
            elif move_type < 0.8:
                tmp = h[i:i+block_len].copy()
                h[i:i+block_len] = h[j:j+block_len]
                h[j:j+block_len] = tmp
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, 0.02)
            no_improve = 0
            stall_counter += 1
            if stall_counter > 5:
                break
        
        T = max(T * cool, T_min)
    
    return best_h, best_c5

            c5_curr = best_c5
            T = min(T * 2.5, 0.018)
            no_improve = 0
            stall += 1
            ones_list = np.where(h == 1.0)[0]
            zeros_list = np.where(h == 0.0)[0]
            if stall > 8:
                break
        T = max(T * cool, T_min)
    return best_h, best_c5

def continuous_refine(h0, n, target_sum, t_end, alpha_init=50):
    h = h0.astype(float).copy()
    h = project_to_sum(h, target_sum)
    h_mom = np.zeros(n)
    alpha = alpha_init
    lr = 0.05
    best_obj = compute_c5(h)
    for it in range(5000):
        if time.time() > t_end:
            break
        obj, grad = smooth_max_obj_grad(h, alpha)
        h_mom = 0.9 * h_mom + 0.1 * grad
        h_new = h - lr * h_mom
        h_new = project_to_sum(h_new, target_sum)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj + 0.01:
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                lr = min(lr * 1.1, 0.2)
            else:
                lr = max(lr * 0.95, 0.005)
        else:
            lr = max(lr * 0.8, 0.003)
        if it % 200 == 0:
            alpha = min(alpha * 1.3, 5000.0)
    best_obj = compute_c5(h)
    return h, best_obj

def interp_to_resolution(h_src, n_src, n_dst):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def greedy_binary(h, n, target_sum, rng, max_rounds=50):
    h = h.copy()
    c5_curr = compute_c5(h)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    for _ in range(max_rounds):
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_gain = 1e-15
        best_i = -1
        best_j = -1
        n_try = min(200, len(ones_idx) * len(zeros_idx))
        for _ in range(n_try):
            ci = ones_idx[rng.randint(len(ones_idx))]
            cj = zeros_idx[rng.randint(len(zeros_idx))]
            h[ci] = 0.0
            h[cj] = 1.0
            c5_try = compute_c5(h)
            h[ci] = 1.0
            h[cj] = 0.0
            gain = c5_curr - c5_try
            if gain > best_gain:
                best_gain = gain
                best_i = ci
                best_j = cj
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5_curr = compute_c5(h)
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
        else:
            break
    return h, c5_curr

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    h_bin = make_binary(best_h, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    h_bin, c5_greedy = greedy_binary(h_bin, n_main, int(ts_main), rng, max_rounds=30)
    if c5_greedy < best_c5:
        best_c5 = c5_greedy
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.40 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h_bin = h_sa.copy()

    h_bin, c5_g2 = greedy_binary(h_bin, n_main, int(ts_main), rng, max_rounds=40)
    if c5_g2 < best_c5:
        best_c5 = c5_g2
        best_h = h_bin.copy()

    h_pol, c5_pol = continuous_refine(best_h, n_main, ts_main, t_start + 0.55 * budget_s, alpha_init=80)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    h_bin2 = make_binary(best_h, int(ts_main), rng)
    sa_end2 = t_start + 0.72 * budget_s
    if remaining() > 60:
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, int(ts_main), sa_end2, rng, T_init=0.008)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    for n_try in [250, 260, 280, 300]:
        if remaining() < 80:
            break
        ts_try = n_try / 2.0
        h_interp = interp_to_resolution(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try_bin = make_binary(h_interp, int(ts_try), rng)
        c5_try = compute_c5(h_try_bin)
        h_try_bin, c5_try = greedy_binary(h_try_bin, n_try, int(ts_try), rng, max_rounds=20)
        sa_t_end = min(t_start + 0.92 * budget_s, time.time() + min(remaining() * 0.8, 60))
        h_try_binary, c5_try_sa = binary_sa(h_try_bin, n_try, int(ts_try), sa_t_end, rng, T_init=0.01)
        if c5_try_sa < best_c5:
            best_c5 = c5_try_sa
            best_h = h_try_binary.copy()
            best_n = n_try

    if remaining() > 30:
        h_pol2, c5_pol2 = continuous_refine(best_h, best_n, best_n / 2.0, t_start + 0.97 * budget_s, alpha_init=200)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
