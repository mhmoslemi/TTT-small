One approach: change h[j] by δ and h[j'] by -δ for some j' (pair swap). Then the sum is preserved.
Change in corr[k] = δ * dc/dh[j][k] - δ * dc/dh[j'][k] = δ * (dc/dh[j] - dc/dh[j'])[k]
New C5 = max(old_corr + δ * (dc_j - dc_j')) * dx

This is O(n) per evaluation. And we can try many (j, j', δ) combinations.

But this is more complex. Let me stick with the simpler approach: change h[j] by δ, then project. The projection is O(n) (just redistribute the excess among interior points). Then compute C5 in O(n²) via np.correlate.

Actually, for the coordinate descent, let me use a hybrid:
- For each j, precompute the "effect vector" e_j = dc/dh[j] (length 2n-1)
- When changing h[j] by δ (and projecting):
  - If no projection needed (h[j]+δ in [0,1] and we can adjust one other point): fast
  - Otherwise: full recompute

For simplicity, let me just do the full recompute but make it as fast as possible. For n=360, np.correlate takes about 0.05ms, so 2160 evaluations per round takes about 0.1s. 200 rounds = 20s. That's fine.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    Nc = len(corr)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / Nc) / alpha
    g = np.zeros(n)
    j_arr = np.arange(n)
    for j in range(n):
        lo1 = max(0, j - (n - 1))
        hi1 = min(2 * n - 2, j + (n - 1))
        if hi1 >= lo1:
            ks = np.arange(lo1, hi1 + 1)
            t_vals = j + (n - 1) - ks
            valid = (t_vals >= 0) & (t_vals < n)
            if np.any(valid):
                g[j] += np.dot(w[ks[valid]], (1.0 - h[t_vals[valid]]))
        lo2 = max(0, j - (n - 1))
        hi2 = min(2 * n - 2, j + (n - 1))
        if hi2 >= lo2:
            ks2 = np.arange(lo2, hi2 + 1)
            u_vals = j - (n - 1) + ks2
            valid2 = (u_vals >= 0) & (u_vals < n)
            if np.any(valid2):
                g[j] -= np.dot(w[ks2[valid2]], h[u_vals[valid2]])
    g /= Nc
    return sm_val, g

_g_ts = 0
_g_n = 0
_g_alpha = 1.0

def _obj(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _g_alpha)
    return v

def _grad(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _g_alpha)
    return g

def _con_fun(x):
    return np.sum(x) - _g_ts

def _con_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _g_ts, _g_n, _g_alpha
    _g_ts = target_sum
    _g_n = n
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_to_sum(res.x, target_sum)
    except Exception:
        pass
    return project_to_sum(h0, target_sum)

def coord_descent_fast(h0, n, target_sum, max_rounds, rng, base_delta, n_delta_levels):
    h = h0.copy()
    c5_cur = compute_c5(h)
    delta_vals = np.array([-4.0, -3.0, -2.0, -1.5, -1.0, -0.5,
                           0.5, 1.0, 1.5, 2.0, 3.0, 4.0]) * base_delta
    delta_vals = delta_vals[:min(len(delta_vals), n_delta_levels * 2)]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in delta_vals:
                nv = h[i] + d
                if nv < -1e-14 or nv > 1.0 + 1e-14:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def coord_descent_blocks(h0, n, target_sum, max_rounds, rng, base_delta, block_size):
    h = h0.copy()
    c5_cur = compute_c5(h)
    delta_vals = np.array([-2.0, -1.0, -0.5, 0.5, 1.0, 2.0]) * base_delta
    for rnd in range(max_rounds):
        improved = False
        start = rng.randint(0, n)
        for rep in range(n // block_size):
            idx = (start + rep * block_size) % n
            if idx + block_size > n:
                break
            for d in delta_vals:
                h_new = h.copy()
                h_new[idx:idx+block_size] = np.clip(h_new[idx:idx+block_size] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-14:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def pair_swap_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0]) * base_delta
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n // 2):
            i1, i2 = rng.choice(n, size=2, replace=False)
            for d in deltas:
                nv1 = h[i1] + d
                nv2 = h[i2] - d
                if nv1 < -1e-14 or nv1 > 1.0 + 1e-14:
                    continue
                if nv2 < -1e-14 or nv2 > 1.0 + 1e-14:
                    continue
                h_new = h.copy()
                h_new[i1] = nv1
                h_new[i2] = nv2
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-14:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def binary_push(h, target_sum, threshold_low, threshold_high):
    h_new = h.copy()
    mask_low = h < threshold_low
    mask_high = h > threshold_high
    h_new[mask_low] = 0.0
    h_new[mask_high] = 1.0
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1300, **kwargs):
