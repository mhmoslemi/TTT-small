import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, k, n):
    """Gradient of R_k with respect to h."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def top_k_subgradient(h, top_k=5, temperature=10.0):
    """Compute weighted subgradient using top-k correlation values."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Softmax weights over top-k
    top_indices = np.argsort(corr)[::-1][:top_k]
    weights = np.exp(temperature * (corr[top_indices] - m))
    weights = weights / np.sum(weights)
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        grad += w * corr_gradient(h, idx, n)
    return grad, m

def projected_subgradient_step(h, grad, target_sum, step_size):
    """Take a projected gradient step maintaining sum constraint."""
    h_new = h - step_size * grad
    return project_to_sum(h_new, target_sum)

def subgradient_method(h0, target_sum, budget_check, max_iters=10000):

                return best_h, best_c5

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def lbfgs_smooth_max(h0, n, target_sum, alpha, maxiter, ftol=1e-14):
    h = h0.copy()
    def fun(x):
        obj, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * 1e6 * s_dev * s_dev
    
    def jac(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + 1e6 * s_dev * np.ones(n)
    
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, k, n):
    shift = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-12
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    h = h0.copy()
    def fun(x):
        obj, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * penalty * s_dev * s_dev
    def jac(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + penalty * s_dev * np.ones(n)
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

def nesterov_projected_gradient(h0, n, target_sum, alpha, n_iters, base_lr):
    """Nesterov accelerated projected gradient for smooth max."""
    h = h0.copy()
    h_prev = h0.copy()
    momentum = 0.9
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for t in range(n_iters):
        # Nesterov lookahead
        h_nest = h + momentum * (h - h_prev)
        h_nest = np.clip(h_nest, 0.0, 1.0)
        
        obj, grad = smooth_max_obj_and_grad(h_nest, alpha)
        s_dev = np.sum(h_nest) - target_sum
        grad_full = grad + 1e6 * s_dev * np.ones(n)
        
        lr = base_lr / (1.0 + 0.01 * t)
        
        h_new = h - lr * grad_full
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
    
    return best_h, best_c5

def exact_max_polish(h, n, target_sum, max_rounds=500):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(5, len(corr))]
        best_improve = 0
        best_h = h
        for k_max in top_k:
            grad = corr_gradient(h, k_max, n)
            for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-14:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is h:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur

def binary_local_search(h_bin, n, target_sum, rng, max_rounds=300):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for r in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        n_trials = min(500, max(100, len(ones) * len(zeros) // 10))
        best_trial_c5 = c5_cur
        best_swap = None
        
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            c5_test = compute_c5(h_test)
            if c5_test < best_trial_c5 - 1e-14:
                best_trial_c5 = c5_test
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            c5_cur = best_trial_c5
            improved = True
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def make_random_binary(n, target_sum, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=int(target_sum), replace=False)
    h[positions] = 1.0
    return h
