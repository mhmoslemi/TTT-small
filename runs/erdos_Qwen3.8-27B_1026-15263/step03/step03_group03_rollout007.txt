import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def compute_B(h, j, n):
    """Compute B vector: corr_new = corr_old + delta * B when h[j] += delta, h[k] -= delta"""
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    # h[j] increases by delta:
    # corr[m] has term h[i]*(1-h[i+m-nm1])
    # When i=j: corr += delta*(1-h[j+m-nm1]) for valid (j+m-nm1)
    # When i+m-nm1=j, i.e., m = j+nm1-i: corr += -delta*h[i] for valid i
    # j+m-nm1 valid: 0 <= j+m-nm1 < n, so nm1-j <= m <= nm1-j+n-1
    lo1 = max(0, nm1 - j)
    hi1 = min(2*n-2, nm1 - j + n - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1+1)
        B[lo1:hi1+1] += (1.0 - h[j + idx - nm1])
    # For the second part: i+m-nm1 = j, i = j+nm1-m

    lo2 = max(0, j + nm1 - n + 1)
    hi2 = min(2*n-2, j + nm1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2+1)
        B[lo2:hi2+1] += -h[j + nm1 - idx]
    return B

def solve_1d(A, D, lo, hi):
    """Minimize max(A + D*t) over t in [lo, hi]"""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Grid search then bisection
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=8):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Choose candidates: neighbors, far neighbors, random
        k_set = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n]
        for _ in range(max(0, n_cand - 4)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:

            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:

                best_delta = delta_opt
                best_D = D

            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)

    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    """Smooth max of corr values and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    # Gradient
    nm1 = n - 1
    grad = np.zeros(n)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    # corr[m] = sum_i h[i] * (1-h[i+m-nm1])
    # d(corr[m])/d(h[j]) = (1-h[j+m-nm1]) if j+m-nm1 in [0,n-1]
    #                     - h[j+m-nm1... wait
    # Actually: corr[m] = sum_i h[i]*(1-h[i+m-nm1])
    # d/dh[j]: 
    #   As h[i] with i=j: (1-h[j+m-nm1]) if 0<=j+m-nm1<n
    #   As h[i+m-nm1] with i+m-nm1=j, i=j+nm1-m: -h[j+nm1-m] if 0<=j+nm1-m<n
    first_idx = j_idx + m_idx - nm1  # j + m - nm1
    second_idx = j_idx + nm1 - m_idx  # j + nm1 - m
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n-1)

    dcorr_dh = valid_f * (1.0 - h[fc]) - valid_s * h[sc]
    # grad[j] = sum_m w[m] * dcorr_dh[j,m]
    grad = np.sum(w[None, :] * dcorr_dh, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

        # corr[k] = sum_i h[i]*(1-h[i+k-nm1])
        # d/dh[j]: (1-h[j+k-nm1]) if 0<=j+k-nm1<n, minus h[j+nm1-k] if 0<=j+nm1-k<n

        valid_f = (first >= 0) & (first < n)
        valid_s = (second >= 0) & (second < n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    lo1 = max(0, nm1 - j)
    hi1 = min(2 * n - 2, nm1 - j + n - 1)
    if lo1 <= hi1:
        B[lo1:hi1+1] += (1.0 - h[j + np.arange(lo1, hi1+1) - nm1])
    lo2 = max(0, j + nm1 - n + 1)
    hi2 = min(2 * n - 2, j + nm1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2+1)
        B[lo2:hi2+1] += -h[j + nm1 - idx]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):

        else:

        best_D = None
        best_k = -1
        k_set = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n]

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
        stall = 0
        while time.time() < t_end2:
            h_hi, corr_hi, _ = pair_descent_sweep(h_hi, corr_hi, rng, n_cand=8)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5 - 1e-11:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
                stall = 0
            else:
                stall += 1
            if stall == 60:
                h_hi = project_to_sum(h_hi + rng.uniform(-0.03, 0.03, n_hi), n_hi / 2.0)
                corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode="full")
                stall = 0

    # Phase 3: SLSQP at both resolutions (25%)
    t_end3 = t_start + 0.70 * budget_s
    alphas = [80, 150, 300, 600, 1200]
    for alpha in alphas:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-8:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    # Also try SLSQP at n=240 version
    if time.time() < t_end3 and best_n != 240:
        h_240 = upsample_h(best_h, best_n, 240)
        for alpha in [100, 300, 800]:
            if time.time() >= t_end3:
                break
            h_pol2 = slsqp_optimize(h_240, 240, 120.0, alpha, maxiter=400)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5 - 1e-8:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
                best_n = 240

    # Phase 4: Subgradient method (10%)
    t_end4 = t_start + 0.80 * budget_s
    if time.time() < t_end4:
        n_sg = best_n
        h_sg = best_h.copy()
        corr_sg = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        c5_sg = compute_c5(h_sg)
        for _ in range(8):
            if time.time() >= t_end4:
                break
            h_sg, c5_sg2 = subgradient_phase(h_sg, corr_sg, n_sg, max_iter=300, alpha_init=1.0)
            if c5_sg2 < best_c5 - 1e-9:
                best_c5 = c5_sg2
                best_h = h_sg.copy()
                best_n = n_sg

    # Phase 5: Final intensive pair descent (15%)
    t_end5 = t_start + budget_s - 3
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall = 0
    while time.time() < t_end5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=14)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-11:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall = 0
        else:
            stall += 1
        if stall == 40:
            h_fin = project_to_sum(h_fin + rng.uniform(-0.015, 0.015, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
