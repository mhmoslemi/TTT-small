    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        grad += wi * subgrad_for_lag(x, shift, n)
    return obj, grad

def lbfgs_smooth(h0, n, target_sum, alpha, maxiter=500, pen=1e6):
    def fun(x):
        obj, sg = smooth_max_obj_grad(x, n, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * pen * s_dev * s_dev
    
    def jac(x):
        _, sg = smooth_max_obj_grad(x, n, alpha)
        s_dev = np.sum(x) - target_sum
        return sg + pen * s_dev
    
    res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    return project_to_sum(res.x, target_sum)

def subgrad_polish(h, target_sum, n, max_rounds=1000, top_k=10, lr0=0.1):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_lb = 0.3  # rough lower bound
    lr = lr0
    for r in range(max_rounds):
        corr = compute_corr(h)
        idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for k in idx:
            shift = k - (n - 1)
            grad += subgrad_for_lag(h, shift, n)
        grad /= top_k
        
        gnorm = np.max(np.abs(grad)) + 1e-10
        if best_c5 > c5_lb:
            lr_polyak = (best_c5 - c5_lb) / (gnorm * gnorm)
            lr = max(0.01, min(0.5, lr * 0.5 + lr_polyak * 0.5))
        else:
            lr *= 0.999
        
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if lr < 1e-7:
            break
    return best_h, best_c5

def coordinate_descent(h, target_sum, n, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Optimize h[i] given all others
            # corr[k] involves pairs (i, i+shift) and (i-shift, i)
            # The contribution of h[i] to corr is:
            # For each shift s: h[i]*(1-h[i+s]) + h[i-s]*(1-h[i]) (where valid)
            # = h[i] - h[i]*h[i+s] + h[i-s] - h[i-s]*h[i]
            # = h[i]*(1 - h[i+s] - h[i-s]) + h[i-s]
            # So d(corr[s])/dh[i] = (1 - h[i+s]) - h[i-s] (for the relevant shifts)
            # The max over s of corr[s] is what we want to minimize
            # But for coordinate descent, we can try to minimize the sum of top correlations
            # Simpler: just try h[i] in {0, current, 1} and a few in between
            
            old_val = best_h[i]
            best_val = old_val
            best_c5_local = best_c5
            
            candidates = [0.0, 1.0, old_val]
            if 0 < old_val < 1:
                candidates.extend([old_val - 0.1, old_val + 0.1, 0.25, 0.75])
            
            for cv in candidates:
                if cv < 0 or cv > 1:
                    continue
                if cv == old_val:
                    continue
                test_h = best_h.copy()
                test_h[i] = cv
                test_h = project_to_sum(test_h, target_sum)
                c5_test = compute_c5(test_h)
                if c5_test < best_c5_local - 1e-14:
                    best_c5_local = c5_test
                    best_val = cv
            if best_val != old_val:
                best_h[i] = best_val
                best_h = project_to_sum(best_h, target_sum)
                best_c5 = best_c5_local
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_search_patterns(n, target_sum, rng, n_trials=2000):
    """Search for good binary patterns."""
    best_h = None
    best_c5 = compute_c5(np.full(n, 0.5))
    for _ in range(n_trials):
        h = np.zeros(n)
        ones = rng.choice(n, size=int(target_sum), replace=False)
        h[ones] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Also try structured patterns
    for period in range(2, 15):
        for offset in range(period):
            h = np.zeros(n)
            for i in range(n):
                if i % period == offset:
                    h[i] = 1.0
            if abs(h.sum() - target_sum) < 0.5:
                # Adjust
                if h.sum() > target_sum:
                    excess = int(h.sum() - target_sum)
                    ones_idx = np.where(h > 0)[0]
                    h[rng.choice(ones_idx, excess, replace=False)] = 0
                elif h.sum() < target_sum:
                    deficit = int(target_sum - h.sum())
                    zeros_idx = np.where(h < 1)[0]
                    h[rng.choice(zeros_idx, deficit, replace=False)] = 1
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    if best_h is None:
        best_h = project_to_sum(np.full(n, 0.5), target_sum)
        best_c5 = compute_c5(best_h)
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(x, shift, n):
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - x[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= x[lo2 - shift:hi2 - shift]
    return grad

def smooth_max_obj_grad(x, n, alpha):
    corr = compute_corr(x)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = max(1e-10, 0.01 / s)
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        grad += wi * subgrad_for_lag(x, shift, n)
    return obj, grad

    def fun(x):
        obj, sg = smooth_max_obj_grad(x, n, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * pen * s_dev * s_dev
    def jac(x):
        _, sg = smooth_max_obj_grad(x, n, alpha)
        s_dev = np.sum(x) - target_sum
        return sg + pen * s_dev
    try:
        res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_lb = 0.28
    h_prev = h.copy()
    y = h.copy()
    momentum = 0.0
    lr = lr0
    stagnant = 0
    for r in range(max_rounds):
        if r % 50 == 0:
            beta = 2.0 / (r + 2.0)
            y = best_h + beta * (best_h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            y = project_to_sum(y, target_sum)
            h_prev = best_h
        corr = compute_corr(y)
        m = np.max(corr) * (2.0 / n)
        idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for k in idx:
            shift = k - (n - 1)
            grad += subgrad_for_lag(y, shift, n)

                if i % period == offset:
                    h[i] = 1.0
            if h.sum() >= target_sum - 0.5 and h.sum() <= target_sum + 0.5:
                if h.sum() > target_sum:
                    excess = int(h.sum() - target_sum)
                    if excess > 0 and excess < len(np.where(h > 0)[0]):
                        idx = rng.choice(np.where(h > 0)[0], excess, replace=False)
                        h[idx] = 0
                elif h.sum() < target_sum:
                    deficit = int(target_sum - h.sum())
                    if deficit > 0 and deficit < len(np.where(h < 1)[0]):
                        idx = rng.choice(np.where(h < 1)[0], deficit, replace=False)
                        h[idx] = 1
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def block_pattern_search(n, target_sum, rng, n_trials=2000):
    best_h = project_to_sum(np.full(n, 0.5), target_sum)
    best_c5 = compute_c5(best_h)
    for trial in range(n_trials):
        nblocks = rng.randint(2, 12)
        h = np.zeros(n)
        boundaries = sorted(rng.choice(range(1, n), size=nblocks - 1, replace=False))
        boundaries = [0] + list(boundaries) + [n]
        for blk in range(nblocks):
            lo = boundaries[blk]
            hi = boundaries[blk + 1]
            if rng.random() < 0.5:
                h[lo:hi] = 1.0
        if abs(h.sum() - target_sum) > 0.5:
            if h.sum() > target_sum:
                excess = int(h.sum() - target_sum)
                if 0 < excess < n:
                    idx = rng.choice(np.where(h > 0)[0], min(excess, n), replace=False)
                    h[idx] = 0
            else:
                deficit = int(target_sum - h.sum())
                if 0 < deficit < n:
                    idx = rng.choice(np.where(h < 1)[0], min(deficit, n), replace=False)
                    h[idx] = 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h0, n, target_sum, alpha, maxiter=500):
    h = h0.copy()
    def fun(x):
        return smooth_max_obj_grad(x, n, alpha)[0]
    def jac(x):
        return smooth_max_obj_grad(x, n, alpha)[1]
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun, h, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    ts = best_n / 2.0

    for a in [20, 100, 500, 2000, 8000, 30000]:
        if remaining() < 60:
            break
        h_opt = lbfgs_smooth(best_h, best_n, ts, a, maxiter=300)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    if remaining() > 100:
        h_pol, c5_pol = nesterov_subgrad(best_h, ts, best_n, max_rounds=2500, top_k=15, lr0=0.12)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if remaining() > 80:
        h_opt = slsqp_refine(best_h, best_n, ts, alpha=5000, maxiter=400)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    if remaining() > 40:
        h_cd, c5_cd = coord_descent_boundary(best_h, ts, best_n, max_sweeps=15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if remaining() > 250:
        for n_try in [40, 60, 80, 100, 120, 160, 200]:
            if remaining() < 50:
                break
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-9:
                continue
            h_bin, c5_bin = binary_pattern_search(n_try, ts_try, rng, n_trials=4000)
            x_old = np.linspace(0, 2, n_try, endpoint=False)
            x_new = np.linspace(0, 2, best_n, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_bin, period=2.0)
            h_interp = project_to_sum(h_interp, ts)
            for a in [200, 2000, 20000]:
                if remaining() < 25:
                    break
                h_opt = lbfgs_smooth(h_interp, best_n, ts, a, maxiter=200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    h_interp = h_opt
            if remaining() > 30:
                h_pol2, c5_pol2 = nesterov_subgrad(h_interp, ts, best_n, max_rounds=500, top_k=12, lr0=0.08)
                if c5_pol2 < best_c5:
                    best_c5 = c5_pol2
                    best_h = h_pol2

    if remaining() > 200:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        for a in [200, 2000, 10000, 40000]:
            if remaining() < 50:
                break
            h_opt = lbfgs_smooth(h_interp, n2, ts2, a, maxiter=250)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n2
                ts = ts2
                h_interp = h_opt
        if remaining() > 80:
            h_pol, c5_pol = nesterov_subgrad(best_h, ts, best_n, max_rounds=2000, top_k=15, lr0=0.06)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        if remaining() > 40:
            h_cd, c5_cd = coord_descent_boundary(best_h, ts, best_n, max_sweeps=10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    if remaining() > 150:
        h_blk, c5_blk = block_pattern_search(best_n, ts, rng, n_trials=2000)
        for a in [500, 5000]:
            if remaining() < 30:
                break
            h_opt = lbfgs_smooth(h_blk, best_n, ts, a, maxiter=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    if remaining() > 50:
        h_pol, c5_pol = nesterov_subgrad(best_h, ts, best_n, max_rounds=1500, top_k=20, lr0=0.05)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if remaining() > 25:
        h_opt = lbfgs_smooth(best_h, best_n, ts, 50000, maxiter=300)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    if remaining() > 15:
        h_cd, c5_cd = coord_descent_boundary(best_h, ts, best_n, max_sweeps=8)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
