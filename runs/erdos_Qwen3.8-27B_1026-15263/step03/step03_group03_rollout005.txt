```python

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    
    grad = np.zeros(n)
    nm1 = n - 1

def projected_gradient_step(h, grad, lr, target_sum):
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def pg_optimize(h, target_sum, alpha, n_iters, dt_check):
    """Projected gradient with adaptive step size."""
    n = len(h)
    x = h.copy()
    obj_prev, _ = smooth_max_and_grad(x, alpha)
    
    # Estimate Lipschitz constant as alpha * n * 2
    L = alpha * n * 2.0
    lr = 1.0 / L

        if time.time() - dt_check > 3.0:
            break
        _, grad = smooth_max_and_grad(x, alpha)
        x_new = projected_gradient_step(x, grad, lr, target_sum)
        obj_new, _ = smooth_max_and_grad(x_new, alpha)
        
        if obj_new < obj_prev:
            x = x_new
            obj_prev = obj_new
            # Increase step size slightly
            lr = min(lr * 1.1, 4.0 / L)  # cap at 4/L
        else:

def compute_B_j(h, j, n):
    """Compute the effect of changing h[j] on all correlations."""
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        indices = np.arange(k_start, k_end + 1)
        B[indices] += (1.0 - h[j - nm1 + indices])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        indices = np.arange(k_start2, k_end2 + 1)
        B[indices] -= h[j + nm1 - indices]
    return B

def solve_1d_max(A, D, lo, hi):
    """Minimize max(A + D*t) over t in [lo, hi]."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Coarse grid
    t_grid = np.linspace(lo, hi, 8)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(25):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=15, window=5):
    """One pass of pair coordinate descent."""
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    count = 0
    
    for j in order:
        B_j = compute_B_j(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Structured neighbors + random
        k_set = list(range(max(0, j - window), min(n, j + window + 1)))
        k_set += [(j + 1) % n, (j + n // 2) % n, (j + n // 3) % n]
        k_set += [rng.randint(n) for _ in range(max(0, n_cand - len(k_set)))]
        
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B_k(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_max(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
            count += 1
    
    return h, corr, improved, count

def block_swap_move(h, corr, rng, block_len=2):
    """Try to improve by swapping two blocks of values."""
    n = len(h)
    a = rng.randint(0, n - block_len)
    b = rng.randint(0, n - block_len)
    if abs(a - b) < block_len:
        return h, corr, False
    
    # Swap h[a:a+block_len] with h[b:b+block_len]
    # Preserve sum: swap values directly
    new_h = h.copy()
    new_h[a:a+block_len] = h[b:b+block_len]
    new_h[b:b+block_len] = h[a:a+block_len]
    
    new_corr = compute_corr(new_h)
    if np.max(new_corr) < np.max(corr) - 1e-12:
        return new_h, new_corr, True
    return h, corr, False
