Hmm, this is getting complicated and might be slow with Python loops. Let me think of a vectorized approach.

Actually, the gradient of corr[j] with respect to h is:
For corr computed as np.correlate(h, 1-h, mode='full'):
corr[t] = Σ_i h[i] * (1-h[i-(t-(n-1))]) for valid i

Let me denote lag = t - (n-1). Then:
corr[t] = Σ_i h[i] * (1-h[i-lag]) for valid i (i and i-lag both in [0,n-1])

If lag ≥ 0: valid i are [lag, n-1], so corr[t] = Σ_{i=lag}^{n-1} h[i](1-h[i-lag])
If lag < 0: valid i are [0, n-1+lag], so corr[t] = Σ_{i=0}^{n-1+lag} h[i](1-h[i-lag]) = Σ_{i=0}^{n-1+lag} h[i](1-h[i+|lag|])

In both cases, ∂corr[t]/∂h[a] involves h[a] appearing in two roles:
1. As the first factor: h[a] * (1-h[a-lag]) → derivative w.r.t. h[a] is (1-h[a-lag])
2. As the second factor: h[a+lag] * (1-h[a]) → derivative w.r.t. h[a] is -h[a+lag]

So the total gradient w.r.t. h[a] is:
(1-h[a-lag]) - h[a+lag] (for valid indices)

This can be computed efficiently! For the max over active lags:
g[a] = Σ_{t ∈ active} w[t] * [(1-h[a-lag_t]) - h[a+lag_t]]

But doing this for all a and all active t is still O(n * n_active). For n=720 and n_active=100, that's 72000 operations, which is fine.

Let me vectorize this:

For a given lag L ≥ 0:
- For a ∈ [L, n-1]: contribution from first role: (1 - h[a-L])
- For a ∈ [0, n-1-L]: contribution from second role: -h[a+L]

So the gradient vector for lag L is:
g_L[a] = (1-h[a-L]) for a ∈ [L, n-1], plus (-h[a+L]) for a ∈ [0, n-1-L]
       = 0 elsewhere

We can compute this with shifts:
g_L = zeros(n)
g_L[L:] += (1 - h[:n-L])
g_L[:n-L] -= h[L:]

For lag L < 0 (let L' = -L):
g_{-L'}[a]: 
- First role: h[a] * (1-h[a+L']), valid for a ∈ [0, n-1-L']
  derivative: (1-h[a+L'])
- Second role: h[a-L'] * (1-h[a]), valid for a ∈ [L', n-1]
  derivative: -h[a-L']

So:
g_{-L'}[:n-L'] += (1 - h[L':])
g_{-L'}[L':] -= h[:n-L']

Great, this is very efficient.

Now, for the optimization, I'll use:
1. Compute the correlation
2. Find the top K active lags
3. Compute the gradient from active lags (weighted by softmax weights)
4. Take a gradient step with line search
5. Project to feasible set

This is essentially a projected gradient method with focus on active constraints.

Let me also add a **momentum** term (Nesterov acceleration) for faster convergence.

And I'll add **adaptive step size** based on the curvature estimate.

from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, corr, dx, n_active, alpha):
    """Compute smoothed max gradient focused on active lags."""
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    
    # Select top n_active for efficiency
    if n_active < len(corr):
        top_idx = np.argsort(corr)[-n_active:]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()
        top_idx = top_idx[np.argsort(w_top)]  # sort by weight
    else:
        top_idx = np.arange(len(corr))
        w_top = w
    
    g = np.zeros(n)
    for idx, wt in zip(top_idx, w_top):
        if wt < 1e-10:
            continue
        lag = int(idx - (n - 1))
        if lag >= 0:
            g[lag:] += wt * (1.0 - h[:n - lag])
            g[:n - lag] -= wt * h[lag:]
        else:
            L = -lag
            g[:n - L] += wt * (1.0 - h[L:])
            g[L:] -= wt * h[:n - L]
    return g

def gradient_step(h, n, target_sum, alpha, n_active, lr, n_iters, time_fn):
    """Nesterov-accelerated projected gradient with active lag focus."""
    h_curr = h.copy()
    y = h.copy()
    t_val = 1.0
    h_best = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best
    
    for it in range(n_iters):
        if time_fn() < 10:
            break
        
        corr = compute_corr(y)
        g = active_gradient(y, corr, 2.0/n, n_active, alpha)
        
        # Nesterov step
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        
        # Backtracking line search
        step_lr = lr
        while c5_new > c5_cur + 1e-12 and step_lr > 1e-8:
            step_lr *= 0.5
            h_new = np.clip(y - step_lr * g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
        
        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h_curr)
        y = np.clip(y, 0.0, 1.0)
        
        h_curr = h_new
        t_val = t_new
        c5_cur = c5_new
        
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
        
        # Increase alpha over time (focus more on active lags)
        # alpha is updated by caller
    
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask) and abs(diff) <= 0.5 * np.sum(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, corr, alpha, n_active):
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    if n_active < len(corr):
        top_idx = np.argsort(corr)[-n_active:]
        w_top = w[top_idx].copy()
        w_top_sum = w_top.sum()
        if w_top_sum > 0:
            w_top /= w_top_sum
        else:
            w_top = np.ones(len(top_idx)) / len(top_idx)
        order = np.argsort(w_top)[::-1]
        top_idx = top_idx[order]
        w_top = w_top[order]
    else:
        top_idx = np.arange(len(corr))
        w_top = w
    g = np.zeros(n)
    for idx, wt in zip(top_idx, w_top):
        if wt < 1e-12:
            continue
        lag = int(idx - (n - 1))
        if lag >= 0:
            g[lag:] += wt * (1.0 - h[:n - lag])
            g[:n - lag] -= wt * h[lag:]
        else:
            L = -lag
            g[:n - L] += wt * (1.0 - h[L:])
            g[L:] -= wt * h[:n - L]
    return g

def nesterov_optimize(h0, n, target_sum, alpha, n_active, lr, n_iters, time_fn):
    h_curr = h0.copy()
    y = h0.copy()
    t_val = 1.0
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    c5_cur = c5_best
    for it in range(n_iters):
        if time_fn() < 5:
            break
        corr = compute_corr(y)
        g = active_gradient(y, corr, alpha, n_active)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        step_lr = lr
        while c5_new > c5_cur + 1e-14 and step_lr > 1e-10:
            step_lr *= 0.5
            if step_lr < 1e-10:
                break
            h_new = np.clip(y - step_lr * g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h_curr)
        y = np.clip(y, 0.0, 1.0)
        h_curr = h_new
        t_val = t_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.03, 0.8)
        elif it > 50:
            lr *= 0.95
    return h_best, c5_best

def coord_descent_focused(h0, n, target_sum, n_rounds, rng, n_vars_per_round):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr = compute_corr(h)
    for rnd in range(n_rounds):
        improved = False
        m = np.max(corr)
        active_threshold = m - 0.02 * m
        active_lags = np.where(corr >= active_threshold)[0]
        active_vars = set()
        for idx in active_lags:
            lag = int(idx - (n - 1))
            if lag >= 0:
                active_vars.update(range(0, n - lag))
            else:
                L = -lag
                active_vars.update(range(L, n))
        active_vars = list(active_vars)
        rng.shuffle(active_vars)
        extra = list(rng.choice(n, size=min(10, n), replace=False))
        all_vars = active_vars + extra
        seen = set()
        unique_vars = []
        for v in all_vars:
            if v not in seen:
                seen.add(v)
                unique_vars.append(v)
        for i in unique_vars[:n_vars_per_round]:
            lo = max(0.0, h[i] - 0.4)
            hi = min(1.0, h[i] + 0.4)
            if hi - lo < 1e-8:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-7})
                if res.fun < c5_cur - 1e-14:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def fourier_perturb(h, n, J, rng, scale):
    x = np.linspace(0, 2, n, endpoint=False)
    pert = np.zeros(n)
    for j in range(1, J + 1):
        a = rng.uniform(-scale, scale)
        b = rng.uniform(-scale, scale)
        pert += a * np.cos(j * np.pi * x) + b * np.sin(j * np.pi * x)
    h_new = h + pert
    h_new = project_to_sum(h_new, n / 2.0)
    return h_new
