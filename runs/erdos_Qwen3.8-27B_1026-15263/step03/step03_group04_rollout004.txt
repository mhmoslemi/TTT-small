        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -200, 0)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-12
    active = np.where(w > thresh)[0]
    for k in active:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def exact_subgradient(h, k_val):
    n = len(h)
    shift = k_val - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def backtracking_polish(h, n, target_sum, max_rounds=2000):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        active_mask = corr > max_corr - 1e-5
        active_ks = np.where(active_mask)[0]
        best_improve = 0.0
        best_h = h
        if len(active_ks) > 0:
            avg_grad = np.zeros(n)
            for k in active_ks:
                avg_grad += exact_subgradient(h, k)
            avg_grad /= len(active_ks)
            lr = 0.1
            for _ in range(25):
                h_new = h - lr * avg_grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-16:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                lr *= 0.5
                if lr < 1e-8:
                    break
        top_k = np.argsort(corr)[::-1][:6]
        for k in top_k:
            grad = exact_subgradient(h, k)
            lr = 0.05
            for _ in range(25):
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-16:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                lr *= 0.5
                if lr < 5e-9:
                    break
        if best_h is h:
            break
        h = best_h
        c5_cur = c5_cur - best_improve
    return h, c5_cur

def optimize_lbfgsb(h0, n, target_sum, alpha, maxiter=400):
    h = h0.copy()
    pen = 5000.0
    def fun_pen(x):
        obj, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return obj + 0.5 * pen * s_dev * s_dev
    def jac_pen(x):
        _, grad = smooth_max_obj_and_grad(x, alpha)
        s_dev = np.sum(x) - target_sum
        return grad + pen * s_dev * np.ones(n)
    try:
        res = minimize(fun_pen, h, method='L-BFGS-B', jac=jac_pen,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 100})
        h = res.x
    except Exception:
        pass
    return project_to_sum(h, target_sum)

def coordinate_descent_polish(h, n, target_sum, max_sweeps=5, rng=None):
    h = h.copy()
    c5_cur = compute_c5(h)
    deltas = [0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05]
    for sweep in range(max_sweeps):
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            best_val = c5_cur
            best_hi = h[i]
            for d in deltas:
                trial = h[i] + d
                if trial < 0 or trial > 1:
                    continue
                h_try = h.copy()
                h_try[i] = trial
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_val - 1e-14:
                    best_val = c5_try
                    best_hi = trial
            if abs(best_hi - h[i]) > 1e-10:
                h[i] = best_hi
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_local_search(h, n, target_sum, rng, max_rounds=150):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    guard = 0
    while h.sum() < target_sum - 0.5 and guard < 5000:
        idx = rng.choice(np.where(h < 1)[0])
        h[idx] = 1.0
        guard += 1
    guard = 0
    while h.sum() > target_sum + 0.5 and guard < 5000:
        idx = rng.choice(np.where(h > 0)[0])
        h[idx] = 0.0
        guard += 1
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        best_improve = 0.0
        best_pair = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(800, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_try = h.copy()
            h_try[i] = 0.0
            h_try[j] = 1.0
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - best_improve - 1e-14:
                best_improve = c5_cur - c5_try
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            h = project_to_sum(h, target_sum)
            c5_cur = compute_c5(h)
        else:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def remaining():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    ts = best_n / 2.0
    for a in [10, 50, 200, 500, 2000, 5000, 20000, 50000, 100000, 200000]:
        if remaining() < 25:
            break
        h_opt = optimize_lbfgsb(best_h, best_n, ts, a, maxiter=400)
        c5 = compute_c5(h_opt)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_opt.copy()
    if remaining() > 40:
        h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=3000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent_polish(best_h, best_n, ts, max_sweeps=5, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    for a in [50000, 100000, 300000]:
        if remaining() < 30:
            break
        h_opt = optimize_lbfgsb(best_h, best_n, ts, a, maxiter=500)
        c5 = compute_c5(h_opt)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_opt.copy()
        if remaining() > 40:
            h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=2000)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    if remaining() > 120:
        n2 = best_n * 2
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, ts2)
        for a in [50, 500, 5000, 20000, 50000]:
            if remaining() < 25:
                break
            h_opt = optimize_lbfgsb(h_interp, n2, ts2, a, maxiter=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n2
                ts = ts2
                h_interp = h_opt.copy()
        if remaining() > 40:
            h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=1000)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    if remaining() > 80:
        for n_try in [120, 160, 200]:
            if remaining() < 25:
                break
            ts_try = n_try / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_down = np.interp(x_new, x_old, best_h, period=2.0)
            h_down = project_to_sum(h_down, ts_try)
            for a in [100, 1000, 10000]:
                if remaining() < 20:
                    break
                h_opt = optimize_lbfgsb(h_down, n_try, ts_try, a, maxiter=200)
                c5 = compute_c5(h_opt)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n_try
                    ts = ts_try
                    h_down = h_opt.copy()
            if remaining() > 20:
                h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=500)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
    if remaining() > 50:
        h_bin, c5_bin = binary_local_search(best_h, best_n, ts, rng, max_rounds=100)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if remaining() > 30:
            h_opt = optimize_lbfgsb(best_h, best_n, ts, 5000, maxiter=300)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=1000)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    if remaining() > 60:
        n_best = best_n
        ts_best = n_best / 2.0
        x = np.linspace(0, 2, n_best, endpoint=False)
        for freq in [2, 3, 5, 7]:
            for amp in [0.35, 0.45]:
                for phase in [0, np.pi / 2, np.pi]:
                    if remaining() < 15:
                        break
                    s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                    h_s = project_to_sum(s, ts_best)
                    h_opt = optimize_lbfgsb(h_s, n_best, ts_best, 2000, maxiter=100)
                    c5 = compute_c5(h_opt)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h_opt.copy()
    if remaining() > 20:
        h_pol, c5_pol = backtracking_polish(best_h, best_n, ts, max_rounds=5000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_cd, c5_cd = coordinate_descent_polish(best_h, best_n, ts, max_sweeps=10, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
