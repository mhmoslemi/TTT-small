def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Fast gradient using FFT convolution
    # g = conv_w_omh[n-1:2n-1] - conv_w_h[n-1:2n-1]
    conv1 = fftconvolve(w, one_mh, mode='full')
    conv2 = fftconvolve(w, h, mode='full')
    g = conv1[n-1:2*n-1].copy() - conv2[n-1:2*n-1].copy()
    
    return sm_val, g

def grad_corr_single(h, k, n):
    """Gradient of corr[k] with respect to h."""
    s = k - n + 1  # shift
    g = np.zeros(n)
    # First term: (1-h[j-s]) for valid j-s
    idx1_min = max(0, s)  # j-s >= 0 → j >= s
    idx1_max = min(n-1, n-1+s)  # j-s <= n-1 → j <= n-1+s
    j1 = np.arange(idx1_min, idx1_max+1)
    g[j1] += (1.0 - h[j1-s])
    
    # Second term: -h[j+s] for valid j+s
    idx2_min = max(0, -s)  # j+s >= 0 → j >= -s
    idx2_max = min(n-1, n-1-s)  # j+s <= n-1 → j <= n-1-s
    j2 = np.arange(idx2_min, idx2_max+1)
    g[j2] -= h[j2+s]
    
    return g

def projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        _, g = smooth_max_and_grad(h, alpha)
        # Backtracking line search
        c5_cur = compute_c5(h)
        step = lr
        for _ in range(30):
            h_try = h - step * g
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
            step *= 0.5
        else:
            # No improvement, reduce lr
            lr *= 0.9
        if it % 10 == 0:
            alpha *= 1.5  # slowly increase alpha
    
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, base_delta, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-3.0, -2.0, -1.0, 0.5, 1.0, 2.0, 3.0]) * base_delta
    
    for rd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h[i] = np.clip(nv, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-14:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[i] = hi
                    h = project_to_sum(h, target_sum)
            # After trying all deltas, ensure h[i] is at best position
            h[i] = np.clip(hi, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
        if not improved:
            break
    
    return h, c5_cur
