    top_idx = np.argsort(corr)[-top_k:]
    return np.sort(top_idx)

def pnrm_grad(h, active_idx, p):
    """Compute gradient of p-norm of correlation values at active lags."""

    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, top_k=40):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    return np.sort(top_idx)

def pnrm_grad(h, active_idx, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    active_corr = corr[active_idx]
    active_corr = np.maximum(active_corr, 1e-15)
    weights = np.power(active_corr, p - 1)
    s = np.sum(np.power(active_corr, p))
    norm_factor = np.power(s, 1.0 / p - 1.0)
    g = np.zeros(n)
    for pos in range(len(active_idx)):
        w = p * weights[pos]
        if w < 1e-15:
            continue
        k = active_idx[pos] - (n - 1)
        if k > 0:
            n_terms = n - k
            g[0:n_terms] += w * (1.0 - h[k:n])
            g[k:n] -= w * h[0:n - k]
        elif k < 0:
            k2 = -k
            g[k2:] += w * (1.0 - h[0:n - k2])
            g[0:n - k2] -= w * h[k2:]
        else:
            g += w * (1.0 - 2.0 * h)
    g *= norm_factor
    return g

def pnrm_value(h, active_idx, p):

    corr = np.correlate(h, 1.0 - h, mode="full")
    active_corr = corr[active_idx]
    active_corr = np.maximum(active_corr, 1e-15)
    return float(np.sum(np.power(active_corr, p)) ** (1.0 / p))

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    g = np.zeros(n)
    for pos in range(len(corr)):
        wp = w[pos]
        if wp < 1e-15:
            continue
        k = pos - (n - 1)
        if k > 0:
            n_terms = n - k
            g[0:n_terms] += wp * (1.0 - h[k:n])
            g[k:n] -= wp * h[0:n - k]
        elif k < 0:
            k2 = -k
            g[k2:] += wp * (1.0 - h[0:n - k2])
            g[0:n - k2] -= wp * h[k2:]
        else:
            g += wp * (1.0 - 2.0 * h)
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, g
