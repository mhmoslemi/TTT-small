def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

    n_old = len(h)

    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        # Find top few lags
        top_k = np.argsort(corr)[::-1][:5]
        for lag_idx in top_k:

            can_dec = h > 1e-10
            if not np.any(can_inc) or not np.any(can_dec):
                break
            g_inc = np.where(can_inc, g, np.inf)

            h[i_dec] -= move

def de_mutation(pop, pop_size, F, rng):
    n = len(pop[0])
    mutant = np.zeros(n)
    while True:
        a = rng.randint(pop_size)
        b = rng.randint(pop_size)
        c = rng.randint(pop_size)
        if a != b and b != c and a != c:
            break
    mutant = pop[a] + F * (pop[b] - pop[c])
    return mutant

def de_crossover(target, mutant, CR, rng):
    n = len(target)
    trial = target.copy()
    r = rng.randint(n)
    for i in range(n):
        if i == r or rng.random() < CR:
            trial[i] = mutant[i]
    return trial

def slsqp_refine(h0, n, target_sum, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val + np.log(np.sum(exp_vals)) / alpha
        # Gradient
        n_x = len(x)
        grad = np.zeros(n_x)
        for lag_idx in range(len(corr)):
            if w[lag_idx] < 1e-10:
                continue
            lag = lag_idx - (n_x - 1)
            g = corr_gradient(x, lag, n_x)
            grad += w[lag_idx] * g
        return obj, grad
    
    def f(x):
        return obj_and_grad(x)[0]
    def g(x):
        return obj_and_grad(x)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

∇C₅_smooth = (dx/alpha) * ∑_k (alpha * exp(alpha*corr[k]) / ∑_j exp(alpha*corr[j])) * ∇corr[k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_at_lag(h, lag_idx, n):
    g = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    else:
        kk = -k
        if kk == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[kk:] += (1.0 - h[:n - kk])
            g[:n - kk] -= h[kk:]
    return g

def sparse_max_grad(h, top_k=5):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    top_indices = np.argsort(corr)[::-1][:top_k]
    shifted = top_k * (corr[top_indices] - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for idx in range(top_k):
        if w[idx] > 1e-12:
            grad += w[idx] * corr_gradient_at_lag(h, int(top_indices[idx]), n)
    obj = m_val + np.log(np.sum(exp_vals)) / top_k
    return obj, grad, m_val

def smooth_max_grad_full(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    active = w > 1e-10
    for idx in np.where(active)[0]:
        grad += w[idx] * corr_gradient_at_lag(h, int(idx), n)
    return obj, grad

def fast_coord_descent(h, target_sum, max_iters=20000, step_frac=0.15, top_k=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        top_indices = np.argsort(corr)[::-1][:top_k]
        improved = False
        for lag_idx in top_indices:
            g = corr_gradient_at_lag(h, int(lag_idx), n)
            can_inc = h < 1.0 - 1e-10
            can_dec = h > 1e-10
            if not np.any(can_inc) or not np.any(can_dec):
                break
            g_inc = np.where(can_inc, g, np.inf)
            g_dec = np.where(can_dec, g, -np.inf)
            i_inc = int(np.argmin(g_inc))
            i_dec = int(np.argmax(g_dec))
            if i_inc == i_dec:
                break
            if g[i_inc] >= g[i_dec] - 1e-12:
                break
            max_move = min(1.0 - h[i_inc], h[i_dec])
            move = step_frac * max_move
            if move < 1e-12:
                break
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new >= c5 - 1e-14:
                h[i_inc] -= move
                h[i_dec] += move
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                stale += 1
            else:
                c5 = c5_new
                stale = 0
                improved = True
        if stale > 50:
            break
    return h, c5

def slsqp_refine(h0, n, target_sum, alpha, maxiter, use_sparse=False, top_k=5):
    bounds = [(0.0, 1.0)] * n

    if use_sparse:
        def f(x):
            return sparse_max_grad(x, top_k)[0]
        def g(x):
            return sparse_max_grad(x, top_k)[1]
    else:
        def f(x):
            return smooth_max_grad_full(x, alpha)[0]
        def g(x):
            return smooth_max_grad_full(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def de_generation(pop, pop_size, F, CR, rng, target_sum):
    n = len(pop[0])
    for i in range(pop_size):
        while True:
            a = rng.randint(pop_size)
            b = rng.randint(pop_size)
            c = rng.randint(pop_size)
            if a != b and b != c and a != c:
                break
        mutant = pop[a] + F * (pop[b] - pop[c])
        trial = pop[i].copy()
        r = rng.randint(n)
        mask = rng.random(n) < CR
        mask[r] = True
        trial[mask] = mutant[mask]
        trial = project_to_sum(trial, target_sum)
    return trial

def run(seed=42, budget_s=1300, **kwargs):

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: Coordinate descent from initial
    if time_left() > 80:

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: SLSQP with sparse gradient at increasing alpha
    for top_k in [3, 5, 8]:
        if time_left() < 40:

        h_ref = slsqp_refine(best_h, n_parent, ts_parent, 0, maxiter=800, use_sparse=True, top_k=top_k)
        c5_ref = compute_c5(h_ref)

            best_h = h_ref.copy()

    pop_size = 20

    target_sum_de = n_de / 2.0

    pop_c5 = np.zeros(pop_size)
    pop[0] = best_h
    pop_c5[0] = best_c5

    for i in range(1, pop_size):
        noise_scale = 0.03 + 0.08 * (i / pop_size)
        h_try = best_h + rng.uniform(-noise_scale, noise_scale, n_de)
        h_try = project_to_sum(h_try, target_sum_de)

        pop_c5[i] = compute_c5(h_try)

            while True:
                a = rng.randint(pop_size)
                b = rng.randint(pop_size)
                c = rng.randint(pop_size)

            trial = project_to_sum(trial, target_sum_de)
            c5_trial = compute_c5(trial)
            if c5_trial < pop_c5[i]:

                pop_c5[i] = c5_trial

        # Refine best every 3 generations
        if gen % 3 == 0:
            best_idx = int(np.argmin(pop_c5))
            h_refine, c5_refine = fast_coord_descent(pop[best_idx].copy(), target_sum_de, max_iters=5000, step_frac=0.1, top_k=3)
            if c5_refine < pop_c5[best_idx]:

                pop_c5[best_idx] = c5_refine
            if c5_refine < best_c5:
                best_c5 = c5_refine
                best_h = h_refine.copy()
            if pop_c5.min() < best_before - 1e-8:

            else:
                de_stagnation += 1

            worst_idx = int(np.argmax(pop_c5))
