```python
def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.asarray(h, dtype=float)
    # Binary search for threshold
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0, 1)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # For numerical stability
    if np.max(shifted) > 50:
        shifted = shifted - np.max(shifted)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    w = exp_v / s
    obj = (m + np.log(s) / alpha)
    # Gradient
    grad = np.zeros(2 * n - 1)  # temp, we'll map to n
    grad_h = np.zeros(n)
    thresh = 1e-12 / s
    active = np.where(w > thresh)[0]
    for k in active:
        wi = w[k]
        shift = k - (n - 1)
        # d(corr[k])/dh[i] = (1-h[i+shift])*I(i+shift in [0,n)) - h[i-shift]*I(i-shift in [0,n))
        # i+shift in [0,n) means i in [-shift, n-shift)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad_h[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        # i-shift in [0,n) means i in [shift, n+shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad_h[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad_h

def projected_gradient_descent(h0, n, target_sum, alpha, n_iter=1000, lr=None):
    h = h0.copy()
    m = np.zeros(n)  # momentum
    v = np.zeros(n)  # velocity
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta = 0.9  # momentum
    
    for it in range(n_iter):
        obj, grad = smooth_max_and_grad(h, alpha)
        if lr is None:
            # Adaptive step size
            gnorm = np.sqrt(np.sum(grad**2))
            if gnorm < 1e-14:
                break
            lr = 0.5 / (alpha * gnorm)
        
        v = beta * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Don't accept, reduce velocity
            v *= 0.5
    
    return best_h, best_c5

def projected_gradient_descent_simple(h0, n, target_sum, alpha, n_iter=2000, lr_init=0.01):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iter):
        obj, grad = smooth_max_and_grad(h, alpha)
        gnorm = np.sqrt(np.sum(grad**2))
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            lr = min(lr * 1.1, 0.5 / max(alpha * gnorm, 1e-10))
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
    
    return best_h, best_c5

def multi_alpha_optimize(h0, n, target_sum, alphas, iters_per_alpha):
    h = h0.copy()
    best_c5 = compute_c5(h)
    for a in alphas:
        h_opt, c5_opt = projected_gradient_descent_simple(h, n, target_sum, a, n_iter=iters_per_alpha)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            h = h_opt
    return h, best_c5

def exact_max_polish(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        # Get top few shifts
        top_k = np.argsort(corr)[::-1][:min(10, len(corr))]
        best_improve = 0
        best_h = None
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                h_new = h - lr * grad
                h_new = project_feasible(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-14:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur

def binary_local_search(h, n, target_sum, n_tries=10000):
    """Local search on binary solutions"""
    rng = np.random.RandomState(12345)
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    attempts = 0
    while abs(h_bin.sum() - target_sum) > 1e-10 and attempts < 50000:
        if h_bin.sum() > target_sum:
            idx = rng.choice(np.where(h_bin > 0)[0])
            h_bin[idx] = 0.0
        else:
            idx = rng.choice(np.where(h_bin < 1)[0])
            h_bin[idx] = 1.0
        attempts += 1
    if abs(h_bin.sum() - target_sum) > 1e-10:
        return h, compute_c5(h)
    
    c5_cur = compute_c5(h_bin)
    for _ in range(n_tries):
        # Pick a random 1 to flip to 0, and a random 0 to flip to 1
        ones = np.where(h_bin > 0)[0]
        zeros = np.where(h_bin < 1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5_cur

    for freq in range(1, 15):

                s = 0.5 + amp * np.sin(np.pi * freq * x + phase)
                starts.append(project_feasible(s, target_sum))
    
    # Block patterns
    for nblks in range(2, 20):
        blk = n // (2 * nblks)
        if blk < 1:

        for offset in range(blk):

    # Random
    for _ in range(3):
        starts.append(project_feasible(rng.uniform(0, 1, n), target_sum))
    
    return starts
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    if np.max(shifted) > 50:
        shifted = shifted - np.max(shifted)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad_h = np.zeros(n)
    thresh = 1e-12
    active = np.where(w > thresh)[0]
    for k in active:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad_h[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad_h[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad_h

def coordinate_descent(h, n, target_sum, alpha, n_sweeps=5, random_order=False, seed=0):
    """Cyclic/random coordinate descent on smooth max of correlations."""
    h = h.copy()
    rng = np.random.RandomState(seed) if random_order else None
    corr = compute_corr(h)
    
    for sweep in range(n_sweeps):
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            # Compute delta: change in corr[k] when h[i] goes from 0 to 1
            delta = np.zeros(2 * n - 1)
            # For shift s = k - (n-1):
            # delta[k] = (1 - h[i+s]) if 0<=i+s<n else 0, minus h[i-s] if 0<=i-s<n else 0
            s_lo = max(0, -i)  # shift s such that i+s >= 0 → s >= -i
            s_hi = min(n - 1 - i, n - 1)  # i+s <= n-1 → s <= n-1-i
            for s in range(s_lo, s_hi + 1):
                k = s + (n - 1)
                delta[k] += (1.0 - h[i + s])
            
            s_lo2 = max(0, i - (n - 1))  # i-s >= 0 → s <= i
            s_hi2 = min(i, n - 1)  # i-s <= n-1 → s >= i-(n-1)
            for s in range(s_lo2, s_hi2 + 1):
                k = s + (n - 1)
                delta[k] -= h[i - s]
            
            # Now minimize 1D: (1/alpha)*log(sum exp(alpha*(corr[k] + delta[k]*t))) for t in [0,1]
            # Since h[i] is currently some value, the current corr already includes h[i]'s contribution
            # corr_current[k] = corr_base[k] + delta[k]*h[i]
            # We want to find t in [0,1] minimizing (1/alpha)*log(sum exp(alpha*(corr_base[k] + delta[k]*t)))
            # where corr_base[k] = corr_current[k] - delta[k]*h[i]
            
            hi_current = h[i]
            corr_base = corr - delta * hi_current
            
            # Bisection on derivative
            # d/dt = sum delta[k] * exp(alpha*(corr_base[k]+delta[k]*t)) / sum exp(...)
            # = E_t[delta] where P_t(k) ∝ exp(alpha*(corr_base[k]+delta[k]*t))
            
            # Check endpoints first
            c5_at_0 = np.max(corr_base)  # since exp(alpha*(corr-m)) >= exp(0)=1 for corr>=m
            c5_at_1 = np.max(corr_base + delta)
            
            if c5_at_0 <= c5_at_1:
                # Minimum is at t=0 (or near it for smooth max)
                t_new = 0.0
            elif c5_at_1 < c5_at_0:
                t_new = 1.0
            else:
                # Interior minimum - use bisection
                lo_t, hi_t = 0.0, 1.0
                for _ in range(40):
                    mid_t = 0.5 * (lo_t + hi_t)
                    vals = corr_base + delta * mid_t
                    m_v = np.max(vals)
                    exp_v = np.exp(alpha * (vals - m_v))
                    w_v = exp_v / exp_v.sum()
                    dval = np.dot(delta, w_v)
                    if dval > 0:
                        hi_t = mid_t
                    else:
                        lo_t = mid_t
                t_new = 0.5 * (lo_t + hi_t)
            
            h[i] = t_new
            corr = corr_base + delta * t_new
        
        h = project_feasible(h, target_sum)
        corr = compute_corr(h)
    
    return h

def projected_gradient_simple(h0, n, target_sum, alpha, n_iter=2000, lr_init=0.1):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iter):
        obj, grad = smooth_max_and_grad(h, alpha)
        gnorm = np.sqrt(np.sum(grad**2))
        if gnorm < 1e-15:
            break
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            lr = min(lr * 1.05, 0.5 / max(gnorm, 1e-10))
        else:
            lr *= 0.6
            if lr < 1e-12:
                break
    
    return best_h, best_c5

def exact_max_polish(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(10, len(corr))]
        best_improve = 0
        best_h = None
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
                h_new = h - lr * grad
                h_new = project_feasible(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - best_improve - 1e-14:
                    best_improve = c5_cur - c5_new
                    best_h = h_new.copy()
                    best_c5_new = c5_new
        if best_h is None:
            break
        h = best_h
        c5_cur = best_c5_new
    return h, c5_cur

def binary_local_search(h, n, target_sum, n_tries=5000, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = (h > 0.5).astype(float)
    attempts = 0
    while abs(h_bin.sum() - target_sum) > 1e-10 and attempts < 50000:
        if h_bin.sum() > target_sum:
            idx = rng.choice(np.where(h_bin > 0)[0])
            h_bin[idx] = 0.0
        else:
            idx = rng.choice(np.where(h_bin < 1)[0])
            h_bin[idx] = 1.0
        attempts += 1
    if abs(h_bin.sum() - target_sum) > 1e-10:
        return h, compute_c5(h)
    
    c5_cur = compute_c5(h_bin)
    for _ in range(n_tries):
        ones = np.where(h_bin > 0)[0]
        zeros = np.where(h_bin < 1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5_cur

def make_starts(n, target_sum, seed, best_h=None):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    
    if best_h is not None:
        if len(best_h) == n:
            starts.append(best_h.copy())
        else:
            starts.append(project_feasible(
                np.interp(x, np.linspace(0, 2, len(best_h), endpoint=False), best_h, period=2.0),
                target_sum))
    
    for freq in range(1, 12):
        for amp in [0.3, 0.4, 0.45, 0.48]:
            for phase in np.linspace(0, 2*np.pi, 4, endpoint=False):
                s = 0.5 + amp * np.sin(np.pi * freq * x + phase)
                starts.append(project_feasible(s, target_sum))
    
    for nblks in range(2, 16):
        total_blk = 2 * nblks
        blk = n // total_blk
        if blk < 1:
            continue
        for offset in range(min(blk, 3)):
            h = np.zeros(n)
            for i in range(offset, n, total_blk):
                h[i:min(i + blk, n)] = 1.0
            starts.append(project_feasible(h, target_sum))
    
    for _ in range(3):
        starts.append(project_feasible(rng.uniform(0, 1, n), target_sum))
    
    return starts
```
