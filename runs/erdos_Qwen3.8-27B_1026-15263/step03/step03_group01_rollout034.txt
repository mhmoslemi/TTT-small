```python
    corr = np.correlate(h, 1.0 - h, mode="full")

    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    g = np.zeros(n)
    for k in range(2*n-1):

        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        # Part 1: g[j] += wk * (1-h[j+offset]) for 0<=j<n and 0<=j+offset<n
        if offset >= 0:

            g[:n-offset] += wk * (1.0 - h[offset:])
        else:
            # j+offset >= 0 => j >= -offset, and j < n
            g[-offset:] += wk * (1.0 - h[:n+offset])

        else:
            # j >= 0 (since offset < 0, j >= offset is automatic) and j < offset+n = n+offset
            g[:n+offset] -= wk * h[-offset:]
    
    return obj, g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            g[:n-offset] += wk * (1.0 - h[offset:])
        else:
            g[-offset:] += wk * (1.0 - h[:n+offset])
        if offset >= 0:
            g[offset:] -= wk * h[:n-offset]
        else:
            g[:n+offset] -= wk * h[-offset:]
    return obj, g

def slsqp_optimize(h0, n, ts, alpha, maxiter=500):
    def fun(x):
        return smooth_max_and_grad(x, alpha)[0]
    def grad(x):
        return smooth_max_and_grad(x, alpha)[1]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(fun, h0.copy(), jac=grad, method='SLSQP',
                          constraints=constraints, bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        return project_to_sum(result.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def subgradient_step(h, n, ts, step_size, n_active=5):
    """Take a subgradient step on the exact max of corr."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top n_active values
    top_idx = np.argsort(corr)[-n_active:]
    w = np.zeros(len(corr))
    w[top_idx] = 1.0 / len(top_idx)
    
    # Compute subgradient (same as smooth max grad but with uniform weights on top lags)
    g = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        offset = k - (n - 1)
        if offset >= 0:
            g[:n-offset] += wk * (1.0 - h[offset:])
        else:
            g[-offset:] += wk * (1.0 - h[:n+offset])
        if offset >= 0:
            g[offset:] -= wk * h[:n-offset]
        else:
            g[:n+offset] -= wk * h[-offset:]
    
    h_new = h - step_size * g
    return project_to_sum(h_new, ts)

def subgradient_optimize(h0, n, ts, n_iter=5000, init_step=0.01):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = init_step
    stall = 0
    for i in range(n_iter):
        h_new = subgradient_step(h, n, ts, step)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.1, 0.1)
            stall = 0
        elif c5_new < best_c5:
            h = h_new
            stall = 0
        else:
            step *= 0.7
            stall += 1
```

**Alternative: Use the fact that the problem is an instance of a max-min problem**

We want to minimize max_k f_k(h) subject to constraints, where f_k(h) = corr_k * dx.

This is equivalent to: find the minimum t such that there exists h with:
- f_k(h) ≤ t for all k
- sum(h) = n/2
- 0 ≤ h ≤ 1

This is a non-convex feasibility problem (because f_k is quadratic in h).

However, we can use a bisection approach: for a given t, check if the feasibility problem is solvable. But checking feasibility of a non-convex problem is NP-hard in general.

**Alternative: Use a penalty/barrier approach with L-BFGS-B**

Instead of SLSQP, use L-BFGS-B with:
- Objective: smooth_max(h, alpha) + mu * (sum(h) - n/2)²
- mu is a large penalty parameter
- L-BFGS-B handles the box constraints natively

This should be faster than SLSQP because:
- L-BFGS-B uses quasi-Newton (BFGS) approximation, which is faster than SLSQP's SQP for large problems
- No need to form the constraint Jacobian

Let me add this as an alternative optimizer:

```python
def lbfgsb_optimize(h0, n, ts, alpha, maxiter=1000, mu=1000):
    def fun(x):
        obj, _ = smooth_max_and_grad(x, alpha)
        penalty = mu * (np.sum(x) - ts) ** 2
        return obj + penalty
    
    def grad(x):
        _, g = smooth_max_and_grad(x, alpha)
        g += 2 * mu * (np.sum(x) - ts) * np.ones(n)
        return g

        result = minimize(fun, h0.copy(), jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})

    except Exception:
        return project_to_sum(h0, ts)
```

Let me also think about what makes the optimum hard to find. The key challenge is that the max is taken over O(n) different lags, and the optimum requires balancing all of them. The smooth max with large alpha approximates this well, but the optimization landscape is very flat in most directions (only the directions that affect the worst lags matter).

A key insight: at the optimum, the top few lags should all be equal (by the KKT conditions for the max). So a good strategy is:
1. Find the current worst lag(s)
2. Take a step that reduces the worst lag while not increasing the other top lags too much
3. Repeat

This is essentially what the subgradient method does, but we can make it more efficient by:
- Using the top k lags (not just the top 1)
- Using a smart step size (backtracking line search)

```python
def smart_subgradient_optimize(h0, n, ts, n_iter=5000, n_active=5, init_step=0.005):
    """Subgradient optimization with backtracking on the exact max."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = init_step
    stall = 0
    
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top n_active lags
        top_idx = np.argsort(corr)[-n_active:]
        
        # Compute subgradient (average of gradients at top lags)
        g = np.zeros(n)
        for k in top_idx:
            wk = 1.0 / n_active
            offset = k - (n - 1)
            if offset >= 0:
                g[:n-offset] += wk * (1.0 - h[offset:])
            else:
                g[-offset:] += wk * (1.0 - h[:n+offset])
            if offset >= 0:
                g[offset:] -= wk * h[:n-offset]
            else:
                g[:n+offset] -= wk * h[-offset:]
        
        # Backtracking line search
        c5_cur = float(np.max(corr) * 2.0 / n)
        accepted = False
        for trial in range(10):
            s = step * (0.5 ** trial)
            h_new = project_to_sum(h - s * g, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                step = s * 1.5  # Increase step if we succeeded
                step = min(step, 0.1)
                accepted = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                stall = 0
                break
        if not accepted:
            step *= 0.5
            stall += 1
            if stall > 50 or step < 1e-12:
                break
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            g[:n-offset] += wk * (1.0 - h[offset:])
            g[offset:] -= wk * h[:n-offset]
        else:
            g[-offset:] += wk * (1.0 - h[:n+offset])
            g[:n+offset] -= wk * h[-offset:]
    return obj, g

def slsqp_solve(h0, ts, alpha, maxiter):
    n = len(h0)
    def fun(x):
        return smooth_max_and_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(fun, h0.copy(), jac=jac, method='SLSQP',
                     constraints=cons, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0, ts)

def lbfgs_solve(h0, ts, alpha, maxiter, mu=500.0):
    n = len(h0)
    def fun(x):
        obj, _ = smooth_max_and_grad(x, alpha)
        return obj + mu * (np.sum(x) - ts) ** 2
    def jac(x):
        _, g = smooth_max_and_grad(x, alpha)
        g = g + 2.0 * mu * (np.sum(x) - ts)
        return g
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(fun, h0.copy(), jac=jac, method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0, ts)

            else:
