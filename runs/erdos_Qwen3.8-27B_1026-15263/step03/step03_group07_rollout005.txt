import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, frac=0.99):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    threshold = frac * m
    active = np.where(corr >= threshold)[0]
    return active

def smooth_max_active_grad_vec(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for i, k in enumerate(active_idx):
        wi = w[i]
        if wi < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            g[:n-d] += wi * (1.0 - h[d:])
            g[d:] -= wi * h[:n-d]
        else:
            d2 = -d
            g[d2:] += wi * (1.0 - h[:d2])
            g[:d2] -= wi * h[d2:]
    return sm_val, g

def slsqp_active(h0, n, target_sum, active_idx, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_active_grad_vec(h, active_idx, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_active_grad_vec(h, active_idx, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_gradient_descent(h0, n, target_sum, active_idx, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_active_grad_vec(h, active_idx, alpha)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, n_try=3):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            # Try several step sizes
            for scale in [0.01, 0.005, 0.002]:
                for sign in [-1, 1]:
                    d = sign * scale
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_val - 1e-14:
                        best_val = c5_new
                        best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def perturb_and_optimize(h0, n, target_sum, rng, alpha_list, maxiter_list, noise_scales):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for noise in noise_scales:
        h_pert = h0.copy()
        h_pert += rng.normal(0, noise, size=n)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = project_to_sum(h_pert, target_sum)
        c5_pert = compute_c5(h_pert)

        for i, alpha in enumerate(alpha_list):

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_pert = h_opt
            active = get_active_lags(h_pert, 0.99)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, frac=0.99):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m <= 0:
        return np.array([n - 1])
    threshold = frac * m
    active = np.where(corr >= threshold)[0]
    return active

def smooth_max_active_grad_vec(h, active_idx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_corr = corr[active_idx]
    m = np.max(active_corr)
    shifted = alpha * (active_corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for i in range(len(active_idx)):
        wi = float(w[i])
        if wi < 1e-15:
            continue
        k = int(active_idx[i])
        d = k - (n - 1)
        if d >= 0:
            g[:n - d] += wi * (1.0 - h[d:])
            g[d:] -= wi * h[:n - d]
        else:
            d2 = -d
            g[d2:] += wi * (1.0 - h[:n - d2])
            g[:n - d2] -= wi * h[d2:]
    return sm_val, g

def slsqp_solve(h0, n, target_sum, active_idx, alpha, maxiter):
    def obj_func(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_active_grad_vec(h, active_idx, alpha)
        return v
    def grad_func(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_active_grad_vec(h, active_idx, alpha)
        return g
    cons_fun = lambda x: np.sum(x) - target_sum
    cons_jac = lambda x: np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbfgs_solve(h0, n, target_sum, active_idx, alpha, maxiter, pen):
    def obj_grad_func(x):
        h = np.clip(x, 0.0, 1.0)
        v, g = smooth_max_active_grad_vec(h, active_idx, alpha)
        sv = np.sum(h) - target_sum
        g = g + pen * sv * np.ones(n)
        return v + 0.5 * pen * sv * sv, g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad_func, h0.copy(), jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_grad_desc(h0, n, target_sum, active_idx, alpha, n_iters, lr):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        _, g = smooth_max_active_grad_vec(h, active_idx, alpha)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.6
            if lr < 1e-9:
                break
    return h_best, c5_cur

def coord_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    steps = [0.008, 0.004, 0.002, 0.001]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for s in steps:
                for sg in [-1, 1]:
                    d = sg * s
                    hn = h.copy()
                    hn[i] = np.clip(hn[i] + d, 0.0, 1.0)
                    hn = project_to_sum(hn, target_sum)
                    cv = compute_c5(hn)
                    if cv < best_val - 1e-14:
                        best_val = cv
                        best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0
    h_cur = best_h.copy()

    if time_left() > 80:
        active = get_active_lags(h_cur, 0.99)
        for alpha in [80.0, 200.0, 500.0, 1000.0]:
            if time_left() < 25:
                break
            mi = min(250, max(40, int(time_left() * 0.12)))
            h_opt = lbfgs_solve(h_cur, n, ts, active, alpha, mi, 50.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            h_cur = h_opt
            active = get_active_lags(h_cur, 0.99)

    if time_left() > 60:
        active = get_active_lags(h_cur, 0.985)
        for alpha in [2000.0, 5000.0, 10000.0]:
            if time_left() < 25:
                break
            hi = min(300, max(50, int(time_left() * 0.15)))
            h_opt = slsqp_solve(h_cur, n, ts, active, alpha, hi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            h_cur = h_opt
            active = get_active_lags(h_cur, 0.985)

    if time_left() > 50:
        active = get_active_lags(h_cur, 0.99)
        n_iters = min(3000, int(time_left() * 3))
        h_pg, c5_pg = projected_grad_desc(h_cur, n, ts, active, 2000.0, n_iters, 0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n
        h_cur = h_pg

    if time_left() > 80:
        for attempt in range(4):
            if time_left() < 30:
                break
            noise = 0.015 * (attempt + 1)
            h_pert = h_cur + rng.normal(0, noise, size=n)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_to_sum(h_pert, ts)
            active_p = get_active_lags(h_pert, 0.99)
            for alpha in [1500.0, 4000.0]:
                if time_left() < 20:
                    break
                mi = min(200, max(40, int(time_left() * 0.1)))
                h_opt = slsqp_solve(h_pert, n, ts, active_p, alpha, mi)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_pert = h_opt
                active_p = get_active_lags(h_pert, 0.99)
            h_cur = h_pert

    if time_left() > 40:
        h_cd, c5_cd = coord_descent(h_cur, n, ts, 40, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        h_cur = h_cd

    if time_left() > 100:
        n_hi = min(n * 2, 1440)
        if n_hi > n:
            ts_hi = n_hi / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_cur)
            h_hi = project_to_sum(h_hi, ts_hi)
            active_hi = get_active_lags(h_hi, 0.99)
            for alpha in [500.0, 2000.0, 5000.0]:
                if time_left() < 25:
                    break
                mi = min(250, max(40, int(time_left() * 0.12)))
                h_opt = lbfgs_solve(h_hi, n_hi, ts_hi, active_hi, alpha, mi, 80.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_hi
                h_hi = h_opt
                active_hi = get_active_lags(h_hi, 0.99)
            if time_left() > 30:
                active_hi2 = get_active_lags(h_hi, 0.985)
                h_opt2 = slsqp_solve(h_hi, n_hi, ts_hi, active_hi2, 8000.0, 200)
                c5_opt2 = compute_c5(h_opt2)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n_hi
                h_hi = h_opt2
            if time_left() > 25:
                active_hi3 = get_active_lags(h_hi, 0.99)
                n_it2 = min(2000, int(time_left() * 2))
                h_pg2, c5_pg2 = projected_grad_desc(h_hi, n_hi, ts_hi, active_hi3, 3000.0, n_it2, 0.015)
                if c5_pg2 < best_c5:
                    best_c5 = c5_pg2
                    best_h = h_pg2.copy()
                    best_n = n_hi

    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        active_f = get_active_lags(h_f, 0.995)
        for alpha in [8000.0, 15000.0]:
            if time_left() < 20:
                break
            h_opt = slsqp_solve(h_f, n_f, ts_f, active_f, alpha, 250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_f
            h_f = h_opt
            active_f = get_active_lags(h_f, 0.995)

    if best_h is not None and time_left() > 20:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        h_cd_f, c5_cd_f = coord_descent(h_f, n_f, ts_f, 60, rng)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()
            best_n = n_f

    if best_h is not None and time_left() > 15:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        active_f = get_active_lags(h_f, 0.99)
        n_it3 = min(2000, int(time_left() * 2))
        h_pg3, c5_pg3 = projected_grad_desc(h_f, n_f, ts_f, active_f, 5000.0, n_it3, 0.01)
        if c5_pg3 < best_c5:
            best_c5 = c5_pg3
            best_h = h_pg3.copy()
            best_n = n_f

    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
