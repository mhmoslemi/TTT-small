        grad = exact_max_grad(h, n, epsilon=eps)
        step = base_step / (1.0 + 0.01 * it)
        h_new = h - step * grad
        h_new = project_polytope(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    return best_h, best_c5


def exact_max_line_search(h, n, target_sum, max_rounds=800):
    h = h.copy()
    c5_cur = compute_c5(h)
    step_sizes = [0.0001, 0.0003, 0.001, 0.002, 0.003, 0.005, 0.008, 0.01,
                  0.015, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]
    for r in range(max_rounds):
        corr = compute_corr(h)
        m = np.max(corr)
        active = np.where(corr >= m - 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        top_active = active[:min(12, len(active))]
        best_improve = 0.0
        best_h = h
        best_c5 = c5_cur
        for k_max in top_active:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            for lr in step_sizes:
                h_new = h - lr * grad
                h_new = project_polytope(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_improve = best_c5 - c5_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
        if best_h is h or best_improve < 1e-15:
            break
        h = best_h
        c5_cur = best_c5
    return h, c5_cur


def sa_binary(h_bin, n, t0, t_end, iterations, rng, time_limit=None, start_time=None):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for it in range(iterations):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        T = max(t0 * (t_end / t0) ** (it / max(iterations, 1)), 1e-12)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


def local_search_binary(h_bin, n, rng, max_rounds=300, n_trials=1000):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        nt = min(n_trials, len(ones) * len(zeros))
        for _ in range(nt):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, c5_cur


def make_binary_from_h(h, n):
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin


def make_symmetric_h(n, structure_indices):
    h = np.zeros(n)
    half = n // 2
    for i in structure_indices:
        h[i] = 1.0
        if n - 1 - i != i:
            h[n - 1 - i] = 1.0
    h = project_to_sum(h, n / 2.0)
    h_bin = make_binary_from_h(h, n)
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    time_start = time.time()

    def time_elapsed():
        return time.time() - time_start

    def remaining():
        return budget_s - time_elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    target = best_n / 2.0
    h_bin = make_binary_from_h(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if remaining() > 120:
        configs = [(1.0, 0.001, 8000), (0.5, 0.0005, 10000), (0.3, 0.0001, 12000)]
        for t0_sa, t_end, iters in configs:
            if remaining() < 40:
                break
            tl = min(remaining() - 20, 100)
            if tl < 5:
                break
            h_sa, c5_sa = sa_binary(h_bin, best_n, t0_sa, t_end, iters, rng,
                                    time_limit=tl, start_time=time.time())
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h_bin = h_sa.copy()

    if remaining() > 60:
        h_ls, c5_ls = local_search_binary(h_bin, best_n, rng, max_rounds=300, n_trials=2000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h_bin = h_ls.copy()

    if remaining() > 90:
        for alpha in [50, 200, 1000, 5000, 20000]:
            if remaining() < 30:
                break
            h_opt = optimize_lbfgs_multi_penalty(best_h, best_n, target, alpha, 200,
                                                  penalties=[5000, 50000, 500000])
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if remaining() > 80:
        h_pol, c5_pol = exact_max_line_search(best_h, best_n, target, max_rounds=600)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if remaining() > 200:
        for cycle in range(15):
            if remaining() < 60:
                break
            K = rng.randint(3, 12)
            h_pert = h_bin.copy()
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
            for _ in range(K):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert > 0.5)[0]
                zeros = np.where(h_pert < 0.5)[0]
            h_ls2, c5_ls2 = local_search_binary(h_pert, best_n, rng, max_rounds=200, n_trials=1500)
            if remaining() > 40:
                tl = min(20, remaining() - 20)
                if tl > 5:
                    h_sa2, c5_sa2 = sa_binary(h_pert, best_n, 0.3, 0.0005, 5000, rng,
                                              time_limit=tl, start_time=time.time())
                    if c5_sa2 < c5_ls2:
                        h_ls2, c5_ls2 = h_sa2, c5_sa2
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
                h_bin = h_ls2.copy()
            if remaining() > 30:
                h_opt = optimize_lbfgs(best_h, best_n, target, 5000, 150, penalty=100000)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                h_pol, c5_pol = exact_max_line_search(best_h, best_n, target, max_rounds=100)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    if remaining() > 30:
        h_sg, c5_sg = subgrad_detailed(best_h, best_n, target, n_iter=1000, base_step=0.02)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    if remaining() > 100:
        n2 = best_n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_to_sum(h_interp, target2)
        for alpha in [100, 500, 2000, 10000]:
            if remaining() < 50:
                break
            h_opt = optimize_lbfgs_multi_penalty(h_interp, n2, target2, alpha, 150,
                                                  penalties=[5000, 50000])
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n2
                target = target2
                h_interp = h_opt
        if remaining() > 50:
            h_pol, c5_pol = exact_max_line_search(best_h, best_n, target, max_rounds=400)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if remaining() > 40:
        h_sg2, c5_sg2 = subgrad_detailed(best_h, best_n, target, n_iter=1500, base_step=0.01)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    if remaining() > 30:
        h_pol, c5_pol = exact_max_line_search(best_h, best_n, target, max_rounds=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if remaining() > 20:
        for alpha in [10000, 50000]:
            if remaining() < 10:
                break
            h_opt = optimize_lbfgs(best_h, best_n, target, alpha, 200, penalty=100000)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_pol, c5_pol = exact_max_line_search(best_h, best_n, target, max_rounds=200)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    best_h = project_polytope(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
