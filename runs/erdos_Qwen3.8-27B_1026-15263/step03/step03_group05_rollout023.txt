import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    """Compute smooth max value and correct gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * dx
    
    # Correct gradient using O(n^2) loop
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        s = t - (n - 1)
        # First term: (1-h[j+s]) for valid j
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wt * (1.0 - h[j_lo + s : j_hi + s])
        # Second term: -h[j-s] for valid j
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wt * h[j_lo2 - s : j_hi2 - s]
    grad *= dx
    return obj, grad

def pd_solve(h0, n, alphas, max_iter_total, time_budget_fn, lr_base=0.1):
    """Projected gradient descent with alpha continuation and correct gradient."""
    h = h0.copy()
    best_obj = smooth_max_val_grad(h, alphas[0])[0]
    best_h = h.copy()
    iters_per_alpha = max_iter_total // len(alphas)
    
    for alpha in alphas:
        if time_budget_fn() < 30:
            break
        lr = lr_base / (1.0 + alpha * 0.01)
        for it in range(iters_per_alpha):
            if time_budget_fn() < 10:
                break
            obj, grad = smooth_max_val_grad(h, alpha)
            # Backtracking line search
            step = lr
            for _ in range(10):
                h_new = project_box_sum(h - step * grad, n / 2.0)
                obj_new = smooth_max_val_grad(h_new, alpha)[0]
                if obj_new < obj:
                    break
                step *= 0.5
            else:
                h_new = h  # no improvement
                obj_new = obj
            h = h_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
    return best_h

def local_lag_search(h, rng, max_rounds=200, time_budget_fn=None):
    """Focus on reducing the maximum correlation lag."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    c5_cur = compute_c5(h)
    
    for _ in range(max_rounds):
        if time_budget_fn and time_budget_fn() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        t_star = np.argmax(corr)
        s_star = t_star - (n - 1)
        
        # Find positions where we can flip to reduce corr[t_star]
        # corr[t_star] = sum_i h[i] * (1-h[i+s_star])
        # To reduce it: change h[i] from 1 to 0 where h[i+s_star]=0, 
        #              or change h[i+s_star] from 0 to 1 where h[i]=1
        # But we need to maintain sum constraint (swap a 1 with a 0)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = c5_cur
        best_swap = None
        
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            if c < best_improvement - 1e-12:
                best_improvement = c
                best_swap = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = best_improvement
    
    return h, c5_cur

            best_h = h_new.copy()
    
    # Phase 2: Higher resolution
    if time_left() > 150:
        n2 = min(480, best_n * 2)

        x_old = np.linspace(0, 2, best_n, endpoint=False)

        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = project_box_sum(h2, n2 / 2.0)

        h2_new = pd_solve(h2, n2, alphas2, 8000, time_left, lr_base=0.03)
        c5_2 = compute_c5(h2_new)
        if c5_2 < best_c5:
