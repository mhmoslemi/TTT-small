    H = np.fft.fft(h)
    # Keep only the first n_fourier modes (and their conjugates)
    mask = np.zeros(n, dtype=bool)
    for k in range(n_fourier + 1):
        mask[k] = True
        mask[n-k] = True
    H[~mask] = 0
    return np.real(np.fft.ifft(H))

One natural candidate is h(x) = 1 on [0, 1/2] and 0 on (1/2, 2]. Then:
- ∫ h = 1/2 * 1 = 0.5 ≠ 1. Not valid.

Let me try h(x) = 1 on [0, 1] and 0 on (1, 2]. Then ∫ h = 1. ✓
- For k=0: ∫ h(1-h) = ∫₀¹ 1*0 dx + ∫₁² 0*1 dx = 0
- For k=1: ∫ h(x)(1-h(x+1)) dx. The overlap of [0,1] and [1,2] (shifted by 1) is... 
  h(x) is 1 on [0,1], h(x+1) is 1 on [-1,0]∩[0,2] = empty for x in [0,1]... 
  
  Actually, for the discrete version with the "full" correlate, we need to be more careful about boundary effects.

In the discrete version with n points on [0,2]:
h = [1, 1, ..., 1, 0, 0, ..., 0] with n/2 ones and n/2 zeros.

corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])

For k = n-1 (lag 0): corr = sum of h[i]*(1-h[i]) = 0 ✓
For k = n-1+1 = n (lag 1): corr = sum_{i=0}^{n/2-2} 1*(1-0) + sum_{i=n/2-1}^{n/2-1} 1*(1-1) = (n/2-1) + 0 = n/2-1
  Hmm, that's not great.

Actually, the "full" correlate has boundary effects that make this trickier. The maximum will likely occur at small lags where the "step" in h causes partial overlap.

Let me think about this differently. The best known constructions for the Erdős problem use a "staircase" or "sawtooth" function. The key is to have the function transition smoothly between 0 and 1 in a way that minimizes the worst-case overlap.

For our discrete problem, I think the optimal solution has h values that are close to 0 or 1 (bimodal), with some intermediate values at the transition regions.

from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_free = np.sum(mask)
        if n_free > 0 and abs(diff) <= 0.5 * n_free + 1e-10:
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    r = alpha * w
    
    hg = np.zeros(n)
    for k in range(2*n - 1):
        rk = r[k]
        if rk < 1e-12:
            continue
        d = (n - 1) - k
        if d >= 0:
            hg[0:n-d] += rk * (1.0 - h[d:n])
            hg[d:n] += -rk * h[0:n-d]
        else:
            dd = -d
            hg[dd:n] += rk * (1.0 - h[0:n-dd])
            hg[0:n-dd] += -rk * h[dd:n]
    
    return hg

def soft_max_val(h, alpha):

        if f_new <= f0 + c1 * t * np.dot(grad, h_new - h):
            return h_new, t
