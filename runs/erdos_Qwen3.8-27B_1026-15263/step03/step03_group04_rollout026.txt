    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-15:
            break

def corr_grad(h, shift):
    """Gradient of corr[shift] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def smooth_max_obj_and_grad(h, alpha):

def slsqp_refine(h, n, target_sum, alpha, maxiter=500):

        return smooth_max_obj_and_grad(x, alpha)[0]

        return smooth_max_obj_and_grad(x, alpha)[1]

             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(fun, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        count = np.sum(mask)
        if count > 0:
            step = diff / count
            step = max(min(step, (1.0 - 1e-14) - h[mask].max()), h[mask].min() - (0.0 + 1e-14))
            h[mask] += step
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, shift):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return grad

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-12
    sig = np.where(w > thresh)[0]
    for k in sig:
        shift = k - (n - 1)
        grad += w[k] * corr_grad_single(h, shift)
    return obj, grad

    h = h.copy()
    c5_cur = compute_c5(h)
    lr = init_lr
    stall = 0
    for it in range(max_iter):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        a = 80.0
        exp_v = np.exp(a * (top_vals - np.max(top_vals)))
        w = exp_v / np.sum(exp_v)
        grad = np.zeros(n)
        for kk, wi in zip(top_idx, w):
            shift = kk - (n - 1)
            grad += wi * corr_grad_single(h, shift)
        found = False
        for lr_try in [lr * 0.05, lr * 0.1, lr * 0.3, lr * 0.5, lr, lr * 2.0]:
            h_new = h - lr_try * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5_new
                lr = max(lr_try * 1.3, 1e-6)
                found = True
                break
        if not found:
            lr = max(lr * 0.25, 1e-8)
            stall += 1
            if stall > 200:
                break
        stall = 0
    return h, c5_cur

def binary_local_search(h, n, target_sum, rng, max_rounds):

                h[i] = np.clip(h_old + d, 0.0, 1.0)
