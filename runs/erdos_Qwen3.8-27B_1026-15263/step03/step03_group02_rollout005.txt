    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -400.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    w = exp_vals / exp_sum
    obj = m_val + np.log(exp_sum) / alpha
    
    j_idx = np.arange(n)[:, None]
    d_idx = np.arange(2*n-1)[None, :]
    s_idx = d_idx - (n - 1)

    M = valid1 * (1.0 - h[tc1]).astype(float) - valid2 * h[tc2].astype(float)
    grad = M @ w
    
    return obj, grad

        return smooth_max_and_grad(h, alpha)[0]

        return smooth_max_and_grad(h, alpha)[1]

        c5_new = compute_c5(h)
        delta = c5_new - c5_curr

        accept = delta < 0.0
        if not accept and T > 1e-12:
            log_u = np.log(max(rng.random(), 1e-300))
            accept = (delta / T) + log_u < 0.0

        if accept:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if r < 0.55:
                h[i], h[j] = h[j], h[i]
            elif r < 0.75:
                h[i:j] = saved_block
            else:
                h[i], h[j], h[k] = hi, hj, hk

        if no_improve > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.025)
            no_improve = 0

        T = max(T * 0.99998, min_T)

    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0
    n_ones = int(ts_main)

    h_bin = make_binary(best_h, n_main, n_ones, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.30 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, n_ones, sa_end1, rng, t_init=0.012)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h_bin = h_sa.copy()

    h_work = project_to_sum(h_bin.astype(float), ts_main)
    alpha_schedule = [8, 30, 120, 500, 2000, 8000, 30000, 100000]
    for i_alpha, alpha in enumerate(alpha_schedule):
        if elapsed() > 0.60 * budget_s:
            break
        maxit = 400 + 100 * i_alpha
        h_work = slsqp_optimize(h_work, n_main, ts_main, alpha, maxiter=maxit)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    h_bin2 = make_binary(best_h, n_main, n_ones, rng)
    c5_bin2 = compute_c5(h_bin2)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()

    sa_end2 = t_start + 0.78 * budget_s
    if remaining() > 60:
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, n_ones, sa_end2, rng, t_init=0.006)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            h_bin2 = h_sa2.copy()

    h_work = project_to_sum(best_h.astype(float), ts_main)
    for alpha in [50000, 200000, 800000]:
        if elapsed() > 0.90 * budget_s:
            break
        h_work = slsqp_optimize(h_work, n_main, ts_main, alpha, maxiter=1000)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    for n_try in [280, 320, 360, 400, 480]:
        if remaining() < 120:
            break
        if n_try % 2 != 0:
            n_try += 1
        ts_try = float(n_try) / 2.0
        n_ones_try = int(ts_try)
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_try = h_interp.copy()
        h_bin_try = make_binary(h_try, n_try, n_ones_try, rng)

        sa_t_end = min(t_start + 0.95 * budget_s, t_start + elapsed() + 40)
        h_try, _ = binary_sa(h_bin_try, n_try, n_ones_try, sa_t_end, rng, t_init=0.01)
        c5_try = compute_c5(h_try)

        for alpha in [50, 300, 3000]:
            if remaining() < 40:
                break
            h_cont = project_to_sum(h_try.astype(float), ts_try)
            h_cont = slsqp_optimize(h_cont, n_try, ts_try, alpha, maxiter=400)
            c5_cont = compute_c5(h_cont)
            if c5_cont < c5_try:
                h_try = h_cont
                c5_try = c5_cont

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    if remaining() > 60:
        h_final_bin = make_binary(best_h, best_n, best_n // 2, rng)
        c5_fb = compute_c5(h_final_bin)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin.copy()
            best_n = len(h_final_bin)

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
