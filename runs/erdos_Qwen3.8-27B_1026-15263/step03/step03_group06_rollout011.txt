To minimize the overlap at small lags, we need adjacent values to be correlated. Specifically, if h(i)=1, we want h(i+1) to also be 1 (so that h(i)(1-h(i+1))=0).

This suggests a "blocky" structure where h stays at 0 or 1 for a while before switching. But as we saw, long blocks create problems at larger lags.

The optimal structure is somewhere in between: blocks that are long enough to reduce small-lag overlap, but arranged aperiodically to reduce large-lag overlap.

This is essentially the **Erdős problem** itself! The optimal h is not known in closed form.

                h = project_to_sum(h, target)
                c = compute_c5(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[i] = best_v
            h = project_to_sum(h, target)
            if best_c < c5_cur - 1e-15:
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 720
        best_h = np.full(best_n, 0.5)

    n1 = best_n
    target1 = n1 / 2.0
    best_h = project_to_sum(best_h, target1)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    idx1, idx2 = precompute_indices(n1)

    def tl():
        return budget_s - (time.time() - t_start)

    phase1_time = min(250, tl() * 0.2)
    if tl() > phase1_time + 50:
        alphas_p1 = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
        h_p1, c5_p1 = proj_gradient(best_h, n1, target1, alphas_p1, 300, 0.5, idx1, idx2, phase1_time)
        if c5_p1 < best_c5:
            best_c5 = c5_p1
            best_h = h_p1

    phase2_time = min(400, tl() * 0.3)
    if tl() > phase2_time + 80:
        for alpha in [3000.0, 8000.0, 20000.0]:
            if tl() < 60:
                break
            h_p2, c5_p2 = lbfgs_run(best_h, n1, target1, alpha, idx1, idx2, min(150, tl() * 0.3), maxiter=1500)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2

    phase3_time = min(400, tl() * 0.35)
    if tl() > 30:
        n_rounds = max(20, int(phase3_time * 1.2))
        h_p3, c5_p3 = coord_descent_fast(best_h, n1, target1, n_rounds, rng, min(phase3_time * 0.6, tl() - 30), fine=False)
        if c5_p3 < best_c5:
            best_c5 = c5_p3
            best_h = h_p3

    if tl() > 60:
        n_rounds2 = max(20, int(min(200, tl() * 0.3)))
        h_p3b, c5_p3b = coord_descent_fast(best_h, n1, target1, n_rounds2, rng, min(150, tl() - 30), fine=True)
        if c5_p3b < best_c5:
            best_c5 = c5_p3b
            best_h = h_p3b

    if tl() > 80:
        n_ls = max(5, int(min(30, tl() * 0.05)))
        h_p3c, c5_p3c = coord_descent_ls(best_h, n1, target1, n_ls, rng, min(120, tl() - 30))
        if c5_p3c < best_c5:
            best_c5 = c5_p3c
            best_h = h_p3c

    if tl() > 60:
        alpha_polish = max(20000.0, 50.0 * n1)
        h_pol, c5_pol = lbfgs_run(best_h, n1, target1, alpha_polish, idx1, idx2, min(120, tl() - 20), maxiter=1000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if tl() > 40:
        for d_fine in [0.0005, 0.0002, 0.0001]:
            if tl() < 20:
                break
            h_ff, c5_ff = coord_descent_fast(best_h, n1, target1, 50, rng, min(20, tl() - 10), fine=True)
            if c5_ff < best_c5:
                best_c5 = c5_ff
                best_h = h_ff

    if tl() > 30:
        for alpha_final in [50000.0, 100000.0]:
            if tl() < 15:
                break
            alphas_p4 = [alpha_final]
            h_p4, c5_p4 = proj_gradient(best_h, n1, target1, alphas_p4, 500, 0.1, idx1, idx2, min(25, tl() - 10))
            if c5_p4 < best_c5:
                best_c5 = c5_p4
                best_h = h_p4

    if tl() > 20:
        h_ls_final, c5_ls_final = coord_descent_ls(best_h, n1, target1, 5, rng, min(15, tl() - 5))
        if c5_ls_final < best_c5:
            best_c5 = c5_ls_final
            best_h = h_ls_final

    best_h = project_to_sum(best_h, target1)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n1)
