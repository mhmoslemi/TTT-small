            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    """Compute smooth max of correlation*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)

    # Gradient: d/dh [m + log(s)/alpha] * dx
    # d/dh log(s) = (1/s) * sum_k w[k] * d(corr[k])/dh * alpha
    # = sum_k w[k] * d(corr[k])/dh
    g = np.zeros(n)

        wl = w[l + n - 1]
        if wl < 1e-14:

        # d/dh[j] of this = (1-h[j-l]) if j-l valid, minus h[j+l] if j+l valid
        for j in range(n):
            # contribution from h[j] as the first factor
            jl = j - l
            if 0 <= jl < n:
                lo = max(0, l)
                hi = min(n, n + l)
                # j is in [lo, hi) and jl = j-l is in [0, n)
                # This means j is contributing (1-h[j-l])
                pass
        # Vectorized version:
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            # h[lo:hi] are the h[i] values where i is in valid range
            # (1-h[lo-l:hi-l]) are the (1-h[i-l]) values
            g[lo:hi] += wl * (1.0 - h[lo - l: hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l: hi2 + l]
    return val, g * dx

    def obj(x):
        h = project_to_sum(x, target_sum)
        v, _ = smooth_max_obj_and_grad(h, alpha)
        return v
    
    def grad(x):
        h = project_to_sum(x, target_sum)
        _, g = smooth_max_obj_and_grad(h, alpha)
        # Project gradient to the tangent space of the constraint
        # The constraint is sum(h) = target_sum, so tangent space is {v: sum(v)=0}

        # Also zero out where h is at bounds

        at_lower = h <= 1e-10
        g[at_upper & (g > 0)] = 0
        g[at_lower & (g < 0)] = 0
        return g

                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10,
                            'maxls': 50, 'maxfun': maxiter * 10})
    return project_to_sum(res.x, target_sum)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = (m + np.log(s) / alpha) * dx
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-15:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l: hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l: hi2 + l]
    return val, g * dx

def fista_smooth_max(h0, n, target_sum, alpha, n_iters, lr_init, t_start, time_limit):
    h = h0.copy()
    z = h0.copy()
    z_prev = h0.copy()
    t_val = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        
        # Nesterov extrapolation
        t_prev = t_val
        t_val = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        beta = (1.0 - t_prev) / t_val
        y = z + beta * (z - z_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient at y
        _, g = smooth_max_val_grad(y, alpha)
        
        # Project gradient to feasible directions
        g = g - np.mean(g)
        at_u = y >= 1.0 - 1e-10
        at_l = y <= 1e-10
        g[at_u & (g > 0)] = 0
        g[at_l & (g < 0)] = 0
        
        # Gradient step + projection
        h_new = np.clip(z - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        # Track best
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        z_prev = z
        z = h_new
        
        # Adaptive lr
        if c5_new < best_c5 + 1e-10:
            lr = min(lr * 1.005, 5.0)
        else:
            lr *= 0.95
    
    return best_h, best_c5

def top_k_focused_descent(h0, n, target_sum, n_iters, lr_init, top_k, t_start, time_limit):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    dx = 2.0 / n

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]

            lo = max(0, l)
            hi = min(n, n + l)

            lo2 = max(0, -l)
            hi2 = min(n, n - l)

                g[lo2:hi2] -= h[lo2 + l: hi2 + l]
        g = g / top_k * dx
        g = g - np.mean(g)

            h = h_new
            lr = min(lr * 1.01, 2.0)
        else:

    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps, t_start, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    steps_all = list(base_steps)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            best_c5_i = c5_cur
            best_h_i = h.copy()
            for d in steps_all:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5_i - 1e-15:
                    best_c5_i = c5_try
                    best_h_i = h_try
            if best_c5_i < c5_cur - 1e-15:
                h = best_h_i
                c5_cur = best_c5_i
                improved = True
        if not improved:
            break
    return h, c5_cur

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, t_start, time_limit):
    def obj_proj(x):
        h = project_to_sum(x, target_sum)
        v, _ = smooth_max_val_grad(h, alpha)
        return v
    def grad_proj(x):
        h = project_to_sum(x, target_sum)

        return g

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return h0, compute_c5(h0)

The gradient is: dV/dh = dx * sum_k w[k] * d(corr[k])/dh

The d(corr[k])/dh is a vector where each component is bounded by 1 (since corr[k] is a sum of products h[i]*(1-h[j]), the partial derivative w.r.t. h[i] is at most the number of terms involving h[i], which is at most n).

So ||dV/dh|| <= dx * sum_k w[k] * ||d(corr[k])/dh|| <= dx * 1 * n * (2n-1) = dx * n * (2n-1)

For n=480: dx = 2/480 = 0.0042, so ||dV/dh|| <= 0.0042 * 480 * 959 ≈ 1930

The Lipschitz constant of the gradient is roughly the Hessian norm, which is bounded by something like O(n^2 * dx * alpha) for the smooth max. But this is a rough bound and the actual Lipschitz constant is much smaller in practice.

For the step size, I'll use an adaptive approach: start with a reasonable lr and adjust based on whether the objective decreases.

3. **Numerical issues**: With very high alpha (e.g., 20000), the exp values can overflow. I'm already clipping the shifted values to [-700, 700], which should prevent this.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = (m + np.log(s) / alpha) * dx
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-15:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return val, g * dx

def momentum_smooth_max(h0, n, target_sum, alpha, n_iters, lr_init, t_start, time_limit):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    c5_cur = best_c5
    lr = lr_init
    mom = 0.9
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        _, g = smooth_max_val_grad(h, alpha)
        g = g - np.mean(g)
        v = mom * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= c5_cur * 1.00001:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.008, 0.5)
            mom = min(mom + 0.0005, 0.95)
        else:
            lr *= 0.8
            v *= 0.3
            h = h_new
            c5_cur = c5_new
    return best_h, best_c5

def topk_descent(h0, n, target_sum, n_iters, lr_init, top_k, t_start, time_limit):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    c5_cur = best_c5
    lr = lr_init
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            l = idx - (n - 1)
            lo = max(0, l)
            hi = min(n, n + l)
            if lo < hi:
                g[lo:hi] += (1.0 - h[lo - l:hi - l])
            lo2 = max(0, -l)
            hi2 = min(n, n - l)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2 + l:hi2 + l]
        g = g / top_k * dx
        g = g - np.mean(g)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= c5_cur:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.005, 1.0)
        else:
            lr *= 0.9
            h = h_new
            c5_cur = c5_new
    return best_h, best_c5

def nesterov_smooth_max(h0, n, target_sum, alpha, n_iters, lr_init, t_start, time_limit):
    h = h0.copy()
    z = h0.copy()
    z_prev = h0.copy()
    t_val = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        t_prev = t_val
        t_val = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        beta = (1.0 - t_prev) / t_val
        y = z + beta * (z - z_prev)
        y = np.clip(y, 0.0, 1.0)
        _, g = smooth_max_val_grad(y, alpha)
        g = g - np.mean(g)
        at_u = y >= 1.0 - 1e-10
        at_l = y <= 1e-10
        g[at_u & (g > 0)] = 0
        g[at_l & (g < 0)] = 0
        h_new = np.clip(z - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        z_prev = z
        z = h_new
        lr = min(lr * 1.003, 1.0)
    return best_h, best_c5

def _lbfgs_obj(x, target_sum, alpha):
    h = project_to_sum(x, target_sum)
    v, _ = smooth_max_val_grad(h, alpha)
    return v

def _lbfgs_grad(x, target_sum, alpha):
    h = project_to_sum(x, target_sum)
    _, g = smooth_max_val_grad(h, alpha)
    g = g - np.mean(g)
    return g

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, t_start, time_limit):
    if time.time() - t_start > time_limit:
        return h0.copy(), compute_c5(h0)
    try:
        res = minimize(_lbfgs_obj, h0.copy(), jac=_lbfgs_grad, method='L-BFGS-B',
                       args=(target_sum, alpha),
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': maxiter * 20})
        h_opt = project_to_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return h0.copy(), compute_c5(h0)

def coordinate_descent(h0, n, target_sum, max_rounds, base_steps, t_start, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    steps_all = list(base_steps)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            best_c5_i = c5_cur
            best_h_i = h.copy()
            for d in steps_all:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5_i - 1e-15:
                    best_c5_i = c5_try
                    best_h_i = h_try
            if best_c5_i < c5_cur - 1e-15:
                h = best_h_i
                c5_cur = best_c5_i
                improved = True
        if not improved:
            break
    return h, c5_cur

def random_perturb_restart(h0, n, target_sum, alpha, lr, n_iters, t_start, time_limit, rng, sigma):
    h_pert = h0 + rng.normal(0, sigma, n)
    h_pert = project_to_sum(h_pert, target_sum)
    h_new, c5_new = momentum_smooth_max(h_pert, n, target_sum, alpha, n_iters, lr, t_start, time_limit)
    return h_new, c5_new
