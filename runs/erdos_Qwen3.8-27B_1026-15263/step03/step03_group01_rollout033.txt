import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    # Bisection on lambda
    # h[i] = clip(h[i] + shift, 0, 1) where we adjust to hit target_sum
    # Actually, use the standard approach: find lambda such that sum(clip(h_center - lambda, 0, 1)) = target
    # where h_center is some center point
    # Simpler: iterative adjustment with bisection on the shift amount
    
    # Use bisection on a scalar shift
    # We want: sum(clip(h + s, 0, 1)) = target_sum for some s
    # But this only works if the solution is a uniform shift, which it isn't always
    
    # Better: use the dual approach
    # h[i] = min(1, max(0, a[i] - lambda)) where a[i] = h[i] initially
    # Find lambda by bisection
    
    a = h.copy()
    lo, hi = -1.0, 1.0  # range for lambda
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(a - lam, 0.0, 1.0))
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(hi - lo) < 1e-15:
            break
    
    lam = (lo + hi) / 2.0
    result = np.clip(a - lam, 0.0, 1.0)
    
    # Handle residual
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    
    return result

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    shifted = alpha * (corr - m)
    log_sum_exp = m + np.log(np.sum(np.exp(shifted))) / alpha
    # Weights for the soft max
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of smooth max with respect to h
    # corr[k] = sum_i h[i+k'] * (1-h[i]) for valid indices
    # d(corr[k])/d(h[j]) = sum over terms where h[j] appears
    # = (1-h[j-k]) * 1[valid] - h[j+k'] * 1[valid]
    # 
    # For the full correlation (2n-1 values), index m corresponds to lag (m - (n-1))
    # corr_full[m] = sum_i h[i] * (1-h)[i + m - (n-1)]
    # = sum_i h[i] * (1-h[i + m - n + 1])
    # 
    # d(corr_full[m])/d(h[j]):
    # - h[j] appears in the first factor when i = j: contributes (1-h[j + m - n + 1]) if valid
    # - h[j] appears in (1-h[...]) when j = i + m - n + 1, i.e., i = j - m + n - 1: contributes -h[j - m + n - 1] if valid
    
    # Vectorized computation
    nj = np.arange(n)[:, None]  # (n, 1)
    nm = np.arange(2 * n - 1)[None, :]  # (1, 2n-1)
    
    # First term: (1-h[j + m - n + 1]) valid when 0 <= j + m - n + 1 < n
    idx1 = nj + nm - n + 1  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # Second term: -h[j - m + n - 1] valid when 0 <= j - m + n - 1 < n
    idx2 = nj - nm + n - 1  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    clipped1 = np.clip(idx1, 0, n-1)
    clipped2 = np.clip(idx2, 0, n-1)
    
    grad = np.sum(valid1 * (1.0 - h[clipped1]) * w[None, :] - valid2 * h[clipped2] * w[None, :], axis=1)
    
    return log_sum_exp, grad

def proj_grad_descent(h0, n, ts, alpha, n_iter=5000, lr_init=0.1, lr_min=1e-8):
    h = h0.copy()
    lr = lr_init
    obj, _ = smooth_max_obj_grad(h, alpha)
    stall = 0
    
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        # Gradient step
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        
        if obj_new < obj - 1e-18:
            h = h_new
            obj = obj_new
            stall = 0
            lr = min(lr * 1.01, 1.0)
        else:
            stall += 1
            lr = max(lr * 0.7, lr_min)
            if stall > 100:
                break
    
    return h

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[0]
    def g(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def lag_balance_opt(h, n, ts, alpha, n_rounds=50):

    h = h.copy()

        threshold = m - 0.02 * m
        hot_lags = np.where(corr >= threshold)[0]
        
        # Compute gradient targeting only hot lags
        # Weight = 1 for hot lags, 0 otherwise (or soft weighting)
        w = np.zeros(len(corr))
        w[hot_lags] = 1.0 / len(hot_lags)
        
        # This is like a weighted correlation minimization
        # grad_j = sum_m w[m] * d(corr[m])/d(h[j])
        nj = np.arange(n)[:, None]
        nm = np.arange(2 * n - 1)[None, :]
        idx1 = nj + nm - n + 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = nj - nm + n - 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        clipped1 = np.clip(idx1, 0, n-1)
        clipped2 = np.clip(idx2, 0, n-1)
        
        grad = np.sum(valid1 * (1.0 - h[clipped1]) * w[None, :] - valid2 * h[clipped2] * w[None, :], axis=1)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n // 2.0)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    return project_to_sum(h_up, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    a = h.copy()
    lo, hi = -1.0, 1.0
    def sum_at_lambda(lam):
        return np.sum(np.clip(a - lam, 0.0, 1.0))
    s_lo = sum_at_lambda(lo)
    s_hi = sum_at_lambda(hi)
    if s_lo < target_sum:
        lo = -2.0
        while sum_at_lambda(lo) < target_sum and lo > -10.0:
            lo *= 2.0
    if s_hi > target_sum:
        hi = 2.0
        while sum_at_lambda(hi) > target_sum and hi < 10.0:
            hi *= 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(hi - lo) < 1e-16:
            break
    lam = (lo + hi) / 2.0
    result = np.clip(a - lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    idx1 = nj + nm - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = nj - nm + n - 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    clipped1 = np.clip(idx1, 0, n - 1)
    clipped2 = np.clip(idx2, 0, n - 1)
    grad = np.sum(valid1 * (1.0 - h[clipped1]) * w[None, :] - valid2 * h[clipped2] * w[None, :], axis=1)
    return obj, grad

def proj_grad_descent(h0, n, ts, alpha, n_iter=2000, lr_init=0.1, lr_min=1e-9, momentum=0.0):
    h = h0.copy()
    lr = lr_init
    v = np.zeros(n)
    obj, _ = smooth_max_obj_grad(h, alpha)
    stall = 0
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        if momentum > 0:
            v = momentum * v + (1 - momentum) * grad
            step = lr * v
        else:
            step = lr * grad
        h_new = project_to_sum(h - step, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj - 1e-16:
            h = h_new
            obj = obj_new
            stall = 0
            lr = min(lr * 1.02, 1.0)
        else:
            stall += 1
            lr = max(lr * 0.6, lr_min)
            if momentum > 0:
                v *= 0.5
            if stall > 150:
                break
    return h

def proj_grad_continuation(h0, n, ts, alphas, iters_per_alpha=1500):
    h = h0.copy()
    for alpha in alphas:
        h = proj_grad_descent(h, n, ts, alpha, n_iter=iters_per_alpha, lr_init=0.08)
    return h

def slsqp_opt(h0, n, ts, alpha, maxiter=400):
    def f(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[0]
    def g(x):
        return smooth_max_obj_grad(np.clip(x, 0, 1), alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def lag_balance_opt(h, n, ts, n_rounds=40):
    h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(n_rounds):
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.05 * m
        hot_lags = np.where(corr >= threshold)[0]
        if len(hot_lags) == 0:
            break
        w = np.zeros(len(corr))
        w[hot_lags] = 1.0 / len(hot_lags)
        nj = np.arange(n)[:, None]
        nm = np.arange(2 * n - 1)[None, :]
        idx1 = nj + nm - n + 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = nj - nm + n - 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        clipped1 = np.clip(idx1, 0, n - 1)
        clipped2 = np.clip(idx2, 0, n - 1)
        grad = np.sum(valid1 * (1.0 - h[clipped1]) * w[None, :] - valid2 * h[clipped2] * w[None, :], axis=1)
        for lr_try in [0.1, 0.05, 0.01, 0.005]:
            h_new = project_to_sum(h - lr_try * grad, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                break
    return h

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

        if time_left() > 25:
            h_sl = slsqp_opt(h_pg, n_b, ts_b, 5000, maxiter=300)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    # Phase 4: Diverse restarts
    n_e = best_n
    ts_e = n_e / 2.0
    n_diverse = 12 if time_left() > 250 else 6
    for trial in range(n_diverse):
        if time_left() < 35:
            break
        starts = make_starts(n_e, ts_e, rng, best_h, best_n)
        for h_s in starts[:6]:
            if time_left() < 20:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_pg = proj_grad_continuation(h_s, n_e, ts_e, [200, 1000, 3000], iters_per_alpha=800)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            if time_left() > 15:
                h_sl = slsqp_opt(h_pg, n_e, ts_e, 5000, maxiter=300)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()

    # Phase 5: Lag balancing
    if time_left() > 50:
        h_lb = lag_balance_opt(best_h, best_n, best_n / 2.0, n_rounds=50)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Phase 6: Final high-alpha polish
    if time_left() > 40:
        for alpha in [5000, 10000, 20000]:
            if time_left() < 20:
                break
            h_f = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=800)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 7: Final gradient squeeze
    if time_left() > 25:
        h_pg = proj_grad_descent(best_h, best_n, best_n / 2.0, 20000, n_iter=3000, lr_init=0.02)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
