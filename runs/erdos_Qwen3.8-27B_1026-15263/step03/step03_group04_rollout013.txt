            continue
        j = np.arange(n)
        # Condition: 0 <= j+ks < n
        valid1 = (j + ks >= 0) & (j + ks < n)
        # Condition: 0 <= j-ks < n
        valid2 = (j - ks >= 0) & (j - ks < n)
        
        a_vals = np.zeros(n)
        a_vals[valid1] += (1.0 - h[j[valid1] + ks])
        a_vals[valid2] -= h[j[valid2] - ks]
        A[:, k_idx] = a_vals
    return A

def compute_B(h, n, A):
    """B[j, k_idx] = corr_k - A[j,k_idx]*h[j]"""
    corr = compute_corr(h)  # shape (2n-1,)
    # B[j,k] = corr[k] - A[j,k]*h[j]
    B = corr[None, :] - A * h[:, None]  # shape (n, 2n-1)
    return B

def solve_1d_vectorized(A, B, lam, n, n_inner=30):
    """Solve min_{x in [0,1]} max_k (A[j,k]+lam)*x + B[j,k] for all j simultaneously."""
    M = A + lam  # shape (n, 2n-1)
    lo = np.zeros(n)
    hi = np.ones(n)
    for _ in range(n_inner):
        mid = (lo + hi) / 2.0  # shape (n,)
        V = M * mid[:, None] + B  # shape (n, 2n-1)
        K = V.argmax(axis=1)  # shape (n,)
        m_star = M[np.arange(n), K]  # shape (n,)
        # If m_star > 0: f increasing at mid → x* < mid → hi = mid
        # If m_star < 0: f decreasing at mid → x* > mid → lo = mid
        hi = np.where(m_star > 0, mid, hi)
        lo = np.where(m_star < 0, mid, lo)
    return (lo + hi) / 2.0

def lagrangian_coord_descent(h0, n, target_sum, n_sweeps=50, n_inner=30, n_outer=40):
    """Lagrangian coordinate descent (Jacobi) for k≠0 terms."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        A = compute_A(h, n)
        B = compute_B(h, n, A)
        
        # Bisection on λ
        lam_lo, lam_hi = -3.0, 3.0
        for _ in range(n_outer):
            lam = (lam_lo + lam_hi) / 2.0
            x_opt = solve_1d_vectorized(A, B, lam, n, n_inner)
            S = x_opt.sum()
            if S > target_sum + 1e-10:
                lam_lo = lam
            elif S < target_sum - 1e-10:
                lam_hi = lam
            else:
                break
        
        h_new = solve_1d_vectorized(A, B, (lam_lo + lam_hi) / 2.0, n, n_inner)
        h_new = project_to_sum(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h = h_new
        else:
            # No improvement, try with the new h anyway (might help in next sweep)
            h = h_new
        
        if sweep > 5 and abs(compute_c5(h) - best_c5) < 1e-8:
            break
    
    return h, best_c5

def smooth_max_obj_and_grad(h, alpha, n):
    """Smooth max objective and gradient (including all k)."""
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    dx = 2.0 / n
    obj = (m + np.log(s) / alpha) * dx
    
    grad = np.zeros(n)
    thresh = 1e-10
    for k_idx in np.where(w > thresh)[0]:
        wi = w[k_idx]
        shift = k_idx - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    grad *= dx
    return obj, grad

def optimize_smooth_max(h0, n, target_sum, alphas, maxiter_per_level):
    """Continuation method with smooth max."""
    h = h0.copy()
    for alpha in alphas:
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, alpha, n)
            s_dev = np.sum(x) - target_sum
            return obj + 0.5 * 1000.0 * s_dev * s_dev
        
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, alpha, n)
            s_dev = np.sum(x) - target_sum
            return grad + 1000.0 * s_dev * np.ones(n)
        
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_level, 'ftol': 1e-15, 'gtol': 1e-8})
            h = project_to_sum(res.x, target_sum)
        except Exception:
            pass
    return h

def exact_max_polish(h, n, target_sum, max_rounds=200, top_k=10):
    """Polish using exact max: gradient descent on the top k overlaps."""
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_indices = np.argsort(corr)[::-1][:top_k]
        best_improve = 0
        best_h = h
        
        # Combine gradients from top k
        combined_grad = np.zeros(n)
        for k_idx in top_indices:
            shift = k_idx - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
            combined_grad += grad
        
        # Line search
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            h_new = h - lr * combined_grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                best_improve = c5_cur - c5_new
                best_h = h_new.copy()
        
        if best_improve < 1e-12:
            break
        h = best_h
        c5_cur = compute_c5(h)
    
    return h, c5_cur
