```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on shift."""
    h = h.astype(float)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def correlate_fft(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    g = 1.0 - h
    # correlate(h, g) = convolve(h, g[::-1])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    G_rev = np.fft.rfft(g[::-1], size)
    corr = np.fft.irfft(H * G_rev, size)[:2*n-1]
    return corr

def subgradient_for_lag(h, lag, dx):
    """Compute subgradient of corr[lag_index] w.r.t. h, where lag_index = n-1+lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        # corr[n-1+lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # d/dh[j]: (1-h[j-lag])*1(lag<=j<=n-1) + (-h[j+lag])*1(0<=j<=n-1-lag)
        if lag < n:
            j1 = np.arange(lag, n)
            g[j1] += (1.0 - h[j1 - lag])
            j2 = np.arange(0, n - lag)
            g[j2] -= h[j2 + lag]
    else:
        m = -lag

        # d/dh[j]: (1-h[j+m])*1(0<=j<=n-1-m) + (-h[j-m])*1(m<=j<=n-1)
        if m < n:
            j1 = np.arange(0, n - m)
            g[j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            g[j2] -= h[j2 - m]
    return g * dx

def active_set_subgradient(h, K=20, alpha=1.0):
    """Compute smoothed subgradient using top K lags with softmax weights."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h)
    # Top K indices
    top_k = np.argsort(corr)[-K:]
    top_vals = corr[top_k]
    # Softmax weights
    shifted = alpha * (top_vals - top_vals.max())
    w = np.exp(shifted)
    w /= w.sum()

def backtracking_line_search(h, g, target_sum, c5_cur, max_steps=50):
    """Find step size using backtracking."""
    n = len(h)
    dx = 2.0 / n
    lr = 0.1
    for _ in range(max_steps):
        h_new = h - lr * g
        h_new = project_to_box_sum(h_new, target_sum)
        # Need to check if c5 decreased
        # For efficiency, just compute the max
        corr = correlate_fft(h_new)
        c5_new = np.max(corr) * dx
        if c5_new < c5_cur:
            return h_new, c5_new, lr
        lr *= 0.5
    return h, c5_cur, lr

def coord_descent_polish(h, target_sum, n_rounds=5, rng=None):
    """Polish using 1D coordinate descent on the most impactful coordinates."""
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(42)
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        # Try perturbing random coordinates
        idxs = rng.choice(n, size=min(50, n), replace=False)
        for i in idxs:
            v0 = h[i]
            lo = max(0.0, v0 - 0.3)
            hi = min(1.0, v0 + 0.3)
            if hi - lo < 1e-6:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_box_sum(h_tmp, target_sum)
                corr = correlate_fft(h_tmp)
                return np.max(corr) * (2.0 / n)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-5, 'maxiter': 50})
                if res.fun < c5_cur - 1e-12:
                    h[i] = res.x
                    h = project_to_box_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def project_to_box_sum(h, target_sum):
    h = h.astype(float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def correlate_fft(h):
    n = len(h)
    g = 1.0 - h
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    G_rev = np.fft.rfft(g[::-1], size)
    corr = np.fft.irfft(H * G_rev, size)[:2 * n - 1]
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h)
    return float(np.max(corr) * dx)

def raw_subgradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            j1 = np.arange(lag, n)
            g[j1] += (1.0 - h[j1 - lag])
            j2 = np.arange(0, n - lag)
            g[j2] -= h[j2 + lag]
    else:
        m = -lag
        if m < n:
            j1 = np.arange(0, n - m)
            g[j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            g[j2] -= h[j2 - m]
    return g

def fast_smooth_max_grad(h, K, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h)
    if K >= len(corr):
        top_k = np.arange(len(corr))
        top_vals = corr
    else:
        top_k = np.argsort(corr)[-K:]
        top_vals = corr[top_k]
    m = top_vals.max()
    shifted = alpha * (top_vals - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    sm_val = (m + np.log(s / len(exp_vals)) / alpha) * dx
    w = exp_vals / s
    g = np.zeros(n)
    for idx, weight in zip(top_k, w):
        if weight < 1e-10:
            continue
        lag = idx - (n - 1)
        g += weight * raw_subgradient(h, lag)
    g *= dx
    return sm_val, g

def slsqp_fast(h0, target_sum, K, alpha, maxiter):
    n = len(h0)
    def obj(x):
        v, _ = fast_smooth_max_grad(x, K, alpha)
        return v
    def grad(x):
        _, g = fast_smooth_max_grad(x, K, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_box_sum(res.x, target_sum)
    except Exception:
        return project_to_box_sum(h0.copy(), target_sum)

def nesterov_fast(h0, target_sum, K, alpha, n_iters, lr_init):
    n = len(h0)
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    for it in range(n_iters):
        _, g = fast_smooth_max_grad(y, K, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-12:
            g = g / g_norm
        h_new = project_to_box_sum(y - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_cur * 1.05 + 1e-8:
            lr *= 0.5
            if lr < 1e-7:
                break
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = np.clip(h_new + mo * (h_new - h), 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.92
    return h_best, c5_best

def binary_local_search(h0, target_sum, n_swaps, rng):
    n = len(h0)
    h_bin = (h0 >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = int(round(target_sum))
    if n_ones > target_ones:
        one_idx = np.where(h_bin == 1.0)[0]
        rng.shuffle(one_idx)
        for i in range(n_ones - target_ones):
            h_bin[one_idx[i]] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(h_bin == 0.0)[0]
        rng.shuffle(zero_idx)
        for i in range(target_ones - n_ones):
            h_bin[zero_idx[i]] = 1.0
    c5_cur = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for _ in range(n_swaps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur - 1e-12:
            c5_cur = c5_new
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5_cur

def coord_descent_polish(h, target_sum, n_rounds, rng, n_try=30):
    n = len(h)
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        idxs = rng.choice(n, size=min(n_try, n), replace=False)
        for i in idxs:
            v0 = h[i]
            lo = max(0.0, v0 - 0.4)
            hi = min(1.0, v0 + 0.4)
            if hi - lo < 1e-6:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_box_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-5, 'maxiter': 30})
                if res.fun < c5_cur - 1e-13:
                    h[i] = res.x
                    h = project_to_box_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur
```
