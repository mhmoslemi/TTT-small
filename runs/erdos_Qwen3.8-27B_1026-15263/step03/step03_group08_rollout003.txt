import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, l, n):
    """Gradient of correlation at lag l w.r.t. h."""
    g = np.zeros(n)
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    else:
        m = -l
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def nesterov_topk(h0, n, target_sum, n_iters, K, lr_init, momentum, time_limit):
    """Fast Nesterov with top-K subgradient."""
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-K:]
        
        # Softmax-like weights
        vals = corr[top_idx]
        max_val = np.max(vals)
        exps = np.exp(-(vals - max_val) * 0.1)  # temperature scaling
        wts = exps / np.sum(exps)
        
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            if wts[i] < 1e-6:
                continue
            l = k - (n - 1)
            g += wts[i] * grad_lag(h, l, n)
        
        g -= np.mean(g)
        
        h_old = h.copy()
        h += momentum * v - lr * g
        np.clip(h, 0.0, 1.0, out=h)
        h = project_to_sum(h, target_sum)
        v = h - h_old
        
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            stall += 1
            if stall > 100:
                lr *= 0.9
                v *= 0.3
                stall = 0
    
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, steps, time_limit):
    """Coordinate descent with multiple step sizes per coordinate."""
    t0 = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            orig = h[i]
            for d in steps:
                h[i] = np.clip(orig + d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-14:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[i] = orig
                    # Quick projection undo
                    h = project_to_sum(h, target_sum)
                    break
            else:
                h[i] = orig
                h = project_to_sum(h, target_sum)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, l, n):
    g = np.zeros(n)
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    else:
        m = -l
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def nesterov_opt(h0, n, target_sum, n_iters, K, lr, mu, t_limit):
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_lr = lr
    stall = 0
    
    for it in range(n_iters):
        if time.time() - t0 > t_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-K:]
        vals = corr[idx]
        mx = vals.max()
        # Temperature for softmax weighting
        temp = max((mx - vals.min()) * 0.5, 0.01)
        wts = np.exp((vals - mx) / temp)
        wts /= wts.sum()
        
        g = np.zeros(n)
        for i in range(K):
            if wts[i] < 1e-8:
                continue
            l = idx[i] - (n - 1)
            g += wts[i] * grad_lag(h, l, n)
        g -= g.mean()
        
        h_old = h.copy()
        h += mu * v - cur_lr * g
        np.clip(h, 0.0, 1.0, out=h)
        h = project_to_sum(h, target_sum)
        v = h - h_old

            stall = 0
