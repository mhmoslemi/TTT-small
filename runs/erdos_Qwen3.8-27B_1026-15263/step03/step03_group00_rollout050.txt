```python
import numpy as np
from scipy.optimize import minimize
import time
import math

_slsqp_alpha = 100.0
_slsqp_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(m):
            h[m] += d / np.count_nonzero(m)
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    n = len(h)
    k = int(round(target))
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    return hb

def corr_gradient_lag(h, lag):
    """Gradient of the overlap at a specific lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_obj(h, alpha):
    corr = get_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    e = np.exp(s)
    return m + np.log(e.sum()) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    
    grad = np.zeros(n)
    idx = np.where(w > 1e-5)[0]
    j_arr = np.arange(n)
    for k in idx:
        wk = w[k]
        b1 = j_arr + k - (n - 1)
        v1 = (b1 >= 0) & (b1 < n)
        b1c = np.clip(b1, 0, n - 1)
        grad += wk * v1 * (1.0 - h[b1c])
        
        b2 = j_arr - k + (n - 1)
        v2 = (b2 >= 0) & (b2 < n)
        b2c = np.clip(b2, 0, n - 1)
        grad -= wk * v2 * h[b2c]
    
    return grad

def slsqp_obj(x):
    return smooth_max_obj(x, _slsqp_alpha)

def slsqp_grad(x):
    return smooth_max_grad(x, _slsqp_alpha)

def slsqp_con_fun(x):
    return np.sum(x) - _slsqp_target

def slsqp_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, alpha, target, maxiter):
    global _slsqp_alpha, _slsqp_target
    _slsqp_alpha = alpha
    _slsqp_target = target
    n = len(h0)
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h0.copy(), method='SLSQP', jac=slsqp_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0.copy(), target)

def binary_local_search(hb, max_iters=100000, rng=None, pair_prob=0.3):

def binary_sa(hb, T0, T1, n_iters, rng):

    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()

            c5 = new_c5

        else:

            h[j] = 0.0

def coord_descent(h, target, max_iters=10000, step_frac=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = get_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target)
            stale += 1
            if stale > 100:
                break
    return h, c5

def interp_to_n(h, n_new):

    x_ext = np.concatenate([x_old, [2.0]])
    h_ext = np.concatenate([h, [h[0]]])
    return np.interp(x_new, x_ext, h_ext)
