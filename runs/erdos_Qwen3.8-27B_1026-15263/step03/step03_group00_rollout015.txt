def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_and_grad(x, alpha)[1]

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        g[:m] += 1.0
        g[:m] -= h[lag:]
        g[lag:] -= h[:m]
    else:
        k = -lag
        m = n - k
        g[k:] += 1.0
        g[k:] -= h[:m]
        g[:m] -= h[k:]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break

        h[i_dec] -= move

            stale = 0
    return h, c5

def multi_lag_polish(h, target_sum, max_rounds=100, top_k=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for round_idx in range(max_rounds):
        corr = compute_corr(h)
        top_idxs = np.argsort(corr)[-top_k:][::-1]
        improved = False
        for m_idx in top_idxs:
            lag = m_idx - (n - 1)
            g = corr_gradient(h, lag)
            for step in [0.005, 0.002, 0.0005]:
                h_new = h - step * g
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-14:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def stochastic_coord_desc(h, target_sum, rng, max_iters=50000, step_frac=0.05):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        candidates_inc = np.where(can_inc)[0]
        candidates_dec = np.where(can_dec)[0]
        i_inc = int(rng.choice(candidates_inc))
        i_dec = int(rng.choice(candidates_dec))
        if g[i_inc] >= g[i_dec]:
            continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            continue
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    return h, c5

def blend_crossover(p1, p2, rng):
    alpha = rng.uniform(0.2, 0.8)
    return alpha * p1 + (1.0 - alpha) * p2

def mutate_h(h, rng, noise_scale):
    h_mut = h + rng.uniform(-noise_scale, noise_scale, len(h))
    return np.clip(h_mut, 0.0, 1.0)

def tournament_select(pop, fitnesses, rng, k=3):
    idxs = rng.randint(0, len(pop), k)
    return int(min(idxs, key=lambda i: fitnesses[i]))

def genetic_search(h_init, n, target_sum, rng, pop_size, n_gen, noise_scale, t_start, time_limit):
    pop = [h_init.copy()]
    fitnesses = [compute_c5(h_init)]
    for i in range(1, pop_size):
        h_mut = h_init + rng.uniform(-noise_scale, noise_scale, n)
        h_mut = np.clip(h_mut, 0.0, 1.0)
        h_mut = project_to_sum(h_mut, target_sum)
        h_mut, _ = zigzag_basic(h_mut, target_sum, max_iters=500, init_step=0.005)
        pop.append(h_mut)
        fitnesses.append(compute_c5(h_mut))
    
    best_idx = int(np.argmin(fitnesses))
    best_h = pop[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    for gen in range(n_gen):
        if time.time() - t_start > time_limit:
            break
        
        p1_idx = tournament_select(pop, fitnesses, rng)
        p2_idx = tournament_select(pop, fitnesses, rng)
        child = blend_crossover(pop[p1_idx], pop[p2_idx], rng)
        if rng.random() < 0.6:
            child = mutate_h(child, rng, noise_scale * 0.5)
        child = np.clip(child, 0.0, 1.0)
        child = project_to_sum(child, target_sum)
        
        child, c5_child = zigzag_basic(child, target_sum, max_iters=800, init_step=0.005)
        fitnesses_child = compute_c5(child)
        
        worst_idx = int(np.argmax(fitnesses))
        if fitnesses_child < fitnesses[worst_idx]:
            pop[worst_idx] = child
            fitnesses[worst_idx] = fitnesses_child
        
        if fitnesses_child < best_c5:
            best_c5 = fitnesses_child
            best_h = child.copy()
        
        if gen % 8 == 4:
            h_new = h_init + rng.uniform(-noise_scale * 1.5, noise_scale * 1.5, n)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            h_new, _ = zigzag_basic(h_new, target_sum, max_iters=400, init_step=0.005)
            fitness_new = compute_c5(h_new)
            worst_idx = int(np.argmax(fitnesses))
            if fitness_new < fitnesses[worst_idx]:
                pop[worst_idx] = h_new
                fitnesses[worst_idx] = fitness_new
                if fitness_new < best_c5:
                    best_c5 = fitness_new
                    best_h = h_new.copy()
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def local_polish(h, n, target_sum, t_start, time_budget):
    ts = target_sum
    c5 = compute_c5(h)
    
    if time.time() - t_start > time_budget * 0.1:
        return h, c5
    h, c5 = zigzag_momentum(h, ts, max_iters=2000, init_step=0.008, momentum=0.3)
    
    if time.time() - t_start > time_budget * 0.25:
        return h, c5
    h, c5 = multi_lag_polish(h, ts, max_rounds=50, top_k=3)
    
    if time.time() - t_start > time_budget * 0.4:
        return h, c5
    h_opt = slsqp_smooth(h, n, ts, 200, maxiter=400)
    c5_opt = compute_c5(h_opt)
    if c5_opt < c5:
        h = h_opt
        c5 = c5_opt
    
    if time.time() - t_start > time_budget * 0.6:
        return h, c5
    h, c5 = coordinate_descent(h, ts, max_iters=5000, step_frac=0.12)
    
    if time.time() - t_start > time_budget * 0.8:
        return h, c5
    h_opt = slsqp_smooth(h, n, ts, 1000, maxiter=500)
    c5_opt = compute_c5(h_opt)
    if c5_opt < c5:
        h = h_opt
        c5 = c5_opt
    h, c5 = coordinate_descent(h, ts, max_iters=3000, step_frac=0.08)
    
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 300
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n_parent = best_n
    ts_parent = n_parent / 2.0
    
    if time_left() < 60:
        best_h = project_to_sum(best_h, ts_parent)
        best_c5 = compute_c5(best_h)
        return (best_h, best_c5, best_n)
    
    phase1_budget = min(300, time_left() * 0.3)
    best_h, best_c5 = local_polish(best_h, n_parent, ts_parent, t_start, phase1_budget)
    
    if time_left() > 200:
        ga_time_limit = time.time() + min(350, time_left() - 50)
        h_ga, c5_ga = genetic_search(best_h, n_parent, ts_parent, rng,
                                      pop_size=12, n_gen=100,
                                      noise_scale=0.04,
                                      t_start=t_start, time_limit=ga_time_limit)
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga.copy()
    
    if time_left() > 100:
        resolutions = [420, 480, 540, 600]
        for n_tr in resolutions:
            if time_left() < 80:
                break
            if n_tr % 2 != 0:
                continue
            ts_tr = n_tr / 2.0
            h_interp = interp_to_n(best_h, n_tr)
            h_interp = project_to_sum(h_interp, ts_tr)
            c5_interp = compute_c5(h_interp)
            h_res, c5_res = local_polish(h_interp, n_tr, ts_tr, t_start, min(150, time_left() - 30))
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_tr
    
    if time_left() > 60:
        n_final = best_n
        ts_final = n_final / 2.0
        h_pol, c5_pol = local_polish(best_h, n_final, ts_final, t_start, min(120, time_left() - 20))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    if time_left() > 40:
        h_st, c5_st = stochastic_coord_desc(best_h, ts_final, rng, max_iters=30000, step_frac=0.04)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
        if time_left() > 15:
            h_cd, c5_cd = coordinate_descent(best_h, ts_final, max_iters=8000, step_frac=0.05)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    if time_left() > 20:
        h_ml, c5_ml = multi_lag_polish(best_h, ts_final, max_rounds=200, top_k=5)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()
    
    if time_left() > 15:
        for alpha in [5000, 20000]:
            if time_left() < 10:
                break
            h_s = slsqp_smooth(best_h, best_n, ts_final, alpha, maxiter=1000)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
    
    if time_left() > 10:
        h_zz, c5_zz = zigzag_basic(best_h, ts_final, max_iters=5000, init_step=0.002)
        if c5_zz < best_c5:
            best_c5 = c5_zz
            best_h = h_zz.copy()
        h_cd, c5_cd = coordinate_descent(best_h, ts_final, max_iters=15000, step_frac=0.03)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
