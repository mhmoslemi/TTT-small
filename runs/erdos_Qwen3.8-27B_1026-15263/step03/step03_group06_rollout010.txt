import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s_total = np.sum(exp_vals)
    w = exp_vals / s_total
    sm_val = m + np.log(s_total / (2*n-1)) / alpha
    
    h_rev = h[::-1].copy()
    omh = 1.0 - h.reshape(1, -1)  # not needed, just use 1-h
    
    g = np.empty(n)
    for j in range(n):
        g[j] = np.dot(w[n-1-j : 2*n-1-j], 1.0-h) - np.dot(w[j : n+j], h_rev)
    g *= dx
    return sm_val, g

def nesterov_optimize(h0, n, target_sum, alphas, steps_per_alpha, lr_base):
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    
    for alpha in alphas:
        y = h.copy()
        t_val = 1.0
        lr = lr_base / alpha  # scale with alpha
        lr = max(lr, 0.001)
        
        for it in range(steps_per_alpha):
            _, g = smooth_max_and_grad(y, alpha)
            h_new = y - lr * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)

            h = h_new
            t_val = t_new

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s_total = np.sum(exp_vals)
    w = exp_vals / s_total
    sm_val = m + np.log(s_total / (2 * n - 1)) / alpha
    h_rev = h[::-1]
    omh = 1.0 - h
    g = np.empty(n)
    for j in range(n):
        g[j] = np.dot(w[n - 1 - j : 2 * n - 1 - j], omh) - np.dot(w[j : n + j], h_rev)
    g *= dx
    return sm_val, g

def nesterov_optimize(h0, n, target_sum, alphas, steps_per_alpha, lr_base, time_limit=None, t_start=0):
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    for alpha in alphas:
        if time_limit is not None and (time.time() - t_start) > time_limit - 30:
            break
        y = h.copy()
        t_val = 1.0
        lr = lr_base / max(1.0, np.sqrt(alpha))
        for it in range(steps_per_alpha):
            if time_limit is not None and (time.time() - t_start) > time_limit - 30:
                return h_best, c5_best
            _, g = smooth_max_and_grad(y, alpha)
            h_new = y - lr * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new > c5_best * 1.02:
                lr *= 0.7
                if lr < 1e-6:
                    break
                continue
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            mu = (t_val - 1.0) / t_new
            y = h_new + mu * (h_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = h_new
            t_val = t_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                lr = min(lr * 1.02, 0.5)
    return h_best, c5_best

def slsqp_refine(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_fine(h0, n, target_sum, n_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def fourier_perturb(h, rng, amp, n_modes):
    n = len(h)
    freqs = np.fft.rfftfreq(n, d=1.0 / n)
    H = np.fft.rfft(h)
    for k in range(1, min(n_modes, len(H))):
        H[k] += amp * rng.randn() * (1.0 + rng.rand())
        H[n - k] += amp * rng.randn() * (1.0 + rng.rand())
    h_pert = np.fft.irfft(H, n=n)
    return project_to_sum(h_pert, n / 2.0)

def block_perturb(h, rng, n_blocks, amp):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, n // 2)
        length = rng.randint(5, max(6, n // 10))
        end = min(start + length, n)
        sig = 1.0 if rng.rand() > 0.5 else -1.0
        envelope = np.sin(np.pi * np.arange(end - start) / max(1, end - start))
        h_new[start:end] += sig * amp * envelope
    return project_to_sum(h_new, n / 2.0)

def interpolate_to_resolution(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h, period=2)
    return project_to_sum(h_interp, n_new / 2.0)

    if best_h is None:
        best_n = 360

    n1 = best_n

    alphas_p1 = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
