def smooth_max_grad_efficient(h, alpha):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -50, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n-1)) / alpha
    
    # Gradient: g[i] = sum_t w[n-1+t] * ((1-h[i+t]) - h[i-t])
    # where t ranges over -(n-1)..(n-1)
    # Valid: 0 <= i+t <= n-1 for first term, 0 <= i-t <= n-1 for second term
    g = np.zeros(n)
    for t in range(-(n-1), n):
        wt = w[n-1+t]
        if wt < 1e-15:
            continue
        # First term: (1-h[i+t]) for valid i
        # i+t in [0, n-1] => i in [-t, n-1-t] ∩ [0, n-1]
        i_start1 = max(0, -t)
        i_end1 = min(n-1, n-1-t)
        if i_start1 <= i_end1:
            g[i_start1:i_end1+1] += wt * (1.0 - h[i_start1+t:i_end1+t+1])
        # Second term: -h[i-t] for valid i
        # i-t in [0, n-1] => i in [t, t+n-1] ∩ [0, n-1]
        i_start2 = max(0, t)
        i_end2 = min(n-1, t+n-1)
        if i_start2 <= i_end2:
            g[i_start2:i_end2+1] -= wt * h[i_start2-t:i_end2-t+1]
    return sm_val, g

```python
def project_hypersimplex(x, s):
    """Project x onto {0 <= x_i <= 1, sum(x) = s}"""
    x = np.clip(x, 0.0, 1.0)
    # If already at target sum, done
    if abs(np.sum(x) - s) < 1e-12:
        return x
    # Bisection on tau
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        sp = np.sum(proj)
        if sp > s:
            lo = mid
        else:
            hi = mid
        if abs(sp - s) < 1e-12:
            break
    return np.clip(x - mid, 0.0, 1.0)
```

This is O(n) per bisection step, and 100 steps is more than enough for double precision.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_hypersimplex(x, s):
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - s) < 1e-12:
        return x
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        sp = np.sum(proj)
        if sp > s:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -40.0, 400.0)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk < 1e-12:
            continue
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - k:i_hi - k + 1])
        i_lo2 = max(0, -k)
        i_hi2 = min(n - 1, n - 1 - k)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 + k:i_hi2 + k + 1]
    return sm_val, g

def active_lags_smooth(h, active_idx, alpha):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    ac = corr[active_idx]
    m = np.max(ac)
    shifted = np.clip(alpha * (ac - m), -40.0, 400.0)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    sm_val = m + np.log(s / len(ac)) / alpha
    g = np.zeros(n)
    for pos, m_idx in enumerate(active_idx):
        wk = w[pos]
        if wk < 1e-12:
            continue
        k = m_idx - (n - 1)
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - k:i_hi - k + 1])
        i_lo2 = max(0, -k)
        i_hi2 = min(n - 1, n - 1 - k)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 + k:i_hi2 + k + 1]
    return sm_val, g

def nesterov_pg(h0, n, ts, alpha, n_iters, step_init, time_budget):
    t_local = time.time()
    h = h0.copy()
    y = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    for k in range(n_iters):
        if time.time() - t_local > time_budget:
            break
        if k > 1:
            mu = (k - 1.0) / (k + 2.0)
        else:
            mu = 0.0
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, g = smooth_max_val_grad(y, alpha)
        h_new = y - step * g
        h_new = project_hypersimplex(h_new, ts)
        c5_new = compute_c5(h_new)
        sm_y, _ = smooth_max_val_grad(y, alpha)
        if c5_new > sm_y * (2.0 / n) + 0.01:
            step *= 0.7
            h_new = y - step * g
            h_new = project_hypersimplex(h_new, ts)
            c5_new = compute_c5(h_new)
        else:
            step = min(step * 1.05, 0.6)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h
        h = h_new
    return best_h, best_c5

def slsqp_active_opt(h0, n, ts, active_idx, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = active_lags_smooth(h, active_idx, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = active_lags_smooth(h, active_idx, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_hypersimplex(res.x, ts)
    except Exception:
        return project_hypersimplex(h0, ts)

def slsqp_full_opt(h0, n, ts, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',

                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_hypersimplex(res.x, ts)
    except Exception:
        return project_hypersimplex(h0, ts)

def get_active_lags(h, frac=0.98):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    m = np.max(corr)
    active = np.where(corr >= frac * m)[0]
    return active, m

def coord_descent(h0, n, ts, max_rounds, rng, time_budget):
    t_local = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
    for rnd in range(max_rounds):
        if time.time() - t_local > time_budget:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() - t_local > time_budget:
                break
            for d in deltas:
                if h[i] + d < 0 or h[i] + d > 1:
                    continue
                h_new = h.copy()
                h_new[i] += d
                h_new = project_hypersimplex(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def random_perturb_restart(h0, n, ts, rng, n_starts, alpha, iters_per, time_budget):
    t_local = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for s in range(n_starts):
        if time.time() - t_local > time_budget:
            break
        eps = rng.randn(n) * 0.03
        h_pert = project_hypersimplex(h0 + eps, ts)
        h_res, c5_res = nesterov_pg(h_pert, n, ts, alpha, iters_per, 0.15,
                                    time_budget - (time.time() - t_local))
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res
    return best_h, best_c5

        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_hypersimplex(h_hi, ts_hi)
        for alpha_hi in [100.0, 500.0, 2000.0]:
            if tl() < 100:
                break
            h_hi, c5_hi = nesterov_pg(h_hi, n_hi, ts_hi, alpha_hi, 2000, 0.2,
                                      tl() * 0.8)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
        if tl() > 60:
            active_hi, _ = get_active_lags(h_hi, 0.97)
            h_hi2 = slsqp_active_opt(h_hi, n_hi, ts_hi, active_hi, 2000.0, 200)
            c5_hi2 = compute_c5(h_hi2)
            if c5_hi2 < best_c5:
                best_c5 = c5_hi2
                best_h = h_hi2.copy()
                best_n = n_hi
        if tl() > 40:
            h_hi3, c5_hi3 = coord_descent(h_hi, n_hi, ts_hi, 20, rng, tl() * 0.7)
            if c5_hi3 < best_c5:
                best_c5 = c5_hi3
                best_h = h_hi3.copy()
                best_n = n_hi

    # Phase 8: Final high-alpha Nesterov + SLSQP on best
    n_f = best_n
    ts_f = n_f / 2.0
    h_f = best_h.copy()
    if tl() > 80:
        h_f, c5_f = nesterov_pg(h_f, n_f, ts_f, 5000.0, 3000, 0.08, tl() * 0.8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_f
    if tl() > 50:
        active_f, _ = get_active_lags(h_f, 0.96)
        h_f2 = slsqp_active_opt(h_f, n_f, ts_f, active_f, 5000.0, 250)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            best_n = n_f
    if tl() > 30:
        h_f3, c5_f3 = coord_descent(best_h, best_n, best_n / 2.0, 30, rng, tl() * 0.7)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()
            best_n = best_n

    # Final validation and return
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_hypersimplex(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
