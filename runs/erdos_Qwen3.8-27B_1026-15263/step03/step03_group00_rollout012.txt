    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / alpha

def smooth_max_grad(h, alpha):

    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):

        return smooth_max_grad(x, alpha)

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)

    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_base = best_n
    ts_base = n_base / 2.0

    # Phase 1: DE for global exploration
    pop_size = 25
    F = 0.7
    CR = 0.6
    pop = np.zeros((pop_size, n_base))
    pop_c5 = np.full(pop_size, 1e10)
    pop[0] = best_h
    pop_c5[0] = best_c5

    for i in range(1, 5):
        scale = 0.08 + 0.12 * (i / 5.0)
        h_noise = best_h + rng.uniform(-scale, scale, n_base)
        pop[i] = project_fast(h_noise, ts_base)
        pop_c5[i] = compute_c5(pop[i])

    for i in range(5, pop_size):
        t = np.linspace(0, 2.0, n_base, endpoint=False)
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h_struct = 0.5 + 0.3 * np.sin(freq * np.pi * t + phase)
        h_struct = h_struct + rng.uniform(-0.05, 0.05, n_base)
        pop[i] = project_fast(h_struct, ts_base)
        pop_c5[i] = compute_c5(pop[i])

    for i in range(pop_size):
        if pop_c5[i] < best_c5:
            best_c5 = pop_c5[i]
            best_h = pop[i].copy()

    stuck_count = 0
    for gen in range(600):
        remaining = budget_s - (time.time() - t_start)
        if remaining < 350:
            break
        improved = 0
        for i in range(pop_size):
            idx = list(range(pop_size))
            idx.remove(i)
            a, b, c = rng.choice(idx, size=3, replace=False)
            if gen % 5 == 0 and rng.random() < 0.3:
                best_idx = int(np.argmin(pop_c5))
                mutant = pop[best_idx] + F * (pop[a] - pop[b])
            else:
                mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            j_rand = rng.randint(n_base)
            mask = rng.random(n_base) < CR
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = project_fast(trial, ts_base)
            c5_trial = compute_c5(trial)
            if c5_trial < pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                improved += 1
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
        if improved == 0:
            stuck_count += 1
            if stuck_count > 30 and gen % 50 == 0:
                for i in range(pop_size):
                    scale = 0.1 + 0.2 * rng.random()
                    h_noise = best_h + rng.uniform(-scale, scale, n_base)
                    pop[i] = project_fast(h_noise, ts_base)
                    pop_c5[i] = compute_c5(pop[i])
                stuck_count = 0
        else:
            stuck_count = 0

    # Phase 2: Gradient refinement with increasing alpha
    for alpha in [30, 100, 500, 2000, 10000, 50000]:
        remaining = budget_s - (time.time() - t_start)
        if remaining < 60:
            break
        h_new = slsqp_refine(best_h, n_base, ts_base, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Coordinate descent
    remaining = budget_s - (time.time() - t_start)
    if remaining > 120:
        h_cd, c5_cd = coordinate_descent(best_h, ts_base, max_iters=15000, step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Multi-lag coordinate descent
    remaining = budget_s - (time.time() - t_start)
    if remaining > 100:
        h_ml, c5_ml = multi_lag_coord_descent(best_h, ts_base, max_iters=8000, n_lags=3)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 5: Multi-resolution
    for n_tr in [420, 480, 600, 720, 900]:
        remaining = budget_s - (time.time() - t_start)
        if remaining < 200:
            break
        if n_tr % 2 != 0:
            n_tr += 1
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

        remaining = budget_s - (time.time() - t_start)
        if remaining < 120:
            continue

        pop2_size = 12
        pop2 = np.zeros((pop2_size, n_tr))
        pop2[0] = h_interp
        pop2_c5 = np.full(pop2_size, 1e10)
        pop2_c5[0] = c5_interp
        for i in range(1, pop2_size):
            h_n = h_interp + rng.uniform(-0.04, 0.04, n_tr)
            pop2[i] = project_fast(h_n, ts_tr)
            pop2_c5[i] = compute_c5(pop2[i])

        stuck2 = 0
        for gen in range(40):
            remaining = budget_s - (time.time() - t_start)
            if remaining < 100:
                break
            imp2 = 0
            for i in range(pop2_size):
                idx2 = list(range(pop2_size))
                idx2.remove(i)
                a2, b2, c2 = rng.choice(idx2, size=3, replace=False)
                mutant2 = pop2[a2] + 0.6 * (pop2[b2] - pop2[c2])
                trial2 = pop2[i].copy()
                j_r = rng.randint(n_tr)
                msk = rng.random(n_tr) < 0.6
                msk[j_r] = True
                trial2[msk] = mutant2[msk]
                trial2 = project_fast(trial2, ts_tr)
                c5_t = compute_c5(trial2)
                if c5_t < pop2_c5[i]:
                    pop2[i] = trial2
                    pop2_c5[i] = c5_t
                    imp2 += 1
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = trial2.copy()
                        best_n = n_tr
            if imp2 == 0:
                stuck2 += 1
                if stuck2 > 15:
                    break
            else:
                stuck2 = 0

        remaining = budget_s - (time.time() - t_start)
        if remaining > 60:
            h_btr = pop2[np.argmin(pop2_c5)].copy()
            for alpha2 in [200, 2000, 10000]:
                remaining = budget_s - (time.time() - t_start)
                if remaining < 30:
                    break
                h_opt = slsqp_refine(h_btr, n_tr, ts_tr, alpha2, maxiter=600)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                    h_btr = h_opt
            remaining = budget_s - (time.time() - t_start)
            if remaining > 30:
                h_cd2, c5_cd2 = coordinate_descent(best_h, best_n / 2.0, max_iters=5000, step_frac=0.08)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()

    # Phase 6: Final polish
    n_final = best_n
    ts_final = n_final / 2.0

    for alpha in [50000, 200000]:
        remaining = budget_s - (time.time() - t_start)
        if remaining < 25:
            break
        h_pol = slsqp_refine(best_h, n_final, ts_final, alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    remaining = budget_s - (time.time() - t_start)
    if remaining > 30:
        h_fc, c5_fc = coordinate_descent(best_h, ts_final, max_iters=15000, step_frac=0.05)
        if c5_fc < best_c5:
            best_c5 = c5_fc
            best_h = h_fc.copy()

    remaining = budget_s - (time.time() - t_start)
    if remaining > 20:
        h_ml2, c5_ml2 = multi_lag_coord_descent(best_h, ts_final, max_iters=8000, n_lags=4)
        if c5_ml2 < best_c5:
            best_c5 = c5_ml2
            best_h = h_ml2.copy()

    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
