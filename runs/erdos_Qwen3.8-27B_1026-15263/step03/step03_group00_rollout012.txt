        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)

        obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * (2.0/n)

        return obj, grad

        res = minimize(lambda x: f_and_g(x)[0], h.copy(), method='SLSQP',
                       jac=lambda x: f_and_g(x)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def sg_descent(h, target_sum, n_iters, eta0, top_k, check_interval=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for t in range(n_iters):
        corr = compute_corr(h)
        if top_k <= 1:
            j_star = int(np.argmax(corr))
            lag = j_star - (n - 1)
            g = corr_gradient(h, lag)
        elif top_k <= 10:
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
            g = np.zeros(n)
            for idx in top_idx:
                lag = int(idx) - (n - 1)
                g += corr_gradient(h, lag)
            g /= float(top_k)
        else:
            top_idx = np.argpartition(corr, -20)[-20:]
            m_max = float(np.max(corr))
            w = np.exp(50.0 * (corr[top_idx] - m_max))
            w /= np.sum(w)
            g = np.zeros(n)
            for i in range(len(top_idx)):
                if w[i] < 1e-6:
                    continue
                lag = int(top_idx[i]) - (n - 1)
                g += w[i] * corr_gradient(h, lag)
        eta = eta0 / (1.0 + t / 8000.0)
        step_scale = eta * dx
        h = h - step_scale * g
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        if (t + 1) % check_interval == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def slsqp_smooth(h, target_sum, alpha, maxiter=400):
    n = len(h)
    dx = 2.0 / n

    def obj_fn(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = alpha * (corr - m)
        sh -= np.max(sh)
        ev = np.exp(sh)
        return (m + np.log(np.sum(ev)) / alpha) * dx

    def grad_fn(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = alpha * (corr - m)
        sh -= np.max(sh)
        ev = np.exp(sh)
        w = ev / np.sum(ev)
        g = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-7:
                continue
            lag = j - (n - 1)
            g += w[j] * corr_gradient(x, lag)
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h.copy(), target_sum)

def multi_lag_gradient(h, top_k):
    n = len(h)
    corr = compute_corr(h)
    if top_k <= 10:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += corr_gradient(h, lag)
        g /= float(top_k)
    else:
        top_idx = np.argpartition(corr, -25)[-25:]
        m_max = float(np.max(corr))
        w = np.exp(40.0 * (corr[top_idx] - m_max))
        w /= np.sum(w)
        g = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-6:
                continue
            lag = int(top_idx[i]) - (n - 1)
            g += w[i] * corr_gradient(h, lag)
    return g

def accelerated_sg(h, target_sum, n_iters, eta0, top_k, check_interval=30):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    y = h.copy()
    t_val = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for t in range(n_iters):
        t_val = t + 1
        g = multi_lag_gradient(y, top_k)
        eta = eta0 / (1.0 + t_val / 5000.0)
        h_new = y - eta * dx * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = 1.0 - t_val / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target_sum)
        h = h_new
        if (t + 1) % check_interval == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

            break
        if trial == 0:

            W = n0 // 6

        else:

                            h_bin[idx] = 0.0
            h_pert = h_bin * 0.3 + best_h * 0.7

        h_pert = project_to_sum(h_pert, ts0)
