def smooth_max_and_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)

    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]

    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

_OPT_PARAMS = {'alpha': 100.0, 'target_sum': 120.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def _squared_sum_err(x):
    return np.sum(x) - _OPT_PARAMS['target_sum']

def _squared_sum_err_jac(x):
    return np.ones(len(x))

def _smooth_obj(x):
    return smooth_max_and_grad(x, _OPT_PARAMS['alpha'])[0]

def _smooth_grad(x):
    return smooth_max_and_grad(x, _OPT_PARAMS['alpha'])[1]

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800, t_end=None):
    _OPT_PARAMS['alpha'] = alpha
    _OPT_PARAMS['target_sum'] = target_sum
    cons = [{'type': 'eq', 'fun': _squared_sum_err, 'jac': _squared_sum_err_jac}]
    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': maxiter, 'ftol': 1e-15}
    if t_end is not None:
        opts['maxiter'] = maxiter  # will check time in run
    try:
        res = minimize(_smooth_obj, h0.copy(), method='SLSQP', jac=_smooth_grad,
                       bounds=bounds, constraints=cons, options=opts)
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.05
    no_improve = 0
    min_T = 0.001
    
    while time.time() < t_end:
        move_type = rng.randint(3)
        accepted = False
        if move_type == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            accepted = True
        elif move_type == 1:
            blen = rng.randint(2, max(3, n // 30))
            i = rng.randint(max(1, n - blen))
            j = rng.randint(max(1, n - blen))
            if abs(i - j) <= blen:
                continue
            block_sum_i = int(np.sum(h[i:i+blen]))
            block_sum_j = int(np.sum(h[j:j+blen]))
            if block_sum_i != block_sum_j:
                continue
            tmp = h[i:i+blen].copy()
            h[i:i+blen] = h[j:j+blen]
            h[j:j+blen] = tmp
            accepted = True
        else:
            blen = rng.randint(3, max(4, n // 20))
            i = rng.randint(max(1, n - blen))
            h[i:i+blen] = h[i:i+blen][::-1]
            accepted = True
        
        if not accepted:
            continue
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / T) + np.log(rng.uniform()) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if move_type == 0:
                h[i], h[j] = h[j], h[i]
            elif move_type == 1:
                tmp = h[i:i+blen].copy()
                h[i:i+blen] = h[j:j+blen]
                h[j:j+blen] = tmp
            else:
                h[i:i+blen] = h[i:i+blen][::-1]
        
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.03)
            no_improve = 0
        T = max(T * 0.999998, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

        h_960 = interpolated_h(best_h, n_480, n_960)
        h_960 = project_to_sum(h_960, ts_960)
        c5_960 = compute_c5(h_960)
        if c5_960 < best_c5:
            best_c5 = c5_960
            best_h = h_960.copy()
            best_n = n_960

        for alpha in [100, 500, 3000]:
            if time.time() > t_start + 0.82 * budget_s:
                break
            h_opt960 = slsqp_optimize(best_h, n_960, ts_960, alpha, maxiter=800)
            c5_opt960 = compute_c5(h_opt960)
            if c5_opt960 < best_c5:
                best_c5 = c5_opt960
                best_h = h_opt960.copy()

    if time.time() < t_start + budget_s - 50:
        if best_n != n_cur:
            h_down = interpolated_h(best_h, best_n, n_cur)
            h_down = project_to_sum(h_down, ts_cur)
        else:
            h_down = best_h.copy()
        h_bin_final = make_binary(h_down, n_cur, int(ts_cur), rng)
        sa_end2 = min(t_start + budget_s - 20, time.time() + 80)
        h_final_sa, c5_final_sa = binary_sa(h_bin_final, n_cur, int(ts_cur), sa_end2, rng)
        if c5_final_sa < best_c5:
            best_c5 = c5_final_sa
            best_h = h_final_sa.copy()
            best_n = n_cur

    if time.time() < t_start + budget_s - 5:
        for alpha in [30000]:
            if time.time() > t_start + budget_s - 2:
                break
            h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=1500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
