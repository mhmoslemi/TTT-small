```python
def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj, w

def smooth_max_grad(h, alpha):
    n = len(h)
    obj, w = smooth_max_obj(h, alpha)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_step(h0, n, target_sum, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj(x, alpha)[0]
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0.copy()
```

```python
    # c[k] = sum_j h[j + offset] * (1-h[j]) for 0 <= j+offset < n and 0 <= j < n
    
    offset = k_star - (n - 1)
    # Valid j: max(0, -offset) <= j <= min(n-1, n-1-offset)
    j_min = max(0, -offset)
    j_max = min(n - 1, n - 1 - offset)
    
    # Grad w.r.t. h[i]:
    # Term 1: i = j + offset, so j = i - offset. Valid if j_min <= i-offset <= j_max
    #   contribution: -(1-h[j]) = -(1-h[i-offset])
    # Term 2: i = j, so i is in [j_min, j_max]
    #   contribution: -h[i+offset]
    
    for j in range(j_min, j_max + 1):
        i1 = j + offset  # where h appears as first factor
        i2 = j           # where (1-h) appears
        if 0 <= i1 < n:
            grad[i1] -= (1.0 - h[j])
        if 0 <= i2 < n:
            grad[i2] -= h[i1]
    
    # Project gradient descent step
    h_new = h - step_size * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new
```

```python
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        n_free = np.sum((h > 1e-12) & (h < 1.0 - 1e-12))
        if n_free > 0:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            h[mask] += diff / n_free
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    if n_ones == target_ones:
        return h_bin
    if n_ones > target_ones:

        order = np.argsort(-h[zeros_idx])[:deficit]

def smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(s)
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    return obj, w

def smooth_max_grad(h, alpha):
    n = len(h)
    obj, w = smooth_max(h, alpha)
    j = np.arange(n)[:, None]
    m = np.arange(2 * n - 1)[None, :]
    fi = j - m + n - 1
    si = j + m - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    grad = cf.sum(axis=1) - cs.sum(axis=1)
    return obj, grad

def slsqp(h0, n, target_sum, alpha, maxiter=1000):
    def f(x):
        return smooth_max(x, alpha)[0]
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0.copy()
```

```python
    grad = cf.sum(axis=1) - cs.sum(axis=1)
    return grad

def slsqp_step(h0, n, target_sum, alpha, maxiter=600):
    def f(x):
        return smooth_max_obj(x, alpha)
    def g(x):
        return smooth_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0.copy()

def interp_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def multi_start_sa(h_starts, n, target_ones, t_end, rng, T_init=0.010):
    best_h = None
    best_c5 = float('inf')
    per_start_time = max(1.0, (t_end - time.time()) / len(h_starts))
    for h_s in h_starts:
        t_sub = time.time() + per_start_time
        if time.time() >= t_end:
            break
        h_res, c5_res = sa_optimize(h_s, n, target_ones, t_sub, rng, T_init=T_init)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = random_binary(best_n, best_n // 2, rng)
        best_c5 = compute_c5(best_h)

    n0 = best_n
    ts0 = n0 / 2.0
    target_ones0 = int(ts0)

    # Phase 1: SA at base resolution from parent
    h_bin = make_binary(best_h, n0, target_ones0, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    t_sa1_end = t0 + 0.30 * budget_s
    h_sa1, c5_sa1 = sa_optimize(h_bin, n0, target_ones0, t_sa1_end, rng, T_init=0.012)
    if c5_sa1 < best_c5:
        best_c5 = c5_sa1
        best_h = h_sa1.copy()

    # Phase 2: SLSQP refinement at base resolution
    for alpha in [30, 100, 500, 2000, 8000]:
        if time_left() < 25:
            break
        h_opt = slsqp_step(best_h, n0, ts0, alpha, maxiter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Multi-start SA at base resolution with different inits
    if time_left() > 100:
        starts = [best_h.copy()]
        for _ in range(3):
            noise = rng.uniform(-0.15, 0.15, n0)
            h_noisy = project_to_sum(best_h + noise, ts0)
            starts.append(make_binary(h_noisy, n0, target_ones0, rng))
        t_sa2_end = t0 + 0.48 * budget_s
        h_ms, c5_ms = multi_start_sa(starts, n0, target_ones0, t_sa2_end, rng, T_init=0.009)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 4: Higher resolution attempts
    for n_try in [360, 480, 600]:
        if time_left() < 150:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        target_ones_try = int(ts_try)
        h_interp = interp_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        h_bin_try = make_binary(h_interp, n_try, target_ones_try, rng)

        sa_time = min(0.12 * budget_s, time_left() - 80)
        if sa_time < 30:
            continue
        t_sa_hr_end = time.time() + sa_time
        h_sa_hr, c5_sa_hr = sa_optimize(h_bin_try, n_try, target_ones_try, t_sa_hr_end, rng, T_init=0.008)

        h_cand = h_sa_hr if c5_sa_hr <= compute_c5(h_bin_try) else h_bin_try
        for alpha in [30, 200, 1000]:
            if time_left() < 25:
                break
            h_cand2 = slsqp_step(h_cand, n_try, ts_try, alpha, maxiter=500)
            c5_cand2 = compute_c5(h_cand2)
            if c5_cand2 < best_c5:
                best_c5 = c5_cand2
                best_h = h_cand2.copy()
                best_n = n_try
                h_cand = h_cand2.copy()

    # Phase 5: Final SA + SLSQP at best resolution
    n_now = best_n
    ts_now = n_now / 2.0
    target_ones_now = int(ts_now)

    if time_left() > 80:
        h_bin_now = make_binary(best_h, n_now, target_ones_now, rng)
        t_final_sa_end = min(t0 + budget_s - 40, time.time() + 0.10 * budget_s)
        h_fs, c5_fs = sa_optimize(h_bin_now, n_now, target_ones_now, t_final_sa_end, rng, T_init=0.005)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    for alpha in [10000, 30000, 80000]:
        if time_left() < 15:
            break
        h_fin = slsqp_step(best_h, best_n, ts_now, alpha, maxiter=1200)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
```
