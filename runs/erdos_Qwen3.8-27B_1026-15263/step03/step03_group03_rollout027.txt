            continue
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] += -wk * h[:-s]
        else:
            sa = -s
            g[sa:] += wk * (1.0 - h[:n - sa])
            g[:n - sa] += -wk * h[sa:]
    return g

def build_hessian_toeplitz(w, n):
    diag = np.zeros(n)
    diag[0] = -2.0 * w[n - 1]
    for s in range(1, n):
        diag[s] = -(w[n - 1 + s] + w[n - 1 - s])
    idx = np.arange(n)
    return diag[np.abs(idx[:, None] - idx[None, :])]

def _obj_func(h):
    return np.dot(_g_w, np.correlate(h, 1.0 - h, mode="full"))

def _grad_func(h):
    return compute_grad(h, _g_w, _g_n)

def _hess_func(h):
    return _g_H

def _con_fun(h):
    return np.sum(h) - _g_n * 0.5

def _con_jac(h):
    return np.ones(_g_n)

def solve_qp(h0, w, n, maxiter=250):
    global _g_w, _g_n, _g_H
    _g_w = w
    _g_n = n
    _g_H = build_hessian_toeplitz(w, n)
    target = n * 0.5
    h0 = project_to_sum(h0, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_func, h0.copy(), jac=_grad_func, hess=_hess_func,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = np.clip(res.x, 0.0, 1.0)
        return project_to_sum(hx, target)
    except Exception:
        return project_to_sum(h0, target)

def _compute_pair_D(h, j, k, n):
    nm1 = n - 1
    D = np.zeros(2 * n - 1)
    for kk in range(2 * n - 1):
        s = kk - nm1
        dj = 0.0
        dk = 0.0
        if s == 0:
            dj = 1.0 - 2.0 * h[j]
            dk = 1.0 - 2.0 * h[k]
        else:
            if 0 <= j + s < n:
                dj += (1.0 - h[j + s])
            if 0 <= j - s < n:
                dj += (-h[j - s])
            if 0 <= k + s < n:
                dk += (1.0 - h[k + s])
            if 0 <= k - s < n:
                dk += (-h[k - s])
        D[kk] = dj - dk
    return D

def _solve_1d(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(30):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5
