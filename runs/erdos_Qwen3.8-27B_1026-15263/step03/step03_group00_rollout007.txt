        obj_new = obj_fn(h_new)

def lbfgs_smooth_max(h0, n, target_sum, alpha, topk, max_iter=500):
    def f(x):
        return smooth_max_obj_grad(x, alpha, topk)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha, topk)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def structured_perturb(h, rng, n, target_sum, pert_type):
    """Apply structured perturbation."""
    h_new = h.copy()
    if pert_type == 0:
        # Local flip: pick a block and flip
        block_size = rng.randint(n // 20, n // 5)
        start = rng.randint(0, n - block_size)
        h_new[start:start+block_size] = 1.0 - h_new[start:start+block_size]
    elif pert_type == 1:
        # Shift: circular shift by small amount
        shift = rng.randint(1, max(2, n // 100))
        h_new = np.roll(h, shift)
    elif pert_type == 2:
        # Gaussian noise
        sigma = 0.05 + 0.1 * rng.random()
        h_new = h + rng.normal(0, sigma, n)
    elif pert_type == 3:
        # Local noise
        n_points = rng.randint(n // 10, n // 3)
        indices = rng.choice(n, n_points, replace=False)
        h_new[indices] += rng.uniform(-0.2, 0.2, n_points)
    elif pert_type == 4:
        # Bimodal push
        h_new = np.where(h > 0.5, np.minimum(1.0, h + 0.1), np.maximum(0.0, h - 0.1))
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def compute_corr(h):

def project_to_sum(h, target_sum):

            break

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_obj_grad(h, alpha, topk=None):
    n = len(h)
    corr = compute_corr(h)
    total_lags = len(corr)
    if topk is not None and topk < total_lags:
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:topk]
        active_corr = corr[active]
    else:
        active = np.arange(total_lags)
        active_corr = corr
    m_val = np.max(active_corr)
    shifted = alpha * (active_corr - m_val)
    max_shifted = np.max(shifted)
    shifted -= max_shifted
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    g = np.zeros(n)
    for idx, wi in zip(active, w):
        if wi > 1e-14:
            lag = idx - (n - 1)
            g += wi * lag_gradient(h, lag)
    return obj, g

        h_next = h_curr - gamma * g
        h_next = np.clip(h_next, 0.0, 1.0)
        h_next = project_to_sum(h_next, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_curr = h_next + ((t_k - 1.0) / t_new) * (h_next - h_prev)
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = project_to_sum(h_curr, target_sum)
        t_k = t_new
        h_prev = h_next
    return h_curr

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts = n_cur / 2.0

    # Phase 1: L-BFGS smooth max with increasing alpha and decreasing topk
    alpha_topk_schedule = [
        (40, 200), (80, 120), (150, 60), (300, 40),
        (600, 30), (1200, 20), (3000, 15), (8000, 10),
        (20000, 7), (50000, 5), (100000, 3), (300000, 2),
    ]
    for alpha, topk in alpha_topk_schedule:
        if time_left() < 20:
            break
        h_opt = lbfgs_smooth_max(best_h, n_cur, ts, alpha, topk, max_iter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Coordinate descent with top-k awareness
    if time_left() > 60:
        h_cd, c5_cd = coord_descent_topk(best_h, ts, max_iters=10000, step_frac=0.15, topk=25)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: FISTA at medium alpha
    if time_left() > 40:
        h_fista = fista_smooth_max(best_h, ts, alpha=500, topk=30, max_iters=400)
        c5_f = compute_c5(h_fista)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fista.copy()

    # Phase 4: Second L-BFGS pass with refined parameters
    for alpha, topk in [(1000, 20), (5000, 8), (20000, 4), (100000, 2)]:
        if time_left() < 20:
            break
        h_opt = lbfgs_smooth_max(best_h, n_cur, ts, alpha, topk, max_iter=500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time_left() > 30:
            h_cd, c5_cd = coord_descent_topk(best_h, ts, max_iters=6000, step_frac=0.1, topk=10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 5: Periodic structure search
    if time_left() > 100:
        divisors = []
        for p in range(2, n_cur // 2 + 1):
            if n_cur % p == 0:
                divisors.append(p)
        divisors.sort(key=lambda x: -x)
        for period in divisors[:15]:
            if time_left() < 30:
                break
            if period > n_cur // 3:
                continue
            h_per = periodic_search(best_h, n_cur, ts, period, alpha=2000, topk=15, max_iter=200)
            if h_per is not None:
                c5_per = compute_c5(h_per)
                if c5_per < best_c5:
                    best_c5 = c5_per
                    best_h = h_per.copy()

    # Phase 6: Multi-resolution refinement
    for n_tr in [720, 900]:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        h_tr = interp_to_n(best_h, n_tr)
        h_tr = project_to_sum(h_tr, ts_tr)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr
            n_cur = n_tr
            ts = ts_tr
        for alpha, topk in [(300, 50), (2000, 15), (20000, 5)]:
            if time_left() < 40:
                break
            h_opt = lbfgs_smooth_max(h_tr, n_tr, ts_tr, alpha, topk, max_iter=350)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_tr = h_opt
        if time_left() > 30:
            h_cd, c5_cd = coord_descent_topk(h_tr, ts_tr, max_iters=6000, step_frac=0.1, topk=15)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr

    # Phase 7: Structured perturbations with local optimization
    n_perturb = 18
    for trial in range(n_perturb):
        if time_left() < 35:
            break
        pert_type = trial % 6
        h_pert = structured_perturb(best_h, rng, best_n, ts, pert_type)
        h_cd, c5_cd = coord_descent_topk(h_pert, ts, max_iters=4000, step_frac=0.12, topk=15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            if time_left() > 30:
                h_opt = lbfgs_smooth_max(h_cd, best_n, ts, 5000, 5, max_iter=400)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 8: Aggressive final L-BFGS at very high alpha
    for alpha, topk in [(50000, 3), (200000, 2), (1000000, 1)]:
        if time_left() < 15:
            break
        h_opt = lbfgs_smooth_max(best_h, best_n, ts, alpha, topk, max_iter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final coordinate descent polish
    if time_left() > 25:
        h_cd, c5_cd = coord_descent_topk(best_h, ts, max_iters=15000, step_frac=0.05, topk=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Another L-BFGS after final CD
    if time_left() > 20:
        h_opt = lbfgs_smooth_max(best_h, best_n, ts, 500000, 1, max_iter=500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time_left() > 15:
            h_cd, c5_cd = coord_descent_topk(best_h, ts, max_iters=10000, step_frac=0.03, topk=3)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final projection and verification
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
