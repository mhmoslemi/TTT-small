    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        if abs(diff) < 1e-13:

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:

        h = np.clip(h, 0.0, 1.0)

def compute_all_B(h, n):
    """Precompute B[j] for all j."""
    nm1 = n - 1
    B = np.zeros((n, 2 * n - 1))
    for j in range(n):
        k_start = max(0, nm1 - j)
        k_end = min(2 * n - 2, 2 * nm1 - j)
        if k_start <= k_end:
            B[j, k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
        k_start2 = max(0, j)
        k_end2 = min(2 * n - 2, j + nm1)
        if k_start2 <= k_end2:
            B[j, k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d_fast(A, D, lo, hi):
    """Find delta that minimizes max(A + D*delta) over delta in [lo, hi]."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Quick check at endpoints
    val_lo = np.max(A + D * lo)
    val_hi = np.max(A + D * hi)
    # Subgradient check
    imax_lo = np.argmax(A + D * lo)
    imax_hi = np.argmax(A + D * hi)
    if D[imax_lo] >= 0:
        return lo  # function is increasing at lo
    if D[imax_hi] <= 0:
        return hi  # function is decreasing at hi
    # Grid search for better starting point
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        val_mid = np.max(A + D * mid)
        if val_mid <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

    n = len(h)

        current_max = np.max(corr)
        best_val = current_max
        best_delta = 0.0
        best_k = -1
        for k in range(n):
            if k == j:

            D = B_all[j] - B_all[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:

            delta_opt = solve_1d_fast(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)

                best_delta = delta_opt

            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
            best_improvement = max(best_improvement, current_max - best_val)
    return h, corr, improved, best_improvement

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_B_vec(h):
    n = len(h)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    idx1 = j_idx + m_idx - n + 1
    idx2 = j_idx - m_idx + n + 1  # Wait, let me recheck
    # B[j, m]: change in corr[m] when h[j] increases by 1
    # Term 1: (1 - h[j+m-n+1]) for valid j+m-n+1
    # Term 2: -h[j-m+n-1] for valid j-m+n-1
    idx1 = j_idx + m_idx - (n - 1)
    idx2 = j_idx - m_idx + (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    ci1 = np.clip(idx1, 0, n - 1)
    ci2 = np.clip(idx2, 0, n - 1)
    B = (1.0 - h[ci1]) * valid1.astype(float) - h[ci2] * valid2.astype(float)
    return B

def solve_1d_fast(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Quick subgradient check at endpoints
    val_lo = np.max(A + D * lo)
    imax_lo = np.argmax(A + D * lo)
    if D[imax_lo] >= 0:
        return lo
    val_hi = np.max(A + D * hi)
    imax_hi = np.argmax(A + D * hi)
    if D[imax_hi] <= 0:
        return hi
    # Grid search
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def full_pair_sweep(h, corr, B_all):
    n = len(h)
    improved = False
    total_improvement = 0.0
    for j in range(n):
        current_max = corr.max()
        best_val = current_max
        best_delta = 0.0
        best_k = -1
        Bj = B_all[j]
        for k in range(n):
            if k == j:
                continue
            D = Bj - B_all[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_fast(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            D = Bj - B_all[best_k]
            corr += best_delta * D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
            total_improvement += (current_max - best_val)
    return h, corr, improved, total_improvement

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    idx1 = j_idx + m_idx - (n - 1)
    idx2 = j_idx - m_idx + (n - 1)
    valid1 = ((idx1 >= 0) & (idx1 < n)).astype(float)
    valid2 = ((idx2 >= 0) & (idx2 < n)).astype(float)
    ci1 = np.clip(idx1, 0, n - 1)
    ci2 = np.clip(idx2, 0, n - 1)
    # d(corr[m])/d(h[j]):
    # = (1-h[j+m-n+1]) * valid1 - h[j-m+n-1] * valid2  (this is B[j,m])
    B_mat = (1.0 - h[ci1]) * valid1 - h[ci2] * valid2
    # grad[j] = sum_m w[m] * d(corr[m])/d(h[j])
    # But corr[m] = sum_i h[i]*(1-h[i+m-n+1])
    # d(corr[m])/d(h[j]) = (1-h[j+m-n+1])*[valid] + sum over i where i+m-n+1=j of h[i]*(-1)*[valid]
    # = (1-h[j+m-n+1])*[valid1] - h[j-m+n-1]*[valid2]
    # So grad = B_mat.T @ w
    grad = B_mat.T @ w
    return obj, grad

def slsqp_polish(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
