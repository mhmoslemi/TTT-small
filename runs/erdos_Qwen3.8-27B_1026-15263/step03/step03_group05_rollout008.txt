    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_value(corr, alpha):
    max_c = np.max(corr)
    shifted = corr - max_c
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    logsum = np.log(np.sum(exp_vals))
    return max_c + max_shift + logsum / alpha

def smooth_max_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = corr - max_c
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    return exp_vals / np.sum(exp_vals)

def compute_grad_smooth(h, w):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n-1):
        if abs(w[m]) < 1e-12:
            continue
        k = m - (n-1)
        lo = max(0, -k)
        hi = min(n, n-k)
        j = np.arange(lo, hi)
        grad[j] += w[m] * ((1.0 - h[j]) - h[j+k])
    return grad * (2.0 / n)

def compute_grad_at_shifts(h, shifts, n):
    # Gradient of sum of corr at specific shifts
    n = len(h)
    grad = np.zeros(n)
    for k in shifts:
        lo = max(0, -k)
        hi = min(n, n-k)
        j = np.arange(lo, hi)
        grad[j] += (1.0 - h[j]) - h[j+k]
    return grad * (2.0 / n)

def top_k_shifts(corr, K):
    # Return the top K shift indices
    idx = np.argsort(corr)[-K:]
    return idx

def flip_improve(h, n, rng, max_rounds=200, time_fn=None, tol=1e-6):
    h = h.copy()

    ts = n / 2.0

        if time_fn and time_fn() < 3:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_k = max_idx - (n-1)
        best_c5 = c5_cur
        best_pair = None
        # Try flips that affect the critical shift
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0

                best_pair = (i, j)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.5
    hi = 1.5 - np.min(y)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_value(corr, alpha):
    max_c = np.max(corr)
    shifted = corr - max_c
    s = np.log(np.sum(np.exp(alpha * shifted)))
    return max_c + s / alpha

def smooth_max_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_vals = np.exp(alpha * shifted)
    return exp_vals / np.sum(exp_vals)

def compute_grad_smooth(h, w):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if abs(wm) < 1e-10:
            continue
        k = m - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo >= hi:
            continue
        j = np.arange(lo, hi)
        grad[j] += wm * ((1.0 - h[j]) - h[j + k])
    return grad * (2.0 / n)

def compute_grad_shifts(h, shift_indices):
    n = len(h)
    grad = np.zeros(n)
    for m in shift_indices:

    m_state = beta1 * m_state + (1.0 - beta1) * grad
    v_state = beta2 * v_state + (1.0 - beta2) * grad * grad
    m_hat = m_state / (1.0 - beta1 ** t)

    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_box_sum(h_new, target_sum)
    return h_new, m_state, v_state

                    if time.time() - t_start > budget_s - 20:
                        break
                    t_adam += 1
                    corr2 = compute_corr(h_init)
                    w2 = smooth_max_weights(corr2, alpha2)
                    grad2 = compute_grad_smooth(h_init, w2)
                    h_init, m_state, v_state = adam_step(h_init, grad2, m_state, v_state, t_adam, lr2, 0.85, 0.999, 1e-8, n / 2.0)
                c5_opt = compute_c5(h_init)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_init.copy()

    # Phase 5: Upsampling to higher resolution
    if time.time() - t_start < budget_s - 100:
        n_up = min(480, n * 2)
        if n_up % 2 != 0:
            n_up += 1
        if n_up > n:
            ts_up = n_up / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_up = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_up, x_old, best_h, period=2.0)
            h_up = project_box_sum(h_up, ts_up)
            
            m_state_up = np.zeros(n_up)
            v_state_up = np.zeros(n_up)
            t_adam_up = 0
            for alpha3 in [50, 500, 5000, 20000]:
                if time.time() - t_start > budget_s - 30:
                    break
                lr3 = 0.02 / (1.0 + 0.02 * np.log1p(alpha3))
                for it3 in range(2000):
                    if time.time() - t_start > budget_s - 10:
                        break
                    t_adam_up += 1
                    corr3 = compute_corr(h_up)
                    w3 = smooth_max_weights(corr3, alpha3)
                    grad3 = compute_grad_smooth(h_up, w3)
                    h_up, m_state_up, v_state_up = adam_step(h_up, grad3, m_state_up, v_state_up, t_adam_up, lr3, 0.85, 0.999, 1e-8, ts_up)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up.copy()
                    best_n = n_up
                    n = n_up

    # Phase 6: Final subgradient polish + flips
    if time.time() - t_start < budget_s - 30:
        h_pol, c5_pol = subgradient_optimize(best_h, best_n, t_start, budget_s, max_iters=3000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() - t_start < budget_s - 15:
        h_fl2, c5_fl2 = binary_flip_improve(best_h, best_n, rng, max_rounds=200, t_start=t_start, budget_s=budget_s)
        if c5_fl2 < best_c5:
            best_c5 = c5_fl2
            best_h = h_fl2.copy()

    # Final: one more high-alpha Adam pass
    if time.time() - t_start < budget_s - 10:
        n_fin = best_n
        h_fin = best_h.copy()
        m_state_f = np.zeros(n_fin)
        v_state_f = np.zeros(n_fin)
        t_f = 0
        for alpha_f in [50000, 200000]:
            if time.time() - t_start > budget_s - 5:
                break
            lr_f = 0.005
            for it_f in range(1000):
                if time.time() - t_start > budget_s - 3:
                    break
                t_f += 1
                corr_f = compute_corr(h_fin)
                w_f = smooth_max_weights(corr_f, alpha_f)
                grad_f = compute_grad_smooth(h_fin, w_f)
                h_fin, m_state_f, v_state_f = adam_step(h_fin, grad_f, m_state_f, v_state_f, t_f, lr_f, 0.85, 0.999, 1e-8, n_fin / 2.0)
            c5_f = compute_c5(h_fin)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_fin.copy()

    # Final validation and cleanup
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
