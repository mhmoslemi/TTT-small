    # Term B: correlate(h, w_pos)[n-1:2n-1]

    # Term C: correlate(1-h, w_neg)[n-1:2n-1]

    # Term D: conv(w_neg, h)[:n]
    term_d = np.convolve(w_neg, h)[:n]
    
    grad = term_a - term_b + term_c - term_d
    return sm_val, grad

def adam_projected(h0, n, target_sum, alpha, n_iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):

    h = h0.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_mask = np.sum(mask)
        if n_mask > 0:
            adj = diff / n_mask
            if np.all(h[mask] + adj >= 0) and np.all(h[mask] + adj <= 1):
                h[mask] += adj
            else:
                h[mask] += np.sign(adj) * min(abs(adj), 1e-6)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    """Compute smooth max value and gradient, vectorized."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n - 1)) / alpha
    
    w_pos = w[n-1:]
    w_neg = np.zeros(n)
    w_neg[1:] = w[:n-1][::-1]
    
    omh = 1.0 - h
    term_a = np.convolve(w_pos, omh)[:n]
    term_b = np.correlate(h, w_pos, mode='full')[n-1:]
    term_c = np.correlate(omh, w_neg, mode='full')[n-1:]
    term_d = np.convolve(w_neg, h)[:n]
    
    grad = term_a - term_b + term_c - term_d
    return sm_val, grad

def adam_loop(h0, n, target_sum, alpha, n_iters, lr, seed=0):
    h = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for t in range(1, n_iters + 1):
        _, grad = smooth_max_grad(h, alpha)
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad * grad
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_refine(h0, n, target_sum, alpha, maxiter=100):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad(h, alpha)
        return v
    def grad_f(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_f, method='SLSQP',
                      bounds=bounds, constraints=constraints, maxiter=maxiter, ftol=1e-16)
        return project_to_sum(res.x, target_sum)
    except:
        return h0.copy()

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur
