from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha, active_lags=None):
    """
    Compute smooth max and its gradient.
    active_lags: array of offsets d (can be negative or positive)
    If None, uses all lags.
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if active_lags is not None:
        # active_lags are offsets d, corresponding to indices n-1+d in corr
        idx = n - 1 + active_lags
        idx = idx[(idx >= 0) & (idx < len(corr))]
        vals = corr[idx]
    else:
        vals = corr
    
    m = np.max(vals)
    shifted = alpha * (vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(vals)) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    if active_lags is not None:
        valid_idx = (n - 1 + active_lags >= 0) & (n - 1 + active_lags < len(corr))
        d_vals = active_lags[valid_idx]
        w_vals = w[valid_idx]
    else:
        d_vals = np.arange(-(n-1), n)
        w_vals = w
    
    for d, wd in zip(d_vals, w_vals):
        if wd < 1e-15:
            continue
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        elif d > 0:
            # First term: (1-h[i+d]) for i in 0..n-1-d
            grad[:n-d] += wd * (1.0 - h[d:])
            # Second term: -h[i-d] for i in d..n-1
            grad[d:] -= wd * h[:n-d]
        else:

            # First term: (1-h[i+d2]) for i in d2..n-1
            grad[d2:] += wd * (1.0 - h[:n-d2])
            # Second term: -h[i-d2]... wait, for d<0, -d = d2
            # shift_left(h, d)[i] = h[i-d] = h[i+d2] for i in 0..n-1-d2
            grad[:n-d2] -= wd * h[d2:]
    
    return sm_val, grad

def projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    """Projected gradient with backtracking"""
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        obj, g = smooth_max_obj_and_grad(h, alpha)
        
        # Backtracking line search
        step = lr
        for bt in range(30):
            h_new = project_to_sum(h - step * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:

            step *= 0.5

        c5_new = compute_c5(h_new)

            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(step * 1.5, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return h_best, c5_cur

    active = np.where(corr >= threshold * max_corr)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_polytope(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum} using bisection on lambda"""
    h = h.astype(float).copy()
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-12:
        mask = (projected > 1e-14) & (projected < 1.0 - 1e-14)
        if np.any(mask):
            projected[mask] += diff / np.sum(mask)
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def smooth_max_and_grad(h, alpha, d_list=None):
    """
    Compute smooth max of f_d(h) and gradient.
    d_list: list of lags to consider (offsets from 0). If None, all lags.
    Returns (smooth_max_value, gradient_vector)
    """
    n = len(h)
    if d_list is None:
        d_list = np.arange(-(n-1), n)
    
    # Compute f_d for each d
    # f_d = sum of h[j]*(1-h[j+d]) for valid j
    # In terms of correlation: f_d = corr[n-1+d]
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get values for our d_list
    idx = n - 1 + d_list
    idx_valid = (idx >= 0) & (idx < len(corr))
    d_valid = d_list[idx_valid]
    vals = corr[idx[idx_valid]]
    
    # Softmax weights
    m = np.max(vals)
    shifted = alpha * (vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(vals)) / alpha
    
    # Compute gradient: grad[i] = sum_d w_d * [(1-h[i+d])*1[0≤i+d<n] - h[i-d]*1[0≤i-d<n]]
    grad = np.zeros(n)
    for k in range(len(d_valid)):
        d = int(d_valid[k])
        wd = w[k]
        if wd < 1e-15:
            continue
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += wd * (1.0 - h[d:])
            grad[d:] -= wd * h[:n-d]
        else:
            d2 = -d
            grad[d2:] += wd * (1.0 - h[:n-d2])
            grad[:n-d2] -= wd * h[d2:]
    
    return sm_val, grad

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init, d_list=None):
    """Nesterov accelerated gradient with projection"""
    h = h0.copy()
    prev_h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    
    for it in range(n_iters):
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        # Nesterov lookahead
        h_look = h + (t/t_new - 1) * (h - prev_h)
        
        obj, g = smooth_max_and_grad(h_look, alpha, d_list)
        
        # Gradient step with projection
        h_new = project_onto_polytope(h_look - lr_init * g, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
        prev_h = h
        h = h_new
        t = t_new
    
    return h_best, c5_cur

def slsqp_with_active(h0, n, target_sum, d_list, alpha, maxiter):
    """SLSQP optimization using active lags only"""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha, d_list)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha, d_list)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_onto_polytope(res.x, target_sum)
    except Exception:
        return project_onto_polytope(h0, target_sum)

def get_active_lags(h, threshold=0.99):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= threshold * max_corr)[0]
    return active - (n - 1)

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    """Fine coordinate descent"""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try small perturbations
            best_val = c5_cur
            best_i_val = h[i]
            for delta in [-0.001, 0.001, -0.0005, 0.0005]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                h_new = project_onto_polytope(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_i_val = np.clip(h[i] + delta, 0.0, 1.0)
            if best_i_val != h[i]:
                h[i] = best_i_val
                h = project_onto_polytope(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

    """Compute smooth max value (fast, only needs d_list lags)."""
    vals = compute_corr_lags(h, d_list)
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)

    sm_val = m + np.log(s / len(vals)) / alpha
    return sm_val, vals, s

def fast_smooth_max_grad(h, alpha, d_list):
    """Compute smooth max value and gradient (fast)."""
    n = len(h)
    vals = compute_corr_lags(h, d_list)
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(vals)) / alpha
    
    grad = np.zeros(n)
    for k in range(len(d_list)):
        d = int(d_list[k])
        wd = w[k]
        if wd < 1e-14:
            continue
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += wd * (1.0 - h[d:])
            grad[d:] -= wd * h[:n-d]
        else:
            d2 = -d
            grad[d2:] += wd * (1.0 - h[:n-d2])
            grad[:n-d2] -= wd * h[d2:]
    return sm_val, grad
