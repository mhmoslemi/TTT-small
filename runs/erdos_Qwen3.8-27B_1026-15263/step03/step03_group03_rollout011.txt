    nm1 = n - 1
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    n_improvements = 0
    
    for j in order:
        B_j = get_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Candidate selection: neighbors, antipodal, random
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (2 * j) % n, (3 * j + 1) % n]
        for _ in range(max(0, n_cand - 7)):
            k_set.append(rng.randint(n))
        
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = get_B(h, k, n)
            D = B_j - B_k
            d_lo = max(-h[j], h[k] - 1.0)
            d_hi = min(1.0 - h[j], h[k])
            if d_hi - d_lo < 1e-14:
                continue
            delta_opt = solve_1d_minmax(corr, D, d_lo, d_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
            n_improvements += 1
    
    # Recompute to avoid accumulation errors
    corr = np.correlate(h, 1.0 - h, mode="full")
    return h, corr, improved, n_improvements

def pair_sweep_fast(h, corr, rng, n_cand=12):
    n = len(h)
    nm1 = n - 1
    L = 2*n - 1
    offset = np.arange(L) - nm1
    
    # Precompute all B
    B_all = np.zeros((n, L))
    for j in range(n):
        ip = j + offset
        vp = (ip >= 0) & (ip < n)
        B_all[j, vp] = 1.0 - h[ip[vp]]
        im = j - offset
        vm = (im >= 0) & (im < n)
        B_all[j, vm] -= h[im[vm]]
    
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for j in order:
        best_val = np.max(corr)
        best_d = 0.0
        best_k = -1
        
        ks = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n,
             (j+2*n//3)%n, (2*j+1)%n, (3*j)%n, (j//2)%n]
        for _ in range(max(0, n_cand - 8)):
            ks.append(rng.randint(n))
        
        seen = set()
        for k in ks:
            if k == j or k in seen: continue
            seen.add(k)
            D = B_all[j] - B_all[k]
            dlo = max(-h[j], h[k]-1.0)
            dhi = min(1.0-h[j], h[k])
            if dhi - dlo < 1e-14: continue
            dopt = solve_1d(corr, D, dlo, dhi)
            vopt = np.max(corr + D*dopt)
            if vopt < best_val - 1e-12:
                best_val = vopt
                best_d = dopt
                best_k = k
        
        if best_k >= 0 and abs(best_d) > 1e-12:
            corr += best_d * (B_all[j] - B_all[best_k])
            h[j] = np.clip(h[j] + best_d, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_d, 0.0, 1.0)
            # Update B_all for j and best_k
            # h[j] changed by best_d, h[best_k] changed by -best_d
            hj_new = h[j]
            hk_new = h[best_k]
            # Recompute B_all[j] and B_all[best_k]
            for jj in [j, best_k]:
                ip = jj + offset
                vp = (ip >= 0) & (ip < n)
                B_all[jj, vp] = 1.0 - h[ip[vp]]
                im = jj - offset
                vm = (im >= 0) & (im < n)
                B_all[jj, vm] -= h[im[vm]]
            improved = True
    
    corr = np.correlate(h, 1-h, mode="full")
    return h, corr, improved

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def solve_1d(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return 0.5 * (lo + hi)
    ng = 24
    tg = np.linspace(lo, hi, ng)
    vals = np.max(corr[None, :] + D[None, :] * tg[:, None], axis=1)
    bi = int(np.argmin(vals))
    if bi == 0:
        return lo
    if bi == ng - 1:
        return hi
    lo_r, hi_r = tg[bi - 1], tg[bi + 1]
    for _ in range(28):
        mid = 0.5 * (lo_r + hi_r)
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return 0.5 * (lo_r + hi_r)

def pair_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    nm1 = n - 1
    L = 2 * n - 1
    offset = np.arange(L) - nm1
    B_all = np.empty((n, L))
    for j in range(n):
        ip = j + offset
        vp = (ip >= 0) & (ip < n)
        im = j - offset
        vm = (im >= 0) & (im < n)
        row = np.zeros(L)
        row[vp] = 1.0 - h[ip[vp]]
        row[vm] -= h[im[vm]]
        B_all[j] = row
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        best_val = np.max(corr)
        best_d = 0.0
        best_k = -1
        ks = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
              (j + n // 3) % n, (j + 2 * n // 3) % n,
              (2 * j + 1) % n, (3 * j + 7) % n, (j * 2 + 3) % n]
        for _ in range(max(0, n_cand - 8)):
            ks.append(rng.randint(n))
        seen = set()
        for k in ks:
            if k == j or k in seen:
                continue
            seen.add(k)
            D = B_all[j] - B_all[k]
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-14:
                continue
            dopt = solve_1d(corr, D, dlo, dhi)
            vopt = np.max(corr + D * dopt)
            if vopt < best_val - 1e-12:
                best_val = vopt
                best_d = dopt
                best_k = k
        if best_k >= 0 and abs(best_d) > 1e-12:
            corr += best_d * (B_all[j] - B_all[best_k])
            h[j] = np.clip(h[j] + best_d, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_d, 0.0, 1.0)
            for jj in (j, best_k):
                ip = jj + offset
                vp = (ip >= 0) & (ip < n)
                im = jj - offset
                vm = (im >= 0) & (im < n)
                row = np.zeros(L)
                row[vp] = 1.0 - h[ip[vp]]
                row[vm] -= h[im[vm]]
                B_all[jj] = row
            improved = True
    corr = np.correlate(h, 1.0 - h, mode="full")
    return h, corr, improved

def fourier_perturb(h, rng, amp):
    n = len(h)
    n_modes = max(3, n // 14)
    pert = np.zeros(n)
    idx = np.arange(n)
    for m in rng.choice(n, size=n_modes, replace=False):
        if m == 0:
            continue
        ph = rng.uniform(0.0, 2.0 * np.pi)
        c = rng.uniform(-1.0, 1.0) * amp
        pert += c * np.cos(2.0 * np.pi * m * idx / n + ph)
    return h + pert

def upstride(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_sum(h_new, n_new / 2.0)

def make_checkerboard(n):
    blk = max(1, n // 8)
    h = np.zeros(n)
    for i in range(0, n, 2 * blk):
        h[i:min(i + blk, n)] = 1.0
    return h

        if stall >= 300:
