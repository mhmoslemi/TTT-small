    exp_vals = np.exp(alpha * (corr - m_val))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Coarse grid
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12):
    n = len(h)
    improved = False
    order = np.arange(n)
    rng.shuffle(order)
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Candidate k values: neighbors, structured, random
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n,
                 (j + n // 4) % n, (j + 3 * n // 4) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def subgradient_step(h, alpha, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid

            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def subgradient_iteration(h, target_sum, alpha, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, grad = smooth_max_grad(corr, h, alpha)
    h_new = h - step_size * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def symmetrize(h, n):
    h_flip = h[n - 1 - np.arange(n)].copy()
    h_sym = 0.5 * (h + h_flip)
    return h_sym

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m_val))
        return m_val + np.log(np.sum(exp_vals)) / alpha
    def g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, grad = smooth_max_grad(corr, x, alpha)
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir():
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Subgradient + gradient-guided pair descent (50% budget)
    t_end1 = t_start + 0.50 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    target = n_w / 2.0
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    alpha = 40.0
    step = 0.025
    iter_cnt = 0

    while time.time() < t_end1:
        if iter_cnt % 15 == 0:
            h_work = subgradient_iteration(h_work, target, alpha, step)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
        else:
            h_work, corr_w, _ = gradient_guided_sweep(h_work, corr_w, rng, alpha)

        iter_cnt += 1
        if iter_cnt % 25 == 0:
            c5_w = compute_c5(h_work)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_work.copy()
                best_n = n_w

        if iter_cnt % 150 == 0:
            step *= 0.75
            alpha *= 1.4
            if step < 0.002:
                step = 0.002

    # Phase 2: Intensive pair descent (30% budget)
    t_end2 = t_start + 0.80 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    target = n_w / 2.0
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0

    while time.time() < t_end2:
        h_work, corr_w, imp = pair_descent_sweep(h_work, corr_w, rng, n_cand=14)
        if not imp:
            h_work, corr_w, imp = gradient_guided_sweep(h_work, corr_w, rng, alpha=80.0)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-12:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall > 200:
            if stall > 600:
                noise = 0.04
                h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), target)
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
                stall = 0
            elif stall % 100 == 0:
                h_work = project_to_sum(h_work + rng.uniform(-0.08, 0.08, n_w), target)
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    # Phase 3: Symmetrization + refinement (10% budget)
    t_end3 = t_start + 0.90 * budget_s
    n_w = best_n
    target = n_w / 2.0
    h_sym = symmetrize(best_h, n_w)
    h_sym = project_to_sum(h_sym, target)
    corr_sym = np.correlate(h_sym, 1.0 - h_sym, mode="full")
    for _ in range(40):
        if time.time() >= t_end3:
            break
        h_sym, corr_sym, _ = pair_descent_sweep(h_sym, corr_sym, rng, n_cand=12)
    c5_sym = compute_c5(h_sym)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym.copy()
        best_n = n_w

    # Phase 4: Multi-resolution (7% budget)
    t_end4 = t_start + 0.97 * budget_s
    for n_tr in [200, 280, 320, 360, 400]:
        if time.time() >= t_end4:
            break
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_tr = np.interp(x_new, x_old, best_h, period=2.0)
        h_tr = project_to_sum(h_tr, ts_tr)
        corr_tr = np.correlate(h_tr, 1.0 - h_tr, mode="full")
        max_sw = min(25, max(3, int(max(0, t_end4 - time.time()) / 4.0)))
        for _ in range(max_sw):
            if time.time() >= t_end4:
                break
            h_tr, corr_tr, _ = pair_descent_sweep(h_tr, corr_tr, rng, n_cand=10)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr

    # Phase 5: SLSQP + final polish (3% budget)
    t_end5 = t_start + budget_s - 5
    n_f = best_n
    ts_f = n_f / 2.0
    if time.time() < t_end5 - 60:
        h_pol = slsqp_local(best_h, n_f, ts_f, 100, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_end5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=16)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
