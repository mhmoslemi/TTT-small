        return x[-1]

        h = x[:-1]
        t = x[-1]
        cc = compute_correlations(h)
        return t - cc[top_k]

        return np.sum(x[:-1]) - n / 2.0

        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

        return h_out, compute_c5_from_h(h_out)

from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.5)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.5)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def grad_correlation_shift(shift, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        ve = n - shift
        if ve <= 0:
            return grad
        grad[:ve] = 1.0 - h[shift:shift + ve]
        grad[shift:] -= h[:ve]
    else:
        s = -shift
        grad[s:] = 1.0 - h[:n - s]
        grad[:n - s] -= h[s:]
    return grad * dx


def subgrad_momentum(h, n, max_iter, active_k, lr0, beta, time_limit):

    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    c = compute_correlations(h)
    active = np.argsort(c)[-active_k:]
    v = np.zeros(n)
    lr = lr0
    nogan = 0

        if time.time() - t_start > time_limit:

        grad = np.zeros(n)
        for k in active:
            shift = k - (n - 1)
            grad += grad_correlation_shift(shift, h, n)
        grad /= active_k

        return x[-1]

    def ineq(x):
        hh = x[:-1]
        t = x[-1]
        cc = compute_correlations(hh)
        return t - cc[top_k]

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    x0 = np.concatenate([h_init, [c_max + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(c_max * 0.3, c_max + t_margin * 2)]
    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq, 'jac': eq_jac}]

    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})

    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def slsqp_all(h_init, n, maxiter, t_margin):
    c = compute_correlations(h_init)
    c_max = c.max()

    def obj(x):
        return x[-1]

    def ineq(x):
        hh = x[:-1]
        t = x[-1]
        cc = compute_correlations(hh)
        return t - cc

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    x0 = np.concatenate([h_init, [c_max + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(c_max * 0.3, c_max + t_margin * 2)]
    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq, 'jac': eq_jac}]

    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})

    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def logsumexp_optimize(h_init, n, T, penalty, maxiter):
    target = n / 2.0

    def smooth_obj(x):
        c = compute_correlations(x)
        c_max = c.max()
        shifted = T * (c - c_max)
        shifted = np.clip(shifted, -400, 400)
        ls = c_max + np.log(np.sum(np.exp(shifted))) / T
        sv = np.sum(x) - target
        return ls + penalty * sv * sv

    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h_init, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})

    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)

    # Phase 1: SLSQP with all constraints (larger budget)
    if time.time() - t0 < budget_s - 500:
        best_h, best_c5 = slsqp_all(best_h, n, maxiter=4000, t_margin=0.12)

    # Phase 2: Subgradient with momentum
    remaining = budget_s - (time.time() - t0)
    if remaining > 200:
        h_sg, c5_sg = subgrad_momentum(best_h, n, max_iter=60000, active_k=100,
                                       lr0=0.025, beta=0.94, time_limit=remaining * 0.25)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 3: SLSQP top-k refinement
    if time.time() - t0 < budget_s - 350:
        best_h, best_c5 = slsqp_solve(best_h, n, k_top=180, maxiter=4000, t_margin=0.08)

    # Phase 4: Subgradient finer
    remaining = budget_s - (time.time() - t0)
    if remaining > 150:
        h_sg, c5_sg = subgrad_momentum(best_h, n, max_iter=50000, active_k=60,
                                       lr0=0.012, beta=0.96, time_limit=remaining * 0.3)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 5: Log-sum-exp smooth optimization
    if time.time() - t0 < budget_s - 200:
        h_ls, c5_ls = logsumexp_optimize(best_h, n, T=80.0, penalty=200.0, maxiter=2000)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 6: SLSQP polish
    if time.time() - t0 < budget_s - 120:
        best_h, best_c5 = slsqp_solve(best_h, n, k_top=120, maxiter=4000, t_margin=0.05)

    # Phase 7: Perturbation + SLSQP
    if time.time() - t0 < budget_s - 60:
        for attempt in range(8):
            if time.time() - t0 > budget_s - 25:
                break
            scale = 0.008 + 0.012 * (1.0 - attempt / 8.0) ** 1.5
            h_pert = best_h + scale * rng.standard_normal(n)
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = slsqp_solve(h_pert, n, k_top=100, maxiter=2500, t_margin=0.04)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new

    # Phase 8: Final subgradient (very fine)
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h_sg, c5_sg = subgrad_momentum(best_h, n, max_iter=40000, active_k=40,
                                       lr0=0.005, beta=0.97, time_limit=remaining * 0.4)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 9: Try higher resolution
    if time.time() - t0 < budget_s - 80:
        for n_try in [350, 450]:
            if time.time() - t0 > budget_s - 40:
                break
            n_cur = len(best_h)
            x_old = np.linspace(0, 2, n_cur, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
            h_interp = np.clip(f_interp(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_try)
            if time.time() - t0 < budget_s - 20:
                h_new, c5_new = slsqp_solve(h_interp, n_try, k_top=120, maxiter=2500, t_margin=0.1)
                if c5_new < best_c5 - 5e-5:
                    best_h, best_c5 = h_new, c5_new

    # Phase 10: Final SLSQP
    if time.time() - t0 < budget_s - 10:
        n_final = len(best_h)
        best_h, best_c5 = slsqp_solve(best_h, n_final, k_top=100, maxiter=3000, t_margin=0.02)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
