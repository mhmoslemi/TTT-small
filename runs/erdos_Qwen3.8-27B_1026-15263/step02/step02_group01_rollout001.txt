    h = np.clip(h, 0.0, 1.0)
    # Bisect on lambda
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Tiny adjustment
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(mask):
            proj[mask] += diff / np.sum(mask)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(mask):
            proj[mask] += diff / np.sum(mask)
    return np.clip(proj, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    m_max = np.max(corr)
    shifted = alpha * (corr - m_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m_max + np.log(s / n_corr) / alpha
    g = np.zeros(n)
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-13:
            continue
        offset = idx - (n - 1)
        lo1 = max(0, offset)
        hi1 = min(n, n + offset)
        if lo1 < hi1:
            g[lo1:hi1] += wi * (1.0 - h[lo1 - offset:hi1 - offset])
        lo2 = max(0, -offset)
        hi2 = min(n, n - offset)
        if lo2 < hi2:
            g[lo2:hi2] -= wi * h[lo2 + offset:hi2 + offset]
    return sm_val, g

def nesterov_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_end):
    h = h0.copy()
    y = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    t_param = 1.0
    lr = lr_init
    for it in range(n_iters):
        sm, g = smooth_max_grad(y, alpha)
        step = y - lr * g
        h_new = project_sum(step, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        momentum = (t_param - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_param = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.85
        if time.time() > t_end:
            break
    return h_best, c5_best

def coord_desc(h0, n, target_sum, max_rounds, rng, t_end):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        idx_order = rng.permutation(n)
        ss_idx = min(rnd // 5, len(step_sizes) - 1)
        for ss_i in range(ss_idx, len(step_sizes)):
            ss = step_sizes[ss_i]
            for i in idx_order:
                if time.time() > t_end:
                    break
                for sign in [1, -1]:
                    d = sign * ss
                    h_new = h.copy()
                     old_val = h_new[i]
                    h_new[i] = old_val + d
                    if h_new[i] < 0.0 or h_new[i] > 1.0:
                        h_new[i] = np.clip(h_new[i], 0.0, 1.0)
                    else:
                        pass
                    h_new = project_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        h_best = h.copy()
                        improved = True
                        break
        if not improved and rnd > 10:
            break
    return h_best, c5_cur

def focused_descent(h0, n, target_sum, n_iters, t_end, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        offset = k_star - (n - 1)
        # Find which h indices contribute most to this correlation
        if offset >= 0:
            js = np.arange(0, n - offset)
            contrib = h[js] * (1.0 - h[js + offset])
        else:
            js = np.arange(0, n + offset)
            contrib = h[js - offset] * (1.0 - h[js])
        # Perturb the top contributors
        n_perturb = min(n // 10, max(5, n - abs(offset)))
        top_idx = np.argsort(-contrib)[:n_perturb]
        for j in top_idx:
            if time.time() > t_end:
                break
            if offset >= 0:
                i = j
            else:
                i = j - offset
            delta = 0.01 * rng.choice([-1, 1])
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new = project_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                h_best = h.copy()
    return h_best, c5_cur

def slsqp_refine(h0, n, target_sum, alpha, maxiter, t_end):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        remaining = max(10, int((t_end - time.time()) * 0.8))
        mi = min(maxiter, remaining)
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': mi, 'ftol': 1e-15, 'iprint': 0})
        return project_sum(res.x, target_sum)
    except Exception:
        return project_sum(h0, target_sum)
