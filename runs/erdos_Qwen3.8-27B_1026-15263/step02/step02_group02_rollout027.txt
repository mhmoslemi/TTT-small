        ones = np.where(h_bin == 1)[0]

            current += 1
            if current == target:
                break

    n = len(x)

    c = np.correlate(x, 1.0 - x, mode="full") * dx

    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_viol = np.sum(x) - n / 2.0
    return lse + penalty * sum_viol * sum_viol

    return x[-1]

        {'type': 'eq', 'fun': slsqp_sum_eq, 'jac': slsqp_sum_eq_jac}

def binary_local_search_v2(h_bin, rng, time_budget=60):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    corrs = np.correlate(h, 1.0 - h, mode="full")
    current_max = float(corrs.max())
    best_max = current_max
    best_h = h.copy()
    dx = 2.0 / n
    t_start = time.time()
    
    # Track the "hot" lags
    threshold = 0.95 * current_max
    
    while time.time() - t_start < time_budget:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        hot_mask = corrs >= threshold
        n_hot = np.sum(hot_mask)
        
        improved = False
        for _ in range(200):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            
            h[i] = 0.0
            h[j] = 1.0
            new_corrs = np.correlate(h, 1.0 - h, mode="full")
            new_max = float(new_corrs.max())
            
            if new_max < current_max - 1e-14:
                corrs = new_corrs
                current_max = new_max
                threshold = 0.95 * current_max
                improved = True
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            # Simulated annealing: accept with some probability
            if rng.random() < 0.05:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_corrs = np.correlate(h, 1.0 - h, mode="full")
                current_max = float(new_corrs.max())
                corrs = new_corrs
    
    return best_h, best_max * dx

The terms involving position i:

1. p = i: changes from 1*(1-h[m-i]) to 0*(1-h[m-i]) = 0. Δ = -(1-h[m-i])
2. p = m-i: changes from h[m-i]*0 to h[m-i]*1. Δ = +h[m-i]

After changing h[j] from 0 to 1:
1. p = j: changes from 0*(1-h[m-j]) to 1*(1-h[m-j]). Δ = +(1-h[m-j])
2. p = m-j: changes from h[m-j]*1 to h[m-j]*0. Δ = -h[m-j]

Note: if i = m-j (i.e., m = i+j), then terms 1 and 2 interact. Similarly if j = m-i.

For simplicity, let's assume i ≠ j and i ≠ m-j and j ≠ m-i (which is true for most m).

Total Δc[m] = -(1-h[m-i]) + h[m-i] + (1-h[m-j]) - h[m-j]

            = -1 + 2h[m-i] + 1 - 2h[m-j]
            = 2(h[m-i] - h[m-j])

So Δc[m] = 2*(h[m-i] - h[m-j]) (for m where both m-i and m-j are in [0,n)).

This is a very clean formula! For binary h, h[m-i] - h[m-j] ∈ {-1, 0, 1}, so Δc[m] ∈ {-2, 0, 2}.

With this formula, I can compute the change in c[m] for all m in O(n) time (just shift h by i and j and subtract).

This means each "try" in the local search is O(n) instead of O(n²), which is much faster!
