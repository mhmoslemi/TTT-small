            hh[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(hh, target_sum))
    for _ in range(5):
        a = rng.uniform(0.1, 0.45)
        b = rng.uniform(-0.3, 0.3)
        f1 = int(rng.choice([1, 2, 3, 4, 5]))
        f2 = int(rng.choice([2, 3, 4, 5, 6, 7, 8, 10]))
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h4 = 0.5 + a * np.sin(2.0 * np.pi * f1 * x / 2.0 + p1) + b * np.sin(2.0 * np.pi * f2 * x / 2.0 + p2)
        starts.append(project_to_sum(h4, target_sum))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_to_sum(np.interp(xn, xo, best_h, period=2.0), target_sum))
    hb = (rng.rand(n) > 0.5).astype(float)
    att = 0
    while abs(hb.sum() - target_sum) > 1e-10 and att < 10000:
        if hb.sum() > target_sum:
            idx = int(rng.choice(np.where(hb > 0)[0]))
            hb[idx] = 0.0
        else:
            idx = int(rng.choice(np.where(hb < 1)[0]))
            hb[idx] = 1.0
        att += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def tleft():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [3, 8, 20, 50, 150, 500, 1500, 5000]:
        if tleft() < 30:
            break
        hn = slsqp_optimize(best_h, n1, ts1, alpha, maxiter=800)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()

    resolutions = [200, 240, 280, 320, 360, 400, 480, 560]
    for n_tr in resolutions:
        if tleft() < 90:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        starts = make_starts(n_tr, ts_tr, rng, best_h, best_n)
        for hs in starts:
            if tleft() < 25:
                break
            hs = project_to_sum(hs, ts_tr)
            cs = compute_c5(hs)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()
                best_n = n_tr
            for alpha in [10, 50, 300, 1000]:
                if tleft() < 18:
                    break
                ho = slsqp_optimize(hs, n_tr, ts_tr, alpha, maxiter=500)
                co = compute_c5(ho)
                if co < best_c5:
                    best_c5 = co
                    best_h = ho.copy()
                    best_n = n_tr
                hs = ho

    n_b = best_n
    ts_b = n_b / 2.0
    h_bin_init = to_binary(best_h, ts_b)
    h_bf, c5_bf = binary_hill_climb(h_bin_init, n_b, ts_b, rng, max_rounds=150, n_trials=400)
    if c5_bf < best_c5:
        best_c5 = c5_bf
        best_h = h_bf.copy()

    if tleft() > 60:
        h_sa, c5_sa = binary_sa(h_bf, n_b, ts_b, rng, n_steps=4000, t0=1.0, t1=0.001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        c5_sa2 = compute_c5(h_sa)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa.copy()

    if tleft() > 50:
        h_heal = slsqp_optimize(best_h, best_n, ts_b, 200, maxiter=800)
        ch = compute_c5(h_heal)
        if ch < best_c5:
            best_c5 = ch
            best_h = h_heal.copy()

    for trial in range(25):
        if tleft() < 35:
            break
        noise = 0.12 * (1.0 - trial / 25.0) + 0.03
        hsk = best_h + rng.uniform(-noise, noise, best_n)
        hsk = project_to_sum(hsk, ts_b)
        for alpha in [30, 150, 800]:
            if tleft() < 20:
                break
            ho = slsqp_optimize(hsk, best_n, ts_b, alpha, maxiter=500)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5 = co
                best_h = ho.copy()
            hsk = ho

    if tleft() > 60:
        h_bin2 = to_binary(best_h, ts_b)
        h_bf2, c5_bf2 = binary_hill_climb(h_bin2, best_n, ts_b, rng, max_rounds=100, n_trials=300)
        if c5_bf2 < best_c5:
            best_c5 = c5_bf2
            best_h = h_bf2.copy()
        if tleft() > 40:
            h_sa2, c5_sa3 = binary_sa(h_bf2, best_n, ts_b, rng, n_steps=3000, t0=0.8, t1=0.001)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa2.copy()

    if tleft() > 40:
        for alpha in [2000, 10000, 50000]:
            if tleft() < 25:
                break
            hp = slsqp_optimize(best_h, best_n, ts_b, alpha, maxiter=1500)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
