def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * shifted - alpha * max_shift)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + max_shift + np.log(np.sum(exp_vals)) / alpha
    # Actually, let me redo this properly
    obj = m_val + np.log(np.sum(np.exp(alpha * (corr - m_val)))) / alpha
    
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)

The key insight is: we want the correlation to be as "flat" as possible (no large peaks). This is like designing a sequence with good autocorrelation properties (like a Barker sequence or m-sequence).

For binary sequences, the optimal autocorrelation properties are achieved by sequences like:
- m-sequences (maximal length LFSR sequences): have three-valued autocorrelation
- Barker sequences: have minimum sidelobes
- Welch sequences: have good cross-correlation

But our problem is slightly different because we're minimizing the maximum of one-sided correlation (h with 1-h, not h with h).

Let me think... ∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - ∫h(x)h(x+k)dx.

So minimizing max_k ∫h(x)(1-h(x+k))dx is equivalent to minimizing max_k (1 - ∫h(x)h(x+k)dx), which is equivalent to maximizing min_k ∫h(x)h(x+k)dx.

So we want to maximize the minimum autocorrelation! This is a different (but related) formulation.

For binary h with sum = n/2, ∫h(x)h(x+k)dx is the (normalized) autocorrelation. We want the minimum autocorrelation to be as large as possible.

This is the "maximum minimum autocorrelation" problem, which is related to sequence design!

For a binary sequence of length n with n/2 ones, the autocorrelation at lag k is:
R(k) = (1/n) * sum_{i=0}^{n-1} h[i] * h[(i+k) mod n]

We want to maximize min_k R(k).

This is well-studied in coding theory. The best known bounds are related to the Welch bound and Plotkin bound.
