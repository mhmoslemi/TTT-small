def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def top_k_smooth_max_grad(h, alpha, top_k):
    """Compute smooth max value and gradient using only top-k shifts."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    if top_k is None or top_k >= len(corr):
        top_k = len(corr)
    
    # Get top-k indices
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_corr = corr[top_idx]
    
    # Softmax weights
    shifted = alpha * (top_corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / top_k) / alpha
    
    # Compute gradient: g = sum_k w[k] * grad(corr[k])/dh
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-15:
            continue
        s_shift = idx - (n - 1)
        if s_shift >= 0:
            # g[0:n-s] += wi * (1 - h[s:])
            # g[0:n-s] -= wi * h[s:]  -- wait no
            # For s >= 0:
            #   g[0:n-s] += wi * 1
            #   g[0:n-s] -= wi * h[s:s+n-s]  = wi * h[s:]
            #   g[s:] -= wi * h[0:n-s]
            ns = n - s_shift
            g[:ns] += wi
            g[:ns] -= wi * h[s_shift:]
            g[s_shift:] -= wi * h[:ns]

            ns = n - sn
            g[sn:] += wi
            g[sn:] -= wi * h[:ns]
            g[:ns] -= wi * h[sn:]
    
    return sm_val, g

        v, _ = top_k_smooth_max_grad(h, alpha, top_k)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = top_k_smooth_max_grad(h, alpha, top_k)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return project_to_sum(h0, target_sum)
