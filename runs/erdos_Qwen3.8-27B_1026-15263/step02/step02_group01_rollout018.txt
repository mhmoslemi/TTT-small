    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k=10):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top K lags
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = np.sort(top_idx)
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for m in top_idx:
            # d(corr[m])/d(h[j]) for each j
            jj1 = np.arange(n) + m - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            
            jj2 = np.arange(n) - m + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        
        # Step
        lr = lr_init / (1.0 + 0.05 * it)
        h_new = h - lr * g / top_k
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_target_sum = 0.0
_g_n_pts = 0
_g_alpha = 1.0

def _c_fun(x):
    return np.sum(x) - _g_target_sum

def _c_jac(x):
    return np.ones(_g_n_pts)

def _obj(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(h, _g_alpha)
    return v

def _grad(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(h, _g_alpha)
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kmax1 = min(j, n - 1)
        if kmax1 >= 0:
            ks = np.arange(kmax1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kmax2 = n - j
        if kmax2 > 0:
            ks2 = np.arange(kmax2)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _g_target_sum, _g_n_pts, _g_alpha
    _g_target_sum = target_sum
    _g_n_pts = n
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def adam_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for _ in range(n_iters):
        t += 1
        _, g = smooth_max_and_grad(h, alpha)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * g * g
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.97
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, n_iters, lr_init, top_k):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    j_arr = np.arange(n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            jj1 = j_arr + m - n + 1
            mask1 = (jj1 >= 0) & (jj1 < n)
            g[mask1] += (1.0 - h[jj1[mask1]])
            jj2 = j_arr - m + n - 1
            mask2 = (jj2 >= 0) & (jj2 < n)
            g[mask2] -= h[jj2[mask2]]
        lr = lr_init / (1.0 + 0.03 * it)
        h_new = h - lr * g / float(top_k)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h0, n, target_sum, max_rounds, rng, base_delta):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-3.0, -2.0, -1.0, 1.0, 2.0, 3.0]) * base_delta
    for _ in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            hi = h[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-15 or nv > 1.0 + 1e-15:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(nv, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def _tl(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts0 = best_n / 2.0
            best_h = project_to_sum(best_h, ts0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_cand

    n_targets = [360, 480, 600]

    for n in n_targets:
        if _tl(t_start, budget_s) < 100:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue

        starts = []
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(project_to_sum(h_interp, ts))

        starts.append(np.full(n, 0.5))
        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_to_sum(hr, ts))

        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [2, 3, 4, 5, 7]:
            for phase in [0.0, np.pi / 4.0, np.pi / 2.0]:
                for amp in [0.3, 0.4, 0.5]:
                    hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                    starts.append(project_to_sum(hs, ts))

        for h_start in starts:
            if _tl(t_start, budget_s) < 50:
                break
            h_start = project_to_sum(h_start, ts)
            try_update(h_start, n)

            h_cur = h_start.copy()
            alphas = [5.0, 15.0, 40.0, 100.0, 250.0, 500.0]
            for alpha in alphas:
                if _tl(t_start, budget_s) < 30:
                    break
                maxiter = max(40, min(600, int(_tl(t_start, budget_s) * 1.2)))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                try_update(h_opt, n)
                h_cur = h_opt

            if _tl(t_start, budget_s) > 50:
                n_adam = min(3000, max(500, int(_tl(t_start, budget_s) * 2.0)))
                h_a, c5_a = adam_optimize(h_cur, n, ts, 100.0, n_adam, 0.08)
                try_update(h_a, n)
                h_cur = h_a

            if _tl(t_start, budget_s) > 40:
                n_sub = min(5000, max(1000, int(_tl(t_start, budget_s) * 2.0)))
                h_sg, c5_sg = topk_subgradient(h_cur, n, ts, n_sub, 0.05, 10)
                try_update(h_sg, n)
                h_cur = h_sg

            if _tl(t_start, budget_s) > 30:
                n_cd = max(20, min(100, int(_tl(t_start, budget_s) / 2.0)))
                h_cd, _ = coord_descent(h_cur, n, ts, n_cd, rng, 0.005)
                try_update(h_cd, n)
                h_cur = h_cd

            if _tl(t_start, budget_s) > 20:
                h_cd2, _ = coord_descent(h_cur, n, ts, 50, rng, 0.001)
                try_update(h_cd2, n)
                h_cur = h_cd2

    if best_h is not None and _tl(t_start, budget_s) > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [500.0, 1000.0, 3000.0]:
            if _tl(t_start, budget_s) < 15:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 800)
            try_update(h_pol, n_f)
            best_h = h_pol

        if _tl(t_start, budget_s) > 25:
            n_adam_f = min(4000, max(500, int(_tl(t_start, budget_s) * 2.0)))
            h_af, _ = adam_optimize(best_h, n_f, ts_f, 500.0, n_adam_f, 0.03)
            try_update(h_af, n_f)
            best_h = h_af

        if _tl(t_start, budget_s) > 15:
            n_sub_f = min(8000, max(1000, int(_tl(t_start, budget_s) * 2.0)))
            h_sg, _ = topk_subgradient(best_h, n_f, ts_f, n_sub_f, 0.03, 20)
            try_update(h_sg, n_f)
            best_h = h_sg

        if _tl(t_start, budget_s) > 15:

    n_final = len(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, best_c5, n_final)
