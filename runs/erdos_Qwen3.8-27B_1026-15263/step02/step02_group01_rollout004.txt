                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    n_list = [360, 480, 600]

    for n in n_list:
        if time_left() < 100:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0

        starts = []
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        else:
            n_src = len(best_h)
            x_old = np.linspace(0, 2, n_src, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, np.append(best_h, best_h[0]))
            starts.append(project_to_sum(h_interp, ts))

        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1, 2, 3, 4, 5]:
            for phase in [0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi]:
                for amp in [0.2, 0.35, 0.5]:
                    hs = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
                    starts.append(project_to_sum(hs, ts))

        h_blk = np.zeros(n)
        period = max(4, n // 40)
        for s in range(0, n, period * 2):
            h_blk[s:s + period] = 1.0
        starts.append(project_to_sum(h_blk, ts))

        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_to_sum(hr, ts))

        for h_start in starts:
            if time_left() < 60:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()
            alphas = [5.0, 20.0, 80.0, 200.0]
            for alpha in alphas:
                if time_left() < 40:
                    break
                maxiter = max(50, min(500, int(time_left() * 2.0)))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt

            if time_left() > 80:
                h_opt2 = lbfgsb_optimize(h_cur, n, ts, 150.0, 200, 10.0)
                c5_opt2 = compute_c5(h_opt2)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n
                h_cur = h_opt2

            if time_left() > 50:
                n_pg = min(2000, max(200, int(time_left() * 2)))
                h_pg, c5_pg = nesterov_gradient(h_cur, n, ts, 100.0, n_pg, 0.05)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                h_cur = h_pg

            if time_left() > 40:
                h_pc, c5_pc = pairwise_descent(h_cur, n, ts, 30, rng)
                if c5_pc < best_c5:
                    best_c5 = c5_pc
                    best_h = h_pc.copy()
                    best_n = n

            if time_left() > 30:
                h_cd, c5_cd = coordinate_descent(h_cur, n, ts, 20, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    if best_h is not None and time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [400.0, 1000.0, 3000.0]:
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 800)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 30:
            h_lbs = lbfgsb_optimize(best_h, n_f, ts_f, 500.0, 400, 50.0)
            c5_lbs = compute_c5(h_lbs)
            if c5_lbs < best_c5:
                best_c5 = c5_lbs
                best_h = h_lbs.copy()

        if time_left() > 25:
            n_pg = min(4000, max(500, int(time_left() * 2)))
            h_pg, c5_pg = nesterov_gradient(best_h, n_f, ts_f, 200.0, n_pg, 0.03)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n_f

        if time_left() > 20:
            h_pw, c5_pw = pairwise_descent(best_h, n_f, ts_f, 50, rng)
            if c5_pw < best_c5:
                best_c5 = c5_pw
                best_h = h_pw.copy()

        if time_left() > 15:
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 80, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 10:
            h_tp, c5_tp = top_peak_reduction(best_h, n_f, ts_f, 20, rng)
            if c5_tp < best_c5:
                best_c5 = c5_tp
                best_h = h_tp.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
