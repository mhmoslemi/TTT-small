import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        kmin = max(-(n-1), -j)
        kmax = min(n-1, n-1-j)
        if kmax >= kmin:
            ks = np.arange(kmin, kmax+1)
            g[j] += np.dot(w[n-1+ks], 1.0 - h[j-ks])
        kmin2 = max(-(n-1), -j)
        kmax2 = min(n-1, n-1-j)
        if kmax2 >= kmin2:
            ks2 = np.arange(kmin2, kmax2+1)
            g[j] -= np.dot(w[n-1+ks2], h[j+ks2])
    return sm_val, g

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_value_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_value_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_snap(h, target_sum):
    """Round h to binary (0 or 1) with exact sum."""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    # Fix the sum
    current_sum = np.sum(h_bin)
    target = int(target_sum)
    if current_sum > target:
        # Need to flip some 1s to 0s
        excess = int(current_sum - target)
        ones = np.where(h_bin == 1.0)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones])[:excess]
        h_bin[ones[order]] = 0.0
    elif current_sum < target:
        deficit = int(target - current_sum)
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        h_bin[zeros[order]] = 1.0
    return h_bin

def binary_local_search(h, target_sum, max_trials, rng):
    """Stochastic local search on binary h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = np.max(corr)
    best_c5 = current_max * dx
    h_best = h.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kmin = max(-(n-1), -j)
        kmax = min(n-1, n-1-j)
        if kmax >= kmin:
            ks = np.arange(kmin, kmax+1)
            g[j] += np.dot(w[n-1+ks], 1.0 - h[j-ks])
        if kmax >= kmin:
            ks2 = np.arange(kmin, kmax+1)
            g[j] -= np.dot(w[n-1+ks2], h[j+ks2])
    return sm_val, g

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_value_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_value_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    best_c5 = current_max * dx
    h_best = h.copy()
    ks = np.arange(-(n-1), n)
    
    for trial in range(max_trials):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

            delta[va] += -(1.0 - h[idx])
        # h[i] appears as h[t] when t = i, valid if 0<=i+k<=n-1
        vb = (i + ks >= 0) & (i + ks < n)
        if np.any(vb):

            delta[vb] += h[idx]
        
        # Compute delta for j: 0->1
        # h[j] appears as h[t+k] when t = j-k, valid if 0<=j-k<=n-1
        vc = (j - ks >= 0) & (j - ks < n)
        if np.any(vc):
            idx = j - ks[vc]
            delta[vc] += (1.0 - h[idx])
        # h[j] appears as h[t] when t = j, valid if 0<=j+k<=n-1
