```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    # Gradient using FFT-based correlate
    first = np.correlate(w, 1.0 - h, mode='valid')[::-1]
    second = np.correlate(w, h[::-1], mode='valid')
    grad = first - second
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=600, ftol=1e-12):
    def obj_func(h):
        return smooth_max_and_grad(h, alpha)[0]
    def grad_func(h):
        return smooth_max_and_grad(h, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    first = np.correlate(w, 1.0 - h, mode='valid')[::-1]
    second = np.correlate(w, h[::-1], mode='valid')
    grad = first - second
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def to_binary(h, target_sum, rng):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    diff = int(round(target_sum - h.sum()))
    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for i in range(min(diff, len(zeros))):

    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for i in range(min(-diff, len(ones))):
            h[ones[i]] = 0.0
    return h

        improved = False
        ones = list(np.where(h == 1)[0])

def simulated_annealing_binary(h, n, target_sum, rng, t0_temp=1.0, t_min=0.01, n_iter=50000, time_limit=60):

            ones.append(j)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

            # Reject
            h[i] = 1.0
            h[j] = 0.0
```

```python
def generate_m_sequence(m, rng=None):
    """Generate a maximal-length LFSR sequence of length 2^m - 1."""
    # Use a primitive polynomial
    # For m=8: x^8 + x^4 + x^3 + x^2 + 1 (0x1D in Galois representation)
    # Actually, let me use a known primitive polynomial
    # For m=7: 0x65 (x^7 + x^3 + 1)
    # For m=8: 0xC7 (x^8 + x^6 + x^5 + x^4 + 1) or 0x1D
    
    # Let me use a simple LFSR
    # State: m bits
    # Tap positions (0-indexed from LSB):
    taps = {
        4: [4, 3],
        5: [5, 3],
        6: [6, 5],
        7: [7, 1],
        8: [8, 6, 5, 4],
    }
    
    # Initialize state to all 1s
    state = (1 << m) - 1
    seq = []
    length = (1 << m) - 1
    
    # Tap feedback: XOR of tapped bits
    tap_positions = taps[m]
    
    for _ in range(length):
        # Output the LSB
        seq.append(state & 1)
        # Compute feedback
        feedback = 0
        for t in tap_positions:
            feedback ^= (state >> t) & 1
        # Shift right and insert feedback at MSB
        state = (state >> 1) | (feedback << (m - 1))
    
    return np.array(seq, dtype=float)
```
