    if lr_schedule is None:
        lr_schedule = None
    
    best_obj = compute_c5(h)
    best_h = h.copy()
    
    for it in range(maxiter):
        alpha = alpha_schedule[min(it, len(alpha_schedule)-1)]
        obj, grad = smooth_max_and_grad(h, alpha)
        
        # Adaptive step size
        if lr_schedule is not None:
            lr = lr_schedule[min(it, len(lr_schedule)-1)]
        else:
            # Backtracking line search
            lr = 0.01
            for _ in range(20):
                h_new = project_to_sum(h - lr * grad, target_sum)
                obj_new, _ = smooth_max_and_grad(h_new, alpha)
                if obj_new < obj - 1e-4 * lr * np.sum(grad**2):
                    break
                lr *= 0.5
        
        h_new = project_to_sum(h - lr * grad, target_sum)
        
        # Check actual C5 improvement
        c5_actual = compute_c5(h_new)
        if c5_actual < best_obj:
            best_obj = c5_actual
            best_h = h_new.copy()
        
        h = h_new
        
        # Convergence check
        if it > 10 and abs(obj - smooth_max_and_grad(h, alpha)[0]) < 1e-10:
            break
    
    return best_h, best_obj

For the LP-based approach:
- At each iteration, linearize h[i]*h[i+k] around h₀:
  h[i]*h[i+k] ≈ h₀[i]*h[i+k] + h[i]*h₀[i+k] - h₀[i]*h₀[i+k]
- The constraint becomes:
  sum_i [h[i] - h[i]*h[i+k]] * dx ≤ t
  ≈ sum_i [h[i] - h₀[i]*h[i+k] - h[i]*h₀[i+k] + h₀[i]*h₀[i+k]] * dx ≤ t
  = sum_i [(1-h₀[i+k])*h[i] - h₀[i]*h[i+k]] * dx + sum_i h₀[i]*h₀[i+k]*dx ≤ t

Wait, let me be more careful. The correlation at lag k (shift = k-(n-1)):

Linearizing h[i]*h[i+shift] around h₀:
h[i]*h[i+shift] ≈ h₀[i]*h₀[i+shift] + h₀[i]*(h[i+shift]-h₀[i+shift]) + h₀[i+shift]*(h[i]-h₀[i])
= h₀[i]*h₀[i+shift] + h₀[i]*h[i+shift] - h₀[i]*h₀[i+shift] + h₀[i+shift]*h[i] - h₀[i]*h₀[i+shift]

Hmm, that's not right. The first-order Taylor expansion of f(x,y) = x*y at (x₀,y₀) is:
f(x₀,y₀) + (x-x₀)*y₀ + (y-y₀)*x₀
= x₀*y₀ + (x-x₀)*y₀ + (y-y₀)*x₀
= x₀*y₀ + x*y₀ - x₀² + y*x₀ - y₀*x₀
= x*y₀ + y*x₀ - x₀*y₀

So: h[i]*h[i+shift] ≈ h[i]*h₀[i+shift] + h[i+shift]*h₀[i] - h₀[i]*h₀[i+shift]

Then: corr[m] ≈ sum_i h[i] - sum_i [h[i]*h₀[i+shift] + h[i+shift]*h₀[i] - h₀[i]*h₀[i+shift]]
= sum_i h[i] - sum_i h[i]*h₀[i+shift] - sum_i h[i+shift]*h₀[i] + sum_i h₀[i]*h₀[i+shift]

For the LP, we need this to be ≤ t/dx for all m.
The terms involving h are:
sum_i h[i]*(1-h₀[i+shift]) - sum_i h[i+shift]*h₀[i]

Let me reindex the second sum: let j = i+shift, i = j-shift:
sum_j h[j]*h₀[j-shift]

So: corr[m]*dx ≈ sum_j h[j]*[(1-h₀[j+shift]) - h₀[j-shift]] * dx + sum_i h₀[i]*h₀[i+shift]*dx

The constant term is sum_i h₀[i]*h₀[i+shift]*dx.
The linear part is sum_j h[j]*[(1-h₀[j+shift]) - h₀[j-shift]]*dx.

So the constraint is:
sum_j h[j]*[(1-h₀[j+shift]) - h₀[j-shift]]*dx ≤ t/dx - sum_i h₀[i]*h₀[i+shift]*dx

This is a linear constraint in h! We can solve:
minimize t
subject to: for all m: sum_j h[j]*a_m[j]*dx + b_m ≤ t
            sum(h) = n/2
            0 ≤ h ≤ 1

where a_m[j] = (1-h₀[j+shift_m]) - h₀[j-shift_m] and b_m = sum_i h₀[i]*h₀[i+shift_m]*dx

This is a linear program! We can solve it with CVXPY or scipy's linprog.

This gives us a sequence of LPs to solve, each giving a better approximation to the true optimum. This is essentially a sequential linear programming (SLP) approach.

The advantage of this approach is that LPs can be solved very efficiently, and the solution is guaranteed to be on the constraint boundary.

Let me combine both approaches in the final algorithm:
1. Phase 1: SLP (sequential LP) approach for a few iterations
2. Phase 2: Projected gradient descent with smooth max for refinement
3. Phase 3: Random restarts + best of both
4. Phase 4: Final polish

import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    """Compute smooth max of correlation and its gradient using FFT."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / exp_vals.sum()
    obj = m_val + np.log(exp_vals.sum()) / alpha
    
    # Gradient: grad = 1 - C1 - C2
    # C1[j] = sum_m w[m]*h[j+n-1-m] = convolve(h, w, 'full')[j+n-1]
    # C2[j] = sum_m w[m]*h[j+m-n+1] = convolve(h, w[::-1], 'full')[j+n-1]
    conv1 = np.convolve(h, w, 'full')
    conv2 = np.convolve(h, w[::-1], 'full')
    C1 = conv1[(n-1):(2*n-1)]
    C2 = conv2[(n-1):(2*n-1)]
    grad = 1.0 - C1 - C2
    return obj, grad

def slp_solve(h0, n, target_sum, max_iter=50, solver='cvxpy', time_limit=30):
    """Sequential Linear Programming approach."""
    h = h0.copy()
    t_start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        dx = 2.0 / n
        corr = compute_corr(h)
        n_corr = 2 * n - 1
        
        # Build LP: minimize t
        # subject to: for each m: sum_j h[j]*a[m,j]*dx ≤ t - b[m]
        #   sum(h) = target_sum
        #   0 ≤ h ≤ 1
        #   t ≥ 0
        
        # a[m, j] = (1 - h[j+shift]) - h[j-shift] where shift = m-(n-1)
        # b[m] = sum_i h[i]*h[i+shift] * dx
        
        # Using CVXPY
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum, t_var >= 0]
        
        for m in range(n_corr):
            shift = m - (n - 1)
            # a[j] = (1-h[j+shift]) - h[j-shift]
            # h[j+shift] valid when 0<=j+shift<n
            # h[j-shift] valid when 0<=j-shift<n
            a = np.zeros(n)
            # (1-h[j+shift]) contributes 1 to all j, and -h[j+shift] where valid
            a[:] = 1.0
            if shift >= 0:
                a[:n-shift] -= h[shift:]  # -h[j+shift] for j < n-shift
                a[shift:] -= h[:n-shift]  # -h[j-shift] for j >= shift
            else:
                a[:n+shift] -= h[-shift:]  # -h[j-shift]=h[j+|shift|] for j < n+shift
                a[-shift:] -= h[:n+shift]  # -h[j+shift]=h[j-|shift|] for j >= |shift|
            
            # b[m] = sum_i h[i]*h[i+shift]*dx
            if shift >= 0:
                b = np.sum(h[:n-shift] * h[shift:]) * dx
            else:
                b = np.sum(h[-shift:] * h[:n+shift]) * dx
            
            constraints.append(cp.sum(a * h_var) * dx <= t_var - b)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000, verbose=False)
            if h_var.value is not None:
                h_new = np.clip(h_var.value, 0, 1)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            break
    
    return best_h, best_c5

def fista(h0, n, target_sum, alpha, maxiter=3000, lr=None):

    h = h0.copy()
    h_prev = h0.copy()
    t_k = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    if lr is None:
        # Estimate Lipschitz constant
        # The gradient of the smooth max has Lipschitz constant ~ O(alpha * n)
        # Use a conservative estimate
        lr = 1.0 / (alpha * n)
    
    for it in range(maxiter):
        obj, grad = smooth_max_grad(h, alpha)
        
        # FISTA update
        y = h_prev + ((t_k - 1) / (t_k + 1)) * (h_prev - h)  # Wait, this isn't right
        # Actually: y_k = x_k + (t_{k-1} - 1)/(t_k) * (x_k - x_{k-1})
        # where t_k = (1 + sqrt(1 + 4*t_{k-1}^2))/2
        
        # Let me use the standard FISTA:
        # y = h + ((t_prev - 1) / t_curr) * (h - h_prev_old)
        # h_new = project(h - lr * grad, sum=target)
        # t_curr = (1 + sqrt(1 + 4*t_prev^2)) / 2
        # h_prev_old = h_prev
        # h_prev = h
        # h = h_new
        
        # Actually let me just use standard gradient descent with momentum (which is similar)
        # Or better, use a simple backtracking line search gradient method.
        
        # Simple projected gradient with backtracking:
        step = lr
        for _ in range(30):
            h_candidate = project_to_sum(h - step * grad, target_sum)
            obj_candidate, _ = smooth_max_grad(h_candidate, alpha)
            if obj_candidate <= obj - 0.5 * step * np.sum(grad**2):
                break
            step *= 0.5
        
        h_prev = h.copy()
        h = h_candidate
        
        c5_actual = compute_c5(h)
        if c5_actual < best_c5:
            best_c5 = c5_actual
            best_h = h.copy()
    
    return best_h, best_c5

def gradient_descent(h0, n, target_sum, alpha, maxiter=5000, lr=None, momentum=0.9):
    """Projected gradient descent with Nesterov acceleration."""
    h = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    if lr is None:
        lr = 2.0 / (alpha * n)  # Conservative step size
    
    for it in range(maxiter):
        # Nesterov: evaluate gradient at z = h + momentum_factor * (h - h_prev)
        obj, grad = smooth_max_grad(z, alpha)
        
        # Update
        h_new = project_to_sum(z - lr * grad, target_sum)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        momentum_factor = (t_k - 1.0) / t_new
        z = h_new + momentum_factor * (h_new - h)
        
        h = h_new
        t_k = t_new
        
        c5_actual = compute_c5(h)
        if c5_actual < best_c5:
            best_c5 = c5_actual
            best_h = h.copy()
    
    return best_h, best_c5
