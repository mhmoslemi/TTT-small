            best_val = val
            corr = compute_corr(best)
            top_idx = np.argsort(corr)[::-1][:n_active]
            top_vals = corr[top_idx]
            grad = np.zeros(n)
            for m in top_idx:
                shift = m - (n - 1)
                weight = 1.0 / (1.0 + np.exp(-(corr[m] - top_vals[-1]) * 100.0))
                grad += weight * gradient_for_lag(h, shift)
            grad *= dx
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.8
            if lr < 1e-6:
                break
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
    else:
        n = 77
        best_h = project_feasible(np.full(n, 0.5), n / 2.0)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h)
    target_sum = n / 2.0
    dx = 2.0 / n

    phase1_end = start_time + budget_s * 0.20
    phase2_end = start_time + budget_s * 0.45
    phase3_end = start_time + budget_s * 0.70
    phase4_end = start_time + budget_s * 0.92
    phase5_end = start_time + budget_s * 0.99

    if best_c5 > 0.38085 and time.time() < phase1_end:
        for topk in [5, 3, 2]:
            if time.time() > phase1_end:
                break
            remaining = phase1_end - time.time()
            est_per_iter = n * topk / 5e5
            n_iter = max(500, int(remaining / max(0.0001, est_per_iter)))
            n_iter = min(n_iter, 8000)
            h_new, c5_new = projected_subgradient(best_h, dx, topk, 0.008, n_iter, 0.85)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    if best_c5 > 0.38085 and time.time() < phase2_end:
        h_new, c5_new = projected_subgradient(best_h, dx, 1, 0.004, 5000, 0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() < phase3_end:
        h_new, c5_new = fine_coord_descent(best_h, n_sweeps=8)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() < phase4_end:
        h_new, c5_new = active_lag_balance(best_h, n_active=3)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new, c5_new = fine_coord_descent(best_h, n_sweeps=5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() < phase5_end:
        h_new, c5_new = slsqp_polish(best_h, n_iter=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new, c5_new = fine_coord_descent(best_h, n_sweeps=5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time.time() < start_time + budget_s * 0.995:
        perturb = project_feasible(best_h + rng.randn(n) * 0.003, target_sum)
        h_new, c5_new = projected_subgradient(perturb, dx, 2, 0.003, 2000, 0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new, c5_new = fine_coord_descent(best_h, n_sweeps=3)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
