Actually, I realize I should also think about numerical stability. When α is large, the softmax weights w become very peaked (close to one-hot at the maximum). Let me also think about whether I can use a different surrogate. Instead of soft-max, I could use:
- Power sum: (sum corr_k^p)^(1/p) for large p (approximates max)
- This has a well-known gradient: (1/p) * (sum corr_k^p)^(1/p-1) * p * corr_k^(p-1) = (sum corr_k^p)^(1/p-1) * corr_k^(p-1)

The power sum might be easier to optimize because it's smooth everywhere (for p >= 1) and its gradient doesn't involve exponentials.

But for very large p, the power sum is also close to a one-hot, so it has the same issues as soft-max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for k in range(n):
        wk = w[n-1+k]
        if wk > 1e-15:
            g[k:] += wk * (1.0 - h[:n-k])
            g[:n-k] -= wk * h[k:]
        if k >= 1:
            wk2 = w[n-1-k]
            if wk2 > 1e-15:
                g[:n-k] += wk2 * (1.0 - h[k:])
                g[k:] -= wk2 * h[:n-k]
    return sm_val, g

def objective_and_grad(x, alpha, target_sum, penalty):
    h = np.clip(x, 0.0, 1.0)
    val, g = smooth_max_and_grad(h, alpha)
    # Add penalty for sum constraint
    sum_diff = np.sum(h) - target_sum
    val += 0.5 * penalty * sum_diff**2
    g += penalty * sum_diff * np.ones(len(h))
    return val, g

def lbfgsb_optimize(h0, n, target_sum, alpha, penalty, maxiter, time_limit):
    t0 = time.time()
    def obj(x):
        v, _ = objective_and_grad(x, alpha, target_sum, penalty)
        return v
    def grad(x):
        _, g = objective_and_grad(x, alpha, target_sum, penalty)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter*5,
                                           'ftol': 1e-14, 'gtol': 1e-8,

def coordinate_descent(h0, n, target_sum, max_rounds, rng):

    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False

        for i in idx_order:
            # Try moving h[i] in both directions
            best = c5_cur
            best_v = h[i]

                    step = frac * 0.01 * direction
                    h_new = h.copy()
                    h_new[i] = np.clip(h[i] + step, 0.0, 1.0)

                        best_v = h_new[i]

                h = project_to_sum(h, target_sum)
                c5_cur = best
                improved = True
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha

    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-15:
            g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] -= wk * h[k:]
        if k >= 1:
            wk2 = w[n - 1 - k]
            if wk2 > 1e-15:
                g[:n - k] += wk2 * (1.0 - h[k:])
                g[k:] -= wk2 * h[:n - k]
    return sm_val, g


def obj_grad(x, alpha, target_sum, penalty):
    h = np.clip(x, 0.0, 1.0)
    val, g = smooth_max_and_grad(h, alpha)
    sd = np.sum(h) - target_sum
    val += 0.5 * penalty * sd * sd
    g += penalty * sd * np.ones(len(h))
    return val, g


def lbfgsb_opt(h0, n, target_sum, alpha, penalty, maxiter, t_start, budget_s):
    def _remaining():
        return budget_s - (time.time() - t_start)

    if _remaining() < 30:
        return h0.copy()

    def f(x):
        v, _ = obj_grad(x, alpha, target_sum, penalty)
        return v

    def gf(x):
        _, g = obj_grad(x, alpha, target_sum, penalty)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), jac=gf, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'maxfun': maxiter * 10,
                                'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 40,
                                'iprint': -1})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
