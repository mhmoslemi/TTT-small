def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        frac = h - np.floor(h)
        mask = (frac > 1e-12) & (frac < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):

def smooth_max_val_grad(h, alpha):
    n = len(h)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        frac = h - np.floor(h)
        mask = (frac > 1e-10) & (frac < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    j = np.arange(n)
    mags = np.arange(-(n - 1), n)
    jm = j[:, None] + mags[None, :]
    jM = j[:, None] - mags[None, :]
    vm1 = (jm >= 0) & (jm < n)
    vm2 = (jM >= 0) & (jM < n)
    T1 = np.zeros((n, 2 * n - 1))
    T2 = np.zeros((n, 2 * n - 1))
    idx1 = np.where(vm1)
    idx2 = np.where(vm2)
    T1[idx1] = 1.0 - h[jm[idx1]]
    T2[idx2] = h[jM[idx2]]
    grad = (T1 - T2).dot(w)
    return sm_val, grad

def nesterov_optimize(h0, n, ts, alpha, n_iters, lr_init, t_start, deadline):
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, grad_y = smooth_max_val_grad(y, alpha)
        h_cand = np.clip(y - lr * grad_y, 0.0, 1.0)
        h_cand = project_to_sum(h_cand, ts)
        c5_cand = compute_c5(h_cand)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        if c5_cand < best_c5 - 1e-14:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            y = np.clip(h_cand + beta * (h_cand - h), 0.0, 1.0)
            h = h_cand
            t_k = t_new
            lr = min(lr * 1.005, 10.0 / alpha)
            stall = 0
        else:
            lr *= 0.85
            if lr < 1e-7:
                lr = 1e-7
            y = np.clip(h_cand + beta * (h_cand - h), 0.0, 1.0)
            h = h_cand
            t_k = t_new
            stall += 1
            if stall > 500:
                break
    return best_h, best_c5

def exchange_local_search(h0, n, ts, n_rounds, rng, t_start, deadline):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(3 * n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] == h[j]:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]
            h_new = project_to_sum(h_new, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                c5_cur = c5_new
                improved = True
        if not improved:
            break
    return h, c5_cur

def coord_descent_fast(h0, n, ts, n_rounds, rng, t_start, deadline):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.06, -0.06, 0.01, -0.01]
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in rng.permutation(n):
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def generate_starts(n, ts, rng, best_h, best_n):
    starts = []
    if best_n == n and best_h is not None:
        starts.append(best_h.copy())
    else:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        starts.append(project_to_sum(hi, ts))
    starts.append(np.full(n, 0.5))
    hr = rng.uniform(0.0, 1.0, n)
    starts.append(project_to_sum(hr, ts))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7]:
        for phase in [0.0, 0.5 * np.pi, np.pi / 3.0]:
            hs = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
            starts.append(project_to_sum(hs, ts))
    h_blk = np.zeros(n)
    period = max(2, n // 40)
    for s in range(0, n, period * 2):
        h_blk[s:min(s + period, n)] = 1.0
    starts.append(project_to_sum(h_blk, ts))
    h_bin = (rng.rand(n) < 0.5).astype(float)
    h_bin = project_to_sum(h_bin, ts)
    starts.append(h_bin)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    n_work = best_n
    ts = n_work / 2.0

    starts = generate_starts(n_work, ts, rng, best_h, best_n)

    for start_idx, h_start in enumerate(starts):
        if time.time() > deadline:
            break
        if time.time() > deadline - 80:
            break
        h_cur = h_start.copy()
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

        alpha_list = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
        for alpha in alpha_list:
            if time.time() > deadline:

            remaining = deadline - time.time()
            if remaining < 30:

            n_iters = max(50, int(remaining * 0.2))

            if c5_cur < best_c5:

        if time.time() < deadline - 20:
