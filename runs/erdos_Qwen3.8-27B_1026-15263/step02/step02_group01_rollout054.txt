Let me think about proximal gradient more carefully:

The objective is f(h) = smooth_max(c(h); α) which is smooth (for α > 0).
The constraint is h ∈ C = {h : 0 ≤ h ≤ 1, sum(h) = n/2}.

Proximal gradient: h_{k+1} = prox_C(η_k * L)(h_k - η_k * ∇f(h_k))

where prox_C is the Euclidean projection onto C, and L is a Lipschitz constant of ∇f.

Estimating L: L = ||∇²f||_∞ (spectral norm of Hessian). For the smooth max of bilinear functions, this is bounded by α * max_k (something) * n... it's hard to estimate a priori.

I'll use backtracking line search:
- Start with η = η_0
- If f(h_new) > f(h_old) + ∇f(h_old)·(h_new - h_old) + (1/(2η))||h_new - h_old||², halve η
- Otherwise, accept the step

But the projection makes this complicated. Let me just use a simpler approach:

**Accelerated proximal gradient (FISTA)**:
y_k = h_k + (t_k - 1)/(t_{k+1}) * (h_k - h_{k-1})
h_{k+1} = P_C(y_k - η * ∇f(y_k))
t_{k+1} = (1 + sqrt(1 + 4*t_k²))/2

With η chosen by backtracking or fixed.

Let me also consider: since the constraint set C = {0 ≤ h ≤ 1} ∩ {sum(h) = n/2} is the intersection of a box and a hyperplane, the projection is:
1. Project onto the box: clip to [0,1]
2. Project onto the hyperplane: adjust to satisfy sum = n/2 (within the box)

This is what `project_to_sum` does (iteratively, but it converges fast).

For efficiency, I can use a closed-form projection:
- Binary search on a threshold τ such that sum(clip(h - τ, 0, 1)) = n/2

This is much faster than the iterative method.

Let me implement this:

```python
def project_box_sum(v, target_sum):
    """Project v onto {h: 0 <= h <= 1, sum(h) = target_sum} using binary search on Lagrange multiplier."""
    n = len(v)
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)
```

This is O(n * log(precision)) which is much faster than the iterative method.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, target_sum):
    n = len(v)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    c = corr * dx
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    ev = np.exp(shifted)
    s = np.sum(ev)
    w = ev / s
    sm_val = m + np.log(s) / (alpha * (2 * n - 1))
    
    grad = np.zeros(n)
    for j in range(n):
        # T1[j]
        qlen = n - j
        if qlen > 0:
            T1 = np.dot(w[n-1-j : 2*n-1-2*j], omh[:qlen])
        else:
            T1 = 0.0
        # T2[j]
        T2 = np.dot(w[n-1+j : j-1 : -1], h)
        grad[j] = dx * (T1 - T2)
    
    return sm_val, grad

def proximal_gradient(h0, n, target_sum, alpha, n_iters, eta_init, t_start, time_budget):
    h = h0.copy()
    h_prev = h.copy()
    t_k = 1.0
    eta = eta_init
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iters):
        if time.time() - t_start > time_budget:
            break
        _, grad = smooth_max_val_grad(h, alpha)
        h_new = project_box_sum(h - eta * grad, target_sum)
        
        # Check improvement
        f_new, _ = smooth_max_val_grad(h_new, alpha)
        f_old, _ = smooth_max_val_grad(h, alpha)
        
        if f_new <= f_old + 1e-10:
            # Accept
            h_prev = h.copy()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            y = h_new + momentum * (h_new - h_prev)
            y = np.clip(y, 0.0, 1.0)
            h_prev = h
            h = project_box_sum(y, target_sum)
            t_k = t_new
            eta = min(eta * 1.05, 5.0)
        else:
            eta *= 0.7
            continue
        
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
    
    return h_best, c5_best

def slsqp_run(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        v, _ = smooth_max_val_grad(x, alpha)
        return v
    def jac(x):
        _, g = smooth_max_val_grad(x, alpha)
        return g
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_box_sum(res.x, target_sum)
    except:
        return project_box_sum(h0, target_sum)

def coordinate_descent(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    sizes = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]
    
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            best_c5 = c5_cur
            best_h = None
            hv = h[i]
            for s in sizes:
                for sign in [1, -1]:
                    h_test = h.copy()
                    h_test[i] = hv + sign * s
                    h_test = project_box_sum(h_test, target_sum)
                    c5_test = compute_c5(h_test)
                    if c5_test < best_c5 - 1e-15:
                        best_c5 = c5_test
                        best_h = h_test.copy()
            if best_h is not None:
                h = best_h
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
