    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = project_feasible(res.x, n)
        return h_opt, evaluate_c5(h_opt)
    except Exception:
        return h, evaluate_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_c5 = float('inf')

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)

    if best_h is None:
        n = 77
        best_h = np.full(n, 0.5)
        best_h = project_feasible(best_h, n)
        best_c5 = evaluate_c5(best_h)

    phase1_deadline = start_time + budget_s * 0.18
    phase2_deadline = start_time + budget_s * 0.68
    phase3_deadline = start_time + budget_s * 0.92
    final_deadline = start_time + budget_s * 0.97

    n_candidates = []
    if parent_h is not None:
        n_candidates.append(len(parent_h))
    for nn in (73, 75, 76, 77, 78, 79, 80, 81, 83, 85):
        if nn not in n_candidates:
            n_candidates.append(nn)

    for n in n_candidates:
        if time.time() > phase1_deadline:
            break
        if parent_h is not None and len(parent_h) == n:
            h0 = parent_h.copy()
        elif parent_h is not None:
            xn = np.linspace(0, 2, n, endpoint=False)
            xo = np.linspace(0, 2, len(parent_h), endpoint=False)
            h0 = np.interp(xn, xo, parent_h)
        else:
            h0 = np.full(n, 0.5)
        h0 = project_feasible(h0, n)
        for attempt in range(3):
            if time.time() > phase1_deadline:
                break
            if attempt == 0:
                h_try = h0.copy()
            else:
                h_try = project_feasible(h0 + rng.randn(n) * 0.04, n)
            h_opt, c5 = subgradient_descent(h_try, n, 400, 0.006, 0.999, 3, rng, phase1_deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    n_best = len(best_h)

    sg_configs = [
        (400, 0.005, 0.9995, 1),
        (400, 0.004, 0.9995, 2),
        (400, 0.003, 0.9995, 3),
        (300, 0.005, 0.999, 5),
        (300, 0.002, 0.999, 2),
    ]

    while time.time() < phase2_deadline:
        for num_it, lr, dec, tk in sg_configs:
            if time.time() > phase2_deadline:
                break
            h_opt, c5 = subgradient_descent(best_h, n_best, num_it, lr, dec, tk, rng, phase2_deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        h_coord, c5_coord = coordinate_descent(best_h, n_best, 3, (0.04, 0.02, 0.01, 0.005), phase2_deadline)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

        h_pair, c5_pair = pair_transfer_search(best_h, n_best, 200, rng, phase2_deadline)
        if c5_pair < best_c5:
            best_c5 = c5_pair
            best_h = h_pair.copy()

        h_rd, c5_rd = random_direction_search(best_h, n_best, 40, rng, phase2_deadline)
        if c5_rd < best_c5:
            best_c5 = c5_rd
            best_h = h_rd.copy()

    while time.time() < phase3_deadline:
        h_opt, c5 = subgradient_descent(best_h, n_best, 600, 0.0015, 0.9999, 1, rng, phase3_deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

        h_coord, c5_coord = coordinate_descent(best_h, n_best, 2, (0.02, 0.01, 0.005, 0.002), phase3_deadline)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

        h_pair, c5_pair = pair_transfer_search(best_h, n_best, 300, rng, phase3_deadline)
        if c5_pair < best_c5:
            best_c5 = c5_pair
            best_h = h_pair.copy()

    if time.time() < final_deadline:
        h_opt, c5 = slsqp_polish(best_h, n_best, 0.001, 300, final_deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    h_coord, c5_coord = coordinate_descent(best_h, n_best, 3, (0.01, 0.005, 0.002, 0.001, 0.0005), start_time + budget_s * 0.99)
    if c5_coord < best_c5:
        best_c5 = c5_coord
        best_h = h_coord.copy()

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    if time.time() > start_time + budget_s:
        pass

    return best_h, best_c5, len(best_h)
