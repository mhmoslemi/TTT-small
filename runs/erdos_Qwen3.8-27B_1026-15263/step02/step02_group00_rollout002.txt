import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    n_corr = 2 * n - 1
    delta = np.arange(n_corr) - (n - 1)
    p = np.arange(n)[:, None]
    first = p + delta[None, :]
    second = p - delta[None, :]
    valid_f = (first >= 0) & (first < n)
    valid_s = (second >= 0) & (second < n)
    fc = np.clip(first, 0, n - 1)
    sc = np.clip(second, 0, n - 1)
    grad = np.sum(valid_f * (1.0 - h[fc]) * w[None, :] - valid_s * h[sc] * w[None, :], axis=1)
    return grad

def lbfgs_step(h0, n, target_sum, alpha, maxiter=1500, pen_lam=50.0):
    def f(x):
        return smooth_max_val(x, alpha) + pen_lam * (np.sum(x) - target_sum) ** 2
    def g(x):
        return smooth_max_grad(x, alpha) + 2.0 * pen_lam * (np.sum(x) - target_sum)
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 20})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slsqp_step(h0, n, target_sum, alpha, maxiter=500):
    def f(x):
        return smooth_max_val(x, alpha)
    def g(x):
        return smooth_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_swaps(h, n, target_sum, rng, max_rounds=2000):
    h = h.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_swap = None
        n_trials = min(300, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_test = compute_c5(h)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_swap = (i, j)
            else:
                h[i] = 1.0
                h[j] = 0.0
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = best_c5
    return h, c5_cur

def fourier_perturb(h, n, rng, max_mode=15, amp_max=0.3):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h_new = h.copy()
    n_modes = rng.randint(2, max_mode + 1)
    for _ in range(n_modes):
        freq = rng.randint(1, max_mode + 1)
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = rng.uniform(0.02, amp_max)
        h_new += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_to_sum(h_new, n / 2.0)

def block_permute(h, n, rng):
    h_new = h.copy()
    n_blocks = rng.randint(3, 9)
    bsz = n // n_blocks
    blocks = [h_new[i*bsz:min((i+1)*bsz, n)] for i in range(n_blocks) if i*bsz < n]
    rng.shuffle(blocks)
    idx = 0
    for blk in blocks:
        h_new[idx:idx+len(blk)] = blk
        idx += len(blk)
    return project_to_sum(h_new, n / 2.0)

def resample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_to_sum(np.interp(x_new, x_old, h, period=2.0), n_new / 2.0)

def run(seed=42, budget_s=1300, **kwargs):

    def tleft():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_solution(h, n):
        nonlocal best_h, best_c5, best_n
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, n / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        return h, c5

    def continuation_optimize(h, n, start_alpha=5, end_alpha=3000, n_alphas=8):
        h_cur = h.copy()
        alphas = np.logspace(np.log10(start_alpha), np.log10(end_alpha), n_alphas)
        for alpha in alphas:
            if tleft() < 20:
                break
            maxit = 2000 if alpha > 500 else 1000
            h_cur = lbfgs_step(h_cur, n, n / 2.0, float(alpha), maxiter=maxit)
            try_solution(h_cur, n)
            if tleft() < 30:
                break
        return h_cur

    # Phase 1: Binary optimization
    if tleft() > 100:
        h_bin = (best_h > 0.5).astype(float)
        ts = best_n / 2.0
        attempts = 0
        while abs(h_bin.sum() - ts) > 1e-10 and attempts < 10000:
            if h_bin.sum() > ts:
                h_bin[rng.choice(np.where(h_bin > 0)[0])] = 0.0
            else:
                h_bin[rng.choice(np.where(h_bin < 1)[0])] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts) < 1e-10:
            h_bin, c5_bin = binary_swaps(h_bin, best_n, ts, rng, max_rounds=1500)
            try_solution(h_bin, best_n)

    # Phase 2: Continuation from current best
    if tleft() > 150:
        h_opt = continuation_optimize(best_h, best_n, start_alpha=5, end_alpha=5000, n_alphas=10)

    # Phase 3: Multi-resolution search
    for n_tr in [120, 160, 200, 280, 320]:
        if tleft() < 80:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        h_init = resample(best_h, best_n, n_tr)
        h_opt = continuation_optimize(h_init, n_tr, start_alpha=5, end_alpha=2000, n_alphas=7)
        try_solution(h_opt, n_tr)

    # Phase 4: Fourier perturbation restarts at best resolution
    for trial in range(12):
        if tleft() < 40:
            break
        h_pert = fourier_perturb(best_h, best_n, rng, max_mode=12, amp_max=0.25)
        h_opt = continuation_optimize(h_pert, best_n, start_alpha=10, end_alpha=2000, n_alphas=6)

    # Phase 5: Block perturbations
    for trial in range(6):
        if tleft() < 40:
            break
        h_blk = block_permute(best_h, best_n, rng)
        h_opt = continuation_optimize(h_blk, best_n, start_alpha=10, end_alpha=1500, n_alphas=5)

    # Phase 6: SLSQP polish at high alpha

        for alpha in [500, 2000, 8000]:
            if tleft() < 15:
                break
            h_pol = slsqp_step(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
            try_solution(h_pol, best_n)

    # Phase 7: Final binary refinement on best solution
    if tleft() > 40:
        h_bin = (best_h > 0.5).astype(float)
        ts = best_n / 2.0
        attempts = 0
        while abs(h_bin.sum() - ts) > 1e-10 and attempts < 10000:
            if h_bin.sum() > ts:
                h_bin[rng.choice(np.where(h_bin > 0)[0])] = 0.0
            else:
                h_bin[rng.choice(np.where(h_bin < 1)[0])] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts) < 1e-10:

    # Final validation

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
