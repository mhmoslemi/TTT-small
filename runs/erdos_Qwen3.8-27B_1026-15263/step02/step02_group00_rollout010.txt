import numpy as np
from scipy.optimize import minimize
import time

_alpha_global = 100.0
_target_sum_global = 0.0
_n_global = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    m_idx = np.arange(2 * n - 1)[None, :]
    j_idx = np.arange(n)[:, None]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term1 = valid_s * (1.0 - h[sc]) * w[None, :]
    term2 = valid_f * h[fc] * w[None, :]
    grad = np.sum(term1, axis=1) - np.sum(term2, axis=1)
    return obj, grad

def _obj_fun(x):
    return smooth_max_obj_and_grad(x, _alpha_global)[0]

def _jac_fun(x):
    return smooth_max_obj_and_grad(x, _alpha_global)[1]

def _cons_fun(x):
    return np.sum(x) - _target_sum_global

def _cons_jac(x):
    return np.ones(_n_global)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    global _alpha_global, _target_sum_global, _n_global
    _alpha_global = alpha
    _target_sum_global = target_sum
    _n_global = n
    cons = {'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_fun, h0.copy(), method='SLSQP', jac=_jac_fun,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_step(h, n, target_sum, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    i = np.arange(n)
    s1 = i + k_max - n + 1
    s2 = i - k_max + n - 1
    valid_s1 = (s1 >= 0) & (s1 < n)
    valid_s2 = (s2 >= 0) & (s2 < n)
    grad = np.zeros(n)
    grad[valid_s1] += (1.0 - h[s1[valid_s1]])
    if np.any(valid_s2):
        grad[valid_s2] -= h[s2[valid_s2]]
    grad *= (2.0 / n)
    h_new = h - lr * grad
    return project_to_sum(h_new, target_sum)

def momentum_subgradient(h, n, target_sum, steps=200, base_lr=0.02, momentum=0.9, time_limit=60):
    h = h.copy()
    v = np.zeros(n)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(steps):
        if time.time() - t0 > time_limit:
            break
        lr = base_lr / (1.0 + 0.01 * step)
        h_new = subgradient_step(h, n, target_sum, lr)
        h_new = project_to_sum(h_new, target_sum)
        v = momentum * v + (1 - momentum) * (h_new - h)
        h = project_to_sum(h - v, target_sum)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def binary_2swap_search(h, n, target_sum, max_rounds=30, time_limit=40):
    h = h.copy()
    t0 = time.time()
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corr_cur = np.correlate(h, 1.0 - h, mode="full")

        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new_max = max_val
        best_i = -1
        best_j = -1
        n_ones = len(ones)
        n_zeros = len(zeros)

                    new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))

    return int(val) % 1000000

def make_fourier_h(n, M, coeffs, target_sum):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        if 2*m-2 < len(coeffs):
            h += coeffs[2*m-2] * np.cos(np.pi * m * x)
        if 2*m-1 < len(coeffs):
            h += coeffs[2*m-1] * np.sin(np.pi * m * x)
    return project_to_sum(h, target_sum)
