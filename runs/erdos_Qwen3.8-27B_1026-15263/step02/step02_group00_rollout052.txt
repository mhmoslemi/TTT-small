    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    """One step of projected subgradient with momentum."""
    n = len(h)
    corr = compute_corr(h) * (2.0 / n)
    m_star = np.argmax(corr)
    # Compute gradient of c(m_star)
    m = m_star - (n - 1)  # actual lag
    g = np.zeros(n)

    idx_direct = np.arange(max(0, -m), min(n, n - m))
    g[idx_direct] += (1.0 - h[idx_direct + m])
    # Indirect term: -h[j-m] for j where j-m is valid
    idx_indirect = np.arange(max(0, m), min(n, n + m))

        v = 0.9 * v + lr * g
        h_new = h - v
    else:

    h_new = project_to_sum(h_new, target_sum)
    return h_new, (lr * g)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.1):
    n = len(h)
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = np.argmax(corr)
        m = m_star - (n - 1)
        
        # Compute gradient
        g = np.zeros(n)
        lo_d = max(0, -m)
        hi_d = min(n, n - m)
        g[lo_d:hi_d] += (1.0 - h[lo_d + m:hi_d + m])
        lo_i = max(0, m)
        hi_i = min(n, n + m)
        g[lo_i:hi_i] -= h[lo_i - m:hi_i - m]
        
        # Find best pair: increase where g is most negative, decrease where g is most positive
        # But we need to respect bounds [0, 1]
        # Find the index with most negative g that can be increased (h < 1)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_clipped_inc = np.where(can_inc, g, np.inf)
        g_clipped_dec = np.where(can_dec, g, -np.inf)
        
        i_inc = np.argmin(g_clipped_inc)  # most negative g, can increase
        i_dec = np.argmax(g_clipped_dec)  # most positive g, can decrease
        
        if g[i_inc] >= g[i_dec]:
            break  # no improvement possible
        
        # How much to move
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new >= c5:

            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            break
        c5 = c5_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def stochastic_coord_desc(h, target_sum, rng, max_iters=5000, step_frac=0.08):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        candidates_inc = np.where(can_inc)[0]
        candidates_dec = np.where(can_dec)[0]
        i_inc = int(rng.choice(candidates_inc))
        i_dec = int(rng.choice(candidates_dec))
        if g[i_inc] >= g[i_dec]:
            continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            continue
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    if best_h is None:

    n_parent = best_n

    if time_left() > 60:

        if c5_cd < best_c5:
            best_c5 = c5_cd

    resolutions = []
    for nr in [n_parent, 300, 360, 420, 480, 540]:

    for n_tr in resolutions:

        ts_tr = n_tr / 2.0

        if c5_interp < best_c5:
