    n = len(h)
    g = np.zeros(n)
    if k == 0:

    else:
        g[k:] += (1.0 - h[:n-k])

def smooth_max_obj_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    total = np.sum(exps)
    w = exps / total
    obj = m + np.log(total) / alpha
    
    grad = np.zeros(n)
    # Only compute for significant weights
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
    return obj, grad

def slsqp_opt(h0, target_sum, alpha, maxiter, deadline):
    n = len(h0)
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < f(h0) + 1e-10:
