import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        remaining = diff
        for idx in order:
            if abs(remaining) < 1e-15:
                break
            if remaining > 0:
                add = min(remaining, 1.0 - result[idx])
                result[idx] += add
                remaining -= add
            else:
                sub = min(-remaining, result[idx])
                result[idx] -= sub
                remaining += sub
    return result

def slsqp_obj(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def slsqp_constraint(x, n):
    return np.sum(x) - n / 2.0

def slsqp_optimize(h0, n, tau_list, max_iter=80):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'args': (n,)}]

            )

                current = project_sum(res.x, target)
        except Exception:
            pass
    return current

    target = n / 2.0

    for sweep in range(n_sweeps):
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for i in order:
            old_val = best[i]
            best_local_c5 = best_c5
            best_local_val = old_val
            best_local_h = None
            
            for d in scales:
                for sign in (1, -1):
                    new_val = old_val + sign * d
                    if new_val < 1e-10 or new_val > 1.0 - 1e-10:

                    h_try = best.copy()
                    h_try[i] = new_val
