    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # distribute diff among entries closest to 0.5
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result

def obj_and_grad(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    vals = corr / tau
    vmax = np.max(vals)
    shifted = np.exp(vals - vmax)
    s = np.sum(shifted)
    obj = tau * (vmax + np.log(s)) * dx
    weights = shifted / s
    # Gradient

        if w < 1e-15:

        if j3 < j4:

    for tau in tau_list:

                method='L-BFGS-B',

    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0

        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:

    for _ in range(3):
        starts.append(rng.rand(n))
    for freq in [2, 3, 4, 6]:
        h_tri = np.abs(np.mod(ar / n * freq, 2.0) - 1.0)
        starts.append(np.clip(0.1 + 0.8 * h_tri, 0, 1))
    for _ in range(2):
        h_bin = (rng.rand(n) > 0.5).astype(float)
        h_bin = np.convolve(h_bin, np.ones(3) / 3, mode='same')
        starts.append(np.clip(0.1 + 0.8 * h_bin, 0, 1))
    return starts

def pair_perturb_optimize(h, n, num_iters, time_limit, start_time, rng):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for _ in range(num_iters):
        if time.time() - start_time > time_limit:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        for delta in [0.003, -0.003, 0.005, -0.005, 0.01, -0.01]:
            h_new = best.copy()
            h_new[i] += delta
            h_new[j] -= delta
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                continue
            c5 = evaluate_c5(h_new)
            if c5 < best_c5 - 1e-14:
                best = h_new
                best_c5 = c5
                break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    def remaining():
        return max(0, budget_s - (time.time() - start_time))

    phase1_end = start_time + budget_s * 0.22
    phase2_end = start_time + budget_s * 0.55
    phase3_end = start_time + budget_s * 0.78

    n_candidates = []
    if parent_h is not None:
        n_candidates.append(len(parent_h))
    for nn in [41, 43, 45, 47, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69, 71, 73, 75, 77, 79, 81, 83, 85, 87, 89, 91, 93, 95, 97, 99, 101, 103, 105]:
        if nn not in n_candidates:
            n_candidates.append(nn)

    for n in n_candidates:
        if time.time() > phase1_end:
            break
        starts = generate_starts(n, parent_h, rng)
        max_starts = max(3, min(len(starts), int(remaining() / 8)))
        for h0 in starts[:max_starts]:
            if time.time() > phase1_end or remaining() < 3:
                break
            tau_sched = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
            h_opt, c5 = optimize_lbfgsb(h0, n, tau_sched, 80)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is None and parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)

    if best_h is not None:
        n = len(best_h)
        for n_try in [n, n - 2, n + 2, n - 4, n + 4, n + 8, n + 12, n + 20, n + 32]:
            if time.time() > phase2_end:
                break
            if n_try < 30 or n_try > 200:
                continue
            xp = np.linspace(0, 2, n, endpoint=False)
            xh = np.linspace(0, 2, n_try, endpoint=False)
            f = interp1d(xp, best_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
            h_init = np.clip(f(xh), 0, 1)
            h_init = project_feasible(h_init, n_try)
            tau_sched = [3.0, 1.5, 0.7, 0.3, 0.15, 0.07, 0.03]
            h_opt, c5 = optimize_lbfgsb(h_init, n_try, tau_sched, 120)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                n = n_try

        while time.time() < phase2_end and remaining() > 5:
            n = len(best_h)
            frac = (time.time() - start_time) / (phase2_end - start_time)
            noise = 0.003 + 0.015 * max(0.0, 1.0 - frac)
            perturb = best_h + rng.randn(n) * noise
            perturb = project_feasible(perturb, n)
            tau_sched = [1.0, 0.5, 0.2, 0.1, 0.05]
            h_opt, c5 = optimize_lbfgsb(perturb, n, tau_sched, 100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is not None and time.time() < phase3_end:
        n = len(best_h)
        time_limit_cd = phase3_end - time.time()
        h_cd, c5_cd = coordinate_descend(best_h, n, max_sweeps=15, time_limit=time_limit_cd, start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        if time.time() < phase3_end:
            h_pp, c5_pp = pair_perturb_optimize(best_h, n, num_iters=5000, time_limit=phase3_end - start_time, start_time=start_time, rng=rng)
            if c5_pp < best_c5:
                best_c5 = c5_pp
                best_h = h_pp

    if best_h is not None and time.time() < start_time + budget_s * 0.93:
        n = len(best_h)
        current = best_h.copy()
        end_t = start_time + budget_s * 0.93
        while time.time() < end_t:
            try:
                for tau in [0.05, 0.02, 0.01]:
                    res = minimize(
                        obj_and_grad, current,
                        method='L-BFGS-B',
                        jac=True,
                        args=(n, tau),
                        options={'maxiter': 150, 'ftol': 1e-16, 'gtol': 1e-12}
                    )
                    current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
                if time.time() > end_t:
                    break
                current = project_feasible(current + rng.randn(n) * 0.002, n)
            except Exception:
                break

    if best_h is not None and remaining() > 10:
        n = len(best_h)
        h_final, c5_final = coordinate_descend(best_h, n, max_sweeps=20, time_limit=remaining() - 3, start_time=start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        if remaining() > 10:
            h_pp2, c5_pp2 = pair_perturb_optimize(best_h, n, num_iters=3000, time_limit=remaining(), start_time=start_time, rng=rng)
            if c5_pp2 < best_c5:
                best_c5 = c5_pp2
                best_h = h_pp2

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 50
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
