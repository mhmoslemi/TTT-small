```python
def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def c5_logsumexp(h, T):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

    """Compute gradient of c_shift with respect to h.
    shift is the actual lag (can be negative).
    c_shift = dx * sum_{valid n} h[n+shift] * (1-h[n])
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:

        #          from -(h[n]) term: -h[n+shift] when n=j, valid if 0<=j<=n-1-shift
```

- The objective sum_k w_k * c_k is simpler (no max or softmax)
- The weights adapt to the current solution
- It naturally focuses on the worst peaks

Let me implement this:

```python
def mw_objective(h, w, penalty):
    """Objective for multiplicative weights method."""
    n = len(h)
    c = compute_all_c(h)
    obj = w @ c
    sum_violation = np.sum(h) - n / 2.0
    return obj + penalty * sum_violation ** 2
```

But wait, for L-BFGS-B to work, I need to pass `w` as an argument. That's fine.

The multiplicative weights algorithm:
1. Initialize w = uniform
2. For t = 1, 2, ...:
   a. Minimize sum_k w_k * c_k(h) + penalty * (sum(h) - n/2)² using L-BFGS-B
   b. Update w_k ∝ w_k * exp(η * c_k(h))
   c. Normalize w
3. Return best h found

The step size η controls exploration (small η = more uniform weights = more exploration).

This is actually a well-known technique for min-max optimization. Let me implement it.

Actually, I realize there's a subtlety: the multiplicative weights method gives a bound on the min-max value. Specifically, after T rounds:
min_h max_k c_k(h) ≤ (1/T) sum_t max_k c_k(h_t)

So if we track the average of the max values, we get an upper bound on C5.

But what we want is the actual min over h of max_k c_k(h), not the average. So we should return the best single h found, not the average.

Let me just use it as a heuristic: run multiplicative weights for many iterations, and keep the best h found.

1. L-BFGS-B with log-sum-exp (temperature annealing)
2. Peak reduction (cyclical gradient descent)
3. Multiplicative weights
4. Resolution increase
5. Final polish

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def lse_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sv = np.sum(h) - n / 2.0
    return float(lse + penalty * sv * sv)


def lbfgsb_optimize(h_init, n, T, penalty, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lse_obj, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 200}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def compute_peak_grad(h, k_idx, n):
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad


def peak_reduction(h, n, max_iter=500, tol=1e-8):
    h = h.copy()
    best_c5 = compute_c5_from_h(h)
    dx = 2.0 / n
    for it in range(max_iter):
        c_all = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = int(np.argmax(c_all))
        grad = compute_peak_grad(h, k_max, n)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        best_alpha = 0.0
        best_c5_trial = best_c5
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            step = alpha / (gnorm * n * 0.5)
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5_from_h(h_trial)
            if c5_trial < best_c5_trial:
                best_alpha = alpha
                best_c5_trial = c5_trial
                h_trial_best = h_trial.copy()
        if best_alpha > 0 and best_c5_trial < best_c5 - tol:
            h = h_trial_best
            best_c5 = best_c5_trial
        else:
            if it > 50:
                break
    return h, best_c5


def mw_objective(h, w, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    obj = float(w @ c)
    sv = np.sum(h) - n / 2.0
    return obj + penalty * sv * sv


def multiplicative_weights(h_init, n, n_rounds=30, eta=50.0, penalty=200.0, t0=time.time(), budget=1300):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    n_k = 2 * n - 1
    w = np.ones(n_k) / n_k
    bounds = [(0.0, 1.0)] * n

    for rnd in range(n_rounds):
        if time.time() - t0 > budget - 60:
            break
        result = minimize(
            mw_objective, h, args=(w, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12}
        )
        if result.x is not None:
            h = project_feasible(result.x, n)
        c5_curr = compute_c5_from_h(h)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        c_all = compute_all_c(h)
        w = w * np.exp(eta * (c_all - c_all.max()))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(n_k) / n_k
        else:
            w = w / w_sum

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: L-BFGS-B temperature annealing
    if tl() > 500:
        schedule = [
            (30.0, 50.0, 4000),
            (80.0, 100.0, 3500),
            (200.0, 200.0, 3000),
            (500.0, 500.0, 2500),
        ]
        for T, pen, iters in schedule:
            if tl() < 450:
                break
            h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=T, penalty=pen, maxiter=iters)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Peak reduction
    if tl() > 350:
        best_h, best_c5 = peak_reduction(best_h, n_best, max_iter=300)

    # Phase 3: Multiplicative weights
    if tl() > 250:
        h_mw, c5_mw = multiplicative_weights(best_h, n_best, n_rounds=20, eta=30.0,
                                             penalty=300.0, t0=t_start, budget=budget_s)
        if c5_mw < best_c5:
            best_h = h_mw
            best_c5 = c5_mw
        best_h, best_c5 = peak_reduction(best_h, n_best, max_iter=200)

    # Phase 4: Multi-restart with diverse perturbations
    if tl() > 200:
        n_restarts = 35
        for attempt in range(n_restarts):
            if tl() < 150:
                break
            frac = attempt / max(1, n_restarts)
            scale = 0.015 + 0.06 * (1.0 - frac) ** 0.7
            r = rng.random()

            if r < 0.25:
                h_pert = best_h.copy()
                n_modes = int(rng.integers(1, 4))
                for _ in range(n_modes):
                    freq = int(rng.integers(1, max(2, n_best // 4)))
                    phase = rng.random() * 2.0 * np.pi
                    h_pert += (scale / np.sqrt(n_modes)) * np.cos(2.0 * np.pi * freq * np.arange(n_best) / n_best + phase)
            elif r < 0.45:
                h_pert = best_h.copy()
                n_blocks = int(rng.integers(2, 6))
                for _ in range(n_blocks):
                    bsize = int(rng.integers(3, max(4, n_best // 8)))
                    start = int(rng.integers(0, max(1, n_best - bsize)))
                    h_pert[start:start + bsize] += scale * 2.0 * (rng.random(bsize) - 0.5)
            elif r < 0.65:
                h_pert = best_h + scale * rng.standard_normal(n_best)
            elif r < 0.8:
                c_all = compute_all_c(best_h)
                top_ks = np.argsort(c_all)[-3:]
                h_pert = best_h.copy()
                for ki in top_ks:
                    shift = ki - (n_best - 1)
                    if shift >= 0:
                        lo = max(0, shift - 5)
                        hi = min(n_best, n_best - shift + 5)
                    else:
                        lo = max(0, -shift - 5)
                        hi = min(n_best, n_best + shift + 5)
                    h_pert[lo:hi] += scale * 1.5 * rng.standard_normal(hi - lo)
            else:
                h_pert = best_h.copy()
                n_push = max(5, int(n_best * 0.2))
                push_idx = rng.choice(n_best, n_push, replace=False)
                for idx in push_idx:
                    if rng.random() < 0.5:
                        h_pert[idx] = min(1.0, h_pert[idx] + scale * 4.0)
                    else:
                        h_pert[idx] = max(0.0, h_pert[idx] - scale * 4.0)

            h_pert = project_feasible(h_pert, n_best)
            T = 40.0 + 400.0 * (1.0 - frac)
            h_new, c5_new = lbfgsb_optimize(h_pert, n_best, T=T, penalty=300.0, maxiter=2000)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                if tl() > 160:
                    best_h, best_c5 = peak_reduction(best_h, n_best, max_iter=80)

    # Phase 5: Resolution increase
    if tl() > 180:
        for n_try in [540, 405]:
            if tl() < 130:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
            h_interp = np.clip(f_interp(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_try)
            h_new, c5_new = lbfgsb_optimize(h_interp, n_try, T=100.0, penalty=500.0, maxiter=3000)
            if tl() > 80:
                h_new2, c5_new2 = lbfgsb_optimize(h_new, n_try, T=500.0, penalty=1000.0, maxiter=2000)
                if c5_new2 < c5_new:
                    h_new, c5_new = h_new2, c5_new2
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                n_best = n_try
                if tl() > 70:
                    best_h, best_c5 = peak_reduction(best_h, n_best, max_iter=150)

    # Phase 6: Final high-temperature polish
    if tl() > 40:
        h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=1000.0, penalty=2000.0, maxiter=5000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 7: Final peak reduction
    if tl() > 15:
        best_h, best_c5 = peak_reduction(best_h, n_best, max_iter=200)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
```
