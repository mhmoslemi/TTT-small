import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s / len(exp_vals)) / alpha) * dx
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        i_lo = max(0, n-1-k)
        i_hi = min(n-1, n-1+k)
        idx = np.arange(i_lo, i_hi+1)
        idx1 = idx + k - n + 1
        term1 = 1.0 - h[idx1]
        idx2 = idx - k + n - 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(i_hi - i_lo + 1)
        term2[valid2] = h[idx2[valid2]]
        grad[i_lo:i_hi+1] += wk * (term1 - term2)
    grad *= dx
    return sm_val, grad

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_and_grad(h, alpha)
        return g
    
    # Sum constraint gradient
    sum_constraint = [target_sum - np.sum(h0)]
    
    constraints = [
        {
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)
        }
    ]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def pair_swap_local_search(h, n, target_sum, n_rounds, rng, base_step=0.01):
    """Local search using pair swaps with adaptive step size."""
    h = h.copy()
    c5_cur, corr_cur = compute_c5(h)
    h_best = h.copy()
    step = base_step
    
    for rnd in range(n_rounds):
        improved = False
        n_no_improve = 0
        for _ in range(200):  # attempts per round
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            direction = 1 if rng.rand() < 0.5 else -1
            h_new = h.copy()
            h_new[i] += direction * step
            h_new[j] -= direction * step
            h_new = np.clip(h_new, 0.0, 1.0)

            c5_new, _ = compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s / len(exp_vals)) / alpha) * dx
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        i_lo = max(0, n-1-k)
        i_hi = min(n-1, n-1+k)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        idx1 = idx + k - n + 1
        term1 = 1.0 - h[idx1]
        idx2 = idx - k + n - 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(len(idx))
        term2[valid2] = h[idx2[valid2]]
        grad[i_lo:i_hi+1] += wk * (term1 - term2)
    grad *= dx
    return sm_val, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_and_grad(h, alpha)
        return g
    
    constraints = [
        {
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)
        }
    ]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def local_search(h, n, target_sum, n_iters, rng, base_step=0.01):
    h = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    step = base_step
    
    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1.0 if rng.rand() < 0.5 else -1.0
        h_new = h.copy()
        h_new[i] += direction * step
        h_new[j] -= direction * step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
            # Occasionally increase step
            if rng.rand() < 0.1:
                step = min(step * 1.2, 0.1)
        else:
            # Decrease step
            if rng.rand() < 0.05:
                step = max(step * 0.8, 0.001)
    
    return h_best, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = (m + np.log(s / len(exp_vals)) / alpha) * dx
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        i_lo = max(0, n-1-k)
        i_hi = min(n-1, n-1+k)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        idx1 = idx + k - n + 1
        term1 = 1.0 - h[idx1]
        idx2 = idx - k + n - 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(len(idx))
        term2[valid2] = h[idx2[valid2]]
        grad[i_lo:i_hi+1] += wk * (term1 - term2)
    grad *= dx
    return sm_val, grad

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h_c = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_and_grad(h_c, alpha)
        return v
    def grad(x):
        h_c = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_and_grad(h_c, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    h = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    step = base_step
    no_improve = 0
    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * (1.0 if rng.rand() < 0.5 else -1.0)
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
            no_improve = 0
            if rng.rand() < 0.15:

        else:
            no_improve += 1
            if no_improve > 200:
                step = max(step * 0.85, 0.002)
                no_improve = 0
    return h_best, c5_cur
