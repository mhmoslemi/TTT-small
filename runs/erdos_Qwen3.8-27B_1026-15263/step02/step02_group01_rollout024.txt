1. Compute all correlations
2. Find the argmax shift(s)
3. Compute the subgradient of the max at those shifts
4. Take a step in the -subgradient direction
5. Project to [0,1] and sum constraint
6. Use step size decay (e.g., 1/sqrt(t) or backtracking)

The subgradient of max_k c_k(h) at h is the subgradient of c_{k*}(h) where k* is the argmax (or a convex combination if multiple argmax).

dc_k/dh[j] can be computed in O(k) or O(n) time for a single k.
Computing the subgradient for the top few k values is O(n * top_k).

This is much faster than computing all correlations!

Let me implement this:

```python
def subgradient_step(h, n, top_k=5, lr=0.01):
    """One step of projected subgradient method."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k shifts
    top_indices = np.argsort(corr)[-top_k:]
    # Weights (equal for simplicity, or proportional to value)
    # Use softmax for smoothness
    top_vals = corr[top_indices]
    w_k = np.exp(100 * (top_vals - top_vals.max()))
    w_k /= w_k.sum()
    
    g = np.zeros(n)
    for idx, ki in enumerate(top_indices):
        s = ki - (n - 1)  # shift
        # c_k = sum_{i: valid} h[i] * (1-h[i+s])
        # dc_k/dh[j] = (1-h[j+s]) if 0<=j+s<n, -h[j-s] if 0<=j-s<n
        for j in range(n):
            val = 0.0
            if 0 <= j + s < n:
                val += (1.0 - h[j + s])
            if 0 <= j - s < n:
                val -= h[j - s]
            g[j] += w_k[idx] * val
    
    return g
```

```python
def subgradient_step(h, n, top_k=5, lr=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_k:]
    top_vals = corr[top_indices]
    w_k = np.exp(100 * (top_vals - top_vals.max()))
    w_k /= w_k.sum()
    
    g = np.zeros(n)
    for idx, ki in enumerate(top_indices):
        s = ki - (n - 1)
        # dc/dh[j] = (1-h[j+s]) * 1_{0<=j+s<n} - h[j-s] * 1_{0<=j-s<n}
        # Vectorized:
        # Term 1: for j where 0<=j+s<n, add (1-h[j+s])
        # j ranges from max(0,-s) to min(n-1, n-1-s)
        j_lo1 = max(0, -s)
        j_hi1 = min(n-1, n-1-s)
        if j_hi1 >= j_lo1:
            g[j_lo1:j_hi1+1] += w_k[idx] * (1.0 - h[j_lo1+s:j_hi1+s+1])
        
        # Term 2: for j where 0<=j-s<n, subtract h[j-s]
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, n-1+s)
        if j_hi2 >= j_lo2:
            g[j_lo2:j_hi2+1] -= w_k[idx] * h[j_lo2-s:j_hi2-s+1]
    
    return g
```

This is O(n * top_k) per step, which is fast. With top_k=5 and n=360, that's 1800 operations per step. We can do millions of steps in 1300 seconds.

    h_best = h.copy()
    lr = lr_init
    t = 1.0
    y = h.copy()  # for FISTA-like acceleration
    
    for it in range(n_iters):
        g = subgradient_grad(h, n)
        # FISTA step
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        coeff = (t - 1.0) / t_new
        y = h_new + coeff * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        if it % 10 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            # Adaptive learning rate
            if c5_cur < c5_best + 1e-6:
                lr = lr * 1.01
            else:
                lr = lr * 0.9
    
    return h_best, c5_best

def project_to_sum(h, target_sum):

        else:
