def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-15:

            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000, ftol=1e-14):
    def obj_fn(h):
        return smooth_max_and_grad(h, alpha)[0]
    def grad_fn(h):
        return smooth_max_and_grad(h, alpha)[1]
    def cons_fn(x):
        return np.sum(x) - target_sum
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter=5000):
    def obj_fn(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target_sum - np.sum(x)
        pen = 0.0
        pen_g = np.zeros(n-1)
        if h[n-1] < -0.01:
            pen = 1000.0 * h[n-1] * h[n-1]
            pen_g[:] = -2000.0 * h[n-1]
        elif h[n-1] > 1.01:
            pen = 1000.0 * (h[n-1]-1.0) * (h[n-1]-1.0)
            pen_g[:] = 2000.0 * (h[n-1]-1.0)
        o, g_full = smooth_max_and_grad(h, alpha)
        g = g_full[:n-1] + pen_g
        return o + pen, g
    def obj_only(x):
        return obj_fn(x)[0]
    def grad_only(x):
        return obj_fn(x)[1]
    x0 = h0[:n-1].copy()
    bounds = [(0.0, 1.0)] * (n-1)
    try:
        res = minimize(obj_only, x0, method='L-BFGS-B', jac=grad_only,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_full = np.empty(n)
        h_full[:n-1] = res.x
        h_full[n-1] = target_sum - np.sum(res.x)
        return project_to_sum(h_full, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def topk_grad_step(h, K, step_size):
    n = len(h)
    corr = compute_corr(h)
    top_k = np.argsort(corr)[-K:]
    j_idx = np.arange(n)[:, None]
    m_idx = top_k[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc])
    cs = valid_s * h[sc]
    grad = np.sum(cf - cs, axis=1) / K
    h_new = h - step_size * grad
    return h_new

def binary_anneal(n, target_sum, rng, n_iter=4000):
    h = np.zeros(n)
    ones_pos = rng.choice(n, size=int(round(target_sum)), replace=False)
    h[ones_pos] = 1.0
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    t_cur = 0.3
    for it in range(n_iter):
        t_cur = 0.3 * (0.01 / 0.3) ** (it / max(n_iter, 1))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(t_cur, 1e-8)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def fourier_h(n, coeffs, phases, target_sum):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for i in range(len(coeffs)):
        h += coeffs[i] * np.sin(2.0 * np.pi * (i+1) * x / 2.0 + phases[i])
    return project_to_sum(h, target_sum)

def fourier_optimize(n, target_sum, n_modes, rng, maxiter=500):
    def fourier_obj(params):
        coeffs = params[:n_modes]
        phases = params[n_modes:]
        h = np.linspace(0, 2.0, n, endpoint=False)
        hv = np.full(n, 0.5)
        for i in range(n_modes):
            hv += coeffs[i] * np.sin(2.0 * np.pi * (i+1) * x / 2.0 + phases[i])
        hv = project_to_sum(hv, target_sum)
        return compute_c5(hv)
    x = np.linspace(0, 2.0, n, endpoint=False)
    def fourier_obj2(params):
        coeffs = params[:n_modes]
        phases = params[n_modes:]
        hv = np.full(n, 0.5)
        for i in range(n_modes):
            hv += coeffs[i] * np.sin(2.0 * np.pi * (i+1) * x / 2.0 + phases[i])
        hv = project_to_sum(hv, target_sum)
        return compute_c5(hv)
    best_h = None
    best_c5 = float('inf')
    for trial in range(10):
        p0 = np.zeros(2 * n_modes)
        p0[:n_modes] = rng.uniform(-0.3, 0.3, n_modes)
        p0[n_modes:] = rng.uniform(0, 2*np.pi, n_modes)
        try:
            res = minimize(fourier_obj2, p0, method='Nelder-Mead',
                          options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
            coeffs = res.x[:n_modes]
            phases = res.x[n_modes:]
            hv = np.full(n, 0.5)
            for i in range(n_modes):
                hv += coeffs[i] * np.sin(2.0 * np.pi * (i+1) * x / 2.0 + phases[i])
            hv = project_to_sum(hv, target_sum)
            c5_val = compute_c5(hv)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = hv.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0

    if time_left() > 100:
        for alpha in [100, 500, 2000, 10000]:
            if time_left() < 30:
                break
            h_new = slsqp_optimize(best_h, n1, ts1, alpha, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if time_left() > 100:
        for alpha in [200, 1000, 5000]:
            if time_left() < 30:
                break
            h_new = lbfgsb_optimize(best_h, n1, ts1, alpha, maxiter=5000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if time_left() > 80:
        h_work = best_h.copy()
        c5_work = compute_c5(h_work)
        for step in range(3000):
            if time_left() < 20:
                break
            decay = 1.0 / (1.0 + step * 0.005)
            K = max(3, int(15 * decay) + 3)
            ss = 0.003 * decay + 0.0005
            h_new = topk_grad_step(h_work, K, ss)
            h_new = project_to_sum(h_new, ts1)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-12:
                h_work = h_new
                c5_work = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_work.copy()
            elif step % 200 == 0:
                h_work = best_h + rng.normal(0, 0.02, n1)
                h_work = project_to_sum(h_work, ts1)
                c5_work = compute_c5(h_work)

    if time_left() > 120:
        for n_try in [320, 400, 480]:
            if time_left() < 100:
                break
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = project_to_sum(np.interp(x_new, x_old, best_h), ts_try)
            c5_interp = compute_c5(h_interp)
            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_interp.copy()
                best_n = n_try
            for alpha in [100, 500, 3000]:
                if time_left() < 25:
                    break
                h_opt = slsqp_optimize(h_interp, n_try, ts_try, alpha, maxiter=1000)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try
                h_interp = h_opt

    if time_left() > 150:
        n_bin = best_n
        ts_bin = n_bin / 2.0
        for trial in range(3):
            if time_left() < 40:
                break
            h_bin, c5_bin = binary_anneal(n_bin, ts_bin, rng, n_iter=3000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            h_smooth = slsqp_optimize(h_bin, n_bin, ts_bin, 500, maxiter=1500)
            c5_smooth = compute_c5(h_smooth)
            if c5_smooth < best_c5:
                best_c5 = c5_smooth
                best_h = h_smooth.copy()

    if time_left() > 100:
        n_p = best_n
        ts_p = n_p / 2.0
        for trial in range(60):
            if time_left() < 20:
                break
            noise_scale = 0.03 + 0.12 * (1.0 - trial / 60.0)
            h_shake = best_h + rng.normal(0, noise_scale, n_p)
            h_shake = project_to_sum(h_shake, ts_p)
            for alpha in [200, 1500]:
                if time_left() < 15:
                    break
                h_opt = slsqp_optimize(h_shake, n_p, ts_p, alpha, maxiter=1000)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_p
                h_shake = h_opt

    if time_left() > 80:
        n_f = best_n
        ts_f = n_f / 2.0
        x_f = np.linspace(0, 2.0, n_f, endpoint=False)
        for n_modes in [3, 5, 7]:
            if time_left() < 30:
                break
            h_four, c5_four = fourier_optimize(n_f, ts_f, n_modes, rng, maxiter=300)
            if h_four is not None and c5_four < best_c5:
                best_c5 = c5_four
                best_h = h_four.copy()
            if h_four is not None:
                h_pol = slsqp_optimize(h_four, n_f, ts_f, 1000, maxiter=1000)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    if time_left() > 40:
        for alpha in [20000, 100000]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=5000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time_left() > 20:
        h_final = best_h.copy()
        c5_final = compute_c5(h_final)
        for _ in range(500):
            if time_left() < 5:
                break
            corr = compute_corr(h_final)
            worst_k = np.argmax(corr)
            s = worst_k - (best_n - 1)
            n_f = best_n
            j_idx = np.arange(n_f)[:, None]
            m_idx = worst_k
            first_idx = j_idx - m_idx + n_f - 1
            second_idx = j_idx + m_idx - n_f + 1
            valid_f = (first_idx >= 0) & (first_idx < n_f)
            valid_s = (second_idx >= 0) & (second_idx < n_f)
            fc = np.clip(first_idx, 0, n_f - 1)
            sc = np.clip(second_idx, 0, n_f - 1)
            cf = valid_f * (1.0 - h_final[fc])
            cs = valid_s * h_final[sc]
            grad = (cf - cs).flatten()
            g_proj = grad - np.mean(grad)
            for ss in [0.001, 0.005, 0.01, 0.02]:
                h_try = h_final - ss * g_proj
                h_try = project_to_sum(h_try, best_n / 2.0)
                c5_try = compute_c5(h_try)
                if c5_try < c5_final - 1e-12:
                    h_final = h_try
                    c5_final = c5_try
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final.copy()
                    break

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
