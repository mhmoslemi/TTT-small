import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s / len(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        g[k:] += wk * (1.0 - h[:n-k])
        g[:n-k] -= wk * h[k:]
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_value(h, alpha)
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fista_optimize(h0, n, target_sum, alpha, n_iters, lr_init, t_start, time_limit):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        _, g = smooth_max_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        if it % 200 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_best = h.copy()
        # Backtracking line search every 100 iters
        if it % 100 == 0 and it > 0:
            c5_test = compute_c5(h)
            if c5_test > c5_cur * 1.01:
                lr *= 0.7
    c5_final = compute_c5(h)
    if c5_final < c5_cur:
        c5_cur = c5_final
        h_best = h.copy()
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, time_start, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
    for rnd in range(max_rounds):
        if time.time() - time_start > time_limit:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_local_search(h0, n, rng, time_start, time_limit):
    # Project to binary
    h_bin = (h0 > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove some ones (those closest to 0.5)
        ones_idx = np.where(h_bin == 1.0)[0]
        probs = h0[ones_idx]  # lower is more likely to be 0
        remove_order = np.argsort(probs)[:n_ones - target_ones]
        h_bin[ones_idx[remove_order]] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0.0)[0]
        probs = h0[zeros_idx]  # higher is more likely to be 1
        add_order = np.argsort(-probs)[:target_ones - n_ones]
        h_bin[zeros_idx[add_order]] = 1.0
    
    c5_cur = compute_c5(h_bin)
    
    # Local search: try flipping pairs
    for it in range(5000):
        if time.time() - time_start > time_limit:
            break
        improved = False
        for _ in range(50):
            if time.time() - time_start > time_limit or improved:
                break
            i, j = rng.choice(n, 2, replace=False)
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c5_new = compute_c5(h_bin)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                improved = True
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        return m
    return m + np.log(s / len(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[k:] += wk * (1.0 - h[:n-k])
        g[:n-k] -= wk * h[k:]
    return sm_val, g

def augmented_lagrange_optimize(h0, n, target_sum, alpha_sm, t_start, time_limit, max_iters=500):
    # Augmented Lagrangian: f(h) + nu*(sum(h)-t) + mu/2*(sum(h)-t)^2
    # Optimized via L-BFGS-B with box constraints
    nu = 0.0
    mu = 1.0
    h = h0.copy()
    
    for al_iter in range(10):
        if time.time() - t_start > time_limit:
            break
        
        def obj(x):
            h_local = np.clip(x, 0.0, 1.0)
            sm, _ = smooth_max_value(h_local, alpha_sm), 0
            penalty = nu * (np.sum(x) - target_sum) + 0.5 * mu * (np.sum(x) - target_sum) ** 2
            return sm + penalty
        
        def grad(x):
            h_local = np.clip(x, 0.0, 1.0)
            _, g_sm = smooth_max_grad(h_local, alpha_sm)
            s = np.sum(x) - target_sum
            g_pen = np.full(n, nu + mu * s)
            return g_sm + g_pen
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iters, 'gtol': 1e-10})
            h = res.x
        except Exception:
            pass
        
        # Update multipliers
        s = np.sum(h) - target_sum
        nu = nu + mu * s
        mu = min(mu * 5.0, 1e6)
        
        if abs(s) < 1e-10:
            break
    
    h = project_to_sum(h, target_sum)
    return h

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_value(h, alpha)
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
