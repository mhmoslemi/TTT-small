        hi2 = min(2*(n-1), j + (n-1))
        if hi2 >= lo2:
            h_idx = j - np.arange(lo2, hi2+1) + (n-1)
            g[j] -= np.dot(w[lo2:hi2+1], h[h_idx])
    
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fine_coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_h_i = h[i]
            for ss in step_sizes:
                for sign in [1.0, -1.0]:
                    new_val = h[i] + sign * ss
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    h_new = h.copy()
                    h_new[i] = new_val
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_val - 1e-15:
                        best_val = c5_new
                        best_h_i = new_val
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n - 1)) / alpha
    
    g = np.ones(n)
    for j in range(n):
        lo1 = max(0, (n-1) - j)
        hi1 = min(2*(n-1), 2*(n-1) - j)
        if hi1 >= lo1:
            h_idx = j + np.arange(lo1, hi1+1) - (n-1)
            g[j] -= np.dot(w[lo1:hi1+1], h[h_idx])
        lo2 = max(0, j)
        hi2 = min(2*(n-1), j + (n-1))
        if hi2 >= lo2:
            h_idx = j - np.arange(lo2, hi2+1) + (n-1)
            g[j] -= np.dot(w[lo2:hi2+1], h[h_idx])
    
    return sm_val, g

def slsqp_opt(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    g = np.ones(n)
    for j in range(n):
        lo1 = max(0, (n - 1) - j)
        hi1 = min(2 * (n - 1), 2 * (n - 1) - j)
        if hi1 >= lo1:
            h_idx = j + np.arange(lo1, hi1 + 1) - (n - 1)
            g[j] -= np.dot(w[lo1:hi1 + 1], h[h_idx])
        lo2 = max(0, j)
        hi2 = min(2 * (n - 1), j + (n - 1))
        if hi2 >= lo2:
            h_idx = j - np.arange(lo2, hi2 + 1) + (n - 1)
            g[j] -= np.dot(w[lo2:hi2 + 1], h[h_idx])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent(h0, n, target_sum, max_rounds, rng, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            best_val = c5_cur
            best_nv = h[i]
            for s in step_sizes:
                for sign in [1.0, -1.0]:
                    nv = h[i] + sign * s
                    if nv < 0.0 or nv > 1.0:
                        continue
                    h_new = h.copy()
                    h_new[i] = nv
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_val - 1e-15:
                        best_val = c5_new
                        best_nv = nv
            if best_nv != h[i]:
                h[i] = best_nv
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
