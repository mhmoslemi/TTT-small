import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return m + np.log(s / len(exp_vals)) / alpha

def fast_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    W = np.exp(shifted)
    W = W / W.sum()
    A = np.convolve(W, np.ones(n), 'valid')
    B = np.correlate(W, h[::-1], 'full')[n-1:2*n-1]
    C = np.correlate(h, W[n-1:], 'full')[n-1:2*n-1]
    return A - B - C

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_val(h, alpha)
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        return fast_gradient(h, alpha)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter, penalty):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        base = smooth_max_val(h, alpha)
        pen = penalty * (np.sum(h) - target_sum) ** 2
        return base + pen
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        g = fast_gradient(h, alpha)
        g += 2.0 * penalty * (np.sum(h) - target_sum) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def projected_gradient_fast(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    y = h.copy()
    lr = lr_init
    for it in range(n_iters):
        g = fast_gradient(y, alpha)
        g_sum = np.sum(g)
        g_centered = g - g_sum / n
        h_new = np.clip(y - lr * g_centered, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.85
            if lr < 1e-6:
                break
    return h_best, c5_cur

def coordinate_descent_fast(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01, -0.02]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def generate_starts(n, rng, best_h, best_n):
    starts = []
    ts = n / 2.0
    x = np.linspace(0, 2, n, endpoint=False)
    
    if best_h is not None:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            h_interp = interpolate_to_resolution(best_h, best_n, n)
            starts.append(project_to_sum(h_interp, ts))
    
    starts.append(np.full(n, 0.5))
    
    hr = rng.uniform(0.0, 1.0, n)
    starts.append(project_to_sum(hr, ts))
    
    for freq in [1, 2, 3, 4, 5, 7]:
        for phase in [0.0, 0.3, 0.7, 1.2]:
            amp = 0.3 + 0.2 * rng.random()
            hs = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            hs = project_to_sum(hs, ts)
            starts.append(hs)
    
    h_blk = np.zeros(n)
    for period in [max(2, n//40), max(3, n//20), max(4, n//12)]:
        h_b = np.zeros(n)
        for s in range(0, n, period * 2):
            h_b[s:s+period] = 1.0
        starts.append(project_to_sum(h_b, ts))
    
    h_tri = np.zeros(n)
    for period in [max(4, n//15), max(6, n//10)]:
        h_t = np.zeros(n)
        for s in range(0, n, period):
            half = period // 2
            ramp_up = np.linspace(0, 1, half)
            ramp_down = np.linspace(1, 0, period - half)
            seg = np.concatenate([ramp_up, ramp_down])
            h_t[s:s+period] = seg[:period]
        starts.append(project_to_sum(h_t, ts))
    
    n_start = min(len(starts), 12)
    return starts[:n_start]

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    resolutions = []
    if best_n <= 360:
        resolutions = [360, 480, 600]
    elif best_n <= 480:
        resolutions = [480, 600, 720]
    else:
        resolutions = [best_n, best_n + 120, best_n + 240]

    for n in resolutions:
        if time_left() < 150:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
        continue
        ts = round(ts)

        starts = generate_starts(n, rng, best_h, best_n)

        for h_start in starts:
            if time_left() < 60:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()
            
            alpha_schedule = [5.0, 15.0, 40.0, 100.0, 250.0, 600.0, 1200.0]
            for alpha in alpha_schedule:
                if time_left() < 40:
                    break
                if n <= 480:
                    maxiter = max(50, min(300, int(time_left() * 0.8)))
                else:
                    maxiter = max(30, min(200, int(time_left() * 0.5)))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt
            
            if time_left() > 80:
                lr_init = 0.1 if n <= 480 else 0.05
                n_pg = min(2000, int(time_left() * 2))
                h_pg, c5_pg = projected_gradient_fast(h_cur, n, ts, 100.0, n_pg, lr_init)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                h_cur = h_pg

            if time_left() > 50:
                h_cd, c5_cd = coordinate_descent_fast(h_cur, n, ts, 30, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    if best_h is not None and time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        ts_f = round(ts_f)

        for alpha in [500.0, 1000.0, 2000.0, 5000.0]:
            if time_left() < 25:
                break
            maxiter = max(50, min(400, int(time_left() * 1.2)))
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, maxiter)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 30:
            h_cd, c5_cd = coordinate_descent_fast(best_h, n_f, ts_f, 80, rng,
                                                   step_sizes=[0.005, 0.002, 0.001, 0.0005,
                                                              -0.0005, -0.001, -0.002, -0.005])
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 20:
            n_pg = min(3000, int(time_left() * 1.5))
            h_pg, c5_pg = projected_gradient_fast(best_h, n_f, ts_f, 200.0, n_pg, 0.02)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

            h_cd2, c5_cd2 = coordinate_descent_fast(best_h, n_f, ts_f, 40, rng,
                                                     step_sizes=[0.002, 0.001, 0.0005, 0.0002,
                                                                -0.0002, -0.0005, -0.001, -0.002])
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    ts_final = round(ts_final)
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
