    n = len(h)

    # Average gradient over top indices
    g = np.zeros(n)

        g += corr_gradient_single(h, k)
    g /= len(top_idx)
    h_new = h - step_size * g
    return project_to_sum(h_new, target_sum)

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_alpha(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # For numerical stability
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return m_val
    return m_val + (max_shifted + np.log(sum_exp)) / alpha

def smooth_max_grad_alpha(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def obj(x):
        return smooth_max_obj_alpha(x, alpha)
    def grad(x):
        return smooth_max_grad_alpha(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter=500, penalty=50.0):
    def obj_pen(x):
        return smooth_max_obj_alpha(x, alpha) + penalty * (np.sum(x) - target_sum) ** 2
    def grad_pen(x):
        return smooth_max_grad_alpha(x, alpha) + 2.0 * penalty * (np.sum(x) - target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(obj_pen, h0.copy(), fprim=grad_pen,
                            pgtol=1e-14, maxiter=maxiter, iprint=0,
                            bounds=bounds)
        return project_to_sum(res, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def subgradient_optimize(h0, target_sum, n_iter=3000, top_frac=0.05):

        n = len(h)

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_alpha(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return m_val
    return m_val + (max_shifted + np.log(sum_exp)) / alpha

def smooth_max_grad_alpha(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def obj(x):
        return smooth_max_obj_alpha(x, alpha)
    def grad(x):
        return smooth_max_grad_alpha(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_optimize(h0, n, target_sum, alpha, maxiter=800, penalty=200.0):
    def obj_pen(x):
        return smooth_max_obj_alpha(x, alpha) + penalty * (np.sum(x) - target_sum) ** 2
    def grad_pen(x):
        return smooth_max_grad_alpha(x, alpha) + 2.0 * penalty * (np.sum(x) - target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(obj_pen, h0.copy(), fprim=grad_pen,
                            pgtol=1e-14, maxiter=maxiter, iprint=0,
                            bounds=bounds)
        return project_to_sum(res, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def multi_top_subgradient(h0, target_sum, n_iter=3000, top_k=3, step_init=0.1):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            i = np.arange(n)
            idx1 = i + lag
            idx2 = i - lag
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            g[valid1] += (1.0 - h[idx1[valid1]])
            g[valid2] -= h[idx2[valid2]]
        g /= top_k
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        step = step_init / (1.0 + 0.005 * it)
        h_new = h - step * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def binary_flip_search(h, n, target_sum, rng, max_rounds=60):
    h = (h > 0.5).astype(float)
    attempts = 0
    while abs(h.sum() - target_sum) > 1e-10 and attempts < 20000:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h > 0)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h < 1)[0])
            h[idx] = 1.0
        attempts += 1
    if abs(h.sum() - target_sum) > 1e-10:
        return h, compute_c5(h)
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_trials = min(300, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        c5_cur = best_c5
    return h, c5_cur

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def make_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.4, 0.45]:
            s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))
            starts.append(project_to_sum(s, target_sum))
    for nblks in [3, 4, 5, 6, 8]:
        blk = max(1, n // nblks)
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    for _ in range(5):
        f1 = rng.choice([1, 2, 3])
        f2 = rng.choice([2, 3, 4, 5, 6, 7])
        a1 = rng.uniform(0.15, 0.4)
        a2 = rng.uniform(0.05, 0.2)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        s = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2 + p2)
        starts.append(project_to_sum(s, target_sum))
    for _ in range(3):
        hb = (rng.rand(n) > 0.5).astype(float)
        attempts = 0
        while abs(hb.sum() - target_sum) > 1e-10 and attempts < 20000:
            if hb.sum() > target_sum:
                idx = rng.choice(np.where(hb > 0)[0])
                hb[idx] = 0.0
            else:
                idx = rng.choice(np.where(hb < 1)[0])
                hb[idx] = 1.0
            attempts += 1
        if abs(hb.sum() - target_sum) < 1e-10:
            starts.append(hb.copy())
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            h_interp = interpolate_h(best_h, best_n, n)
            starts.append(project_to_sum(h_interp, target_sum))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)

    n2 = int(n1 * 1.5)

    if time_left() > 120:

                c5_opt = compute_c5(h_opt)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
