import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = alpha * (corr - M)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    W = exp_vals / s
    sm_val = M + np.log(s / (2*n-1)) / alpha
    
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w_val = W[n-1+d]
        if w_val < 1e-14:
            continue
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += w_val * (1.0 - h[i_lo+d:i_hi+d+1])
        i_lo2 = max(0, d)
        i_hi2 = min(n-1, n-1+d)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w_val * h[i_lo2-d:i_hi2-d+1]
    
    return sm_val, grad

def nesterov_projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init, time_limit=60):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t0 = time.time()
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        _, g = smooth_max_and_grad(y, alpha)
        # Add gradient of constraint penalty
        # Actually, let's handle constraint via projection
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = np.clip(h_new + mo * (h_new - h), 0.0, 1.0)
        h = h_new
        t = t_new
        if it % 50 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_best = h.copy()
                lr = min(lr * 1.02, 0.3)
            else:
                lr *= 0.9
    c5_final = compute_c5(h)
    if c5_final < c5_cur:
        c5_cur = c5_final
        h_best = h.copy()
    return h_best, c5_cur

def slaqp_step(h0, n, target_sum, alpha, maxiter, time_limit=60):
    # Use L-BFGS-B with a penalty for the sum constraint
    # penalty: mu * (sum(h) - target_sum)^2
    mu = 100.0
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        penalty = mu * (np.sum(h) - target_sum) ** 2
        return v + penalty
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        g = g + 2.0 * mu * (np.sum(h) - target_sum)
        return g
    
    t0 = time.time()
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                               'maxfun': int(time_limit * 20)})
        h_res = project_to_sum(res.x, target_sum)
        return h_res
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_local_search(h0, n, target_sum, max_rounds, rng, time_limit=120):
    h = h0.copy()
    c5_cur = compute_c5(h)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        # Try flipping pairs (i: 0->1, j: 1->0) to preserve sum
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Random pair flips
        n_tries = min(500, len(zeros) * len(ones) // 10)
        improved = False
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j:
                continue
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            # Actually, for binary local search, set them to 0 and 1
            h[i] = 1.0
            h[j] = 0.0
            c5_new = compute_c5(h)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                improved = True
            else:
                h[i] = 0.0
                h[j] = 1.0

            for _ in range(200):

def coordinate_descent(h0, n, target_sum, max_rounds, rng, time_limit=60):

            break

def generate_thue_morse(n, rng):
    # Generate a Thue-Morse-like sequence of length n with exactly n/2 ones
    t = np.zeros(n, dtype=int)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # This gives roughly n/2 ones
    # Adjust if needed
    ones = np.sum(t == 1)
    target = n // 2
    if ones > target:
        # Flip some 1s to 0s
        idx = np.where(t == 1)[0]
        flip_count = ones - target
        idx_to_flip = rng.choice(idx, flip_count, replace=False)
        t[idx_to_flip] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        flip_count = target - ones
        idx_to_flip = rng.choice(idx, flip_count, replace=False)
        t[idx_to_flip] = 1
    return t.astype(float)
