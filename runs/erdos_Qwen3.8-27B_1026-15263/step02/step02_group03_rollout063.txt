def pair_coordinate_step(h, i, j, n):
    """Optimize h[i] and h[j] simultaneously, maintaining sum constraint."""
    dx = 2.0 / n
    c = h[i] + h[j]  # must maintain this sum
    # h[j] = c - h[i]
    # h[i] in [max(0, c-1), min(1, c)]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if lo >= hi:
        return h[i]
    
    # For each lag m, corr_m as function of x = h[i]:
    # h[i] = x, h[j] = c - x
    # corr_m = sum_{valid i'} h'[i'] * (1 - h'[i'+m]) * dx
    # where h' differs from h only at positions i and j
    
    # Terms involving position i:
    #   i'=i: h'[i]*(1-h'[i+m]) = x*(1-h'[i+m])
    #   i'+m=i, i'=i-m: h'[i-m]*(1-h'[i]) = h'[i-m]*(1-x)
    # Terms involving position j:
    #   i'=j: h'[j]*(1-h'[j+m]) = (c-x)*(1-h'[j+m])
    #   i'+m=j, i'=j-m: h'[j-m]*(1-h'[j]) = h'[j-m]*(1-(c-x))
    
    # For each m, corr_m is quadratic in x (or linear, or constant)
    # Let's compute coefficient a (of x^2), b (of x), c_const for each m
    
    # Actually, let's just evaluate at several points and find the minimum
    # Since it's max of quadratics, it's convex in x, so we can use ternary search
    
    def obj_at(x):
        h_new = h.copy()
        h_new[i] = x
        h_new[j] = c - x
        g_new = 1.0 - h_new
        corr = np.correlate(h_new, g_new, mode="full")
        return np.max(corr) * dx
    
    # Ternary search on [lo, hi]
    a, b = lo, hi
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if obj_at(m1) < obj_at(m2):
            b = m2
        else:
            a = m1
    
    x_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = x_opt
    h_new[j] = c - x_opt
    return h_new

def pair_coordinate_descent(h, n, sweeps=5, pairs_per_sweep=50):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for sweep in range(sweeps):
        improved = False
        rng_pairs = np.random.RandomState(42 + sweep)
        for _ in range(pairs_per_sweep):
            i = rng_pairs.randint(n)
            j = rng_pairs.randint(n)
            if i == j:
                continue
            h_new = pair_coordinate_step(best, i, j, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best = h_new
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best, best_c5

The terms involving h'[j] or h'[k] (where k is the compensation):

If h is a "Heaviside" step function that's 1 on some sets and 0 elsewhere, then h(x)(1-h(x+k)) = 1 when h(x)=1 and h(x+k)=0, and 0 otherwise. So the integral counts the "boundary" between the 1-region and the 0-region at shift k.

For a single interval [a,b] where h=1:
∫_0^2 h(x)(1-h(x+k))dx = measure of {x: x ∈ [a,b] and x+k ∉ [a,b]}
= measure of [a,b] \ [a-k, b-k]
= (b-a) - measure of [a,b] ∩ [a-k, b-k]

This is maximized when the overlap between [a,b] and [a-k, b-k] is minimized, which happens when |k| is close to (b-a) or (b-a)+something.

For a more complex h with multiple intervals, the structure is more complex.

    corr = np.correlate(h, g, mode="full")

        if 0 <= j + m < n:

        if 0 <= j - m < n:
            s -= h[j - m]
        grad[j] = s * dx
    return grad, max_idx, corr[max_idx] * dx
