            c = np.correlate(h_pert, 1.0 - h_pert, mode="full")
            top_idx = np.argsort(c)[-15:]
            h_opt, c5_opt = slsqp_reduced(h_pert, n, top_idx, maxiter=2500, t_margin=0.08)

        delta = c5_opt - best_c5
        if delta < -1e-12:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            no_improve = 0
        else:
            no_improve += 1
            if delta < T * 15 and rng.random() < np.exp(-delta / T):
                best_h = h_opt.copy()
                best_c5 = c5_opt
                no_improve = 0

    if time.time() - t0 < budget_s - 150:
        for n_try in [180, 360, 200]:
            if time.time() - t0 > budget_s - 80:
                break
            n_alt = n_try
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_alt, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
            h_interp = np.clip(f_interp(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_alt)
            h_new, c5_new = active_set_optimize(h_interp, n_alt, t0, budget_s - 60, n_active=20, max_inner=4)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                n = n_alt

    if time.time() - t0 < budget_s - 120:
        n_starts = 4
        for s in range(n_starts):
            if time.time() - t0 > budget_s - 80:
                break
            h_rand = random_binary_start(n, rng, target_frac=0.5)
            h_rand = h_rand + 0.05 * rng.standard_normal(n)
            h_rand = project_feasible(h_rand, n)
            h_try, c5_try = active_set_optimize(h_rand, n, t0, budget_s - 70, n_active=20, max_inner=3)
            if c5_try < best_c5 - 1e-10:
                best_h = h_try
                best_c5 = c5_try

    if time.time() - t0 < budget_s - 60:
        for attempt in range(15):
            if time.time() - t0 > budget_s - 15:
                break
            sigma = 0.001 + 0.004 * (1.0 - attempt / 15.0)
            h_pert = best_h + sigma * rng.standard_normal(n)
            h_pert = project_feasible(h_pert, n)
            c = np.correlate(h_pert, 1.0 - h_pert, mode="full")
            top_idx = np.argsort(c)[-20:]
            h_opt, c5_opt = slsqp_reduced(h_pert, n, top_idx, maxiter=3000, t_margin=0.02)
            if c5_opt < best_c5 - 1e-13:
                best_h = h_opt
                best_c5 = c5_opt

    if time.time() - t0 < budget_s - 3:
        best_h, best_c5 = slsqp_full(best_h, n, maxiter=8000, t_margin=0.01)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, n
