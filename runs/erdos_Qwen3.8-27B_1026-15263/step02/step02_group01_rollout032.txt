The key insight is that we want to minimize the MAX of the correlations. At the optimum, multiple correlations should be at (or near) the maximum. The subgradient of the max is the average of the gradients of the active (at-max) functions.

2. Compute the average of their gradients

    h_cur = h_interp.copy()
    alphas = [20.0, 50.0, 100.0, 300.0]
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        maxiter = max(50, min(500, int((time_limit - (time.time() - t_start)) * 1.0)))
        h_opt = slsqp_optimize(h_cur, n_target, target_sum, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < c5_inter:
            c5_inter = c5_opt
            h_interp = h_opt
        h_cur = h_opt

    if time.time() - t_start < time_limit - 30:
        h_mom, c5_mom = momentum_active_set(h_cur, n_target, target_sum,
                                            time_limit - (time.time() - t_start),
                                            lr_init=0.03, k_active=80)
        if c5_mom < c5_inter:
            c5_inter = c5_mom
            h_interp = h_mom

    return h_interp, c5_inter

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    target_sum = n / 2.0

    # Phase 1: Quick diversification with structured perturbations
    if time_left() > 200:
        perturb_starts = []
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1, 2, 3, 4, 5, 7, 10]:
            for amp in [0.05, 0.1, 0.15]:
                for phase in [0.0, 0.5, 1.0]:
                    pert = best_h + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
                    perturb_starts.append(project_to_sum(pert, target_sum))
        for _ in range(5):
            pert = best_h + rng.randn(n) * 0.03
            perturb_starts.append(project_to_sum(pert, target_sum))

        for h_start in perturb_starts:
            if time_left() < 150:
                break
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
            h_mom, c5_mom = momentum_active_set(h_start, n, target_sum,
                                                min(15, time_left() - 100),
                                                lr_init=0.03, k_active=40)
            if c5_mom < best_c5:
                best_c5 = c5_mom
                best_h = h_mom.copy()

    # Phase 2: Main momentum/active-set optimization
    if time_left() > 200:
        budget_phase2 = min(500, time_left() - 150)
        h_mom, c5_mom = momentum_active_set(best_h, n, target_sum,
                                            budget_phase2 * 0.5,
                                            lr_init=0.04, k_active=50, eps_frac=0.02)
        if c5_mom < best_c5:
            best_c5 = c5_mom
            best_h = h_mom.copy()

    if time_left() > 150:
        budget_nest = min(300, time_left() - 120)
        h_nest, c5_nest = nesterov_active_set(best_h, n, target_sum,
                                               budget_nest,
                                               k_active=50, eps_frac=0.02, lr_scale=5.0)
        if c5_nest < best_c5:
            best_c5 = c5_nest
            best_h = h_nest.copy()

    # Phase 3: SLSQP polishing with increasing alpha
    if time_left() > 80:
        alphas = [30.0, 80.0, 200.0, 500.0, 1000.0]
        for alpha in alphas:
            if time_left() < 50:
                break
            maxiter = max(100, min(800, int(time_left() * 0.8)))
            h_opt = slsqp_optimize(best_h, n, target_sum, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Coordinate descent fine-tuning
    if time_left() > 40:
        h_cd, c5_cd = coordinate_descent_fine(best_h, n, target_sum,
                                              min(120, time_left() - 20), rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Final SLSQP at highest alpha
    if time_left() > 30:
        h_opt = slsqp_optimize(best_h, n, target_sum, 2000.0, 500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Try higher resolution if time allows
    if time_left() > 120:
        n_high = 480
        h_high, c5_high = try_higher_resolution(best_h, n, n_high,
                                                 min(100, time_left() - 30), rng)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = n_high

    # Phase 7: Another round of momentum on the best
    if time_left() > 60:
        n_final_try = len(best_h)
        ts_final_try = n_final_try / 2.0
        h_mom2, c5_mom2 = momentum_active_set(best_h, n_final_try, ts_final_try,
                                               min(50, time_left() - 20),
                                               lr_init=0.02, k_active=80, eps_frac=0.01)
        if c5_mom2 < best_c5:
            best_c5 = c5_mom2
            best_h = h_mom2.copy()

    # Final validation
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
