    t_grid = np.linspace(0.0, 1.0, 201)
    f = np.max(a[:, None] * t_grid[None, :] + b[:, None], axis=0)
    idx = int(np.argmin(f))
    t_star = t_grid[idx]
    
    if 0 < idx < 200:
        t_fine = np.linspace(t_grid[idx-1], t_grid[idx+1], 101)
        f_f = np.max(a[:, None] * t_fine[None, :] + b[:, None], axis=0)
        if f_f.min() < f[idx]:
            t_star = t_fine[int(np.argmin(f_f))]
    
    return t_star

def local_search_cd(h, n, n_sweeps, rng=None):
    """Exact coordinate descent."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    max_no_improve = 5
    
    for sweep in range(n_sweeps):
        if no_improve >= max_no_improve:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for j in order:
            t_star = coord_step_fast(h, j, n)
            if abs(t_star - h[j]) > 1e-10:
                h[j] = t_star
                h = project_feasible(h, n)
                improved = True
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5
