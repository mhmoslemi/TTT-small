    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shift = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shift))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + max_shift + np.log(np.sum(exp_vals)) / alpha) * (2.0 / n)
    v = 1.0 - 2.0 * h
    grad = np.correlate(w, v, mode="valid") * (2.0 / n)
    return obj, grad

        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)

                        except Exception:
                            pass

    # Phase 4: shake and re-optimize
    if time_left() > 80:
        n_sh = best_n
        for trial in range(20):
            if time_left() < 30:
                break
            noise_scale = 0.2 * (1.0 - trial / 20.0) + 0.02
            h_shake = best_h + rng.uniform(-noise_scale, noise_scale, n_sh)
            h_shake = project_box_sum(h_shake, n_sh / 2.0)
            for alpha in [50, 300, 1500]:
                if time_left() < 15:
                    break
                h_opt, _ = projected_gradient_descent(h_shake, n_sh, alpha, 2000, 0.05, time_left)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_shake = h_opt

    # Phase 5: High-resolution upsample + optimize
    if time_left() > 120:
        n_up = min(480, max(360, best_n * 2))
        if n_up % 2 != 0:
            n_up += 1
        ts_up = n_up / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_up = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old, best_h, period=2.0)
        h_up = project_box_sum(h_up, ts_up)
        
        for alpha in [50, 200, 1000, 5000]:
            if time_left() < 30:
                break
            h_opt, _ = projected_gradient_descent(h_up, n_up, alpha, 3000, 0.03, time_left)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_up

    # Phase 6: Final fine-tuning
    if time_left() > 30:
        n_fin = best_n
        for alpha in [5000, 20000, 100000]:
            if time_left() < 15:
                break
            h_fin, _ = projected_gradient_descent(best_h, n_fin, alpha, 5000, 0.01, time_left)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
