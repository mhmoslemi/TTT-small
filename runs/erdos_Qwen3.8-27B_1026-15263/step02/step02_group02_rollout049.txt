```python
def smooth_gradient(h, beta, dx):
    n = len(h)
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    
    # Softmax weights
    shifted = beta * corr - beta * corr.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient: for each j, sum_s w_s * [(1-h[j+s]) - h[j-s]] with bounds
    # shifts range from -(n-1) to (n-1), indexed as s_idx - (n-1)
    n_corrs = len(corr)  # 2n-1
    grad = np.zeros(n)
    for s_idx in range(n_corrs):
        s = s_idx - (n - 1)  # shift value
        w = weights[s_idx]
        if s >= 0:
            # (1-h[j+s]) for j where j+s < n, i.e., j < n-s
            valid_j = np.arange(0, n - s)
            grad[valid_j] += w * (1.0 - h[valid_j + s])
            # -h[j-s] for j where j-s >= 0, i.e., j >= s (but s>=0 and j<n always)
            # Actually for s >= 0: j-s >= 0 means j >= s
            valid_j2 = np.arange(s, n)
            grad[valid_j2] -= w * h[valid_j2 - s]
        else:
            t = -s  # positive
            # (1-h[j+s]) = (1-h[j-t]) for j where j-t >= 0, i.e., j >= t
            valid_j = np.arange(t, n)
            grad[valid_j] += w * (1.0 - h[valid_j - t])
            # -h[j-s] = -h[j+t] for j where j+t < n, i.e., j < n-t
            valid_j2 = np.arange(0, n - t)
            grad[valid_j2] -= w * h[valid_j2 + t]
    
    return dx * grad
```

```python
def project_onto_simplex_box(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)

        s = projected.sum()
        if s > target_sum:

        else:
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def c5_smooth_obj_grad(h, beta, dx, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = corr.max()
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    logsum = np.log(exp_vals.sum())
    obj = dx * (c_max + logsum / beta) + penalty * (h.sum() - n / 2.0) ** 2
    weights = exp_vals / exp_vals.sum()
    active = np.where(weights > 1e-9)[0]
    grad_noise = np.zeros(n)
    for idx in active:
        s = int(idx) - (n - 1)
        w = weights[idx]
        if s >= 0:
            grad_noise[:n - s] += w * (1.0 - h[s:])
            if s > 0:
                grad_noise[s:] -= w * h[:n - s]
        else:
            t = -s
            grad_noise[t:] += w * (1.0 - h[:n - t])
            grad_noise[:n - t] -= w * h[t:]
    grad = dx * grad_noise + 2.0 * penalty * (h.sum() - n / 2.0)
    return obj, grad


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_obj(x):
    return x[-1]


def slsqp_optimize(h_init, maxiter=3000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def lgbfgs_smooth(h_init, beta, penalty, maxiter=2000):
    n = len(h_init)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_grad(x):
        return c5_smooth_obj_grad(x, beta, dx, penalty)

    result = minimize(obj_grad, h_init, jac=True, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                               'maxcor': 30, 'maxls': 60})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def exact_subgrad_step(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    s_star = int(np.argmax(corr))
    s = s_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n - s] += (1.0 - h[s:])
        if s > 0:
            grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[t:] += (1.0 - h[:n - t])
        grad[:n - t] -= h[t:]
    return dx * grad


def nesterov_subgrad(h_init, steps=500, lr0=0.015):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)

    for k in range(steps):
        grad = exact_subgrad_step(y, dx)
        momentum = 2.0 / (k + 3.0)
        lr = lr0 / (np.sqrt(k + 1.0))
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        y = h_new + momentum * (h_new - h)
        y = project_feasible(y, n)
        h = h_new

        c5_h = compute_c5_from_h(h)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()
        c5_y = compute_c5_from_h(y)
        if c5_y < best_c5:
            best_c5 = c5_y
            best_h = y.copy()

    return best_h, best_c5


def upsample(h, target_n):
    n = len(h)
    if target_n == n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    f_interp = interp1d(x_old, h, kind='cubic', fill_value="extrapolate", bounds_error=False)
    h_new = np.clip(f_interp(x_new), 0.0, 1.0)
    return project_feasible(h_new, target_n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    best_n = n

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP on parent
    if remaining() > 400:
        best_h, best_c5 = slsqp_optimize(best_h, maxiter=5000, t_margin=0.08)
        best_n = len(best_h)

    # Phase 2: Upsample to 540 and optimize with smooth gradient
    if remaining() > 300:
        h_540 = upsample(best_h, 540)
        cur_h = h_540
        cur_c5 = compute_c5_from_h(cur_h)

        for beta in [20.0, 50.0, 100.0, 300.0, 800.0]:
            if remaining() < 120:
                break
            maxiter = 1500
            h_new, c5_new = lgbfgs_smooth(cur_h, beta=beta, penalty=200.0, maxiter=maxiter)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = 540
            cur_h = h_new
            cur_c5 = c5_new

        if remaining() > 150 and cur_c5 < best_c5 + 0.001:
            h_new, c5_new = slsqp_optimize(cur_h, maxiter=3000, t_margin=0.04)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)
            cur_h = h_new

    # Phase 3: Try n=1080
    if remaining() > 350:
        h_1080 = upsample(cur_h, 1080)
        cur_h = h_1080

        for beta in [50.0, 200.0, 600.0, 2000.0]:
            if remaining() < 150:
                break
            h_new, c5_new = lgbfgs_smooth(cur_h, beta=beta, penalty=500.0, maxiter=1200)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = 1080
            cur_h = h_new

        if remaining() > 120:
            h_new, c5_new = slsqp_optimize(cur_h, maxiter=2500, t_margin=0.03)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)

    # Phase 4: Nesterov on the exact objective at current best
    if remaining() > 80:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        steps = min(1500, int(remaining() * 2))
        h_new, c5_new = nesterov_subgrad(best_h, steps=steps, lr0=0.012)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = len(best_h)

    # Phase 5: Perturbation + local refinement
    if remaining() > 60:
        n_cur = len(best_h)
        for attempt in range(15):
            if remaining() < 25:
                break
            scale = 0.005 + 0.02 * (1.0 - attempt / 15.0) ** 1.5
            h_pert = best_h + scale * rng.standard_normal(n_cur)
            h_pert = project_feasible(h_pert, n_cur)
            if remaining() > 40:
                h_new, c5_new = slsqp_optimize(h_pert, maxiter=1500, t_margin=0.02)
            else:
                h_new, c5_new = h_pert, compute_c5_from_h(h_pert)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)

    # Phase 6: Final polish
    if remaining() > 20:
        maxiter = min(6000, int(remaining() * 2))
        h_new, c5_new = slsqp_optimize(best_h, maxiter=maxiter, t_margin=0.01)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
```
