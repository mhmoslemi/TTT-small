4. Flip and repeat

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        nmask = np.sum(mask)
        if nmask > 0:
            h[mask] += diff / nmask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -100, 100)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    obj = m + np.log(np.sum(exp_s)) / alpha
    
    j = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    shift = m_idx - (n - 1)
    
    fp = j - shift
    vf = (fp >= 0) & (fp < n)
    fc = np.clip(fp, 0, n - 1)
    
    sp = j + shift
    vs = (sp >= 0) & (sp < n)
    sc = np.clip(sp, 0, n - 1)
    
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, target_sum, alpha, maxiter, penalty):
    def f(x):
        x_c = np.clip(x, 0, 1)
        obj, _ = smooth_max_obj_grad(x_c, alpha)
        obj += penalty * (np.sum(x_c) - target_sum) ** 2
        return obj
    def g(x):
        x_c = np.clip(x, 0, 1)
        _, grad = smooth_max_obj_grad(x_c, alpha)
        grad += 2 * penalty * (np.sum(x_c) - target_sum) * np.ones(n)
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_opt(h0, n, target_sum, alpha, maxiter, penalty):
    def f(x):
        x_c = np.clip(x, 0, 1)
        obj, _ = smooth_max_obj_grad(x_c, alpha)
        obj += penalty * (np.sum(x_c) - target_sum) ** 2
        return obj
    def g(x):
        x_c = np.clip(x, 0, 1)
        _, grad = smooth_max_obj_grad(x_c, alpha)
        grad += 2 * penalty * (np.sum(x_c) - target_sum) * np.ones(n)
        return grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds, maxiter=maxiter,
                       options={'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_gradient_flip(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        corr = compute_corr(h)
        dx = 2.0 / n
        critical_lags = np.where(corr > np.max(corr) - dx * 0.001)[0]
        best_improvement = 0
        best_pair = None
        for m_idx in critical_lags:
            s = m_idx - (n - 1)
            j = np.arange(n)
            fp = j - s
            vf = (fp >= 0) & (fp < n)
            fc = np.clip(fp, 0, n - 1)
            sp = j + s
            vs = (sp >= 0) & (sp < n)
            sc = np.clip(sp, 0, n - 1)
            grad = vf * (1.0 - h[fc]) - vs * h[sc]
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            pos_ones = ones[grad[ones] > 0.01]
            neg_zeros = zeros[grad[zeros] < -0.01]
            if len(pos_ones) == 0 or len(neg_zeros) == 0:
                continue
            best_g = 0
            for i in pos_ones[:50]:
                for j2 in neg_zeros[:50]:
                    improvement = grad[i] + (-grad[j2])
                    if improvement > best_g:
                        best_g = improvement
                        if best_pair is None or best_g > best_improvement:
                            best_improvement = best_g
                            best_pair = (i, j2)
        if best_pair is None:
            break
        i, j2 = best_pair
        h[i] = 0.0
        h[j2] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 >= c5_cur - 1e-12:
            h[i] = 1.0
            h[j2] = 0.0
            break
        c5_cur = new_c5
    return h, c5_cur

def fft_structured_start(n, target_sum, mode_idx, amplitude, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    n_modes = min(mode_idx, n // 2)
    for k in range(1, n_modes + 1):
        a = amplitude / (k ** 0.7) * rng.choice([-1, 1])
        phi = rng.uniform(0, 2 * np.pi)
        h += a * np.sin(2 * np.pi * k * x / 2.0 + phi)
    return project_to_sum(h, target_sum)

def make_diverse_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    
    starts.append(np.full(n, 0.5))

            s = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
            starts.append(project_to_sum(s, target_sum))
    
    for nblks in [2, 3, 4, 5, 6, 8, 10, 12]:
        blk = max(1, n // nblks)
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    
    for _ in range(6):
        starts.append(fft_structured_start(n, target_sum, 
                                            int(rng.randint(2, min(10, n//4))), 
                                            float(rng.uniform(0.15, 0.4)), rng))
    
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), target_sum))
    
    hb = (rng.rand(n) > 0.5).astype(float)
    attempts = 0
    while abs(hb.sum() - target_sum) > 1e-10 and attempts < 20000:
        if hb.sum() > target_sum:

            hb[idx] = 0.0
        else:
            idx = rng.choice(np.where(hb < 1)[0])
            hb[idx] = 1.0
        attempts += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())
    
    return starts

def multi_alpha_optimize(h0, n, target_sum, alpha_list, method='slsqp', maxiter=1000):
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    for alpha in alpha_list:
        if method == 'slsqp':
            h_new = slsqp_opt(h_best, n, target_sum, alpha, maxiter, penalty=0)
        else:
            h_new = lbfgsb_opt(h_best, n, target_sum, alpha, maxiter, penalty=1000.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        else:
            h_best = h_new.copy()
    return h_best, c5_best
