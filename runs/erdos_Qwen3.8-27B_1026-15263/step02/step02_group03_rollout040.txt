def evaluate_c5(h):

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                step = min(diff, 1.0 - result[idx])
            else:
                step = min(-diff, result[idx])
            result[idx] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-14:
                break
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def slsqp_objective(x, n, tau):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, tau) * dx

    return np.sum(x) - n / 2.0

def grad_c_k(h, n, k):
    lag = k - (n - 1)
    grad = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    for j in range(j_lo, j_hi + 1):
        grad[j] += (1.0 - h[j - lag])
        grad[j - lag] -= h[j]
    return grad

def compute_active_gradient(h, n, num_active=5):
    corrs = get_corrs(h)
    max_val = np.max(corrs)
    threshold = max_val * 0.95
    active = np.where(corrs >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    if len(active) > num_active:
        sorted_idx = active[np.argsort(corrs[active])[::-1]][:num_active]
    else:
        sorted_idx = active
    grad = np.zeros(n)
    for k in sorted_idx:
        grad += grad_c_k(h, n, int(k))
    grad /= max(1, len(sorted_idx))
    return grad, corrs, sorted_idx
