import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.05):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.0001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None and result.fun >= 0:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter, t_margin):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.0001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 0.02:
            return h_out, c5_out
    return h_init, c5_init

def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    if c_max < -1000:
        return float(c_max)
    shifted = np.clip(T * (c - c_max), -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

def smooth_obj(x, T, penalty):
    n = len(x)
    c5_val = c5_logsumexp(x, T)
    sum_viol = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_viol * sum_viol

def lbfgsb_optimize(h_init, n, T, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

from scipy.optimize import minimize

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize(h_init, n, maxiter, t_margin):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.0001, c5_init * 0.3), 2.0)]
    constraints = [
        {"type": "ineq", "fun": slsqp_ineq},
        {"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac},
    ]
    result = minimize(
        slsqp_obj, x0, method="SLSQP", bounds=bounds,
        constraints=constraints,
        options={"maxiter": maxiter, "ftol": 1e-16},
    )
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 0.02:
            return h_out, c5_out
    return h_init, c5_init


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -1000:
        return float(c_max)
    shifted = np.clip(T * (c - c_max), -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def smooth_obj(x, T, penalty):
    n = len(x)
    c5_val = c5_logsumexp(x, T)
    sv = np.sum(x) - n / 2.0
    return c5_val + penalty * sv * sv


def smooth_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    if c_max < -1000:
        return np.zeros(n)
    shifted = np.clip(T * (c - c_max), -400, 400)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = i - lo
            kval = k - n + 1 + j
            gc = -np.correlate(x, 1.0 - x, mode="full")
            grad[i] += w * gc[k] * 0
    sv = np.sum(x) - n / 2.0
    grad = grad + 2.0 * penalty * sv
    return grad

    fine_n = 12
    for it in range(fine_n):
        if budget_s - (time.time() - t0) < 35:
            break
        scale = 0.002 * (1 - it / fine_n) + 0.0003
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        h_new, c5_new = slsqp_optimize(h_pert, n_best, 3000, 0.005)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            n_best = len(best_h)

    if budget_s - (time.time() - t0) > 18:
        best_h, best_c5 = slsqp_optimize(best_h, n_best, 8000, 0.002)
        n_best = len(best_h)

    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, n_best
