def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):

        else:

    """Projected subgradient method with the active lag."""

    return best, best_c5

def alternating_reweight(h, n, max_iter, time_limit=None):
    current = h.copy()
    best = current.copy()
    best_c5 = evaluate_c5(best)
    dx = 2.0 / n
    tau_schedule = [0.05, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0008]
    tau_idx = 0
    for t in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break
        g = 1.0 - current
        corr = np.correlate(current, g, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best = current.copy()
        if tau_idx < len(tau_schedule) and t % max(1, max_iter // len(tau_schedule)) == 0:
            tau = tau_schedule[tau_idx]
            tau_idx += 1
        else:
            tau = max(0.0005, tau_schedule[min(tau_idx, len(tau_schedule) - 1)] * 0.85)
        v = corr / tau
        vmax = np.max(v)
        exp_v = np.exp(v - vmax)
        w = exp_v / np.sum(exp_v)
        current = slsqp_weighted(current, n, w, dx, max_iter=60)
    return best, best_c5

def generate_starts(n, parent, rng, num_starts=10):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
            starts.append(project_feasible(parent + rng.randn(n) * 0.05, n))
            starts.append(project_feasible(parent + rng.randn(n) * 0.15, n))
        else:
            xp = np.linspace(0, 2, len(parent), endpoint=False)
            f = interp1d(xp, parent, kind='cubic', fill_value='extrapolate', bounds_error=False)
            starts.append(project_feasible(f(x), n))
            starts.append(project_feasible(f(x) + rng.randn(n) * 0.08, n))
    for _ in range(3):
        h = rng.rand(n)
        starts.append(project_feasible(h, n))
    for freq in [2.0, 3.0, 4.0]:
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
        starts.append(project_feasible(h, n))
    for _ in range(2):
        h = 0.5 + 0.3 * rng.randn(n)
        starts.append(project_feasible(h, n))
    h_bin = (rng.rand(n) > 0.5).astype(float)
    starts.append(project_feasible(h_bin, n))
    h_bin2 = (rng.rand(n) > 0.4).astype(float)
    starts.append(project_feasible(h_bin2, n))
    return starts[:num_starts]

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        best_h = project_feasible(parent.copy(), n)
        best_c5 = evaluate_c5(best_h)
    else:
        n = 77
        best_h = np.full(n, 0.5)
        best_c5 = evaluate_c5(best_h)

    phase1_end = start_time + budget_s * 0.15
    phase2_end = start_time + budget_s * 0.70
    phase3_end = start_time + budget_s * 0.88
    phase4_end = start_time + budget_s * 0.99

    # Phase 1: Quick SLSQP from parent and a few random starts
    starts = generate_starts(n, parent, rng, num_starts=6)
    for h0 in starts:
        if time.time() > phase1_end:
            break
        for tau in [0.02, 0.008]:
            h_new = slsqp_smooth(h0, n, tau, max_iter=150)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 2: Alternating reweighting (main optimization)
    h_alt, c5_alt = alternating_reweight(best_h, n, max_iter=300, time_limit=phase2_end)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = h_alt.copy()

    # Phase 2b: Subgradient refinement
    if time.time() < phase2_end:
        for attempt in range(15):
            if time.time() > phase2_end:
                break
            noise = 0.005 * max(0.0, 1.0 - (time.time() - start_time) / (budget_s * 0.7))
            perturbed = project_feasible(best_h + rng.randn(n) * noise, n)
            h_sub, c5_sub = subgradient_method(perturbed, n, max_iter=800,
                                               step_scale=0.03, time_limit=phase2_end)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
            # Quick SLSQP polish after subgradient
            h_pol = slsqp_smooth(best_h, n, tau=0.003, max_iter=100)
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 3: Coordinate descent
    if time.time() < phase3_end:
        h_cd, c5_cd = coordinate_descent_fine(best_h, n, sweeps=4, time_limit=phase3_end)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Final polish - multiple SLSQP with decreasing tau + subgradient
    if time.time() < phase4_end:
        tau_list = [0.002, 0.001, 0.0008, 0.0005, 0.0003, 0.0002]
        for tau in tau_list:
            if time.time() > phase4_end:
                break
            h_new = slsqp_smooth(best_h, n, tau, max_iter=300)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            # Subgradient between SLSQP calls
            if time.time() < phase4_end:
                h_sg, c5_sg = subgradient_method(best_h, n, max_iter=500,
                                                 step_scale=0.01, time_limit=phase4_end)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()

    # Final coordinate descent
    if time.time() < start_time + budget_s * 0.995:
        h_cd2, c5_cd2 = coordinate_descent_fine(best_h, n, sweeps=2,
                                                 time_limit=start_time + budget_s * 0.995)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
