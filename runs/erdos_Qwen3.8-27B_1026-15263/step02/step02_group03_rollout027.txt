    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def objective_func(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def constraint_func(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = s - target - 1.0, s - target + 1.0
    # We want to find shift lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Adjust the element closest to 0.5
        idx = int(np.argmin(np.abs(result - 0.5)))

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return float(tau * (vmax + np.log(np.sum(np.exp(v - vmax)))))

    """
    Compute the correlation vector and the gradient of its maximum.
    Returns (corr_values, gradient, max_index).
    """
    dx = 2.0 / n
    g = 1.0 - h

    max_idx = int(np.argmax(corr))
    
    # Gradient of the max (subgradient using the argmax)
    k = max_idx - (n - 1)
    grad = np.zeros(n)

        grad[mask1] += (1.0 - h[i[mask1] - m])
        grad[mask2] -= h[i[mask2] + m]
    
    grad *= dx
    return corr, grad, max_idx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return float(tau * (vmax + np.log(np.sum(np.exp(v - vmax)))))

def objective_slsqp(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def constraint_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def active_set_subgradient(h, n, tol=1e-6):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        grad += f_lag_grad(h, k, n)
    grad /= len(active)
    return grad, max_val * dx, active

def coordinate_descent(h0, n, max_time):
    start_time = time.time()
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    sweeps = 0
    no_improve = 0
    while time.time() - start_time < max_time:
        sweeps += 1
        improved = False
        for i in range(n):
            if time.time() - start_time > max_time:
                break
            for d in deltas:
                new_val = best[i] + d
                if new_val < 0.0 or new_val > 1.0:
                    continue
                trial = best.copy()
                trial[i] = new_val
                trial = project_feasible(trial, n)
                c5 = evaluate_c5(trial)
                if c5 < best_c5 - 1e-15:
                    best = trial
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 2:
                deltas = [d * 0.5 for d in deltas]
                no_improve = 0
            if abs(deltas[0]) < 1e-7:
                break
    return best, best_c5

def binarize_push(h, n):
    target = n / 2.0
    h_sorted_idx = np.argsort(h)
    new_h = np.zeros(n)
    num_ones = int(np.floor(target))
    for i in range(num_ones):
        new_h[h_sorted_idx[n - 1 - i]] = 1.0
    rem = target - np.sum(new_h)
    if abs(rem) > 1e-12:
        candidate = h_sorted_idx[num_ones]
        new_h[candidate] = min(1.0, max(0.0, rem))
    return new_h

    if time.time() < phase3_end:
        remaining = phase3_end - time.time()
        h_res, c5_res = nesterov_pursuit_hybrid(best_h, n, max_time=remaining)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

        # Also try pure pursuit from best
        if time.time() < phase3_end:
            h_res2, c5_res2 = pursue_method(best_h, n, max_time=min(60.0, phase3_end - time.time()))
            if c5_res2 < best_c5:
                best_c5 = c5_res2
                best_h = h_res2.copy()

    # Phase 3b: Binary push + re-optimize
    if time.time() < phase3_end:
        for _ in range(3):
            if time.time() > phase3_end:
                break
            h_bin = binarize_push(best_h, n)
            h_res, c5_res = nesterov_exact_max(
                h_bin, n, max_time=8.0, subgrad_tol=1e-8
            )
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    # Phase 4: Coordinate descent polish
    if time.time() < phase4_end:
        remaining = phase4_end - time.time()
        h_coord, c5_coord = coordinate_descent(best_h, n, max_time=remaining)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    # Phase 5: Final Nesterov micro-optimization
    final_end = start_time + budget_s * 0.99
    if time.time() < final_end:
        remaining = final_end - time.time()
        h_res, c5_res = nesterov_exact_max(
            best_h, n, max_time=remaining,
            init_alpha=float(n)/40.0, subgrad_tol=1e-10
        )
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
