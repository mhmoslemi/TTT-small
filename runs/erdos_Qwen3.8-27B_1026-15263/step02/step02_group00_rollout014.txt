def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    sz = np.max(shifted)
    ev = np.exp(shifted - sz)
    se = np.sum(ev)
    w = ev / se
    obj = (m_val + sz / alpha + np.log(se) / alpha) * dx
    neg_h = 1.0 - h
    grad = np.zeros(n)
    scale = dx / alpha
    for k in range(2 * n - 1):
        wk = w[k] * scale
        if wk < 1e-18:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n - 1, 2 * n - 2 - k)
        if lo <= hi:
            grad[lo:hi + 1] += wk * neg_h[lo + k - n + 1:hi + k - n + 2]
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wk * h[lo2 - k + n - 1:hi2 - k + n + 1]
    return obj, grad

        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(500, len(ones) * len(zeros))
        best_delta = 0
        best_pair = None
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_test = compute_c5(h)
            delta = c5_cur - c5_test
            h[i] = 1.0
            h[j] = 0.0
            if delta > best_delta:
                best_delta = delta
                best_pair = (i, j)
        if best_pair is not None and best_delta > 1e-12:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5_cur = compute_c5(h)
            improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def tl():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Alternating LP from parent
    n1 = best_n
    ts1 = n1 / 2.0
    h_lp, c5_lp = alternating_lp_solve(best_h, n1, max_rounds=300, time_budget=tl() * 0.3)
    if c5_lp < best_c5 - 1e-10:
        best_c5 = c5_lp
        best_h = h_lp.copy()
        best_n = n1

    # Phase 2: Continuation L-BFGS-B polish
    if tl() > 60:
        h_cont = continuation_opt(best_h, best_n, best_n / 2.0,
                                  alphas=(20, 60, 200, 600, 2000, 6000), maxiter=800)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5 - 1e-10:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 3: Multi-resolution alternating LP
    for n_tr in [120, 180, 200, 240, 300, 360, 420]:
        if tl() < 50:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 0.5:
            continue
        h_interp = project_to_sum(interpolate_h(best_h, best_n, n_tr), ts_tr)
        h_lp2, c5_lp2 = alternating_lp_solve(h_interp, n_tr, max_rounds=200,
                                             time_budget=min(40.0, tl() * 0.15))
        if c5_lp2 < best_c5 - 1e-10:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()
            best_n = n_tr

    # Phase 4: Random restarts + alternating LP
    for trial in range(20):
        if tl() < 30:
            break
        n_tr = best_n
        ts_tr = n_tr / 2.0
        if trial % 3 == 0:
            h_st = make_binary_start(n_tr, rng)
        elif trial % 3 == 1:
            h_st = best_h + rng.uniform(-0.15, 0.15, n_tr)
            h_st = project_to_sum(h_st, ts_tr)
        else:
            x = np.linspace(0, 2, n_tr, endpoint=False)
            h_st = 0.5 + 0.35 * np.sin(2 * np.pi * rng.randint(1, 6) * x / 2.0 + rng.uniform(0, 2 * np.pi))
            h_st = project_to_sum(h_st, ts_tr)
        h_st = project_to_sum(h_st, ts_tr)
        h_new, c5_new = alternating_lp_solve(h_st, n_tr, max_rounds=150,
                                             time_budget=min(25.0, tl() * 0.1))
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_tr

    # Phase 5: Binary local search
    if tl() > 40:
        n5 = best_n
        ts5 = n5 / 2.0
        h_bin = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin.sum() - ts5) > 1e-10 and attempts < 10000:
            if h_bin.sum() > ts5:
                idx = rng.choice(np.where(h_bin > 0)[0])
                h_bin[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin < 1)[0])
                h_bin[idx] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts5) < 1e-10:
            h_bf, c5_bf = binary_local_search(h_bin, n5, ts5, rng, max_rounds=50,
                                              time_budget=min(30.0, tl() * 0.2))
            if c5_bf < best_c5 - 1e-10:
                best_c5 = c5_bf
                best_h = h_bf.copy()
            h_sm = lbfgsb_opt(h_bf, n5, ts5, 500, 800)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5 - 1e-10:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 6: Active-lag focused optimization
    if tl() > 40:
        n6 = best_n
        ts6 = n6 / 2.0
        h6 = best_h.copy()
        for itera in range(8):
            if tl() < 15:
                break
            dx6 = 2.0 / n6
            corr6 = np.correlate(h6, 1.0 - h6, mode='full')
            m6 = np.max(corr6)
            thresh = m6 - 0.003
            active = corr6 > thresh
            # Optimize sum of active correlations via L-BFGS-B
            h6 = lbfgsb_opt(h6, n6, ts6, 200, 400)
            c5_6 = compute_c5(h6)
            if c5_6 < best_c5 - 1e-10:
                best_c5 = c5_6
                best_h = h6.copy()
                best_n = n6

    # Phase 7: Final high-alpha polish
    if tl() > 25:
        for alpha in [3000, 10000, 30000]:
            if tl() < 15:
                break
            h_pol = lbfgsb_opt(best_h, best_n, best_n / 2.0, alpha, 1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5 - 1e-10:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
