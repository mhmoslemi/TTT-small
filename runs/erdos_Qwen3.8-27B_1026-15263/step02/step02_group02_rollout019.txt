def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            step = min(step, (1.0 - h[mask]).min() * 0.999)
            h[mask] += step * 0.5  # damped
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask)
            step = min(step, h[mask].min() * 0.999)
            h[mask] -= step * 0.5  # damped
    return np.clip(h, 0.0, 1.0)

def subgradient_c5(h, active_indices, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active_indices:
        k = m - n + 1
        # For each i, contribution is (1-h[i+k]) - h[i-k] (with 0 for OOB)
        # h[i+k] for i in range(n): this is h shifted
        # Let's build it vectorially
        if k >= 0:
            # h[i+k] is valid for i in [0, n-1-k]
            # h[i-k] is valid for i in [k, n-1]
            grad[:n-k] += (1.0 - h[k:])  # (1-h[i+k]) for i in [0, n-1-k]
            grad[k:] -= h[:n-k]  # -h[i-k] for i in [k, n-1]
        else:
            k_abs = -k
            # h[i+k] = h[i-k_abs] is valid for i in [k_abs, n-1]
            # h[i-k] = h[i+k_abs] is valid for i in [0, n-1-k_abs]
            grad[k_abs:] += (1.0 - h[:n-k_abs])  # (1-h[i-k_abs]) for i in [k_abs, n-1]
            grad[:n-k_abs] -= h[k_abs:]  # -h[i+k_abs] for i in [0, n-1-k_abs]
    return grad * dx / len(active_indices)

    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            max_step = (1.0 - h[mask]).min()
            step = min(step, max_step)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask)
            max_step = h[mask].min()
            step = min(step, max_step)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active_indices:
        k = m - n + 1
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            ka = -k
            grad[ka:] += (1.0 - h[:n-ka])
            grad[:n-ka] -= h[ka:]
    return grad * dx / len(active_indices)

    """SLSQP with only active constraints."""
    corr = compute_correlations(h_init)
    c_max = corr.max()
    
    # Find active constraints
    active = np.where(corr > c_max - active_tol)[0]
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = compute_correlations(h)
        return t - c[active]
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    def eq_jac(x):
        jac = np.zeros(len(x))
        jac[:-1] = 1.0
        return jac
    
    x0 = np.concatenate([h_init, [c_max + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(c_max * 0.5, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    
    if result.x is not None and result.success:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out), corr
    return h_init, compute_c5_from_h(h_init), corr

            best_c5 = c_max

        active = np.where(corr >= c_max - active_tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute subgradient
        grad = subgradient_c5(h, active, n)
        
        # Skip zero gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        
        # Normalized gradient step with backtracking
        alpha = step
        for _ in range(30):
            h_trial = h - alpha * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5_from_h(h_trial)
            if c5_trial <= c_max + 1e-12:
                break
            alpha *= 0.5
        
        h = h_trial
        # Adapt step size
        step = alpha * 1.05
        
        # Periodic polish
        if (it + 1) % poll_interval == 0:
            h_pol, c5_pol, _ = slsqp_active(h, n, active_tol=active_tol, maxiter=poll_maxiter)
            if c5_pol < best_c5 - 1e-10:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h = h_pol.copy()
    
    return best_h, best_c5
