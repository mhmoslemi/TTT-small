import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            if room.min() <= 0:
                break
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            if room.min() <= 0:
                break
            step = min(-diff / np.sum(mask), room.min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def c5_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -1000:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def smooth_obj(x, T, penalty):
    n = len(x)
    c5_val = c5_smooth(x, T)
    sum_violation = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_violation ** 2


def lbfgsb_optimize(h_init, n, T, penalty, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-15}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def fourier_to_h(params, n):
    K = len(params) // 2
    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = params[2 * (k - 1)]
        b = params[2 * (k - 1) + 1]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return h


def de_objective(params, n):
    h = fourier_to_h(params, n)
    h = project_feasible(h, n)
    return compute_c5_from_h(h)

        for k in range(1, K + 1):

            b_k = 2.0 * np.sum((h_init_from_parent - a0) * np.sin(np.pi * k * x)) / n
            coeffs.extend([a_k, b_k])
        x0 = np.array(coeffs)

def block_coordinate_optimize(h_init, n, n_blocks=10, max_rounds=200, t_margin=0.05):
    h = h_init.copy()
    best_c5 = compute_c5_from_h(h)
    block_size = n // n_blocks
    rng = np.random.default_rng(12345)
    for rnd in range(max_rounds):
        order = rng.permutation(n_blocks)
        for b in order:
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            h_block = h[start:end].copy()
            h_frozen = h.copy()
            sub_n = end - start
            x0 = np.concatenate([h_block, [compute_c5_from_h(h) + t_margin]])
            bounds_sub = [(0.0, 1.0)] * sub_n + [(max(0.001, best_c5 * 0.3), 2.0)]
            
            def sub_ineq(x):
                hs = x[:-1]
                t = x[-1]
                h_full = h_frozen.copy()
                h_full[start:end] = hs
                dd = 2.0 / n
                c = np.correlate(h_full, 1.0 - h_full, mode="full")
                return t - c * dd
            
            def sub_eq(x):
                hs = x[:-1]
                h_full = h_frozen.copy()
                h_full[start:end] = hs
                return np.sum(h_full) - n / 2.0
            
            def sub_eq_jac(x):
                jac = np.zeros(len(x))
                jac[:sub_n] = 1.0
                return jac
            
            cons = [
                {'type': 'ineq', 'fun': sub_ineq},
                {'type': 'eq', 'fun': sub_eq, 'jac': sub_eq_jac}
            ]
            res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds_sub,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None:
                h[start:end] = project_feasible(np.concatenate([h[start:end], np.zeros(0)]), 1).shape[0] and res.x[:-1]
                h[start:end] = np.clip(res.x[:-1], 0, 1)
                h = project_feasible(h, n)
                c5_cur = compute_c5_from_h(h)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
        if best_c5 < compute_c5_from_h(h_init) - 0.01:
            break
    h = project_feasible(h, n)
    return h, compute_c5_from_h(h)
