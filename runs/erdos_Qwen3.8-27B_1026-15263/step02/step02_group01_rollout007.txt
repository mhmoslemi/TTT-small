    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_and_grad(h, alpha)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return h_best, c5_cur

def binary_push(h0, n, target_sum, strength, n_iter):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for it in range(n_iter):
        h_pulled = h + strength * (2.0 * (h - 0.5))
        h_pulled = np.clip(h_pulled, 0.0, 1.0)
        h_pulled = project_to_sum(h_pulled, target_sum)
        c5_new = compute_c5(h_pulled)
        if c5_new < c5_cur:
            h = h_pulled
            c5_cur = c5_new
            strength *= 0.9
        else:
            h_coord = h + strength * 0.01 * np.sign(2.0 * h - 1.0)
            h_coord = np.clip(h_coord, 0.0, 1.0)
            h_coord = project_to_sum(h_coord, target_sum)
            c5_coord = compute_c5(h_coord)
            if c5_coord < c5_cur:
                h = h_coord
                c5_cur = c5_coord
            strength *= 0.85
    return h, c5_cur

def coordinate_descent_fine(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def pairwise_exchange(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = h[i] - h[j]
            if abs(delta) < 1e-8:
                continue
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] - delta * 0.5, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] + delta * 0.5, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_resolutions = [360, 540, 720, 900]
    time_per_n = [250, 200, 400, 250]

    for res_idx, n in enumerate(n_resolutions):
        if time_left() < time_per_n[res_idx]:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0

        h_cur = None
        if best_n == n:
            h_cur = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_cur = project_to_sum(h_interp, ts)
            c5_interp = compute_c5(h_cur)
            if c5_interp < best_c5:
                best_c5 = c5_interp
                best_h = h_cur.copy()
                best_n = n

        alphas_stage1 = [5.0, 15.0, 40.0, 100.0]
        for alpha in alphas_stage1:
            if time_left() < 30:
                break
            maxiter = max(50, min(500, int(time_left() * 0.8)))
            h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            h_cur = h_opt

        alphas_stage2 = [200.0, 500.0, 1000.0, 2000.0]
        for alpha in alphas_stage2:
            if time_left() < 25:
                break
            maxiter = max(50, min(300, int(time_left() * 0.5)))
            h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            h_cur = h_opt

        if time_left() > 40:
            c5_check, active = compute_c5_and_active_lags(h_cur)
            if len(active) > 0 and len(active) < n:
                for alpha_a in [500.0, 1500.0]:
                    if time_left() < 20:
                        break
                    h_act = slsqp_active(h_cur, n, ts, active, alpha_a, 200)
                    c5_act = compute_c5(h_act)
                    if c5_act < best_c5:
                        best_c5 = c5_act
                        best_h = h_act.copy()
                        best_n = n
                    h_cur = h_act
                    _, active = compute_c5_and_active_lags(h_cur)
                    if len(active) == 0:
                        break

        if time_left() > 30:
            n_pg = min(2000, int(time_left() * 2))
            h_pg, c5_pg = projected_subgradient(h_cur, n, ts, n_pg, 0.05, 200.0)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n
            h_cur = h_pg

        if time_left() > 25:
            h_bp, c5_bp = binary_push(h_cur, n, ts, 0.05, 30)
            if c5_bp < best_c5:
                best_c5 = c5_bp
                best_h = h_bp.copy()
                best_n = n
            h_cur = h_bp

        if time_left() > 50:
            h_cd, c5_cd = coordinate_descent_fine(h_cur, n, ts, 30, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n
            h_cur = h_cd

        if time_left() > 40:
            h_pe, c5_pe = pairwise_exchange(h_cur, n, ts, 20, rng)
            if c5_pe < best_c5:
                best_c5 = c5_pe
                best_h = h_pe.copy()
                best_n = n
            h_cur = h_pe

        if time_left() > 60 and n <= 540:
            n_hires = n * 2
            ts_hires = n_hires / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_hires, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_cur)
            h_hi = project_to_sum(h_hi, ts_hires)
            for alpha in [100.0, 500.0, 1500.0]:
                if time_left() < 30:
                    break
                h_hi_opt = slsqp_optimize(h_hi, n_hires, ts_hires, alpha, 200)
                c5_hi = compute_c5(h_hi_opt)
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = h_hi_opt.copy()
                    best_n = n_hires
                h_hi = h_hi_opt

    if best_h is not None and time_left() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = best_h.copy()

        for alpha in [3000.0, 8000.0]:
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(h_f, n_f, ts_f, alpha, 400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f
            h_f = h_pol

        if time_left() > 30:
            c5_chk, active_f = compute_c5_and_active_lags(h_f)
            if len(active_f) > 0 and len(active_f) < n_f:
                h_act2 = slsqp_active(h_f, n_f, ts_f, active_f, 3000.0, 300)
                c5_act2 = compute_c5(h_act2)
                if c5_act2 < best_c5:
                    best_c5 = c5_act2
                    best_h = h_act2.copy()
                    best_n = n_f
                h_f = h_act2

        if time_left() > 25:
            h_cd_f, c5_cd_f = coordinate_descent_fine(h_f, n_f, ts_f, 80, rng)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = h_cd_f.copy()
                best_n = n_f
            h_f = h_cd_f

        if time_left() > 20:
            n_pg_f = min(3000, int(time_left() * 1.5))
            h_pg_f, c5_pg_f = projected_subgradient(h_f, n_f, ts_f, n_pg_f, 0.02, 500.0)
            if c5_pg_f < best_c5:
                best_c5 = c5_pg_f
                best_h = h_pg_f.copy()
                best_n = n_f

        if time_left() > 15:
            h_pe_f, c5_pe_f = pairwise_exchange(h_f, n_f, ts_f, 30, rng)
            if c5_pe_f < best_c5:
                best_c5 = c5_pe_f
                best_h = h_pe_f.copy()
                best_n = n_f

    if best_h is not None:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_to_sum(best_h, ts_final)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
