def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(values, tau):
    """Returns smooth max value and its gradient."""
    v = values / tau
    vmax = v.max()
    exps = np.exp(v - vmax)
    weights = exps / exps.sum()
    max_val = tau * (vmax + np.log(exps.sum()))
    grad = weights / tau
    return max_val, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Distribute the difference to elements closest to 0.5
        dists = np.abs(result - 0.5)
        idx = int(np.argmin(dists))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    val, _ = smooth_max_grad(corr, tau)
    return val * dx

def smooth_max_objective_grad(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    val, weights = smooth_max_grad(corr, tau)
    # Gradient of corr w.r.t. x
    # corr[j] for j >= n-1: lag = j-(n-1) >= 0
    #   corr[j] = sum_{i=0}^{n-1-lag} x[i+lag] * (1-x[i])
    #   d(corr[j])/dx[m] = sum_{i: i+lag=m} (1-x[i]) - sum_{i: i=m} x[i+lag]
    # For m: first term is non-zero when i = m-lag >= 0 and i < n-lag, i.e., m-lag valid
    #   (1-x[m-lag]) if 0 <= m-lag < n-lag
    # Second term: i=m, need m < n-lag, i.e., lag < n-m
    #   -x[m+lag] if m+lag < n
    # Similar for j < n-1 (negative lags)
    
    grad = np.zeros(n)
    for j in range(2*n-1):
        if weights[j] < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                # term: x[i+lag]*(1-x[i])
                # d/dx[i+lag] = (1-x[i])
                # d/dx[i] = -x[i+lag]
                grad[i + lag] += (1.0 - x[i]) * weights[j]
                if i != i + lag:
                    grad[i] -= x[i + lag] * weights[j]
        else:
            # lag < 0, let m = -lag > 0
            m = -lag
            for i in range(m, n):
                # term: x[i-m]*(1-x[i])
                # d/dx[i-m] = (1-x[i])
                # d/dx[i] = -x[i-m]
                grad[i - m] += (1.0 - x[i]) * weights[j]
                grad[i] -= x[i - m] * weights[j]
    return val * dx, grad * dx

from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        dists = np.abs(result - 0.5)
        idx = int(np.argmin(dists))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def to_near_binary(h, n):
    """Round to near-binary: (n-1)/2 ones, (n-1)/2 zeros, 1 half."""
    n1 = (n - 1) // 2
    idx_desc = np.argsort(-h)
    result = np.zeros(n)
    result[idx_desc[:n1]] = 1.0
    result[idx_desc[n1]] = 0.5
    return result

def sturmian_pattern(n, alpha, beta):
    """Generate a Sturmian/Beatty pattern."""
    x = np.arange(n) * alpha
    return (x % 1.0 < beta).astype(float)

def generate_constructive_starts(n, rng, parent):
    """Generate diverse starting points."""
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    
    # Parent-based
    if parent is not None:
        starts.append(parent.copy())
        if len(parent) != n:
            xp = np.linspace(0, 2, len(parent), endpoint=False)
            f = interp1d(xp, parent, kind='cubic', fill_value='extrapolate', bounds_error=False)
            starts.append(np.clip(f(x), 0, 1))
        else:
            for amp in [0.05, 0.1, 0.15]:
                starts.append(np.clip(parent + rng.randn(n) * amp, 0, 1))
    
    # Sturmian patterns (various slopes)
    alphas = [0.38, 0.39, 0.40, 0.41, 0.42, 0.45, 0.50, 0.52, 0.55]
    for alpha in alphas:
        beta = 0.5  # for balance
        h = sturmian_pattern(n, alpha, beta)
        starts.append(h)
    
    # Sinusoidal patterns
    for freq in [1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
        for phase in [0, np.pi/4, np.pi/2, 3*np.pi/4]:
            h = 0.5 + 0.45 * np.sin(2*np.pi*freq*x/2.0 + phase)
            starts.append(np.clip(h, 0, 1))
    
    # Random
    for _ in range(5):
        starts.append(rng.rand(n))
    
    return starts

def slsqp_optimize(h0, n, tau_schedule, max_iter_each, time_limit=None, start_time=None):
    """SLSQP with smooth max."""
    bounds = [(0.0, 1.0)] * n
    current = project_feasible(h0.copy(), n)
    
    for tau in tau_schedule:
        if time_limit is not None and time.time() > start_time + time_limit:
            break
        def obj(x):
            dx = 2.0 / n
            g = 1.0 - x
            corr = np.correlate(x, g, mode="full")
            v = corr / tau
            vmax = v.max()
            return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx
        
        def cons_fun(x):
            return np.sum(x) - n / 2.0
        
        try:
            res = minimize(
                obj, current,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons_fun}],
                options={'maxiter': max_iter_each, 'ftol': 1e-12}
            )
            current = project_feasible(res.x, n)
        except Exception:
            pass
    
    return current, evaluate_c5(current)

def binary_local_search(h_init, n, time_limit, seed=12345):
    """Local search on near-binary space with SA."""
    h = to_near_binary(h_init, n)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    start = time.time()
    rng = np.random.RandomState(seed)
    T = 0.015
    iterations = 0
    stuck_count = 0
    
    while time.time() - start < time_limit:
        iterations += 1
        S_idx = np.where(h > 0.7)[0]
        F_idx = np.where(h < 0.3)[0]
        H_idx = np.where(np.abs(h - 0.5) < 0.3)[0]
        
        if len(H_idx) != 1:
            fracs = np.where((h > 0.1) & (h < 0.9))[0]
            if len(fracs) > 0:
                H_idx = np.array([fracs[np.argmin(np.abs(h[fracs] - 0.5))]])
            else:
                break
        
        best_move_c5 = best_c5 - 1e-15
        best_move = None
        
        # Try all swaps (S -> F and F -> S)
        for a in S_idx:
            for b in F_idx:
                h[a] = 0.0
                h[b] = 1.0
                c5_try = evaluate_c5(h)
                h[b] = 0.0
                h[a] = 1.0
                if c5_try < best_move_c5:
                    best_move_c5 = c5_try
                    best_move = (a, b, 'swap')
        
        # Try half moves
        if len(H_idx) == 1:
            p = int(H_idx[0])
            for a in S_idx:
                old_p, old_a = h[p], h[a]
                h[p] = 1.0
                h[a] = 0.5
                c5_try = evaluate_c5(h)
                h[p], h[a] = old_p, old_a
                if c5_try < best_move_c5:
                    best_move_c5 = c5_try
                    best_move = (p, a, 'half_S')
            for b in F_idx:
                old_p, old_b = h[p], h[b]
                h[p] = 0.0
                h[b] = 0.5
                c5_try = evaluate_c5(h)
                h[p], h[b] = old_p, old_b
                if c5_try < best_move_c5:
                    best_move_c5 = c5_try
                    best_move = (p, b, 'half_F')
        
        if best_move is not None:
            a, b, mtype = best_move
            if mtype == 'swap':
                h[a] = 0.0
                h[b] = 1.0
            elif mtype == 'half_S':
                h[p] = 1.0
                h[a] = 0.5
            elif mtype == 'half_F':
                h[p] = 0.0
                h[b] = 0.5
            best_c5 = evaluate_c5(h)
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 20:
                # Restart with random perturbation
                h = best_h.copy()
                n_flip = rng.randint(2, 5)
                flip_pos = rng.choice(n, size=n_flip, replace=False)
                for pos in flip_pos:
                    h[pos] = 1.0 - h[pos]
                h = to_near_binary(h, n)
                stuck_count = 0
            else:
                # Try a random move (SA)
                if len(S_idx) > 0 and len(F_idx) > 0:
                    a = int(rng.choice(S_idx))
                    b = int(rng.choice(F_idx))
                    h[a] = 0.0
                    h[b] = 1.0
                    T = max(0.001, T * 0.99)
        
        # Cool down
        T = max(0.001, T * 0.999)
    
    return best_h, best_c5
