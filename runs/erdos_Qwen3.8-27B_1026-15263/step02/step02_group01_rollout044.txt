import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(n)[None, :]
    m1 = k_idx - i_idx + (n - 1)
    mask1 = (m1 >= 0) & (m1 < 2 * n - 1)
    w_mat1 = np.where(mask1, w[m1], 0.0)
    part1 = w_mat1.dot(1.0 - h)
    
    m2 = i_idx - k_idx + (n - 1)
    mask2 = (m2 >= 0) & (m2 < 2 * n - 1)
    w_mat2 = np.where(mask2, w[m2], 0.0)
    part2 = w_mat2.dot(h)
    
    grad = (part1 - part2) / alpha
    return sm_val, grad

def l_bfgs_optimize(h0, n, target_sum, alpha, penalty, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_and_grad(h, alpha)
        return v + penalty * (np.sum(h) - target_sum) ** 2
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        v, g = smooth_max_val_and_grad(h, alpha)
        g += 2 * penalty * (np.sum(h) - target_sum) * np.ones(n)
        return g
    
    res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
    return project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)

def fista_optimize(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iters):
        _, g = smooth_max_val_and_grad(y, alpha)
        # Sum constraint gradient
        sum_diff = np.sum(y) - target_sum
        g = g + sum_diff / n  # projection gradient approximation
        
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
    
    return h_best, c5_cur

def active_lag_descent(h0, n, target_sum, n_iters, top_k, lr):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        
        # Gradient from top lags only
        grad = np.zeros(n)
        for m in top_idx:
            offset = m - n + 1
            for i in range(n):
                if 0 <= i + offset < n:
                    grad[i] += (1.0 - h[i + offset])
                if 0 <= i - offset < n:
                    grad[i] -= h[i - offset]

        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)

        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
        else:
            lr *= 0.8
    
    return h_best, c5_cur

Actually, for the gradient from a single lag m:
∂c[m]/∂h[i] = [0≤i+offset<n] * (1-h[i+offset]) - [0≤i-offset<n] * h[i-offset]

where offset = m - n + 1.

I can compute this for all i in vectorized form:
```python
def grad_for_lag(h, m):
    n = len(h)
    offset = m - n + 1
    grad = np.zeros(n)
    # Part 1: (1-h[i+offset]) for valid i
    i_range = np.arange(n)
    idx1 = i_range + offset
    mask1 = (idx1 >= 0) & (idx1 < n)
    if np.any(mask1):
        grad[mask1] += (1.0 - h[idx1[mask1]])
    # Part 2: -h[i-offset] for valid i
    idx2 = i_range - offset
    mask2 = (idx2 >= 0) & (idx2 < n)
    if np.any(mask2):
        grad[mask2] -= h[idx2[mask2]]
    return grad
```

This is O(n) per lag, so O(n * top_k) total. For n=360, top_k=5, that's 1800 ops per iteration. Very fast.

    for rnd in range(max_rounds):
        improved = False

        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
