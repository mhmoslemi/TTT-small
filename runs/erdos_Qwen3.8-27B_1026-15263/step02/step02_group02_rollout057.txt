def build_grad_matrix(h0, n):
    dx = 2.0 / n
    g = n - 1
    grad = np.zeros((2*g+1, n))
    for idx in range(2*g+1):
        k = idx - g
        for j in range(n):

            if 0 <= j+k < n:
                val -= h0[j+k]
            if 0 <= j-k < n:
                val -= h0[j-k]

    return grad

def build_grad_matrix(h0, n):
    dx = 2.0 / n
    g = n - 1
    grad = np.ones((2 * g + 1, n)) * dx
    for idx in range(2 * g + 1):
        k = idx - g
        if k >= 0:
            # h0[j+k]: j from 0 to n-1-k, values h0[k:n]
            grad[idx, :n-k] -= dx * h0[k:n]
            # h0[j-k]: j from k to n-1, values h0[0:n-k]
            grad[idx, k:n] -= dx * h0[0:n-k]
        else:
            ak = -k
            # h0[j+k] = h0[j-ak]: j from ak to n-1, values h0[0:n-ak]
            grad[idx, ak:n] -= dx * h0[0:n-ak]
            # h0[j-k] = h0[j+ak]: j from 0 to n-1-ak, values h0[ak:n]
            grad[idx, 0:n-ak] -= dx * h0[ak:n]
    return grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import cvxpy as cp

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total_room = room.sum()
            if total_room < 1e-15:
                break
            # Proportional allocation
            alloc = diff * room / total_room
            h[mask] += alloc
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            total = h[mask].sum()
            if total < 1e-15:
                break
            h[mask] += diff * h[mask] / total
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h0, n):
    dx = 2.0 / n
    g = n - 1
    grad = np.ones((2 * g + 1, n)) * dx
    for idx in range(2 * g + 1):
        k = idx - g
        if k >= 0:
            if n - k > 0:
                grad[idx, :n-k] -= dx * h0[k:n]
                grad[idx, k:n] -= dx * h0[0:n-k]
        else:
            ak = -k
            if n - ak > 0:
                grad[idx, ak:n] -= dx * h0[0:n-ak]
                grad[idx, 0:n-ak] -= dx * h0[ak:n]
    return grad

def solve_qp(h0, n, lam):
    dx = 2.0 / n
    c0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    grad = build_grad_matrix(h0, n)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + lam / 2.0 * cp.sum_squares(delta))
    cons = [
        t - grad @ delta >= c0,
        cp.sum(delta) == 0.0,
        delta >= -h0,
        delta <= 1.0 - h0
    ]
    
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    
    if delta.value is None:
        return None
    d = np.array(delta.value).flatten().copy()
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    return d

def qp_optimize(h_init, n, lam_schedule, max_iters_total=200, time_limit_s=60):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    t0 = time.time()
    
    for lam in lam_schedule:
        for _ in range(max_iters_total // len(lam_schedule)):
            if time.time() - t0 > time_limit_s:
                return best_h, best_c5
            delta = solve_qp(h, n, lam)
            if delta is None:
                break
            delta_norm = np.linalg.norm(delta)
            if delta_norm < 1e-10:
                break
            # Line search
            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
                h_try = h + alpha * delta
                if h_try.min() < -0.01 or h_try.max() > 1.01:
                    continue
                h_try = np.clip(h_try, 0, 1)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5_from_h(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                break
    return h, compute_c5_from_h(h)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t_val = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t_val - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter=3000):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.1]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init
