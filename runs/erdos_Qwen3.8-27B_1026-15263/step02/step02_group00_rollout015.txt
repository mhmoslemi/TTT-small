```python
        first_contrib = np.where(first_valid, 1.0 - h[np.clip(j_idx + s_idx, 0, n-1)], 0.0)
        second_contrib = np.where(second_valid, h[np.clip(j_idx - s_idx, 0, n-1)], 0.0)
        grad = w[None, :] @ (first_contrib - second_contrib)
    
    return obj, grad

def lbfgs_optimize(h0, target_sum, alpha, maxiter=2000, ftol=1e-14):
    n = len(h0)
    best_x = h0.copy()
    best_f = compute_c5(h0)
    
    def fun_g(x):
        obj, grad = smooth_max_and_grad(x, alpha)
        # Add penalty for constraint violation
        penalty = 100.0 * (np.sum(x) - target_sum) ** 2
        grad = grad + 200.0 * (np.sum(x) - target_sum) * np.ones(n)
        return obj + penalty, grad
    
    try:
        res = minimize(fun_g, h0.copy(), method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10,
                                'maxls': 50, 'maxfun': 5000})
        h_new = project_to_sum(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_f:
            best_x = h_new
            best_f = c5_new
    except Exception:
        pass
    
    return best_x, best_f
```

2. Adjust to satisfy sum constraint (like project_to_sum)

```python
            h[j] = 0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_w(corr, alpha):
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    return obj, w

def grad_from_weights(h, w):
    n = len(h)
    # grad[i] = sum_m w[m] * [(1-h[i-m-(n-1)])*I(0<=i-m-(n-1)<n) - h[i+m-(n-1)]*I(0<=i+m-(n-1)<n)]
    # where m is index into w (0 to 2n-2), corresponding to shift s = m-(n-1)
    # Let s = m-(n-1), so m = s+(n-1)
    # grad[i] = sum_s w[s+(n-1)] * [(1-h[i-s])*I(0<=i-s<n) - h[i+s]*I(0<=i+s<n)]
    
    i_idx = np.arange(n)[:, None]  # shape (n, 1)
    s_idx = np.arange(-(n-1), n)[None, :]  # shape (1, 2n-1), s from -(n-1) to n-1
    
    # First term: (1-h[i-s]) * I(0<=i-s<n)
    first_arg = i_idx - s_idx  # i-s
    first_valid = (first_arg >= 0) & (first_arg < n)
    first_contrib = np.where(first_valid, 1.0 - h[np.clip(first_arg, 0, n-1)], 0.0)
    
    # Second term: h[i+s] * I(0<=i+s<n)
    second_arg = i_idx + s_idx  # i+s
    second_valid = (second_arg >= 0) & (second_arg < n)
    second_contrib = np.where(second_valid, h[np.clip(second_arg, 0, n-1)], 0.0)
    
    grad = (w[None, :] * (first_contrib - second_contrib)).sum(axis=1)
    return grad

def smooth_obj_grad(h, alpha):
    corr = compute_corr(h)
    obj, w = smooth_max_val_and_w(corr, alpha)
    grad = grad_from_weights(h, w)
    return obj, grad

def pgd_optimize(h0, target_sum, alpha, n_iter=2000, lr_init=None, time_limit=30):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if lr_init is None:
        lr_init = 0.1 / np.sqrt(n)
    
    t0 = time.time()
    lr = lr_init
    stall = 0
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        obj, grad = smooth_obj_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 1.0)
            stall = 0
        else:
            stall += 1
            if stall > 20:
                lr *= 0.5
                stall = 0
            h = h_new  # still move, just don't update best
    return best_h, best_c5

def lbfgs_opt(h0, target_sum, alpha, maxiter=2000):
    n = len(h0)
    
    def fun_g(x):
        obj, grad = smooth_obj_grad(x, alpha)
        # Penalty for sum constraint
        s = np.sum(x) - target_sum
        obj_p = obj + 50.0 * s * s
        grad_p = grad + 100.0 * s * np.ones(n)
        return obj_p, grad_p
    
    try:
        res = minimize(fun_g, h0.copy(), method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 5000})
        h_new = project_to_sum(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_opt(h0, target_sum, alpha, maxiter=500):
    n = len(h0)
    
    def fun(x):
        return smooth_obj_grad(x, alpha)[0]
    
    def grad(x):
        return smooth_obj_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h0.copy(), method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_to_sum(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return project_to_sum(h0, target_sum), compute_c5(h0)

def binary_sa(h0, target_sum, rng, max_iters=50000, t_init=0.05, t_min=0.001, time_limit=60):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    # Fix sum
    attempts = 0
    while abs(h.sum() - target_sum) > 1e-10 and attempts < 10000:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        attempts += 1
    if abs(h.sum() - target_sum) > 1e-10:
        return h0, compute_c5(h0)
    
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    dx = 2.0 / n
    
    t0 = time.time()
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        T = t_init * max(t_min / t_init, (t_min / t_init) ** (it / max_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        corr_new = compute_corr(h)
        c5_new = float(np.max(corr_new) * dx)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def make_diverse_starts(n, target_sum, rng, best_h=None, best_n=0):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    
    # Uniform
    starts.append(np.full(n, 0.5))
    
    # Random
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    
    # Sine harmonics
    for freq in [1, 2, 3, 4, 5, 6, 8, 10]:
        for amp in [0.3, 0.4, 0.45]:
            s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
            starts.append(project_to_sum(s, target_sum))
    
    # Two-tone
    for f1, f2 in [(1,2), (1,3), (2,3), (2,5), (3,4), (3,5), (4,5), (5,7)]:
        a = rng.uniform(0.15, 0.35)
        b = rng.uniform(0.1, 0.3)
        p1 = rng.uniform(0, 2*np.pi)
        p2 = rng.uniform(0, 2*np.pi)
        s = 0.5 + a*np.sin(2*np.pi*f1*x/2 + p1) + b*np.sin(2*np.pi*f2*x/2 + p2)
        starts.append(project_to_sum(s, target_sum))
    
    # Block patterns
    for nblk in [2, 3, 4, 5, 6, 8]:
        blk = n // nblk
        h = np.zeros(n)
        for i in range(0, n, 2*blk):
            h[i:min(i+blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    
    # Thue-Morse inspired
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    starts.append(project_to_sum(tm, target_sum))
    starts.append(project_to_sum(1-tm, target_sum))
    
    # Best solution projected
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            starts.append(project_to_sum(np.interp(x, x_old, best_h, period=2.0), target_sum))
    
    # Binary with correct count
    hb = np.zeros(n)
    idx = rng.choice(n, size=int(target_sum), replace=False)
    hb[idx] = 1.0
    starts.append(hb.copy())
    
    return starts
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_from_weights(h, w):
    n = len(h)
    i_idx = np.arange(n)[:, None]
    s_idx = np.arange(-(n - 1), n)[None, :]
    first_arg = i_idx - s_idx
    first_valid = (first_arg >= 0) & (first_arg < n)
    first_contrib = np.where(first_valid, 1.0 - h[np.clip(first_arg, 0, n - 1)], 0.0)
    second_arg = i_idx + s_idx
    second_valid = (second_arg >= 0) & (second_arg < n)
    second_contrib = np.where(second_valid, h[np.clip(second_arg, 0, n - 1)], 0.0)
    grad = (w[None, :] * (first_contrib - second_contrib)).sum(axis=1)
    return grad

def smooth_obj_grad(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    grad = grad_from_weights(h, w)
    return obj, grad

def pgd_optimize(h0, target_sum, alpha, n_iter=3000, lr_init=None, time_limit=40):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if lr_init is None:
        lr_init = 0.5 / np.sqrt(n)
    t0 = time.time()
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:

        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                stall = 0
            h = h_new
    return best_h, best_c5

def slsqp_opt(h0, target_sum, alpha, maxiter=500):
    n = len(h0)

    def fun(x):
        return smooth_obj_grad(x, alpha)[0]

    def grad(x):
        return smooth_obj_grad(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h0.copy(), method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_to_sum(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return project_to_sum(h0, target_sum), compute_c5(h0)

    except Exception:
        return h0, compute_c5(h0)

def binary_sa(h0, target_sum, rng, max_iters=60000, t_init=0.04, t_min=0.001, time_limit=80):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    attempts = 0
    while abs(h.sum() - target_sum) > 1e-10 and attempts < 10000:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1.0
        attempts += 1
    if abs(h.sum() - target_sum) > 1e-10:
        return h0, compute_c5(h0)
    dx = 2.0 / n
    c5_cur = float(np.max(compute_corr(h)) * dx)
    best_h = h.copy()
    best_c5 = c5_cur
    t0 = time.time()
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        T = t_init * (t_min / t_init) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        corr_new = compute_corr(h)
        c5_new = float(np.max(corr_new) * dx)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def make_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10, 12]:
        s = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(s, target_sum))
    for f1, f2 in [(1, 2), (1, 3), (2, 3), (2, 5), (3, 4), (3, 5), (4, 6), (5, 7), (1, 4), (2, 7)]:
        a = rng.uniform(0.15, 0.4)
        b = rng.uniform(0.05, 0.3)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        s = 0.5 + a * np.sin(2 * np.pi * f1 * x / 2 + p1) + b * np.sin(2 * np.pi * f2 * x / 2 + p2)
        starts.append(project_to_sum(s, target_sum))
    for nblk in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        blk = max(1, n // nblk)
        hh = np.zeros(n)
        for i in range(0, n, 2 * blk):
            hh[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(hh, target_sum))
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    starts.append(project_to_sum(tm, target_sum))
    starts.append(project_to_sum(1.0 - tm, target_sum))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            starts.append(project_to_sum(np.interp(x, x_old, best_h, period=2.0), target_sum))
    hb = np.zeros(n)
    idx = rng.choice(n, size=int(target_sum), replace=False)
    hb[idx] = 1.0
    starts.append(hb.copy())
    hb2 = np.zeros(n)
    idx2 = rng.permutation(n)[:int(target_sum)]
    hb2[idx2] = 1.0
    starts.append(hb2.copy())
    return starts

def refine_continuous(h, n, target_sum, time_limit=50):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alphas = [5, 20, 80, 300]
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        h_new, c5_new = pgd_optimize(best_h, target_sum, alpha, n_iter=2000,
                                      time_limit=time_limit - (time.time() - t0))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    if time.time() - t0 < time_limit - 15:
        h_new, c5_new = slsqp_opt(best_h, target_sum, 200, maxiter=400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        x = np.linspace(0, 2, best_n, endpoint=False)
        best_h = project_to_sum(0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x),
                                best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 1: Continuation optimization at parent resolution
    n1 = best_n
    ts1 = n1 / 2.0
    h_cur, c5_cur = refine_continuous(best_h, n1, ts1, time_limit=min(150, time_left() - 50))
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur
        best_n = n1

    # Phase 2: Binary SA + continuous re-optimize
    if time_left() > 120:
        h_bin, c5_bin = binary_sa(best_h, ts1, rng, max_iters=100000,
                                   t_init=0.05, t_min=0.002, time_limit=min(90, time_left() - 60))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n1
        h_ref, c5_ref = refine_continuous(h_bin, n1, ts1, time_limit=min(60, time_left() - 30))
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = n1

    # Phase 3: Multi-resolution exploration
    resolutions = [40, 60, 80, 100, 120, 150, 180, 220, 240, 300]
    for n_tr in resolutions:
        if time_left() < 100:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        starts = make_starts(n_tr, ts_tr, rng, best_h, best_n)
        for h_start in starts[:8]:
            if time_left() < 70:
                break
            h_s = project_to_sum(h_start, ts_tr)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_tr
            h_r, c5_r = refine_continuous(h_s, n_tr, ts_tr, time_limit=40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_tr

    # Phase 4: L-BFGS-B refinement at best resolution
    if time_left() > 100:
        n_f = best_n
        ts_f = n_f / 2.0
        h_lb, c5_lb = lbfgs_opt(best_h, ts_f, 100, maxiter=3000)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb
        h_lb2, c5_lb2 = lbfgs_opt(best_h, ts_f, 500, maxiter=2000)
        if c5_lb2 < best_c5:
            best_c5 = c5_lb2
            best_h = h_lb2

    # Phase 5: Shake-and-reoptimize (diversity)
    for trial in range(15):
        if time_left() < 50:
            break
        noise_scale = 0.08 * (1.0 - trial / 15.0) + 0.02
        h_shake = best_h + rng.uniform(-noise_scale, noise_scale, best_n)
        h_shake = project_to_sum(h_shake, best_n / 2.0)
        h_r, c5_r = refine_continuous(h_shake, best_n, best_n / 2.0, time_limit=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 6: Second round binary SA at potentially improved starting point
    if time_left() > 100:
        h_bin2, c5_bin2 = binary_sa(best_h, best_n / 2.0, rng, max_iters=80000,
                                     t_init=0.03, t_min=0.001, time_limit=min(70, time_left() - 50))
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        if time_left() > 40:
            h_r2, c5_r2 = refine_continuous(best_h, best_n, best_n / 2.0, time_limit=min(50, time_left() - 20))
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2

    # Phase 7: High-alpha SLSQP final polish
    if time_left() > 40:
        h_f, c5_f = slsqp_opt(best_h, best_n / 2.0, 800, maxiter=1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left() > 25:
        h_f2, c5_f2 = slsqp_opt(best_h, best_n / 2.0, 2000, maxiter=1000)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    # Final validation and cleanup
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
