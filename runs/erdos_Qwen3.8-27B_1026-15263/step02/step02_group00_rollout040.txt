```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    corr_shifted = corr - m_val
    exp_vals = np.exp(alpha * corr_shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    i_idx = np.arange(n)[:, None]
    s_idx = np.arange(-(n-1), n)
    M_mask = (i_idx + s_idx[None, :] >= 0) & (i_idx + s_idx[None, :] < n)
    N_mask = (i_idx - s_idx[None, :] >= 0) & (i_idx - s_idx[None, :] < n)
    hs_idx = np.clip(i_idx + s_idx[None, :], 0, n-1)
    h_idx = np.clip(i_idx - s_idx[None, :], 0, n-1)
    term1 = M_mask * w[None, :] * (1.0 - h[hs_idx])
    term2 = N_mask * w[None, :] * h[h_idx]
    grad = np.sum(term1 - term2, axis=1)
    return obj, grad

def local_optimize(h0, n, target_sum, alpha, maxiter=1000):
    """Local optimization using SLSQP with smooth max."""
    def f(h):
        return smooth_max_obj_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_and_grad(h, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    h = (h0 > 0.5).astype(float)

    attempts = 0
    while not np.isclose(h.sum(), target_sum) and attempts < 100000:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h > 0.5)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h < 0.5)[0])
            h[idx] = 1.0
        attempts += 1
    
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    n_total = max(100000, int(time_limit * 5000))  # Estimate ~5000 steps/sec
    T_start = 0.005
    T_end = 0.0001
    cooling = (T_end / T_start) ** (1.0 / n_total)
    T = T_start
    
    for step in range(n_total):
        if time.time() - t_start > time_limit:
            break
        if step % 5000 == 0:
            T = max(T * cooling**5000, T_end)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
```

```python
def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):

        if abs(diff) < 1e-15:
            break

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    corr_shifted = corr - m_val
    exp_vals = np.exp(alpha * corr_shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    i_idx = np.arange(n)[:, None]
    s_idx = np.arange(-(n-1), n)
    M_mask = (i_idx + s_idx[None, :] >= 0) & (i_idx + s_idx[None, :] < n)
    N_mask = (i_idx - s_idx[None, :] >= 0) & (i_idx - s_idx[None, :] < n)
    hs_idx = np.clip(i_idx + s_idx[None, :], 0, n-1)
    h_idx = np.clip(i_idx - s_idx[None, :], 0, n-1)
    term1 = M_mask.astype(float) * w[None, :] * (1.0 - h[hs_idx])
    term2 = N_mask.astype(float) * w[None, :] * h[h_idx]
    grad = np.sum(term1 - term2, axis=1)
    return obj, grad
```
