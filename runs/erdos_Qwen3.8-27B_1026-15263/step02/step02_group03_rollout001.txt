            shift = lag_idx - (n - 1)
            if shift == 0:
                continue
            
            # For each pair (i, i+shift), compute the "conflict"
            if shift > 0:
                pairs_i = np.arange(0, n - shift)
                pairs_j = pairs_i + shift
            else:
                pairs_i = np.arange(-shift, n)
                pairs_j = pairs_i + shift
            
            # Conflict: h[i]*(1-h[j]) + h[j]*(1-h[i]) is high when they disagree
            # We want to reduce this
            conflicts = best[pairs_i] * (1 - best[pairs_j]) + best[payers_j] * (1 - best[pairs_i])
            
            # Find most conflicting pair and try to fix it
            worst = np.argmax(conflicts)
            i_idx = pairs_i[worst]
            j_idx = pairs_j[worst]
            
            # Try to make them agree: if h[i] > h[j], decrease h[i] or increase h[j]
            # Or vice versa
            h_new = best.copy()
            if best[i_idx] > best[j_idx]:
                # Move h[i] down or h[j] up
                delta = 0.01 * rng.rand()
                if rng.rand() < 0.5:
                    h_new[i_idx] -= delta
                else:
                    h_new[j_idx] += delta
            else:
                delta = 0.01 * rng.rand()
                if rng.rand() < 0.5:
                    h_new[j_idx] -= delta
                else:
                    h_new[i_idx] += delta
            
            h_new = project_feasible(h_new, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best = h_new
                best_c5 = c5
                improved = True
        
        if not improved:
            # Try broader random moves
            h_new = project_feasible(best + rng.randn(n) * 0.01, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best = h_new
                best_c5 = c5
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx


def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))


def smax_obj(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx


def smax_obj2(x, n, tau, topk):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full") * dx
    if topk <= len(corr) and topk > 1:
        return smooth_max(corr, tau) * 0.5 + np.sort(corr)[-topk:].mean() * 0.5
    return smooth_max(corr, tau)


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result


def slsqp_refine(h, n, tau_list, time_budget, max_iter_each=150):
    current = h.copy()
    best = current.copy()
    best_c5 = evaluate_c5(best)
    elapsed = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    for tau in tau_list:
        if time.time() - elapsed > time_budget * 0.95:
            break
        for _ in range(2):
            if time.time() - elapsed > time_budget * 0.95:
                break
            try:
                res = minimize(
                    smax_obj, current,
                    method='SLSQP',
                    args=(n, tau),
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iter_each, 'ftol': 1e-14}
                )
                current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best = current.copy()
            except Exception:
                pass
    return best, best_c5


def lag_targeted_optimize(h, n, rng, time_budget):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    elapsed = time.time()
    while time.time() - elapsed < time_budget:
        corr = get_corr(best)
        top_lags = np.argsort(corr)[-4:]
        improved = False
        for lag_idx in top_lags:
            shift = int(lag_idx - (n - 1))
            if shift == 0:
                continue
            if shift > 0:
                pairs_i = np.arange(0, n - shift)
                pairs_j = pairs_i + shift
            else:
                pairs_i = np.arange(-shift, n)
                pairs_j = pairs_i + shift
            if len(pairs_i) == 0:
                continue
            conflicts = (best[pairs_i] * (1.0 - best[pairs_j]) +
                         best[pairs_j] * (1.0 - best[pairs_i]))
            top_conflicts = np.argsort(conflicts)[-min(6, len(conflicts)):]
            for tc in top_conflicts:
                i_idx = int(pairs_i[tc])
                j_idx = int(pairs_j[tc])

                        h_new[j_idx] = min(1.0, h_new[j_idx] + delta)
                else:
                    if rng.rand() < 0.6:
                        h_new[j_idx] = max(0.0, h_new[j_idx] - delta)
                    else:
                        h_new[i_idx] = min(1.0, h_new[i_idx] + delta)
                h_new = project_feasible(h_new, n)

                    best_c5 = c5

def fine_coord_descent(h, n, time_budget, sweeps=15):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    elapsed = time.time()
    base_deltas = np.array([-0.03, -0.015, -0.007, -0.003, -0.001, 0.001, 0.003, 0.007, 0.015, 0.03])

            break

                break

def simulated_anneal(h, n, rng, time_budget, t_start=0.03, t_end=0.0003):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    current = best.copy()
    current_c5 = best_c5
    elapsed = time.time()
    total_steps_est = int(time_budget / 0.01)
    if total_steps_est < 10:
        total_steps_est = 10
    cool_rate = (t_end / t_start) ** (1.0 / total_steps_est)
    T = t_start
    block_choices = np.array([1, 2, 3, 4, 6, 10, 15])

    while time.time() - elapsed < time_budget and T > t_end:
        bs = int(rng.choice(block_choices))
        if bs >= n:
            bs = max(1, n // 4)
        start_idx = rng.randint(0, n - bs + 1)
        mask = np.zeros(n)
        mask[start_idx:start_idx + bs] = 1.0
        scale = 0.01 + T * 2.0
        noise = rng.randn(n) * scale * mask
        candidate = project_feasible(current + noise, n)
        c5 = evaluate_c5(candidate)
        delta = c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            current = candidate
            current_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best = candidate.copy()
        T *= cool_rate
    return best, best_c5


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h_old, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new)
