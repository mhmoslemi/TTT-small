```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj_grad(h, alpha, penalty, target_sum):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    m = np.max(corr)
    exp_shifted = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_shifted)
    logsumexp = m + np.log(sum_exp) / alpha
    w = exp_shifted / sum_exp
    
    sum_diff = np.sum(h) - target_sum
    obj = logsumexp + penalty * sum_diff * sum_diff
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        if k < n:
            grad[:n - k] -= wk * h[k:]
        if k > 0:
            grad[k:] += wk * one_minus_h[:n - k]
        else:
            grad += wk * one_minus_h
    
    grad += 2.0 * penalty * sum_diff
    return obj, grad

def lbfgs_optimize(h0, n, target_sum, alpha, penalty, maxiter=1000):
    """Run L-BFGS-B optimization with penalty for sum constraint."""
    def fun(x):
        return compute_obj_grad(x, alpha, penalty, target_sum)[0]
    def grad(x):
        return compute_obj_grad(x, alpha, penalty, target_sum)[1]
    
    res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_to_sum(res.x, target_sum)
```

This makes sense: the problem is bilinear (quadratic overall), not separable.

So I should stick with the smooth max + L-BFGS-B approach, or use a different strategy.

Let me think about what else I can try:

1. **Nelder-Mead (simplex) method**: Derivative-free, can handle the non-smooth max directly. But very slow in high dimensions.

2. **Simulated annealing**: Random perturbations with acceptance criterion. Could escape local optima.

3. **Mixing approach**: Take a convex combination of the current best h and a random h, then optimize.

4. **Multi-objective scan**: Instead of smooth max, directly optimize the sum of the top-K correlations. This is smoother and might find better optima.

5. **p-norm approximation**: Instead of max or smooth max, use the p-norm of the correlations (raised to power p, summed, then p-th root). As p → ∞, this approaches the max.

Let me try approach 5: p-norm with large p.

obj_p(h) = (Σ_k C(k)^p)^(1/p)

For large p, this is close to max_k C(k). The gradient is:
∂obj_p/∂h[i] = (1/p) * (Σ_k C(k)^p)^(1/p - 1) * Σ_k p * C(k)^(p-1) * ∂C(k)/∂h[i]
= (Σ_k C(k)^p)^(1/p - 1) * Σ_k C(k)^(p-1) * ∂C(k)/∂h[i]

This is well-defined and smooth for all h. For p = 100 or 200, it should be a good approximation of the max.

The advantage over smooth max: no free parameter (alpha), and the gradient is always well-defined (no risk of numerical issues with the exp).

Let me implement this:

```python
def pnorm_obj_grad(h, p, penalty, target_sum):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    
    # p-norm (unnormalized, we'll multiply by dx at the end)
    corr_p = np.abs(corr) ** p  # corr should be non-negative since h in [0,1]
    sum_corr_p = np.sum(corr_p)
    obj_pnorm = sum_corr_p ** (1.0 / p) * (2.0 / n)  # normalize by dx
    
    # Actually, let's work with the unnormalized version for simplicity
    # and just remember to multiply by dx at the end for reporting
    obj_pnorm = (sum_corr_p) ** (1.0 / p)
    
    # Weights for gradient
    if sum_corr_p > 1e-30:
        w = p * corr ** (p - 1) / (p * sum_corr_p ** (1 - 1.0/p))
        # Actually: d/dh (sum C^p)^(1/p) = (1/p) (sum C^p)^(1/p - 1) * sum p C^(p-1) dC/dh
        # = (sum C^p)^(1/p - 1) * sum C^(p-1) dC/dh
        norm_factor = sum_corr_p ** (1.0/p - 1.0)
        w = corr ** (p - 1) * norm_factor
    else:
        w = np.ones(2*n-1) / (2*n-1)
    
    sum_diff = np.sum(h) - target_sum
    obj = obj_pnorm + penalty * sum_diff * sum_diff
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        if k < n:
            grad[:n - k] -= wk * h[k:]
        if k > 0:
            grad[k:] += wk * one_minus_h[:n - k]
        else:
            grad += wk * one_minus_h
    
    grad += 2.0 * penalty * sum_diff
    return obj, grad
```

Hmm, but the p-norm with large p might have numerical issues (overflow in corr^p for large p). Let me use log-space:

log(obj_pnorm) = (1/p) * log(sum exp(p * log(corr)))

And for the gradient, I can use the softmax trick:
w[k] = exp(p * (log(corr[k]) - max_log)) / sum(exp(p * (log(corr) - max_log)))

This is essentially the same as the smooth max approach! The smooth max with parameter alpha IS the p-norm with p = alpha (approximately, for large alpha).

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj_grad(h, alpha, penalty, target_sum):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    m = np.max(corr)
    exp_shifted = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_shifted)
    logsumexp = m + np.log(sum_exp) / alpha
    w = exp_shifted / sum_exp
    
    sum_diff = np.sum(h) - target_sum
    obj = logsumexp + penalty * sum_diff * sum_diff
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        if k < n:
            grad[:n - k] -= wk * h[k:]
        if k > 0:
            grad[k:] += wk * one_minus_h[:n - k]
        else:
            grad += wk * one_minus_h
    
    grad += 2.0 * penalty * sum_diff
    return obj, grad

def lbfgs_optimize(h0, target_sum, alpha, penalty, maxiter=2000):
    n = len(h0)
    def fun(x):
        return compute_obj_grad(x, alpha, penalty, target_sum)[0]
    def grad(x):
        return compute_obj_grad(x, alpha, penalty, target_sum)[1]
    
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 50000})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def interp_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points using periodic interpolation."""
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def make_structured_start(n, target_sum, rng, mode):
    """Generate structured starting points."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    if mode == 0:

    elif mode == 1:

        return project_to_sum(h, target_sum)
    elif mode == 2:

        f2 = rng.choice([4, 5, 6, 7, 8])

        p2 = rng.uniform(0, 2 * np.pi)

        return project_to_sum(h, target_sum)
    elif mode == 3:

        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        return project_to_sum(h, target_sum)
    elif mode == 4:

        h = (rng.rand(n) > 0.5).astype(float)

        attempts = 0

            if h.sum() > target_sum:
                idx = rng.choice(np.where(h > 0)[0])
                h[idx] = 0
            else:
                idx = rng.choice(np.where(h < 1)[0])
                h[idx] = 1
            attempts += 1
        return project_to_sum(h, target_sum)
    else:
        return project_to_sum(rng.uniform(0.2, 0.8, n), target_sum)
```
