        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, h_best, kind='linear', fill_value='extrapolate')
        h_new = np.clip(f_interp(x_new), 0, 1)
        h_new = project(h_new, n_try)
        
        # Optimize at new resolution
        h_opt, c5_opt = paired_cd(h_new, n_try, n_pairs_per_pass=200,
                                   max_passes=100, time_limit=60, rng=rng)
        if c5_opt < c5_best:
            h_best, c5_best, n = h_opt, c5_opt, n_try
    
    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 30:
        h_sl, c5_sl = slsqp_refine(h_best, len(h_best), maxiter=8000)
        if c5_sl < c5_best:
            h_best, c5_best = h_sl, c5_sl
    
    # Final projection
    h_best = project(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    
    return h_best, c5_best, len(h_best)

Actually, for a piecewise-linear convex function with at most O(n) pieces (one per correlation value), we can find the exact minimum by:
1. Computing the breakpoints (where the active constraint changes)
2. Evaluating at each breakpoint and at the interval endpoints
3. Taking the minimum

The breakpoints occur where c_k(δ) = c_l(δ) for some pair (k,l). Since c_k and c_l are both linear in δ, the breakpoint is at δ = (c_k(0) - c_l(0)) / (slope_l - slope_k).

But computing all O(n^2) breakpoints is expensive. Let me just use minimize_scalar.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def get_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
            if not m.any():
                break
            s = min(d / m.sum(), (1.0 - h[m]).min() * 0.999)
            h[m] += s
        else:
            m = h > 0.0
            if not m.any():
                break
            s = min(-d / m.sum(), h[m].min() * 0.999)
            h[m] -= s
    return np.clip(h, 0.0, 1.0)


def paired_objective(delta, h, i, j, n):
    h[i] += delta
    h[j] -= delta
    val = np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)
    h[i] -= delta
    h[j] += delta
    return val


def paired_cd_h(h0, n, n_pairs_per_pass=300, max_passes=300, time_limit=120.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(0)
    t_start = time.time()
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_total = 2 * n - 1  # number of correlation values
    
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        
        for _ in range(n_pairs_per_pass):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue
            
            res = minimize_scalar(
                paired_objective, args=(h, i, j, n),
                bounds=(lo, hi), method='bounded',
                options={'xatol': 1e-9}
            )
            if res.success and res.fun < best_c5 - 1e-10:
                h[i] = np.clip(h[i] + res.x, 0, 1)
                h[j] = np.clip(h[j] - res.x, 0, 1)
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        
        if not improved and p > 20:
            break
    
    return best_h, best_c5


def slsqp_refine(h0, n, maxiter=5000, time_limit=60.0):
    t_start = time.time()
    t_init = compute_c5(h0)
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h_c = x[:-1]
        return x[-1] - np.correlate(h_c, 1.0 - h_c, mode="full") * (2.0 / n)
    
    def eq(x):
        return x[:-1].sum() - n / 2.0
    
    def eqj(x):
        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j
    
    x0 = np.append(h0.copy(), t_init + 0.1)
    bnds = [(0.0, 1.0)] * n + [(max(0.001, t_init * 0.3), 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq, 'jac': eqj}]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if res.x is not None and time.time() - t_start < time_limit:
        h_out = project(res.x[:-1], n)
        c5_out = compute_c5(h_out)
        return h_out, c5_out
    return h0, t_init
