from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)

def lse_objective(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    c_shifted = T * (c - c_max)
    c_shifted = np.clip(c_shifted, -500, 500)
    obj = c_max + np.log(np.sum(np.exp(c_shifted))) / T
    sum_viol = np.sum(h) - n / 2.0
    obj += penalty * sum_viol * sum_viol
    return obj

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    c_shifted = T * (c - c_max)

    idx = np.arange(n)

    threshold = 1e-10

        m = k - n + 1

        lo1 = max(0, m)
        hi1 = min(n, m + n)

        # Term 2: -h_ext[i+m]
        t2 = np.zeros(n)
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            t2[lo2:hi2] = -h[lo2 + m: hi2 + m]
        grad += w[k] * (t1 + t2)
    
    return grad

def lse_obj_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    c_shifted = T * (c - c_max)
    c_shifted = np.clip(c_shifted, -500, 500)
    obj = c_max + np.log(np.sum(np.exp(c_shifted))) / T
    
    sum_viol = np.sum(x) - n / 2.0
    obj += penalty * sum_viol * sum_viol
    grad_sum = 2.0 * penalty * sum_viol

    c = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)


def lse_obj_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    c_shifted = T * (c - c_max)
    c_shifted = np.clip(c_shifted, -500, 500)
    obj = c_max + np.log(np.sum(np.exp(c_shifted))) / T
    sum_viol = np.sum(x) - n / 2.0
    obj += penalty * sum_viol * sum_viol
    grad_sum = 2.0 * penalty * sum_viol
    w = np.exp(c_shifted)
    w /= w.sum()
    idx = np.arange(n)
    grad = np.full(n, grad_sum)
    threshold = 1e-10
    active_ki = np.where(w > threshold)[0]
    for ki in active_ki:
        k = ki - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += w[ki] * (1.0 - x[lo1 - k:hi1 - k])
        if k >= 0:
            lo2 = 0
            hi2 = n - k
        else:
            lo2 = -k
            hi2 = n
        if lo2 < hi2:
            grad[lo2:hi2] += w[ki] * (-x[lo2 + k:hi2 + k])
    return obj, grad


def lbfgsb_optimize(h_init, T, penalty, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lse_obj_and_grad, h_init, args=(T, penalty),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def generate_perturbation(h, scale, rng, n):
    r = rng.random()
    if r < 0.2:
        h_p = h.copy()
        n_modes = rng.integers(1, 4)
        for _ in range(n_modes):
            freq = rng.integers(1, max(2, n // 4))
            phase = rng.random() * 2 * np.pi
            h_p += (scale / np.sqrt(n_modes)) * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif r < 0.4:
        h_p = h.copy()
        n_blocks = rng.integers(2, max(3, n // 15))
        for _ in range(n_blocks):
            bsize = rng.integers(3, max(4, n // 5))
            start = rng.integers(0, max(1, n - bsize))
            h_p[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 2
    elif r < 0.6:
        h_p = h + scale * rng.standard_normal(n)
    elif r < 0.8:
        h_p = h.copy()
        n_push = max(3, int(n * 0.1))
        push_idx = rng.choice(n, min(n_push, n), replace=False)
        for idx in push_idx:
            if rng.random() < 0.5:
                h_p[idx] = min(1.0, h_p[idx] + scale * 2.5)
            else:
                h_p[idx] = max(0.0, h_p[idx] - scale * 2.5)
    else:
        h_p = h + scale * 0.3 * rng.standard_normal(n)
    return project_feasible(h_p, n)

    h_cur, c5_cur = lbfgsb_optimize(best_h, T=80.0, penalty=20.0, maxiter=1500)

    h_cur, c5_cur = lbfgsb_optimize(best_h, T=300.0, penalty=30.0, maxiter=1500)
    if c5_cur < best_c5:

    if c5_cur < best_c5:
        best_h, best_c5, n_best = h_cur, c5_cur, len(h_cur)
    if time.time() - t0 > budget_s - 200:
        return best_h, best_c5, n_best

        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)

    for attempt in range(n_fine_perturb):
        if time.time() - t0 > budget_s - 60:
            break
        scale = 0.002 + 0.012 * (1 - attempt / n_fine_perturb) ** 2
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        T_fine = 500.0 + 5000.0 * (1 - attempt / n_fine_perturb)
        h_cur, c5_cur = lbfgsb_optimize(h_pert, T=T_fine, penalty=50.0, maxiter=600)
        if c5_cur < best_c5 - 1e-13:
