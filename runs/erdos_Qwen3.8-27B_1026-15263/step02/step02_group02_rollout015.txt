```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_c5_vec(h):
    """Compute all correlation values (not just max)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            vals = h[mask]
            room = 1.0 - vals
            step = min(diff / np.sum(mask), room.min() * 0.99)
            h[mask] += step
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            vals = h[mask]
            step = min(-diff / np.sum(mask), vals.min() * 0.99)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_bin, n, max_flips=5000, rng=None):
    """Simulated annealing on binary h"""
    if rng is None:
        rng = np.random.default_rng()
    h = h_bin.copy()
    c5_vec = compute_c5_vec(h)
    current_max = c5_vec.max()
    
    best_h = h.copy()
    best_c5 = current_max
    
    T = 0.1
    T_min = 0.001
    cooling = 0.999
    
    k_max = n - 1  # max lag index in correlate output
    n_lags = 2 * n - 1
    
    for step in range(max_flips):
        j = rng.integers(0, n)
        h[j] = 1.0 - h[j]
        
        # Recompute correlation (expensive but simple)
        c5_vec_new = compute_c5_vec(h)
        new_max = c5_vec_new.max()
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            c5_vec = c5_vec_new
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            h[j] = 1.0 - h[j]  # revert
        
        T *= cooling
        if T < T_min:
            T = T_min
    
    return best_h, best_c5

def active_set_optimize(h, n, n_active=10, maxiter=2000):
    """Optimize focusing on the top n_active lags"""
    c5_vec = compute_c5_vec(h)
    # Find the top lags
    top_indices = np.argsort(c5_vec)[-n_active:]
    
    def obj(x):
        c = compute_c5_vec(x)
        # Soft max over active set
        active = c[top_indices]
        # Use softmax with high temperature for focus
        T = 0.01
        shift = np.max(active)
        return float(shift + np.log(np.sum(np.exp((active - shift) / T))) * T)
    
    # Actually, let's just use the max of the active set as objective
    def obj_simple(x):
        c = compute_c5_vec(x)
        active = c[top_indices]
        m = np.max(active)
        # Soft smoothing
        return float(m + 0.01 * np.log(np.sum(np.exp((active - m) / 0.01))))
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_simple, h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    
    if result.success or result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h, compute_c5_from_h(h)

def multi_scale_active(h, n, n_active_list, maxiter_list):
    """Multi-scale: start with fewer active lags, increase"""

        if c5 < best_c5:

    return best, best_c5

def anti_periodic_init(n, rng):
    """Create anti-periodic h: h[i+n//2] = 1 - h[i]"""
    half = n // 2
    # Random first half
    h_first = rng.random(half)
    # Make it roughly binary
    h_first = (h_first > 0.3).astype(float) * 0.8 + (h_first > 0.6).astype(float) * 0.2
    h = np.zeros(n)
    if n % 2 == 0:
        h[:half] = h_first
        h[half:] = 1.0 - h_first
    else:
        h[:half] = h_first
        h[half:n] = 1.0 - h_first[:n-half]
    # Fix the sum
    target = n / 2.0
    # For anti-periodic, sum is automatically n/2 (if n is even)
    # For odd n, we need to adjust
    if n % 2 != 0:
        # Adjust middle element
        h[half] = 0.5
    h = project_feasible(h, n)
    return h

def sine_init(n, freq, rng):
    """Initialize with a sinusoidal pattern"""
    x = np.arange(n) / n
    h = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
    # Make it more binary

    h = project_feasible(h, n)
    return h

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    
    def slsqp_obj(x):
        return x[-1]
    
    def slsqp_ineq(x):
        h = x[:-1]
        t = x[-1]
        n_local = len(h)
        dx = 2.0 / n_local
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * dx
    
    def slsqp_eq(x):
        h = x[:-1]
        return np.sum(h) - len(h) / 2.0
    
    def slsqp_eq_jac(x):
        n_loc = len(x) - 1
        jac = np.zeros(len(x))
        jac[:n_loc] = 1.0
        return jac
    
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            max_step = (1.0 - h[mask]).min()
            h[mask] += min(step, max_step * 0.99)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            max_step = h[mask].min()
            h[mask] -= min(step, max_step * 0.99)
    return np.clip(h, 0.0, 1.0)

def active_set_grad(h, active_full_indices, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for ki in active_full_indices:
        k = ki - (n - 1)
        if k >= 0:
            nk = n - k
            grad[:nk] += dx * (1.0 - h[k:k+nk])
            grad[k:] += -dx * h[:nk]
        else:
            kp = -k
            nk = n - kp
            grad[kp:] += dx * (1.0 - h[:nk])
            grad[:nk] += -dx * h[kp:]
    return grad

def projected_gradient(h_init, n, n_steps=5000, lr_init=0.5, active_margin=0.005, active_max=20):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    
    for step in range(n_steps):
        c5_vec = compute_c5_vec(h)
        cmax = c5_vec.max()
        threshold = cmax - active_margin
        active_idx = np.where(c5_vec >= threshold)[0]
        
        if len(active_idx) > active_max:
            top_order = np.argsort(c5_vec[active_idx])[-active_max:]
            active_idx = active_idx[top_order]
        elif len(active_idx) == 0:
            active_idx = np.argsort(c5_vec)[-5:]
        
        grad = active_set_grad(h, active_idx, n)
        
        # Adaptive learning rate
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        lr = lr_init / (1.0 + 0.005 * step)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * (2.0 / len(h))
    
    def eq(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0
    
    def eq_jac(x):
        jacc = np.zeros(len(x))
        jacc[:-1] = 1.0
        return jacc
    
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def binary_sa(h_init, n, max_flips=5000, rng=None, T_init=0.05):
    """Simulated annealing on binary representation"""
    if rng is None:
        rng = np.random.default_rng()
    h = h_init.copy()
    # Make it binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_feasible(h, n)
    
    c5_vec = compute_c5_vec(h)
    cur_max = c5_vec.max()
    best_h = h.copy()
    best_c5 = cur_max
    
    T = T_init
    for step in range(max_flips):
        j = rng.integers(0, n)
        h[j] = 1.0 - h[j]
        
        # Recompute (fast for n=270)
        c5_vec_new = compute_c5_vec(h)
        new_max = c5_vec_new.max()
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta * 100 / max(T, 1e-10)):
            cur_max = new_max
            c5_vec = c5_vec_new
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            h[j] = 1.0 - h[j]
        
        T *= 0.9995
        if T < 0.001:
            T = 0.001
    
    # Project back and refine
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_from_h(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            max_step = (1.0 - h[mask]).min()
            h[mask] += min(step, max_step * 0.99)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            max_step = h[mask].min()
            h[mask] -= min(step, max_step * 0.99)
    return np.clip(h, 0.0, 1.0)


def active_set_grad(h, active_full_indices, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for ki in active_full_indices:
        k = ki - (n - 1)
        if k >= 0:
            nk = n - k
            if nk <= 0:
                continue
            grad[:nk] += dx * (1.0 - h[k:k + nk])
            grad[k:] += -dx * h[:nk]
        else:
            kp = -k
            nk = n - kp
            if nk <= 0:
                continue
            grad[kp:] += dx * (1.0 - h[:nk])
            grad[:nk] += -dx * h[kp:]
    return grad


def projected_gradient_mom(h_init, n, n_steps=10000, lr_init=0.3, mom=0.85, active_max=15, margin=0.005):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    for step in range(n_steps):
        c5_vec = compute_c5_vec(h)
        cmax = c5_vec.max()
        threshold = cmax - margin
        active_idx = np.where(c5_vec >= threshold)[0]
        if len(active_idx) > active_max:
            order = np.argsort(c5_vec[active_idx])[-active_max:]
            active_idx = active_idx[order]
        elif len(active_idx) == 0:
            active_idx = np.argsort(c5_vec)[-5:]
        grad = active_set_grad(h, active_idx, n)
        lr = lr_init / (1.0 + 0.008 * step)
        v = mom * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            v *= 0.3
        h = h_new
    return best_h, best_c5


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)

    def _slsqp_obj(x):
        return x[-1]

    def _slsqp_ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * (2.0 / len(h))

    def _slsqp_eq(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

    def _slsqp_eq_jac(x):
        jacc = np.zeros(len(x))
        jacc[:-1] = 1.0
        return jacc

    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    cons = [{'type': 'ineq', 'fun': _slsqp_ineq}, {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    result = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def binary_sa(h_init, n, max_flips=5000, rng=None, T_init=0.03):
    if rng is None:
        rng = np.random.default_rng()
    h = h_init.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_feasible(h, n)
    c5_vec = compute_c5_vec(h)
    cur_max = c5_vec.max()
    best_h = h.copy()
    best_c5 = cur_max
    T = T_init
    for step in range(max_flips):
        j = rng.integers(0, n)
        h[j] = 1.0 - h[j]
        c5_vec_new = compute_c5_vec(h)
        new_max = c5_vec_new.max()
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta * 200.0 / max(T, 1e-12)):
            cur_max = new_max
            c5_vec = c5_vec_new
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            h[j] = 1.0 - h[j]
        T *= 0.9997
        if T < 0.002:
            T = 0.002
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_from_h(best_h)


def pair_flip_search(h_init, n, n_rounds=5000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    h = h_init.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    for _ in range(n_rounds):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        c5_new = compute_c5_from_h(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_from_h(best_h)


def lbfgsb_smooth(h_init, n, T_smooth, penalty, maxiter=2000):
    def _smooth_obj(x):
        c = compute_c5_vec(x)
        cmax = c.max()
        shifted = T_smooth * (c - cmax)
        shifted = np.clip(shifted, -400, 400)
        lse = cmax + np.log(np.sum(np.exp(shifted))) / T_smooth
        sv = np.sum(x) - n / 2.0
        return float(lse + penalty * sv * sv)

    bounds = [(0.0, 1.0)] * n
    result = minimize(_smooth_obj, h_init, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def anti_periodic_init(n, rng):
    half = n // 2
    h_first = rng.random(half)
    h_first = np.where(h_first > 0.4, 0.8, 0.2)
    h = np.zeros(n)
    if n % 2 == 0:
        h[:half] = h_first
        h[half:] = 1.0 - h_first
    else:
        h[:half] = h_first
        h[half:n] = 1.0 - h_first[:n - half]
        h[half] = 0.5
    h = project_feasible(h, n)
    return h


def block_flip(h, n, bsize, rng):
    h = h.copy()
    start = rng.integers(0, n - bsize + 1)
    h[start:start + bsize] = 1.0 - h[start:start + bsize]
    return h


def run(seed=42, budget_s=1300, **kwargs):

        if c5_pg < best_c5:

            best_c5 = c5_pg

    # Phase 6: PG again after binary search

        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg

    # Phase 7: L-BFGS-B smooth multi-T
    if elapsed() < budget_s - 150:

    # Phase 8: Multiple perturbation rounds
    n_perturb = 50

    best_c5 = compute_c5_from_h(best_h)
```
