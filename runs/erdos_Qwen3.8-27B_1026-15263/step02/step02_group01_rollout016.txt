At the optimum, suppose lags {k1, k2, ..., kM} achieve the maximum. Then by the KKT conditions (for the max problem), we have:
- The gradient of the objective is a convex combination of the gradients of the active constraints
- Each active constraint is tight: f_kj(h*) = C*

So at the optimum, the weighted average gradient of the active lags is zero (or orthogonal to the feasible set).

This suggests an algorithm:
1. Find the active set (top M lags)
2. Set up the problem: minimize max_{j in active} f_jj(h) subject to constraints
3. Since we want all active f_jj to be equal at the optimum, we can instead minimize:
   (1/M) * ∑_{j in active} f_jj(h) + λ * ∑_{j in active} (f_jj(h) - avg)²
   The first term minimizes the average, the second term equalizes.
   
   Or more simply: minimize ∑_{j in active} f_jj(h)² = ∑ f_jj² which encourages equalization.

Actually, the simplest approach is: minimize the sum of the top M correlations. This is a differentiable objective (if M is fixed and the top M don't change).

Let me implement this:

            for j in range(n_h - abl):

    return obj, g

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    w = exp_v / s
    val = m + np.log(s / len(corr)) / alpha
    g = np.zeros(n)
    threshold = 1.0 / (len(corr) * 500)
    active_idx = np.where(w > threshold)[0]
    for m_idx in active_idx:
        wm = w[m_idx]
        if m_idx >= n - 1:
            lag = m_idx - (n - 1)
            nr = n - lag
            g[:nr] += wm * (1.0 - h[lag:n])
            g[lag:] -= wm * h[:nr]
        else:
            abl = (n - 1) - m_idx
            nr = n - abl
            g[abl:] += wm * (1.0 - h[:nr])
            g[:nr] -= wm * h[abl:]
    return val, g

def active_set_grad(h, active_lags):
    n = len(h)
    g = np.zeros(n)
    for m_idx in active_lags:
        if m_idx >= n - 1:
            lag = m_idx - (n - 1)
            nr = n - lag
            g[:nr] += (1.0 - h[lag:n])
            g[lag:] -= h[:nr]
        else:
            abl = (n - 1) - m_idx
            nr = n - abl
            g[abl:] += (1.0 - h[:nr])
            g[:nr] -= h[abl:]
    return g

def projected_gradient(h0, n, target_sum, alpha, max_iters, lr_init, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = lr_init
    patience = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        val, g = smooth_max_and_grad(h, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        step = lr
        improved = False
        for _ls in range(40):
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            val_new, _ = smooth_max_and_grad(h_new, alpha)
            if val_new < val - 1e-14 * step * g_norm * 0.1:
                improved = True
                break
            step *= 0.5
        if improved:
            h = h_new
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            lr = min(lr * 1.05, 2.0)
        else:
            lr *= 0.6
            patience += 1
        if patience > 200:
            break
    return best_h, best_c5

def active_set_optimize(h0, n, target_sum, M, max_iters, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.02
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-M:]
        top_idx = np.sort(top_idx)
        g = active_set_grad(h, top_idx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        step = lr
        improved = False
        for _ls in range(30):
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            corr_new = compute_corr(h_new)
            obj = np.sum(corr_new[top_idx])
            obj_old = np.sum(corr[top_idx])
            if obj < obj_old - 1e-14 * step * g_norm * 0.05:
                improved = True
                break
            step *= 0.5
        if improved:
            h = h_new
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.7
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, rng, fine=False):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if fine:
        deltas = [-0.005, -0.002, -0.001, -0.0005, 0.0005, 0.001, 0.002, 0.005]
    else:
        deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                if h[i] + d < 0.0 or h[i] + d > 1.0:
                    continue
                h_new = h.copy()
                h_new[i] = h[i] + d
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_smooth(h0, n, target_sum, alpha, maxiter, time_limit):
    t0 = time.time()
    def obj(x):
        if time.time() - t0 > time_limit:
            raise RuntimeError("timeout")
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        if time.time() - t0 > time_limit:
            raise RuntimeError("timeout")
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts_init = best_n / 2.0
            best_h = project_to_sum(best_h, ts_init)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts_cur = n_cur / 2.0

    def time_left():
        return budget_s - (time.time() - t_start)

    phase1_budget = budget_s * 0.35
    phase2_budget = budget_s * 0.25
    phase3_budget = budget_s * 0.25
    phase4_budget = budget_s * 0.15

    # Phase 1: Optimization at current resolution with progressive alpha
    alphas_p1 = [1.0, 2.0, 4.0, 8.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0]
    t_p1_start = time.time()
    for alpha in alphas_p1:
        if time.time() - t_p1_start > phase1_budget:
            break
        remain = phase1_budget - (time.time() - t_p1_start)
        if remain < 15:
            break
        max_it = max(200, int(remain * 2))
        h_opt, c5_opt = projected_gradient(best_h, n_cur, ts_cur, alpha, max_it, 0.05, remain - 5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Active set optimization
    if time_left() > 60:
        for M in [3, 5, 8, 12, 20]:
            if time_left() < 30:
                break
            remain = min(40, time_left() - 10)
            h_as, c5_as = active_set_optimize(best_h, n_cur, ts_cur, M, 2000, remain)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()

    # Phase 2: SLSQP refinement at current resolution
    t_p2_start = time.time()
    alphas_p2 = [100.0, 300.0, 800.0, 2000.0, 5000.0]
    for alpha in alphas_p2:
        if time.time() - t_p2_start > phase2_budget:
            break
        remain = phase2_budget - (time.time() - t_p2_start)
        if remain < 20:
            break
        maxit = max(100, int(remain * 1.5))
        h_slq = slsqp_smooth(best_h, n_cur, ts_cur, alpha, maxit, remain - 5)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Coordinate descent polish at current resolution
    if time_left() > 40:
        h_cd, c5_cd = coordinate_descent(best_h, n_cur, ts_cur, 30, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Try higher resolution
    for n_new in [480, 720]:
        if time_left() < 80:
            break
        if n_new % 2 != 0:
            continue
        ts_new = n_new / 2.0
        if abs(ts_new - round(ts_new)) > 1e-9:
            continue
        h_up = upsample(best_h, n_new)
        h_up = project_to_sum(h_up, ts_new)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_new
        n_cur = n_new
        ts_cur = ts_new

        # Optimize at new resolution
        alphas_p3 = [5.0, 20.0, 50.0, 150.0, 400.0, 1000.0]
        for alpha in alphas_p3:
            if time_left() < 40:
                break
            remain = min(60, time_left() - 20)
            max_it = max(500, int(remain * 3))
            h_opt, c5_opt = projected_gradient(best_h, n_cur, ts_cur, alpha, max_it, 0.03, remain - 5)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_cur

        if time_left() > 50:
            for M in [5, 10, 15]:
                if time_left() < 30:
                    break
                remain = min(30, time_left() - 10)
                h_as, c5_as = active_set_optimize(best_h, n_cur, ts_cur, M, 3000, remain)
                if c5_as < best_c5:
                    best_c5 = c5_as
                    best_h = h_as.copy()
                    best_n = n_cur

    # Phase 4: Final polish
    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0

        # High alpha projected gradient
        for alpha in [2000.0, 5000.0, 10000.0]:
            if time_left() < 20:
                break
            remain = min(30, time_left() - 10)
            h_pg, c5_pg = projected_gradient(best_h, n_f, ts_f, alpha, 5000, 0.01, remain)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        # SLSQP with very high alpha
        if time_left() > 30:
            h_slq = slsqp_smooth(best_h, n_f, ts_f, 10000.0, maxiter=500, time_limit=min(40, time_left() - 10))
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

        # Fine coordinate descent
        if time_left() > 20:
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 50, rng, fine=True)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        # Final active set
        if time_left() > 15:
            for M in [3, 5]:
                if time_left() < 10:
                    break
                h_as, c5_as = active_set_optimize(best_h, n_f, ts_f, M, 5000, min(10, time_left() - 5))
                if c5_as < best_c5:
                    best_c5 = c5_as
                    best_h = h_as.copy()

    # Final projection and verification
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
