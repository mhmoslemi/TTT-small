            starts.append(project_feasible(np.clip(parent + rng.randn(n) * 0.06, 0, 1), n))
            starts.append(project_feasible(np.clip(parent + rng.randn(n) * 0.15, 0, 1), n))
        else:
            xp = np.linspace(0, 2, len(parent), endpoint=False)
            f = interp1d(xp, parent, kind='cubic', fill_value='extrapolate', bounds_error=False)
            starts.append(project_feasible(np.clip(f(x), 0, 1), n))
            starts.append(project_feasible(np.clip(f(x) + rng.randn(n) * 0.1, 0, 1), n))
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.48)
        h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
        starts.append(project_feasible(np.clip(h, 0, 1), n))
    for _ in range(2):
        f1, f2 = rng.choice([1, 2, 3, 4], 2)
        p1, p2 = rng.uniform(0, 2 * np.pi, 2)
        h = 0.5 + 0.28 * np.sin(2*np.pi*f1*x/2.0 + p1) + 0.18 * np.sin(2*np.pi*f2*x/2.0 + p2)
        starts.append(project_feasible(np.clip(h, 0, 1), n))
    for _ in range(2):
        h = np.zeros(n)
        n_ones = n // 2
        pos = rng.choice(n, n_ones, replace=False)
        h[pos] = 1.0
        starts.append(project_feasible(h, n))
    starts.append(project_feasible(rng.rand(n), n))
    starts.append(project_feasible(np.linspace(0.1, 0.9, n), n))
    return starts[:num]

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        best_h = project_feasible(parent_h.copy(), len(parent_h))
        best_c5 = evaluate_c5(best_h)
        n = len(best_h)
    else:
        n = 77
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    p1_end = t0 + budget_s * 0.28
    p2_end = t0 + budget_s * 0.55
    p3_end = t0 + budget_s * 0.75
    p4_end = t0 + budget_s * 0.93

    # Phase 1: SLSQP from parent with fine tau
    tau1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    h_opt, c5 = run_single_slsqp(best_h, n, tau1, 80, p1_end)
    if c5 < best_c5:
        best_c5, best_h = c5, h_opt.copy()

    if time.time() < p1_end:
        starts = generate_starts(n, parent_h, rng, num=5)
        for h0 in starts:
            if time.time() > p1_end:
                break
            tau_short = [0.05, 0.02, 0.01, 0.005, 0.002]
            h_opt, c5 = run_single_slsqp(h0, n, tau_short, 50, p1_end)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()

    # Phase 2: Subgradient on true objective
    if time.time() < p2_end:
        h_sg, c5_sg = subgradient_phase(best_h, n, p2_end, max_iter=800)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()
        tau_refine = [0.002, 0.001, 0.0005, 0.0002]
        h_opt, c5 = run_single_slsqp(best_h, n, tau_refine, 60, p2_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 3: Focused coordinate descent
    if time.time() < p3_end:
        h_cd, c5_cd = coordinate_descent_focused(best_h, n, p3_end, max_sweeps=40)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()
        if time.time() < p3_end:
            tau_fin = [0.001, 0.0005, 0.0002]
            h_opt, c5 = run_single_slsqp(best_h, n, tau_fin, 100, p3_end)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()

    # Phase 4: Random perturbations with quick SLSQP
    if time.time() < p4_end:
        for _ in range(300):
            if time.time() > p4_end:
                break
            frac = (time.time() - t0) / budget_s
            noise = 0.002 + 0.025 * max(0.0, 1.0 - 1.5 * frac)
            perturb = project_feasible(best_h + rng.randn(n) * noise, n)
            tau_q = [0.002, 0.001, 0.0005]
            h_opt, c5 = run_single_slsqp(perturb, n, tau_q, 30, p4_end)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()

    # Phase 5: Final ultra-fine polishing
    fin_end = t0 + budget_s * 0.99
    if best_h is not None and time.time() < fin_end:
        while time.time() < fin_end - 5.0:
            h_opt, c5 = run_single_slsqp(best_h, n, [0.0001, 0.00005], 200, fin_end)
            if c5 >= best_c5 - 1e-15:
                break
            best_c5, best_h = c5, h_opt.copy()
        if time.time() < fin_end:
            h_cd, c5_cd = coordinate_descent_focused(best_h, n, fin_end, max_sweeps=10)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
