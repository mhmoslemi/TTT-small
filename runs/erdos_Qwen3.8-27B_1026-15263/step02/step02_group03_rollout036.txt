from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def slsqp_refine(h, n, tau, max_iter):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max(corr_dx(x), tau)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coordinate_descent(h, n, sweeps=3, step_list=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    if step_list is None:
        step_list = [0.01, 0.02, 0.03, 0.05]
    for sweep in range(sweeps):
        improved = False
        for i in range(n):
            for d in step_list:
                for sign in [-1, 1]:
                    delta = sign * d
                    if best[i] + delta < 0 or best[i] + delta > 1:
                        continue
                    h_new = best.copy()
                    h_new[i] += delta
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-12:

                        best_c5 = c5
                        improved = True
        if not improved:

    return best, best_c5

    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def make_obj_fn(n, tau, active=None):
    def obj_fn(x):
        cv = corr_dx(x)
        if active is not None:
            cv = cv[active]
        return smooth_max(cv, tau)
    return obj_fn

def make_cons_fn(n):
    def cons_fn(x):
        return np.sum(x) - n / 2.0
    return cons_fn

def slsqp_refine(h, n, tau, max_iter, top_m=None):
    bounds = [(0.0, 1.0)] * n
    active = None
    if top_m is not None and top_m < 2 * n - 1:
        cv = corr_dx(h)
        active = np.argsort(cv)[-top_m:]
    obj_fn = make_obj_fn(n, tau, active)
    cons_fn = make_cons_fn(n)
    cons = [{'type': 'eq', 'fun': cons_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-11})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_multi_tau(h, n, tau_list, max_iter_each):
    current = h.copy()
    for tau in tau_list:
        current = slsqp_refine(current, n, tau, max_iter_each)
    return current

def pair_sa(h_init, n, rng, n_iter, t_start, t_end, min_step=0.001, max_step=0.04):
    current = h_init.copy()
    current_c5 = evaluate_c5(current)
    best = current.copy()
    best_c5 = current_c5
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = t_start * (t_end / t_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(min_step, max_step)
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        delta = sign * d
        if current[i] + delta < 0.0 or current[i] + delta > 1.0:
            continue
        if current[j] - delta < 0.0 or current[j] - delta > 1.0:
            continue
        new_h = current.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = evaluate_c5(new_h)
        if new_c5 <= current_c5:
            accept = True
        elif T > 1e-10:
            accept = rng.rand() < np.exp(-(new_c5 - current_c5) / T)
        else:
            accept = False
        if accept:
            current = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best = current.copy()
                best_c5 = current_c5
    return best, best_c5

def coordinate_descent(h, n, sweeps=3, step_list=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    if step_list is None:
        step_list = [0.01, 0.02, 0.05]
    for sweep in range(sweeps):
        improved = False
        for i in range(n):
            for d in step_list:
                for sgn in (-1.0, 1.0):
                    delta = sgn * d
                    if best[i] + delta < 0.0 or best[i] + delta > 1.0:
                        continue
                    h_new = best.copy()
                    h_new[i] += delta
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-12:
                        best = h_new
                        best_c5 = c5
                        improved = True
        if not improved:
            break
    return best, best_c5

def pair_descent(h, n, n_trials, step_list, rng):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice(step_list)
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        delta = sign * d
        if best[i] + delta < 0.0 or best[i] + delta > 1.0:
            continue
        if best[j] - delta < 0.0 or best[j] - delta > 1.0:
            continue
        h_new = best.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new
            best_c5 = c5
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        n = len(parent_h)
        best_h = project_feasible(parent_h, n)
        best_c5 = evaluate_c5(best_h)
    else:
        n = 77
        x = np.linspace(0, 2, n, endpoint=False)
        best_h = np.clip(0.5 + 0.4 * np.sin(np.pi * 2.0 * x), 0, 1)
        best_h = project_feasible(best_h, n)
        best_c5 = evaluate_c5(best_h)

    # Phase 1: Initial SLSQP (10%)
    phase1_end = start_time + budget_s * 0.10
    h_opt = slsqp_multi_tau(best_h, n, [0.05, 0.02, 0.01], 80)
    c5_opt = evaluate_c5(h_opt)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: Iterated Local Search (45%)
    phase2_end = start_time + budget_s * 0.55
    il_best = best_h.copy()
    il_best_c5 = best_c5
    perturbation = 0.012
    accept_temp = 0.002

    while time.time() < phase2_end:
        noisy = il_best + rng.randn(n) * perturbation
        noisy = project_feasible(noisy, n)
        h_opt = slsqp_multi_tau(noisy, n, [0.004, 0.001], 70)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt <= il_best_c5:
            il_best = h_opt
            il_best_c5 = c5_opt
            perturbation = max(0.003, perturbation * 0.92)
        else:
            delta_c5 = c5_opt - il_best_c5
            if rng.rand() < np.exp(-delta_c5 / accept_temp):
                il_best = h_opt
                il_best_c5 = c5_opt
            else:
                perturbation = min(0.06, perturbation * 1.08)
        if il_best_c5 < best_c5:
            best_c5 = il_best_c5
            best_h = il_best.copy()

    # Phase 3: Simulated Annealing with pair transfers (25%)
    phase3_end = start_time + budget_s * 0.80
    if time.time() < phase3_end:
        t0 = time.time()
        for _ in range(30):
            _ = evaluate_c5(best_h)
        time_per_eval = (time.time() - t0) / 30.0
        time_per_iter = max(time_per_eval * 2.5, 5e-5)
        available = phase3_end - time.time()
        n_sa = int(available / time_per_iter)
        n_sa = max(5000, min(n_sa, 5000000))

        t_sa_init = 0.006
        t_sa_end = 0.00005
        chunk = min(n_sa, 150000)
        sa_best = best_h.copy()
        sa_best_c5 = best_c5

        while time.time() < phase3_end:
            this_chunk = min(chunk, n_sa)
            h_sa, c5_sa = pair_sa(sa_best, n, rng, this_chunk, t_sa_init, t_sa_end)
            if c5_sa < sa_best_c5:
                sa_best_c5 = c5_sa
                sa_best = h_sa
            h_ref = slsqp_refine(sa_best, n, 0.002, 80)
            c5_ref = evaluate_c5(h_ref)
            if c5_ref < sa_best_c5:
                sa_best_c5 = c5_ref
                sa_best = h_ref
            if sa_best_c5 < best_c5:
                best_c5 = sa_best_c5
                best_h = sa_best.copy()
            t_sa_init = max(t_sa_end, t_sa_init * 0.35)
            n_sa -= this_chunk

    # Phase 4: Active-set SLSQP (7%)
    phase4_end = start_time + budget_s * 0.87
    if time.time() < phase4_end:
        for m in (12, 18, 8, 25):
            if time.time() > phase4_end:
                break
            h_opt = slsqp_refine(best_h, n, 0.0015, 120, top_m=m)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_h = slsqp_refine(best_h, n, 0.001, 80)
                c5_opt = evaluate_c5(best_h)
                if c5_opt < best_c5:
                    best_c5 = c5_opt

    # Phase 5: Pair descent + coordinate descent (7%)
    phase5_end = start_time + budget_s * 0.94
    if time.time() < phase5_end:
        for steps in ([0.02, 0.01, 0.005], [0.005, 0.002, 0.001], [0.002, 0.001, 0.0005]):
            if time.time() > phase5_end:
                break
            h_pd, c5_pd = pair_descent(best_h, n, 3000, steps, rng)
            if c5_pd < best_c5:
                best_c5 = c5_pd
                best_h = h_pd
            h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=2, step_list=steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 6: Final polish (6%)
    if time.time() < start_time + budget_s * 0.995:
        h_opt = slsqp_refine(best_h, n, 0.0008, 200)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=4, step_list=[0.001, 0.0005, 0.0002])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
