So: ∂c/∂h[j] = (1-h[j+L]) * [j < n-L] - h[j-L] * [j ≥ L]

For L = 0:
c = sum_i h[i](1-h[i]) = sum h - sum h²
∂c/∂h[j] = 1 - 2h[j]

Great. Now for the subgradient method:

At each step:
1. Compute all correlations, find k* = argmax
2. Determine L = |k* - (n-1)| (the absolute lag)
3. Compute gradient of c[k*] w.r.t. h
4. Take step: h_new = h - step_size * grad
5. Project to feasible set (0 ≤ h ≤ 1, sum(h) = n/2)
6. Check if improved

def compute_gradient_at_max_lag(h, n, top_k=1):
    """Compute gradient of the sum of top-k correlations."""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    
    # Find top-k lags
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)

        L = abs(k - (n - 1))
        if L == 0:
            grad += 1.0 - 2.0 * h
        else:

            # grad[j] = (1-h[j+L]) for j < n-L, minus h[j-L] for j >= L
            j1 = np.arange(0, n - L)
            grad[j1] += (1.0 - h[j1 + L])
            j2 = np.arange(L, n)
            grad[j2] -= h[j2 - L]

def penalty_objective(h, n, T, mu):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    excess = np.maximum(0, corr - T + mu)
    return np.sum(excess**2)

def penalty_gradient(h, n, T, mu):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    active = (corr - T + mu) > 0
    if not np.any(active):
        return np.zeros(n)
    # For each active lag k, compute gradient of corr_k
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        L = abs(k - (n - 1))
        if L == 0:
            grad += 2 * (corr[k] - T + mu) * dx * (1 - 2*h)
        else:
            j1 = np.arange(0, n - L)
            grad[j1] += 2 * (corr[k] - T + mu) * dx * (1 - h[j1+L])
            j2 = np.arange(L, n)
            grad[j2] -= 2 * (corr[k] - T + mu) * dx * h[j2-L]
    return grad
