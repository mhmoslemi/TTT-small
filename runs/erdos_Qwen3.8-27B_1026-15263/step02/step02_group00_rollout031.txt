```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)

    # For numerical stability
    max_shifted = np.max(shifted)
    log_sum = m_val + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha
```

```python
import numpy as np
from scipy.optimize import minimize, bounds as scipy_bounds
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    max_s = float(np.max(shifted))
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)
    obj = m_val + (max_s + np.log(np.sum(exp_s))) / alpha
    
    # Gradient
    grad = np.zeros(n)
    # For each lag l = k - (n-1), k in [0, 2n-2]
    # d corr[k] / d h[j] = (1-h[j+l]) if valid - h[j-l] if valid
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        l = k - (n - 1)
        # First term: (1 - h[j+l]) for j where 0 <= j+l < n
        j_min = max(0, -l)
        j_max = min(n, n - l)  # exclusive
        if j_max > j_min:
            grad[j_min:j_max] += wk * (1.0 - h[j_min + l:j_max + l])
        # Second term: -h[j-l] for j where 0 <= j-l < n
        j_min2 = max(0, l)
        j_max2 = min(n, n + l)  # exclusive, since j-l < n => j < n+l
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= wk * h[j_min2 - l:j_max2 - l]
    
    return obj, grad

    def objective(x):
        # x has length n, last variable is eliminated
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target_sum - np.sum(x)
        obj, _ = smooth_max_and_grad(h, alpha)
        return obj
    
    def gradient(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target_sum - np.sum(x)
        _, g = smooth_max_and_grad(h, alpha)
        # Chain rule: d obj / d x[j] = g[j] - g[n-1] for j < n-1
        gx = g[:n-1] - g[n-1]
        return gx
    
    # Bounds for x[0:n-1]: need 0 <= x[j] <= 1 and 0 <= target_sum - sum(x) <= 1
    # So sum(x) is in [target_sum - 1, target_sum]
    # Individual bounds: 0 <= x[j] <= 1

    # For safety, use [0, 1] for each and let the optimizer handle it
    
    x0 = h0[:n-1].copy()
    bnds = [(0.0, 1.0)] * (n - 1)
    
    try:
```

```python
def stochastic_smooth_max(h, alpha, n_lags=None, rng=None):
    n = len(h)
    total_lags = 2 * n - 1
    if n_lags is None or n_lags >= total_lags:
        return smooth_max_and_grad(h, alpha)
    # Sample a subset of lags
    if rng is None:
        rng = np.random
    idx = rng.choice(total_lags, size=n_lags, replace=False)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_sub = corr[idx]
    m_val = float(np.max(corr_sub))
    shifted = alpha * (corr_sub - m_val)
    max_s = float(np.max(shifted))
    exp_s = np.exp(shifted - max_s)
    w_sub = exp_s / np.sum(exp_s)
    obj = m_val + (max_s + np.log(np.sum(exp_s))) / alpha
    # Gradient (only for sampled lags)
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        wk = w_sub[i]
        if wk < 1e-18:
            continue
        lag = k - (n-1)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        if j_max > j_min:
            grad[j_min:j_max] += wk * (1.0 - h[j_min + lag:j_max + lag])
        j_min2 = max(0, lag)
        j_max2 = min(n, n + lag)
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= wk * h[j_min2 - lag:j_max2 - lag]
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    max_s = float(np.max(shifted))
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)
    obj = m_val + (max_s + np.log(np.sum(exp_s))) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        if j_max > j_min:
            grad[j_min:j_max] += wk * (1.0 - h[j_min + lag:j_max + lag])
        j_min2 = max(0, lag)
        j_max2 = min(n, n + lag)
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= wk * h[j_min2 - lag:j_max2 - lag]
    return obj, grad

def lbfgs_optimize(h0, alpha, maxiter=5000, M_pen=500.0):
    n = len(h0)
    target_sum = n / 2.0
    def objective(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target_sum - np.sum(x)
        obj, _ = smooth_max_and_grad(h, alpha)
        hn1 = target_sum - np.sum(x)
        pen = 0.0
        if hn1 < 0.0:
            pen += M_pen * hn1 * hn1
        elif hn1 > 1.0:
            pen += M_pen * (hn1 - 1.0) * (hn1 - 1.0)
        return obj + pen
    def gradient(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target_sum - np.sum(x)
        _, g = smooth_max_and_grad(h, alpha)
        gx = g[:n-1] - g[n-1]
        hn1 = target_sum - np.sum(x)
        if hn1 < 0.0:
            gx += 2.0 * M_pen * hn1 * (-1.0)
        elif hn1 > 1.0:
            gx += 2.0 * M_pen * (hn1 - 1.0) * (-1.0)
        return gx
    x0 = h0[:n-1].copy()
    bnds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': maxiter * 4})
        h = np.empty(n)
        h[:n-1] = res.x
        h[n-1] = target_sum - np.sum(res.x)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        return h
    except Exception:
        return project_to_sum(h0, target_sum)

def fourier_h_from_coeff(n, coeffs, m_max):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, m_max + 1):
        h += coeffs[m-1, 0] * np.cos(2.0 * np.pi * m * x / 2.0)
        h += coeffs[m-1, 1] * np.sin(2.0 * np.pi * m * x / 2.0)
    return h

def h_to_fourier(h, m_max):
    n = len(h)
    coeffs = np.zeros((m_max, 2))
    x = np.linspace(0, 2.0, n, endpoint=False)
    h_ac = h - 0.5
    for m in range(1, m_max + 1):
        coeffs[m-1, 0] = (2.0 / n) * np.sum(h_ac * np.cos(2.0 * np.pi * m * x / 2.0))
        coeffs[m-1, 1] = (2.0 / n) * np.sum(h_ac * np.sin(2.0 * np.pi * m * x / 2.0))
    return coeffs

def fourier_optimize(h0, n, m_max, alpha, maxiter=8000):
    target_sum = n / 2.0
    coeffs0 = h_to_fourier(h0, m_max)
    def objective(c):
        h = fourier_h_from_coeff(n, c, m_max)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        obj, _ = smooth_max_and_grad(h, alpha)
        pen = 100.0 * (np.sum(np.minimum(h, 0.0)**2) + np.sum(np.maximum(h - 1.0, 0.0)**2))
        return obj + pen
    try:
        res = minimize(objective, coeffs0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': maxiter * 5})
        h = fourier_h_from_coeff(n, res.x, m_max)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        return h
    except Exception:
        return h0.copy()

def make_starts(n, target_sum, rng, parent_h, parent_n):
    starts = []
    x = np.linspace(0, 2.0, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for freq in range(1, 9):
        for amp in [0.15, 0.3, 0.4, 0.48]:
            h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
            starts.append(project_to_sum(h, target_sum))
            h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x / 2.0)
            starts.append(project_to_sum(h, target_sum))
    for nblks in range(1, 11):
        blk = n // nblks
        if blk < 1:
            continue
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i+blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    for _ in range(4):
        h = rng.uniform(0, 1, n)
        starts.append(project_to_sum(h, target_sum))
    if parent_h is not None and parent_n > 0:
        if parent_n == n:
            starts.append(parent_h.copy())
        else:
            x_old = np.linspace(0, 2.0, parent_n, endpoint=False)
            h_interp = np.interp(x, x_old, np.tile(parent_h, 2), period=2.0)
            starts.append(project_to_sum(h_interp, target_sum))
        parent_at_n = parent_h if parent_n == n else np.interp(x, np.linspace(0, 2.0, parent_n, endpoint=False), np.tile(parent_h, 2), period=2.0)
        parent_at_n = project_to_sum(parent_at_n, target_sum)
        for nl in [0.05, 0.1, 0.2, 0.35]:
            h_p = parent_at_n + rng.uniform(-nl, nl, n)
            starts.append(project_to_sum(h_p, target_sum))
    return starts

def binary_flip(h, target_sum, rng, max_rounds=60):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_trials = min(400, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        c5_cur = best_c5
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

        if n_tr % 2 != 0:

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
