import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = diff / count
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = -diff / count
            h[mask] = np.clip(h[mask] - step, 0.0, 1.0)
    return h

def c5_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return c_max + np.log(np.sum(np.exp(shifted))) / T

def smooth_obj(x, T, penalty):
    n = len(x)
    val = c5_smooth(x, T)
    sv = np.sum(x) - n / 2.0
    return val + penalty * sv * sv

def lbfgsb_opt(h_init, T, penalty, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h_init, args=(T, penalty),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_opt(h_init, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    res = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,

    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def binary_local_search(h_binary, n, max_iter=5000, rng=None):
    """Local search on binary sequences by moving ones."""
    if rng is None:
        rng = np.random.default_rng(0)
    target = n // 2
    state = h_binary.astype(int).copy()
    ones_pos = np.where(state == 1)[0]
    c5_cur = compute_c5_from_h(state.astype(float))
    best_state = state.copy()
    best_c5 = c5_cur
    best_ones = ones_pos.copy()
    
    for it in range(max_iter):
        # Choose a random one to move
        idx = rng.integers(len(ones_pos))
        old_pos = ones_pos[idx]
        # Choose a random zero to move to
        zeros = np.where(state == 0)[0]
        new_pos = zeros[rng.integers(len(zeros))]
        
        # Make the move
        state[old_pos] = 0
        state[new_pos] = 1
        ones_pos[idx] = new_pos
        ones_pos.sort()
        
        c5_new = compute_c5_from_h(state.astype(float))
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_state = state.copy()
            best_ones = ones_pos.copy()
            c5_cur = c5_new
        else:
            # Revert
            state[new_pos] = 0
            state[old_pos] = 1
            ones_pos[idx] = old_pos
            ones_pos.sort()
    
    return best_state.astype(float), best_c5

    c5_cur = compute_c5_from_h(state.astype(float))

    best_ones = ones_pos.copy()
    
    n_iter = max_iter

        old_pos = ones_pos[idx]
        zeros = np.where(state == 0)[0]
        new_pos = zeros[rng.integers(len(zeros))]
        
        state[old_pos] = 0
        state[new_pos] = 1
        ones_pos[idx] = new_pos
        ones_pos.sort()
        
        c5_new = compute_c5_from_h(state.astype(float))
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):

            if c5_new < best_c5:
                best_c5 = c5_new
                best_state = state.copy()
                best_ones = ones_pos.copy()
        else:

            state[old_pos] = 1

- Our problem: minimize max_k sum_j h_j * (1 - h_{j+k}) = max_k (sum_j h_{j+k} - R(k))

- R(k) is the standard autocorrelation

So our objective is max_k (S(k) - R(k)) where S(k) = sum of h in the valid range for lag k.

This is NOT the same as minimizing max_k R(k) or max_k |R(k)|. It's a different objective.

However, for a "balanced" sequence where the ones are spread out, S(k) ≈ n/2 * (n-k)/n for small k. So S(k) - R(k) ≈ n/2 * (1 - k/n) - R(k).
