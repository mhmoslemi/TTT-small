from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = -diff / count
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)

def active_lag_objective(h, active_indices, weights):
    """Compute weighted sum of f_k for active lags"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.dot(weights, c[active_indices])

def logsumexp_objective(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_violation = np.sum(h) - n / 2.0
    return lse + penalty * sum_violation ** 2

def top_lags_objective(h, M, penalty):
    """Minimize sum of top-M lag values (smooth approximation)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Use soft top-k: approximate by sum of max(0, c_i - threshold)
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[min(M, len(c))-1]
    top_sum = np.sum(np.maximum(c - threshold, 0)) / min(M, len(c))
    sum_violation = np.sum(h) - n / 2.0
    return top_sum + penalty * sum_violation ** 2

    return x[-1]

    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def make_slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq}
    ]

def slsqp_optimize(h_init, maxiter=3000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.1), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

    result = minimize(

    return h_init, compute_c5_from_h(h_init)

    return h_init, compute_c5_from_h(h_init)

∫ h(x)(1-h(x+k)) dx = ∫ h(x)g(x+k) dx

So we want to minimize max_k ∫ h(x)g(x+k) dx where h+g = 1 and ∫h = ∫g = 1.

This is like minimizing the maximum cross-correlation between h and g = 1-h.

If h and g were "uncorrelated" (like independent random variables), the cross-correlation would be E[h]*E[g] * (overlap length) = 0.5 * 0.5 * (2-k) = 0.25*(2-k). The max would be at k=0: 0.5. But we can do better by making h and g "anti-correlated" at the lags where they overlap the most.

This suggests that h should oscillate with a period that makes h(x) high where h(x+k) is low, for the critical values of k.

def project_feasible(h, n):

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def lse_objective(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sv = np.sum(x) - n / 2.0
    return lse + penalty * sv * sv


def lse_gradient(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    A = w[n - 1: 2 * n - 1]
    B = w[n - 2: 0: -1]
    grad = np.zeros(n)
    for l in range(n):
        if A[l] > 1e-18:
            grad[l:] += A[l] * dx * (1.0 - x[:n - l])
            grad[:n - l] -= A[l] * dx * x[l:]
    for p in range(1, n):
        if B[p - 1] > 1e-18:
            grad[:n - p] += B[p - 1] * dx * (1.0 - x[p:])
            grad[p:] -= B[p - 1] * dx * x[:n - p]
    sv = np.sum(x) - n / 2.0
    grad += 2.0 * penalty * sv
    return grad


def lse_lipschitz(x):
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    for l in range(n):
        grad[l:] += dx
        grad[:n - l] -= dx
    for p in range(1, n):
        grad[:n - p] += dx
        grad[p:] -= dx
    return np.max(np.abs(grad))


def lse_optimize(h_init, T, penalty, maxiter=3000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lse_objective, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        jac=lambda x: lse_gradient(x, T, penalty),
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                 'maxfun': maxiter * 5}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_optimize(h_init, maxiter=5000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.01), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq_full},
        {'type': 'eq', 'fun': slsqp_eq}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def coordinate_descent(h, n_active=40, n_sweeps=80, penalty=100.0):
    N = len(h)
    dx = 2.0 / N
    target = N / 2.0
    best_c5 = compute_c5_from_h(h)
    stagnant = 0
    for sweep in range(n_sweeps):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[min(n_active, len(c)) - 1]
        active_idx = np.where(c >= threshold)[0]

        sweep_improved = False
        order = np.arange(N)
        rng_cd = np.random.default_rng(42 + sweep)
        rng_cd.shuffle(order)
        for j in order:
            lo = 0.0
            hi = 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                old_val = h[j]
                h[j] = m1
                c1 = np.correlate(h, 1.0 - h, mode="full") * dx
                obj1 = c1[active_idx].max() + penalty * (np.sum(h) - target) ** 2
                h[j] = m2
                c2 = np.correlate(h, 1.0 - h, mode="full") * dx
                obj2 = c2[active_idx].max() + penalty * (np.sum(h) - target) ** 2
                h[j] = old_val
                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1
            h_new_val = (lo + hi) / 2.0
            old_val = h[j]
            h[j] = h_new_val
            c_new = np.correlate(h, 1.0 - h, mode="full") * dx
            obj_new = c_new[active_idx].max() + penalty * (np.sum(h) - target) ** 2
            h[j] = old_val
            c_old = np.correlate(h, 1.0 - h, mode="full") * dx
            obj_old = c_old[active_idx].max() + penalty * (np.sum(h) - target) ** 2
            if obj_new < obj_old - 1e-14:
                h[j] = h_new_val
                sweep_improved = True
        h = project_feasible(h, N)
        c5_cur = compute_c5_from_h(h)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 15:
            break
    h = project_feasible(h, N)
    return h, compute_c5_from_h(h)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 270
        x = np.linspace(0, 2, n_init, endpoint=False)
        h0 = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)

    if time.time() - t0 < budget_s * 0.15:
        h_new, c5_new = slsqp_optimize(best_h, maxiter=6000, t_margin=0.1)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    T_schedule = [5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
    for Ti in T_schedule:
        if time.time() - t0 > budget_s * 0.45:
            break
        penalty = max(5.0, 2.0 * Ti / 5.0)
        h_new, c5_new = lse_optimize(best_h, Ti, penalty, maxiter=2000)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
        if time.time() - t0 > budget_s * 0.45:
            break
        scale = 0.005
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        h_new2, c5_new2 = lse_optimize(h_pert, Ti, penalty, maxiter=1000)
        if c5_new2 < best_c5 - 1e-12:
            best_h = h_new2
            best_c5 = c5_new2

    if time.time() - t0 < budget_s * 0.70:
        h_cd, c5_cd = coordinate_descent(best_h, n_active=40, n_sweeps=80, penalty=50.0)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    if time.time() - t0 < budget_s * 0.80:
        h_new, c5_new = slsqp_optimize(best_h, maxiter=8000, t_margin=0.05)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s * 0.90:
        h_cd2, c5_cd2 = coordinate_descent(best_h, n_active=60, n_sweeps=60, penalty=200.0)
        if c5_cd2 < best_c5:
            best_h = h_cd2
            best_c5 = c5_cd2

    n_try_list = [200, 320, 400]
    for n_try in n_try_list:
        if time.time() - t0 > budget_s * 0.95:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        h_alt, c5_alt = lse_optimize(h_interp, 2000, 500.0, maxiter=2000)
        if time.time() - t0 < budget_s * 0.97:
            h_alt, c5_alt = slsqp_optimize(h_alt, maxiter=3000, t_margin=0.03)
        if c5_alt < best_c5 - 1e-10:
            best_h = h_alt
            best_c5 = c5_alt
            n = len(best_h)

    if time.time() - t0 < budget_s * 0.97:
        h_new, c5_new = slsqp_optimize(best_h, maxiter=10000, t_margin=0.01)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 5:
        h_cd3, c5_cd3 = coordinate_descent(best_h, n_active=50, n_sweeps=40, penalty=500.0)
        if c5_cd3 < best_c5:
            best_h = h_cd3
            best_c5 = c5_cd3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
