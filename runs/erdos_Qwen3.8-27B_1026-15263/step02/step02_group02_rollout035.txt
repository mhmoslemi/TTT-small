            coeff[lag:] = 1.0 - h[:m]
            const = np.sum(h[lag:lag + m] * h[:m])
            row = np.zeros(n + 1)
            row[:n] = coeff * dx
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-const * dx)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(0, 2)]
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            break
        h_lp = result.x[:n]
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        if c5_new < c5_cur - 1e-14:
            h = h_new
            c5_cur = c5_new
        else:
            if damping > 0.1:
                damping *= 0.7
            else:
                break
    return best_h, best_c5


def logsumexp_obj(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    if c_max < -500:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_v = np.sum(x) - n / 2.0
    return lse + penalty * sum_v ** 2


def lbfgsb_optimize(h_init, T, penalty, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            logsumexp_obj, h_init, args=(T, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        if result.x is not None:
            h_out = project_feasible(result.x, n)
            return h_out, compute_c5_from_h(h_out)
    except Exception:
        pass
    return h_init, compute_c5_from_h(h_init)


def targeted_perturbation(h, c_vec, c_max, rng, scale):
    n = len(h)
    h_new = h.copy()
    threshold = c_max * 0.96
    active = np.where(c_vec >= threshold)[0]
    n_active = len(active)
    if n_active == 0:
        return h_new
    n_targets = min(15, n * 3)
    for _ in range(n_targets):
        lag_idx = int(rng.integers(0, n_active))
        lag = active[lag_idx]
        actual_lag = lag - (n - 1)
        if actual_lag >= 0:
            m = n - actual_lag
            if m <= 0:
                continue
            prod = h[:m] * (1.0 - h[actual_lag:actual_lag + m])
            top_n = min(5, m)
            top_idx = np.argsort(prod)[-top_n:]
            for i in top_idx:
                if rng.random() < 0.5:
                    h_new[i] = max(0.0, h_new[i] - scale)
                else:
                    h_new[i + actual_lag] = min(1.0, h_new[i + actual_lag] + scale)
        else:
            l = -actual_lag
            m = n - l
            if m <= 0:
                continue
            prod = h[:m] * (1.0 - h[l:l + m])
            top_n = min(5, m)
            top_idx = np.argsort(prod)[-top_n:]
            for i in top_idx:
                if rng.random() < 0.5:
                    h_new[i + l] = max(0.0, h_new[i + l] - scale)
                else:
                    h_new[i] = min(1.0, h_new[i] + scale)
    return h_new


def make_block_h(n, blocks, widths):
    h = np.zeros(n)
    for i in range(blocks):
        start = int(i * n / blocks)
        end = int(start + widths[i])
        h[start:min(end, n)] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    # Phase 1: Thorough initial optimization
    if time_left() > 120:
        h_new, c5_new = slsqp_optimize(best_h, n_best, maxiter=8000, t_margin=0.15, ftol=1e-16)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time_left() > 80:
        h_new, c5_new = slp_solve(best_h, n_best, max_iters=20, damping=1.0)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: L-BFGS-B with various temperatures
    if time_left() > 200:
        for T, pen in [(20.0, 80.0), (50.0, 150.0), (100.0, 300.0), (200.0, 600.0), (400.0, 1000.0)]:
            if time_left() < 20:
                break
            h_new, c5_new = lbfgsb_optimize(best_h, T, pen, maxiter=1500)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5, n_best = h_new, c5_new, len(h_new)

    # Phase 3: SLP refinement
    if time_left() > 60:
        h_new, c5_new = slp_solve(best_h, n_best, max_iters=30, damping=0.8)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Multi-start with diverse initializations
    if time_left() > 300:
        alternatives = []
        x = np.linspace(0, 2, n_best, endpoint=False)
        sin_h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        alternatives.append(project_feasible(sin_h, n_best))
        tri_h = 1.0 - np.abs(2 * x - 2)
        alternatives.append(project_feasible(tri_h, n_best))
        block_w = np.array([n_best // 5, n_best // 5, n_best // 5, n_best // 5, n_best // 5])
        alternatives.append(make_block_h(n_best, 5, block_w))
        cos2_h = 0.5 + 0.35 * np.cos(4 * np.pi * x)
        alternatives.append(project_feasible(cos2_h, n_best))

        for alt in alternatives:
            if time_left() < 60:
                break
            h_new, c5_new = slsqp_optimize(alt, n_best, maxiter=5000, t_margin=0.3)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new
            if time_left() > 40:
                h_new2, c5_new2 = slp_solve(h_new, n_best, max_iters=15, damping=0.9)
                if c5_new2 < best_c5 - 1e-12:
                    best_h, best_c5 = h_new2, c5_new2

    # Phase 5: Resolution exploration
    for n_try in [200, 320, 400, 500, 600]:
        if time_left() < 120:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new_grid = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new_grid), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = slsqp_optimize(h_interp, n_try, maxiter=5000, t_margin=0.15)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5, n_best = h_new, c5_new, n_try
        if time_left() > 50 and c5_new <= best_c5:
            h_new2, c5_new2 = slp_solve(h_new, n_try, max_iters=15, damping=0.9)
            if c5_new2 < best_c5 - 1e-12:
                best_h, best_c5, n_best = h_new2, c5_new2, n_try

    # Phase 6: Targeted perturbation loop
    n_perturb = 90
    for attempt in range(n_perturb):
        if time_left() < 25:
            break
        frac = attempt / max(1, n_perturb)
        scale = 0.10 * (1.0 - frac) ** 0.7 + 0.004

        c_vec = compute_c5_vector(best_h)
        c_max = c_vec.max()

        r = rng.random()
        if r < 0.25:
            h_pert = targeted_perturbation(best_h, c_vec, c_max, rng, scale)
        elif r < 0.40:
            h_pert = best_h + scale * 0.6 * rng.standard_normal(n_best)
        elif r < 0.55:
            h_pert = best_h.copy()
            n_blocks = int(rng.integers(3, max(4, n_best // 25)))
            for _ in range(n_blocks):
                bsize = int(rng.integers(max(4, n_best // 12), max(5, n_best // 5)))
                start = int(rng.integers(0, max(1, n_best - bsize)))
                h_pert[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 2
        elif r < 0.70:
            h_pert = best_h.copy()
            n_push = max(5, int(n_best * 0.12))
            push_idx = rng.choice(n_best, n_push, replace=False)
            for idx in push_idx:
                if rng.random() < 0.5:
                    h_pert[idx] = min(1.0, h_pert[idx] + scale * 3.5)
                else:
                    h_pert[idx] = max(0.0, h_pert[idx] - scale * 3.5)
        elif r < 0.85:
            h_pert = best_h.copy()
            n_modes = int(rng.integers(1, 4))
            for _ in range(n_modes):
                freq = int(rng.integers(1, max(2, n_best // 3)))
                phase = float(rng.random() * 2 * np.pi)
                amp = scale / np.sqrt(n_modes)
                h_pert += amp * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
        else:
            h_pert = best_h + scale * 0.4 * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            T_cur = 40.0 + 80.0 * frac
            h_pert, _ = lbfgsb_optimize(h_pert, T_cur, 100.0 + 300.0 * frac, maxiter=800)

        h_pert = project_feasible(h_pert, n_best)

        if time_left() > 50:
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=3000, t_margin=0.04)
        else:
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=1500, t_margin=0.04)

        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new
            if time_left() > 40:
                h_s, c5_s = slp_solve(best_h, n_best, max_iters=10, damping=0.9)
                if c5_s < best_c5 - 1e-12:
                    best_h, best_c5 = h_s, c5_s

    # Phase 7: Fine-tuning
    for attempt in range(50):
        if time_left() < 12:
            break
        scale = 0.002 + 0.010 * (1 - attempt / 50) ** 2
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2500, t_margin=0.02)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new

    # Phase 8: Final SLP + SLSQP
    if time_left() > 40:
        h_new, c5_new = slp_solve(best_h, n_best, max_iters=25, damping=0.7)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new

    if time_left() > 20:
        best_h, best_c5 = slsqp_optimize(best_h, n_best, maxiter=12000, t_margin=0.01, ftol=1e-16)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
