def projected_gradient_smooth_max(h, n, tau, lr, max_iter, top_k_frac=0.1):
    """Projected gradient method for smooth max of correlations."""
    dx = 2.0 / n
    num_lags = 2 * n - 1
    # Use only top fraction of lags for efficiency
    num_active = max(1, int(num_lags * top_k_frac))
    
    for it in range(max_iter):
        # Compute correlations
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full") * dx
        # Get top lags
        top_idx = np.argsort(corr)[-num_active:]
        # Compute softmax weights
        vals = corr[top_idx] / tau
        vals -= np.max(vals)
        weights = np.exp(vals)
        weights /= np.sum(weights)
        # Compute gradient
        grad = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)  # actual lag
            # grad_k[j] = (1-h[j-k]) * 1[0<=j-k<n] - h[j+k] * 1[0<=j+k<n]
            # Vectorized:
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            if k >= 0:
                # j-k valid: j >= k, j-k < n => j < n+k (always true since j < n)
                # j+k valid: j < n-k
                term1[k:] = 1.0 - h[:n-k]  # h[j-k] for j=k..n-1
                term2[:n-k] = h[k:]  # h[j+k] for j=0..n-k-1
            else:
                kk = -k
                # j-k = j+kk valid: j+kk < n => j < n-kk, j >= 0 (always)
                # j+k = j-kk valid: j >= kk
                term1[:n-kk] = 1.0 - h[kk:]  # h[j+kk] for j=0..n-kk-1
                term2[kk:] = h[:n-kk]  # h[j-kk] for j=kk..n-1
            grad += weights[np.where(top_idx == m)[0][0]] * (term1 - term2)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check convergence
        if np.max(np.abs(h_new - h)) < 1e-12:
            break
        h = h_new

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        # Spread the correction across coordinates closest to 0.5
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            adj = min(diff, 1.0 - result[idx]) if diff > 0 else max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-13:
                break
    return result

def smooth_max_grad(h, tau, top_k=30):
    """Gradient of smooth max of correlations * dx."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    
    num_active = min(top_k, 2*n-1)
    top_idx = np.argsort(corr)[-num_active:]
    
    vals = corr[top_idx] / tau
    vals -= np.max(vals)
    weights = np.exp(vals)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for i in range(num_active):
        m = top_idx[i]
        k = int(m) - (n - 1)
        w = weights[i]
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n-kk] += w * (1.0 - h[kk:])
                grad[kk:] -= w * h[:n-kk]
    
    return grad

def projected_grad_opt(h0, n, tau_start, tau_end, lr_start, lr_end, n_iter, top_k=30):
    """Projective gradient with annealing."""
    h = h0.copy()
    best = h0.copy()
    best_c5 = evaluate_c5(h)[0]
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = smooth_max_grad(h, tau, top_k)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = evaluate_c5(h_new)[0]
        if c5 < best_c5:
            best = h_new.copy()
            best_c5 = c5
        h = h_new
    
    return best, best_c5

        return tau * (v.max() + np.log(np.sum(np.exp(v))))

            best_h = h_pg2

    if time.time() < phase3_end:
        n_sa = int((phase3_end - time.time()) * 120)
        n_sa = max(n_sa, 3000)
        h_sa, c5_sa = paired_move_sa(best_h, n, rng, n_steps=n_sa, top_lags=10,
                                     delta_range=(0.005, 0.12))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        h_cd, c5_cd = coord_descent(best_h, n, max_sweeps=40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        h_pg3, c5_pg3 = pg_optimize(best_h, n, n_iter=5000, top_k=50,
                                    tau_start=0.2, tau_end=0.003, lr_start=0.05, lr_end=0.001)
        if c5_pg3 < best_c5:
            best_c5 = c5_pg3
            best_h = h_pg3

    if time.time() < phase4_end:
        for attempt in range(200):
            if time.time() > phase4_end:
                break
            frac = (time.time() - t0) / budget_s
            scale = 0.003 + 0.02 * max(0.0, 1.0 - frac * 2.0)
            h_p = il(scale, n, rng, best_h)
            h_opt, c5_opt = pg_optimize(h_p, n, n_iter=1500, top_k=40,
                                        tau_start=0.05, tau_end=0.002, lr_start=0.03, lr_end=0.001)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        h_cd2, c5_cd2 = coord_descent(best_h, n, max_sweeps=30,
                                      steps=[0.03, 0.015, 0.007, 0.003, 0.001])
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2
        h_r2, c5_r2 = slsqp_refine(best_h, n, max_iter=200, tau=0.005)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2

    if best_h is not None and time.time() < t0 + budget_s * 0.97:
        h_final, c5_final = pg_optimize(best_h, n, n_iter=3000, top_k=60,
                                        tau_start=0.05, tau_end=0.001, lr_start=0.02, lr_end=0.0003)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        h_cd3, c5_cd3 = coord_descent(best_h, n, max_sweeps=20,
                                      steps=[0.01, 0.005, 0.002, 0.001])
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3

    final_c5 = evaluate_c5(best_h)[0]
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
