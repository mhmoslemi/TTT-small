    """Compute subgradient of max_k corr[k] using active set."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    # Active set: lags where corr[k] >= M - tol
    tol = 1e-6 + 0.01 * M
    active = corr >= M - tol
    
    # Gradient of corr[k] w.r.t. h
    # corr[n-1+lag] for lag >= 0: sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
    # d/dh[j] = (1-h[j-lag]) if j>=lag - h[j+lag] if j<=n-1-lag

            # d/dh[j] = (1-h[j-lag]) * I[j>=lag] - h[j+lag] * I[j<=n-1-lag]

            g[m:] -= h[:n-m]  # -h[j-m] for j=m,...,n-1
    
    # Normalize active set weights (uniform for simplicity)

    return M, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):

    w_neg = w[0:n-1][::-1]

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol=1e-12):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init=0.01):
    """Nesterov accelerated gradient with backtracking."""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        g_proj = g - np.mean(g)
        
        h_new = np.clip(y - lr * g_proj, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new

        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.85

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    """Adaptive coordinate descent with line search."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try to optimize coordinate i via small line search
            best_val = c5_cur
            best_d = 0.0
            # Adaptive step sizes
            for scale in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                d = scale * (1 - 2*h[i])  # scale by distance to boundary
                if abs(d) < 1e-6:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def _gradient_from_weights(w, h, n):
    w_pos = w[n - 1:2 * n - 1]
    w_neg = w[0:n - 1][::-1]
    one_minus_h = 1.0 - h
    h_flip = h[::-1]
    one_minus_h_flip = one_minus_h[::-1]
    t1 = np.convolve(w_pos, one_minus_h, mode='full')[:n]
    conv_wh = np.convolve(w_pos, h_flip, mode='full')
    t2 = -conv_wh[n - 1::-1][:n]
    if n > 1:
        conv_t3 = np.convolve(w_neg, one_minus_h_flip, mode='full')
        t3 = np.zeros(n)
        t3[:n - 1] = conv_t3[n - 2::-1][:n - 1]
        conv_t4 = np.convolve(w_neg, h, mode='full')
        t4 = np.zeros(n)
        t4[1:] = -conv_t4[:n - 1]
    else:
        t3 = np.zeros(n)
        t4 = np.zeros(n)
    return t1 + t2 + t3 + t4


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = _gradient_from_weights(w, h, n)
    return sm_val, g


def subgradient_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    tol = 0.05 * (M + 1e-10)
    active = corr >= M - tol
    n_active = np.sum(active)
    if n_active == 0:
        active = np.zeros(2 * n - 1, dtype=bool)
        active[np.argmax(corr)] = True
        n_active = 1
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_active
    g = _gradient_from_weights(w, h, n)
    return M, g


def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol=1e-12):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)


def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init=0.01):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        g_proj = g - np.mean(g)
        h_new = np.clip(y - lr * g_proj, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.85
    return h_best, c5_cur


def subgradient_polish(h0, n, target_sum, n_iters, lr_init=0.005):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = subgradient_step(h, n)
        g_proj = g - np.mean(g)
        h_new = np.clip(h - lr * g_proj, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-16:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.92
            if lr < 1e-9:
                break
    return h_best, c5_cur


def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for scale in [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]:
                d = scale * (1.0 - 2.0 * h[i])
                if abs(d) < 1e-7:
                    continue
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
