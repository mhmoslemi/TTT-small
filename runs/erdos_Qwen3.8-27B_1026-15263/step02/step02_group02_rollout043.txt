def compute_c5_from_h(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), (1.0 - h[mask].max()) * 0.9999)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min() * 0.9999)
            if step <= 0:
                break

def compute_c5_from_h(h):

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            if step <= 0: break
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            if step <= 0: break
            h[mask] -= step

    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()

    for _ in range(500):

        if diff > 0:
            mask = h < 1.0 - 1e-16
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            if room.min() <= 0:
                break
            step = min(diff / np.count_nonzero(mask), room.min() * 0.9999)
            if step <= 0:
                break
            h[mask] += step

            if not np.any(mask):
                break
            room = h[mask]
            if room.min() <= 0:
                break

            if step <= 0:
                break
            h[mask] -= step

def coord_descent(h_init, max_sweeps, n_grid, t0, time_limit, rng, recompute_interval=20):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    k_vals = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        
        if sweep % recompute_interval == 0:
            c_h = np.correlate(h, 1.0 - h, mode="full")
        
        order = rng.permutation(n)
        
        for i in order:
            h_i = h[i]
            diff_k = i - k_vals
            mask1 = (diff_k >= 0) & (diff_k < n)
            sum_k = i + k_vals
            mask2 = (sum_k >= 0) & (sum_k < n)
            
            d = np.zeros(2 * n - 1)
            if np.any(mask1):
                d[mask1] = 1.0 - h[diff_k[mask1]]
            if np.any(mask2):
                d[mask2] -= h[sum_k[mask2]]
            
            T = np.linspace(0.0, 1.0, n_grid)
            delta = T - h_i
            c_T = c_h[None, :] + delta[:, None] * d[None, :]
            c_T[:, n - 1] = c_h[n - 1] + T * (1.0 - T) - h_i * (1.0 - h_i)
            c_max = np.max(c_T, axis=1)
            t_best = T[np.argmin(c_max)]

                c_T2[:, n - 1] = c_h[n - 1] + T2 * (1.0 - T2) - h_i * (1.0 - h_i)
                c_max2 = np.max(c_T2, axis=1)
                t_best = T2[np.argmin(c_max2)]
            
            diff_t = t_best - h_i

                h[i] = t_best
                c_h += diff_t * d
                c_h[n - 1] -= diff_t * d[n - 1]
                c_h[n - 1] += t_best * (1.0 - t_best) - h_i * (1.0 - h_i)
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)

    t1 = min(budget_s * 0.22, 300.0)
    h1, c5_1 = coord_descent(h, max_sweeps=10000, n_grid=300,
                             t0=t0, time_limit=t1, rng=rng)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n

    if time.time() - t0 < budget_s * 0.35:
        h2, c5_2 = slsqp_optimize(best_h, maxiter=4000, t_margin=0.03)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    resolutions = [405, 540, 729, 1080]
    for n_target in resolutions:
        if time.time() - t0 > budget_s * 0.80:
            break

        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_target, endpoint=False)
        fi = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_interp = np.clip(fi(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_target)

        ng = max(120, min(256, 120000 // n_target))
        t_lim = min(budget_s * 0.83, time.time() - t0 + budget_s * 0.10)
        h3, c5_3 = coord_descent(h_interp, max_sweeps=8000, n_grid=ng,
                                 t0=t0, time_limit=t_lim, rng=rng)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n_target

    if time.time() - t0 < budget_s - 50:
        h4, c5_4 = slsqp_optimize(best_h, maxiter=6000, t_margin=0.02)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    if time.time() - t0 < budget_s - 10:
        ng_final = max(120, min(256, 120000 // len(best_h)))
        h5, c5_5 = coord_descent(best_h, max_sweeps=3000, n_grid=ng_final,
                                 t0=t0, time_limit=budget_s - 3, rng=rng)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
