def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

   - (bound constraints handled via active set)
3. Solve the linearized system for δh, δt
4. Take a step

The Jacobian of F_i with respect to h is the gradient of C₅(k_i), which we can compute efficiently.

The Jacobian of F_i with respect to t is -1.

So the system is:
[∇C₅(k_1)  -1] [δh]   [-C₅(k_1) + t]
[∇C₅(k_2)  -1] [δt] = [-C₅(k_2) + t]
[...         ...]
[∇C₅(k_m)  -1]
[1...1       0 ]         [n/2 - Σh]

This is a (m+1) × (n+1) system, which is underdetermined (m+1 << n+1). We need to add more conditions.

Actually, for the minimum-norm solution, we can solve:
min ||δh||² + ||δt||²
s.t. the linearized constraints

This is a standard least-squares problem. The solution is:
[δh; δt] = J' (JJ')^{-1} r

where J is the Jacobian and r is the residual.

- Fix t, find the h that minimizes the maximum violation of C₅(k) ≤ t
- If t is feasible, decrease t; otherwise increase t
- This is a bisection on t

For a fixed t, the feasibility problem is:
Find h such that C₅(k) ≤ t for all k, 0 ≤ h ≤ 1, Σh = n/2

This is a nonconvex feasibility problem (bilinear constraints), so it's not easy to solve directly. But we can use SLSQP to minimize the maximum violation:

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def symmetrize(h):
    n = len(h)
    return 0.5 * (h + h[::-1].copy())


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.05):
    c5_init = compute_c5_from_h(h_init)

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - c

    def eq(x):
        h = x[:-1]
        return np.sum(h) - n / 2.0

    def eq_jac(x):
        jac = np.zeros(n + 1)
        jac[:n] = 1.0
        return jac

    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), c5_init + t_margin)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]

    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})

    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def logsumexp_optimize(h_init, n, T, maxiter=2000):
    def smooth_obj(x):
        h = x
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        c_max = c.max()
        shifted = T * (c - c_max)
        shifted = np.clip(shifted, -400, 400)
        val = c_max + np.log(np.sum(np.exp(shifted))) / T
        sum_v = np.sum(h) - n / 2.0
        return val + 100.0 * sum_v ** 2

    result = minimize(smooth_obj, h_init, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16})

    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def compute_gradient_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(k, n):
        g[j] += dx * (1.0 - h[j - k])
    for j in range(0, n - k):
        g[j] -= dx * h[j + k]
    return g


def active_set_lp_step(h, c5_profile, n, top_k=5, max_step=0.03):
    c5_first_half = c5_profile[:n - 1]
    sorted_idx = np.argsort(c5_first_half)[::-1][:top_k]
    t = c5_first_half[sorted_idx[0]]

    dx = 2.0 / n
    G_active = np.zeros((top_k, n))
    for idx, k in enumerate(sorted_idx):
        for j in range(k, n):
            G_active[idx, j] += dx * (1.0 - h[j - k])
        for j in range(0, n - k):
            G_active[idx, j] -= dx * h[j + k]

    m = top_k
    A_ub = np.hstack([G_active, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-max_step, max_step)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status != 0:
        return None, None

    return res.x[:n], res.x[-1]


def newton_kkt_step(h, n, tol=1e-8):
    c5_prof = compute_c5_profile(h)
    c5_first = c5_prof[:n - 1]
    t = c5_first.max()
    active = np.where(c5_first > t - 0.002)[0]
    m = len(active)
    if m == 0:
        return h, compute_c5_from_h(h)

    J = np.zeros((m + 1, n + 1))
    r = np.zeros(m + 1)
    for i, k in enumerate(active):
        for j in range(k, n):
            J[i, j] += (2.0 / n) * (1.0 - h[j - k])
        for j in range(0, n - k):
            J[i, j] -= (2.0 / n) * h[j + k]
        J[i, n] = -1.0
        r[i] = c5_first[k] - t

    J[m, :n] = 1.0
    J[m, n] = 0.0
    r[m] = np.sum(h) - n / 2.0

    try:
        JJt = J @ J.T
        rhs = J @ r
        delta = np.linalg.solve(JJt + 1e-12 * np.eye(m + 1), rhs)
        step = J.T @ delta
        dh = step[:n]
        dt = step[n]
    except np.linalg.LinAlgError:
        return h, compute_c5_from_h(h)

    alpha = 1.0
    best_h_trial = h
    best_c5_trial = compute_c5_from_h(h)
    for _ in range(20):
        h_trial = project_feasible(h + alpha * dh, n)
        c5_trial = compute_c5_from_h(h_trial)
        if c5_trial < best_c5_trial:
            best_h_trial = h_trial
            best_c5_trial = c5_trial
        alpha *= 0.7
        if alpha < 1e-4:
            break

    return best_h_trial, best_c5_trial


def targeted_perturbation(h, c5_profile, n, rng, scale=0.01):
    c5_first = c5_profile[:n - 1]
    active = np.where(c5_first > c5_first.max() - 0.005)[0]
    h_pert = h.copy()
    for k in active[:3]:
        direction = rng.choice([-1, 1])
        h_pert[k:k + n - k] += direction * scale * 0.1 * rng.random(n - k)
    h_pert += scale * 0.05 * rng.standard_normal(n)
    return project_feasible(h_pert, n)
