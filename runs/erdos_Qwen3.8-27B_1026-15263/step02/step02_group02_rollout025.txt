    h = np.array(h, dtype=float).copy()

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_obj_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode="full")
    c = c_raw * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -400, 400)
    exp_vals = np.exp(shifted)
    s = exp_vals / exp_vals.sum()
    obj = c_max + np.log(exp_vals.sum()) / T

    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = s[k]
        if w < 1e-14:
            continue
        m = lags[k]
        if m >= 0:
            grad[m:] += w * (1.0 - x[:n - m])
            grad[:n - m] -= w * x[m:]
        else:
            mm = -m
            grad[:n - mm] += w * (1.0 - x[mm:])
            grad[mm:] -= w * x[:n - mm]
    return obj, grad * dx


def lse_only(x, T):

            threshold = np.sort(c)[-top_k][0]
            active = np.where(c >= threshold - 1e-6)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        def obj_slsqp(x):
            return x[-1]

        def ineq_slsqp(x):
            h_x = x[:-1]
            t = x[-1]
            c_x = np.correlate(h_x, 1.0 - h_x, mode="full") * (2.0 / n)
            return t - c_x[active]

        def eq_slsqp(x):

        o, g = lse_obj_grad(x, T)
