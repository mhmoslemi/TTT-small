import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_set(h, target_sum):
    """Project onto {0<=x<=1, sum=x_target} using bisection on the Lagrange multiplier"""
    h = h.astype(float).copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - np.sum(h_proj)
    mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
    if np.any(mask) and abs(diff) > 1e-14:
        h_proj[mask] += diff / np.sum(mask)
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_value_and_weights(corr, alpha):
    """Compute smooth max value and weights for the log-sum-exp"""
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    W = exp_vals / s
    sm_val = m + np.log(s) / alpha
    return sm_val, W

def compute_gradient_fft(h, W):
    """Compute gradient using FFT-based correlation: g = correlate(W, 1-2h, 'full')[:n]"""
    n = len(h)
    total_len = len(W) + n - 1
    fW = np.fft.rfft(W, n=total_len)
    f_12h = np.fft.rfft(1.0 - 2.0*h, n=total_len)
    # correlate(W, 1-2h) = ifft(conj(fft(W)) * fft(1-2h))... 
    # Actually, np.correlate(a, v) = ifft(fft(a, N) * conj(fft(v, N)))
    # where N >= len(a) + len(v) - 1
    # But we want correlate(W, 1-2h), so:
    # result = ifft(fft(W, N) * conj(fft(1-2h, N)))
    # Wait no. The cross-correlation R_av[k] = sum_n a[n+k] * conj(v[n])
    # In numpy: np.correlate(a, v, 'full')[k] = sum_n a[n] * v[n+k-(len(v)-1)]
    
    # Let me use the direct relation:
    # np.correlate(a, v)[k] = sum_n a[n] * v[n + k - (len(v)-1)]
    # This equals the convolution of a with flip(v): conv(a, flip(v))[k + len(v)-1]
    # In FFT: fft(conv) = fft(a) * fft(flip(v)) = fft(a) * conj(fft(v)) (for real signals, at least approximately)
    
    # Actually for real signals: ifft(fft(a, N) * conj(fft(v, N))) gives the cross-correlation
    # Let me just use ifft(fft(W) * conj(fft(1-2h)))
    
    result = np.fft.irfft(fW * np.conj(f_12h), n=total_len)
    return result[:n]

def compute_gradient_direct(h, W):
    """Direct computation of gradient (slower but for verification)"""
    n = len(h)
    return np.correlate(W, 1.0 - 2.0*h, mode='full')[:n]

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, ftol=1e-12):
    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm_val, W = smooth_max_value_and_weights(corr, alpha)
        g = compute_gradient_direct(h, W)
        return sm_val, g
    
    # SLSQP needs separate obj and grad
    def obj(x):
        v, _ = obj_and_grad(x)
        return v
    def grad(x):
        _, g = obj_and_grad(x)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def fista_optimize(h0, n, target_sum, alpha, n_iters, init_lr=None):
    """FISTA optimization with smooth max objective"""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    # Estimate Lipschitz constant
    L = 1.0
    if init_lr is not None:
        L = 1.0 / init_lr
    
    for it in range(n_iters):
        g = compute_gradient_direct(y, h, alpha)  # placeholder
        # Actually need to compute properly
        corr = np.correlate(y, 1.0 - y, mode='full')
        _, W = smooth_max_value_and_weights(corr, alpha)
        g = compute_gradient_direct(y, W)
        
        lr = 1.0 / L
        h_new = project_onto_set(y - lr * g, target_sum)
        
        # Backtracking
        # For simplicity, use fixed step with occasional adjustment
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
    
    return h_best, c5_best

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        v, _ = get_smooth_max_and_weights(corr, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        _, W = get_smooth_max_and_weights(corr, alpha)
        return gradient_smooth_max(h, W)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

            lr = min(lr * 1.1, 0.5)
        else:
            stagnant += 1
            lr *= 0.8

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -600.0, 600.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    W = exp_vals / s
    sm_val = m + np.log(s) / alpha
    return sm_val, W

def gradient_smooth_max(h, W):
    n = len(h)
    return np.correlate(W, 1.0 - 2.0 * h, mode='full')[:n]

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=200, ftol=1e-14):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')

        h = np.clip(x, 0.0, 1.0)

        _, W = get_smooth_max_and_weights(corr, alpha)
        return gradient_smooth_max(h, W)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
        if stale > 80:
            break
    return h_best, c5_best

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    deltas = [-0.0005, -0.002, -0.008, 0.0005, 0.002, 0.008]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n_cur = best_n
    ts = n_cur / 2.0
    h_work = best_h.copy()
    alphas = [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    for alpha in alphas:
        if time_left() < 80:
            break
        n_iter = max(50, min(300, int(time_left() * 0.6)))
        h_opt = slsqp_optimize(h_work, n_cur, ts, alpha, n_iter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_cur
        h_work = h_opt
    if time_left() > 80:
        n_it = min(8000, int(time_left() * 4))
        h_f, c5_f = fista_optimize(h_work, n_cur, ts, 2000.0, n_it, 0.005)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_cur
        h_work = h_f
    if time_left() > 60:
        n_it2 = min(5000, int(time_left() * 3))
        h_f2, c5_f2 = fista_optimize(h_work, n_cur, ts, 5000.0, n_it2, 0.002)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            best_n = n_cur
        h_work = h_f2
    if time_left() > 180:
        n_new = max(720, best_n * 2)
        if n_new % 2 == 0:
            ts_new = n_new / 2.0
            h_interp = project_to_sum(interpolate_h(best_h, n_new), ts_new)
            h_res = h_interp
            for alpha in [200.0, 500.0, 1000.0, 3000.0]:
                if time_left() < 60:
                    break
                n_iter = max(30, min(200, int(time_left() * 0.5)))
                h_res = slsqp_optimize(h_res, n_new, ts_new, alpha, n_iter)
                c5_r = compute_c5(h_res)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_res.copy()
                    best_n = n_new
            if time_left() > 60:
                n_it3 = min(4000, int(time_left() * 2))
                h_f3, c5_f3 = fista_optimize(h_res, n_new, ts_new, 3000.0, n_it3, 0.003)
                if c5_f3 < best_c5:
                    best_c5 = c5_f3
                    best_h = h_f3.copy()
                    best_n = n_new
    if time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        max_cd = min(100, int(time_left() * 0.5))
        h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, max_cd, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    if time_left() > 20:
        n_f = best_n
        ts_f = n_f / 2.0
        h_work2 = best_h.copy()
        h_work2 = np.clip(h_work2 + rng.normal(0, 0.01, n_f), 0.0, 1.0)
        h_work2 = project_to_sum(h_work2, ts_f)
        h_opt2 = slsqp_optimize(h_work2, n_f, ts_f, 5000.0, 200)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
    n_final = best_n
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
