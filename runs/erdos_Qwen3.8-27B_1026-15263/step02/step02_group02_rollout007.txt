    # f is convex piecewise linear
    lo, hi = 0.0, 1.0

    f0 = np.max(base)
    f1 = np.max(base + slope)
    if f0 <= f1:
        # Check slope of active line at x=0
        k0 = np.argmax(base)
        if slope[k0] >= 0:
            return 0.0, f0
    else:
        k1 = np.argmax(base + slope)
        if slope[k1] <= 0:
            return 1.0, f1

    for _ in range(50):

        active = np.argmax(base + slope * mid)
        if slope[active] < 0:
            lo = mid  # function still decreasing
        else:
            hi = mid  # function still increasing
    x_star = (lo + hi) / 2
    f_star = np.max(base + slope * x_star)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)

def clip_lookup(h, idx):
    n = len(h)
    result = np.zeros(len(idx))
    mask = (idx >= 0) & (idx < n)
    result[mask] = h[idx[mask]]
    return result

def solve_1d_minmax(base, slope):
    m = len(base)
    f0 = np.max(base)
    f1 = np.max(base + slope)
    
    # Boundary checks
    if f0 <= f1:
        k0 = np.argmax(base)
        if slope[k0] >= -1e-12:
            return 0.0
    else:
        k1 = np.argmax(base + slope)
        if slope[k1] <= 1e-12:
            return 1.0
    
    # Interior: bisection
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        active = np.argmax(base + slope * mid)
        if slope[active] < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def coord_descent_sweep(h, c_cur, n):
    """One pass of Gauss-Seidel coordinate descent."""
    for i in range(n):
        # Compute delta_k for all k
        idx1 = np.arange(i - n + 1, i + n)  # i+k-n+1 for k=0..2n-2
        # idx2 = i-k+n-1 for k=0..2n-2 = i+n-1-k = reverse of idx1
        idx2 = idx1[::-1]
        
        h1 = clip_lookup(h, idx1)
        h2 = clip_lookup(h, idx2)
        delta = 1.0 - h1 - h2
        
        base = c_cur - h[i] * delta
        x_star = solve_1d_minmax(base, delta)
        
        if x_star != h[i]:
            c_cur += (x_star - h[i]) * delta
            h[i] = x_star
    return h, c_cur

import numpy as np
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = min(diff / count, (1.0 - h[mask]).min() * 0.9999)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            count = np.sum(mask)
            step = min(-diff / count, h[mask].min() * 0.9999)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def clip_lookup(h, idx):
    n = len(h)
    result = np.zeros(len(idx))
    mask = (idx >= 0) & (idx < n)
    result[mask] = h[idx[mask]]
    return result


def solve_1d_minmax(base, slope):
    f0 = np.max(base)
    f1 = np.max(base + slope)
    if f0 <= f1:
        k0 = np.argmax(base)
        if slope[k0] >= -1e-14:
            return 0.0
    else:
        k1 = np.argmax(base + slope)
        if slope[k1] <= 1e-14:
            return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = base + slope * mid
        active = np.argmax(vals)
        if slope[active] < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5


def coord_descent_sweep(h, c_cur, n, order):
    m = 2 * n - 1
    for i in order:
        idx1 = np.arange(i - n + 1, i + n)
        h1 = clip_lookup(h, idx1)
        h2 = clip_lookup(h, idx1[::-1])
        delta = 1.0 - h1 - h2
        base = c_cur - h[i] * delta
        x_star = solve_1d_minmax(base, delta)
        diff = x_star - h[i]
        if abs(diff) > 1e-12:
            c_cur += diff * delta
            h[i] = x_star
    return h, c_cur


def run_cd_phase(h, n, max_time, rng, sequential=True):
    t_start = time.time()
    c_cur = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(c_cur) * 2.0 / n)
    best_h = h.copy()
    no_improve = 0
    sweep = 0
    while time.time() - t_start < max_time:
        if sequential:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        h, c_cur = coord_descent_sweep(h, c_cur, n, order)
        h = project_feasible(h, n)
        c_cur = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(c_cur) * 2.0 / n)
        sweep += 1
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
    return best_h, best_c5, n


def try_binarize(h, n):
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n // 2:]] = 1.0
    h_bin = project_feasible(h_bin, n)
    c5_bin = compute_c5_from_h(h_bin)
    return h_bin, c5_bin
