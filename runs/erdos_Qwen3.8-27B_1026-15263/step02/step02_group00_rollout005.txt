import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + (max_shifted + np.log(np.sum(exp_vals))) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    # Simplest: use SLSQP but with tighter tolerances, or projected gradient
    
    # Let's try: minimize smooth_max subject to sum(h)=target using penalty method
    penalty = 1e6
    
    def penalized_obj(h):
        obj, grad = smooth_max_and_grad(h, alpha)
        sum_v = np.sum(h)
        obj += penalty * (sum_v - target_sum) ** 2
        grad += 2 * penalty * (sum_v - target_sum) * np.ones(n)
        return obj, grad
    
    try:
        res = minimize(penalized_obj, h0.copy(), method='L-BFGS-B',
                      jac=True, bounds=[(0.0, 1.0)] * n,

        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    # Projected gradient descent
    h_new = h - lr * grad
    h_new = project_to_sum(h_new, target_sum)
    return h_new, obj

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2*n-1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp(h0, n, ts, alpha, maxiter=800):
    def f(x): return smooth_max_obj_grad(x, alpha)[0]
    def g(x): return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0, ts)

    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev:
            h = h_new
            obj_prev = obj_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h

def freq_perturb(h, rng, n_harmonics, amp):

            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for alpha in [30, 200]:
                if time_left() < 12:
                    break
                h_o = slsqp_opt(h_s, n_e, ts_e, alpha, maxiter=400)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_s = h_o

    # Phase 5: Stochastic coordinate descent (mass transfer) on exact objective
    if time_left() > 80:
        n_cd = best_n
        ts_cd = n_cd / 2.0
        h_cd = best_h.copy()
        c5_cd = compute_c5(h_cd)
        n_cd_steps = 80000
        h_cd, c5_cd = stochastic_cd(h_cd, n_cd, ts_cd, rng, n_steps=n_cd_steps, delta_range=(0.01, 0.4))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        # Polish after stochastic
        if time_left() > 40:
            h_pol = slsqp_opt(best_h, n_cd, ts_cd, 2000, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Second round of stochastic CD with smaller deltas
    if time_left() > 60:
        n_cd2 = best_n
        ts_cd2 = n_cd2 / 2.0
        h_cd2 = best_h.copy()
        h_cd2, c5_cd2 = stochastic_cd(h_cd2, n_cd2, ts_cd2, rng, n_steps=50000, delta_range=(0.005, 0.15))
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        if time_left() > 30:
            h_pol2 = slsqp_opt(best_h, n_cd2, ts_cd2, 5000, maxiter=2000)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    # Phase 7: Projected gradient squeeze
    if time_left() > 40:
        n_pg = best_n
        ts_pg = n_pg / 2.0
        h_pg = proj_grad_opt(best_h, n_pg, ts_pg, 3000, n_iter=3000, lr_init=0.01)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 8: Final high-alpha SLSQP polish
    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [3000, 10000, 30000]:
            if time_left() < 20:
                break
            h_f = slsqp_opt(best_h, n_f, ts_f, alpha, maxiter=2000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 9: Tiny stochastic refinements
    if time_left() > 30:
        n_t = best_n
        ts_t = n_t / 2.0
        h_t, c5_t = stochastic_cd(best_h, n_t, ts_t, rng, n_steps=30000, delta_range=(0.005, 0.1))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
