def pnorm_obj(x, p, penalty):

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj_grad(h, T, penalty=0.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    Tc = T * c
    Tc = np.clip(Tc, -400.0, 400.0)
    max_Tc = Tc.max()
    raw_weights = np.exp(Tc - max_Tc)
    wsum = raw_weights.sum()
    weights = raw_weights / wsum
    
    obj = max_Tc / T + np.log(wsum) / T
    # Add penalty for sum constraint
    sum_h = np.sum(h)
    target = n / 2.0
    obj += penalty * (sum_h - target) ** 2
    
    # Gradient of smooth max
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-18:
            continue
        # A[i] += w * h_clip(i + s)
        if s >= 0:
            A[:n - s] += w * h[s:]
        else:
            A[-s:] += w * h[:n + s]
        # B[i] += w * h_clip(i - s)
        if s <= 0:
            t = -s
            B[:n - t] += w * h[t:]
        else:
            B[-s:] += w * h[:n - s]
    
    grad = dx * (1.0 - A + B)
    grad += 2.0 * penalty * (sum_h - target) * np.ones(n)

    return obj


def grad_wrapper(x, T, penalty):
    _, grad = smooth_max_obj_grad(x, T, penalty)
    return grad

    bounds = [(0.0, 1.0)] * n
    result = minimize(
        obj_wrapper, h_init,
        jac=grad_wrapper,
        args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxfun}
    )

        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

from scipy.interpolate import interp1d
import time

_slsqp_state = {'active': None, 'dx': 0.0}


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    Tc = T * c
    Tc = np.clip(Tc, -350.0, 350.0)
    max_Tc = Tc.max()
    raw_weights = np.exp(Tc - max_Tc)
    wsum = raw_weights.sum()
    weights = raw_weights / wsum
    obj = max_Tc / T + np.log(wsum) / T
    sum_h = np.sum(h)
    target = n / 2.0
    obj += penalty * (sum_h - target) ** 2
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-18:
            continue
        if s >= 0:
            A[:n - s] += w * h[s:]
        else:
            A[-s:] += w * h[:n + s]
        if s <= 0:
            t = -s
            if t == 0:
                B += w * h
            else:
                B[:n - t] += w * h[t:]
        else:
            B[-s:] += w * h[:n - s]
    grad = dx * (1.0 - A + B)
    grad += 2.0 * penalty * (sum_h - target) * np.ones(n)
    return obj, grad


def _obj_wrapper(x, T, penalty):
    obj, _ = smooth_max_obj_grad(x, T, penalty)
    return obj


def _grad_wrapper(x, T, penalty):
    _, grad = smooth_max_obj_grad(x, T, penalty)
    return grad


def lbfgsb_smooth(h_init, T, penalty, maxiter, maxfun):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        _obj_wrapper, h_init, jac=_grad_wrapper, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxfun}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def _slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    c = np.correlate(h, 1.0 - h, mode="full") * _slsqp_state['dx']
    return t - c[_slsqp_state['active']]


def _slsqp_eq(x):
    return np.sum(x[:-1]) - (len(x) - 1) / 2.0


def _slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def _slsqp_obj(x):
    return x[-1]


def slsqp_active(h_init, top_k, maxiter, t_margin, n_iter):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5_from_h(best_h)
    dx = 2.0 / n
    _slsqp_state['dx'] = dx
    for iteration in range(n_iter):
        c = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        actual_k = min(top_k, len(c))
        active = np.argsort(c)[-actual_k:]
        _slsqp_state['active'] = active
        c5_init = best_c5
        x0 = np.concatenate([best_h, [c5_init + t_margin]])
        bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.4), 2.0)]
        constraints = [
            {'type': 'ineq', 'fun': _slsqp_ineq},
            {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}
        ]
        result = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': 1e-19})
        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5_from_h(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
                t_margin = max(0.001, t_margin * 0.5)
    return best_h, best_c5


def generate_perturbation(h, n, rng, mode, scale):
    x = np.arange(n)
    if mode == 0:
        return h + scale * rng.standard_normal(n)
    elif mode == 1:
        out = h.copy()
        n_modes = rng.integers(1, 4)
        for _ in range(n_modes):
            freq = rng.integers(1, max(2, n // 4))
            phase = rng.random() * 2 * np.pi
            out += (scale / np.sqrt(n_modes)) * np.sin(2 * np.pi * freq * x / n + phase)
        return out
    elif mode == 2:
        out = h.copy()
        n_blocks = rng.integers(1, max(2, n // 40))
        for _ in range(n_blocks):
            bsize = rng.integers(5, max(6, n // 8))
            start = rng.integers(0, max(1, n - bsize))
            out[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 2.0
        return out
    elif mode == 3:
        out = h.copy()
        n_push = max(5, int(n * 0.1))
        idx = rng.choice(n, n_push, replace=False)
        for i in idx:
            if rng.random() < 0.5:
                out[i] = min(1.0, out[i] + scale * 2.0)
            else:
                out[i] = max(0.0, out[i] - scale * 2.0)
        return out
    else:
        out = h + scale * rng.standard_normal(n)
        out = 0.7 * out + 0.3 * h
        return out


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    def elapsed():
        return time.time() - t0

    # Phase 1: p-norm L-BFGS-B with increasing T
    if elapsed() < budget_s - 800:
        for T in [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]:
            if elapsed() > budget_s - 800:
                break
            h_new, c5_new = lbfgsb_smooth(best_h, T, 50.0, maxiter=3000, maxfun=50000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Active-set SLSQP refinement
    if elapsed() < budget_s - 600:
        for top_k in [8, 12, 20, 30]:
            if elapsed() > budget_s - 600:
                break
            h_new, c5_new = slsqp_active(best_h, top_k, maxiter=3000, t_margin=0.02, n_iter=2)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: More L-BFGS-B with very high T
    if elapsed() < budget_s - 500:
        for T in [500.0, 1000.0, 2000.0]:
            if elapsed() > budget_s - 500:
                break
            h_new, c5_new = lbfgsb_smooth(best_h, T, 200.0, maxiter=5000, maxfun=100000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Multi-start with diverse perturbations
    if elapsed() < budget_s - 300:
        for attempt in range(40):
            if elapsed() > budget_s - 300:
                break
            frac = attempt / 40.0
            scale = 0.08 * (1.0 - frac) ** 0.5 + 0.005
            mode = int(rng.integers(0, 5))
            h_pert = generate_perturbation(best_h, n_best, rng, mode, scale)
            h_pert = project_feasible(h_pert, n_best)
            T_cur = 50.0 + 200.0 * frac
            h_new, c5_new = lbfgsb_smooth(h_pert, T_cur, 100.0, maxiter=2000, maxfun=30000)
            if c5_new < best_c5 - 1e-12:
                h_new2, c5_new2 = slsqp_active(h_new, 15, maxiter=2000, t_margin=0.02, n_iter=2)
                if c5_new2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_new2
                else:
                    best_h = h_new
                    best_c5 = c5_new
            elif c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Try higher resolution
    if elapsed() < budget_s - 200:
        for n_try in [405, 540]:
            if elapsed() > budget_s - 200:
                break
            x_old = np.linspace(0, 2, n_best, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
            h_interp = np.clip(f_interp(x_new), 0.0, 1.0)
            h_interp = project_feasible(h_interp, n_try)
            h_new, c5_new = lbfgsb_smooth(h_interp, 200.0, 100.0, maxiter=3000, maxfun=50000)
            if c5_new < best_c5 - 1e-10:
                h_new2, c5_new2 = slsqp_active(h_new, 20, maxiter=3000, t_margin=0.02, n_iter=2)
                if c5_new2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_new2
                    n_best = n_try
                else:
                    best_h = h_new
                    best_c5 = c5_new
                    n_best = n_try

    # Phase 6: Final active-set SLSQP with tight tolerances
    if elapsed() < budget_s - 30:
        for top_k in [10, 20, 40]:
            if elapsed() > budget_s - 30:
                break
            h_new, c5_new = slsqp_active(best_h, top_k, maxiter=5000, t_margin=0.005, n_iter=3)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Final L-BFGS-B polish
    if elapsed() < budget_s - 5:
        h_new, c5_new = lbfgsb_smooth(best_h, 2000.0, 500.0, maxiter=5000, maxfun=100000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
