    h = np.array(h, dtype=float).copy()

            if not np.any(mask):

            h[mask] -= step

    shifted = np.clip(T * (c - c_max), -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

def c5_lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -400, 400)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    # w has length 2n-1, indexed 0 to 2n-2
    # g[j] = sum_k w[k] * dc_k/dh[j]
    # dc_k/dh[j] = (1-h[j+s_k])*I[0<=j+s_k<n] - h[j-s_k]*I[0<=j-s_k<n]
    # where s_k = k - (n-1)
    
    g = np.zeros(n)
    # Efficient: use the fact that this is a convolution
    # First part: sum_k w[k] * (1-h[j+k-(n-1)]) * I[0<=j+k-(n-1)<n]
    # = cross-correlation of (1-h) with w
    # Second part: -sum_k w[k] * h[j-(k-(n-1))] * I[0<=j-(k-(n-1))<n]
    # = -cross-correlation of h with w_reversed
    
    # Let's just do it directly for clarity (n is small)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        # (1-h[j+s]) valid for j in [max(0,-s), min(n, n-s))
        js = max(0, -s)
        je = min(n, n - s)
        if js < je:

        # -h[j-s] valid for j in [max(0,s), min(n, n+s))
        js2 = max(0, s)
        je2 = min(n, n + s)
        if js2 < je2:
            g[js2:je2] -= w[k] * h[js2-s:je2-s]

    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, maxiter=5000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_t_con},
        {'type': 'eq', 'fun': slsqp_sum_eq, 'jac': slsqp_sum_jac}
    ]
    res = minimize(slsqp_t_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.success or res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def lbfgsb_lse(h_init, T, maxiter=5000):

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
