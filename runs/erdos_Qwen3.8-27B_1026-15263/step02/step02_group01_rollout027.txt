def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-12:
            continue
        k = j - n + 1
        wt = w[j]
        l1 = max(0, -k)
        l2 = min(n, n - k)
        if l1 < l2:
            g[l1:l2] += wt * (1.0 - h[l1+k:l2+k])
        l3 = max(0, k)
        l4 = min(n, n + k)
        if l3 < l4:
            g[l3:l4] -= wt * h[l3-k:l4-k]
    
    return sm_val, g

def slsqp_solve(h0, n, ts, alpha, maxiter=200, ftol=1e-14):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v

        h = np.clip(x, 0.0, 1.0)

        return project_to_sum(h0, ts)

def subgradient_step(h, ts, step_size):
    """Take a step in the direction that reduces the maximum lag."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - n + 1
    
    # Subgradient of max at the maximizing lag
    g = np.zeros(n)
    l1 = max(0, -k)
    l2 = min(n, n - k)
    if l1 < l2:
        g[l1:l2] += (1.0 - h[l1+k:l2+k])
    l3 = max(0, k)
    l4 = min(n, n + k)
    if l3 < l4:
        g[l3:l4] -= h[l3-k:l4-k]
    
    # Step
    h_new = np.clip(h - step_size * g, 0.0, 1.0)
    h_new = project_to_sum(h_new, ts)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-12:
            continue
        k = j - n + 1
        wt = w[j]
        l1 = max(0, -k)
        l2 = min(n, n - k)
        if l1 < l2:
            g[l1:l2] += wt * (1.0 - h[l1+k:l2+k])
        l3 = max(0, k)
        l4 = min(n, n + k)
        if l3 < l4:
            g[l3:l4] -= wt * h[l3-k:l4-k]
    return sm_val, g

def slsqp_solve(h0, n, ts, alpha, maxiter=200):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)
