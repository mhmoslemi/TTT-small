def fast_smooth_max_gradient(h, alpha, threshold=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Only keep shifts within threshold of max
    active = (m - corr) < threshold * m  # or absolute threshold
    # Or: active = corr > m - 0.01
    
    # Compute weights for active shifts
    shifted = alpha * (corr[active] - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    
    # Smooth max value
    all_shifted = alpha * (corr - m)
    all_shifted = np.clip(all_shifted, -700, 700)
    sm_val = m + np.log(np.sum(np.exp(all_shifted)) / len(corr)) / alpha
    
    # Gradient: for each active shift s_i, compute ∇f_{s_i}
    g = np.zeros(n)
    active_indices = np.where(active)[0]
    for idx in active_indices:
        s = idx - (n - 1)  # actual shift
        # ∇f_s[j] = (1-h[j-s])*I(valid) - h[j+s]*I(valid)
        # h[j-s] valid when 0 <= j-s < n, i.e., s <= j < s+n
        # h[j+s] valid when 0 <= j+s < n, i.e., -s <= j < n-s
        
        # First term: (1-h[j-s]) for j in [max(0,s), min(n, s+n))
        j_start1 = max(0, s)
        j_end1 = min(n, s + n)
        if j_end1 > j_start1:
            g[j_start1:j_end1] += w[np.where(active)[0].tolist().index(idx)] * (1.0 - h[j_start1 - s : j_end1 - s])
        
        # Second term: -h[j+s] for j in [max(0,-s), min(n, n-s))
        j_start2 = max(0, -s)
        j_end2 = min(n, n - s)
        if j_end2 > j_start2:
            g[j_start2:j_end2] -= w[np.where(active)[0].tolist().index(idx)] * h[j_start2 + s : j_end2 + s]
    
    return sm_val, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target_sum):
    # Project h onto {0<=h<=1, sum(h)=target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if diff > 0 and np.any(mask & (h < 0.5)):
            m = mask & (h < 0.5)
            h[m] += diff / np.sum(m)
        elif diff < 0 and np.any(mask & (h > 0.5)):
            m = mask & (h > 0.5)
            h[m] += diff / np.sum(m)
        elif np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def time_remaining(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def fast_smooth_max_grad(h, alpha, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    n_corr = len(corr)
    shifted_all = np.clip(alpha * (corr - m), -500.0, 500.0)
    sm_val = m + np.log(np.sum(np.exp(shifted_all)) / n_corr) / alpha
    if K >= n_corr:
        top_indices = np.arange(n_corr)
        top_corr = corr
    else:
        top_indices = np.argpartition(corr, -K)[-K:]
        top_corr = corr[top_indices]
    shifted_top = np.clip(alpha * (top_corr - m), -500.0, 500.0)
    exp_top = np.exp(shifted_top)
    w = exp_top / np.sum(exp_top)
    g = np.zeros(n)
    for i in range(len(top_indices)):
        s = int(top_indices[i]) - (n - 1)
        wi = w[i]
        if s >= 0:
            L = n - s
            if L > 0:
                g[s:] += wi * (1.0 - h[:L])
                g[:L] -= wi * h[s:]
        else:
            ss = -s
            L = n - ss
            if L > 0:
                g[:L] += wi * (1.0 - h[ss:])
                g[ss:] -= wi * h[:L]
    return sm_val, g

def _slsqp_obj(x):
    n = _opt_state['n']
    alpha = _opt_state['alpha']
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted)) / (2 * n - 1)) / alpha

def _slsqp_grad(x):
    alpha = _opt_state['alpha']
    K = _opt_state['K']
    h = np.clip(x, 0.0, 1.0)
    _, g = fast_smooth_max_grad(h, alpha, K)
    return g

def _slsqp_cfun(x):
    return np.sum(x) - _opt_state['target_sum']

def _slsqp_cjac(x):
    return np.ones(len(x))

def nesterov_loop(h0, n, target_sum, max_iters, alpha, K, lr_init, t_start, budget):
    h = h0.copy()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    eval_count = 0
    for it in range(max_iters):
        if time_remaining(t_start, budget) < 2.0:
            break
        beta = (t - 1.0) / (t + 1.0)
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, g = fast_smooth_max_grad(y, alpha, K)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        h_prev = h
        h = h_new
        t += 1.0
        if it % 30 == 0 or it < 50:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 30
        if patience > 300:
            lr *= 0.92
            patience = 0
            if lr < 1e-7:
                break
        elif patience > 80:
            lr *= 0.98
        elif patience == 0 and lr < 0.4:
            lr *= 1.01
    return best_h, best_c5

def slsqp_polish(h0, n, target_sum, alpha, K, maxiter, t_start, budget):
    if time_remaining(t_start, budget) < 10.0:
        return h0, compute_c5(h0)
    _opt_state['n'] = n
    _opt_state['alpha'] = alpha
    _opt_state['K'] = K
    _opt_state['target_sum'] = target_sum
    h_start = project_to_sum(h0, target_sum)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h_start.copy(), jac=_slsqp_grad,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_to_sum(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        h_out = h_start
    return h_out, compute_c5(h_out)

def coordinate_descent(h0, n, target_sum, max_rounds, rng, t_start, budget):
    h = h0.copy()
    c5_cur = compute_c5(h)
    n_deltas = 10
    for rnd in range(max_rounds):
        if time_remaining(t_start, budget) < 3.0:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            orig = h[i]
            for dr in range(-n_deltas, n_deltas + 1):
                if dr == 0:
                    continue
                d = dr * 0.004
                h[i] = orig + d
                h_c = np.clip(h, 0.0, 1.0)
                h_c = project_to_sum(h_c, target_sum)
                c5_new = compute_c5(h_c)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    h = h_c.copy()
            if best_val < c5_cur - 1e-15:
                improved = True
            else:
                h[i] = orig
            c5_cur = min(c5_cur, best_val)
        if not improved:
            break
    return h, c5_cur

def make_periodic_start(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period + period) % period
        if phase < int(round(duty * period)):
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def make_sinusoidal_start(n, freq, phase, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(360, 0.5)
    
    n = len(best_h)
    target_sum = n / 2.0
    best_h = project_to_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    
    rem = time_remaining(t_start, budget_s)
    phase1_budget = rem * 0.45
    phase2_budget = rem * 0.15
    phase3_budget = rem * 0.12
    phase4_budget = rem * 0.10
    
    t_p1 = t_start + phase1_budget
    t_p2 = t_p1 + phase2_budget
    t_p3 = t_p2 + phase3_budget
    t_p4 = t_p3 + phase4_budget
    
    # Phase 1: Nesterov with moderate alpha, increasing K
    h_cur = best_h.copy()
    for sub in range(4):
        alpha_sub = 20.0 * (sub + 1)
        K_sub = 15 + 15 * sub
        lr_sub = 0.12 / (sub + 1)
        max_it_sub = max(5000, int(phase1_budget * 0.25 * 2000))
        h_cur, c5_tmp = nesterov_loop(h_cur, n, target_sum, max_it_sub,
                                       alpha_sub, K_sub, lr_sub, t_start, t_p1)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_cur.copy()
        if time_remaining(t_start, budget_s) < 30:
            break
    
    # Phase 2: Nesterov with higher alpha
    h_cur = best_h.copy()
    for sub in range(3):
        alpha_sub = 100.0 * (sub + 1)
        K_sub = 40 + 20 * sub
        lr_sub = 0.06 / (sub + 1)
        max_it_sub = max(3000, int(phase2_budget * 0.33 * 1500))
        h_cur, c5_tmp = nesterov_loop(h_cur, n, target_sum, max_it_sub,
                                       alpha_sub, K_sub, lr_sub, t_start, t_p2)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_cur.copy()
        if time_remaining(t_start, budget_s) < 30:
            break
    
    # Phase 3: SLSQP polish
    if time_remaining(t_start, budget_s) > 20:
        h_sls, c5_sls = slsqp_polish(best_h, n, target_sum, 300.0, 80, 300, t_start, t_p3)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()
        if time_remaining(t_start, budget_s) > 15:
            h_sls2, c5_sls2 = slsqp_polish(best_h, n, target_sum, 800.0, 120, 200, t_start, t_p3)
            if c5_sls2 < best_c5:
                best_c5 = c5_sls2
                best_h = h_sls2.copy()
    
    # Phase 4: Coordinate descent
    if time_remaining(t_start, budget_s) > 10:
        cd_rounds = max(20, int(phase4_budget / 3.0))
        h_cd, c5_cd = coordinate_descent(best_h, n, target_sum, cd_rounds, rng, t_start, t_p4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final polish with remaining time
    rem_final = time_remaining(t_start, budget_s)
    if rem_final > 20:
        h_fp, c5_fp = nesterov_loop(best_h, n, target_sum,
                                     max(2000, int(rem_final * 800)),
                                     500.0, 80, 0.02, t_start,
                                     t_start + rem_final * 0.6)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
        rem_after = time_remaining(t_start, budget_s)
        if rem_after > 8:
            h_cd2, c5_cd2 = coordinate_descent(best_h, n, target_sum,
                                                max(5, int(rem_after / 4.0)), rng,
                                                t_start, t_start + rem_after)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
    
    # Try alternative starting points with remaining time
    rem_alt = time_remaining(t_start, budget_s)
    if rem_alt > 40:
        alt_starts = []
        for period in [6, 8, 10, 12, 14, 18]:
            hs = make_periodic_start(n, period, 0.5)
            hs = project_to_sum(hs, target_sum)
            alt_starts.append(hs)
        for freq in [3, 5, 7, 11]:
            for phase in [0.0, 1.0, 2.0]:
                hs = make_sinusoidal_start(n, freq, phase, 0.3)
                hs = project_to_sum(hs, target_sum)
                alt_starts.append(hs)
        
        for hs in alt_starts:
            if time_remaining(t_start, budget_s) < 15:
                break
            c5_s = compute_c5(hs)
            if c5_s > best_c5 + 0.01:
                continue
            h_alt, c5_alt = nesterov_loop(hs, n, target_sum,
                                           max(2000, int(time_remaining(t_start, budget_s) * 500)),
                                           50.0, 30, 0.08, t_start,
                                           t_start + time_remaining(t_start, budget_s) * 0.7)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
    
    # Final validation
    best_h = project_to_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h, best_c5, n)
