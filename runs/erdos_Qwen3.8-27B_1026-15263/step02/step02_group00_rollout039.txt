import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n_corr)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def fista_optimize(h0, alpha, n_iter, target_sum, L=None):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    if L is None:
        L = 4.0 * alpha + 4.0
    
    for i in range(n_iter):
        if i > 0:
            beta = (t - 1.0) / t
            y = h + beta * (h - h_prev)
        else:
            y = h.copy()
        
        obj_y, grad_y = smooth_max_obj_and_grad(y, alpha)
        
        # Gradient step with backtracking
        step = 1.0 / L
        x_new = project_to_sum(y - step * grad_y, target_sum)
        
        # Simple convergence check: if step is too large, increase L
        # (skip for simplicity)
        
        h_prev = h
        h = x_new
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
    
    return h

def fourier_to_h(params, n, M):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        a = params[2 * (m - 1)]
        b = params[2 * (m - 1) + 1]
        h += a * np.cos(2.0 * np.pi * m * i / n) + b * np.sin(2.0 * np.pi * m * i / n)
    return h

def extract_fourier_params(h, n, M):
    params = np.zeros(2 * M)
    i = np.arange(n)
    for m in range(1, M + 1):
        basis_c = np.cos(2.0 * np.pi * m * i / n)
        basis_s = np.sin(2.0 * np.pi * m * i / n)
        params[2 * (m - 1)] = 2.0 * np.sum((h - 0.5) * basis_c) / n
        params[2 * (m - 1) + 1] = 2.0 * np.sum((h - 0.5) * basis_s) / n
    return params

def fourier_obj_and_grad(params, n, M, alpha):
    h = fourier_to_h(params, n, M)
    h_proj = project_to_sum(h, n / 2.0)
    obj, grad_h = smooth_max_obj_and_grad(h_proj, alpha)
    
    # Jacobian: J[i, 2(m-1)] = cos(2*pi*m*i/n), J[i, 2(m-1)+1] = sin(2*pi*m*i/n)
    i = np.arange(n)
    grad_params = np.zeros(2 * M)
    for m in range(1, M + 1):
        angle = 2.0 * np.pi * m * i / n
        grad_params[2 * (m - 1)] = np.sum(grad_h * np.cos(angle))
        grad_params[2 * (m - 1) + 1] = np.sum(grad_h * np.sin(angle))
    
    return obj, grad_params

def fourier_optimize(params0, n, M, alpha, rng, maxiter=500):
    def f(p):

    def g(p):

    try:

        return res.x

        return params0.copy()

            h[j] = h[j]  # restore (this doesn't work, need to save/restore properly)
        
        if best_j == -1:

        h = best_val
        c5_cur = compute_c5(h)

def projected_fista(h0, alpha, n_iter, target_sum):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_obj = smooth_max_obj_and_grad(h, alpha)[0]
    
    L = max(4.0 * alpha, 4.0) + n / max(alpha, 1.0) + 2.0
    
    t = 1.0
    h_prev = h0.copy()
    
    for i in range(n_iter):
        # FISTA momentum
        if i > 0 and i % 100 != 0:  # restart every 100 iterations
            beta = (t - 1.0) / t
            y = h + beta * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)

            y = h.copy()
            t = 1.0
            h_prev = h.copy()

        h = x_new
        if i % 100 != 0:

        # Track best
        obj_h, _ = smooth_max_obj_and_grad(h, alpha)
        if obj_h < best_obj:
            best_obj = obj_h
            best_h = h.copy()
    
    return best_h

But in FISTA, I need to evaluate the gradient at y (the momentum point), not at x. The momentum point y might not satisfy the constraint (0 ≤ y ≤ 1, sum(y) = target_sum). 

In standard FISTA for constrained problems, the typical approach is:
1. y = x + momentum
2. x_new = P(y - η ∇f(y))

where P is the projection onto the constraint set. The gradient is evaluated at y (which is unconstrained), and the projection handles the constraint.

A simpler approach: just use the projected gradient method (no momentum):
x_{k+1} = P(x_k - η ∇f(x_k))

This is simpler and more robust. The convergence rate is O(1/k) instead of O(1/k^2), but for our purposes it should be fine.
