```python
def compute_gradient(h, n, tau, k_active=None):
    """Compute gradient of smoothed max correlation."""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')  # length 2n-1
    
    if tau < 1e-10:
        k_star = np.argmax(corr)
        weights = np.zeros(2*n-1)
        weights[k_star] = 1.0
    else:
        v = corr / tau
        vmax = np.max(v)
        weights = np.exp(v - vmax)
        weights /= np.sum(weights)
    
    # Compute gradient: grad[i] = sum_k weights[k] * dC_k/dh[i]
    # dC_k/dh[i] = dx * (1-h[i+k]) * 1_{valid} - dx * h[i-k] * 1_{valid}
    # where k = k_index - (n-1)
    
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - (n-1)  # actual shift
        # Terms where i is first index: need i+k in [0,n-1]
        # i ranges from max(0,-k) to min(n-1, n-1-k)
        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)
        if i_start <= i_end:
            grad[i_start:i_end+1] += weights[k_idx] * dx * (1 - h[i_start+k:i_end+k+1])
        # Terms where i is second index: h[i-k]*(1-h[i]) -> -h[i-k]
        # i ranges from max(0,k) to min(n-1, n-1+k) [i-k in [0,n-1] means i in [k, k+n-1] intersect [0,n-1]]
        i_start2 = max(0, k)
        i_end2 = min(n-1, n-1+k)
        if i_start2 <= i_end2:
            grad[i_start2:i_end2+1] -= weights[k_idx] * dx * h[i_start2-k:i_end2-k+1]
    
    return grad
```

This is O(n²) per gradient evaluation (loop over 2n-1 shifts, each doing O(n) work). For n=77, that's about 2*77*77 ≈ 12K operations, very fast.

```python
def evaluate_c5(h):

    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference among free variables
        free = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(free) > 0:
            adj = diff / len(free)
            result[free] = np.clip(result[free] + adj, 0, 1)
            # Re-project if needed
            remaining = target - np.sum(result)
            if abs(remaining) > 1e-12:
                # Add to variable closest to 0.5
                idx = free[np.argmin(np.abs(result[free] - 0.5))]
                result[idx] = np.clip(result[idx] + remaining, 0, 1)
    return result

        if w < 1e-12:

        i_hi = min(n - 1, n - 1 - k)

        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w * dx * h[i_lo2-k:i_hi2-k+1]

    h_prev = h0.copy()
    y = h0.copy()

        alpha = (1.0 + np.sqrt(1.0 + 4.0 * (t) ** 2)) / 2.0
        beta = (1.0 - 1.0 / alpha) if t > 0 else 0.0

        # Gradient step
        h_prev = h.copy()
        h = y - lr * grad

        # Track best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
```

```python
    g = 1.0 - h

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        free = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(free) > 0:

                result[idx] += adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
    return result

def compute_subgradient(h, n):
    """Compute subgradient of max_k C_k at h. Returns (grad, k_star, max_val)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    max_val = corr[k_star]
    k = k_star - (n - 1)
    
    grad = np.zeros(n)
    # C_k = sum_{j in S_k} h[j]*(1-h[j+k])*dx
    # dC_k/dh[i] = dx*(1-h[i+k])*1_{i+k valid} - dx*h[i-k]*1_{i-k valid}
    i_lo = max(0, -k)
    i_hi = min(n - 1, n - 1 - k)
    if i_lo <= i_hi:
        grad[i_lo:i_hi+1] += dx * (1.0 - h[i_lo+k:i_hi+k+1])
    i_lo2 = max(0, k)
    i_hi2 = min(n - 1, n - 1 + k)
    if i_lo2 <= i_hi2:
        grad[i_lo2:i_hi2+1] -= dx * h[i_lo2-k:i_hi2-k+1]
    
    return grad, k_star, max_val

def compute_smoothed_subgradient(h, n, tau):
    """Compute gradient of smoothed max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wi = w[k_idx]
        if wi < 1e-10:
            continue
        k = k_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += wi * dx * (1.0 - h[i_lo+k:i_hi+k+1])
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, n - 1 + k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= wi * dx * h[i_lo2-k:i_hi2-k+1]
    
    return grad

def nesterov_optimize(h0, n, tau, num_iters, lr_init=0.1):

    h = h0.astype(float).copy()
    h_prev = h.copy()
    best = h.copy()
    best_c5 = evaluate_c5(best)
    lr = lr_init

        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - 1.0 / alpha) if t > 0 else 0.0
        y = h + beta * (h - h_prev)
        
        grad = compute_smoothed_subgradient(y, n, tau)
        
        h_prev = h.copy()
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            h = h_new.copy()
        else:
            # Step back
            lr *= 0.95
            h = h_prev.copy()
    
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        free = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(free) > 0:
            for idx in free:
                adj = min(max(diff, -(result[idx]), 1.0 - result[idx]), 1.0 - result[idx])
                result[idx] += adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
    return result

def compute_smoothed_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        k = k_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += wi * dx * (1.0 - h[i_lo+k:i_hi+k+1])
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, n - 1 + k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= wi * dx * h[i_lo2-k:i_hi2-k+1]
    return grad

def nesterov_opt(h0, n, tau, num_iters, lr_init=0.05):
    h = h0.astype(float).copy()
    h_prev = h.copy()
    best = h.copy()
    best_c5 = evaluate_c5(best)
    lr = lr_init
    for t in range(num_iters):
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - 1.0 / alpha) if t > 0 else 0.0
        y = h + beta * (h - h_prev)
        grad = compute_smoothed_grad(y, n, tau)
        h_prev = h.copy()
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            h = h_new.copy()
        else:
            lr *= 0.9
            h = h_prev.copy()
    return best, best_c5

def corr_eval(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def pair_sweep(h0, n, rng, deadline, subsample=None):
    best = h0.astype(float).copy()
    best_c5 = evaluate_c5(best)
    stall = 0
    idx_pairs = None
    if subsample is None:
        idx_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    while time.time() < deadline:
        improved = False
        if subsample is not None:
            pairs = [(rng.randint(n), rng.randint(n)) for _ in range(subsample)]
        else:
            order = list(range(n))
            rng.shuffle(order)
            pairs = [(order[a], order[b]) for a in range(n) for b in range(a + 1, n)]
        for i, j in pairs:
            if i == j:
                continue
            d_min = max(-best[i], best[j] - 1.0)
            d_max = min(1.0 - best[i], best[j])
            if d_min >= d_max - 1e-10:
                continue
            span = d_max - d_min
            candidates = [0.5 * (d_min + d_max), d_min + 0.25 * span, d_min + 0.75 * span]
            for d in candidates:
                if d <= d_min + 1e-12 or d >= d_max - 1e-12:
                    continue
                best[i] += d
                best[j] -= d
                c5 = evaluate_c5(best)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                    break
                best[i] -= d
                best[j] += d
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best, best_c5
```
