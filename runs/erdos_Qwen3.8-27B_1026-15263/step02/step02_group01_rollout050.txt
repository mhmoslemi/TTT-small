The problem min_h max_k C(k) where C(k) = sum_{i=0}^{n-1-|k|} h[i+|k|] * (1-h[i]) (for k>0) and similar for k<0.

This is related to the concept of "balanced sequences" or "low-discrepancy sequences" in a sense.

Actually, I recall that for the Erdős problem, the optimal constructions often involve characteristic functions of sets with specific additive properties. The set A = {i : h[i] = 1} should have the property that |A ∩ (A+k)| is as uniform as possible (close to |A|(n-|k|)/n for all k).

Wait, we want to MINIMIZE C(k) = |{i: i+k ∈ A, i ∉ A}| = |(A-k) \ A| (for positive k).
And by symmetry, we also care about |{i: i ∈ A, i+k ∉ A}| = |A \ (A+k)|.

Note that |(A-k) \ A| + |A \ (A-k)| = |A Δ (A-k)| = |A| + |A-k| - 2|A ∩ (A-k)|

For the full correlation (both positive and negative lags), C(k) for k>0 is |(A-k) \ A| and C(-k) is |A \ (A-k)|.

So max_k C(k) = max over positive k of max(|(A-k)\A|, |A\(A-k)|).

Now |(A-k)\A| = |A-k| - |A ∩ (A-k)| ≈ |A| - |A ∩ (A-k)| (for small k)
And |A\(A-k)| = |A| - |A ∩ (A-k)| (exactly, since A and A-k have the same size)

Wait: |A| = |A-k| = n/2 (for the relevant range). And |A ∩ (A-k)| = A(k) (autocorrelation).
So |(A-k)\A| = |A-k| - |A ∩ (A-k)| = n/2 - A(k)
And |A\(A-k)| = |A| - |A ∩ (A-k)| = n/2 - A(k)

They're the same! So C(k) = C(-k) = n/2 - A(k) for 0-1 valued h.

Wait, that's not quite right because of boundary effects. For the finite sum:
C(k) = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
This counts pairs (i, i+k) where i+k is in A and i is not in A, for i in [0, n-1-k].

The total number of such pairs is:
|(A ∩ [k, n-1]) \ ([0, n-1-k] + k)|... this is getting confusing with the finite bounds.

Let me just accept that for practical purposes, C(k) ≈ n/2 - A(k) (for 0-1 h), and the goal is to maximize min_k A(k).

For a "good" set A with |A| = n/2, we want A(k) = |A ∩ (A-k)| to be as large and uniform as possible.

The maximum possible A(0) = |A| = n/2.
The average of A(k) over all k is approximately |A|²/n = n/4.

So we want A(k) to be close to n/4 for all k (but it can't exceed n/2-k for lag k).

If A(k) ≈ n/4 for all k, then C(k) ≈ n/2 - n/4 = n/4, and C5 ≈ (n/4) * (2/n) = 1/2.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_subgradient(h, M=5):
    """Compute subgradient of max_k corr_k using top M active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    
    # Get top M lag indices
    if M >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -M)[-M:]
    
    grad = np.zeros(n)
    n_corr = len(corr)
    n_half = n - 1
    
    for idx in top_idx:
        k = idx - n_half  # lag: negative to positive
        if k == 0:
            # f_0 = sum h[i]*(1-h[i]), grad = 1-2h
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            # f_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j] for j in [k, n-1]: (1-h[j-k])
            # d/dh[j] for j in [0, n-1-k]: -h[j+k]
            if n - k > 0:
                idx1 = np.arange(k, n)
                grad[idx1] += (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= h[idx2 + k]
        else:
            m = -k  # m > 0
            # f_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # d/dh[j] for j in [0, n-1-m]: (1-h[j+m])
            # d/dh[j] for j in [m, n-1]: -h[j-m]
            if n - m > 0:
                idx1 = np.arange(0, n - m)
                grad[idx1] += (1.0 - h[idx1 + m])
                idx2 = np.arange(m, n)
                grad[idx2] -= h[idx2 - m]
    
    grad /= float(len(top_idx))
    return grad, max_val

def active_subgradient_weighted(h, M=10, temperature=1.0):
    """Compute weighted subgradient with softmax weights on active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    
    if M >= len(corr):
        top_idx = np.arange(len(corr))
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        # Get top M
        part_idx = np.argpartition(corr, -M)[-M:]
        part_idx = part_idx[np.argsort(corr[part_idx])[::-1]]
        top_idx = part_idx
        # Softmax weights
        vals = corr[top_idx]
        shifted = (vals - max_val) / temperature
        weights = np.exp(shifted)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx, w in zip(top_idx, weights):
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                idx1 = np.arange(k, n)
                grad[idx1] += w * (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= w * h[idx2 + k]
        else:
            m = -k
            if n - m > 0:
                idx1 = np.arange(0, n - m)
                grad[idx1] += w * (1.0 - h[idx1 + m])
                idx2 = np.arange(m, n)
                grad[idx2] -= w * h[idx2 - m]
    
    return grad, max_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, M=10, temp=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = float(np.max(corr))
    
    n_corr = 2 * n - 1
    if M >= n_corr:
        top_idx = np.arange(n_corr)
        w = np.ones(n_corr) / n_corr
    else:
        top_idx = np.argpartition(corr, -M)[-M:]
        top_vals = corr[top_idx]
        if temp > 1e-10:
            shifted = (top_vals - max_val) / temp
            w = np.exp(shifted)
            w /= w.sum()
        else:
            # Find which ones achieve the max
            max_count = np.sum(np.abs(top_vals - max_val) < 1e-12)
            w = np.zeros(M)
            max_mask = np.abs(top_vals - max_val) < 1e-12
            w[max_mask] = 1.0 / max_count
            if not np.any(max_mask):
                w[np.argmax(top_vals)] = 1.0
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for ii in range(len(top_idx)):
        idx = top_idx[ii]
        kk = w[ii]
        k = idx - n_half
        if k == 0:
            grad += kk * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                grad[k:] += kk * (1.0 - h[:n-k])
                grad[:n-k] -= kk * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += kk * (1.0 - h[m:])
                grad[m:] -= kk * h[:n-m]
    
    return grad, max_val

def nesterov_subgrad(h0, n, target_sum, max_iters, step_init, M=10, temp=0.01):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    step = step_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        grad, c5_y = get_subgradient(y, M, temp)
        # Projected gradient step
        h_new = np.clip(y - step * grad, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        
        h = h_new
        t = t_new
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.01, 1.0)
        else:
            step *= 0.95
        
        # Adaptive temperature: decrease over time for sharper focus
        temp = max(0.005, 0.1 * (1.0 - it / max_iters))
    
    return best_h, best_c5

def slsqp_refine(h0, n, target_sum, alpha, maxiter=200):
    """Use SLSQP with smooth max for final polishing."""
    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_ = corr * (2.0 / n)
        m = np.max(corr_)
        # Smooth max
        shifted = alpha * (corr_ - m)
        shifted = np.clip(shifted, -300.0, 300.0)
        exp_v = np.exp(shifted)
        s = np.sum(exp_v)
        w = exp_v / s
        val = m + np.log(s / len(exp_v)) / alpha
        
        # Gradient using active lags (top few)
        M = max(1, min(20, len(corr_)))
        top_idx = np.argpartition(corr_, -M)[-M:]
        top_vals = corr_[top_idx]
        shifted_t = (top_vals - m) / (1.0/alpha)
        wt = np.exp(shifted_t)
        wt /= wt.sum()
        
        grad = np.zeros(n)
        n_half = n - 1
        for ii in range(len(top_idx)):
            idx = top_idx[ii]
            kk = wt[ii]
            k = idx - n_half
            if k == 0:
                grad += kk * (1.0 - 2.0 * h)
            elif k > 0:
                if n - k > 0:
                    grad[k:] += kk * (1.0 - h[:n-k])
                    grad[:n-k] -= kk * h[k:]
            else:
                m2 = -k
                if n - m2 > 0:
                    grad[:n-m2] += kk * (1.0 - h[m2:])
                    grad[m2:] -= kk * h[:n-m2]
        
        return val, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        f, g = obj_and_grad(h0.copy())
        res = minimize(lambda x: obj_and_grad(x)[0], h0.copy(),
                      jac=lambda x: obj_and_grad(x)[1],
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def coordinate_descent(h0, n, target_sum, max_rounds, rng, n_cand=8):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    # Adaptive coordinate selection: try all coordinates but use larger steps
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_h_i = h[i]
            # Binary search for optimal value at coordinate i
            lo = max(0.0, h[i] - 0.5)
            hi = min(1.0, h[i] + 0.5)
            for _ in range(5):
                mid = (lo + hi) / 2
                h_try = h.copy()
                h_try[i] = mid
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_h_i = mid
                # Simple: just try a few candidate values
            # Try a few specific values
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0, h[i] - 0.1, h[i] + 0.1, h[i]]
            for c in candidates:
                c = np.clip(c, 0.0, 1.0)
                h_try = h.copy()
                h_try[i] = c
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_h_i = c
            
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, M, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = float(np.max(corr))
    n_corr = 2 * n - 1
    n_half = n - 1
    if M >= n_corr:
        top_idx = np.arange(n_corr)
        w = np.ones(n_corr) / n_corr
    else:
        top_idx = np.argpartition(corr, -M)[-M:]
        top_vals = corr[top_idx]
        if temp > 1e-10:
            shifted = (top_vals - max_val) / temp
            w = np.exp(shifted)
            w /= w.sum()
        else:
            max_mask = np.abs(top_vals - max_val) < 1e-12
            w = np.zeros(M)
            cnt = np.sum(max_mask)
            if cnt > 0:
                w[max_mask] = 1.0 / cnt
            else:
                w[np.argmax(top_vals)] = 1.0
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        idx = int(top_idx[ii])
        kk = w[ii]
        k = idx - n_half
        if k == 0:
            grad += kk * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                grad[k:] += kk * (1.0 - h[:n-k])
                grad[:n-k] -= kk * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += kk * (1.0 - h[m:])
                grad[m:] -= kk * h[:n-m]
    return grad, max_val

def nesterov_subgrad(h0, n, target_sum, max_iters, step_init, M, temp0, temp_min):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    step = step_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall_count = 0
    for it in range(max_iters):
        temp = temp_min + (temp0 - temp_min) * max(0.0, 1.0 - it / float(max_iters))
        grad, _ = get_subgradient(y, M, temp)
        h_new = np.clip(y - step * grad, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.02, 2.0)
            stall_count = 0

def smooth_max_obj_grad(h, n, alpha, M):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -300.0, 300.0)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    val = m + np.log(s / len(exp_v)) / alpha
    n_corr = len(corr)
    n_half = n - 1
    if M >= n_corr:
        top_idx = np.arange(n_corr)
        w = exp_v / s
    else:
        top_idx = np.argpartition(corr, -M)[-M:]
        top_vals = corr[top_idx]
        shifted_t = (top_vals - m) / (1.0 / alpha)
        shifted_t = np.clip(shifted_t, -300.0, 300.0)
        wt = np.exp(shifted_t)
        wt /= wt.sum()
        w = wt
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        idx = int(top_idx[ii])
        kk = w[ii]
        k = idx - n_half
        if k == 0:
            grad += kk * (1.0 - 2.0 * h)

                grad[m2:] -= kk * h[:n-m2]
    return val, grad

def slsqp_refine(h0, n, target_sum, alpha, maxiter, M):
    x0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    sum_const = target_sum
    n_ref = n
    alpha_ref = alpha
    M_ref = M
    def obj_func(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_obj_grad(h, n_ref, alpha_ref, M_ref)
        return v
    def grad_func(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_obj_grad(h, n_ref, alpha_ref, M_ref)
        return g
    def cons_fun(x):
        return np.sum(x) - sum_const
    def cons_jac(x):
        return np.ones(n_ref)
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def local_perturb_search(h0, n, target_sum, rng, n_trials, max_rounds):

    for rnd in range(max_rounds):

        for _ in range(n_trials):
            idx = rng.integers(0, n, size=rng.integers(1, n // 10 + 1))
            noise_scale = 0.05 + 0.1 * rng.random()
            h_try = h.copy()
            h_try[idx] += rng.uniform(-noise_scale, noise_scale, size=len(idx))
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-14:
                h = h_try
                c5_cur = c5_try
                improved = True
        if not improved:
            break
    return h, c5_cur

def coordinate_refine(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_h_i = h[i]
            candidates = [0.0, 1.0, h[i] - 0.2, h[i] - 0.1, h[i] - 0.05,
                         h[i] + 0.05, h[i] + 0.1, h[i] + 0.2, 0.5]
            for c in candidates:
                c = float(np.clip(c, 0.0, 1.0))
                h_try = h.copy()
                h_try[i] = c
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_h_i = c
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
```
