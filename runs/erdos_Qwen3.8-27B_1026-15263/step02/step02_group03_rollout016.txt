import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def obj_func(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def constraint_func(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def slsqp_step(h, n, tau, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
    try:
        res = minimize(
            obj_func, h, method='SLSQP', args=(n, tau),
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 0.002)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            valid_i = range(n - s)
            for i in valid_i:
                grad[i] += (1.0 - h[i + s])
                grad[i + s] -= h[i]
        else:
            ss = -s
            valid_i = range(n - ss)
            for i in valid_i:
                grad[i + ss] += (1.0 - h[i])
                grad[i] -= h[i + ss]
    return grad
