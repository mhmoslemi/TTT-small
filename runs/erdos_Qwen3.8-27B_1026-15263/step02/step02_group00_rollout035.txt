- Second factor: h[j] appears as h[i+shift] when j-shift ≥ 0 and j-shift ≤ n-1-shift, i.e., shift ≤ j ≤ n-1, contributes -h[j-shift]

So for shift ≥ 0:

def project_to_sum(h, target_sum):

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):

def make_fourier_starts(n, M, rng, existing_h=None):

    # Zero (constant 0.5)
    starts.append(np.zeros(2 * M))
    # Single frequency

        for amp in [0.3, 0.4, 0.45]:
            p = np.zeros(2 * M)
            p[2*(m-1)] = amp
            starts.append(p.copy())
            p[2*(m-1)] = 0
            p[2*(m-1)+1] = amp
            starts.append(p.copy())
    # Random
    for _ in range(8):

    # From existing_h Fourier coefficients

        h_centered = existing_h - 0.5
        a_m = np.array([np.mean(h_centered * np.cos(np.pi * m * x)) for m in range(1, M+1)])
        b_m = np.array([np.mean(h_centered * np.sin(np.pi * m * x)) for m in range(1, M+1)])
        # Scale to ensure |h| <= 1 approximately
        p = np.zeros(2 * M)
        for m in range(M):
            p[2*m] = a_m[m]
            p[2*m+1] = b_m[m]
        # Clip amplitudes

            if amp > 0.4:
                scale = 0.4 / amp
                p[2*m] *= scale
                p[2*m+1] *= scale
        starts.append(p)
    return starts

def binary_flip_search(h, n, target_sum, rng, max_rounds=100):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_i = -1
        best_j = -1
        n_trials = min(400, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            old_i, old_j = h[i], h[j]
            h[i], h[j] = 0.0, 1.0
            c5_test = compute_c5(h)
            h[i], h[j] = old_i, old_j
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

    for _ in range(3):
        p = rng.uniform(-0.3, 0.3, 2 * M)
        p[2 * (min(M, 3) - 1) + 1] = rng.uniform(0.2, 0.45)
        starts.append(p)
    if existing_h is not None:
        coeffs = get_fourier_coeffs(existing_h, M)
        for m in range(M):
            amp = np.sqrt(coeffs[2 * m] ** 2 + coeffs[2 * m + 1] ** 2)
            if amp > 0.45:
                scale = 0.45 / amp
                coeffs[2 * m] *= scale
                coeffs[2 * m + 1] *= scale
        starts.append(coeffs.copy())
    return starts

def binary_flip_search(h, n, target_sum, rng, max_rounds=100):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_i = -1
        best_j = -1
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            old_i, old_j = h[i], h[j]
            h[i], h[j] = 0.0, 1.0
            c5_test = compute_c5(h)
            h[i], h[j] = old_i, old_j
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

def systematic_binary_search(h, n, target_sum, max_rounds=50):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_i = -1
        best_j = -1
        step = max(1, len(ones) // 20)
        for ii in range(0, len(ones), step):
            i = ones[ii]
            step2 = max(1, len(zeros) // 20)
            for jj in range(0, len(zeros), step2):
                j = zeros[jj]
                h[i], h[j] = 0.0, 1.0
                c5_test = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if c5_test < best_c5 - 1e-12:
                    best_c5 = c5_test
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = 0.5

    n_curr = best_n
    ts = n_curr / 2.0

    for M in [3, 5, 8, 12, 16, 20, 30]:
        if time_left() < 60:
            break
        starts = make_fourier_starts(n_curr, M, rng, best_h)
        for alpha in [10, 40, 150]:
            if time_left() < 20:
                break
            for p_init in starts:
                if time_left() < 10:
                    break
                h_f = fourier_optimize_single(n_curr, M, alpha, p_init.copy(), maxiter=400)
                if h_f is not None:
                    c5_f = compute_c5(h_f)
                    if c5_f < best_c5:
                        best_c5 = c5_f
                        best_h = h_f.copy()

    for alpha in [5, 20, 60, 200, 600, 2000, 6000]:
        if time_left() < 40:
            break
        h_opt = slsqp_optimize(best_h, n_curr, ts, alpha, maxiter=1000)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 120:
        x = np.linspace(0, 2, n_curr, endpoint=False)
        ms_starts = []
        ms_starts.append(best_h.copy())
        for _ in range(4):
            hb = (rng.rand(n_curr) > 0.5).astype(float)
            attempts = 0
            while abs(hb.sum() - ts) > 1e-10 and attempts < 8000:
                if hb.sum() > ts:
                    idx = int(rng.choice(np.where(hb > 0)[0]))
                    hb[idx] = 0.0
                else:
                    idx = int(rng.choice(np.where(hb < 1)[0]))
                    hb[idx] = 1.0
                attempts += 1
            if abs(hb.sum() - ts) < 1e-10:
                ms_starts.append(hb)
        for period in [3, 4, 5, 6, 7, 8, 9]:
            pat = np.zeros(n_curr)
            fill = period // 2
            for i in range(n_curr):
                if (i % period) < fill:
                    pat[i] = 1.0
            ms_starts.append(project_to_sum(pat, ts))
        for freq in [2, 3, 4, 5, 7, 10, 13]:
            s = 0.5 + 0.44 * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            ms_starts.append(project_to_sum(s, ts))
        for _ in range(3):
            noise = rng.uniform(0, 0.3, n_curr)
            h_noise = best_h + noise - 0.15
            ms_starts.append(project_to_sum(h_noise, ts))

        for h_start in ms_starts:
            if time_left() < 30:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
            for alpha in [20, 200]:
                if time_left() < 20:
                    break
                h_opt = slsqp_optimize(h_start, n_curr, ts, alpha, maxiter=600)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_start = h_opt

    if time_left() > 60:
        h_bin = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin.sum() - ts) > 1e-10 and attempts < 10000:
            if h_bin.sum() > ts:
                idx = int(rng.choice(np.where(h_bin > 0)[0]))
                h_bin[idx] = 0.0
            else:
                idx = int(rng.choice(np.where(h_bin < 1)[0]))
                h_bin[idx] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts) < 1e-10:
            h_bf, c5_bf = binary_flip_search(h_bin, n_curr, ts, rng, max_rounds=60)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
            h_sys, c5_sys = systematic_binary_search(h_bf, n_curr, ts, max_rounds=30)
            if c5_sys < best_c5:
                best_c5 = c5_sys
                best_h = h_sys.copy()
            h_sm = slsqp_optimize(h_bf, n_curr, ts, 300, maxiter=800)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    for n_hi in [300, 360, 420, 480]:
        if time_left() < 120:
            break
        ts_hi = n_hi / 2.0
        if abs(ts_hi - round(ts_hi)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, n_curr, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            n_curr = n_hi
            ts = ts_hi
        for alpha in [20, 200, 1000]:
            if time_left() < 50:
                break
            h_opt = slsqp_optimize(h_hi, n_hi, ts_hi, alpha, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n_curr = n_hi
                ts = ts_hi
            h_hi = h_opt
        M_hi = min(30, n_hi // 8)
        if time_left() > 60:
            starts_hi = make_fourier_starts(n_hi, M_hi, rng, best_h)
            for alpha_f in [20, 100]:
                if time_left() < 20:
                    break
                for p_init in starts_hi[:6]:
                    if time_left() < 10:
                        break
                    h_f = fourier_optimize_single(n_hi, M_hi, alpha_f, p_init.copy(), maxiter=300)
                    if h_f is not None:
                        c5_f = compute_c5(h_f)
                        if c5_f < best_c5:
                            best_c5 = c5_f
                            best_h = h_f.copy()
                            n_curr = n_hi
                            ts = ts_hi

    for trial in range(20):
        if time_left() < 30:
            break
        noise_amp = 0.1 * (1.0 - trial / 20.0) + 0.02
        h_shake = best_h + rng.uniform(-noise_amp, noise_amp, n_curr)
        h_shake = project_to_sum(h_shake, ts)
        for alpha in [50, 300, 1500]:
            if time_left() < 20:
                break
            h_opt = slsqp_optimize(h_shake, n_curr, ts, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_shake = h_opt

    if time_left() > 30:
        for alpha in [5000, 20000, 50000]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, n_curr, ts, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_curr)
