```python
    y = h0.copy()
    t = 1.0
    lr = lr_init
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for it in range(n_iters):
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr(y)
        k_top = max(10, int(topk_frac * len(corr)))
        g = topk_grad(corr, y, k_top, alpha)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        frac = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(frac):
            h[frac] += diff / np.count_nonzero(frac)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    ev = np.exp(shifted)
    return m + np.log(np.sum(ev) / len(ev)) / alpha

def lse_grad(corr, h, alpha):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    g = np.zeros(n)
    for i in range(len(corr)):
        if w[i] < 1e-16:
            continue
        d = i - (n - 1)
        k1s = max(0, d)
        k1e = min(n, d + n)
        if k1s < k1e:
            g[k1s:k1e] += w[i] * (1.0 - h[k1s - d:k1e - d])
        k2s = max(0, -d)
        k2e = min(n, n - d)
        if k2s < k2e:
            g[k2s:k2e] -= w[i] * h[k2s + d:k2e + d]
    return g

def topk_grad(corr, h, k_top, alpha):
    n = len(h)
    n_corr = len(corr)
    if k_top >= n_corr:
        return lse_grad(corr, h, alpha)
    idx = np.argsort(corr)[::-1][:k_top]
    vals = corr[idx]
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -500.0, 500.0)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    g = np.zeros(n)
    for ii in range(k_top):
        if w[ii] < 1e-16:
            continue
        i = idx[ii]
        d = i - (n - 1)
        k1s = max(0, d)
        k1e = min(n, d + n)
        if k1s < k1e:
            g[k1s:k1e] += w[ii] * (1.0 - h[k1s - d:k1e - d])
        k2s = max(0, -d)
        k2e = min(n, n - d)
        if k2s < k2e:
            g[k2s:k2e] -= w[ii] * h[k2s + d:k2e + d]
    return g

def nesterov_opt(h0, n, target_sum, alpha, n_iters, lr_init, topk_frac):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    lr = lr_init
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(n_iters):
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr(y)
        k_top = max(20, int(topk_frac * len(corr)))
        g = topk_grad(corr, y, k_top, alpha)
        h_new = y - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.97
    return best_h, best_c5

def fast_coord_descent(h, n, target_sum, max_rounds, rng):
    h = h.copy()
    corr = compute_corr(h)
    max_corr = np.max(corr)
    dx = 2.0 / n
    c5_cur = max_corr * dx
    best_h = h.copy()
    best_c5 = c5_cur
    deltas = [-0.03, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.03]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            sens = np.zeros(2 * n - 1)
            k1s = i
            k1e = min(i + n, 2 * n - 1)
            if k1s < k1e:
                js = i + n - 1 - np.arange(k1s, k1e)
                sens[k1s:k1e] += (1.0 - h[js])
            k2s = max(0, n - 1 - i)
            k2e = min(2 * n - 2 - i + 1, 2 * n - 1)
            if k2s < k2e:
                js2 = i + np.arange(k2s, k2e) - n + 1
                sens[k2s:k2e] -= h[js2]
            best_new_max = max_corr
            best_d = 0.0
            for d in deltas:
                nv = h[i] + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                new_max = np.max(corr + d * sens)
                if new_max < best_new_max - 1e-13:
                    best_new_max = new_max
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                corr = compute_corr(h)
                max_corr = np.max(corr)
                c5_cur = max_corr * dx
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return h, compute_c5(h)

def gauss_start(n, n_pulses, sigma_scale, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for _ in range(n_pulses):
        center = rng.uniform(0, 2)
        amp = rng.uniform(-0.3, 0.3)
        sigma = sigma_scale * 2.0
        h += amp * np.exp(-0.5 * ((x - center) / sigma) ** 2)
    h = project_to_sum(h, n / 2.0)
    return h

def block_start(n, n_blocks, rng):
    h = np.zeros(n)
    total = n // (2 * n_blocks)
    if total < 2:
        total = 2
    for i in range(n_blocks):
        start = int(i * 2 * total + total)
        end = min(start + total, n)
        h[start:end] = 1.0
    return project_to_sum(h, n / 2.0)

def freq_start(n, freq, amp, phase, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_to_sum(h, n / 2.0)

def threshold_01(h, n):
    target = n / 2.0
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    cands = np.where(np.abs(cumsum - target) < 0.01)[0]
    if len(cands) == 0:
        best_k = np.argmin(np.abs(cumsum - target))
        k_count = best_k + 1
    else:
        best_k = cands[0]
        k_count = best_k + 1
    thresh = sorted_h[min(k_count - 1, n - 1)]
    h01 = (h >= thresh).astype(float)
    n_ones = int(np.sum(h01))
    if n_ones != int(target):
        h01 = project_to_sum(np.where(h >= thresh, 0.5, 0.5), target)
        h01 = (h01 > 0.25).astype(float)
        h01 = project_to_sum(h01, target)
    return h01

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_to_sum(h_interp, n_new / 2.0)

def gen_starts(n, best_h, best_n, rng):
    starts = []
    if best_h is not None:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(interpolate_h(best_h, n))
    starts.append(np.full(n, 0.5))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [3, 5, 7]:
        for phase in [0.0, 1.0, 2.0]:
            starts.append(freq_start(n, freq, 0.35, phase, rng))
    starts.append(gauss_start(n, 8, 0.15, rng))
    starts.append(gauss_start(n, 16, 0.08, rng))
    starts.append(block_start(n, 5, rng))
    starts.append(block_start(n, 8, rng))
    hr = rng.uniform(0.0, 1.0, n)
    starts.append(project_to_sum(hr, n / 2.0))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(180, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    n_targets = [360, 480, 600, 720, 900]

    for n in n_targets:
        if time_left() < 80:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        starts = gen_starts(n, best_h, best_n, rng)

        for h_start in starts:
            if time_left() < 50:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()
            alphas = [20.0, 60.0, 150.0, 300.0]
            for alpha in alphas:
                if time_left() < 30:
                    break
                n_iters = min(3000, max(500, int(time_left() * 1.2)))
                h_opt, c5_opt = nesterov_opt(h_cur, n, ts, alpha, n_iters, 0.05, 0.3)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt

            if time_left() > 40:
                n_rounds = min(120, max(30, int(time_left() * 0.4)))
                h_cd, c5_cd = fast_coord_descent(h_cur, n, ts, n_rounds, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    if best_h is not None and time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [500.0, 1000.0]:
            if time_left() < 20:
                break
            n_iters = min(2000, int(time_left() * 0.5))
            h_opt, c5_opt = nesterov_opt(best_h, n_f, ts_f, alpha, n_iters, 0.02, 0.15)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        if time_left() > 20:
            h_cd, c5_cd = fast_coord_descent(best_h, n_f, ts_f, 200, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 15:
            h01 = threshold_01(best_h, n_f)
            h01 = project_to_sum(h01, ts_f)
            c5_01 = compute_c5(h01)
            if c5_01 < best_c5:
                best_c5 = c5_01
                best_h = h01.copy()
                if time_left() > 15:
                    h_cd, c5_cd = fast_coord_descent(best_h, n_f, ts_f, 100, rng)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()

    if time_left() > 25:
        n_up = min(1200, best_n * 2)
        if n_up % 2 == 0 and n_up > best_n:
            h_up = interpolate_h(best_h, n_up)
            ts_up = n_up / 2.0
            alphas = [80.0, 200.0, 500.0]
            for alpha in alphas:
                if time_left() < 15:
                    break
                n_iters = min(2000, int(time_left() * 0.4))
                h_opt, c5_opt = nesterov_opt(h_up, n_up, ts_up, alpha, n_iters, 0.03, 0.25)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_up
                h_up = h_opt
            if time_left() > 15:
                h_cd, c5_cd = fast_coord_descent(h_up, n_up, ts_up, 80, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n_up

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
```
