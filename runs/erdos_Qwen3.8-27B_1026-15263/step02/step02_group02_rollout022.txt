            if step <= 0:

        else:

            step = min(-diff / np.sum(mask), room.min() * 0.999)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, top_m=20):
    """Compute subgradient of max overlap at h"""
    c = compute_all_overlaps(h)
    c_max = c.max()
    # Find active shifts (within epsilon of max)
    active = np.where(c >= c_max - 1e-8)[0]
    if len(active) > top_m:
        active = np.argsort(c)[-top_m:]
    
    # Compute subgradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active:
        # k in [0, 2n-2], shift = k - (n-1)
        shift = k - (n - 1)
        # ∂g_k/∂h[j] = dx * [(1-h[j-shift]) * 1_{valid} - h[j+shift] * 1_{valid}]
        for j in range(n):
            coeff = 0.0
            # First term: j - shift in [0, n-1]
            idx1 = j - shift
            if 0 <= idx1 < n:
                coeff += (1.0 - h[idx1])
            # Second term: j + shift in [0, n-1]
            idx2 = j + shift
            if 0 <= idx2 < n:
                coeff -= h[idx2]
            grad[j] += coeff
    grad *= dx
    # Normalize to be a proper subgradient (convex combination)
    n_active = len(active)
    grad /= n_active
    return grad, c_max, active

def proj_subgradient(h0, n, max_iter=5000, top_m=20, t_end=None):
    """Project subgradient method with adaptive step size"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    
    D = np.sqrt(n)  # diameter bound
    alpha = D  # step size parameter
    
    for t in range(1, max_iter + 1):
        if t_end is not None and time.time() > t_end:
            break
        
        step_size = alpha / np.sqrt(t)
        grad, c_max, active = subgradient_step(h, n, top_m)
        
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
    
    return best_h, best_c5

        return x[-1]

        return t - c

            if 0 <= idx2 < n:
                val -= h[idx2]
            jac[i, j] = -val * dx
        jac[i, -1] = 1.0
    return jac


def active_set_slsqp(h_init, n, max_rounds=20, m_top=25, maxiter_per_round=3000, time_limit=None):
    h = h_init.copy()
    best_c5 = compute_c5_from_h(h)

    for round_num in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        c = compute_all_overlaps(h)
        c_max = c.max()
        c5_cur = compute_c5_from_h(h)
        top_idx = np.argsort(c)[-m_top:]

        x0 = np.concatenate([h, [c5_cur + 0.08]])
        bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_cur * 0.3), 2.0)]

        def obj(x):
            return x[-1]

        def ineq_nojac(x):
            h = x[:-1]
            t = x[-1]
            c = compute_all_overlaps(h)
            return t - c[top_idx]

        def ineq_jac_fn(x):
            return ineq_jac_active(x, top_idx, n)

        def eq_fn(x):
            return np.sum(x[:-1]) - n / 2.0

        def eq_jac_fn(x):
            j = np.zeros(len(x))
            j[:-1] = 1.0
            return j

        constraints = [
            {'type': 'ineq', 'fun': ineq_nojac, 'jac': ineq_jac_fn},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}
        ]

        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': maxiter_per_round, 'ftol': 1e-13})

        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5_from_h(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
            else:
                m_top = min(len(c), m_top + 15)
                if m_top >= len(c) - 10:
                    break
    return h, best_c5


def slsqp_full(h_init, n, maxiter=5000, t_margin=0.05, time_limit=None):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]

    def obj(x):
        return x[-1]

    def ineq_fn(x):
        h = x[:-1]
        t = x[-1]
        c = compute_all_overlaps(h)
        return t - c

    def eq_fn(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac_fn(x):
        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j

    constraints = [
        {'type': 'ineq', 'fun': ineq_fn},
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}
    ]

    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                     constraints=constraints,
                     options={'maxiter': maxiter, 'ftol': 1e-14})

    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def logsumexp_optimize(h_init, n, T_list, penalty, maxiter=2000, time_limit=None):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)

    for T in T_list:
        if time_limit is not None and time.time() > time_limit:
            break

        def obj(x):
            c5_val = smooth_c5(x, T)
            sum_v = np.sum(x) - n / 2.0
            return c5_val + penalty * sum_v * sum_v

        bounds = [(0.0, 1.0)] * n
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-12})

        if result.x is not None:
            h_new = project_feasible(result.x, n)
            c5_new = compute_c5_from_h(h_new)
            if c5_new < best_c5:
                h = h_new.copy()
                best_h = h_new.copy()
                best_c5 = c5_new

    return best_h, best_c5


def subgradient_polish(h0, n, max_iter=3000, top_m=15, time_limit=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    dx = 2.0 / n

    alpha = np.sqrt(n) * 0.5
    for t in range(1, max_iter + 1):
        if time_limit is not None and time.time() > time_limit:
            break
        step_size = alpha / np.sqrt(t)
        c = compute_all_overlaps(h)
        c_max = c.max()
        active = np.argsort(c)[-top_m:]

        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            for j in range(n):
                val = 0.0
                idx1 = j - s
                if 0 <= idx1 < n:
                    val += (1.0 - h[idx1])
                idx2 = j + s
                if 0 <= idx2 < n:
                    val -= h[idx2]
                grad[j] += val * dx
        grad /= len(active)

        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new

    return best_h, best_c5


def fourier_init(n, rng, n_modes=5):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for m in range(1, n_modes + 1):
        amp = 0.15 / np.sqrt(m)
        phase = rng.random() * 2 * np.pi
        h += amp * np.cos(2 * np.pi * m * x / n + phase)
    return project_feasible(h, n)


def targeted_perturb(h, n, rng, scale):
    c = compute_all_overlaps(h)
    top_k = np.argsort(c)[-3:]
    h_pert = h.copy()
    for k in top_k:
        s = k - (n - 1)
        for j in range(n):
            idx1 = j - s
            idx2 = j + s
            if 0 <= idx1 < n and 0 <= idx2 < n:
                prod = h[j] * (1.0 - h[idx1])
                if prod > 0.1:
                    if rng.random() < 0.5:
                        h_pert[j] -= scale * 0.5
                    else:
                        h_pert[idx1] += scale * 0.5
    return project_feasible(h_pert, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    # Phase 1: Active-set SLSQP from parent (25% budget)
    tl1 = t0 + budget_s * 0.25
    h_new, c5_new = active_set_slsqp(best_h, n_best, max_rounds=25, m_top=25,
                                      maxiter_per_round=3000, time_limit=tl1)
    if c5_new < best_c5:
        best_h = h_new
        best_c5 = c5_new

    # Phase 2: Log-sum-exp optimization (15% budget)
    tl2 = t0 + budget_s * 0.40
    if time.time() < tl2:
        h_new, c5_new = logsumexp_optimize(best_h, n_best,
                                           T_list=[30, 80, 150, 300],
                                           penalty=200.0, maxiter=2000, time_limit=tl2)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Targeted perturbations + local optimize (25% budget)
    tl3 = t0 + budget_s * 0.65
    n_perturb = 40
    for attempt in range(n_perturb):
        if time.time() > tl3:
            break
        frac = attempt / max(1, n_perturb)
        scale = 0.12 * (1.0 - frac) ** 0.7 + 0.008

        r = rng.random()
        if r < 0.30:
            h_pert = targeted_perturb(best_h, n_best, rng, scale)
        elif r < 0.50:
            freq = rng.integers(2, max(3, n_best // 5))
            phase = rng.random() * 2 * np.pi
            h_pert = best_h + scale * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            h_pert = project_feasible(h_pert, n_best)
        elif r < 0.65:
            h_pert = best_h.copy()
            n_blocks = rng.integers(2, max(3, n_best // 40))
            for _ in range(n_blocks):
                bsize = rng.integers(5, max(6, n_best // 10))
                start = rng.integers(0, max(1, n_best - bsize))
                h_pert[start:start + bsize] += scale * (rng.random() - 0.5) * 2
            h_pert = project_feasible(h_pert, n_best)
        elif r < 0.80:
            h_pert = best_h + scale * 0.4 * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
        else:
            h_pert = fourier_init(n_best, rng, n_modes=rng.integers(3, 8))

        if attempt % 4 == 0:
            h_new, c5_new = active_set_slsqp(h_pert, n_best, max_rounds=8, m_top=20,
                                              maxiter_per_round=2000, time_limit=tl3)
        else:
            h_new, c5_new = logsumexp_optimize(h_pert, n_best, T_list=[50, 100],
                                               penalty=300.0, maxiter=1500, time_limit=tl3)

        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Resolution experiments (15% budget)
    tl4 = t0 + budget_s * 0.80
    for n_try in [200, 240, 270, 320, 360, 400]:
        if time.time() > tl4:
            break
        if n_try == n_best:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)

        h_new, c5_new = active_set_slsqp(h_interp, n_try, max_rounds=12, m_top=25,
                                          maxiter_per_round=2500, time_limit=tl4)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            n_best = n_try

    # Phase 5: Subgradient polish + final SLSQP (15% budget)
    tl5 = t0 + budget_s * 0.95
    if time.time() < tl5:
        h_new, c5_new = subgradient_polish(best_h, n_best, max_iter=2000, top_m=15,
                                            time_limit=min(tl5 - 30, t0 + budget_s * 0.90))
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() < tl5:
        h_new, c5_new = active_set_slsqp(best_h, n_best, max_rounds=15, m_top=40,
                                          maxiter_per_round=4000, time_limit=tl5)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final: try full SLSQP if n is small enough
    if time.time() < t0 + budget_s * 0.98 and n_best <= 300:
        h_new, c5_new = slsqp_full(best_h, n_best, maxiter=5000, t_margin=0.01,
                                    time_limit=t0 + budget_s * 0.99)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
