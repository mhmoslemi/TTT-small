def evaluate_c5(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def compute_gradients(h, top_m=5):
    """Compute averaged gradient over top-m active lags."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    C = corr * dx
    
    # Find top m lags
    top_idx = np.argsort(C)[-top_m:]
    
    # Compute gradient for each active lag
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)  # actual lag
        for j in range(n):
            if k >= 0:
                # dC/dh[j] = (1-h[j-k]) * [k<=j<=n-1] - h[j+k] * [0<=j<=n-1-k]
                if k <= j <= n - 1:
                    grad[j] += (1 - h[j - k])
                if 0 <= j <= n - 1 - k:
                    grad[j] -= h[j + k]
            else:
                kp = -k
                if kp <= j <= n - 1:
                    grad[j] += (1 - h[j + kp])
                if 2 * kp <= j <= n - 1 - kp:
                    grad[j] -= h[j - kp]
        grad[j] *= dx  # scale by dx
    
    return grad

- From (1-h[i]) where i = j: valid when 0 <= j <= n-1-k.
  Contribution: dx * (-h[j+k])

So dC(k)/dh[j] = dx * [(1-h[j-k]) * 1_{k<=j<=n-1} - h[j+k] * 1_{0<=j<=n-1-k}]

For lag k < 0, let k' = -k > 0: C(-k') = dx * sum_{i=k'}^{n-1} h[i] * (1-h[i+k'])

dC(-k')/dh[j]:
- From h[i] where i = j: valid when k' <= j <= n-1.
  Contribution: dx * (1-h[j+k'])
- From (1-h[i+k']) where i+k' = j, i = j-k': valid when k' <= j-k' <= n-1, i.e., 2k' <= j <= n-1-k'.
  Contribution: dx * (-h[j-k'])

So dC(-k')/dh[j] = dx * [(1-h[j+k']) * 1_{k'<=j<=n-1} - h[j-k'] * 1_{2k'<=j<=n-1-k'}]

Great, now I can vectorize this.

For the active set approach:
1. Compute all C(k)
2. Pick top m
3. Sum their gradients (or average)
4. Take a step: h_new = h - alpha * grad
5. Project to feasibility (clip to [0,1] and adjust sum)
6. Repeat with decreasing alpha

This is essentially a projected subgradient method for the max of linear (well, bilinear) functions.

Let me also add a **momentum** term (like in deep learning) to help with convergence:
v = beta * v - alpha * grad
h_new = h + v

And a **learning rate schedule**: alpha_t = alpha_0 / (1 + t * decay)

For the final polish, I'll use coordinate descent.

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_active_gradient(h, m=5):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    C = corr * dx
    top_idx = np.argsort(C)[-m:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:] += (1.0 - h[:nk])
                grad[:nk] -= h[k:]
        else:
            kp = -k
            width = n - 2 * kp
            if width > 0:
                grad[kp:n-kp] += (1.0 - h[2*kp:n])
                grad[2*kp:n-kp] -= h[kp:n-kp]
    grad *= dx
    return grad

def projected_gradient_opt(h0, n, max_iter=200, m=5, alpha0=0.05, momentum_beta=0.7):
    h = project_feasible(h0.copy(), n)
    v = np.zeros(n)
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for it in range(max_iter):
        alpha = alpha0 / (1.0 + it * 0.05)
        grad = compute_active_gradient(h, m=m)
        v = momentum_beta * v - alpha * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best = h_new.copy()
            best_c5 = c5
        else:
            v *= 0.5
            h_new = h - 0.5 * alpha * grad
            h_new = project_feasible(h_new, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best = h_new.copy()
                best_c5 = c5
                h = h_new
            else:
                alpha0 *= 0.9
                if alpha0 < 1e-6:
                    break
                h = best.copy()
        h = h_new if best_c5 == c5 else best.copy()
    return best, best_c5

def pairwise_descend(h, n, sweeps=3, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]
    best = project_feasible(h.copy(), n)
    best_c5 = evaluate_c5(best)
    for step in step_sizes:
        for sweep in range(sweeps):
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if abs(best[i] - 0.0) < 1e-9 or abs(1.0 - best[i]) < 1e-9:
                        continue
                    if abs(best[j] - 0.0) < 1e-9 or abs(1.0 - best[j]) < 1e-9:
                        continue
                    for d in [step, -step]:

                            continue

                        h_try[j] -= d
                        c5 = evaluate_c5(h_try)
                        if c5 < best_c5 - 1e-15:
                            best = h_try
                            best_c5 = c5
                            improved = True
                            break
            if not improved:
                break
    return best, best_c5

    target = n / 2.0
    deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]

                else:
