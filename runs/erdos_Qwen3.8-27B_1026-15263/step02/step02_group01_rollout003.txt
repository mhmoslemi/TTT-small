import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(corr)) / alpha
    g = np.zeros(n)
    for j in range(n):
        kmax1 = min(j, n - 1)
        if kmax1 >= 0:
            ks = np.arange(0, kmax1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kmax2 = n - j - 1
        if kmax2 >= 0:
            ks2 = np.arange(0, kmax2 + 1)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_val_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_val_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def projected_gradient_nesterov(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    y = h.copy()
    lr = lr_init
    stale = 0
    for it in range(n_iters):
        _, g = smooth_max_val_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-12:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.4)
            stale = 0
        else:
            stale += 1
            if stale > 50:
                lr *= 0.7
                stale = 0
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, fine=True):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if fine:
        deltas = [-0.001, -0.0005, -0.0001, 0.0001, 0.0005, 0.001]
    else:
        deltas = [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def stochastic_perturb(h, sigma, rng):
    h_new = h + sigma * rng.randn(len(h))
    return project_to_sum(np.clip(h_new, 0.0, 1.0), np.sum(h))

def try_binary_periodic(n, period, n_ones_per_period, n_periods_needed):
    h = np.zeros(n)
    if period <= 0:
        return h
    for p in range(n_periods_needed):
        base = p * period
        if base + period > n:
            break
        ones = n_ones_per_period
        zeros = period - n_ones_per_period
        block = np.array([1.0]*ones + [0.0]*zeros)
        if base + period <= n:
            h[base:base+period] = block
    return h
