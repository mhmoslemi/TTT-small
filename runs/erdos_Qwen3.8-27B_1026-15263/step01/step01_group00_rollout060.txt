import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_simplex_bbox(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        if np.sum(y) > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(y)
    mask = (y > 1e-12) & (y < 1.0 - 1e-12)
    if np.any(mask):
        y[mask] += diff / np.sum(mask)
        y = np.clip(y, 0.0, 1.0)
    return y

def gradient_smooth_max_c5(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

                grad[:n-s] += wk * (1.0 - h[s:])

            if n - t > 0:
                grad[t:] += wk * (1.0 - h[:n-t])

    grad = gradient_smooth_max_c5(h, alpha)
    h_new = h - lr * grad
    h_new = project_simplex_bbox(h_new, target_sum)
    return h_new

    n = len(h)
    target_sum = n / 2.0
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    lr = lr_init
    t0 = time.time()
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:

        h_new = projected_gradient_step(h, alpha, lr, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:

            c5_cur = c5_new
            if c5_new < c5_best:
