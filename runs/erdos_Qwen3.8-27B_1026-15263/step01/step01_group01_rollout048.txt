    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12

                break
            idx = np.where(mask)[0]

            per_point = diff / len(idx)
            h[idx] = np.minimum(h[idx] + per_point, 1.0)

                break

    dx = 2.0 / n
    h0 = project_feasible(np.asarray(h0, dtype=float).copy(), n)
    def obj(x):

        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 500)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

        return np.sum(x) - n / 2.0

        try:

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                return project_feasible(h_new, n), float(t_var.value)
        except Exception:
            continue
    return h_old, None


def sca_loop(h0, n, max_iters=25):
    h = project_feasible(np.asarray(h0, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_t = None
    for it in range(max_iters):
        h_new, t_val = solve_sca_lp(h, n)
        if t_val is None:
            break
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        if prev_t is not None and abs(t_val - prev_t) < 1e-7:
            break
        prev_t = t_val
        h = h_new
    return best_h, best_c5


def slsqp_refine(h0, n, max_iter=200, beta=100.0):
    dx = 2.0 / n
    h0 = project_feasible(np.asarray(h0, dtype=float).copy(), n)

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500.0, 500.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    def con_eq(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def make_fourier_init(n, rng, n_modes=3, amp_range=(0.1, 0.35)):
    x = np.arange(n, dtype=float)
    h = np.ones(n) * 0.5
    for _ in range(n_modes):
        f = rng.randint(1, max(2, n // 4))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2.0 * np.pi * f * x / n + phase)
    return project_feasible(h, n)


def make_pseudo_binary_init(n, rng):
    block_lens = rng.randint(2, 8, size=40)
    h = np.zeros(200)
    pos = 0
    val = 1.0
    for bl in block_lens:
        h[pos:pos + bl] = val
        pos += bl
        val = 1.0 - val
        if pos >= n:
            break
    h = h[:n]
    return project_feasible(h, n)


def simulated_anneal(h, n, rng, n_steps=200, t_start=0.15, t_end=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current = h.copy()
    current_c5 = best_c5
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(1, n_steps - 1))
        n_flips = rng.randint(1, 5)
        candidate = current.copy()
        for _ in range(n_flips):
            idx = rng.randint(n)
            candidate[idx] = 1.0 - candidate[idx]
        candidate = project_feasible(candidate, n)
        cand_c5 = compute_c5(candidate)
        delta = cand_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_h = current.copy()
                best_c5 = cand_c5
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init
    else:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    n = best_n

    # Phase 1: SCA from initial
    if time.time() - t0 < budget_s - 150:
        h_sca, c5_sca = sca_loop(best_h, n, max_iters=25)
        if c5_sca < best_c5:
            best_h, best_c5 = h_sca, c5_sca

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s - 120:
        h_opt, c5_opt = slsqp_refine(best_h, n, max_iter=250, beta=80.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: SCA again after SLSQP
    if time.time() - t0 < budget_s - 100:
        h_sca, c5_sca = sca_loop(best_h, n, max_iters=20)
        if c5_sca < best_c5:
            best_h, best_c5 = h_sca, c5_sca

    # Phase 4: Multi-restart with diverse initializations
    no_improve = 0
    attempt = 0
    while time.time() - t0 < budget_s - 200 and no_improve < 25:
        attempt += 1
        strat = attempt % 5

        if strat == 0:
            h_start = project_feasible(best_h + rng.normal(0, 0.06, n), n)
        elif strat == 1:
            h_start = make_fourier_init(n, rng, n_modes=rng.randint(2, 5))
        elif strat == 2:
            h_start = make_pseudo_binary_init(n, rng)
            h_start = 0.6 * best_h + 0.4 * h_start
            h_start = project_feasible(h_start, n)
        elif strat == 3:
            h_best_sa, c5_sa = simulated_anneal(best_h, n, rng, n_steps=100)
            h_start = h_best_sa
            if c5_sa < best_c5:
                best_h, best_c5 = h_start, c5_sa
        else:
            h_start = project_feasible(best_h + rng.normal(0, 0.2, n), n)

        h_sca, c5_sca = sca_loop(h_start, n, max_iters=15)
        if c5_sca < best_c5:
            best_h, best_c5 = h_sca, c5_sca
            no_improve = 0
        else:
            no_improve += 1

    # Phase 5: SLSQP on best
    if time.time() - t0 < budget_s - 80:
        h_opt, c5_opt = slsqp_refine(best_h, n, max_iter=300, beta=150.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Try higher resolution
    if time.time() - t0 < budget_s - 100:
        for n_try in [284, 380]:
            if time.time() - t0 > budget_s - 50:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new_arr = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new_arr, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_sca, c5_sca = sca_loop(h_interp, n_try, max_iters=15)
            if c5_sca < best_c5:
                h_opt2, c5_opt2 = slsqp_refine(h_sca, n_try, max_iter=200)
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_h, best_c5, best_n = h_opt2, c5_opt2, n_try
                else:
                    best_h, best_c5, best_n = h_sca, c5_sca, n_try

    # Phase 7: Final SCA + SLSQP
    if time.time() - t0 < budget_s - 30:
        h_sca, c5_sca = sca_loop(best_h, best_n, max_iters=20)
        if c5_sca < best_c5:
            best_h, best_c5 = h_sca, c5_sca

    if time.time() - t0 < budget_s - 15:
        h_opt, c5_opt = slsqp_refine(best_h, best_n, max_iter=150, beta=200.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
