    n = len(h)
    c = fftconvolve(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

    n = len(h)
    c = fftconvolve(h, 1.0 - h, mode='full')
    return c * (2.0 / n)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = float(np.sum(mask))

            h[mask] += min(diff / cnt, room)
        else:
            mask = h > 0.0
            if not np.any(mask):

            room = h[mask].min()
            h[mask] -= min(-diff / cnt, room)
