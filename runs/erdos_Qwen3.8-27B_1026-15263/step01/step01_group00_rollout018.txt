            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except:
            pass
    
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # List of n values to try
    n_list = [180, 200, 240, 270, 300, 360, 450]
    
    # Filter to valid n (n/2 must be integer)
    n_list = [n for n in n_list if n % 2 == 0]
    
    # Ensure we include the current best n
    if best_n not in n_list:
        n_list.insert(0, best_n)
    
    for n in n_list:
        if time_left() < 30:
            break
        ts = n / 2.0
        
        # Generate starting points
        starts = []
        
        # Interpolate from best
        if best_h is not None and best_n != n:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(project_to_sum(h_interp, ts))
        
        # Uniform
        starts.append(np.full(n, 0.5))
        
        # From best if same n
        if best_n == n:
            starts.append(best_h.copy())
        
        # Sinusoidal patterns
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1, 2, 3]:
            for amp in [0.2, 0.3, 0.4]:
                hs = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
                starts.append(project_to_sum(hs, ts))
        
        # Random
        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_to_sum(hr, ts))
        
        for h_start in starts:
            if time_left() < 25:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n
            
            # Phase 1: SLSQP with increasing alpha
            for alpha in [5.0, 20.0, 50.0, 150.0, 400.0]:
                if time_left() < 20:
                    break
                h_opt = optimize_slsqp(h_start, n, ts, alpha, maxiter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_start = h_opt
            
            # Phase 2: Multiplicative weights
            if time_left() > 60:
                n_outer = min(15, int(time_left() / 5))
                h_mw, c5_mw = multiplicative_weights_opt(
                    h_start, n, ts, n_outer, beta=3.0, max_inner=30, lr0=0.02, rng=rng)
                if c5_mw < best_c5:
                    best_c5 = c5_mw
                    best_h = h_mw.copy()
                    best_n = n
                h_start = h_mw
            
            # Phase 3: Fine coordinate descent
            if time_left() > 40:
                max_cd_rounds = min(30, int(time_left() / 5))
                h_cd, c5_cd = coordinate_descent_fine(h_start, n, ts, max_cd_rounds)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n
    
    # Final polish on the best solution
    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        
        # High-alpha SLSQP
        for alpha in [200.0, 500.0, 1000.0, 2000.0]:
            if time_left() < 15:
                break
            h_pol = optimize_slsqp(best_h, n_f, ts_f, alpha, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        
        # Very fine coordinate descent
        if time_left() > 20:
            fine_steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
            max_cd = min(50, int(time_left() / 3))
            h_cd, c5_cd = coordinate_descent_fine(best_h, n_f, ts_f, max_cd, fine_steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
        
        # One more SLSQP pass
        if time_left() > 15:
            h_pol = optimize_slsqp(best_h, n_f, ts_f, 3000.0, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Final validation
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
