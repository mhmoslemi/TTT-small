import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    # grad[j] = sum_t w[t] * ( [0<=j-lag_t<n] * (1-h[j-lag_t]) - [0<=j+lag_t<n] * h[j+lag_t] )
    # where lag_t = t - (n-1)
    
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)

        wt = w[t]

        # Term 1: [0<=j-k<n] * (1-h[j-k])

        else:
            js = 0
            je = min(n, n+k)  # j from 0 to n+k-1

        else:
            js = -k
            je = n  # j from -k to n-1
            if js < je:
                grad[js:je] -= wt * h[0:je-js]
    
    return grad

def projected_gradient_descent(h0, target_sum, alpha, max_iters, t_start, time_limit):
    h = h0.copy()
    f_cur = smooth_max_val(np.correlate(h, 1.0-h, mode='full'), alpha)
    step = 0.1  # initial step size
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        g = compute_grad(h, alpha)
        
        # Check gradient norm
        if np.max(np.abs(g)) < 1e-10:
            break
        
        # Project gradient: subtract mean (for sum constraint)
        g_proj = g - np.mean(g)
        
        # Backtracking line search
        accepted = False
        for _ in range(30):
            h_trial = h - step * g_proj
            h_trial = project_to_sum(h_trial, target_sum)
            f_trial = smooth_max_val(np.correlate(h_trial, 1.0-h_trial, mode='full'), alpha)
            if f_trial < f_cur - 1e-12:
                h = h_trial
                f_cur = f_trial
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            break
        
        # Reset step size after successful step (or increase slightly)
        step = min(step * 1.1, 0.5)
    
    return h

def coordinate_descent(h0, n, target_sum, rng, max_rounds, t_start, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() - t_start > time_limit:
                break
            for delta in [-0.03, -0.015, -0.007, 0.007, 0.015, 0.03]:
                h_new = h.copy()
                old_val = h_new[i]
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)

                c5_new = compute_c5(h_new)

                    c5_cur = c5_new
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
        if not improved:
            break
    return best_h, best_c5

        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n

    # Phase 4: SLSQP with proper gradient
    deadline4 = t_start + budget_s * 0.87
    n = best_n
    ts = n / 2.0

    def obj_fn(h_flat):
        hc = np.clip(h_flat, 0, 1)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        return smooth_max_val(corr, 200)

    def grad_fn(h_flat):
        hc = np.clip(h_flat, 0, 1)
        return compute_grad(hc, 200)

    try:
        res = minimize(obj_fn, best_h.copy(), jac=grad_fn, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': 500, 'ftol': 1e-15})
        h_sls = project_to_sum(res.x, ts)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()
            best_n = n
    except:
        pass

    # Phase 5: Second SLSQP with higher alpha
    if time.time() < deadline4:
        try:
            def obj_fn2(h_flat):
                hc = np.clip(h_flat, 0, 1)
                corr = np.correlate(hc, 1.0 - hc, mode='full')
                return smooth_max_val(corr, 500)

            def grad_fn2(h_flat):
                hc = np.clip(h_flat, 0, 1)
                return compute_grad(hc, 500)

            res2 = minimize(obj_fn2, best_h.copy(), jac=grad_fn2, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_sls2 = project_to_sum(res2.x, ts)
            c5_sls2 = compute_c5(h_sls2)
            if c5_sls2 < best_c5:
                best_c5 = c5_sls2
                best_h = h_sls2.copy()
                best_n = n
        except:
            pass

    # Phase 6: Coordinate descent refinement
    n = best_n
    ts = n / 2.0
    deadline5 = t_start + budget_s * 0.97
    h_cd, c5_cd = coord_desc(best_h, n, ts, rng, 300, deadline5)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        best_n = n

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
