    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.maximum(h, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def obj_full(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def obj_active(x, n, active_lags, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    subset = corr[active_lags]
    return smooth_max(subset, beta) * (2.0 / n)


def slsqp_optimize(h0, n, beta, max_iter, active_lags=None):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    if active_lags is not None:
        fun = lambda x: obj_active(x, n, active_lags, beta)
    else:
        fun = lambda x: obj_full(x, n, beta)
    try:
        res = minimize(fun, h0, method='SLSQP', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def trust_constr_optimize(h0, n, beta, max_iter):
    h0 = project_feasible(h0, n)
    from scipy.optimize import LinearConstraint
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    cons = [
        LinearConstraint(np.eye(n), 0.0, 1.0),
        LinearConstraint(A_eq, n / 2.0, n / 2.0)
    ]
    fun = lambda x: obj_full(x, n, beta)
    try:
        res = minimize(fun, h0, method='trust-constr', constraints=cons,
                       options={'maxiter': max_iter, 'gtol': 1e-10, 'xtol': 1e-14})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def find_active_lags(h, n, tol=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    max_val = np.max(overlaps)
    active = np.where(overlaps >= max_val - tol * max_val)[0]
    if len(active) == 0:
        active = np.array([np.argmax(overlaps)])
    return active


def projected_subgradient(h, n, max_iters, alpha_init):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    stuck = 0

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        k_max = int(np.argmax(overlaps))
        k_rel = k_max - (n - 1)

        grad = np.zeros(n)
        ones_minus_h = 1.0 - h
        for i in range(n):
            val = 0.0
            j1 = i + k_rel
            if 0 <= j1 < n:
                val += ones_minus_h[i]
            j2 = i - k_rel
            if 0 <= j2 < n:
                val -= ones_minus_h[i]
            grad[i] = val
        grad *= dx

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stuck = 0
            if alpha < 0.5:
                alpha *= 1.05
        else:
            alpha *= 0.7
            stuck += 1
            if stuck > 50:
                break

    return best_h, best_c5


def interpolate_to_n(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    def still_running(margin=30):
        return time.time() - t0 < budget_s - margin

    # Phase 1: Progressive smooth max at current resolution
    for beta in [40, 80, 150, 300]:
        if not still_running(300):
            break
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, beta, 400)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Active-set focused optimization
    if still_running(300):
        active = find_active_lags(best_h, best_n, tol=0.02)
        for _ in range(3):
            if not still_running(300):
                break
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, 200, 400, active_lags=active)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
                active = find_active_lags(best_h, best_n, tol=0.02)

    # Phase 3: Projected subgradient method for escape
    if still_running(350):
        h_pgd, c5_pgd = projected_subgradient(best_h, best_n, 3000, alpha_init=0.3)
        if c5_pgd < best_c5:
            best_h, best_c5 = h_pgd, c5_pgd
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, 200, 300)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Multi-restart with adaptive perturbation + SLSQP
    no_improve = 0
    max_no_improve = 80
    while still_running(250) and no_improve < max_no_improve:
        if rng.random() < 0.08:
            sigma = 0.25 + 0.2 * rng.random()
        else:
            sigma = 0.02 + 0.12 * (0.92 ** min(no_improve, 30))

        h_pert = best_h + rng.normal(0, sigma, best_n)
        h_pert = project_feasible(h_pert, best_n)

        if rng.random() < 0.3:
            active = find_active_lags(h_pert, best_n, tol=0.03)
            beta = int(80 + 200 * rng.random())
            h_opt, c5_opt = slsqp_optimize(h_pert, best_n, beta, 250, active_lags=active)
        else:
            beta = int(100 + 300 * rng.random())
            h_opt, c5_opt = slsqp_optimize(h_pert, best_n, beta, 250)

        if h_opt is not None and c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

    # Phase 5: Try higher resolution
    for n_new in [238, 286]:
        if not still_running(200):
            break
        h_interp = interpolate_to_n(best_h, best_n, n_new)
        h_interp = project_feasible(h_interp, n_new)

        for beta in [60, 120, 250]:
            if not still_running(200):
                break
            h_opt, c5_opt = slsqp_optimize(h_interp, n_new, beta, 350)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_new
                h_interp = h_opt

    # Phase 6: Trust-constr attempt
    if still_running(150):
        h_opt, c5_opt = trust_constr_optimize(best_h, best_n, 200, 500)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Very high beta fine-tuning
    if still_running(80):
        for beta in [400, 700, 1200]:
            if not still_running(40):
                break
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, beta, 600)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 8: Tiny perturbations + final polish
    if still_running(30):
        for _ in range(5):
            if not still_running(20):
                break
            h_pert = best_h + rng.normal(0, 0.005, best_n)
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = slsqp_optimize(h_pert, best_n, 800, 400)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
