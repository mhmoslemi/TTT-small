    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max(vals, tau):
    v = vals / tau
    vmax = np.max(v)
    return float(tau * (vmax + np.log(np.sum(np.exp(v - vmax)))))

def obj_penalty(h, n, tau):
    h = np.asarray(h).ravel()
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    dx = 2.0 / n
    val = smooth_max(corr, tau) * dx
    s = np.sum(h) - n / 2.0
    val += 1e4 * s * s
    return val

def obj_direct(h, n, tau):
    h = np.asarray(h).ravel()
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    dx = 2.0 / n
    return smooth_max(corr, tau) * dx

def constraint_sum(h, n):
    return np.sum(h) - n / 2.0

def constraint_grad(h, n):
    return np.ones(len(h))

def jitter_search(h, n, rng, n_steps, t_limit):
    best = h.copy()
    best_val = evaluate_c5(best)
    for step in range(n_steps):
        if time.time() > t_limit:
            break
        j = rng.randint(n)
        delta = rng.uniform(-0.04, 0.04)
        cand = best.copy()
        cand[j] = np.clip(cand[j] + delta, 0.0, 1.0)
        diff = n / 2.0 - np.sum(cand)
        if abs(diff) > 1e-10:
            if diff > 0:
                pool = np.where(cand < 0.95)[0]
            else:
                pool = np.where(cand > 0.05)[0]
            pool = pool[pool != j]
            if len(pool) > 0:
                k = pool[rng.randint(len(pool))]
                cand[k] = np.clip(cand[k] - diff, 0.0, 1.0)
            diff2 = n / 2.0 - np.sum(cand)
            if abs(diff2) > 1e-10:
                idx = int(np.argmin(np.abs(cand - 0.5)))
                cand[idx] = np.clip(cand[idx] + diff2, 0.0, 1.0)
        new_val = evaluate_c5(cand)
        if new_val < best_val - 1e-13:
            best = cand
            best_val = new_val
    return best, best_val

def targeted_local_search(h, n, rng, n_steps, t_limit):
    """Targeted search: identify worst correlations and try to reduce them specifically."""
    best = h.copy()
    best_val = evaluate_c5(best)
    m = 2 * n - 1

    for step in range(n_steps):
        if time.time() > t_limit:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        # Find top-3 worst correlations
        top_idx = np.argsort(corr)[-3:]
        # Try to reduce them by adjusting coordinates
        improved = False
        n_try = 15
        for _ in range(n_try):
            j = rng.randint(n)
            # Determine good direction based on contribution to worst correlations
            # For corr[k] = sum_i h[i]*(1-h[i+k-(n-1)]), changing h[j] affects:
            # corr[k] where k = n-1-j+j+k' ... it's complex, so try both directions
            base = best[j]
            for direction in [1, -1]:
                delta = direction * rng.uniform(0.005, 0.04)
                cand = best.copy()
                cand[j] = np.clip(cand[j] + delta, 0.0, 1.0)
                diff = n / 2.0 - np.sum(cand)
                if abs(diff) > 1e-10:
                    if diff > 0:
                        pool = np.where(cand < 0.95)[0]
                    else:
                        pool = np.where(cand > 0.05)[0]
                    pool = pool[pool != j]
                    if len(pool) == 0:
                        continue
                    k = pool[rng.randint(len(pool))]
                    cand[k] = np.clip(cand[k] - diff, 0.0, 1.0)
                    diff2 = n / 2.0 - np.sum(cand)
                    if abs(diff2) > 1e-10:
                        idx = int(np.argmin(np.abs(cand - 0.5)))
                        cand[idx] = np.clip(cand[idx] + diff2, 0.0, 1.0)
                new_val = evaluate_c5(cand)
                if new_val < best_val - 1e-13:
                    best = cand
                    best_val = new_val
                    improved = True
                    break
            if improved:
                break
    return best, best_val

def lbb_optimize(h, n, tau, bounds, max_iter=2000):
    try:
        res = minimize(
            obj_penalty, h.copy(),
            method='L-BFGS-B',
            args=(n, tau),
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-18, 'maxfun': max_iter * 10}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_optimize(h, n, tau, bounds, max_iter=300):
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,), 'jac': constraint_grad}]
    try:
        res = minimize(
            obj_direct, h.copy(),
            method='SLSQP',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        n = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
    else:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B progressive tau annealing (main optimization)
    t_phase1 = t0 + 0.38 * budget_s
    taus_main = [0.25, 0.15, 0.1, 0.07, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.004, 0.003, 0.002, 0.001, 0.0005]
    for tau in taus_main:
        if time.time() > t_phase1:
            break
        h_new = lbb_optimize(best_h, n, tau, bounds, max_iter=1500)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: SLSQP with explicit constraint for precision
    t_phase2 = t0 + 0.58 * budget_s
    if time.time() < t_phase2:
        for tau in [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]:
            if time.time() > t_phase2:
                break
            h_new = slsqp_optimize(best_h, n, tau, bounds, max_iter=200)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 3: Multi-restart perturbation + re-optimize
    t_phase3 = t0 + 0.82 * budget_s
    if time.time() < t_phase3:
        scales = [0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.01, 0.005, 0.003, 0.02, 0.03, 0.005]
        for i, scale in enumerate(scales):
            if time.time() > t_phase3:
                break
            noise = rng.randn(n) * scale
            perturbed = project_feasible(best_h + noise, n)
            tau_opt = 0.002 if i > 5 else 0.005
            h_new = lbb_optimize(perturbed, n, tau_opt, bounds, max_iter=800)
            h_new = slsqp_optimize(h_new, n, 0.001, bounds, max_iter=100)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Targeted local search on worst correlations
    t_phase4 = t0 + 0.93 * budget_s
    if time.time() < t_phase4:
        est_steps = int((t_phase4 - time.time()) * 800)
        best_h, best_c5 = targeted_local_search(best_h, n, rng, est_steps, t_phase4)

    # Phase 5: Final jitter search
    t_phase5 = t0 + 0.985 * budget_s
    if time.time() < t_phase5:
        est_steps = int((t_phase5 - time.time()) * 2000)
        best_h, best_c5 = jitter_search(best_h, n, rng, est_steps, t_phase5)

    # Final SLSQP polish if time allows
    if time.time() < t0 + 0.99 * budget_s:
        h_new = slsqp_optimize(best_h, n, 0.0005, bounds, max_iter=100)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
