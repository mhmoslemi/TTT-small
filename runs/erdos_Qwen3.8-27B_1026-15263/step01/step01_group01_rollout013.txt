```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def lbfgsb_objective(x, n, beta, mu):
    h = x
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    dx = 2.0 / n
    sm = smooth_max_val(corr, beta) * dx
    s = np.sum(h) - n / 2.0
    return sm + mu * s * s

def lbfgsb_gradient(x, n, beta, mu):
    h = x
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    dx = 2.0 / n
    w = smooth_max_weights(corr, beta)
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        k = p - (n - 1)
        # Term 1: (1-h[m-k]) for m-k in [0,n-1] → m in [k, n-1+k] ∩ [0,n-1]
        m1_lo = max(0, k)
        m1_hi = min(n - 1, n - 1 + k)
        if m1_lo <= m1_hi:
            mr = np.arange(m1_lo, m1_hi + 1)
            grad[mr] += w[p] * (1.0 - h[mr - k])
        # Term 2: -h[m+k] for m+k in [0,n-1] → m in [-k, n-1-k] ∩ [0,n-1]
        m2_lo = max(0, -k)
        m2_hi = min(n - 1, n - 1 - k)
        if m2_lo <= m2_hi:
            mr = np.arange(m2_lo, m2_hi + 1)
            grad[mr] -= w[p] * h[mr + k]
    
    s = np.sum(h) - n / 2.0
    grad = dx * grad + 2.0 * mu * s
    return grad

def run_lbfgsb(h0, n, max_iter, beta, mu_start, mu_end, n_mu_steps):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    
    for step in range(n_mu_steps):
        frac = step / max(1, n_mu_steps - 1)
        mu = mu_start * (mu_end / mu_start) ** frac
        iters = max(50, max_iter // n_mu_steps)
        try:
            res = minimize(
                lbfgsb_objective, h0, jac=lbfgsb_gradient,
                args=(n, beta, mu),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0, 'm': 30}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h0 = h_new
        except Exception:
            break
    
    return best_h, best_c5

def local_search(h, n, window=15, max_rounds=5):
    h = project_feasible(h.copy(), n)
    current = compute_c5(h)
    
    for _ in range(max_rounds):
        improved = False
        # Try small transfers
        for i in range(n):
            if h[i] <= 1e-8:
                continue
            for j in range(max(0, i - window), min(n, i + window + 1)):
                if j == i or h[j] >= 1.0 - 1e-8:
                    continue
                eps = 0.01
                h2 = h.copy()
                h2[i] = max(0.0, h2[i] - eps)
                h2[j] = min(1.0, h2[j] + eps)
                h2 = project_feasible(h2, n)
                c52 = compute_c5(h2)
                if c52 < current - 1e-12:
                    h = h2
                    current = c52
                    improved = True
        # Try swaps
        for i in range(0, n, 2):
            for j in range(i + 1, min(i + window, n), 2):
                h[i], h[j] = h[j], h[i]
                h2 = project_feasible(h, n)
                c52 = compute_c5(h2)
                if c52 < current - 1e-12:
                    h = h2
                    current = c52
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            break
    
    return project_feasible(h, n), current

def make_binary_start(n, rng):
    """Create a random binary sequence with n/2 ones and n/2 zeros."""
    h = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return project_feasible(h, n)

def make_smooth_binary(n, rng, smooth=0.1):
    h = make_binary_start(n, rng)
    h = h + rng.normal(0, smooth, n)
    return project_feasible(h, n)

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    ev = np.exp(shifted)
    return ev / ev.sum()


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def lbfgsb_obj(x, n, beta, mu):
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    sm = smooth_max_val(corr, beta) * dx
    s = np.sum(h) - n / 2.0
    return sm + mu * s * s


def lbfgsb_grad(x, n, beta, mu):
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        k = p - (n - 1)
        m1_lo = max(0, k)
        m1_hi = min(n - 1, n - 1 + k)
        if m1_lo <= m1_hi:
            mr = np.arange(m1_lo, m1_hi + 1)
            grad[mr] += wp * (1.0 - h[mr - k])
        m2_lo = max(0, -k)
        m2_hi = min(n - 1, n - 1 - k)
        if m2_lo <= m2_hi:
            mr = np.arange(m2_lo, m2_hi + 1)
            grad[mr] -= wp * h[mr + k]
    s = np.sum(h) - n / 2.0
    grad = dx * grad + 2.0 * mu * s
    return grad


def run_lbfgsb(h0, n, max_iter, beta, mu_start, mu_end, n_mu_steps):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    for step in range(n_mu_steps):
        frac = step / max(1, n_mu_steps - 1)
        mu = mu_start * (mu_end / mu_start) ** frac
        iters = max(40, max_iter // n_mu_steps)
        try:
            res = minimize(
                lbfgsb_obj, h0, jac=lbfgsb_grad,
                args=(n, beta, mu),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-13, 'iprint': 0, 'm': 40}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h0 = h_new
        except Exception:
            break
    return best_h, best_c5


def local_search(h, n, window=12, max_rounds=4, eps=0.015):
    h = project_feasible(h.copy(), n)
    current = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            if h[i] < 1e-8:
                continue
            lo = max(0, i - window)
            hi = min(n, i + window + 1)
            for j in range(lo, hi):
                if j == i or h[j] > 1.0 - 1e-8:
                    continue
                h2 = h.copy()
                h2[i] = max(0.0, h2[i] - eps)
                h2[j] = min(1.0, h2[j] + eps)
                h2 = project_feasible(h2, n)
                c52 = compute_c5(h2)
                if c52 < current - 1e-13:
                    h = h2
                    current = c52
                    improved = True
        for i in range(0, n, 3):
            for j in range(i + 1, min(i + window, n), 3):
                h[i], h[j] = h[j], h[i]
                h2 = project_feasible(h, n)
                c52 = compute_c5(h2)
                if c52 < current - 1e-13:
                    h = h2
                    current = c52
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return project_feasible(h, n), current


def subgradient_polish(h, n, steps=200, alpha_start=0.3, alpha_end=0.02):
    h = h.copy()
    current = compute_c5(h)
    for s in range(steps):
        t = s / max(1, steps - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        m1_lo = max(0, k)
        m1_hi = min(n - 1, n - 1 + k)
        if m1_lo <= m1_hi:
            mr = np.arange(m1_lo, m1_hi + 1)
            grad[mr] += (1.0 - h[mr - k])
        m2_lo = max(0, -k)
        m2_hi = min(n - 1, n - 1 - k)
        if m2_lo <= m2_hi:
            mr = np.arange(m2_lo, m2_hi + 1)
            grad[mr] -= h[mr + k]
        grad *= dx
        sv = np.sum(h) - n / 2.0
        grad += 200.0 * sv
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < current:
            h = h_new
            current = c5_new
        else:
            h = h_new
    return project_feasible(h, n), current


def make_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)


def make_smooth_start(n, rng, base_smooth=0.12):
    h = make_binary_start(n, rng)
    h = h + rng.normal(0.0, base_smooth, n)
    return project_feasible(h, n)


def make_wave_start(n, freq, amp, rng, noise=0.06):
    i = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * i / freq)
    h = h + rng.normal(0.0, noise, n)
    return project_feasible(h, n)


def make_block_start(n, rng, block=8):
    h = np.zeros(n)
    n_blocks = n // block
    for b in range(n_blocks):
        if rng.random() < 0.5:
            h[b * block:(b + 1) * block] = 1.0
    h = h + rng.normal(0.0, 0.08, n)
    return project_feasible(h, n)


def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new)
```
