import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            room = (1.0 - h[idx])
            total_room = room.sum()
            if total_room <= 0:
                break
            h[idx] += diff * room / total_room
            h = np.clip(h, 0, 1)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            room = h[idx]
            total_room = room.sum()
            if total_room <= 0:
                break
            h[idx] -= (-diff) * room / total_room
            h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = get_corr(h)
    return smooth_max(corr, beta) * (2.0 / n)

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def optimize_slsqp(h0, n, max_iter, beta):
    h0 = project_h(h0, n).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_h(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None

def subgradient_refine(h, n, n_steps, lr_init, top_k):
    """Refine h using subgradient descent on the max correlation."""
    h_curr = h.copy()
    best_local = compute_c5(h_curr)
    lr = lr_init
    for _ in range(n_steps):
        corr = get_corr(h_curr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k_full in top_idx:
            lag = k_full - (n - 1)
            for j in range(n):
                g = 0.0
                jl = j - lag
                jlp = j + lag
                if 0 <= jl < n:
                    g += (1.0 - h_curr[jl])
                if 0 <= jlp < n:
                    g += (-h_curr[jlp])
                grad[j] += g
        grad /= len(top_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        h_new = h_curr - lr * grad / gnorm * (gnorm / n)
        # Scale step: lr * (gradient direction) where gradient is normalized
        # Actually let me just use: h_new = h_curr - lr * grad
        # But scale lr appropriately
        h_new = project_h(h_curr - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            h_curr = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr = max(lr * 0.9, 0.001)
            h_curr = h_new
    return h_curr, compute_c5(h_curr)

def fourier_perturb(h, n, rng, n_freqs=3, amplitude=0.1):
    """Perturb h in the frequency domain."""
    H = np.fft.rfft(h)
    freqs_to_perturb = rng.choice(range(1, len(H)), size=min(n_freqs, len(H)-1), replace=False)
    for f in freqs_to_perturb:
        H[f] += amplitude * (rng.randn() + 1j * rng.randn())
        if len(H) - 1 - f >= 1:  # conjugate对称
            H[len(H)-1-f] = np.conj(H[f])
    h_new = np.fft.irfft(H, n=n)
    return project_h(h_new, n)

    return np.interp(x_new, x_old, h_old)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            room = (1.0 - h[idx])
            total_room = room.sum()
            if total_room <= 0:
                break
            h[idx] += diff * room / total_room
            h = np.clip(h, 0, 1)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            room = h[idx]
            total_room = room.sum()
            if total_room <= 0:
                break
            h[idx] -= (-diff) * room / total_room
            h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

    h = np.clip(x, 0.0, 1.0)
    corr = get_corr(h)
    return smooth_max(corr, beta) * (2.0 / n)


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def optimize_slsqp(h0, n, max_iter, beta):
    h0 = project_h(h0, n).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_h(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None

        return project_h(h, n)

                best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 10:
        h_ref, c5_ref = subgradient_refine(best_h, best_n, 8000, 0.5, 1)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
