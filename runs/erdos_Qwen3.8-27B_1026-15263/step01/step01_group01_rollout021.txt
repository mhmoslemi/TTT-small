def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_all_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n, target_sum=None):
    if target_sum is None:
        target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling projection
    for _ in range(200):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return h

def gradient_of_fk(h, k, n):

    g = np.zeros(n)

        # g[j] -= h[j-k] for j >= k
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    return g

def projected_subgradient(h, n, n_iter=5000, verbose=False):

    dx = 2.0 / n
    h = project_feasible(h.copy(), n)

    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[-n_active:]
    g = np.zeros(n)
    for p in top_idx:
        lag = int(p) - (n - 1)
        g += gradient_at_lag(h, lag, n)
    g /= float(len(top_idx))
    return g, float(np.max(overlaps))


def projected_subgradient(h0, n, n_iter=20000, n_active=10, momentum_init=0.6):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h)
    vel = np.zeros(n)
    momentum = momentum_init
    alpha = 0.5 / np.sqrt(n)
    prev_h = h.copy()
    prev_g = None
    bb_count = 0

    for it in range(n_iter):
        g, cur_val = subgradient_max(h, n, n_active)

        if prev_g is not None:
            y = h - prev_h
            r = g - prev_g
            ydot = np.dot(y, y)
            rdot = np.dot(r, d = r) if False else np.dot(r, y)
            if ydot > 1e-16 and abs(rdot) > 1e-16:
                bb_alpha = ydot / abs(rdot)
                bb_alpha = np.clip(bb_alpha, 0.1 / np.sqrt(n), 10.0 / np.sqrt(n))
                alpha = 0.8 * alpha + 0.2 * bb_alpha
                bb_count += 1

        vel = momentum * vel + (1.0 - momentum) * g
        h = project_feasible(h - alpha * vel, n)

        prev_h = h.copy()
        prev_g = g.copy()

        if it % 500 == 0:
            momentum = 0.9
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

    return best_h, best_val


def slsqp_penalty_obj(x, n, beta, gamma):
    h = np.clip(x, 0.0, 1.0)
    ov = compute_overlaps(h)
    return smooth_max(ov, beta) + gamma * (np.sum(h) - n / 2.0) ** 2


def optimize_lbfgs(h0, n, max_iter=1000, beta=200.0, gamma=500.0):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(slsqp_penalty_obj, h0, method='L-BFGS-B',
                   bounds=bounds, args=(n, beta, gamma),
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0})
    h_res = project_feasible(res.x, n)
    return h_res, compute_c5(h_res)


def optimize_slsqp(h0, n, max_iter=300, beta=100.0):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return slsqp_penalty_obj(x, n, beta, 1e6)

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        ov = compute_overlaps(h)
        m = np.max(ov)
        shifted = np.clip(beta * (ov - m), -500.0, 500.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        g = np.zeros(n)
        for i, p in enumerate(range(len(ov))):
            if weights[i] > 1e-10:
                lag = p - (n - 1)
                g += weights[i] * gradient_at_lag(h, lag, n)
        g += 2.0 * 1e6 * (np.sum(h) - n / 2.0)
        return g

    try:
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, args=(),
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def binary_push(h, n, threshold=0.1):
    h2 = h.copy()
    h2[h2 > 0.5 + threshold] = 1.0
    h2[h2 < 0.5 - threshold] = 0.0
    h2[(h2 >= 0.5 - threshold) & (h2 <= 0.5 + threshold)] = 0.5
    return project_feasible(h2, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 190
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Polish with L-BFGS-B (10% time)
    if remaining() > 130:
        h_try, c5_try = optimize_lbfgs(best_h, best_n, max_iter=500, beta=150.0, gamma=1000.0)
        if h_try is not None and c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 2: Projected subgradient with momentum (30% time)
    if remaining() > 260:
        t_limit = 0.30 * budget_s
        n_iter = int(15000)
        h_try, c5_try = projected_subgradient(best_h, best_n, n_iter=n_iter, n_active=10, momentum_init=0.7)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 3: L-BFGS-B with higher beta (15% time)
    if remaining() > 130:
        h_try, c5_try = optimize_lbfgs(best_h, best_n, max_iter=800, beta=500.0, gamma=2000.0)
        if h_try is not None and c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 4: Binary push + re-optimize (15% time)
    if remaining() > 130:
        h_bin = binary_push(best_h, best_n, threshold=0.05)
        h_try, c5_try = optimize_lbfgs(h_bin, best_n, max_iter=600, beta=200.0, gamma=1000.0)
        if h_try is not None and c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 5: Multi-start with perturbations (20% time)
    if remaining() > 100:
        for attempt in range(8):
            if remaining() < 60:
                break
            sigma = 0.15 * (0.85 ** attempt)
            h_pert = best_h + rng.normal(0.0, sigma, best_n)
            h_pert = project_feasible(h_pert, best_n)
            h_try, c5_try = optimize_lbfgs(h_pert, best_n, max_iter=400, beta=200.0, gamma=1000.0)
            if h_try is not None and c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
                best_n = len(best_h)

    # Phase 6: Try higher resolution
    for n_hi in [380, 470]:
        if remaining() < 150:
            break
        h_interp = interpolate_h(best_h, best_n, n_hi)
        h_interp = project_feasible(h_interp, n_hi)
        h_try, c5_try = optimize_lbfgs(h_interp, n_hi, max_iter=400, beta=100.0, gamma=500.0)
        if h_try is not None:
            h_try2, c5_try2 = projected_subgradient(h_try, n_hi, n_iter=5000, n_active=8, momentum_init=0.7)
            if c5_try2 < c5_try:
                h_try, c5_try = h_try2, c5_try2
        if h_try is not None and c5_try < best_c5:
            best_h, best_c5, best_n = h_try, c5_try, n_hi

    # Phase 7: Final polish (remaining time)
    if remaining() > 30:
        h_try, c5_try = optimize_lbfgs(best_h, best_n, max_iter=1000, beta=1000.0, gamma=5000.0)
        if h_try is not None and c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    if remaining() > 20:
        h_try, c5_try = projected_subgradient(best_h, best_n, n_iter=5000, n_active=5, momentum_init=0.8)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
