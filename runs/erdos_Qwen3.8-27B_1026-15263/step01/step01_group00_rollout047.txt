import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        # Project: adjust all non-saturated values
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            adj = diff / np.sum(free)
            h[free] += adj
        else:
            # All saturated, just distribute
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(c, alpha):
    m = np.max(c)
    exp_c = np.exp(alpha * (c - m))
    w = exp_c / np.sum(exp_c)
    val = m + np.log(np.sum(exp_c) / len(exp_c)) / alpha

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)

def smooth_max_value(c, alpha):
    m = np.max(c)
    exp_c = np.exp(alpha * (c - m))
    sm = m + np.log(np.sum(exp_c) / len(c)) / alpha
    return sm

    m = np.max(c)
    exp_c = np.exp(alpha * (c - m))
    return exp_c / np.sum(exp_c)

        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: Main gradient optimization from parent
    h_work = best_h.copy()
    alphas_main = [3.0, 6.0, 12.0, 25.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    for alpha in alphas_main:
        if time.time() - t_start > budget_s - 80:
            break
        h_opt = lbfgsb_optimize(h_work, n, ts, alpha, 5000.0, 600, 3000)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        h_work = h_opt

    # Phase 2: Multiple diverse restarts
    if time.time() - t_start < budget_s - 120:
        n_restarts = 4
        for ri in range(n_restarts):
            if time.time() - t_start > budget_s - 50:
                break
            if ri == 0:
                h_s = best_h + rng.normal(0, 0.02, n)
                h_s = project_to_sum(np.clip(h_s, 0.0, 1.0), ts)
            elif ri == 1:
                h_s = fourier_start(n, rng, n_modes=8)
            elif ri == 2:
                h_s = np.full(n, 0.5)
                third = n // 3
                h_s[0:third] = 0.9
                h_s[2 * third:3 * third] = 0.1
                h_s = project_to_sum(h_s, ts)
            else:
                h_s = np.zeros(n)
                blk = n // 7
                for bi in [0, 2, 4, 5]:
                    h_s[bi * blk:(bi + 1) * blk] = 1.0
                h_s = project_to_sum(h_s, ts)

            for alpha in [20.0, 60.0, 150.0, 400.0]:
                if time.time() - t_start > budget_s - 30:
                    break
                h_opt = lbfgsb_optimize(h_s, n, ts, alpha, 5000.0, 400, 2000)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                h_s = h_opt

    # Phase 3: Fine local search
    if time.time() - t_start < budget_s - 40:
        remaining = budget_s - (time.time() - t_start) - 40
        max_rounds = int(remaining * 8)
        max_rounds = max(max_rounds, 100)
        h_fine, c5_fine = fine_local_search(best_h, n, ts, rng, max_rounds)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Phase 4: Stochastic refinement
    if time.time() - t_start < budget_s - 15:
        remaining = budget_s - (time.time() - t_start) - 15
        n_iters = int(remaining * 80)
        n_iters = min(n_iters, 15000)
        h_st, c5_st = stochastic_refine(best_h, n, ts, rng, n_iters)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()

    # Phase 5: Final gradient polish
    if time.time() - t_start < budget_s - 3:
        for alpha in [1000.0, 2000.0]:
            if time.time() - t_start > budget_s - 3:
                break
            h_pol = lbfgsb_optimize(best_h, n, ts, alpha, 10000.0, 300, 1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final projection
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), ts)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
