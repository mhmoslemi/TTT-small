```python
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

    dx = 2.0 / n
```

```python
def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smoothed_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    # Use log-sum-exp trick
    max_s = np.max(shifted)
    log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    return c_max + log_sum / beta

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(diff / float(np.sum(mask)), (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = min(-diff / float(np.sum(mask)), h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

    return np.sum(x) - _CON_N / 2.0

    global _CON_N
    _CON_N = n
    return [{'type': 'eq', 'fun': _eq_fun}]

    _CON_N = n

        return smoothed_c5(x, beta)
```

```python
def gradient_smoothed_c5(h, beta):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_scaled = beta * c
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)  # weights, length 2N-1
    
    grad = np.zeros(N)
    for j in range(N):
        # First term: sum_k w[k] * (1-h[j+k-(N-1)]) for valid indices
        # j+k-(N-1) in [0, N-1] means k in [(N-1)-j, (N-1)+j]
        k_min = max(0, (N-1) - j)
        k_max = min(2*N-2, (N-1) + j)
        if k_min <= k_max:
            idx = np.arange(k_min, k_max+1)
            h_idx = j + idx - (N-1)  # indices into h, should be 0 to N-1
            term1 = np.dot(w[k_min:k_max+1], 1.0 - h[h_idx])
        else:
            term1 = 0.0
        
        # Second term: sum_k w[k] * h[j-k+(N-1)] for valid indices
        # j-k+(N-1) in [0, N-1] means k in [j, j+(N-1)]
        k_min2 = max(0, j)
        k_max2 = min(2*N-2, j + N - 1)
        if k_min2 <= k_max2:
            idx2 = np.arange(k_min2, k_max2+1)
            h_idx2 = j - idx2 + (N-1)  # indices into h
            term2 = np.dot(w[k_min2:k_max2+1], h[h_idx2])
        else:
            term2 = 0.0
        
        grad[j] = dx * (term1 - term2)
    
    return grad
```
