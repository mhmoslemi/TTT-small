import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_vals(vals, tau):
    v = vals / tau
    v -= np.max(v)
    return tau * (np.max(v) + np.log(np.sum(np.exp(v))))

def power_mean_vals(vals, p):
    return float(np.sum(np.abs(vals) ** p) ** (1.0 / p))

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max_vals(corr, tau) * dx

def obj_pen(h, n, tau, lam):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    smax = smooth_max_vals(corr, tau)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return smax * dx + penalty

def obj_pow(h, n, p):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return power_mean_vals(corr, p) * dx

def sum_constraint(h, n):
    return np.sum(h) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            rv = result[idx] + diff
            if 0.0 <= rv <= 1.0:
                result[idx] = rv
                break
        else:
            result[order[0]] = np.clip(result[order[0]] + diff, 0.0, 1.0)
    return result

def coordinate_descent_pass(h, n, max_iter=50):
    target = n / 2.0
    best_c5 = evaluate_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            saved = h[i]
            other_sum = target - saved
            lo = max(0.0, other_sum - 1.0 * (n - 1))
            hi = min(1.0, other_sum)
            if lo > hi:
                continue
            best_val = saved
            best_obj = best_c5
            n_evals = 16
            for t in np.linspace(lo, hi, n_evals):
                h[i] = t
                c = evaluate_c5(h)
                if c < best_obj:
                    best_obj = c
                    best_val = t
            h[i] = best_val
            if best_val != saved:
                improved = True
                if best_obj < best_c5:
                    best_c5 = best_obj
        if not improved:
            break
    return h, best_c5

def binary_refine(h, n, rng, max_iters=200):
    target = n / 2.0
    h = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        h = project_feasible(h, n)
        c = evaluate_c5(h)
        if c < best_c5:
            best_c5 = c
        else:
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
    return h, best_c5
