import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            # Distribute evenly among non-saturated points
            cnt = int(np.sum(mask))
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def coord_descent_pass(h, n_passes=10, n_ternary=60):
    n = len(h)
    dx = 2.0 / n
    for _ in range(n_passes):
        for j in range(n):
            v = h[j]
            # Compute D = d(corr)/dh[j]
            # D[k] = (1-h[n-1-k+j]) * I(max(0,k-n+1) <= j <= min(n-1,k)) 
            #        - h[j+k-n+1] * I(max(0,k-n+1) <= j+k-n+1 <= min(n-1,k))
            
            k_vals = np.arange(2*n-1)
            # First term validity: max(0, k-n+1) <= j <= min(n-1, k)
            # i.e., j >= k-n+1 and j <= k and j >= 0 and j <= n-1
            # Since 0 <= j <= n-1 always, we need: k-n+1 <= j <= k
            # i.e., k <= j+n-1 and k >= j
            valid1 = (k_vals >= j) & (k_vals <= j + n - 1)
            
            # Second term validity: max(0, k-n+1) <= j+k-n+1 <= min(n-1, k)
            # i.e., j+k-n+1 >= 0 and j+k-n+1 <= n-1 and k-n+1 <= j+k-n+1 and j+k-n+1 <= k
            # j+k-n+1 >= 0 → k >= n-1-j
            # j+k-n+1 <= n-1 → k <= 2n-2-j
            # k-n+1 <= j+k-n+1 → 0 <= j (always true)
            # j+k-n+1 <= k → j <= n-1 (always true)
            valid2 = (k_vals >= n - 1 - j) & (k_vals <= 2*n - 2 - j)
            
            # First term: 1-h[n-1-k+j] for valid k
            idx1 = n - 1 - k_vals + j  # this is the index into h
            # For valid1, idx1 should be in [0, n-1]
            first_term = np.zeros(2*n-1)
            first_term[valid1] = 1.0 - h[idx1[valid1]]
            
            # Second term: -h[j+k-n+1] for valid k
            idx2 = j + k_vals - n + 1
            second_term = np.zeros(2*n-1)
            second_term[valid2] = -h[idx2[valid2]]
            
            D = first_term + second_term
            
            # Current corr
            corr = np.correlate(h, 1.0 - h, mode="full")
            
            # Solve min_t max_k (corr[k] + t * D[k]) for t in [-v, 1-v]
            lo = -v
            hi = 1.0 - v
            
            # Ternary search
            for _ in range(n_ternary):
                t1 = lo + (hi - lo) / 3.0
                t2 = hi - (hi - lo) / 3.0
                f1 = np.max(corr + t1 * D)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_slsqp(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def optimize_slsqp(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

def coord_descent(h, n_cd, n_grid=200):
    n = len(h)
    dx = 2.0 / n
    for _ in range(n_cd):
        improved = False
        for j in range(n):
            v = h[j]
            corr = np.correlate(h, 1.0 - h, mode="full")
            
            # Compute D = d(corr)/dh[j]
            k_vals = np.arange(2*n-1)
            valid1 = (k_vals >= j) & (k_vals <= j + n - 1)
            valid2 = (k_vals >= n - 1 - j) & (k_vals <= 2*n - 2 - j)
            
            idx1 = n - 1 - k_vals + j
            idx2 = j + k_vals - n + 1
            
            D = np.zeros(2*n-1)
            D[valid1] = 1.0 - h[idx1[valid1]]
            D[valid2] += -h[idx2[valid2]]
            
            # Grid search for best t in [-v, 1-v]
            t_vals = np.linspace(-v, 1.0 - v, n_grid)
            # f(t) = max( max_{k!=n-1}(corr[k]+t*D[k]), corr[n-1]+t*D[n-1]-t^2 )
            
            # Vectorized: compute corr + t*D for all t
            # Shape: (n_grid, 2n-1)
            base = corr[None, :] + np.outer(t_vals, D)
            # Fix the n-1 component
            base[:, n-1] = corr[n-1] + t_vals * D[n-1] - t_vals**2
            f_vals = np.max(base, axis=1)
            
            best_idx = int(np.argmin(f_vals))
            t_opt = t_vals[best_idx]
            new_v = v + t_opt
            
            if abs(new_v - v) > 1e-12:
                h[j] = new_v
                improved = True
        
        if not improved:
            break
    
    return h

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted)))
