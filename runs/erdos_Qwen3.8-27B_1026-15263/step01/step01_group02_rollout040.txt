import cvxpy as cp

def active_set_lp(h, n, top_k=None):
    """Solve LP relaxation at current point h"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c5_values = c * dx
    
    # Select top constraints
    K = len(c5_values)
    if top_k is None:
        top_k = min(K, 50)  # use top 50 constraints
    active_idx = np.argsort(c5_values)[-top_k:]
    
    # Linearize f_k at h
    # f_k(h) = dx * sum_i h[i] * (1-h[i+lag_k])
    # ∇f_k(h)[j] = dx * [(1-h[j+lag_k])*I(0<=j+lag_k<n) - h[j-lag_k)*I(0<=j-lag_k<n)]
    
    lags = np.arange(2*n-1) - (n-1)
    grads = np.zeros((K, n))
    for idx in range(K):
        lag = lags[idx]
        if lag >= 0:
            valid1 = n - lag
            if valid1 > 0:
                grads[idx, 0:valid1] += dx * (1.0 - h[lag:n])
                grads[idx, lag:n] += dx * (-h[0:valid1])
        else:
            alag = -lag
            valid1 = n - alag
            if valid1 > 0:
                grads[idx, alag:n] += dx * (1.0 - h[0:valid1])
                grads[idx, 0:valid1] += dx * (-h[alag:n])
    
    # Build LP
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2
    ]

        constraints.append(cp.sum(grads[idx] * (h_var - h)) + c5_values[idx] <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:

    except:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            return None
    
    if h_var.value is not None:
        return h_var.value, t_var.value
    return None, None

def paired_coord_descent(h, n, max_passes=5):
    """Coordinate descent with paired updates to maintain sum constraint"""
    h = h.copy()
    current_c5 = compute_c5_from_h(h)
    
    for _ in range(max_passes):
        improved = False
        # Shuffle order for diversity
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            # Find partner k (different from j)
            k = (j + 1) % n
            while k == j:
                k = (k + 1) % n
            
            orig_j = h[j]
            orig_k = h[k]
            
            # We want to search over t in [max(0, orig_j-orig_k), min(1, orig_j+1-orig_k)]
            # such that h[j] = orig_j + t, h[k] = orig_k - t
            # maintaining h[j] + h[k] = const
            
            t_lo = max(-orig_j, -(1.0 - orig_k))
            t_hi = min(1.0 - orig_j, orig_k)
            
            if t_hi - t_lo < 1e-10:
                continue
            
            best_t = 0.0
            best_val = compute_c5_from_h(h)
            
            # Try a few values of t

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total = np.sum(room)
            if total < 1e-14:
                break
            scale = min(1.0, diff / total)
            h[mask] += room * scale
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            total = np.sum(h[mask])
            if total < 1e-14:
                break
            scale = min(1.0, -diff / total)
            h[mask] -= h[mask] * scale
    return np.clip(h, 0.0, 1.0)


def smooth_c5_val(h, T):
    c5_vals = compute_c5_values(h)
    c_max = np.max(c5_vals)
    return c_max + T * np.log(np.sum(np.exp((c5_vals - c_max) / T)))


def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c5_vals = c * dx
    c_max = np.max(c5_vals)
    w = np.exp((c5_vals - c_max) / T)
    w /= np.sum(w)
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        lag = lags[idx]
        wi = w[idx] * dx
        if lag >= 0:
            valid = n - lag
            if valid > 0:
                grad[0:valid] += wi * (1.0 - h[lag:n])
                grad[lag:n] += wi * (-h[0:valid])
        else:
            alag = -lag
            valid = n - alag
            if valid > 0:
                grad[alag:n] += wi * (1.0 - h[0:valid])
                grad[0:valid] += wi * (-h[alag:n])
    return grad


def slsqp_smooth(h0, n, T, maxiter=2000):
    def obj(x):
        return smooth_c5_val(x, T)

    def obj_grad(x):
        return smooth_c5_grad(x, T)

    def eq_con(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        if result.x is not None:
            return project_feasible(result.x, n)
    except Exception:
        pass
    return h0


def paired_coord_descent(h, n, max_passes=5, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = h.copy()
    current_c5 = compute_c5_from_h(h)
    for _ in range(max_passes):
        improved = False
        order = rng.permutation(n)
        for j in order:
            k = (j + 7) % n
            if k == j:
                k = (j + 1) % n
            orig_j = h[j]
            orig_k = h[k]
            t_lo = max(-orig_j, -(1.0 - orig_k))
            t_hi = min(1.0 - orig_j, orig_k)
            if t_hi - t_lo < 1e-10:
                continue
            best_t = 0.0
            best_val = compute_c5_from_h(h)
            n_trials = 15
            for t_candidate in np.linspace(t_lo, t_hi, n_trials):
                h[j] = orig_j + t_candidate
                h[k] = orig_k - t_candidate
                val = compute_c5_from_h(h)
                if val < best_val:
                    best_val = val
                    best_t = t_candidate
            h[j] = orig_j + best_t
            h[k] = orig_k - best_t
            h = np.clip(h, 0.0, 1.0)
            if abs(best_t) > 1e-12:
                improved = True
            new_c5 = compute_c5_from_h(h)
            if new_c5 < current_c5:
                current_c5 = new_c5
        if not improved:
            break
    return h


def lp_direction(h, n, top_k=30):
    if not HAS_CVXPY:
        return None
    try:
        c5_vals = compute_c5_values(h)
        K = len(c5_vals)
        n_active = min(top_k, K)
        active_idx = np.argsort(c5_vals)[-n_active:]
        lags = np.arange(2 * n - 1) - (n - 1)
        dx = 2.0 / n
        grads = []
        for idx in active_idx:
            lag = lags[idx]
            g = np.zeros(n)
            if lag >= 0:
                valid = n - lag
                if valid > 0:
                    g[0:valid] = dx * (1.0 - h[lag:n])
                    g[lag:n] -= dx * h[0:valid]
            else:
                alag = -lag
                valid = n - alag
                if valid > 0:
                    g[alag:n] = dx * (1.0 - h[0:valid])
                    g[0:valid] -= dx * h[alag:n]
            grads.append(g)
        grads = np.array(grads)
        x_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            x_var >= 0,
            x_var <= 1,
            cp.sum(x_var) == n / 2
        ]
        for i, idx in enumerate(active_idx):
            constraints.append(c5_vals[idx] + grads[i] @ (x_var - h) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        if x_var.value is not None:
            return project_feasible(x_var.value, n)
    except Exception:
        pass
    return None


def multi_resolution(best_h, best_c5, n_best, rng, time_limit):
    for n_alt in [180, 225, 270]:
        if time.time() > time_limit:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='cubic', fill_value=0,
                            bounds_error=False)
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_alt)
        c5_interp = compute_c5_from_h(h_interp)

            c5_opt = compute_c5_from_h(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n_best = n_alt
                h_interp = best_h.copy()
