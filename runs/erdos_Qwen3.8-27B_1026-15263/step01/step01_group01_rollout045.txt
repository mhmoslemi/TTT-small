```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-15:

            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + diff, 0.0)
    return h

    best_h = h.copy()
    lr = lr0

        max_val = np.max(corr)

            best_h = h.copy()
```

```python
    # For each lag k (index in full array), lag = k - (n-1)
    # ∂(corr_k * dx)/∂h[m] = dx * [(1-h[m+lag])*I(m+lag<n, m>=0) - h[m-lag]*I(m-lag>=0, m<n)]
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # h[m] * (1-h[m+lag]) for m = 0..n-1-lag
            # ∂/∂h[m]:
            #   (1-h[m+lag]) for m < n-lag (first factor)
            #   -h[m-lag] for m >= lag (second factor, with sign)
            g = np.zeros(n)
            g[:n-lag] += (1.0 - h[lag:])  # first factor
            g[lag:] -= h[:n-lag]  # second factor
        else:
            L = -lag
            g = np.zeros(n)
            g[L:] += (1.0 - h[:n-L])  # first factor
            g[:n-L] -= h[L:]  # second factor
        grad += w[k] * dx * g
    
    return S, grad
```
