def project_feasible(h, n, tolerance=1e-6):
    # Ensuring 0 <= h_i <= 1 and sum(h) = n/2
    h_array = np.array(h)
    target_sum = n/2.0
    while True:
        # Enforce box constraints
        h_clipped = np.clip(h_array, 0, 1)
        diff = np.abs(np.sum(h_clipped) - target_sum)
        if diff < tolerance:
            return h_clipped
        # Phase 1: Clip extreme values
        sort_order = np.argsort(np.abs(h_array - 0.5))
        for idx in sort_order:
            if h_clipped[idx] > 1:
                h_clipped[idx] = 1
            elif h_array[idx] < 0:
                h_clipped[idx] = 0
        # Phase 2: Final adjustment
        new_sum = np.sum(h_clipped)
        target_idx = np.argmin(np.abs([np.sum(h_array[:i+1]) for i in range(n)]))

        h_array = h_clipped.copy()
        h_array[target_idx] = min(1, max(0, h_array[target_idx] + adjustment))
        return project_feasible(h_array, n, tolerance)
