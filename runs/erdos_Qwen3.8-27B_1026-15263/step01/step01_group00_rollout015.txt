def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    sm_val = m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha
    
    # Compute gradient
    g = np.zeros(n)
    for j in range(n):
        # Positive lags k = 0..n-1 (index n-1+k in corr)
        # Term: w[n-1+k] * (1-h[j-k]) for j>=k
        # Term: -w[n-1+k] * h[j+k] for j+k<n
        kmax1 = min(j, n-1)
        if kmax1 >= 0:
            ks = np.arange(0, kmax1+1)
            g[j] += np.dot(w[n-1+ks], 1.0 - h[j-ks])
        kmax2 = n - j
        if kmax2 > 0:
            ks2 = np.arange(0, kmax2)
            g[j] -= np.dot(w[n-1+ks2], h[j+ks2])
        # Negative lags l = 1..n-1 (index n-1-l in corr)
        # Term: w[n-1-l] * (1-h[j+l]) for j<n-l
        # Term: -w[n-1-l] * h[j-l] for j>=l
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1+1)
            g[j] += np.dot(w[n-1-ls], 1.0 - h[j+ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2+1)
            g[j] -= np.dot(w[n-1-ls2], h[j-ls2])
    
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0, 1)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0, 1)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def subgradient_step(h, alpha, lr, n, target_sum):

    k_star = np.argmax(corr)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha

    g = np.zeros(n)
    for j in range(n):
        kmax1 = min(j, n - 1)
        if kmax1 >= 0:
            ks = np.arange(0, kmax1 + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        kmax2 = n - j
        if kmax2 > 0:
            ks2 = np.arange(0, kmax2)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        lmax1 = n - j - 1
        if lmax1 >= 1:
            ls = np.arange(1, lmax1 + 1)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        lmax2 = j
        if lmax2 >= 1:
            ls2 = np.arange(1, lmax2 + 1)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    y = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)

            best_val = c5_cur
            best_delta = 0.0

                h_new = h.copy()

                h_new = project_to_sum(h_new, target_sum)

                h = project_to_sum(h, target_sum)

                improved = True
        if not improved:

    return h, c5_cur

                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()
            alphas = [3.0, 8.0, 20.0, 50.0, 100.0, 200.0, 400.0]
            for alpha in alphas:
                if time_left() < 40:
                    break
                maxiter = max(30, min(400, int(time_left() * 1.5)))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt

            if time_left() > 60:
                n_pg = min(3000, int(time_left() * 3))
                h_pg, c5_pg = projected_gradient(h_cur, n, ts, 50.0, n_pg, 0.05)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                h_cur = h_pg

            if time_left() > 40:
                h_cd, c5_cd = coordinate_descent(h_cur, n, ts, 40, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [300.0, 800.0, 2000.0]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 600)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 20:
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 100, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 15:
            n_pg = min(5000, int(time_left() * 2))
            h_pg, c5_pg = projected_gradient(best_h, n_f, ts_f, 100.0, n_pg, 0.02)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
