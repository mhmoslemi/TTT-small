```python
def compute_smoothed_gradient(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    # Smoothed max weights
    log_w = tau * c * dx - np.max(tau * c * dx)  # for numerical stability
    w = np.exp(log_w) / np.sum(np.exp(log_w))
    
    # Gradient of smoothed max
    grad = np.zeros(n)
    for j in range(n):
        for m in range(2*n-1):
            rel = m - (n-1)
            term = 0.0
            if 0 <= j - rel < n:
                term += (1.0 - h[j - rel])
            if 0 <= j + rel < n:
                term -= h[j + rel]
            grad[j] += w[m] * term
    return grad * dx
```
