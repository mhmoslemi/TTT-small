def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def corr_vector(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-16:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_subgradient(h, n, active_frac=0.1):
    """
    Compute (sub)gradient of C5 = max_m corr[m] * dx
    """
    dx = 2.0 / n
    corr = corr_vector(h)  # already scaled by dx
    cmax = np.max(corr)
    # Active set: indices where corr is within a fraction of max
    threshold = cmax - active_frac * (cmax - np.min(corr))
    active = np.where(corr >= threshold)[0]

    g *= dx / len(active)

    g_h = 1.0 - h

    cmax = np.max(corr_scaled)
    cmin = np.min(corr_scaled)
    range_val = cmax - cmin
    if range_val < 1e-15:
        return np.zeros(n)
    threshold = cmax - active_frac * range_val
    active = np.where(corr_scaled >= threshold)[0]
    
    g = np.zeros(n)
    for m in active:

        # d(c_m)/dh[j] = (1-h[j-k]) * I(0<=j-k<n) - h[j+k] * I(0<=j+k<n)
        # First term: j-k is valid, so j ranges from k to k+n-1 (clipped to [0,n-1])
        # Second term: j+k is valid, so j ranges from -k to n-1-k (clipped to [0,n-1])

    return g

    v = corr / tau

    return tau * (v_max + np.log(np.sum(np.exp(v - v_max))))

    def obj(h):
        return smooth_max_obj(h, n, tau)
    
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h_start
