```python
def project_onto_constraint(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = h.copy()
    target = n / 2.0
    # Binary search on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_proj)

        if diff > 0:
            mask = h_proj < 1.0 - 1e-10
        else:

        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
```

So: ∂corr[idx]/∂h[j] = (1-h[j-r]) * I[0≤j-r<n] - h[j+r] * I[0≤j+r<n]

where r = idx-(n-1).

Great, so the gradient formula is:
g[j] = (2/n) * sum_idx w[idx] * [(1-h[j-r_idx]) * I[0≤j-r_idx<n] - h[j+r_idx] * I[0≤j+r_idx<n]]

where r_idx = idx-(n-1).

Now let me vectorize this. For a fixed idx (with r = r_idx):
- Term 1: (1-h[j-r]) for j in [r, n-1] if r ≥ 0, or j in [0, n-1+r] if r < 0
  Actually: 0 ≤ j-r < n means r ≤ j < r+n. Combined with 0≤j<n:
  j in [max(0, r), min(n-1, r+n-1)]
  
- Term 2: -h[j+r] for j in [-r, n-1-r] if r ≥ 0, or j in [0, n-1-r] if r < 0
  Actually: 0 ≤ j+r < n means -r ≤ j < n-r. Combined with 0≤j<n:
  j in [max(0, -r), min(n-1, n-r-1)]

For r ≥ 0:
  Term 1: j in [r, n-1], value = (1-h[j-r]) = (1-h[0:n-r])
  Term 2: j in [0, n-1-r], value = -h[j+r] = -h[r:n]

For r < 0, let s = -r > 0:
  Term 1: j in [0, n-1-s], value = (1-h[j+s]) = (1-h[s:n])
  Term 2: j in [s, n-1], value = -h[j-s] = -h[0:n-s]

```python
def compute_gradient(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    g = np.zeros(n)
    for idx in range(len(corr)):
        r = idx - (n - 1)
        wm = w[idx]
        if wm < 1e-15:
            continue
        if r >= 0:
            if n - r > 0:
                g[r:] += wm * (1.0 - h[:n-r])  # Term 1
                g[:n-r] -= wm * h[r:]  # Term 2
        else:
            s = -r
            if n - s > 0:
                g[:n-s] += wm * (1.0 - h[s:])  # Term 1
                g[s:] -= wm * h[:n-s]  # Term 2
    
    return g * (2.0 / n)
```

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2} using bisection on shift."""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # If already feasible
    if np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and abs(np.sum(h) - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    # Bisection on lambda: find lambda s.t. sum(clip(h+lambda, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_mid) > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_gradient(h, n, beta):
    """Compute gradient of (2/n) * log_sum_exp(corr(h), beta)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    
    g = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        r = idx - (n - 1)
        wm = w[idx]

def projected_gradient_solve(h0, n, time_budget, beta0=10.0, beta_final=500.0):
    """Solve using projected gradient descent with adaptive beta and line search."""
    h = project_onto_constraint(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t_start = time.time()
    total_iterations = 0
    lr = 0.5
    
    while time.time() - t_start < time_budget:
        # Adaptive beta
        progress = min(1.0, total_iterations / max(1, time_budget * 50))
        beta = beta0 * (beta_final / beta0) ** progress
        
        g = compute_gradient(h, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            h_new = project_onto_constraint(h - alpha * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:

                h_rand = project_onto_constraint(h + 0.1 * np.random.randn(n), n)

                if c5_rand < best_c5:
                    h = h_rand
                    best_h = h_rand.copy()
                    best_c5 = c5_rand
                    break
            else:
                # No improvement, reduce learning rate
                lr *= 0.7
                if lr < 1e-6:
                    break
        
        total_iterations += 1
    
    return best_h, best_c5
