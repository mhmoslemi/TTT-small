def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

def compute_gradient_for_lag(h, n, m):
    """Compute gradient of corr[n-1+m] with respect to h."""
    grad = np.zeros(n)
    if m >= 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[-m:] += (1.0 - h[:n+m])
        grad[:n+m] -= h[-m:]
    return grad

    """Compute average gradient over active lags."""
    grad = np.zeros(n)
    for m in active_lags:
        grad += compute_gradient_for_lag(h, n, m)
    grad /= len(active_lags)
    return grad

        v = momentum * v - step * grad
        h = h + v
        h = project_feasible(h, n)

            grad[m:] -= h[:n - m]
    else:

        if m_neg < n:
            grad[m_neg:] += (1.0 - h[:n - m_neg])
            grad[:n - m_neg] -= h[m_neg:]

        elif kind == 1:
            pert = 0.015 * rng.standard_normal(n_best)
        elif kind == 2:
            block = max(1, n_best // 5)
            pert = np.zeros(n_best)
            b = int(rng.integers(0, max(1, n_best // block)))
            sl = slice(b * block, min((b + 1) * block, n_best))
            pert[sl] = 0.04 * rng.standard_normal(sl.stop - sl.start)
        elif kind == 3:
            pert = np.cumsum(0.004 * rng.standard_normal(n_best))
            pert -= pert.mean()
        else:
            # Two-frequency sinusoid
            f1, f2 = rng.uniform(1, 4), rng.uniform(4, 10)
            amp = rng.uniform(0.02, 0.05)
            x_grid = np.linspace(0, 2, n_best)
            pert = amp * (np.sin(2 * np.pi * f1 * x_grid) + np.sin(2 * np.pi * f2 * x_grid))

        h_pert = project_feasible(best_h + pert, n_best)
        h_ref, c5_ref = slsqp_mm(h_pert, n_best, maxiter=1200, ftol=1e-13)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref
            n_best = n_best

    # Phase 5: Try different n values via interpolation
    for n_try in [90, 180, 225, 270, 315]:
        if time.time() - t0 > budget_s - 80:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        fi = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate', assume_sorted=True)
        h_int = np.clip(fi(x_new), 0.0, 1.0)
        h_int = project_feasible(h_int, n_try)
        h_ref, c5_ref = slsqp_mm(h_int, n_try, maxiter=1500, ftol=1e-13)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref
            n_best = n_try

    # Phase 6: Final aggressive polish
    if time.time() - t0 < budget_s - 15:
        best_h, best_c5 = slsqp_mm(best_h, n_best, maxiter=5000, ftol=1e-15)

    # Phase 7: One more round of small perturbations on final
    if time.time() - t0 < budget_s - 5:
        for _ in range(10):
            if time.time() - t0 > budget_s - 2:
                break
            h_pert = project_feasible(best_h + 0.005 * rng.standard_normal(n_best), n_best)
            h_ref, c5_ref = slsqp_mm(h_pert, n_best, maxiter=1000, ftol=1e-14)
            if c5_ref < best_c5:
                best_h = h_ref
                best_c5 = c5_ref

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
