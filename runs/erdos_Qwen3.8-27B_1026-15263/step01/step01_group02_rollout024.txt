    # Actually, let me compute the gradient numerically for each active lag
    
    # For efficiency, let's compute the subgradient analytically
    # c_full[s] = sum_i h[i] * (1-h[i+s-(n-1)]) for s in [0, 2n-2]

    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)


def fourier_to_h(params, n):
    K = len(params) // 2
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(K):
        h += params[2 * k] * np.cos(np.pi * (k + 1) * x)
        h += params[2 * k + 1] * np.sin(np.pi * (k + 1) * x)
    return h


def slsqp_smooth(h, n, alpha, maxiter):
    def obj(x):
        return smooth_max_obj(x, alpha)

    def eq(x):
        return np.sum(x) - n / 2.0

    cons = [{'type': 'eq', 'fun': eq}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_fft(h_out)
    return h, compute_c5_fft(h)


def subgradient_step(h, n, lr):
    dx = 2.0 / n
    c = fftconvolve(h, 1.0 - h, mode='full') * dx
    max_val = np.max(c)
    eps = 1e-4
    active = c >= max_val - eps
    active_idx = np.where(active)[0]
    if len(active_idx) == 0:
        return h.copy()
    grad = np.zeros(n)
    for s in active_idx:
        shift = s - (n - 1)
        if shift >= 0:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] -= h[shift:]
        else:
            s2 = -shift
            grad[s2:] += (1.0 - h[:n - s2])
            grad[:n - s2] -= h[s2:]
    grad *= dx / len(active_idx)
    h_new = h - lr * grad
    return project_feasible(h_new, n)


def de_fourier_obj(params, n):
    h = fourier_to_h(params, n)
    h = project_feasible(h, n)
    return compute_c5_fft(h)


def de_full_obj(params, n):
    h = np.clip(params, 0.0, 1.0)
    h = project_feasible(h, n)
    return compute_c5_fft(h)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    n = 135
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        h0 = project_feasible(h0, n)
    else:
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5_fft(h0)
    n_best = n

    # Phase 1: DE with Fourier parameterization (multiple K values)
    for K in [5, 7, 9, 12]:
        if time.time() - t0 > budget_s - 250:
            break
        dim = 2 * K
        bounds_f = [(-2.0, 2.0)] * dim
        remaining = budget_s - (time.time() - t0)
        max_iter_de = min(400, int(remaining * 8))
        if max_iter_de < 20:
            continue
        de_res = differential_evolution(
            de_fourier_obj, bounds_f, args=(n_best,),
            strategy='best1bin', max_iter=max_iter_de,
            tol=1e-15, mutation=(0.15, 0.5), recombination=0.9,
            seed=seed + K * 37, popsize=25, polish=False
        )
        if de_res.x is not None:
            h_cand = fourier_to_h(de_res.x, n_best)
            h_cand = project_feasible(h_cand, n_best)
            c5_cand = compute_c5_fft(h_cand)
            if c5_cand < best_c5:
                best_h = h_cand.copy()
                best_c5 = c5_cand

    # Phase 2: DE on full h vector
    if time.time() - t0 < budget_s - 150:
        bounds_full = [(0.0, 1.0)] * n_best
        remaining = budget_s - (time.time() - t0)
        max_iter_de2 = min(150, int(remaining * 3))
        if max_iter_de2 >= 10:
            de_res2 = differential_evolution(
                de_full_obj, bounds_full, args=(n_best,),
                strategy='best1bin', max_iter=max_iter_de2,
                tol=1e-15, mutation=(0.1, 0.3), recombination=0.9,
                seed=seed + 7777, popsize=15, polish=False
            )
            if de_res2.x is not None:
                h_cand = project_feasible(de_res2.x, n_best)
                c5_cand = compute_c5_fft(h_cand)
                if c5_cand < best_c5:
                    best_h = h_cand.copy()
                    best_c5 = c5_cand

    # Phase 3: Continuation SLSQP with increasing alpha
    alphas = [10, 30, 100, 300, 1000, 3000]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 40:
            break
        h_new, c5_new = slsqp_smooth(best_h, n_best, alpha, maxiter=1500)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Subgradient descent (many cheap steps)
    if time.time() - t0 < budget_s - 30:
        lr = 0.002
        for i in range(2000):
            if time.time() - t0 > budget_s - 25:
                break
            h_new = subgradient_step(best_h, n_best, lr)
            c5_new = compute_c5_fft(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
            else:
                lr *= 0.95
            if lr < 1e-7:
                break
        # Follow up with SLSQP
        if time.time() - t0 < budget_s - 20:
            h_new, c5_new = slsqp_smooth(best_h, n_best, 500, maxiter=1000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Perturbation + SLSQP
    n_pert = 60
    if time.time() - t0 < budget_s - 30:
        for i in range(n_pert):
            if time.time() - t0 > budget_s - 20:
                break
            frac = i / max(1, n_pert)
            scale = 0.05 * (1.0 - frac) + 0.003
            h_pert = best_h + scale * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            alpha_pert = 200 + 200 * rng.random()
            h_new, c5_new = slsqp_smooth(h_pert, n_best, alpha_pert, maxiter=400)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                # Quick subgradient polish
                for _ in range(200):
                    if time.time() - t0 > budget_s - 15:
                        break
                    h_tmp = subgradient_step(best_h, n_best, 0.001)
                    c5_tmp = compute_c5_fft(h_tmp)
                    if c5_tmp < best_c5:
                        best_h = h_tmp
                        best_c5 = c5_tmp
                    else:
                        break

    # Phase 6: Higher resolution refinement
    for n_try in [270, 405, 540]:
        if time.time() - t0 > budget_s - 50:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_int = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
        h_interp = np.clip(f_int(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = slsqp_smooth(h_interp, n_try, 100, maxiter=1000)
        h_new, c5_new = slsqp_smooth(h_new, n_try, 500, maxiter=1500)
        h_new, c5_new = slsqp_smooth(h_new, n_try, 2000, maxiter=1000)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            n_best = n_try

    # Phase 7: Final aggressive refinement
    if time.time() - t0 < budget_s - 10:
        for alpha_final in [500, 2000, 5000]:
            if time.time() - t0 > budget_s - 5:
                break
            h_new, c5_new = slsqp_smooth(best_h, n_best, alpha_final, maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # One last subgradient sweep
    if time.time() - t0 < budget_s - 3:
        lr = 0.0005
        for _ in range(1000):
            if time.time() - t0 > budget_s - 2:
                break
            h_new = subgradient_step(best_h, n_best, lr)
            c5_new = compute_c5_fft(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
            else:
                lr *= 0.9

    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5_fft(best_h)

    return best_h, best_c5, n_best
