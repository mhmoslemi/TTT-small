import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on lambda."""
    h = np.array(h, dtype=float)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_vals(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    sum_exp = exp_vals.sum()
    smax = m + max_s / alpha + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    return smax, w

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    smax, w = smooth_max_vals(corr, alpha)
    
    j_idx = np.arange(n)[:, None]
    p_idx = np.arange(n)[None, :]
    idx1 = j_idx + (n - 1) - p_idx
    idx2 = p_idx - j_idx + (n - 1)
    
    first = w[idx1] @ (1.0 - h)
    second = w[idx2] @ h
    grad = first - second
    return smax, grad, corr

def pgd_step(h, alpha, lr, max_lr=1.0):
    """One projected gradient descent step with backtracking."""
    n = len(h)
    target_sum = n / 2.0
    smax, grad, corr = compute_grad(h, alpha)
    
    # Backtracking line search
    f_cur = smax
    step = min(lr, max_lr)
    for _ in range(30):
        h_new = project_feasible(h - step * grad, target_sum)
        f_new = compute_c5(h_new)  # Use actual max, not smooth
        if f_new < f_cur or step < 1e-6:
            return h_new, f_new
        step *= 0.5
    return h_new, f_new

def optimize_pgd(h0, alpha_schedule, n_steps_budget, t_start, time_limit):
    """Run projected gradient descent with alpha schedule."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    f_cur = compute_c5(h)
    best_h = h.copy()
    best_f = f_cur
    
    for alpha, lr, n_steps in alpha_schedule:
        for it in range(n_steps):
            if time.time() - t_start > time_limit - 10:
                return best_h, best_f
            h_new, f_new = pgd_step(h, alpha, lr)
            if f_new < f_cur:
                h = h_new
                f_cur = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
            else:
                # No improvement, break inner loop
                break
    return best_h, best_f

def perturb_block(h, rng, block_size=None):
    """Perturb a random block of consecutive values."""
    n = len(h)
    if block_size is None:
        block_size = rng.randint(3, max(4, n // 10))
    start = rng.randint(0, n - block_size + 1)
    h_new = h.copy()
    delta = rng.uniform(-0.1, 0.1)
    h_new[start:start+block_size] += delta
    target_sum = n / 2.0
    return project_feasible(h_new, target_sum)

def perturb_random(h, rng, strength=0.05):
    """Random perturbation."""
    n = len(h)
    h_new = h + rng.randn(n) * strength
    target_sum = n / 2.0
    return project_feasible(h_new, target_sum)

def interpolate_h(h_low, n_high):
    """Interpolate from lower to higher resolution."""
    n_low = len(h_low)
    x_low = np.linspace(0, 2, n_low, endpoint=False)
    x_high = np.linspace(0, 2, n_high, endpoint=False)
    h_high = np.interp(x_high, x_low, h_low, period=2.0)
    return project_feasible(h_high, n_high / 2.0)

    time_limit = t_start + budget_s

    for n in n_list:

        else:

        seg = n // 6
        for k in [0, 2, 4]:
            h_blk[k*seg:(k+1)*seg] = 1.0
        starts.append(project_feasible(h_blk, n/2.0))

            hs = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x)
            starts.append(project_feasible(hs, n/2.0))

        starts = starts[:6]

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_vals(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    sum_exp = exp_vals.sum()
    smax = m + max_s / alpha + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    return smax, w

def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    smax, w = smooth_max_vals(corr, alpha)
    j_idx = np.arange(n)[:, None]
    p_idx = np.arange(n)[None, :]
    idx1 = j_idx + (n - 1) - p_idx
    idx2 = p_idx - j_idx + (n - 1)
    first = w[idx1] @ (1.0 - h)
    second = w[idx2] @ h
    return smax, first - second, corr

def slsqp_optimize(h0, alpha, maxiter, t_start, time_limit):
    n = len(h0)
    target_sum = n / 2.0

    def obj(x):
        return compute_gradient(x, alpha)[0]

    def obj_grad(x):
        f, g = compute_gradient(x, alpha)[:2]
        return f, g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        return h_out
    except:
        return project_feasible(h0, target_sum)

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    return project_feasible(h_dst, n_dst / 2.0)

def perturb_block(h, rng, max_block=None):
    n = len(h)
    if max_block is None:
        max_block = max(3, n // 15)
    bs = rng.randint(2, max_block + 1)
    start = rng.randint(0, n - bs + 1)
    h_new = h.copy()
    h_new[start:start + bs] += rng.uniform(-0.15, 0.15)
    return project_feasible(h_new, n / 2.0)

def perturb_gaussian(h, rng, sigma=0.03):
    n = len(h)
    h_new = h + rng.randn(n) * sigma
    return project_feasible(h_new, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    time_limit = t_start + budget_s

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            h_p = np.array(initial_h_values, dtype=float)
            n_p = len(h_p)
            h_p = project_feasible(h_p, n_p / 2.0)
            best_h = h_p.copy()
            best_c5 = compute_c5(h_p)
            best_n = n_p
        except:
            pass

    if best_h is None:

        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def try_optimize(h_start, n, label=""):

        h_start = project_feasible(h_start, n / 2.0)

            best_n = n

            c5_new = compute_c5(h_new)

                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
            else:
                h_cur = h_new
        return h_cur, c5_cur

    n_targets = sorted(set([best_n] + [90, 120, 144, 180]))
    n_targets = [n for n in n_targets if n % 2 == 0]

        h_blk = np.zeros(n)

        for k in [0, 2, 4]:
            h_blk[k * seg:(k + 1) * seg] = 1.0
        starts.append(project_feasible(h_blk, n / 2.0))

        starts = starts[:7]

        for h_s in starts:
            if time.time() > time_limit - 40:
                break
            h_opt, c5_opt = try_optimize(h_s, n)

            for pr in range(8):
                if time.time() > time_limit - 40:
                    break
                if pr < 4:
                    h_pert = perturb_block(h_opt, rng)
                else:
                    h_pert = perturb_gaussian(h_opt, rng, sigma=0.02)
                h_re, c5_re = try_optimize(h_pert, n)
                if c5_re < compute_c5(h_opt):
                    h_opt = h_re
                elif pr > 2:
                    break

    if best_h is not None and time.time() < time_limit - 30:
        n_f = best_n

        for alpha in [500.0, 1000.0, 3000.0]:
            if time.time() > time_limit - 20:

            c5_new = compute_c5(h_final)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_final.copy()

    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
