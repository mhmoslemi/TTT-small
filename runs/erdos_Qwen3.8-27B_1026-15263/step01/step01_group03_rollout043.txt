import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.linalg import toeplitz
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max_value(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def compute_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    w = exp_v / exp_v.sum()
    obj = smooth_max_value(corr, tau) * dx
    
    # Build Toeplitz matrices
    first_row_A = w[n-1:]
    first_col_A = w[n-1::-1]
    A = toeplitz(first_col_A, first_row_A)
    
    first_col_B = w[n-1:]
    first_row_B = w[n-1::-1]
    B = toeplitz(first_col_B, first_row_B)
    
    s1 = A @ h
    s2 = B @ h
    
    g = (dx / tau) * (1.0 - s1 - s2)
    return obj, g

def gradient_descent(h, n, tau_schedule, rng, time_limit):
    start_time = time.time()
    h = h.copy()
    best_local = h.copy()
    best_local_obj = float('inf')
    
    lr = 0.05
    min_improve = 1e-8
    
    for tau in tau_schedule:
        for it in range(2000):
            if time.time() > time_limit:
                break
            obj, g = compute_grad(h, n, tau)
            
            # Backtracking line search
            for ls in range(20):
                h_new = h - lr * g
                h_new = project_feasible(h_new, n)
                obj_new, _ = compute_grad(h_new, n, tau)
                if obj_new < obj - min_improve * (1 + abs(obj)):
                    h = h_new
                    break
                lr *= 0.5
            else:
                # Even the smallest step didn't help
                lr = max(lr * 1.5, 0.01)
                continue
            
            if obj < best_local_obj:
                best_local_obj = obj
                best_local = h.copy()
            
            # Adaptive learning rate
            if ls < 5:
                lr = min(lr * 1.1, 0.5)
        
        if time.time() > time_limit:
            break
    
    return best_local, best_local_obj

def slsqp_polish(h, n, tau, maxiter=100):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_value(corr, tau) * dx
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    res = minimize(
        objective, h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def interpolate_parent(parent_h, n):
    if parent_h is None:
        return None
    n_p = len(parent_h)
    if n_p == n:
        return parent_h.copy()
    x_p = np.linspace(0, 2, n_p, endpoint=False) + 1.0/n_p
    x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    f = interp1d(x_p, parent_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    return np.clip(f(x_new), 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    if parent_h is not None:
        parent_h = np.clip(parent_h, 0, 1)
        best_c5 = evaluate_c5(parent_h)
        best_h = parent_h.copy()
    else:
        best_c5 = float('inf')
        best_h = None
    
    n_list = []
    if parent_h is not None:
        n_list.append(len(parent_h))
    for nn in [79, 99, 119, 149, 199]:
        if nn not in n_list:
            n_list.append(nn)
    
    # Phase 1: Gradient descent on each n
    for n in n_list:
        if time.time() - start_time > budget_s * 0.75:
            break
        
        # Multiple starting points
        starts = []
        if parent_h is not None:
            h_init = interpolate_parent(parent_h, n)
            starts.append(h_init)
            # Perturbed versions
            for scale in [0.05, 0.1, 0.2]:
                perturbed = h_init + rng.randn(n) * scale
                starts.append(np.clip(perturbed, 0, 1))
        
        # Random starts
        for _ in range(2):
            starts.append(rng.rand(n))
        
        # Structured starts
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1.5, 2.0, 2.5]:
            for phase in [0, np.pi/2, np.pi]:
                h_struct = 0.5 + 0.35 * np.sin(2*np.pi*freq*x + phase)
                starts.append(np.clip(h_struct, 0, 1))
        
        for h0 in starts:
            if time.time() - start_time > budget_s * 0.75:
                break
            
            h0 = project_feasible(h0, n)
            
            # Gradient descent with tau annealing
            tau_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            time_limit = time.time() + min(60, (budget_s * 0.75 - (time.time() - start_time)) / max(1, len(starts)))

            c5 = evaluate_c5(h_opt)
            
            if c5 < best_c5:

                best_h = h_opt.copy()

        best_h = project_feasible(np.full(n, 0.5), n)

    return best_h, best_c5, len(best_h)

def toeplitz_matvec(h, n, w):

    result = np.zeros(n)

            if 0 <= idx <= 2*n-2:

    return result

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.linalg import toeplitz
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max_value(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def compute_obj_and_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    w = exp_v / exp_v.sum()
    obj = smooth_max_value(corr, tau) * dx
    fr = w[n - 1:]
    fc = w[n - 1::-1]
    A = toeplitz(fc, fr)
    frB = w[n - 1::-1]
    fcB = w[n - 1:]
    B = toeplitz(fcB, frB)
    s1 = A.dot(h)
    s2 = B.dot(h)
    g = (dx / tau) * (1.0 - s1 - s2)
    return obj, g

def gradient_phase(h, n, tau_levels, max_iter_per_tau, deadline):
    h = h.copy()
    best_h = h.copy()
    best_obj = float('inf')
    lr = 0.1
    for tau in tau_levels:
        if time.time() > deadline:
            break
        lr = 0.1
        for it in range(max_iter_per_tau):
            if time.time() > deadline:
                break
            obj, g = compute_obj_and_grad(h, n, tau)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-10:
                break
            g = g / g_norm * g_norm
            accepted = False
            for ls in range(25):
                h_new = h - lr * g
                h_new = project_feasible(h_new, n)
                obj_new, _ = compute_obj_and_grad(h_new, n, tau)
                if obj_new < obj - 1e-12:
                    h = h_new
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                lr = 0.05
                h_pert = h + np.random.randn(n) * 0.005
                h = project_feasible(h_pert, n)
                continue
            if ls < 8:
                lr = min(lr * 1.05, 1.0)
        if time.time() > deadline:
            break
    return best_h

def slsqp_refine(h, n, tau, maxiter):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_value(c, tau) * dx

    def constr(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constr}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interp_parent(parent_h, n):
    if parent_h is None:
        return None
    np_ = len(parent_h)
    if np_ == n:
        return parent_h.copy()
    xp = np.linspace(0.0, 2.0, np_, endpoint=False) + 1.0 / np_
    xn = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    f = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate')
    return np.clip(f(xn), 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)

    except Exception:

    best_h = None
    best_c5 = float('inf')
    if parent_h is not None:
        best_c5 = evaluate_c5(parent_h)
        best_h = parent_h.copy()

    n_list = []
    if parent_h is not None:

    for nn in [79, 99, 119, 149, 199]:
        if nn not in n_list:
            n_list.append(nn)

    phase1_deadline = start_time + budget_s * 0.70
    for n in n_list:
        if time.time() > phase1_deadline:
            break
        starts = []
        if parent_h is not None:
            h_i = interp_parent(parent_h, n)
            starts.append(h_i)
            for sc in [0.03, 0.08, 0.15]:
                starts.append(np.clip(h_i + rng.randn(n) * sc, 0.0, 1.0))
        for _ in range(2):
            starts.append(rng.rand(n))
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        for freq in [1.3, 1.7, 2.3, 2.8]:
            for ph in [0.0, 1.0, 2.1]:
                starts.append(np.clip(0.5 + 0.38 * np.sin(2.0 * np.pi * freq * x + ph), 0.0, 1.0))

        remaining = max(10.0, phase1_deadline - time.time())
        per_start_time = remaining / max(1, len(starts))
        for h0 in starts:
            if time.time() > phase1_deadline:
                break
            h0 = project_feasible(h0, n)
            tau_sched = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
            max_it = max(500, int(per_start_time * 50))
            dl = time.time() + per_start_time
            h_opt = gradient_phase(h0, n, tau_sched, max_it, dl)
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is not None and time.time() < start_time + budget_s * 0.88:
        n = len(best_h)

        for tau in [0.001, 0.0005, 0.0002, 0.0001]:

                c5 = evaluate_c5(h_p)

                    best_h = h_p.copy()

    if best_h is None:

        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
