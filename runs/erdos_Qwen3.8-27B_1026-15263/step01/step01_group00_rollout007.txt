import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        kp = -k
        g[:n-kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n-kp]
    return g

def subgradient_step(h, top_m=10, concentration=30.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shift = n - 1
    max_corr = np.max(corr)
    
    # Top M indices
    top_idx = np.argsort(corr)[-top_m:]
    
    # Weighted gradient
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - shift
        w = np.exp(concentration * (corr[k_idx] - max_corr))
        grad += w * grad_corr_k(h, k, n)
    
    return grad, max_corr

def refine_subgrad(h, target_sum, max_iters=500, t_end=None):
    h = project_to_sum(h.copy(), target_sum)
    c5_best = compute_c5(h)
    h_best = h.copy()
    step = 0.01
    n = len(h)
    
    for it in range(max_iters):
        if t_end is not None and time.time() > t_end:
            break
        grad, max_corr = subgradient_step(h, top_m=8, concentration=20.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Try several step sizes
        improved = False
        for s in [step, step*0.5, step*2, step*0.25]:
            h_trial = h - s * grad / gnorm
            h_trial = project_to_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                h = h_trial
                c5_best = c5_trial
                h_best = h.copy()
                step = s
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    
    return h_best, c5_best

    pop = np.zeros((pop_size, n))
    pop[0] = project_to_sum(h_init.copy(), target_sum)
    
    # Diverse starts
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.3:
            # Perturbation of initial
            pop[i] = project_to_sum(h_init + rng.normal(0, 0.1, n), target_sum)
        elif r < 0.6:

            x = np.arange(n) / n * 2.0
            freq = rng.randint(1, 6)
            pop[i] = project_to_sum(np.clip(0.5 + 0.35*np.sin(2*np.pi*freq*x/2.0 + rng.uniform(0, 2*np.pi)), 0, 1), target_sum)
        else:
            # Random
            pop[i] = project_to_sum(rng.uniform(0, 1, n), target_sum)
    
    fitnesses = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitnesses)
    best_h = pop[best_idx].copy()
    best_f = fitnesses[best_idx]
    
    elite = min(8, pop_size // 5)
    
    for gen in range(max_gens):
        if time.time() > t_end - 5:
            break
        
        new_pop = np.zeros((pop_size, n))
        # Elitism
        sort_idx = np.argsort(fitnesses)
        new_pop[:elite] = pop[sort_idx[:elite]]
        new_fitnesses = np.zeros(pop_size)
        new_fitnesses[:elite] = fitnesses[sort_idx[:elite]]
        
        for i in range(elite, pop_size):
            # Tournament select 2 parents
            c1 = rng.randint(0, pop_size, 4)
            p1 = pop[c1[np.argmin(fitnesses[c1])]]
            c2 = rng.randint(0, pop_size, 4)
            p2 = pop[c2[np.argmin(fitnesses[c2])]]
            
            # BLX-alpha crossover
            alpha = 0.5
            lo = np.minimum(p1, p2) - alpha * np.abs(p1 - p2)
            hi = np.maximum(p1, p2) + alpha * np.abs(p1 - p2)

            best_h = pop[bi].copy()
    
    return best_h, best_f

import time

    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if np.any(mask):
            h[mask] += diff / np.sum(mask)

        if time.time() > t_end - 2:
            break
        sort_idx = np.argsort(fitnesses)
        new_pop = np.zeros((pop_size, n))
        new_fit = np.zeros(pop_size)
        for ei in range(elite):
            new_pop[ei] = pop[sort_idx[ei]]
            new_fit[ei] = fitnesses[sort_idx[ei]]
        for i in range(elite, pop_size):
            c1 = rng.randint(0, pop_size, 4)
            p1 = pop[c1[np.argmin(fitnesses[c1])]]
            c2 = rng.randint(0, pop_size, 4)
            p2 = pop[c2[np.argmin(fitnesses[c2])]]
            alpha_cx = 0.4
            lo = np.minimum(p1, p2) - alpha_cx * np.abs(p1 - p2)
            hi = np.maximum(p1, p2) + alpha_cx * np.abs(p1 - p2)
            child = rng.uniform(np.maximum(0.0, lo), np.minimum(1.0, hi))
            sigma = 0.03 * (0.997 ** gen) + 0.003
            mut_mask = rng.random(n) < 0.35
            child[mut_mask] += rng.normal(0, sigma, mut_mask.sum())
            child = np.clip(child, 0.0, 1.0)
            child = project_to_sum(child, target_sum)
            new_pop[i] = child
            new_fit[i] = compute_c5(child)
        pop = new_pop
        fitnesses = new_fit
        bi = int(np.argmin(fitnesses))
        if fitnesses[bi] < best_f:
            best_f = fitnesses[bi]
            best_h = pop[bi].copy()
    return best_h, best_f

def smooth_max_optimize(h_init, n, target_sum, alpha, maxiter, t_end):
    def objective(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        m = np.max(corr)
        ev = np.exp(alpha * (corr - m))
        return m + np.log(np.sum(ev)) / alpha
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    h0 = project_to_sum(h_init.copy(), n / 2.0)
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
        if time.time() < t_end:
            return project_to_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
    except:
        pass
    return project_to_sum(h0, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            parent = np.array(initial_h_values, dtype=float)
            n_p = len(parent)
            parent = project_to_sum(parent, n_p / 2.0)
            best_h = parent.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_p
        except:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n0 = best_n
    ts0 = n0 / 2.0

    t_phase1 = t_start + budget_s * 0.30
    h_ga, f_ga = ga_optimize(best_h, n0, ts0, seed, 400, 90, t_phase1)
    if f_ga < best_c5:
        best_c5 = f_ga
        best_h = h_ga.copy()
        best_n = n0

    t_phase2 = t_start + budget_s * 0.45
    h_sg, f_sg = subgradient_refine(best_h, ts0, 3000, t_phase2, top_m=10, conc=25.0)
    if f_sg < best_c5:
        best_c5 = f_sg
        best_h = h_sg.copy()

    t_phase2b = t_start + budget_s * 0.52
    for alpha in [80.0, 200.0]:
        if time.time() > t_phase2b:
            break
        h_sm = smooth_max_optimize(best_h, n0, ts0, alpha, 200, t_phase2b)
        f_sm = compute_c5(h_sm)
        if f_sm < best_c5:
            best_c5 = f_sm
            best_h = h_sm.copy()

    n_hi = 360
    if time.time() < t_start + budget_s * 0.78:
        ts_hi = n_hi / 2.0
        x_old = np.arange(best_n) / best_n * 2.0
        x_new = np.arange(n_hi) / n_hi * 2.0
        h_hi = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]), period=2.0)
        h_hi = h_hi[:n_hi]
        h_hi = project_to_sum(h_hi, ts_hi)
        f_hi = compute_c5(h_hi)
        if f_hi < best_c5:
            best_c5 = f_hi
            best_h = h_hi.copy()
            best_n = n_hi
        t_phase3 = t_start + budget_s * 0.78
        h_ga2, f_ga2 = ga_optimize(h_hi, n_hi, ts_hi, seed + 50, 150, 60, t_phase3)
        if f_ga2 < best_c5:
            best_c5 = f_ga2
            best_h = h_ga2.copy()
            best_n = n_hi

    n_hi2 = 500
    if time.time() < t_start + budget_s * 0.88:
        ts_hi2 = n_hi2 / 2.0
        x_old = np.arange(best_n) / best_n * 2.0
        x_new = np.arange(n_hi2) / n_hi2 * 2.0
        h_hi2 = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]), period=2.0)
        h_hi2 = h_hi2[:n_hi2]
        h_hi2 = project_to_sum(h_hi2, ts_hi2)
        f_hi2 = compute_c5(h_hi2)
        if f_hi2 < best_c5:
            best_c5 = f_hi2
            best_h = h_hi2.copy()
            best_n = n_hi2
        t_phase3b = t_start + budget_s * 0.88
        h_sg2, f_sg2 = subgradient_refine(h_hi2, ts_hi2, 1000, t_phase3b, top_m=12, conc=30.0)
        if f_sg2 < best_c5:
            best_c5 = f_sg2
            best_h = h_sg2.copy()
            best_n = n_hi2

    t_phase4 = t_start + budget_s * 0.95
    h_cd, f_cd = coordinate_descent(best_h, best_n / 2.0, 60, t_phase4, rng)
    if f_cd < best_c5:
        best_c5 = f_cd
        best_h = h_cd.copy()

    t_phase5 = t_start + budget_s * 0.98
    h_fin, f_fin = subgradient_refine(best_h, best_n / 2.0, 5000, t_phase5, top_m=12, conc=35.0)
    if f_fin < best_c5:
        best_c5 = f_fin
        best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, int(best_n))
