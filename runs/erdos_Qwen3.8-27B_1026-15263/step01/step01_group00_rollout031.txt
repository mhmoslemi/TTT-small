    for _ in range(100):

For v=n, k=n/2, we need λ = k(k-1)/(v-1) = (n/2)(n/2-1)/(n-1) ≈ n/4 for large n.

Existence of difference sets: (v, k, λ) where v = k² - k + 1 (Singer difference sets) gives k = k²-k+1 → k²-k-k+1=0... that's not right.

Actually, for a (v, k, λ) difference set: k(k-1) = λ(v-1). So λ = k(k-1)/(v-1).

For v = n, k = n/2: λ = (n/2)(n/2-1)/(n-1) = n(n-2)/(4(n-1)) ≈ n/4 - 1/2.

For this to be an integer, we need specific n values. But we don't need exact difference sets - we just need "approximately" uniform differences.

Known constructions: If n-1 is a prime power, there exist Singer difference sets with v = (q²+q+1), k = (q+1), λ = 1. These have very specific sizes.

For our purposes, a simpler approach: use a "twisted" pattern. For example:
- For n = 2m, place ones at positions {i² mod (2m) : i = 0, ..., 2m-1} ∩ [0, 2m-1]... this doesn't give exactly m ones in general.

                c5_best = c5_new
                h_best = h.copy()
        else:
            lr *= lr_decay
            stall_count += 1
            if stall_count > stall_threshold:
                idx = rng.randint(len(h))
                h[idx] = np.clip(h[idx] + rng.uniform(-0.15, 0.15), 0.0, 1.0)
                h = project_feasible(h, S)
                c5_cur = compute_c5(h)
                v = np.zeros_like(h)
                lr = lr_init * 0.5
                stall_count = 0

        if it % 1000 == 999:
            alpha = min(alpha * 1.8, 2000.0)

    return h_best, c5_best

def simulated_annealing(h0, S, n, rng, n_iters, T0, Tend, binary_alpha=0.2):
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    T = T0
    cool = (Tend / T0) ** (1.0 / max(1, n_iters))
    corr = compute_corr(h)
    w = smooth_max_weights(corr, 50.0)
    grad = grad_smooth_max(h, w)
    grad_centered = grad - np.mean(grad)

    for it in range(n_iters):
        if it % 200 == 0:
            corr = compute_corr(h)
            w = smooth_max_weights(corr, 50.0)
            grad = grad_smooth_max(h, w)
            grad_centered = grad - np.mean(grad)

        r = rng.random()
        if r < 0.35:
            idx = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.06, 0.06)
            h_new = h.copy()
            h_new[idx[0]] = np.clip(h_new[idx[0]] + delta, 0.0, 1.0)
            h_new[idx[1]] = np.clip(h_new[idx[1]] - delta, 0.0, 1.0)
            h_new = project_feasible(h_new, S)
        elif r < 0.55:
            idx = rng.randint(n)
            delta = rng.uniform(-0.08, 0.08)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
            h_new = project_feasible(h_new, S)
        elif r < 0.75:
            idx = rng.randint(n)
            step = np.sign(-grad_centered[idx]) * rng.uniform(0.02, 0.08)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + step, 0.0, 1.0)
            h_new = project_feasible(h_new, S)
        elif r < 0.90:
            idx = rng.randint(n)
            h_new = h.copy()
            if h_new[idx] > 0.3:
                h_new[idx] = max(0.0, h_new[idx] - 0.3)
            else:
                h_new[idx] = min(1.0, h_new[idx] + 0.3)
            h_new = project_feasible(h_new, S)
        else:
            idx = rng.choice(n, size=3, replace=False)
            h_new = h.copy()
            d1 = rng.uniform(-0.04, 0.04)
            d2 = rng.uniform(-0.04, 0.04)
            h_new[idx[0]] = np.clip(h_new[idx[0]] + d1, 0.0, 1.0)
            h_new[idx[1]] = np.clip(h_new[idx[1]] - d1, 0.0, 1.0)
            h_new[idx[2]] = np.clip(h_new[idx[2]] + d2, 0.0, 1.0)
            h_new = project_feasible(h_new, S)

        c5_new = compute_c5(h_new)
        accept = c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10))
        if accept:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()

        T = max(T * cool, Tend)

    return h_best, c5_best

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new / 2.0)

def binary_starts(n, rng):
    starts = []
    S = n // 2
    for trial in range(4):
        h = np.zeros(n)
        indices = rng.choice(n, size=S, replace=False)
        h[indices] = 1.0
        starts.append(h)
    for trial in range(3):
        base_positions = np.arange(n) * S // n
        base_positions = np.unique(base_positions[:S])
        if len(base_positions) < S:
            available = np.setdiff1d(np.arange(n), base_positions)
            extra = rng.choice(available, S - len(base_positions), replace=False)
            base_positions = np.sort(np.concatenate([base_positions, extra]))
        jitter = rng.randint(-1, 2, size=S)
        positions = np.clip(base_positions + jitter, 0, n - 1)
        positions = np.unique(positions)
        h = np.zeros(n)
        h[positions[:S]] = 1.0
        if len(positions) < S:
            available = np.setdiff1d(np.arange(n), positions)
            h[rng.choice(available, S - len(positions), replace=False)] = 1.0
        starts.append(h)
    return starts

def fourier_start(n, rng):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for mode in range(1, min(12, n // 3)):
        freq = mode * rng.randint(1, 3)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.15) / mode
        h += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h, n / 2.0)

def weighted_binary_round(h, S):
    n = len(h)
    h_sorted_idx = np.argsort(h)[::-1]
    h_new = np.zeros(n)
    h_new[h_sorted_idx[:S]] = 1.0
    if h_new.sum() != S:
        diff = S - h_new.sum()
        remaining = np.argsort(h)[::-1][S:]
        if diff > 0 and len(remaining) >= diff:
            h_new[remaining[:diff]] = 1.0
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=float)
            n_p = len(parent)
            S_p = n_p / 2.0
            if np.abs(S_p - round(S_p)) < 1e-6:
                parent = project_feasible(parent, S_p)
                best_h = parent.copy()
                best_c5 = compute_c5(best_h)
                best_n = n_p
    except:
        pass

    if best_h is None:
        n_init = 50
        best_n = n_init
        best_h = project_feasible(np.full(n_init, 0.5), n_init / 2.0)
        best_c5 = compute_c5(best_h)

    c5_target = 0.3808
    n_values = sorted(set([best_n] + [v for v in [176, 178, 180, 182, 184, 186, 188, 190, 192, 194, 196, 200, 210, 220] if v % 2 == 0]))

    for n in n_values:
        if time_left() < 40:
            break
        S = n / 2.0

        starts = []
        if best_n == n:
            starts.append(best_h.copy())
        elif best_h is not None:
            starts.append(interpolate_h(best_h, n))
        starts.append(fourier_start(n, rng))
        starts.extend(binary_starts(n, rng)[:2])
        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_feasible(hr, S))

        for h_start in starts:
            if time_left() < 25:
                break
            h_start = project_feasible(h_start, S)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            if time_left() > 20:
                h_pgd, c5_pgd = pgd_optimize(h_start, S, alpha=15.0, n_iter=min(8000, int(time_left() * 3)),
                                             lr_init=0.015, lr_decay=0.95, rng=rng, momentum=0.85)
                if c5_pgd < best_c5:
                    best_c5 = c5_pgd
                    best_h = h_pgd.copy()
                    best_n = n
                h_start = h_pgd

            if time_left() > 30:
                h_sa, c5_sa = simulated_annealing(h_start, S, n, rng, min(5000, int(time_left() * 2)), 0.04, 0.001)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n
                h_start = h_sa

            if time_left() > 20:
                h_pgd2, c5_pgd2 = pgd_optimize(h_start, S, alpha=50.0, n_iter=min(5000, int(time_left() * 2)),
                                               lr_init=0.008, lr_decay=0.95, rng=rng, momentum=0.8)
                if c5_pgd2 < best_c5:
                    best_c5 = c5_pgd2
                    best_h = h_pgd2.copy()
                    best_n = n

        if best_c5 < c5_target:
            break

    if best_h is not None and time_left() > 40:
        n_f = best_n
        S_f = n_f / 2.0

        for alpha in [20, 60, 200, 600, 1500]:
            if time_left() < 25:
                break
            max_iter = min(15000, int(time_left() * 8))
            h_opt, c5_opt = pgd_optimize(best_h, S_f, alpha=alpha, n_iter=max_iter,
                                         lr_init=0.01, lr_decay=0.96, rng=rng, momentum=0.9)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        if time_left() > 30:
            h_sa, c5_sa = simulated_annealing(best_h, S_f, n_f, rng,
                                              min(15000, int(time_left() * 6)),
                                              0.025, 0.0003)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

        if time_left() > 20:
            h_fin, c5_fin = pgd_optimize(best_h, S_f, alpha=2000, n_iter=min(10000, int(time_left() * 10)),
                                         lr_init=0.003, lr_decay=0.97, rng=rng, momentum=0.95)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

        if time_left() > 15:
            h_bin = weighted_binary_round(best_h, int(S_f))
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin

            h_pgd3, c5_pgd3 = pgd_optimize(h_bin, S_f, alpha=50, n_iter=min(5000, int(time_left() * 8)),
                                           lr_init=0.01, lr_decay=0.95, rng=rng, momentum=0.85)
            if c5_pgd3 < best_c5:
                best_c5 = c5_pgd3
                best_h = h_pgd3.copy()

    if best_h is not None and time_left() > 10:
        n_f = best_n
        S_f = n_f / 2.0
        for _ in range(3):
            if time_left() < 10:
                break
            h_pol, c5_pol = pgd_optimize(best_h, S_f, alpha=10000, n_iter=5000,
                                         lr_init=0.001, lr_decay=0.98, rng=rng, momentum=0.95)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    n_final = len(best_h)
    S_final = n_final / 2.0
    best_h = project_feasible(best_h, S_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
