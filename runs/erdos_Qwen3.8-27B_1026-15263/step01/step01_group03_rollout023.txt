from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx

def focused_obj(h, n, tau, k_indices):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    ac = corr[k_indices]
    v = ac / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-13:

    return result

    return np.clip(f(xn), 0, 1)

            noise = 0.004 * (attempt + 1)
            h0 = project_feasible(h_work + rng.randn(n_p) * noise, n_p)
            h_new = slsqp_step(h0, n_p, tau, 200, 1e-13, focus_ks=active)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_work:
                c5_work = c5_new
                h_work = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()
        # Also try full (unfocused) at this tau
        h_new = slsqp_step(h_work, n_p, tau, 150, 1e-13)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_work:
            c5_work = c5_new
            h_work = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_work.copy()

    # Phase 3: Fine tau annealing
    fine_taus = [0.007, 0.006, 0.005, 0.0045, 0.004, 0.0035, 0.003, 0.0025, 0.002]
    for tau in fine_taus:
        if not budget_f(0.55):
            break
        active = get_active_ks(h_work, n_p, 0.2)
        for attempt in range(3):
            if not budget_f(0.55):
                break
            noise = 0.002 * (attempt + 1)
            h0 = project_feasible(h_work + rng.randn(n_p) * noise, n_p)
            h_new = slsqp_step(h0, n_p, tau, 250, 1e-14, focus_ks=active)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_work:
                c5_work = c5_new
                h_work = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()

    # Phase 4: Ultra-fine with very small tau
    ultra_taus = [0.0018, 0.0015, 0.0012, 0.001, 0.0008, 0.0006]
    for tau in ultra_taus:
        if not budget_f(0.7):
            break
        active = get_active_ks(h_work, n_p, 0.15)
        for attempt in range(4):
            if not budget_f(0.7):
                break
            noise = 0.001 * (attempt + 1)
            h0 = project_feasible(h_work + rng.randn(n_p) * noise, n_p)
            h_new = slsqp_step(h0, n_p, tau, 350, 1e-15, focus_ks=active)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_work:
                c5_work = c5_new
                h_work = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()

    # Phase 5: Coordinate descent on best
    if budget_f(0.8):
        coord_time = min(60.0, budget_s * 0.1)
        h_cd, c5_cd = coord_refine(best_h, len(best_h), coord_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 6: Try nearby grid sizes
    for n_try in [n_p - 2, n_p + 2, n_p + 4, n_p - 4]:
        if not budget_f(0.85):
            break
        if n_try < 30:
            continue
        h0 = project_feasible(interp_to_n(parent_h, n_try), n_try)
        for tau in [0.03, 0.015, 0.008, 0.004, 0.002]:
            if not budget_f(0.88):
                break
            active = get_active_ks(h0, n_try, 0.25)
            h_new = slsqp_step(h0, n_try, tau, 200, 1e-13, focus_ks=active)
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h0):
                h0 = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h0.copy()

    # Phase 7: Final ultra-fine polish on the overall best
    if best_h is not None and budget_f(0.92):
        n_best = len(best_h)
        h_pol = best_h.copy()
        for tau in [0.001, 0.0007, 0.0005, 0.0004]:
            if not budget_f(0.95):
                break
            active = get_active_ks(h_pol, n_best, 0.15)
            for attempt in range(3):
                if not budget_f(0.95):
                    break
                h0 = project_feasible(h_pol + rng.randn(n_best) * 0.001, n_best)
                h_new = slsqp_step(h0, n_best, tau, 400, 1e-15, focus_ks=active)
                c5_new = evaluate_c5(h_new)
                if c5_new < evaluate_c5(h_pol):
                    h_pol = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_pol.copy()

        if budget_f(0.98):
            h_cd2, c5_cd2 = coord_refine(best_h, len(best_h), 30.0)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    if best_h is None:
        best_h = project_feasible(parent_h.copy(), n_p)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
