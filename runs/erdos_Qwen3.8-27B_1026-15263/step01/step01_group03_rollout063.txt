import cvxpy as cp

def lp_linearize(h0, n):
    dx = 2.0 / n
    g0 = 1.0 - h0
    corr0 = np.correlate(h0, g0, mode="full")
    
    # Compute gradient of each f(k) w.r.t. h
    # grad[k, j] = (1-h0[j+k-(n-1)]) - h0[j-k+(n-1)] (with range checks)
    nfull = 2*n - 1
    grad = np.zeros((nfull, n))
    for k in range(nfull):
        for j in range(n):
            term = 0.0
            idx1 = j + k - (n-1)
            if 0 <= idx1 < n:
                term += (1.0 - h0[idx1])
            idx2 = j - k + (n-1)
            if 0 <= idx2 < n:
                term -= h0[idx2]
            grad[k, j] = term
    
    # LP:
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    f_lin = corr0 * dx + dx * (grad @ (h_var - h0))
    
    constraints = [
        f_lin <= t_var,
        cp.sum(h_var) == n * 0.5,
        h_var >= 0,
        h_var <= 1
    ]
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    solver = cp.GLOP if 'GLOP' in [s.name for s in cp.installed_solvers()] else cp.CBC
    prob.solve(solver=solver)
    
    return h_var.value

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    v = corr / tau
    v = v - v.max()
    return tau * (corr.max() / tau + np.log(np.sum(np.exp(v))))

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, tau) * dx

def constraint_fn(h, n):
    return h.sum() - n * 0.5

def proj_feas(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - r.sum()
    if abs(d) > 1e-14:
        oi = np.argsort(np.abs(r - 0.5))
        for i in oi:
            if d > 0:
                a = min(d, 1.0 - r[i])
            else:
                a = min(-d, r[i])
            if a <= 0:
                continue
            r[i] += (1.0 if d > 0 else -1.0) * a
            d -= (1.0 if d > 0 else -1.0) * a
            if abs(d) < 1e-14:
                break
    return r

def slsqp_refine(h0, n, tau_list, maxiter):
    current = proj_feas(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn, 'args': (n,)}]
    for tau in tau_list:
        try:
            res = minimize(obj_smooth, current, method='SLSQP',
                           args=(n, tau), bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            current = proj_feas(res.x, n)
        except Exception:
            pass
    return current

def sa_search(h_init, n, n_iter, T_init, T_final, seed_local, deadline):
    rng_sa = np.random.RandomState(seed_local)
    current = h_init.copy()
    cur_c5 = evaluate_c5(current)
    best_local = current.copy()
    best_c5_local = cur_c5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        ratio = it / max(n_iter - 1, 1)
        T = T_init * (T_final / T_init) ** ratio
        j, k = rng_sa.choice(n, 2, replace=False)
        eps = rng_sa.uniform(-0.04, 0.04)
        nj = current[j] + eps
        nk = current[k] - eps
        if nj < 0 or nj > 1 or nk < 0 or nk > 1:
            continue
        trial = current.copy()
        trial[j] = nj
        trial[k] = nk
        nc5 = evaluate_c5(trial)
        delta = nc5 - cur_c5
        if delta < 0 or rng_sa.random() < np.exp(-delta / max(T, 1e-12)):
            current = trial
            cur_c5 = nc5
            if cur_c5 < best_c5_local:
                best_c5_local = cur_c5
                best_local = current.copy()
    return best_local, best_c5_local

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    v = corr / tau
    v = v - v.max()
    return tau * (corr.max() / tau + np.log(np.sum(np.exp(v))))

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, tau) * dx

def constraint_fn(h, n):
    return h.sum() - n * 0.5

def proj_feas(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - r.sum()
    if abs(d) > 1e-14:
        oi = np.argsort(np.abs(r - 0.5))
        for i in oi:
            if d > 0:
                a = min(d, 1.0 - r[i])
            else:
                a = min(-d, r[i])
            if a <= 0:
                continue
            r[i] += (1.0 if d > 0 else -1.0) * a
            d -= (1.0 if d > 0 else -1.0) * a
            if abs(d) < 1e-14:
                break
    return r

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            best_h = proj_feas(np.full(59, 0.5), 59)
            best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
