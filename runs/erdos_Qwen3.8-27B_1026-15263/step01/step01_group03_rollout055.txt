    lo = -3.0
    hi = 3.0

        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:

    diff = target - np.sum(result)

                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:

    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")

    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max(corr, tau) * dx

    return np.sum(h) - n / 2.0

    lo = -3.0
    hi = 3.0

        proj = np.clip(h + mid, 0.0, 1.0)

                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub

                break

def nlp_eq(x, n):
    h = x[:n]
    return np.sum(h) - n / 2.0

def nlp_full_ineq(x, n):
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    return t - corr

def generate_diverse_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)

    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
            starts.append(np.clip(parent_h + rng.randn(n) * 0.03, 0, 1))
            starts.append(np.clip(parent_h + rng.randn(n) * 0.08, 0, 1))
        else:
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            f = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate')
            starts.append(np.clip(f(x), 0, 1))
            f2 = interp1d(xp, np.clip(parent_h + rng.randn(len(parent_h)) * 0.05, 0, 1),
                          kind='cubic', fill_value='extrapolate')
            starts.append(np.clip(f2(x), 0, 1))

    for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.45)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + ph), 0, 1))

    for _ in range(4):
        starts.append(rng.rand(n))

    starts.append(np.linspace(0.05, 0.95, n))
    starts.append(np.linspace(0.95, 0.05, n))

    for freq in [1.0, 2.0, 3.0]:
        ph = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        amp1 = 0.25
        amp2 = 0.15
        starts.append(np.clip(0.5 + amp1 * np.sin(2 * np.pi * freq * x + ph) +
                              amp2 * np.sin(2 * np.pi * (freq + 1) * x + ph2), 0, 1))

    saw = np.mod(ar / n * 3.0, 2.0)
    saw = np.where(saw < 1.0, saw, 2.0 - saw)
    starts.append(np.clip(0.5 + 0.3 * (saw - 0.5), 0, 1))

    h_bin = (rng.rand(n) > 0.5).astype(float)
    starts.append(np.clip(h_bin * 0.7 + 0.15, 0, 1))

    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_list = []
    if parent_h is not None:
        n_list.append(len(parent_h))
    for nn in [41, 43, 45, 47, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69, 71, 73, 75, 77]:
        if nn not in n_list:
            n_list.append(nn)

    n_list_sorted = sorted(n_list, key=lambda x: (0 if parent_h is not None and x == len(parent_h) else 1, abs(x - 55)))

    time_limit_start = budget_s * 0.80

    for n in n_list_sorted:
        if time.time() - t0 > time_limit_start:
            break

        starts = generate_diverse_starts(n, parent_h, rng)
        bounds = [(0.0, 1.0)] * n
        cons_eq = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]

        for h0 in starts:
            if time.time() - t0 > time_limit_start:
                break

            h0 = project_feasible(h0, n)
            current_h = h0.copy()

            taus = [0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003]
            for tau in taus:
                if time.time() - t0 > time_limit_start:
                    break
                try:
                    res = minimize(obj_smooth, current_h, method='SLSQP',
                                   args=(n, tau), bounds=bounds,
                                   constraints=cons_eq,
                                   options={'maxiter': 100, 'ftol': 1e-11})
                    current_h = project_feasible(res.x, n)
                    c5 = evaluate_c5(current_h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = current_h.copy()
                except Exception:
                    pass

            if time.time() - t0 > time_limit_start:
                break

            if time.time() - t0 < time_limit_start:
                try:
                    t_init = max(evaluate_c5(current_h) * 1.4, 0.05)
                    x_init = np.concatenate([current_h, [t_init]])
                    n_bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                    k = min(25, 2 * n - 1)
                    cons_ineq = [{'type': 'ineq', 'fun': nlp_topk_ineq, 'args': (n, k)}]
                    cons_all = cons_ineq + [{'type': 'eq', 'fun': nlp_eq, 'args': (n,)}]
                    res = minimize(nlp_topk_objective, x_init, method='SLSQP',
                                   args=(n, k), bounds=n_bounds,
                                   constraints=cons_all,
                                   options={'maxiter': 200, 'ftol': 1e-13})
                    h_nlp = project_feasible(res.x[:n], n)
                    c5 = evaluate_c5(h_nlp)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_nlp.copy()
                except Exception:
                    pass

            if time.time() - t0 > time_limit_start:
                break

            for attempt in range(6):
                if time.time() - t0 > time_limit_start:
                    break
                scale = 0.04 * (0.6 ** attempt)
                perturb = current_h + rng.randn(n) * scale
                perturb = project_feasible(perturb, n)
                for tau in [0.005, 0.002]:
                    if time.time() - t0 > time_limit_start:
                        break
                    try:
                        res = minimize(obj_smooth, perturb, method='SLSQP',
                                       args=(n, tau), bounds=bounds,
                                       constraints=cons_eq,
                                       options={'maxiter': 80, 'ftol': 1e-12})
                        h_new = project_feasible(res.x, n)
                        c5 = evaluate_c5(h_new)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_new.copy()
                        if c5 < evaluate_c5(current_h):
                            current_h = h_new
                    except Exception:
                        pass

    if best_h is not None and time.time() - t0 < budget_s * 0.93:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons_eq = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]

        for tau in [0.002, 0.001, 0.0005]:
            if time.time() - t0 > budget_s * 0.96:
                break
            for attempt in range(8):
                if time.time() - t0 > budget_s * 0.96:
                    break
                scale = 0.015 * (0.5 ** attempt)
                perturb = best_h + rng.randn(n) * scale
                perturb = project_feasible(perturb, n)
                try:
                    res = minimize(obj_smooth, perturb, method='SLSQP',
                                   args=(n, tau), bounds=bounds,
                                   constraints=cons_eq,
                                   options={'maxiter': 200, 'ftol': 1e-14})
                    h_pol = project_feasible(res.x, n)
                    c5 = evaluate_c5(h_pol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_pol.copy()
                except Exception:
                    pass

        if time.time() - t0 < budget_s * 0.96:
            try:
                t_init = max(evaluate_c5(best_h) * 1.15, 0.02)
                x_init = np.concatenate([best_h, [t_init]])
                n_bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                k = min(30, 2 * n - 1)
                cons_ineq = [{'type': 'ineq', 'fun': nlp_topk_ineq, 'args': (n, k)}]
                cons_all = cons_ineq + [{'type': 'eq', 'fun': nlp_eq, 'args': (n,)}]
                res = minimize(nlp_topk_objective, x_init, method='SLSQP',
                               args=(n, k), bounds=n_bounds,
                               constraints=cons_all,
                               options={'maxiter': 400, 'ftol': 1e-15})
                h_nlp = project_feasible(res.x[:n], n)
                c5 = evaluate_c5(h_nlp)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_nlp.copy()
            except Exception:
                pass

        if time.time() - t0 < budget_s * 0.97:
            try:
                t_init = max(evaluate_c5(best_h) * 1.1, 0.01)
                x_init = np.concatenate([best_h, [t_init]])
                n_bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                cons_ineq_full = [{'type': 'ineq', 'fun': nlp_full_ineq, 'args': (n,)}]
                cons_all_full = cons_ineq_full + [{'type': 'eq', 'fun': nlp_eq, 'args': (n,)}]
                res = minimize(nlp_topk_objective, x_init, method='SLSQP',
                               args=(n, 2 * n - 1), bounds=n_bounds,
                               constraints=cons_all_full,
                               options={'maxiter': 300, 'ftol': 1e-15})
                h_nlp = project_feasible(res.x[:n], n)
                c5 = evaluate_c5(h_nlp)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_nlp.copy()
            except Exception:
                pass

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 47
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
