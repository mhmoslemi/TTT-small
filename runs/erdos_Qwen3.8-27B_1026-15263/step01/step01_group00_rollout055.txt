import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_weights(corr, alpha):
    """Compute smooth max and softmax weights"""
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    return m + np.log(np.sum(exp_vals)) / alpha, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth_max(c(h), alpha)"""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-16:
            continue
        if lag >= 0:
            if lag < n:
                grad[lag:] += wk * (1.0 - h[:n - lag])
            if lag > 0:
                grad[:n - lag] -= wk * h[lag:]
        else:

            if lag_abs < n:
                grad[:n - lag_abs] += wk * (1.0 - h[lag_abs:])
            if lag_abs > 0:
                grad[lag_abs:] -= wk * h[:n - lag_abs]

    n = len(h)

    best_h = h.copy()
    best_val = compute_c5(h)

        if time.time() - t_start > time_limit:
            break
        val, grad = compute_gradient(h, alpha)

            h_new = h - step_size * grad
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_val or c5_new < val:
                h = h_new
                if c5_new < best_val:
                    best_val = c5_new
                    best_h = h.copy()
                accepted = True
                break
        
        if not accepted:

            h = project_feasible(h + 0.01 * np.random.randn(n), target_sum)

    v = 1.0 - h
