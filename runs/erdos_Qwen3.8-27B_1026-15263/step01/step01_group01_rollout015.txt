```python
def project_feasible_fast(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    # Use bisection on the dual variable
    # min ||h - h0||^2 s.t. 0 <= h <= 1, sum(h) = n/2
    # KKT: h_i = clip(h0_i - lambda, 0, 1)
    lo, hi = -1.0, 2.0
    target = n / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
            if mask.sum() > 0:
                add = diff / mask.sum()
                h_proj[mask] = np.minimum(h_proj[mask] + add, 1.0)
        else:
            mask = h_proj > 0.0
            if mask.sum() > 0:
                sub = -diff / mask.sum()
                h_proj[mask] = np.maximum(h_proj[mask] - sub, 0.0)
    return h_proj
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def compute_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for m_idx in range(ncorr):
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            grad[js:je] += wm * (1.0 - h[js+k:je+k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2-k:je2-k]
    grad *= dx
    obj = (m + np.log(exps.sum()) / beta) * dx
    return obj, grad

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    # Fine adjustment
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = hp < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt > 0:
                hp[mask] = np.minimum(hp[mask] + diff / cnt, 1.0)
        else:
            mask = hp > 1e-15
            cnt = mask.sum()
            if cnt > 0:
                hp[mask] = np.maximum(hp[mask] + diff / cnt, 0.0)
    # Final clip to ensure feasibility
    np.clip(hp, 0.0, 1.0, out=hp)
    return hp

def fista_optimize(h0, n, beta, max_iter, lr=None):
    """FISTA with projection"""
    x = project_feasible(h0.copy(), n).copy()
    x_prev = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr is None:
        lr = 0.1
    
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)  # keep in box for gradient eval
        
        obj_y, grad_y = compute_obj_grad(y, n, beta)
        
        # Line search / backtracking
        step = lr
        for _ in range(30):
            x_new = project_feasible(y - step * grad_y, n)
            break  # use fixed step for simplicity
        
        x_new = project_feasible(y - step * grad_y, n)
        obj_x = compute_c5(x_new)
        
        if obj_x < best_obj:
            best_obj = obj_x
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
        t = t_new
        
        # Check convergence
        if it > 10 and abs(obj_x - obj_y) < 1e-12:
            break
    
    return best_x, best_obj
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def compute_obj_grad(h, n, beta):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for m_idx in range(ncorr):
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if wm < 1e-13:
            continue
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            grad[js:je] += wm * (1.0 - h[js+k:je+k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2-k:je2-k]
    grad *= dx
    obj = (m + np.log(exps.sum()) / beta) * dx
    return obj, grad

            lo = mid
        else:
            hi = mid

    x = project_feasible(h0.copy(), n)
    x_prev = x.copy()
    t = 1.0
    best_x = x.copy()

    lr = 0.5  # will be adjusted

        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)

        # Backtracking line search
        step = lr
        x_new = project_feasible(y - step * grad_y, n)

        x = x_new
        t = t_new
        lr = max(0.1, lr * 0.999)

            best_x = x.copy()
    
    return best_x, best_c5

def compute_V(h, n):
    """Compute V[m, j] = d(corr[m])/d(h[j]) for all m, j"""
    ncorr = 2 * n - 1
    V = np.zeros((ncorr, n))
    k_arr = np.arange(ncorr) - (n - 1)
    for m_idx in range(ncorr):
        k = k_arr[m_idx]
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            V[m_idx, js:je] += (1.0 - h[js+k:je+k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            V[m_idx, js2:je2] -= h[js2-k:je2-k]
    return V
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def compute_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for m_idx in range(ncorr):
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            grad[js:je] += wm * (1.0 - h[js + k:je + k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2 - k:je2 - k]
    grad *= dx
    obj = (m + np.log(exps.sum()) / beta) * dx
    return obj, grad


def project_feasible(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = hp < 1.0 - 1e-15
            cnt = int(mask.sum())
            if cnt > 0:
                hp[mask] = np.minimum(hp[mask] + diff / cnt, 1.0)
        else:
            mask = hp > 1e-15
            cnt = int(mask.sum())
            if cnt > 0:
                hp[mask] = np.maximum(hp[mask] + diff / cnt, 0.0)
    np.clip(hp, 0.0, 1.0, out=hp)
    return hp


def fista_optimize(h0, n, beta, n_iter):
    x = project_feasible(h0.copy(), n)
    x_prev = x.copy()
    t = 1.0
    best_x = x.copy()
    best_c5 = compute_c5(x)
    lr = 0.8

    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        np.clip(y, 0.0, 1.0, out=y)

        obj_y, grad_y = compute_obj_grad(y, n, beta)

        step = lr / (1.0 + 0.01 * np.log1p(it))
        x_new = project_feasible(y - step * grad_y, n)

        x_prev = x
        x = x_new
        t = t_new

        c5_curr = compute_c5(x)
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_x = x.copy()

    return best_x, best_c5


def compute_V(h, n):
    ncorr = 2 * n - 1
    V = np.zeros((ncorr, n))
    for m_idx in range(ncorr):
        k = m_idx - (n - 1)
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            V[m_idx, js:je] += (1.0 - h[js + k:je + k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            V[m_idx, js2:je2] -= h[js2 - k:je2 - k]
    return V


def binary_local_search(h_bin, n, max_sweeps, budget_end, sa_temp):
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_c5 = float(np.max(corr) * dx)
    rng_local = np.random.RandomState(int(current_c5 * 1e6))

    for sweep in range(max_sweeps):
        if budget_end is not None and time.time() > budget_end:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        V = compute_V(h, n)
        V_ones = V[:, ones]
        V_zeros = V[:, zeros]
        new_corrs = corr[None, None, :] - V_ones[:, :, None] + V_zeros[:, None, :]
        new_c5s = new_corrs.max(axis=0)

        best_flat = int(np.argmin(new_c5s))
        best_new_c5 = float(new_c5s.ravel()[best_flat] * dx)

        if best_new_c5 < current_c5 - 1e-13:
            best_idx = np.unravel_index(best_flat, new_c5s.shape)
            i = ones[best_idx[0]]
            j = zeros[best_idx[1]]
            h[i] = 0.0
            h[j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode="full")
            current_c5 = best_new_c5
        else:
            if sa_temp > 0.001 and sweep > max_sweeps // 3:
                n_kicks = min(3, len(ones) * len(zeros))
                for _ in range(n_kicks):
                    ri = rng_local.randint(len(ones))
                    rj = rng_local.randint(len(zeros))
                    i = ones[ri]
                    j = zeros[rj]
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    c5_kick = float(np.max(corr) * dx)
                    if c5_kick < current_c5 or rng_local.random() < np.exp(-(c5_kick - current_c5) / sa_temp):
                        current_c5 = c5_kick
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = np.correlate(h, 1.0 - h, mode="full")
                sa_temp *= 0.9
            else:
                break

    return h, current_c5


def fractional_refine(h_base, n, budget_end):
    dx = 2.0 / n
    best_h = h_base.astype(float).copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(h_base > 0.5)[0]
    zeros = np.where(h_base < 0.5)[0]

    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c5

    h_f = best_h.copy()
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    a_vals = np.linspace(0.05, 0.95, 19)

    n_check_ones = min(15, len(ones))
    n_check_zeros = min(15, len(zeros))
    ones_sub = ones[:n_check_ones]
    zeros_sub = zeros[:n_check_zeros]

    for i in ones_sub:
        for j in zeros_sub:
            if budget_end is not None and time.time() > budget_end:
                return best_h, best_c5
            for a in a_vals:
                h_try = h_f.copy()
                h_try[i] = a
                h_try[j] = 1.0 - a
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_f = h_try.copy()
                    corr = np.correlate(h_f, 1.0 - h_f, mode="full")

    return best_h, best_c5


def make_periodic_binary(n, period, pattern, rng):
    h = np.zeros(n)
    for p in range(period):
        if pattern[p % len(pattern)]:
            h[p::period] = 1.0
    n_ones = int(np.sum(h > 0.5))
    target = n // 2
    if n_ones > target:
        ones = np.where(h > 0.5)[0]
        flip = ones[rng.choice(len(ones), n_ones - target, replace=False)]
        h[flip] = 0.0
    else:
        zeros = np.where(h < 0.5)[0]
        need = target - n_ones
        if need > 0 and len(zeros) > 0:
            flip = zeros[rng.choice(len(zeros), min(need, len(zeros)), replace=False)]
            h[flip] = 1.0
    return h


def adjust_binary_count(h, n):
    n_ones = int(np.sum(h > 0.5))
    target = n // 2
    if n_ones > target:
        ones = np.where(h > 0.5)[0]
        flip = ones[np.argsort(h[ones])[:n_ones - target]]
        h[flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h < 0.5)[0]
        need = target - n_ones
        flip = zeros[np.argsort(-h[zeros])[:min(need, len(zeros))]]
        h[flip] = 1.0
    return h
```
