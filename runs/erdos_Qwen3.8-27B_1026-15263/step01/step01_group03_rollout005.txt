Actually, here's another idea: instead of searching over a grid of t values, I can observe that for a fixed pair (i,j) and fixed other coordinates, the objective as a function of t=h[i] (with h[j]=pair_sum-t) is:

f(t) = max_k g_k(t)

where g_k(t) is at most quadratic in t (since corr involves products h[i]*h[j]). The maximum of quadratics is a convex piecewise-quadratic function. I can find its minimum by:
1. Evaluating at a few points
2. Doing a ternary/golden-section search

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-16:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def obj_func(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max(corr, tau) * dx

def constraint_func(h, n):
    return np.sum(h) - n / 2.0

def coord_descent(h, n, max_iter=200, search_range=0.2, n_search=31):
    h = h.copy()
    current_obj = evaluate_c5(h)
    for it in range(max_iter):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            h_i_old = h[i]
            h_j_old = h[j]
            pair_sum = h_i_old + h_j_old
            t_lo = max(0.0, pair_sum - 1.0)
            t_hi = min(1.0, pair_sum)
            if t_hi - t_lo < 1e-14:
                continue
            center = h_i_old
            r = search_range
            s_lo = max(t_lo, center - r)
            s_hi = min(t_hi, center + r)
            if s_hi - s_lo < 1e-12:
                continue
            ts = np.linspace(s_lo, s_hi, n_search)
            best_t = h_i_old
            best_obj = current_obj
            for t in ts:
                if abs(t - h_i_old) < 1e-13:
                    continue
                tj = pair_sum - t
                if tj < -1e-12 or tj > 1.0 + 1e-12:
                    continue
                h[i] = t
                h[j] = tj
                obj = evaluate_c5(h)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_t = t
            if best_t != h_i_old:
                h[i] = best_t
                h[j] = pair_sum - best_t
                current_obj = best_obj
                improved = True
        if not improved:
            break
    return h, current_obj

def coord_descent_block(h, n, max_iter=100, search_range=0.15, n_search=21):

            best_obj = current_obj

    return h, current_obj

def slsqp_optimize(h0, n, tau, maxiter=80, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP',
                       args=(n, tau), bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def nelder_mead_opt(h0, n, maxiter=300):
    def obj_p(h):
        pen = 0.0
        if abs(np.sum(h) - n / 2.0) > 1e-6:
            pen += 100.0 * (np.sum(h) - n / 2.0) ** 2
        pen += 1000.0 * np.sum(np.maximum(0, h - 1.0) ** 2)
        pen += 1000.0 * np.sum(np.maximum(0, -h) ** 2)
        return evaluate_c5(np.clip(h, 0, 1)) + pen
    try:
        res = minimize(obj_p, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h0
