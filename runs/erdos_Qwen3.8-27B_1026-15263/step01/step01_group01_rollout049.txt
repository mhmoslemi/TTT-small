def slp_step(h0, n, delta, solver='GLPK'):
    # Linearize corr around h0
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    dx = 2.0 / n
    
    # Compute gradients of each corr[k] with respect to h
    # dc[m]/dh[p] = (1-h0[p+s_m])*1[valid] - h0[p-s_m]*1[valid]
    # where s_m = m - (n-1)
    
    num_corr = 2*n - 1
    # Build the Jacobian matrix J: J[p, m] = dc_m/dh_p
    J = np.zeros((n, num_corr))
    for s in range(-(n-1), n):
        m = s + n - 1
        ps = max(0, -s)
        pe = min(n, n - s)
        if ps < pe:
            J[ps:pe, m] += (1.0 - h0[ps + s:pe + s])
        ps2 = max(0, s)
        pe2 = min(n, n + s)
        if ps2 < pe2:
            J[ps2:pe2, m] += (-h0[ps2 - s:pe2 - s])
    
    # LP: minimize t
    # subject to:
    #   sum_m J[:,m] * (h - h0) + corr0[m] <= t  for all m
    #   sum(h) = n/2
    #   0 <= h <= 1
    #   ||h - h0||_inf <= delta (optional, for trust region)
    
    # Variables: h (n vars) + t (1 var) = n+1 vars
    import cvxpy as cp
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    # Linearized correlation constraints
    for m in range(num_corr):
        linearized = corr0[m] + J[:, m] @ (h_var - h0)
        constraints.append(linearized * dx <= t_var)
    
    # Trust region (optional)
    # constraints.append(cp.norm(h_var - h0, inf) <= delta)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_feasible(h_new, n)
            return h_new, t_var.value
    except:
        pass
    return None, None

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if res.status == 0 and res.x is not None:
                h_lp = np.clip(res.x[:n], 0.0, 1.0)
                h_lp = project_feasible(h_lp, n)
                c5_lp = compute_c5(h_lp)
                
                alpha_list = [1.0, 0.5, 0.25, 0.125, 0.0625]
                for alpha in alpha_list:
                    h_trial = (1.0 - alpha) * h + alpha * h_lp
                    h_trial = project_feasible(h_trial, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        h = h_trial
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        break
        except Exception:
            pass
    
    return best_h, best_c5

def make_perturbed(h, sigma, rng):
    return project_feasible(h + rng.normal(0.0, sigma, len(h)), len(h))

def make_structured_start(n, rng):
    i = np.arange(n, dtype=float)
    start_type = rng.randint(5)
    if start_type == 0:
        freq = rng.randint(3, 12)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * freq * i / n + phase)
    elif start_type == 1:
        freq1 = rng.randint(2, 8)
        freq2 = rng.randint(9, 20)
        h = 0.5 + 0.15 * np.sin(2 * np.pi * freq1 * i / n) + 0.1 * np.sin(2 * np.pi * freq2 * i / n)
    elif start_type == 2:
        h = 0.5 + rng.normal(0, 0.1, n)
    elif start_type == 3:
        period = rng.randint(4, 15)
        h = np.tile(np.array([0.3, 0.5, 0.7, 0.5][:period // 2 + 1]), n // (period // 2 + 1) + 1)[:n]
        h = h + rng.normal(0, 0.05, n)
    else:
        h = np.ones(n) * 0.5
        n_blocks = rng.randint(5, 20)
        block_size = n // n_blocks
        for b in range(n_blocks):
            val = rng.uniform(0.2, 0.8)
            start = b * block_size
            end = min(start + block_size, n)
            h[start:end] = val
    return project_feasible(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    else:
        best_n = 190
        best_h = np.ones(best_n) * 0.5
    
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B with moderate beta
    if time.time() - t0 < budget_s - 500:
        t_limit = min(150.0, budget_s - 500 - (time.time() - t0))
        h_opt, c5_opt = lbfgsb_optimize(best_h, best_n, beta=80, penalty=500, max_iter=3000, time_limit=t_limit)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: SLP refinement
    if time.time() - t0 < budget_s - 400:
        t_remaining = budget_s - 400 - (time.time() - t0)
        slp_max = int(min(80, t_remaining / 3.0))
        h_slp, c5_slp = slp_step(best_h, best_n, max_iter=slp_max, time_limit=t_remaining)
        if c5_slp < best_c5:
            best_h, best_c5 = h_slp, c5_slp
    
    # Phase 3: L-BFGS-B polish after SLP
    if time.time() - t0 < budget_s - 300:
        t_limit = min(120.0, budget_s - 300 - (time.time() - t0))
        h_opt, c5_opt = lbfgsb_optimize(best_h, best_n, beta=200, penalty=2000, max_iter=3000, time_limit=t_limit)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 4: Multi-start exploration
    no_improve = 0
    while time.time() - t0 < budget_s - 250:
        if no_improve > 25:
            break
        r = rng.random()
        if r < 0.4:
            sigma = 0.08 + 0.04 * rng.random()
            h_start = make_perturbed(best_h, sigma, rng)
        elif r < 0.7:
            h_start = make_structured_start(best_n, rng)
        else:
            sigma = 0.3 + 0.2 * rng.random()
            h_start = make_perturbed(best_h, sigma, rng)
        
        t_limit = min(30.0, budget_s - 250 - (time.time() - t0))
        if t_limit < 5:
            break
        h_opt, c5_opt = lbfgsb_optimize(h_start, best_n, beta=150, penalty=1000, max_iter=1500, time_limit=t_limit)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
    
    # Phase 5: SLP from best found
    if time.time() - t0 < budget_s - 100:
        t_remaining = budget_s - 100 - (time.time() - t0)
        slp_max = int(min(60, t_remaining / 3.0))
        h_slp, c5_slp = slp_step(best_h, best_n, max_iter=slp_max, time_limit=t_remaining)
        if c5_slp < best_c5:
            best_h, best_c5 = h_slp, c5_slp
    
    # Phase 6: Try different n (finer grid)
    for n_try in [285, 380]:
        if time.time() - t0 > budget_s - 80:
            break
        if best_n != n_try:
            h_interp = interpolate_h(best_h, best_n, n_try)
            t_limit = min(40.0, budget_s - 80 - (time.time() - t0))
            if t_limit < 5:
                break
            h_opt, c5_opt = lbfgsb_optimize(h_interp, n_try, beta=100, penalty=1000, max_iter=2000, time_limit=t_limit)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try
    
    # Phase 7: Final high-beta polish
    if time.time() - t0 < budget_s - 20:
        t_limit = min(30.0, budget_s - 20 - (time.time() - t0))
        if t_limit > 5:
            h_opt, c5_opt = lbfgsb_optimize(best_h, best_n, beta=500, penalty=5000, max_iter=5000, time_limit=t_limit)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    
    # Final SLP polish
    if time.time() - t0 < budget_s - 5:
        t_remaining = budget_s - 5 - (time.time() - t0)
        if t_remaining > 5:
            slp_max = int(min(30, t_remaining / 3.0))
            h_slp, c5_slp = slp_step(best_h, best_n, max_iter=slp_max, time_limit=t_remaining)
            if c5_slp < best_c5:
                best_h, best_c5 = h_slp, c5_slp
    
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
