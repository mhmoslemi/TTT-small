```python
def compute_gradient(h, w, dx):
    n = len(h)
    # w has length 2n-1, indexed by s' = 0..2n-2, corresponding to shift s = s'-(n-1)
    # grad[j] = dx * sum_{s'} w[s'] * [(1-h[j+s'-（n-1)])*I[valid] - h[j-(s'-(n-1))]*I[valid]]
    
    # Create shift indices
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    j = np.arange(n)  # shape (n,)
    
    # For each (j, s) pair, check validity
    # j + s in [0, n-1]: j + s >= 0 and j + s <= n-1
    js = j[:, None] + s[None, :]  # shape (n, 2n-1)
    valid1 = (js >= 0) & (js < n)
    
    js2 = j[:, None] - s[None, :]  # shape (n, 2n-1)
    valid2 = (js2 >= 0) & (js2 < n)
    
    # A[j] = sum_s' w[s'] * (1-h[j+s])*valid1
    # B[j] = sum_s' w[s'] * h[j-s]*valid2
    
    # For A: (1-h[j+s]) where valid
    # h[j+s] for valid positions
    h_js = np.zeros_like(js, dtype=float)
    h_js[valid1] = h[js[valid1]]
    A = np.sum(w[None, :] * (1 - h_js) * valid1, axis=1)
    
    # For B: h[j-s] where valid
    h_js2 = np.zeros_like(js2, dtype=float)
    h_js2[valid2] = h[js2[valid2]]
    B = np.sum(w[None, :] * h_js2 * valid2, axis=1)
    
    return dx * (A - B)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_vector(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx


def project_onto_set(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient(h, w, dx):
    n = len(h)
    s = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    idx1 = j[:, None] + s[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j[:, None] - s[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    w_exp = w[None, :]
    h_at_1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    A = np.sum(w_exp * (1.0 - h_at_1) * valid1, axis=1)
    h_at_2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    B = np.sum(w_exp * h_at_2 * valid2, axis=1)
    return dx * (A - B)


def soft_max_objective(h, dx, eps):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_shifted = c - c.max()
    return eps * np.log(np.sum(np.exp(c_shifted / eps))) + c.max()


def soft_max_gradient(h, dx, eps):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_shifted = c - c.max()
    w = np.exp(c_shifted / eps)
    w /= w.sum()
    return compute_gradient(h, w, dx), w


def projected_gradient_phase(h, n, target_sum, dx, max_iters, eps_start=0.5, eps_end=0.002, t_limit=None, t0=None):
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_eps = 30
    eps_schedule = np.geomspace(eps_start, eps_end, n_eps)

    for eps in eps_schedule:
        if t_limit is not None and (time.time() - t0) > t_limit:
            break
        alpha = 1.0 / (3.0 * n)
        momentum = 0.7
        v = np.zeros(n)
        f_curr = soft_max_objective(h, dx, eps)

        for it in range(max_iters // n_eps):
            if t_limit is not None and (time.time() - t0) > t_limit:
                break
            grad, w = soft_max_gradient(h, dx, eps)
            v = momentum * v + (1.0 - momentum) * grad
            alpha_ls = alpha
            improved = False
            for ls in range(30):
                h_try = project_onto_set(h - alpha_ls * v, target_sum)
                f_try = soft_max_objective(h_try, dx, eps)
                if f_try < f_curr - 1e-12:
                    h = h_try
                    f_curr = f_try
                    improved = True
                    break
                alpha_ls *= 0.5
            if not improved:
                break
            c5_curr = compute_c5_from_h(h)
            if c5_curr < best_c5 - 1e-10:
                best_c5 = c5_curr
                best_h = h.copy()
            alpha = max(alpha * 1.05, 1e-6)
        h = best_h.copy()
    return best_h, best_c5


def subgradient_phase(h, n, target_sum, dx, max_iters, t_limit, t0):
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    rng = np.random.default_rng(12345)
    step_scale = 0.5

    for it in range(max_iters):
        if (time.time() - t0) > t_limit:
            break
        c_vec = compute_c5_vector(h)
        c_max = c_max = np.max(c_vec)
        tol = 1e-8
        active = np.where(c_vec >= c_max - tol)[0]
        s_active = active[rng.integers(len(active))]
        s_val = s_active - (n - 1)
        j = np.arange(n)
        grad = np.zeros(n)

def l_bfgs_polish(h, n, target_sum, dx, eps, max_iter=500):
    def obj_and_grad(x):
        h_x = np.clip(x, 1e-8, 1.0 - 1e-8)
        c = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        c_shifted = c - c.max()
        w = np.exp(c_shifted / eps)
        w /= w.sum()
        f = eps * np.log(np.sum(np.exp(c_shifted / eps))) + c.max()
        pen = 500.0 * (np.sum(h_x) - n * 0.5) ** 2
        f_total = f + pen
        grad = compute_gradient(h_x, w, dx)
        grad += 2.0 * 500.0 * (np.sum(h_x) - n * 0.5)
        return f_total, grad

    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_and_grad, x0, method="L-BFGS-B",
                      bounds=bounds, options={"maxiter": max_iter, "ftol": 1e-15, "gtol": 1e-10})
    if result.x is not None:
        h_out = project_onto_set(result.x, n * 0.5)
        return h_out
    return h

    f = interp1d(x_old, h_old, kind="linear", fill_value="extrapolate", bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_onto_set(h_new, n_new / 2.0)

    return best_h, best_c5, len(best_h)
```
