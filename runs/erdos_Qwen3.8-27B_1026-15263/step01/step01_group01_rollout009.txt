            h[np.where(h > 1.0)] = 1.0
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)

            h[np.where(h < 0.0)] = 0.0

            diff -= add

    v = np.zeros(n)
    dx = 2.0 / n
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    
    for t in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = np.max(corr)
        # Active set: top active_frac of values
        sorted_idx = np.argsort(corr)[::-1]
        n_active = max(1, int(len(corr) * active_frac))
        active_set = sorted_idx[:n_active]

        grad = np.zeros(n)
        for p in active_set:

                    grad[k:] += (1.0 - h[:n-k])

            else:

                    grad[:n-q] += (1.0 - h[q:])
                    grad[q:] += (-h[:n-q])

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-13
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            for _ in range(5):
                idx = int(np.argmax(1.0 - h))
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            for _ in range(5):
                idx = int(np.argmax(h))
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h


def smax_value(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -300.0, 300.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smax_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -300.0, 300.0)
    exp_vals = np.exp(shifted)
    sw = np.sum(exp_vals)
    weights = exp_vals / sw
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for p in active:
        w = weights[p]
        shift = int(p) - (n - 1)
        if shift >= 0:
            k = shift
            if k < n:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] += w * (-h[k:])
        else:
            q = -shift
            if q < n:
                grad[:n - q] += w * (1.0 - h[q:])
                grad[q:] += w * (-h[:n - q])
    return grad * dx


def slsqp_f_val(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    return smax_value(h, beta) * (2.0 / n)


def slsqp_f_grad(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    return smax_grad(h, n, beta)


def run_slsqp(h0, n, max_iter, beta, time_limit=None):
    t_start = time.time()
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    constraints = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(
            lambda x: slsqp_f_val(x, n, beta),
            h0,
            jac=lambda x: slsqp_f_grad(x, n, beta),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0}
        )
        if time_limit is not None and time.time() - t_start > time_limit:
            return None, None
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def run_subgradient(h, n, n_steps, lr0, momentum, active_frac, time_limit):
    t_start = time.time()
    v = np.zeros(n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()

    for t in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
        c_max = np.max(corr)
        n_active = max(1, int(len(corr) * active_frac))
        sorted_idx = np.argsort(corr)[::-1][:n_active]

        grad = np.zeros(n)
        for p in sorted_idx:
            shift = int(p) - (n - 1)
            if shift >= 0:
                k = shift
                if k < n:
                    grad[k:] += (1.0 - h_cur[:n - k])
                    grad[:n - k] += (-h_cur[k:])
            else:
                q = -shift
                if q < n:
                    grad[:n - q] += (1.0 - h_cur[q:])
                    grad[q:] += (-h_cur[:n - q])
        grad /= float(len(sorted_idx))
        grad *= dx

        v = momentum * v + (1.0 - momentum) * grad
        lr = lr0 / (1.0 + 0.05 * t)
        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_cur = h_new

    return best_h, best_c5


def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_perturbed(h, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, len(h))
    return project_feasible(h_p, len(h))
