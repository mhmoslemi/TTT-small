            grad[0:n-t] += w * dx * (-h[t:n])
    return grad

def adam_subgradient(h, n, n_iter, top_m, tau, lr_init, rng):
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)

        v_hat = v_arr / (1.0 - beta2 ** t_idx)

        h = project_feasible(h, n)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            best_at = t
        else:
            patience += 1
            if patience > n_iter // 4 and t > n_iter // 2:
                break
    
    return best_h, best_c5

def binary_snap(h, n):
    """Try to snap values close to 0 or 1 to exactly 0 or 1"""
    h_snap = h.copy()
    h_snap[h_snap < 0.15] = 0.0
    h_snap[h_snap > 0.85] = 1.0
    h_snap = project_feasible(h_snap, n)

def coordinate_descent_local(h, n, n_sweeps, rng):

            if old < 1: candidates.append(min(1, old + 0.15))

            if old < 0.5: candidates.append(old + 0.3)

            break

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(result - 0.5))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def thue_morse_binary(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def subgradient_lse(h, n, top_m, tau, dx):
    lags = compute_lags(h)
    top_idx = np.argsort(lags)[::-1][:top_m]
    top_vals = lags[top_idx]
    v = top_vals / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    weights = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        s = int(k) - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += w * dx * (1.0 - h[s:])
                grad[s:] += w * dx * (-h[:ns])
        else:
            t_val = -s
            nt = n - t_val
            grad[t_val:] += w * dx * (1.0 - h[:nt])
            grad[:nt] += w * dx * (-h[t_val:])
    return grad

def adam_subgradient(h, n, n_iter, top_m, tau, lr_init, rng):
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    dx = 2.0 / n
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for t in range(n_iter):
        lr = lr_init / (1.0 + 0.02 * t)
        grad = subgradient_lse(h, n, top_m, tau, dx)
        t_idx = t + 1
        m_arr = beta1 * m_arr + (1.0 - beta1) * grad
        v_arr = beta2 * v_arr + (1.0 - beta2) * grad * grad
        m_hat = m_arr / (1.0 - beta1 ** t_idx)
        v_hat = v_arr / (1.0 - beta2 ** t_idx)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_snap(h, n):
    h2 = h.copy()
    h2[h2 < 0.2] = 0.0
    h2[h2 > 0.8] = 1.0

def coordinate_descent_local(h, n, n_sweeps, rng):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for idx in order:
            old = h[idx]
            cands = [0.0, 1.0, 0.5, max(0.0, old - 0.2), min(1.0, old + 0.2)]
            best_val = old
            best_c = best_c5
            for c in cands:
                if abs(c - old) < 1e-6 or c < 0 or c > 1:
                    continue
                h_t = h.copy()
                h_t[idx] = c
                h_t = project_feasible(h_t, n)
                c5_t = evaluate_c5(h_t)
                if c5_t < best_c:
                    best_c = c5_t
                    best_val = c
            if best_val != old:
                h[idx] = best_val
                h = project_feasible(h, n)

                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, n, tau, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    def obj(x):
        lags = compute_lags(x)
        top = lags[np.argsort(lags)[::-1][:5]]
        v = top / tau
        return tau * (np.max(v) + np.log(np.sum(np.exp(v - np.max(v)))))
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_pol = project_feasible(res.x, n)
        return h_pol, evaluate_c5(h_pol)
    except Exception:
        return h, evaluate_c5(h)
