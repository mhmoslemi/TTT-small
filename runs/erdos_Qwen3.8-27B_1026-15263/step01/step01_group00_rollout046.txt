def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_objective(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    e = np.exp(alpha * (corr - max_c))
    w = e / np.sum(e)
    
    grad = np.zeros(n)

        js2 = max(0, -m)
        je2 = min(n, n - m)

    return grad

            h_start[idx] += rng.uniform(-0.3, 0.3)
            h_start = project_box_sum(np.clip(h_start, 0, 1), ts)
        elif r < 0.5:
            # Block perturbation
            h_start = best_h.copy()
            bs = rng.randint(3, max(4, n // 5))
            st = rng.randint(0, n - bs)
            h_start[st:st+bs] += rng.uniform(-0.3, 0.3)
            h_start = project_box_sum(np.clip(h_start, 0, 1), ts)
        elif r < 0.75:
            # Binary push
            h_start = perturb_binary_push(best_h, rng, ts)
        else:
            # Random restart from scratch (interpolated)
            x = np.linspace(0, 2, n, endpoint=False)
            h_rand = 0.5 + 0.4 * np.sin(2*np.pi*rng.uniform(1,5)*x/2.0 + rng.uniform(0, 2*np.pi))
            h_start = project_box_sum(np.clip(h_rand, 0, 1), ts)
        
        # Optimize the perturbed solution
        for alpha in [50, 200, 1000]:
            if tl() < 15:
                break
            step = 1.0 / (4.0 * alpha * n)
            max_iter = min(5000, int(tl() * 2))
            h_opt, c5_opt = fista_opt(h_start, ts, alpha, max_iter, step, tl)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_start = h_opt
    
    # Phase 3: Coordinate descent polish
    if tl() > 30:
        h_cd, c5_cd = coord_descent(best_h, ts, rng, min(30, int(tl() / 2)), tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: Try other resolutions
    for n_try in [120, 160, 200, 240, 280]:
        if tl() < 30:
            break
        ts_try = n_try / 2.0
        if n_try % 2 != 0:
            continue
