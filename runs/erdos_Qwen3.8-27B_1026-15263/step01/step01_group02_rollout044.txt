    h = np.zeros(n)
    for i in range(n):
        t = (x[i] / period) % 1.0
        h[i] = 1.0 - 2.0 * abs(t - 0.5)  # triangular wave from 0 to 1
    # Center around 0.5
    h = (h - 0.5) + 0.5  # already centered
    return project_feasible(h, n)

def make_staircase(n, n_steps):
    """Create a staircase pattern."""
    h = np.zeros(n)
    for i in range(n):
        step = int(i * n_steps / n)
        h[i] = (step + 0.5) / n_steps
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            room = (1.0 - h[mask]).min()
            step = min(diff / np.sum(mask), room)
            h[mask] += step
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            room = h[mask].min()
            step = min(-diff / np.sum(mask), room)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h_ = x[:-1]
    t = x[-1]
    n = len(h_)
    c = np.correlate(h_, 1.0 - h_, mode='full')
    return t - c * 2.0 / n


def slsqp_eq(x):
    n = len(x) - 1
    return np.sum(x[:-1]) - n / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]


def slsqp_refine(h, maxiter=2000, ftol=1e-14):

    x0 = np.concatenate([h, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.7), 2.0)]
    constraints = slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})

        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h, c5_init
