            if n - 1 - k >= 0:
                grad[:n-k] += wi * (1.0 - h[k:n])
            # Second term: j in [k, n-1], contributes -h[j-k]
            if k < n:
                grad[k:] -= wi * h[:n-k]
        else:
            ka = -k  # positive
            # First term: j in [ka, n-1], contributes (1-h[j+k]) = (1-h[j-ka])
            if ka < n:
                grad[ka:] += wi * (1.0 - h[:n-ka])
            # Second term: j in [0, n-1-ka], contributes -h[j-k] = -h[j+ka]
            if n - 1 - ka >= 0:
                grad[:n-ka] -= wi * h[ka:]
    
    return grad

def nesterov_projected_gradient(h0, alpha, lr, momentum, max_iters, t_check):
    """Nesterov accelerated projected gradient descent."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = smooth_max_val(get_corr(h), alpha)
    
    for it in range(max_iters):
        if time.time() > t_check:
            break
        # Nesterov step
        y = h + momentum * v  # extrapolation
        grad = grad_smooth_max(y, alpha)
        
        # Gradient step
        v_new = momentum * v - lr * grad
        h_trial = project_bounded(h + v_new, target_sum)
        
        # Line search (backtracking)
        val_trial = smooth_max_val(get_corr(h_trial), alpha)
        if val_trial > best_val * 1.01 and it > 10:
            # Try smaller step
            h_trial = project_bounded(h - 0.5 * lr * grad, target_sum)
            val_trial = smooth_max_val(get_corr(h_trial), alpha)
        
        if val_trial < best_val:
            v = h_trial - h
            h = h_trial
            best_val = val_trial
            best_h = h.copy()
        else:
            # Decay learning rate
            lr *= 0.95
            v = momentum * v
        
        # Adaptive learning rate
        if it % 100 == 99:
            lr = min(2 * lr, 1.0)
    
    return best_h, best_val

def coordinate_descent(h0, max_rounds, t_check):
    """Coordinate descent: try to improve by adjusting pairs of coordinates."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.astype(float).copy()
    c5 = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > t_check:
            break
        improved = False
        # Try small pairwise adjustments
        for _ in range(50):
            i, j = np.random.choice(n, 2, replace=False)
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                h_new = h.copy()

                h_new[j] -= delta
                if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                    c5_new = compute_c5(h_new)
                    if c5_new < c5 - 1e-12:
                        h = h_new
                        c5 = c5_new
                        improved = True
                        break
        if not improved and rnd > 5:
            break
    return h, c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -40, 40)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    if s < 1e-300:
        return m
    return m + np.log(s / len(exp_v)) / alpha

def grad_smooth_max(h, alpha, top_k=None):

    corr = get_corr(h)

    shifted = np.clip(alpha * (corr - m), -40, 40)
