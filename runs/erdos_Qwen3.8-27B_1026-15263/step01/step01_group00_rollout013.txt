    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(80):

        h_p[mask] += diff / mask.sum()
        h_p = np.clip(h_p, 0.0, 1.0)

def smooth_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    se = np.exp(shifted)
    return (m + np.log(se.sum() / len(se)) / alpha) * dx, se / se.sum()

def compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    se = np.exp(shifted)
    w = se / se.sum()
    
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        i_lo = max(0, n - 1 - t)
        i_hi = min(n - 1, 2 * n - 2 - t)
        if i_lo > i_hi:
            continue
        idx1 = np.arange(i_lo, i_hi + 1)
        idx2 = idx1 + t - n + 1
        grad[idx1] += wt * (1.0 - h[idx2])
        grad[idx2] -= wt * h[idx1]
    return grad * dx

        g = compute_grad(y, alpha)

            continue
        step = min(step * 1.1, 0.1)
        
        # Decay momentum
        momentum = 0.9 * momentum + 0.0625
    
    return best_h, best_c5

    best_h = h.copy()

    step = 0.005
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        # Find all active lags (within 1e-6 of max)
        active = np.where(corr > max_val - 1e-6)[0]

        # Or just the max one
        k_star = active[0]  # or average over active

        # grad = average of grad of active corr_k
        # But that's expensive. Just use k_star.

        g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h_p.sum()
    mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
    if mask.sum() > 0:
        h_p[mask] += diff / mask.sum()
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    se = np.exp(shifted)
    w = se / se.sum()
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-14:
            continue
        i_lo = max(0, n - 1 - t)
        i_hi = min(n - 1, 2 * n - 2 - t)
        if i_lo > i_hi:
            continue
        idx1 = np.arange(i_lo, i_hi + 1)
        idx2 = idx1 + t - n + 1
        grad[idx1] += wt * (1.0 - h[idx2])
        grad[idx2] -= wt * h[idx1]
    return grad * dx

def smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    se = np.exp(shifted)
    return (m + np.log(se.sum() / len(se)) / alpha) * dx

def projected_gradient(h0, n, target_sum, alpha, max_iter, t_start, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.04
    prev_c5 = best_c5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_grad(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        c5_cur = compute_c5(h)
        accepted = False
        for _ in range(25):
            h_new = project_box_sum(h - step * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-5 * step * gn * gn:
                h = h_new
                step = min(step * 1.15, 0.15)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step = max(step * 0.8, 1e-9)
            if step < 1e-9:
                break
    return best_h, best_c5

def active_lag_polish(h0, n, target_sum, max_iter, t_start, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 0.004
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-8)[0]
        g = np.zeros(n)
        for t in active:
            i_lo = max(0, n - 1 - t)
            i_hi = min(n - 1, 2 * n - 2 - t)
            if i_lo > i_hi:
                continue
            idx1 = np.arange(i_lo, i_hi + 1)
            idx2 = idx1 + t - n + 1
            g[idx1] += (1.0 - h[idx2])
            g[idx2] -= h[idx1]
        g /= len(active)
        c5_cur = max_val * dx
        accepted = False
        for _ in range(15):
            h_new = project_box_sum(h - step * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-10:
                h = h_new
                step = min(step * 1.03, 0.04)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step = max(step * 0.7, 1e-10)
            if step < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    deadline = t_start + budget_s - 15
    rng = np.random.RandomState(seed)

    best_h = None

    if 'initial_h_values' in globals() and initial_h_values is not None:

        if alphas.index(alpha) < 2:
