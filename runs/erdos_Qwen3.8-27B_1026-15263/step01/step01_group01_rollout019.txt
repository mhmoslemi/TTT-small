Let me also consider: should I use the smooth max or the actual max? The smooth max is differentiable everywhere, which is important for L-BFGS-B. The actual max is only differentiable where the maximizer is unique.

With smooth max and large beta, we approximate the max well. The issue is that for very large beta, the gradient becomes concentrated on the maximizing k, and the optimization can get "stuck" trying to reduce just that one term while others increase.

Strategy: use progressively larger beta, starting small (where the objective is more "flat") and ending large (where it approximates the max well).

Let me also think about the step size. L-BFGS-B uses its own line search (Wolfe conditions), so I don't need to worry about that.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    scaled = beta * corr * dx
    scaled -= np.max(scaled)
    w = np.exp(scaled) / np.sum(np.exp(scaled))
    
    grad = np.zeros(n)
    l_arr = np.arange(n)
    for j in range(n):
        k1 = j + n - 1 - l_arr
        term1 = np.dot(w[k1], g)
        k2 = l_arr - j + n - 1
        term2 = np.dot(w[k2], h)
        grad[j] = dx * (term1 - term2)
    return grad

def obj_and_grad(x, n, beta, lam):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * (2.0/n)
    scaled -= np.max(scaled)
    val = float(np.max(corr) * (2.0/n) + (2.0/(n*beta)) * np.log(np.sum(np.exp(scaled))))
    
    grad = smooth_max_grad(h, beta)
    
    sum_h = np.sum(h)
    target = n / 2.0
    diff = sum_h - target
    val += lam * diff * diff
    grad = grad + 2.0 * lam * diff * np.ones(n)
    
    return val, grad

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final exact fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:int(np.ceil(abs(diff)))]
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
        else:
            idx = np.argsort(h)[:int(np.ceil(abs(diff)))]
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

            h_new = project_feasible(res.x, best_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    scaled = beta * corr * dx
    scaled -= np.max(scaled)
    w = np.exp(scaled) / np.sum(np.exp(scaled))
    grad = np.zeros(n)
    l_arr = np.arange(n)
    for j in range(n):
        k1 = j + n - 1 - l_arr
        term1 = np.dot(w[k1], g)
        k2 = l_arr - j + n - 1
        term2 = np.dot(w[k2], h)
        grad[j] = dx * (term1 - term2)
    return grad

def obj_and_grad(x, n, beta, lam):
