import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_polytope(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(vals, alpha):
    m = np.max(vals)
    if alpha < 1e10:
        exp_vals = np.exp(alpha * (vals - m))
        return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha
    return m

def smooth_max_weights(vals, alpha):
    m = np.max(vals)
    if alpha < 1e10:
        exp_vals = np.exp(alpha * (vals - m))
        return exp_vals / np.sum(exp_vals)
    w = np.zeros(len(vals))
    w[np.argmax(vals)] = 1.0
    return w

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    k_indices = np.arange(2 * n - 1)
    for m in range(n):
        a = m + k_indices - (n - 1)
        valid_a = (a >= 0) & (a < n)
        if np.any(valid_a):
            term1 = np.sum(w[valid_a] * (1.0 - h[a[valid_a]]))
        else:
            term1 = 0.0
        b = m - k_indices + (n - 1)
        valid_b = (b >= 0) & (b < n)
        if np.any(valid_b):
            term2 = np.sum(w[valid_b] * h[b[valid_b]])
        else:
            term2 = 0.0
        grad[m] = term1 - term2
    return grad

def gradient_topk(h, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[-topk:]
    w = np.zeros(2 * n - 1)
    w[top_indices] = 1.0 / topk
    grad = np.zeros(n)
    k_indices = np.arange(2 * n - 1)
    for m in range(n):
        a = m + k_indices - (n - 1)
        valid_a = (a >= 0) & (a < n)
        if np.any(valid_a):
            term1 = np.sum(w[valid_a] * (1.0 - h[a[valid_a]]))
        else:
            term1 = 0.0
        b = m - k_indices + (n - 1)
        valid_b = (b >= 0) & (b < n)
        if np.any(valid_b):
            term2 = np.sum(w[valid_b] * h[b[valid_b]])
        else:
            term2 = 0.0
        grad[m] = term1 - term2
    return grad

def projected_gradient_descent(h0, n, target_sum, alpha, n_iters, lr=None, momentum=0.9):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if lr is None:
        lr = 0.5 / (np.linalg.norm(gradient_smooth_max(h, alpha)) + 1e-8)
    
    for it in range(n_iters):
        grad = gradient_smooth_max(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_to_polytope(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 50 == 0 and it > 0:
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-8:
                lr = lr * 0.95
    return best_h, best_c5

def topk_subgradient(h0, n, target_sum, topk, n_iters, lr=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if lr is None:
        lr = 0.3
    lr_decay = 0.999
    
    for it in range(n_iters):
        grad = gradient_topk(h, topk)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        h_new = h - lr * grad / grad_norm
        h_new = project_to_polytope(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            lr *= 0.8
        h = h_new
        lr *= lr_decay
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, rng, max_rounds=100, deltas=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if deltas is None:
        deltas = np.array([0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01])
    
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            best_delta = 0.0
            best_c5 = c5_cur
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_polytope(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_polytope(h, target_sum)
                c5_cur = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, c5_cur

def pair_swap_search(h0, n, target_sum, rng, n_trials=5000):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_trials):
        i, j = rng.choice(n, size=2, replace=False)
        di = rng.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + di, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - di, 0.0, 1.0)
        h_new = project_to_polytope(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=200):
    def objective(x):
        h_c = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        return smooth_max(corr, alpha)
    
    def grad(x):
        h_c = np.clip(x, 0.0, 1.0)
        return gradient_smooth_max(h_c, alpha)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0.copy(), Jac=grad, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_to_polytope(res.x, target_sum)
    except:
        return project_to_polytope(h0, target_sum)

def simulated_annealing(h0, n, target_sum, rng, max_iters=5000, T0=0.01, T_end=0.0001):
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    T = T0
    cool = (T_end / T0) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if rng.random() < 0.3:
            i, j = rng.choice(n, size=2, replace=False)
            d = rng.uniform(-0.03, 0.03)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - d, 0.0, 1.0)
        elif rng.random() < 0.5:
            idx = rng.randint(n)
            d = rng.uniform(-0.05, 0.05)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + d, 0.0, 1.0)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            d = rng.uniform(-0.01, 0.01)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - d, 0.0, 1.0)
        
        h_new = project_to_polytope(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-12)):
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        T = max(T * cool, T_end)
    return h_best, c5_best
