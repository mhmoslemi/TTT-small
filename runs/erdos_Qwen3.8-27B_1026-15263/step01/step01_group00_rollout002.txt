import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(v, target_sum, lo=0.0, hi=1.0):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    t_lo = np.min(v) - (hi - lo) - 1.0
    t_hi = np.max(v) + (hi - lo) + 1.0
    for _ in range(80):
        t_mid = 0.5 * (t_lo + t_hi)
        s = np.sum(np.clip(v - t_mid, lo, hi))
        if s > target_sum:
            t_lo = t_mid
        else:
            t_hi = t_mid
    tau = 0.5 * (t_lo + t_hi)
    x = np.clip(v - tau, lo, hi)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-13:
        mask = (x > lo + 1e-10) & (x < hi - 1e-10)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, lo, hi)
    return x

def smoothed_obj_and_grad(h, alpha):
    n = len(h)
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = cmax + np.log(np.sum(exp_vals) / (2 * n - 1)) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wi * (1.0 - h_c[j_lo - k:j_hi - k])
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wi * h_c[j_lo2 + k:j_hi2 + k]
    return obj, grad

def fista(h0, target_sum, alpha, max_iter, time_limit, t_start):
    n = len(h0)
    h = project_simplex(h0.copy(), target_sum)
    z = h.copy()
    t_param = 1.0
    best_h = h.copy()
    best_f = compute_c5(h)
    L = 2.0 * n
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        obj, grad = smoothed_obj_and_grad(z, alpha)
        step = 1.0 / L
        z_trial = project_simplex(z - step * grad, target_sum)
        f_trial = compute_c5(z_trial)
        for _ in range(15):
            if f_trial <= obj + 0.5 * step * np.dot(grad, grad) + 1e-9:
                break
            step *= 0.5
            L *= 1.3
            z_trial = project_simplex(z - step * grad, target_sum)
            f_trial = compute_c5(z_trial)
        if f_trial < best_f:
            best_f = f_trial
            best_h = z_trial.copy()
        t_old = t_param
        t_param = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        h = z_trial + ((t_old - 1.0) / t_param) * (z_trial - h)
        h = project_simplex(h, target_sum)
        z = h.copy()
    return best_h, best_f

def subgradient_polish(h0, target_sum, time_limit, t_start, max_iter=5000):
    n = len(h0)
    h = project_simplex(h0.copy(), target_sum)
    best_h = h.copy()
    best_f = compute_c5(h)
    step = 0.3
    patience = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)
        sg = np.zeros(n)
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            sg[j_lo:j_hi] += (1.0 - h[j_lo - k:j_hi - k])
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            sg[j_lo2:j_hi2] -= h[j_lo2 + k:j_hi2 + k]
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        sg = sg / sg_norm
        h_trial = project_simplex(h - step * sg, target_sum)
        f_trial = compute_c5(h_trial)
        if f_trial < best_f - 1e-12:
            best_f = f_trial
            best_h = h_trial.copy()
            h = h_trial
            patience = 0
            step = min(step * 1.05, 0.8)
        else:
            patience += 1
            step *= 0.85
            if step < 1e-5:
                break
            if patience > 60:
                break
    return best_h, best_f

def fine_coordinate(h0, target_sum, time_limit, t_start, n_rounds=200):
    n = len(h0)
    h = project_simplex(h0.copy(), target_sum)
    best_f = compute_c5(h)
    rng = np.random.RandomState(12345)
    deltas = [0.001, -0.001, 0.003, -0.003, 0.005, -0.005, 0.01, -0.01]
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_simplex(h_new, target_sum)
                f_new = compute_c5(h_new)
                if f_new < best_f - 1e-13:
                    h = h_new
                    best_f = f_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_f

def multi_lag_subgradient(h0, target_sum, time_limit, t_start, max_iter=3000):
    n = len(h0)
    h = project_simplex(h0.copy(), target_sum)
    best_h = h.copy()
    best_f = compute_c5(h)
    step = 0.15
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        thresh = cmax - 0.5
        active = np.where(corr > thresh)[0]
        sg = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            j_lo = max(0, k)
            j_hi = min(n, n + k)
            if j_lo < j_hi:
                sg[j_lo:j_hi] += (1.0 - h[j_lo - k:j_hi - k])
            j_lo2 = max(0, -k)
            j_hi2 = min(n, n - k)
            if j_lo2 < j_hi2:
                sg[j_lo2:j_hi2] -= h[j_lo2 + k:j_hi2 + k]
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        sg = sg / (sg_norm * max(1, len(active)))
        h_trial = project_simplex(h - step * sg, target_sum)
        f_trial = compute_c5(h_trial)
        if f_trial < best_f - 1e-13:
            best_f = f_trial
            best_h = h_trial.copy()
            h = h_trial
            step = min(step * 1.02, 0.5)
        else:
            step *= 0.9
            if step < 1e-5:
                break
    return best_h, best_f

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, np.concatenate([h_src, [h_src[0]]]))
    return h_dst

def make_block_start(n, rng):
    pattern = []
    block = max(1, n // 12)
    for i in range(n):
        phase = (i // block) % 4
        if phase in (0, 2):
            pattern.append(1.0)
        else:
            pattern.append(0.0)
    h = np.array(pattern[:n], dtype=float)
    h += rng.uniform(-0.05, 0.05, n)
    return h

def make_sinusoidal_start(n, rng, freq):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0)
    h += rng.uniform(-0.03, 0.03, n)
    return h
