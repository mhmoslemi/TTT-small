def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute among all
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha

def smooth_max_weighted(corr, alpha, dx):
    # Returns the smooth max objective (scaled by dx) and the softmax weights
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    obj = (m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha) * dx
    return obj, w, m
