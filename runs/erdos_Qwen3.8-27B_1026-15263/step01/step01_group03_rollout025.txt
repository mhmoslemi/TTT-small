import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def solve_waterfill(a, target):
    n = len(a)
    lo = -np.max(a) - 0.5
    hi = 1.0 - np.min(a) + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(a + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def a_to_h(a, n):
    target = n / 2.0
    t = solve_waterfill(a, target)
    h = np.clip(a + t, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def obj_smooth(a, n, tau):
    h = a_to_h(a, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, tau) * (2.0 / n)

def obj_lag(a, n, k_star):
    h = a_to_h(a, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[k_star] * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def pair_swap_descent(h, n, n_rounds=300, swap_base=0.15):
    dx = 2.0 / n
    best_c = evaluate_c5(h)
    stall = 0
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        c_max = corr[k_star] * dx
        if c_max < best_c - 1e-12:
            best_c = c_max
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
        lag = k_star - (n - 1)
        if lag >= 0:
            idx_range = np.arange(n - lag)
            left = h[idx_range]
            right = h[idx_range + lag]
        else:
            idx_range = np.arange(-lag, n)
            left = h[idx_range + lag]
            right = h[idx_range]
        contrib = left * (1.0 - right)
        top_k = min(5, len(contrib))
        top_indices = np.argsort(contrib)[::-1][:top_k]
        improved = False
        swap_amt = swap_base * (1.0 - 0.5 * stall / 30.0)
        for ti in top_indices:
            i = int(idx_range[ti])
            h_new = h.copy()
            if lag >= 0:
                h_new[i] = np.clip(h_new[i] - swap_amt, 0, 1)
                h_new[i + lag] = np.clip(h_new[i + lag] + swap_amt, 0, 1)
            else:
                h_new[i + lag] = np.clip(h_new[i + lag] - swap_amt, 0, 1)
                h_new[i] = np.clip(h_new[i] + swap_amt, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = evaluate_c5(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            stall += 2
    return h, best_c

def coordinate_refine(h, n, n_passes=5):
    for _ in range(n_passes):
        improved_any = False
        for j in range(n):
            best_val = h[j]
            best_c = evaluate_c5(h)
            for delta in [-0.08, -0.04, -0.02, 0.02, 0.04, 0.08]:
                h[j] = np.clip(h[j] + delta, 0, 1)
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 0.3:
                    h[j] = best_val
                    continue
                h[j] = np.clip(h[j] + diff, 0, 1)
                c = evaluate_c5(h)
                h[j] = best_val
                if c < best_c:
                    best_c = c
                    best_val = h[j]
                    h[j] = np.clip(h[j] + delta, 0, 1)
                    diff2 = n / 2.0 - np.sum(h)
                    h[j] = np.clip(h[j] + diff2, 0, 1)
                    improved_any = True
        diff = n / 2.0 - np.sum(h)
        if abs(diff) > 1e-10:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
        if not improved_any:
            break
    return h

def generate_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)
    if parent_h is not None:
        if n == len(parent_h):
            starts.append(parent_h.copy())
            for scale in [0.03, 0.06, 0.1, 0.15]:
                starts.append(np.clip(parent_h + rng.randn(n) * scale, 0, 1))
        else:
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            f = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate')
            starts.append(np.clip(f(x), 0, 1))
            for scale in [0.05, 0.1, 0.15]:
                h_p = np.clip(parent_h + rng.randn(len(parent_h)) * scale, 0, 1)
                h_p = project_feasible(h_p, len(parent_h))
                f2 = interp1d(xp, h_p, kind='cubic', fill_value='extrapolate')
                starts.append(np.clip(f2(x), 0, 1))
    for _ in range(5):
        starts.append(rng.rand(n))
    for freq in [1.2, 1.5, 1.8, 2.1, 2.4, 2.8, 3.2, 3.7]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.25, 0.49)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), 0, 1))
    for freq in [1.0, 1.5, 2.0]:
        phase1 = rng.uniform(0, 2 * np.pi)
        phase2 = rng.uniform(0, 2 * np.pi)
        amp1 = rng.uniform(0.15, 0.3)
        amp2 = rng.uniform(0.1, 0.25)
        starts.append(np.clip(0.5 + amp1 * np.sin(2 * np.pi * freq * x + phase1) +
                              amp2 * np.sin(2 * np.pi * freq * 2 * x + phase2), 0, 1))
    starts.append(np.linspace(0.02, 0.98, n))
    starts.append(np.linspace(0.98, 0.02, n))
    h_tri = np.mod(ar / n * 2.0, 2.0)
    h_tri = np.where(h_tri < 1.0, h_tri, 2.0 - h_tri)
    starts.append(np.clip(h_tri, 0, 1))
    h_block = np.zeros(n)
    block_size = n // 5
    for i in range(5):
        h_block[i * block_size:(i + 1) * block_size] = 0.9
    starts.append(h_block)
    return starts
