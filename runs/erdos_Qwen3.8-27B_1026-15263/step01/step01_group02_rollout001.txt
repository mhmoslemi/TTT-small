    h = np.array(h, dtype=float)

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx


def exact_project(h, target_sum):
    h = np.array(h, dtype=float)
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-15:
        mask = (result > 0.0) & (result < 1.0)
        if np.any(mask):
            result[mask] += residual / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result


def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    exps = np.exp(beta * (c - c_max))
    w_sum = np.sum(exps)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = exps / w_sum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        r = m - n + 1
        i_lo = max(0, r)
        i_hi = min(n - 1, n - 1 + r)
        js = np.arange(i_lo, i_hi + 1)
        jr = js - r

        if js2_lo <= js2_hi:
            js2 = np.arange(js2_lo, js2_hi + 1)
            grad[js2] -= w * dx * h[js2 + r]
    return grad

    c = compute_c5_vals(h)
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

    h_prev = h0.copy()

        if time_limit_fn():
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        m = (it + 1) / (it + 2.0)
        h_nest = h_curr + m * (h_curr - h_prev)
        grad = smooth_gradient(h_nest, beta)
        f_old = smooth_obj(h_nest, beta)
        h_trial = h_nest - lr * grad
        h_trial = exact_project(h_trial, target_sum)
        f_trial = smooth_obj(h_trial, beta)
        backtracks = 0
        while f_trial > f_old + 1e-12 and backtracks < 20:

            h_trial = h_nest - lr * grad
            h_trial = exact_project(h_trial, target_sum)
            f_trial = smooth_obj(h_trial, beta)
            backtracks += 1
        h_prev = h_curr
        h_curr = h_trial
        if backtracks < 5:
            lr = min(lr * 1.05, lr_init * 2.0)
    return h_curr


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
