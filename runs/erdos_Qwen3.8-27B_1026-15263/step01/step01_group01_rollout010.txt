```python
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
```

```python
def compute_subgradient(h, n, top_fraction=0.1):
    """Compute subgradient of max_k c_k"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Active: top fraction of correlations
    n_active = max(1, int(len(corr) * top_fraction))
    # Get indices of top n_active
    active_indices = np.argsort(corr)[-n_active:]
    active_corrs = corr[active_indices]
    # Weight: how much each active corr is near the max
    weights = np.minimum(active_corrs / max_val, 1.0)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for idx in active_indices:
        lag = idx - (n - 1)
        for m in range(n):
            contrib = 0.0
            if 0 <= m + lag < n:
                contrib += (1.0 - h[m + lag])
            if 0 <= m - lag < n:
                contrib -= h[m - lag]
            g[m] += weights[active_indices == idx][0] * contrib
    
    return g, corr, max_val
```

```python
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        m1_lo = max(0, -lag)
        m1_hi = min(n-1, n-1-lag)
        if m1_lo <= m1_hi:
            g[m1_lo:m1_hi+1] += weights[k] * (1.0 - h[m1_lo+lag:m1_hi+1+lag])
        m2_lo = max(0, lag)
        m2_hi = min(n-1, n-1+lag)
        if m2_lo <= m2_hi:
            g[m2_lo:m2_hi+1] -= weights[k] * h[m2_lo-lag:m2_hi+1-lag]
    
    return g * (2.0/n), corr, m
```

For n=190, the loop over 379 corrs with vectorized inner ops should be fast enough.

Now, the projected subgradient method:
1. Compute gradient g of the smooth max objective
2. Project g onto the constraint sum(v)=0: v = g - mean(g)*1
3. Update: h_new = h - α * v
4. Project h_new to [0,1] and fix sum

Let me integrate this into my run function.

Let me structure the algorithm:

2. **Subgradient descent** (30% of time):

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 0.0)
    return m + np.sum(np.exp(shifted)) / beta


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
        else:
            mask = h > 1e-13
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / np.sum(mask), 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def smooth_max_grad_h(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += w[k] * (1.0 - h[lo1 + lag:hi1 + 1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= w[k] * h[lo2 - lag:hi2 + 1 - lag]
    return g * (2.0 / n), corr, m


def subgradient_descent(h, n, rng, time_budget, alpha0, alpha_end, beta):
    t_start = time.time()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_total = max(1, int(time_budget * 800))
    n_step = max(1, n_total // 100)
    for it in range(n_total):
        if time.time() - t_start > time_budget:
            break
        frac = it / max(1, n_total)
        alpha = alpha0 * (alpha_end / alpha0) ** frac
        g, corr, mval = smooth_max_grad_h(h, n, beta)
        v = g - np.mean(g)
        norm_v = np.linalg.norm(v)
        if norm_v < 1e-15:
            break
        v = v / norm_v * alpha
        h_new = h - v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def local_search(h, n, rng, time_budget, delta_list):
    t_start = time.time()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    idx_list = list(range(len(delta_list)))
    cached_c5 = best_c5
    while time.time() - t_start < time_budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        di = delta_list[rng.choice(idx_list)]
        hi_new = h[i] + di
        hj_new = h[j] - di
        if hi_new < -1e-12 or hi_new > 1.0 + 1e-12:
            continue
        if hj_new < -1e-12 or hj_new > 1.0 + 1e-12:
            continue
        hi_new = min(max(hi_new, 0.0), 1.0)
        hj_new = min(max(hj_new, 0.0), 1.0)
        h[i], h[j] = hi_new, hj_new
        c = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if c < best_c5 - 1e-12:
            best_c5 = c
        else:
            h[i], h[j] = h[i] - di, h[j] + di
    return h, best_c5


def slsqp_optimize(h0, n, max_iter, beta, ftol, time_limit):
    t_start = time.time()
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        if time.time() - t_start > time_limit:
            raise ValueError("timeout")
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max_val(corr, beta) * (2.0 / n)

    def sum_eq(x):
        return np.sum(x) - n / 2.0

    def sum_eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return h0, compute_c5(h0)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_n = 190
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient descent (25% of budget)
    remaining = budget_s - elapsed()
    limit1 = remaining * 0.25
    if limit1 > 30:
        h_sg, c5_sg = subgradient_descent(best_h, best_n, rng, limit1,
                                          alpha0=0.05, alpha_end=0.001, beta=100.0)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 2: Local search (20% of budget)
    remaining = budget_s - elapsed()
    limit2 = remaining * 0.20
    if limit2 > 20:
        deltas = [0.01, 0.02, 0.05, 0.1, 0.15]
        h_ls, c5_ls = local_search(best_h, best_n, rng, limit2, deltas)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 3: SLSQP refinement (15% of budget)
    remaining = budget_s - elapsed()
    limit3 = remaining * 0.15
    if limit3 > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 300, 150.0, 1e-14, min(limit3, 180))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Fine local search (15% of budget)
    remaining = budget_s - elapsed()
    limit4 = remaining * 0.15
    if limit4 > 20:
        deltas_fine = [0.005, 0.01, 0.02, 0.05]
        h_ls, c5_ls = local_search(best_h, best_n, rng, limit4, deltas_fine)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 5: Subgradient with higher beta (10% of budget)
    remaining = budget_s - elapsed()
    limit5 = remaining * 0.10
    if limit5 > 20:
        h_sg, c5_sg = subgradient_descent(best_h, best_n, rng, limit5,
                                          alpha0=0.01, alpha_end=0.0005, beta=300.0)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 6: SLSQP with very high beta (10% of budget)
    remaining = budget_s - elapsed()
    limit6 = remaining * 0.10
    if limit6 > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 250, 400.0, 1e-14, min(limit6, 150))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Very fine local search (5% of budget)
    remaining = budget_s - elapsed()
    limit7 = remaining * 0.05
    if limit7 > 10:
        deltas_veryfine = [0.003, 0.005, 0.01, 0.02]
        h_ls, c5_ls = local_search(best_h, best_n, rng, limit7, deltas_veryfine)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 8: Try higher resolution
    remaining = budget_s - elapsed()
    if remaining > 120:
        n_new = 380
        h_interp = interpolate_h(best_h, best_n, n_new)
        h_interp = project_feasible(h_interp, n_new)
        time_left = budget_s - elapsed()
        h_opt, c5_opt = slsqp_optimize(h_interp, n_new, 200, 100.0, 1e-12, min(time_left * 0.5, 100))
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_new
            remaining = budget_s - elapsed()
            if remaining > 40:
                h_ls, c5_ls = local_search(best_h, best_n, rng, min(remaining * 0.5, 60),
                                            [0.005, 0.01, 0.02])
                if c5_ls < best_c5:
                    best_h, best_c5 = h_ls, c5_ls

    # Phase 9: Multiple SLSQP restarts with perturbation
    remaining = budget_s - elapsed()
    no_improve = 0
    while remaining > 60 and no_improve < 15:
        sigma = 0.03 + 0.05 * (0.9 ** no_improve)
        h_start = best_h + rng.normal(0, sigma, best_n)
        h_start = project_feasible(h_start, best_n)
        time_left = budget_s - elapsed()
        if time_left < 40:
            break
        h_opt, c5_opt = slsqp_optimize(h_start, best_n, 200, 200.0, 1e-13, min(time_left * 0.6, 80))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
        remaining = budget_s - elapsed()

    # Final local search
    remaining = budget_s - elapsed()
    if remaining > 15:
        h_ls, c5_ls = local_search(best_h, best_n, rng, min(remaining * 0.7, 30),
                                    [0.003, 0.005, 0.01])
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
```
