import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    seq = np.zeros(n)
    for i in range(1, n):
        # t(i) = parity of number of 1s in binary representation of i
        v = i
        p = 0
        while v:
            p ^= (v & 1)
            v >>= 1
        seq[i] = p
    return seq

def generate_diverse_starts(n, target_sum, rng, count=20):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    
    # Uniform
    starts.append(np.full(n, 0.5))
    
    # Random
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    
    # Binary random
    h = (rng.rand(n) > 0.5).astype(float)
    starts.append(project_to_sum(h, target_sum))
    
    # Thue-Morse
    tm = thue_morse(n)
    starts.append(project_to_sum(tm, target_sum))
    
    # Sinusoidal various frequencies
    for freq in [1, 2, 3, 4, 5]:
        hs = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(hs, target_sum))
    
    # Block structures
    for nblocks in [2, 3, 4, 5, 6]:
        h4 = np.zeros(n)
        period = n // nblocks
        for b in range(nblocks):
            h4[b*period:(b+1)*period] = 1.0
        # Alternate: fill every other block
        h4 = np.zeros(n)
        for b in range(0, nblocks, 2):
            h4[b*period:(b+1)*period] = 1.0
        starts.append(project_to_sum(h4, target_sum))
    
    # Triangular wave
    for freq in [1, 2, 3]:
        t = np.abs(2 * (freq * x / 2.0 - np.floor(freq * x / 2.0 + 0.5)))
        starts.append(project_to_sum(t, target_sum))
    
    # Smooth step
    for center in [0.5, 1.0, 1.5]:
        hs = 1.0 / (1.0 + np.exp(-10 * (x - center)))
        starts.append(project_to_sum(hs, target_sum))
    
    while len(starts) < count:
        h = rng.uniform(0.1, 0.9, n)
        starts.append(project_to_sum(h, target_sum))
    
    return starts

def ga_optimize(n, target_sum, rng, budget_s, t_start, pop_size=60, elite=2):

    # Initialize population
    population = []
    population_f = []
    
    # Add diverse starts
    starts = generate_diverse_starts(n, target_sum, rng, count=min(pop_size, 20))
    for s in starts:
        population.append(s.copy())
        population_f.append(compute_c5(s))
    
    # Fill remaining with random perturbations of best
    while len(population) < pop_size:
        best_idx = np.argmin(population_f)
        pert = population[best_idx].copy()
        idx = rng.randint(n)
        pert[idx] = np.clip(pert[idx] + rng.gauss(0, 0.1), 0, 1)
        pert = project_to_sum(pert, target_sum)
        population.append(pert)
        population_f.append(compute_c5(pert))
    
    best_h = population[np.argmin(population_f)].copy()
    best_f = min(population_f)
    
    generation = 0
    while time.time() - t_start < budget_s - 30:
        generation += 1
        # Sort by fitness
        order = np.argsort(population_f)
        population = [population[i].copy() for i in order]

        new_population = []
        new_f = []
        
        # Elitism

            new_population.append(population[i].copy())

def fine_local_search(h0, n, target_sum, rng, max_rounds=200):
    """Very fine coordinate descent with adaptive step sizes."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in step_sizes:
        for rnd in range(max_rounds // len(step_sizes)):
            improved = False
            idx_order = rng.permutation(n)
            for i in idx_order:
                # Try moving h[i] up and down
                for direction in [1, -1]:
                    delta = direction * step
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-13:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    
    return h, c5_cur

def pairwise_swap_search(h0, n, target_sum, rng, max_rounds=100):

        improved = False
        for _ in range(50):

            if c5_new < c5_cur - 1e-13:
                h = h_new
                c5_cur = c5_new
                improved = True

def upscale_h(h_old, n_old, n_new):
    """Interpolate from n_old to n_new points (periodic)."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Periodic interpolation
    # Extend for wrapping
    x_ext = np.concatenate([x_old - 2, x_old, x_old + 2])
    h_ext = np.concatenate([h_old, h_old, h_old])
    h_new = np.interp(x_new, x_ext, h_ext)
    return h_new

            break
        order = np.argsort(pop_f)
        pop = [pop[i].copy() for i in order]
        pop_f = [pop_f[i] for i in order]
        new_pop = [pop[0].copy()]
        new_f = [pop_f[0]]
        while len(new_pop) < len(pop):
            i1, i2 = rng.choice(4, 2, replace=False)
            child = (pop[i1] + pop[i2]) / 2.0
            nmut = rng.randint(1, 3)
            for _ in range(nmut):
                mi = rng.randint(n)
                child[mi] = np.clip(child[mi] + rng.gauss(0, 0.015), 0, 1)
            child = project_to_sum(child, target_sum)
            new_pop.append(child)
            new_f.append(compute_c5(child))
        pop, pop_f = new_pop, new_f
        cb = min(pop_f)
        if cb < best_f - 1e-14:
            best_f = cb
            best_h = pop[np.argmin(pop_f)].copy()
    return best_h, best_f

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        _ihv = initial_h_values
        if _ihv is not None:
            _ihv = np.array(_ihv, dtype=float)
            _n = len(_ihv)
            _ts = _n / 2.0
            _ihv = project_to_sum(_ihv, _ts)
            best_h = _ihv.copy()
            best_c5 = compute_c5(best_h)
            best_n = _n
    except (NameError, TypeError, AttributeError):
        pass
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    t_limit_ga = t_start + budget_s * 0.30
    h_ga, c5_ga = ga_run(best_h, best_n, best_n / 2.0, rng, t_start, t_limit_ga, pop_size=40)
    if c5_ga < best_c5:
        best_c5 = c5_ga
        best_h = h_ga.copy()
    t_limit_n = t_start + budget_s * 0.50
    n_cands = [n for n in [156, 160, 164, 168, 172, 176, 180, 184, 188, 192, 196, 200, 204, 208, 216, 224, 240] if n % 2 == 0]
    n_cands.sort(key=lambda x: abs(x - best_n))
    for n_tri in n_cands:
        if time.time() > t_limit_n:
            break
        if n_tri == best_n:
            continue
        ts_t = n_tri / 2.0
        h_t = upscale_h(best_h, best_n, n_tri)
        h_t = project_to_sum(h_t, ts_t)
        h_t = bottleneck_refine(h_t, n_tri, ts_t, rng, t_start, t_limit_n)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_tri
    t_limit_ns = t_start + budget_s * 0.70
    while time.time() < t_limit_ns:
        h_nb, c5_nb = neighborhood_ga(best_h, best_n, best_n / 2.0, rng, t_start, t_limit_ns)
        if c5_nb < best_c5:
            best_c5 = c5_nb
            best_h = h_nb.copy()
        else:
            break
    t_limit_br = t_start + budget_s * 0.85
    h_br, c5_br = bottleneck_refine(best_h, best_n, best_n / 2.0, rng, t_start, t_limit_br)
    if c5_br < best_c5:
        best_c5 = c5_br
        best_h = h_br.copy()
    t_end = t_start + budget_s - 10
    h_cf, c5_cf = coord_fine(best_h, best_n, best_n / 2.0, rng, t_start, t_end)
    if c5_cf < best_c5:
        best_c5 = c5_cf
        best_h = h_cf.copy()
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
