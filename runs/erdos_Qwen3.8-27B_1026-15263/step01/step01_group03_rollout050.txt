    cons = [{'type': 'eq', 'fun': constraint_fun, 'args': (n,)}]
    res = minimize(
        obj_and_grad, h0,
        method='SLSQP',
        args=(n, tau, 0.0),
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)

    if parent_h is not None:
        pn = len(parent_h)
        n_sizes = sorted(set([pn] + list(range(max(42, pn - 10), min(80, pn + 11)))))
    else:
        n_sizes = [48, 50, 52, 54, 56, 58, 60, 62, 64]

    for n in n_sizes:
        if time.time() - start_time > budget_s * 0.78:
            break
        starts = generate_starts(n, parent_h, rng)
        for h0 in starts:
            if time.time() - start_time > budget_s * 0.84:
                break
            h_curr = h0.copy()
            taus = [0.1, 0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002]
            for tau in taus:
                if time.time() - start_time > budget_s * 0.90:
                    break
                maxit = 80 if tau > 0.02 else 120
                h_curr = optimize_lbfgs(h_curr, n, tau, maxit, 20.0)
                c5 = evaluate_c5(h_curr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()

    if best_h is not None and time.time() - start_time < budget_s * 0.90:
        n = len(best_h)
        for tau in [0.005, 0.003, 0.002]:
            if time.time() - start_time > budget_s * 0.94:
                break
            h_pol = slsqp_polish(best_h, n, tau, 300)
            c5 = evaluate_c5(h_pol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pol

    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        n = len(best_h)
        for attempt in range(10):
            if time.time() - start_time > budget_s * 0.98:
                break
            noise_scale = 0.002 + 0.001 * attempt
            h_pert = project_feasible(best_h + rng.randn(n) * noise_scale, n)
            h_pert = optimize_lbfgs(h_pert, n, 0.003, 100, 20.0)
            h_pert = slsqp_polish(h_pert, n, 0.003, 200)
            c5 = evaluate_c5(h_pert)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pert

    if best_h is None:
        n = 50
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
