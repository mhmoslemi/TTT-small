```python
def compute_gradient_c5(h, alpha):
    """Compute gradient of smooth_max(correlate(h, 1-h, 'full'), alpha) w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # smooth max weights
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    # Now compute gradient
    # grad[i] = sum_k w[k] * d(corr[k])/d(h[i])
    # d(corr[k])/d(h[i]) = (1-h[i+k-(n-1)])*1[i+k-(n-1) valid] - h[i-k+(n-1)]*1[i-k+(n-1) valid]

    n_half = n - 1  # center index in full correlation
    
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue

        i_start1 = max(0, n_half - k)
        i_end1 = min(n, n_half - k + n)  # exclusive
        if i_start1 < i_end1:
            idx1 = i_start1 + np.arange(i_end1 - i_start1)
            j1 = idx1 + k - n_half  # j1 = i + k - n_half, should be in [0, n-1]
            grad[idx1] += w[k] * (1.0 - h[j1])
        
        # i - k + n_half in [0, n-1] means i in [k - n_half, k - n_half + n - 1]
        i_start2 = max(0, k - n_half)
        i_end2 = min(n, k - n_half + n)  # exclusive
        if i_start2 < i_end2:
            idx2 = i_start2 + np.arange(i_end2 - i_start2)
            j2 = idx2 - k + n_half  # j2 = i - k + n_half
            grad[idx2] -= w[k] * h[j2]
    
    return grad
```

```python
    """Project h onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded_sum(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_and_weights(vals, alpha):
    m = np.max(vals)
    exp_vals = np.exp(alpha * (vals - m))
    sm = m + np.log(np.sum(exp_vals)) / alpha
    w = exp_vals / np.sum(exp_vals)
    return sm, w

def compute_smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = smooth_max_and_weights(corr, alpha)
    
    # Gradient: grad[i] = sum_k w[k] * (d c[k]/d h[i])
    # d c[k]/d h[i] = (1-h[i+k-(n-1)])*1[0<=i+k-(n-1)<n] - h[i-k+(n-1)]*1[0<=i-k+(n-1)<n]
    nh = n - 1
    grad = np.zeros(n)
    
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-8:
            continue
        # Term 1: i + k - nh in [0, n-1] => i in [nh-k, nh-k+n-1]
        i_lo1 = max(0, nh - k)
        i_hi1 = min(n - 1, nh - k + n - 1)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            j = idx + k - nh
            grad[idx] += wk * (1.0 - h[j])
        
        # Term 2: i - k + nh in [0, n-1] => i in [k-nh, k-nh+n-1]
        i_lo2 = max(0, k - nh)
        i_hi2 = min(n - 1, k - nh + n - 1)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            j = idx - k + nh
            grad[idx] -= wk * h[j]
    
    return sm, grad

def projected_gradient_descent(h0, target_sum, alpha, n_iters, lr_init, lr_decay):
    h = h0.copy().astype(float)
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        sm, grad = compute_smooth_max_grad(h, alpha)
        # Projected gradient step
        h_new = project_bounded_sum(h - lr * grad, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
    
    return best_h, best_c5

def local_search_pair_swap(h, target_sum, n_rounds, max_delta=0.05):
    """Local search: try pair swaps (i,j) with small delta, check exact c5 change"""
    n = len(h)
```

```python
import numpy as np
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded_sum(h, target_sum):
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if np.any(mask):
            h_p[mask] += diff / np.sum(mask)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s_exp = np.sum(exp_v)
    w = exp_v / s_exp
    sm = m + np.log(s_exp) / alpha
    nh = n - 1
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-7:
            continue
        i_lo1 = max(0, nh - k)
        i_hi1 = min(n - 1, nh - k + n - 1)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            j = idx + k - nh
            grad[idx] += wk * (1.0 - h[j])
        i_lo2 = max(0, k - nh)
        i_hi2 = min(n - 1, k - nh + n - 1)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            j = idx - k + nh
            grad[idx] -= wk * h[j]
    return sm, grad

def pgd_solve(h0, target_sum, alpha, n_iters, step_init, step_decay):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    for it in range(n_iters):
        sm, grad = compute_grad_smooth_max(h, alpha)
        h_new = project_bounded_sum(h - step * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > sm * (2.0 / n):
            step *= 0.5
        h = h_new
        step *= step_decay
    return best_h, best_c5

def directed_local_search(h, target_sum, n_rounds):
    n = len(h)
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(777)
    nh = n - 1
    for rnd in range(n_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_best = int(np.argmax(corr))
        grad_k = np.zeros(n)
        i_lo1 = max(0, nh - k_best)
        i_hi1 = min(n - 1, nh - k_best + n - 1)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            j = idx + k_best - nh
            grad_k[idx] += (1.0 - h[j])
        i_lo2 = max(0, k_best - nh)
        i_hi2 = min(n - 1, k_best - nh + n - 1)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            j = idx - k_best + nh
            grad_k[idx] -= h[j]
        n_tries = n * 5
        for _ in range(n_tries):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if grad_k[i] < grad_k[j]:
                delta = rng.uniform(0.003, 0.04)
            else:
                delta = -rng.uniform(0.003, 0.04)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new = project_bounded_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                c5_cur = c5_new
                improved = True
```
