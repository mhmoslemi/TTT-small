            if not np.any(mask):

            increment = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + increment, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            decrement = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - decrement, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
            if h[idx] < 1.0:
                h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = np.argmax(h)
            if h[idx] > 0.0:
                h[idx] = max(h[idx] + diff, 0.0)
    return h


def _obj(x):
    return x[-1]


def _ineq(x):
    h = x[:-1]
    t = x[-1]
    dx = 2.0 / _current_n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - np.max(c)


def _eq(x):
    return np.sum(x[:-1]) - _current_n / 2.0


def _eq_jac(x):
    jac = np.zeros(_current_n + 1)
    jac[:_current_n] = 1.0
    return jac


def _get_constraints():
    return [
        {'type': 'ineq', 'fun': _ineq},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
    ]


def slsqp_optimize(h_init, n, maxiter, t_margin):
    global _current_n
    _current_n = n
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = _get_constraints()
    try:
        result = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
        if result.x is not None:
            h_out = project_feasible(result.x[:-1], n)
            return h_out, compute_c5_from_h(h_out)
    except Exception:
        pass
    return h_init.copy(), c5_init


def make_perturbation(n, rng, style):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    if style == 0:
        n_modes = int(rng.integers(1, 5))
        for _ in range(n_modes):
            freq = int(rng.integers(1, 15))
            phase = rng.uniform(0, 2.0 * np.pi)
            amp = rng.uniform(0.01, 0.10)
            pert += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    elif style == 1:
        n_modes = int(rng.integers(2, 7))
        for _ in range(n_modes):
            freq = int(rng.integers(1, 10))
            phase = rng.uniform(0, 2.0 * np.pi)
            amp = rng.uniform(0.02, 0.12)
            pert += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    elif style == 2:
        n_modes = int(rng.integers(2, 6))
        for _ in range(n_modes):
            freq = int(rng.integers(1, 12))
            phase = rng.uniform(0, 2.0 * np.pi)
            amp = rng.uniform(0.01, 0.09)
            if rng.random() < 0.5:
                pert += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
            else:
                pert += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    elif style == 3:
        n_modes = int(rng.integers(1, 4))
        for _ in range(n_modes):
            freq = int(rng.integers(1, 8))
            phase = rng.uniform(0, 2.0 * np.pi)
            amp = rng.uniform(0.03, 0.15)
            pert += amp * np.sin(2.0 * np.pi * freq * x / n + phase) ** 2
    elif style == 4:
        pert = rng.standard_normal(n) * 0.04
        smooth = np.convolve(pert, np.ones(5) / 5.0, mode='same')
        pert = smooth
    else:
        pert = rng.standard_normal(n) * 0.03
    return pert


def transfer_resolution(h_src, n_src, n_dst):
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    f_interp = interp1d(x_old, h_src, kind='linear', fill_value="extrapolate")
    h_dst = np.clip(f_interp(x_new), 0, 1)
    return project_feasible(h_dst, n_dst)


def subgradient_step(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(c))
    offset = n - 1
    shift = k_star - offset
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if shift >= 0:
            if j >= shift and j < n:
                val += (1.0 - h[j])
            if 0 <= j - shift and j - shift < n:
                val -= h[j - shift]
        else:
            if 0 <= j and j < n + shift:
                val += (1.0 - h[j])
            if 0 <= j + (-shift) and j + (-shift) < n:
                val -= h[j - shift]
        grad[j] = val * dx
    h_new = h - alpha * grad
    return project_feasible(h_new, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        h0 = project_feasible(h0, n)
    else:
        n = 135
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)

    if time.time() - t0 < budget_s - 120:
        best_h, best_c5 = slsqp_optimize(best_h, n, maxiter=5000, t_margin=0.02)

    if time.time() - t0 < budget_s - 150:
        n_coarse = 30
        h_coarse = transfer_resolution(best_h, n, n_coarse)
        c5_coarse = compute_c5_from_h(h_coarse)
        for attempt in range(20):
            if time.time() - t0 > budget_s - 80:
                break
            style = attempt % 5
            pert = make_perturbation(n_coarse, rng, style)
            h_cp = project_feasible(h_coarse + pert, n_coarse)
            h_cn, c5_cn = slsqp_optimize(h_cp, n_coarse, maxiter=2000, t_margin=0.03)
            if compute_c5_from_h(h_cn) < c5_coarse - 1e-5:
                h_coarse = h_cn
                c5_coarse = compute_c5_from_h(h_coarse)
        h_transferred = transfer_resolution(h_coarse, n_coarse, n)
        h_new, c5_new = slsqp_optimize(h_transferred, n, maxiter=2000, t_margin=0.02)
        if c5_new < best_c5 - 1e-8:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 100:
        for attempt in range(60):
            if time.time() - t0 > budget_s - 30:
                break
            style = attempt % 5
            pert = make_perturbation(n, rng, style)
            h_pert = project_feasible(best_h + pert, n)
            h_new, c5_new = slsqp_optimize(h_pert, n, maxiter=1500, t_margin=0.02)
            if c5_new < best_c5 - 1e-9:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 80:
        for attempt in range(80):
            if time.time() - t0 > budget_s - 20:
                break
            alpha = 0.1 * (1.0 - attempt / 100.0) + 0.01
            h_sg = subgradient_step(best_h, n, alpha)
            h_sg = project_feasible(h_sg, n)
            if compute_c5_from_h(h_sg) < best_c5 - 1e-8:
                best_h = h_sg
                best_c5 = compute_c5_from_h(best_h)

    for n_up in [270, 405]:
        if time.time() - t0 > budget_s - 280:
            break
        h_up = transfer_resolution(best_h, len(best_h), n_up)
        if time.time() - t0 < budget_s - 130:
            h_new, c5_new = slsqp_optimize(h_up, n_up, maxiter=2500, t_margin=0.02)
            if c5_new < best_c5 - 1e-9:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 20:
        n_cur = len(best_h)
        for attempt in range(40):
            if time.time() - t0 > budget_s - 5:
                break
            scale = rng.uniform(0.003, 0.012)
            h_pert = project_feasible(best_h + scale * rng.standard_normal(n_cur), n_cur)
            h_new, c5_new = slsqp_optimize(h_pert, n_cur, maxiter=800, t_margin=0.01)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
