def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    sum_exp = np.sum(exp_shifted)
    smax = m + np.log(sum_exp) / beta
    weights = exp_shifted / sum_exp
    return smax, weights

def gradient_of_objective(h, beta, lambda_penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    smax, weights = smooth_max_and_weights(corr, beta)
    
    # Compute gradient of each f_d w.r.t. h[j]
    # f_d(h) = sum_{i valid} h[i] * (1 - h[i+d])
    # df_d/dh[j] = (1-h[j+d])*1[valid] - h[j-d]*1[valid']
    
    grad = np.zeros(n)
    # d = k - (n-1) for k = 0, ..., 2n-2
    for k in range(2*n - 1):
        d = k - (n + 1)  # offset
        w = weights[k] * dx  # weight for this lag in the smooth max
        # For this lag d, gradient contribution:
        # For each j where both j and j+d are valid: (1-h[j+d])
        # For each j where both j and j-d are valid: -h[j-d]
        # Combined: for j in [0, n-1]:
        #   if j+d in [0, n-1]: add (1-h[j+d])
        #   if j-d in [0, n-1]: add -h[j-d]

        # Second term: h[j+d] * (1-h[j]) contributes -h[j] to dh[j+d]
        # So dh[j] += (1-h[j+d]) [if j+d valid] + (-h[j-d]) [if j-d valid]
        
        # Vectorized:
        if d >= 0:
            # j+d valid means j <= n-1-d, i.e., j in [0, n-1-d]
            # j-d valid means j >= d, i.e., j in [d, n-1]
            n_valid1 = n - d
            grad[:n_valid1] += w * (1.0 - h[d:n])  # (1-h[j+d]) for j in [0, n-1-d]
            if d > 0:
                grad[d:] += w * (-h[:n-d])  # -h[j-d] for j in [d, n-1]

            d_abs = -d
            # j+d valid means j >= d_abs, i.e., j in [d_abs, n-1]

            grad[d_abs:] += w * (1.0 - h[:n-d_abs])  # (1-h[j+d]) for j in [d_abs, n-1]
            if d_abs > 0:
                grad[:n-d_abs] += w * (-h[d_abs:n])  # -h[j-d] for j in [0, n-1-d_abs]
    
    # Add penalty gradient for sum constraint
    grad += 2.0 * lambda_penalty * (np.sum(h) - n/2.0)
    
    return smax * dx + lambda_penalty * (np.sum(h) - n/2.0)**2, grad

def objective_and_grad(x, beta, lambda_penalty):

    h = np.clip(x, 0.0, 1.0)
    obj, grad = gradient_of_objective(h, beta, lambda_penalty)
    # Adjust gradient for clipping
    mask = (x >= 0.0) & (x <= 1.0)
    grad = grad * mask
    return obj, grad

        smax_k = full_m
    s_val = np.sum(h) - n / 2.0
    obj = smax_k * dx + lam * s_val * s_val
    grad = np.zeros(n)
    d_vals = np.arange(2 * n - 1) - (n - 1)
    for idx_k in range(len(topk_idx)):
        k = topk_idx[idx_k]
        w = dx * weights_k[idx_k]
        if w < 1e-16:
            continue
        d = d_vals[k]
        if d >= 0:
            grad[0:n - d] += w * (1.0 - h[d:n])
            if d > 0:
                grad[d:n] += w * (-h[0:n - d])
        else:
            da = -d
            grad[da:n] += w * (1.0 - h[0:n - da])
            if da > 0:
                grad[0:n - da] += w * (-h[da:n])
    grad += 2.0 * lam * s_val
    mask = (x > 0.0) & (x < 1.0)
    grad = grad * mask
    return obj, grad


def lbfgs_solve(h0, n, max_iter, beta, lam):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: obj_grad_full(x, n, beta, lam)[0],
            h0,
            lambda x: obj_grad_full(x, n, beta, lam)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50, 'maxcor': 30}
        )
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def lbfgs_solve_topk(h0, n, max_iter, beta, lam, topk_idx):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: obj_grad_topk(x, n, beta, lam, topk_idx)[0],
            h0,
            lambda x: obj_grad_topk(x, n, beta, lam, topk_idx)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50, 'maxcor': 30}
        )
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def subgradient_step(h, n, alpha, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_vals = np.sort(corr)[::-1]
    n_active = int(np.sum(top_vals > top_vals[0] - 0.05))
    n_active = max(1, min(n_active, 2 * n - 1))
    active_mask = corr > (top_vals[n_active - 1] - 0.05)
    d_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if not active_mask[k]:
            continue
        d = d_vals[k]
        w = dx / n_active
        if d >= 0:
            grad[0:n - d] += w * (1.0 - h[d:n])
            if d > 0:
                grad[d:n] += w * (-h[0:n - d])
        else:
            da = -d
            grad[da:n] += w * (1.0 - h[0:n - da])
            if da > 0:
                grad[0:n - da] += w * (-h[da:n])
    h_new = h + alpha * grad + rng.normal(0, 0.01, n)
    return project_feasible(h_new, n)


def find_topk_indices(h, n, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    topk = np.argsort(corr)[::-1][:K]
    return topk


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_directional_perturb(h, n, rng, scale):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    d = k_max - (n - 1)
    d_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    d = d_vals[k_max]
    if d >= 0:
        grad[0:n - d] += (1.0 - h[d:n])
        if d > 0:
            grad[d:n] += (-h[0:n - d])
    else:
        da = -d
        grad[da:n] += (1.0 - h[0:n - da])
        if da > 0:
            grad[0:n - da] += (-h[da:n])
    norm = np.linalg.norm(grad)
    if norm > 1e-10:
        grad = grad / norm
    h_new = h + scale * grad + rng.normal(0, scale * 0.3, n)
    return project_feasible(h_new, n)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float).copy()
            n_init = len(h_init)
            best_h = project_feasible(h_init, n_init)
            best_n = n_init
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_h = np.ones(47) * 0.5
        best_n = 47
        best_c5 = compute_c5(best_h)

    # Phase 1: Progressive beta optimization from parent
    betas_phase1 = [25, 40, 60, 100, 150, 250]
    for beta in betas_phase1:
        if time.time() - t0 > budget_s - 150:
            break
        h_opt, c5_opt = lbfgs_solve(best_h, best_n, 400, beta, 80.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Top-K focused optimization
    if time.time() - t0 < budget_s - 150:
        for K in [5, 10, 20, 40]:
            if time.time() - t0 > budget_s - 100:
                break
            topk_idx = find_topk_indices(best_h, best_n, K)
            h_opt, c5_opt = lbfgs_solve_topk(best_h, best_n, 300, 200, 80.0, topk_idx)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Multi-restart with diverse perturbations
    no_improve = 0
    max_no_improve = 50
    attempt = 0
    while time.time() - t0 < budget_s - 250 and no_improve < max_no_improve:
        attempt += 1
        r = rng.random()
        if r < 0.4:
            sigma = 0.03 + 0.08 * (0.92 ** min(no_improve, 25))
            h_start = project_feasible(best_h + rng.normal(0, sigma, best_n), best_n)
        elif r < 0.7:
            h_start = make_directional_perturb(best_h, best_n, rng, 0.08 + 0.05 * rng.random())
        elif r < 0.9:
            sigma = 0.2 + 0.15 * rng.random()
            h_start = project_feasible(best_h + rng.normal(0, sigma, best_n), best_n)
        else:
            period = rng.choice([3, 4, 5, 6, 7, 8])
            i = np.arange(best_n, dtype=float)
            h_struct = 0.5 + 0.3 * np.sin(2.0 * np.pi * i / period + rng.uniform(0, 2 * np.pi))
            h_struct = 0.6 * project_feasible(h_struct, best_n) + 0.4 * best_h
            h_start = project_feasible(h_struct, best_n)

        h_opt, c5_opt = lbfgs_solve(h_start, best_n, 250, 120, 80.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                h_sg = subgradient_step(best_h, best_n, 0.02, rng)
                h_opt2, c5_opt2 = lbfgs_solve(h_sg, best_n, 200, 120, 80.0)
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2
                    no_improve = 0

    # Phase 4: Higher resolution attempts
    for n_try in [285, 380, 475]:
        if time.time() - t0 > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = lbfgs_solve(h_interp, n_try, 250, 80, 60.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 5: Final polish with high beta
    if time.time() - t0 < budget_s - 40:
        for beta_final in [300, 500, 800]:
            if time.time() - t0 > budget_s - 20:
                break
            h_opt, c5_opt = lbfgs_solve(best_h, best_n, 500, beta_final, 100.0)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 6: One more round of top-K refinement
    if time.time() - t0 < budget_s - 15:
        for K in [3, 5, 8]:
            if time.time() - t0 > budget_s - 5:
                break
            topk_idx = find_topk_indices(best_h, best_n, K)
            h_opt, c5_opt = lbfgs_solve_topk(best_h, best_n, 300, 500, 100.0, topk_idx)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
