import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Euclidean projection onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        tau = 0.5 * (lo + hi)
        h_p = np.clip(h - tau, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def smooth_max_corr_grad(h, alpha):
    """Smooth max of correlations and its gradient."""
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")  # length 2n-1
    
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient computation
    # shifts[k] = k - (n-1) for k=0,...,2n-2
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    # For each j in 0..n-1 and each shift s:
    # d(corr[n-1+s])/d(h[j]) = (1-h[j-s])*I(0<=j-s<n) - h[j+s]*I(0<=j+s<n)
    
    # Vectorized:
    J = np.arange(n)[:, None]  # (n,1)
    S = shifts[None, :]  # (1, 2n-1)
    
    idx_a = J - S  # j - s, need 0 <= idx_a < n
    idx_b = J + S  # j + s, need 0 <= idx_b < n
    
    valid_a = (idx_a >= 0) & (idx_a < n)
    valid_b = (idx_b >= 0) & (idx_b < n)
    
    # Clamp for safe indexing
    idx_a_c = np.clip(idx_a, 0, n - 1)
    idx_b_c = np.clip(idx_b, 0, n - 1)
    
    term_a = np.where(valid_a, 1.0 - h[idx_a_c], 0.0)
    term_b = np.where(valid_b, h[idx_b_c], 0.0)
    
    grad = (w[None, :] * (term_a - term_b)).sum(axis=1)
    
    return obj, grad

def slsqp_optimize(h0, target_sum, alpha, maxiter, ftol):
    n = len(h0)
    
    def objective(x):
        h = np.clip(x, 0, 1)
        obj, _ = smooth_max_corr_grad(h, alpha)
        return obj
    
    def grad(x):
        h = np.clip(x, 0, 1)
        _, g = smooth_max_corr_grad(h, alpha)
        return g
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    def constraint_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def projected_gradient(h0, target_sum, alpha, n_iters, lr_init, momentum=0.9):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        obj, grad = smooth_max_corr_grad(h, alpha)
        
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()
        
        # Simple adaptive lr
        obj_curr = compute_c5(h)
        if c5 > obj_curr * 1.01:
            lr *= 0.7
            v *= 0.5
        elif c5 < obj_curr * 0.99:
            lr = min(lr * 1.05, lr_init * 3)
        
        h = h_new
    
    return h_best, c5_best

def multi_start_optimize(h0, n, target_sum, rng, budget_remaining):
    """Try multiple alphas and methods from this starting point."""
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # SLSQP with increasing alpha
    for alpha in [30.0, 80.0, 200.0]:
        if time.time() - t_global > 1300 - 30:
            break
        h_opt = slsqp_optimize(best_h, target_sum, alpha, 300, 1e-10)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Projected gradient
    for alpha in [50.0, 150.0]:
        if time.time() - t_global > 1300 - 30:
            break
        h_pg, c5_pg = projected_gradient(best_h, target_sum, alpha, 500, 0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    return best_h, best_c5

def perturb_and_reopt(h, n, target_sum, rng, n_perturb=5):
    """Perturb the solution and re-optimize to escape local minima."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_perturb):

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

t_global = 0

import numpy as np
from scipy.optimize import minimize
import time

GLOBAL_ALPHA = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        tau = 0.5 * (lo + hi)
        h_p = np.clip(h - tau, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def smooth_max_corr_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    
    shifts = np.arange(2 * n - 1) - (n - 1)
    J = np.arange(n)[:, None]
    S = shifts[None, :]
    idx_a = J - S
    idx_b = J + S
    valid_a = (idx_a >= 0) & (idx_a < n)
    valid_b = (idx_b >= 0) & (idx_b < n)
    idx_a_c = np.clip(idx_a, 0, n - 1)
    idx_b_c = np.clip(idx_b, 0, n - 1)
    term_a = np.where(valid_a, 1.0 - h[idx_a_c], 0.0)
    term_b = np.where(valid_b, h[idx_b_c], 0.0)
    grad = (w[None, :] * (term_a - term_b)).sum(axis=1)
    return obj, grad

def obj_fn(x):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_corr_grad(h, GLOBAL_ALPHA)[0]

def grad_fn(x):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_corr_grad(h, GLOBAL_ALPHA)[1]

    GLOBAL_TARGET_SUM = target_sum

def projected_momentum(h0, target_sum, alpha, n_iters, lr_init, momentum=0.8):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    c5_prev = c5_best
    
    for it in range(n_iters):
        obj, grad = smooth_max_corr_grad(h, alpha)
        v = momentum * v + lr * grad  # gradient ascent on obj (we minimize)
        # Wait, we want to minimize, so step = -lr * grad
        v = momentum * v - lr * grad
        h_candidate = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_candidate)

            h_best = h_candidate.copy()
        
        # Adaptive lr

            lr = max(lr * 0.7, 1e-6)
            v *= 0.5
        elif c5_new < c5_prev * 0.995:
            lr = min(lr * 1.3, lr_init * 5.0)
        
        h = h_candidate
        c5_prev = compute_c5(h)
    
    return h_best, c5_best

def local_search(h0, target_sum, rng, n_rounds=200):

    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
