```python
def sla_step(h_current, n, max_lp_iter=1000):
    """One step of successive linear approximation."""
    dx = 2.0 / n
    c_vals = np.correlate(h_current, 1.0 - h_current, mode="full") * dx
    t_current = np.max(c_vals)
    
    # Compute gradients of each c_k with respect to h
    # c_k = sum_i h[i] * (1-h[i+k]) for valid i
    # dc_k/dh[j] = (1-h[j-k]) - h[j+k] (with boundary handling)
    
    n_corr = 2 * n - 1  # number of correlation lags
    c_offsets = np.arange(n_corr) - (n - 1)  # lags from -(n-1) to n-1
    
    # Build the LP
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize: t (i.e., [0, ..., 0, 1])
    # Subject to:
    #   For each k: sum_j (dc_k/dh_j) * h_j - t ≤ -c_k(h*)
    #   sum(h) = n/2
    #   0 ≤ h_j ≤ 1
    
    # Compute gradient matrix A where A[k, j] = dc_k/dh_j
    A = np.zeros((n_corr, n))
    for k_idx, lag in enumerate(c_offsets):
        for j in range(n):
            grad = 0.0
            if 0 <= j - lag <= n - 1:
                grad += (1.0 - h_current[j - lag])
            if 0 <= j + lag <= n - 1:
                grad -= h_current[j + lag]
            A[k_idx, j] = grad
    
    # LP: min t
    # s.t. A @ h - t ≤ -c_vals  (i.e., A @ h - [0,...,0,1] @ [h, t] ≤ -c_vals)
    #      sum(h) = n/2
    #      0 ≤ h ≤ 1
    #      t ≥ 0 (or some lower bound)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.ones((n_corr, 1))])
    b_ub = -c_vals
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    bounds = [(0, 1)] * n + [(0, None)]

    if res.success:
        h_new = res.x[:n]
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5_from_h(h_new)
    return h_current, compute_c5_from_h(h_current)
```

```python
            if total < 1e-15:
                break
            h[mask] += diff * h[mask] / total

def compute_c5_and_grads(h, n):
    """Compute C5 and the gradient matrix A where A[k,j] = dc_k/dh_j."""
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, mode="full") * dx
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)
    
    # Compute gradient: A[k, j] = dc_k/dh_j
    # For lag L: c_L = sum_i h[i]*(1-h[i+L]) for valid i
    # dc_L/dh[j] = (1-h[j-L]) [if j-L valid] - h[j+L] [if j+L valid]
    A = np.zeros((n_corr, n))
    for k_idx, L in enumerate(lags):
        # Term 1: (1-h[j-L]) for j-L in [0, n-1]
        j_min1 = max(0, -L)
        j_max1 = min(n - 1, n - 1 - L)
        if j_min1 <= j_max1:
            A[k_idx, j_min1:j_max1+1] += (1.0 - h[j_min1-L:j_max1+1-L])
        # Term 2: -h[j+L] for j+L in [0, n-1]
        j_min2 = max(0, -L)
        j_max2 = min(n - 1, n - 1 - L)
        if j_min2 <= j_max2:
            A[k_idx, j_min2:j_max2+1] -= h[j_min2+L:j_max2+1+L]
    
    return c_vals, A

    c_vals = np.correlate(h, 1.0 - h, mode="full") * dx

    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)

    for k_idx, L in enumerate(lags):
        # dc_L/dh[j] = (1-h[j-L]) - h[j+L] with boundary handling
        for j in range(n):
            g = 0.0
            if 0 <= j - L:
                g += (1.0 - h[j - L])
            if 0 <= j + L < n:
                g -= h[j + L]
            A[k_idx, j] = g

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    if res.success:
        h_new = project_feasible(res.x[:n], n)
        return h_new, compute_c5_from_h(h_new)
```

```python
    for k_idx in range(n_corr):
        L = k_idx - (n - 1)

        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if L >= 0:
            # j-L >= 0 means j >= L; j-L < n means j < n+L (always true since j < n)
            term1[L:] = 1.0 - h[:n-L]
            # j+L < n means j < n-L
            term2[:n-L] = h[L:]

            # j-L >= 0 always (since L < 0, j-L = j+|L| >= 0); j-L < n means j < n+L
            term1[:n+L] = 1.0 - h[-L:]
            # j+L >= 0 means j >= -L; j+L < n always
            term2[-L:] = h[:n+L]
        A[k_idx, :] = term1 - term2
    return A
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
```
