    """Compute gradient of c_d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            # c_d = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
            # d c_d / d h[i] = (1-h[i+d]) for i < n-d
            # d c_d / d h[i+d] = -h[i] for i+d >= d, i.e., for indices d to n-1
            valid = np.arange(n - d)
            grad[valid] += (1 - h[valid + d])
            grad[d:] += (-h[:n-d])
    else:
        e = -d
        if e < n:
            # c_d = sum_{i=e}^{n-1} h[i] * (1-h[i-e])
            valid = np.arange(e, n)
            grad[valid] += (1 - h[valid - e])
            grad[:n-e] += (-h[e:])
    return grad

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx), c * dx

def compute_all_correlations(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            max_step = 1.0 - h[mask].max()
            h[mask] += min(step, max_step)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            max_step = h[mask].max()
            h[mask] -= min(step, max_step)
    return np.clip(h, 0.0, 1.0)

def gradient_at_lag(h, d):
    """Vectorized gradient of correlation at shift d."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            m = n - d
            grad[:m] += (1.0 - h[d:])
            grad[d:] += (-h[:m])
    else:
        e = -d
        if e < n:
            m = n - e
            grad[e:] += (1.0 - h[:m])
            grad[:m] += (-h[e:])
    return grad

def subgradient_descent(h, max_iters=50000, lr0=0.01, momentum=0.9):
    """Subgradient descent with momentum."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)[0]
    v = np.zeros(n)
    beta = momentum
    
    for it in range(max_iters):
        c_vals = compute_all_correlations(h)
        c5_cur = np.max(c_vals)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Find active lags (within 5% of max)
        threshold = best_c5 * 0.99
        active = np.where(c_vals >= threshold)[0]
        
        # Compute average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            grad += gradient_at_lag(h, d)
        grad /= len(active) * dx
        
        # Nesterov momentum step
        v = beta * v - lr0 * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        h = h_new
        
        # Adaptive learning rate
        c5_new = compute_c5_from_h(h)[0]
        if c5_new > best_c5 * 1.05:
            lr0 *= 0.95
        elif it > 100 and c5_new < best_c5:
            lr0 = min(lr0 * 1.02, 0.1)
    
    return best_h, best_c5

def objective_slsqp(x):
    return x[-1]

def constraint_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def constraint_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_polish(h, maxiter=2000, t_margin=0.02):
    n = len(h)
    c5 = compute_c5_from_h(h)[0]
    x0 = np.concatenate([h, [c5 + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5 * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_ineq},
        {'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}
    ]

                      constraints=constraints,

    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)[0]
    return h, c5

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            max_step = 1.0 - h[mask].max()
            h[mask] += min(step, max_step)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            max_step = h[mask].max()
            h[mask] -= min(step, max_step)

    else:

        return x[-1]

        t_ = x[-1]

        dx_ = 2.0 / n_
        c_ = np.correlate(h_, 1.0 - h_, mode="full")

        h_ = x[:-1]
        return np.sum(h_) - len(h_) / 2.0

        n_ = len(x) - 1

        jac[:n_] = 1.0

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_h(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
            if not m.any():
                break
            s = diff / m.sum()
            h[m] += min(s, 1.0 - h[m].max())
        else:
            m = h > 0.0
            if not m.any():
                break
            s = -diff / m.sum()
            h[m] -= min(s, h[m].max())
    return np.clip(h, 0.0, 1.0)


def grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if 0 <= d < n:
        m = n - d
        g[:m] += 1.0 - h[d:]
        g[d:] -= h[:m]
    elif -n < d < 0:
        e = -d
        m = n - e
        g[e:] += 1.0 - h[:m]
        g[:m] -= h[e:]
    return g


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    return t - np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def slsqp_eq(x):
    return x[:-1].sum() - (len(x) - 1) / 2.0


def slsqp_eqj(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def slsqp(h, maxit=2000, marg=0.02):
    n = len(h)
    c5 = compute_c5(h)
    x0 = np.append(h, c5 + marg)
    bnds = [(0.0, 1.0)] * n + [(max(0.001, c5 * 0.7), 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eqj}
    ]
    r = minimize(slsqp_obj, x0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14})
    if r.success or r.x is not None:
        hh = project_h(r.x[:-1], n)
        return hh, compute_c5(hh)
    return h, c5


def subgrad_descent(h, n_iters, lr0, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    for it in range(n_iters):
        cvals = compute_corrs(h)
        c5 = cvals.max()
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        # Use top 3 active lags
        idx = np.argsort(cvals)[-3:]
        g = np.zeros(n)
        for k in idx:
            d = k - (n - 1)
            g += grad_lag(h, d)
        g /= 3.0
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        v = momentum * v - lr * g
        h = project_h(h + v, n)
        # Adaptive lr
        c5_new = compute_c5(h)
        if c5_new > best_c5 * 1.01:
            lr *= 0.9
        elif c5_new < best_c5:
            lr = min(lr * 1.01, 0.05)
    return best, best_c5


def coord_desc(h, n_steps, step_size):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    for s in range(n_steps):
        i = s % n
        # Try small increase
        if best[i] < 1.0:
            ht = best.copy()
            step = min(step_size, 1.0 - ht[i])
            ht[i] += step
            ht = project_h(ht, n)
            c = compute_c5(ht)
            if c < best_c5:

        # Try small decrease
        if best[i] > 0.0:
            ht = best.copy()
            step = min(step_size, ht[i])
            ht[i] -= step
            ht = project_h(ht, n)
            c = compute_c5(ht)
            if c < best_c5:
                best, best_c5 = ht, c
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_h(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_s = np.clip(h + mid, 0.0, 1.0)
        if h_s.sum() > target:
            hi = mid
        else:
            lo = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_out.sum()
    if abs(diff) > 1e-10:
        for _ in range(200):
            if diff > 0:
                m = h_out < 1.0
            else:
                m = h_out > 0.0
            if not m.any():
                break
            step = diff / m.sum()
            if diff > 0:
                step = min(step, 1.0 - h_out[m].max())
            else:
                step = -min(-step, h_out[m].max())
            h_out[m] += step
            diff = target - h_out.sum()
            if abs(diff) < 1e-14:
                break
    return np.clip(h_out, 0.0, 1.0)


def grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if 0 <= d < n:
        m = n - d
        g[:m] += 1.0 - h[d:]
        g[d:] -= h[:m]
    elif -n < d < 0:
        e = -d
        m = n - e
        g[e:] += 1.0 - h[:m]
        g[:m] -= h[e:]
    return g


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    return t - np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def slsqp_eq(x):
    return x[:-1].sum() - (len(x) - 1) / 2.0


def slsqp_eqj(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def slsqp(h, maxit=2000, marg=0.02):
    n = len(h)
    c5 = compute_c5(h)
    x0 = np.append(h, c5 + marg)
    bnds = [(0.0, 1.0)] * n + [(max(0.001, c5 * 0.7), 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eqj}
    ]
    r = minimize(slsqp_obj, x0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14})
    if r.x is not None:
        hh = project_h(r.x[:-1], n)
        return hh, compute_c5(hh)
    return h, c5


def subgrad_phase(h, n_iters, lr0=0.008, momentum=0.82):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    for it in range(n_iters):
        cvals = compute_corrs(h)
        c5 = float(cvals.max())
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        top_k = min(5, len(cvals))
        idx = np.argsort(cvals)[-top_k:]
        g = np.zeros(n)
        for k in idx:
            d = k - (n - 1)
            g += grad_lag(h, d)
        g /= top_k
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        v = momentum * v - lr * g
        h = project_h(h + v, n)
        c5_new = compute_c5(h)
        if c5_new > best_c5 * 1.02:
            lr *= 0.9
        elif c5_new < best_c5 - 1e-6:
            lr = min(lr * 1.005, 0.04)
    return best, best_c5


def coord_desc(h, n_steps, step0=0.03):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for s in range(n_steps):
        i = s % n
        improved = False
        for direction in [step, -step]:
            ht = best.copy()
            val = ht[i] + direction
            if 0.0 <= val <= 1.0:
                ht[i] = val
                ht = project_h(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-12:
                    best = ht
                    best_c5 = c
                    improved = True
                    break
        if not improved and s > 0 and s % n == 0:
            step = max(step * 0.85, 0.001)
    return best, best_c5
