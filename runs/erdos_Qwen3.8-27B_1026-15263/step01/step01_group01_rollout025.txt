```python
import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def objective_smooth(h, n, beta):
    corr_vals = compute_all_corr(h)
    return smooth_max(corr_vals, beta)

def gradient_corr_at_lag(h, k):
    """Compute gradient of correlation at lag k with respect to h.
    k is the lag (can be negative), 0-indexed from the center.
    """
    n = len(h)
    g = np.zeros(n)
    # corr[k] = sum_i h[i] * (1-h[i+k]) for valid i, i+k
    # d/dh[j] = (1-h[j+k]) if j+k in [0,n-1] minus h[j-k] if j-k in [0,n-1]
    # Term 1: for j where j+k is valid
    if k >= 0:
        # j ranges from 0 to n-1-k
        g[:n-k] += (1.0 - h[k:])  # (1-h[j+k]) for j=0..n-1-k
        # j ranges from k to n-1 (where j-k is valid)
        g[k:] -= h[:n-k]  # -h[j-k] for j=k..n-1
    else:
        kk = -k
        # j ranges from 0 to n-1-kk
        g[:n-kk] += (1.0 - h[kk:])  # (1-h[j+k]) where j+k = j-kk
        # j ranges from kk to n-1
        g[kk:] -= h[:n-kk]
    return g

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        step = diff / n
        h = np.clip(h + step, 0.0, 1.0)

            mask = h < 1.0
            if np.any(mask):
                idx = np.where(mask)[0]
                h[idx[0]] = min(h[idx[0]] + diff, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                idx = np.where(mask)[0]
                h[idx[0]] = max(h[idx[0]] + diff, 0.0)
    return h

    """One subgradient step: minimize max_k corr[k]*dx"""
    corr_vals = compute_all_corr(h)
    k_star = int(np.argmax(corr_vals))
    # k_star is the index in the full correlation array
    lag = k_star - (n - 1)
    g = gradient_corr_at_lag(h, lag)
    h_new = h - alpha * g
    h_new = project_feasible(h_new, n)
    return h_new, k_star, corr_vals[k_star]

    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    alpha = alpha0
    for i in range(max_iter):

        if c5_now < best_c5:

            best_h = h.copy()
        alpha *= cooling
```

```python
        p = probs[k_idx]
        if p < 1e-12:
```

```python
        if lag >= 0:
            g[:n-lag] += p * (1.0 - h[lag:])
            g[lag:] -= p * h[:n-lag]
        else:
            kk = -lag
            g[kk:] += p * (1.0 - h[:n-kk])
            g[:n-kk] -= p * h[kk:]
    return g * dx
```

```python
    h = np.asarray(h, dtype=float).copy()

    for _ in range(300):
        diff = n / 2.0 - np.sum(h)

    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff > -1e-15:
                    break

        m = np.max(corr_vals)
        shifted = np.clip(beta * (corr_vals - m), -500.0, 500.0)
        return m + np.log(np.sum(np.exp(shifted))) / beta

        corr_vals = np.correlate(x, 1.0 - x, mode="full") * (2.0/n)
        m = np.max(corr_vals)
        shifted = np.clip(beta * (corr_vals - m), -500.0, 500.0)

        return np.sum(x) - n / 2.0
```

One final consideration: the `project_feasible` function. My current implementation does:

2. Iteratively adjust to meet the sum constraint

The iterative adjustment (adding diff/n to all elements) is slow to converge if some elements are saturated. A better approach:
1. Clip to [0,1]
2. Compute the deficit/surplus
3. Distribute only among non-saturated elements

```python
def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break
    return h
```

This should work well. The iterative part handles most of the adjustment, and the final correction handles the remaining tiny error.

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_all_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def grad_corr_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        kk = -lag
        if kk < n:
            g[kk:] += (1.0 - h[:n - kk])
            g[:n - kk] -= h[kk:]
    return g


def smooth_max_value(corr_vals, beta):
    m = np.max(corr_vals)
    shifted = np.clip(beta * (corr_vals - m), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    return m + np.log(exp_shifted.sum()) / beta


def smooth_max_probs(corr_vals, beta):
    m = np.max(corr_vals)
    shifted = np.clip(beta * (corr_vals - m), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    return exp_shifted / exp_shifted.sum()


def full_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_vals = np.correlate(h, 1.0 - h, mode="full") * dx
    probs = smooth_max_probs(corr_vals, beta)
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        p = probs[k_idx]
        if p < 1e-14:
            continue
        lag = k_idx - (n - 1)
        g += p * grad_corr_lag(h, lag)
    return g * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break
    return h


def slsqp_obj(x, n, beta):
    corr_vals = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    return smooth_max_value(corr_vals, beta)


def slsqp_grad(x, n, beta):
    return full_smooth_gradient(x, beta)


def sum_eq_fun(x, n):
    return np.sum(x) - n / 2.0


def sum_eq_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h0, n, beta, max_iter=200, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]

    def obj(x):
        return slsqp_obj(x, n, beta)

    def obj_grad(x):
        return slsqp_grad(x, n, beta)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def subgradient_refine(h0, n, max_iter=5000, alpha0=0.003, cooling=0.9998, top_m=5):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    alpha = alpha0
    no_improve = 0

    for i in range(max_iter):
        corr_vals = compute_all_corr_scaled(h)
        max_val = corr_vals.max()

        if top_m <= len(corr_vals):
            top_indices = np.argsort(corr_vals)[-top_m:]
        else:
            top_indices = np.arange(len(corr_vals))

        weights = np.zeros(len(corr_vals))
        for idx in top_indices:
            weights[idx] = np.exp(40.0 * (corr_vals[idx] - max_val))
        wsum = weights.sum()
        if wsum < 1e-30:
            weights[top_indices[0]] = 1.0
            wsum = 1.0
        weights /= wsum

        g = np.zeros(n)
        for idx in top_indices:
            w = weights[idx]
            if w < 1e-14:
                continue
            lag = idx - (n - 1)
            g += w * grad_corr_lag(h, lag)
        g *= (2.0 / n)

        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new
        alpha *= cooling

        if no_improve > 3000:
            break

    return best_h, best_c5


def lbfgs_penalty_optimize(h0, n, beta, max_iter=300, penalty=100.0):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr_vals = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        sm = smooth_max_value(corr_vals, beta)
        penalty_term = 0.5 * penalty * (np.sum(x) - n / 2.0) ** 2
        return sm + penalty_term

    def obj_grad(x):
        g = full_smooth_gradient(x, beta)
        penalty_grad = penalty * (np.sum(x) - n / 2.0) * np.ones(n)
        return g + penalty_grad

    try:
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new)


def make_perturbed(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def elapsed_time(t0):
    return time.time() - t0
```
