from scipy.optimize import differential_evolution, minimize
from scipy.signal import fftconvolve
from time import time
import math

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    # correlate(h, 1-h, 'full') == convolve(h, flip(1-h), 'full')
    c = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return float(np.max(c) * dx)

def compute_c5_vector(h):
    """Return the full correlation vector times dx"""
    n = len(h)
    dx = 2.0 / n
    c = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask.astype(float)), room.min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]

        active_k = target_k

        return x[-1]

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            free = h[mask]
            step = min(diff / len(free), (1.0 - free).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            free = h[mask]
            step = min(-diff / len(free), free.min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

        return x[-1]

        h = x[:-1]
        t = x[-1]

        return t - c[active_k]

        return np.sum(x[:-1]) - n / 2.0

        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j

    bounds = [(0.0, 1.0)] * n + [(c5_init * 0.3, 1.0)]

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5(h_out)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            free = h[mask]
            step = min(diff / len(free), (1.0 - free).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            free = h[mask]
            step = min(-diff / len(free), free.min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def make_sin_bump(n, p=2.0):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / (2 * n)
    h = np.sin(np.pi * x / 2.0) ** p
    return project_feasible(h, n)

        h = h + a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h_init, n, maxiter=3000):
    dx = 2.0 / n
    c5_init = compute_c5(h_init)
    c_vec = compute_c5_vec(h_init)
    max_c = np.max(c_vec)
    active_k = np.where(c_vec >= 0.96 * max_c)[0]
    if len(active_k) < 2:
        active_k = np.where(c_vec >= 0.5 * max_c)[0]

    def obj_t(x):
        return x[-1]

    def ineq_active(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - c[active_k]

    def eq_sum(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j

    x0 = np.concatenate([h_init, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_active},
        {'type': 'eq', 'fun': eq_sum, 'jac': eq_jac}
    ]
    res = minimize(obj_t, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_out = project_feasible(res.x[:-1], n)
        c5_out = compute_c5(h_out)
        return h_out, c5_out
    return h_init, c5_init

                c5_hi = compute_c5(h_hi)
                if c5_hi < best_c5 + 0.003:
                    h_hi, c5_hi = slsqp_refine(h_hi, n_hi, maxiter=3000)
                    if c5_hi < best_c5:
                        best_h = h_hi
                        best_c5 = c5_hi

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
