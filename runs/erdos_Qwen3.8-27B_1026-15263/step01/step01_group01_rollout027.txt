def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            # Water-fill: increase from the bottom
            idx = np.where(mask)[0]
            # Sort by h value, increase smallest first
            order = idx[np.argsort(h[idx])]
            per_val = diff / len(order)
            h[order] = np.minimum(h[order] + per_val, 1.0)
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            order = idx[np.argsort(-h[idx])]  # decrease largest first
            per_val = diff / len(order)  # diff is negative
            h[order] = np.maximum(h[order] + per_val, 0.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def solve_1d_max_of_lines(consts, slopes, lo=0.0, hi=1.0, tol=1e-12):
    """Find v in [lo, hi] minimizing max_j (consts[j] + slopes[j]*v)."""
    # Filter out zero-slope entries (they just constrain the minimum value)
    zero_mask = np.abs(slopes) < 1e-15
    if np.any(zero_mask):
        zero_max = np.max(consts[zero_mask])
        # The answer is at least zero_max
    
    pos_mask = slopes > 1e-15
    neg_mask = slopes < -1e-15
    
    # Binary search on t (the value of the max)
    # For a given t, v must satisfy:
    #   v <= (t - consts[j])/slopes[j] for all j with slopes[j] > 0
    #   v >= (t - consts[j])/slopes[j] for all j with slopes[j] < 0
    #   lo <= v <= hi
    
    def feasible(t):
        v_lo = lo
        v_hi = hi
        if np.any(neg_mask):
            bounds = (t - consts[neg_mask]) / slopes[neg_mask]
            v_lo = max(v_lo, np.max(bounds))
        if np.any(pos_mask):
            bounds = (t - consts[pos_mask]) / slopes[pos_mask]
            v_hi = min(v_hi, np.min(bounds))
        return v_lo <= v_hi, v_lo, v_hi
    
    # Find bounds for t
    t_lo = max(lo, np.max(consts + slopes * np.array([lo, hi])))  # rough lower bound
    # Actually t_lo should be the minimum possible max value
    # At v=0: max_j consts[j]
    # At v=1: max_j (consts[j] + slopes[j])
    # The minimum is somewhere in between
    t_lo = min(np.max(consts), np.max(consts + slopes))
    t_hi = max(np.max(consts), np.max(consts + slopes))
    
    # Also account for zero-slope entries
    if np.any(zero_mask):
        t_lo = max(t_lo, np.max(consts[zero_mask]))
    
    # Binary search
    for _ in range(80):
        t_mid = (t_lo + t_hi) / 2
        ok, v_lo_opt, v_hi_opt = feasible(t_mid)
        if ok:
            t_hi = t_mid
            best_v = (v_lo_opt + v_hi_opt) / 2
        else:
            t_lo = t_mid
            best_v = np.clip(v_lo_opt, lo, hi) if not np.isinf(v_lo_opt) else lo
        if t_hi - t_lo < tol:
            break
    
    # Final: find the v at t_hi
    ok, v_lo_opt, v_hi_opt = feasible(t_hi)
    if ok:
        best_v = (v_lo_opt + v_hi_opt) / 2
    else:
        # Shouldn't happen, but fallback
        best_v = np.clip((lo + hi) / 2, lo, hi)
    
    return np.clip(best_v, lo, hi)

def coordinate_descent_em(h, n, max_sweeps=200, tol=1e-10, verbose=False):
    """Coordinate descent with exact max evaluation."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        # corr has length 2n-1, corr[k] corresponds to lag j = k - (n-1)
        
        improved = False
        for i in range(n):
            # Compute slopes and constants for all lags
            # corr[k] = sum valid h[n+k-(n-1)] * (1-h[n])
            # Let j = k - (n-1), so corr[k] = corr_j
            # h[i] appears in corr_j when:
            #   As h[n+j] with n+j=i, so n=i-j: valid if 0<=i-j<n
            #   As (1-h[n]) with n=i: valid if 0<=i+j<n (and 0<=i<n always)
            
            j_range = np.arange(-(n-1), n)  # all possible lags j
            k_range = j_range + (n-1)  # corresponding indices in corr array
            
            A = ((i - j_range >= 0) & (i - j_range < n)).astype(float)  # h[i] as h[n+j]
            B = ((i + j_range >= 0) & (i + j_range < n)).astype(float)  # h[i] as (1-h[n])
            
            # c1 = h[i-j] where valid, 0 otherwise
            idx_c1 = i - j_range
            c1 = np.where((idx_c1 >= 0) & (idx_c1 < n), h[np.clip(idx_c1, 0, n-1)], 0.0)
            c1 = c1 * A  # zero out where not valid
            
            # c2 = h[i+j] where valid, 0 otherwise
            idx_c2 = i + j_range
            c2 = np.where((idx_c2 >= 0) & (idx_c2 < n), h[np.clip(idx_c2, 0, n-1)], 0.0)
            c2 = c2 * B  # zero out where not valid
            
            # Current contribution of h[i] to corr_j:
            # A*h[i]*(1-c1) + B*c2*(1-h[i])
            current_contrib = A * h[i] * (1.0 - c1) + B * c2 * (1.0 - h[i])
            
            # Base correlation (without h[i]'s contribution)
            base = corr[k_range] - current_contrib
            
            # corr_j as function of v = h[i]:
            # corr_j(v) = base + B*c2 + v*(A*(1-c1) - B*c2)
            consts = base + B * c2
            slopes = A * (1.0 - c1) - B * c2
            
            # Solve 1D problem
            v_opt = solve_1d_max_of_lines(consts, slopes)
            
            if abs(v_opt - h[i]) > 1e-12:
                h[i] = v_opt
                improved = True
        
        # Project onto sum constraint
        h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if not improved or (sweep > 5 and abs(best_c5 - c5) < tol):
            # Check if we're making progress
            if c5 >= best_c5 - tol:
                # Try a few more sweeps
                pass
            if sweep > 50 and c5 >= best_c5 - 1e-12:
                break
    
    return best_h, best_c5
