    # Phase 7: Second DE round on best resolution
    if elapsed() < budget_s - 300:
        h_de2, c5_de2 = do_de(n_best, seed + 13, 80, 25, t0, budget_s)
        if h_de2 is not None:
            h_mix = project_feasible(0.4 * best_h + 0.6 * h_de2, n_best)
            h_new, c5_new = do_slsqp(h_mix, 1500)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)
            if c5_de2 < best_c5 - 1e-10:
                best_h = h_de2
                best_c5 = c5_de2
                n_best = len(best_h)
                if elapsed() < budget_s - 100:
                    best_h, best_c5 = do_slsqp(best_h, 1500)
                    n_best = len(best_h)

    # Phase 8: Fine-tuning
    if elapsed() < budget_s - 40:
        for _ in range(20):
            if elapsed() > budget_s - 15:
                break
            h_pert = best_h + 0.003 * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = do_slsqp(h_pert, 600)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)

    # Phase 9: Final SLSQP polish
    if elapsed() < budget_s - 20:
        best_h, best_c5 = do_slsqp(best_h, 500)
        n_best = len(best_h)

    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_best
