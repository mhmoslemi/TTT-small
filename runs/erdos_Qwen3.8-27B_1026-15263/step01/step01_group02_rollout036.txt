def compute_grad_c(h):
    """Compute gradients of all c(k) w.r.t. h.
    Returns array of shape (len(c), n) where c has length 2n-1.
    grad[k, j] = dc(k)/dh[j] * dx
    """
    n = len(h)
    dx = 2.0 / n

    # where s ranges from 0 to 2n-2
    # shift = s - (n-1) ranges from -(n-1) to (n-1)

    # c(s) = sum over i where 0<=i<n and 0<=i+d<n of h[i]*(1-h[i+d])
    # = sum h[i] - sum h[i]*h[i+d] over valid ranges
    # 
    # dc(s)/dh[j] = sum of contributions:
    # From h[j] as the first factor: (1-h[j+d]) for valid j+d
    # From h[j] as the second factor (i.e., i+d=j, so i=j-d): -h[j-d] for valid j-d
    
    # Let's build the gradient matrix
    # Actually this is O(n^2) which for n=135 is about 18000 - manageable
    
    shifts = np.arange(-(n-1), n)  # length 2n-1
    n_shifts = len(shifts)
    
    grad = np.zeros((n_shifts, n))
    for idx, d in enumerate(shifts):
        # Valid i: 0 <= i < n and 0 <= i+d < n
        # So i ranges from max(0, -d) to min(n, n-d) - 1
        i_start = max(0, -d)
        i_end = min(n, n - d)
        if i_start >= i_end:

        # For j in [i_start, i_end):
        # contribution from h[j] as first factor: (1 - h[j+d])
        for j in range(i_start, i_end):
            grad[idx, j] = (1.0 - h[j + d]) * dx
        # For j where j-d is valid (i.e., j = i+d for valid i):
        # j ranges from i_start+d to i_end+d-1 = max(0,-d)+d to min(n,n-d)+d-1
        # = max(d, 0) to min(n+d, n) - 1
        j_start2 = max(0, d)
        j_end2 = min(n, n + d)
        if j_start2 >= j_end2:
            continue
        for j in range(j_start2, j_end2):

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

            break

            mask = h < 1.0
            if not np.any(mask):
                break
            # Distribute proportionally to available room
            room = 1.0 - h[mask]
            total_room = room.sum()
            h[mask] += diff * room / total_room
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            # Distribute proportionally to available mass
            mass = h[mask]
            total_mass = mass.sum()
            h[mask] -= (-diff) * mass / total_mass

def smoothed_max(c, beta):
    """Compute smoothed max using log-sum-exp trick"""
    max_c = np.max(c)
    if max_c < -1000:
        return max_c
    shifted = c - max_c
    exp_vals = np.exp(beta * shifted)
    return max_c + np.log(np.sum(exp_vals)) / beta

    max_c = np.max(c)
    shifted = c - max_c
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def smoothed_max_val(c, beta):
    max_c = np.max(c)
    if max_c < -500:
        return max_c
    shifted = c - max_c
    log_sum = np.log(np.sum(np.exp(beta * shifted)))
    return max_c + log_sum / beta


def smoothed_max_weights(c, beta):
    max_c = np.max(c)
    if max_c < -500:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    shifted = c - max_c
    e = np.exp(beta * shifted)
    return e / np.sum(e)

    n = len(h)

    obj = smoothed_max_val(c, beta)

        for scale in [0.01, 0.02, 0.005]:
            h_try = best_h + scale * direction
            h_try = project_feasible(h_try, n_dir)
            c5_try = compute_c5_from_h(h_try)
            if c5_try < best_c5 - 1e-10:
                best_h = h_try
                best_c5 = c5_try
                break
        if time.time() > t_limit_dir:
            break
        h_pert = best_h + 0.015 * rng.standard_normal(n_dir)
        h_pert = project_feasible(h_pert, n_dir)
        h_res, _ = projected_grad_descent(h_pert, 200, 100, n_dir / 2.0, t0, t_limit_dir)
        c5_res = compute_c5_from_h(h_res)
        if c5_res < best_c5 - 1e-10:
            best_h = h_res
            best_c5 = c5_res

    t_limit_res = t0 + budget_s * 0.95
    for n_try in [270, 405, 200]:
        if time.time() > t_limit_res:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0.0, 1.0)
        h_interp = project_feasible(h_interp, n_try)
        c5_interp = compute_c5_from_h(h_interp)
        if c5_interp < best_c5:
            best_h = h_interp
            best_c5 = c5_interp
        for beta_tr in [20, 50, 100]:
            if time.time() > t_limit_res - 10:
                break
            h_res, _ = projected_grad_descent(best_h, beta_tr, 200, n_try / 2.0, t0, t_limit_res)
            c5_res = compute_c5_from_h(h_res)
            if c5_res < best_c5 - 1e-10:
                best_h = h_res
                best_c5 = c5_res

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
