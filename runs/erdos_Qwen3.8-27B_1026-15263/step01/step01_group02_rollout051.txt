```python
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_hi)
        c_interp = compute_c5_from_h(h_interp)
        
        h_new, c_new = slsqp_refine(h_interp, n_hi, deadline=deadline - 60)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
    
    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    
    return best_h, best_c5, len(best_h)
```

Now let me think about the SLSQP formulation. The key issue is: how to formulate the objective for SLSQP?

Option A: Minimize t with constraint t ≥ max(c*dx)
- Problem: max is not differentiable
- Workaround: use the value at the current point as a linear approximation

Option B: Minimize the soft-max of c*dx
- Smooth, differentiable
- Approximates max for large β
- SLSQP can handle the gradient

Option C: Minimize the sum of the top-k values of c*dx
- Somewhat differentiable (use soft-top-k)

I'll go with Option B (soft-max) for the main SLSQP, and also try Option A with a trick.

For Option A, I can use the following trick: at each SLSQP evaluation, compute max(c*dx) and use it as the objective. SLSQP uses finite differences for the gradient, so it will "see" the kinks but might still converge (just slowly).

Actually, the previous code uses a different trick: it adds t as a variable and uses the inequality t - c*dx ≥ 0. But c*dx is a single number (the max), so this inequality is just t ≥ max(c*dx). SLSQP will try to minimize t, and the inequality will push t up to max(c*dx). The gradient of the constraint with respect to h will be the gradient of max(c*dx) (via finite differences). This should work.

Let me use a slightly different formulation that I think will work better:

Minimize: max(c*dx)  [just compute it directly]
Subject to: sum(h) = n/2
             0 ≤ h[i] ≤ 1

SLSQP will use finite differences to approximate the gradient of max(c*dx).

Actually, I just realized: scipy's SLSQP can handle non-smooth objectives (it uses finite differences). The issue is that at the kink points, the finite difference gradient might not point in the right direction. But in practice, SLSQP often works fine for max-type objectives.

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def soft_max_gradient(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    exp_vals = np.exp(beta * (c - c_max))
    p = exp_vals / exp_vals.sum()
    s_vals = np.arange(2 * n - 1) - (n - 1)
    sig = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-10:
            continue
        s = s_vals[k_idx]
        j_lo = max(0, s)
        j_hi = min(n - 1, n - 1 + s)
        if j_lo <= j_hi:
            sig[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - s:j_hi - s + 1])
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            sig[j_lo2:j_hi2 + 1] -= w * h[j_lo2 + s:j_hi2 + s + 1]
    return dx * sig

        c = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(c) * (2.0 / n))
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun}]
    
    result = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)
