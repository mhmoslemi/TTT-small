                                                  max_iter=200, eta=1.0,
                                                  mu=0.9,
                                                  time_limit=min(remaining() * 0.1, 60))
                if val_try < best_c5:
                    best_c5 = val_try
                    best_h = h_try
            h_out = slsqp_refine(best_h, len(best_h), 0.008,
                                 max_iter=200,
                                 time_limit=min(remaining() * 0.05, 40))
            c5_out = evaluate_c5(h_out)
            if c5_out < best_c5:
                best_c5 = c5_out
                best_h = h_out

    # Phase 5: Coordinate descent polish
    if remaining() > 20:
        h_out, val_out = coordinate_descent(best_h, len(best_h),
                                            max_rounds=30,
                                            time_limit=min(remaining() * 0.4, 120))
        if val_out < best_c5:
            best_c5 = val_out
            best_h = h_out

    # Phase 6: Final SLSQP with very small tau
    if remaining() > 15:
        n_final = len(best_h)
        for tau in [0.002, 0.001, 0.0005]:
            if remaining() < 10:
                break
            h_out = slsqp_refine(best_h, n_final, tau, max_iter=200,
                                 time_limit=min(remaining() * 0.5, 60))
            c5 = evaluate_c5(h_out)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_out

    # Phase 7: One more coordinate descent
    if remaining() > 10:
        h_out, val_out = coordinate_descent(best_h, len(best_h),
                                            max_rounds=10,
                                            time_limit=min(remaining() * 0.8, 60))
        if val_out < best_c5:
            best_c5 = val_out
            best_h = h_out

    return best_h, best_c5, len(best_h)
