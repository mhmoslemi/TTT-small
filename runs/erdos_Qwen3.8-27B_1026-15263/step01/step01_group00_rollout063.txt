    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(overlaps, alpha):
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    w = exp_vals / np.sum(exp_vals)
    val = m + np.log(np.sum(exp_vals)) / alpha
    return val, w

    Returns (value, gradient).

    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    val, w = smooth_max_and_weights(overlaps, alpha)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_grad(h, alpha):
    """Compute smooth max of overlaps and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    w = exp_vals / np.sum(exp_vals)
    val = m + np.log(np.sum(exp_vals)) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            # full_corr[k] = sum_{i=0}^{n-1-offset} h[i]*(1-h[i+offset])
            grad[:n-offset] += wk * (1 - h[offset:])
            grad[offset:] -= wk * h[:n-offset]

            grad[j:] += wk * (1 - h[:n-j])
            grad[:n-j] -= wk * h[j:]
    
    return val, grad

    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        count = int(np.sum(mask))
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    w = exp_vals / np.sum(exp_vals)
    val = m + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    cutoff = 1e-10
    for k in range(len(overlaps)):
        wk = w[k]
        if wk < cutoff:
            continue
        wk_dx = wk * dx
        offset = k - (n - 1)
        if offset >= 0:
            sz = n - offset
            grad[:sz] += wk_dx * (1.0 - h[offset:])
            grad[offset:] -= wk_dx * h[:sz]
        else:
            j = -offset
            sz = n - j
            grad[j:] += wk_dx * (1.0 - h[:sz])
            grad[:sz] -= wk_dx * h[j:]
    return val, grad

def fista_optimize(h0, ts, alpha, max_iter, step_init, time_left_fn):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    step = step_init
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(max_iter):
        if time_left_fn() < 30:
            break
        val, grad = smooth_max_value_grad(y, alpha)
        step_try = step
        for _ in range(30):
            h_cand = np.clip(y - step_try * grad, 0.0, 1.0)
            h_cand = project_to_sum(h_cand, ts)
            val_cand, _ = smooth_max_value_grad(h_cand, alpha)
            if val_cand <= val - 1e-5 * step_try * np.dot(grad, grad):
                break
            step_try *= 0.5
        h_new = h_cand
        step = step_try * 1.1
        if step > 1.0:
            step = 1.0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_k = t_new
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def pair_cd(h, ts, max_rounds, n_pairs, rng, time_left_fn):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time_left_fn() < 25:
            break
        improved = False
        for _ in range(n_pairs):
            i, j = rng.choice(n, size=2, replace=False)
            a, b = h[i], h[j]
            t_lo = max(-a, b - 1.0) + 1e-10
            t_hi = min(1.0 - a, b) - 1e-10
            if t_hi - t_lo < 1e-8:
                continue
            t_vals = np.linspace(t_lo, t_hi, 15)
            best_t = 0.0
            best_c = c5_cur
            for t in t_vals:
                h[i] = a + t
                h[j] = b - t
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_t != 0.0:
                h[i] = a + best_t
                h[j] = b - best_t
                c5_cur = best_c
                improved = True
            else:
                h[i] = a
                h[j] = b
        if not improved:
            break
    return h, c5_cur

def upsample_periodic(h_low, n_high):
    n_low = len(h_low)
    x_low = np.linspace(0, 2, n_low, endpoint=False)
    x_high = np.linspace(0, 2, n_high, endpoint=False)
    extended = np.concatenate([h_low, h_low, h_low])
    x_ext = np.concatenate([x_low, x_low + 2, x_low + 4])
    h_high = np.interp(x_high, x_ext, extended)
    return h_high

def make_fourier_init(n, rng):
    x = np.arange(n)
    h = np.zeros(n)
    n_modes = min(8, n // 3)
    for m in range(1, n_modes + 1):
        freq = m * 2 / n
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.15) / np.sqrt(m)
        h += amp * np.cos(2 * np.pi * freq * x + phase)
    h += 0.5 * np.ones(n)
    return h

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 7)
    block_size = n // n_blocks
    for _ in range(n_blocks // 2):
        start = rng.randint(0, n - block_size)
        h[start:start + block_size] += 1.0
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            h_parent = np.array(initial_h_values, dtype=float)
            n_parent = len(h_parent)
            ts_parent = n_parent / 2.0
            h_parent = project_to_sum(h_parent, ts_parent)
            c5_parent = compute_c5(h_parent)
            best_h = h_parent.copy()
            best_c5 = c5_parent
            best_n = n_parent
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    h_work = best_h.copy()

    alphas_main = [40, 100, 250, 500, 1000]
    for alpha in alphas_main:
        if time_left() < 80:
            break
        step_init = 2.0 / (alpha * n)
        h_new, c5_new = fista_optimize(h_work, ts, alpha, 4000, step_init, time_left)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_work = h_new

    if time_left() > 100:
        h_cd, c5_cd = pair_cd(best_h, ts, 300, 150, rng, time_left)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Multi-start exploration
    n_starts = 12
    time_per_start = max(30, (time_left() - 120) / n_starts)

    for si in range(n_starts):
        if time_left() < 100:
            break
        if si == 0:
            h_start = np.full(n, 0.5)
        elif si == 1:
            h_start = make_fourier_init(n, rng)
        elif si == 2:
            h_start = make_block_init(n, rng)
        elif si == 3:
            h_start = project_to_sum(rng.uniform(0, 1, n), ts)
        elif si == 4:
            x = np.linspace(0, 2, n, endpoint=False)
            freq = rng.choice([2, 3, 4, 5, 6])
            phase = rng.uniform(0, 2 * np.pi)
            h_start = np.clip(0.5 + 0.2 * np.sin(2 * np.pi * freq * x / 2.0 + phase), 0, 1)
        elif si == 5:
            n_low = 36
            ts_low = n_low / 2.0
            h_low = project_to_sum(rng.uniform(0, 1, n_low), ts_low)
            step_l = 2.0 / (50 * n_low)
            h_low, _ = fista_optimize(h_low, ts_low, 50, 1500, step_l, time_left)
            h_low, _ = fista_optimize(h_low, ts_low, 200, 1000, 2.0 / (200 * n_low), time_left)
            h_start = upsample_periodic(h_low, n)
        elif si == 6:
            n_low = 60
            ts_low = n_low / 2.0
            h_low = make_fourier_init(n_low, rng)
            h_low = project_to_sum(h_low, ts_low)
            h_low, _ = fista_optimize(h_low, ts_low, 50, 1500, 2.0 / (50 * n_low), time_left)
            h_low, _ = fista_optimize(h_low, ts_low, 200, 1000, 2.0 / (200 * n_low), time_left)
            h_start = upsample_periodic(h_low, n)
        elif si == 7:
            h_start = project_to_sum(upsample_periodic(best_h, n), ts)
            h_start += rng.normal(0, 0.02, n)
        elif si == 8:
            n_low = 45
            ts_low = n_low / 2.0
            x_low = np.linspace(0, 2, n_low, endpoint=False)
            h_low = 0.5 + 0.25 * np.cos(2 * np.pi * 3 * x_low / 2.0)
            h_low = project_to_sum(np.clip(h_low, 0, 1), ts_low)
            h_low, _ = fista_optimize(h_low, ts_low, 50, 1500, 2.0 / (50 * n_low), time_left)
            h_low, _ = fista_optimize(h_low, ts_low, 300, 1000, 2.0 / (300 * n_low), time_left)
            h_start = upsample_periodic(h_low, n)
        elif si == 9:
            h_start = np.zeros(n)
            quarter = n // 4
            h_start[0:quarter] = 0.8
            h_start[quarter:2 * quarter] = 0.3
            h_start[2 * quarter:3 * quarter] = 0.7
            h_start[3 * quarter:] = 0.4
        elif si == 10:
            h_start = project_to_sum(rng.uniform(0.2, 0.8, n), ts)
            h_start = h_start + 0.1 * np.sin(2 * np.pi * 4 * np.arange(n) / n)
        else:
            n_low = 30
            ts_low = n_low / 2.0
            h_low = project_to_sum(rng.uniform(0, 1, n_low), ts_low)
            for al in [30, 100, 400]:
                h_low, _ = fista_optimize(h_low, ts_low, al, 1000, 2.0 / (al * n_low), time_left)
            h_start = upsample_periodic(h_low, n)

        h_start = project_to_sum(np.clip(h_start, 0, 1), ts)
        c5_s = compute_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()

        for alpha in [50, 200, 600]:
            if time_left() < 50:
                break
            step_init = 2.0 / (alpha * n)
            h_start, _ = fista_optimize(h_start, ts, alpha, 2000, step_init, time_left)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()

        if time_left() > 50:
            h_cd, c5_cd = pair_cd(h_start, ts, 150, 100, rng, time_left)
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final refinement
    if time_left() > 40:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        for alpha in [500, 1500, 3000]:
            if time_left() < 20:
                break
            step_init = 2.0 / (alpha * n_f)
            h_f, _ = fista_optimize(best_h, ts_f, alpha, 3000, step_init, time_left)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        if time_left() > 20:
            h_f, c5_f = pair_cd(best_h, ts_f, 500, 250, rng, time_left)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
