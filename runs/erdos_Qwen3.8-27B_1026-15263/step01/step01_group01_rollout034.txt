def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_grad(h, p):
    n = len(h)
    c = get_corr(h)
    c_max = np.max(c)
    if c_max < 1e-15:
        return np.zeros(n), c
    if p >= 100:
        c_norm = c / c_max
        thresh = 1.0 - 3.0 / p
        w = np.where(c_norm >= thresh, c_norm ** 20, 0.0)
        ws = np.sum(w)
        if ws < 1e-15:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = w / ws
    else:
        c_norm = c / c_max
        total = np.sum(c_norm ** p)
        if total < 1e-30:
            return np.zeros(n), c
        w = (c_norm ** (p - 1)) * total ** (1.0 / p - 1.0)
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-18:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            i_lo = max(0, k)
            i_hi = min(n - 1, n - 1 + k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo - k:i_hi + 1 - k])
            i_lo2 = max(0, -k)
            i_hi2 = min(n - 1, n - 1 - k)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 + k:i_hi2 + 1 + k]
    return grad, c


def nesterov_optimize(h, n, p, max_iters, time_limit, lr_init=0.1):
    t0 = time.time()
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    momentum = 0.9
    lr = lr_init
    stagnant = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        y = h_curr + momentum * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, n)
        grad, c = compute_grad(y, p)
        f_curr = compute_c5(h_curr)
        alpha = lr
        h_new = None
        for _ in range(20):
            candidate = y - alpha * grad
            candidate = np.clip(candidate, 0.0, 1.0)
            candidate = project_feasible(candidate, n)
            f_new = compute_c5(candidate)
            if f_new < f_curr - 1e-14:
                h_new = candidate
                break
            alpha *= 0.5
        if h_new is not None:
            h_new_c5 = compute_c5(h_new)
            if h_new_c5 < best_c5:
                best_c5 = h_new_c5
                best_h = h_new.copy()
                stagnant = 0
            else:
                stagnant += 1
            lr = min(lr * 1.03, 0.5)
        else:
            stagnant += 1
            lr = max(lr * 0.8, 1e-8)
            if stagnant > 100:
                break
        h_prev = h_curr.copy()
        h_curr = h_new if h_new is not None else h_curr
    return best_h, best_c5


def gaussian_kick(h, n, sigma, rng):
    h_kick = h + rng.normal(0, sigma, n)
    return project_feasible(h_kick, n)


def block_flip_kick(h, n, rng):
    block_len = rng.randint(max(5, n // 15), max(6, n // 4))
    start = rng.randint(0, n - block_len)
    h_new = h.copy()
    h_new[start:start + block_len] = 1.0 - h[start:start + block_len]
    return project_feasible(h_new, n)


def sinusoidal_kick(h, n, rng):
    freq = rng.uniform(1.0, n / 4.0)
    amp = rng.uniform(0.05, 0.25)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h_new = h + amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, n)


def multi_sine_kick(h, n, rng):
    h_new = h.copy()
    for _ in range(rng.randint(1, 4)):
        freq = rng.uniform(1.0, n / 3.0)
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, n)


def block_swap_kick(h, n, rng):
    block_len = rng.randint(max(5, n // 10), max(6, n // 3))
    s1 = rng.randint(0, n - block_len)
    s2 = rng.randint(0, n - block_len)
    if abs(s1 - s2) < block_len:
        return h.copy()
    h_new = h.copy()
    temp = h_new[s1:s1 + block_len].copy()
    h_new[s1:s1 + block_len] = h_new[s2:s2 + block_len]
    h_new[s2:s2 + block_len] = temp
    return project_feasible(h_new, n)


def interpolate_to(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
