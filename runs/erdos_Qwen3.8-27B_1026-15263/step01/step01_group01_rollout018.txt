def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.maximum(h, 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff
        for idx in order:

                break
            if remaining > 0:
                step = min(remaining, 1.0 - h[idx])
            else:
                step = max(remaining, -h[idx])
            h[idx] += step
            remaining -= step
    return h

    shifted = np.clip(beta * (f - m), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = m + np.log(sum_exp) / beta
    s = np.sum(h) - n / 2.0
    obj += lambda_sum * s * s
    p = exp_shifted / sum_exp
    grad = np.zeros(n)
    for m_idx in range(len(p)):

        grad[vi] += pm * dx * (1.0 - h[vik])
        grad[vik] += pm * dx * (-h[vi])
    grad += 2.0 * lambda_sum * s
    return obj, grad
