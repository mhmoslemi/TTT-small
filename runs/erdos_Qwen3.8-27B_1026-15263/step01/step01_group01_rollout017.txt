    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            if not np.any(mask):

            idx = np.where(mask)[0]

            h[idx] = np.minimum(h[idx] + add, 1.0)

            h[idx] = np.maximum(h[idx] - sub, 0.0)

            h[idx] = max(0.0, h[idx] + diff)
    return h

    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            h[idx] += diff / len(idx)
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            h[idx] += diff / len(idx)
            np.clip(h, 0.0, 1.0, out=h)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def _obj_1d(v, c_base, alpha, beta, n):
    corr = c_base + v * alpha
    corr[n - 1] = c_base[n - 1] + v * (1.0 - v)
    return smooth_max(corr, beta) * (2.0 / n)


def coord_descent(h_init, n, max_sweeps, beta, time_limit, t0):
    h = project_feasible(h_init.copy(), n)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            h[j] = 0.0
            c_base = np.correlate(h, 1.0 - h, mode="full")
            k = np.arange(-(n - 1), n)
            alpha = np.zeros(2 * n - 1, dtype=float)
            mask1 = (j - k >= 0) & (j - k < n)
            alpha[mask1] += (1.0 - h[j - k[mask1]])
            mask2 = (j + k >= 0) & (j + k < n)
            alpha[mask2] -= h[j + k[mask2]]
            res = minimize_scalar(_obj_1d, args=(c_base, alpha, beta, n),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            h[j] = res.x
        h = project_feasible(h, n)
    return h, compute_c5(h)


def anneal_coord_descent(h_init, n, time_limit, t0):
    h = project_feasible(h_init.copy(), n)
    schedule = [(5, 8), (10, 12), (20, 15), (50, 25),
                (100, 40), (200, 60), (500, 100), (1000, 150), (2000, 200)]
    for beta, sweeps in schedule:
        if time.time() - t0 > time_limit:
            break
        h, _ = coord_descent(h, n, sweeps, beta, time_limit, t0)
    return h, compute_c5(h)


def make_perturbed(h, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, len(h))
    return project_feasible(h_p, len(h))


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_blocked(n, n_blocks, rng):
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        h[start:end] = rng.uniform(0.1, 0.9)
    return project_feasible(h, n)


def make_ripple(n, freq, amp, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x)
    h = h + rng.normal(0, 0.03, n)
    return project_feasible(h, n)


def slsqp_polish(h_init, n, beta, max_iter, time_limit, t0):
    if time.time() - t0 > time_limit:
        return h_init, compute_c5(h_init)

    def obj_slsqp(x):
        h_c = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        return smooth_max(corr, beta) * (2.0 / n)

    def constraint_sum(x):
        return np.sum(x) - n / 2.0

    h0 = project_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:
        res = minimize(obj_slsqp, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return h_init, compute_c5(h_init)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.ones(60) * 0.5
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    if time.time() - t0 < budget_s - 200:
        h_test, c5_test = anneal_coord_descent(best_h, best_n, budget_s - 300, t0)
        if c5_test < best_c5:
            best_h, best_c5 = h_test, c5_test

    no_improve = 0
    max_no_improve = 25
    while time.time() - t0 < budget_s - 350 and no_improve < max_no_improve:
        sigma_val = 0.08 + 0.12 * (0.92 ** no_improve)
        if rng.random() < 0.15:
            sigma_val = 0.35 + 0.2 * rng.random()
        h_start = make_perturbed(best_h, sigma_val, rng)
        remaining = budget_s - (time.time() - t0) - 350
        if remaining > 40:
            h_test, c5_test = anneal_coord_descent(h_start, best_n,
                                                   t0 + remaining, t0)
            if c5_test < best_c5:
                best_h, best_c5 = h_test, c5_test
                no_improve = 0
            else:
                no_improve += 1

    if no_improve >= max_no_improve and time.time() - t0 < budget_s - 300:
        for trial in range(5):
            if time.time() - t0 > budget_s - 300:
                break
            r = rng.random()
            if r < 0.33:
                h_start = make_blocked(best_n, int(rng.randint(3, 15)), rng)
            elif r < 0.66:
                h_start = make_ripple(best_n, int(rng.randint(1, 8)),
                                      0.3 + 0.2 * rng.random(), rng)
            else:
                h_start = make_perturbed(best_h, 0.5, rng)
            remaining = budget_s - (time.time() - t0) - 300
            if remaining > 50:
                h_test, c5_test = anneal_coord_descent(h_start, best_n,
                                                       t0 + remaining, t0)
                if c5_test < best_c5:
                    best_h, best_c5 = h_test, c5_test

    for n_try in [240, 300, 380, 470]:
        if time.time() - t0 > budget_s - 200:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_interp = project_feasible(h_interp, n_try)
        remaining = budget_s - (time.time() - t0) - 150
        if remaining > 60:
            h_test, c5_test = anneal_coord_descent(h_interp, n_try,
                                                   t0 + remaining, t0)
            if c5_test < best_c5:
                best_h, best_c5, best_n = h_test, c5_test, n_try

    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_test, c5_test = coord_descent(best_h, best_n, 300, 3000,
                                        t0 + remaining - 10, t0)
        if c5_test < best_c5:
            best_h, best_c5 = h_test, c5_test

    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_test, c5_test = slsqp_polish(best_h, best_n, 500, 300,
                                       t0 + remaining - 5, t0)
        if c5_test < best_c5:
            best_h, best_c5 = h_test, c5_test

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
