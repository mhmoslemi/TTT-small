def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_c5(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    if tau < 1.0:
        return c_max
    exps = np.exp((c - c_max) * tau)
    s = np.sum(exps)
    return c_max + np.log(s) / tau


def smooth_objective(h, tau, lam):
    n = len(h)
    base = soft_max_c5(h, tau)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return base + penalty

    n_h = n

    def obj(x):
        return x[-1]

        t = x[-1]
        dx = 2.0 / n_h
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * dx

    def eq_con(x):
        h = x[:-1]
        return np.sum(h) - n_h / 2.0

        j = np.zeros(n_h + 1)
        j[:n_h] = 1.0
        return j

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(h < 1.0) if diff > 0 else np.argmax(h > 0.0)
        if 0 <= idx < n:
            h[idx] += diff
    h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_c5(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    if tau < 1.0:
        return c_max
    scaled = (c - c_max) * tau
    if np.max(scaled) > 500:
        return c_max
    exps = np.exp(scaled)
    s = np.sum(exps)
    if s <= 0 or not np.isfinite(s):
        return c_max
    return c_max + np.log(s) / tau


def smooth_objective(h, tau, lam):
    n = len(h)
    base = soft_max_c5(h, tau)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return base + penalty


def lbfgsb_smooth(h_init, n, time_limit):
    tau_schedule = [3, 8, 20, 50, 100, 200]
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    for tau in tau_schedule:
        if time.time() > time_limit:
            break
        for lam in [100.0, 1000.0]:
            if time.time() > time_limit:
                break
            h_p = project_feasible(best_h, n)
            try:
                result = minimize(smooth_objective, h_p, args=(tau, lam),
                                  method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * n,
                                  options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10})
            except Exception:
                continue
            if result.x is not None:
                h_new = project_feasible(result.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
    return best_h, best_c5


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n_h = len(h)
    dx = 2.0 / n_h
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n_h = len(x) - 1
    j = np.zeros(n_h + 1)
    j[:n_h] = 1.0
    return j


def slsqp_solve(h_init, n, maxiter=1500):
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.85), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    try:
        result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
    except Exception:
        return h_init, c5_init
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h_init, c5_init


def pair_descent(h_init, n, time_limit):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    deltas = [0.02, 0.05, 0.1, 0.15, 0.2]
    radius = max(5, n // 20)
    for _ in range(30):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            j_lo = max(0, i - radius)
            j_hi = min(n, i + radius + 1)
            for j in list(range(j_lo, i)) + list(range(i + 1, j_hi)):
                if time.time() > time_limit:
                    break
                oi, oj = best_h[i], best_h[j]
                for d in deltas:
                    for s in [1.0, -1.0]:
                        ni = oi + s * d
                        nj = oj - s * d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            best_h[i] = ni
                            best_h[j] = nj
                            c5_t = compute_c5(best_h)
                            if c5_t < best_c5 - 1e-13:
                                best_c5 = c5_t
                                improved = True
                                break
                            best_h[i] = oi
                            best_h[j] = oj
                    else:
                        continue
                    break
                best_h[i] = oi
                best_h[j] = oj
            if time.time() > time_limit:
                break
        if not improved:
            break
    return best_h, best_c5


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)


def make_sinusoidal_h(n, freq, amp, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)


def make_binary_h(n, seed_rng):
    h = (seed_rng.random(n) < 0.5).astype(float)
    target = n / 2.0
    current = np.sum(h)
    if current > target:
        idx = np.where(h == 1.0)[0]
        for k in range(int(current - target + 0.5)):
            h[idx[k]] = 0.0
    elif current < target:
        idx = np.where(h == 0.0)[0]
        for k in range(int(target - current + 0.5)):
            h[idx[k]] = 1.0
    frac = target - np.sum(h)
    if abs(frac) > 1e-10:
        non_boundary = np.where((h > 0.0) & (h < 1.0))[0]
        if len(non_boundary) > 0:
            h[non_boundary[0]] += frac
        else:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += frac
    return project_feasible(h, n)
