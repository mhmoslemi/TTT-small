def project_feasible(v, target_sum):
    """Project v onto {h: 0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(v)
    # Bisection on lambda
    lo = v.min() - target_sum / n
    hi = v.max()
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(v, target_sum):
    n = len(v)
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo = v.min() - target_sum / n - 1
    hi = v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_smoothed_grad(h, tau):
    """Compute gradient of log-sum-exp smoothed objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(tau * (corr - max_c))
    weights /= weights.sum()
    
    # grad[i] = sum_k weights[k] * ((1-h)[i+k-n+1] - h[i-k+n-1])
    # with out-of-bounds treated as 0
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for i in range(n):
        # First part: sum_k weights[k] * (1-h)[i+k-n+1]
        # Valid k: 0 <= i+k-n+1 < n => n-1-i <= k <= n-1+i
        k_lo = max(0, n - 1 - i)
        k_hi = min(2 * n - 2, n - 1 + i)
        if k_lo <= k_hi:
            j = i + np.arange(k_lo, k_hi + 1) - (n - 1)
            grad[i] += np.sum(weights[k_lo:k_hi+1] * one_minus_h[j])
        
        # Second part: -sum_k weights[k] * h[i-k+n-1]
        # Valid k: 0 <= i-k+n-1 < n => i-n+1 <= k <= i+n-1
        k_lo2 = max(0, i - n + 1)
        k_hi2 = min(2 * n - 2, i + n - 1)
        if k_lo2 <= k_hi2:
            j = i - np.arange(k_lo2, k_hi2 + 1) + (n - 1)
            grad[i] -= np.sum(weights[k_lo2:k_hi2+1] * h[j])
    
    return grad

def local_search(h, target_sum, max_iters=2000, time_limit=None, start_time=None):
    """Local search with random perturbations"""
    n = len(h)
    h = h.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        
        # Random perturbation
        h_try = h.copy()
        n_perturb = np.random.randint(1, 5)
        idx = np.random.choice(n, n_perturb, replace=False)
        h_try[idx] += np.random.uniform(-0.05, 0.05, n_perturb)
        h_try = np.clip(h_try, 0, 1)
        h_try = project_feasible(h_try, target_sum)
        
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    
    return best_h, best_c5
