import numpy as np
import time


def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def pair_optimize_step(h, i, j, n):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return np.max(np.correlate(h, 1.0 - h, mode='full'))

    n_full = 2 * n - 1
    t_mid = (lo + hi) * 0.5

    h[i] = t_mid
    h[j] = s - t_mid
    corr = np.correlate(h, 1.0 - h, mode='full')

    eps = 1e-5
    tp = min(hi, t_mid + eps)
    tm = max(lo, t_mid - eps)
    actual_eps = (tp - tm) * 0.5
    if actual_eps < 1e-12:
        return np.max(corr)

    h[i] = tp
    h[j] = s - tp
    corr_p = np.correlate(h, 1.0 - h, mode='full')

    h[i] = tm
    h[j] = s - tm
    corr_m = np.correlate(h, 1.0 - h, mode='full')

    second_diff = (corr_p - 2.0 * corr + corr_m) / (actual_eps * actual_eps)
    first_deriv = (corr_p - corr_m) / (2.0 * actual_eps)

    quad_mask = np.abs(second_diff) > 0.5

    c2 = np.where(quad_mask, 1.0, 0.0)
    c1 = np.zeros(n_full)
    c0 = np.zeros(n_full)

    qi = np.where(quad_mask)[0]
    for k in qi:
        c1[k] = first_deriv[k] - 2.0 * t_mid
        c0[k] = corr[k] - t_mid * t_mid - c1[k] * t_mid

    li = np.where(~quad_mask)[0]
    for k in li:
        c1[k] = first_deriv[k]
        c0[k] = corr[k] - c1[k] * t_mid

    # Find all pairwise intersection points within [lo, hi]
    candidates = [lo, hi]
    for k1 in range(n_full):
        for k2 in range(k1 + 1, n_full):
            a2 = c2[k1] - c2[k2]
            a1 = c1[k1] - c1[k2]
            a0 = c0[k1] - c0[k2]
            if abs(a2) < 1e-14:
                if abs(a1) > 1e-14:
                    tc = -a0 / a1
                    if lo < tc < hi:
                        candidates.append(tc)
            else:
                disc = a1 * a1 - 4.0 * a2 * a0
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    tc1 = (-a1 + sq) / (2.0 * a2)
                    tc2 = (-a1 - sq) / (2.0 * a2)
                    if lo < tc1 < hi:
                        candidates.append(tc1)
                    if lo < tc2 < hi:
                        candidates.append(tc2)

    candidates = np.array(candidates)
    tc = candidates[:, None]
    vals = c2[None, :] * tc * tc + c1[None, :] * tc + c0[None, :]
    max_vals = np.max(vals, axis=1)
    best_idx = np.argmin(max_vals)
    best_t = candidates[best_idx]

    h[i] = best_t
    h[j] = s - best_t
    return np.max(np.correlate(h, 1.0 - h, mode='full'))


def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idxs = np.where(h < 1.0)[0]
        for idx in idxs:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        idxs = np.where(h > 0.0)[0]
        for idx in idxs:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return h


def make_random_feasible(n, rng):
    target_sum = n / 2.0
    h = rng.uniform(0.0, 1.0, n)
    fix_sum(h, target_sum)
    return h


def make_spiky_feasible(n, rng):
    target_sum = n / 2.0
    h = np.zeros(n)
    n_ones = int(target_sum)
    frac = target_sum - n_ones
    idxs = rng.permutation(n)
    h[idxs[:n_ones]] = 1.0
    if frac > 0:
        h[idxs[n_ones]] = frac
    return h


def coordinate_descent(h, n, n_pairs, pairs, best_c5, best_h, dx, rng, max_sweeps):
    t0 = time.time()
    sweep = 0
    stall = 0

    while sweep < max_sweeps:
        perm = rng.permutation(n_pairs)
        improved = False

        for idx in perm:
            i, j = pairs[idx]
            h_save_i = h[i]
            h_save_j = h[j]
            max_corr = pair_optimize_step(h, i, j, n)
            c5 = max_corr * dx
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                improved = True
            else:
                h[i] = h_save_i
                h[j] = h_save_j

        sweep += 1

        if not improved:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0

        if time.time() - t0 > 1200:
            break

    return h, best_c5, best_h
