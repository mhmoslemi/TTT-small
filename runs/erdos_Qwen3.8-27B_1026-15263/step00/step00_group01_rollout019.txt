import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 1.0
_g_target = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5
        return h
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-15:
        if d > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                adj = min(d, room[idx])
                h[idx] += adj
                d -= adj
                if abs(d) < 1e-15:
                    break
        else:
            room = h.copy()
            order = np.argsort(-room)
            for idx in order:
                adj = min(-d, room[idx])
                h[idx] -= adj
                d += adj
                if abs(d) < 1e-15:
                    break
    return np.clip(h, 0.0, 1.0)

def smax(v, a):
    m = float(v.max())
    x = np.clip((v - m) * a, -50.0, 50.0)
    return m + float(np.log(np.mean(np.exp(x)) + 1e-300)) / a

def _smax_obj(x):
    return smax(corr_vals(x, _g_n), _g_alpha)

def _sum_con(x):
    return x.sum() - _g_target

def _sum_jac(x):
    return np.ones(_g_n)

def _pair_swap_refine(best, n, budget_end, deltas, passes_per_delta=3):
    bc5 = compute_c5(best, n)
    for dval in deltas:
        if time.time() > budget_end:
            break
        for _p in range(passes_per_delta):
            if time.time() > budget_end:
                break
            for i in range(n):
                if time.time() > budget_end:
                    break
                for j in range(i + 1, n):
                    if time.time() > budget_end:
                        break
                    found = False
                    ht = best.copy()
                    ht[i] += dval
                    ht[j] -= dval
                    if 0.0 <= ht[i] <= 1.0 and 0.0 <= ht[j] <= 1.0:
                        ct = compute_c5(ht, n)
                        if ct < bc5 - 1e-14:
                            best = ht
                            bc5 = ct
                            found = True
                    if not found:
                        ht = best.copy()
                        ht[i] -= dval
                        ht[j] += dval
                        if 0.0 <= ht[i] <= 1.0 and 0.0 <= ht[j] <= 1.0:
                            ct = compute_c5(ht, n)
                            if ct < bc5 - 1e-14:
                                best = ht
                                bc5 = ct
    return best, bc5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_alpha, _g_target
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    tgt = n * 0.5
    _g_n = n
    _g_target = tgt
    h = repair(h, n)
    best = h.copy()
    bc5 = compute_c5(best, n)

    alphas = [8, 15, 25, 40, 60, 90, 130, 180, 250, 350, 500]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 60:
            break
        _g_alpha = alpha
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            r = minimize(_smax_obj, best, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 5000, 'ftol': 1e-15})
            if np.all(np.isfinite(r.x)):
                hn = repair(r.x, n)
                c = compute_c5(hn, n)
                if c < bc5:
                    best, bc5 = hn, c
        except Exception:
            pass

    step = 0.06
    max_anneal_iters = 8000
    for it in range(max_anneal_iters):
        if time.time() - t0 > budget_s - 30:
            break
        if it % 300 == 0 and step > 0.002:
            step *= 0.88
        ht = best + rng.randn(n) * step
        ht = repair(ht, n)
        ct = compute_c5(ht, n)
        if ct < bc5:
            best, bc5 = ht, ct
        elif rng.random() < np.exp(-(ct - bc5) / max(step * 0.25, 1e-10)):
            best = ht

    budget_end = t0 + budget_s - 5
    if time.time() < budget_end:
        coarse_deltas = [0.1, 0.05, 0.02, 0.01]
        best, bc5 = _pair_swap_refine(best, n, budget_end, coarse_deltas, 2)

    if time.time() < budget_end:

    bc5 = compute_c5(best, n)
