    # This is a convex piecewise linear function in t
    # The minimum is at t=0, t=1, or at a breakpoint
    
    # Breakpoints: for each pair (a,b), find t where corr0[a] + t*slopes[a] = corr0[b] + t*slopes[b]
    # t = (corr0[b] - corr0[a]) / (slopes[a] - slopes[b])
    
    candidates = [0.0, 1.0]
    
    # Find all breakpoints
    for a in range(len(slopes)):
        for b in range(a+1, len(slopes)):
            denom = slopes[a] - slopes[b]
            if abs(denom) > 1e-12:
                t = (corr0[b] - corr0[a]) / denom
                if 0 <= t <= 1:
                    candidates.append(t)
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        val = np.max(corr0 + t * slopes)
        if val < best_val:
            best_val = val
            best_t = t
    
    h[i] = best_t
    return best_val

import numpy as np
import time


def project_constraint(v, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        projected = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def eval_c5(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return dx * np.max(corr)


def compute_smooth_grad(h, dx, temp, n):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full') * dx
    m = np.max(corr)
    exps = np.exp((corr - m) / temp)
    s = np.sum(exps)
    if s < 1e-30:
        return np.zeros(n)
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wgt = weights[idx]
        if wgt < 1e-12:
            continue
        shift = idx - (n - 1)
        for i in range(n):
            j1 = i + shift
            if 0 <= j1 < n:
                grad[i] += wgt * dx * (1.0 - h[j1])
            j2 = i - shift
            if 0 <= j2 < n:
                grad[i] -= wgt * dx * h[j2]
    return grad


def gradient_phase(h, n, dx, target_sum, t0, deadline, rng):
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    c5_cur = best_c5

    for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline:
            break
        lr = 0.008
        for _ in range(5000):
            if time.time() > deadline:
                break
            grad = compute_smooth_grad(h, dx, temp, n)
            h_new = h - lr * grad
            h_new = project_constraint(h_new, target_sum)
            c5_new = eval_c5(h_new, dx)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                lr = min(lr * 1.04, 0.03)
            else:
                lr *= 0.82
            if lr < 1e-9:
                break
    return best_h, best_c5


def sa_phase(h, n, dx, target_sum, t0, deadline, rng, best_h, best_c5):

    T_min = 1e-7
    cooling = 0.99985
    c5_cur = eval_c5(h, dx)
    iter_count = 0

def coordinate_descent(h, n, dx, target_sum, t0, deadline):
    h = h.copy()
    c5 = eval_c5(h, dx)
    for _ in range(200):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            orig = h[i]
            cands = [0.0, 1.0, max(0.0, orig - 0.15), max(0.0, orig - 0.07),
                     min(1.0, orig + 0.07), min(1.0, orig + 0.15),
                     max(0.0, orig - 0.03), min(1.0, orig + 0.03)]
            for c in cands:
                h[i] = c
                hp = project_constraint(h, target_sum)
                c_try = eval_c5(hp, dx)
                if c_try < c5 - 1e-12:
                    c5 = c_try
                    h[:] = hp
                    improved = True
                    break
            if not improved:
                h[i] = orig
        if not improved:
            break
    return h, c5
