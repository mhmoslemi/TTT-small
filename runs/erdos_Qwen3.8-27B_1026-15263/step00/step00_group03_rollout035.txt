from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_unscaled(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)

def optimize_pair(h, i, j, dx):
    """Find best delta to transfer from j to i (or vice versa)"""
    h_i, h_j = h[i], h[j]
    
    def obj(delta):
        h[i] += delta
        h[j] -= delta
        val = compute_c5_unscaled(h)
        h[i] -= delta
        h[j] += delta
        return val

    # and h[i]+delta <= 1 and h[j]-delta <= 1
    lo = max(-h_i, h_j - 1)
    hi = min(1 - h_i, h_j)

    if result.x != 0:
        h[i] += result.x
        h[j] -= result.x
