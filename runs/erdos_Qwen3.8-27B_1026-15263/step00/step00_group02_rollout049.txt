import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def objective_smooth(x, n, beta, penalty_w):
    target = n / 2.0
    corr = compute_corr(x)
    sm = smooth_max_val(corr, beta) * (2.0 / n)
    penalty = penalty_w * (np.sum(x) - target) ** 2
    return sm + penalty

def coordinate_descent(h, n, max_iter=100):
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        improved = False
        for i in range(n):
            orig = h[i]
            # Try a few values
            candidates = [0.0, 1.0, orig - 0.1, orig - 0.05, orig - 0.01,
                         orig, orig + 0.01, orig + 0.05, orig + 0.1]
            # Also try to help with sum constraint
            sum_others = target - orig
            best_val = orig
            best_obj = compute_c5(h)
            for v in candidates:
                v = np.clip(v, 0, 1)
                h[i] = v
                c5 = compute_c5(h)
                if c5 < best_obj - 1e-7:
                    best_obj = c5
                    best_val = v
            h[i] = best_val
            if best_val != orig:
                improved = True
        # Project to sum constraint
        s = np.sum(h)
        if abs(s - target) > 1e-10:
            diff = (target - s) / n
            h = np.clip(h + diff, 0, 1)
        if not improved:
            break
    
    return h

def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()
    
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    dx = 2.0 / n
    target = n / 2.0
    
    # Ensure initial feasibility
    h0 = np.clip(h0, 0, 1)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    penalty_weights = [10, 50, 100, 500, 1000]
    betas = [10, 30, 100, 300]
    
    for pw in penalty_weights:
        for beta in betas:
            if time.time() - start_time > budget_s * 0.7:
                break
            x0 = best_h.copy()
            # Add small perturbation
            if np.random.rand() < 0.3:
                x0 = np.clip(x0 + 0.05 * np.random.randn(n), 0, 1)
            
            bounds = [(0, 1)] * n
            res = minimize(objective_smooth, x0, args=(n, beta, pw),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-12})
            
            x_opt = np.clip(res.x, 0, 1)
            # Check sum constraint
            if abs(np.sum(x_opt) - target) < 0.5:
                c5_val = compute_c5(x_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = x_opt.copy()
    
    # Coordinate descent refinement
    while time.time() - start_time < budget_s * 0.85:
        h_cd = coordinate_descent(best_h, n, max_iter=50)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        else:
            break
    
    # Random restarts
    while time.time() - start_time < budget_s * 0.9:
        x0 = best_h + 0.1 * np.random.randn(n)
        x0 = np.clip(x0, 0, 1)
        # Project to sum constraint
        s = x0.sum()
        x0 = np.clip(x0 + (target - s) / n, 0, 1)
        
        c5_val = compute_c5(x0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = x0.copy()
    
    return (best_h.tolist(), best_c5, n)
