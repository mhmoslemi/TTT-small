def compute_subgrad(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    K = idx_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        i1 = j - K
        if 0 <= i1 < n:
            grad[j] += (1.0 - h[i1])
        i2 = j + K
        if 0 <= i2 < n:
            grad[j] -= h[i2]

        try:
            res = do_slsqp(noisy, n, dx, 200)
            if res.x is not None:
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc, dx)
                if c5c < best_c5:
                    best_h = hc
                    best_c5 = c5c
        except Exception:
            pass

    # Phase 6: Try different n values
    for n_try in [60, 80, 100, 50, 140, 90, 110, 40, 120]:
        if remaining() < 40:
            break
        dx_try = 2.0 / n_try
        x_orig = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_orig, best_h)
        h_try = project_feasible(h_try, n_try)
        try:
            res = do_slsqp(h_try, n_try, dx_try, 300)
            if res.x is not None:
                hc = project_feasible(res.x, n_try)
                c5c = compute_c5(hc, dx_try)
                if c5c < best_c5:
                    best_h = hc
                    best_c5 = c5c
                    n = n_try
                    dx = dx_try
        except Exception:
            pass

    # Phase 7: Final SLSQP + coordinate descent
    if remaining() > 15:
        try:
            res = do_slsqp(best_h, n, dx, 300)
            if res.x is not None:
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc, dx)
                if c5c < best_c5:
                    best_h = hc
                    best_c5 = c5c
        except Exception:
            pass

    if remaining() > 10:
        hp = coord_descent(best_h, n, dx, 3)
        c5p = compute_c5(hp, dx)
        if c5p < best_c5:
            best_h = hp
            best_c5 = c5p

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
