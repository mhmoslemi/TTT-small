    h = cp.Variable(n_points)

    dx = 2.0 / n_points
    corr_full = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr_full) * dx
    # Find active lags
    active = np.where(np.isclose(corr_full * dx, c5, atol=1e-10))[0]
    
    # Compute subgradient (average over active lags)
    grad = np.zeros(n_points)
    for m in active:
        d = m - (n_points - 1)  # actual lag
        for j in range(n_points):
            g = 0.0
            # h[j] as first factor: j in V(d) means 0<=j<n and 0<=j+d<n

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = float(np.max(corr))
    m = int(np.argmax(corr))
    d = m - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + d < n:
            g += (1.0 - h[j + d])
        if 0 <= j - d < n:
            g -= h[j - d]
        grad[j] = g * dx
    return grad, c5


def solve_lp_given(q, n, solver):
    if not HAS_CVXPY:
        return None
    dx = 2.0 / n
    var = cp.Variable(n)
    t = cp.Variable()
    cons = [var >= 0, var <= 1, cp.sum(var) == n / 2.0]
    for d in range(-(n - 1), n):
        il = max(0, -d)
        ih = min(n - 1, n - 1 - d)
        if il > ih:
            continue
        cons.append(cp.sum(var[il:ih + 1] * q[il + d:ih + d + 1]) * dx <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if var.value is None:
        return None
    return np.array(var.value).flatten()

    return np.repeat(h, factor)

    h = project_feasible(h, n)
    c5 = compute_c5(h, n)

            c5 = c5_new
