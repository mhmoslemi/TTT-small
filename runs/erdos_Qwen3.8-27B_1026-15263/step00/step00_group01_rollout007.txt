_G = {}

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, n):
    return (2.0 / n) * np.max(_corr(h))

def _fix(h, tgt):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(m.sum())
        if nf == 0:
            i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
            h[i] += 0.01 * (1.0 if d > 0 else -1.0)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[m] += d / nf
            h = np.clip(h, 0.0, 1.0)
    return h
