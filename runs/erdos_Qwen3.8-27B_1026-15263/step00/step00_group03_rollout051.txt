import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_values(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def compute_c5(h, n_points):
    return np.max(compute_c5_values(h, n_points))

def project_feasible(h, S):
    """Project h onto {x : 0<=x<=1, sum(x)=S}"""
    n = len(h)
    # Bisection on lambda for x_i = clip(h_i - lambda, 0, 1)
    lo = -2.0
    hi = 2.0
    
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    return result

def gradient_single_shift(h, s, n):
    """Gradient of overlap(s) w.r.t. h for a given shift s"""
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - s < n:
            g += (1.0 - h[j - s])
        if 0 <= j + s < n:
            g -= h[j + s]
        grad[j] = g
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)
    
    S = n_points / 2.0
    dx = 2.0 / n_points
    
    # Ensure initial is feasible
    h_init = project_feasible(h_init, S)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_points)
    
    def elapsed():
        return time.time() - start_time
    
    # Projected gradient descent
    alpha = 0.1  # initial step size
    
    while elapsed() < budget_s - 5.0:
        h = best_h.copy()
        
        # Find worst shift (0-indexed in full correlation, center is n_points-1)
        c5_vals = compute_c5_values(h, n_points)
        worst_idx = np.argmax(c5_vals)
        s = worst_idx - (n_points - 1)  # actual shift
        
        # Compute gradient
        grad = gradient_single_shift(h, s, n_points)
        
        # Also consider small perturbations to escape local optima
        if rng.random() < 0.1:
            # Add noise
            grad += 0.01 * rng.standard_normal(n_points)
        
        # Backtracking line search
        success = False
        for _ in range(50):

            c5_trial = compute_c5(h_trial, n_points)
            if c5_trial < best_c5 - 1e-10:
                best_h = h_trial
                best_c5 = c5_trial
                success = True
                break
            alpha *= 0.5

            # Try a different shift or add perturbation

        for _ in range(30):
            h_new = project_feasible(h - a * avg_grad, S)
            c5_new = compute_c5(h_new, n_points)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                alpha = a
                improved = True
                break
            a *= 0.5
        
        if improved:
            h = best_h
            stall = 0
        else:
            stall += 1
            alpha = min(1.0, alpha * 1.5)
            if stall % 15 == 0:
                h = best_h + 0.08 * rng.standard_normal(n_points)
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, S)
                stall = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)

    best_c5 = compute_c5(best_h, n_points)
    
    # Phase 1: Optimize from initial
    time_for_phase1 = min(400, budget_s * 0.4)
    h1, c5_1 = optimize_from(h_init, S, n_points, time_for_phase1, rng)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

        h_pert = best_h + 0.1 * rng.standard_normal(n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = project_feasible(h_pert, S)
        time_for_phase2 = min(200, (budget_s - 200) * 0.5)
        h2, c5_2 = optimize_from(h_pert, S, n_points, time_for_phase2, rng)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 3: Nelder-Mead
    if time.time() - start_time < budget_s - 30:
        def nm_obj(x):
            return compute_c5(project_feasible(x, S), n_points)
        try:
            res = minimize(nm_obj, best_h, method='Nelder-Mead',
                          options={'maxiter': 10000, 'xatol': 1e-8, 'fatol': 1e-10, 'adaptive': True})
            if res.fun < best_c5:
                best_h = project_feasible(res.x, S)
                best_c5 = compute_c5(best_h, n_points)
        except Exception:

    # Final validation
    best_h = project_feasible(best_h, S)
    best_c5 = compute_c5(best_h, n_points)
