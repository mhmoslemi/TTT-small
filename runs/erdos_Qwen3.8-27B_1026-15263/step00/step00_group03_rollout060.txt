import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h, n_points):
    target = n_points / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - proj
            mask = room > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                proj[mask] = np.minimum(proj[mask] + diff / cnt, 1.0)
        else:
            mask = proj > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                proj[mask] = np.maximum(proj[mask] + diff / cnt, 0.0)
    return proj

def local_search_01(h, n_points, max_rounds=500, n_tries=300):

        j = zeros.pop(np.random.randint(len(zeros)))
        h[j] = 1
        ones.append(j)

        best_val = current
        best_flip = None

        zeros_arr = np.array(zeros) if zeros else np.array([], dtype=int)

def simulated_annealing_01(h, n_points, start_temp=0.01, end_temp=0.001, max_iter=5000):
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    target = n_points // 2

        h[i] = 0
        zeros.append(i)
    while len(ones) < target:
        j = zeros.pop(np.random.randint(len(zeros)))
        h[j] = 1
        ones.append(j)

    best_h = h.copy()
    best_val = current
    n_iter = max_iter
    cooling = (end_temp / start_temp) ** (1.0 / n_iter)

        if proj.sum() > target:

        else:
