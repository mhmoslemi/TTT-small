import numpy as np
import time


def _repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        h[:] = target / len(h)
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    resid = target - np.sum(h)
    if resid > 1e-10:
        for i in range(len(h)):
            d = min(resid, 1.0 - h[i])
            if d <= 0:
                break
            h[i] += d
            resid -= d
            if resid < 1e-12:
                break
    elif resid < -1e-10:
        for i in range(len(h)):
            d = min(-resid, h[i])
            if d <= 0:
                break
            h[i] -= d
            resid += d
            if resid > -1e-12:
                break
    return h


def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
