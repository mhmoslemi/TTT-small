from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def logsumexp_obj_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr(h)  # shape (2n-1,)
    scaled = tau * corr * dx
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = np.log(np.sum(exp_scaled)) + max_scaled
    obj = log_sum / tau
    weights = exp_scaled / np.sum(exp_scaled)  # softmax weights
    
    # Gradient: dx * sum_k weights[k] * grad_c_k
    grad = np.zeros(n)
    # Only compute gradient for lags with significant weight
    significant = weights > 1e-8
    if not np.any(significant):
        return obj, grad
    
    for k in range(len(corr)):
        if not significant[k]:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += weights[k] * (1.0 - h[lag:])
            grad[lag:] += weights[k] * (-h[:n-lag])
        else:
            m = -lag
            grad[m:] += weights[k] * (1.0 - h[:n-m])
            grad[:n-m] += weights[k] * (-h[m:])
