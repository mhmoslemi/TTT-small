def compute_weighted_gradient(h, weights, dx):
    """
    weights: array of length 2n-1, representing softmax weights for each lag
    Returns gradient of sum_k weights[k] * corr[k] * dx with respect to h
    """
    n = len(h)
    grad = np.zeros(n)
    # corr_full[m] = sum_i h[i] * (1-h[i-(m-n+1)]) for valid i
    # Let l = m - n + 1 (the lag)
    # corr_full[n+l] = sum_i h[i+l]*(1-h[i]) for 0<=i<n, 0<=i+l<n
    # 
    # d/dh[j] of corr_full[n+l]:
    #   From h[i+l] term: h[j] = h[i+l] when i=j-l, contrib: (1-h[i]) = (1-h[j-l]) if 0<=j-l<n
    #   From (1-h[i]) term: h[j] = h[i] when i=j, contrib: -h[i+l] = -h[j+l] if 0<=j+l<n
    
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        l = m - n + 1  # lag
        # First term: sum over j of w * (1-h[j-l]) where 0<=j-l<n
        # This adds to grad[j] the value w*(1-h[j-l]) for j where 0<=j-l<n
        # i.e., j ranges from max(0,l) to min(n-1, n-1+l)
        j_start = max(0, l)
        j_end = min(n, n + l)
        if j_start < j_end:
            grad[j_start:j_end] += w * (1 - h[j_start - l : j_end - l])
        # Second term: sum over j of -w * h[j+l] where 0<=j+l<n
        # i.e., j ranges from max(0, -l) to min(n-1, n-1-l)
        j_start2 = max(0, -l)
        j_end2 = min(n, n - l)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2 + l : j_end2 + l]
    
    return grad * dx

def compute_weighted_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for m in range(2 * n - 1):
        w = weights[m]

        js2 = max(0, l)
        je2 = min(n, n + l)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 - l:je2 - l]
    return grad * dx
