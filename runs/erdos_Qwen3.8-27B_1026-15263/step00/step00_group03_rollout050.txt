            best_h, best_c5 = h.copy(), c5

    if time.time() - t0 < budget_s - 120:
        def nm_obj(x):
            return compute_overlap(project_onto_feasible(x, n))
        try:
            res = minimize(nm_obj, best_h.copy(), method='Nelder-Mead',
                          options={'maxiter': 30000, 'xatol': 1e-10, 'fatol': 1e-12, 'maxfev': 15000})
            h_nm = project_onto_feasible(res.x, n)
            c5_nm = compute_overlap(h_nm)
            if c5_nm < best_c5:
                best_h, best_c5 = h_nm, c5_nm
        except Exception:
            pass

    if time.time() - t0 < budget_s - 60:
        h = best_h.copy()
        h = projected_gradient_step(h, n, alpha=80, init_step=0.003, max_iter=30000, time_fn=tl)
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    if time.time() - t0 < budget_s - 40:
        h_bin = binarize(best_h, n)
        h_bin, c5_bin = binary_local_search(h_bin, n, rng, max_iters=100000, time_fn=tl)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c5_bin
            h = projected_gradient_step(best_h, n, alpha=50, init_step=0.005, max_iter=20000, time_fn=tl)
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    if time.time() - t0 < budget_s - 20:
        for restart in range(200):
            if tl():
                break
            scale = rng.choice([0.01, 0.02, 0.03, 0.05, 0.08])
            noise = rng.standard_normal(n) * scale
            h_re = project_onto_feasible(best_h + noise, n)
            h_re = projected_gradient_step(h_re, n, alpha=60, init_step=0.005, max_iter=5000, time_fn=tl)
            c5_re = compute_overlap(h_re)
            if c5_re < best_c5:
                best_h, best_c5 = h_re.copy(), c5_re
            if time.time() - t0 > budget_s - 20:
                break

    if time.time() - t0 < budget_s - 5:
        if best_c5 > 0.3808:
            h_bin = binarize(best_h, n)
            h_bin, c5_bin = binary_local_search(h_bin, n, rng, max_iters=50000, time_fn=tl)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin.copy(), c5_bin

    best_h = project_onto_feasible(best_h, n)
    best_c5 = compute_overlap(best_h)
    return best_h, best_c5, n
