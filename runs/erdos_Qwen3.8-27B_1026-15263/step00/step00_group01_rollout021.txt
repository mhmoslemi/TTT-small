import numpy as np
import time
from scipy.optimize import minimize

def project_to_simplex(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    
    # Use bisection on the threshold
    lo, hi = 0, 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

def compute_c5(h):
    """Compute C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def subgradient_descent(h, n, max_iter=1000, lr=None, budget_s=60):
    """Optimize h using subgradient method with projection"""
    h = h.copy().astype(float)
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    if lr is None:
        lr = 0.5
    
    for it in range(max_iter):
        if time.time() - start > budget_s:
            break
        
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full") * dx
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active lags (within 1% of max)
        threshold = c5 * 0.99
        active = corr >= threshold
        
        # Compute subgradient
        # For each active lag, compute gradient
        # corr[j] corresponds to lag j - (n-1)
        grad = np.zeros(n)
        for j in np.where(active)[0]:
            lag = j - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                # f = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
                # d/dh[i] = (1-h[i+lag]) for i <= n-1-lag, minus h[i-lag] for i >= lag
                valid_i = np.arange(0, n - lag)
                g[valid_i] += (1 - h[valid_i + lag])
                g[lag:] -= h[:n-lag]
            else:
                lag = -lag
                # f = sum_{i=lag}^{n-1} h[i] * (1-h[i-lag])
                valid_i = np.arange(lag, n)
                g[valid_i] += (1 - h[valid_i - lag])
                g[:n-lag] -= h[lag:]
            grad += g
    
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
        
        # Update
        h = h - lr * grad
        h = project_to_simplex(h, target_sum)
        
        # Adaptive learning rate
        lr = max(0.01, lr * 0.999)
    
    return best_h, best_c5
