    return max_v + np.log(np.sum(np.exp(shifted))) / beta

    return float(np.max(corr) * dx)

def project_onto_set(y, n, target_sum):
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    h = np.clip(y - tau, 0.0, 1.0)
    diff = target_sum - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        order = np.argsort(-room) if diff > 0 else np.argsort(room)
        count = max(1, min(n, int(abs(diff) * 50) + 1))
        step = diff / count
        for idx in order[:count]:
            h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    return h

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_v = float(np.max(vals))
    log_w = beta * (vals - max_v)
    w = np.exp(log_w)
    w /= float(w.sum())
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k] * dx
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            term = np.zeros(n)
            term[:n - L] = 1.0 - h[L:]
            term[L:] -= h[:n - L]
            grad += wk * term
        else:
            Lp = -L
            term = np.zeros(n)
            term[Lp:] = 1.0 - h[:n - Lp]
            term[:n - Lp] -= h[Lp:]
            grad += wk * term
    return grad

def gd_phase(h, n, start_time, time_limit, beta, lr, momentum):
    target_sum = n / 2.0
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    count = 0
    while time.time() - start_time < time_limit:
        g = smooth_grad(h, beta)
        v = momentum * v - lr * g
        h_new = project_onto_set(h + v, n, target_sum)
        h = h_new
        count += 1
        if count % 100 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def subgradient_phase(h, n, start_time, time_limit, lr_decay, lr0):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    count = 0
    lr = lr0
    avg_h = h.copy()
    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        idx_max = int(np.argmax(corr))
        L = idx_max - (n - 1)
        g = np.zeros(n)
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:
            g[:n - L] = 1.0 - h[L:]
            g[L:] = -h[:n - L]
        else:
            Lp = -L
            g[Lp:] = 1.0 - h[:n - Lp]
            g[:n - Lp] = -h[Lp:]
        h_new = project_onto_set(h - lr * g, n, target_sum)
        avg_h = 0.99 * avg_h + 0.01 * h_new
        h = h_new
        count += 1
        lr = lr0 / (1.0 + lr_decay * count)
        if count % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            c5_avg = compute_c5(avg_h)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = avg_h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def sa_phase(h, n, start_time, time_limit, temp0):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    temp = temp0
    batch_count = 0
    while time.time() - start_time < time_limit:
        for _ in range(80):
            n_pert = int(np.random.randint(1, 9))
            idx = np.random.choice(n, n_pert, replace=False)
            h_new = h.copy()
            perturb = np.random.randn(n_pert) * 0.04
            h_new[idx] += perturb
            h_new = project_onto_set(h_new, n, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - curr_c5
            if delta < 0 or (temp > 1e-8 and np.random.random() < np.exp(-max(delta, -30) / temp)):
                h = h_new
                curr_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        batch_count += 1
        if batch_count % 8 == 0:
            temp *= 0.92
    return best_h, best_c5

def targeted_sa(h, n, start_time, time_limit, temp0):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    temp = temp0
    batch_count = 0
    while time.time() - start_time < time_limit:
        for _ in range(60):
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_k = int(np.random.randint(1, 4))
            top_indices = np.argsort(-corr)[:top_k]
            k_pick = top_indices[int(np.random.randint(0, len(top_indices)))]
            L = k_pick - (n - 1)
            h_new = h.copy()
            if L == 0:
                n_pairs = int(np.random.randint(2, 6))
                for _p in range(n_pairs):
                    i = int(np.random.randint(0, n))
                    h_new[i] -= 0.03 * np.random.random()
            elif L > 0:
                n_pairs = int(np.random.randint(2, 8))
                valid_i = np.arange(0, n - L)
                pairs = np.random.choice(valid_i, min(n_pairs, len(valid_i)), replace=False)
                for i in pairs:
                    amt = 0.02 * np.random.random()
                    h_new[i] -= amt
                    h_new[i + L] -= amt
            else:
                Lp = -L
                n_pairs = int(np.random.randint(2, 8))
                valid_i = np.arange(0, n - Lp)
                pairs = np.random.choice(valid_i, min(n_pairs, len(valid_i)), replace=False)
                for i in pairs:
                    amt = 0.02 * np.random.random()
                    h_new[i + Lp] -= amt
                    h_new[i] -= amt
            h_new = project_onto_set(h_new, n, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - curr_c5
            if delta < 0 or (temp > 1e-8 and np.random.random() < np.exp(-max(delta, -30) / temp)):
                h = h_new
                curr_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        batch_count += 1
        if batch_count % 10 == 0:
            temp *= 0.93
    return best_h, best_c5

def local_refine(h, n, start_time, time_limit):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    delta = 0.015
    while time.time() - start_time < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            old_val = h[i]
            for d in [delta, -delta, 0.5 * delta, -0.5 * delta, 0.25 * delta, -0.25 * delta]:
                h[i] = np.clip(old_val + d, 0.0, 1.0)
                h = project_onto_set(h, n, target_sum)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    old_val = h[i]
                    improved = True
                    break
            h[i] = old_val
            h = project_onto_set(h, n, target_sum)
        if not improved:
            delta *= 0.85
            if delta < 0.0005:
                break
        else:
            delta = max(0.0005, delta * 0.97)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    target_sum = n / 2.0
    h = project_onto_set(h, n, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    h1, c5_1 = gd_phase(h, n, start_time, 150, 40.0, 0.08, 0.85)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    h2, c5_2 = gd_phase(best_h, n, start_time, 350, 100.0, 0.03, 0.9)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    h3, c5_3 = subgradient_phase(best_h, n, start_time, 500, 0.5, 0.05)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    h4, c5_4 = sa_phase(best_h, n, start_time, 850, 0.025)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    h5, c5_5 = targeted_sa(best_h, n, start_time, 1100, 0.02)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    h6, c5_6 = local_refine(best_h, n, start_time, budget_s - 5)
    if c5_6 < best_c5:
        best_h, best_c5 = h6, c5_6

    best_h = project_onto_set(best_h, n, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
