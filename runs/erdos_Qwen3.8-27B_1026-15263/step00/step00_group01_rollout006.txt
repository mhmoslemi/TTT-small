            stall = 0
            lr = min(lr * 1.005, 0.2)
        else:
            stall += 1
            lr = max(lr * 0.99, 0.0005)
            if stall > 100:
                h = h + np.random.randn(n) * 0.03
                h = _project(h, n)
                lr = 0.01
                stall = 0
                mom = np.zeros(n)
    return h, best

def _coord_search(h, n, max_rounds, deadline):
    h = h.copy()
    best = _compute_c5(h)
    deltas = [0.0005, 0.001, 0.002, 0.005, 0.01, -0.0005, -0.001, -0.002, -0.005, -0.01]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for d in deltas:
                ht = h.copy()
                ht[i] = np.clip(ht[i] + d, 0.0, 1.0)
                ht = _project(ht, n)
                v = _compute_c5(ht)
                if v < best - 1e-14:
                    best = v
                    h = ht
                    improved = True
                    break
        if not improved:
            break
    return h, best

def _binary_swap(h, n, max_iters, deadline):
    h = h.copy()
    best = _compute_c5(h)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-9:
            continue
        ht = h.copy()
        ht[i], ht[j] = ht[j], ht[i]
        ht = _project(ht, n)
        v = _compute_c5(ht)
        if v < best - 1e-14:
            best = v
            h = ht
    return h, best

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 47
        h0 = np.full(n0, 0.5)

    best_h = _project(h0.copy(), n0)
    best_c5 = _compute_c5(best_h)

    dl1 = t0 + budget_s * 0.25
    h1, c51 = _local_opt(best_h, n0, 100000, dl1)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    dl2 = t0 + budget_s * 0.40
    h2, c52 = _coord_search(best_h, n0, 50, dl2)
    if c52 < best_c5:
        best_h, best_c5 = h2, c52

    dl3 = t0 + budget_s * 0.55
    h3, c53 = _local_opt(best_h, n0, 100000, dl3)
    if c53 < best_c5:
        best_h, best_c5 = h3, c53

    dl4 = t0 + budget_s * 0.65
    n0f = len(best_h)
    for r in range(40):
        if time.time() > dl4:
            break
        scale = 0.04 + 0.02 * r
        hp = best_h + np.random.randn(n0f) * scale
        hp = _project(hp, n0f)
        hd, cd = _local_opt(hp, n0f, 20000, min(dl4, time.time() + budget_s * 0.01))
        if cd < best_c5:
            best_h, best_c5 = hd, cd

    n0f = len(best_h)
    dl5 = t0 + budget_s * 0.70
    h5, c55 = _coord_search(best_h, n0f, 80, dl5)
    if c55 < best_c5:
        best_h, best_c5 = h5, c55

    dl6 = t0 + budget_s * 0.92
    n_list = [48, 49, 50, 51, 52, 54, 56, 200]
    for n_try in n_list:
        if time.time() > dl6:
            break
        if n_try == n0f:
            continue
        x_old = np.linspace(0, 2, n0f, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = _project(hi, n_try)

        h_try, c5_try = _local_opt(hi, n_try, 50000, dl6)
        h_try2, c5_try2 = _coord_search(h_try, n_try, 30, dl6)
        if c5_try2 < c5_try:
            h_try, c5_try = h_try2, c5_try2

        if n_try % 2 == 0 and time.time() < dl6:
            hb = np.zeros(n_try)
            idx = np.random.permutation(n_try)
            hb[idx[:n_try // 2]] = 1.0
            hb_c, c5b = _binary_swap(hb, n_try, 80000, dl6)
            if c5b < c5_try:
                h_try, c5_try = hb_c, c5b
            hb3 = (hi > 0.5).astype(float)
            hb3 = _project(hb3, n_try)
            hb4, c5b4 = _binary_swap(hb3, n_try, 80000, dl6)
            if c5b4 < c5_try:
                h_try, c5_try = hb4, c5b4

        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    nf = len(best_h)
    dl_final = t0 + budget_s * 0.97
    best_h, best_c5 = _local_opt(best_h, nf, 100000, dl_final)
    best_h, best_c5 = _coord_search(best_h, nf, 30, dl_final)

    nf = len(best_h)
    best_c5 = _compute_c5(best_h)
    return (best_h, best_c5, nf)
