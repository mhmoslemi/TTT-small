But actually, there's an even simpler approach for the 1D minimization of max of lines: use binary search. Since f(d) = max_i (c_i + a_i * d) is convex, we can find the minimum by:
- If the subgradient at d=0 contains 0, then d=0 is optimal
- Otherwise, binary search for where the subgradient changes sign

The subgradient of f at d is the convex hull of {a_i : c_i + a_i * d = f(d)}, i.e., the set of slopes of lines that achieve the maximum at d.

So: compute f(d) = max_i (c_i + a_i * d), find which i achieves it (there might be multiple), and the subgradient is the range of their a_i values. If 0 is in this range, we're done. Otherwise, if all achieving a_i > 0, we need to decrease d. If all < 0, increase d.

```python
def minimize_max_lines(c, a, d_min, d_max):
    """Minimize max_i (c[i] + a[i] * d) over d in [d_min, d_max]."""
    # Check boundaries
    f_min = np.max(c + a * d_min)
    f_max = np.max(c + a * d_max)
    
    # Binary search
    lo, hi = d_min, d_max
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        vals = c + a * mid
        m = np.max(vals)
        # Find achieving indices
        achieving = np.where(np.abs(vals - m) < 1e-12)[0]
        slopes = a[achieving]
        if slopes.min() <= 0 <= slopes.max():
            return mid
        elif slopes.min() > 0:
            hi = mid
        else:
            lo = mid
    
    # Return best of the three
    d_candidates = [d_min, d_max, (lo + hi) / 2]
    best_d = d_candidates[0]
    best_val = np.max(c + a * best_d)
    for d in d_candidates[1:]:
        val = np.max(c + a * d)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d
```

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14 * n:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        count = int(np.sum(mask))
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h


def minimize_max_lines(c, a, d_lo, d_hi):
    """Minimize max_i (c[i] + a[i]*d) over d in [d_lo, d_hi]."""
    if d_lo >= d_hi:
        return d_lo
    # Check boundaries
    best_d = d_lo
    best_val = float(np.max(c + a * d_lo))
    v_hi = float(np.max(c + a * d_hi))
    if v_hi < best_val:
        best_val = v_hi
        best_d = d_hi
    # Binary search for zero subgradient
    lo, hi = d_lo, d_hi
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = c + a * mid
        m = float(np.max(vals))
        if m < best_val:
            best_val = m
            best_d = mid
        # Subgradient: set of a[i] where vals[i] ≈ m
        thr = m - 1e-9 * (abs(m) + 1.0)
        achieving = vals >= thr
        if not np.any(achieving):
            break
        slopes = a[achieving]
        if slopes.min() <= 0.0 <= slopes.max():
            return best_d
        elif slopes.min() > 0.0:
            hi = mid
        else:
            lo = mid
    # Check midpoint
    mid = 0.5 * (lo + hi)
    v = float(np.max(c + a * mid))
    if v < best_val:
        best_val = v
        best_d = mid
    return best_d


def compute_A(j, h, n):
    """Compute the coefficient vector A for coordinate j."""
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    L_all = np.arange(-(n - 1), n)  # lags
    
    hj = h[j]
    
    # Positive lags: L = 0, 1, ..., n-1 (indices n-1 to 2n-2 in A)
    L_pos = np.arange(0, n)  # 0 to n-1
    idx_pos = n - 1 + L_pos  # indices into A
    
    # Term 1: (1-h[j-L]) for L in [0, min(j, n-1)] and j-L in [0, n-1]
    v1 = (L_pos <= j) & (L_pos <= n - 1)
    # j-L >= 0 is guaranteed by L <= j, j-L <= n-1 is guaranteed by j <= n-1
    A[idx_pos[v1]] += (1.0 - h[j - L_pos[v1]])
    
    # Term 2: -h[j+L] for L in [0, min(n-1-j, n-1)] and j+L in [0, n-1]
    v2 = (L_pos <= n - 1 - j) & (L_pos <= n - 1)
    A[idx_pos[v2]] -= h[j + L_pos[v2]]
    
    # Negative lags: L = -1, -2, ..., -(n-1) (indices 0 to n-2 in A)
    # M = -L = 1, 2, ..., n-1
    M = np.arange(1, n)  # 1 to n-1
    idx_neg = n - 1 - M  # indices into A (0 to n-2)
    
    # Term 1: (1-h[j+M]) for M <= n-1-j and j+M in [0, n-1]
    w1 = (M <= n - 1 - j) & (M <= n - 1)
    A[idx_neg[w1]] += (1.0 - h[j + M[w1]])
    
    # Term 2: -h[j-M] for M <= j and j-M in [0, n-1]
    w2 = (M <= j) & (M <= n - 1)
    A[idx_neg[w2]] -= h[j - M[w2]]
    
    return A


def coordinate_descent(h, n, max_sweeps=500, tol=1e-12, verbose=False):
    dx = 2.0 / n
    h = np.asarray(h, dtype=float).copy()
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        # Randomize sweep direction occasionally to avoid cycling
        if sweep % 10 == 5:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            A = compute_A(j, h, n)
            hj = h[j]
            d_lo = -hj
            d_hi = 1.0 - hj
            best_d = minimize_max_lines(corr, A, d_lo, d_hi)
            new_hj = np.clip(hj + best_d, 0.0, 1.0)
            delta = new_hj - hj
            if abs(delta) > max_change:
                max_change = abs(delta)
            if abs(delta) > 1e-15:
                h[j] = new_hj
                corr += A * delta
        
        # Periodic recomputation to avoid drift
        if sweep % 50 == 49:
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        
        if max_change < tol:
            break
    
    c5 = float(np.max(corr) * dx)
    return h, c5


def smooth_lbfobs(h, n, dx, T=20.0, penalty=1000.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    exp_vals = np.exp((vals - m) * T)
    smax = m + np.log(np.sum(exp_vals)) / T
    penalty_term = penalty * (np.sum(h) - n / 2.0) ** 2
    return smax + penalty_term
