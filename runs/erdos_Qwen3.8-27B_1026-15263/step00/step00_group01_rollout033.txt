def c5_value(h, dx):
    """Compute C5 for a given h vector."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def gradient_c5(h, dx):
    """Compute the gradient of C5 (subgradient at the max)."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_idx = np.argmax(corr)
    n = len(h)
    # c[k] = sum_j h[j] * one_minus_h[j - (max_idx - (n-1))]
    # The shift k = max_idx - (n-1)
    k = max_idx - (n - 1)

    # c_k = sum_j h[j+k] * (1-h[j]) (with valid indices)
    # d/dh[i] = (1-h[i-k]) * I(valid i-k) - h[i+k] * I(valid i+k)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # i as the "first" index: h[i] appears as h[j+k] when j+k=i, j=i-k

def smooth_max(c, beta):
    """Smooth maximum using log-sum-exp."""
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp((c - c_max) * beta))) / beta

def objective(h, beta, dx, n):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c = corr * dx

from scipy.optimize import minimize

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum h = target_sum} using bisection."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp((c - c_max) * beta))) / beta

def subgradient(h, dx, n):
    """Compute subgradient of max_k c[k] at h."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i - k < n:
            g += (1.0 - h[i - k])
        if 0 <= i + k < n:
            g -= h[i + k]
        grad[i] = g
    return dx * grad
