    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h = h.copy()
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _con_fun(h):
    return np.sum(h) - _G_TARGET

def _con_jac(h):
    return np.ones(len(h))

def _nm_obj(h):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * dx)
    return val + 10.0 * (np.sum(h) - _G_TARGET) ** 2

def _lse_obj(h):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / _G_N
    T = _G_T
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    s = np.clip((corr - mc) / T, -400.0, 400.0)
    return (mc + T * np.log(np.sum(np.exp(s)))) * dx

def _do_nm(h0, maxiter):
    r = minimize(_nm_obj, h0, method='Nelder-Mead',
                 options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
    return _project(np.clip(r.x, 0.0, 1.0), _G_TARGET)

def _do_slsqp(h0, T, maxiter):
    global _G_T
    _G_T = T
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * _G_N
    r = minimize(_lse_obj, h0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return _project(np.clip(r.x, 0.0, 1.0), _G_TARGET)

def _swap_search(h, max_iters):
    h = h.copy()
    c5c = _c5_eval(h)
    n = len(h)
    for _ in range(max_iters):
        a = int(np.random.randint(n))
        b = int(np.random.randint(n))
        if a == b:
            continue
        h[a], h[b] = h[b], h[a]
        c5n = _c5_eval(h)
        if c5n <= c5c:
            c5c = c5n
        else:
            h[a], h[b] = h[b], h[a]
    return h, c5c
