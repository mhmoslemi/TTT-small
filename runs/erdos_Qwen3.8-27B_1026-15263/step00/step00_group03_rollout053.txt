def repair_h(h, n_points):
    target = n_points / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    s = np.sum(h)
    if s > 1e-10:
        h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h


def coordinate_descent(h, n_points, max_passes=300, tol=1e-10):
    best_c5 = evaluate_c5(h, n_points)
    grid = np.linspace(0.0, 1.0, 31)

    for p in range(max_passes):
        improved = False
        for i in range(n_points):
            old_val = h[i]
            h_backup = h.copy()
            best_obj = best_c5
            best_h = h.copy()
            best_t = old_val

            for t in grid:
                if abs(t - old_val) < 1e-8:
                    continue
                h_try = h_backup.copy()
                h_try[i] = t
                h_rep = repair_h(h_try, n_points)
                obj = evaluate_c5(h_rep, n_points)
                if obj < best_obj - tol:
                    best_obj = obj
                    best_t = t
                    best_h = h_rep

            if best_t != old_val:
                lo = max(0.0, best_t - 0.03)
                hi = min(1.0, best_t + 0.03)
                fine_grid = np.linspace(lo, hi, 15)
                for t in fine_grid:
                    h_try = h_backup.copy()
                    h_try[i] = t
                    h_rep = repair_h(h_try, n_points)
                    obj = evaluate_c5(h_rep, n_points)
                    if obj < best_obj - tol:
                        best_obj = obj
                        best_t = t
                        best_h = h_rep
                h = best_h
                improved = True
                best_c5 = best_obj

        if not improved:
            break

    return h, best_c5


def nm_objective(x, n_points):
    x = repair_h(x, n_points)
    return evaluate_c5(x, n_points)


def nelder_mead_optimize(h0, n_points, max_iter=300):
    h0 = h0.copy()
    result = minimize(
        nm_objective, h0, args=(n_points,),
        method='Nelder-Mead',
        options={'maxiter': max_iter, 'xatol': 1e-7, 'fatol': 1e-9,
                 'initial_simplex': h0 + 0.01 * np.ones(n_points)}
    )
    h_out = repair_h(result.x, n_points)
    c5_out = evaluate_c5(h_out, n_points)
    return h_out, c5_out


def random_01_start(n_points, rng):
    idx = rng.permutation(n_points)
    h = np.zeros(n_points)
    h[idx[:n_points // 2]] = 1.0
    return h


def periodic_start(n_points, period, duty, rng):
    h = np.zeros(n_points)
    for block_start in range(0, n_points, period):
        end = min(block_start + int(period * duty), n_points)
        noise = rng.uniform(-0.1, 0.1, size=min(period, n_points - block_start))
        h[block_start:end] = np.clip(1.0 + noise, 0, 1)
    return repair_h(h, n_points)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    n_points = 70

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) != n_points:
            n_points = len(h0)
    else:
        h0 = np.ones(n_points) * 0.5

    h0 = repair_h(h0, n_points)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, n_points)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Coordinate descent from initial
    if budget_s - elapsed() > 5:
        h_cd, c5_cd = coordinate_descent(h0, n_points, max_passes=200)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 2: Nelder-Mead from best
    if budget_s - elapsed() > 20:
        h_nm, c5_nm = nelder_mead_optimize(best_h, n_points, max_iter=200)
        if c5_nm < best_c5:
            best_h = h_nm
            best_c5 = c5_nm
        h_cd, c5_cd = coordinate_descent(best_h, n_points, max_passes=100)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 3: Multiple random restarts
    restart_count = 0
    while budget_s - elapsed() > 40:
        restart_count += 1
        # Generate diverse starting points
        if restart_count % 3 == 1:
            h_start = random_01_start(n_points, rng)
        elif restart_count % 3 == 2:
            period = rng.integers(5, 20)
            duty = rng.uniform(0.3, 0.7)
            h_start = periodic_start(n_points, period, duty, rng)
        else:
            # Perturb best solution
            strength = max(0.05, 0.3 * (1.0 - restart_count / 30.0))
            h_start = best_h + rng.uniform(-strength, strength, size=n_points)
            h_start = repair_h(h_start, n_points)

        # Coordinate descent
        h_try, c5_try = coordinate_descent(h_start, n_points, max_passes=150)

        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            # Nelder-Mead refinement if time allows
            if budget_s - elapsed() > 30:
                h_nm, c5_nm = nelder_mead_optimize(best_h, n_points, max_iter=150)
                if c5_nm < best_c5:
                    best_h = h_nm
                    best_c5 = c5_nm
                h_cd, c5_cd = coordinate_descent(best_h, n_points, max_passes=100)
                if c5_cd < best_c5:
                    best_h = h_cd
                    best_c5 = c5_cd

    # Phase 4: Final intensive polish
    if budget_s - elapsed() > 5:
        h_final, c5_final = coordinate_descent(best_h, n_points, max_passes=500)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Try Nelder-Mead one more time if time allows
    if budget_s - elapsed() > 30:
        h_nm, c5_nm = nelder_mead_optimize(best_h, n_points, max_iter=200)
        if c5_nm < best_c5:
            best_h = h_nm
            best_c5 = c5_nm
        h_cd, c5_cd = coordinate_descent(best_h, n_points, max_passes=200)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    best_h = repair_h(best_h, n_points)
    best_c5 = evaluate_c5(best_h, n_points)

    return best_h, best_c5, n_points
