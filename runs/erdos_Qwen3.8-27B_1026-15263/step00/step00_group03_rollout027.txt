        if abs(t) < 1e-12:
            continue
        val = float(np.max(g + t * d))
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    return best_t, best_val

def local_search(h, lags, n, pair_list, time_limit=None, start_time=None):
    g = compute_lags(h, n)
    cur_max = float(np.max(g))
    
    for i, j in pair_list:
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        t_lo = max(-h[j], h[i] - 1.0)
        t_hi = min(h[i], 1.0 - h[j])
        if t_hi - t_lo < 1e-10:
            continue
        d = grad_at_position(h, j, lags, n) - grad_at_position(h, i, lags, n)
        best_t, best_val = find_best_transfer(g, d, t_lo, t_hi)
        if abs(best_t) > 1e-10:
            h[i] -= best_t
            h[j] += best_t
            g += best_t * d
            cur_max = best_val
    return h, g, cur_max
