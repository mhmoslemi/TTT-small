from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 = dx * max(corr)"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_onto_constraint(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment for exact sum
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        if np.any(mask):
            h_p[mask] += diff / np.sum(mask)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def smooth_c5(h, n, dx, tau):
    """Smooth C5 using log-sum-exp"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * corr
    m = np.max(scaled)
    return dx * (m + np.log(np.sum(np.exp(scaled - m)))) / tau

def smooth_c5_grad(h, n, dx, tau):
    """Gradient of smooth C5"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * corr
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-12:
            continue
        d = p - (n - 1)
        if d >= 0:
            idx = np.arange(n - d)
            grad[idx] += w * (1.0 - h[idx + d])
            grad[idx + d] -= w * h[idx]
        else:
            m2 = -d
            idx = np.arange(m2, n)
            grad[idx] += w * (1.0 - h[idx - m2])
            grad[idx - m2] -= w * h[idx]

def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
