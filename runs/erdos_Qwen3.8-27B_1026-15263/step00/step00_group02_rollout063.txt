```python
def project_to_bounded_simplex(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x_i <= 1}"""
    h = h.copy()
    # Binary search for the water level
    lo, hi = -target_sum, 1.0 + target_sum  # wide bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    return projected
```
