import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max_c5(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = tau * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / tau

def smooth_max_c5_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(tau * (corr - m))
    w = exps / np.sum(exps)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        j = k - (n - 1)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if j >= 0:
            term1[:n - j] = one_minus_h[j:]
            term2[j:] = h[:n - j]
        else:
            jj = -j
            term1[jj:] = one_minus_h[:n - jj]
            term2[:n - jj] = h[jj:]
        grad += wk * (term1 - term2)
    return grad

    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            h += room * (diff / total_room)
        else:
            room = h.copy()
            room[h <= 0.0] = 0.0
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            h += room * (diff / total_room)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

                h[idx] += add
                diff -= add
                if abs(diff) < 1e-13:

        else:
            order = np.argsort(h)
            for idx in order:
                rem = max(diff, -(h[idx]))
                h[idx] += rem
                diff -= rem
                if abs(diff) < 1e-13:

    return h

def try_optimize(h_start, n, tau=100.0, maxiter=400, t_start=None, budget_s=None):
    h_start = project_to_feasible(h_start, n)
    f0 = smooth_max_c5(h_start, tau)

    def obj(x):
        return smooth_max_c5(x, tau)

    def grad(x):
        return smooth_max_c5_grad(x, tau)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj, h_start, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.fun < f0:
        h_opt = project_to_feasible(result.x, n)
        return h_opt
    return h_start

def generate_random_h(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        num_ones = int(np.floor(n / 2.0))
        frac = n / 2.0 - num_ones
        idx = rng.choice(n, size=num_ones, replace=False)
        h[idx] = 1.0
        if frac > 0:
            remaining = np.setdiff1d(np.arange(n), idx)
            h[rng.choice(remaining, size=1, replace=False)] = frac
    elif mode == 1:

        h = 0.5 + r
    elif mode == 2:

    elif mode == 3:

            h[idx] = 0.5
