```python
import time

    v_max = np.max(v)
    shifted = alpha * (v - v_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    e = np.exp(shifted)
    return v_max + np.log(np.sum(e)) / alpha

def smooth_max_grad(v, alpha):
    v_max = np.max(v)
    shifted = alpha * (v - v_max)
    shifted = np.clip(shifted, -100, 100)
    e = np.exp(shifted)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_val(corr, alpha):
    v_max = np.max(corr)
    shifted = alpha * (corr - v_max)
    shifted = np.clip(shifted, -50, 100)
    e = np.exp(shifted)
    return v_max + np.log(e.sum()) / alpha

def fix_sum(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    current = np.sum(h)
    deficit = target_sum - current
    if abs(deficit) < 1e-14:
        return h
    if deficit > 0:
        slack = 1.0 - h
        total = np.sum(slack)
        if total <= 1e-12:
            return np.clip(h, 0, 1)
        h += deficit * slack / total
    else:
        total = np.sum(h)
        if total <= 1e-12:
            return np.zeros_like(h)
        h += deficit * h / total
    return np.clip(h, 0, 1)

def obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    v_max = np.max(corr)
    shifted = alpha * (corr - v_max)
    shifted = np.clip(shifted, -50, 100)
    e = np.exp(shifted)
    w = e / e.sum()
    
    val = (v_max + np.log(e.sum()) / alpha) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sw = w[k]
        s = k - (n - 1)
        if s == 0:
            grad += sw * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += sw * one_minus_h[s:]
            grad[s:] += sw * (-h[:n-s])
        else:
            t = -s
            grad[t:] += sw * one_minus_h[:n-t]
            grad[:n-t] += sw * (-h[t:])
    
    grad *= dx
    return val, grad

def projected_gradient_descent(h, alpha, max_iter=500, lr=0.01):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    best_val = compute_c5(h, dx)
    
    for it in range(max_iter):
        val, grad = obj_and_grad(h, alpha)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project: clip to [0,1] and fix sum
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, target_sum)
        
        # Line search
        for _ in range(20):
            candidate = h_new
            cand_val = compute_c5(candidate, dx)
            if cand_val < best_val:
                h = candidate
                best = candidate.copy()
                best_val = cand_val
                break
            else:
                h_new = h - lr * 0.5 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = fix_sum(h_new, target_sum)
                cand_val = compute_c5(h_new, dx)
                if cand_val < best_val:
                    h = h_new
                    best = h_new.copy()
                    best_val = cand_val
                    break
                lr *= 0.5
        
        if lr < 1e-10:
            break
    
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h0 = fix_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [5, 10, 25, 50, 100, 200, 500, 1000, 2000, 5000]
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.5:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0-x, mode='full')
            return smooth_max_val(corr, alpha) * (2.0/len(x))
        
        def grad_obj(x):
            _, g = obj_and_grad(x, alpha)
            return g
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(obj, best_h, jac=grad_obj, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            new_h = fix_sum(res.x, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        except Exception:
            pass
    
    # Phase 2: Projected gradient descent refinement
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.8:
        for alpha in [100, 500, 2000]:
            if time.time() - t_start > budget_s * 0.8:
                break
            h_new, c5_new = projected_gradient_descent(best_h, alpha, max_iter=300, lr=0.005)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 3: Random restarts

    if elapsed < budget_s * 0.9:

            # Random perturbation

                        _, g = obj_and_grad(x, alpha)
                        return g
                    
                    res = minimize(obj_r, h_pert, jac=grad_r, method='SLSQP',
                                  bounds=bounds, constraints=cons,

                    new_h = fix_sum(res.x, target_sum)

                        best_c5 = new_c5
                        h_pert = best_h.copy()
                except:
                    pass
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)
    deficit = target_sum - np.sum(h)
    if abs(deficit) < 1e-14:
        return h
    if deficit > 0:

        total = slack.sum()
        if total <= 1e-12:
            return h
        h += deficit * slack / total
    else:
        total = h.sum()
        if total <= 1e-12:
            return np.zeros_like(h)
        h += deficit * h / total
    return np.clip(h, 0, 1)

def obj_grad(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    v_max = np.max(corr)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)
    deficit = target_sum - np.sum(h)
    if abs(deficit) < 1e-14:
        return h
    if deficit > 0:
        slack = 1.0 - h
        total = slack.sum()
        if total <= 1e-12:
            return h
        h += deficit * slack / total
    else:
        total = h.sum()
        if total <= 1e-12:
            return np.zeros_like(h)
        h += deficit * h / total
    return np.clip(h, 0, 1)

def obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v_max = np.max(corr)
    shifted = alpha * (corr - v_max)
    shifted = np.clip(shifted, -50, 100)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (v_max + np.log(e.sum()) / alpha) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo > m_hi:
            continue
        idx_start = k - m_hi
        idx_end = k - m_lo + 1
        grad[m_lo:m_hi+1] += w[k] * (1.0 - 2.0 * h[idx_start:idx_end][::-1])
    grad *= dx
    return val, grad

def slsqp_optimize(h0, target_sum, n, max_time):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    alphas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for alpha in alphas:
        if time.time() - t0 > max_time:
            break
        try:
            def obj(x, a=alpha):
                return obj_grad(x, a)[0]
            def jac(x, a=alpha):
                return obj_grad(x, a)[1]
            res = minimize(obj, best_h, jac=jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            new_h = fix_sum(res.x, target_sum)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        except Exception:
            pass
    return best_h, best_c5

def nelder_mead_refine(h0, target_sum, n, max_time):
    t0 = time.time()
    def obj(x):
        return compute_c5(fix_sum(x, target_sum))
    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                      options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11,
                               'adaptive': True})
        new_h = fix_sum(res.x, target_sum)
        new_c5 = compute_c5(new_h)
        return new_h, new_c5
    except Exception:
        return h0, compute_c5(h0)

def projected_gd(h0, target_sum, n, alpha, max_time):
    t0 = time.time()
    h = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    lr = 0.01
    for it in range(3000):
        if time.time() - t0 > max_time:
            break
        _, g = obj_grad(h, alpha)
        h_new = fix_sum(h - lr * g, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best = h_new.copy()
            best_c5 = c5
            h = h_new.copy()
        else:
            lr *= 0.5
            if lr < 1e-13:
                break
    return best, best_c5
```
