import numpy as np
from scipy.optimize import minimize
import time


def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def nelder_mead_wrapper(obj, x0, n, maxiter):
    def inner(x):
        return obj(x)
    res = minimize(inner, x0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
    return res.x


def coordinate_descent(h, n, steps, max_sweeps, deadline):
    h_work = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False

            if time.time() > deadline:
                break
            cur = compute_c5(h_work, n)

                trial[j] = np.clip(trial[j] + s, 0.0, 1.0)
                trial = repair(trial, n)
                val = compute_c5(trial, n)
                if val < cur - 1e-14:
                    h_work = trial.copy()
                    improved = True
                    break
        if not improved:
            break
    return h_work
