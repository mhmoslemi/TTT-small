    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def project_sum(h, target):
    """Project h to have sum = target, maintaining [0,1] bounds."""
    h = h.copy()
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: increase entries that are < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            # Proportional adjustment
            room = 1.0 - h[mask]
            if np.sum(room) < diff:
                h[mask] = 1.0
            else:
                h[mask] += diff / np.sum(room) * room
        else:
            # Need to decrease sum: decrease entries that are > 0
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            if np.sum(room) < -diff:
                h[mask] = 0.0
            else:
                h[mask] += diff / np.sum(room) * room
    return np.clip(h, 0.0, 1.0)

def pairwise_cd(h, target, max_passes=20, tol=1e-10):
    """Pairwise coordinate descent maintaining sum constraint."""
    n = len(h)
    h = h.copy()
    best_obj = compute_c5_raw(h)
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Try moving mass from i to j
                # h[i] + t, h[j] - t
                # t range: t_min = max(-h[i], -(1.0 - h[j])), t_max = min(1.0 - h[i], h[j])
                t_min = max(-h[i], -(1.0 - h[j]))
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-12:
                    continue
                # Golden section search
                best_t = 0.0
                best_val = best_obj
                # Try a few values
                for t in [t_min, t_max, 0.0, (t_max - t_min) * 0.25, (t_max - t_min) * 0.5, (t_max - t_min) * 0.75]:
                    t = np.clip(t, t_min, t_max)
                    h[i], h[j] = h[i] + t, h[j] - t
                    val = compute_c5_raw(h)
                    if val < best_val:
                        best_val = val
                        best_t = t
                    h[i], h[j] = h[i] - t, h[j] + t
                if best_t != 0.0:
                    h[i] += best_t
                    h[j] -= best_t
                    if best_val < best_obj - tol:
                        improved = True
                    best_obj = best_val
        if not improved:
            break
    
    return h
