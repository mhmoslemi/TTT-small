        T = max(T * cool, T_floor)
    return best_local, best_local_c5


def transfer_sa(h, n, dx, target, rng, t_limit, T_init=0.01, T_floor=0.0002, cool=0.9998):
    h = h.copy()
    cur_c5 = compute_c5(h, dx)
    best_local = h.copy()
    best_local_c5 = cur_c5
    T = T_init
    while time.time() < t_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amt = rng.uniform(-0.5, 0.5)
        if h[i] + amt < 0 or h[j] - amt < 0 or h[i] + amt > 1 or h[j] - amt > 1:
            continue
        oi = h[i]
        oj = h[j]
        h[i] += amt
        h[j] -= amt
        new_c5 = compute_c5(h, dx)
        d = new_c5 - cur_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_c5 = new_c5
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local = h.copy()
        else:
            h[i] = oi
            h[j] = oj
        T = max(T * cool, T_floor)
    return best_local, best_local_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except (NameError, UnboundLocalError):
        n = 47
        h_init = make_binary_initial(n, rng)

    dx = 2.0 / n
    target = n / 2.0
    h_init = fix_sum(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)

    t_phase1 = t0 + budget_s * 0.45
    t_phase2 = t0 + budget_s * 0.65
    t_phase3 = t0 + budget_s * 0.82
    t_phase4 = t0 + budget_s * 0.98

    # Phase 1: Swap SA (preserves sum exactly, stays near-binary)
    if time.time() < t_phase1:
        h1, c5_1 = swap_sa(best_h, n, dx, target, rng, t_phase1, T_init=0.025, T_floor=0.0001, cool=0.9997)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Transfer SA (continuous, preserves sum)
    if time.time() < t_phase2:
        h2, c5_2 = transfer_sa(best_h, n, dx, target, rng, t_phase2, T_init=0.008, T_floor=0.0001, cool=0.9997)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: L-BFGS-B continuous optimization
    if time.time() < t_phase3:
        def obj(x):
            hx = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            c5val = np.max(corr) * dx
            pen = 200.0 * (np.sum(hx) - target) ** 2
            return c5val + pen

        bounds = list(zip(np.zeros(n), np.ones(n)))
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-15})
            nh = fix_sum(np.clip(res.x, 0, 1), target)
            nc5 = compute_c5(nh, dx)
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = nh
        except Exception:
            pass

    # Phase 4: Multi-restart with local SA
    t_recommend = t0 + budget_s * 0.985
    while time.time() < t_recommend:
        # Random perturbation
        sigma = rng.uniform(0.05, 0.25)
        h_try = best_h + rng.randn(n) * sigma
        h_try = fix_sum(h_try, target)
        c5_try = compute_c5(h_try, dx)

        # Quick local swap SA from perturbed point
        h_local = h_try.copy()
        c5_local = c5_try
        T = 0.008
        n_inner = 8000
        for _ in range(n_inner):
            if time.time() > t_recommend:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_local[i], h_local[j] = h_local[j], h_local[i]
            nc = compute_c5(h_local, dx)
            dd = nc - c5_local
            if dd < 0 or rng.random() < np.exp(-dd / T):
                c5_local = nc
            else:
                h_local[i], h_local[j] = h_local[j], h_local[i]
            T = max(T * 0.9995, 0.0001)

        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()

        # Also try a pure binary restart occasionally
        if rng.random() < 0.1:
            h_bin = make_binary_initial(n, rng)
            h_bin = fix_sum(h_bin, target)
            h_bin2, c5_bin = swap_sa(h_bin, n, dx, target, rng,
                                     min(t_recommend, time.time() + 5),
                                     T_init=0.015, T_floor=0.0001, cool=0.9996)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin2

    # Final verification
    best_h = fix_sum(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
