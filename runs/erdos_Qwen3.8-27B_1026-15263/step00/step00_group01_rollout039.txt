def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr)) * dx

def repair_constraint(h, n):
    """Repair h so that sum(h) == n/2 exactly and 0<=h<=1."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add diff total; add to elements with room (h[i] < 1)
        order = np.argsort(1.0 - h)  # smallest room first... no, we want to add to those with most room
        # Actually, we want to distribute the addition. Simple approach: add to the element with most room.
        # Better: add proportionally to room.
        room = 1.0 - h
        total_room = room.sum()
        if total_room < diff - 1e-12:
            # Can't fit; set all to 1
            h = np.ones(n)
            # This shouldn't happen if target = n/2
            return h
        h = h + room * (diff / total_room)
    else:
        # Need to subtract; subtract from elements with excess (h[i] > 0)
        excess = h
        total_excess = excess.sum()
        if total_excess < -diff - 1e-12:
            h = np.zeros(n)
            return h
        h = h - excess * ((-diff) / total_excess)
    
    h = np.clip(h, 0, 1)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        i = np.argmax(1 - h) if diff > 0 else np.argmax(h)
        h[i] += diff
        h[i] = np.clip(h[i], 0, 1)
    return h

def coord_descent_sweep(h):
    """One full sweep of coordinate descent. Returns updated h."""
    n = len(h)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    
    for j in range(n):
        # Compute A vector for coordinate j
        # For lag s (output index k = s + n - 1):
        # contribution to c[k] from h[j]:
        #   if j-s in [0,n-1]: (1-h[j-s]) * h[j]
        #   if j+s in [0,n-1]: -h[j+s] * h[j] + h[j+s]
        # So A[k] = (1-h[j-s])*[valid] - h[j+s]*[valid]
        #    C contribution from this term = h[j+s]*[valid]
        
        A = np.zeros(2*n-1)
        for k in range(2*n-1):
            s = k - (n-1)
            js = j - s
            jp = j + s
            val = 0.0
            if 0 <= js < n:
                val += (1.0 - h[js])
            if 0 <= jp < n:
                val -= h[jp]
            A[k] = val
        
        # C = corr - A * h[j] gives the part not involving h[j]
        C = corr - A * h[j]
        
        # Find t in [0,1] minimizing max(A*t + C)
        # Candidates: 0, 1, and intersection points
        m = 2*n - 1
        
        # Vectorized intersection computation
        # t[i,j] = (C[j] - C[i]) / (A[i] - A[j])
        Ai = A[:, None]  # (m, 1)
        Aj = A[None, :]  # (1, m)
        Ci = C[:, None]
        Cj = C[None, :]
        denom = Ai - Aj  # (m, m)
        numer = Cj - Ci  # (m, m)
        mask = np.abs(denom) > 1e-14
        with np.errstate(divide='ignore', invalid='ignore'):
            t_mat = np.where(mask, numer / np.where(mask, denom, 1.0), np.nan)
        
        valid = (t_mat >= -1e-10) & (t_mat <= 1 + 1e-10) & mask
        candidates_1d = t_mat[valid]
        candidates_1d = np.clip(candidates_1d, 0, 1)
        candidates = np.unique(np.round(candidates_1d, 12))
        candidates = np.concatenate(([0.0, 1.0], candidates))
        
        # Evaluate max(A*t + C) at each candidate

        if abs(best_t - h[j]) > 1e-12:
            delta = best_t - h[j]
            corr += A * delta
            h[j] = best_t
    
    return h

def compute_c5(h):

    dx = 2.0 / n

def repair_constraint(h, n):

def coord_descent_sweep(h):
    n = len(h)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    for j in range(n):
        js = j - lags
        jp = j + lags
        A = np.zeros(m)
        mask_js = (js >= 0) & (js < n)
        mask_jp = (jp >= 0) & (jp < n)
        if np.any(mask_js):
            A[mask_js] += (1.0 - h[js[mask_js]])
        if np.any(mask_jp):
            A[mask_jp] -= h[jp[mask_jp]]
        C = corr - A * h[j]
        Ai = A[:, None]
        Aj = A[None, :]
        denom = Ai - Aj
        numer = C[None, :] - C[:, None]
        mask = np.abs(denom) > 1e-14
        with np.errstate(divide='ignore', invalid='ignore'):
