    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0

    best_h = h_init.copy()
    best_c5 = _c5_value(best_h, dx)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x, t=target: np.sum(x) - t,
             'jac': lambda x, nn=n: np.ones(nn)}]

    def obj_smooth(h, alpha=150.0):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return _smooth_max(corr, alpha) * dx

    def obj_max(h):
        return _c5_value(h, dx)

    # Generate diverse starts
    starts = [h_init.copy()]
    for _ in range(20):
        h_p = h_init + np.random.normal(0, 0.08, n)
        h_p = _make_feasible(h_p, n, target)
        starts.append(h_p)
    for _ in range(8):
        h_p = h_init.copy()
        k = np.random.choice(n, size=max(1, n // 6), replace=False)
        h_p[k] = np.random.uniform(0, 1, size=len(k))
        h_p = _make_feasible(h_p, n, target)
        starts.append(h_p)
    for _ in range(5):
        h_p = np.random.uniform(0, 1, n)
        h_p = _make_feasible(h_p, n, target)
        starts.append(h_p)

    # Phase 1: SLSQP from multiple starts
    for i, h_s in enumerate(starts):
        el = time.time() - t0
        if el > budget_s - 150:
            break
        try:
            res = minimize(obj_smooth, h_s, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            h_opt = _make_feasible(res.x, n, target)
            c5 = _c5_value(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent refinement
    def coord_descent_step(h, n, target, dx, max_passes=50):
        for p in range(max_passes):
            improved = False
            for j in range(n):
                old_val = h[j]
                obj_old = _c5_value(h, dx)
                # Try several candidate values for h[j]
                candidates = [0.0, 1.0, old_val * 0.5, old_val * 1.5,
                             old_val + 0.1, old_val - 0.1,
                             old_val + 0.3, old_val - 0.3]
                best_v = old_val
                best_o = obj_old
                for cv in candidates:
                    cv = np.clip(cv, 0.0, 1.0)
                    h[j] = cv
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    o = np.max(corr) * dx
                    if o < best_o - 1e-12:
                        best_o = o
                        best_v = cv
                if best_v != old_val:
                    h[j] = best_v
                    improved = True
            h = _make_feasible(h, n, target)
            c5_now = _c5_value(h, dx)
            if not improved and c5_now >= best_c5 - 1e-13:
                break
        return h

    el = time.time() - t0
    if el < budget_s - 200:
        h_cd = best_h.copy()
        h_cd = coord_descent_step(h_cd, n, target, dx, max_passes=30)
        c5_cd = _c5_value(h_cd, dx)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Try different n values
    for n_try in [60, 90, 120, 50]:
        el = time.time() - t0
        if el > budget_s - 180:
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        bounds_t = [(0.0, 1.0)] * n_try
        cons_t = [{'type': 'eq',
                   'fun': lambda x, tt=target_t: np.sum(x) - tt,
                   'jac': lambda x, nn=n_try: np.ones(nn)}]

        # Interpolate from best solution
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = _make_feasible(h_interp, n_try, target_t)

        def obj_t(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            alpha = 120.0
            return _smooth_max(corr, alpha) * dx_t

        try:
            res = minimize(obj_t, h_interp, method='SLSQP',
                           bounds=bounds_t, constraints=cons_t,
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_opt = _make_feasible(res.x, n_try, target_t)
            c5 = _c5_value(h_opt, dx_t)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            pass

        # Also try a random start at this n
        el = time.time() - t0
        if el < budget_s - 150:
            h_rand = np.random.uniform(0, 1, n_try)
            h_rand = _make_feasible(h_rand, n_try, target_t)
            try:
                res2 = minimize(obj_t, h_rand, method='SLSQP',
                                bounds=bounds_t, constraints=cons_t,
                                options={'maxiter': 200, 'ftol': 1e-12})
                h_opt2 = _make_feasible(res2.x, n_try, target_t)
                c5_2 = _c5_value(h_opt2, dx_t)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2.copy()
            except Exception:
                pass

    # Phase 4: Final Nelder-Mead polish on best
    el = time.time() - t0
    if el < budget_s - 60:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        target_cur = n_cur / 2.0
        try:
            simplex = best_h.copy()
            simplex[0] += 0.02
            simplex = _make_feasible(simplex, n_cur, target_cur)
            res = minimize(obj_max, best_h.copy(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-10,
                                    'fatol': 1e-12, 'adaptive': True})
            h_nm = _make_feasible(res.x, n_cur, target_cur)
            c5_nm = _c5_value(h_nm, dx_cur)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()
        except Exception:
            pass

    # Final coordinate descent polish
    el = time.time() - t0
    if el < budget_s - 20:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        target_cur = n_cur / 2.0
        h_final = best_h.copy()
        h_final = coord_descent_step(h_final, n_cur, target_cur, dx_cur, max_passes=20)
        c5_final = _c5_value(h_final, dx_cur)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = _make_feasible(best_h, n_final, target_final)
    best_c5 = _c5_value(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
