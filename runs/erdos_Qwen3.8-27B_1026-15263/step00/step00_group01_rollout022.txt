import numpy as np
import time

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h, dx):
    corr = compute_correlation(h)
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.any(mask):
        h[mask] += diff / np.sum(mask)
    else:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] += min(diff, 1.0 - h[idx])
        else:
            idx = np.argmax(h)
            h[idx] += max(diff, -(1.0 - h[idx]))
    h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, dx, t0, time_budget):
    n = len(h)
    total_deadline = t0 + time_budget
    corr = compute_correlation(h)
    best_c5 = float(np.max(corr)) * dx
    target = n / 2.0

    epsilons = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]

    for eps in epsilons:
        if time.time() > total_deadline:
            break
        sweep_count = 0
        while time.time() < total_deadline:
            sweep_count += 1
            c5_raw = float(np.max(corr))
            best_inc = 0.0
            best_i = -1
            best_j = -1

            for i in range(n):
                if h[i] < eps:
                    continue
                for j in range(n):
                    if i == j or h[j] > 1.0 - eps:
                        continue
                    h[i] -= eps
                    h[j] += eps
                    nc = compute_correlation(h)
                    new_max = float(np.max(nc))
                    h[i] += eps
                    h[j] -= eps
                    inc = c5_raw - new_max
                    if inc > best_inc + 1e-13:
                        best_inc = inc
                        best_i = i
                        best_j = j

            if best_i < 0:
                break
            h[best_i] -= eps
            h[best_j] += eps
            corr = compute_correlation(h)
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5:

def sim_annealing(h, dx, t0, time_budget, seed_val=99):
    n = len(h)
    rng = np.random.RandomState(seed_val)
    total_deadline = t0 + time_budget
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    curr_c5 = best_c5
    T = 0.005
    cool = 0.9998
    target = n / 2.0

    while time.time() < total_deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 0.005:
            continue
        eps = rng.uniform(0.01, max_eps)
        h[i] -= eps
        h[j] += eps
        h = fix_sum(h, target)
        new_c5 = evaluate_c5(h, dx)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] += eps
            h[j] -= eps
            h = fix_sum(h, target)

        T *= cool

    return best_h, best_c5

def make_random_h(n, rng):
    target = n / 2.0
    for _ in range(200):
        h = rng.uniform(0.0, 1.0, n)
