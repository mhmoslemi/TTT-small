Minimizing C₅ is equivalent to maximizing min_k |A ∩ (A+k)|.

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def penalized_obj(h, n):
    h_c = np.clip(h, 0.0, 1.0)
    base = compute_c5(h_c, n)
    pen = 500.0 * (np.sum(h_c) - n / 2.0) ** 2
    return base + pen

def nm_obj(h, n):
    return compute_c5(project_feasible(h, n), n)

def smooth_obj(h, n):
    h_c = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
    T = 100.0
    cmax = np.max(corr)
    vals = T * (corr - cmax)
    vals[vals > 500] = 500
    vals[vals < -500] = -500
    lse = cmax + np.log(np.sum(np.exp(vals))) / T
    pen = 500.0 * (np.sum(h_c) - n / 2.0) ** 2
    return lse + pen

def get_elapsed(start_time):
    return time.time() - start_time

    best_c5 = compute_c5(best_h, n)
    
    bounds = [(0.0, 1.0)] * n

    if budget_s - get_elapsed(start_time) > 120:
        res = minimize(penalized_obj, best_h, method='SLSQP', args=(n,),
                      bounds=bounds, maxiter=10000, ftol=1e-14)
        if res.x is not None:
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    
    # Phase 2: Smoothed SLSQP
    if budget_s - get_elapsed(start_time) > 120:

        if res.x is not None:
