import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    if beta < 1e-6:
        return float(m)
    return float(m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute to points not at boundary
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Final fix: adjust one element
        residual = target - np.sum(h)
        if abs(residual) > 1e-12:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
            # Renormalize
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                interior2 = (h > 1e-10) & (h < 1.0 - 1e-10)
