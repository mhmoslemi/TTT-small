import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def _evaluate(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return float(np.max(corr) * dx)


def _obj_smooth(h, dx, tau):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip((corr - c_max) * tau, -50, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / tau


def _con_func(h, target):
    return np.sum(h) - target


def _fix_sum(h, target):
    h = h.copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h


def _slsqp_run(h0, n, dx, target, tau, maxiter):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_func, 'args': (target,)}]
    result = minimize(
        _obj_smooth, h0, args=(dx, tau),
        method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return result.x


def _coord_1d_func(t, a_s, b_s, sum_excl, target, penalty):
    corr_vals = a_s * t + b_s
    max_corr = np.max(corr_vals)
    sum_pen = penalty * (sum_excl + t - target) ** 2
    return max_corr + sum_pen


def _coord_descent(h, n, dx, target, rng, deadline, penalty_weight):
    h = h.copy()
    for iteration in range(200000):
        if time.time() >= deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            h0c = h.copy()
            h1c = h.copy()
            h0c[i] = 0.0
            h1c[i] = 1.0
            c0 = np.correlate(h0c, 1.0 - h0c, mode='full') * dx
            c1 = np.correlate(h1c, 1.0 - h1c, mode='full') * dx
            a_s = c1 - c0
            b_s = c0
            sum_excl = np.sum(h) - h[i]
            result = minimize_scalar(
                _coord_1d_func, bounds=(0.0, 1.0),
                args=(a_s, b_s, sum_excl, target, penalty_weight),
                method='bounded', options={'xatol': 1e-10}
            )
            t_opt = result.x
            if abs(t_opt - old_val) > 1e-10:
                h[i] = t_opt
                improved = True
        if not improved:
            break
    return h


def _generate_structured(n, target, rng):
    points = []
    h_block = np.zeros(n)
    n_ones = int(target)
    h_block[:n_ones] = 1.0
    h_block[n_ones] = target - n_ones
    for _ in range(3):
        points.append(h_block[rng.permutation(n)].copy())
    h_sine = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(n) / n)
    points.append(h_sine.copy())
    h_cos = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) / n)
    points.append(h_cos.copy())
    h_tri = np.abs(2.0 * (np.arange(n) / n - 0.5)) * 2.0
    points.append(h_tri.copy())
    return points


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = _fix_sum(h0, target)
    best_score = _evaluate(best_h, dx)

    tau_schedule = [20, 40, 80, 160, 320, 640]
    for tau in tau_schedule:
        if time.time() - start_time >= budget_s * 0.2:
            break
        try:
            x_opt = _slsqp_run(best_h, n, dx, target, tau, 1000)
            x_opt = _fix_sum(x_opt, target)
            score = _evaluate(x_opt, dx)
            if score < best_score:
                best_score = score
                best_h = x_opt
        except Exception:
            pass

    deadline_cd = start_time + budget_s * 0.50
    if time.time() < deadline_cd:
        h_cd = _coord_descent(best_h, n, dx, target, rng, deadline_cd, 5000.0)
        h_cd = _fix_sum(h_cd, target)
        score = _evaluate(h_cd, dx)
        if score < best_score:
            best_score = score
            best_h = h_cd

    structured = _generate_structured(n, target, rng)
    phase3_deadline = start_time + budget_s * 0.82
    attempt = 0
    while time.time() < phase3_deadline:
        attempt += 1
        if attempt <= len(structured):
            h_start = structured[attempt - 1].copy()
        elif attempt % 4 == 1:
            h_start = best_h + rng.normal(0, 0.04, n)
        elif attempt % 4 == 2:
            h_start = best_h + rng.normal(0, 0.10, n)
        elif attempt % 4 == 3:
            h_start = best_h.copy()
            n_flip = rng.randint(2, 7)
            flip_idx = rng.choice(n, n_flip, replace=False)
            h_start[flip_idx] = 1.0 - h_start[flip_idx]
        else:
            h_start = rng.uniform(0, 1, n)

        h_start = np.clip(h_start, 0, 1)
        h_start = _fix_sum(h_start, target)

        start_score = _evaluate(h_start, dx)
        tau = [40, 80, 160, 320][attempt % 4]
        try:
            x_opt = _slsqp_run(h_start, n, dx, target, tau, 500)
            x_opt = _fix_sum(x_opt, target)
            score = _evaluate(x_opt, dx)
        except Exception:
            score = start_score
            x_opt = h_start

        if score < best_score:
            best_score = score
            best_h = x_opt

        cd_deadline = min(phase3_deadline, time.time() + 25.0)
        if time.time() < cd_deadline:
            h_cd = _coord_descent(x_opt, n, dx, target, rng, cd_deadline, 10000.0)
            h_cd = _fix_sum(h_cd, target)
            score2 = _evaluate(h_cd, dx)
            if score2 < best_score:
                best_score = score2
                best_h = h_cd

    final_deadline = start_time + budget_s - 3.0
    if time.time() < final_deadline:
        h_final = _coord_descent(best_h, n, dx, target, rng, final_deadline, 50000.0)
        h_final = _fix_sum(h_final, target)
        score = _evaluate(h_final, dx)
        if score < best_score:
            best_score = score
            best_h = h_final

    return best_h, best_score, n
