    best_t = 0.0
    best_val = compute_c5(h)

        h_trial[j] += t
        c5_t = compute_c5(h_trial)

            best_val = c5_t
            best_t = t
    
    # Refine around best_t
    if np.abs(best_t) < np.abs(t_lo) and np.abs(best_t) < np.abs(t_hi):
        span = (t_hi - t_lo) / n_grid

                h_trial[i] -= t
                h_trial[j] += t
                c5_t = compute_c5(h_trial)
                if c5_t < best_val:
                    best_val = c5_t
                    best_t = t
            span /= 3
    
    h[i] -= best_t
    h[j] += best_t
    return h, best_val

        if c5_alt < best_c5:
            best_h = h_alt.copy()
            best_c5 = c5_alt
            n = n_alt
            dx = dx_alt
            target = target_alt

    # Phase 8: Final fine-tuning with many small pair transfers
    for _final in range(200):
        if time.time() > deadline:
            break
        best_h, imp = pair_optimize_pass(best_h, n, dx, target)
        best_c5 = compute_c5(best_h)
        if not imp:
            break

    # Final verification
    best_h = project_h(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
