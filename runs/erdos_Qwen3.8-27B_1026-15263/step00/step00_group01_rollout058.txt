            h_rnd = rng.uniform(0.3, 0.7, n)
            h_rnd = fix_sum(h_rnd, n)
            starts.append(h_rnd)
        
        per_start = time_alloc / len(starts)
        for h0 in starts:
            if time.time() - start_time > budget_s * 0.85:
                break
            if per_start < 2:
                continue
            h_opt, c5_opt = optimize_single(h0, n, max_time=per_start)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Final refinement of best solution
    remain = budget_s - (time.time() - start_time)
    if remain > 10 and best_h is not None:
        n_best = len(best_h)
        h_ref, c5_ref = optimize_single(best_h, n_best, max_time=remain * 0.8)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
    
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
