```python
    dx = 2.0 / n

    corr = np.correlate(h, w, mode='full') * dx

    # Smooth max approximation
    shifted = corr * tau - cmax * tau
    logsumexp = np.log(np.sum(np.exp(shifted))) / tau + cmax
    return logsumexp

def project_to_sum(h, target):

    h = np.clip(h, 0, 1)

        if abs(diff) < 1e-10:

        if diff > 0:

        else:
            free = h > 1e-10
        if not np.any(free):

        h[free] += diff / np.sum(free)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0
_DEADLINE = 1e18
_TAU = 200.0


def _c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _c5_smooth(h, tau=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip((corr - cmax) * tau, -500.0, 0.0)
    return cmax + np.log(np.sum(np.exp(shifted)) / len(corr)) / tau


def _sum_eq(x):
    return np.sum(x) - _TARGET_SUM


def _sum_eq_jac(x):
    return np.ones(len(x))


def _obj_func(x):
    if time.time() > _DEADLINE:
        return 1e10
    return _c5_smooth(x, tau=_TAU)


def _slsqp(h0, maxiter, tau, deadline):
    global _TARGET_SUM, _DEADLINE, _TAU
    n = len(h0)
    _TARGET_SUM = n / 2.0
    _DEADLINE = deadline
    _TAU = tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    res = minimize(_obj_func, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x


def _fix_sum(h, target, tol=1e-12):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < tol:
            break
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def _random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    rng.shuffle(h)
    return h


def _thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h


def _pair_search(h, rng, max_iters, deadline):
    n = len(h)
    h = h.copy()
    cur = _c5_eval(h)
    stalling = 0
    step_lo, step_hi = 0.005, 0.08
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        step = rng.uniform(step_lo, step_hi) * (1.0 if rng.rand() < 0.5 else -1.0)
        old_i, old_j = h[i], h[j]
        h[i] = old_i + step
        h[j] = old_j - step
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c = _c5_eval(h)
            if c < cur - 1e-12:
                cur = c
                stalling = 0
            else:
                h[i], h[j] = old_i, old_j
                stalling += 1
        else:
            h[i], h[j] = old_i, old_j
            stalling += 1
        if stalling > 2000:
            step_hi = min(step_hi * 2.0, 0.3)
            step_lo = min(step_lo * 1.5, step_hi * 0.5)
            stalling = 0
            if step_hi > 0.25:
                break
    return h, cur


def _binary_round(h, target):
    n = len(h)
    h = h.copy()
    for threshold in [0.1, 0.05, 0.02]:
        h2 = h.copy()
        h2[h2 < threshold] = 0.0
        h2[h2 > 1.0 - threshold] = 1.0
        h2 = _fix_sum(h2, target)
        c5_2 = _c5_eval(h2)
        c5_orig = _c5_eval(h)
        if c5_2 < c5_orig:
            h = h2
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline_total = t0 + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = _random_binary(45, rng)

    best_h = h_init.copy()
    best_c5 = _c5_eval(best_h)
    n_best = len(best_h)

    def time_left(frac):
        return time.time() < t0 + budget_s * frac

    # Phase 1: SLSQP from initial
    if time_left(0.3):
        try:
            dl = t0 + budget_s * 0.32
            h_opt = _slsqp(h_init, maxiter=600, tau=200.0, deadline=dl)
            h_opt = _fix_sum(h_opt, n_best / 2.0)
            c5 = _c5_eval(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt.copy(), c5
        except Exception:
            pass

    # Phase 2: Pair search
    if time_left(0.5):
        try:
            dl = t0 + budget_s * 0.52
            h_ps, c5_ps = _pair_search(best_h, rng, 50000, dl)
            if c5_ps < best_c5:
                best_h, best_c5 = h_ps.copy(), c5_ps
        except Exception:
            pass

    # Phase 3: SLSQP polish after pair search
    if time_left(0.65):
        try:
            dl = t0 + budget_s * 0.68
            h_opt = _slsqp(best_h, maxiter=400, tau=300.0, deadline=dl)
            h_opt = _fix_sum(h_opt, len(h_opt) / 2.0)
            c5 = _c5_eval(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt.copy(), c5
        except Exception:
            pass

    # Phase 4: Binary rounding attempt
    if time_left(0.72):
        h_br = _binary_round(best_h, len(best_h) / 2.0)
        c5_br = _c5_eval(h_br)
        if c5_br < best_c5:
            best_h, best_c5 = h_br.copy(), c5_br
            # Re-optimize from rounded
            if time_left(0.8):
                try:
                    dl = t0 + budget_s * 0.82
                    h_opt = _slsqp(best_h, maxiter=300, tau=250.0, deadline=dl)
                    h_opt = _fix_sum(h_opt, len(h_opt) / 2.0)
                    c5 = _c5_eval(h_opt)
                    if c5 < best_c5:
                        best_h, best_c5 = h_opt.copy(), c5
                except Exception:
                    pass

    # Phase 5: Thue-Morse start with SLSQP
    n_tm = n_best
    if time_left(0.85):
        try:
            h_tm = _thue_morse(n_tm)
            if n_tm % 2 == 1:
                h_tm[n_tm // 2] = 0.5
                n_ones = np.sum(h_tm)
                if n_ones > n_tm / 2.0:
                    idx = np.where(h_tm == 1.0)[0]
                    if len(idx) > 0:
                        h_tm[idx[0]] = 0.0
            h_tm = _fix_sum(h_tm, n_tm / 2.0)
            dl = t0 + budget_s * 0.87
            h_opt = _slsqp(h_tm, maxiter=300, tau=150.0, deadline=dl)
            h_opt = _fix_sum(h_opt, n_tm / 2.0)
            c5 = _c5_eval(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt.copy(), c5
        except Exception:
            pass

    # Phase 6: Random binary restarts with SLSQP (different n)
    for n_try in [n_best, n_best - 1, n_best + 1, 48, 50]:
        if not time_left(0.95):
            break
        if n_try < 20 or n_try > 100:
            continue
        for att in range(2):
            if time.time() > t0 + budget_s * 0.97:
                break
            h_try = _random_binary(n_try, rng)
            h_try += rng.normal(0, 0.05, n_try)
            h_try = _fix_sum(h_try, n_try / 2.0)
            try:
                dl = t0 + budget_s * 0.97
                h_opt = _slsqp(h_try, maxiter=200, tau=150.0, deadline=dl)
                h_opt = _fix_sum(h_opt, n_try / 2.0)
                c5 = _c5_eval(h_opt)
                if c5 < best_c5:
                    best_h, best_c5 = h_opt.copy(), c5
            except Exception:
                pass

    # Final pair search polish if time
    if time.time() < deadline_total - 10:
        try:
            dl = deadline_total - 1
            h_ps, c5_ps = _pair_search(best_h, rng, 20000, dl)
            if c5_ps < best_c5:
                best_h, best_c5 = h_ps.copy(), c5_ps
        except Exception:
            pass

    # Final fix
    best_h = _fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = _c5_eval(best_h)

    return best_h, best_c5, len(best_h)
```
