import numpy as np
import time

def project_box_sum(x, S):
    """Project x onto {0 <= x <= 1, sum(x) = S}"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def gradient_objective(h, tau):
    """Compute gradient of smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += wk * (1.0 - h[:n - d])
            grad[:n - d] -= wk * h[d:]
        else:
            a = -d
            grad[:n - a] += wk * (1.0 - h[a:])
            grad[a:] -= wk * h[:n - a]
    return grad * dx

def gradient_descent(h0, S, budget_end, rng):
    """Run smoothed gradient descent."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    tau = 0.05
    base_lr = 0.05
    stagnant = 0
    
    while time.time() < budget_end:
        g = gradient_objective(h, tau)
        
        # Backtracking line search
        lr = base_lr
        found = False
        for _ in range(30):
            h_new = project_box_sum(h - lr * g, S)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                stagnant = 0
                found = True
                break
            lr *= 0.5
        
        if not found:
            stagnant += 1
            # Decrease tau to sharpen the max approximation
            tau *= 0.9
            if tau < 0.001:
                tau = 0.001
                # Try perturbation to escape local minimum
                h = project_box_sum(h + 0.05 * rng.randn(n), S)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            if stagnant > 100:
                # Try a bigger perturbation
                h = project_box_sum(h + 0.2 * rng.randn(n), S)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                stagnant = 0
                tau = 0.05

        tau = max(tau * 0.999, 0.001)

        delta = rng.uniform(-0.1, 0.1)

        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_box_sum(h_new, S)
        c5 = compute_c5(h_new)
        if c5 < best_c5:

            best_c5 = c5

        sorted_idx = np.argsort(h)[::-1]

        h_bin[sorted_idx[:k]] = 1.0

        h_bin[sorted_idx[:k]] = 1.0
        h_bin[sorted_idx[k]] = 1.0  # this one will be fractional after projection
        h_bin = project_box_sum(h_bin, S)
        c5 = compute_c5(h_bin)
        return h_bin, c5
