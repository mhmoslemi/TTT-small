import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0≤x[i]≤1, sum(x)=target_sum}"""
    # Use bisection on threshold
    # We want sum(clamp(h - tau, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        candidate = np.clip(h - mid, 0.0, 1.0)
        s = candidate.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def gradient_c5(h):
    """Compute gradient of C5(h) using finite differences"""
    n = len(h)
    dx = 2.0 / n
    c5_0 = compute_c5(h)
    grad = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - eps)
        grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (h_plus[i] - h_minus[i])
    return grad

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def compute_all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx

def gradient_of_lag(h, j):
    """Gradient of the j-th lag value (output[j] * dx) with respect to h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, j - n + 1)
    i_hi = min(j, n - 1)
    for i in range(i_lo, i_hi + 1):
        k = i
        j_minus_i = j - i
        if 0 <= j_minus_i < n:
            grad[k] += (1 - h[j_minus_i])
            grad[j_minus_i] += (-h[k])
    return grad * dx

def gradient_of_c5(h):
    """Gradient of C5 = max_j (lag_values[j])"""
    n = len(h)
    lag_values = compute_all_lag_values(h)
    j_max = np.argmax(lag_values)
    return gradient_of_lag(h, j_max), lag_values

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target_sum} using bisection"""
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        candidate = np.clip(h - mid, 0.0, 1.0)
        s = candidate.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def projected_gradient_step(h, grad, alpha, target_sum, momentum=None, beta=0.9):
    if momentum is None:
        m = -alpha * grad
    else:
        m = beta * momentum - alpha * grad
    h_new = project_onto_constraint(h + m, target_sum)
    return h_new, m

def optimize(h0, n, max_time, start_time, alpha0=0.05, verbose=False):
    """Optimize using projected gradient descent with momentum"""
    target_sum = n / 2.0
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = alpha0
    momentum = np.zeros(n)
    beta = 0.8
    patience = 0
    best_patience_c5 = best_c5
    
    while time.time() - start_time < max_time:
        grad, lag_values = gradient_of_c5(h)
        h_new, momentum = projected_gradient_step(h, grad, alpha, target_sum, momentum, beta)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            # Increase alpha slightly
            alpha = min(0.1, alpha * 1.05)
        else:
            patience += 1
            alpha *= 0.8
            # Try smaller step
            h_new2, _ = projected_gradient_step(h, grad, alpha * 0.1, target_sum, None)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < compute_c5(h):
                h = h_new2
            else:
                h = project_onto_constraint(h + 0.1 * (h_new - h), target_sum)
        
        if patience > 50:
            # Reset with random perturbation
            h = h + 0.01 * np.random.randn(n)
            h = project_onto_constraint(h, target_sum)
            alpha = alpha0
            momentum = np.zeros(n)
            patience = 0
        
        alpha = max(1e-5, alpha)
    
    return best_h, best_c5
```

The term -h[i]*h[j-i] is concave (it's a negative product, and the product of two non-negative variables is not convex or concave in general... actually h[i]*h[j-i] is not convex in (h[i], h[j-i]) - its Hessian is [[0,1],[1,0]] which is indefinite).

So f_j is not convex in general. The problem is non-convex.

However, we can still use gradient methods effectively.

2. SLSQP for local polishing

Let me also add a "soft max" trick: instead of the hard max, use a smoothed max (log-sum-exp) with a temperature parameter that decreases over time. This gives a smoother objective and can escape local minima of the hard max.

Actually, for the gradient of the soft max:
soft_max_T(v) = (1/T) * log(sum(exp(T*v_j)))
d(soft_max_T)/dv = v - soft_max_T(v) (in the gradient of the composition)

Wait, let me be more careful:
g(T) = (1/T) * log(sum_j exp(T * f_j))
dg/dT = ... (not useful)
dg/dh = (1/T) * sum_j exp(T*f_j) * T * (df_j/dh) / sum_j exp(T*f_j)
       = sum_j softmax_j * df_j/dh

where softmax_j = exp(T*f_j) / sum_k exp(T*f_k)

So the gradient of the soft max is a convex combination of the individual gradients, weighted by the softmax. This is nice!

def soft_max_c5(h, T=10):
    n = len(h)
    dx = 2.0 / n
    lag_values = compute_all_lag_values(h)
    # Soft max
    max_val = np.max(lag_values)
    exps = np.exp(T * (lag_values - max_val))  # shift for numerical stability
    weights = exps / np.sum(exps)
    c5_soft = max_val + np.log(np.sum(exps)) / T
    return c5_soft, weights

def gradient_soft_max_c5(h, T=10):

    max_val = np.max(lag_values)
    exps = np.exp(T * (lag_values - max_val))
    weights = exps / np.sum(exps)
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if weights[j] > 1e-8:
            grad += weights[j] * gradient_of_lag(h, j)
    return grad

This is better because it smooths out the max and gives a more informative gradient (averaging over near-active constraints).

import numpy as np
from scipy.optimize import minimize
import time

def compute_all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx

def compute_c5(h):
    return np.max(compute_all_lag_values(h))

def gradient_of_lag(h, j):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, j - n + 1)
    i_hi = min(j, n - 1)
    for i in range(i_lo, i_hi + 1):
        ji = j - i
        grad[i] += (1.0 - h[ji])
        grad[ji] += (-h[i])
    return grad * dx

def gradient_soft_max(h, T=20.0):
    n = len(h)
    lag_values = compute_all_lag_values(h)
    max_val = np.max(lag_values)
    exps = np.exp(T * (lag_values - max_val))
    total = np.sum(exps)
    weights = exps / total
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w > 1e-10:
            grad += w * gradient_of_lag(h, j)
    return grad

def gradient_hard_max(h):
    n = len(h)
    lag_values = compute_all_lag_values(h)
    j_max = np.argmax(lag_values)
    return gradient_of_lag(h, j_max)

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        candidate = np.clip(h - mid, 0.0, 1.0)
        s = candidate.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)

def soft_max_value(h, T=20.0):
    n = len(h)
    lag_values = compute_all_lag_values(h)
    max_val = np.max(lag_values)
    exps = np.exp(T * (lag_values - max_val))
    return max_val + np.log(np.sum(exps)) / T

def optimize_softmax(h0, n, duration, start_time, T0=50.0, T1=5.0):
    target_sum = n / 2.0
    h = project_onto_constraint(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = 0.02
    momentum = np.zeros(n)
    beta = 0.85
    
    elapsed_0 = time.time() - start_time
    total_elapsed = 0.0
    
    while time.time() - start_time < start_time + duration:
        frac = (time.time() - start_time - elapsed_0) / max(duration, 1e-6)
        T = T0 * (T1 / T0) ** min(frac, 1.0)
        
        grad = gradient_soft_max(h, T)
        momentum = beta * momentum - alpha * grad
        h_new = project_onto_constraint(h + momentum, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(0.05, alpha * 1.02)
        else:
            alpha *= 0.7
            h = project_onto_constraint(h + 0.3 * momentum, target_sum)
        
        if alpha < 1e-6:
            # Random restart within local neighborhood
            h = best_h + 0.02 * np.random.randn(n)
            h = project_onto_constraint(h, target_sum)
            alpha = 0.02
            momentum = np.zeros(n)
    
    return best_h, best_c5

def slsqp_polish(h0, n, max_iter=500):
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0.astype(float).copy(), target_sum)
    
    def obj(h):
        return compute_c5(h)
    
    def obj_grad(h):
        return gradient_hard_max(h)
    
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-12})
        if result.success or result.fun < obj(h0):
            h_res = project_onto_constraint(result.x, target_sum)
            return h_res, compute_c5(h_res)
    except:
        pass
    return h0, obj(h0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    # Hmm, interpolation on a step function is tricky. Let me just use linear interp
    h_new = np.interp(x_new, x_old, h)
    return h_new

def make_random_h(n, seed=None):

    h = np.random.uniform(0, 1, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx


def compute_c5(h):
    return np.max(compute_all_lag_values(h))


def gradient_of_lag(h, j):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, j - n + 1)
    i_hi = min(j, n - 1)
    for i in range(i_lo, i_hi + 1):
        ji = j - i
        grad[i] += (1.0 - h[ji])
        grad[ji] += (-h[i])
    return grad * dx


def gradient_soft_max(h, T=20.0):
    n = len(h)
    lag_values = compute_all_lag_values(h
    max_val = np.max(lag_values)
    exps = np.exp(T * (lag_values - max_val))
    total = np.sum(exps)
    weights = exps / total
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        i_lo = max(0, j - n + 1)
        i_hi = min(j, n - 1)
        for i in range(i_lo, i_hi + 1):
            ji = j - i
            grad[i] += w * (1.0 - h[ji])
            grad[ji] += w * (-h[i])
    return grad * (2.0 / n)


def gradient_hard_max(h):
    n = len(h)
    lag_values = compute_all_lag_values(h)
    j_max = int(np.argmax(lag_values))
    return gradient_of_lag(h, j_max)


def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        candidate = np.clip(h - mid, 0.0, 1.0)
        s = candidate.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)


def optimize_softmax(h0, n, duration, start_time, T0=50.0, T1=5.0, alpha_init=0.03):
    target_sum = n / 2.0
    h = project_onto_constraint(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    momentum = np.zeros(n)
    beta = 0.85
    end_time = start_time + duration
    stuck_count = 0

    while time.time() < end_time:
        frac = min(1.0, (time.time() - start_time) / max(duration, 1e-6))
        T = T0 * (T1 / T0) ** frac
        grad = gradient_soft_max(h, T)
        momentum = beta * momentum - alpha * grad
        h_new = project_onto_constraint(h + momentum, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stuck_count = 0
            alpha = min(0.08, alpha * 1.03)
        else:
            stuck_count += 1
            alpha *= 0.75
            h = project_onto_constraint(h + 0.2 * momentum, target_sum)
        if stuck_count > 80:
            h = best_h.copy()
            h += 0.05 * (np.random.random(n) - 0.5)
            h = project_onto_constraint(h, target_sum)
            alpha = alpha_init
            momentum = np.zeros(n)
            stuck_count = 0
        alpha = max(1e-5, alpha)

    return best_h, best_c5


def slsqp_polish(h0, n, max_iter=300):
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0.astype(float).copy(), target_sum)

    def obj(h):
        return compute_c5(h)

    def obj_grad(h):
        return gradient_hard_max(h)

    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
        h_res = project_onto_constraint(np.clip(result.x, 0, 1), target_sum)
        c5_res = compute_c5(h_res)
        if c5_res < obj(h0):
            return h_res, c5_res
    except Exception:
        pass
    return h0, obj(h0)


def upsample_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new


def make_random_h(n):
    h = np.random.uniform(0.0, 1.0, n)
    return project_onto_constraint(h, n / 2.0)


def make_sine_h(n, freq=1):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
    return project_onto_constraint(h, n / 2.0)
