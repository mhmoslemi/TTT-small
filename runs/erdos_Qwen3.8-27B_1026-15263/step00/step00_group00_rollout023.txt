    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_smooth_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    m = np.max(corr)
    exp_vals = np.exp(tau * (corr - m))
    s = np.sum(exp_vals)
    p = exp_vals / s
    
    f = (m + np.log(s) / tau) * dx
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        w = p[k_idx]
        if w < 1e-14:
            continue
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            m_val = -k
            if m_val < n:
                grad[:n-m_val] += w * (1.0 - h[m_val:])
                grad[m_val:] -= w * h[:n-m_val]
    
    return f, grad * dx

def enforce_sum(h, target):
    """Project h to have the correct sum while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the correction
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        if not np.any(mask):
            return h
        n_add = np.sum(mask)
        add = diff / n_add
        if add > 1.0 - h[mask].max():
            # Can't add uniformly, need iterative
            h[mask] = np.minimum(h[mask] + add, 1.0)
            # Redistribute remaining
            remaining = target - np.sum(h)
            mask2 = h < 1.0
            while remaining > 1e-10 and np.any(mask2):
                n_add2 = np.sum(mask2)
                add2 = remaining / n_add2
                h[mask2] = np.minimum(h[mask2] + add2, 1.0)
                remaining = target - np.sum(h)
                mask2 = h < 1.0
        else:

            h = np.clip(h, 0.0, 1.0)

        mask = h > 0.0
        if not np.any(mask):
            return h
        n_sub = np.sum(mask)
        sub = -diff / n_sub
        if sub > h[mask].min():
            h[mask] = np.maximum(h[mask] - sub, 0.0)
            remaining = np.sum(h) - target
            mask2 = h > 0.0
            while remaining > 1e-10 and np.any(mask2):
                n_sub2 = np.sum(mask2)
                sub2 = remaining / n_sub2
                h[mask2] = np.maximum(h[mask2] - sub2, 0.0)
                remaining = np.sum(h) - target
                mask2 = h > 0.0
        else:
            h[mask] -= sub
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_smooth_val_and_grad(x, tau, target_sum, pw):
    n = len(x)
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode="full")

    exp_vals = np.exp(tau * (corr - m))
    s = np.sum(exp_vals)
    p = exp_vals / s
    f = (m + np.log(s) / tau) * dx
    penalty = pw * (np.sum(x) - target_sum) ** 2
    f += penalty
    
    grad = np.zeros(n)

        k = k_idx - (n - 1)

        if w < 1e-14:

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Coordinate-wise fine tuning
    if time_left() > 15:
        n_cur = len(best_h)
        ts = n_cur / 2.0
        for _ in range(3):
            if time_left() < 10:
                break
            h_tmp = best_h.copy()
            for j in range(n_cur):
                for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                    h_try = h_tmp.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = enforce_sum(h_try, ts)
                    c5_try = compute_c5(h_try)
                    if c5_try < compute_c5(h_tmp) - 1e-10:
                        h_tmp = h_try.copy()
                        break
            if compute_c5(h_tmp) < best_c5:
                best_c5 = compute_c5(h_tmp)
                best_h = h_tmp.copy()

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
