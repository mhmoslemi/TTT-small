    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') 
    return np.max(corr) * dx

def repair_exact(h, n):
    """Repair h to have sum exactly n/2 and values in [0,1]"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass
        room = 1.0 - h
        mask = room > 1e-12
        if not np.any(mask):
            return h
        # Distribute proportionally to room
        proportional = diff * room[mask] / np.sum(room[mask])
        h[mask] += proportional
    else:
        # Need to remove mass
        mask = h > 1e-12
        if not np.any(mask):
            return h
        proportional = (-diff) * h[mask] / np.sum(h[mask])
        h[mask] -= proportional
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment: tweak one element
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return h

    sum_viol = np.sum(h) - n / 2.0

        res = minimize(nm_objective, best_h, method='Nelder-Mead', args=(n,),
                      options={'maxiter': 5000, 'xatol': 1e-11, 'fatol': 1e-13,
                               'initial_simplex': simplex})
        h_try = repair_exact(res.x, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    best_h = repair_exact(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
