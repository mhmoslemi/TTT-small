def project_feasible(h, n):
    """Project h onto {h: 0<=h[i]<=1, sum(h)=n/2}."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    # Water filling: adjust to hit the target
    # If current > target, need to decrease; if current < target, need to increase
    # Use bisection to find the threshold
    if current > target:
        # Need to decrease: clamp values above a threshold
        lo, hi = 0.0, 1.0
        for _ in range(100):
            mid = (lo + hi) / 2
            adjusted = np.minimum(h, mid)
            if np.sum(adjusted) > target:
                lo = mid
            else:
                hi = mid
        # Now fill the gap
        threshold = (lo + hi) / 2
        adjusted = np.minimum(h, threshold)
        remaining = target - np.sum(adjusted)
        # Distribute remaining to elements that were below threshold
        mask = h < threshold
        if np.sum(mask) > 0:

            adjusted = np.clip(adjusted, 0, 1)

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def local_search_swaps(h, n, max_rounds=200, epsilons=None):
