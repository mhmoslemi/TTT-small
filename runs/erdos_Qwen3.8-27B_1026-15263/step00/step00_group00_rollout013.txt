```python
def _grad_smooth_c5(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0, 0.0)
    w = np.exp(p * shifted)
    s = np.sum(w)
    if s == 0:
        return np.zeros(n)
    w = w / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i] * dx
        for t in range(n):
            g = 0.0
            # Case 1: t as first factor h[t+i-n+1]
            # Valid when t <= i <= t+n-1, i.e., i-t in [0, n-1]
            if 0 <= i - t < n:
                # Second factor index: t - i + n - 1
                idx2 = t - i + n - 1
                if 0 <= idx2 < n:
                    g += 1.0 - x[idx2]
            # Case 2: t in second factor (1-h[t])
            # Valid when n-1-t <= i <= 2n-2-t
            if n - 1 - t <= i <= 2 * n - 2 - t:
                # First factor index: t + i - n + 1
                idx1 = t + i - n + 1
                if 0 <= idx1 < n:
                    g += -x[idx1]
            grad[t] += wi * g
    return grad
```

```python
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0, 0.0)
    lse = np.log(np.mean(np.exp(p * shifted))) / p
    return (max_c + lse) * dx

def _smooth_c5_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0, 0.0)
    w = np.exp(p * shifted)
    s = np.sum(w)
    if s < 1e-300:
        return np.zeros(n)
    w = w / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i] * dx
        if wi < 1e-15:
            continue
        for t in range(n):
            g = 0.0
            if 0 <= i - t < n:
                idx2 = t - i + n - 1
                if 0 <= idx2 < n:
                    g += 1.0 - x[idx2]
            if n - 1 - t <= i <= 2 * n - 2 - t:
                idx1 = t + i - n + 1
                if 0 <= idx1 < n:
                    g += -x[idx1]
            grad[t] += wi * g
    return grad

def _repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                if diff <= 0:
                    break
                add = min(room[idx], diff)
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= 0:
                    break
                sub = min(h[idx], -diff)
                h[idx] -= sub
                diff += sub
    return h

def _sum_con_fun(x):
    return np.sum(x) - _TARGET_SKETCH

def _sum_con_jac(x):
    return np.ones(len(x))

def _slsqp_step(h, n, target, p, maxiter):
    global _TARGET_SKETCH
    _TARGET_SKETCH = target
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        result = opt.minimize(_smooth_c5_obj, h, args=(p,),
                              jac=_smooth_c5_grad, bounds=bounds,
                              constraints=constraints, method='SLSQP',
                              options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair_sum(result.x, target)
    except Exception:
        return h

def _subgradient_refine(h, n, target, n_iter, eps0):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    eps = eps0
    for it in range(n_iter):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = int(np.argmax(corr))
        m = n - 1
        g = np.zeros(n)
        for t in range(n):
            gi = 0.0
            if 0 <= k_star - t < n:
                idx2 = t - k_star + n - 1
                if 0 <= idx2 < n:
                    gi += 1.0 - best_h[idx2]
            if n - 1 - t <= k_star <= 2 * n - 2 - t:
                idx1 = t + k_star - n + 1
                if 0 <= idx1 < n:
                    gi += -best_h[idx1]
            g[t] = gi
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        test = best_h - eps * g
        test = _repair_sum(test, target)
        c5_test = _compute_c5(test)
        if c5_test < best_c5:
            best_c5 = c5_test
            if it % 20 == 0:
                eps *= 0.8
        else:
            eps *= 0.5
            if eps < 1e-8:
                break
    return best_h

def _coord_descent(h, n, target, time_limit):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    deadline = time.time() + time_limit
    step = 0.04
    while time.time() < deadline and step > 1e-7:
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            k = (j + 1) % n
            if j == k:
                continue
            for delta in [step, -step]:
                new_j = best_h[j] + delta
                new_k = best_h[k] - delta
                if new_j < 0 or new_j > 1 or new_k < 0 or new_k > 1:
                    continue
                test = best_h.copy()
                test[j] = new_j
                test[k] = new_k
                c5_test = _compute_c5(test)
                if c5_test < best_c5 - 1e-15:
                    best_h = test
                    best_c5 = c5_test
                    improved = True
                    break
        if not improved:
            step *= 0.7
    return best_h

def _random_feasible(n, target, rng):
    h = rng.random(n)
    h /= h.sum()
    h = h * (2.0 * n / (2.0 * n))
    h = rng.random(n)
    h = np.clip(h, 0, 1)
    h = _repair_sum(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _TARGET_SKETCH
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def _remaining():
        return budget_s - (time.time() - t_start)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 45
        h0 = np.zeros(n)
        h0[0::2] = 1.0
        h0[-1] = 0.5
        h0 = _repair_sum(h0, n / 2.0)

    n = len(h0)
    target = n / 2.0
    _TARGET_SKETCH = target
    h0 = _repair_sum(h0, target)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    p_schedule = [200, 100, 60, 40, 25, 15, 10, 7, 5, 3, 2]
    for p in p_schedule:
        if _remaining() < 120:
            break
        maxiter = min(500, max(100, int(_remaining() * 0.3)))
        candidate = _slsqp_step(best_h, n, target, p, maxiter)
        c5_val = _compute_c5(candidate)
        if c5_val < best_c5:
            best_h = candidate.copy()
            best_c5 = c5_val

    if _remaining() > 100:
        n_sub = min(5000, int(_remaining() * 3))
        best_h = _subgradient_refine(best_h, n, target, n_sub, 0.02)
        best_c5 = _compute_c5(best_h)

    if _remaining() > 80:
        coord_time = min(_remaining() - 40, 200)
        best_h = _coord_descent(best_h, n, target, coord_time)
        best_c5 = _compute_c5(best_h)

    n_restarts = 25
    for r in range(n_restarts):
        if _remaining() < 80:
            break
        scale = 0.05 + 0.15 * rng.rand()
        pert = rng.randn(n) * scale
        candidate = best_h + pert
        candidate = _repair_sum(candidate, target)
        for p in [40, 15, 8]:
            if _remaining() < 40:
                break
            candidate = _slsqp_step(candidate, n, target, p, 150)
        c5_val = _compute_c5(candidate)
        if c5_val < best_c5:
            best_h = candidate.copy()
            best_c5 = c5_val
            if _remaining() > 60:
                best_h = _subgradient_refine(best_h, n, target, 2000, 0.01)
                best_c5 = _compute_c5(best_h)

    if _remaining() > 40:
        coord_time = min(_remaining() - 10, 150)
        best_h = _coord_descent(best_h, n, target, coord_time)
        best_c5 = _compute_c5(best_h)

    best_h = _repair_sum(best_h, target)
    best_c5 = _compute_c5(best_h)

    return (best_h, best_c5, n)
```
