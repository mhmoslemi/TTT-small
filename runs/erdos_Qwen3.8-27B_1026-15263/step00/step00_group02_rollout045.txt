The objective is max_k c_k * dx, which is the maximum of quadratic forms. This is a non-convex optimization problem.

    weights /= np.sum(weights)

    w = 1.0 - h

    w = 1.0 - h
    corr = np.correlate(h, w, mode='full') * dx

    penalty = lam * (np.sum(h) - n / 2.0) ** 2

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt > 0:
