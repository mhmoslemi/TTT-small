def _binary_refine(h, target_sum, n):

    h = h.copy()

    dists = np.minimum(h, 1-h)
    order = np.argsort(dists)

                h[idx] = 1
                current_sum += diff
        else:

import numpy as np
from scipy.optimize import minimize
import time


def _smooth_obj(x, temp, dx):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if temp < 1e-10:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def _sum_fn(x, target):
    return np.sum(x) - target


def _sum_jac(x, target):
    return np.ones(len(x))

    corr = np.correlate(h, 1.0 - h, mode="full")
