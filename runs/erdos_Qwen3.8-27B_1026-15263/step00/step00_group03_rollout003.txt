import numpy as np
import time


def repair(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n, dtype=np.float64)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def coord_descent(h, n, max_passes=500, deadline=None):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            orig = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = orig
            B = c0.astype(np.float64)
            S = c1.astype(np.float64) - c0.astype(np.float64)
            t_grid = np.linspace(0.0, 1.0, 257)
            V = B[None, :] + t_grid[:, None] * S[None, :]
            mv = V.max(axis=1)
            bi = int(mv.argmin())
            if 0 < bi < 256:
                tl, th = float(t_grid[bi - 1]), float(t_grid[bi + 1])
            elif bi == 0:
                tl, th = 0.0, float(t_grid[1])
            else:
                tl, th = float(t_grid[-2]), 1.0
            tf = np.linspace(tl, th, 1001)
            Vf = B[None, :] + tf[:, None] * S[None, :]
            best_t = float(tf[int(Vf.max(axis=1).argmin())])
            h[i] = best_t
    return h


def multi_repair(h, n):
    h = repair(h, n)
    h = repair(h, n)
    return h


def try_n_points(h_init, n_target, deadline):
    if time.time() > deadline:
        return None
    if n_target == len(h_init):
        h_start = h_init.copy()
    else:
        idx = np.linspace(0, len(h_init) - 1, n_target)
        lo = np.floor(idx).astype(int)
        hi = np.minimum(lo + 1, len(h_init) - 1)
        frac = (idx - lo).astype(float)
        h_start = h_init[lo] * (1.0 - frac) + h_init[hi] * frac
        h_start = repair(h_start, n_target)
    h_opt = coord_descent(h_start, n_target, max_passes=300, deadline=deadline)
    h_opt = multi_repair(h_opt, n_target)
    return h_opt, compute_c5(h_opt)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    n = 70
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = len(h_init)
    else:
        h_init = np.full(n, 0.5)

    h_init = multi_repair(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_n = n

    deadline_cd = t_start + budget_s * 0.25
    h_cd = coord_descent(best_h, n, max_passes=2000, deadline=deadline_cd)
    h_cd = multi_repair(h_cd, n)
    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        best_n = n

    t_restart_start = time.time()
    restart_deadline = t_start + budget_s * 0.65
    sigmas = [0.15, 0.1, 0.05, 0.03, 0.02, 0.01]
    sigma_idx = 0

    while time.time() < restart_deadline:
        sigma = sigmas[min(sigma_idx, len(sigmas) - 1)]
        h_pert = best_h + np.random.randn(best_n) * sigma
        h_pert = multi_repair(h_pert, best_n)
        h_opt = coord_descent(h_pert, best_n, max_passes=500, deadline=restart_deadline)
        h_opt = multi_repair(h_opt, best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n
            sigma_idx = 0
        else:
            if sigma_idx < len(sigmas) - 1 and (time.time() - t_restart_start) % 120 > 90:
                sigma_idx += 1

    fine_deadline = t_start + budget_s * 0.85
    while time.time() < fine_deadline:
        h_pert = best_h + np.random.randn(best_n) * 0.005
        h_pert = multi_repair(h_pert, best_n)
        h_opt = coord_descent(h_pert, best_n, max_passes=1000, deadline=fine_deadline)
        h_opt = multi_repair(h_opt, best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n

    if time.time() < t_start + budget_s - 60:
        for n_alt in [140, 100, 90]:
            if time.time() > t_start + budget_s - 30:
                break
            result = try_n_points(best_h, n_alt, t_start + budget_s - 30)
            if result is not None:
                h_alt, c5_alt = result
                if c5_alt < best_c5:
                    best_c5 = c5_alt
                    best_h = h_alt
                    best_n = n_alt

    if time.time() < t_start + budget_s - 10:
        h_final_polish = best_h + np.random.randn(best_n) * 0.002
        h_final_polish = multi_repair(h_final_polish, best_n)
        h_final_polish = coord_descent(h_final_polish, best_n, max_passes=2000,
                                       deadline=t_start + budget_s - 5)
        h_final_polish = multi_repair(h_final_polish, best_n)
        c5_final = compute_c5(h_final_polish)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_polish.copy()
            best_n = best_n

    best_h = multi_repair(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
