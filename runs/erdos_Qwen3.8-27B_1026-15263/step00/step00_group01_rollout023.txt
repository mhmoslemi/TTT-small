from scipy.optimize import minimize
import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h to {x: 0<=x<=1, sum(x)=target_sum} using bisection on lambda"""
    h = np.array(h, dtype=float)
    def projected(lam):
        return np.clip(h - lam, 0.0, 1.0)
    # Find lambda such that sum(projected(lam)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(projected(mid))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return projected((lo + hi) / 2.0)

def c5_subgradient(h):
    """Return (c5, subgradient) using the active max term"""
    n = len(h)
    dx = 2.0 / n
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode="full")
    k_max = np.argmax(corr)
    c5 = float(corr[k_max] * dx)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # f = sum_{m=0}^{n-1-lag} h[m+lag]*(1-h[m])
        # d/dh[i] = [i>=lag]*(1-h[i-lag]) - [i<=n-1-lag]*h[i+lag]
        for i in range(lag, n):
            grad[i] += (1.0 - h[i - lag])
        for i in range(0, n - lag):
            grad[i] -= h[i + lag]
    else:
        L = -lag
        # f = sum_{m=0}^{n-1-L} h[m]*(1-h[m+L])
        # d/dh[i] = [i<=n-1-L]*(1-h[i+L]) - [i>=L]*h[i-L]
        for i in range(0, n - L):
            grad[i] += (1.0 - h[i + L])
        for i in range(L, n):
            grad[i] -= h[i - L]
    grad *= dx
    return c5, grad

def projected_gradient_descent(h_start, target_sum, n_steps=10000, lr_init=0.01):
    """PGD with decreasing step size"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for step in range(n_steps):
        c5, grad = c5_subgradient(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = lr_init / (1.0 + step * 0.01)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new
    return best_h, best_c5

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def c5_and_active_lags(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    c5 = float(np.max(corr) * dx)
    # Get top k active lags
    idx = np.argsort(corr)[::-1][:top_k]
    lags = idx - (n - 1)
    return c5, corr, lags

def subgradient_at_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        # f = sum_{m=0}^{n-1-lag} h[m+lag]*(1-h[m])
        # df/dh[i] = [i>=lag]*(1-h[i-lag]) - [i<n-lag]*h[i+lag]
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        L = -lag
        # f = sum_{m=0}^{n-1-L} h[m]*(1-h[m+L])
        # df/dh[i] = [i<n-L]*(1-h[i+L]) - [i>=L]*h[i-L]
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_and_top_lags(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    c5 = float(np.max(corr) * dx)
    idx = np.argsort(corr)[::-1][:top_k]
    lags = idx - (n - 1)
    return c5, corr, lags

def subgradient_at_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    lag = int(lag)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        L = -lag
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    return g * dx

def subgradient_max(h, top_k=3):
    c5, corr, lags = c5_and_top_lags(h, top_k=top_k)
    active_mask = np.abs(corr - np.max(corr)) < 1e-10
    n = len(h)
    all_lags = np.arange(2 * n - 1) - (n - 1)
    active = all_lags[active_mask]
    if len(active) == 0:
        active = lags[:1]
    g = np.zeros(n)
    for lag in active:
        g += subgradient_at_lag(h, lag)
    g /= len(active)
    return c5, g

def pgd_momentum(h_init, target, n_iter, lr, momentum, top_k=3, check_interval=500):
    h = h_init.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    for i in range(n_iter):
        c5, g = subgradient_max(h, top_k=top_k)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000 and i > 1000:
            break
        v = momentum * v + (1.0 - momentum) * g
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def soft_max_objective(h, target, T=50.0):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    corr_scaled = corr * dx * T
    mx = np.max(corr_scaled)
    if mx > 500:
        return 100.0
    return np.log(np.sum(np.exp(corr_scaled - mx))) / T + mx / T
