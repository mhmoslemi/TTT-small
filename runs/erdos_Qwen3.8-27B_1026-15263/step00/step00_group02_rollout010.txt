import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, dx, alpha=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx)

def project_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h[i] <= 1."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0
        if np.sum(mask) > 0:
            avail = np.minimum(1.0 - h, np.inf)
            avail[~mask] = 0
            total_avail = np.sum(avail)
            if total_avail > 0:
                scale = min(1.0, diff / total_avail)
                h += avail * scale
    elif diff < 0:
        # Need to decrease sum
        mask = h > 0.0
        if np.sum(mask) > 0:
            avail = h.copy()
            avail[~mask] = 0
            total_avail = np.sum(avail)
            if total_avail > 0:
                scale = min(1.0, -diff / total_avail)
                h -= avail * scale
    return h

            if cp < best_c5:
                best_c5 = cp
                best_h = hp
                dxp = 2.0 / best_n

    if time.time() < deadline - 3:
        res = minimize(slsqp_obj, best_h,
                       args=(best_n, 2.0 / best_n, 100.0, 1000.0),
                       method='SLSQP', bounds=[(0.0, 1.0)] * best_n,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            ht = project_feasible(res.x, best_n)
            ct = compute_c5(ht, 2.0 / best_n)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht

    dx_final = 2.0 / best_n
    c5_final = compute_c5(best_h, dx_final)
    return best_h, c5_final, best_n
