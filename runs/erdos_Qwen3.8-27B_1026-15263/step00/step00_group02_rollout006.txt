def get_subgradient(h, dx):
    """Compute a subgradient of C5(h) = max_k corr_k * dx"""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = np.max(corr)
    
    active = np.where(np.abs(corr - max_val) < 1e-8)[0]
    
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        for j in range(n):
            val = 1.0
            if 0 <= j + k < n:
                val -= h[j + k]
            if 0 <= j - k < n:
                val -= h[j - k]
            grad[j] += val
    grad /= len(active)
    return max_val * dx, grad * dx

def project_h(h, target_sum):
