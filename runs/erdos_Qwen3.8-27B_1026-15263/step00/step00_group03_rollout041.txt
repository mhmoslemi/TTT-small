        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            max_add = (1.0 - h[mask]).max()
            if per > max_add:
                per = max_add
            if per <= 0:
                break
            h[mask] += per
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = (-diff) / cnt
            min_sub = h[mask].min()
            if per > min_sub:
                per = min_sub
            if per <= 0:
                break
            h[mask] -= per
    return h
