import numpy as np
from scipy.optimize import minimize
import time

def objective_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 0) & (x < 1)
        if len(free) > 0:
            x[free] += diff / len(free)
    return x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h onto [0,1]^n ∩ {sum = target_sum} using bisection."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix small residual
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        free = (x > 1e-10) & (x < 1 - 1e-10)
        if len(free) > 0:
            x[free] += diff / len(free)
            x = np.clip(x, 0, 1)
    return x

def obj_and_grad(h, n, dx):
    """Objective and (numerical) gradient."""
    corr = np.correlate(h, 1 - h, mode='full')
    val = np.max(corr) * dx
    # Numerical gradient
    grad = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        h[i] += eps
        c_plus = np.max(np.correlate(h, 1-h, mode='full')) * dx
        h[i] -= 2*eps
        c_minus = np.max(np.correlate(h, 1-h, mode='full')) * dx
        h[i] += eps
        grad[i] = (c_plus - c_minus) / (2*eps)
    return val, grad

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_init = 70
        h0 = np.ones(n_init) * 0.5
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h0 = project_h(h0, target_sum)
    best_h = h0.copy()
    best_val = compute_c5(h0)
    
    def objective(x):
        return np.max(np.correlate(x, 1-x, mode='full')) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    # SLSQP from initial
    result = minimize(objective, h0, method='SLSQP', bounds=bounds,
                     constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
    if result.fun < best_val:
        best_h = project_h(result.x, target_sum)
        best_val = compute_c5(best_h)
    
    # Multiple restarts

    while time.time() - start_time < budget_s * 0.8:
        # Try different perturbation sizes

        h_start = best_h + rng.randn(n) * scale
        h_start = project_h(h_start, target_sum)
        
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-12})
        new_val = compute_c5(project_h(result.x, target_sum))
        if new_val < best_val:
            best_h = project_h(result.x, target_sum)
            best_val = new_val
    
    # Coordinate descent polish
    elapsed = time.time() - start_time
    if time.time() - start_time < budget_s * 0.95:
        for sweep in range(20):
            if time.time() - start_time > budget_s * 0.95:

            improved = False
            for i in range(n):
                old = best_h[i]
                # 1D optimization

                    best_h[i] = x
                    return compute_c5(best_h)
                
                res = minimize_scalar(f_1d, bounds=(0, 1), method='bounded',

    # Final projection and evaluation
    best_h = project_h(best_h, target_sum)
    final_val = compute_c5(best_h)
    
    return (best_h, final_val, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _n, _dx
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    _n = n
    _dx = 2.0 / n
    target_sum = n / 2.0

    h0 = _project(h0, target_sum)
    best_h = h0.copy()
    best_val = _compute_c5(h0)

    # Phase 1: SLSQP from initial
    if time.time() - t0 < budget_s * 0.3:
        x_res, f_res = _slsqp_optimize(h0, maxiter=5000)
        cand = _project(x_res, target_sum)
        val = _compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val

    # Phase 2: Coordinate descent
    if time.time() - t0 < budget_s * 0.5:
        best_h = _coord_descent(best_h, target_sum, max_sweeps=200,
                                time_limit=budget_s * 0.5, t0=t0)
        val = _compute_c5(best_h)
        if val < best_val:
            best_val = val

    # Phase 3: SLSQP refinement
    if time.time() - t0 < budget_s * 0.65:
        x_res, f_res = _slsqp_optimize(best_h, maxiter=8000)
        cand = _project(x_res, target_sum)
        val = _compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val

    # Phase 4: Coordinate descent again
    if time.time() - t0 < budget_s * 0.75:
        best_h = _coord_descent(best_h, target_sum, max_sweeps=200,
                                time_limit=budget_s * 0.75, t0=t0)
        val = _compute_c5(best_h)
        if val < best_val:
            best_val = val

    # Phase 5: Multi-start with perturbations
    if time.time() - t0 < budget_s * 0.92:
        for attempt in range(100):
            if time.time() - t0 > budget_s * 0.92:
                break
            scale = 0.005 + 0.03 * rng.random()
            h_start = best_h + rng.randn(n) * scale
            h_start = _project(h_start, target_sum)
            x_res, f_res = _slsqp_optimize(h_start, maxiter=2000)
            cand = _project(x_res, target_sum)
            val = _compute_c5(cand)
            if val < best_val:
                best_h, best_val = cand, val

    # Phase 6: Final coordinate descent polish
    if time.time() - t0 < budget_s * 0.98:
        best_h = _coord_descent(best_h, target_sum, max_sweeps=300,
                                time_limit=budget_s * 0.98, t0=t0)
        val = _compute_c5(best_h)
        if val < best_val:
            best_val = val

    # Final projection
    best_h = _project(best_h, target_sum)
    final_val = _compute_c5(best_h)

    return (best_h, final_val, n)
