                        delta = lo + frac * (hi - lo)
                        best_h[j] += delta
                        best_h[i] -= delta
                        c = compute_c5(best_h)
                        if c < best_c5:
                            best_c5 = c
                            improved = True
                        else:
                            best_h[j] -= delta
                            best_h[i] += delta
        
        if not improved:
            if no_improve >= max_no_improve:
                break
            # Escape: random perturbation
            h_try = best_h + rng.uniform(-0.15, 0.15, n)
            h_try = fix_sum(h_try, target_sum)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try
                no_improve = 0
    
    return best_h, best_c5
