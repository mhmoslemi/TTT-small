def project_to_feasible(h, n):
    """Project h to satisfy sum = n/2 and 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = n/2 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly
        h += diff / n
        h = np.clip(h, 0, 1)
        # Re-adjust
        diff = n/2 - np.sum(h)
        if len(h) > 0:
            # Add to the most "free" elements
            if diff > 0:
                order = np.argsort(1 - h)
            else:
                order = np.argsort(h)
            h[order] += diff / len(order)
            h = np.clip(h, 0, 1)
    return h

def slp_step(h, n, target_sum=None):
    """One step of Sequential Linear Programming"""
    if target_sum is None:
        target_sum = n / 2.0
    
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f_values = corr  # f_k values (before multiplying by dx)
    
    n_corr = len(corr)  # 2n-1
    # corr[j] = f_k where k = j - (n-1)
    
    # Build LP
    # Variables: h[0], ..., h[n-1], t
    # n+1 variables
    nvar = n + 1
    
    # Objective: minimize t
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: f_k(h) ≈ f_k(h_t) + grad_k · (h - h_t) ≤ t
    # grad_k[j] = ∂f_k/∂h[j]
    
    # For corr index j, lag k = j - (n-1)
    A_rows = []
    b_rows = []
    
    for j in range(n_corr):
        k = j - (n - 1)  # lag
        f_k = corr[j]
        
        # Compute gradient of f_k with respect to h
        grad = np.zeros(n)
        
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            for i in range(n - k):
                # ∂f_k/∂h[i+k] += (1-h[i])
                grad[i+k] += (1 - h[i])
                # ∂f_k/∂h[i] += -h[i+k]
                grad[i] += -h[i+k]
        else:
            m = -k
            # f_k = sum_{i=m}^{n-1} h[i-m] * (1-h[i])
            for i in range(m, n):
                # ∂f_k/∂h[i-m] += (1-h[i])
                grad[i-m] += (1 - h[i])
                # ∂f_k/∂h[i] += -h[i-m]
                grad[i] += -h[i-m]
        
        # Linearization: f_k(h_t) + grad·(h - h_t) ≤ t
        # f_k(h_t) + grad·h - grad·h_t ≤ t
        # grad·h - t ≤ f_k(h_t) - grad·h_t
        rhs = f_k - np.dot(grad, h)
        
        row = np.zeros(nvar)
        row[:n] = grad
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum h = target_sum
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: 0 ≤ h[j] ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        new_h = res.x[:n]
        return new_h, res.x[-1]
    else:
        return h, None

def smooth_max(corr, alpha=10.0):
    """Smooth approximation of max"""
    shifted = corr - np.max(corr)  # numerical stability
    return np.max(corr) + (1.0/alpha) * np.log(np.sum(np.exp(alpha * shifted)))

def smooth_c5(h, n, alpha=10.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return smooth_max(corr, alpha)

def gradient_smooth_c5(h, n, alpha=10.0):
    """Gradient of smooth C5"""
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1 - h, mode='full')
    corr = corr_raw * dx
    
    # Numerical gradient
    grad = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        grad[i] = (smooth_c5(h_plus, n, alpha) - smooth_c5(h_minus, n, alpha)) / (2*eps)
    return grad

    n_corr = len(corr)
    nvar = n + 1
    
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    
    # Precompute gradients for all lags

    A_ub_list = []
    b_ub_list = []

            grad[m:] += -h[:n-m]

        row[-1] = -1.0

        b_ub_list.append(rhs)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
