def snap_to_binary(h, n, threshold=0.3):
    """Snap values close to 0 or 1, then adjust to meet sum constraint."""
    h_new = h.copy()
    h_new[h_new < threshold] = 0.0
    h_new[h_new > 1.0 - threshold] = 1.0
    # Now adjust the remaining values (those between threshold and 1-threshold)
    mid_mask = (h_new >= 0.0) & (h_new <= 1.0) & (h_new > 0.0) & (h_new < 1.0)
    if np.sum(mid_mask) > 0:
        target_sum = n * 0.5
        fixed_sum = np.sum(h_new[~mid_mask])
        remaining = target_sum - fixed_sum
        cnt = int(np.sum(mid_mask))
        if remaining >= 0 and remaining <= cnt:
            h_new[mid_mask] = remaining / cnt
        else:
            # Need to adjust binary values too
            if remaining > cnt:
                # Convert some 0s to 1s
                extra = remaining - cnt
                h_new[mid_mask] = 1.0
                zeros = np.where(h_new == 0.0)[0]
                if len(zeros) >= int(np.ceil(extra)):
                    h_new[zeros[:int(np.ceil(extra))]] = 1.0
                    # The last one might need to be fractional
                    frac = extra - int(np.floor(extra))
                    if frac > 0 and int(np.floor(extra)) < len(zeros):
                        h_new[zeros[int(np.floor(extra))]] = frac
            elif remaining < 0:
                # Convert some 1s to 0s
                deficit = -remaining
                h_new[mid_mask] = 0.0
                ones = np.where(h_new == 1.0)[0]
                if len(ones) >= int(np.ceil(deficit)):

                    frac = deficit - int(np.floor(deficit))
                    if frac > 0 and int(np.floor(deficit)) < len(ones):
                        h_new[ones[int(np.floor(deficit))]] = 1.0 - frac
    h_new = project_feasible(h_new, n)
    return h_new
