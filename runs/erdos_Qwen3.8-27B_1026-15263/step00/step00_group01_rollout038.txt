def local_search_pair_swap(h, n, rng, max_time=5.0):
    """Local search by swapping pairs of 1s and 0s."""
    start = time.time()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    h = h.copy()
    patience = 0
    max_patience = 500
    
    while time.time() - start < max_time:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a random few swaps
        improved = False
        for _ in range(50):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            
            if new_c5 < best_c5 - 1e-10:
                best_c5 = new_c5
                best_h = h.copy()
                current_c5 = new_c5
                improved = True
                patience = 0
                # Update ones and zeros
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                break
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            patience += 1
            if patience > max_patience:
                break
    
    return best_h, best_c5

def simulated_annealing(h, n, rng, max_time=10.0, T0=0.5, T1=0.01):
    """Simulated annealing for binary sequence optimization."""
    start = time.time()
    h = h.copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    
    elapsed = 0
    n_steps = 0
    while time.time() - start < max_time:
        T = T0 * (T1/T0) ** (elapsed / max_time)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        elapsed = time.time() - start
        n_steps += 1
    
    return best_h, best_c5

    """Optimize the relaxed problem h in [0,1] with sum(h)=n/2."""
    from scipy.optimize import minimize as sp_minimize
    
    def obj_and_grad(x):
        # x has length n-1, last element determined by constraint
        h = np.zeros(n)
        h[:n-1] = x
        h[n-1] = n/2.0 - np.sum(x)
        h = np.clip(h, 0, 1)
        # Re-enforce constraint after clipping
        if abs(np.sum(h) - n/2) > 1e-6:
            h = h / np.sum(h) * (n/2)
            h = np.clip(h, 0, 1)

                         bounds=bounds,

        h[n-1] = n/2.0 - np.sum(result.x)

        diff = n/2 - np.sum(h)

                h[mask] += diff / np.sum(mask)

For s=24: expected B(24) = 24/4 = 6
For s=47: expected B(47) = 1/4 = 0.25

The maximum is at small s. So the bottleneck is small shifts.

To minimize B(s) for small s, we need adjacent 1s to be followed by 1s (not 0s). This suggests clustering 1s together, but that would make B(s) large for large s (where clusters overlap with the other clusters).

The optimal structure is a balance between these competing constraints. This is exactly what the Erdős problem is about.

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def local_search(h, n, rng, max_time=10.0, tabu_size=100):
    """Tabu local search by swapping pairs."""
    start = time.time()
    h = h.copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    
    tabu = {}  # (i,j) -> time when it became tabu
    tabu_tenure = 50
    
    iteration = 0
    while time.time() - start < max_time:
        iteration += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swaps, keep the best non-tabu one
        best_delta = 0
        best_swap = None
        n_tries = min(200, len(ones) * len(zeros) // 10)
        
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            
            key = (min(i,j), max(i,j))
            if key in tabu and iteration - tabu[key] < tabu_tenure:
                continue
            
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            delta = new_c5 - current_c5
            h[i], h[j] = h[j], h[i]
            
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, n)
            tabu[(min(i,j), max(i,j))] = iteration
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # No improving move found, break
            # Try a random move (diversification)
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, n)
            tabu[(min(i,j), max(i,j))] = iteration

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def local_search(h, n, rng, max_time=10.0, n_tries_per_iter=100):
    start = time.time()
    h = h.copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    iter_count = 0
    stall = 0
    
    while time.time() - start < max_time:
        iter_count += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        n_tries = min(n_tries_per_iter, len(ones) * len(zeros))
        best_delta = 0
        best_i = -1
        best_j = -1
        
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            delta = new_c5 - current_c5
            h[i], h[j] = h[j], h[i]
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j
        
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            current_c5 += best_delta
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 200:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, n)
    
    return best_h, best_c5

def periodic_search(n, rng, max_time=5.0):
    best_h = None
    best_c5 = float('inf')
    start = time.time()
    
    divisors = [d for d in range(4, n + 1) if n % d == 0 and d % 2 == 0]
    
    for d in divisors:
        if time.time() - start > max_time:
            break
        half = d // 2
        n_reps = n // d
        
        if half > 12:
            n_trials = 5000
        elif half > 8:
            n_trials = 20000
        elif half > 5:
            n_trials = 100000
        else:
            n_trials = 200000
        
        for _ in range(n_trials):
            if time.time() - start > max_time:
                break
            block = np.zeros(d)
            perm = rng.permutation(d)
            block[perm[:half]] = 1.0
            h = np.tile(block, n_reps)
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    if best_h is not None:
        best_h, best_c5 = local_search(best_h, n, rng, max_time=min(5, max_time))
    
    return best_h, best_c5

def simulated_annealing(h, n, rng, max_time=15.0, T0=0.3, T1=0.001):
    start = time.time()
    h = h.copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    total_steps = 0

        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        total_steps += 1
    
    return best_h, best_c5
