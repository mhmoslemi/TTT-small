import numpy as np
from scipy.optimize import minimize
import time

_g_n = 45
_g_dx = 2.0 / _g_n
_g_target = _g_n / 2.0


def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * _g_dx)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(h):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    n = _g_n
    target = _g_target
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _sum_constraint(x):
    return x.sum() - _g_target


def _sobj(x, tau):
    h = np.clip(x, 0.0, 1.0)
    cr = _corr(h)
    cm = cr.max()
    sm = cm + tau * np.log(np.sum(np.exp((cr - cm) / tau)))
    pen = 500.0 * (h.sum() - _g_target) ** 2
    return sm * _g_dx + pen

    n = _g_n
    target = _g_target
    idx = np.argsort(h0)[::-1]
    n_ones = int(target)
    frac = target - n_ones
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    hb[idx[n_ones]] = frac
    c = _c5(hb)
    T = 0.015

    for it in range(max_iters):

            zeros = np.where(h_tr < 0.001)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_tr[i] = 0.0
            h_tr[j] = 1.0
        elif r < 0.85:

            j = np.random.randint(n)

            c = c_new
        T *= 0.9997
