import numpy as np
import time


def _eval_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def _make_binary(h, n, target_ones, rng):
    h = (h > 0.5).astype(np.float64)
    s = int(np.sum(h))
    if s > target_ones:
        idx = np.where(h == 1)[0]
        idx = idx[:s - target_ones]
        h[idx] = 0.0
    elif s < target_ones:
        idx = np.where(h == 0)[0]
        idx = idx[:target_ones - s]
        h[idx] = 1.0
    return h


def _nl_search(h, n, target_ones, rng, t_end):
    best_c5 = _eval_c5(h, n)
    best_h = h.copy()
    while time.time() < t_end:
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if time.time() > t_end:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = _eval_c5(h, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    ones.remove(i)
                    zeros.remove(j)
                    zeros.append(i)
                    ones.append(j)
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            h = best_h.copy()
            best_c5 = _eval_c5(h, n)
            n_perturb = rng.randint(3, 12) + 1
            for _ in range(n_perturb):
                o = np.where(h > 0.5)[0]
                z = np.where(h < 0.5)[0]
                if len(o) == 0 or len(z) == 0:
                    break
                ii = int(rng.choice(o))
                jj = int(rng.choice(z))
                h[ii] = 0.0
                h[jj] = 1.0
        else:
            h = best_h.copy()
    return best_h, best_c5


def _nonbinary_refine(h0, n, t_end, rng):
    target_sum = n / 2.0
    best_c5 = _eval_c5(h0, n)
    best_h = h0.copy()
    h = h0.copy()
    step = 0.05
    while time.time() < t_end:
        step *= 0.95
        if step < 0.001:
            step = 0.001
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(n):
                if i == j:
                    continue
                for frac in (0.3, 1.0):
                    d = step * frac
                    if h[i] - d < -1e-9 or h[j] + d > 1.0 + 1e-9:
                        continue
                    h[i] -= d
                    h[j] += d
                    c5 = _eval_c5(h, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
        if not improved:
            h = best_h.copy()
            for _ in range(5):
                ii = int(rng.randint(n))
                jj = int(rng.randint(n))
                if ii != jj:
                    dd = rng.uniform(-step, step)
                    h[ii] = np.clip(h[ii] + dd, 0, 1)
                    h[jj] = np.clip(h[jj] - dd, 0, 1)
            s = np.sum(h)
            if s > 1e-9:
                h *= target_sum / s
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            diff = target_sum - s
            if abs(diff) > 1e-12:
                if diff > 0:
                    cap = 1.0 - h
                    tc = np.sum(cap)
                    if tc > 0:
                        h += diff * cap / tc
                else:
                    tc = np.sum(h)
                    if tc > 0:
                        h += diff * h / tc
                h = np.clip(h, 0, 1)
    return best_h, best_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _eval_c5(best_h, n)

    return (best_h, final_c5, n)
