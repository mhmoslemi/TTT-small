def local_search_fast(h, n, target_sum, time_limit):
    """Fast local search for binary-like sequences."""
    best_h = h.copy()
    best_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = np.max(best_corr)
    start = time.time()
    
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    
    no_improve = 0
    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a random move
        oi = rng.choice(ones)
        zi = rng.choice(zeros)
        
        # Make the move
        best_h[oi] -= 1.0  # was 1, now 0 (approximately)
        best_h[zi] += 1.0  # was 0, now 1
        # Now: h[oi] was ~1, now ~0; h[zi] was ~0, now ~1
        
        new_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        new_max = np.max(new_corr)
        
        if new_max < best_max:
            best_max = new_max
            best_corr = new_corr
            # Update sets
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            no_improve = 0
        else:
            # Undo
            best_h[oi] += 1.0
            best_h[zi] -= 1.0
            no_improve += 1
            if no_improve > 10000:
                break
    
    return best_h, best_max
