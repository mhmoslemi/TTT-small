```python
        grad[:n-kp] += (1.0 - h[kp:])
        # p in [kp, n-1]: grad[p] -= h[p-kp]
        grad[kp:] -= h[:n-kp]
    return grad
```

This is O(n) per lag, very fast.

- For the worst lag k*, find the pair (i, j) that would most reduce corr_{k*} if we moved mass from i to j
- Specifically: moving δ from h[i] to h[j] changes corr_{k*} by:
  δ * (grad_{k*}[j] - grad_{k*}[i])
- We want this to be negative, so we want grad_{k*}[j] < grad_{k*}[i]
- But we also don't want to increase other lags too much

Actually, let me think about this more carefully. Moving δ from position i to position j:
- h[i] decreases by δ, h[j] increases by δ
- Change in corr_m for any lag m: δ * (dc_m/dh[j] - dc_m/dh[i]) = δ * (grad_m[j] - grad_m[i])
- For the objective max_m corr_m, we want to reduce the active lags

For a single active lag k*:
- We want grad_{k*}[j] - grad_{k*}[i] < 0, i.e., grad_{k*}[j] < grad_{k*}[i]
- So move from high gradient to low gradient positions

For the "worst" move that reduces corr_{k*} the most:
- Maximize grad_{k*}[i] - grad_{k*}[j] subject to 0≤h[i]-δ and h[j]+δ≤1

This is an assignment problem, but we can approximate by just taking the max and min.

Let me implement this as a "greedy projected gradient" step:
1. Compute gradient of the worst lag
2. Find argmax and argmin of the gradient
3. Move mass from argmax to argmin (clipped to maintain feasibility)
4. Check if other lags got worse; if so, revert or reduce δ

This should be much faster and more effective than random moves.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lsm_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    return m_val + (max_sh + np.log(np.sum(exp_v))) / alpha

def lsm_grad_full(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    nm1 = n - 1
    grad = np.zeros(n)
    for p in range(n):
        k_lo = max(p - n + 1, -nm1)
        k_hi = min(p, nm1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            idx_h = p - ks
            grad[p] += np.dot(w[ks + nm1], 1.0 - h[idx_h])
        k_lo2 = max(-p, -nm1)
        k_hi2 = min(nm1 - p, nm1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            idx_h2 = p + ks2
            grad[p] -= np.dot(w[ks2 + nm1], h[idx_h2])
    return grad

def lbfgsb_opt(h0, n, ts, alpha, pen, maxiter=300):
    def f(x):
        obj = lsm_obj(x, alpha)
        s = np.sum(x) - ts
        return obj + pen * s * s
    def g(x):
        grad = lsm_grad_full(x, alpha)
        s = np.sum(x) - ts
        return grad + 2.0 * pen * s
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.astype(float).copy(), method='L-BFGS-B', jac=g, bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40, 'maxfun': 6000})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def corr_grad_single(h, m_idx):
    n = len(h)
    nm1 = n - 1
    k = m_idx - nm1
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif 0 < k < n:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    elif -n < k < 0:
        kp = -k
        grad[:n - kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n - kp]
    return grad

def targeted_opt(h0, n, ts, alpha, pen, maxiter=200, n_worst=5):
    h = h0.copy()
    for _ in range(3):
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idxs = np.argsort(corr)[-n_worst:]
        worst_count = len(worst_idxs)
        ls = 1.0 / worst_count
        def f_t(x):
            obj = lsm_obj(x, alpha)
            for m_i in worst_idxs:
                s = np.correlate(x, 1.0 - x, mode="full")[m_i]
                obj += ls * s * (alpha * 0.1)
            s_sum = np.sum(x) - ts
            return obj + pen * s_sum * s_sum
        def g_t(x):
            grad = lsm_grad_full(x, alpha)
            for m_i in worst_idxs:
                grad += ls * (alpha * 0.1) * corr_grad_single(x, m_i)
            s_sum = np.sum(x) - ts
            grad += 2.0 * pen * s_sum
            return grad
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(f_t, h.astype(float).copy(), method='L-BFGS-B', jac=g_t, bounds=bnds,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40, 'maxfun': 4000})
            h = project_to_sum(res.x, ts)
        except Exception:
            break
    return h

def greedy_lag_steps(h, n, ts, n_steps=20000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    c5_cur = compute_c5(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_m = int(np.argmax(corr))
        g = corr_grad_single(h, worst_m)
        top_k = 8
        src_candidates = np.argsort(g)[-top_k:]
        dst_candidates = np.argsort(g)[:top_k:]
        i = int(rng.choice(src_candidates))
        j = int(rng.choice(dst_candidates))
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.15)
        if h[i] < delta:
            delta = h[i] * 0.9
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.9
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            if rng.random() < np.exp(-(c5_new - c5_cur) / 0.005):
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
    return h, c5_cur

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)
```
