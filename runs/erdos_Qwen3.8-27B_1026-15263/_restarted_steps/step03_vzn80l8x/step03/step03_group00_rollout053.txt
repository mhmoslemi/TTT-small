def binary_projection(h, target_sum):
    """Project h to binary, fixing sum."""
    n = len(h)
    h = h.copy()
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    current_sum = np.sum(h_bin)
    target = target_sum
    if current_sum > target:
        # Need to flip some 1s to 0s
        n_flip = int(current_sum - target)
        # Flip the ones closest to 0.5 (i.e., most "uncertain")
        ones = np.where(h_bin == 1)[0]
        uncertainty = 1.0 - h[ones]  # distance from 0.5 (well, from 1)
        flip_indices = ones[np.argsort(uncertainty)][:n_flip]
        h_bin[flip_indices] = 0
    elif current_sum < target:
        n_flip = int(target - current_sum)
        zeros = np.where(h_bin == 0)[0]
        uncertainty = h[zeros]  # distance from 0.5
        flip_indices = zeros[np.argsort(uncertainty, kind='stable')[::-1][:n_flip]]
        h_bin[flip_indices] = 1
    return h_bin

def generalized_coord_descent(h, target_sum, max_iters=10000, step_frac=0.15, top_k_lags=3):
    """Coordinate descent that considers multiple worst lags."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = np.max(corr)
        
        # Use top-k lags
        top_indices = np.argsort(corr)[::-1][:top_k_lags]
        
        # Combined gradient
        grad = np.zeros(n)
        for idx in top_indices:
            lag = idx - (n - 1)
            grad += corr_gradient(h, lag)
        grad /= len(top_indices)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, grad, np.inf)
        g_dec = np.where(can_dec, grad, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break
        if grad[i_inc] >= grad[i_dec] - 1e-12:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    
    return h, c5

def fista_smooth_max(h0, n, target_sum, alpha, max_iters=5000, t_init=1.0):
    """FISTA (accelerated proximal gradient) for smooth max."""
    h = h0.copy()
    h_prev = h0.copy()
    t = t_init
    L = alpha * 2 * n  # Lipschitz constant estimate (rough)
    step = 1.0 / L
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        # Compute gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val + np.log(np.sum(exp_vals)) / alpha
        
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(2 * n - 1)[None, :]
        first_idx = j_idx - m_idx + n - 1
        second_idx = j_idx + m_idx - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        cf = valid_f * (1.0 - h[fc]) * w[None, :]
        cs = valid_s * h[sc] * w[None, :]
        grad = np.sum(cf - cs, axis=1)
        
        # Project gradient onto null space of sum constraint
        grad -= np.mean(grad)
        
        # FISTA update
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        
        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_mom = h_new + ((t - 1) / t_new) * (h_new - h)
        h_mom = np.clip(h_mom, 0, 1)
        h_mom = project_to_sum(h_mom, target_sum)
        
        h = h_mom
        t = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
