        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:
                break
    return h

def active_set_descent(h, n, ts, n_iters=2000, lr_init=0.1, top_k=20):
    """Greedy descent on the most violated lags."""
    h = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_vals = corr * (2.0 / n)
        # Find top k lags
        idx = np.argsort(c5_vals)[-top_k:]
        # Average gradient of top lags
        grad = np.zeros(n)
        w_sum = 0
        for j in idx:
            s = j - (n - 1)  # shift
            # Compute gradient of corr[j] w.r.t. h
            # d(corr[j])/dh[m] = (1-h[m-s])*[0<=m-s<n] - h[m+s]*[0<=m+s<n]
            # (using the convention where corr[j] = sum_i h[i]*(1-h[i-s]))
            m = np.arange(n)
            a = m - s
            b = m + s
            va = (a >= 0) & (a < n)
            vb = (b >= 0) & (b < n)
            g_j = (1.0 - h[np.clip(a, 0, n-1)]) * va - h[np.clip(b, 0, n-1)] * vb
            w = c5_vals[j]  # weight by how close to max
            grad += w * g_j
            w_sum += w
        grad /= (w_sum + 1e-30)
        # Projected step
        h_new = project_to_sum(h - lr_init * grad, ts)
        c5_new = compute_c5(h_new)
        c5_old = c5_vals.max()
        if c5_new < c5_old - 1e-15:
            h = h_new
        else:
            lr_init *= 0.8
            if lr_init < 1e-14:
                break
    return h

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_gamma(h, alpha):
    """Returns (objective_value, gradient) for smooth max of dx*corr."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    # Gradient via matrix method
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def _obj_sl(h, alpha):
    return smooth_max_obj_gamma(h, alpha)[0]

def _grad_sl(h, alpha):
    return smooth_max_obj_gamma(h, alpha)[1]

def _sum_fun(h, ts):
    return np.sum(h) - ts

def _sum_jac(h, ts):
    return np.ones(len(h))

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    _obj_sl.outer_alpha = alpha  # Can't do this without closure...
    # Use args parameter
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (ts,)}]
    try:
        r = minimize(_obj_sl, h0.copy(), method='SLSQP', jac=_grad_sl, 
                     args=(alpha,), bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gamma(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def _slof(x, alpha):
    return smooth_max_gamma(x, alpha)[0]

def _slg(x, alpha):
    return smooth_max_gamma(x, alpha)[1]

def _sfun(x, ts):
    return np.sum(x) - ts

def _sjac(x, ts):
    return np.ones(len(x))

def _sopf(x, alpha, pen):
    o, _ = smooth_max_gamma(x, alpha)
    s = np.sum(x) - len(x) / 2.0
    return o + 0.5 * pen * s * s

def _sog(x, alpha, pen):
    _, g = smooth_max_gamma(x, alpha)
    s = np.sum(x) - len(x) / 2.0
    return g + pen * s * np.ones(len(x))

def do_slsqp(h0, n, ts, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfun, 'jac': _sjac, 'args': (ts,)}]
    try:
        r = minimize(_slof, h0.copy(), method='SLSQP', jac=_slg,
                     args=(alpha,), bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def do_lbfgs(h0, n, ts, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    pen = 1e5
    try:
        r = minimize(_sopf, h0.copy(), method='L-BFGS-B', jac=_sog,
                     args=(alpha, pen), bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def do_pgrad(h0, n, ts, alpha, n_iter, lr_init):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_gamma(h, alpha)
    stall = 0
    for _ in range(n_iter):
        _, grad = smooth_max_gamma(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_gamma(h_new, alpha)
        if obj_new < obj_prev - 1e-15:
            h = h_new
            obj_prev = obj_new
            stall = 0
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.5
            stall += 1
            if stall > 150 or lr < 1e-14:
                break
    return h

def active_grad_step(h, n, ts, lr, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    w_sum = 0.0
    for j in idx:
        s = j - (n - 1)
        m = np.arange(n)
        a = m - s
        b = m + s
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        g_j = (1.0 - h[np.clip(a, 0, n - 1)]) * va - h[np.clip(b, 0, n - 1)] * vb
        wt = corr[j]
        grad += wt * g_j
        w_sum += wt
    if w_sum > 0:
        grad /= w_sum
    h_new = project_to_sum(h - lr * grad, ts)
    return h_new, corr.max()

def do_active_descent(h, n, ts, n_iters, lr_init, top_k):
    h = h.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    for _ in range(n_iters):
        h_new, c5_cur = active_grad_step(h, n, ts, lr, top_k)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-13:
                break
    return h

def stochastic_cd(h, n, ts, rng, n_steps, d_lo, d_hi, sa=0.25):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = sa
    for step in range(n_steps):
        ones_idx = np.where(h > 0.03)[0]
        zeros_idx = np.where(h < 0.97)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(d_lo, d_hi)
        if h[i] < delta:
            delta = h[i] * 0.4
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.4
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-6, sa * np.exp(-step / 4000.0))
    return h, c5_cur

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def make_perturbed(h, rng, n):
    x = np.linspace(0, 2, n, endpoint=False)
    results = []
    nf = rng.randint(2, 5)
    freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
    h_p = h.copy()
    for f in freqs:
        h_p += rng.uniform(0.03, 0.15) * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 6.28))
    results.append(project_to_sum(h_p, n / 2.0))
    h_b = h.copy()
    mask = rng.rand(n) < 0.1
    h_b[mask] += rng.uniform(-0.3, 0.3, size=np.sum(mask))
    results.append(project_to_sum(h_b, n / 2.0))
    h_c = 0.5 + 0.3 * np.sin(2.0 * np.pi * rng.randint(1, 10) * x / 2.0 + rng.uniform(0, 6.28))
    h_c = 0.5 * h + 0.5 * h_c
    results.append(project_to_sum(h_c, n / 2.0))
    return results
