def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum} using bisection."""
    h = h.astype(float)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mu = (lo + hi) * 0.5
        h_p = np.clip(h - mu, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) * 0.5
    return np.clip(h - mu, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient computation
    j_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2 * n - 1)[None, :]  # (1, 2n-1)
    first_idx = j_idx - m_idx + n - 1  # (n, 2n-1)
    second_idx = j_idx + m_idx - n + 1  # (n, 2n-1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad, corr

    alpha = alpha_init

            lr = lr * 0.5
            v = momentum * v - lr * grad

        else:
            lr = lr * 2.0
            break

        actual_lr = lr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = h.astype(float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mu = (lo + hi) * 0.5
        h_p = np.clip(h - mu, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) * 0.5
    return np.clip(h - mu, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad, corr

def momentum_gd(h0, alpha, lr_init, momentum, max_iter, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        obj, grad, corr = smooth_max_grad(h, alpha)
        v = momentum * v - lr * grad
        h_trial = project_box_sum(h + v, target_sum)
        obj_trial, _, _ = smooth_max_grad(h_trial, alpha)
        if obj_trial < obj - 1e-15:
            h = h_trial
            lr = min(lr * 1.02, lr_init * 3.0)
        else:
            lr = lr * 0.5
            if lr < 1e-8:
                break
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n]
        for _ in range(max(0, n_cand - 4)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            t_grid = np.linspace(delta_lo, delta_hi, 10)
            vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
            best_idx = int(np.argmin(vals))
            if best_idx == 0:
                delta_opt = delta_lo
            elif best_idx == len(t_grid) - 1:
                delta_opt = delta_hi
            else:
                lo_r = t_grid[best_idx - 1]
                hi_r = t_grid[best_idx + 1]
                for _ in range(25):
                    mid = (lo_r + hi_r) * 0.5
                    if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
                        lo_r = mid
                    else:
                        hi_r = mid
                delta_opt = (lo_r + hi_r) * 0.5
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h = np.clip(h, 0.0, 1.0)
            improved = True
    return h, corr, improved

    if best_h is None:

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
