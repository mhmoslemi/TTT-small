def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, top_k=40):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(corr)[-k:]
    shifted = alpha * (corr[top_idx] - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = m_val + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for ii in range(k):
        m = top_idx[ii]
        w = exp_vals[ii] / sum_exp
        if w < 1e-15:
            continue
        l = m - (n - 1)
        if l == 0:
            grad += w * (1.0 - 2.0 * h)
        elif l > 0:
            if n - l > 0:
                grad[:n-l] += w * (1.0 - h[l:])
            if l < n:
                grad[l:] -= w * h[:n-l]
        else:
            lp = -l
            if n - lp > 0:
                grad[lp:] += w * (1.0 - h[:n-lp])
            if lp < n:
                grad[:n-lp] -= w * h[lp:]
    return obj, grad

def thue_morse_init(n, n_ones):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    ones_mask = tm == 1
    n_ones_actual = np.sum(ones_mask)
    if n_ones_actual < n_ones:
        zeros_idx = np.where(ones_mask == 0)[0]
        extra = n_ones - n_ones_actual
        if extra > 0 and len(zeros_idx) >= extra:
            chosen = zeros_idx[:extra]
            ones_mask[chosen] = 1
    elif n_ones_actual > n_ones:
        ones_idx = np.where(ones_mask == 1)[0]
        extra = n_ones_actual - n_ones
        if extra > 0 and len(ones_idx) >= extra:
            chosen = ones_idx[:extra]
            ones_mask[chosen] = 0
    return ones_mask.astype(float)

        result_t2 = phase_end['p2']
        h_sg, c5_sg = subgrad_nesterov(best_h, target_sum, t_start, result_t2, lr_init=0.005, alpha_init=80, momentum=0.88)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        h_sg2, c5_sg2 = subgrad_nesterov(best_h, target_sum, t_start, result_t2, lr_init=0.003, alpha_init=200, momentum=0.92)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    # Phase 3: Coordinate descent (deterministic + simulated annealing)
    if time_left() > 80:
        h_cd, c5_cd = coord_desc_active(best_h, target_sum, 80000, t_start, phase_end['p3'], step_frac=0.25)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() > 60:
            h_sa, c5_sa = coord_desc_sa(best_h, target_sum, 50000, t_start, phase_end['p3'], t0=0.015, t_min=0.0005)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: SLSQP high alpha + coordinate descent
    if time_left() > 120:
        for alpha in [3000, 15000, 60000]:
            if time.time() - t_start > phase_end['p4'] - 30:
                break
            h_opt = slsqp_optimize(best_h, alpha, maxiter=1200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cd, c5_cd = coord_desc_active(best_h, target_sum, 30000, t_start, phase_end['p4'] - 30, step_frac=0.15)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 5: Multi-start with structured perturbations
    n_starts = 6
    for trial in range(n_starts):
        if time.time() - t_start > phase_end['p5']:
            break
        if trial == 0:
            tm = thue_morse_init(n, int(n / 2))
            h_try = tm.copy()
            h_try = project_to_sum(h_try, target_sum)
        elif trial == 1:
            h_try = best_h + rng.uniform(-0.04, 0.04, n)
            h_try = project_to_sum(h_try, target_sum)
        elif trial == 2:
            h_try = best_h.copy()
            block_size = n // 12
            for b in range(0, n, block_size * 2):
                if (b // block_size) % 2 == 0:
                    h_try[b:b+block_size] += 0.08
                else:
                    h_try[b:b+block_size] -= 0.08
            h_try = project_to_sum(h_try, target_sum)
        elif trial == 3:
            freq = 3.0 + 2.0 * trial
            pert = 0.06 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
            h_try = best_h + pert
            h_try = project_to_sum(h_try, target_sum)
        else:
            noise_level = 0.03 + 0.05 * (1.0 - trial / float(n_starts))
            h_try = best_h + rng.uniform(-noise_level, noise_level, n)
            h_try = project_to_sum(h_try, target_sum)

        c5_try = compute_c5(h_try)
        h_cd, c5_cd = coord_desc_active(h_try, target_sum, 40000, t_start, phase_end['p5'], step_frac=0.2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time.time() - t_start > phase_end['p5'] - 20:
            break
        h_opt = slsqp_optimize(best_h, 5000, maxiter=600)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cd2, c5_cd2 = coord_desc_active(best_h, target_sum, 20000, t_start, phase_end['p5'], step_frac=0.12)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 6: Final polishing
    if time_left() > 40:
        h_cd, c5_cd = coord_desc_active(best_h, target_sum, 50000, t_start, phase_end['p6'], step_frac=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        h_sa, c5_sa = coord_desc_sa(best_h, target_sum, 30000, t_start, phase_end['p6'], t0=0.008, t_min=0.0002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Final SLSQP with highest alpha
    if time_left() > 25:
        h_final = slsqp_optimize(best_h, 100000, maxiter=3000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Absolute final coordinate descent
    if time_left() > 10:
        h_fin, c5_fin = coord_desc_active(best_h, target_sum, 30000, t_start, t_start + max(0, time_left() - 2), step_frac=0.08)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
