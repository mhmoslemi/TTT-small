3. Project to feasible set: [0,1]^n ∩ {sum = target}

The projection step is crucial. The standard projection onto [0,1]^n is just clipping. But we also need the sum constraint. The projection onto [0,1]^n ∩ {sum = s} can be done as follows:
- Start with h - lr * grad
- Find λ such that sum(clip(x + λ, 0, 1)) = s where x = h - lr * grad
- This is a 1D root-finding problem (bisection works)

Let me implement this properly:

```python
def project_box_sum(x, target_sum):
    """Project x onto {z : 0 <= z_i <= 1, sum(z) = target_sum}"""
    n = len(x)
    # Bisection on lambda
    lo, hi = -np.max(x) - 1, 1 - np.min(x)
    for _ in range(100):
        mid = (lo + hi) / 2
        z = np.clip(x + mid, 0, 1)
        s = np.sum(z)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(x + (lo + hi) / 2, 0, 1)
    # Final adjustment
    diff = target_sum - np.sum(z)
    if abs(diff) > 1e-15:
        mask = (z > 1e-10) & (z < 1 - 1e-10)
        if np.any(mask):
            z[mask] += diff / np.sum(mask)
            z = np.clip(z, 0, 1)
    return z
```

For the line search, I'll use a simple backtracking method.

Let me also think about the SA moves. Instead of binary swaps, I can:
- Pick two random positions
- Swap a small amount of "mass" between them
- Or: pick one position, perturb it, and adjust another to maintain the sum

Actually, the most effective moves for this problem are probably:
1. **Binary swap**: Swap h[i] and h[j] (only useful for binary solutions)
2. **Value transfer**: Move a small amount ε from h[i] to h[j] (maintains sum)
3. **Pair flip**: Change h[i] from 0→1 and h[j] from 1→0 (binary, maintains sum)
4. **Gradient step**: Move in the negative gradient direction

For the continuous optimization, the gradient step is most efficient. For escape from local optima, the value-based perturbations help.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(x, target_sum):
    n = len(x)
    lo, hi = -np.max(x) - 1.0, 1.0 - np.min(x)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(z)
    if abs(diff) > 1e-15:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        if np.any(mask):
            z[mask] += diff / np.sum(mask)
            z = np.clip(z, 0.0, 1.0)
    return z

def soft_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    lags = np.arange(n_corr) - (n - 1)
    j_idx = np.arange(n)[:, None]
    
    idx1 = j_idx + lags[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    term1 = np.sum(valid1 * (1.0 - h[idx1c]) * w[None, :], axis=1)
    
    idx2 = j_idx - lags[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n - 1)
    term2 = np.sum(valid2 * h[idx2c] * w[None, :], axis=1)
    
    grad = term1 - term2
    return obj, grad

def projected_gradient(h0, n, target_sum, alphas, t_end, rng):
    h = h0.copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    
    for alpha in alphas:
        t_alpha_start = time.time()
        lr = 1.0 / (alpha * 0.1)
        n_iter = 0
        while time.time() < t_end and n_iter < 5000:
            obj, grad = soft_max_obj_grad(h, alpha)
            
            # Backtracking line search
            step_dir = -grad
            # Normalize step
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step_dir = step_dir / grad_norm
            
            lr_try = lr
            obj_cur = obj
            for backtrack in range(30):
                h_new = project_box_sum(h + lr_try * step_dir, target_sum)
                obj_new, _ = soft_max_obj_grad(h_new, alpha)
                if obj_new < obj_cur:
                    break
                lr_try *= 0.5
            
            h = h_new
            n_iter += 1
            
            c5 = compute_c5(h)
            if c5 < best_obj - 1e-12:
                best_obj = c5
                best_h = h.copy()
            
            lr = max(lr_try, 1e-8)
        
        # After each alpha, do a quick SA to escape
        if time.time() < t_end - 30:
            h = quick_value_sa(h, n, target_sum, 10000, 0.05, 0.005, rng)
            c5 = compute_c5(h)
            if c5 < best_obj:
                best_obj = c5
                best_h = h.copy()
    
    return best_h, best_obj

def quick_value_sa(h0, n, target_sum, n_iter, t_init, t_min, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_curr = best_c5
    T = t_init
    for i in range(n_iter):
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        eps = rng.uniform(0.01, 0.15)
        if h[i1] >= eps and h[i2] <= 1.0 - eps:
            h_old1, h_old2 = h[i1], h[i2]
            h[i1] -= eps
            h[i2] += eps
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or rng.uniform() < np.exp(-delta / T):
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                h[i1], h[i2] = h_old1, h_old2
        T = max(T * 0.999, t_min)
    return h

def binary_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    no_improve = 0
    min_T = 0.004
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / T) + np.log(rng.uniform()) < 0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.02)
            no_improve = 0
        T = max(T * 0.999998, min_T)
    return best_h, best_c5

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def interpolate_up(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def make_binary(h, n, target_sum_int, rng):
