    # f is convex piecewise linear
    lo, hi = 0.0, 1.0

        mid = (lo + hi) / 2
        # Subgradient at mid: D[k] for k achieving the max
        vals = C + D * mid
        k_max = np.argmax(vals)
        # But there might be multiple k achieving the max
        mask = np.abs(vals - vals[k_max]) < 1e-12
        # Subgradient is in [min D[k] for k in mask, max D[k] for k in mask]
        sg_lo = D[mask].min()
        sg_hi = D[mask].max()
        if sg_lo >= 0:
            hi = mid  # f is increasing, minimum is to the left
        elif sg_hi <= 0:
            lo = mid  # f is decreasing, minimum is to the right
        else:
            # The minimum is at mid (subgradient contains 0)
            return mid
    return (lo + hi) / 2

The streaming approach is better because:

- It naturally interleaves with other methods
- It's very fast per step (O(n) per coordinate update)

- Computing C (O(n))

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    O = np.zeros(n)
    for k in range(n):
        O[k] = dx * np.dot(h[:n-k], one_minus_h[k:])
    return O

def solve_1d_min(C, D):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C + D * mid
        vmax = np.max(vals)
        mask = vals >= vmax - 1e-14
        sg_min = np.min(D[mask])
        sg_max = np.max(D[mask])
        if sg_min > 1e-15:
            hi = mid
        elif sg_max < -1e-15:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def coord_step(h, O, j, n, dx):
    """Perform one exact coordinate descent step at index j."""
    old_hj = h[j]
    # Compute D[k] = d(O[k])/d(h[j])
    D = np.zeros(n)
    # First term: dx * (1-h[j+k]) for k=0,...,n-1-j
    D[0:n-j] += dx * (1.0 - h[j:n])
    # Second term: -dx * h[j-k] for k=0,...,j
    D[0:j+1] -= dx * h[j::-1]
    
    C = O - D * old_hj
    x_new = solve_1d_min(C, D)
    delta = x_new - old_hj
    if abs(delta) > 1e-15:
        h[j] = x_new
        O += D * delta
    return h, O, abs(delta)

def streaming_coord_descent(h, O, n, dx, n_steps, rng, time_fn=None, min_time=10):
    """Streaming coordinate descent with random coordinate selection."""
    best_obj = np.max(O)
    for step in range(n_steps):
        if time_fn is not None and time_fn() < min_time:
            break
        j = rng.randint(n)
        h, O, delta = coord_step(h, O, j, n, dx)
        if step % 100 == 0:
            cur_obj = np.max(O)
            if cur_obj < best_obj:
                best_obj = cur_obj
    return h, O

def cyclic_coord_descent(h, O, n, dx, n_passes, time_fn=None, min_time=10):
    """Cyclic coordinate descent."""
    for p in range(n_passes):
        if time_fn is not None and time_fn() < min_time:
            break
        improved = False
        for j in range(n):
            h, O, delta = coord_step(h, O, j, n, dx)
            if delta > 1e-13:
                improved = True
        if not improved:
            break
    return h, O

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    # Gradient

    # Let's use the same formula as before
    nj = np.arange(n)[:, None]

    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return obj, grad

        return smooth_max_obj_grad(x, alpha)[0]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
