import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / max(1, len(h))
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones"""
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        idx = np.where(h_bin == 1.0)[0]
        # Flip the ones with smallest h values (closest to 0.5)
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_bin[idx[order[i]]] = 0.0
    elif ones < target:
        idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_bin[idx[order[i]]] = 1.0
    return h_bin

def binary_sa(h_bin, n, rng, n_iter, t0, t1):
    """Simulated annealing on binary pattern"""
    h = h_bin.copy()
    ts = n // 2
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    for it in range(n_iter):
        # Temperature schedule
        frac = it / n_iter
        temp = t0 * (t1 / t0) ** frac
        
        i = rng.randint(n)
        old = h[i]
        h[i] = 1.0 - old
        new_c5 = compute_c5(h)
        
        if new_c5 < c5 or rng.random() < np.exp(-(new_c5 - c5) / max(temp, 1e-15)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = old
    
    return best_h, best_c5

def refined_optimize(h0, n, alpha, maxiter=500):
    """Optimize using smooth max with SLSQP"""
    ts = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = alpha * (corr - m)
        ms = np.max(sh)
        ev = np.exp(sh - ms)
        return m + (ms + np.log(np.sum(ev))) / alpha
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = alpha * (corr - m)
        ms = np.max(sh)
        ev = np.exp(sh - ms)
        w = ev / np.sum(ev)
        
        # Gradient: d/dx_i of sum_j w_j * corr_j
        # corr_j = sum_t x[t] * (1-x[t-(j-n+1)])
        # d corr_j / dx_i = (1-x[i-(j-n+1)]) - x[i+(j-n+1)]  (with bounds)
        # Actually: corr_j = sum_t x[t]*(1-x[t-s]) where s = j-(n-1)
        # d/dx_i: 
        #   if i is the first factor: (1-x[i-s]) for valid i-s
        #   if i is the second factor: -x[i+s] for valid i+s
        # So d corr_j/dx_i = (1-x[i-s])*I(0<=i-s<n) - x[i+s]*I(0<=i+s<n)
        
        # Vectorized: for each i, sum over j of w_j * [(1-x[i-s_j])*I - x[i+s_j]*I]
        # s_j = j-(n-1), ranging from -(n-1) to n-1
        
        nj = np.arange(n)[:, None]  # i index
        sj = np.arange(2*n-1)[None, :] - (n-1)  # shift for each j
        fi = (nj + sj)  # i - s = i + (-(j-(n-1))) = i - s_j... 
        # Wait: i - s where s = j-(n-1)
        # i - (j-(n-1)) = i + (n-1) - j
        fi = nj + (n-1) - np.arange(2*n-1)[None, :]
        si = nj + np.arange(2*n-1)[None, :] - (n-1)  # i + s = i + j-(n-1)
        
        vf = (fi >= 0) & (fi < n)
        vs = (si >= 0) & (si < n)
        fci = np.clip(fi, 0, n-1)
        sci = np.clip(si, 0, n-1)
        
        grad = np.sum(vf * (1.0 - x[fci]) * w[None, :] - vs * x[sci] * w[None, :], axis=1)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h0.copy(), method='SLSQP', jac=obj_grad, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        h_res = project_to_sum(r.x, ts)
    except:

    return h_res
