    # Wait no: c[k'] = sum_n a[n+k'] * b[n] = sum_n a[n+k'] * (1-a[n])
    # d(c[k'])/d(a[i]) = (1-a[i-k']) [from a[n+k'] term] - a[i+k'] [from (1-a[n]) term, where n=j and i=j+k']
    #                    = (1-a[i-k']) - a[i+k'] (with validity conditions)
    
    # So: grad[i] = sum_{k' valid for i} w[k'+n-1] * ((1-a[i-k']) - a[i+k'])
    
    # This is O(n²) to compute directly. For n=240, that's 57600 operations, which is fine.
    
    # Let me compute it using the matrix approach:
    # Define a matrix M where M[i, m] = 1 if position i participates in correlation entry m
    # Then grad = M^T * (w element-wise * ((1-h_shifted) - h_shifted))
    
    # Actually, let me just do it with loops for clarity (n=240 is small enough):
    for i in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            lag = m - n + 1
            # i as first index: n = i - lag, valid if 0 <= i-lag < n
            n1 = i - lag
            if 0 <= n1 < n:
                g += w[m] * (1.0 - h[n1])
            # i as second index: n = i, first index = i + lag, valid if 0 <= i+lag < n
            # contribution from -a[n] term: -a[i+lag]
            n2 = i + lag
            if 0 <= n2 < n:
                g -= w[m] * h[n2]
        grad[i] = g
    
    dx = 2.0 / n
    return obj * dx, grad * dx

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj(x, alpha)[0]
    def g(x):
        return smooth_max_obj(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)
