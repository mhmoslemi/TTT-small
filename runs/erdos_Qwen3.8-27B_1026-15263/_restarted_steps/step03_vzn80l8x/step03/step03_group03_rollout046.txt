def project_to_sum(h, target_sum):

    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break

        if np.any(mask):

def smooth_max_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr) * dx
    exp_vals = np.exp(alpha * (corr * dx - m))
    sm = m + np.log(np.sum(exp_vals)) / alpha
    return sm

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr) * dx
    exp_vals = np.exp(alpha * (corr * dx - m_val))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        if k < n:
            for j in range(k, n):
                grad[j] += wk * dx * (1.0 - h[j - k])
            for j in range(0, n - k):
                grad[j] += wk * dx * (-h[j + k])
        else:
            kk = 2 * n - 2 - k
            for j in range(kk, n):
                grad[j] += wk * dx * (1.0 - h[j - kk])
            for j in range(0, n - kk):
                grad[j] += wk * dx * (-h[j + kk])
    return grad

def smooth_max_grad_fast(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr) * dx
    exp_vals = np.exp(alpha * (corr * dx - m_val))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + (n - 1)
    second_idx = j_idx + m_idx - (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * (w * dx)[None, :]
    cs = valid_s * h[sc] * (w * dx)[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def subgradient_method(h0, n, target_sum, alpha, n_iters, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    for t in range(1, n_iters + 1):
        grad = smooth_max_grad_fast(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_sum(h_new, target_sum)
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:

    return best_h, best_c5

def pair_descent_sweep(h, corr, rng, n_cand=10):

def slsqp_polish(h0, n, target_sum, alpha, maxiter=500):
    def f(hx):
        return smooth_max_objective(hx, alpha)
    def g(hx):
        return smooth_max_grad_fast(hx, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

            best_h = h_res.copy()
            best_n = n_main

    # Phase 2: Multi-resolution exploration (25% of time)
    t_end2 = t_start + 0.70 * budget_s
    resolutions = [160, 200, 240, 280, 320, 360, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
        h_tr, c5_tr = adam_multistage(h_interp, n_tr, ts_tr, [50, 100, 200], 400)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr

    # Phase 3: SLSQP polish (15% of time)
    t_end3 = t_start + 0.85 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [20, 50, 100, 300, 800]:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_polish(best_h, n_f, ts_f, alpha, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 4: Intensive pair descent (15% of time)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall = 0
    while time.time() < t_start + budget_s - 3:
        h_fin, corr_fin, imp = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall = 0
        else:
            stall += 1
        if stall >= 100:
            noise = 0.03
            h_fin = project_to_sum(h_fin + rng.uniform(-noise, noise, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            stall = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
