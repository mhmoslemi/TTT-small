```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):

    for _ in range(500):

        if abs(diff) < 1e-14:

        else:

    #   Also j+p <= n-1, i.e., p <= n-1-j (same condition)
```

```python
def compute_gradient_of_lag(h, p, n):
    """Compute ∂corr[p]/∂h for a given lag p."""
    grad = np.zeros(n)
    if p >= 0:
        # First term: j in [0, n-1-p], (1-h[j+p])
        j_range = np.arange(0, n - p)
        grad[j_range] += (1.0 - h[j_range + p])
        # Second term: j in [p, n-1], -h[j-p]
        j_range2 = np.arange(p, n)
        grad[j_range2] += -h[j_range2 - p]
    else:
        q = -p
        # First term: j in [q, n-1] and j+q... wait
        # p < 0, valid_i_range = [-p, n-1] = [q, n-1]
        # First term: j in [q, n-1] and j+p in [0, n-1], i.e., j-q in [0, n-1]... 
        # wait, j+p = j-q, need 0 <= j-q < n, so q <= j < n+q, but j < n, so q <= j < n
        # Also j in valid range: j >= q

        # Also i in [0, n-1]: j+q <= n-1, j <= n-1-q
        # Contribution: -h[i] = -h[j+q]
        j_range2 = np.arange(0, n - q)
        grad[j_range2] += -h[j_range2 + q]
```

```python
def subgradient_method(h, n, target_sum, max_iter, alpha_init, top_k=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        # Find top K lags
        top_indices = np.argsort(corr)[-top_k:]
        
        # Compute subgradient (average of gradients of top K lags)
        grad = np.zeros(n)
        for r in top_indices:
            p = int(r) - (n - 1)
            grad += compute_gradient_of_lag(h, p, n)
        grad /= len(top_indices)
        
        # Take step
        v = h - alpha * grad
        h_new = project_onto_feasible(v, target_sum)
        
        # Check if improved
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Slow down alpha
            alpha *= 0.99
        else:
            # Backtrack
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        h = h_new  # Always move (even if not improved, to escape local optima?)
        # Actually, for subgradient methods, we should always move
        # Let's only keep the best but always update h
    
    return best_h, best_c5
```

```python
    grad = np.zeros(n)
    if p >= 0:
        if p < n:
            j1 = np.arange(0, n - p)
            if len(j1) > 0:
                grad[j1] += (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            if len(j2) > 0:
                grad[j2] += -h[j2 - p]
    else:
        q = -p
        if q < n:
            j1 = np.arange(q, n)
            if len(j1) > 0:
                grad[j1] += (1.0 - h[j1 - q])
            j2 = np.arange(0, n - q)
            if len(j2) > 0:
                grad[j2] += -h[j2 + q]
    return grad

def solve_1d_convex(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    # f(delta) = max(corr + delta * D) is convex in delta
    # Use ternary search
    a, b = lo, hi
    for _ in range(60):
        if b - a < 1e-13:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = float(np.max(corr + m1 * D))
        f2 = float(np.max(corr + m2 * D))
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def full_pair_sweep(h, corr, rng, n_cand=None, n_skip=1):
    """One full sweep of pair descent.
    If n_cand is None, try all k for each j.
    If n_skip > 1, skip some j's.
    """
    n = len(h)
    if n_cand is None:
        n_cand = n  # try all
    
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for jj in range(len(order)):
        j = order[jj]
        if n_skip > 1 and jj % n_skip != 0:
            continue
        
        # Compute B_j
        B_j = np.zeros(2 * n - 1)
        p_min = -j
        p_max = n - 1 - j
        for p in range(p_min, p_max + 1):
            B_j[n - 1 + p] += (1.0 - h[j + p])
        for p in range(-(n - 1), n):
            i = j - p
            if 0 <= i < n:
                if p >= 0:
                    if 0 <= i <= n - 1 - p:
                        B_j[n - 1 + p] += -h[i]
                else:
                    if -p <= i <= n - 1:
                        B_j[n - 1 + p] += -h[i]
        
        best_val = float(np.max(corr))
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Generate candidate k's
        if n_cand >= n:
            k_candidates = list(range(n))
            k_candidates.remove(j)
            rng.shuffle(k_candidates)
        else:
            k_candidates = list(rng.choice(np.setdiff1d(np.arange(n), j), size=min(n_cand, n-1), replace=False))
        
        for k in k_candidates:
            # Compute B_k
            B_k = np.zeros(2 * n - 1)
            p_min_k = -k
            p_max_k = n - 1 - k
            for p in range(p_min_k, p_max_k + 1):
                B_k[n - 1 + p] += (1.0 - h[k + p])
            for p in range(-(n - 1), n):
                i = k - p
                if 0 <= i < n:
                    if p >= 0:
                        if 0 <= i <= n - 1 - p:
                            B_k[n - 1 + p] += -h[i]
                    else:
                        if -p <= i <= n - 1:
                            B_k[n - 1 + p] += -h[i]
            
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            delta_opt = solve_1d_convex(corr, D, delta_lo, delta_hi)
            val_opt = float(np.max(corr + delta_opt * D))
            
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    
    return h, corr, improved
```

So C₅ = max(corr) * (2/n). For a "similar" function at different resolutions, max(corr) scales roughly as n, so C₅ is roughly the same. import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_feasible(v, target_sum):
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = 1.0 - float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = float(np.sum(x))
        if abs(s - target_sum) < 1e-13:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def precompute_B_all(h, n):
    B_all = np.zeros((n, 2 * n - 1))
    one_minus_h = 1.0 - h
    for j in range(n):
        B_all[j, n - 1 - j : 2 * n - 1 - j] += one_minus_h
        B_all[j, n - 1 : n + j + 1] += -h[0:j+1][::-1]
        if j < n - 1:
            B_all[j, j : n - 1] += -h[j+1:n][::-1]
    return B_all

def solve_1d_max(corr, D, lo, hi):
    """Find delta in [lo,hi] minimizing max(corr + delta * D)."""
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    a, b = lo, hi
    for _ in range(40):
        if b - a < 1e-13:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = float(np.max(corr + m1 * D))
        f2 = float(np.max(corr + m2 * D))
        if f1 <= f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def pair_descent_sweep(h, corr, B_all, rng, n_cand):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for j in order:
        B_j = B_all[j]
        best_val = float(np.max(corr))
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        if n_cand >= n:
            k_candidates = np.delete(np.arange(n), j)
            rng.shuffle(k_candidates)
        else:
            k_candidates = rng.choice(np.delete(np.arange(n), j), 
                                      size=min(n_cand, n - 1), replace=False)
        
        for k in k_candidates:
            D = B_j - B_all[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_max(corr, D, delta_lo, delta_hi)
            val_opt = float(np.max(corr + delta_opt * D))
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = int(k)
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            # Update B_all for changed coordinates
            B_all[j] = precompute_B_all_row(h, j, n)
            B_all[best_k] = precompute_B_all_row(h, best_k, n)
            improved = True
    
    return h, corr, B_all, improved

def precompute_B_all_row(h, j, n):
    B = np.zeros(2 * n - 1)
    one_minus_h = 1.0 - h
    B[n - 1 - j : 2 * n - 1 - j] += one_minus_h
