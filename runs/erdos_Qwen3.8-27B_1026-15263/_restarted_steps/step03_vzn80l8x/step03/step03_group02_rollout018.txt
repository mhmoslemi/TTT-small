from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = 1.0 - h.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def pgd_optimize(h0, n, target_sum, alpha, lr, momentum, max_iter, t_limit):

    v = np.zeros(n)
    best_h = h.copy()
    best_true_c5 = compute_c5(h)
    last_improve = 0

        obj, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad

def make_binary(h, n, target_count, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_count:
        excess = n_ones - target_count
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0

        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex(h, target_sum):
    lo = h.min() - 1.0
    hi = 1.0 - h.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    v = np.zeros(n)

    for it in range(max_iter):

            break
        obj, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = project_onto_simplex(h + v, target_sum)
        h = h_new

```python
def topk_subgrad(h, k=5):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[-k:]
    n = len(h)
    grad = np.zeros(n)
    for m in top_indices:
        j_idx = np.arange(n)
        first_idx = j_idx - m + n - 1
        second_idx = j_idx + m - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        grad += valid_f * (1.0 - h[fc]) - valid_s * h[sc]
    grad /= k
    return float(np.max(corr)), grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_onto_simplex(h, target_sum):
    lo = h.min() - 1.0
    hi = 1.0 - h.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def topk_subgrad(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    top_indices = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    j_arr = np.arange(n)
    for m in top_indices:
        first_idx = j_arr - m + (n - 1)
        second_idx = j_arr + m - (n - 1)
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        grad += valid_f.astype(float) * (1.0 - h[fc]) - valid_s.astype(float) * h[sc]
    grad /= k
    return m_val, grad

def pgd_full(h0, n, target_sum, alpha, lr, momentum, max_iter, t_limit, check_interval=300):
    h = h0.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        obj, grad = smooth_max_obj_grad(h, alpha)
        v = momentum * v - lr * grad
        h = project_onto_simplex(h + v, target_sum)
        if it % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def pgd_topk(h0, n, target_sum, k, lr, momentum, max_iter, t_limit, check_interval=500):

def make_binary(h, target_count):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_count:
        excess = n_ones - target_count
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target_count:
        deficit = target_count - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def interp_to_n(h_src, n_src, n_dst):

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)
