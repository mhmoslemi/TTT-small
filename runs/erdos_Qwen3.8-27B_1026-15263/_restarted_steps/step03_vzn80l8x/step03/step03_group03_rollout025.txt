import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_matrix(h, n):
    B = np.zeros((n, 2 * n - 1))
    nm1 = n - 1
    for j in range(n):
        k_start = max(0, nm1 - j)
        k_end = min(2 * n - 2, 2 * nm1 - j)
        if k_start <= k_end:
            idx = np.arange(k_start, k_end + 1)
            B[j, k_start:k_end + 1] += (1.0 - h[j - nm1 + idx])
        k_start2 = max(0, j)
        k_end2 = min(2 * n - 2, j + nm1)
        if k_start2 <= k_end2:
            B[j, k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d_precise(corr, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        lo_r, hi_r = lo, t_grid[1]
    elif best_idx == len(t_grid) - 1:
        lo_r, hi_r = t_grid[-2], hi
    else:
        lo_r, hi_r = t_grid[best_idx - 1], t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def full_pair_sweep(h, corr, B, n, rng, top_k=None):
    improved = False
    total_improvement = 0.0
    for j in range(n):
        best_val = np.max(corr)
        best_delta = 0.0
        best_k = -1
        if top_k is not None and top_k < n - 1:
            scores = np.zeros(n)
            top_indices = np.argsort(corr)[-10:]
            D_all = B[j][None, :] - B
            for k in range(n):
                if k == j:
                    scores[k] = -np.inf
                    continue
                D_k = D_all[k]
                scores[k] = -np.sum(D_k[top_indices] * corr[top_indices])
            candidates = np.argsort(scores)[::-1][:top_k]
            candidates = [c for c in candidates if c != j]
        else:
            candidates = [k for k in range(n) if k != j]
        for k in candidates:
            D = B[j] - B[k]
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d_precise(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            D = B[j] - B[best_k]
            corr += best_delta * D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            total_improvement += (np.max(corr - best_delta * D) - np.max(corr))
            improved = True
    return h, corr, improved

def block_move(h, corr, B, n, rng, block_size=3):
    j = rng.randint(n)
    k_start = rng.randint(0, n - block_size + 1)
    j_start = j
    if j_start + block_size > n:
        j_start = n - block_size
    D_block = np.zeros(2 * n - 1)
    for jj in range(block_size):
        kk = (k_start + jj) % n
        if kk != j_start + jj:
            D_block += B[j_start + jj] - B[kk]
    delta_lo = -min(h[j_start:j_start + block_size])
    delta_hi = min(1.0 - h[j_start:j_start + block_size], h[k_start:k_start + block_size])
    if delta_hi - delta_lo < 1e-14:
        return h, corr, False
    delta_opt = solve_1d_precise(corr, D_block, delta_lo, delta_hi)
    val_opt = np.max(corr + D_block * delta_opt)
    if val_opt < np.max(corr) - 1e-13:
        corr += delta_opt * D_block
        for jj in range(block_size):
            kk = (k_start + jj) % n
            h[j_start + jj] = np.clip(h[j_start + jj] + delta_opt, 0.0, 1.0)
            h[kk] = np.clip(h[kk] - delta_opt, 0.0, 1.0)
        return h, corr, True
    return h, corr, False

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_polish(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

        h_fin = project_to_sum(h_fin, n_fin / 2.0)
        corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
