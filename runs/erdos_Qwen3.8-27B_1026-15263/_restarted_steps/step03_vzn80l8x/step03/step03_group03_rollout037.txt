    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Coarse grid
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=15, accept_worse=False, temperature=0.0):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Candidate set: neighbors, far positions, random
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, 
                 (j + n // 3) % n, (j + 2 * n // 3) % n, (j + n // 4) % n,
                 (j + 3 * n // 4) % n]
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_minimax(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            # Simulated annealing acceptance
            if accept_worse and val_opt > best_val and temperature > 0:
                if rng.random() < np.exp(-(val_opt - best_val) / temperature):
                    if val_opt < best_val - 1e-13:
                        best_val = val_opt
                        best_delta = delta_opt
                        best_D = D
                        best_k = k
                # Also accept worsening with probability
            elif val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr = corr + best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: w[m] * d(corr[m])/dh[j] * dx... wait, the objective is max(corr)*dx
    # But for optimization, dx is constant, so we can ignore it.
    # Actually, let's work with the unscaled value and multiply by dx at the end.
    
    # d(corr[k])/dh[j]:
    # corr[k] = sum_n h[n] * (1-h[n-m]) where m = k-(n-1)
    # d/dh[j]: (1-h[j-m]) * [valid] - h[j+m] * [valid]
    grad = np.zeros(n)
    nm1 = n - 1
    
    for j in range(n):
        # First term: (1-h[j-m]) for k where 0<=j-m<n, m=k-nm1
        # k from j to j+nm1 (clipped)
        k1 = max(0, j)
        k2 = min(2*n-2, j + nm1)
        if k1 <= k2:
            m = np.arange(k1, k2+1) - nm1
            grad[j] += np.sum(w[k1:k2+1] * (1.0 - h[j - m]))
        # Second term: -h[j+m] for k where 0<=j+m<n, m=k-nm1
        k1 = max(0, nm1 - j)
        k2 = min(2*n-2, 2*nm1 - j)
        if k1 <= k2:
            m = np.arange(k1, k2+1) - nm1
            grad[j] += np.sum(-w[k1:k2+1] * h[j + m])
    
    return obj, grad

    h_new = h - lr * grad
    return project_to_sum(h_new, np.sum(h))

    elif pattern_type == 2:
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(np.pi * x + phase) + 0.1 * np.sin(2 * np.pi * x + 2 * phase)
    elif pattern_type == 3:
        h = np.zeros(n)
        blk = n // 4
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk + rng.randint(-2, 3), n)] = 1.0
    elif pattern_type == 4:
        h = 0.5 + 0.45 * np.tanh(3.0 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi)))
    else:
        h = rng.uniform(0, 1, n)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Intensive pair descent with annealing (40% budget)
    t_end1 = t_start + 0.40 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    target = n_w / 2.0
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0
    temperature = max(0.003, (best_c5 - 0.36) * 0.4)

    while time.time() < t_end1:
        temperature = max(temperature * 0.997, 1e-8)
        h_work, corr_w, _ = pair_descent_sweep(h_work, corr_w, rng, n_cand=25, temperature=temperature)
        h_work = project_to_sum(h_work, target)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall > 150:
            noise = 0.08 * min(stall // 150, 3)
            h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), target)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            temperature = max(0.002, (best_c5 - 0.36) * 0.5)
            stall = 0

    # Phase 2: Gradient polish (15% budget)
    t_end2 = t_start + 0.55 * budget_s
    n_g = best_n
    target_g = n_g / 2.0
    for alpha in [50, 200, 800, 2000]:
        if time.time() >= t_end2:
            break
        h_grad = gradient_polish(best_h, target_g, alpha, maxiter=400, tol=1e-14)
        c5_grad = compute_c5(h_grad)
        if c5_grad < best_c5:
            best_c5 = c5_grad
            best_h = h_grad.copy()
            best_n = n_g

    # Phase 3: Multi-resolution exploration (25% budget)
    t_end3 = t_start + 0.80 * budget_s
    x_best = np.linspace(0, 2.0, best_n, endpoint=False)
    resolutions = [160, 200, 240, 280, 320, 400]
    for n_tr in resolutions:
        if time.time() >= t_end3:
            break
        target_tr = n_tr / 2.0
        if abs(target_tr - round(target_tr)) > 1e-6:
            continue
        x_new = np.linspace(0, 2.0, n_tr, endpoint=False)
        starts = []
        starts.append(project_to_sum(np.interp(x_new, x_best, best_h, period=2.0), target_tr))
        for pt in range(3):
            starts.append(project_to_sum(make_periodic_start(n_tr, pt + rng.randint(0, 2), rng), target_tr))

        for h_st in starts:
            if time.time() >= t_end3:
                break
            n_sweeps = max(10, int((t_end3 - time.time()) / 3.0))
            h_res, c5_res = multi_start_descent(h_st, n_tr, target_tr, rng, n_sweeps, n_cand=15, t_end=t_end3)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_tr

    # Phase 4: Final intensive pair descent (15% budget)
    t_end4 = t_start + budget_s - 10
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    target_fin = n_fin / 2.0
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    temp_fin = 0.001
    stall_fin = 0

    while time.time() < t_end4:
        temp_fin = max(temp_fin * 0.999, 1e-10)
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=30, temperature=temp_fin)
        h_fin = project_to_sum(h_fin, target_fin)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            stall_fin += 1
        if stall_fin > 100:
            h_fin = project_to_sum(h_fin + rng.uniform(-0.03, 0.03, n_fin), target_fin)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            temp_fin = 0.002
            stall_fin = 0

    # Phase 5: Binary rounding attempt (last 5%)
    t_end5 = t_start + budget_s - 2
    h_bin = (best_h > 0.5).astype(float)
    n_br = len(h_bin)
    target_br = n_br / 2.0
    ones_count = int(np.sum(h_bin))
    if ones_count != int(target_br):
        if ones_count > int(target_br):
            ones_idx = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[ones_idx])
            flips = ones_idx[order[:ones_count - int(target_br)]]
            h_bin[flips] = 0
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros_idx])
            flips = zeros_idx[order[:int(target_br) - ones_count]]
            h_bin[flips] = 1

    corr_br = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    improved_br = True
    while improved_br and time.time() < t_end5:
        improved_br = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for j in ones:
            B_j = compute_B(h_bin, j, n_br)
            for k in zeros:
                if k == j:
                    continue
                B_k = compute_B(h_bin, k, n_br)
                D = B_j - B_k
                val = np.max(corr_br + D)
                if val < np.max(corr_br) - 1e-12:
                    h_bin[j] = 0
                    h_bin[k] = 1
                    corr_br = corr_br + D
                    improved_br = True
                    break
            if time.time() >= t_end5:
                break

    c5_br = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode="full")) * 2.0 / n_br)
    if c5_br < best_c5:
        best_c5 = c5_br
        best_h = h_bin.copy()
        best_n = n_br

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
