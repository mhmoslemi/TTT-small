        return project_to_sum(h0.copy(), tsum)

def coord_desc(h, tsum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = subgrad_at_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, tsum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, tsum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def simulated_anneal(h, tsum, rng, n_iter, t0, t1):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = project_to_sum(h, tsum)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    one_set = np.where(h == 1.0)[0]
    zero_set = np.where(h == 0.0)[0]
    for it in range(n_iter):
        t = t0 * (t1 / t0) ** (it / max(n_iter - 1, 1))
        if len(one_set) < 1 or len(zero_set) < 1:
            break
        i1 = one_set[rng.randint(len(one_set))]
        i2 = zero_set[rng.randint(len(zero_set))]
        h[i1] = 0.0
        h[i2] = 1.0
        h = project_to_sum(h, tsum)
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-10)):
            c5 = c5_new
            one_set = np.where(h == 1.0)[0]
            zero_set = np.where(h == 0.0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i2] = 0.0
            h = project_to_sum(h, tsum)
    return best_h, best_c5

    return (best_h, best_c5, best_n)
