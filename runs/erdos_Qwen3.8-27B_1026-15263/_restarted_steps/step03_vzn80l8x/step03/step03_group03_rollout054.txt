    if bi == 0:
        return d_lo
    if bi == len(d_grid) - 1:
        return d_hi
    lo_r = d_grid[bi - 1]
    hi_r = d_grid[bi + 1]
    for _ in range(28):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent(h, corr, rng, n_cand, active_set=None):
    n = len(h)
    nm1 = n - 1
    current_max = np.max(corr)
    improved = False
    order = np.arange(n)
    rng.shuffle(order)
    for j in order:
        best_delta = 0.0
        best_val = current_max
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n, (j + n // 4) % n]
        if active_set is not None:
            for ak in active_set:
                k_set.append((j + ak) % n)
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set([j])
        for k in k_set:
            if k in seen:
                continue
            seen.add(k)
            D = compute_D(h, j, k, n)
            d_lo = max(-h[j], h[k] - 1.0)
            d_hi = min(1.0 - h[j], h[k])
            if d_hi - d_lo < 1e-13:
                continue
            d_opt = solve_1d_minmax(corr, D, d_lo, d_hi)
            val_opt = np.max(corr + D * d_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = d_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            new_max = np.max(corr)
            if new_max < current_max:
                current_max = new_max
                improved = True
    return h, corr, improved

def triple_descent(h, corr, rng, n_trials):
    n = len(h)
    current_max = np.max(corr)
    improved = False
    for _ in range(n_trials):
        j, k, l = rng.choice(n, size=3, replace=False)
        sign_j = 1
        sign_k = -1
        sign_l = 0
        mask_pos = (h[j] < 1.0 - 1e-6)
        mask_neg_j = (h[k] > 1e-6)
        if not (mask_pos and mask_neg_j):
            continue
        d_lo = max(-h[j] * sign_j, h[k] * sign_k, h[l] * (-1))
        d_hi = min((1.0 - h[j]) * sign_j, h[k] * (-sign_k), (1.0 - h[l]))
        if d_hi - d_lo < 1e-12:
            continue
        D = compute_D(h, j, k, n)
        D_l = np.zeros(2 * n - 1)
        nm1 = n - 1
        lo1 = max(0, nm1 - l)
        hi1 = min(2 * n - 2, 2 * nm1 - l)
        if lo1 <= hi1:
            m_vals = np.arange(lo1, hi1 + 1)
            pidx = m_vals + l - nm1
            D_l[lo1:hi1 + 1] += (1.0 - h[pidx])
        lo2 = max(0, l - nm1)
        hi2 = min(2 * n - 2, l)
        if lo2 <= hi2:
            m_vals2 = np.arange(lo2, hi2 + 1)
            pidx2 = l - m_vals2 + nm1
            D_l[lo2:hi2 + 1] -= h[pidx2]
        D_total = D + D_l
        for _ in range(15):
            mid = (d_lo + d_hi) * 0.5
            if np.max(corr + D_total * mid) <= np.max(corr + D_total * d_lo):
                d_lo = mid
            else:
                d_hi = mid
        d_opt = (d_lo + d_hi) * 0.5
        h_new = h.copy()
        h_new[j] = np.clip(h_new[j] + sign_j * d_opt, 0.0, 1.0)
        h_new[k] = np.clip(h_new[k] + sign_k * d_opt, 0.0, 1.0)
        h_new[l] = np.clip(h_new[l] - d_opt, 0.0, 1.0)
        if np.abs(np.sum(h_new) - np.sum(h)) > n * 1e-10:
            h_new = project_to_sum(h_new, np.sum(h))
        corr_new = compute_corr(h_new)
        if np.max(corr_new) < current_max - 1e-12:
            h[:] = h_new
            corr[:] = corr_new
            current_max = np.max(corr)
            improved = True
    return h, corr, improved

def block_move(h, corr, rng, n_trials):
    n = len(h)
    current_max = np.max(corr)
    improved = False
    for _ in range(n_trials):
        size = rng.randint(2, max(3, n // 20))
        pos = rng.randint(0, n - size)
        shift = rng.choice([-1, -2, -3, 1, 2, 3])
        h_new = h.copy()
        block_vals = h_new[pos:pos + size].copy()
        for i in range(size):
            src = pos + i
            dst = (pos + i + shift) % n
            if 0 <= dst < n:
                h_new[dst] = np.clip(h_new[dst] + h_new[src] - 0.5, 0.0, 1.0)
        h_new = project_to_sum(h_new, np.sum(h))
        corr_new = compute_corr(h_new)
        if np.max(corr_new) < current_max - 1e-10:
            h[:] = h_new
            corr[:] = corr_new
            current_max = np.max(corr)
            improved = True
    return h, corr, improved

def active_set_sw(h, corr, n_active=10):
    top_idx = np.argsort(corr)[-n_active:]
    shifts = top_idx - (len(h) - 1)
    return shifts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    n0 = best_n
    ts0 = n0 / 2.0
    h_work = best_h.copy()
    corr_work = compute_corr(h_work)

    # Phase 1: Intensive pair descent (45% budget)
    t_end1 = t_start + 0.45 * budget_s
    stall = 0
    while time.time() < t_end1:
        act = active_set_sw(h_work, corr_work, 8)
        h_work, corr_work, imp = pair_descent(h_work, corr_work, rng, n_cand=18, active_set=act)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n0
            stall = 0
        else:
            stall += 1
        if stall >= 300:
            noise_scale = min(0.02, 0.005 * (stall // 100))
            h_work = project_to_sum(h_work + rng.uniform(-noise_scale, noise_scale, n0), ts0)
            corr_work = compute_corr(h_work)
            stall = 0
        elif stall >= 150 and stall % 50 == 0:
            h_work = project_to_sum(h_work + rng.uniform(-0.01, 0.01, n0), ts0)
            corr_work = compute_corr(h_work)

    # Phase 2: Triple descent + more pair descent (20% budget)
    t_end2 = t_start + 0.65 * budget_s
    h_work = best_h.copy()
    corr_work = compute_corr(h_work)
    while time.time() < t_end2:
        act = active_set_sw(h_work, corr_work, 6)
        h_work, corr_work, _ = triple_descent(h_work, corr_work, rng, 30)
        h_work, corr_work, _ = pair_descent(h_work, corr_work, rng, n_cand=20, active_set=act)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n0

    # Phase 3: Multi-resolution exploration (20% budget)
    t_end3 = t_start + 0.85 * budget_s
    resolutions = [120, 160, 240, 320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end3:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        for h_st in [h_interp,
                     project_to_sum(rng.uniform(0, 1, n_tr), ts_tr)]:
            if time.time() >= t_end3:
                break
            corr_st = compute_corr(h_st)
            max_sweeps = min(40, max(3, int((t_end3 - time.time()) / 3.0)))
            for _ in range(max_sweeps):
                if time.time() >= t_end3:
                    break
                act_tr = active_set_sw(h_st, corr_st, 5)
                h_st, corr_st, _ = pair_descent(h_st, corr_st, rng, n_cand=12, active_set=act_tr)
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 4: Adam smooth-max polish (10% budget)
    t_end4 = t_start + 0.95 * budget_s
    n_pol = best_n
    ts_pol = n_pol / 2.0
    h_pol = best_h.copy()
    for alpha in [50, 500, 5000]:
        if time.time() >= t_end4:
            break
        lr = 0.01
        for it in range(300):
            if time.time() >= t_end4:
                break
            corr_pol = compute_corr(h_pol)
            _, w = smooth_max_and_weights(corr_pol, alpha)
            grad = compute_grad(h_pol, w)
            gnorm = np.linalg.norm(grad)
            if gnorm > 0.5:
                grad *= 0.5 / gnorm
            h_pol = project_to_sum(h_pol - lr * grad, ts_pol)
            if it % 50 == 0:
                lr *= 0.85
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-11:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_pol

    # Phase 5: Final intensive pair descent (5% budget)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = compute_corr(h_fin)
    while time.time() < t_start + budget_s - 3:
        act_fin = active_set_sw(h_fin, corr_fin, 8)
        h_fin, corr_fin, _ = pair_descent(h_fin, corr_fin, rng, n_cand=22, active_set=act_fin)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-11:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
        elif time.time() < t_start + budget_s - 3:
            h_fin = project_to_sum(h_fin + rng.uniform(-0.003, 0.003, n_fin), n_fin / 2.0)
            corr_fin = compute_corr(h_fin)

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
