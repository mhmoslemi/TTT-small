import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    max_improve = 0.0
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        if n_cand >= n:
            k_set = [k for k in range(n) if k != j]
        else:
            k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                     (j + n // 3) % n, (j + 2 * n // 3) % n]
            for _ in range(max(0, n_cand - 5)):
                k_set.append(rng.randint(n))
            k_set = [k for k in k_set if k != j]
        seen = set()
        for k in k_set:
            if k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
            max_improve = max(max_improve, best_val and (np.max(corr) - best_val) or 0)
    return h, corr, improved

def lse_obj_grad_vectorized(h, alpha):
    corr = get_corr(h)
    n = len(h)
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for i in range(n):
        j_lo = max(0, i - n + 1)
        j_hi = min(n - 1, i)
        if j_lo <= j_hi:
            ks = i - np.arange(j_lo, j_hi + 1)
            grad[i] += np.dot(w[ks], (1.0 - h[j_lo:j_hi + 1]))
        k_hi = min(n - 1 - i, 2 * n - 2)
        if k_hi >= 0:
            ks2 = np.arange(0, k_hi + 1)
            grad[i] -= np.dot(w[ks2], h[i:i + k_hi + 1])
    return obj, grad

def gradient_optimize(h0, target_sum, alphas, max_iter_each, lbfgs_m=50):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        def fun(x):
            o, _ = lse_obj_grad_vectorized(x, alpha)
            return o
        def fprime(x):
            _, g = lse_obj_grad_vectorized(x, alpha)
            return g
        try:
            res = minimize(fun, h.copy(), jac=fprime, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter_each, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'm': lbfgs_m})

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
        except Exception:
            pass
        # Also try a few gradient steps with projection
        for _ in range(20):
            _, grad = lse_obj_grad_vectorized(h, alpha)
            lr = 0.1 / alpha
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                lr *= 0.5
    return best_h, best_c5

def targeted_critical_shift(h, corr, target_sum, n_crit):
    n = len(h)
    c5_val = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    crit_shifts = sorted_idx[:n_crit]
    for k_star in crit_shifts:
        # Find positions where h[j+k_star]=1 and h[j]=0 (bad pairs for this shift)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, n - 1 - k_star)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        bad_js = js[(h[js + k_star] > 0.5) & (h[js] < 0.5)]
        for j in bad_js:
            # Try reducing h[j+k_star] or increasing h[j]
            idx_a = j + k_star
            idx_b = j
            for idx_hi_v in [idx_a, idx_b]:
                for idx_lo_v in [idx_b, idx_a]:
                    if idx_hi_v == idx_lo_v:
                        continue
                    B_hi = compute_B(h, idx_hi_v, n)
                    B_lo = compute_B(h, idx_lo_v, n)
                    D = B_hi - B_lo
                    delta_lo = max(-h[idx_hi_v], h[idx_lo_v] - 1.0)
                    delta_hi = min(1.0 - h[idx_hi_v], h[idx_lo_v])
                    if delta_hi - delta_lo < 1e-13:
                        continue
                    delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
                    new_corr = corr + delta_opt * D
                    if np.max(new_corr) < c5_val - 1e-12:
                        corr = new_corr
                        h[idx_hi_v] = np.clip(h[idx_hi_v] + delta_opt, 0.0, 1.0)
                        h[idx_lo_v] = np.clip(h[idx_lo_v] - delta_opt, 0.0, 1.0)
                        c5_val = np.max(corr)
                        break
    return h, corr

def interpolate_to_n(h_old, n_old, n_new):

def make_block_init(n, pattern_frac, rng):
    h = np.zeros(n)

    rng.shuffle(positions)
