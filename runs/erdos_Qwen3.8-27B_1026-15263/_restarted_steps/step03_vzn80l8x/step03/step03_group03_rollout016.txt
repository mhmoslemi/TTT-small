def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_max_obj(x, alpha)
    def g(x):
        return smooth_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    # Phase 3: Multi-resolution exploration (15%)
    t_end3 = t_start + 0.60 * budget_s
    resolutions = [160, 200, 240, 320, 360, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end3:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        h_rand = project_to_sum(rng.uniform(0, 1, n_tr), ts_tr)
        blk = max(1, n_tr // 8)
        h_blk = np.zeros(n_tr)
        for i in range(0, n_tr, 2 * blk):
            h_blk[i:min(i + blk, n_tr)] = 1.0
        h_blk = project_to_sum(h_blk, ts_tr)
        h_sin = np.sin(np.pi * x_new) ** 2
        h_sin = project_to_sum(h_sin, ts_tr)

        for h_st in [h_interp, h_rand, h_blk, h_sin]:
            if time.time() >= t_end3:
                break
            h_st = h_st.copy()
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            max_sweeps = min(40, max(3, int((t_end3 - time.time()) / 3.0)))
            for _ in range(max_sweeps):
                if time.time() >= t_end3:
                    break
                h_st, corr_st, _ = pair_descent_sweep(h_st, corr_st, rng, n_cand=15)
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 4: Higher resolution polish (25%)
    t_end4 = t_start + 0.85 * budget_s
    n_hi = 480
    if best_n < n_hi:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), n_hi / 2.0)
    else:
        h_hi = best_h.copy()
        n_hi = best_n

    corr_hi = np.correlate(h_hi, 1.0 - h_hi, mode="full")
    while time.time() < t_end4:
        h_hi, corr_hi, _ = pair_descent_sweep(h_hi, corr_hi, rng, n_cand=20)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 5: SLSQP polish (10%)
    t_end5 = t_start + 0.95 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [50, 200, 500, 1000]:
        if time.time() >= t_end5:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Final pair descent (5%)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=25)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
