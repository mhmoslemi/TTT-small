import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    # Use ternary search with more iterations for precision
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(A + D * m1)
        f2 = np.max(A + D * m2)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=20):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = list(set([(j + d) % n for d in [1, -1, 2, -2, 3, n//2, n//3, 2*n//3, n//4, 3*n//4]]))
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr = corr + best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def triple_descent_step(h, corr, rng, n_trials=5):
    """Try 3-point moves: adjust h[j], h[k], h[l] with delta1 at j, delta2 at k, -(delta1+delta2) at l"""
    n = len(h)
    best_val = np.max(corr)
    best_move = None
    for _ in range(n_trials):
        j = rng.randint(n)
        k = (j + rng.randint(1, n)) % n
        l = (j + rng.randint(1, n)) % n
        if len({j, k, l}) < 3:
            continue
        B_j = compute_B(h, j, n)
        B_k = compute_B(h, k, n)
        B_l = compute_B(h, l, n)
        # We want to find delta1, delta2 that minimize max(corr + delta1*(B_j-B_l) + delta2*(B_k-B_l))
        # subject to constraint bounds
        D1 = B_j - B_l
        D2 = B_k - B_l
        # Simple grid search in 2D
        d1_lo = max(-h[j], 0.0)  # delta1 >= -h[j] (so h[j]+delta1 >= 0)
        d1_hi = min(1.0-h[j], h[l])  # delta1 <= 1-h[j] and -(delta1+delta2) <= h[l] means delta1+delta2 >= -h[l]
        # This is getting complex. Let's use a simpler approach:
        # Just do a 1D search along a random direction in (delta1, delta2) space
        angle = rng.uniform(0, 2*np.pi)
        dir1 = np.cos(angle)
        dir2 = np.sin(angle)
        # Find max step size
        s_max = 1e-3
        for s_eval in [0.01, 0.005, 0.001]:
            d1 = s_eval * dir1
            d2 = s_eval * dir2
            if h[j] + d1 < -1e-10 or h[j] + d1 > 1 + 1e-10:
                continue
            if h[k] + d2 < -1e-10 or h[k] + d2 > 1 + 1e-10:
                continue
            if h[l] - d1 - d2 < -1e-10 or h[l] - d1 - d2 > 1 + 1e-10:
                continue
            s_max = s_eval
            break
        if s_max <= 0:
            continue
        # 1D search along (s*dir1, s*dir2) for s in [-s_max, s_max]
        D_eff = s_max * (dir1 * D1 + dir2 * D2)
        s_opt = solve_1d(corr, D_eff, -1.0, 1.0)
        val_opt = np.max(corr + s_opt * D_eff)
        if val_opt < best_val - 1e-12:
            d1 = s_opt * s_max * dir1
            d2 = s_opt * s_max * dir2
            if 0 <= h[j] + d1 <= 1 and 0 <= h[k] + d2 <= 1 and 0 <= h[l] - d1 - d2 <= 1:
                best_move = (j, k, l, d1, d2)
                best_val = val_opt
    if best_move is not None:
        j, k, l, d1, d2 = best_move
        h[j] += d1
        h[k] += d2
        h[l] -= d1 + d2
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return h, corr, True
    return h, corr, False

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient computation
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter=800):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f_and_g(h):
        return smooth_max_and_grad(h, alpha)
    
    try:
        res = minimize(lambda x: smooth_max_and_grad(x, alpha)[0], h0.copy(),
                       method='SLSQP', jac=lambda x: smooth_max_and_grad(x, alpha)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def structured_perturb(h, rng, n):
    """Apply a structured perturbation: flip signs of a random subset"""
    n_flip = rng.randint(n // 10, n // 4)
    flip_idx = rng.choice(n, size=n_flip, replace=False)
    h_pert = h.copy()
    h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
    return project_to_sum(h_pert, n / 2.0)

def binary_round_and_refine(h, n, rng, n_sweeps=30):
    """Round to binary, then refine with pair descent"""
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Remove excess ones from positions with lowest h values
        ones_idx = np.where(h_bin == 1.0)[0]
        order = ones_idx[np.argsort(h[ones_idx])]
        h_bin[order[:n_ones - n//2]] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = zeros_idx[np.argsort(-h[zeros_idx])]
        h_bin[order[:n//2 - n_ones]] = 1.0
    h_bin = project_to_sum(h_bin, n / 2.0)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    for _ in range(n_sweeps):
        h_bin, corr, _ = pair_descent_sweep(h_bin, corr, rng, n_cand=15)
    return h_bin, compute_c5(h_bin)

    return np.sum(x) - _slsqp_target

def _slsqp_cjac(x):
    return np.ones(_slsqp_n)

def _slsqp_f(x):
    return smooth_max_and_grad(x, _slsqp_alpha)[0]

def _slsqp_g(x):
    return smooth_max_and_grad(x, _slsqp_alpha)[1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(600):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    n_grid = 24
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == n_grid - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(60):
        m1 = lo_r + (hi_r - lo_r) / 3.0
        m2 = hi_r - (hi_r - lo_r) / 3.0
        if np.max(A + D * m1) <= np.max(A + D * m2):
            hi_r = m2
        else:
            lo_r = m1
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=25):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        offsets = [1, -1, 2, -2, 3, -3, 5, -5, n//4, -n//4, n//3, -n//3,
                   n//2, n//5, -n//5, 2*n//5, 3*n//5, n//6, 5*n//6]
        k_set = list(set([(j + off) % n for off in offsets]))
        n_rand = max(0, n_cand - len(k_set))
        for _ in range(n_rand):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter=600):
    global _slsqp_target, _slsqp_alpha, _slsqp_n
    _slsqp_target = target_sum
    _slsqp_alpha = alpha
    _slsqp_n = n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_f, h0.copy(), method='SLSQP', jac=_slsqp_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def structured_perturb(h, rng, n, strength=0.3):
    n_flip = max(3, int(n * strength * rng.uniform(0.5, 1.5)))
    flip_idx = rng.choice(n, size=min(n_flip, n), replace=False)
    h_pert = h.copy()
    h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
    return project_to_sum(h_pert, n / 2.0)

def block_perturb(h, rng, n):
    n_blocks = rng.randint(2, 8)
    block_sizes = rng.dirichlet(np.ones(n_blocks) * 0.5)
    h_pert = np.zeros(n)
    pos = 0
    val = 1.0
    for i in range(n_blocks):
        bs = max(1, int(n * block_sizes[i]))
        h_pert[pos:pos+bs] = val
        pos += bs
        val = 1.0 - val
    if pos < n:
        h_pert[pos:] = val
    return project_to_sum(h_pert, n / 2.0)

def sine_perturb(h, rng, n):
    freq = rng.uniform(1.0, 8.0)
    phase = rng.uniform(0, 2 * np.pi)
    h_pert = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h_pert = np.clip(h_pert, 0.0, 1.0)
    return project_to_sum(h_pert, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Aggressive pair descent from initial (45% budget)
    t_end1 = t_start + 0.45 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0
    stall_thresh = 50

    while time.time() < t_end1 and time_left() > 30:
        h_work, corr_w, imp = pair_descent_sweep(h_work, corr_w, rng, n_cand=25)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1
        if stall >= stall_thresh:
            r = rng.uniform()
            if r < 0.4:
                h_work = structured_perturb(h_work, rng, n_w, strength=0.15)
            elif r < 0.7:
                h_work = project_to_sum(h_work + rng.uniform(-0.08, 0.08, n_w), n_w / 2.0)
            elif r < 0.85:
                h_work = block_perturb(h_work, rng, n_w)
            else:
                h_work = sine_perturb(h_work, rng, n_w)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
            stall_thresh = max(30, stall_thresh + 5)

    # Phase 2: Multi-resolution and multi-start (25% budget)
    t_end2 = t_start + 0.70 * budget_s
    resolutions = [240, 360, 480]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h), ts_tr)
        starts = [h_interp,
                  project_to_sum(rng.uniform(0, 1, n_tr), ts_tr),
                  block_perturb(h_interp, rng, n_tr)]
        for h_st in starts:
            if time.time() >= t_end2:
                break
            h_st = project_to_sum(h_st, ts_tr)
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            max_sweeps = min(50, max(5, int((t_end2 - time.time()) / 4.0)))
            st_stall = 0
            for _ in range(max_sweeps):
                if time.time() >= t_end2:
                    break
                h_st, corr_st, imp_st = pair_descent_sweep(h_st, corr_st, rng, n_cand=20)
                c5_st = compute_c5(h_st)
                if c5_st < best_c5 - 1e-11:
                    best_c5 = c5_st
                    best_h = h_st.copy()
                    best_n = n_tr
                    st_stall = 0
                else:
                    st_stall += 1
                if st_stall >= 30:
                    h_st = structured_perturb(h_st, rng, n_tr, strength=0.1)
                    corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
                    st_stall = 0
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 3: SLSQP polishing (15% budget)
    t_end3 = t_start + 0.85 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [30, 100, 300, 1000, 3000]:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 4: Final intensive pair descent (15% budget)
    t_end4 = t_start + budget_s - 5
    h_fin = best_h.copy()
    n_fin = len(h_fin)
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    fin_stall = 0
    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=30)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            fin_stall = 0
        else:
            fin_stall += 1
        if fin_stall >= 40:
            h_fin = project_to_sum(h_fin + rng.uniform(-0.03, 0.03, n_fin), n_fin / 2.0)
            corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
            fin_stall = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
