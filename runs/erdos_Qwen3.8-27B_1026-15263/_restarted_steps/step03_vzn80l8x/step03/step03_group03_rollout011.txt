def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(corr, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 10)
    vals = np.max(corr[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(24):
        mid = (lo_r + hi_r) * 0.5
        if np.max(corr + D * mid) <= np.max(corr + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_sweep(h, corr, n, B_all, rng, n_cand, recompute_every=10):
    """One sweep of pairwise coordinate descent."""
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    moves = 0
    
    for j in order:
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        
        # Generate candidates
        cands = list(set([(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n]))
        remaining = n_cand - len(cands)
        if remaining > 0:
            cands.extend(rng.choice(n, size=remaining, replace=False).tolist())
        
        seen = set([j])
        for k in cands:
            if k in seen:
                continue
            seen.add(k)
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            D = B_all[j] - B_all[k]
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k

            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
            moves += 1
            B_all[j] = compute_B(h, j, n)
            B_all[best_k] = compute_B(h, best_k, n)
    
    return h, corr, improved, moves

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = m_val + np.log(sum_exp) / alpha
    
    # Gradient: d(c_m)/dh[i] for the weighted sum
    # c_m = sum_j h[j+m-(n-1)]*(1-h[j]) for valid j
    # d(c_m)/dh[i] = sum over m where i appears
    # i appears in (1-h[j]) term when j = i: contributes -h[j+m-(n-1)] = -h[i+m-n+1]
    # i appears in h[j+m-(n-1)] term when j+m-n+1 = i, i.e., j = i-m+n-1: contributes (1-h[j])
    
    grad = np.zeros(n)
    for i in range(n):
        # i in the (1-h) position: j=i, m ranges over valid m where 0<=i+m-n+1<n
        # corr term: -h[i+m-n+1] * w[m]
        # i in the h position: j=i-m+n-1, m ranges valid
        # contribution: (1-h[j]) * w[m]
        # Combined: for each m where both terms are valid:
        #   w[m] * ((1-h[j]) - h[j+m-n+1]) where j is such that i = j or i = j+m-n+1
        
        # Actually, let me think of it differently.
        # c_m = sum_j h[j+m-n+1]*(1-h[j]) for valid j (0<=j<n and 0<=j+m-n+1<n)
        # d(c_m)/dh[i] = (1-h[i]) if m-n+1+i in [0,n-1] (i is in the h position with j=i-m+n-1)
        #              + (-h[i+m-n+1]) if i in [0,n-1] (i is in the (1-h) position with j=i)
        # Hmm, let me be more careful.
        # c_m = sum_{j: valid} h[j+m-n+1]*(1-h[j])
        # ∂c_m/∂h[i] = (1-h[j]) * [j+m-n+1 = i] + h[j+m-n+1] * (-1) * [j = i]
        #            = (1-h[i-m+n-1]) * [i-m+n-1 in [0,n-1]] - h[i+m-n+1] * [i+m-n+1 in [0,n-1]]
        #            (where the second term has j=i, so we need i in [0,n-1] which is always true)
        
        # For term 1: j = i-m+n-1, need 0 <= i-m+n-1 <= n-1
        #   → m-n+1 <= i-m+n-1... wait: 0 <= i-m+n-1 <= n-1
        #   → m-n+1 <= i <= m (from 0 <= i-m+n-1) and i-m+n-1 <= n-1 → i <= m
        #   So: m-n+1 <= i <= m, i.e., i ∈ [m-n+1, m] ∩ [0, n-1]
        
        # For term 2: j = i, need 0 <= i+m-n+1 <= n-1
        #   → n-1-i <= m <= n-1+i
        #   So: m ∈ [n-1-i, n-1+i] ∩ [0, 2n-2]
        
        # ∂c_m/∂h[i] = (1-h[i-m+n-1]) * I(m-n+1<=i<=m) - h[i+m-n+1] * I(n-1-i<=m<=n-1+i)
    
    # This is O(n²) to compute the gradient. For n=240, that's 57600 which is fine.
    # But let me vectorize.
    
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    
    # Term 1: (1-h[i-m+n-1]) valid when 0 <= i-m+n-1 < n
    j1 = i_idx - m_idx + n - 1
    valid1 = (j1 >= 0) & (j1 < n)
    term1 = valid1 * (1.0 - h[np.clip(j1, 0, n-1)])
    
    # Term 2: -h[i+m-n+1] valid when 0 <= i+m-n+1 < n
    j2 = i_idx + m_idx - n + 1
    valid2 = (j2 >= 0) & (j2 < n)
    term2 = valid2 * h[np.clip(j2, 0, n-1)]
    
    grad = np.sum((term1 - term2) * w[None, :], axis=1)
    
    return obj, grad
