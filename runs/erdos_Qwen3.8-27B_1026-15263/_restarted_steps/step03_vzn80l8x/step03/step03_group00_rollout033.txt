def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    N = 2 * n
    H = fft(h, N)
    H1 = fft(1.0 - h, N)
    corr = ifft(H * np.conj(H1)).real[:2*n-1]
    return corr

def compute_corr_direct(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:

        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def frank_wolfe(h0, n, target_sum, max_iter, gamma_factor=2.0):

    h = project_to_sum(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()

        m_star = int(np.argmax(corr))
        c5 = float(corr[m_star]) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        lag = m_star - (n - 1)
        g = subgradient_corr(h, lag)

        h = project_to_sum(h, target_sum)

    target = n // 2
    ones = np.where(h > 0.5)[0]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h


def subgradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g


def subgradient_matrix(h):
    n = len(h)
    ks = np.arange(2 * n - 1)
    sg = np.zeros((n, 2 * n - 1))
    for i in range(n):
        a = i + ks - (n - 1)
        b = i - ks + (n - 1)
        valid_a = (a >= 0) & (a < n)
        valid_b = (b >= 0) & (b < n)
        sg[i, valid_a] += (1.0 - h[a[valid_a]])
        sg[i, valid_b] -= h[b[valid_b]]
    return sg


def frank_wolfe(h0, n, target_sum, max_iter, verbose=False):
    h = h0.astype(float).copy()
    h = project_to_sum(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n

    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        c5 = float(corr[m_star]) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

        lag = m_star - (n - 1)
        g = subgradient_single(h, lag)

        idx = np.argsort(g)[:int(target_sum)]
        s = np.zeros(n)
        s[idx] = 1.0

        gamma = 2.0 / (t + 2.0)
        gamma = min(gamma, 1.0)
        h = (1.0 - gamma) * h + gamma * s
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)

    return best_h, best_c5


def binary_local_search(h_bin, n, max_iter=200):

    c5 = compute_c5(h)
    target = int(round(np.sum(h > 0.5)))
    if target > n // 2:
        target = n // 2

    corr = np.correlate(h, 1.0 - h, mode="full")
    sg = subgradient_matrix(h)

    for it in range(max_iter):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        A = sg[ones, :]
        B = sg[zeros, :]

        best_val = c5 / (2.0 / n)
        best_i = -1
        best_j = -1

        nj = len(zeros)
        chunk = max(1, nj // 50)
        for j_start in range(0, nj, chunk):

            B_chunk = B[j_start:j_end, :]
            C = corr[None, :] + B_chunk
            vals = np.max(C - A, axis=1)
            flat_idx = np.argmin(vals)
            gi = flat_idx % len(ones)
            gj = flat_idx // len(ones) + j_start

                best_i = gi

            sg = subgradient_matrix(h)
        else:
            h[i_pos] = 1.0

def slsqp_refine(h0, n, target_sum, alpha, maxiter=800):
    def smooth_max_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        return m_val + np.log(np.sum(exp_vals)) / alpha

    def smooth_max_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m_val = np.max(corr)
        shifted = alpha * (corr - m_val)
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            if lag == 0:
                g += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[:n - lag] += w[k] * (1.0 - x[lag:])
                g[lag:] -= w[k] * x[:n - lag]
            else:
                kk = -lag
                g[:n - kk] += w[k] * (1.0 - x[kk:])
                g[kk:] -= w[k] * x[:n - kk]
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(smooth_max_obj, h0.copy(), method='SLSQP',
                       jac=smooth_max_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

        g = subgradient_single(h_start, lag)
        idx = np.argsort(g)[:int(ts_cur)]
        s = np.zeros(n_cur)
        s[idx] = 1.0
        h_alt = 0.3 * h_start + 0.7 * s
        h_alt = project_to_sum(h_alt, ts_cur)
        h_fw2, c5_fw2 = frank_wolfe(h_alt, n_cur, ts_cur, 30000)
        if c5_fw2 < best_c5:
            best_c5 = c5_fw2
            best_h = h_fw2.copy()

    # Phase 2: Binary local search
    if time_left() > 100:
        h_bin = (best_h > 0.4).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != n_cur // 2:
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if n_ones > n_cur // 2:
                for _ in range(n_ones - n_cur // 2):
                    i = ones[rng.randint(len(ones))]
                    h_bin[i] = 0
                    ones = np.where(h_bin == 1)[0]
            elif n_ones < n_cur // 2:
                for _ in range(n_cur // 2 - n_ones):
                    j = zeros[rng.randint(len(zeros))]
                    h_bin[j] = 1
                    zeros = np.where(h_bin == 0)[0]
        h_bls, c5_bls = binary_local_search(h_bin, n_cur, max_iter=150)
        if c5_bls < best_c5:
            best_c5 = c5_bls
            best_h = h_bls.copy()

    # Phase 3: SLSQP refinement from best
    if time_left() > 80:
        h_smooth = best_h.copy()
        for alpha in [200, 1000, 5000]:
            if time_left() < 30:
                break
            h_opt = slsqp_refine(h_smooth, n_cur, ts_cur, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_smooth = h_opt.copy()

    # Phase 4: Try different starting points (structured perturbations)
    n_starts = 5
    for si in range(n_starts):
        if time_left() < 60:
            break
        # Shift
        shift = rng.randint(1, n_cur // 3)
        h_shift = np.roll(best_h, shift)
        h_shift = project_to_sum(h_shift, ts_cur)
        h_fw_s, c5_fw_s = frank_wolfe(h_shift, n_cur, ts_cur, 20000)
        if c5_fw_s < best_c5:
            best_c5 = c5_fw_s
            best_h = h_fw_s.copy()

        # Bit flip perturbation on best binary
        h_bin2 = (best_h > 0.4).astype(float)
        n_flip = max(1, n_cur // 20)
        ones = np.where(h_bin2 == 1)[0]
        zeros = np.where(h_bin2 == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            for _ in range(n_flip):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin2[i] = 0
                h_bin2[j] = 1
                ones = np.where(h_bin2 == 1)[0]
                zeros = np.where(h_bin2 == 0)[0]
            h_bls2, c5_bls2 = binary_local_search(h_bin2, n_cur, max_iter=80)
            if c5_bls2 < best_c5:
                best_c5 = c5_bls2
                best_h = h_bls2.copy()

    # Phase 5: Resolution increase
    if time_left() > 120:
        n_high = n_cur * 2
        ts_high = n_high / 2.0
        h_up = interp_to_n(best_h, n_high)
        h_up = project_to_sum(h_up, ts_high)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_high

        if time_left() > 60:
            h_fw_hi, c5_fw_hi = frank_wolfe(h_up, n_high, ts_high, 30000)
            if c5_fw_hi < best_c5:
                best_c5 = c5_fw_hi
                best_h = h_fw_hi.copy()
                best_n = n_high

            if time_left() > 30:
                h_bin_hi = (best_h > 0.4).astype(float)
                n_ones_hi = int(np.sum(h_bin_hi))
                if n_ones_hi != n_high // 2:
                    ones_hi = np.where(h_bin_hi == 1)[0]
                    zeros_hi = np.where(h_bin_hi == 0)[0]
                    if n_ones_hi > n_high // 2:
                        for _ in range(n_ones_hi - n_high // 2):
                            i = ones_hi[rng.randint(len(ones_hi))]
                            h_bin_hi[i] = 0
                            ones_hi = np.where(h_bin_hi == 1)[0]
                    elif n_ones_hi < n_high // 2:
                        for _ in range(n_high // 2 - n_ones_hi):
                            j = zeros_hi[rng.randint(len(zeros_hi))]
                            h_bin_hi[j] = 1
                            zeros_hi = np.where(h_bin_hi == 0)[0]
                h_bls_hi, c5_bls_hi = binary_local_search(h_bin_hi, n_high, max_iter=100)
                if c5_bls_hi < best_c5:
                    best_c5 = c5_bls_hi
                    best_h = h_bls_hi.copy()
                    best_n = n_high

            if time_left() > 30:
                h_opt_hi = slsqp_refine(best_h, n_high, ts_high, 1000, maxiter=400)
                c5_opt_hi = compute_c5(h_opt_hi)
                if c5_opt_hi < best_c5:
                    best_c5 = c5_opt_hi
                    best_h = h_opt_hi.copy()
                    best_n = n_high

    # Phase 6: Final polish
    n_final = best_n
    ts_final = n_final / 2.0

    if time_left() > 40:
        h_fw_final, c5_fw_final = frank_wolfe(best_h, n_final, ts_final, 30000)
        if c5_fw_final < best_c5:
            best_c5 = c5_fw_final
            best_h = h_fw_final.copy()

    if time_left() > 30:
        h_bin_f = (best_h > 0.4).astype(float)
        n_ones_f = int(np.sum(h_bin_f))
        if n_ones_f != n_final // 2:
            ones_f = np.where(h_bin_f == 1)[0]
            zeros_f = np.where(h_bin_f == 0)[0]
            if n_ones_f > n_final // 2:
                for _ in range(n_ones_f - n_final // 2):
                    i = ones_f[rng.randint(len(ones_f))]
                    h_bin_f[i] = 0
                    ones_f = np.where(h_bin_f == 1)[0]
            elif n_ones_f < n_final // 2:
                for _ in range(n_final // 2 - n_ones_f):
                    j = zeros_f[rng.randint(len(zeros_f))]
                    h_bin_f[j] = 1
                    zeros_f = np.where(h_bin_f == 0)[0]
        h_bls_f, c5_bls_f = binary_local_search(h_bin_f, n_final, max_iter=100)
        if c5_bls_f < best_c5:
            best_c5 = c5_bls_f
            best_h = h_bls_f.copy()

    if time_left() > 20:
        h_slsqp_final = slsqp_refine(best_h, n_final, ts_final, 3000, maxiter=500)
        c5_final = compute_c5(h_slsqp_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_slsqp_final.copy()

    # Final projection and evaluation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
