def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_binary_sa(h_init, n, target_ones, t_end, rng, T_init=0.008, T_min=0.001, cool=0.99999):

    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    c5_max = np.max(corr)
    
    best_c5 = c5_max
    best_h = h.copy()

        idx_b_mk = b - k_vals

        m2 = (idx_a_mk >= 0) & (idx_a_mk < n)

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj

def project_box_sum(h, S):
    """Project h onto {0<=x<=1, sum(x)=S}."""
    n = len(h)
    # If sum is already S and h is in [0,1], return h
    if abs(np.sum(h) - S) < 1e-12 and np.all(h >= -1e-12) and np.all(h <= 1+1e-12):
        return np.clip(h, 0, 1)
    
    # Bisection for lambda
    lo, hi = -1.0, 1.0  # lambda range: h - lambda should span [0,1]
    # Actually, lambda can be outside [-1, 1] in principle, but:
    # sum(clip(h-lambda, 0, 1)) is monotonically decreasing in lambda
    # When lambda -> -inf: all clips become 1, sum = n
    # When lambda -> +inf: all clips become 0, sum = 0
    # We want sum = S, so lambda is finite.
    
    # Better bounds:
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment for numerical accuracy
    diff = S - np.sum(h_proj)
    mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
    if np.any(mask) and abs(diff) > 1e-10:
        h_proj[mask] += diff / np.sum(mask)
        h_proj = np.clip(h_proj, 0, 1)

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(h, S):
    n = len(h)
    if abs(np.sum(h) - S) < 1e-12 and np.min(h) >= -1e-12 and np.max(h) <= 1.0 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = S - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def fast_binary_sa(h_init, n, target_ones, t_end, rng, T_init=0.006, T_min=0.0005, cool_rate=0.999995, restart_thresh=15000):
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    c5_max = np.max(corr)
    best_c5 = c5_max
    best_h = h.copy()
    T = T_init
    no_improve = 0
    k_vals = np.arange(-(n - 1), n)
    while time.time() < t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        idx1 = a + k_vals
        idx2 = a - k_vals
        idx3 = b + k_vals
        idx4 = b - k_vals
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        m3 = (idx3 >= 0) & (idx3 < n)
        m4 = (idx4 >= 0) & (idx4 < n)
        delta = np.zeros(2 * n - 1)
        if np.any(m1):
            delta[m1] += h[idx1[m1]]
        if np.any(m2):
            delta[m2] -= (1.0 - h[idx2[m2]])
        if np.any(m3):
            delta[m3] -= h[idx3[m3]]
        if np.any(m4):
            delta[m4] += (1.0 - h[idx4[m4]])
        new_max = np.max(corr + delta)
        delta_max = new_max - c5_max
        if delta_max < 0 or rng.uniform() < np.exp(delta_max / T):
            h[a] = 0.0
            h[b] = 1.0
            corr += delta
            c5_max = new_max
            if c5_max < best_c5 - 1e-12:
                best_c5 = c5_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > restart_thresh:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            c5_max = np.max(corr)
            T = min(T * 3.0, T_init)
            no_improve = 0
        T = max(T * cool_rate, T_min)
    return best_h, best_c5 * dx

def smooth_max_val_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j = np.arange(n)[:, None]
    p = np.arange(2 * n - 1)[None, :]
    a_idx = j - p + (n - 1)
    b_idx = j + p - (n - 1)
    va = (a_idx >= 0) & (a_idx < n)
    vb = (b_idx >= 0) & (b_idx < n)
    a_c = np.clip(a_idx, 0, n - 1)
    b_c = np.clip(b_idx, 0, n - 1)
    term1 = va.astype(np.float64) * (1.0 - h[a_c])
    term2 = vb.astype(np.float64) * (-h[b_c])
    grad = np.sum(w[None, :] * (term1 + term2), axis=1)
    return obj, grad

def projected_gradient_opt(h0, n, target_sum, alpha, t_end, lr_init=0.003, maxiter=10000):
    h = h0.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(maxiter):
        if time.time() > t_end:
            break
        _, grad = smooth_max_val_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.85
        h = h_new
    return best_h, best_c5

def slsqp_refine(h0, n, target_sum, alpha, maxiter=500):
    j_arr = np.arange(n)
    p_arr = np.arange(2 * n - 1)
    def obj_func(x):
        return smooth_max_val_grad(x, alpha)[0]
    def grad_func(x):
        return smooth_max_val_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_ones, rng):
    h_bin = (h > 0.5).astype(np.float64)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_ones:
        return h_bin
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def random_binary(n, target_ones, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=target_ones, replace=False)
    h[idx] = 1.0
    return h

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def iterative_sa(h_init, n, target_ones, t_end, rng, n_restarts=5):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    time_per = (t_end - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        seg_end = time.time() + time_per
        if r > 0:
            h_start = best_h.copy()
            n_flips = rng.randint(max(2, n // 20), max(3, n // 10))
            flips = rng.choice(n, size=n_flips, replace=False)
            for fi in flips:
                h_start[hi if len(hi:=np.where(h_start==0)[0])>0 else fi] = 0.0
            h_start = make_binary(h_start, n, target_ones, rng)
        else:
            h_start = h_init.copy()
        h_res, c5_res = fast_binary_sa(h_start, n, target_ones, seg_end, rng,
                                       T_init=0.005 + 0.003 * rng.rand(),
                                       T_min=0.0003 + 0.0005 * rng.rand(),
                                       cool_rate=0.99999 - 0.00002 * rng.rand(),
                                       restart_thresh=10000 + 10000 * rng.randint(0, 3))
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = random_binary(best_n, best_n // 2, rng)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    target_ones = n_main // 2

    h_bin = make_binary(best_h, n_main, target_ones, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Iterative SA from initial (35% budget)
    phase1_end = t_start + 0.35 * budget_s
    h_sa1, c5_sa1 = iterative_sa(h_bin, n_main, target_ones, phase1_end, rng, n_restarts=4)
    if c5_sa1 < best_c5:
        best_c5 = c5_sa1
        best_h = h_sa1.copy()

    # Phase 2: Multi-start from random (25% budget)
    phase2_end = t_start + 0.60 * budget_s
    num_random = 3
    time_per_random = (phase2_end - time.time()) / max(num_random, 1)
    for rs in range(num_random):
        seg_end = time.time() + time_per_random
        if seg_end > phase2_end:
            seg_end = phase2_end
        h_rand = random_binary(n_main, target_ones, rng)
        h_res, c5_res = fast_binary_sa(h_rand, n_main, target_ones, seg_end, rng,
                                       T_init=0.010, T_min=0.0008, cool_rate=0.99999, restart_thresh=20000)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 3: Long focused SA from best (20% budget)
    phase3_end = t_start + 0.80 * budget_s
    if elapsed() < phase3_end:
        h_bin3 = make_binary(best_h, n_main, target_ones, rng)
        h_sa3, c5_sa3 = fast_binary_sa(h_bin3, n_main, target_ones, phase3_end, rng,
                                       T_init=0.003, T_min=0.0002, cool_rate=0.999997, restart_thresh=12000)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()

    # Phase 4: Continuous refinement (10% budget)
    if remaining() > 50:
        cont_end = t_start + 0.93 * budget_s
        ts = n_main / 2.0
        h_cont = best_h.copy()
        for alpha in [100, 500, 2000]:
            if time.time() > cont_end or remaining() < 15:
                break
            h_opt, c5_opt = projected_gradient_opt(h_cont, n_main, ts, alpha, cont_end, lr_init=0.002)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cont = h_opt.copy()

    # Phase 5: Higher resolution (7% budget)
    if remaining() > 60:
        for n_try in [360, 480]:
            if remaining() < 40:
                break
            ts_try = n_try / 2.0
            h_interp = interpolated_h(best_h, best_n, n_try)
            h_interp = project_to_sum(h_interp, ts_try)
            h_bin_try = make_binary(h_interp, n_try, n_try // 2, rng)
            t_try_end = time.time() + min(remaining() * 0.4, 50)
            h_try, c5_try = fast_binary_sa(h_bin_try, n_try, n_try // 2, t_try_end, rng,
                                           T_init=0.004, T_min=0.001, cool_rate=0.99999, restart_thresh=10000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                c5_interp = compute_c5(h_interp)
                if c5_interp < best_c5:
                    best_c5 = c5_interp
                    best_h = h_interp.copy()
                    best_n = n_try

    # Final SA polish if time remains
    if remaining() > 30 and best_h is not None:
        n_pol = len(best_h) if best_h.dtype == np.float64 else best_n
        h_pol_bin = make_binary(best_h, n_pol, n_pol // 2, rng)
        h_final, c5_final = fast_binary_sa(h_pol_bin, n_pol, n_pol // 2, time.time() + remaining() * 0.8, rng,
                                           T_init=0.002, T_min=0.0002, cool_rate=0.999998, restart_thresh=8000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            best_n = n_pol

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
