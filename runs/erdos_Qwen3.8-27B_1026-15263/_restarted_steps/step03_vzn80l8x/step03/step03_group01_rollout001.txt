import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    # Gradient: d/dh_i of sum_k w_k * corr_k
    # corr[k] = sum_i h[i] * (1-h[i + k - n + 1])

    nm = np.arange(2 * n - 1)[None, :]

    si = nj + nm - n + 1  # index for (1-h[.]) * h[i]: the . is i - (k-(n-1)) = i + si... wait

        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def adam_opt(h0, n, ts, alpha, n_iter=5000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    for i in range(n_iter):
        t += 1
        _, grad = smooth_max_obj_grad(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        h_new = project_to_sum(h_new, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-18:

            obj_prev = obj_new
        else:

            lr = lr * 0.8

def subgrad_step(h, ts, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find all indices that achieve (or nearly achieve) the max
    max_val = np.max(corr)
    threshold = max_val - 1e-10
    active = np.where(corr >= threshold)[0]
    
    # Subgradient: average over active lags
    # For lag k (0-indexed in full), j = k - (n-1)
    # d/dh[m] corr[k] = (1-h[m+j]) - h[m-j] (where valid)
    nj = np.arange(n)[:, None]
    j_active = (active - (n - 1))[None, :]
    fi = nj + j_active
    si = nj - j_active
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    
    # Average subgradient over active lags
    grad = np.sum(vf * (1.0 - h[fc]) - vs * h[sc], axis=1) / len(active)
    
    h_new = project_to_sum(h - lr * grad, ts)
    return h_new

    lr = lr_init
    c5_prev = compute_c5(h)
    stall = 0
    for i in range(n_iter):
        h_new = subgrad_step(h, ts, lr)
        c5_new = compute_c5(h_new)
        if c5_new < c5_prev - 1e-15:

            stall = 0
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.7
            stall += 1
            if stall > 200 or lr < 1e-14:
                break
    return h

        starts.append(project_to_sum(h_b, ts))
    
    # Random with smoothing
    for _ in range(6):

        # Smooth
        kernel = np.ones(5) / 5
        h_s = np.convolve(h_r, kernel, mode='same')
        starts.append(project_to_sum(h_s, ts))

def stochastic_cd(h, n, ts, rng, n_steps=50000, delta_range=(0.01, 0.3)):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.3
    for step in range(n_steps):
        ones_idx = np.where(h > 0.05)[0]
        zeros_idx = np.where(h < 0.95)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-4:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-5, 0.3 * np.exp(-step / 8000.0))
    return h, c5_cur

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def adam_projected(h0, n, ts, alpha, n_iter=5000, lr=0.02):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_cur = lr
    obj_prev, _ = smooth_max_obj_grad(h, alpha)
    patience = 0
    for i in range(n_iter):
        _, grad = smooth_max_obj_grad(h, alpha)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - b1**(i+1))
        v_hat = v / (1 - b2**(i+1))
        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_sum(h_new, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            patience = 0
            if i % 500 == 0 and patience == 0:
                lr_cur = min(lr_cur * 1.05, 0.3)
        else:
            patience += 1
            lr_cur *= 0.9
            if patience > 500 or lr_cur < 1e-14:
                break
    return h

def exact_subgrad_step(h, ts, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-8)[0]
    nj = np.arange(n)[:, None]
    j_act = (active - (n - 1))[None, :]
    fi = nj + j_act
    si = nj - j_act
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = (np.sum(vf * (1.0 - h[fc]), axis=1) - np.sum(vs * h[sc], axis=1)) / len(active)
    return project_to_sum(h - lr * grad, ts)

def subgrad_descent(h0, ts, n_iter=10000, lr_init=0.03):
    h = h0.copy()
    lr = lr_init
    c5_prev = compute_c5(h)
    patience = 0
    for i in range(n_iter):
        h_new = exact_subgrad_step(h, ts, lr)
        c5_new = compute_c5(h_new)
        if c5_new < c5_prev - 1e-14:
            h = h_new
            c5_prev = c5_new
            patience = 0
            lr = min(lr * 1.01, 0.3)
        else:
            lr *= 0.85
            patience += 1
            if patience > 300 or lr < 1e-14:
                break
    return h

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n // 2.0)

def make_starts(n, rng, best_h, best_n):
    ts = n / 2.0
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for f in [1, 2, 3, 4, 5, 7, 8, 10]:
        s = 0.5 + 0.35 * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(s, ts))
    for _ in range(6):
        nf = rng.randint(2, 5)
        freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
        h_mix = np.zeros(n)
        for f in freqs:
            h_mix += rng.uniform(0.05, 0.3) * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
        h_mix += 0.5
        starts.append(project_to_sum(h_mix, ts))
    for nb in [4, 6, 8, 10, 12]:
        blk = n // nb
        h_b = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_b[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h_b, ts))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_o = np.linspace(0, 2, best_n, endpoint=False)
            h_up = np.interp(x, x_o, best_h, period=2.0)
            starts.append(project_to_sum(h_up, ts))
    for _ in range(4):
        h_r = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        h_s = np.convolve(h_r, kernel, mode='same')
        starts.append(project_to_sum(h_s, ts))
    return starts

def stochastic_cd(h, n, ts, rng, n_steps=50000, delta_range=(0.01, 0.3)):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.25
    cool_rate = n_steps / 10000.0
    for step in range(n_steps):
        ones_idx = np.where(h > 0.05)[0]
        zeros_idx = np.where(h < 0.95)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.5
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.5
        if delta < 1e-4:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        if step % 1000 == 0:
            temp = max(1e-5, 0.25 * np.exp(-step / cool_rate))
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load initial
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass
    
    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Adam optimization on initial resolution with alpha continuation
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [20, 80, 300, 1000, 3000, 10000]:
        if time_left() < 40:
            break
        h_a = adam_projected(best_h, n1, ts1, alpha, n_iter=4000, lr=0.03)
        c5_a = compute_c5(h_a)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
    
    # Phase 2: Subgradient on exact max
    if time_left() > 60:
        h_sg = subgrad_descent(best_h, ts1, n_iter=15000, lr_init=0.03)
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 3: SLSQP polish at current resolution
    for alpha in [500, 2000, 8000]:
        if time_left() < 30:
            break
        h_s = slsqp_opt(best_h, n1, ts1, alpha, maxiter=1000)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 4: Upsample to higher resolution
    n_up = n1 * 2  # e.g., 960
    ts_up = n_up / 2.0
    if time_left() > 120:
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
        # Adam at higher resolution
        for alpha in [50, 300, 2000]:
            if time_left() < 60:
                break
            h_au = adam_projected(h_up, n_up, ts_up, alpha, n_iter=3000, lr=0.02)
            c5_au = compute_c5(h_au)
            if c5_au < best_c5:
                best_c5 = c5_au
                best_h = h_au.copy()
                best_n = n_up
            h_up = h_au
        # Subgradient at higher resolution
        if time_left() > 80:
            h_sgu = subgrad_descent(h_up, ts_up, n_iter=10000, lr_init=0.02)
            c5_sgu = compute_c5(h_sgu)
            if c5_sgu < best_c5:
                best_c5 = c5_sgu
                best_h = h_sgu.copy()
                best_n = n_up
    
    # Phase 5: Shake and bake (frequency perturbations + re-optimize)
    n_b = best_n
    ts_b = n_b / 2.0
    for trial in range(40):
        if time_left() < 30:
            break
        amp = 0.12 * (1.0 - trial / 50.0) + 0.02
        nf = max(5, n_b // 20)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        # Quick optimization
        h_opt = adam_projected(h_sh, n_b, ts_b, 500, n_iter=2000, lr=0.02)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_sh = h_opt
        # Subgradient polish
        h_opt2 = subgrad_descent(h_sh, ts_b, n_iter=5000, lr_init=0.02)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
    
    # Phase 6: Diverse restarts
    n_e = best_n
    ts_e = n_e / 2.0
    n_restarts = 20
    for trial in range(n_restarts):
        if time_left() < 40:
            break
        starts = make_starts(n_e, rng, best_h, best_n)
        for h_s in starts[:5]:
            if time_left() < 20:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_o = adam_projected(h_s, n_e, ts_e, 200, n_iter=2000, lr=0.03)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
    
    # Phase 7: Stochastic CD
    if time_left() > 100:
        h_cd, c5_cd = stochastic_cd(best_h, n_e, ts_e, rng, n_steps=60000, delta_range=(0.01, 0.35))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 8: Post-stochastic polish
    if time_left() > 50:
        h_pol = adam_projected(best_h, n_e, ts_e, 5000, n_iter=3000, lr=0.01)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_sg2 = subgrad_descent(best_h, ts_e, n_iter=8000, lr_init=0.015)
        c5_sg2 = compute_c5(h_sg2)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
    
    # Phase 9: Second stochastic round (finer)
    if time_left() > 60:
        h_cd2, c5_cd2 = stochastic_cd(best_h, n_e, ts_e, rng, n_steps=40000, delta_range=(0.005, 0.15))
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Phase 10: Final SLSQP + subgradient
    if time_left() > 40:
        h_f = slsqp_opt(best_h, n_e, ts_e, 5000, maxiter=1500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        h_f2 = subgrad_descent(best_h, ts_e, n_iter=5000, lr_init=0.01)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
    
    # Final projection and validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h <
