    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
        else:
            h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = alpha * (corr - mc)
    ms = np.max(sh)
    ev = np.exp(sh - ms)
    se = np.sum(ev)
    w = ev / se
    obj = mc + (ms + np.log(se)) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        m1 = max(-j, -(n-1))
        m2 = min(n-1-j, n-1)
        if m2 >= m1:
            idx1 = np.arange(m1 + n - 1, m2 + n)
            h1 = np.arange(j + m1, j + m2 + 1)
            grad[j] += np.dot(w[idx1], 1.0 - h[h1])

            h2 = np.arange(j - m3, j - m4 + 1)
            grad[j] -= np.dot(w[idx2], h[h2])
    return obj, grad

def lbfgs_opt(h0, n, ts, alpha, maxiter=500, ftol=1e-14):

        return lse_obj_grad(x, alpha)[0]

        return lse_obj_grad(x, alpha)[1]

                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    return project_feasible(res.x, ts)

    h = h0.copy()
    lr = lr_init
    c5_prev = compute_c5(h)
    stall = 0
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # Subgradient of max at the active lag
        grad = np.zeros(n)
        for j in range(n):
            # dC_lag/dh[j] = (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
            if 0 <= j + lag < n:

            if 0 <= j - lag < n:
                grad[j] -= h[j - lag]
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_prev - 1e-16:
            h = h_new
            c5_prev = c5_new
            stall = 0
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.5
            stall += 1
            if stall > 100 or lr < 1e-14:

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new / 2.0)

def stochastic_cd(h0, n, ts, rng, n_steps=20000, delta_range=(0.01, 0.3)):
    h = h0.copy()
    c5_cur = compute_c5(h)
    temp = 0.2

    best_c5 = c5_cur

        delta = rng.uniform(delta_range[0], delta_range[1])

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    sh = alpha * (corr - mc)
    ms = float(np.max(sh))
    ev = np.exp(sh - ms)
    se = float(np.sum(ev))
    w = ev / se
    obj = mc + (ms + np.log(se)) / alpha
    grad = np.zeros(n)
    for j in range(n):
        m1 = max(-j, -(n - 1))
        m2 = min(n - 1 - j, n - 1)
        if m2 >= m1:
            idx1 = np.arange(m1 + n - 1, m2 + n)
            h1 = np.arange(j + m1, j + m2 + 1)
            grad[j] += np.dot(w[idx1], 1.0 - h[h1])
        m3 = max(j - n + 1, -(n - 1))
        m4 = min(j, n - 1)
        if m4 >= m3:
            idx2 = np.arange(m3 + n - 1, m4 + n)
            h2 = np.arange(j - m3, j - m4 + 1)
            grad[j] -= np.dot(w[idx2], h[h2])
    return obj, grad

def lbfgs_opt(h0, n, ts, alpha, maxiter=400):
    def obj_fn(x):
        return lse_obj_grad(x, alpha)[0]
    def grad_fn(x):
        return lse_obj_grad(x, alpha)[1]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='L-BFGS-B', jac=grad_fn,
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter * 3})
        return project_feasible(res.x, ts)
    except Exception:
        return project_feasible(h0, ts)

def topk_subgrad(h, n, ts, k=3, n_iter=3000, lr_init=0.01):
    h = h.copy()
    lr = lr_init
    c5_prev = compute_c5(h)
    stall = 0
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:k]
        grad = np.zeros(n)
        for ti in top_idx:
            lag = int(ti) - (n - 1)
            for j in range(n):
                if 0 <= j + lag < n:
                    grad[j] += (1.0 - h[j + lag])
                if 0 <= j - lag < n:
                    grad[j] -= h[j - lag]
        grad /= float(k)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_prev - 1e-16:
            h = h_new
            c5_prev = c5_new
            stall = 0
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            stall += 1
            if stall > 150 or lr < 1e-14:
                break
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new / 2.0)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = float(np.mean(h))
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for kk in range(1, upper + 1):
        H[kk] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_feasible(h_new, n // 2.0)

def stochastic_cd(h0, n, ts, rng, n_steps=30000, drange=(0.01, 0.3)):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    temp = 0.15
    for step in range(n_steps):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = rng.uniform(drange[0], drange[1])
        if h[i] < delta:
            delta = h[i] * 0.8
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.8
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        np.clip(h, 0.0, 1.0, out=h)
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            if temp > 1e-9:
                acc = np.exp(-(c5_new - c5_cur) / temp)
                if rng.random() < acc:
                    c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                np.clip(h, 0.0, 1.0, out=h)
                h = project_feasible(h, ts)
        temp = max(1e-9, 0.15 * np.exp(-step / 5000.0))
    return best_h, best_c5

def make_diverse_starts(n, ts, rng, best_h, best_n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    starts = []
    starts.append(project_feasible(rng.uniform(0, 1, n), ts))
    for f in [2, 3, 5, 7, 11]:
        ph = rng.uniform(0, 2 * np.pi)
        s = 0.5 + 0.3 * np.sin(2.0 * np.pi * f * x / 2.0 + ph)
        starts.append(project_feasible(s, ts))
    for nb in [3, 5, 7]:
        blk = n // nb
        h_b = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_b[i:min(i + blk, n)] = 1.0
        starts.append(project_feasible(h_b, ts))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(upsample_h(best_h, n))
    for _ in range(3):
        nf = int(rng.randint(3, 8))
        freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
        h_mix = np.zeros(n)
        for f in freqs:
            h_mix += rng.uniform(0.05, 0.2) * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
        h_mix += 0.5
        starts.append(project_feasible(h_mix, ts))
    return starts

def check_time(t_start, budget_s, margin):
    return (time.time() - t_start) < (budget_s - margin)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in dir() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at current resolution with L-BFGS-B
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [30, 100, 400, 1500]:
        if not check_time(t_start, budget_s, 50):
            break
        h_opt = lbfgs_opt(best_h, n1, ts1, alpha, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n1

    # Phase 2: Increase resolution
    target_ns = [480, 600, 720, 960]
    for n_tr in target_ns:
        if n_tr <= best_n:
            continue
        if not check_time(t_start, budget_s, 120):
            break
        ts_tr = n_tr / 2.0
        h_up = upsample_h(best_h, n_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr

        for alpha in [50, 200, 800, 3000]:
            if not check_time(t_start, budget_s, 60):
                break
            h_opt = lbfgs_opt(h_up, n_tr, ts_tr, alpha, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_up = h_opt

        if check_time(t_start, budget_s, 50):
            h_tk = topk_subgrad(h_up, n_tr, ts_tr, k=5, n_iter=2000, lr_init=0.005)
            c5_tk = compute_c5(h_tk)
            if c5_tk < best_c5:
                best_c5 = c5_tk
                best_h = h_tk.copy()
                best_n = n_tr
                h_up = h_tk

    # Phase 3: Frequency perturbation + re-optimize
    n3 = best_n
    ts3 = n3 / 2.0
    for trial in range(40):
        if not check_time(t_start, budget_s, 40):
            break
        amp = 0.06 * (1.0 - trial / 50.0) + 0.01
        nf = max(3, n3 // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        for alpha in [200, 1000]:
            if not check_time(t_start, budget_s, 25):
                break
            h_ot = lbfgs_opt(h_sh, n3, ts3, alpha, maxiter=300)
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            h_sh = h_ot

    # Phase 4: Diverse restarts
    n4 = best_n
    ts4 = n4 / 2.0
    for trial in range(15):
        if not check_time(t_start, budget_s, 60):
            break
        starts = make_diverse_starts(n4, ts4, rng, best_h, best_n)
        for h_s in starts[:5]:
            if not check_time(t_start, budget_s, 30):
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for alpha in [50, 300]:
                if not check_time(t_start, budget_s, 20):
                    break
                h_o = lbfgs_opt(h_s, n4, ts4, alpha, maxiter=250)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_s = h_o

    # Phase 5: Stochastic coordinate descent (exact objective)
    if check_time(t_start, budget_s, 120):
        n5 = best_n
        ts5 = n5 / 2.0
        h_s, c5_s = stochastic_cd(best_h, n5, ts5, rng, n_steps=40000, drange=(0.01, 0.35))
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        if check_time(t_start, budget_s, 50):
            h_pol = lbfgs_opt(best_h, n5, ts5, 5000, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Top-k subgradient refinement
    if check_time(t_start, budget_s, 60):
        n6 = best_n
        ts6 = n6 / 2.0
        h_tk = topk_subgrad(best_h, n6, ts6, k=5, n_iter=3000, lr_init=0.005)
        c5_tk = compute_c5(h_tk)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()

    # Phase 7: Fine stochastic + polish
    if check_time(t_start, budget_s, 80):
        n7 = best_n
        ts7 = n7 / 2.0
        h_f, c5_f = stochastic_cd(best_h, n7, ts7, rng, n_steps=30000, drange=(0.005, 0.15))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if check_time(t_start, budget_s, 40):
            h_p2 = lbfgs_opt(best_h, n7, ts7, 10000, maxiter=800)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()

    # Phase 8: Final top-k squeeze
    if check_time(t_start, budget_s, 40):
        n8 = best_n
        ts8 = n8 / 2.0
        h_squeezed = topk_subgrad(best_h, n8, ts8, k=3, n_iter=2000, lr_init=0.003)
        c5_sq = compute_c5(h_squeezed)
        if c5_sq < best_c5:
            best_c5 = c5_sq
            best_h = h_squeezed.copy()

    # Final validation and cleanup
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / int(mask.sum())
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
