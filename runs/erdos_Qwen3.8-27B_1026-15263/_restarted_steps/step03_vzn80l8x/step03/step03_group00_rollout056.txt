import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    ev = np.exp(shifted)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-7:
            continue
        lag = j - (n - 1)
        grad += w[j] * dx * corr_grad_single(x, lag)
    return grad

def slsqp_opt(h0, n, target_sum, alpha, maxiter=500):
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h0.copy(), method='SLSQP',
                       jac=lambda x: smooth_max_grad(x, n, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def momentum_gd(h0, n, target_sum, alpha, lr=None, momentum=0.9, max_iters=5000):
    if lr is None:
        lr = 0.5 / alpha
    h = h0.copy()

    best_h = h.copy()

    for it in range(max_iters):
        obj = smooth_max_obj(h, n, alpha)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
        grad = smooth_max_grad(h, n, alpha)
        v = momentum * v - lr * grad

        lr *= (1.0 - it / (3.0 * max_iters))

    return best_h

def thue_morse(n):
    """Generate first n terms of Thue-Morse sequence."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def halton_1d(n, base=2):
    """Generate first n Halton sequence values in [0,1)."""
    result = np.zeros(n)
    for i in range(1, n + 1):
        f = 1.0 / base
        result[i-1] = 0.0
        val = i
        while val > 0:
            result[i-1] += f * (val % base)
            val //= base
            f /= base
    return result
