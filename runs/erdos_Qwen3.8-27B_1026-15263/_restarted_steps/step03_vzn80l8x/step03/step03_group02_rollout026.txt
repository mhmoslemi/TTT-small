import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)[:target_ones]
    h_bin[idx] = 1.0
    return h_bin

def steepest_descent_fast(h, n, t_end, rng, max_passes=5000):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_curr
    stagnant = 0
    n_corr = 2 * n - 1
    
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Build contrib matrix
        p_idx = np.arange(n)[:, None].astype(int)
        k_idx = np.arange(n_corr)[None, :].astype(int)
        s_k = k_idx - (n - 1)
        pos_plus = p_idx + s_k
        pos_minus = p_idx - s_k
        valid_plus = (pos_plus >= 0) & (pos_plus < n)
        valid_minus = (pos_minus >= 0) & (pos_minus < n)
        contrib = np.where(valid_plus, h[np.clip(pos_plus, 0, n-1)], 0.0) + \
                  np.where(valid_minus, h[np.clip(pos_minus, 0, n-1)], 0.0)
        
        best_new_max = np.max(corr)
        best_pair = None
        
        for i in ones:
            ci = contrib[i]
            for j in zeros:
                delta = ci - contrib[j]
                k_plus = n - 1 + (i - j)
                k_minus = n - 1 + (j - i)
                if 0 <= k_plus < n_corr:
                    delta[k_plus] += 2.0
                if 0 <= k_minus < n_corr:
                    delta[k_minus] += 2.0
                new_max = np.max(corr + delta)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pair = (i, j)
        
        if best_pair is None:
            stagnant += 1
            if stagnant > 5:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_curr = np.max(corr) * dx
            continue
        
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = np.max(corr) * dx
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            best_h = h.copy()
            stagnant = 0
    
    return best_h, best_c5

def binary_sa(h_start, n, t_end, rng, T_init=0.05):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = np.max(corr) * dx
    best_h = h_start.copy()
    best_c5 = c5_curr
    h = h_start.copy()
    T = T_init
    no_improve = 0
    min_T = 0.0001
    
    while time.time() < t_end:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_new = np.max(corr) * dx
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < delta / T:

            if c5_new < best_c5 - 1e-15:

                no_improve = 0

            T = min(T * 2.0, T_init)
            no_improve = 0
        T = max(T * 0.999999, min_T)

            km = n - 1 + zeros - i  # (n_zeros,)
            valid_kp = (kp >= 0) & (kp < n_corr)
            valid_km = (km >= 0) & (km < n_corr)
            if np.any(valid_kp):
                np.add.at(diff, (np.where(valid_kp)[0], kp[valid_kp]), 2.0)
            if np.any(valid_km):
                np.add.at(diff, (np.where(valid_km)[0], km[valid_km]), 2.0)
            
            result = diff.max(axis=1)  # (n_zeros,)
            min_j = np.argmin(result)
            if result[min_j] < best_new_max:
                best_new_max = result[min_j]
                best_pair = (i, zeros[min_j])
        
        if best_pair is None or best_new_max >= np.max(corr) - 1e-14:
            stagnant += 1
            if stagnant > 5:
                break
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_curr = np.max(corr) * dx
            continue
        
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = np.max(corr) * dx
        
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            best_h = h.copy()
            stagnant = 0
    
    return best_h, best_c5
```

This should be correct and fast. Let me estimate: per pass, the main cost is:
- Building contrib: O(n * n_corr) = O(240 * 479) ≈ 115K
- Inner loop over n_ones=120, each doing (n_zeros * n_corr) = 120*479 ≈ 57K ops + np.add.at overhead
- Total: 120 * 57K ≈ 7M ops per pass

In practice with NumPy, each inner iteration might take 20-50μs (due to memory access patterns and the np.add.at calls). So per pass: 120 * 35μs ≈ 4.2ms. We can do ~300K passes in 1300s. Way more than enough.

Now let me think about the overall algorithm structure:

1. **Phase 1** (30% of time): Steepest descent from initial_h_values
2. **Phase 2** (20% of time): SA from Phase 1 result
3. **Phase 3** (20% of time): Steepest descent from Phase 2 result
4. **Phase 4** (25% of time): Try different n values (300, 360, 420, 480)
5. **Phase 5** (5% of time): Final polish

For Phase 4, when trying a new n:
- Interpolate the current best to the new n
- Binary-ize
- Run steepest descent + SA
- Compare

Let me also add a "random restart" component: occasionally, create a completely random binary sequence and optimize it. If it beats the current best, use it.

Actually, random sequences have C₅ ≈ 0.5, so they're much worse than our current 0.38. They won't help unless the optimization is very good. Let me skip this.

Instead, let me try "perturbed restarts": take the current best, make L random swaps (L=5, 10, 20), then optimize. This explores the neighborhood of the current best.

Let me write the final version:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones):
    h_bin = np.zeros(n, dtype=float)
    idx = np.argsort(-h)[:target_ones]
    h_bin[idx] = 1.0
    return h_bin

def steepest_descent(h, n, t_end, rng, max_passes=50000):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_curr
    stagnant = 0
    n_corr = 2 * n - 1
    
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        p_idx = np.arange(n)[:, None].astype(np.int64)
        k_idx = np.arange(n_corr)[None, :].astype(np.int64)
        s_k = k_idx - (n - 1)
        pos_plus = p_idx + s_k
        pos_minus = p_idx - s_k
        valid_plus = (pos_plus >= 0) & (pos_plus < n)
        valid_minus = (pos_minus >= 0) & (pos_minus < n)
        contrib = np.where(valid_plus, h[np.clip(pos_plus, 0, n-1)], 0.0) + \
                  np.where(valid_minus, h[np.clip(pos_minus, 0, n-1)], 0.0)
        
        ci = contrib[ones]
        cj = contrib[zeros]
        
        best_new_max = np.inf
        best_pair = None
        
        for ii in range(n_ones):
            base = corr + ci[ii]
            diff = base[None, :] - cj
            i = ones[ii]
            kp = (n - 1) + i - zeros
            km = (n - 1) + zeros - i
            vkp = (kp >= 0) & (kp < n_corr)
            vkm = (km >= 0) & (km < n_corr)
            if np.any(vkp):
                idx_kp = np.where(vkp)[0]
                diff[idx_kp, kp[vkp]] += 2.0
            if np.any(vkm):
                idx_km = np.where(vkm)[0]
                diff[idx_km, km[vkm]] += 2.0
            result = diff.max(axis=1)
            min_j = np.argmin(result)
            if result[min_j] < best_new_max:
                best_new_max = result[min_j]
                best_pair = (i, zeros[min_j])
        
        if best_pair is None or best_new_max >= np.max(corr) - 1e-14:
            stagnant += 1
            if stagnant > 10:
                break
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_curr = np.max(corr) * dx
            continue
        
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = np.max(corr) * dx
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            best_h = h.copy()
            stagnant = 0
    
    return best_h, best_c5

def sa_optimize(h_start, n, t_end, rng):
    dx = 2.0 / n
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_curr
    T = 0.03
    no_improve = 0
    min_T = 0.00005
    max_T = 0.05
    
    while time.time() < t_end:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_new = np.max(corr) * dx
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < delta / T:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, max_T)
            no_improve = 0
        T = max(T * 0.999998, min_T)
    
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - t_start)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 240
        h_init = np.zeros(best_n)
        pos = rng.choice(best_n, best_n // 2, replace=False)
        h_init[pos] = 1.0
        best_h = h_init
        best_c5 = compute_c5(best_h)
    
    n_main = best_n
    target_ones = n_main // 2
    
    # Phase 1: Steepest descent (30% of time)
    h_bin = make_binary(best_h, n_main, target_ones)
    t1_end = t_start + 0.25 * budget_s
    h_opt, c5_opt = steepest_descent(h_bin, n_main, t1_end, rng, max_passes=100000)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
    
    # Phase 2: SA (20% of time)
    h_sa = make_binary(best_h, n_main, target_ones)
    t2_end = t_start + 0.45 * budget_s
    h_sa, c5_sa = sa_optimize(h_sa, n_main, t2_end, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    
    # Phase 3: Steepest descent again (15% of time)
    h_bin = make_binary(best_h, n_main, target_ones)
    t3_end = t_start + 0.60 * budget_s
    h_opt, c5_opt
