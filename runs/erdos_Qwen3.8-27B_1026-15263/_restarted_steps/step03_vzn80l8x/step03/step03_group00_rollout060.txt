import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def binary_push(h, target_sum, threshold=0.5):
    h_bin = (h >= threshold).astype(float)
    h_bin = project_to_sum(h_bin, target_sum)
    return h_bin

def sla_setup(h0, n, target_sum, alpha, maxiter=500):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def lbfgs_smooth_max(h0, n, target_sum, alpha, maxiter=200):
    from scipy.optimize import minimize as sp_min
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]

        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:

def projected_subgradient(h0, target_sum, steps=5000, lr_init=0.01):

    c5 = compute_c5(h)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def sa_block_moves(h, target_sum, rng, steps=3000, T_init=0.05, T_min=0.001):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    T = T_init
    cooling = (T_min / T_init) ** (1.0 / steps)
    for it in range(steps):
        T *= cooling
        # Choose a move type
        move_type = rng.randint(4)
        h_try = h.copy()
        if move_type == 0:
            # Block swap
            block_len = rng.randint(max(2, n//50), max(3, n//10))
            i = rng.randint(0, n - block_len)
            j = rng.randint(0, n - block_len)
            if j < i:
                i, j = j, i
            if j - i < block_len:
                continue
            h_try[i:i+block_len], h_try[j:j+block_len] = h_try[j:j+block_len].copy(), h_try[i:i+block_len].copy()
        elif move_type == 1:
            # Block flip
            block_len = rng.randint(max(2, n//30), max(3, n//5))
            i = rng.randint(0, n - block_len)
            h_try[i:i+block_len] = 1.0 - h_try[i:i+block_len]
        elif move_type == 2:
            # Single point swap
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            h_try[i], h_try[j] = h_try[j], h_try[i]
        else:

            block_len = rng.randint(max(3, n//20), max(5, n//8))
            i = rng.randint(0, n - block_len)
            noise = rng.normal(0, 0.1, block_len)
            h_try[i:i+block_len] = h_try[i:i+block_len] + noise
        
        h_try = np.clip(h_try, 0, 1)
        h_try = project_to_sum(h_try, target_sum)
        c5_try = compute_c5(h_try)
        delta = c5_try - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            c5 = c5_try
    return h, c5

    grad = np.zeros(n)
    for k_idx, wi in zip(lags, w):
        lag = k_idx - (n - 1)
        grad += wi * corr_gradient(h, lag)
    return obj, grad

Let me also think about one more idea: **exploiting the fact that the optimum likely has a specific symmetry or periodicity**.

For the problem on [0,2] with the "full" correlation (which effectively treats the function as zero outside [0,2]), the optimal solution might not be periodic. But it might have some symmetry like h(x) = 1 - h(x+1) (anti-periodicity with period 1).

If h(x+1) = 1 - h(x), then:
- At lag k=1: ∫h(x)(1-h(x+1))dx = ∫h(x)·h(x)dx = ∫h² dx ≥ 1/2
- This is already ≥ 0.5, which is worse than our target of 0.38.

So anti-periodicity is NOT good. The optimal h should be more "correlated" with its shifts.

What about h(x+0.5) = 1 - h(x)? (anti-periodicity with period 0.5)
- At lag k=0.5: ∫h(x)²dx ≥ 0.5. Same problem.

So the optimal h should NOT have anti-periodicity. It should be "smooth" in some sense, or have a complex structure.

Actually, thinking about it more, the optimal h should be close to binary (to minimize the k=0 term) but with the 1s arranged so that for every shift k, the set {x: h(x)=1} overlaps significantly with {x: h(x+k)=1}.

This is like a "difference set" problem. In the discrete case, a (v, k, λ) difference set has the property that every non-zero element of the group is the difference of exactly λ pairs from the set. This would give constant correlation for all non-zero lags.

In the continuous case, we want a set A of measure 1 in [0,2] such that |A ∩ (A-k)| is as constant as possible (and as large as possible) for all k.

The maximum possible |A ∩ (A-k)| for a set of measure 1 in [0,2] is... well, by the Cauchy-Schwarz inequality:
|A ∩ (A-k)| ≤ √(|A| · |A-k|) = √(1 · 1) = 1 (but this is only achieved when A = A-k a.e., which is impossible for k ≠ 0 in a bounded domain).

More realistically, for a "random" set of measure 1, |A ∩ (A-k)| ≈ 1/2 (for small k) and decreases as |k| increases.

For a carefully constructed set, we can do better. The ideal would be:
|A ∩ (A-k)| ≈ 1 - |k|/2 for |k| ≤ 2 (which is what a single interval gives, but it's not constant).

Hmm, I think the optimal structure is something like a "fractal" or "multi-scale" pattern where the 1s are distributed at multiple scales to maintain good overlap at all lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_full(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def smooth_max_k_lags(h, lag_indices, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    vals = corr[lag_indices]
    m_val = np.max(vals)
    shifted = alpha * (vals - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for k_idx, wi in zip(lag_indices, w):
        lag = k_idx - (n - 1)
        grad += wi * corr_gradient(h, lag)
    return obj, grad

def slsqp_full(h0, n, target_sum, alpha, maxiter=400):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_max_full(x, alpha)[0]
    def g(x):
        return smooth_max_full(x, alpha)[1]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def slsqp_k_lags(h0, n, target_sum, lag_indices, alpha, maxiter=600):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_max_k_lags(x, lag_indices, alpha)[0]
    def g(x):
        return smooth_max_k_lags(x, lag_indices, alpha)[1]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 60:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def sa_block_moves(h, target_sum, rng, steps=5000, T_init=0.08, T_min=0.002):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(steps, 1))
    best_h = h.copy()
    best_c5 = c5
    for it in range(steps):
        T *= cooling
        move_type = rng.randint(5)
        h_try = h.copy()
        if move_type == 0:
            block_len = rng.randint(max(2, n // 60), max(4, n // 8))
            i = rng.randint(0, n - block_len)
            j = rng.randint(0, n - block_len)
            if j < i:
                i, j = j, i
            if j - i < block_len:
                continue
            temp = h_try[i:i + block_len].copy()
            h_try[i:i + block_len] = h_try[j:j + block_len]
            h_try[j:j + block_len] = temp
        elif move_type == 1:
            block_len = rng.randint(max(2, n // 40), max(4, n // 4))
            i = rng.randint(0, n - block_len)
            h_try[i:i + block_len] = 1.0 - h_try[i:i + block_len]
        elif move_type == 2:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            h_try[i], h_try[j] = h_try[j], h_try[i]
        elif move_type == 3:
            block_len = rng.randint(max(3, n // 25), max(5, n // 6))
            i = rng.randint(0, n - block_len)
            noise = rng.normal(0, 0.12, block_len)
            h_try[i:i + block_len] += noise
        else:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            scan = min(n, max(0, min(i, j)))
            scan_end = min(n, max(i, j) + 1)
            h_try[scan:scan_end] = 1.0 - h_try[scan:scan_end]

        h_try = np.clip(h_try, 0, 1)
        h_try = project_to_sum(h_try, target_sum)
        c5_try = compute_c5(h_try)
        delta = c5_try - c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            h = h_try
            c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def top_k_lags(corr, k):
    total = len(corr)
    k_actual = min(k, total)
    top_indices = np.argsort(corr)[-k_actual:]
    return top_indices

def binary_push(h, target_sum, threshold=0.5):
    h_bin = (h >= threshold).astype(float)
    h_bin = project_to_sum(h_bin, target_sum)
    return h_bin

def multi_binary_push(h, target_sum, rng):
    n = len(h)
    candidates = []
    for thresh in [0.4, 0.45, 0.5, 0.55, 0.6]:
        h_bin = (h >= thresh).astype(float)
        h_bin = project_to_sum(h_bin, target_sum)
        candidates.append(h_bin)
    for _ in range(4):
        noise = rng.uniform(0.3, 0.7, n)
        h_rand = (h > noise).astype(float)
        h_rand = project_to_sum(h_rand, target_sum)
        candidates.append(h_rand)
    return candidates
