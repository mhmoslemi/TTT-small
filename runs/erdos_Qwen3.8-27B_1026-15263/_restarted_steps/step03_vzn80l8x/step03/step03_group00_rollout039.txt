    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP at original resolution (n=540)
    for alpha in [200, 1000, 5000, 30000]:
        if time_left() < 80:
            break
        h_new = slsqp_opt(best_h, best_n / 2.0, alpha, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent at original n
    if time_left() > 80:
        h_cd, c5_cd = coord_descent(best_h, best_n / 2.0, max_iters=8000, step_frac=0.1)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 3: Upscale to higher resolutions
    n_targets = [1080, 1620, 2160]
    for n_target in n_targets:
        if time_left() < 150:
            break
        ts = n_target / 2.0
        h_up = project_to_sum(interp_to_n(best_h, n_target), ts)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_h, best_c5, best_n = h_up.copy(), c5_up, n_target

        h_opt, c5_opt = optimize_at_n(h_up, n_target, ts, t_start, budget_s, min_time_left=80)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_target

        if time_left() > 60:
            h_cd, c5_cd = coord_descent(h_opt, ts, max_iters=4000, step_frac=0.08)
            if c5_cd < best_c5:
                best_h, best_c5, best_n = h_cd, c5_cd, n_target
            h_opt = h_cd

        # Second round of optimization
        if time_left() > 80:
            h_opt2, c5_opt2 = optimize_at_n(h_opt, n_target, ts, t_start, budget_s, min_time_left=60)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2, c5_opt2, n_target

    # Phase 4: Random restarts with Fourier perturbations
    n_cur = best_n
    ts_cur = n_cur / 2.0
    n_restarts = 12
    for r in range(n_restarts):
        if time_left() < 70:
            break
        intensity = 0.04 + 0.06 * (1.0 - r / float(n_restarts))
        h_pert = project_to_sum(fourier_perturb(best_h, rng, intensity), ts_cur)
        h_pert = push_binary(h_pert, ts_cur, strength=0.3)
        h_opt, c5_opt = optimize_at_n(h_pert, n_cur, ts_cur, t_start, budget_s, min_time_left=50)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Stochastic coordinate descent for escape from local optima
    if time_left() > 80:
        h_stoch, c5_stoch = stochastic_cd(best_h, ts_cur, rng, max_iters=15000, step_frac=0.06)
        if c5_stoch < best_c5:
            best_h, best_c5 = h_stoch, c5_stoch

    # Phase 6: High-alpha polishing
    for alpha in [50000, 200000]:
        if time_left() < 40:
            break
        h_pol = project_to_sum(lbfgsb_opt(best_h, ts_cur, alpha, mu=50000, maxiter=800), ts_cur)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 7: Final coordinate descent
    if time_left() > 40:
        h_final, c5_final = coord_descent(best_h, ts_cur, max_iters=12000, step_frac=0.05)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    # Final verification
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
