import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):

def active_smooth_max_obj_grad(h, alpha, top_k=15):

    m_val = np.max(corr)
    idx = np.argsort(corr)[-top_k:]
    corr_active = corr[idx]
    shifted = alpha * (corr_active - np.max(corr_active))

    h_prev = h0.copy()
    t_mom = 1.0
    lr = lr_init
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    stall = 0
    obj_fn = active_smooth_max_obj_grad if use_active else smooth_max_obj_grad
    for i in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / (t_new + 1.0)
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = obj_fn(y, alpha, top_k) if use_active else (None, smooth_max_obj_grad(y, alpha)[1])

def slsqp_opt(h0, n, ts, alpha, maxiter=600):
    def f_and_g(x):
        return smooth_max_obj_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        obj0, g0 = smooth_max_obj_grad(h0, alpha)
        r = minimize(lambda x: smooth_max_obj_grad(x, alpha)[0], h0.copy(), method='SLSQP',
                     jac=lambda x: smooth_max_obj_grad(x, alpha)[1],
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def fourier_perturb(h, rng, n_max_harm, amp, n_harmonics):

def upsample_periodic(h_old, n_new):

    h_up = np.clip(h_up, 0.0, 1.0)

        h_fin, c5_fin = nesterov_proj_grad(best_h, best_n, best_n / 2.0, 50000, n_iter=5000, lr_init=0.003)
        c5_fin_real = compute_c5(h_fin)
        if c5_fin_real < best_c5:
            best_c5 = c5_fin_real
            best_h = h_fin.copy()

    # Phase 9: Tiny stochastic touches
    if time_left() > 30:
        h_t, c5_t = stochastic_cd_structured(best_h, best_n, best_n / 2.0, rng, min(30000, int(time_left() * 4)), (0.003, 0.1), best_c5)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 5e-11) & (best_h < 1.0 - 5e-11)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
