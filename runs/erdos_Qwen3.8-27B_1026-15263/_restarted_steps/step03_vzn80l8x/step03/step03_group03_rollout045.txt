import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    lo = j
    hi = min(2 * n - 2, j + nm1)
    if lo <= hi:
        idx = j + nm1 - np.arange(lo, hi + 1)
        B[lo:hi + 1] += (1.0 - h[idx])
    lo2 = max(0, nm1 - j)
    hi2 = min(2 * n - 2, nm1 - j + nm1)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1) - nm1 + j
        B[lo2:hi2 + 1] += (-h[idx2])
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[max(0, best_idx - 1)]
    hi_r = t_grid[min(len(t_grid) - 1, best_idx + 1)]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand, structured=True):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    best_max_before = np.max(corr)
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = []
        if structured:
            k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                     (j + n // 3) % n, (j + 2 * n // 3) % n,
                     (j + n // 4) % n, (j + n // 6) % n]
        n_rand = max(0, n_cand - len(k_set))
        for _ in range(n_rand):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0:
                h[j] = 0.0
            elif h[j] > 1:
                h[j] = 1.0
            if h[best_k] < 0:
                h[best_k] = 0.0
            elif h[best_k] > 1:
                h[best_k] = 1.0
            improved = True
    return h, corr, improved, np.max(corr) < best_max_before

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = m_val + np.log(np.sum(np.exp(shifted))) / alpha
    return obj, corr, m_val

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h_flat):
        return smooth_max_obj(h_flat, alpha)[0]
    def g(h_flat):
        return smooth_max_grad(h_flat, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': maxiter * 3})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def gradient_descent_step(h, alpha, lr, momentum, vel, target_sum):
    grad = smooth_max_grad(h, alpha)
    vel = momentum * vel - lr * grad
    h_new = h + vel
    h_new = project_to_sum(np.clip(h_new, 0, 1), target_sum)
    return h_new, vel
