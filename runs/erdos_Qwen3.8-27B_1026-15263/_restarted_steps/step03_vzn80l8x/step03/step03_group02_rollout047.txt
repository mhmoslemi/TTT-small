    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

class _Obj:
    def __init__(self, alpha):
        self.alpha = alpha
    def f(self, x):
        return smooth_max_and_grad(x, self.alpha)[0]
    def g(self, x):
        return smooth_max_and_grad(x, self.alpha)[1]

class _Con:
    def __init__(self, target_sum, n):
        self.target_sum = target_sum
        self.n = n
    def fun(self, x):
        return np.sum(x) - self.target_sum
    def jac(self, x):
        return np.ones(self.n)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    obj = _Obj(alpha)
    con = _Con(target_sum, n)
    cons = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj.f, h0.copy(), method='SLSQP', jac=obj.g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_sa(h_start, n, target_sum, t_end, rng, T_init=0.02, T_min=0.002,
              restart_thresh=30000, cool_rate=0.999995):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0

    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / T) + np.log(rng.uniform()) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]

        if no_improve > restart_thresh:
            r = rng.random()
            if r < 0.35:
                h = best_h.copy()
                c5_curr = best_c5
                T = min(T * 2.5, 0.03)
            elif r < 0.65:
                h = best_h.copy()
                n_pert = rng.randint(3, 10)
                for _ in range(n_pert):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                c5_curr = compute_c5(h)
                T = 0.018
            else:
                T = min(T * 1.8, 0.028)
            no_improve = 0

        T = max(T * cool_rate, T_min)

    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    phase1_end = t_start + 0.52 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), phase1_end, rng,
                            T_init=0.022, restart_thresh=25000)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        h_bin = h_sa.copy()
    else:
        h_bin = h_sa.copy()

    phase2_end = t_start + 0.65 * budget_s
    alphas = [30, 100, 500, 2000]
    h_current = best_h.copy()
    for alpha in alphas:
        if elapsed() > phase2_end - t_start:
            break
        h_pol = slsqp_optimize(h_current, n_main, ts_main, alpha, maxiter=1000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_current = h_pol.copy()
        else:
            h_current = h_pol.copy()

    h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
    phase3_end = t_start + 0.78 * budget_s
    if remaining() > 60:
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, int(ts_main), phase3_end, rng,
                                  T_init=0.015, restart_thresh=20000,
                                  cool_rate=0.99999)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    phase4_end = t_start + 0.90 * budget_s
    if remaining() > 80:
        n_hi = 480
        ts_hi = n_hi / 2.0
        h_hi = interpolated_h(best_h, best_n, n_hi)
        h_hi = project_to_sum(h_hi, ts_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

        h_hi_bin = make_binary(h_hi, n_hi, int(ts_hi), rng)
        c5_hi_bin = compute_c5(h_hi_bin)
        if c5_hi_bin < best_c5:
            best_c5 = c5_hi_bin
            best_h = h_hi_bin.copy()
            best_n = n_hi

        sa_hi_end = min(phase4_end, t_start + 0.88 * budget_s)
        if time.time() < sa_hi_end:
            h_hi_sa, c5_hi_sa = binary_sa(h_hi_bin, n_hi, int(ts_hi), sa_hi_end, rng,
                                          T_init=0.02, restart_thresh=15000)
            if c5_hi_sa < best_c5:
                best_c5 = c5_hi_sa
                best_h = h_hi_sa.copy()
                best_n = n_hi

    if remaining() > 30:
        alphas_final = [5000, 20000]
        for alpha in alphas_final:
            if remaining() < 15:
                break
            h_fin = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=1200)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    if remaining() > 20:
        h_final_bin = make_binary(best_h, best_n, int(best_n / 2.0), rng)
        h_final_sa, c5_final = binary_sa(h_final_bin, best_n, int(best_n / 2.0),
                                         t_start + budget_s - 5, rng,
                                         T_init=0.012, restart_thresh=10000,
                                         cool_rate=0.99999)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_sa.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
