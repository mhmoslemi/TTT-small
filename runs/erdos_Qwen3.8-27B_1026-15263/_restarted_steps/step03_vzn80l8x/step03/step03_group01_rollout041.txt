def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :] - (n - 1)
    fi = nj - k
    vf = (fi >= 0) & (fi < n)
    fc = np.clip(fi, 0, n - 1)
    si = nj + k
    vs = (si >= 0) & (si < n)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def lbfgsb_opt(h0, n, ts, alpha, maxiter=300, penalty=1e8):
    def f(x):
        obj, _ = smooth_max_obj_grad(x, alpha)
        cv = np.sum(x) - ts
        return obj + penalty * cv * cv
    def g(x):
        _, grad_om = smooth_max_obj_grad(x, alpha)
        cv = np.sum(x) - ts
        return grad_om + 2.0 * penalty * cv
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='L-BFGS-B', jac=g, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0, ts)

def slsqp_opt(h0, n, ts, alpha, maxiter=500):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def con_f(x):
        return np.sum(x) - ts
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except:
        return project_to_sum(h0, ts)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)

    h_new = h - lr * grad
    return project_to_sum(h_new, ts)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :] - (n - 1)
    fi = nj - k
    vf = (fi >= 0) & (fi < n)
    fc = np.clip(fi, 0, n - 1)
    si = nj + k
    vs = (si >= 0) & (si < n)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad
