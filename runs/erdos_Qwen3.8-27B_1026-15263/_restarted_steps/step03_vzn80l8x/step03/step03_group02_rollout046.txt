def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    Gradient descent that focuses on the top-k lags.

    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def obj_func(h):
        return smooth_max_and_grad(h, alpha)[0]
    def grad_func(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def topk_gradient(h, n, top_k=5):
    """Compute gradient of sum of top-k correlation values."""
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        l = idx - (n - 1)
        if l >= 0:
            js = np.arange(l, n)
            grad[js] += (1.0 - h[js - l])
            grad[js - l] -= h[js]
        else:
            js = np.arange(0, n + l)
            grad[js] += (1.0 - h[js - l])
            grad[js - l] -= h[js]
    return grad, corr

def focused_grad_descent(h0, n, target_sum, n_steps=5000, lr_init=0.05, top_k=5):
    """Gradient descent focusing on top-k lags with adaptive learning rate."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        grad, corr = topk_gradient(h, n, top_k)
        c5_curr = np.max(corr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        lr = lr_init * (0.9995 ** step)
        h = h - lr * grad / grad_norm * grad_norm  # = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
    return best_h, best_c5

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin
