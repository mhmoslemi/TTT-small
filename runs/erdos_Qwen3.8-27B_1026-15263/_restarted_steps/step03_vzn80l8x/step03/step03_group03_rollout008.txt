def compute_c5(h):

def project_to_sum(h, target_sum):

def compute_B_correct(h, j, n):

    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: ∂corr[t]/∂h[j] * w[t] summed over t
    # ∂corr[t]/∂h[j] = (1-2*h[t-j]) * 1[0≤t-j<n]

        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < f(h0):
            return project_to_sum(res.x, target_sum)
    except Exception:
        pass
    return project_to_sum(h0, target_sum)
