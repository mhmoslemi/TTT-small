```python
def project_to_sum(h, target_sum):

    k_end = min(2 * n - 2, 2 * nm1 - j)

def solve_1d(A, D, lo, hi):
    """Minimize max(A + D*t) over t in [lo, hi]."""
    if hi - lo < 1e-14:

    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)

        return lo
    if best_idx == len(t_grid) - 1:
```

The objective is f(h) = max_k corr[k] * dx. The subgradient of max is the indicator of the argmax. So a subgradient step would be:
h_new = h - η * e_k* (where k* is the argmax)

But this is too crude. A better approach is the smooth max (log-sum-exp):
f_α(h) = (1/α) * log(sum_k exp(α*corr[k]*dx))

The gradient of this is:
∇f_α(h)[i] = sum_k w_k * ∂corr[k]/∂h[i]

where w_k = exp(α*corr[k]*dx) / sum_j exp(α*corr[j]*dx)

This is essentially a weighted average of the "influence vectors" B_i, weighted by how close each corr[k] is to the maximum.

So a gradient step would be:
h_new = h - η * Σ_k w_k * B_i[k] ... no, that's not right either.

Actually, the gradient of corr[k] with respect to h[i] depends on k and i. The influence vector B_i tells us how all corr values change when h[i] changes. So:

∂f_α/∂h[i] = Σ_k w_k * B_i[k]

where B_i[k] is the k-th component of the influence vector for position i.

This is exactly what the previous program computes in `smooth_max_and_grad`. The SLSQP phase uses this.

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def beatty_init(n, alpha, rng):
    """Create a Beatty-sequence-like initialization."""
    h = np.zeros(n)
    # Beatty sequence: position i is 1 if {i * alpha} < 0.5
    # Adjust alpha so that approximately n/2 positions are 1
    # For a truly balanced sequence, use the golden ratio conjugate
    x = np.arange(n)
    vals = np.mod(x * alpha, 1.0)
    h[vals < 0.5] = 1.0
    h[vals >= 0.5] = 0.0
    # Adjust to get exactly n/2 ones
    current_sum = np.sum(h)
    target = n / 2.0
    if current_sum < target:
        need = int(target - current_sum)
        # Add 1 to the positions closest to threshold
        sorted_idx = np.sort(vals[vals >= 0.5])[:need]
        for idx in np.where(vals >= 0.5)[0][np.argsort(vals[vals >= 0.5])[:need]]:
            h[idx] = 1.0
    elif current_sum > target:
        excess = int(current_sum - target)
        for idx in np.where(vals < 0.5)[0][np.argsort(-vals[vals < 0.5])[:excess]]:
            h[idx] = 0.0
    # Add some noise to break symmetry
    h += rng.uniform(-0.1, 0.1, n)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, target)
    return h

def smooth_max_obj_grad(h, alpha):

    a = min(alpha, 500.0 / max(m_val, 1e-10))
    exp_vals = np.exp(a * (corr - m_val))

    dx = 2.0 / n
    obj = (m_val + np.log(np.sum(exp_vals)) / a) * dx

    # grad[i] = dx * sum_k w[k] * B_i[k]
    grad = np.zeros(n)

    # Actually, let's compute it the "naive" way: for each k, weight w[k], 
    # and B_i[k] is the k-th component of the influence vector for position i
    # B_i[k] = (1-h[i-k+nm1])*valid1 - h[k+nm1-i]*valid2
    
    # Alternative: use the fact that B_i[k] depends on i and k in a structured way
    # Let's just loop over i (n iterations, each O(n) for the B vector)
    # Total: O(n²) which is fine for n=240
    nm1 = n - 1
    for i in range(n):

        # B_i[k] for k = 0 to 2n-2
        # Term 1: (1-h[i-k+nm1]) for k where 0 <= i-k+nm1 < n
        #   → i+nm1-n < k <= i+nm1
        #   → k in [i, i+nm1] (clipped to [0, 2n-2])
```

```python
def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad
```
