import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_fast(h):
    """Optimized for binary or near-binary h"""
    n = len(h)
    dx = 2.0 / n
    # For efficiency, use the fact that 1-h is just the complement
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = m_val + np.log(total) / alpha
    # Gradient
    # c[k] = sum_i h[i+k*(+1 or -1)] * (1-h[i])
    # d(c[k])/d(h[j]) = (1-h[j]) if j appears in h-index, -h[j] if j appears in (1-h) index
    # For full correlate, lag structure:
    # c[m] = sum_i a[i+m-(n-1)] * b[i] for m in [0, 2n-2]
    # where a = h, b = 1-h
    # So c[m] = sum_i h[i + m - (n-1)] * (1-h[i])
    # valid when 0 <= i+m-(n-1) < n and 0 <= i < n
    
    j_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2 * n - 1)[None, :]  # (1, 2n-1)
    # c[m] involves h[j] where j = i + m - (n-1) for some valid i
    # i = j - m + (n-1), need 0 <= i < n
    # (1-h)[i] where i = j - m + (n-1)
    # h[j] appears in c[m] when 0 <= j - m + (n-1) < n, i.e., m-(n-1) <= j <= m
    # The contribution to c[m] from h[j] is: (1 - h[j - m + (n-1)])
    # And (1-h)[j] = 1-h[j] appears in c[m] when h indices include j:
    #   i + m - (n-1) = j => i = j - m + (n-1)
    #   contribution is -(h[j]) * (since derivative of (1-h[i]) w.r.t. h[j] is -1 if i=j)
    
    # Let me redo this more carefully.
    # c[m] = sum_i h[i + m - (n-1)] * (1 - h[i])  [product rule for gradient]
    # Let p = i + m - (n-1), q = i. Then p - q = m - (n-1)
    # dc[m]/dh[j] = sum over (p,q) with p-q = m-(n-1) of [1{j=p}*(1-h[q]) + 1{j=q}*(-h[p])]
    #             = (1-h[q]) if j=p=q+m-(n-1) valid, plus -h[p] if j=q=j valid and p=j+m-(n-1) valid
    
    # For efficiency, let's compute:
    # grad[j] = sum_m w[m] * dc[m]/dh[j]
    
    # dc[m]/dh[j]:
    # Term 1: j appears as h-index: i = j - m + (n-1), need 0<=i<n
    #   contribution: w[m] * (1-h[i]) where i = j-m+(n-1)
    # Term 2: j appears in (1-h) index: i = j, h-index is j+m-(n-1), need 0<=j+m-(n-1)<n
    #   contribution: w[m] * (-h[j+m-(n-1)])
    
    # This is O(n * 2n) which is O(n²) - fine for n up to ~600
    
    # Precompute for speed
    # For each j, the valid m range is m in [max(0, j-(n-1)), min(j, 2n-2)]... 
    # Actually m-(n-1) <= j <= m => j-(n-1) <= m <= j, and 0<=m<=2n-2
    # And 0 <= j-m+(n-1) < n => j-(n-1) <= m <= j+n-1
    
    # Let me just do it with the matrix approach for clarity
    # But that's O(n * 2n) memory... for n=480 that's 480*959 ≈ 460K floats, fine.
    
    a_idx = j_idx - (m_idx - (n - 1))  # i = j - m + (n-1), shape (n, 2n-1)
    b_idx = j_idx + (m_idx - (n - 1))  # p = j + m - (n-1), shape (n, 2n-1)
    valid_a = (a_idx >= 0) & (a_idx < n)
    valid_b = (b_idx >= 0) & (b_idx < n)
    a_c = np.clip(a_idx, 0, n-1)
    b_c = np.clip(b_idx, 0, n-1)
    
    # grad[j] = sum_m w[m] * [(1-h[a_c[j,m]]) * valid_a[j,m] + (-h[b_c[j,m]]) * valid_b[j,m]]
    #         = sum_m w[m] * (1-h[a_c]) - sum_m w[m] * h[b_c]
    # where we only include valid terms
    
    term1 = valid_a * (1.0 - h[a_c]) * w[None, :]  # (n, 2n-1)
    term2 = valid_b * h[b_c] * w[None, :]
    grad = np.sum(term1 - term2, axis=1)
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa_improved(h_start, n, target_sum, t_end, rng, T_init=0.015, T_min=0.002, decay=0.99999):
    h = h_start.copy().astype(float)
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = T_init
    no_improve = 0
    count = 0
    while time.time() < t_end:
        count += 1
        move_type = rng.randint(4)
        if move_type == 0:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif move_type == 1:
            # 2-swap: swap i<->j and k<->l
            idx = rng.choice(n, size=4, replace=False)
            i, j, k, l = idx
            if h[i] == h[j] and h[k] == h[l]:
                continue
            h[i], h[j], h[k], h[l] = h[j], h[i], h[l], h[k]
        elif move_type == 2:
            # Block flip: reverse a block of length 2-5
            blen = rng.randint(2, 6)
            start = rng.randint(0, n - blen)
            h[start:start+blen] = h[start:start+blen][::-1]
        else:
            # Jump: move a 1 to a random 0 position (or vice versa)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            if rng.rand() < 0.5:
                src = ones[rng.randint(len(ones))]
                dst = zeros[rng.randint(len(zeros))]
            else:
                src = zeros[rng.randint(len(zeros))]
                dst = ones[rng.randint(len(ones))]
            h[src], h[dst] = h[dst], h[src]
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta > 0 and (delta / max(T, 1e-10)) + np.log(max(rng.uniform(), 1e-15)) < 0):
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Undo
            if move_type == 0:
                h[i], h[j] = h[j], h[i]
            elif move_type == 1:
                h[i], h[j], h[k], h[l] = h[j], h[i], h[l], h[k]
            elif move_type == 2:
                h[start:start+blen] = h[start:start+blen][::-1]
            else:
                h[src], h[dst] = h[dst], h[src]
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, 0.025)
            no_improve = 0

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = m_val + np.log(total) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    a_off = m_idx - (n - 1)
    a_idx = j_idx - a_off
    b_idx = j_idx + a_off
    valid_a = (a_idx >= 0) & (a_idx < n)
    valid_b = (b_idx >= 0) & (b_idx < n)
    a_c = np.clip(a_idx, 0, n - 1)
    b_c = np.clip(b_idx, 0, n - 1)
    term1 = valid_a * (1.0 - h[a_c]) * w[None, :]
    term2 = valid_b * h[b_c] * w[None, :]
    grad = np.sum(term1 - term2, axis=1)
    return obj, grad

        return smooth_max_obj_grad(x, alpha)[1]

             'jac': lambda x: np.ones(n)}]

            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
