            hi = mid

def compute_corr(h):

def project_feasible(h, target_sum):

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_obj_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

    new_h = project_feasible(new_h, target_sum)
    new_obj = smooth_max_obj_and_grad(new_h, alpha)[0]

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa_block(h_start, n, target_sum, t_end, rng, block_max=8):
    """SA with mixed single swaps and block moves"""
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    no_improve = 0
    min_T = 0.002
    n_lags = 2 * n - 1
    
    while time.time() < t_end:
        move_type = rng.random()
        if move_type < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            # Block move: pick a block and either reverse it or shift it
            block_len = rng.randint(2, block_max + 1)
            if block_len >= n:
                block_len = n // 2
            start = rng.randint(0, n - block_len + 1)
            block = h[start:start+block_len].copy()
            
            move_sub = rng.random()
            if move_sub < 0.5:
                # Reverse the block
                h[start:start+block_len] = block[::-1]
            else:
                # Shift the block by 1 (circular within the block region)
                h[start:start+block_len] = np.roll(block, 1)
            
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[start:start+block_len] = block
            
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.025)
            no_improve = 0
        
        T = max(T * 0.999995, min_T)
    
    return best_h, best_c5

def iterated_sa(h_start, n, target_sum, t_end, rng):
    """Iterated local search: SA + perturbation + SA"""
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    
    perturbation_size = 5
    
    while time.time() < t_end:
        # Run short SA
        sa_t_end = min(time.time() + 30, t_end - 60)
        h_sa, c5_sa = binary_sa_block(best_h, n, target_sum, sa_t_end, rng)
        
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        
        # Perturb
        h_pert = best_h.copy()
        for _ in range(perturbation_size):
            i = rng.randint(n)
            j = rng.randint(n)
            if h_pert[i] != h_pert[j]:
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        
        # Increase perturbation over time
        perturbation_size = min(perturbation_size + 1, 15)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):

    return np.interp(x_dst, x_src, h_src, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    """Compute smooth max objective and its correct gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Correct gradient computation
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    
    # d(corr[k])/d(h[i]) = (1-h[second_idx])*valid_s - h[first_idx]*valid_f
    cs = valid_s * (1.0 - h[sc]) * w[None, :]
    cf = valid_f * h[fc] * w[None, :]
    grad = np.sum(cs - cf, axis=1)
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

def pgd_optimize(h0, n, target_sum, alpha, t_end, lr_init=0.005):
    h = h0.copy()
    obj = smooth_max_obj_grad(h, alpha)[0]
    best_h = h.copy()
    best_obj = obj
    lr = lr_init
    no_improve = 0
    
    while time.time() < t_end:
        obj_new_raw, grad = smooth_max_obj_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        obj_new = smooth_max_obj_grad(h_new, alpha)[0]
        
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            no_improve = 0
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.7
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng, block_max=6):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.012
    no_improve = 0
    min_T = 0.003
    
    while time.time() < t_end:
        r = rng.random()
        if r < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            blen = rng.randint(2, block_max + 1)
            if blen >= n - 1:
                blen = max(2, n // 4)
            start = rng.randint(0, n - blen)
            block = h[start:start+blen].copy()
            if rng.random() < 0.5:
                h[start:start+blen] = block[::-1]
            else:
                h[start:start+blen] = np.roll(block, rng.randint(-2, 3))
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[start:start+blen] = block
        
        if no_improve > 25000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.02)
            no_improve = 0
        T = max(T * 0.99999, min_T)
    
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cs = valid_s * (1.0 - h[sc]) * w[None, :]
    cf = valid_f * h[fc] * w[None, :]
    grad = np.sum(cs - cf, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=400):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

def pgd_optimize(h0, n, target_sum, alpha, t_end, lr_init=0.003):
    h = h0.copy()
    obj = smooth_max_obj_grad(h, alpha)[0]
    best_h = h.copy()
    best_obj = obj
    lr = lr_init
    no_improve = 0
    while time.time() < t_end:
        _, grad = smooth_max_obj_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        obj_new = smooth_max_obj_grad(h_new, alpha)[0]
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            no_improve = 0
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
            lr = min(lr * 1.01, 0.04)
        else:
            lr *= 0.75
            no_improve += 1
            if no_improve > 800:
                break
    return best_h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa(h_start, n, target_sum, t_end, rng, block_max=7):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.014
    no_improve = 0
    min_T = 0.0025
    while time.time() < t_end:
        r = rng.random()
        if r < 0.65:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        else:
            blen = rng.randint(2, block_max + 1)
            if blen >= n - 1:
                blen = max(2, n // 5)
            start = rng.randint(0, n - blen)
            block = h[start:start + blen].copy()
            if rng.random() < 0.5:
                h[start:start + blen] = block[::-1]
            else:
                shift = rng.randint(-2, 3)
                h[start:start + blen] = np.roll(block, shift)
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[start:start + blen] = block
        if no_improve > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.025)
            no_improve = 0
        T = max(T * 0.99998, min_T)
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    # Phase 1: Binary SA (20%)
    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sa_end1 = t_start + 0.18 * budget_s
    h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Continuous optimization with corrected gradient (35%)
    h_cont = best_h.copy().astype(float)
    h_cont = project_feasible(h_cont, ts_main)
    cont_end = t_start + 0.55 * budget_s
    alphas_phase2 = [15, 50, 200, 800, 3000]
    for alpha in alphas_phase2:
        if elapsed() > cont_end - t_start:
            break
        if remaining() < 40:
            break
        t_pg = min(time.time() + 12, cont_end)
        h_cont = pgd_optimize(h_cont, n_main, ts_main, alpha, t_pg)
        c5_pg = compute_c5(h_cont)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_cont.copy()
        if remaining() < 30:
            break
        t_sls = min(time.time() + 35, cont_end)
        h_sls = slsqp_optimize(h_cont, n_main, ts_main, alpha, maxiter=350)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()
            h_cont = h_sls.copy()
        else:
            h_cont = project_feasible(h_cont, ts_main)

    # Phase 3: Iterated SA from best (15%)

    c5_bin2 = compute_c5(h_bin2)
    if c5_bin2 < best_c5:

        best_h = h_bin2.copy()
    sa_end2 = t_start + 0.72 * budget_s
    if remaining() > 80:
        h_sa2, c5_sa2 = binary_sa(h_bin2, n_main, int(ts_main), sa_end2, rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Second round continuous optimization (18%)
    h_fin = best_h.copy().astype(float)
    h_fin = project_feasible(h_fin, ts_main)
    fin_end = t_start + 0.90 * budget_s
    alphas_phase4 = [1000, 5000, 20000]
    for alpha in alphas_phase4:
        if elapsed() > fin_end - t_start:
            break
        if remaining() < 25:
            break
        t_pg = min(time.time() + 12, fin_end)
        h_fin = pgd_optimize(h_fin, n_main, ts_main, alpha, t_pg)

        if remaining() < 20:
            break
        t_sls = min(time.time() + 25, fin_end)
        h_sls = slsqp_optimize(h_fin, n_main, ts_main, alpha, maxiter=300)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()
            h_fin = h_sls.copy()
        else:
            h_fin = project_feasible(h_fin, ts_main)

    # Phase 5: Shake and reoptimize
    if remaining() > 50:
        for trial in range(4):
            if remaining() < 25:
                break
            noise = 0.05 + 0.03 * trial
            h_shake = best_h + rng.uniform(-noise, noise, n_main)
            h_shake = project_feasible(h_shake, ts_main)

                if c5_try < best_c5:

                    best_h = h_interp.copy()
                    best_n = n_try

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
