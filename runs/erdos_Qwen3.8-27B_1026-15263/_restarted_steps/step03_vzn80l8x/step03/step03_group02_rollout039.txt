import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):
    """Convert to binary with exactly target_ones ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target_ones:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        # Add the zeros with largest h values
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def sa_optimize(h0, n, target_ones, t_end, rng, initial_T=None):
    """Simulated annealing for binary sequence optimization."""
    h = h0.copy()
    dx = 2.0 / n
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    if initial_T is None:
        initial_T = 0.01
    T = initial_T
    min_T = 0.001
    cooling = 0.99999
    no_improve = 0
    max_no_improve = 50000
    
    ones_indices = np.where(h == 1.0)[0]
    zeros_indices = np.where(h == 0.0)[0]
    
    while time.time() < t_end:
        # Choose move
        move_type = rng.random()
        if move_type < 0.7:
            # Single swap: move one 1 to a 0 position
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                continue
            i = ones_indices[rng.randint(len(ones_indices))]
            j = zeros_indices[rng.randint(len(zeros_indices))]
            if i == j:
                continue
            # Perform swap
            h[i], h[j] = h[j], h[i]
            # Update index arrays
            ones_indices = np.where(h == 1.0)[0]
            zeros_indices = np.where(h == 0.0)[0]
        else:
            # Double swap
            if len(ones_indices) < 2 or len(zeros_indices) < 2:
                continue
            i1 = ones_indices[rng.randint(len(ones_indices))]
            i2 = ones_indices[rng.randint(len(ones_indices))]
            if i1 == i2:
                continue
            j1 = zeros_indices[rng.randint(len(zeros_indices))]
            j2 = zeros_indices[rng.randint(len(zeros_indices))]
            if j1 == j2:
                continue
            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            ones_indices = np.where(h == 1.0)[0]
            zeros_indices = np.where(h == 0.0)[0]
        
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):

            else:

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):

def slsqp_refine(h0, n, target_sum, alpha=200, maxiter=500):
    dx = 2.0 / n
    def obj_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_val = float(np.max(corr))
        shifted = np.clip(alpha * (corr * dx - m_val * dx), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val * dx + np.log(np.sum(exp_vals)) / alpha
        npts = len(h)
        j_idx = np.arange(npts)[:, None]
        m_idx = np.arange(2 * npts - 1)[None, :]
        first_idx = j_idx - m_idx + npts - 1
        second_idx = j_idx + m_idx - npts + 1
        valid_f = (first_idx >= 0) & (first_idx < npts)
        valid_s = (second_idx >= 0) & (second_idx < npts)
        fc = np.clip(first_idx, 0, npts - 1)
        sc = np.clip(second_idx, 0, npts - 1)
        cf = valid_f * (1.0 - h[fc]) * w[None, :]
        cs = valid_s * h[sc] * w[None, :]
        grad = np.sum(cf - cs, axis=1)
        return obj, grad
    try:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        def f(h):
            return obj_and_grad(h)[0]
        def g(h):
            return obj_and_grad(h)[1]
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
