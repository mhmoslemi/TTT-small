import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr_array(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def topk_smooth_max_and_grad(h, alpha, topk):
    n = len(h)
    corr = compute_corr_array(h)
    # Sum of top-k with smooth approximation
    # Use soft-topk: topk = sum of exp(alpha * (corr - threshold)) / alpha
    # where threshold is the k-th largest value
    sorted_corr = np.sort(corr)[::-1]
    if topk >= len(corr):
        threshold = sorted_corr[-1]
    else:
        threshold = sorted_corr[topk - 1]
    shifted = alpha * (corr - threshold)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    obj = np.sum(exp_vals) / alpha + threshold
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1

    valid_s = (second_idx >= 0) & (second_idx < n)

    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, topk=None):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        if topk is None:
            return smooth_max_and_grad(x, alpha)[0]
        return topk_smooth_max_and_grad(x, alpha, topk)[0]

        if topk is None:
            return smooth_max_and_grad(x, alpha)[1]
        return topk_smooth_max_and_grad(x, alpha, topk)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

        if abs(diff) < 1e-14:

            h[mask] += diff / np.sum(mask)
        else:

        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= shifted.max()
    ev = np.exp(shifted)
    w = ev / ev.sum()
    obj = m_val + np.log(ev.sum()) / alpha
    # Gradient
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    fi = j_idx - m_idx + n - 1
    si = j_idx + m_idx - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    cf = vf * (1.0 - h[fc]) * w[None, :]
    cs = vs * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_solve(h0, n, ts, alpha, maxiter):
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except:
        return project_to_sum(h0.copy(), ts)

    n = len(h)

    c5 = compute_c5(h)
    stale = 0

        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)

        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:

                g[lag:] += (1.0 - h[:n-lag])
                g[:n-lag] -= h[lag:]

            g[k:] -= h[:n-k]

        can_dec = h > 1e-12
        if not np.any(can_inc) or not np.any(can_dec):
            break
        gi = np.where(can_inc, g, np.inf)
        gd = np.where(can_dec, g, -np.inf)
        ii = int(np.argmin(gi))
        id_ = int(np.argmax(gd))
        if ii == id_:
            break
        if g[ii] >= g[id_] - 1e-14:
            break

        max_step = min(1.0 - h[ii], h[id_])
