def solve_1d(A, D, lo, hi):

        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)

    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(36):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = corr.max()
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    obj = m_val + np.log(exp_vals.sum()) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad


def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_obj_grad(h, alpha)[0]
    def g(h):
        return smooth_max_obj_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def upscale_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:

    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

            if c5_w < best_c5 - 1e-12:
                best_c5 = c5_w
                best_h = h_work.copy()
                best_n = n_w
                stall = 0

    # Phase 2: Multi-resolution escalation (40% budget)
    t_end2 = t_start + 0.75 * budget_s
    resolutions = [320, 400, 480, 560, 640, 720, 800]
    for n_tr in resolutions:
        if time_left() < 30:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_tr, endpoint=False)
        h_tr = np.interp(x_new, x_old, best_h, period=2.0)
        h_tr = project_to_sum(h_tr, ts_tr)
        corr_tr = corr_full(h_tr)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr

        n_sweeps_target = max(10, int((t_end2 - time.time()) * 0.15 / max(0.1, n_tr / 240.0 / 60.0)))
        n_sweeps_target = min(n_sweeps_target, 200)
        stall_tr = 0
        for _ in range(n_sweeps_target):
            if time_left() < 30:
                break
            c_lags = get_critical_lags(corr_tr, top_frac=0.05)
            h_tr, corr_tr, _ = pair_descent_sweep_v2(h_tr, corr_tr, rng, n_cand=8, critical_lags=c_lags)
            c5_tr = compute_c5(h_tr)
            if c5_tr < best_c5 - 1e-12:
                best_c5 = c5_tr
                best_h = h_tr.copy()
                best_n = n_tr
                stall_tr = 0
            else:
                stall_tr += 1
            if stall_tr > 100:
                h_tr = project_to_sum(h_tr + rng.normal(0, 0.03, n_tr), ts_tr)
                corr_tr = corr_full(h_tr)
                stall_tr = 0

        # Try a structured start at this resolution
        if time_left() > 60:
            h_alt = make_structured_start(n_tr, rng, style=rng.randint(0, 5))
            h_alt = project_to_sum(h_alt, ts_tr)
            corr_alt = corr_full(h_alt)
            n_sweeps_alt = max(5, n_sweeps_target // 3)
            for _ in range(n_sweeps_alt):
                if time_left() < 30:
                    break
                c_lags = get_critical_lags(corr_alt, top_frac=0.05)
                h_alt, corr_alt, _ = pair_descent_sweep_v2(h_alt, corr_alt, rng, n_cand=8, critical_lags=c_lags)
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_tr

    # Phase 3: SLSQP polish (10% budget)
    t_end3 = t_start + 0.88 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [30, 80, 200]:
        if time_left() < 15:
            break
        t_pol_start = time.time()
        maxit = max(100, int(min(500, (t_end3 - t_start) * 0.3 / max(0.01, (time.time() - t_start) / budget_s + 0.01))))
        h_pol = slsqp_local(best_h.copy(), n_f, ts_f, alpha, maxiter=min(maxit, 400))
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 4: Final intensive pair descent (12% budget, reserve 2s)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = corr_full(h_fin)
    final_stall = 0
    while time_left() > 3:
        c_lags = get_critical_lags(corr_fin, top_frac=0.08)
        h_fin, corr_fin, _ = pair_descent_sweep_v2(h_fin, corr_fin, rng, n_cand=14, critical_lags=c_lags)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            final_stall = 0
        else:
            final_stall += 1
        if final_stall > 200:
            h_fin = project_to_sum(h_fin + rng.normal(0, 0.02, n_fin), n_fin / 2.0)
            corr_fin = corr_full(h_fin)
            final_stall = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
