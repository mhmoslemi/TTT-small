        idx = np.arange(s1, e1 + 1)
        B[s1:e1 + 1] = (1.0 - h[j - nm1 + idx])
    s2 = max(0, j)
    e2 = min(2 * n - 2, j + nm1)
    if s2 <= e2:
        idx = np.arange(s2, e2 + 1)
        B[s2:e2 + 1] += (-h[j + nm1 - idx])
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = np.max(A + D * c)
    fd = np.max(A + D * d)
    for _ in range(45):
        if hi - lo < 1e-16:
            break
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = np.max(A + D * c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = np.max(A + D * d)
    return (lo + hi) * 0.5

def pair_sweep(h, corr, rng, n, n_cand=20):
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    cur_max = np.max(corr)
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = cur_max
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Build candidate set
        k_set = [
            (j + 1) % n, (j - 1 + n) % n,
            (j + n // 4) % n, (j + n // 3) % n,
            (j + n // 2) % n, (j + 2 * n // 3) % n,
            (j + 3 * n // 4) % n,
        ]
        for _ in range(n_cand - len(k_set)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            k = k % n
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-14:
                continue
            delta = solve_1d(corr, D, dlo, dhi)
            val = np.max(corr + D * delta)
            if val < best_val - 1e-13:
                best_val = val
                best_delta = delta
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            cur_max = best_val
            improved = True
    return h, corr, improved

def block_swap(h, corr, rng, n, block_len, max_tries=20):
    """Try to swap a block of values between two positions."""
    improved = False
    for _ in range(max_tries):
        s = rng.randint(0, n - block_len)
        t = rng.randint(0, n - block_len)
        if abs(s - t) < block_len:
            continue
        # Compute B for all indices in both blocks
        B_s = np.zeros(2 * n - 1)
        B_t = np.zeros(2 * n - 1)
        for jj in range(block_len):
            B_s += compute_B(h, s + jj, n)
            B_t += compute_B(h, t + jj, n)
        D = B_s - B_t
        # The move is: h[s:s+L] += δ, h[t:t+L] -= δ
        # Constraints: 0 ≤ h[s+j] + δ ≤ 1 and 0 ≤ h[t+j] - δ ≤ 1 for all j
        dlo = max(h[t + jj] - 1.0 for jj in range(block_len))
        dhi = min(1.0 - h[s + jj] for jj in range(block_len))
        # Wait, that's wrong. dlo should be max of -h[s+j] and h[t+j]-1
        dlo = max(max(-h[s + jj] for jj in range(block_len)), max(h[t + jj] - 1.0 for jj in range(block_len)))
        dhi = min(min(1.0 - h[s + jj] for jj in range(block_len)), min(h[t + jj] for jj in range(block_len)))
        if dhi - dlo < 1e-14:
            continue
        delta = solve_1d(corr, D, dlo, dhi)
        val = np.max(corr + D * delta)
        if val < np.max(corr) - 1e-12:
            corr += delta * D
            for jj in range(block_len):
                h[s + jj] = np.clip(h[s + jj] + delta, 0.0, 1.0)
                h[t + jj] = np.clip(h[t + jj] - delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: d corr[m] / d h[j]
    # corr[m] = sum_i h[i] * (1-h[i+m-n+1])
    # d corr[m] / d h[j] = (1-h[j+m-n+1]) * [valid] + (-h[j-m+n-1]) * [valid]
    # grad[j] = sum_m w[m] * d corr[m] / d h[j]
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    # First term: h[j] * (1-h[j+m-n+1])
    i1 = j_idx + m_idx - n + 1
    v1 = (i1 >= 0) & (i1 < n)
    i1c = np.clip(i1, 0, n - 1)
    # Second term: -h[j] from (1-h[i+m-n+1]) where i+m-n+1 = j
    i2 = j_idx - m_idx + n - 1
    v2 = (i2 >= 0) & (i2 < n)
    i2c = np.clip(i2, 0, n - 1)
    
    grad = np.sum((v1 * (1.0 - h[i1c]) - v2 * h[i2c]) * w[None, :], axis=1)
    return obj, grad

def slsqp_polish(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
