**Alternative: Direct subgradient / cutting plane method**

The idea: we want to minimize f(h) = max_k corr_k(h). 

A cutting plane method would:
1. Start with some h
2. Evaluate corr_k(h) for all k
3. Add a "cut": the new h must satisfy corr_k(h) <= t for the active k's
4. Solve the resulting problem (minimize t subject to corr_k(h) <= t for active k, 0<=h<=1, sum(h)=n/2)
5. Update active set

But step 4 is a non-convex QP (since corr_k is quadratic in h with negative definite Hessian). So this won't work directly.

However, if we linearize corr_k around the current point h_0:
corr_k(h) ≈ corr_k(h_0) + grad_k(h_0)^T (h - h_0)

Then the problem becomes:
minimize t
subject to corr_k(h_0) + grad_k(h_0)^T (h - h_0) <= t for active k
0 <= h <= 1
sum(h) = n/2

This IS a linear program! We can solve it with any LP solver.

Then we update h and repeat. This is essentially the **linearized cutting plane method** or **outer approximation**.

This could be very effective because:
- Each LP is fast (can use scipy or cvxpy with a fast LP solver)
- We only need to maintain the active set (subset of lags)
- The method converges to a local optimum

Let me implement this:

```python
from scipy.optimize import linprog

def cuttingplane_step(h, active_lags, n):
    # Linearize corr_k at h for each active lag k
    # corr_k(h') ≈ corr_k(h) + g_k^T (h' - h)
    # where g_k = gradient of corr_k at h
    
    n_vars = n + 1  # h_0, ..., h_{n-1}, t
    # Minimize t (last variable)
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    # Constraints: corr_k(h) + g_k^T (h' - h) <= t  =>  g_k^T h' - t <= corr_k(h) - g_k^T h
    # i.e., [g_k, -1] @ [h', t] <= corr_k(h) - g_k^T h
    
    A_ub = []
    b_ub = []
    
    for lag in active_lags:
        g = corr_gradient(h, lag)
        rhs = np.dot(g, h) - g.dot(h)  # = 0? No...
        # corr_k(h) + g^T (h' - h) <= t
        # g^T h' - t <= corr_k(h) - g^T h
        # But g^T h = sum_i g[i]*h[i], and corr_k(h) is the actual value
        # Actually: corr_k(h') ≈ corr_k(h) + g^T(h'-h)
        # So: corr_k(h) + g^T(h'-h) <= t
        # => g^T h' - t <= corr_k(h) - g^T h
        # But corr_k(h) = sum_j h[j](1-h[j+lag]), and g = grad of this
        # g^T h is not simply related to corr_k(h)
        
        g = corr_gradient(h, lag)
        corr_val = # compute corr at this lag
        rhs = corr_val - g.dot(h)
        A_ub.append(np.append(g, -1.0))
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: 0 <= h_i <= 1, t free (or t >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = n / 2.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    return h, None
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    i_idx = np.arange(n)[:, None]
    lags = (np.arange(2 * n - 1) - (n - 1))[None, :]
    first_idx = i_idx + lags
    valid_f = (first_idx >= 0) & (first_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    second_idx = i_idx - lags
    valid_s = (second_idx >= 0) & (second_idx < n)
    sc = np.clip(second_idx, 0, n - 1)
    wf = w[None, :]
    term1 = np.where(valid_f, 1.0 - h[fc], 0.0) * wf
    term2 = np.where(valid_s, h[sc], 0.0) * wf
    grad = np.sum(term1, axis=1) - np.sum(term2, axis=1)
    return obj, grad

def slsqp_optimize(h0, target_sum, alpha, maxiter=600, ftol=1e-14):
    n = len(h0)
    def obj_func(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def grad_func(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def adam_optimize(h0, target_sum, alpha, lr=0.005, max_steps=3000, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr_cur = lr
    for step in range(max_steps):
        obj, grad = smooth_max_obj_grad(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (step + 1))
        v_hat = v / (1.0 - beta2 ** (step + 1))
        step_size = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step_size
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr_cur = min(lr_cur * 1.05, lr * 3.0)
        else:
            h_restored = project_to_sum(h, target_sum)
            lr_cur *= 0.9
            if lr_cur < 1e-8:
                break
    return best_h, best_c5

def corr_gradient_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def coord_descent_v2(h, target_sum, max_iters=10000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)

            stale += 1
            if stale > 150:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_adam, c5_adam = adam_optimize(h_new, ts, alpha=5000, lr=0.002, max_steps=1000)
                if c5_adam < best_c5:
                    best_c5 = c5_adam
                    best_h = h_adam.copy()

    if time_left() > 60:
        h_final, c5_final = coord_descent_v2(best_h, ts, max_iters=20000, step_frac=0.05)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if time_left() > 40:
        h_ml2, c5_ml2 = multi_lag_coord_descent(best_h, ts, max_iters=10000, step_frac=0.05, n_top=5)
        if c5_ml2 < best_c5:
            best_c5 = c5_ml2
            best_h = h_ml2.copy()

    if time_left() > 30:
        for alpha in [50000, 200000]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, ts, alpha, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time_left() > 20:
        h_adam3, c5_adam3 = adam_optimize(best_h, ts, alpha=10000, lr=0.001, max_steps=2000)
        if c5_adam3 < best_c5:
            best_c5 = c5_adam3
            best_h = h_adam3.copy()

    if time_left() > 15:
        h_final2, c5_final2 = coord_descent_v2(best_h, ts, max_iters=15000, step_frac=0.03)
        if c5_final2 < best_c5:
            best_c5 = c5_final2
            best_h = h_final2.copy()

    best_h = project_to_sum(best_h, ts)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
```
