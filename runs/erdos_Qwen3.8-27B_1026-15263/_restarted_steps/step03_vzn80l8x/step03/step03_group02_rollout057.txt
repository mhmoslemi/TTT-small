def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    shift = m_idx - (n - 1)
    first_idx = j_idx + shift
    second_idx = j_idx - shift
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term1 = valid_f * (1.0 - h[fc]) * w[None, :]
    term2 = valid_s * h[sc] * w[None, :]
    grad = np.sum(term1 - term2, axis=1)
    return obj, grad

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    return h

def make_binary(h, n, target_sum, rng):

    if n_ones == target_sum:
        return h_bin

        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def get_gradient(h, alpha=100.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    shift = m_idx - (n - 1)
    first_idx = j_idx + shift
    second_idx = j_idx - shift
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)

    term2 = valid_f * h[fc] * w[None, :]
    grad = np.sum(term1 - term2, axis=1)
    return grad, corr, m_val

        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Sort by gradient: most positive grad for 1s (bad, should decrease)
        # most negative grad for 0s (bad, should increase)
        top_ones = ones[np.argsort(-grad[ones])[:K]]
        top_zeros = zeros[np.argsort(grad[zeros])[:K]]
        best_c5 = c5
        best_swap = None
        for i in top_ones:
            for j in top_zeros:

        T = max(T * decay, Tmin)
    return best_h, best_c5

def proj_grad_optimize(h0, n, target_sum, alpha, maxiter):
    h = h0.copy().astype(float)
    lr = 0.3
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stale = 0
    for it in range(maxiter):
        grad, corr, m_val = get_grad_and_corr(h, alpha)
        step = lr
        improved = False
        for _ in range(12):
            h_new = project_to_sum(h - step * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if improved:
            stale = 0
            lr = min(lr * 1.02, 1.5)
        else:
            stale += 1
            lr *= 0.85
        if stale > 150:
            break
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def perturb_solution(h, n, target_sum, rng, n_kicks):
    h_pert = h.copy()
    ones = np.where(h_pert > 0.5)[0]
    zeros = np.where(h_pert < 0.5)[0]
    for _ in range(n_kicks):
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
    return h_pert
