def compute_corr(h):

def project_to_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-12:

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def corr_gradient_single_lag(h, lag):

    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:

def informed_coord_desc(h, target_sum, max_iters=10000, step_frac=0.1, use_momentum=False):
    """Coordinate descent that targets the worst lag with momentum."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    velocity = np.zeros(n) if use_momentum else None
    beta = 0.9 if use_momentum else 0
    stale = 0

    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)

        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single_lag(h, lag)

        can_inc = (h < 1.0 - 1e-10) & (g < 0)  # positions where decreasing helps
        can_dec = (h > 1e-10) & (g > 0)  # positions where increasing helps

        # - Increase h where g < 0 (since grad tells us direction of increase in corr)
        # - Decrease h where g > 0

        # So we want to increase h[j] where g[j] < 0 and decrease h[j] where g[j] > 0

        inc_candidates = np.where((h < 1.0 - 1e-10) & (g < 0))[0]
        dec_candidates = np.where((h > 1e-10) & (g > 0))[0]

            # Try random move
            inc_candidates = np.where(h < 1.0 - 1e-10)[0]
            dec_candidates = np.where(h > 1e-10)[0]
        
        if len(inc_candidates) == 0 or len(dec_candidates) == 0:
            break
        
        # Weight by |gradient|
        w_inc = np.abs(g[inc_candidates])
        w_dec = np.abs(g[dec_candidates])
        w_inc = w_inc / w_inc.sum()
        w_dec = w_dec / w_dec.sum()
        
        i = inc_candidates[rng.choice(len(inc_candidates), p=w_inc)]
        j = dec_candidates[rng.choice(len(dec_candidates), p=w_dec)]
        
        if i == j:
            continue
        
        # Step size
        max_move = min(1.0 - h[i], h[j])
        move = min(0.1 * max_move, 0.05)
        
        h[i] += move
        h[j] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)

            h[j] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    
    return best_h, best_c5

**Key new idea: "Pressure" method**
- Compute all correlations
- Identify the top-K lags (those within, say, 1% of the maximum)
- Compute a weighted gradient that pushes down ALL of these lags simultaneously
- The weight for each lag is proportional to (corr[k] - threshold) where threshold is slightly below the max
- This creates "pressure" on all near-maximal lags

This is different from just using the single worst lag (which might shift the max) or using log-sum-exp with very high alpha (which is similar to single lag).

def pressure_gradient(h, top_k_frac=0.1, alpha=100):
    """Compute gradient that pushes down the top fraction of correlations."""
    corr = compute_corr(h)
    n = len(h)
    # Use log-sum-exp for smoothness
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)  # numerical stability
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

        g += w[m_idx] * corr_gradient_single_lag(h, lag)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def pressure_gradient(h, alpha):
    """Compute weighted gradient using log-sum-exp over all lags."""
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    g = np.zeros(n)
    significant = np.where(w > 1e-8)[0]
    for m_idx in significant:
        lag = m_idx - (n - 1)
        g += w[m_idx] * corr_gradient_single_lag(h, lag)
    
    obj = m + np.log(exp_vals.sum()) / alpha
    return obj, g, corr

def slsqp_pressure(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return pressure_gradient(x, alpha)[0]
    def g(x):
        return pressure_gradient(x, alpha)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_desc_with_backtrack(h, target_sum, max_iters=20000, base_step=0.15):
    """Coordinate descent with backtracking line search."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single_lag(h, lag)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            stale += 1
            if stale > 100:
                break
            continue
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = base_step * max_move
        
        # Backtracking
        accepted = False
        for _bt in range(5):
            if move < 1e-12:
                break
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                accepted = True
                break
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            move *= 0.5
        
        if not accepted:
            stale += 1
            if stale > 100:
                break
        else:
            stale = 0
    
    return h, c5

def annealing_with_pressure(h, target_sum, rng, max_iters=30000, 
                            T_start=0.005, T_end=0.0001, alpha=50):
    """Simulated annealing using pressure gradient for proposals."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Use pressure gradient for proposals
        obj, g, corr = pressure_gradient(h, alpha)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        
        # Weight proposals by gradient magnitude
        w_inc = np.abs(g[can_inc])
        w_dec = np.abs(g[can_dec])
        
        if w_inc.sum() < 1e-10 or w_dec.sum() < 1e-10:
            break
        
        w_inc = w_inc / w_inc.sum()
        w_dec = w_dec / w_dec.sum()
        
        inc_idx = np.where(can_inc)[0]
        dec_idx = np.where(can_dec)[0]
        
        i = inc_idx[rng.choice(len(inc_idx), p=w_inc)]
        j = dec_idx[rng.choice(len(dec_idx), p=w_dec)]
        
        if i == j:
            continue
        
        max_move = min(1.0 - h[i], h[j])
        move = min(0.1 * max_move, 0.03)
        
        h[i] += move
        h[j] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        delta = c5_new - c5
        
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= move
            h[j] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    
    return best_h, best_c5

def block_swap_anneal(h, target_sum, rng, max_iters=20000, block_sizes=[2,3,4,5,8]):
    """Annealing with block swap moves."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        # Choose block size
        bs = block_sizes[rng.randint(len(block_sizes))]
        if bs >= n // 2:
            bs = 2
        
        # Choose two non-overlapping blocks to swap
        i = rng.randint(0, n - bs)
        # Choose j far from i
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        
        # Swap blocks
        h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
        h = np.clip(h, 0.0, 1.0)
        # Note: swapping preserves sum, so no projection needed
        
        c5_new = compute_c5(h)
        delta = c5_new - c5
        
        if delta < 0:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Reject
            h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_cons_target_sum = 0.0
_cons_n = 0

def _cons_fun(x):
    return np.sum(x) - _cons_target_sum

def _cons_jac(x):
    return np.ones(_cons_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def pressure_gradient(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    significant = np.where(w > 1e-7)[0]
    for m_idx in significant:
        lag = m_idx - (n - 1)
        g += w[m_idx] * corr_gradient_single_lag(h, lag)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    return obj, g, corr

def slsqp_pressure(h0, n, target_sum, alpha, maxiter):
    global _cons_target_sum, _cons_n
    _cons_target_sum = target_sum
    _cons_n = n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    def _f(x):
        return pressure_gradient(x, alpha)[0]
    def _g(x):
        return pressure_gradient(x, alpha)[1]
    try:
        res = minimize(_f, h0.copy(), method='SLSQP', jac=_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_subgrad(h, target_sum, alpha, max_iters, lr0=0.02):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    lr = lr0
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iters):
        obj, g, corr = pressure_gradient(h, alpha)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return best_h, best_c5

def coord_desc_backtrack(h, target_sum, max_iters=15000, base_step=0.12):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient_single_lag(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            stale += 1
            if stale > 150:
                break
            continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = base_step * max_move
        accepted = False
        for _bt in range(5):
            if move < 1e-13:
                break
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                accepted = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            move *= 0.5
        if not accepted:
            stale += 1
            if stale > 150:
                break
        else:
            stale = 0
    return best_h, best_c5

def annealing_pressure(h, target_sum, rng, max_iters=25000, T_start=0.004, T_end=0.00003, alpha=40):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)
        if T < 1e-7:
            break
        obj, g, corr = pressure_gradient(h, alpha)
        can_inc = (h < 1.0 - 1e-10) & (g < -1e-10)
        can_dec = (h > 1e-10) & (g > 1e-10)
        if not np.any(can_inc) or not np.any(can_dec):
            can_inc = h < 1.0 - 1e-10
            can_dec = h > 1e-10
            if not np.any(can_inc) or not np.any(can_dec):
                break
        w_inc = np.abs(g[can_inc])
        w_dec = np.abs(g[can_dec])
        if w_inc.sum() < 1e-12:
            w_inc = np.ones(np.sum(can_inc))
        if w_dec.sum() < 1e-12:
            w_dec = np.ones(np.sum(can_dec))
        w_inc = w_inc / w_inc.sum()
        w_dec = w_dec / w_dec.sum()
        inc_idx = np.where(can_inc)[0]
        dec_idx = np.where(can_dec)[0]
        i = inc_idx[rng.choice(len(inc_idx), p=w_inc)]
        j = dec_idx[rng.choice(len(dec_idx), p=w_dec)]
        if i == j:
            continue
        max_move = min(1.0 - h[i], h[j])
        move = min(0.08 * max_move, 0.02)
        h[i] += move
        h[j] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= move
            h[j] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
    return best_h, best_c5

def binary_block_anneal(h, target_sum, rng, max_iters=20000):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    ones_needed = int(target_sum)
    curr_ones = int(np.sum(h_bin))
    if curr_ones > ones_needed:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) >= curr_ones - ones_needed:
            idx = rng.choice(np.where(h_bin == 1)[0], curr_ones - ones_needed, replace=False)
            h_bin[idx] = 0
    elif curr_ones < ones_needed:
        if len(np.where(h_bin == 0)[0]) >= ones_needed - curr_ones:
            idx = rng.choice(np.where(h_bin == 0)[0], ones_needed - curr_ones, replace=False)
            h_bin[idx] = 1
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    block_sizes = [1, 2, 3, 4, 5, 6, 8, 12]
    for it in range(max_iters):
        bs = block_sizes[rng.randint(len(block_sizes))]
        if bs >= n // 2:
            bs = 2
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
        c5_new = compute_c5(h)
        delta = c5_new - c5
        T = 0.002 * (0.5 ** (it / max(float(max_iters), 1)))
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)
