def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if abs(diff) < 1e-15:

def smooth_max_and_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m_val + max_sh / alpha + np.log(np.sum(exp_v)) / alpha)
    n = len(h)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)

        x_new2 = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new2, x_new, h_sub, period=2.0)
        h = project_to_sum(h, ts)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5, _ = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5, _ = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: FISTA with smooth max at current resolution
    for alpha in [50, 150, 500, 1500, 5000]:
        if time_left() < 30:
            break
        n_iter = max(300, int(time_left() * 2))
        h_new = fista_adaptive(best_h, n, ts, alpha, n_iter=n_iter)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Binary projection + local search
    if time_left() > 60:
        h_bin = binary_project(best_h, n)
        c5_bin, _ = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_ls, c5_ls = local_search_binary(h_bin, n, max_rounds=100, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        h_tls, c5_tls = targeted_local_search(h_ls, n, max_rounds=80, rng=rng)
        if c5_tls < best_c5:
            best_c5 = c5_tls
            best_h = h_tls.copy()

    # Phase 3: Continue FISTA from binary-informed start
    if time_left() > 60:
        h_cont = best_h.copy()
        for alpha in [200, 800, 3000, 10000]:
            if time_left() < 20:
                break
            h_new = fista_adaptive(h_cont, n, ts, alpha, n_iter=800)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cont = h_new

    # Phase 4: Multi-scale approach
    if time_left() > 120:
        for n_sub in [120, 160, 200, 240]:
            if time_left() < 40:
                break
            if n_sub % 2 != 0:
                continue
            ts_sub = n_sub / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_sub, endpoint=False)
            h_sub = np.interp(x_new, x_old, best_h, period=2.0)
            h_sub = project_to_sum(h_sub, ts_sub)
            c5_sub, _ = compute_c5(h_sub)
            for alpha in [50, 200, 800]:
                if time_left() < 15:
                    break
                h_sub = fista_adaptive(h_sub, n_sub, ts_sub, alpha, n_iter=400)
            h_sub_bin = binary_project(h_sub, n_sub)
            h_sub_bin, c5_sub_bin = local_search_binary(h_sub_bin, n_sub, max_rounds=50, rng=rng)
            if c5_sub_bin < compute_c5(h_sub)[0]:
                h_sub = h_sub_bin
            x_up = np.linspace(0, 2, n, endpoint=False)
            h_up = np.interp(x_up, x_new, h_sub, period=2.0)
            h_up = project_to_sum(h_up, ts)
            c5_up, _ = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
                n = n
            for alpha in [500, 2000]:
                if time_left() < 15:
                    break
                h_ref = fista_adaptive(h_up, n, ts, alpha, n_iter=500)
                c5_ref, _ = compute_c5(h_ref)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()

    # Phase 5: Higher resolution attempt
    if time_left() > 150:
        for n_hi in [600, 720]:
            if time_left() < 60:
                break
            if n_hi % 2 != 0:
                continue
            ts_hi = n_hi / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, best_h, period=2.0)
            h_hi = project_to_sum(h_hi, ts_hi)
            c5_hi, _ = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                n = n_hi
                ts = ts_hi
            for alpha in [100, 500, 2000]:
                if time_left() < 30:
                    break
                h_hi = fista_adaptive(h_hi, n_hi, ts_hi, alpha, n_iter=500)
                c5_hi, _ = compute_c5(h_hi)
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = h_hi.copy()
                    n = n_hi
                    ts = ts_hi
            h_hi_bin = binary_project(h_hi, n_hi)
            h_hi_bin, c5_hi_bin = local_search_binary(h_hi_bin, n_hi, max_rounds=50, rng=rng)
            if c5_hi_bin < best_c5:
                best_c5 = c5_hi_bin
                best_h = h_hi_bin.copy()
                n = n_hi
                ts = ts_hi

    # Phase 6: Simulated annealing on exact objective
    if time_left() > 80:
        h_sa = best_h.copy()
        h_sa_bin = binary_project(h_sa, n)
        n_steps = min(60000, int(time_left() * 3))
        h_sa, c5_sa = simulated_annear(h_sa_bin, n, rng, n_steps=n_steps, t_start=0.03, t_end=0.001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 7: Fine local search
    if time_left() > 40:
        h_bin = binary_project(best_h, n)
        h_tls, c5_tls = targeted_local_search(h_bin, n, max_rounds=150, rng=rng)
        if c5_tls < best_c5:
            best_c5 = c5_tls
            best_h = h_tls.copy()
        h_ls, c5_ls = local_search_binary(h_tls, n, max_rounds=100, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 8: Final FISTA polish
    if time_left() > 30:
        for alpha in [2000, 5000, 15000]:
            if time_left() < 15:
                break
            h_pol = fista_adaptive(best_h, n, ts, alpha, n_iter=1000)
            c5_pol, _ = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 9: Final tiny local search
    if time_left() > 20:
        h_bin = binary_project(best_h, n)
        h_final, c5_final = targeted_local_search(h_bin, n, max_rounds=100, rng=rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final validation and projection
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5, _ = compute_c5(best_h)
    return (best_h, best_c5, best_n)
