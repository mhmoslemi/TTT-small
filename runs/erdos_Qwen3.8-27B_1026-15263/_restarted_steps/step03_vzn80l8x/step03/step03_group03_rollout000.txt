def pair_descent(h, corr, rng, n_cand=12):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = list({(j + 1) % n, (j - 1 + n) % n,
                      (j + n // 2) % n, (j + n // 3) % n,
                      (j + 2 * n // 3) % n, (j + n // 4) % n,
                      (j + 3 * n // 4) % n})
        for _ in range(max(0, n_cand - len(k_set))):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:

                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            if delta_opt <= delta_lo + 1e-14 or delta_opt >= delta_hi - 1e-14:
                continue
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-12:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def block_move(h, corr, rng, max_block=8):
    # Try rotating a block or swapping two blocks
    n = len(h)
    improved = False
    for _ in range(5):
        blen = rng.randint(2, max_block + 1)
        start = rng.randint(0, n - blen)
        # Try swapping with another block of same length
        start2 = rng.randint(0, n - blen)
        if abs(start - start2) < blen:
            continue
        h_new = h.copy()
        h_new[start:start+blen], h_new[start2:start2+blen] = h[start2:start2+blen].copy(), h[start:start+blen].copy()
        # Check sum constraint
        if abs(np.sum(h_new) - np.sum(h)) < 1e-12:

            if np.max(corr_new) < np.max(corr) - 1e-12:
                h = h_new
                corr = corr_new
                improved = True
    return h, corr, improved

def simulated_annealing_step(h, corr, T, rng, target_sum):
    n = len(h)
    # Random pair swap
    j = rng.randint(n)
    k = rng.randint(n)
    if j == k:
        return h, corr, False
    delta = rng.uniform(-0.3, 0.3)
    h[j] += delta
    h[k] -= delta
    if h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
        return h, corr, False
    h = project_to_sum(h, target_sum)
    corr_new = np.correlate(h, 1.0 - h, mode="full")
    delta_obj = np.max(corr_new) - np.max(corr)
    if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (T + 1e-10)):
        corr = corr_new
        return h, corr, True
    return h, corr, False

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h_f):
        return smooth_max_and_grad(h_f, alpha)[0]
    def g(h_g):
        return smooth_max_and_grad(h_g, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
