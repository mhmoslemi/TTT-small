import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        nc = np.sum(mask)
        if nc > 0:
            h[mask] += diff / nc
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s_exp = np.sum(exp_v)
    w = exp_v / s_exp
    obj = m_val + (max_sh + np.log(s_exp)) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=1000):
    def f(x):
        return smooth_max_grad(x, alpha)[0]
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def incremental_stochastic_cd(h, n, ts, rng, n_steps=100000, delta_lo=0.005, delta_hi=0.3, sa_temp=0.05):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_cur = np.max(corr)
    temp = sa_temp
    
    for step in range(n_steps):
        if c5_cur < 0:
            break
        # Pick a random "hot" index (where h is neither 0 nor 1)
        hot = (h > 0.02) & (h < 0.98)
        n_hot = np.sum(hot)
        if n_hot < 2:
            break
        hot_idx = np.where(hot)[0]
        m = hot_idx[rng.randint(n_hot)]
        # Pick another hot index to transfer to/from
        m2 = hot_idx[rng.randint(n_hot)]
        if m == m2:
            continue
        
        delta = rng.uniform(delta_lo, delta_hi)
        # Determine direction: try to reduce overlap
        # If h[m] > h[m2], move mass from m to m2 (or vice versa)
        if h[m] > h[m2]:
            dm = -delta  # decrease h[m]
            dm2 = delta   # increase h[m2]
        else:
            dm = delta
            dm2 = -delta
        
        # Clamp
        if h[m] + dm < 0:
            dm = -h[m]
            dm2 = -dm
        if h[m] + dm > 1:
            dm = 1 - h[m]
            dm2 = -dm
        if h[m2] + dm2 < 0:
            dm2 = -h[m2]
            dm = -dm2
        if h[m2] + dm2 > 1:
            dm2 = 1 - h[m2]
            dm = -dm2
        
        if abs(dm) < 1e-6:
            continue
        
        old_c5 = c5_cur
        # Update h
        h[m] += dm
        h[m2] += dm2
        
        # Update corr incrementally
        # For changing h[m] by dm:
        #   corr[m_idx] changes by: dm*(1-h[m-k]) - dm*h[m+k] for valid indices
        # where k = m_idx - (n-1)
        
        # For h[m] changed by dm:
        #   h[i+k] term: i = m-k, valid if 0 <= m-k < n → k in [m-n+1, m]
        #     corr[n-1+k] += dm * (1 - h[m-k])
        #   (1-h[i]) term: i = m, valid if 0 <= m+k < n → k in [-m, n-1-m]
        #     corr[n-1+k] -= dm * h[m+k]
        
        # Range of k for h[m] in h[i+k]: k from max(-(n-1), m-(n-1)) to min(n-1, m)
        #   i = m-k in [0,n): k in [m-(n-1), m] intersect [-(n-1), n-1]
        #   = [max(m-(n-1), -(n-1)), min(m, n-1)]
        
        k_lo1 = max(m - (n-1), -(n-1))
        k_hi1 = min(m, n-1)
        for k in range(k_lo1, k_hi1 + 1):
            i = m - k
            corr[n-1+k] += dm * (1.0 - h[i])
        
        k_lo2 = max(-m, -(n-1))
        k_hi2 = min(n-1-m, n-1)
        for k in range(k_lo2, k_hi2 + 1):
            j = m + k
            corr[n-1+k] -= dm * h[j]
        
        # For h[m2] changed by dm2:
        k_lo1 = max(m2 - (n-1), -(n-1))
        k_hi1 = min(m2, n-1)
        for k in range(k_lo1, k_hi1 + 1):
            i = m2 - k
            corr[n-1+k] += dm2 * (1.0 - h[i])
        
        k_lo2 = max(-m2, -(n-1))
        k_hi2 = min(n-1-m2, n-1)
        for k in range(k_lo2, k_hi2 + 1):
            j = m2 + k
            corr[n-1+k] -= dm2 * h[j]
        
        c5_new = np.max(corr)
        
        if c5_new < old_c5:
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - old_c5) / max(temp, 1e-10))
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                # Revert
                h[m] -= dm
                h[m2] -= dm2
                # Revert corr
                for k in range(k_lo1, k_hi1 + 1):
                    i = m - k
                    corr[n-1+k] -= dm * (1.0 - h[i])
                for k in range(k_lo2, k_hi2 + 1):
                    j = m + k
                    corr[n-1+k] += dm * h[j]
                k_lo1 = max(m2 - (n-1), -(n-1))
                k_hi1 = min(m2, n-1)
                for k in range(k_lo1, k_hi1 + 1):
                    i = m2 - k
                    corr[n-1+k] -= dm2 * (1.0 - h[i])
                k_lo2 = max(-m2, -(n-1))
                k_hi2 = min(n-1-m2, n-1)
                for k in range(k_lo2, k_hi2 + 1):
                    j = m2 + k
                    corr[n-1+k] += dm2 * h[j]
        
        # Cool down
        if step % 5000 == 0:
            temp *= 0.85
    
    h = project_to_sum(h, ts)
    return h, c5_cur

  h[::-1] = [d, c, b, a], 1-h[::-1] = [1-d, 1-c, 1-b, 1-a]. ✓
  
  My formula says: corr[m:m+n] = corr[1:5] += dm * (1-h[::-1]) = dm * [1-d, 1-c, 1-b, 1-a]. ✓!

- (1-h[i]) = (1-h[1]) when i=1:
  k=-3: i+k = 1-3 = -2 (invalid) → no
  k=-2: i+k = 1-2 = -1 (invalid) → no
  k=-1: i+k = 0: h[0]=a → corr[2] gets -dm*h[0] = -dm*a
  k=0: i+k = 1: h[1]=b → corr[3] gets -dm*h[1] = -dm*b
  k=1: i+k = 2: h[2]=c → corr[4] gets -dm*h[2] = -dm*c
  k=2: i+k = 3: h[3]=d → corr[5] gets -dm*h[3] = -dm*d
  k=3: i+k = 4 (invalid) → no
  
  So corr[2:6] gets -dm * [a, b, c, d] = -dm * h
  My formula says: corr[n-1-m : 2*n-1-m] = corr[4-1-1 : 8-1-1] = corr[2:6] -= dm * h. ✓!

corr[m : m+n] += dm * (1.0 - h[::-1])
corr[n-1-m : 2*n-1-m] -= dm * h

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        nc = np.sum(mask)
        if nc > 0:
            h[mask] += diff / nc
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s_exp = np.sum(exp_v)
    w = exp_v / s_exp
    obj = m_val + (max_sh + np.log(s_exp)) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def smooth_max_obj_fn(x, alpha):
    return smooth_max_grad(x, alpha)[0]

def smooth_max_grd_fn(x, alpha):
    return smooth_max_grad(x, alpha)[1]

def sum_eq_fn(x, ts):
    return np.sum(x) - ts

def sum_eq_jac_fn(x, ts):
    return np.ones(len(x))

def slsqp_opt(h0, n, ts, alpha, maxiter=1000):
    f = partial(smooth_max_obj_fn, alpha=alpha)
    g = partial(smooth_max_grd_fn, alpha=alpha)
    cfun = partial(sum_eq_fn, ts=ts)
    cjac = partial(sum_eq_jac_fn, ts=ts)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def fast_stochastic_cd(h_in, n, ts, rng, n_steps=300000, delta_lo=0.005, delta_hi=0.25, temp_init=0.04, cool_rate=0.97, cool_period=5000):
    h = h_in.copy()
    h_rev = h[::-1].copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_cur = np.max(corr)
    temp = temp_init
    hot = (h > 0.01) & (h < 0.99)
    hot_idx = np.where(hot)[0]
    for step in range(n_steps):
        if len(hot_idx) < 2:
            break
        m = hot_idx[rng.randint(len(hot_idx))]
        m2 = hot_idx[rng.randint(len(hot_idx))]
        if m == m2:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        if h[m] > h[m2]:
            dm = -min(delta, h[m] * 0.8)
            dm2 = min(-dm, (1.0 - h[m2]) * 0.8)
        else:
            dm = min(delta, (1.0 - h[m]) * 0.8)
            dm2 = min(-dm, h[m2] * 0.8)
        if abs(dm) < 1e-7:
            continue
        old_c5 = c5_cur
        h_rev = h[::-1].copy()
        corr[m : m + n] += dm * (1.0 - h_rev)
        corr[n - 1 - m : 2 * n - 1 - m] -= dm * h
        h[m] += dm
        h[m2] += dm2
        h_rev = h[::-1].copy()
        corr[m2 : m2 + n] += dm2 * (1.0 - h_rev)
        corr[n - 1 - m2 : 2 * n - 1 - m2] -= dm2 * h
        c5_new = np.max(corr)
        if c5_new < old_c5:
            c5_cur = c5_new
            hot = (h > 0.01) & (h < 0.99)
            hot_idx = np.where(hot)[0]
        else:
            accept = np.exp(-(c5_new - old_c5) / max(temp, 1e-12))
            if rng.random() < accept:
                c5_cur = c5_new
                hot = (h > 0.01) & (h < 0.99)
                hot_idx = np.where(hot)[0]
            else:
                h[m] -= dm
                h[m2] -= dm2
                h_rev = h[::-1].copy()
                corr[m : m + n] -= dm * (1.0 - h_rev)
                corr[n - 1 - m : 2 * n - 1 - m] += dm * h
                corr[m2 : m2 + n] -= dm2 * (1.0 - h_rev)
                corr[n - 1 - m2 : 2 * n - 1 - m2] += dm2 * h
        if step % cool_period == 0 and step > 0:
            temp *= cool_rate
    h = project_to_sum(h, ts)
    return h, c5_cur

def grad_descent_step(h, n, ts, alpha, lr, n_inner=50):
    h = h.copy()
    for _ in range(n_inner):
        _, grad = smooth_max_grad(h, alpha)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new = smooth_max_grad(h_new, alpha)[0]
        obj_old = smooth_max_grad(h, alpha)[0]
        if obj_new < obj_old:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h

def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_to_sum(h_new, n_new / 2.0)
    return h_new

        starts.append(project_to_sum(h_mix, ts))
