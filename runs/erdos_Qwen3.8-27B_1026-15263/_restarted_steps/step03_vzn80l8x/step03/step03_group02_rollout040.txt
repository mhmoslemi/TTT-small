```python
def update_corr_after_swap(corr, h, i, j, n):
    """Update correlation array after swapping h[i]=1 to 0 and h[j]=0 to 1."""
    # Before swap: h[i]=1, h[j]=0
    # After swap: h[i]=0, h[j]=1
    
    # c[k] = sum_{valid p} h[p] * (1-h[p+lag])
    # where lag = k - (n-1), k in 0..2n-2
    
    # Effect of h[i]: 1 -> 0
    # For lags where i is the "first" element: c decreases by (1-h[i+lag])
    # For lags where i is the "second" element: c increases by h[i-lag] (was 0, now (1-0)=1 * h[i-lag])
    
    # Effect of h[j]: 0 -> 1
    # For lags where j is the "first" element: c increases by (1-h[j+lag])
    # For lags where j is the "second" element: c decreases by h[j-lag] (was h[j-lag]*1, now h[j-lag]*0)
    
    delta = np.zeros(2*n-1)
    # Lag = k - (n-1)
    # i as first: lag = L means i+L is in [0,n), so L in [-i, n-1-i]
    #   k = L + (n-1), so k in [n-1-i, 2n-2-i]
    #   values: (1-h[i+L]) = (1-h[i+L]) for L = -i, ..., n-1-i

    #   k = L + (n-1), so k in [i, i+n-1]
    #   values: h[i-L] for L = i-(n-1), ..., i
```

Let me now think about the continuous polishing step. The SLSQP approach from the previous program works but is slow for large n. Let me use a different approach: **projected subgradient with line search** or **FISTA** (fast iterative shrinkage-thresholding).

Actually, for the smooth max approximation, the problem is:
minimize f(h) = (1/α) log(sum_k exp(α * c_k(h)))
subject to: sum(h) = n/2, 0 ≤ h ≤ 1

This is a smooth, convex-in-spirit problem (though not actually convex due to the bilinear terms). I can use:
1. L-BFGS-B with the smooth max (fast, works well for smooth problems)
2. Projected gradient descent with backtracking line search

```python
def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -500, 0)
    vals = np.exp(shifted)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient of c_k with respect to h[i]
    # c_k = sum_p h[p] * (1-h[p+L]) where L = k-(n-1)
    # d(c_k)/d(h[i]) = (1-h[i+L]) * [valid] - h[i-L] * [valid]
    
    L_arr = np.arange(-(n-1), n)
    idx_ip = np.arange(n)[:, None] + L_arr[None, :]
    idx_im = np.arange(n)[:, None] - L_arr[None, :]
    
    m1 = (idx_ip >= 0) & (idx_ip < n)
    m2 = (idx_im >= 0) & (idx_im < n)
    ci = np.clip(idx_ip, 0, n-1)
    cj = np.clip(idx_im, 0, n-1)
    
    # grad[i] = sum_k w[k] * ((1-h[ci]) * m1 - h[cj] * m2)
    part1 = m1 * (1.0 - h[ci])  # shape (n, 2n-1)
    part2 = m2 * h[cj]          # shape (n, 2n-1)
    grad = (part1 - part2).T @ w  # shape (n,)
    
    return grad
```

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha, precomputed):

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary_from_scores(h, n, target_ones):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:target_ones]] = 1.0
    return h_bin

def is_prime(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True

def gen_quadratic_residue(n, rng):
    p = n + 2
    while not is_prime(p):
        p += 1
    seq = np.zeros(n)
    for i in range(n):
        x = (i + 1) % p
        if x != 0 and pow(x, (p - 1) // 2, p) == 1:
            seq[i] = 1.0
    return seq

def gen_random_binary(n, target_ones, rng):
    h = np.zeros(n)

def precompute_grad_indices(n):
    L_arr = np.arange(-(n - 1), n)
    i_arr = np.arange(n)
    idx_ip = i_arr[:, None] + L_arr[None, :]
    idx_im = i_arr[:, None] - L_arr[None, :]
    m1 = (idx_ip >= 0) & (idx_ip < n)
    m2 = (idx_im >= 0) & (idx_im < n)
    ci = np.clip(idx_ip, 0, n - 1)
    cj = np.clip(idx_im, 0, n - 1)
    return m1, m2, ci, cj

def smooth_max_obj(h, alpha, pre):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(corr.max())
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    ev = np.exp(shifted)
    return m_val + np.log(ev.sum()) / alpha

def smooth_max_grad(h, alpha, pre):
    n = len(h)
    m1, m2, ci, cj = pre
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(corr.max())
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    ev = np.exp(shifted)
    w = ev / ev.sum()
    part1 = m1 * (1.0 - h[ci])
    part2 = m2 * h[cj]
    grad = (part1 - part2).T @ w
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, pre):
    def obj(x):
        return smooth_max_obj(x, alpha, pre)
    def grad(x):
        return smooth_max_grad(x, alpha, pre)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
```
