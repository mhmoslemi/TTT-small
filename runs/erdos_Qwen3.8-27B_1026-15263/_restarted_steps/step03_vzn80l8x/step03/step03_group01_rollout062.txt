def binary_local_search(h_in, n, ts, rng, max_rounds=300, pool=16):
    h = (h_in >= 0.5).astype(float)
    h = project_to_sum(h, ts)
    c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(corr)
        active_thresh = max_c - 2.0
        active = np.where(corr >= active_thresh)[0]
        pressure = np.zeros(n)
        for j in active:
            k = j - (n - 1)
            if k >= 0:
                r = np.arange(0, n - k)
                pressure[r] += h[r + k]
                pressure[r + k] += (1.0 - h[r])
            else:
                mv = -k
                r = np.arange(mv, n)
                pressure[r] += (1.0 - h[r - mv])
                pressure[r - mv] += h[r]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_ones = ones[np.argsort(pressure[ones])[::-1][:pool]]
        low_zeros = zeros[np.argsort(pressure[zeros])[:pool]]
        n_rand = pool // 2
        if len(ones) > pool and len(zeros) > pool:
            rand_ones = rng.choice(ones, size=n_rand, replace=False)
            rand_zeros = rng.choice(zeros, size=n_rand, replace=False)
        else:
            rand_ones = np.array([], dtype=int)
            rand_zeros = np.array([], dtype=int)
        all_ones = np.unique(np.concatenate([top_ones, rand_ones]))
        all_zeros = np.unique(np.concatenate([low_zeros, rand_zeros]))
        best_imp = 0.0
        best_pair = None
        for oi in all_ones:
            for zj in all_zeros:
                h[oi] = 0.0
                h[zj] = 1.0
                nc = compute_c5(h)
                imp = c5 - nc
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (oi, zj)
                h[oi] = 1.0
                h[zj] = 0.0
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            c5 = compute_c5(h)
        else:
            max_rounds = rnd + 1
            break
    return h, c5

def binary_2swap_search(h_in, n, ts, rng, max_rounds=100, pool=8):
    h = (h_in >= 0.5).astype(float)
    h = project_to_sum(h, ts)
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(corr)
        active = np.where(corr >= max_c - 3.0)[0]
        pressure = np.zeros(n)
        for j in active:
            k = j - (n - 1)
            if k >= 0:
                r = np.arange(0, n - k)
                pressure[r] += h[r + k]
                pressure[r + k] += (1.0 - h[r])
            else:
                mv = -k
                r = np.arange(mv, n)
                pressure[r] += (1.0 - h[r - mv])
                pressure[r - mv] += h[r]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        top_ones = ones[np.argsort(pressure[ones])[::-1][:pool]]
        low_zeros = zeros[np.argsort(pressure[zeros])[:pool]]
        best_imp = 0.0
        best_quad = None
        for i1 in top_ones:
            for i2 in top_ones:
                if i1 == i2:
                    continue
                for j1 in low_zeros:
                    for j2 in low_zeros:
                        if j1 == j2:
                            continue
                        h[i1] = 0.0
                        h[i2] = 0.0
                        h[j1] = 1.0
                        h[j2] = 1.0
                        nc = compute_c5(h)
                        imp = c5 - nc
                        if imp > best_imp:
                            best_imp = imp
                            best_quad = (i1, i2, j1, j2)
                        h[i1] = 1.0
                        h[i2] = 1.0
                        h[j1] = 0.0
                        h[j2] = 0.0
        if best_quad is not None:
            i1, i2, j1, j2 = best_quad
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c5 = compute_c5(h)
        else:
            break
    return h, c5

def subgradient_exact(h0, n, ts, n_iter=5000, lr_init=0.05):
    h = h0.copy()
    lr = lr_init
    f_prev = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_cur = np.max(corr) * (2.0 / n)
        active = np.where(corr >= np.max(corr) - 0.5)[0]
        grad = np.zeros(n)
        for j in active:
            k = j - (n - 1)
            if k >= 0:
                grad[0:n-k] += (1.0 - h[k:n])
                grad[k:n] -= h[0:n-k]
            else:
                mv = -k
                grad[mv:n] += (1.0 - h[0:n-mv])
                grad[0:n-mv] -= h[mv:n]
        grad /= max(len(active), 1)
        grad *= (2.0 / n)
        h_new = project_to_sum(h - lr * grad, ts)
        f_new = compute_c5(h_new)
        if f_new < f_cur - 1e-15:
            h = h_new
            f_prev = f_new
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.7
            stall += 1
            if stall > 80 or lr < 1e-11:
                break
    return h

def kronecker_init(n, ts, rng):
    h = np.zeros(n)
    alpha = 0.5 * (1.0 + np.sqrt(5.0))
    positions = (np.arange(n // 2) * alpha + rng.uniform(0, 1)).astype(int) % n
    positions = np.unique(positions)
    if len(positions) < n // 2:
        remaining = np.setdiff1d(np.arange(n), positions)
        extra = rng.choice(remaining, size=n // 2 - len(positions), replace=False)
        positions = np.concatenate([positions, extra])
    elif len(positions) > n // 2:
        positions = positions[:n // 2]
    h[positions] = 1.0
    return project_to_sum(h, ts)

def upsample_periodic(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def tleft():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass
    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [30, 100, 400, 1500, 5000]:
        if tleft() < 30:
            break
        h_new = slsqp_opt(best_h, n1, ts1, alpha, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    if tleft() > 40:
        h_pg = proj_grad_descend(best_h, n1, ts1, 3000, n_iter=2000, lr_init=0.02)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    if tleft() > 60:
        h_sg = subgradient_exact(best_h, n1, ts1, n_iter=3000, lr_init=0.04)
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    # Binary local search at current resolution
    if tleft() > 60:
        h_bin, c5_bin = binary_local_search(best_h, n1, ts1, rng, max_rounds=200, pool=14)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    if tleft() > 40:
        h_2s, c5_2s = binary_2swap_search(best_h, n1, ts1, rng, max_rounds=50, pool=6)
        if c5_2s < best_c5:
            best_c5 = c5_2s
            best_h = h_2s.copy()
    if tleft() > 30:
        h_bin2, c5_bin2 = binary_local_search(best_h, n1, ts1, rng, max_rounds=150, pool=12)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
    # SLSQP polish after binary
    if tleft() > 40:
        for alpha in [2000, 8000]:
            if tleft() < 25:
                break
            h_pol = slsqp_opt(best_h, n1, ts1, alpha, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    # Try nearby resolutions
    for n_try in [best_n - 1, best_n + 1, best_n - 2, best_n + 2, best_n - 3, best_n + 3]:
        if tleft() < 40:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        h_up = upsample_periodic(best_h, best_n, n_try)
        h_up = project_to_sum(h_up, ts_try)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_try
        for alpha in [200, 1000, 4000]:
            if tleft() < 20:
                break
            h_o = slsqp_opt(h_up, n_try, ts_try, alpha, maxiter=400)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = n_try
            h_up = h_o
        if tleft() > 20:
            h_b, c5_b = binary_local_search(h_up, n_try, ts_try, rng, max_rounds=80, pool=10)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_try
    # Higher resolutions
    for n_tr in [best_n + 32, best_n + 64, best_n + 96]:
        if tleft() < 80:
            break
        if n_tr % 2 != 0:
            n_tr += 1
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr
        for alpha in [100, 500, 2000]:
            if tleft() < 30:
                break
            h_o = slsqp_opt(h_up, n_tr, ts_tr, alpha, maxiter=500)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = n_tr
            h_up = h_o
        if tleft() > 30:
            h_pg = proj_grad_descend(h_up, n_tr, ts_tr, 3000, n_iter=1500, lr_init=0.015)
            c5_pg = compute_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n_tr
        if tleft() > 40:
            h_b, c5_b = binary_local_search(h_up, n_tr, ts_tr, rng, max_rounds=100, pool=12)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_tr
    # Alternative initializations at best resolution
    n_alt = best_n
    ts_alt = n_alt / 2.0
    for trial in range(8):
        if tleft() < 35:
            break
        alpha_b = 0.5 * (1.0 + np.sqrt(5.0)) + trial * 0.137
        h_k = kronecker_init(n_alt, ts_alt, rng)
        c5_k = compute_c5(h_k)
        if c5_k < best_c5:
            best_c5 = c5_k
            best_h = h_k.copy()
        for alpha in [200, 1000]:
            if tleft() < 20:
                break
            h_o = slsqp_opt(h_k, n_alt, ts_alt, alpha, maxiter=400)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            h_k = h_o
        if tleft() > 15:
            h_b, c5_b = binary_local_search(h_k, n_alt, ts_alt, rng, max_rounds=60, pool=10)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
    # Frequency perturbation + re-optimize
    for trial in range(10):
        if tleft() < 30:
            break
        amp = 0.06 * (1.0 - trial / 12.0) + 0.01
        n_harm = max(3, best_n // 30)
        mean_h = np.mean(best_h)
        H = np.fft.rfft(best_h - mean_h)
        upper = min(n_harm, best_n // 2 - 1)
        for kk in range(1, upper + 1):
            H[kk] += amp * (rng.randn() + 1j * rng.randn())
        h_pert = np.fft.irfft(H, n=best_n) + mean_h
        h_pert = project_to_sum(h_pert, best_n / 2.0)
        c5_p = compute_c5(h_pert)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pert.copy()
        for alpha in [300, 1500]:
            if tleft() < 20:
                break
            h_o = slsqp_opt(h_pert, best_n, best_n / 2.0, alpha, maxiter=400)
            c5_o = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            h_pert = h_o
    # Final aggressive binary search
    if tleft() > 50:
        h_fin, c5_fin = binary_local_search(best_h, best_n, best_n / 2.0, rng, max_rounds=300, pool=18)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if tleft() > 30:
        h_2f, c5_2f = binary_2swap_search(best_h, best_n, best_n / 2.0, rng, max_rounds=80, pool=7)
        if c5_2f < best_c5:
            best_c5 = c5_2f
            best_h = h_2f.copy()
    # Final SLSQP polish
    if tleft() > 30:
        for alpha in [5000, 15000, 50000]:
            if tleft() < 15:
                break
            h_f = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=1500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
    # Final subgradient
    if tleft() > 25:
        h_sf = subgradient_exact(best_h, best_n, best_n / 2.0, n_iter=2000, lr_init=0.02)
        c5_sf = compute_c5(h_sf)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()
    # Final binary pass
    if tleft() > 30:
        h_bf, c5_bf = binary_local_search(best_h, best_n, best_n / 2.0, rng, max_rounds=200, pool=14)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
    # Validation
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
