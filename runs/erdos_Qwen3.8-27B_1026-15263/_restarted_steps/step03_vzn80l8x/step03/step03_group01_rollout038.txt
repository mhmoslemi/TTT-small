def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m_val = np.max(scaled)
    shifted = alpha * (scaled - m_val)
    max_shifted = np.max(shifted)
    exp_v = np.exp(shifted - max_shifted)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_shifted + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    lags = np.arange(2 * n - 1) - (n - 1)
    fi = nj + lags[None, :]
    si = nj - lags[None, :]
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = dx * (vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :]).sum(axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def active_lag_direction(h, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    max_idx = np.argmax(corr)
    l = max_idx - (n - 1)
    d = np.zeros(n)
    for j in range(n):
        jp = j + l
        jm = j - l
        if 0 <= jp < n:
            d[j] += (1.0 - h[jp])
        if 0 <= jm < n:
            d[j] -= h[jm]
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm > 1e-15:
        d /= norm
