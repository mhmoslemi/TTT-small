                h[:] = orig
        else:
            blk = rng.randint(5, 20)
            if blk >= n:
                continue
            i = rng.randint(n - blk)
            h[i:i + blk] = 1.0 - h[i:i + blk]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0.0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i:i + blk] = 1.0 - h[i:i + blk]
        if no_improve > 25000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.0, 0.025)
            no_improve = 0
        T = max(T * 0.99998, min_T)
    return best_h, best_c5

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start
    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            ihv = initial_h_values
            best_h = np.array(ihv, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    # Phase 1: SLSQP chain with increasing alpha
    p1_end = t_start + 0.28 * budget_s
    alphas_p1 = [20, 50, 150, 500, 2000]
    for alpha in alphas_p1:
        if elapsed() > p1_end:
            break
        h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=1500, ftol=1e-15)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        else:
            if c5_opt > best_c5 + 1e-8:
                break

    # Phase 2: Face descent
    p2_end = t_start + 0.38 * budget_s
    h_fd, c5_fd = face_descent(best_h, ts_main, eps=1.0, max_iter=5000, lr=0.004)
    if c5_fd < best_c5:
        best_c5 = c5_fd
        best_h = h_fd.copy()

    # Phase 3: Binary SA with diverse moves
    p3_end = t_start + 0.60 * budget_s
    if remaining() > 60:
        h_bin = make_binary(best_h, n_main, int(ts_main), rng)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_sa, c5_sa = binary_sa(h_bin, n_main, int(ts_main), p3_end, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: SLSQP after SA with high alpha
    p4_end = t_start + 0.78 * budget_s
    if remaining() > 40:
        alphas_p4 = [100, 500, 2000, 10000, 50000]
        for alpha in alphas_p4:
            if elapsed() > p4_end:
                break
            h_opt = slsqp_optimize(best_h, n_main, ts_main, alpha, maxiter=2000, ftol=1e-15)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Tight face descent
    p5_end = t_start + 0.84 * budget_s
    if remaining() > 30:
        h_fd2, c5_fd2 = face_descent(best_h, ts_main, eps=0.1, max_iter=5000, lr=0.002)
        if c5_fd2 < best_c5:
            best_c5 = c5_fd2
            best_h = h_fd2.copy()
        h_fd3, c5_fd3 = face_descent(best_h, ts_main, eps=0.01, max_iter=3000, lr=0.001)
        if c5_fd3 < best_c5:
            best_c5 = c5_fd3
            best_h = h_fd3.copy()

    # Phase 6: Multi-start with diverse perturbations
    p6_end = t_start + 0.94 * budget_s
    perturbations = []
    for trial in range(8):
        if remaining() < 25:
            break
        ptype = trial % 4
        if ptype == 0:
            h_shake = best_h + rng.uniform(-0.04, 0.04, n_main)
        elif ptype == 1:
            h_shake = best_h.copy()
            blen = rng.randint(8, 30)
            bstart = rng.randint(0, n_main - blen)
            h_shake[bstart:bstart + blen] = 1.0 - h_shake[bstart:bstart + blen]
        elif ptype == 2:
            freq = rng.uniform(0.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.08 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n_main) + phase)
            h_shake = best_h + pert
        else:
            # Swap a random contiguous segment with another
            h_shake = best_h.copy()
            blen = rng.randint(5, 15)
            i = rng.randint(n_main - blen)
            j = rng.randint(n_main - blen)
            if abs(i - j) > blen:
                h_shake[i:i + blen], h_shake[j:j + blen] = h_shake[j:j + blen].copy(), h_shake[i:i + blen].copy()
        h_shake = project_to_sum(h_shake, ts_main)
        for alpha in [1000, 10000]:
            if remaining() < 20:
                break
            h_opt = slsqp_optimize(h_shake, n_main, ts_main, alpha, maxiter=1200, ftol=1e-14)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Higher resolution attempt
    if remaining() > 80:
        for n_try in [360, 480]:
            if remaining() < 60:
                break
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            h_interp = interpolated_h(best_h, best_n, n_try)
            h_interp = project_to_sum(h_interp, ts_try)
            c5_interp = compute_c5(h_interp)
            h_cur = h_interp
            for alpha in [100, 1000, 10000]:
                if remaining() < 30:
                    break
                h_opt = slsqp_optimize(h_cur, n_try, ts_try, alpha, maxiter=1200, ftol=1e-14)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try
                    h_cur = h_opt
                else:
                    h_cur = h_opt

    # Final polish with very high alpha
    if remaining() > 30:
        h_final = slsqp_optimize(best_h, best_n, best_n / 2.0, 100000, maxiter=2000, ftol=1e-15)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
