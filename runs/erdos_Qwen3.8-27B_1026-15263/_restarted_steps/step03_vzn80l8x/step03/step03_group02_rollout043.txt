def incremental_swap(h, p, q, corr, n):
    """
    Compute new correlation after swapping h[p] and h[q].
    Assumes h[p] != h[q] (one is 0, other is 1).
    h: current binary array
    p, q: positions to swap (h[p]=1, h[q]=0 or vice versa)
    corr: current full correlation array (length 2n-1)
    Returns: new_corr (length 2n-1)
    """
    new_corr = corr.copy()
    s = np.arange(-(n-1), n)  # shifts, length 2n-1
    # s[k] = k - (n-1), so index k corresponds to shift s[k]
    
    # Term from i=p: h[p] changes
    # If h[p] was 1 and becomes 0: change = -(1-h[p-s]) for valid p-s
    # If h[p] was 0 and becomes 1: change = +(1-h[p-s]) for valid p-s
    sign_p = 1.0 if h[p] == 0 else -1.0  # +1 if h[p] goes 0->1, -1 if 1->0
    valid_p_s = (p - s >= 0) & (p - s < n)
    new_corr += sign_p * (1 - h[np.clip(p - s, 0, n-1)]) * valid_p_s
    
    # Term from i=q: h[q] changes
    sign_q = 1.0 if h[q] == 0 else -1.0
    valid_q_s = (q - s >= 0) & (q - s < n)
    new_corr += sign_q * (1 - h[np.clip(q - s, 0, n-1)]) * valid_q_s
    
    # Term from i=p+s: (1-h[i-s]) = (1-h[p]) changes
    # If h[p] was 1: (1-h[p]) goes 0->1, change = +h[p+s]
    # If h[p] was 0: (1-h[p]) goes 1->0, change = -h[p+s]
    sign_ps = -sign_p  # opposite sign because (1-h) flips opposite to h
    valid_ps = (p + s >= 0) & (p + s < n)
    # Exclude s=0 (i=p+s=p, already counted) and p+s=q (overlaps with i=q term)
    valid_ps &= (s != 0) & (p + s != q)
    new_corr += sign_ps * h[np.clip(p + s, 0, n-1)] * valid_ps
    
    # Term from i=q+s: (1-h[i-s]) = (1-h[q]) changes
    sign_qs = -sign_q
    valid_qs = (q + s >= 0) & (q + s < n)
    valid_qs &= (s != 0) & (q + s != p)
    new_corr += sign_qs * h[np.clip(q + s, 0, n-1)] * valid_qs
    
    return new_corr

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

    valid_ps = (p_ps >= 0) & (p_ps < n) & (s != 0) & (p_ps != q)
    valid_qs = (q_qs >= 0) & (q_qs < n) & (s != 0) & (q_qs != p)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)

        return project_to_sum(np.clip(h0.copy(), 0, 1), target_sum)
