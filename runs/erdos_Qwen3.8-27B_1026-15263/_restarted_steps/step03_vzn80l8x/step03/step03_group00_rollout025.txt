def improved_coord_desc(h, target_sum, max_iters=5000):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = compute_corr(h)
        # Use softmax weighting over lags
        alpha = 50.0
        shifted = alpha * (corr - np.max(corr))
        shifted -= np.max(shifted)
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Compute weighted gradient
        g = np.zeros(n)
        for lag_idx in range(len(corr)):
            if w[lag_idx] < 1e-8:
                continue
            lag = lag_idx - (n - 1)
            g += w[lag_idx] * corr_gradient(h, lag)
        
        # Find best pair to move
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-14:
            break
        
        # Small line search
        best_move = 0
        best_val = c5
        for frac in [0.01, 0.03, 0.1, 0.3, 0.5]:
            max_move = min(1.0 - h[i_inc], h[i_dec])
            move = frac * max_move
            if move < 1e-12:
                continue
            h_test = h.copy()
            h_test[i_inc] += move
            h_test[i_dec] -= move
            h_test = project_to_sum(h_test, target_sum)
            val = compute_c5(h_test)
            if val < best_val:
                best_val = val
                best_move = move
        
        if best_move == 0:
            break
        
        h[i_inc] += best_move
        h[i_dec] -= best_move
        h = project_to_sum(h, target_sum)
        c5 = best_val
    
    return h, c5

    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]

    second_idx = j_idx + m_idx - n + 1

    valid_s = (second_idx >= 0) & (second_idx < n)

    cs = valid_s * h[sc] * w[None, :]

    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def _slsqp_fun(x):
    return smooth_max_obj_grad(x, _g_alpha)[0]

def _slsqp_jac(x):
    return smooth_max_obj_grad(x, _g_alpha)[1]

def _slsqp_eq(x):
    return np.sum(x) - _g_target

def _slsqp_eq_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    global _g_alpha, _g_target, _g_n
    _g_alpha = alpha
    _g_target = target_sum
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_fun, h0.copy(), method='SLSQP',
                       jac=_slsqp_jac, bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def proj_grad_desc(h, target_sum, max_iters=3000, alpha=500):
    n = len(h)
    h = h.copy()
    lr = 0.005
    best_obj, _ = smooth_max_obj_grad(h, alpha)
    for it in range(max_iters):
        obj, grad = smooth_max_obj_grad(h, alpha)
        if obj - best_obj > 1.0:
            break
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-9:
                break
    return h, compute_c5(h)

def coord_desc(h, target_sum, max_iters=5000, alpha_w=50.0):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        shifted = alpha_w * (corr - np.max(corr))
        shifted -= np.max(shifted)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(2 * n - 1)[None, :]
        first_idx = j_idx - m_idx + n - 1
        second_idx = j_idx + m_idx - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        cf = valid_f * (1.0 - h[fc]) * w[None, :]
        cs = valid_s * h[sc] * w[None, :]
        g = np.sum(cf - cs, axis=1)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-14:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        best_move = 0.0
        best_val = c5
        for frac in [0.01, 0.03, 0.1, 0.3]:
            move = frac * max_move
            if move < 1e-12:
                continue
            h_test = h.copy()
            h_test[i_inc] += move
            h_test[i_dec] -= move
            h_test = project_to_sum(h_test, target_sum)
            val = compute_c5(h_test)
            if val < best_val:
                best_val = val
                best_move = move
        if best_move == 0.0:
            stale += 1
            if stale > 100:
                break
            continue
        stale = 0
        h[i_inc] += best_move
        h[i_dec] -= best_move
        h = project_to_sum(h, target_sum)
        c5 = best_val
    return h, c5

def fourier_to_h(params, n):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, params[0])
    n_freqs = (len(params) - 1) // 2
    for k in range(n_freqs):
        h += params[1 + 2 * k] * np.cos(np.pi * k * x)
        h += params[2 + 2 * k] * np.sin(np.pi * k * x)
    return np.clip(h, 0.0, 1.0)

def _fourier_obj(params):
    h = fourier_to_h(params, _g_no)
    h = project_to_sum(h, _g_tso)
    return compute_c5(h)

def pc_to_h(heights, n):
    M = len(heights)
    pts_per = n // M
    h = np.repeat(heights, pts_per)
    if len(h) < n:
        h = np.concatenate([h, np.full(n - len(h), heights[-1])])
    return np.clip(h[:n], 0.0, 1.0)

def _pc_obj(params):
    h = pc_to_h(params, _g_no)
    h = project_to_sum(h, _g_tso)
    return compute_c5(h)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def extract_fourier(h, n_freq):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    coeffs = [float(np.mean(h))]
    for k in range(1, n_freq + 1):
        a = float(np.mean(h * np.cos(np.pi * k * x)))
        s = float(np.mean(h * np.sin(np.pi * k * x)))
        coeffs.extend([a, s])
    return np.array(coeffs)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_no, _g_tso
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_h[best_n // 4:3 * best_n // 4] = 1.0
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    n0 = best_n
    ts0 = n0 / 2.0
    _g_no = n0
    _g_tso = ts0

    for alpha in [100, 500, 2000]:
        if time_left() < 30:
            break
        h_new = slsqp_optimize(best_h, n0, ts0, alpha, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 180:
        n_freq = 7
        n_params = 1 + 2 * n_freq
        bounds_f = [(0.2, 0.8)]
        for k in range(n_freq):
            scale = 0.5 / (1.0 + 0.3 * k)
            bounds_f.extend([(-scale, scale)] * 2)
        try:
            de_res = differential_evolution(_fourier_obj, bounds_f, seed=seed,
                                             maxiter=100, popsize=20,
                                             tol=1e-9, mutation=(0.3, 1.2),
                                             recombination=0.8,
                                             polish=False, init='latinhypercube')
            h_de = fourier_to_h(de_res.x, n0)
            h_de = project_to_sum(h_de, ts0)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
        except Exception:
            pass

    if time_left() > 150:
        M = 24
        bounds_pc = [(0.0, 1.0)] * M
        try:
            de_res2 = differential_evolution(_pc_obj, bounds_pc, seed=seed + 100,
                                              maxiter=80, popsize=20,
                                              tol=1e-9, mutation=(0.3, 1.2),
                                              recombination=0.8,
                                              polish=False, init='latinhypercube')
            h_pc = pc_to_h(de_res2.x, n0)
            h_pc = project_to_sum(h_pc, ts0)
            c5_pc = compute_c5(h_pc)
            if c5_pc < best_c5:
                best_c5 = c5_pc
                best_h = h_pc.copy()
        except Exception:
            pass

    if time_left() > 120:
        fourier_center = extract_fourier(best_h, 7)
        n_params2 = 1 + 14
        bounds_f2 = [(0.2, 0.8)]
        for k in range(7):
            scale = 0.4 / (1.0 + 0.2 * k)
            bounds_f2.extend([(-scale, scale)] * 2)
        init_pop = fourier_center + rng.normal(0, 0.05, n_params2)
        init_pop = np.clip(init_pop, [b[0] for b in bounds_f2], [b[1] for b in bounds_f2])
        n_init = min(20, int((bounds_f2[0][1] - bounds_f2[0][0]) / 0.01))
        init_mat = np.tile(init_pop, (n_init, 1))
        init_mat += rng.normal(0, 0.03, init_mat.shape)
        for i in range(init_mat.shape[1]):
            init_mat[:, i] = np.clip(init_mat[:, i], bounds_f2[i][0], bounds_f2[i][1])
        try:
            de_res3 = differential_evolution(_fourier_obj, bounds_f2, seed=seed + 200,
                                              maxiter=60, popsize=20,
                                              tol=1e-9, mutation=(0.2, 0.9),
                                              recombination=0.9,
                                              polish=False, init=init_mat)
            h_de3 = fourier_to_h(de_res3.x, n0)
            h_de3 = project_to_sum(h_de3, ts0)
            c5_de3 = compute_c5(h_de3)
            if c5_de3 < best_c5:
                best_c5 = c5_de3
                best_h = h_de3.copy()
        except Exception:
            pass

    for alpha in [200, 1000, 5000]:
        if time_left() < 60:
            break
        h_new = slsqp_optimize(best_h, n0, ts0, alpha, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time_left() > 80:
        h_pg, c5_pg = proj_grad_desc(best_h, ts0, max_iters=2000, alpha=1000)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if time_left() > 80:
        h_cd, c5_cd = coord_desc(best_h, ts0, max_iters=3000, alpha_w=50.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    for n_tr in [600, 720, 840]:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        h_int = interp_to_n(best_h, n_tr)
        h_int = project_to_sum(h_int, ts_tr)
        c5_int = compute_c5(h_int)
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_int.copy()
            best_n = n_tr
        for alpha in [200, 1000, 3000]:
            if time_left() < 40:
                break
            h_opt = slsqp_optimize(h_int, n_tr, ts_tr, alpha, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_int = h_opt
        if time_left() > 50:
            h_cd, c5_cd = coord_desc(h_int, ts_tr, max_iters=2000, alpha_w=50.0)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr

    n_cur = best_n
    ts_cur = n_cur / 2.0
    for trial in range(10):
        if time_left() < 40:
            break
        noise = 0.02 + 0.04 * (1.0 - trial / 10.0)
        h_shake = best_h + rng.uniform(-noise, noise, n_cur)
        h_shake = project_to_sum(h_shake, ts_cur)
        h_cd, c5_cd = coord_desc(h_shake, ts_cur, max_iters=2000, alpha_w=30.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() < 30:
            break
        h_opt = slsqp_optimize(h_cd, n_cur, ts_cur, 2000, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    for alpha in [5000, 20000, 80000]:
        if time_left() < 20:
            break
        h_pol = slsqp_optimize(best_h, n_cur, ts_cur, alpha, maxiter=1000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 30:
        h_final, c5_final = coord_desc(best_h, n_cur, ts_cur, max_iters=15000, alpha_w=100.0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if time_left() > 20:
        h_pol2 = slsqp_optimize(best_h, n_cur, ts_cur, 100000, maxiter=1500)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    best_h = project_to_sum(best_h, n_cur / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
