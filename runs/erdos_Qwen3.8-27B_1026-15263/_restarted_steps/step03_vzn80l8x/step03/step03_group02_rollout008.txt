import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            adj = diff / np.sum(mask)
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
    # Final fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff)):]
            for i in idx:
                need = min(diff, 1.0 - h[i])
                h[i] += need
                diff -= need
        else:
            idx = np.argsort(h)[int(abs(diff)):]
            for i in idx:
                need = min(-diff, h[i])
                h[i] -= need
                diff += need
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target_sum:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def sa_binary(h_start, n, target_sum, t_end, rng, verbose=False):
    """Improved simulated annealing for binary solutions."""
    h = h_start.copy()
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))  # without dx factor for comparison
    best_c5 = c5_curr
    best_h = h.copy()
    
    T = 0.8  # initial temperature (in units of correlation value)
    T_min = 0.1
    cool_rate = 0.999995  # per iteration
    no_improve = 0
    improve_threshold = 40000
    n_ones = target_sum
    
    ones_set = set(np.where(h == 1.0)[0].tolist())
    
    while time.time() < t_end:
        # Choose move type
        if rng.random() < 0.85:
            # Single swap: exchange one 1 with one 0
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            # Swap
            h[i], h[j] = h[j], h[i]
            if i in ones_set:
                ones_set.remove(i)
                ones_set.add(j)
            else:
                ones_set.remove(j)
                ones_set.add(i)
            
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            
            if delta < 0 or (delta / max(T, 1e-10)) + np.log(max(rng.random(), 1e-300)) < 0:
                c5_curr = c5_new
                corr = corr_new
                if c5_new < best_c5 - 1e-6:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Revert
                h[i], h[j] = h[j], h[i]
                if i in ones_set:
                    ones_set.remove(i)
                    ones_set.add(j)
                else:
                    ones_set.remove(j)
                    ones_set.add(i)
        else:
            # Double swap: exchange two 1s with two 0s
            if len(ones_set) < 2 or n - len(ones_set) < 2:
                continue
            ones_list = list(ones_set)
            zeros_list = [k for k in range(n) if k not in ones_set]
            if len(ones_list) < 2 or len(zeros_list) < 2:
                continue
            i1, i2 = rng.choice(len(ones_list), size=2, replace=False)
            j1, j2 = rng.choice(len(zeros_list), size=2, replace=False)
            a, b = ones_list[i1], ones_list[i2]
            c, d = zeros_list[j1], zeros_list[j2]
            if len({a, b, c, d}) < 4:
                continue
            
            h[a] = 0.0; h[b] = 0.0; h[c] = 1.0; h[d] = 1.0
            ones_set.discard(a); ones_set.discard(b); ones_set.add(c); ones_set.add(d)
            
            corr_new = compute_corr(h)
            c5_new = float(np.max(corr_new))
            delta = c5_new - c5_curr
            
            if delta < 0 or (delta / max(T, 1e-10)) + np.log(max(rng.random(), 1e-300)) < 0:
                c5_curr = c5_new
                corr = corr_new
                if c5_new < best_c5 - 1e-6:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[a] = 1.0; h[b] = 1.0; h[c] = 0.0; h[d] = 0.0
                ones_set.discard(c); ones_set.discard(d); ones_set.add(a); ones_set.add(b)
        
        T = max(T * cool_rate, T_min)
        
        if no_improve > improve_threshold:
            # Restart from best with perturbed temperature
            h = best_h.copy()
            # Randomly flip a few positions
            n_flips = rng.randint(2, 6)
            for _ in range(n_flips):
                flip_pos = rng.randint(n)
                if h[flip_pos] == 1.0:
                    # Find a 0 to swap with
                    zeros = [k for k in range(n) if h[k] == 0.0]
                    if zeros:
                        swap_with = rng.choice(zeros)
                        h[flip_pos] = 0.0
                        h[swap_with] = 1.0
                        ones_set.remove(flip_pos)
                        ones_set.add(swap_with)
                else:
                    ones = [k for k in ones_set]
                    if ones:
                        swap_with = rng.choice(ones)
                        h[flip_pos] = 1.0
                        h[swap_with] = 0.0
                        ones_set.remove(swap_with)
                        ones_set.add(flip_pos)
            c5_curr = float(np.max(compute_corr(h)))
            T = min(T * 2.0, 1.0)
            no_improve = 0
    
    return best_h, best_c5

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter, t_check):
    """L-BFGS-B optimization with penalty for sum constraint."""
    penalty = 100.0
    
    def f_g(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        obj = m_val + np.log(np.sum(exp_vals)) / alpha
        pen = penalty * (np.sum(h) - target_sum) ** 2
        total = obj + pen
        
        # Gradient
        j = np.arange(n)[:, None]
        k = np.arange(2 * n - 1)[None, :]
        first_idx = j - k + n - 1
        second_idx = j + k - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        cf = valid_f * (1.0 - h[fc]) * w[None, :]
        cs = valid_s * h[sc] * w[None, :]
        grad_smooth = np.sum(cf - cs, axis=1)
        grad_pen = 2 * penalty * (np.sum(h) - target_sum) * np.ones(n)
        grad = grad_smooth + grad_pen
        
        # Clip gradient for out-of-bounds
        grad[x < 0] = 0
        grad[x > 1] = 0
        
        if callable(t_check) and t_check():
            raise ValueError("timeout")
        return total, grad
    
    try:
        res = minimize(f_g, h0.copy(), method='L-BFGS-B',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_sum(h_opt, target_sum)
        return h_opt
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
