    step = 0
    while time.time() < t_end:
        step += 1
        mt = step % 5
        if mt == 0:
            if len(ones_cache) == 0 or len(zeros_cache) == 0:
                continue
            i = ones_cache[rng.randint(len(ones_cache))]
            j = zeros_cache[rng.randint(len(zeros_cache))]
            h[i], h[j] = h[j], h[i]
            ones_cache = np.where(h == 1.0)[0]
            zeros_cache = np.where(h == 0.0)[0]
        elif mt == 1:
            if len(ones_cache) == 0:
                continue
            i = ones_cache[rng.randint(len(ones_cache))]
            j = i + 1 if rng.randint(2) else i - 1
            if j < 0 or j >= n or h[j] != 0.0:
                continue
            h[i], h[j] = h[j], h[i]
            ones_cache = np.where(h == 1.0)[0]
            zeros_cache = np.where(h == 0.0)[0]
        elif mt == 2:
            blen = rng.randint(2, 8)
            start = rng.randint(max(1, n - blen))
            h[start:start + blen] = h[start:start + blen][::-1]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            if delta < 0 or (delta / T) + np.log(rng.uniform()) < 0.0:
                c5_curr = c5_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[start:start + blen] = h[start:start + blen][::-1]
            continue
        elif mt == 3:
            nswaps = rng.randint(2, 5)
            for _ in range(nswaps):
                if len(ones_cache) == 0 or len(zeros_cache) == 0:
                    break
                i = ones_cache[rng.randint(len(ones_cache))]
                j = zeros_cache[rng.randint(len(zeros_cache))]
                h[i], h[j] = h[j], h[i]
            ones_cache = np.where(h == 1.0)[0]
            zeros_cache = np.where(h == 0.0)[0]
        else:
            blen = rng.randint(3, 10)
            start = rng.randint(max(1, n - blen))
            block = h[start:start + blen].copy()
            if np.all(block == 0.0) or np.all(block == 1.0):
                continue
            new_start = rng.randint(max(1, n - blen))
            if abs(new_start - start) < 2:
                continue
            old_region = h[new_start:new_start + blen].copy()
            h[new_start:new_start + blen] = block
            h[start:start + blen] = old_region
            ones_cache = np.where(h == 1.0)[0]
            zeros_cache = np.where(h == 0.0)[0]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta / T) + np.log(rng.uniform()) < 0.0:
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if mt == 0:
                h[i], h[j] = h[j], h[i]
                ones_cache = np.where(h == 1.0)[0]
                zeros_cache = np.where(h == 0.0)[0]
            elif mt == 1:
                h[i], h[j] = h[j], h[i]
                ones_cache = np.where(h == 1.0)[0]
                zeros_cache = np.where(h == 0.0)[0]
            elif mt == 3:
                for _ in range(nswaps):
                    if len(ones_cache) == 0 or len(zeros_cache) == 0:
                        break
                    oi = np.where(h == 1.0)[0]
                    oz = np.where(h == 0.0)[0]
                    ii = oi[rng.randint(len(oi))]
                    jj = oz[rng.randint(len(oz))]
                    h[ii], h[jj] = h[jj], h[ii]
                ones_cache = np.where(h == 1.0)[0]
                zeros_cache = np.where(h == 0.0)[0]
            elif mt == 4:
                tmp = h[new_start:new_start + blen].copy()
                h[new_start:new_start + blen] = old_region
                h[start:start + blen] = tmp
                ones_cache = np.where(h == 1.0)[0]
                zeros_cache = np.where(h == 0.0)[0]
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 3.0, 0.012)
            no_improve = 0
            frac = epoch(rand_val)
            for jj in range(n):
                if rng.uniform() < frac:
                    h[jj] = 1.0 - h[jj]
            h = make_binary(h, n, target_sum, rng)
            c5_curr = compute_c5(h)
        T = max(T * decay, T_min)
    return best_h, best_c5

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter, t_end, penalty=50.0):
    def f(x):
        base, _ = c5_smooth_and_grad(x, alpha)
        return base + penalty * (np.sum(x) - target_sum) ** 2
    def g(x):
        _, grad = c5_smooth_and_grad(x, alpha)
        return grad + 2.0 * penalty * (np.sum(x) - target_sum) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 40})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, t_end):
    def f(x):
        return c5_smooth_and_grad(x, alpha)[0]
    def g(x):
        return c5_smooth_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

    return 1.0

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n_main = best_n
    ts_main = n_main / 2.0
    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    h_sa_start = h_bin.copy()
    sa_end1 = t_start + 0.40 * budget_s
    h_sa, c5_sa = binary_sa(h_sa_start, n_main, int(ts_main), sa_end1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin2 = compute_c5(h_bin2)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()
    grad_end1 = t_start + 0.70 * budget_s
    alphas = [30, 100, 400, 2000, 10000]
    for alpha in alphas:
        if time.time() > grad_end1:
            break
        remaining_frac = (grad_end1 - time.time()) / max(grad_end1 - t_start, 1.0)
        maxiter = min(2000, int(300 + 1500 * remaining_frac))
        h_pol = lbfgs_optimize(best_h, n_main, ts_main, alpha, maxiter, time.time() + 60.0)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_bin2 = make_binary(h_pol, n_main, int(ts_main), rng)
        c5_b2 = compute_c5(h_bin2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2.copy()
    sa_end2 = t_start + 0.82 * budget_s
    if time.time() < sa_end2:
        h_bin3 = make_binary(best_h, n_main, int(ts_main), rng)
        h_sa2, c5_sa2 = binary_sa(h_bin3, n_main, int(ts_main), sa_end2, rng, T_init=0.003, T_min=0.0005)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
    grad_end2 = t_start + 0.92 * budget_s
    if time.time() < grad_end2:
        h_pol2 = slsqp_optimize(best_h, n_main, ts_main, 50000, 800, grad_end2)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
    for n_try in [360, 480]:
        if time.time() > budget_s - 80:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        h_interp = interpolated_h(best_h, best_n, n_try)
        h_interp = project_to_sum(h_interp, ts_try)
        c5_interp = compute_c5(h_interp)
        h_try = h_interp.copy()
        for alpha_sq in [50, 500, 5000]:
            if time.time() > budget_s - 30:
                break
            h_try = lbfgs_optimize(h_try, n_try, ts_try, alpha_sq, 500, time.time() + 40.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
