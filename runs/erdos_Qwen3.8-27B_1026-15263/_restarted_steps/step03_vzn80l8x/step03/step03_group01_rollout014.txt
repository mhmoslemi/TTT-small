    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    """Log-sum-exp approximation of max(corr * dx) with gradient."""

    vals = corr * dx

    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m_val = np.max(vals)
    shifted = alpha * (vals - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    lse_sum = max_sh + np.log(np.sum(exp_v))
    obj = m_val + lse_sum / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = dx * np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def hard_max_obj_and_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    obj = np.max(vals)
    idx_max = np.where(vals == obj)[0]
    w = np.zeros(len(vals))
    w[idx_max] = 1.0 / len(idx_max)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = dx * np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def lbfgs_opt(h0, n, ts, alpha, maxiter=3000):
    def f(x):
        return lse_obj_and_grad(x, alpha)[0]
    def g(x):
        return lse_obj_and_grad(x, alpha)[1]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='L-BFGS-B', jac=g, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        hx = np.clip(r.x, 0.0, 1.0)
        hx = project_to_sum(hx, ts)
        return hx
    except Exception:
        return project_to_sum(h0, ts)

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return lse_obj_and_grad(x, alpha)[0]
    def g(x):
        return lse_obj_and_grad(x, alpha)[1]
    def c_fun(x):
        return np.sum(x) - ts
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def interp_up(h, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    return project_to_sum(h_up, n_new / 2.0)

def freq_perturb_structured(h, rng, n_modes, amp):

    H = np.fft.rfft(h - mean_h)

        H[k] += dk * (rng.randn() + 1j * rng.randn())

def peak_shave_coord_descent(h, n, ts, rng, n_steps=20000, top_m=5):
    h = h.copy()
    c5_cur = compute_c5(h)
    for step in range(n_steps):
        overlaps = compute_all_overlaps(h)
        top_idx = np.argsort(overlaps)[-top_m:]
        avg_grad = np.zeros(n)
        for k in top_idx:
            m = k - (n - 1)
            for i in range(n):
                ia = i - m
                ib = i + m
                if 0 <= ia < n:
                    avg_grad[i] += (1.0 - h[ia])
                if 0 <= ib < n:
                    avg_grad[i] -= h[ib]
        avg_grad /= top_m
        dx_lr = 0.02 / np.max(np.abs(avg_grad)) if np.max(np.abs(avg_grad)) > 0 else 0.02
        h_new = h - dx_lr * avg_grad
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def two_level_fourier_opt(h0, n, ts, rng, n_modes=25, n_inner=200):
    mean_h = 0.5
    H = np.fft.rfft(h0 - mean_h)

        h_f = np.fft.irfft(H_new, n=n) + mean_h

        if c5_f < best_c5:

            old_a = amplitudes[k-1]
            old_p = phases[k-1]
            amplitudes[k-1] = max(0, old_a + da)
            phases[k-1] = old_p + dp
            H_test = np.fft.rfft(np.zeros(n))

    H_final = np.fft.rfft(np.zeros(n))

    return project_to_sum(h_out, ts)

def stochastic_refine(h, n, ts, rng, n_steps=30000):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.1
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.2)
        if h[i] < delta:
            delta = max(h[i] * 0.5, 0.001)
        if 1.0 - h[j] < delta:
            delta = max((1.0 - h[j]) * 0.5, 0.001)
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(0.005, 0.1 * np.exp(-step / n_steps))
    return h, c5_cur

    # Phase 3: Peak-shave coordinate descent
    if time_left() > 120:
        h_ps = best_h.copy()
        h_ps, c5_ps = peak_shave_coord_descent(h_ps, n, ts, rng, n_steps=15000, top_m=5)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
        h = best_h.copy()
        if time_left() > 60:
            h_new = lbfgs_opt(h, n, ts, 50000, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    # Phase 4: Upgrade to n=512
    if time_left() > 200:
        n2 = 512
        ts2 = n2 / 2.0
        h2 = interp_up(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_n = n2
        for alpha in [50, 200, 800, 3000, 10000, 30000]:
            if time_left() < 45:
                break
            h2_new = lbfgs_opt(h2, n2, ts2, alpha, maxiter=1500)
            c5_2 = compute_c5(h2_new)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2_new.copy()
                best_n = n2
            h2 = h2_new
        if time_left() > 60:
            h2_new = slsqp_opt(h2, n2, ts2, 50000, maxiter=800)
            c5_2 = compute_c5(h2_new)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2_new.copy()
                best_n = n2
        n = best_n
        ts = n / 2.0
    # Phase 5: Structured frequency perturbations + re-optimize
    for trial in range(12):
        if time_left() < 50:
            break
        amp = 0.06 * (1.0 - trial / 15.0) + 0.01
        nf = max(5, n // 20)
        h_sh = freq_perturb_structured(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        for alpha in [500, 5000, 20000]:
            if time_left() < 25:
                break
            h_ot = lbfgs_opt(h_sh, n, ts, alpha, maxiter=800)
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            h_sh = h_ot
    # Phase 6: Two-level Fourier optimization
    if time_left() > 100:
        h_fl = two_level_fourier_opt(best_h, n, ts, rng, n_modes=30, n_inner=150)
        c5_fl = compute_c5(h_fl)
        if c5_fl < best_c5:
            best_c5 = c5_fl
            best_h = h_fl.copy()
        if time_left() > 40:
            h_pol = lbfgs_opt(h_fl, n, ts, 10000, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    # Phase 7: Stochastic refinement
    if time_left() > 80:
        h_sr, c5_sr = stochastic_refine(best_h, n, ts, rng, n_steps=25000)
        if c5_sr < best_c5:
            best_c5 = c5_sr
            best_h = h_sr.copy()
        if time_left() > 40:
            h_pol = lbfgs_opt(best_h, n, ts, 50000, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    # Phase 8: Second stochastic with smaller moves
    if time_left() > 60:
        h_sr2, c5_sr2 = stochastic_refine(best_h, n, ts, rng, n_steps=20000)
        if c5_sr2 < best_c5:
            best_c5 = c5_sr2
            best_h = h_sr2.copy()
        if time_left() > 30:
            h_pol2 = slsqp_opt(best_h, n, ts, 100000, maxiter=1000)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
    # Phase 9: Final peak-shave + polish
    if time_left() > 50:
        h_fs, c5_fs = peak_shave_coord_descent(best_h, n, ts, rng, n_steps=10000, top_m=8)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()
        if time_left() > 30:
            h_final = lbfgs_opt(best_h, n, ts, 100000, maxiter=2000)
            c5_final = compute_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    # Final projection and validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(200):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
