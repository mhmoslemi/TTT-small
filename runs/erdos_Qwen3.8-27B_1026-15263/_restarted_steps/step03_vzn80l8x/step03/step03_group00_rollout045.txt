import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    """Gradient of corr at specific lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-lag] += 1.0
            g[:n-lag] -= h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[k:] += 1.0
            g[k:] -= h[:n-k]
            g[:n-k] -= h[k:]
    return g

def topk_smooth_max(h, alpha, k=5):
    """Smooth max focusing on top-k lags."""
    corr = compute_corr(h)
    n_corr = len(corr)
    # Softmax over all lags but with emphasis on top-k
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj, w

    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    n = len(h)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)

        if abs(wk) < 1e-14:

        k = k_idx - (n-1)

            else:

                grad[kk:] -= wk * h[:n-kk]

    return obj, grad

def project_step(h, grad, step, target_sum):
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def slsqp_wrap(h0, n, target_sum, alpha, maxiter=300):
    def f(x):
        return full_smooth_grad(x, alpha)[0]
    def g(x):
        return full_smooth_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
