def compute_B(h, j, n):

    idx_b = j - s

    valid_b = (idx_b >= 0) & (idx_b < n)

    B[valid_b] -= h[idx_b[valid_b]]

    best_idx = int(np.argmin(vals))

        return lo
    if best_idx == len(t_grid) - 1:

    lo_r = t_grid[max(0, best_idx - 1)]

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)

    m_idx = np.arange(2*n-1)[None, :]
    s = m_idx - (n-1)
    idx_a = j_idx + s
    idx_b = j_idx - s
    valid_a = (idx_a >= 0) & (idx_a < n)
    valid_b = (idx_b >= 0) & (idx_b < n)
    J_n = n - 1

    grad_b = valid_b * h[fb] * w[None, :]
    grad = np.sum(grad_a - grad_b, axis=1)
    return obj, grad

    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]

**Idea: Gradient-based pair descent**
Instead of trying random k values, use the gradient to determine which k values are most promising.

For position j, the gradient component is:
g[j] = Σ_k w[k] * B[j][k]

where w[k] are the softmax weights. A large |g[j]| means h[j] is far from optimal.

For the pair (j, k), the improvement is proportional to g[j] * (h[k] - 1) + g[k] * h[j] (roughly). So we want to pair positions with opposite gradient signs.

_g_n = 0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    k = np.arange(2 * n - 1)
    s = k - nm1
    idx_a = j + s
    idx_b = j - s
    va = (idx_a >= 0) & (idx_a < n)
    vb = (idx_b >= 0) & (idx_b < n)
    B = np.ones(2 * n - 1)
    B[va] -= h[idx_a[va]]
    B[vb] -= h[idx_b[vb]]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return 0.5 * (lo + hi)
    t = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t[:, None], axis=1)
    bi = int(np.argmin(vals))
    if bi == 0:
        return lo
    if bi == len(t) - 1:
        return hi
    lo_r, hi_r = t[bi - 1], t[bi + 1]
    for _ in range(40):
        mid = 0.5 * (lo_r + hi_r)
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return 0.5 * (lo_r + hi_r)

def pair_sweep(h, corr, rng, n_cand=15):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        bv = np.max(corr)
        bdelta = 0.0
        bD = None
        bk = -1
        cands = [(j+1)%n, (j+n-1)%n, (j+2)%n, (j+n-2)%n,
                 (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n,
                 (j+n//4)%n, (j+3*n//4)%n]
        while len(cands) < n_cand:
            cands.append(int(rng.randint(n)))
        seen = set()
        for k in cands:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-13:
                continue
            d = solve_1d(corr, D, dlo, dhi)
            v = np.max(corr + D * d)
            if v < bv - 1e-13:
                bv = v
                bdelta = d
                bD = D
                bk = k
        if bk >= 0 and abs(bdelta) > 1e-13:
            corr += bdelta * bD
            h[j] = np.clip(h[j] + bdelta, 0.0, 1.0)
            h[bk] = np.clip(h[bk] - bdelta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    ev = np.exp(alpha * (corr - m_val))
    w = ev / np.sum(ev)
    obj = m_val + np.log(np.sum(ev)) / alpha
    ji = np.arange(n)[:, None]
    mi = np.arange(2 * n - 1)[None, :]
    s = mi - (n - 1)
    ia = ji + s
    ib = ji - s
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    fa = np.clip(ia, 0, n - 1)
    fb = np.clip(ib, 0, n - 1)
    ga = va * (1.0 - h[fa]) * w[None, :]
    gb = vb * h[fb] * w[None, :]
    grad = np.sum(ga - gb, axis=1)
    return obj, grad

def _obj(x):
    return smooth_max_and_grad(x, _g_alpha)[0]

def _grad(x):
    return smooth_max_and_grad(x, _g_alpha)[1]

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    global _g_target, _g_n, _g_alpha
    _g_target = target_sum
    _g_n = n
    _g_alpha = alpha
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h0.copy(), method='SLSQP', jac=_grad,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fourier_perturb(h, rng, n_modes=3, amp=0.02):
    n = len(h)
    x = np.arange(n)
    delta = np.zeros(n)
    for _ in range(n_modes):
        hm = int(rng.randint(2, max(3, n // 3)))
        ph = rng.uniform(0, 2 * np.pi)
        delta += amp * np.sin(np.pi * hm * x / n + ph)
    return project_to_sum(h + delta, n // 2.0)
