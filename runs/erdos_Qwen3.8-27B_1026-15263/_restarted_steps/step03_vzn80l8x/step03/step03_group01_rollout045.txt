        s = max_idx - (n - 1)  # the worst lag
        
        # For this lag, find the worst positions
        if s >= 0:
            # O(s) = sum_{i=0}^{n-1-s} h[i](1-h[i+s])
            terms = h[:n-s] * (1.0 - h[s:])
        else:
            p = -s
            # O(-p) = sum_{i=p}^{n-1} h[i](1-h[i-p])
            terms = h[p:] * (1.0 - h[:n-p])
        
        # Find top n_swaps positions with largest terms
        top_k = min(n_swaps, len(terms))
        top_idx = np.argsort(terms)[-top_k:]
        
        # For each, try to move mass
        for ti in top_idx[:n_swaps]:
            delta = min(terms[ti], 0.05)  # move at most 0.05
            if delta < 1e-4:
                continue
            
            h_new = h.copy()
            if s >= 0:
                i_src = ti
                i_dst = ti + s
                h_new[i_src] -= delta
                h_new[i_dst] += delta
