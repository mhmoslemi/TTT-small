def slsqp_step(h0, n, target_sum, alpha, maxiter=600):
    def sml_f(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        return m_val + np.log(s) / alpha

    def sml_g(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(2 * n - 1)[None, :]
        fi = j_idx - m_idx + n - 1
        si = j_idx + m_idx - n + 1
        vf = (fi >= 0) & (fi < n)
        vs = (si >= 0) & (si < n)
        fc = np.clip(fi, 0, n - 1)
        sc = np.clip(si, 0, n - 1)
        cf = vf * (1.0 - h[fc]) * w[None, :]
        cs = vs * h[sc] * w[None, :]
        return np.sum(cf - cs, axis=1)

    def cons_fun(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sml_f, h0.copy(), method='SLSQP', jac=sml_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

                              t_init=0.004, t_min=0.0008, patience=120000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    if best_n % 2 == 0:
        best_h = make_binary(best_h, best_n, best_n // 2, rng)
    else:
        best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
