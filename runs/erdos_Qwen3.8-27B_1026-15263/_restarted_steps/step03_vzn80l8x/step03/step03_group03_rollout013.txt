```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_correlation_matrix(n):
    """Build the matrix M where M[m, j] = 1 if h[j] participates in corr[m]."""
    # corr[m] = sum_j h[j] * (1-h)[j-m+n-1] for valid indices
    # For fixed g = 1-h: corr[m] = sum_j h[j] * g[j-m+n-1] * indicator(valid)
    M = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        j_lo = max(0, m - n + 1)
        j_hi = min(n-1, m)
        for j in range(j_lo, j_hi+1):
            idx = j - m + n - 1  # index into g
            if 0 <= idx < n:
                M[m, j] = 1.0
    return M

def solve_lp_h(g, n, dx):
    """Given g=1-h, find h minimizing max_m sum_j h[j]*g[j-m+n-1]*dx."""
    M = build_correlation_matrix(n)
    B = M * g[None, :] * dx  # B[m, j] = g[j-m+n-1]*dx if valid
    
    # Variables: [h(0), ..., h(n-1), T]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize T
    
    # Inequality: B @ h - T <= 0, i.e., B @ h + (-1)*T <= 0
    A_ub = np.hstack([B, -np.ones(2*n-1).reshape(-1,1)])
    b_ub = np.zeros(2*n-1)
    
    # sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]/dx
    return None, None
```

```python
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_B(factor, n, dx):
    """Build matrix B where B[m, j] = factor[j-m+n-1] * dx if valid."""
    B = np.zeros((2*n-1, n))
    fdx = factor * dx
    for m in range(2*n-1):
        s = m - (n-1)
        if s >= 0:
            B[m, s:n] = fdx[:n-s]
        else:
            B[m, 0:n+s] = fdx[-s:n]
    return B

def solve_overlap_lp(factor, n, target_sum):
    """Given factor (= 1-h or = h), solve LP for the other variable."""
    dx = 2.0 / n

    c[-1] = 1.0
```

```python
def build_B(factor, n):
    dx = 2.0 / n
    B = np.zeros((2*n-1, n))
    fdx = factor * dx
    for k in range(2*n-1):
        j_lo = max(0, (n-1) - k)
        length = min(n-1, 2*n-2-k) - j_lo + 1
        if length > 0:
            idx_start = j_lo + k - (n-1)
            B[k, j_lo:j_lo+length] = fdx[idx_start:idx_start+length]
    return B
```

This should be fast enough.

Actually, I can vectorize the entire B construction using broadcasting:
```python
def build_B(factor, n):
    dx = 2.0 / n
    k_idx = np.arange(2*n-1)[:, None]  # shape (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # shape (1, n)
    idx = j_idx + k_idx - (n-1)  # shape (2n-1, n)
    valid = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n-1)
    B = np.where(valid, factor[idx_clipped] * dx, 0.0)
    return B
```

This is much faster (pure numpy, no Python loop). For n=480, the arrays are 959×480 ≈ 460K elements. Should run in <5ms.
