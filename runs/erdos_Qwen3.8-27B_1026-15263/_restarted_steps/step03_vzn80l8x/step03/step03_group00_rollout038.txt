But if I fix all variables except h[j], then c_k becomes linear in h[j]:
c_k(h) = ... + h[j]*(1-h[j-k]) + h[j+k]*(1-h[j]) + ...

       = ... + h[j]*(1 - h[j-k] - h[j+k]) + const

So c_k is linear in h[j] (with coefficient (1 - h[j-k] - h[j+k])). This means the max of linear functions is convex in h[j], and I can find the optimal h[j] analytically or with a 1D search!

This is a great insight for coordinate descent. For each coordinate j, I can:
1. Compute c_k as a function of h[j] (linear in h[j] for each k)
2. Find the h[j] that minimizes max_k c_k(h)

The optimal h[j] is at a point where two or more of the c_k functions intersect (or at the boundary 0 or 1).

This gives an exact coordinate update (like in proximal methods) that should be much more effective than fixed-step coordinate descent.

    h[j] = t_opt

```python
def exact_coordinate_descent(h, target_sum, max_passes=100, tol=1e-10):
    """Exact coordinate descent using 1D minimax subproblems."""
    from scipy.optimize import minimize_scalar
    n = len(h)
    h = h.astype(float).copy()
    h = project_to_sum(h, target_sum)
    
    # Precompute correlation
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    c5 = np.max(corr) * (2.0/n)
    
    # Precompute B matrix: B[m, j] = coefficient of h[j] in corr_m
    # B[m, j] = (1-h[j-m])*1[0<=j-m<n] - h[j+m]*1[0<=j+m<n]
    # where m = k - (n-1) for correlation index k
    
    # This depends on h, so we need to recompute it as h changes.
    # But we can compute it on-the-fly for each j.
    
    for pass_num in range(max_passes):
        max_c5_before = np.max(corr)
        improved = False
        
        for j in range(n):
            old_val = h[j]
            
            # Compute B for coordinate j (given current h)
            B = np.zeros(2*n-1)
            lags = np.arange(2*n-1) - (n-1)
            # B[k] = (1-h[j-lag_k])*1[j-lag_k valid] - h[j+lag_k]*1[j+lag_k valid]
            for k in range(2*n-1):
                m = lags[k]
                val = 0.0
                im1 = j - m
                im2 = j + m
                if 0 <= im1 < n:
                    val += (1.0 - h[im1])
                if 0 <= im2 < n:
                    val -= h[im2]
                B[k] = val
            
            # Now corr_m(t) = corr_m + (t - old_val) * B_m
            # Minimize max_m (corr_m + (t - old_val) * B_m) over t ∈ [0,1]
            # Equivalently: minimize max_m ((corr_m - old_val*B_m) + t * B_m)
            base = corr - old_val * B
            
            def f1d(t):
                return np.max(base + t * B)
            
            result = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                     options={'xatol': 1e-10})
            t_opt = result.x
            
            if abs(t_opt - old_val) > 1e-10:
                # Update correlation incrementally
                delta = t_opt - old_val
                corr += delta * B
                h[j] = t_opt
                improved = True
        
        # Project to maintain sum constraint
        h = project_to_sum(h, target_sum)
        # Recompute correlation after projection
        corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        
        c5_new = np.max(corr) * (2.0/n)
        if c5_new < c5 - tol:
            c5 = c5_new
        else:
            c5 = c5_new
        
        if not improved or np.max(corr) - max_c5_before < tol:
            break
    
    return h, c5
```

B[j, k] = (1-h[j-lag_k])*1[j-lag_k valid] - h[j+lag_k]*1[j+lag_k valid]

For a fixed j, B[k] depends on h[j - lag_k] and h[j + lag_k].

lag_k = k - (n-1) for k = 0, ..., 2n-2
So lag ranges from -(n-1) to n-1.

j - lag_k = j - (k - (n-1)) = j + (n-1) - k
j + lag_k = j + (k - (n-1)) = j + k - (n+1)

So:
B[k] = (1-h[j+(n-1)-k])*1[0 ≤ j+(n-1)-k < n] - h[j+k-(n+1)]*1[0 ≤ j+k-(n+1) < n]

For the first term: j+(n-1)-k ranges from j+(n-1) to j as k goes from 0 to 2n-2.
It's valid when 0 ≤ j+(n-1)-k < n, i.e., j+(n-1)-n < k ≤ j+(n-1), i.e., j-1 < k ≤ j+n-1, i.e., k ∈ [j, j+n-1] (integer).

For the second term: j+k-(n+1) ranges from j-(n+1) to j+n-2 as k goes from 0 to 2n-2.
It's valid when 0 ≤ j+k-(n+1) < n, i.e., n+1-j ≤ k < 2n+1-j, i.e., k ∈ [max(0,n+1-j), min(2n-2, 2n-j-2)].

So B is a "banded" structure - for each j, only O(n) entries of B are non-zero.

But computing B for a specific j still takes O(n) time (to fill the non-zero entries). And we do this for all j, giving O(n^2) per pass for just computing B.

```python
def compute_B_for_j(h, j, n):
    """Compute B vector for coordinate j."""
    lags = np.arange(2*n-1) - (n-1)
    im1 = j - lags  # j - lag_k for each k
    im2 = j + lags  # j + lag_k for each k
    valid1 = (im1 >= 0) & (im1 < n)
    valid2 = (im2 >= 0) & (im2 < n)
    B = np.zeros(2*n-1)
    B[valid1] += (1.0 - h[im1[valid1]])
    B[valid2] -= h[im2[valid2]]
    return B
```

Now, the 1D minimization: minimize max(base + t*B) over t ∈ [0,1].

The optimal t is where two lines in the upper envelope intersect at the minimum point.

- This is: for all k with B[k] > 0: t ≤ (V - base[k])/B[k]

    # concave in t (max of linear functions is convex, so min of convex is at most
    # the min of endpoints... wait no, max of linear is convex, so min of convex is

    # And v_opt ≥ min_k max(0, base[k]+t*B[k]) for any t...

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minimax(base, B, n_corr):
    """Minimize max_k (base[k] + t*B[k]) over t in [0,1].
    Uses bisection on the optimal value V."""
    B = B[:n_corr]
    base = base[:n_corr]
    
    eps_B = 1e-14
    zero_mask = np.abs(B) < eps_B
    pos_mask = B > eps_B
    neg_mask = B < -eps_B
    
    # Check zero-b entries
    if np.any(zero_mask):
        if np.max(base[zero_mask]) > np.max(base[pos_mask]) if np.any(pos_mask) else False:
            pass  # handled in feasibility check

    val_at_1 = np.max(base + B)

    hi = max(val_at_0, val_at_1)

        if val_at_0 <= val_at_1:
            return 0.0
        else:

    def feasible(V):
        """Return t if V is achievable, else None."""
        # Check zero-b entries
        if np.any(zero_mask):

        else:

def exact_coord_descent(h, target_sum, max_passes=50, tol=1e-11):
    """Exact coordinate descent for the Erdős problem."""
    n = len(h)
    n_corr = 2 * n - 1
    h = h.astype(float).copy()
    h = project_to_sum(h, target_sum)
    corr = compute_corr(h).astype(float)
    c5 = float(np.max(corr) * (2.0 / n))
    
    lags = np.arange(n_corr) - (n - 1)
    
    for p in range(max_passes):
        max_before = float(np.max(corr))
        total_change = 0.0
        
        for j in range(n):
            old_val = h[j]
            
            # Compute B for coordinate j
            im1 = j - lags  # index for h[j - lag]
            im2 = j + lags  # index for h[j + lag]
            valid1 = (im1 >= 0) & (im1 < n)
            valid2 = (im2 >= 0) & (im2 < n)
            
            B = np.zeros(n_corr)
            if np.any(valid1):
                B[valid1] += (1.0 - h[im1[valid1]])
            if np.any(valid2):
                B[valid2] -= h[im2[valid2]]
            
            # Check if B is essentially zero
            if np.max(np.abs(B)) < 1e-14:
                continue
            
            # base[k] = corr[k] - old_val * B[k]
            base = corr - old_val * B
            
            t_opt = solve_1d_minimax(base, B, n_corr)
            
            if abs(t_opt - old_val) > 1e-12:
                delta = t_opt - old_val
                corr += delta * B
                h[j] = t_opt
                total_change += abs(delta)
        
        # Project to maintain sum constraint
        h = project_to_sum(h, target_sum)
        corr = compute_corr(h).astype(float)
        c5_new = float(np.max(corr) * (2.0 / n))
        
        if c5_new < c5:
            c5 = c5_new
        else:
            c5 = c5_new
        
        if total_change < tol or c5 >= max_before * (2.0/n) - 1e-12:
            # Converged
            break
    
    return h, c5

def smooth_max_grad(h, alpha):
    """Compute smooth max of correlation and gradient, vectorized."""
    n = len(h)
    n_corr = 2 * n - 1
    corr = compute_corr(h).astype(float)
    
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = float(m_val + np.log(np.sum(exp_vals)) / alpha)
    
    # Gradient: grad[j] = sum_k w[k] * [(1-h[j-lag_k])*1[valid] - h[j+lag_k]*1[valid]]
    lags = np.arange(n_corr) - (n - 1)
    
    # Vectorized computation
    # For each j, im1 = j - lags, im2 = j + lags
    # This is an (n, n_corr) matrix operation - might be memory intensive for large n
    # For n=540, that's 540 * 1079 ≈ 583K entries - fine
    # For n=1000, that's 1000 * 1999 ≈ 2M entries - still fine
    
    j_idx = np.arange(n)[:, None]
    im1 = j_idx - lags[None, :]  # (n, n_corr)
    im2 = j_idx + lags[None, :]
    
    valid1 = (im1 >= 0) & (im1 < n)
    valid2 = (im2 >= 0) & (im2 < n)
    
    # Clip for safe indexing
    im1_clipped = np.clip(im1, 0, n-1)
    im2_clipped = np.clip(im2, 0, n-1)
    
    term1 = valid1 * (1.0 - h[im1_clipped]) * w[None, :]
    term2 = valid2 * h[im2_clipped] * w[None, :]
    
    grad = np.sum(term1 - term2, axis=1)
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500):
    """SLSQP optimization with smooth max objective."""
    h0 = h0.astype(float).copy()
    h0 = project_to_sum(h0, target_sum)
    
    def f(x):
        return smooth_max_grad(x, alpha)[0]
    
    def g(x):
        return smooth_max_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minimax(base, B):
    n_corr = len(base)
    eps_B = 1e-13
    zero_mask = np.abs(B) < eps_B
    pos_mask = B > eps_B
    neg_mask = B < -eps_B

    val_at_0 = float(np.max(base))
    val_at_1 = float(np.max(base + B))

    lo = min(val_at_0, val_at_1)
    hi = max(val_at_0, val_at_1)

    if hi - lo < 1e-13:
        return 0.0 if val_at_0 <= val_at_1 else 1.0

    def feasible(V):
        if np.any(zero_mask):
            if float(np.max(base[zero_mask])) > V + 1e-13:
                return None
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg_mask):
            bounds = (V - base[neg_mask]) / B[neg_mask]
            t_lo = max(t_lo, float(np.max(bounds)))
        if np.any(pos_mask):
            bounds = (V - base[pos_mask]) / B[pos_mask]
            t_hi = min(t_hi, float(np.min(bounds)))
        if t_lo <= t_hi:
            return (t_lo + t_hi) * 0.5
        return None

    t_opt = 0.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        t = feasible(mid)
        if t is not None:
            hi = mid
            t_opt = t
        else:
            lo = mid

    t = feasible(hi)
    if t is not None:
        t_opt = t
    else:
        t_opt = 0.0 if val_at_0 <= val_at_1 else 1.0

    return float(np.clip(t_opt, 0.0, 1.0))

def exact_coord_descent(h, target_sum, max_passes=50, tol=1e-12):
    n = len(h)
    n_corr = 2 * n - 1
    h = h.astype(float).copy()
    h = project_to_sum(h, target_sum)
    corr = compute_corr(h).astype(float)
    c5 = float(np.max(corr) * (2.0 / n))
    lags = np.arange(n_corr) - (n - 1)

    for p in range(max_passes):
        max_before = float(np.max(corr))
        total_change = 0.0

        for j in range(n):
            old_val = h[j]
            im1 = j - lags
            im2 = j + lags
            valid1 = (im1 >= 0) & (im1 < n)
            valid2 = (im2 >= 0) & (im2 < n)
            B = np.zeros(n_corr)
            if np.any(valid1):
                B[valid1] += (1.0 - h[im1[valid1]])
            if np.any(valid2):
                B[valid2] -= h[im2[valid2]]
            if float(np.max(np.abs(B))) < 1e-14:
                continue
            base = corr - old_val * B
            t_opt = solve_1d_minimax(base, B)
            if abs(t_opt - old_val) > 1e-12:
                delta = t_opt - old_val
                corr += delta * B
                h[j] = t_opt
                total_change += abs(delta)

        h = project_to_sum(h, target_sum)
        corr = compute_corr(h).astype(float)
        c5_new = float(np.max(corr) * (2.0 / n))
        if c5_new < c5:
            c5 = c5_new
        else:
            c5 = c5_new
        if total_change < tol:
            break

    return h, c5

def smooth_max_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = compute_corr(h).astype(float)
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + float(np.log(np.sum(exp_vals))) / alpha
    lags = np.arange(n_corr) - (n - 1)
    j_idx = np.arange(n)[:, None]
    im1 = j_idx - lags[None, :]
    im2 = j_idx + lags[None, :]
    valid1 = (im1 >= 0) & (im1 < n)
    valid2 = (im2 >= 0) & (im2 < n)
    im1c = np.clip(im1, 0, n - 1)
    im2c = np.clip(im2, 0, n - 1)
    term1 = valid1.astype(float) * (1.0 - h[im1c]) * w[None, :]
    term2 = valid2.astype(float) * h[im2c] * w[None, :]
    grad = np.sum(term1 - term2, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=400):
    h0 = h0.astype(float).copy()
    h0 = project_to_sum(h0, target_sum)

    def obj_fn(x):
        return smooth_max_grad(x, alpha)[0]

    def grad_fn(x):
        return smooth_max_grad(x, alpha)[1]

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h0

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old, x_old + 2.0, x_old + 4.0])
    h_new = np.interp(x_new, x_ext, h_ext)
    return h_new
