import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    """Project h onto {h >= 0, h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_max_corr(h, target_sum):
    """Compute subgradient of max_k corr_k(h) at h.
    Returns (objective, subgradient, argmax_lag)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m_idx = int(np.argmax(corr))
    obj = float(corr[m_idx] * dx)
    lag = m_idx - (n - 1)
    
    # Gradient of corr[lag] with respect to h
    # corr[lag] = sum_i h[i] * (1 - h[i + lag]) for valid i
    # For lag >= 0: corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
    # d/dh[j] = (1-h[j+lag]) for j <= n-1-lag, and -h[j-lag] for j >= lag
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    
    # Scale by dx
    g *= dx
    return obj, g, m_idx

def nesterov_subgrad(h0, target_sum, step_size, max_iters, momentum=0.9):
    """Nesterov-style accelerated subgradient method."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    
    for it in range(max_iters):
        # Nesterov lookahead
        y = h + momentum * v
        
        # Subgradient at y
        _, g, _ = subgradient_max_corr(y, target_sum)
        
        # Update
        v = momentum * v - step_size * g
        h_new = y + v
        h_new = project_feasible(h_new, target_sum)
        
        # Check improvement
        obj = compute_c5(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_obj

def backtracking_subgrad(h0, target_sum, max_iters, step_size0=0.01, c=0.5, rho=0.5):
    """Subgradient method with backtracking line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    step_size = step_size0
    
    for it in range(max_iters):
        obj, g, m_idx = subgradient_max_corr(h, target_sum)
        
        # Try to decrease the objective
        h_trial = h - step_size * g
        h_trial = project_feasible(h_trial, target_sum)
        obj_trial = compute_c5(h_trial)
        
        if obj_trial < obj or it == 0:
            h = h_trial
            step_size = min(step_size * 1.05, 1.0)
        else:
            step_size = step_size * rho
        
        obj = compute_c5(h)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj

def annealed_subgrad(h0, target_sum, rng, max_iters, temp0=0.1, temp_min=0.001):
    """Subgradient method with simulated annealing for step size."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    step_size = 0.01
    temp = temp0

        h_trial = project_feasible(h_trial, target_sum)
        obj_trial = compute_c5(h_trial)
        
        delta = obj_trial - obj
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            h = h_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_h = h_trial.copy()

    """Fast coordinate descent focusing on the worst lag."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)

            if lag == 0:

            g[k:] -= h[:n-k]

        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break

def smooth_max_obj_grad(h, alpha):
    """Smooth max approximation with gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted - np.max(shifted))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * dx
    
    # Gradient
    g = np.zeros(n)
    j = np.arange(n)
    for k_idx in range(2*n-1):
        lag = k_idx - (n-1)
        if w[k_idx] < 1e-10:
            continue
        if lag >= 0:
            if lag == 0:
                g[:] += w[k_idx] * (1.0 - 2.0 * h)
            else:
                g[:n-lag] += w[k_idx] * (1.0 - h[lag:])
                g[lag:] -= w[k_idx] * h[:n-lag]
        else:
            kk = -lag
            g[:n-kk] += w[k_idx] * (1.0 - h[kk:])
            g[kk:] -= w[k_idx] * h[:n-kk]
    g *= dx
    return obj, g

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    """SLSQP with smooth max objective."""
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 500.0
_g_target_sum = 270.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_and_grad_fast(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * dx
    g = np.zeros(n)
    threshold = 1e-7
    for k_idx in range(2 * n - 1):
        if w[k_idx] < threshold:
            continue
        lag = k_idx - (n - 1)
        gk = corr_grad_at_lag(h, lag)
        g += w[k_idx] * gk
    g *= dx
    return obj, g

def _slsqp_f(x):
    return smooth_max_and_grad_fast(x, _g_alpha)[0]

def _slsqp_grad(x):
    return smooth_max_and_grad_fast(x, _g_alpha)[1]

def _slsqp_con_fun(x):
    return np.sum(x) - _g_target_sum

def _slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    global _g_alpha, _g_target_sum
    _g_alpha = alpha
    _g_target_sum = target_sum
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_f, h0.copy(), method='SLSQP', jac=_slsqp_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

def coord_desc_topk(h, target_sum, max_iters=10000, step_frac_init=0.15, top_k=4):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    step_frac = step_frac_init
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        g = np.zeros(n)
        for m_idx in top_idx:
            lag = m_idx - (n - 1)
            g += corr_grad_at_lag(h, lag)
        g /= float(top_k)
        g *= dx
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stale = 0
            step_frac = min(step_frac * 1.05, 0.4)

            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            step_frac = max(step_frac * 0.85, 0.04)
            stale += 1
            if stale > 100:
                break
    return h, c5

def nesterov_subgrad(h0, target_sum, max_iters=5000, step0=0.008, momentum=0.88):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    step = step0
    stale = 0
    for it in range(max_iters):
        y = h + momentum * v
        corr_y = compute_corr(y)
        m_idx = int(np.argmax(corr_y))
        lag = m_idx - (n - 1)
        g = corr_grad_at_lag(y, lag) * dx
        v = momentum * v - step * g
        h_new = y + v
        h_new = project_feasible(h_new, target_sum)
        obj = compute_c5(h_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
            if stale > 300:
                step *= 0.85
                if step < 1e-6:
                    break
        h = h_new
    return best_h, best_obj

def stochastic_coord(h, target_sum, rng, max_iters=8000, step_frac=0.08):

            break

            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def block_binary_init(n, rng, n_blocks):
    h = np.zeros(n)
    block_size = max(1, n // n_blocks)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        if rng.rand() < 0.5:
            h[start:end] = 1.0
    h = project_feasible(h, n / 2.0)
    return h
