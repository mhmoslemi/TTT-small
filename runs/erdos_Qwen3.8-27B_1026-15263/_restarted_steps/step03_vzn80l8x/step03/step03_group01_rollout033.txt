```python
def targeted_local_search(h, n, ts, n_rounds=100):
    """Find the worst shift and try to reduce it via targeted moves."""
    h = h.copy()
    best_c5 = compute_c5(h)[0]
    
    for round in range(n_rounds):
        # Find the worst shift
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        current_max = corr[k_star]
        
        if current_max <= best_c5:
            best_c5 = current_max
            # Keep h as is
        
        # Get the shift value
        shift = k_star - (n - 1)
        
        # For this shift, the overlap is:
        # sum_{i: valid} h[i] * (1 - h[i-shift]) * dx
        # To reduce it, we want to:
        #   - Decrease h[i] where h[i] is large and h[i-shift] is small (or vice versa)
        #   - Or increase h[i-shift] where h[i] is large
        
        # Find the indices that contribute most to the overlap
        if shift >= 0:
            # valid i: 0 <= i < n and 0 <= i-shift < n => shift <= i < n
            i_start = max(0, shift)
            i_end = n
            contrib = h[i_start:i_end] * (1 - h[i_start - shift:i_end - shift])
        else:
            # shift < 0, let s = -shift > 0
            s = -shift
            # valid i: 0 <= i < n and 0 <= i+s < n => 0 <= i < n-s
            i_start = 0
            i_end = n - s
            contrib = h[i_start:i_end] * (1 - h[i_start + s:i_end + s])
        
        # Find the top contributor
        if len(contrib) == 0:
            break
        max_idx = np.argmax(contrib)
        i_best = i_start + max_idx
        
        # Determine the move:
        # If h[i_best] > h[i_best - shift], decrease h[i_best]
        # If h[i_best] < h[i_best - shift], increase h[i_best - shift]
        # Choose the partner j that doesn't affect this shift
        
        # Simple move: transfer mass from the "problem" region to a "safe" point
        # A "safe" point is one that's not in the active range for shift k_star
        
        if shift >= 0:
            active = set(range(shift, n))  # indices i that are "first factor"
            active2 = set(range(0, n - shift))  # indices that are "second factor" (i-shift)
            problematic = active | active2
        else:
            s = -shift
            active = set(range(0, n - s))
            active2 = set(range(s, n))
            problematic = active | active2
        
        safe_indices = [j for j in range(n) if j not in problematic]
        if not safe_indices:
            safe_indices = list(range(n))  # fallback
        
        # Choose a safe index to receive mass
        j_safe = safe_indices[np.argmax(h[j] < 0.5 for j in safe_indices)] if safe_indices else 0
        
        # Transfer
        delta = min(0.05, h[i_best] * 0.5)  # conservative
        h[i_best] -= delta
        h[j_safe] += delta
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, ts)
        
        c5_new = compute_c5(h)[0]
        if c5_new > best_c5 * 1.05:
            # Revert
            h[i_best] += delta
            h[j_safe] -= delta
            h = np.clip(h, 0, 1)
            h = project_to_sum(h, ts)
    
    return h, compute_c5(h)[0]
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_v = np.exp(shifted - max_s)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_s + np.log(np.sum(exp_v))) / alpha
    
    grad = np.zeros(n)
    # Only iterate over shifts with significant weight
    threshold = 1e-8
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < threshold:
            continue
        s = i - (n - 1)
        # First term: (1-h[j-s]) for j where 0<=j-s<n
        j_min = max(0, s)
        j_max = min(n - 1, s + n - 1)
        if j_min <= j_max:
            grad[j_min:j_max+1] += wi * (1.0 - h[j_min - s : j_max - s + 1])
        # Second term: -h[j+s] for j where 0<=j+s<n
        j_min2 = max(0, -s)
        j_max2 = min(n - 1, n - 1 - s)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= wi * h[j_min2 + s : j_max2 + s + 1]
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_v = np.exp(shifted - max_s)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_s + np.log(np.sum(exp_v))) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < thresh:
            continue
        s = i - (n - 1)
        j_min1 = max(0, s)
        j_max1 = min(n - 1, s + n - 1)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1 + 1] += wi * (1.0 - h[j_min1 - s:j_max1 - s + 1])
        j_min2 = max(0, -s)
        j_max2 = min(n - 1, n - 1 - s)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2 + 1] -= wi * h[j_min2 + s:j_max2 + s + 1]
    return obj, grad

def lbfgsb_run(h0, n, ts, alpha, maxiter, penalty):
    dx = 2.0 / n

    def obj_func(x):
        o, _ = smooth_max_obj_grad(x, alpha, n)
        pen = penalty * (np.sum(x) - ts) ** 2
        return o + pen

    def grad_func(x):
        _, g = smooth_max_obj_grad(x, alpha, n)
        g = g + 2.0 * penalty * (np.sum(x) - ts) * np.ones(n)
        return g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='L-BFGS-B', jac=grad_func,
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def fourier_init(n, rng, n_harm, amp_range):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_harm + 1):
        a = rng.uniform(*amp_range)
        phi = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(np.pi * k * x + phi)
    return project_to_sum(h, n / 2.0)

def targeted_cd(h, n, ts, rng, n_steps, delta):
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        if s >= 0:
            i_lo = max(0, s)
            i_hi = n
            contrib = h[i_lo:i_hi] * (1.0 - h[i_lo - s:i_hi - s])
        else:
            ss = -s
            i_lo = 0
            i_hi = n - ss
            contrib = h[i_lo:i_hi] * (1.0 - h[i_lo + ss:i_hi + ss])
        if len(contrib) == 0:
            break
        top_idx = np.argmax(contrib)
        i_best = i_lo + top_idx
        if s >= 0:
            active1 = set(range(s, n))
            active2 = set(range(0, n - s))
        else:
            ss = -s
            active1 = set(range(0, n - ss))
            active2 = set(range(ss, n))
        problematic = active1 | active2
        safe = [j for j in range(n) if j not in problematic and h[j] < 0.6]
        if not safe:
            safe = [j for j in range(n) if h[j] < 0.8]
        if not safe:
            safe = list(range(n))
        j_safe = int(safe[rng.randint(0, len(safe))])
        d = delta * rng.uniform(0.5, 1.5)
        if h[i_best] > d * 0.5:
            h[i_best] -= d
            h[j_safe] += d
        else:
            h[j_safe] -= d
            h[i_best] += d
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new > c5_cur * 1.02:
            if h[i_best] > d * 0.5:
                h[i_best] += d
                h[j_safe] -= d
            else:
                h[j_safe] += d
                h[i_best] -= d
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
            c5_new = c5_cur
    return h, compute_c5(h)

def random_cd(h, n, ts, rng, n_steps, temp0, cool):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = temp0
    for step in range(n_steps):

        d = rng.uniform(0.005, 0.15)

            h[j] -= d
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
        temp = max(1e-7, temp0 * cool ** step)
    return h, c5_cur

    if time_left() > 80:

        ts_sa = n_sa / 2.0
        h_sa = best_h.copy()
        h_sa, c5_sa = random_cd(h_sa, n_sa, ts_sa, rng, n_steps=40000, temp0=0.02, cool=0.9995)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        if time_left() > 30:
            h_pol = lbfgsb_run(best_h, n_sa, ts_sa, 8000, maxiter=500, penalty=1e4)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Another round of targeted CD with smaller steps
    if time_left() > 60:
        n_t2 = best_n
        ts_t2 = n_t2 / 2.0
        h_t2 = best_h.copy()
        h_t2, c5_t2 = targeted_cd(h_t2, n_t2, ts_t2, rng, n_steps=40000, delta=0.03)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    # Phase 7: Final high-alpha polish
    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [5000, 20000, 50000]:
            if time_left() < 20:
                break
            h_f = lbfgsb_run(best_h, n_f, ts_f, alpha, maxiter=400, penalty=1e5)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 8: Tiny random refinements
    if time_left() > 30:
        n_r = best_n
        ts_r = n_r / 2.0
        h_r = best_h.copy()
        h_r, c5_r = random_cd(h_r, n_r, ts_r, rng, n_steps=20000, temp0=0.005, cool=0.9998)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        if time_left() > 15:
            h_pol = lbfgsb_run(best_h, n_r, ts_r, 30000, maxiter=300, penalty=1e5)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation and cleanup
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
