def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s = np.sum(exp_v)
    obj = (m_val + (max_sh + np.log(s)) / alpha)
    w = exp_v / s
    return obj, w, corr

    n = len(h)
    # Gradient using the formula from the previous code
    nj = np.arange(n)[:, None]

    return obj, grad

        # First factor: h[j] appears as h[fi] where fi = i - offset, so i = fi + offset
        # When j = fi, the contribution is (1 - h[si]) where si = j + offset
        # Valid when 0 <= j + offset < n
        if offset >= 0:
            # j ranges from 0 to n-1-offset
            valid_j = np.arange(0, n - offset)
            grad[valid_j] += (1 - h[valid_j + offset])
        else:
            # j ranges from -offset to n-1
            valid_j = np.arange(-offset, n)
            grad[valid_j] += (1 - h[valid_j + offset])
        # Second factor: (1-h[si]) where si = i + offset, so i = si - offset
        # When j = si, contribution is -h[fi] where fi = j - offset
        # Valid when 0 <= j - offset < n
        if offset >= 0:
            valid_j = np.arange(offset, n)
            grad[valid_j] -= h[valid_j - offset]
        else:
            valid_j = np.arange(0, n + offset)
            grad[valid_j] -= h[valid_j - offset]
    return grad * (2.0 / n)  # multiply by dx

        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

def get_corr(h):

def smax_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    sh = alpha * (corr - m)
    ms = np.max(sh)
    ev = np.exp(sh - ms)
    s = np.sum(ev)
    obj = m + (ms + np.log(s)) / alpha
    w = ev / s

    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
