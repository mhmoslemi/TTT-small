def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    # Project h to {0<=h<=1, sum(h)=target_sum}
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_active(h, lags):
    # Gradient of sum of corr[k] for k in lags
    n = len(h)
    g = np.zeros(n)
    for lag in lags:

            else:

                g[:n-lag] -= h[lag:]

                g[:] += (1.0 - 2.0 * h)
            else:
                g[:n-k] += (1.0 - h[k:])
                g[k:] -= h[:n-k]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def single_lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        k = -lag
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

def topk_smooth_max_obj(h, alpha, top_k):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    if top_k < len(corr):
        top_idx = np.argsort(corr)[::-1][:top_k]
    else:
        top_idx = np.arange(len(corr))
    corr_sub = corr[top_idx]
    m_sub = np.max(corr_sub)
    shifted = alpha * (corr_sub - m_sub)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_sub + np.log(np.sum(exp_vals)) / alpha
    return obj, top_idx, w

def topk_smooth_max_grad(h, alpha, top_k):
    n = len(h)
    obj, top_idx, w = topk_smooth_max_obj(h, alpha, top_k)
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        lag = int(idx) - (n - 1)
        g += wi * single_lag_grad(h, lag)
    return obj, g

class SLSQPOpt:
    def __init__(self, alpha, top_k, target_sum):
        self.alpha = alpha
        self.top_k = top_k
        self.target_sum = target_sum
    
    def fun(self, x):
        return topk_smooth_max_grad(x, self.alpha, self.top_k)[0]
    
    def jac(self, x):
        return topk_smooth_max_grad(x, self.alpha, self.top_k)[1]
    
    def con_fun(self, x):
        return np.sum(x) - self.target_sum
    
    def con_jac(self, x):
        return np.ones(len(x))

def slsqp_optimize(h0, target_sum, alpha, top_k, maxiter=500):
    n = len(h0)
    opt = SLSQPOpt(alpha, top_k, target_sum)
    cons = [{'type': 'eq', 'fun': opt.con_fun, 'jac': opt.con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(opt.fun, h0.copy(), method='SLSQP', jac=opt.jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_descent(h, target_sum, max_iters, init_step, top_k=5):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    step = init_step
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iters):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_lags = [int(top_idx[i]) - (n-1) for i in range(top_k)]
        g = np.zeros(n)
        for lag in top_lags:
            g += single_lag_grad(h, lag)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-24:
            break
        h_new = h - step * g
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            step = min(step * 1.03, init_step)
        else:
            step *= 0.85
    return best_h, best_c5

def coordinate_descent(h, target_sum, max_iters=5000, step_frac=0.12):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = single_lag_grad(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def active_set_optimize(h, target_sum, n_active=3, n_iters=15, alpha=200, top_k=10):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        corr = compute_corr(h)
        m_val = np.max(corr)
        # Identify active lags
        threshold = m_val - 0.01 * m_val  # within 1% of max
        active_mask = corr >= threshold
        active_lags = np.where(active_mask)[0] - (n-1)
        
        # Optimize to minimize the smooth max of active set
        # Use SLSQP with the active set as top_k
        h_new = slsqp_optimize(h, target_sum, alpha, min(top_k, len(active_lags)) + 3, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Subgradient polish
        h_sub, c5_sub = subgradient_descent(h, target_sum, max_iters=5000, init_step=0.02, top_k=n_active+2)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            h = h_sub
        # Increase alpha for sharper max
        alpha *= 2
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _remaining(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(400):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def single_lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def topk_smooth_max(h, alpha, top_k):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    if top_k < len(corr):
        top_idx = np.argsort(corr)[::-1][:top_k]
    else:
        top_idx = np.arange(len(corr))
    corr_sub = corr[top_idx]
    m_sub = np.max(corr_sub)
    shifted = alpha * (corr_sub - m_sub)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_sub + np.log(np.sum(exp_vals)) / alpha
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        lag = int(idx) - (n - 1)
        g += wi * single_lag_grad(h, lag)
    return obj, g

class _Opt:
    def __init__(self, alpha, top_k, target_sum):
        self.alpha = alpha
        self.top_k = top_k
        self.target_sum = target_sum
    def fun(self, x):
        return topk_smooth_max(x, self.alpha, self.top_k)[0]
    def jac(self, x):
        return topk_smooth_max(x, self.alpha, self.top_k)[1]
    def cfun(self, x):
        return np.sum(x) - self.target_sum
    def cjac(self, x):
        return np.ones(len(x))

def slsqp_optimize(h0, target_sum, alpha, top_k, maxiter=400):
    n = len(h0)
    opt = _Opt(alpha, top_k, target_sum)
    cons = [{'type': 'eq', 'fun': opt.cfun, 'jac': opt.cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(opt.fun, h0.copy(), method='SLSQP', jac=opt.jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_cur)
