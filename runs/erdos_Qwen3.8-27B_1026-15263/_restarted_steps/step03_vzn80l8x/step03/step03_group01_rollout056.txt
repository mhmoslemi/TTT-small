import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_shifted = np.max(shifted)
    exp_v = np.exp(shifted - max_shifted)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_shifted + np.log(np.sum(exp_v))) / alpha
    
    m_idx = np.arange(n)[:, None]
    s_all = (np.arange(2*n-1) - (n-1))[None, :]
    
    ms1 = m_idx + s_all
    valid1 = (ms1 >= 0) & (ms1 < n)
    h1 = np.where(valid1, 1.0 - h[np.clip(ms1, 0, n-1)], 0.0)
    term1 = h1 @ w
    
    ms2 = m_idx - s_all
    valid2 = (ms2 >= 0) & (ms2 < n)
    h2 = np.where(valid2, h[np.clip(ms2, 0, n-1)], 0.0)
    term2 = h2 @ w
    
    grad = term1 - term2
    return obj, grad

def optimize_lbfgs(h0, target_sum, alpha, maxiter=1000, gtol=1e-10):

    h = project_to_sum(h0.copy(), target_sum)
    
    def fun(x):

        return smooth_max_obj_grad(xp, alpha)[0]

        return smooth_max_obj_grad(xp, alpha)[1]
    
    try:
        res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                      options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': maxiter*5})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return h

def subgradient_polish(h0, target_sum, n_steps=5000, lr_init=0.005):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for i in range(n_steps):
        corr = get_correlations(h)
        max_val = np.max(corr)
        tol = 1e-6
        active = np.where(corr >= max_val - tol)[0]
        
        n = len(h)
        m_range = np.arange(n)
        grad = np.zeros(n)
        for k in active:
            s = k - (n-1)
            valid_r = (m_range + s >= 0) & (m_range + s < n)
            valid_l = (m_range - s >= 0) & (m_range - s < n)
            grad[valid_r] += (1.0 - h[m_range[valid_r] + s])
            grad[valid_l] -= h[m_range[valid_l] - s]
