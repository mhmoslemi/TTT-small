def project_to_sum(h, target_sum):

        if abs(diff) < 1e-13:

                for idx in idxs:

            else:

                    room = h[idx]
                    sub = min(room, -diff)
                    h[idx] -= sub
                    diff += sub

    # For n=240, a direct approach is fine.

    # grad[j] = sum_k w[k] * B_j[k]

    partB = valid2 * (-h[idx_h2])

    grad = B_all @ w  # (n,)
    
    return obj, grad

            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = interior.sum()

            h[interior] += diff / cnt
        else:
            if diff > 0:

                    idx = h.argmin()

                    diff -= add

                    sub = min(h[idx], -diff)

def smooth_grad(h, alpha):

    pB = v2 * (-h[np.clip(d2-nm1, 0, nm1)])

def window_sweep(h, corr, rng, n, window_size=3, n_moves=20):
    # For each pair of windows, try shifting mass from one to the other
    for _ in range(n_moves):
        j = rng.randint(n - window_size)
        k = rng.randint(n - window_size)
        if abs(j-k) < window_size:
            continue
        # Moving mass: add delta to h[j:j+ws], subtract from h[k:k+ws]
        # This changes corr by delta * (B_j + B_{j+1} + ... + B_{j+ws-1} - B_k - ...)
        B_sum_j = sum(compute_B(h, jj, n) for jj in range(j, j+window_size))
        B_sum_k = sum(compute_B(h, kk, n) for kk in range(k, k+window_size))
        D = B_sum_j - B_sum_k
        # Constraints: h[j:j+ws] + delta <= 1, h[k:k+ws] - delta >= 0
        dlo = max(h[k:k+window_size] - 1.0).max()  # h[k+k] - delta >= 0 => delta <= min(h[k:k+ws])
        dlo = -min(h[j:j+window_size])  # h[j+j] + delta >= 0 always true since h>=0
        # Actually: h[j+j]+delta <= 1 => delta <= 1-h[j+j], for all j in window
        # h[k+k]-delta >= 0 => delta <= h[k+k], for all k in window
        dhi = min(min(1.0 - h[j:j+window_size]), min(h[k:k+window_size]))
        dlo = 0.0  # We only consider positive delta for simplicity
        if dhi - dlo < 1e-13:
            continue
        delta = solve_1d(corr, D, dlo, dhi)
        if abs(delta) > 1e-13:
            corr += delta * D

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        elif d > 0:
            i = int(np.argmin(h))
            h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + d)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_B(h, n):
    nm1 = n - 1
    ij = np.arange(n)[:, None]
    ik = np.arange(2*n-1)[None, :]
    d1 = ik - ij
    B = np.zeros((n, 2*n-1))
    v1 = (d1 >= 0) & (d1 <= nm1)
    B[v1] += (1.0 - h[np.clip(nm1 - d1, 0, nm1)[v1]])
    d2 = ik + ij
    v2 = (d2 >= nm1) & (d2 <= 2*nm1)
    B[v2] += -h[np.clip(d2 - nm1, 0, nm1)[v2]]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    g = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * g[:, None], axis=1)
    bi = int(np.argmin(vals))
    if bi == 0:
        return lo
    if bi == len(g) - 1:
        return hi
    a, b = g[bi-1], g[bi+1]
    va = float(np.max(A + D * a))
    for _ in range(40):
        m = (a + b) * 0.5
        vm = float(np.max(A + D * m))
        if vm <= va:
            a, va = m, vm
        else:
            b = m
    return (a + b) * 0.5

def pair_sweep(h, corr, B, rng, n, n_cand=25):
    order = np.arange(n)
    rng.shuffle(order)
    for j in order:
        Bj = B[j]
        best_v = float(np.max(corr))
        best_d = 0.0
        best_D = None
        best_k = -1
        ks = []
        for dd in (1, -1, 2, -2, 3, -3, n//4, -n//4, n//3, -n//3, n//2, -n//2):
            ks.append((j + dd) % n)
        for _ in range(max(0, n_cand - len(ks))):
            ks.append(rng.randint(n))
        seen = set()
        for k in ks:
            if k == j or k in seen:
                continue
            seen.add(k)
            D = Bj - B[k]
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-13:
                continue
            d = solve_1d(corr, D, dlo, dhi)
            v = float(np.max(corr + D * d))
            if v < best_v - 1e-14:
                best_v, best_d, best_D, best_k = v, d, D, k
        if best_k >= 0 and abs(best_d) > 1e-13:
            corr += best_d * best_D
            h[j] = float(np.clip(h[j] + best_d, 0.0, 1.0))
            h[best_k] = float(np.clip(h[best_k] - best_d, 0.0, 1.0))
            # Update B for j and best_k
            B[j] = build_B_row(h, j, n)
            B[best_k] = build_B_row(h, best_k, n)

def build_B_row(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    ks = max(0, j)
    ke = min(2*n-2, j + nm1)
    if ks <= ke:
        B[ks:ke+1] += 1.0 - h[j + nm1 - np.arange(ks, ke+1)]
    ks2 = max(0, nm1 - j)
    ke2 = min(2*n-2, 2*nm1 - j)
    if ks2 <= ke2:
        B[ks2:ke2+1] += -h[j + np.arange(ks2, ke2+1) - nm1]
    return B

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    mv = float(np.max(corr))
    sh = alpha * (corr - mv)
    ev = np.exp(sh)
    w = ev / ev.sum()
    obj = mv + np.log(ev.sum()) / alpha
    B = build_B(h, n)
    grad = B @ w
    return obj, grad

def subgrad_step(h0, n, target, alpha, n_iter, lr0, mom):
    h = h0.copy()
    v = np.zeros(n)
    bc = compute_c5(h)
    bh = h.copy()
    for t in range(n_iter):
        _, g = smooth_max_grad(h, alpha)
        lr = lr0 / (np.sqrt(t + 1) + 1.0)
        v = mom * v + lr * g
        h = np.clip(h - v, 0.0, 1.0)
        h = project_to_sum(h, target)
        c5 = compute_c5(h)
        if c5 < bc:
            bc = c5
            bh = h.copy()
        if (t + 1) % 200 == 0:
            # Periodic pair descent micro-step
            corr = compute_corr(h)
            B = build_B(h, n)
            h, corr, _ = pair_sweep(h, corr, B, np.random.RandomState(t), n, n_cand=8)
    return bh, bc

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(m.sum())
        if cnt > 0:
            h[m] += d / cnt
        elif d > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + d)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_B_row(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    ks = max(0, j)
    ke = min(2 * n - 2, j + nm1)
    if ks <= ke:
        B[ks:ke + 1] += 1.0 - h[j + nm1 - np.arange(ks, ke + 1)]
    ks2 = max(0, nm1 - j)
    ke2 = min(2 * n - 2, 2 * nm1 - j)
    if ks2 <= ke2:
        B[ks2:ke2 + 1] += -h[j + np.arange(ks2, ke2 + 1) - nm1]
    return B

def build_B_all(h, n):
    nm1 = n - 1
    ij = np.arange(n)[:, None]
    ik = np.arange(2 * n - 1)[None, :]
    B = np.zeros((n, 2 * n - 1))
    d1 = ik - ij
    v1 = (d1 >= 0) & (d1 <= nm1)
    B[v1] += (1.0 - h[np.clip(nm1 - d1, 0, nm1)[v1]])
    d2 = ik + ij
    v2 = (d2 >= nm1) & (d2 <= 2 * nm1)
    B[v2] += -h[np.clip(d2 - nm1, 0, nm1)[v2]]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    g = np.linspace(lo, hi, 14)
    vals = np.max(A[None, :] + D[None, :] * g[:, None], axis=1)
    bi = int(np.argmin(vals))
    if bi == 0:
        return lo
    if bi == len(g) - 1:
        return hi
    a = float(g[bi - 1])
    b = float(g[bi + 1])
    va = float(np.max(A + D * a))
    for _ in range(36):
        m = (a + b) * 0.5
        vm = float(np.max(A + D * m))
        if vm <= va:
            a, va = m, vm
        else:
            b = m
    return (a + b) * 0.5

def pair_sweep(h, corr, B, rng, n, n_cand=20):
    order = np.arange(n)
    rng.shuffle(order)
    top3 = np.argsort(corr)[-3:]
    top_lags = [(int(l) - (n - 1)) for l in top3]
    for j in order:
        Bj = B[j]
        best_v = float(np.max(corr))
        best_d = 0.0
        best_D = None
        best_k = -1
        ks = []
        for ml in top_lags:
            ks.append((j + ml) % n)
            ks.append((j - ml) % n)
        for dd in (1, -1, 2, -2, n // 4, -n // 4, n // 3, -n // 3, n // 2, -n // 2):
            ks.append((j + dd) % n)
        n_extra = max(0, n_cand - len(ks))
        for _ in range(n_extra):
            ks.append(rng.randint(n))
        seen = set()
        for k in ks:
            if k == j or k in seen:
                continue
            seen.add(k)
            D = Bj - B[k]
            dlo = max(-h[j], h[k] - 1.0)
            dhi = min(1.0 - h[j], h[k])
            if dhi - dlo < 1e-13:
                continue
            d = solve_1d(corr, D, dlo, dhi)
            v = float(np.max(corr + D * d))
            if v < best_v - 1e-14:
                best_v, best_d, best_D, best_k = v, d, D, k
        if best_k >= 0 and abs(best_d) > 1e-13:
            corr += best_d * best_D
            h[j] = float(np.clip(h[j] + best_d, 0.0, 1.0))
            h[best_k] = float(np.clip(h[best_k] - best_d, 0.0, 1.0))
            B[j] = build_B_row(h, j, n)
            B[best_k] = build_B_row(h, best_k, n)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    mv = float(np.max(corr))
    sh = alpha * (corr - mv)
    ev = np.exp(sh)
    w = ev / ev.sum()
    obj = mv + np.log(ev.sum()) / alpha
    Bm = build_B_all(h, n)
    grad = Bm @ w
    return obj, grad

def subgrad_optimize(h0, n, target, alpha, n_iter, lr0, mom, rng_sg):
    h = h0.astype(float).copy()
    v = np.zeros(n)
    bc = compute_c5(h)
    bh = h.copy()
    for t in range(n_iter):
        _, g = smooth_max_grad(h, alpha)
        lr = lr0 / (float(np.sqrt(t + 1)) + 1.0)
        v = mom * v + lr * g
        h_new = np.clip(h - v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target)
        if (t + 1) % 300 == 0:
            corr_sg = compute_corr(h_new)
            B_sg = build_B_all(h_new, n)
            pair_sweep(h_new, corr_sg, B_sg, rng_sg, n, 10)
            h_new = project_to_sum(h_new, target)
        c5n = compute_c5(h_new)
        if c5n < bc:
            bc = c5n
            bh = h_new.copy()
        h = h_new
    return bh, bc
