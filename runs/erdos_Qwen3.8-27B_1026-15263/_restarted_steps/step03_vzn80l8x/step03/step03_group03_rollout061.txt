```python
    idx2 = i_idx + m_vals[None, :]  # shape (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2_clipped = np.clip(idx2, 0, n-1)
    term2 = np.sum(w[None, :] * h[idx2_clipped] * valid2, axis=1)
    
    grad = dx * (term1 - term2)
    return obj, grad
```

```python
            improved = True
```

```python
    h = h0.copy()

    v = np.zeros(n)

    step = 0

def pair_descent(h, corr, rng, n_cand=12, n_rounds=1):
    n = len(h)
    improved_total = False
    for _round in range(n_rounds):
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            B_j = compute_B_vec(h, j, n)
            best_val = np.max(corr)
            best_delta = 0.0
            best_D = None
            best_k = -1
            candidates = [int((j + 1) % n), int((j - 1 + n) % n), int((j + n // 2) % n)]
            for _ in range(max(0, n_cand - 3)):
                candidates.append(rng.randint(n))
            seen = set()

                k = int(k)
                if k == j or k in seen:
                    continue
                seen.add(k)
                B_k = compute_B_vec(h, k, n)
                D = B_j - B_k
                delta_lo = max(-h[j], h[k] - 1.0)
                delta_hi = min(1.0 - h[j], h[k])
                if delta_hi - delta_lo < 1e-14:
                    continue
                delta_opt = solve_1d_max(corr, D, delta_lo, delta_hi)
                val_opt = np.max(corr + D * delta_opt)
                if val_opt < best_val - 1e-14:
                    best_val = val_opt
                    best_delta = delta_opt
                    best_D = D
                    best_k = k
            if best_k >= 0 and abs(best_delta) > 1e-14:
                corr = corr + best_delta * best_D
                h[j] = min(1.0, max(0.0, h[j] + best_delta))
                h[best_k] = min(1.0, max(0.0, h[best_k] - best_delta))
                improved_total = True
    return h, corr, improved_total
```
