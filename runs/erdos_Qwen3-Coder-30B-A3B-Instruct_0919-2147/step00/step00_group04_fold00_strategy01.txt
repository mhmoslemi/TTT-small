def project_box_sum(v, lower, upper, target):
    # Clip to bounds first, then adjust to meet sum with water-filling
    w = np.clip(v, lower, upper)
    current = w.sum()
    if abs(current - target) < 1e-12:
        return w
    # Determine direction
    if current < target:
        # need to increase some components not at upper bound
        free = np.where(w < upper)[0]
        while True:
            if free.size == 0:
                break
            free_vals = w[free]
            inc = (target - w.sum()) / free.size
            # compute max possible increment before hitting upper bound
            max_inc = np.min(upper - free_vals)
            if inc <= max_inc:
                w[free] += inc
                break
            else:
                # fill those that hit bound
                w[free] += max_inc
                # now some may be at upper
                saturated = (w >= upper - 1e-12)
                free = np.where(~saturated)[0]
    else:
        # need to decrease some components not at lower bound
        free = np.where(w > lower)[0]
        while True:
            if free.size == 0:
                break
            free_vals = w[free]
            dec = (w.sum() - target) / free.size
            max_dec = np.min(free_vals - lower)
            if dec <= max_dec:
                w[free] -= dec
                break
            else:
                w[free] -= max_dec
                saturated = (w <= lower + 1e-12)
                free = np.where(~saturated)[0]
    # after adjustments may have tiny floating errors, clip again
    w = np.clip(w, lower, upper)
    # final correction for any small residual by adding to some element
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally but ensure bounds
        # Add diff to first element that can accommodate
        for i in range(len(w)):
            if lower <= w[i] + diff <= upper:
                w[i] += diff
                break
    return w

def gradient_for_shift(h, shift, dx):
    N = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # term1: (1 - h_{i+shift}) for i=0..N-1-shift
        grad[:N-shift] = 1 - h[shift:]
        # term2: h_{i-shift} for i=shift..N-1
        grad[shift:] -= h[:N-shift]
    else:
        s = -shift
        # shift negative: shift = -s
        # term1: (1 - h_{i-s}) for i=s..N-1
        grad[s:] = 1 - h[:N-s]
        # term2: h_{i+s} for i=0..N-s-1
        grad[:N-s] -= h[s:]
    grad *= dx
    return grad

import numpy as np

def compute_c_vec(h, dx):
    """Return the overlap vector C_k for all integer shifts."""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(v, lower, upper, target):
    """Project v onto the hyperplane sum(v)=target with box bounds."""
    w = np.clip(v, lower, upper)
    cur = w.sum()
    eps = 1e-12
    # increase if needed
    if cur < target - eps:
        free = np.where(w < upper - eps)[0]
        while free.size > 0 and cur < target - eps:
            inc = (target - cur) / free.size
            max_inc = np.min(upper - w[free])
            step = min(inc, max_inc)
            w[free] += step
            cur = w.sum()
            free = np.where(w < upper - eps)[0]
    # decrease if needed
    elif cur > target + eps:
        free = np.where(w > lower + eps)[0]
        while free.size > 0 and cur > target + eps:
            dec = (cur - target) / free.size
            max_dec = np.min(w[free] - lower)
            step = min(dec, max_dec)
            w[free] -= step
            cur = w.sum()
            free = np.where(w > lower + eps)[0]
    # final tiny correction
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        for i in range(len(w)):
            if lower <= w[i] + diff <= upper:
                w[i] += diff
                break
    return w

def gradient_for_shift(h, shift, dx):
    """Gradient of C_shift = ∫ h(x)*(1-h(x+shift)) dx w.r.t. h."""
    N = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        # term (1 - h_{i+shift})
        grad[:N - shift] = 1.0 - h[shift:]
        # term -h_{i-shift}
        grad[shift:] -= h[:N - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[:N - s]
        grad[:N - s] -= h[s:]
    grad *= dx
    return grad

def optimize(N, budget, rng):
    """Run many random restarts of a subgradient descent for a given N."""
    dx = 2.0 / N
    target = N / 2.0
    best_h = None
    best_c = np.inf
    start = time.time()
    # parameters
    max_inner_iters = 5000
    stagnation_limit = 2000
    top_k = 3

    while time.time() - start < budget:
        # random feasible start
        h = rng.rand(N)
        h = project_box_sum(h, 0.0, 1.0, target)

        # inner optimization
        it = 0
        last_improve = 0
        c_vec = compute_c_vec(h, dx)
        cur_max = c_vec.max()

        while it < max_inner_iters and time.time() - start < budget:
            # record best globally
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
                last_improve = it

            # pick a shift among the top_k worst
            if top_k >= len(c_vec):
                chosen_idx = np.argmax(c_vec)
            else:
                top_idxs = np.argpartition(-c_vec, top_k)[:top_k]
                chosen_idx = rng.choice(top_idxs)
            shift = chosen_idx - (N - 1)

            # gradient step
            grad = gradient_for_shift(h, shift, dx)
            lr = 0.5 / np.sqrt(it + 1)
            h_new = h - lr * grad
            h_new = project_box_sum(h_new, 0.0, 1.0, target)

            # evaluate
            c_vec = compute_c_vec(h_new, dx)
            new_max = c_vec.max()

            # simple acceptance (always keep the step)
            h = h_new
            cur_max = new_max
            it += 1

            # stagnation detection
            if it - last_improve > stagnation_limit:
                break

    return best_h, best_c

def binary_swap_search(h, dx, budget, rng):
    """Local search on a binary vector using swaps."""
    N = h.shape[0]
    best_h = h.copy()
    best_c = compute_c_vec(best_h, dx).max()
    start = time.time()
    while time.time() - start < budget:
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c_new = compute_c_vec(h_new, dx).max()
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
    return best_h, best_c
