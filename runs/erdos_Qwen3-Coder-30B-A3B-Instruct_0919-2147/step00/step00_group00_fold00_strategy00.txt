```
def project_sum_box(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # We need to find smallest adjustment delta such that sum(h) = target_sum and h stays within [0,1].
    # This is a Euclidean projection onto intersection of hyperplane and box.
    # Can solve using water-filling algorithm: treat each variable allowed to move from its current value to add delta, but bounded.
    # Equivalent to find lambda that when adjusted uniformly for free variables yields sum = target_sum.
    # But easier: Use iterative approach:
    eps = 1e-12
    while True:
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-9:
            break
        # Identify indices not at bounds
        free = (h > 0 + eps) & (h < 1 - eps)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust without violating bounds; break
            break
        # Distribute diff evenly across free indices
        delta = diff / n_free
        # Update h for free indices
        h[free] += delta
        # Clip again any that exceed bounds due to numerical errors
        h = np.clip(h, 0.0, 1.0)
    return h
```

```
def max_overlap_and_shift(h, dx):
    H = np.fft.rfft(h, n=2*N)
    G = np.fft.rfft(1 - h, n=2*N)
    conv = np.fft.irfft(H * np.conj(G), n=2*N)

    idx = np.argmax(conv)
    max_val = conv[idx]
    shift = idx - (N-1)
    return max_val, shift
```
