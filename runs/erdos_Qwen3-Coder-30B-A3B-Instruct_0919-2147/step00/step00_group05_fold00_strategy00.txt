def projection(v):
    low, high = 0., 1.
    S = target_sum
    # Determine bounds for lambda
    lambda_low = np.min(v - high) - 1.0  # ensure all clip(v_i - lambda_low) = high?
    lambda_high = np.max(v - low) + 1.0  # ensure all clip(v_i - lambda_high) = low?
    # Actually as λ→ -∞, v_i - λ → +∞, clipped to high, sum = n*high
    # As λ→ +∞, v_i - λ → -∞, clipped to low, sum = n*low = 0.
    # So we can start with low negative large and high positive large.
    for _ in range(50):
        lam = (lambda_low + lambda_high)/2
        x = np.clip(v - lam, low, high)
        s = x.sum()
        if s > S:
            lambda_low = lam
        else:
            lambda_high = lam
    return np.clip(v - (lambda_low+lambda_high)/2, low, high)

def shift_gradients(h, s):
    n = len(h)
    grad = np.zeros_like(h)
    if s >= 0:
        # a: (1 - h) at indices [s..n-1] assigned to grad[0..n-s-1]
        grad[:n-s] += (1 - h[s:]) 
        # b: -h at indices [0..n-s-1] assigned to grad[s..]

def local_optimize(h, time_remaining, max_iter=20000):
    t_start = time.time()
    n = len(h)
    S = n/2.0
    dx = 2.0 / n
    beta = 5.0
    lr = 5.0  # initial
    best_h = h.copy()
    best_val = evaluate(h)
    iter = 0
    while iter < max_iter and (time.time() - t_start) < time_remaining * 0.99:
        # compute correlation
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_c = np.max(c)
        if max_c < best_val:
            best_val = max_c
            best_h = h.copy()
        # compute weights
        # beta schedule: add small per iter
        if iter % 500 == 0 and beta < 30:
            beta = min(beta + 2, 30)
        exp_term = np.exp(beta * (c - max_c))
        w = exp_term / np.sum(exp_term)
        # compute gradient
        grad = np.zeros_like(h)
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            shift = idx - (n - 1)
            if shift >= 0:
                s = shift
                # term1
                if s < n:
                    grad[:n-s] += weight * (1 - h[s:])
                    # term2
                    grad[s:] += weight * (-h[:n-s])
            else:
                s = -shift
                if s < n:
                    grad[s:] += weight * (1 - h[:n-s])
                    grad[:n-s] += weight * (-h[s:])
        grad *= dx
        # update
        h_new = h - lr * grad
        # project to bounds+sum= S
        h = project_onto_bounds_sum(h_new, S)
        # optionally enforce symmetry
        h = (h + h[::-1]) / 2.0
        # clip (though after projection it's within bounds)
        # optionally adapt lr: if max_c did not change for many steps, reduce lr
        if iter % 500 == 0 and iter > 0:
            # compute new objective
            new_val = evaluate(h)
            if new_val >= best_val:
                lr *= 0.8 # reduce step
            else:
                lr *= 1.05 # slight increase
        iter += 1
    return best_h, best_val
