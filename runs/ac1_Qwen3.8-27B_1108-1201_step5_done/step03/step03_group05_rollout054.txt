import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_descent_direction(a, conv, active_peaks, step_bound):
    """
    Solve LP to find best linearized descent direction.
    minimize t
    s.t. 2 * sum_i a_i * delta[k-i] <= t - conv[k] for k in active_peaks
         sum(delta) = 0
         -a_i <= delta_i <= step_bound
    """
    n = len(a)
    # Variables: delta_0, ..., delta_{n-1}, t
    # t is the last variable (index n)
    
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: 2 * sum_i a_i * delta[k-i] - t <= -conv[k]

    A_ub = np.zeros((len(active_peaks), n + 1))
    b_ub = np.zeros(len(active_peaks))
    
    for idx, k in enumerate(active_peaks):
        k = int(k)
        # 2 * sum_i a_i * delta[k-i] - t <= -conv[k]
        # Coefficients for delta_j: 2 * a[k-j] if 0 <= k-j < n
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A_ub[idx, j] = 2.0 * a[ki]
        A_ub[idx, n] = -1.0  # coefficient of t
        b_ub[idx] = -conv[k]

    bounds = []

    t_opt = result.x[n]

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Linearized LP descent (primary method - solves LP for best first-order descent direction)
2. Smoothed gradient descent with Nesterov momentum (auxiliary)
3. Targeted peak-pair coordinate descent (polish)
4. Multi-phase search across lengths and initializations
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p

    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def lp_descent_direction(a, conv, active_peaks, step_bound):
    """
    Solve LP: minimize t
    s.t. 2 * sum_i a_i * delta[k-i] - t <= -conv[k] for k in active_peaks
         sum(delta) = 0
         -a_i <= delta_i <= step_bound
    Returns (delta, t_opt) or (None, None).
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    m = len(active_peaks)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(active_peaks):
        k = int(k)
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A_ub[idx, j] = 2.0 * a[ki]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -conv[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a[i], step_bound) for i in range(n)]
    bounds.append((None, None))  # t unbounded

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 3})
    except Exception:
        return None, None

    if not result.success:
        return None, None

    delta = result.x[:n].copy()
    t_opt = result.x[n]
    return delta, t_opt


def optimize_lp_linearized(a0, time_limit, seed=42, active_set_size=20, initial_step_bound=0.08):
    """
    Iterative linearized LP-based optimization for minimizing max(conv(a,a)).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    step_bound = initial_step_bound

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        max_val = conv_cur.max()

        # Find active peaks
        k = min(active_set_size, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -k)[-k:]
        threshold = max_val * 0.85
        active = peak_indices[conv_cur[peak_indices] >= threshold]

        if len(active) == 0:
            active = peak_indices[:5]

        # Solve LP
        delta, t_opt = lp_descent_direction(current_a, conv_cur, active, step_bound)

        if delta is None:
            noise = rng.standard_normal(n) * 0.004
            current_a = project_simplex(current_a + noise)
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        improved = False
        alphas = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
        for alpha in alphas:
            a_try = current_a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < max_val * (1.0 - 1e-7):
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 3:
                step_bound = max(0.01, step_bound * 0.8)
            if no_improve > 20:
                noise = rng.standard_normal(n) * 0.003
                current_a = project_simplex(current_a + noise)
                conv_cur = fast_convolve(current_a)
                if conv_cur.max() < best_obj:
                    best_obj = conv_cur.max()
                    best_a = current_a.copy()
                step_bound = initial_step_bound
                no_improve = 0
        else:
            no_improve = 0
            # Gradually increase step bound if improving well
            if iter_count % 50 == 0:
                step_bound = min(0.1, step_bound * 1.05)

        iter_count += 1

    return best_a, best_obj


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=300.0,
                      lr_init=0.015, momentum=0.92, perturb_every=300,
                      perturb_scale=0.004, seed=0):
    """
    Nesterov accelerated projected gradient descent on smoothed objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    perturb_every_cur = perturb_every
    scale_cur = perturb_scale

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = lr_init * (1.0 - 0.6 * frac)
        lr = max(0.001, lr)

        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= perturb_every_cur:
            noise = rng.standard_normal(n) * scale_cur
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every_cur = int(perturb_every_cur * 1.3)
            scale_cur = max(0.001, scale_cur * 0.7)

    return best_a, best_obj


def peak_pair_descent(a0, time_limit, seed=0):
    """
    Targeted peak reduction by adjusting specific element pairs.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Find top peaks
        top_k = min(10, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.92]

        if len(peak_indices) == 0:
            break

        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            if i_start > i_end:
                continue

            # Find the pair with maximum contribution
            indices = range(i_start, i_end + 1)
            contributions = np.array([a[i] * a[k - i] for i in indices])
            max_contrib_idx = np.argmax(contributions)
            i_max = i_start + max_contrib_idx
            j_max = k - i_max

            # Try transferring mass from the larger of the pair to a less critical position
            src = i_max if a[i_max] >= a[j_max] else j_max
            for _ in range(5):
                dst = rng.integers(0, n)
                if abs(dst - src) < 3 or (src == i_max and dst == j_max) or (src == j_max and dst == i_max):
                    continue

                for frac in [0.03, 0.07, 0.12, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-9:
                        break
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-11:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved and iter_count % 1000 == 0:
            # Random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                for frac in [0.01, 0.03, 0.05]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-11:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

        iter_count += 1

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (on simplex)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Bell / inverse quadratic
    center = n / 2.0
    bw = n / 4.0
    tri = 1.0 / (1.0 + ((x - center) / bw) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    # Gaussians
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - center) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal (mass at edges)
    edge = max(2, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Smooth bimodal
    g1 = np.exp(-0.5 * ((x - 0.15 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.85 * n) / (0.1 * n)) ** 2)
    sm_bi = g1 + g2
    sm_bi /= sm_bi.sum()
    inits['smooth_bimodal'] = sm_bi

    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n].copy(), 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. Linearized LP descent (primary: solves LP for best first-order simultaneous peak-reduction direction)
2. Smoothed gradient descent with Nesterov momentum (auxiliary for smoothing and initialization)

4. Multi-phase diverse search across sequence lengths and initializations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a via FFT cross-correlation."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def lp_descent_direction(a, conv, active_peaks, step_bound):
    """
    Solve LP: minimize t
    s.t. conv[k] + 2*sum_i a_i*delta[k-i] <= t  for k in active_peaks
         sum(delta) = 0
         -a_i <= delta_i <= step_bound
    Returns (delta, t_opt) or (None, None).
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    m = len(active_peaks)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(active_peaks):
        k = int(k)
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A_ub[idx, j] = 2.0 * a[ki]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -conv[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a[i], step_bound) for i in range(n)]
    bounds.append((None, None))

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None, None

    if not result.success:
        return None, None

    delta = result.x[:n].copy()
    t_opt = result.x[n]
    return delta, t_opt


def optimize_lp_linearized(a0, time_limit, seed=42, active_set_size=20, initial_step_bound=0.07):
    """Iterative linearized LP-based optimization for minimizing max(conv(a,a))."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    step_bound = initial_step_bound

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        max_val = conv_cur.max()

        k = min(active_set_size, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -k)[-k:]
        threshold = max_val * 0.85
        active = peak_indices[conv_cur[peak_indices] >= threshold]
        if len(active) == 0:
            active = peak_indices[:5]

        delta, t_opt = lp_descent_direction(current_a, conv_cur, active, step_bound)

        if delta is None:
            noise = rng.standard_normal(n) * 0.004
            current_a = project_simplex(current_a + noise)
            no_improve += 1
            if no_improve > 40:
                break
            continue

        improved = False
        alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]
        for alpha in alphas:
            a_try = current_a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < max_val * (1.0 - 1e-8):
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 4:
                step_bound = max(0.01, step_bound * 0.75)
            if no_improve > 25:
                noise = rng.standard_normal(n) * 0.003
                current_a = project_simplex(current_a + noise)
                conv_cur = fast_convolve(current_a)
                if conv_cur.max() < best_obj:
                    best_obj = conv_cur.max()
                    best_a = current_a.copy()
                step_bound = initial_step_bound
                no_improve = 0
        else:
            no_improve = 0
            if iter_count % 80 == 0:
                step_bound = min(0.12, step_bound * 1.04)

        iter_count += 1
        if iter_count % 200 == 0:
            print(f"      LP iter {iter_count}: best={best_obj:.8f}, n={n}", flush=True)

    return best_a, best_obj


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=300.0,
                      lr_init=0.015, momentum=0.92, perturb_every=300,
                      perturb_scale=0.004, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed (log-sum-exp) objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    pe = perturb_every
    ps = perturb_scale

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lr = max(0.001, lr_init * (1.0 - 0.6 * frac))

        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= pe:
            noise = rng.standard_normal(n) * ps
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            pe = int(pe * 1.3)
            ps = max(0.001, ps * 0.7)

    return best_a, best_obj


def peak_pair_descent(a0, time_limit, seed=0):
    """Targeted peak reduction by adjusting contributing element pairs."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        top_k = min(12, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.90]

        if len(peak_indices) == 0:
            break

        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            indices = list(range(i_start, i_end + 1))
            contributions = np.array([a[i] * a[k - i] for i in indices])
            max_idx = np.argmax(contributions)
            i_max = indices[max_idx]
            j_max = k - i_max

            src = i_max if a[i_max] >= a[j_max] else j_max
            for _ in range(4):
                dst = int(rng.integers(0, n))
                if abs(dst - src) < 4 or dst == j_max or dst == i_max:
                    continue
                for frac in [0.04, 0.08, 0.15, 0.25]:
                    amt = a[src] * frac
                    if amt < 1e-9:
                        break
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-11:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved and iter_count % 500 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-6:
                for frac in [0.02, 0.04, 0.06]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-11:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

        iter_count += 1

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n on the simplex."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0

    inits['uniform'] = np.ones(n) / n

    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    bw = n / 4.0
    bell = 1.0 / (1.0 + ((x - center) / bw) ** 2)
    bell /= bell.sum()
    inits['bell'] = bell

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - center) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf * 100)}'] = g

    edge = max(2, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    g1 = np.exp(-0.5 * ((x - 0.15 * n) / (0.12 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.85 * n) / (0.12 * n)) ** 2)
    sm_bi = g1 + g2
    sm_bi /= sm_bi.sum()
    inits['smooth_bimodal'] = sm_bi

    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n].copy(), 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

            if np.isfinite(score_fp) and score_fp < best_score:
                best_score = score_fp
                best_sequence = a_fp.copy()
                print(f"  FinalPeak: {best_score:.6f}", flush=True)

    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}", flush=True)
    return [float(x) for x in best_sequence]
