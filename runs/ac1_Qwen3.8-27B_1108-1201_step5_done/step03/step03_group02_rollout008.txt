import numpy as np

from scipy.fft import rfft, irfft

def fast_autoconv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute the objective value for a normalized sequence (sum=1)."""
    n = len(a)
    c = fast_autoconv(a)
    return 2.0 * n * c.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific lag indices."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)

def lp_solve_for_b(a, M, all_lags=True):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    if all_lags:
        k_indices = list(range(2*n-1))
    else:
        # Only use critical lags (top peaks)
        c = fast_autoconv(a)
        top_k = max(n, 64)
        k_indices = list(np.argpartition(c, -top_k)[-top_k:])
    
    A = build_conv_matrix_rows(a, k_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'mip_rel_gap': 0.1})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.3
    no_improve = 0
    alternating = True  # True: solve for b given a, False: solve for a given b
    
    # We alternate: given current a, find best b, then update a towards b
    # (We could also do the "symmetric" version where we also solve for a given b)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if b_sum < 1e-15:

        b = b / b_sum

def spectral_flatten(a0, time_limit, alpha=0.1, max_iters=1000):
    """Spectral flattening: iteratively modify the spectrum to reduce peaks."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    fft_size = 1 << (2*n-2).bit_length()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autoconv(a)
        M = c.max()
        
        # In the frequency domain, the auto-convolution peak is related to the
        # inverse FFT of |A|^2. We want to flatten this.
        # Strategy: dampen the frequency components that contribute most to the peak.
        A = rfft(a, fft_size)
        power = A.real**2 + A.imag**2
        
        # Modify the power spectrum: reduce power at frequencies where the phase
        # alignment causes constructive interference at the peak.
        # Simple approach: temporarily increase power at frequencies that would
        # cause destructive interference at the peak location.
        
        # Alternative: use the "water-filling" idea
        # c[k] = (1/N) * sum_f |A[f]|^2 * exp(2*pi*i*f*k/N)
        # To reduce c at the peak, we want to reduce |A|^2 at frequencies that
        # contribute constructively there.
        
        # Simpler iterative approach:
        # 1. Find the peak location k_peak
        # 2. Compute the "sensitivity" of c[k_peak] to each frequency component
        # 3. Reduce power at those frequencies
        # 4. Reconstruct a and project to non-negative
        
        k_peak = int(np.argmax(c))
        # Sensitivity: dc[f] = cos(2*pi*f*k_peak/N) (real part of the DFT kernel)
        freqs = np.arange(fft_size // 2 + 1)
        # For rfft, frequencies are 0, 1, ..., N/2
        phase = 2 * np.pi * freqs * k_peak / fft_size
        sensitivity = np.cos(phase)
        
        # Dampen power where sensitivity is high (constructive at peak)
        # and boost where sensitivity is low/negative (destructive at peak)
        weight = 1.0 - alpha * (sensitivity - sensitivity.mean()) / (sensitivity.std() + 1e-15)
        weight = np.maximum(weight, 0.1)
        
        new_power = power * weight
        new_power = np.maximum(new_power, 0)
        
        # Reconstruct: a_new = ifft(sqrt(new_power) * exp(i*angle(A)))
        # Keep the phase of A but modify the magnitude
        magnitude = np.sqrt(new_power)
        phase_A = np.angle(A)
        A_new = magnitude * np.exp(1j * phase_A)
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project to non-negative, normalized
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        c = fast_autoconv(a_new)
        obj = 2.0 * n * c.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Try smaller alpha
            alpha = max(0.01, alpha * 0.9)
        
        if no_improve > 200:
            break
    
    return best_a, best_obj

def fast_autoconv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_rows(a, k_indices):
    """Build rows of convolution matrix for specific lag indices."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            row[js] = a[k - js]
        rows.append(row)
    return np.array(rows)


def lp_solve_direction(a, M, top_k=150, time_limit_lp=15):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for top-k critical lags."""
    n = len(a)
    c_prev = fast_autoconv(a)
    if top_k >= 2 * n - 1:
        k_indices = list(range(2 * n - 1))
    else:
        k_indices = list(np.argpartition(c_prev, -top_k)[-top_k:])
    A = build_conv_rows(a, k_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_lp})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def biconvex_lp_refine(a0, time_limit, top_k=150, step_init=0.3, max_iters=500):
    """Biconvex LP: alternate solving for direction given current sequence."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = c.max()
        b = lp_solve_direction(a, M, top_k=top_k)
        if b is None:
            b = lp_solve_direction(a, M * 1.01, top_k=max(50, top_k // 2))
            if b is None:
                break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        c = fast_autoconv(a_new)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.6, step * 1.01)
        else:
            no_improve += 1
            step = max(0.05, step * 0.97)

        a = a_new

        if no_improve > 150:
            break

    return best_a, best_obj


def smoothed_grad(a, conv, temperature):
    """Compute gradient of softmax-smoothed objective."""
    n = len(a)
    scaled = temperature * conv
    scaled -= scaled.max()
    exp_c = np.exp(scaled)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp

    # grad[j] = 4*n * sum_{m=0}^{n-1} p[j+m] * a[m]
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(a, p[j:j + n])
    grad *= 4.0 * n
    return grad


def optimize_smoothed(a0, time_limit, lr_init=0.003, momentum=0.92, t_init=40.0):
    """Projected gradient on softmax-smoothed objective."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    temp = t_init
    no_improve = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        grad = smoothed_grad(a, c, temp)
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        c = fast_autoconv(a_new)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr = max(0.0001, lr * 0.9)

        if no_improve > 3000:
            break
        if it % 800 == 0:
            temp = min(800.0, temp * 1.15)

    return best_a, best_obj


def spectral_refine(a0, time_limit, alpha_init=0.15, max_iters=500):
    """Spectral shaping: modify magnitude spectrum to flatten auto-convolution."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()

    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autoconv(a)
        k_peak = int(np.argmax(c))

        A = rfft(a, fft_size)
        freqs = np.arange(fft_size // 2 + 1)
        phase_kernel = 2.0 * np.pi * freqs * k_peak / fft_size
        sensitivity = np.cos(phase_kernel)

        sens_mean = sensitivity.mean()
        sens_std = sensitivity.std() + 1e-15
        weight = 1.0 - alpha * (sensitivity - sens_mean) / sens_std
        weight = np.maximum(weight, 0.05)

        power = A.real ** 2 + A.imag ** 2
        new_power = power * weight
        new_power = np.maximum(new_power, 1e-30)

        magnitude = np.sqrt(new_power)
        phase_A = np.angle(A)
        A_new = magnitude * np.cos(phase_A) + 1j * magnitude * np.sin(phase_A)
        a_new = irfft(A_new, fft_size)[:n]

        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        c = fast_autoconv(a_new)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            alpha = max(0.02, alpha * 0.95)

        if no_improve > 100:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, window_size=2, max_iters=200000, rng_seed=123):
    """Local search: optimize small windows of the sequence."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    no_improve = 0
    w = window_size

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        start = int(rng.integers(0, n - w + 1))
        old_vals = a[start:start + w].copy()
        target_sum = old_vals.sum()

        if target_sum < 1e-15:
            continue

        best_window = old_vals.copy()
        best_local = 2.0 * n * c.max()

        # Try several perturbations
        for trial in range(15):
            if trial == 0:
                candidate = old_vals.copy()
            elif trial == 1:
                # Move mass from peak-contributing positions
                candidate = old_vals.copy()
                idx = int(rng.integers(0, w))
                candidate[idx] = 0
                rem = candidate.sum()
                if rem > 1e-15:
                    other = [i for i in range(w) if i != idx]
                    candidate[other[0]] = old_vals[other[0]] + old_vals[idx]
                    candidate = candidate / candidate.sum() * target_sum
                continue
            else:
                noise = rng.standard_normal(w) * target_sum * 0.05
                candidate = np.maximum(old_vals + noise, 0)
                cand_sum = candidate.sum()
                if cand_sum < 1e-15:
                    continue
                candidate = candidate * (target_sum / cand_sum)

            a_test = a.copy()
            a_test[start:start + w] = candidate
            c_test = fast_autoconv(a_test)
            local_obj = 2.0 * n * c_test.max()

            if local_obj < best_local - 1e-13:
                best_local = local_obj
                best_window = candidate

        a[start:start + w] = best_window
        c = fast_autoconv(a)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            c = fast_autoconv(a)
            no_improve = 0
            if time.monotonic() - t0 > time_limit * 0.8:
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos_{freq}'] = a / a_sum

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Raised cosine
    a = 0.5 * (1 + np.cos(2.0 * np.pi * x / n))
    a = np.maximum(a, 1e-10)
    inits['raised_cos'] = a / a.sum()

    # Bimodal
    p1 = int(n * 0.25)
    p2 = int(n * 0.75)
    a = np.zeros(n)
    a = np.exp(-0.5 * ((x - p1) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - p2) / (n * 0.1)) ** 2)
    a /= a.sum()
    inits['bimodal'] = a

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: biconvex LP, smoothed gradient, spectral refine, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")

    # Determine lengths to try
    lengths = [512]
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, min(len(parent), 1024))
    lengths.append(1024)

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 64 <= L <= 2048:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths

    # === Phase 1: Biconvex LP exploration ===
    print("=== Phase 1: Biconvex LP Exploration ===")
    t1_start = time.monotonic()

    for n in lengths:
        remain = deadline - time.monotonic() - 600
        if remain < 60:
            break

        inits = generate_initializations(n, parent, rng)
        # Priority order
        priority = ['parent', 'raised_cos', 'uniform', 'cos_1', 'random', 'gauss']
        names = [nm for nm in priority if nm in inits]
        names += [nm for nm in inits if nm not in priority]
        names = names[:4]

        time_per_segment = max(20.0, remain / (len(names) * 2))

        for name in names:
            remain = deadline - time.monotonic() - 500
            if remain < 15:
                break
            t_alloc = min(time_per_segment, remain)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, _ = biconvex_lp_refine(a0, time_limit=t_alloc, top_k=128, step_init=0.3)
            try_update(a_opt)

            # Second pass with smaller step
            remain = deadline - time.monotonic() - 500
            if remain > 10:
                a_opt2, _ = biconvex_lp_refine(a_opt, time_limit=min(30.0, remain),
                                               top_k=100, step_init=0.15)
                try_update(a_opt2)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Smoothed gradient refinement ===
    print("=== Phase 2: Smoothed Gradient ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        for lr in [0.002, 0.005, 0.001]:
            remain = deadline - time.monotonic() - 300
            if remain < 20:
                break
            t_alloc = min(60.0, remain / 3.0)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_smoothed(best_sequence, time_limit=t_alloc,
                                         lr_init=lr, momentum=0.93, t_init=30.0)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Spectral refinement ===
    print("=== Phase 3: Spectral Refinement ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        for alpha in [0.1, 0.2, 0.05]:
            remain = deadline - time.monotonic() - 200
            if remain < 15:
                break
            t_alloc = min(40.0, remain / 3.0)
            if t_alloc < 8:
                break
            a_opt, _ = spectral_refine(best_sequence, time_limit=t_alloc,
                                       alpha_init=alpha)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Coordinate descent (fine-tuning) ===
    print("=== Phase 4: Coordinate Descent ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        remain = deadline - time.monotonic() - 80
        if remain > 20:
            a_opt, _ = coordinate_descent(best_sequence, time_limit=min(120.0, remain),
                                          window_size=2, rng_seed=99)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Final LP polish on best ===
    print("=== Phase 5: Final LP Polish ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        remain = deadline - time.monotonic() - 30
        if remain > 15:
            a_opt, _ = biconvex_lp_refine(best_sequence, time_limit=min(50.0, remain),
                                          top_k=80, step_init=0.1)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # === Phase 6: Additional coordinate descent if time remains ===
    remain = deadline - time.monotonic() - 10
    if best_sequence is not None and remain > 15:
        print("=== Phase 6: Extended Coordinate Descent ===")
        a_opt, _ = coordinate_descent(best_sequence, time_limit=remain,
                                      window_size=3, rng_seed=777)
        try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = [float(x) for x in best_sequence]

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
