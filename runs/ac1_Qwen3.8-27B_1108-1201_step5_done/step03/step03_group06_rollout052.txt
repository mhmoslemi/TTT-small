def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

def lp_direction(a, M, top_k=None, conv=None):
    """Solve LP for direction."""
    n = len(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    else:
        k_idx = np.arange(num_k)
    
    # Build constraint matrix
    A_ub = np.zeros((len(k_idx), n))
    for row, k in enumerate(k_idx):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[row, idx] = a[k - idx]
    
    b_ub = np.full(len(k_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:

                      options={'presolve': True})
        if res.success and res.fun > -1e10:
            return res.x
    except Exception:
        pass
    return None

Multi-phase optimization for minimum 2*n*max(conv(a,a))/sum(a)^2.
Uses power-mean smooth optimization with Adam, LP-based direction finding,
multi-length exploration, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

# --- Core functions ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_conv_gradient(a, weights):
    """Gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    fft_size = 1 << (2 * (2 * n - 1) - 1).bit_length()  # need size >= 2*(2n-1)-1
    # Actually we need the result to have 2n-1 elements, so fft_size >= 4n-1
    fft_size = 1 << (4 * n - 1).bit_length()
    
    w_p = np.zeros(fft_size)
    w_p[:len(weights)] = weights
    a_p = np.zeros(fft_size)
    a_p[:n] = a
    
    W = np.fft.fft(w_p)
    A = np.fft.fft(a_p)
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # First term: sum_k w[k] * a[k-i]
    # Let j = k-i: sum_j w[j+i] * a[j] = cross-correlation of a and w at lag i
    # = IFFT(FFT(a) * conj(FFT(w)))[i]
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    
    # Second term: sum_k w[k] * a[i-k]
    # Let j = i-k, k = i-j: sum_j w[i-j] * a[j]
    # = convolution of a and w at position i
    # = IFFT(FFT(a) * FFT(w))[i]
    conv_term = np.fft.ifft(A * W).real[:n]
    
    return cross + conv_term

def softmax_weights(conv, beta):
    """Compute softmax weights for approximating max."""
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w

        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                # Perturb and reset
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                lr = max(0.001, lr * 0.6)
        
        if iters % 30 == 0:
            beta = min(beta_max, beta * 1.08)
        
        iters += 1
        if verbose and iters % 200 == 0:
            print(f"      SA it={iters}: obj={best_obj:.7f} beta={beta:.0f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
    
    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()

def lp_solve_direction(a, M, k_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for k in k_indices."""
    n = len(a)
    A_ub = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[row, idx] = a[k - idx]
    
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a0, time_limit, step_init=0.8, top_k_ratio=0.15, seed=0, verbose=False):
    """LP-based optimization with top-k active set."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    num_k = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Select top-k constraints
        top_k = max(10, int(num_k * top_k_ratio))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        b = lp_solve_direction(a, M, peaks)
        if b is None:
            # Try with fewer constraints
            top_k2 = max(5, top_k // 2)
            peaks2 = np.argpartition(conv, -top_k2)[-top_k2:]
            b = lp_solve_direction(a, M, peaks2)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                step = max(0.01, step * 0.7)
            if no_improve > 40:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                step = step_init
                no_improve = 0
                top_k_ratio = min(0.5, top_k_ratio * 1.3)
        
        if iters % 30 == 0:
            step = min(step_init, step * 1.1)
        
        iters += 1
        if verbose and iters % 100 == 0:
            print(f"      LP it={iters}: obj={best_obj:.7f} step={step:.3f} tk={top_k} t={time.monotonic()-t0:.0f}s")
    
    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()

def hybrid_optimize(a0, time_limit, seed=0, verbose=False):
    """Alternate between smooth and LP steps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    num_k = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta = 100
    lr = 0.03
    step = 0.6
    no_improve = 0
    iters = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Smooth step
        conv = np.maximum(fast_convolve(a), 0.0)
        w = softmax_weights(conv, beta)
        grad = compute_conv_gradient(a, w)
        
        t_adam += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1.0 - 0.9 ** t_adam)
        v_hat = v / (1.0 - 0.999 ** t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        
        # LP step every 5 iterations
        if iters % 5 == 0:
            conv_check = fast_convolve(a_new)
            M_check = conv_check.max()
            top_k = max(10, int(num_k * 0.1))
            peaks = np.argpartition(conv_check, -top_k)[-top_k:]
            b = lp_solve_direction(a_new, M_check, peaks)
            if b is not None and b.sum() > 1e-15:
                b = b / b.sum()
                a_new = (1.0 - step) * a_new + step * b
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
            if no_improve > 80:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                no_improve = 0
                beta = min(5000, beta * 2.0)
                step = 0.4 if step > 0.4 else 0.8
        
        if iters % 50 == 0:
            beta = min(5000, beta * 1.1)
        
        iters += 1
        if verbose and iters % 200 == 0:
            print(f"      HY it={iters}: obj={best_obj:.7f} beta={beta:.0f} step={step:.2f} t={time.monotonic()-t0:.0f}s")
    
    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()

def coordinate_descent_polish(a, time_limit, seed=0):
    """Try coordinate-wise and small block updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        improved = False
        
        # Try random single-coordinate changes
        for _ in range(min(50, n)):
            idx = rng.integers(n)
            if a[idx] < 1e-12:
                continue
            
            for frac in [0.6, 0.8, 0.9, 1.1, 1.4]:
                a_try = a.copy()
                delta = a_try[idx] * (frac - 1.0)
                a_try[idx] *= frac
                # Redistribute to others proportionally
                others_mask = np.ones(n, dtype=bool)
                others_mask[idx] = False
                others_vals = a_try[others_mask]
                others_sum = others_vals.sum()
                if others_sum > 1e-15:
                    a_try[others_mask] += -delta * others_vals / others_sum
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            # Try a small random perturbation
            if no_improve % 20 == 0:
                noise = rng.standard_normal(n) * 0.001
                a_try = project_simplex(a + noise)
                conv_try = fast_convolve(a_try)
                if 2.0 * n * conv_try.max() < best_obj - 1e-13:
                    best_obj = 2.0 * n * conv_try.max()
                    best_a = a_try.copy()
                    a = a_try
    
    return best_a, best_obj

def resize_sequence(parent, n_target):
    """Resize parent sequence to target length."""
    if parent is None or len(parent) == 0:
        return None
    p = np.maximum(parent, 0.0).astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        return None
    p = p / s
    if len(p) == n_target:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, len(p))
    x_new = np.linspace(0.0, 1.0, n_target)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 0.0)
    s_new = p_new.sum()
    if s_new < 1e-15:
        return None
    return p_new / s_new

def make_initializations(n, rng, parent=None):
    """Create diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # From parent
    pp = resize_sequence(parent, n)
    if pp is not None:
        inits.append(pp)
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(bi / bi.sum())
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    cos_w = np.maximum(cos_w, 0.001)
    inits.append(cos_w / cos_w.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    
    # Power law
    inits.append(((x + 1) ** 0.3) / ((x + 1) ** 0.3).sum())
    inits.append(((n - x + 1) ** 0.3) / ((n - x + 1) ** 0.3).sum())
    
    # Triangular
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 0.1
    inits.append(tri / tri.sum())
    
    return inits
