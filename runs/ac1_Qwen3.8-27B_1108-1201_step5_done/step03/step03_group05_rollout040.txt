import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_gradient(a, conv, K=5):
    n = len(a)
    actual_K = min(K, len(conv))
    top_idx = np.argpartition(conv, -actual_K)[-actual_K:]
    
    med = np.median(conv)
    excess = np.maximum(conv[top_idx] - med, 1e-10)
    weights = excess / excess.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        k = int(idx)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            m_start = k - j_end
            m_end = k - j_start
            grad[j_start:j_end+1] += 2.0 * w * a[m_start:m_end+1][::-1]
    
    return grad

def freq_init(n, rng, bandwidth=0.8):
    """
    Generate initialization from a flat-ish spectrum.
    The auto-convolution spectrum is |A(w)|^2, so we want |A(w)| to be flat-ish.
    This means A(w) should be roughly constant in magnitude.
    """
    # Create a raised cosine window in frequency domain
    N = 1 << (2 * n - 1).bit_length()  # FFT size
    freqs = np.fft.rfftfreq(N) * N  # 0 to N/2
    # Band-limited: keep only frequencies below bandwidth
    mask = np.abs(freqs) / (N/2) <= bandwidth
    # Raised cosine taper at the edges
    taper = 0.5 * (1 + np.cos(np.pi * np.abs(freqs) / (N/2 * bandwidth)))
    taper[mask] = np.minimum(taper[mask], 1.0)
    taper[~mask] = 0.0
    
    A_mag = taper * (1 + 0.1 * rng.standard_normal(len(taper)))
    A_mag = np.maximum(A_mag, 0)
    
    # Random phase
    phase = rng.uniform(0, 2*np.pi, len(A_mag)) * rng.standard_normal(len(A_mag))
    A_mag[0] = np.abs(A_mag[0])  # DC component positive
    
    A = A_mag * np.exp(1j * phase)
    a = np.fft.irfft(A, N)[:n]
    
    # Project to simplex
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
```

Let me now think about another important aspect: **the choice of n (sequence length)**.

The objective is `2n * max(conv) / (sum(a))^2`. On the simplex (sum=1), this is `2n * max(conv)`.

For a given "shape" of the distribution (e.g., Gaussian with some relative width), increasing n:
- Spreads out the convolution (more terms in the sum)
- But the peak of the convolution scales as... let's see.

If we have a Gaussian-shaped distribution of width `sigma` (in units of n), then the convolution peak is approximately `1/(sigma * sqrt(2*pi))` (for a continuous Gaussian). The objective would be `2n * 1/(sigma * n * sqrt(2*pi)) = 2/(sigma * sqrt(2*pi))`.

So the objective depends on the relative width `sigma/n`, not on n directly! This means that for a fixed shape (relative width), the objective is the same regardless of n.

However, there's a discretization effect: for small n, the discrete approximation to the continuous optimum is better (less quantization error). For very large n, the computation becomes slower.

So the optimal n is somewhere in the "sweet spot" where:
- n is large enough for good discretization
- n is small enough for fast computation

I think n = 256-768 is a good range to explore.

Now, let me also think about a **key insight** that might help: the optimal distribution is NOT a simple Gaussian or bell shape. It's a more complex shape that's specifically designed to flatten the auto-convolution. This is similar to "window design" in signal processing.

The optimal shape can be thought of as the solution to:
minimize max_k conv(a,a)[k]
subject to: a >= 0, sum(a) = 1

This is a semi-infinite optimization problem (infinitely many constraints if we consider the continuous version, but finite for discrete n with 2n-1 constraints).

One approach is to use **Chebyshev approximation** ideas: find the distribution whose auto-convolution deviates least from a constant (in the minimax sense).

Another approach is **iterative reweighting**:
1. Start with a distribution a
2. Compute conv(a,a)
3. The "worst" positions are where conv is largest
4. Reduce the weight at positions that contribute to the worst positions
5. Repeat

This is similar to the "water-filling" or "power allocation" idea.

Actually, one more thought. Let me reconsider the problem from the perspective of the **dual**.

The primal problem (on the simplex) is:
minimize max_k conv(a,a)[k]
subject to: a >= 0, sum(a) = 1

The dual (roughly) involves finding non-negative weights w_k on the convolution positions such that the "effective gradient" `sum_k w_k * grad(conv[k])` is constant (by KKT conditions).

This suggests an algorithm:
1. Start with uniform weights w_k
2. At each step:
   a. Compute the "effective gradient" `g = sum_k w_k * grad(conv[k])`
   b. Find the distribution `a` that makes `g` as constant as possible (subject to a >= 0, sum(a) = 1)
   c. Update weights: increase w_k for positions where conv[k] is large

This is related to the **multiplicative weight update** (MWU) algorithm in game theory and combinatorial optimization.

```python
"""
Minimizes 2n*max(conv(a))/(sum(a))^2 using direct peak-gradient descent
with Nesterov momentum, adaptive peak targeting, frequency-domain
initializations, and multi-phase diverse search with coordinate descent polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_gradient(a, conv, K=5):
    """
    Compute the (sub)gradient of max(conv) approximated by weighted
    sum of gradients at the top-K peaks.
    grad[j] = 2 * sum_{k in topK} w_k * a[k-j] (with bounds)
    """
    n = len(a)
    actual_K = min(K, len(conv))
    top_idx = np.argpartition(conv, -actual_K)[-actual_K:]
    
    # Weight by excess over median (emphasize the true peaks)
    med = np.median(conv)
    excess = np.maximum(conv[top_idx] - med, 1e-10)
    weights = excess / excess.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        k = int(idx)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            m_start = k - j_end
            m_end = k - j_start
            grad[j_start:j_end + 1] += 2.0 * w * a[m_start:m_end + 1][::-1]
    
    return grad


def optimize_peak_grad(a0, time_limit, seed=0, K_init=3, K_max=10,
                       lr_init=0.04, momentum_init=0.85, momentum_end=0.97):
    """
    Nesterov accelerated projected gradient descent using direct peak gradients.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    K = K_init
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        # Adaptive parameters
        momentum = momentum_init + (momentum_end - momentum_init) * frac
        K_cur = min(K_init + int((K_max - K_init) * frac), K_max)
        lr = lr_init * (1.0 - 0.4 * frac) + 0.002
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        conv_y = fast_convolve(y)
        grad = peak_gradient(y, conv_y, K_cur)
        
        # Momentum update
        v = momentum * v + lr * grad
        a_new = project_simplex(a - v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Restart if stuck
        if no_improve > 400:
            scale = 0.002 + 0.003 * rng.random()
            a = best_a + rng.standard_normal(n) * scale
            a = project_simplex(a)
            v = np.zeros(n)
            no_improve = 0
        
        # Occasional aggressive perturbation
        if no_improve > 200 and no_improve % 100 == 0:
            a = project_simplex(a + rng.standard_normal(n) * 0.01)
            v *= 0.3
    
    return best_a, best_obj
