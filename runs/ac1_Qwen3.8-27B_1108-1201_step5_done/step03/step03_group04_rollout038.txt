import time

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Smooth gradient optimization with log-sum-exp smoothing, Adam optimizer,
β-annealing, and multiple restarts to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: using FFT-based cross-correlation for O(n log n) gradient computation,
combined with simplex projection and adaptive step sizes.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def cross_correlate(a, p):
    """Compute (a star p)[j] = sum_m a[m]*p[m+j] for j=0..n-1 via FFT."""
    n = len(a)
    np_ = len(p)
    ar = a[::-1].copy()
    result_len = n + np_ - 1
    fft_size = 1 << (result_len - 1).bit_length()
    Ar = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * P, fft_size)
    return conv[n - 1:n + np_ - 1]


def compute_smooth_gradient(a, beta):
    """Compute gradient of (2n/beta)*log(sum exp(beta*conv_k)) wrt a."""
    n = len(a)
    conv = fast_convolve(a)
    # Numerically stable log-sum-exp
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - max_bc)
    p = exp_bc / np.sum(exp_bc)
    # Gradient = 4n * (a cross-correlate p)[j] for j=0..n-1
    cc = cross_correlate(a, p)
    grad = 4.0 * n * cc[:n]
    smooth_val = (2.0 * n / beta) * (max_bc + np.log(np.sum(exp_bc)))
    return grad, conv, smooth_val


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_optimize(a0, total_time, beta_start=5.0, beta_end=5000.0,
                    n_anneal_steps=200, verbose=False):
    """
    Smooth gradient optimization with Adam and beta annealing.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Track best
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    adam_t = 0
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    total_iters = 0
    no_improve = 0
    max_stall = 50

    while time.monotonic() - t0 < total_time:
        # Anneal beta
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        # Exponential annealing
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient
        grad, conv, smooth_val = compute_smooth_gradient(a, beta)

        # Adam update
        adam_t += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** adam_t)
        v_hat = v_state / (1 - beta2 ** adam_t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a = project_simplex(a_new)

        # Evaluate exact objective
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Stuck handling: perturb and restart Adam
        if no_improve >= max_stall:
            rng = np.random.default_rng(total_iters)
            scale = 0.002 * (1 + no_improve // max_stall)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(a + noise)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            adam_t = 0
            lr = max(0.1, lr * 0.7)
            no_improve = 0
            max_stall = int(max_stall * 1.5)

        # Adapt learning rate
        if total_iters % 100 == 0:
            lr = max(0.05, 0.5 * (0.995 ** (total_iters // 100)))

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SmoothOpt it={total_iters}, obj={best_obj:.6f}, "
                  f"beta={beta:.1f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def lp_refine(a0, M, n_substitutes=3):
    """
    Single LP step: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the LP solution b normalized to sum=1, or None.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a0)
    # Build sparse constraint matrix: A @ b gives conv(a0, b)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a0[k - j])
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs',
                  bounds=bounds, options={'presolve': True, 'time_limit': 10.0})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb > 1e-10:
            return b / sb
    return None


def lp_fw_step(a, time_limit=60.0, verbose=False):
    """Several LP-Frank-Wolfe steps to refine a."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = lp_refine(a, M)
        if b is None:
            break

        # Frank-Wolfe step with line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            candidate = (1 - t) * a + t * b
            conv_c = fast_convolve(candidate)
            obj_c = 2.0 * n * conv_c.max()
            if obj_c < best_obj - 1e-12:
                a = candidate
                best_obj = obj_c
                best_a = a.copy()
                conv = conv_c
                M = conv.max()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-FW it={it}, obj={best_obj:.6f}, M={M:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def get_parent_sequence(n_target):
    """Get parent sequence resized to target length."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            else:

            return p

    return None


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}

    if parent is not None:
        inits['parent'] = parent.copy()

    inits['uniform'] = np.ones(n) / n

    # Smooth random (Gaussian window on random)
    r = rng.random(n)
    x = np.arange(n, dtype=np.float64)
    window = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    smooth_rand = r * window + 0.1 * np.ones(n)
    smooth_rand /= smooth_rand.sum()
    inits['smooth_rand'] = smooth_rand

    # Edge-heavy
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec += 0.05 * rng.random(n)
    ec /= ec.sum()
    inits['edge'] = ec

    # Sine-modulated
    x = np.arange(n, dtype=np.float64)
    sine = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)
    sine = np.maximum(sine, 0.01)
    sine /= sine.sum()
    inits['sine'] = sine

    # Two-hump
    h1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    two = h1 + h2 + 0.05 * np.ones(n)
    two /= two.sum()
    inits['two_hump'] = two

    return inits
