```python
"""
Hybrid LP-Frank-Wolfe + Fast Coordinate Descent optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Improvements over previous approach:
- Spectral and ramp-based initializations with theoretical motivation
- Fast coordinate descent exploiting O(n) convolution update per coordinate change
- Adaptive LP oracle with progressive tightening
- Better time allocation and perturbation strategy
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_conv_max(a):
    """Compute just the max of convolution and sum."""
    n = len(a)
    conv = fast_convolve(a)
    return conv.max(), a.sum(), conv


def coord_descent_fast(a, n_sweeps=20, time_limit=60.0, verbose=False):
    """
    Fast coordinate descent using O(n) convolution update per coordinate change.
    For coordinate i, changing a[i] from x_old to x_new changes conv[k] by:
      - 2*(x_new - x_old)*a[k-i] for 0 <= k-i < n, k != 2i
      - (x_new^2 - x_old^2) for k = 2i (if valid)
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = fast_convolve(a).copy()
    cur_max = conv.max()
    cur_sum = a.sum()
    best_obj = 2.0 * n * cur_max / (cur_sum * cur_sum)
    best_a = a.copy()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[i]
            if x_old < 1e-14:
                continue

            # Precompute the delta coefficients for this coordinate
            # conv[k] changes by: delta_x * (a[k-i] + a[k-i] if k!=2i) + quad_correction
            # delta_x = x_new - x_old
            # For k where 0 <= k-i < n:
            #   if k != 2i: linear coeff = 2*a[k-i], quad coeff = 0
            #   if k == 2i: linear coeff = 2*a[i], quad coeff = -x_old (since x_new^2 - x_old^2 = (x_new-x_old)*(x_new+x_old), and x_new = x_old + delta, so = delta*(2*x_old + delta))
            # Actually let me just use: new_conv[k] = conv[k] + 2*(x_new-x_old)*a[k-i] for k!=2i
            #                                       = conv[k] + (x_new^2 - x_old^2) = conv[k] + (x_new-x_old)*(x_new+x_old) for k==2i

            # Affected indices: k from i to i+n-1 (where k-i is in [0, n-1])
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Try several candidate values
            max_a = a.max()
            candidates = [x_old]
            if x_old > 1e-10:
                candidates.extend([x_old * f for f in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0]])
                candidates.append(max_a * 0.5)
            candidates.append(0.0)

            best_x = x_old
            best_local_obj = best_obj

            # Current objective for reference
            cur_obj = 2.0 * n * cur_max / (cur_sum * cur_sum)

            for x_new in candidates:
                if abs(x_new - x_old) < 1e-12:
                    continue
                dx = x_new - x_old
                cur_sum_new = cur_sum + dx
                if cur_sum_new < 1e-10:
                    continue

                # Update conv in-place (we'll restore later)
                max_new = cur_max
                for k in range(k_start, k_end + 1):
                    ki = k - i  # this is in [0, n-1]
                    if k == 2 * i:
                        # quad term: x_new^2 - x_old^2
                        old_val = conv[k]
                        new_val = old_val + x_new * x_new - x_old * x_old
                        conv[k] = new_val
                    else:
                        old_val = conv[k]
                        new_val = old_val + 2.0 * dx * a[ki]
                        conv[k] = new_val
                    if new_val > max_new:
                        max_new = new_val

                obj_new = 2.0 * n * max_new / (cur_sum_new * cur_sum_new)
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_x = x_new

                # Restore conv
                for k in range(k_start, k_end + 1):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] = conv[k] - (x_new * x_new - x_old * x_old)
                    else:
                        conv[k] = conv[k] - 2.0 * dx * a[ki]
                    # Fix floating point
                    if conv[k] < 0 and abs(conv[k]) < 1e-15:
                        conv[k] = 0.0

            if best_x != x_old:
                # Apply the best change
                dx = best_x - x_old
                cur_sum += dx
                for k in range(k_start, k_end + 1):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] += best_x * best_x - x_old * x_old
                    else:
                        conv[k] += 2.0 * dx * a[ki]
                    if conv[k] < 0 and abs(conv[k]) < 1e-15:
                        conv[k] = 0.0
                a[i] = best_x
                cur_max = conv.max()
                improved = True

                cur_obj = 2.0 * n * cur_max / (cur_sum * cur_sum)
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}, elapsed={elapsed:.1f}s")
        if not improved and sweep > 2:
            break

    return best_a, best_obj


def gen_spectral_init(n, rng):
    """Generate a spectral initialization using random phases."""
    # Create random complex spectrum with random phases
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random magnitudes (use a smooth envelope to get a reasonable temporal shape)
    freqs = np.arange(fft_size // 2 + 1)
    # Envelope: low-pass to ensure smoothness
    envelope = np.exp(-0.5 * (freqs / (0.3 * n)) ** 2)
    magnitudes = envelope * (0.5 + 0.5 * rng.random(fft_size // 2 + 1))
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    if r := True:
        phases[0] = 0  # DC component real
    spectrum = magnitudes * np.exp(1j * phases)
    # Enforce Hermitian symmetry for real IFFT
    if fft_size % 2 == 0:
        spectrum[-1] = np.conj(spectrum[-1])  # Nyquist
    # Full spectrum
    full_spec = np.zeros(fft_size, dtype=complex)
    full_spec[:fft_size // 2 + 1] = spectrum
    for i in range(1, fft_size // 2):
        full_spec[fft_size - i] = np.conj(spectrum[i])
    # IFFT
    sig = np.fft.ifft(full_spec).real
    sig = np.abs(sig[:n])  # Take magnitude to ensure non-negativity
    s = sig.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return sig / s


def gen_ramp_init(n, rng):
    """Generate a ramp-like initialization (theoretically motivated)."""
    # Various ramp profiles
    profiles = []
    # Linear ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    profiles.append(r / r.sum())
    # Power ramp
    for p in [0.5, 2.0, 3.0]:
        r = np.arange(1, n + 1, dtype=np.float64) ** p
        profiles.append(r / r.sum())
    # Exponential ramp
    for alpha in [0.5, 1.0, 2.0]:
        r = np.exp(alpha * np.linspace(0, 1, n))
        profiles.append(r / r.sum())
    # Reversed
    profiles.append(profiles[0][::-1].copy())

    # Pick randomly or use all
    chosen = profiles[rng.integers(len(profiles))]
    # Add small noise
    noise = rng.standard_normal(n) * 0.01
    chosen = np.maximum(chosen + noise, 0)
    chosen /= chosen.sum()
    return chosen


def gen_edge_init(n, rng, frac=0.3):
    """Edge-concentrated initialization."""
    a = np.zeros(n)
    edge_len = max(2, int(n * frac))
    a[:edge_len] = 1.0
    a[-edge_len:] = 1.0
    # Add decay
    x = np.arange(n)
    a *= np.exp(-0.5 * ((np.minimum(x, n - 1 - x)) / (0.2 * n)) ** 2)
    # Randomize edges slightly
    a[:edge_len] *= 0.5 + 0.5 * rng.random(edge_len)
    a[-edge_len:] *= 0.5 + 0.5 * rng.random(edge_len)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def gen_diverse_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    inits['ramp_rev'] = r[::-1].copy()

    # Power ramps
    r2 = np.arange(1, n + 1, dtype=np.float64) ** 2
    r2 /= r2.sum()
    inits['ramp_sq'] = r2.copy()

    # Edge
    inits['edge'] = gen_edge_init(n, rng, frac=0.4)
    inits['edge_strong'] = gen_edge_init(n, rng, frac=0.2)

    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Spectral
    inits['spectral'] = gen_spectral_init(n, rng)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random with concentration
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 0.1)

    # W-shaped (two peaks)
    w1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    w2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    w = w1 + w2
    w /= w.sum()
    inits['wshape'] = w.copy()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def fpw_lp_optimize(a0, time_limit, verbose=False, verbose_every=100):
    """Frank-Wolfe iterations with LP oracle and adaptive M."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M  # with sum=1 normalization
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150
    m_tighten = 1.0  # multiplier on M for constraint tightness

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Adaptive time limit for LP
        lp_time = min(5.0, remaining * 0.05)
        if it < 5:
            lp_time = min(8.0, remaining * 0.1)

        b = lp_oracle(a, M * m_tighten, time_limit=lp_time)
        if b is None:
            # LP failed, try subgradient step
            peaks = np.argmax(conv)
            g = np.zeros(n)
            for k in [peaks]:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * a[k - idx]
            lr = 0.01 / (1 + it * 0.01)
            a_new = np.maximum(a - lr * g / g.max(), 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a = a_new / s_new
            no_improve += 5
            conv = fast_convolve(a)
            M = conv.max()
            it += 1
            continue

        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        # Check if b same as a
        diff = np.abs(a - b).max()
        if diff < 1e-10:
            no_improve += 1
            # Try tightening M
            m_tighten = max(0.85, m_tighten * 0.98)
        else:
            # Line search
            best_t = 0.0
            best_step_obj = 2.0 * n * M
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                candidate = (1 - t) * a + t * b
                c = fast_convolve(candidate)
                step_obj = 2.0 * n * c.max()
                if step_obj < best_step_obj:
                    best_step_obj = step_obj
                    best_t = t

            if best_t < 0.05:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                s_a = a.sum()
                a = a / s_a  # renormalize
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                    # Reset tighten
                    m_tighten = min(1.0, m_tighten * 1.02)
                else:
                    no_improve += 1

        # Stuck handling
        if no_improve >= max_stall:
            rng = np.random.default_rng(it * 7919)
            # Various escape strategies
            strategy = it % 3
            if strategy == 0:
                noise = rng.standard_normal(n) * 0.005
                a = np.maximum(a + noise, 0.0)
            elif strategy == 1:
                # Random reweighting
                w = rng.dirichlet(np.ones(n) * 0.5)
                a = 0.7 * a + 0.3 * w
            else:
                # Shift-based perturbation
                shift = rng.integers(1, max(2, n // 10))
                a_shifted = np.roll(a, shift)
                a_shifted = np.maximum(a_shifted, 0)
                a = 0.8 * a + 0.2 * a_shifted
                a_shifted2 = np.roll(a, -shift)
                a = 0.8 * a + 0.2 * a_shifted2

            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            m_tighten = 1.0

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, no_improve={no_improve}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.92):
    """Polish with projected subgradient on simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Multi-peak subgradient with softmax weighting
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.03, 1e-15)
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            k = int(peaks[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[ii] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
            v *= 0.5
        else:
            a_new = a_new / s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            stall = 0

        lr *= 0.9997
        it += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid LP-FW + coordinate descent search.
    Returns the best sequence found that minimizes the evaluation objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # === Phase 1: Multi-length, multi-init LP-FW scan ===
    print("=== Phase 1: LP-FW Scan ===")
    lengths = [384, 256, 512]

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        inits = gen_diverse_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'ramp', 'ramp_rev', 'ramp_sq',
                    'edge', 'edge_strong', 'wshape', 'spectral',
                    'uniform', 'gauss', 'random', 'random_conc']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Allocate time proportional to remaining budget
        remaining_for_phase = deadline - time.monotonic() - 300  # leave time for later phases
        n_inits_to_try = min(len(init_names), 7)
        time_per_init = max(10.0, remaining_for_phase / n_inits_to_try / 3)

        for name in init_names[:n_inits_to_try]:
            if time.monotonic() > deadline - 300:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 280)
            if t_alloc < 8:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t_alloc={t_alloc:.0f}s")
            a_opt, obj_opt = fpw_lp_optimize(a0, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}, local_obj={obj_opt:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    *** NEW BEST: {score:.6f} ***")

    print(f"After Phase 1: best={best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    # === Phase 2: Coordinate Descent Polish ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 2: Coordinate Descent Polish ===")
        remaining = deadline - time.monotonic() - 200
        t_cd = min(120.0, remaining * 0.4)
        if t_cd > 10:
            a_cd, obj_cd = coord_descent_fast(best_sequence, n_sweeps=30, time_limit=t_cd, verbose=True)
            score = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score):
                print(f"  CD score={score:.6f} (was {best_score:.6f})")
                if score < best_score:
                    best_score = score
                    best_sequence = a_cd.copy()
                    print(f"  *** NEW BEST: {score:.6f} ***")

    # === Phase 3: Subgradient Polish ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 3: Subgradient Polish ===")
        remaining = deadline - time.monotonic() - 150
        t_sg = min(100.0, remaining * 0.4)
        if t_sg > 10:
            a_sg, obj_sg = subgradient_polish(best_sequence, time_limit=t_sg, lr_init=0.003, momentum=0.93)
            score = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score):
                print(f"  SG score={score:.6f} (was {best_score:.6f})")
                if score < best_score:
                    best_score = score
                    best_sequence = a_sg.copy()
                    print(f"  *** NEW BEST: {score:.6f} ***")
            # CD after SG
            if time.monotonic() < deadline - 40:
                t_cd2 = min(30.0, deadline - time.monotonic() - 30)
                if t_cd2 > 5:
                    a_cd2, obj_cd2 = coord_descent_fast(a_sg, n_sweeps=15, time_limit=t_cd2)
                    score2 = float(evaluate_sequence(a_cd2.tolist()))
                    if np.isfinite(score2) and score2 < best_score:
                        best_score = score2
                        best_sequence = a_cd2.copy()
                        print(f"  CD-after-SG score={score2:.6f} *** NEW BEST ***")

    # === Phase 4: Deep LP optimization on best ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 4: Deep LP Optimization ===")
        remaining = deadline - time.monotonic() - 80
        t_deep = remaining * 0.5
        if t_deep > 10:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_deep, verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f} (was {best_score:.6f})")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  *** NEW BEST: {score:.6f} ***")

    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Different perturbation strategies
            strategy = i % 3
            if strategy == 0:
                scale = 0.003 * (1 + i * 0.3)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
            elif strategy == 1:
                # Coordinate swap perturbation
                a_pert = best_sequence.copy()
                idx1 = rng.integers(n_best)
                idx2 = rng.integers(n_best)
                delta = a_pert[idx1] * 0.2
                a_pert[idx1] -= delta
                a_pert[idx2] += delta
                a_pert = np.maximum(a_pert, 0.0)
            else:
                # Block perturbation
                a_pert = best_sequence.copy()
                block_start = rng.integers(0, max(1, n_best - n_best // 4))
                block_end = min(n_best, block_start + n_best // 4)
                w = rng.dirichlet(np.ones(block_end - block_start) * 0.5)
                total = a_pert[block_start:block_end].sum()
                a_pert[block_start:block_end] = w * total
                a_pert = np.maximum(a_pert, 0.0)

            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert

            t_alloc = min(15.0, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break

            # Short LP-FW + CD
            a_opt, _ = fpw_lp_optimize(a_pert, time_limit=t_alloc * 0.6, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {i} (strategy {strategy}): score={score:.6f} *** NEW BEST ***")
            elif time.monotonic() < deadline - 25:
                # Try CD on the perturbed
                a_cd, _ = coord_descent_fast(a_opt, n_sweeps=10, time_limit=min(10, deadline - time.monotonic() - 20))
                score2 = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_cd.copy()
                    print(f"  Restart {i} CD: score={score2:.6f} *** NEW BEST ***")

    # === Phase 6: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 6: Final Polish ===")
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5:
            a_final, _ = coord_descent_fast(best_sequence, n_sweeps=10, time_limit=remaining)
            score = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_final.copy()
                print(f"  Final polish score={score:.6f} *** NEW BEST ***")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
