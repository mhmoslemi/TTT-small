"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with Nesterov acceleration, adaptive temperature scheduling, LP-based
directional refinement, and multi-length multi-start search with
FFT-based gradient computation exploiting the convolution structure.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, t):
    """
    Compute gradient of F(a) = (1/t)*log(sum_k exp(t*c_k)) where c_k = conv(a,a)_k.
    Returns (gradient_vector, softmax_weights, convolution_values).
    """
    n = len(a)
    conv = fast_convolve(a)  # length 2n-1
    max_c = conv.max()
    w = np.exp(t * (conv - max_c))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    # grad[m] = 2 * sum_{j=0}^{n-1} w[j+m] * a[j] for m=0,...,n-1
    # = 2 * conv(a, w_flip)[2n-2-m] ... using the flip trick
    w_flip = w[::-1]
    conv_full = fast_convolve_cross(a, w_flip)  # length 3n-2
    grad = 2.0 * conv_full[n - 1: 2 * n - 1][::-1]
    return grad, w, conv


def smooth_objective(a, t):
    """Compute smoothed max of conv(a,a)."""
    conv = fast_convolve(a)
    max_c = conv.max()
    val = np.log(np.exp(t * (conv - max_c)).sum()) / t + max_c
    return float(val)


def nesterov_smooth_descent(a_init, time_limit, t_init=30.0, t_min=5.0,
                            eta_init=0.05, momentum_init=0.9,
                            restart_every=200, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on smoothed objective.
    Minimizes F(a) = (1/t)*log(sum exp(t*c_k)) over the simplex,
    with adaptive temperature and momentum scheduling.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.time()
    x = a.copy()
    v = np.zeros(n)  # momentum vector
    momentum = momentum_init
    eta = eta_init
    t = t_init
    stall = 0
    prev_best = best_obj

    while time.time() - t0 < time_limit:
        # Nesterov lookahead
        y = x + momentum * v

        # Compute gradient at y
        gy, w, conv_y = compute_smooth_grad(y, t)
        obj_y = 2.0 * n * conv_y.max()

        # Gradient step + projection
        x_new = project_simplex(y - eta * gy)

        # Update momentum
        v = x_new - x
        x = x_new

        # Evaluate
        conv = fast_convolve(x)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive: decrease temperature over time
        elapsed_frac = (time.time() - t0) / time_limit
        t = max(t_min, t_init * (1.0 - 0.7 * elapsed_frac))

        # Adaptive: decrease eta slowly
        eta = max(0.001, eta_init * (1.0 - 0.5 * elapsed_frac))

        # Momentum decay
        momentum = max(0.8, 0.95 - 0.1 * elapsed_frac)

        # Restart if stuck
        if stall > restart_every:
            x = best_a + np.random.default_rng(int(stall)).standard_normal(n) * 0.003
            x = project_simplex(x)
            v = np.zeros(n)
            momentum = momentum_init
            stall = 0

        if verbose and (stall == 0 or stall % 50 == 0):
            print(f"    t={t:.1f} obj={cur_obj:.8f} best={best_obj:.8f} stall={stall}")

    return best_a, best_obj


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g>=0, conv(a,g)_k <= M for all k.
    Returns optimal g (normalized to sum=1) or None.
    """
    n = len(a)
    # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # This makes (A @ g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_refine(a_init, time_limit, max_iters=50):
    """LP-based directional optimization with blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.time()
    blend = 0.3

    for it in range(max_iters):
        if time.time() - t0 > time_limit - 5:
            break
        M = conv.max()
        g = lp_direction(a, M)
        if g is None:
            break
        a_new = project_simplex((1.0 - blend) * a + blend * g)
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still move, but keep best
        blend = max(0.01, blend * 0.9)

    return best_a, best_obj


def multiplicative_weight_update(a_init, time_limit, iterations=500):
    """
    Multiplicative weight / mirror descent approach.
    Updates a_j <- a_j * exp(-eta * grad_j) / Z, which is natural for simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.time()
    t_param = 20.0 / n  # temperature for smoothing
    eta = 1.0 / np.sqrt(n)

    for it in range(iterations):
        if time.time() - t0 > time_limit - 2:
            break
        grad, w, conv = compute_smooth_grad(a, t_param)
        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
        a = a_new

        # Schedule
        t_param = max(5.0 / n, t_param * 0.995)
        eta = eta * 0.999

    return best_a, best_obj


def coordinate_polish(a_in, time_limit=30.0):
    """Coordinate descent: try moving mass between pairs to reduce peak."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    t0 = time.time()

    # Focus on top peaks
    top_k = 10
    for round_num in range(200):
        if time.time() - t0 > time_limit - 2:
            break
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_max = conv.max()

        for k in peaks:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            # Try reducing the peak by moving mass from peak-contributing elements
            # to elements that don't contribute to this peak
            # Simple: try moving a small amount from the largest contributor
            contribs = [(j, a[j] * a[k - j]) for j in range(j_lo, j_hi + 1)]
            if not contribs:
                continue
            contribs.sort(key=lambda x: -x[1])
            j_primary = contribs[0][0]
            if a[j_primary] < 1e-8:
                continue
            # Try moving to a random other position
            for _ in range(5):
                j_target = np.random.randint(n)
                if j_target == j_primary:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    delta = a[j_primary] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[j_primary] -= delta
                    a_test[j_target] += delta
                    if a_test[j_primary] < -1e-12:
                        continue
                    a_test = np.maximum(a_test, 0)
                    st = a_test.sum()
                    if st < 1e-10:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        break

    return a


def gen_initializations(n, parent=None, rng=None):

    # Edge-concentrated

    e = max(2, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['bimodal'] = g2 / g2.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random with structure (low-freq + noise)

    structured = np.abs(np.fft.irfft(np.fft.rfft(np.ones(n) * (1 + freq)))) + 0.01
    inits['structured'] = structured / structured.sum()

    # Parent (if available)
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                inits['parent_ds'] = p_res / p_res.sum()
            else:
                p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                inits['parent_up'] = p_res / p_res.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Multi-start Nesterov smooth descent
    print("=== Phase 1: Nesterov Smooth Descent ===")
    phase1_deadline = deadline - 600.0
    lengths = [384, 512, 640, 768, 1024]

    results = []
    for n in lengths:
        if time.time() > phase1_deadline - 30:
            break
        inits = gen_initializations(n, parent, rng)
        # Priority: parent first, then structured patterns
        names = [nm for nm in ['parent', 'parent_ds', 'parent_up', 'bimodal',
                                'edge', 'gauss', 'uniform', 'triangle',
                                'random', 'structured'] if nm in inits]
        names = names[:4]  # top 4 per length

        t_per = max(8.0, (phase1_deadline - time.time()) / (len(names) * max(1, len(lengths))))
        t_per = min(t_per, 40.0)

        for name in names:
            if time.time() > phase1_deadline - 15:
                break
            t_alloc = min(t_per, phase1_deadline - time.time() - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]
            # Nesterov smooth descent
            a_opt, obj_opt = nesterov_smooth_descent(
                a0, t_alloc,
                t_init=40.0, t_min=8.0,
                eta_init=0.03, momentum_init=0.85,
                restart_every=150
            )
            # Quick MWU polish
            a_opt2, obj2 = multiplicative_weight_update(
                a_opt, min(5.0, t_alloc * 0.2), iterations=200
            )
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: LP refinement + deeper optimization on top candidates
    print("=== Phase 2: LP Refinement ===")
    phase2_deadline = deadline - 200.0
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:10]:
        key = n
        if key not in seen:
            top_list.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = phase2_deadline - time.time()
    t_per_cand = max(20.0, remaining / max(1, len(top_list) * 3))

    for n, a0 in top_list:
        if time.time() > phase2_deadline - 30:
            break

        # LP refinement
        t_lp = min(40.0, t_per_cand * 0.4)
        if t_lp > 5:
            a_lp, obj_lp = lp_refine(a0, t_lp, max_iters=30)
            if obj_lp < 2.0 * n * 1.0:
                score = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_lp.copy()
                    print(f"  LP n={n}: {score:.6f}")
                    a0 = a_lp

        # Deep Nesterov with lower temperature
        t_nest = min(50.0, t_per_cand * 0.5)
        if t_nest > 5:
            a_deep, obj_deep = nesterov_smooth_descent(
                a0, t_nest,
                t_init=15.0, t_min=3.0,
                eta_init=0.015, momentum_init=0.9,
                restart_every=100
            )
            score = float(evaluate_sequence(a_deep.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_deep.copy()
                print(f"  Deep Nesterov n={n}: {score:.6f}")

        # MWU polish
        if time.time() < phase2_deadline - 10:
            a_mw, obj_mw = multiplicative_weight_update(
                best_sequence, min(15.0, t_per_cand * 0.2), iterations=300
            )
            score = float(evaluate_sequence(a_mw.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_mw.copy()
                print(f"  MWU n={n}: {score:.6f}")

    # Phase 3: Polish
    print("=== Phase 3: Polish ===")
    phase3_deadline = deadline - 20.0

    if best_sequence is not None and time.time() < phase3_deadline:
        # Coordinate descent
        t_cd = min(60.0, (phase3_deadline - time.time()) * 0.3)
        if t_cd > 5:
            print(f"  Running CD polish (t={t_cd:.0f}s)...")
            a_cd = coordinate_polish(best_sequence, t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final Nesterov + MWU cycles
        remaining = phase3_deadline - time.time()
        for cycle in range(4):
            if time.time() > phase3_deadline - 10:
                break
            t_cyc = min(remaining / (4 - cycle), 40.0)
            if t_cyc < 8:
                break

            a_opt, _ = nesterov_smooth_descent(
                best_sequence, t_cyc * 0.6,
                t_init=10.0, t_min=2.0,
                eta_init=0.01, momentum_init=0.92,
                restart_every=80
            )
            a_opt2, _ = multiplicative_weight_update(a_opt, min(10.0, t_cyc * 0.3), iterations=200)
            if float(evaluate_sequence(a_opt2.tolist())) < float(evaluate_sequence(a_opt.tolist())):
                a_opt = a_opt2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cycle}: {score:.6f}")

            # Small perturbations
            if time.time() < phase3_deadline - 5:
                n_b = len(best_sequence)
                for ps in [0.002, 0.005]:
                    if time.time() > phase3_deadline - 5:
                        break
                    for _ in range(3):
                        if time.time() > phase3_deadline - 5:
                            break
                        noise = rng.standard_normal(n_b) * ps
                        a_p = project_simplex(best_sequence + noise)
                        a_opt, _ = nesterov_smooth_descent(
                            a_p, 5.0, t_init=8.0, t_min=3.0,
                            eta_init=0.01, momentum_init=0.9
                        )
                        score_p = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score_p) and score_p < best_score:
                            best_score = score_p
                            best_sequence = a_opt.copy()
                            print(f"  Perturb({ps}): {score_p:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(384) / 384
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
