from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a, n):
    """Build matrix C such that C @ g = a * g (full linear convolution). Shape: (2n-1, n)."""
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_direction(a, M, A_ub=None):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (elementwise), g >= 0.
    Returns (g, g_sum) or None on failure.
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g, g_sum
    return None, 0.0


def compute_obj(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max(), conv

"""
Hybrid minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding with parabolic line search and M-annealing
2. Log-sum-exp smoothed subgradient descent with momentum
3. Multi-scale search (n=256..1024) with diverse initializations
4. Adaptive time allocation and perturbative restarts

import numpy as np
import time
from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M, A_ub=None):
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        gs = g.sum()
        if gs > 1e-10:
            return g, gs
    return None, 0.0


def obj_and_conv(a, n):
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max(), conv


def line_search_parabolic(a, d, best_obj, n, t0=1.0, max_evals=15):
    """
    Parabolic line search. Start with t0, evaluate, then fit a parabola.
    Returns (best_t, best_obj).
    """
    best_t = 0.0
    cur_obj = best_obj
    evals = 0
    
    def eval_at(t):
        nonlocal evals
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            return float('inf')
        a_try = a_try / s
        o, _ = obj_and_conv(a_try, n)
        evals += 1
        return o
    
    # Initial evaluations
    t1 = t0
    f1 = eval_at(t1)
    
    if f1 < cur_obj - 1e-14:
        cur_obj = f1
        best_t = t1
    
    t2 = t1 * 0.5
    f2 = eval_at(t2)
    
    if f2 < cur_obj - 1e-14:
        cur_obj = f2
        best_t = t2
    
    # Try a few more points
    candidates = [t1 * 1.5, t1 * 2.0, t1 * 0.25]
    for tc in candidates:
        if evals >= max_evals or tc <= 0 or tc > 2.0:
            continue
        fc = eval_at(tc)
        if fc < cur_obj - 1e-14:
            cur_obj = fc
            best_t = tc
    
    # Parabolic refinement
    # Use three points to fit a parabola
    if evals < max_evals - 2 and best_t > 0:
        # Get three points around best_t
        h = best_t * 0.1
        pts = [(best_t - h, 0), (best_t, cur_obj), (best_t + h, 0)]
        need_eval = 0
        for i, (tp, fp) in enumerate(pts):
            if fp == 0 and tp > 0 and tp < 2.0:
                pts[i] = (tp, eval_at(tp))
        
        (x1, y1), (x2, y2), (x3, y3) = pts
        denom = (x1 - x2) * (x1 - x3) * (x2 - x3)
        if abs(denom) > 1e-20:
            apex = (x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2)) / (2.0 * denom)
            if 0 < apex < 2.0 and evals < max_evals:
                fa = eval_at(apex)
                if fa < cur_obj - 1e-14:
                    cur_obj = fa
                    best_t = apex
    
    return best_t, cur_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based optimization with parabolic line search and M-annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    _, conv = obj_and_conv(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 100
    perturb_count = 0
    
    # M-annealing: start with M * 1.01, tighten to M * 1.0001
    m_margin_start = 1.01
    m_margin_end = 1.0001
    
    while time.monotonic() - t0 < time_limit:
        # Anneal the margin
        progress = min(1.0, iteration / 200.0)
        m_margin = m_margin_start * (m_margin_end / m_margin_start) ** progress
        
        A_ub = build_conv_matrix(a)
        M_lp = M * m_margin
        
        result = solve_lp(a, M_lp, A_ub)
        g, g_sum = result
        
        if g is None or g_sum <= 1.0 + 1e-6:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search_parabolic(a, d, best_obj, n)
                
                if obj_best < best_obj - 1e-13:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    
                    _, conv = obj_and_conv(a, n)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and iteration % 50 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
                else:
                    no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + perturb_count * 1000 + 13)
            scale = 0.003 + 0.005 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv = obj_and_conv(a, n)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(500, 100 + 50 * perturb_count)
            if perturb_count > 50:
                break

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp_init=0.0002):
    """Subgradient descent with momentum and softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    _, conv = obj_and_conv(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    batch = 20
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find top-k peaks
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w = w / w_sum
            
            # Compute subgradient
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]
            
            # Update
            v = momentum * v - lr * sg
            a_new = np.maximum(a + v, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            _, conv = obj_and_conv(a, n)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.998
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 400:
            break
    
    return best_a, best_obj


def make_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp (decreasing)
    ramp = (n - x)
    inits['ramp'] = ramp / ramp.sum()
    
    # Ramp (increasing)
    inits['ramp_inc'] = (x + 1.0) / (x + 1.0).sum()
    
    # Triangle (peaked in middle)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri
