def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = fast_convolve(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using efficient simulated annealing
with O(n) incremental convolution updates, targeted peak-reduction moves,
and multi-length search. Key innovation: vectorized delta-conv computation
allows ~100x more evaluations than FFT-based approaches in the same time.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_delta_conv(a, i, j, d, n):
    """Compute change in convolution when a[i]-=d, a[j]+=d. O(n) vectorized."""
    conv_len = 2 * n - 1
    delta = np.zeros(conv_len, dtype=np.float64)
    # i contribution: -d*(a[k-i] + a[i]) for k in [i, i+n-1]
    ki = np.arange(i, i + n)
    delta[ki] -= d * (a[np.arange(n)] + a[i])
    if 2 * i < conv_len:
        delta[2 * i] += d * d
    # j contribution: +d*(a[k-j] + a[j]) for k in [j, j+n-1]
    kj = np.arange(j, j + n)
    delta[kj] += d * (a[np.arange(n)] + a[j])
    if 2 * j < conv_len:
        delta[2 * j] += d * d
    return delta


def get_peaks(conv, top_k=10):
    """Get indices of top_k largest entries in conv."""
    if len(conv) <= top_k:
        return np.argsort(conv)[::-1]
    idx = np.argpartition(conv, -top_k)[-top_k:]
    return idx[np.argsort(conv[idx])[::-1]]

            cur_obj = 2.0 * n * conv.max()

    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Sine-modulated
    sm = 0.5 + 0.5 * np.sin(2 * np.pi * x / max(1, n // 4))
    sm = np.maximum(sm, 0.01)
    inits['sine'] = sm / sm.sum()

    # Edge concentration
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                inits['parent_resized'] = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                inits['parent_resized'] /= inits['parent_resized'].sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: SA with efficient conv updates + greedy CD polish.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if par_score < best_score:
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Quick SA scan over lengths and initializations
    print("=== Phase 1: SA Scan ===")
    lengths = [256, 384, 512, 640, 768]

    results = []
    phase1_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.30)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        # Priority: parent first, then diverse inits
        prio = ['parent', 'parent_resized', 'random', 'sine', 'bimodal',
                'triangle', 'uniform', 'gauss', 'edge', 'ramp']
        names = [nm for nm in prio if nm in inits]
        names = names[:4]

        t_per = max(15.0, 200.0 / max(1, len(lengths) * len(names)))

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = sa_optimize(a0, t_alloc, rng, n_candidates=12, top_k=10)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:10]:
        key = n
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0
    if remaining < 60:
        remaining = 60

    t_per = max(30.0, remaining / max(1, len(top_list) * 2))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 220:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 180)
        if t_alloc < 15:
            break

        # SA with more candidates and higher top_k
        a_opt, obj_opt = sa_optimize(a0, t_alloc * 0.6, rng,
                                     n_candidates=20, top_k=20)
        # Greedy CD polish
        t_cd = min(20.0, t_alloc * 0.4)
        a_opt2, obj2 = greedy_cd(a_opt, t_cd, rng, n_trials=40)
        if obj2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj2

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")

        # Second round with different params
        if time.monotonic() < deadline - 180:
            a_opt3, obj3 = sa_optimize(a0, min(30.0, t_alloc * 0.3), rng,
                                       n_candidates=25, top_k=15)
            score3 = float(evaluate_sequence(a_opt3.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_sequence = a_opt3.copy()
                print(f"  Deep2 n={n}: {score3:.6f}")

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0

        # Greedy CD with more trials
        t_cd = min(150.0, remaining * 0.5)
        a_cd, obj_cd = greedy_cd(best_sequence, t_cd, rng, n_trials=50)
        score_cd = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score_cd) and score_cd < best_score:
            best_score = score_cd
            best_sequence = a_cd.copy()
            print(f"  CD: {score_cd:.6f}")

        # SA refinement
        remaining = deadline - time.monotonic() - 25.0
        if remaining > 20:
            a_sa, obj_sa = sa_optimize(best_sequence, min(80.0, remaining * 0.6),
                                       rng, n_candidates=25, top_k=25)
            score_sa = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score_sa) and score_sa < best_score:
                best_score = score_sa
                best_sequence = a_sa.copy()
                print(f"  SA final: {score_sa:.6f}")

        # Final greedy CD
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            a_cd2, _ = greedy_cd(best_sequence, remaining, rng, n_trials=60)
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD final: {score_cd2:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
