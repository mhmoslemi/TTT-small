import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective value (assumes sum(a) = 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_direction(a, M, top_k=None):
    """Solve LP to find direction that reduces peak.
    max sum(b) s.t. conv(a,b) <= M, b >= 0
    If top_k is given, only use top_k most important constraints.
    """
    n = len(a)
    conv_ab_matrix = None
    
    # Build constraint matrix efficiently
    num_rows = 2 * n - 1
    # We'll build the matrix column by column
    # (A @ b)[k] = sum_j a[k-j] * b[j] where a[m]=0 for m outside [0,n-1]
    
    if top_k is not None:
        # Only use top_k constraints
        conv_aa = fast_convolve(a)
        # The constraints most likely to be active are those where conv_aa is largest
        top_indices = np.argpartition(conv_aa, -top_k)[-top_k:]
        A = np.zeros((top_k, n))
        for idx, k in enumerate(top_indices):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[idx, j] = a[k - j]
        b_ub = np.full(top_k, M)
    else:
        A = np.zeros((num_rows, n))
        for k in range(num_rows):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[k, j] = a[k - j]
        b_ub = np.full(num_rows, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid LP + subgradient + BFGS approach.
Key improvements over previous: adaptive line search for LP steps, reduced-constraint LPs,
log-sum-exp BFGS polishing, and multi-direction exploration.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_rows(a, rows):
    """Build matrix rows where (A[idx] @ b) = conv(a, b)[rows[idx]]."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[idx, j] = a[k - j]
    return A


def lp_max_sum(a, M, top_k=None):
    """LP: max sum(b) s.t. conv(a,b)[k] <= M for selected k, b >= 0."""
    n = len(a)
    if top_k is not None and top_k < 2 * n - 1:
        conv_aa = fast_convolve(a)
        top_idx = np.argpartition(conv_aa, -top_k)[-top_k:]
        rows = top_idx
    else:
        rows = np.arange(2 * n - 1)
    A = build_conv_matrix_rows(a, rows)
    c = -np.ones(n)
    b_ub = np.full(len(rows), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            return b / b_sum
    return None


def lp_min_peak_full(a):
    """LP: min M' s.t. conv(a,b)[k] <= M' all k, sum(b)=1, b>=0."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix_rows(a, np.arange(num_rows))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            return b / b_sum, res.x[n]
    return None, None


def subgrad_peak(a, conv, top_k=50, temperature=0.001):
    """Weighted subgradient over top-k peaks of convolution."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def logsumexp_smoothed_obj(a, n, t):
    """Smoothed objective using log-sum-exp and its gradient."""
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = (2.0 * n / t) * np.log(sum_exp)
    p = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] = (2.0 * n / t) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return obj, grad

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    step = step_init
    no_improve = 0

            b = lp_max_sum(a, M, top_k=top_k)

        if b is None:
            break
        if np.linalg.norm(b - a) < 1e-12:
            break

        # Adaptive step: try to find good step via a few trials

        candidate_steps = [step, step * 0.5, step * 1.5, step * 0.25]

                continue

            a_try = np.maximum(a_try, 0)

        step = best_step

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

            step = step_init
            no_improve = 0

    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        sg = subgrad_peak(a, conv, top_k, temp)

def generate_diverse_init(n, rng, parent=None):
    """Generate a set of diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 4]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos_{freq}'] = a / a_sum

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Two-peak
    p1 = n // 4
    p2 = 3 * n // 4
    a = np.zeros(n)
    a[max(0, p1 - 5):p1 + 6] = 1.0
    a[max(0, p2 - 5):p2 + 6] = 1.0
    a /= a.sum()
    inits['two_peak'] = a

    return inits


def coordinate_descent(a, time_limit, max_rounds=100):
    """Simple coordinate descent: adjust one element at a time."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        for _ in range(n):
            i = rng.integers(0, n)
            # Try small adjustments
            for delta in [0.001, -0.001, 0.005, -0.005]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + delta)
                a_try = project_simplex(a_try)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = conv_try
                    break
    return best_a, best_obj

        if time.monotonic() < phase3_deadline - 20:
            t_alloc2 = min(40.0, phase3_deadline - time.monotonic())
            if t_alloc2 > 5:
                a_opt2, _ = lp_refine(a_opt, time_limit=t_alloc2,
                                      method='min_peak', top_k=100, step_init=0.2)
                try_update(a_opt2)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: BFGS smoothing polish (next 250s)
    print("=== Phase 4: BFGS Polish ===")
    t4_start = time.monotonic()
    phase4_deadline = time.monotonic() + 230

    for t_init in [30.0, 10.0, 50.0]:
        if time.monotonic() > phase4_deadline:
            break
        t_alloc = min(70.0, phase4_deadline - time.monotonic())
        if t_alloc < 5:
            break
        a_opt, _ = bfgs_smoothed(best_sequence, time_limit=t_alloc, t_init=t_init)
        try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Final perturbations + LP (remaining time)
    print("=== Phase 5: Final Perturbations ===")
    t5_start = time.monotonic()

    n_best = len(best_sequence)
    scales = [0.001, 0.003, 0.0005, 0.005, 0.002]
    for scale in scales:
        remain = deadline - time.monotonic()
        if remain < 10:
            break
        noise = rng.standard_normal(n_best) * scale
        a_pert = project_simplex(best_sequence + noise)
        t_alloc = min(30.0, remain - 5)
        if t_alloc < 5:
            break
        a_opt, _ = lp_refine(a_pert, time_limit=t_alloc, method='max_sum',
                             top_k=100, step_init=0.2)
        try_update(a_opt)

    # One more try: min_peak from best
    remain = deadline - time.monotonic()
    if remain > 15:
        a_opt, _ = lp_refine(best_sequence, time_limit=min(40.0, remain - 5),
                             method='min_peak', top_k=None, step_init=0.1)
        try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final coordinate descent polish
    remain = deadline - time.monotonic()
    if remain > 10:
        print("=== Coordinate Descent Polish ===")
        a_opt, _ = coordinate_descent(best_sequence, time_limit=min(60.0, remain - 3))
        try_update(a_opt)

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
