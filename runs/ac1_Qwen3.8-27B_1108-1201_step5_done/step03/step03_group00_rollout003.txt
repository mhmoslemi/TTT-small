2. Compute `A = FFT(a)`
3. Compute `P = |A|^2` (power spectrum)
4. Compute `c = IFFT(P)` (this is the auto-convolution, up to normalization)
5. For each lag `k` where `c[k] > M` (peak exceeds target), modify `P` to reduce it
   - Specifically, `P_new[k] = P[k] * exp(-alpha * c[k])` for the "excess" part

6. Compute `a_new = IFFT(sqrt(P_new))` (taking the square root because P = |A|^2)
7. Enforce constraints: `a_new = max(a_new, 0)`, normalize
8. Repeat

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Expand symmetric half to full sequence."""
    full = np.empty(n)
    full[:len(half)] = half
    full[len(half):] = half[:n - len(half)][::-1]
    return full

    sym_full = 0.5 * (full + full[::-1])
