def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9):
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    N = 1 << (2 * n - 2).bit_length()
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A.conjugate(), N)[:2*n-1]
        
        cur_obj = 2.0 * n * c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active peaks (top K)
        top_k = min(20, 2*n-1)
        peak_indices = np.argpartition(c, -top_k)[-top_k:]
        peak_values = c[peak_indices]
        
        # Softmax weights for peaks
        temp = max(peak_values.max() * 0.01, 1e-15)
        w = np.exp((peak_values - peak_values.max()) / temp)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-12:
                continue
            k = int(peak_indices[i])
            # sg[j] += w[i] * a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = np.maximum(a + v, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Adaptive learning rate
        lr *= 0.999
        if no_improve > 100:
            lr = max(lr * 0.5, 1e-8)
            no_improve = 0
        
        it += 1
    
    return best_a, best_obj

def fast_convolve(a):

    N = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A.conjugate(), N)[:2 * n - 1]

    """Compute 2*n*max(conv(a,a)) with sum(a)=1 assumed."""

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A.conjugate(), N)[:2 * n - 1]
    return 2.0 * n * c.max(), c


def compute_subgradient(a, c, N, top_k=15, temperature=0.05):
    """Compute weighted subgradient of max(c) using top peaks."""
    n = len(a)
    c_len = len(c)
    k = min(top_k, c_len)
    peak_idx = np.argpartition(c, -k)[-k:]
    peak_vals = c[peak_idx]
    
    temp = max(peak_vals.max() * temperature, 1e-18)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w /= w.sum()

        if j_lo <= j_hi:

            sg[idx] += w[i] * a[kth - idx]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         top_k=15, temperature_init=0.05, verbose=False):
    """Fast subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    N = 1 << (2 * n - 2).bit_length()
    
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    _, c = local_objective(a, N)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    temperature = temperature_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        sg = compute_subgradient(a, c, N, top_k=top_k, temperature=temperature)
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        
        # Project to simplex
        a_candidate = project_simplex(a_new)
        
        # Evaluate
        obj_new, c_new = local_objective(a_candidate, N)
        
        # Backtracking: revert and reduce step if no improvement
        if obj_new < best_obj - 1e-14:
            a = a_candidate
            c = c_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        elif obj_new < best_obj * 1.002:
            # Accept small increases (momentum needs headroom)
            a = a_candidate
            c = c_new
            no_improve += 1
        else:
            # Revert
            no_improve += 1
            lr *= 0.8
            v *= 0.5
        
        # Adaptive adjustments
        if no_improve > 50:
            lr = max(lr * 0.7, 1e-10)
            temperature = max(temperature * 0.9, 0.001)
            no_improve = 0
            best_no_improve = min(best_no_improve, no_improve)
        
        if no_improve > 200:
            # Structured perturbation
            rng = np.random.default_rng(it)
            # Perturb based on current autocorrelation structure
            noise = fast_convolve(a) - c.mean()
            noise = sign_pattern(a, noise, rng)
            a = np.maximum(a + noise * 0.002, 0.0)
            a = a / a.sum()
            _, c = local_objective(a, N)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
        
        lr = min(lr * 1.002, lr_init)  # Slowly recover lr
        it += 1
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG iter={it}, obj={best_obj:.8f}, lr={lr:.2e}, "
                  f"temp={temperature:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def sign_pattern(a, signal, rng):
    """Create a sign pattern perturbation that's structured."""
    n = len(a)
    # Use the signal to guide perturbation direction
    pattern = np.tanh(signal / (np.std(signal) + 1e-15))
    # Add small random component
    pattern += rng.standard_normal(n) * 0.1
    return pattern


def freq_domain_init(n, rng, mode='sinc'):
    """Generate initial sequences using frequency domain construction."""
    N = 1 << (2 * n - 2).bit_length()
    
    if mode == 'flat_phase':
        # Flat magnitude, random phase -> gives a "noise-like" sequence
        A = np.ones(N // 2 + 1, dtype=complex)
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        A *= np.exp(1j * phases)
        # Make conjugate symmetric for real result
        A_full = np.zeros(N, dtype=complex)
        A_full[:N//2+1] = A
        A_full[N//2+1:] = np.conjugate(A[1:N//2][::-1])
        a = np.fft.ifft(A_full).real[:n]
        
    elif mode == 'tapered':
        # Tapered magnitude (raised cosine) with linear phase
        frequencies = np.arange(N // 2 + 1)
        f_norm = frequencies / (N / 2.0)
        mag = 0.5 * (1 + np.cos(np.pi * f_norm))  # raised cosine
        mag[0] = 1.0
        phases = -np.pi * frequencies * (n / 2.0) / (N / 2.0)  # centered
        A = mag * np.exp(1j * phases)
        A_full = np.zeros(N, dtype=complex)
        A_full[:N//2+1] = A
        A_full[N//2+1:] = np.conjugate(A[1:N//2][::-1])
        a = np.fft.ifft(A_full).real[:n]
        
    elif mode == 'two_peak':
        # Two frequency peaks -> gives oscillatory pattern
        center1 = int(N * 0.2)
        center2 = int(N * 0.8)
        width = max(1, N // 32)
        mag = np.zeros(N // 2 + 1)
        for center in [center1, min(center2, N//2)]:
            for i in range(-width, width + 1):
                idx = center + i
                if 0 <= idx <= N // 2:
                    mag[idx] += np.exp(-0.5 * (i / (width * 0.5)) ** 2)
        phases = np.zeros(N // 2 + 1)
        A = mag * np.exp(1j * phases)
        A_full = np.zeros(N, dtype=complex)
        A_full[:N//2+1] = A
        A_full[N//2+1:] = np.conjugate(A[1:N//2][::-1])
        a = np.fft.ifft(A_full).real[:n]
        
    elif mode == 'barker_like':
        # Try to create a sequence with low autocorrelation sidelobes
        # Start with a Barker-like pattern, extend
        a = np.zeros(n)
        # Use a Thue-Morse sequence as base pattern
        base = thue_morse(min(64, n))
        # Interpolate to length n
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(base)), base)
        
    elif mode == 'uniform':
        a = np.ones(n) / n
        return a
        
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a
    
    # Clean up: clip negatives, normalize
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def thue_morse(n):
    """Generate Thue-Morse sequence (0/1 pattern with low autocorrelation)."""
    t = [0]
    for _ in range(10):
        t = t + [1 - x for x in t]
        if len(t) >= n:
            break
    return np.array(t[:n], dtype=float)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramp (asymmetric - important for reducing center peak)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    inits['ramp_rev'] = r[::-1] / r.sum()
    
    # Exponential decay (asymmetric)
    x = np.arange(n, dtype=np.float64)
    decay = np.exp(-0.5 * x / (n * 0.3))
    inits['decay'] = decay / decay.sum()
    inits['decay_rev'] = decay[::-1] / decay.sum()
    
    # Two-hump (W-shape)
    h1 = np.exp(-0.5 * ((x - n * 0.1) / (n * 0.1)) ** 2)
    h2 = np.exp(-0.5 * ((x - n * 0.9) / (n * 0.1)) ** 2)
    two = h1 + h2 + 1e-10
    inits['two_hump'] = two / two.sum()
    
    # Three-hump
    h3 = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.15)) ** 2)
    three = h1 + h3 + h2 + 1e-10
    inits['three_hump'] = three / three.sum()
    
    # Frequency domain inits
    for mode in ['flat_phase', 'tapered', 'two_peak', 'barker_like']:
        inits[f'freq_{mode}'] = freq_domain_init(n, rng, mode)
    
    # Parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                interp = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(parent)), pp)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['parent_interp'] = interp / is_
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    # === Phase 1: Quick scan across lengths and inits ===
    print("=== Phase 1: Quick Scan ===")
    lengths_to_try = []
    
    if parent is not None:
        lengths_to_try.append(len(parent))
    
    # Add other lengths
    for n in [384, 512, 640, 768]:
        if n not in lengths_to_try:
            lengths_to_try.append(n)
    
    # Sort: prioritize parent length first
    lengths_to_try = sorted(set(lengths_to_try))
    
    time_per_scan = max(5.0, (budget_s * 0.15) / len(lengths_to_try))
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Prioritize: parent > freq inits > structural > random
        priority = ['parent', 'parent_interp', 'freq_tapered', 'freq_barker_like',
                    'decay', 'ramp', 'two_hump', 'freq_flat_phase', 'freq_two_peak',
                    'three_hump', 'decay_rev', 'ramp_rev', 'uniform']
        names = list(inits.keys())
        names.sort(key=lambda x: priority.index(x) if x in priority else 99)
        
        for name in names[:5]:
            if time.monotonic() > deadline - 300:
                break
            t_alloc = min(time_per_scan, deadline - time.monotonic() - 300)
            if t_alloc < 2:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = subgradient_optimize(a0, time_limit=t_alloc,
                                                  lr_init=0.003, momentum=0.92,
                                                  top_k=12, temperature_init=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, init={name}: score={score:.6f} (NEW BEST)")
                else:
                    print(f"  n={n}, init={name}: score={score:.6f}")
    
    print(f"After Phase 1: best={best_score:.6f}")
    
    # === Phase 2: Deep optimization on best ===
    if best_sequence is not None:
        print("=== Phase 2: Deep Optimization ===")
        remaining = deadline - time.monotonic() - 120.0
        t_deep = remaining * 0.5
        if t_deep > 30:
            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_deep,
                                                  lr_init=0.002, momentum=0.93,
                                                  top_k=20, temperature_init=0.02,
                                                  verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 3: Try different momentum/lr settings ===
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("=== Phase 3: Hyperparameter Variants ===")
        remaining = deadline - time.monotonic() - 60.0
        configs = [
            (0.001, 0.95, 25, 0.01),   # low lr, high momentum, many peaks
            (0.005, 0.85, 10, 0.05),   # higher lr, low momentum, few peaks
            (0.003, 0.90, 30, 0.03),   # moderate
        ]
        t_per_cfg = remaining / len(configs)
        for lr, mom, tk, temp in configs:
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(t_per_cfg, deadline - time.monotonic() - 60)
            if t_alloc < 5:
                break
            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                  lr_init=lr, momentum=mom,
                                                  top_k=tk, temperature_init=temp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  cfg=({lr},{mom},{tk},{temp}): score={score:.6f} (NEW BEST)")
    
    # === Phase 4: Perturbation and re-optimization ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 4: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 10)
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Use autocorrelation-guided perturbation
            c = fast_convolve(best_sequence)
            c_norm = (c - c.mean()) / (c.std() + 1e-15)
            # Create perturbation that reduces peaks
            pert = -c_norm[:n_best] if len(c_norm) >= n_best else \
                   np.interp(np.linspace(0, 1, n_best), np.linspace(0, 1, len(c_norm)), c_norm)
            scale = 0.001 * (1 + i * 0.3)
            a_pert = np.maximum(best_sequence + scale * pert + rng.standard_normal(n_best) * scale * 0.5, 0.0)
            a_pert = a_pert / a_pert.sum()
            
            t_alloc = min(18.0, deadline - time.monotonic() - 25)
            if t_alloc < 3:
                break
            a_opt, obj_opt = subgradient_optimize(a_pert, time_limit=t_alloc,
                                                  lr_init=0.002, momentum=0.92,
                                                  top_k=15, temperature_init=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (NEW BEST!)")
    
    # === Phase 5: Final fine-tuning ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 5: Final Fine-tuning ===")
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5:
            # Very small steps for polishing
            n_final = len(best_sequence)
            a = best_sequence.copy()
            N = 1 << (2 * n_final - 2).bit_length()
            t0 = time.monotonic()
            lr = 0.0005
            v = np.zeros(n_final)
            
            while time.monotonic() - t0 < remaining:
                _, c = local_objective(a, N)
                sg = compute_subgradient(a, c, N, top_k=8, temperature=0.01)
                v = 0.95 * v - lr * sg
                a_new = project_simplex(a + v)
                obj_new, _ = local_objective(a_new, N)
                obj_cur, _ = local_objective(a, N)
                if obj_new < obj_cur:
                    a = a_new
                else:
                    lr *= 0.9
                    v *= 0.5
                if time.monotonic() - t0 > remaining:
                    break
            
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a.copy()
                print(f"  Phase 5: score={score:.6f} (NEW BEST!)")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
