def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

    """Compute soft max and weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w

def grad_smooth_conv(a, w):
    """Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    grad = np.real(np.fft.ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - 1.0) / np.arange(1, n + 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fista_optimize(a0, beta_start, beta_end, n_iter, lr_init, rho=0.9):

import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def smooth_max_weights(conv, beta):
    cmax = conv.max()
    shifted = np.clip(beta * (conv - cmax), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    w = exp_s / total
    sm = cmax + np.log(total) / beta
    return sm, w


def compute_gradient(a, w):
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    return 2.0 * np.real(ifft(np.conj(A) * W, N))[:n]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_smooth(a0, beta_start, beta_end, n_iter, lr, momentum=True):
    n = len(a0)
    x = project_simplex(a0.copy())
    z = x.copy()
    t = 1.0
    best_obj = 2.0 * n * fast_convolve(x).max()
    best_x = x.copy()
    
    for it in range(n_iter):
        progress = it / max(1, n_iter - 1)
        if beta_end > 0 and beta_start > 0:
            beta = beta_start * (beta_end / beta_start) ** progress
        else:
            beta = beta_start
        
        conv = fast_convolve(z)
        sm, w = smooth_max_weights(conv, beta)
        grad = compute_gradient(z, w)
        
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        
        if momentum:
            y = z - (lr / t) * grad
        else:
            y = z - lr * grad
        
        x_new = project_simplex(y)
        
        if momentum:
            z = x_new + ((t_old - 1.0) / t) * (x_new - x)
        else:
            z = x_new
        x = x_new
        
        # Periodically check actual objective
        if it % 50 == 0 or it == n_iter - 1:
            actual = 2.0 * n * fast_convolve(x).max()
            if actual < best_obj:
                best_obj = actual
                best_x = x.copy()

    actual = 2.0 * n * fast_convolve(x).max()
    if actual < best_obj:
        best_obj = actual
        best_x = x.copy()
    
    return best_x, best_obj

- x_{k+1} = argmin (1/(2L)||x - y_k||^2 + f(x))  (proximal step)
- z_{k+1} = x_{k+1} + ((t_k - 1)/t_{k+1})(x_{k+1} - x_k)
- y_k = z_k  (or y_k = z_k - ... for the lookahead variant)

For our problem, the proximal step is:
argmin_x (1/(2L)||x - y||^2 + f_smooth(x)) subject to x in simplex

Since f_smooth is not separable, we can't use a separable prox. Instead, we approximate the proximal step with a gradient step + projection:
x_{k+1} = P_simplex(y - (1/L) * grad f_smooth(z_k))

This is just projected gradient with momentum, which is what I had.

The issue is that the "Lipschitz constant" L of the smooth objective depends on beta and the current iterate. Let me just use a fixed conservative learning rate.

For the smooth max objective:
f_beta(a) = 2n * (1/beta) * log(sum_k exp(beta * conv_k))

The Hessian of conv_k is 2*P_k (a symmetric permutation matrix), so the Hessian of f_beta is:
2n * [sum_k w_k * 2*P_k + beta * (sum_k w_k * grad_k)(sum_k w_k * grad_k)^T]
= 2n * [2 * sum_k w_k * P_k + beta * outer(sum_k w_k * grad_k, sum_k w_k * grad_k)]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with adaptive smooth max,
diverse asymmetric initializations, multi-length search, multi-stage beta annealing,
and coordinate descent refinement.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def smooth_max_weights(conv, beta):
    """Compute soft max value and softmax weights."""
    cmax = conv.max()
    shifted = np.clip(beta * (conv - cmax), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    w = exp_s / total
    sm = cmax + np.log(total) / beta
    return sm, w


def compute_gradient(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a via FFT cross-correlation."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    return 2.0 * np.real(ifft(np.conj(A) * W, N))[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, beta_schedule, lr_init=1.0, use_momentum=True):
    """
    FISTA (accelerated projected gradient) with adaptive smooth max.
    beta_schedule: list of (n_sub_iter, beta) tuples.
    """
    n = len(a0)
    x = project_simplex(a0.astype(np.float64).copy())
    z = x.copy()
    t = 1.0
    best_obj = 2.0 * n * fast_convolve(x).max()
    best_x = x.copy()
    lr = lr_init

    total_iters = sum(s for s, _ in beta_schedule)
    done = 0

    for n_sub, beta in beta_schedule:
        for it in range(n_sub):
            done += 1
            conv = fast_convolve(z)
            sm, w = smooth_max_weights(conv, beta)
            grad = 2.0 * n * compute_gradient(z, w)

            t_old = t
            t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

            if use_momentum:
                y = z - (lr / t) * grad
            else:
                y = z - lr * grad

            x_new = project_simplex(y)

            if use_momentum:
                z = x_new + ((t_old - 1.0) / t) * (x_new - x)
            else:
                z = x_new
            x = x_new

            # Decay learning rate slowly
            lr *= 0.99995

            if done % 200 == 0 or done == total_iters:
                actual = 2.0 * n * fast_convolve(x).max()
                if actual < best_obj:
                    best_obj = actual
                    best_x = x.copy()

    actual = 2.0 * n * fast_convolve(x).max()
    if actual < best_obj:
        best_obj = actual
        best_x = x.copy()

    return best_x, best_obj

def coord_descent_refine(a, n_sweeps=20, n_trials=5):
    """
    Coordinate descent on the exact (non-smooth) objective.
    For each coordinate, try several scale factors and keep the best.
    """
    n = len(a)
    x = project_simplex(a.astype(np.float64).copy())
    best_obj = 2.0 * n * fast_convolve(x).max()

    scales = np.array([0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.3])

    for sweep in range(n_sweeps):
        improved = False
        order = np.random.default_rng(sweep * 1000 + 42).permutation(n)
        for i in order:
            old = x[i]
            if old < 1e-15:
                # Try adding a small amount
                for s in [0.001, 0.005, 0.01]:
                    x[i] = s
                    x = project_simplex(x)
                    obj = 2.0 * n * fast_convolve(x).max()
                    if obj < best_obj:
                        best_obj = obj
                        improved = True
                        break
                    x[i] = old
            else:
                for s in scales:
                    x[i] = old * s
                    x_temp = project_simplex(x.copy())
                    obj = 2.0 * n * fast_convolve(x_temp).max()
                    if obj < best_obj:
                        best_obj = obj
                        x = x_temp
                        improved = True
                        break
                else:
                    x[i] = old

        if not improved:
            break

    return x, best_obj


def init_flat_spectrum(n, rng):
    """Initialize with flat 2n-point power spectrum."""
    N = 1 << (2 * n - 1).bit_length()
    A = np.zeros(N, dtype=complex)
    A[0] = n  # DC component to get sum ~ n
    for k in range(1, N // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = np.exp(1j * phase)
        if k < N - k:
            A[N - k] = np.exp(-1j * phase)
    a = np.real(ifft(A, N))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def init_optimal_phase(n, rng, n_trials=30):
    """Try multiple phase assignments to find best flat-spectrum init."""
    N = 1 << (2 * n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(N, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, N // 2)
        for k in range(1, N // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[N - k] = np.exp(-1j * phases[k - 1])
        if N % 2 == 0:
            A[N // 2] = np.exp(1j * phases[-1])
        a = np.real(ifft(A, N))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def init_one_sided(n, rng, frac=0.6):
    """Put mass in the first `frac` fraction of the sequence."""
    a = np.zeros(n)
    k = max(1, int(n * frac))
    a[:k] = rng.dirichlet(np.ones(k) * 0.5)
    return a


def init_ramp(n, direction=1):
    """Ramp initialization."""
    a = np.arange(1, n + 1, dtype=np.float64)
    if direction == -1:
        a = a[::-1]
    return a / a.sum()


def init_offset_gaussian(n, center_frac, width_frac):
    """Gaussian bump off-center."""
    x = np.arange(n, dtype=np.float64)
    c = center_frac * (n - 1)
    w = width_frac * n
    a = np.exp(-0.5 * ((x - c) / w) ** 2)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def init_bimodal_edge(n, frac):
    """Mass at both edges."""
    a = np.zeros(n)
    k = max(1, int(n * frac))
    a[:k] = 1.0
    a[-k:] = 1.0
    return a / a.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    inits['ramp_r'] = init_ramp(n, 1)
    inits['ramp_l'] = init_ramp(n, -1)
    inits['one_sided_r'] = init_one_sided(n, rng, 0.6)
    inits['one_sided_l'] = init_one_sided(n, rng, 0.6)[::-1].copy()
    inits['gauss_left'] = init_offset_gaussian(n, 0.25, 0.15)
    inits['gauss_right'] = init_offset_gaussian(n, 0.75, 0.15)
    inits['bimodal'] = init_bimodal_edge(n, 0.25)
    inits['flat_spec'] = init_flat_spectrum(n, rng)
    inits['opt_phase'] = init_optimal_phase(n, rng, 20)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent'] = p / s
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            s = p.sum()
            if s > 1e-10:
                inits['parent_res'] = p / s

    return inits


def multi_stage_fista(a0, total_iter, n_stages=6):
    """FISTA with geometric beta annealing from high to low."""
    betas = np.geomspace(200.0, 2.0, n_stages)
    schedule = []
    per_stage = total_iter // n_stages
    for i, b in enumerate(betas):
        n_sub = per_stage if i < n_stages - 1 else total_iter - per_stage * (n_stages - 1)
        schedule.append((n_sub, b))
    return fista_optimize(a0, schedule, lr_init=1.0, use_momentum=True)

            if t_alloc < 10:
                break

            a0 = inits[name]
            n_iters = int(min(5000, max(500, t_alloc * 50)))

            # Multi-stage FISTA with careful beta annealing
            a_opt, _ = multi_stage_fista(a0, n_iters, n_stages=8)

            # Follow up with lower beta
            remaining = phase2_deadline - time.monotonic() - 10
            if remaining > 20:
                additional = int(min(2000, remaining * 30))
                schedule = [(additional, 8.0), (0, 4.0)]
                a_opt2, _ = fista_optimize(a_opt, [(additional, 5.0)], lr_init=0.5, use_momentum=True)
                score2 = score_and_update(a_opt2)
                if np.isfinite(score2):
                    a_opt = a_opt2

            score = score_and_update(a_opt)
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))

    print("Phase 2 done. Best: %.6f" % best_score)

    # ============ PHASE 3: Refinement ============
    print("=== Phase 3: Refinement ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        a_best = best_sequence / best_sequence.sum()
        remaining = deadline - time.monotonic() - 30.0

        # 3a: Low-beta FISTA polish
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.3)
            n_iters = int(min(3000, t_alloc * 40))
            a_opt, _ = fista_optimize(a_best, [(n_iters // 2, 5.0), (n_iters - n_iters // 2, 2.0)],
                                      lr_init=0.3, use_momentum=True)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30.0

        # 3b: Coordinate descent refinement
        if remaining > 20:
            t_alloc = min(120.0, remaining * 0.3)
            n_sweeps = int(min(100, t_alloc * 2))
            a_coord, _ = coord_descent_refine(a_best, n_sweeps=n_sweeps, n_trials=5)
            score_and_update(a_coord)

            # Re-optimize from coordinate descent result
            if remaining > 20 and best_sequence is not None:
                a_reopt, _ = fista_optimize(a_coord, [(500, 3.0)], lr_init=0.3, use_momentum=True)
                score_and_update(a_reopt)

            remaining = deadline - time.monotonic() - 30.0

        # 3c: Perturbation + re-optimization
        if remaining > 30:
            n_trials = int(min(10, remaining / 20))
            for trial in range(n_trials):
                if remaining < 15:
                    break
                scale = 0.005 + 0.005 * rng.random()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_best + noise)
                n_iters = int(min(1000, max(200, remaining * 0.4)))
                a_opt, _ = multi_stage_fista(a_pert, n_iters, n_stages=4)
                score_and_update(a_opt)
                if best_sequence is not None:
                    a_best = best_sequence / best_sequence.sum()
                remaining = deadline - time.monotonic() - 30.0

        # 3d: Final coordinate descent
        if remaining > 10 and best_sequence is not None:
            a_final, _ = coord_descent_refine(best_sequence / best_sequence.sum(),
                                             n_sweeps=50, n_trials=5)
            score_and_update(a_final)

        # 3e: Ultra-low beta FISTA on the very best
        if remaining > 10 and best_sequence is not None:
            n_iters = int(min(2000, remaining * 30))
            a_ultra, _ = fista_optimize(best_sequence / best_sequence.sum(),
                                       [(n_iters, 1.5)], lr_init=0.2, use_momentum=False)
            score_and_update(a_ultra)

    # Try a different length if time permits
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 60:
            alt_lengths = [l for l in [128, 192, 256, 320, 384, 512] if l != len(best_sequence)]
            for alt_n in alt_lengths[:2]:
                if remaining < 40:
                    break
                inits_alt = generate_inits(alt_n, parent, rng)
                p = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(best_sequence)),
                              np.maximum(best_sequence, 0))
                s = p.sum()
                if s > 1e-10:
                    inits_alt['from_best'] = p / s
                a0 = inits_alt.get('from_best', inits_alt.get('opt_phase', inits_alt['uniform']))
                n_iters = int(min(3000, remaining * 30))
                a_opt, _ = multi_stage_fista(a0, n_iters, n_stages=6)
                score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 20.0

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
