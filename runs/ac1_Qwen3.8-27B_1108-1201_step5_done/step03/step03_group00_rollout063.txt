The key question is: how to efficiently find x*?

Method 1: Grid search (simple, vectorized)
Method 2: Binary search on the subgradient (the subgradient of max_k (C_k + x*d_k) at x is d_{k*(x)} where k*(x) is the argmax)
  - The subgradient is non-decreasing in x (since the function is convex)
  - At the optimum, 0 is in the subgradient, i.e., d_{k*(x*)} ≤ 0 and d_{k***(x*)} ≥ 0 for some argmax indices
  - So we can binary search for the x where the subgradient crosses 0

Method 2 is more efficient. Let me work it out:
- At x, the subgradient is d[k*(x)] where k*(x) = argmax_k (C_k + x*d_k)
- If d[k*(x)] < 0: the function is decreasing at x, so x* > x
- If d[k*(x)] > 0: the function is increasing at x, so x* < x
- If d[k*(x)] = 0: x* = x (or x is in a flat region)

But since the function is piecewise linear, the subgradient jumps at breakpoints. The minimum occurs where the subgradient jumps from negative to positive.

Binary search:
- lo = 0, hi = x_max
- For 50 iterations:

  - If d[k*] < 0: lo = mid
  - Elif d[k*] > 0: hi = mid
  - Else: lo = hi = mid (converged)
- x* = (lo + hi) / 2

            # But if j == k-j (i.e., k=2j), these are the same term, so we count it once.
            
            # Vectorized computation:
            # For k in [0, 2n-2], d_k = f[k-j] if 0<=k-j<n, plus f[k-j] again if k-j != j
            # = 2*f[k-j] if 0<=k-j<n and k-j != j

            k_range = np.arange(2 * n - 1)

            valid = (kj >= 0) & (kj < n)
            d = np.zeros(2 * n - 1)
            d[valid] = 2.0 * f[kj[valid]]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with 
incremental convolution updates, combined with subgradient polishing and 
LP-based global steps for escaping local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def coord_desc(f0, time_limit, rng_seed=42):
    """Coordinate descent for minimizing max(conv(f,f)) given sum(f)=1."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        
        for j in order:
            k_range = np.arange(2 * n - 1)
            kj = k_range - j
            valid = (kj >= 0) & (kj < n)
            d = np.zeros(2 * n - 1)
            d[valid] = 2.0 * f[kj[valid]]
            if 2 * j < 2 * n - 1:
                d[2 * j] = f[j]
            
            C = conv - f[j] * d
            
            # Check boundary x=0
            k_star_0 = np.argmax(C)
            if d[k_star_0] >= -1e-14:
                x_star = 0.0
            else:
                lo = 0.0
                hi = 50.0 / n
                for _ in range(50):
                    mid = (lo + hi) * 0.5
                    vals = C + mid * d
                    k_star = np.argmax(vals)
                    if d[k_star] < 0:
                        lo = mid
                    elif d[k_star] > 0:
                        hi = mid
                    else:
                        break
                x_star = (lo + hi) * 0.5
            
            delta = x_star - f[j]
            if abs(delta) > 1e-16:
                f[j] = x_star
                conv += delta * d
        
        # Normalize
        s = f.sum()
        if s > 1e-15:
            f /= s
            conv = fast_convolve(f)
        
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj, sweep


def sg_polish(f0, time_limit, rng_seed=99):
    """Subgradient polishing with softmax peak weighting."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)
    
    def project_simplex(v):
        n_local = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n_local + 1) > (cssv - 1)
        if not cond.any():
            return np.ones(n_local) / n_local
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    
    v_mom = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = 0.005
    mom = 0.93
    top_k = max(10, n // 6)
    temp = 0.0001
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Get top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v_mom = mom * v_mom - lr * sg
        f_new = project_simplex(f + v_mom)
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            # Don't update f, just keep going
        
        lr *= 0.9997
        temp *= 1.005
        
        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v_mom = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0
    
    return best_f, best_obj


def lp_direction(f, M, top_k_constraints=50):
    """Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for top-k k's, g >= 0."""
    n = len(f)
    conv_f = fast_convolve(f)
    # Get top-k constraint indices
    if top_k_constraints < len(conv_f):
        active = np.argpartition(conv_f, -top_k_constraints)[-top_k_constraints:]
    else:
        active = np.arange(len(conv_f))
    
    # Build LP: max sum(g) s.t. (f*g)[k] <= M for k in active, g >= 0
    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = np.zeros((len(active), n))
    for row, k in enumerate(active):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[row, j] += f[idx]
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def solve_lp_full(f, M, time_per_lp=30):
    """Solve full LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_per_lp})
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with 
incremental convolution updates, alternating with subgradient polishing 
and LP-based global steps. Coordinate descent provides fast, guaranteed 
descent per coordinate; subgradient helps escape local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_desc(f0, time_limit, rng_seed=42):
    """
    Coordinate descent minimizing max(conv(f,f)) subject to f>=0, sum(f)=1.
    Uses incremental convolution updates and binary search for each coordinate.
    """
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    sweep = 0
    no_improve = 0
    k_range = np.arange(2 * n - 1)  # Precompute
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        
        for j in order:
            kj = k_range - j
            valid = (kj >= 0) & (kj < n)
            d = np.zeros(2 * n - 1)
            d[valid] = 2.0 * f[kj[valid]]
            if 2 * j < 2 * n - 1:
                d[2 * j] = f[j]
            
            C = conv - f[j] * d
            
            # Binary search for optimal x
            k_star_0 = int(np.argmax(C))
            if d[k_star_0] >= -1e-14:
                x_star = 0.0
            else:
                lo = 0.0
                hi = 50.0 / n
                for _ in range(35):
                    mid = (lo + hi) * 0.5
                    vals = C + mid * d
                    k_star = int(np.argmax(vals))
                    dk = d[k_star]
                    if dk < 0:
                        lo = mid
                    elif dk > 0:
                        hi = mid
                    else:
                        break
                x_star = (lo + hi) * 0.5
            
            delta = x_star - f[j]
            if abs(delta) > 1e-16:
                f[j] = x_star
                conv += delta * d
        
        # Normalize
        s = f.sum()
        if s > 1e-15:
            f /= s
            conv = fast_convolve(f)  # Recompute to avoid drift
        
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 150:
            f = f.copy()
            f += rng.standard_normal(n) * 0.002
            f = project_simplex(f)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj, sweep


def sg_polish(f0, time_limit, rng_seed=99, lr_init=0.004, mom_init=0.93):
    """Subgradient method with softmax peak weighting and momentum."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)
    
    v_mom = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = lr_init
    mom = mom_init
    top_k = max(8, n // 8)
    temp = 0.00008
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Get top peaks with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Nesterov momentum update
        v_mom = mom * v_mom - lr * sg
        f_new = project_simplex(f + v_mom)
        
        conv_new = fast_convolve(f_new)
        co = conv_new.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        temp *= 1.004
        
        if no_improve > 250:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v_mom = np.zeros(n)
            lr = max(lr * 3, 0.003)
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_reduced(f, M, n_constraints=100, time_limit_lp=20):
    """Solve reduced LP with only top constraints for speed."""
    n = len(f)
    from scipy.optimize import linprog
    
    conv_f = fast_convolve(f)
    if n_constraints < len(conv_f):
        active = np.argpartition(conv_f, -n_constraints)[-n_constraints:]
    else:
        active = np.arange(len(conv_f))
    
    c = -np.ones(n)
    A_ub = np.zeros((len(active), n))
    for row, k in enumerate(active):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k_int - j]
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit_lp})
    if res.success:
        return res.x
    return None
