The Away-Step FW (FWA) update is: a_{t+1} = (1 - gamma_t - sigma_t) * a_t + gamma_t * g_t + sigma_t * (a_t - a_{t,s} e_s) / a_{t,s}

The gs-step FW (FWA/ACD) combines both:
a_{t+1} = (1 - t - s) * a_t + t * g_t + s * support(a_t), where support(a_t) = a_t - a_{t,s} e_s / a_{t,s}

"""
Hybrid optimizer for step function coefficient minimization.
Uses accelerated conditional gradient (Away-Step FW) with smooth gradient,
combined with LP oracle and evolutionary restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a) is close to 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_weights(conv, tau):
    """Compute softmax weights for the log-sum-exp smoothed max."""
    shifted = conv - conv.max()
    w = np.exp(tau * shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def gradient_smooth(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k(a)))."""
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, tau)
    # grad[i] = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i]
    grad = np.zeros(n)
    for i in range(n):
        k_end = min(2 * n - 1, i + n)
        length = k_end - i
        grad[i] = 2.0 * np.dot(w[i:k_end], a[:length])
    return grad, conv, w


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

            no_improve = 0

def generate_initializations(n, parent, rng):

    inits['triangle'] = tri / tri.sum()

    inits['gaussian'] = g / g.sum()
    
    # Two bumps

    b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_bump = b1 + b2
    inits['two_bump'] = two_bump / two_bump.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Cosine
    c = np.cos(np.pi * x / (2 * n)) ** 2 + 0.01
    inits['cosine'] = c / c.sum()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)

            ps = pp.sum()

            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent'] = pp_interp / ps
    
    return inits
