Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with softmax
parameterization, multi-length exploration, and peak-targeting strategies.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def smooth_max_grad(c, beta):
    """Returns (smooth_max, weights) where weights = softmax(beta*(c - max(c)))"""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return c_max, w
    w = exp_shifted / total
    sm = c_max + np.log(total) / beta
    return sm, w

def softmax_from_x(x):
    """Convert log-domain parameters to simplex."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    return a


def softmax_jacobian_grad(x, a, grad_a):
    """
    Given grad_a = d f / d a (where a = softmax(x)), compute d f / d x.
    For softmax: a_i = exp(x_i) / sum(exp(x_j))
    d f / d x_i = sum_j (d f / d a_j) * (d a_j / d x_i)
                 = grad_a_i * a_i - a_i * sum_j (grad_a_j * a_j)
    """
    return a * (grad_a - np.dot(a, grad_a))


class AdamSimplex:
    """Adam optimizer with softmax parameterization."""
    
    def __init__(self, x0, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.x = x0.copy()
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = np.zeros_like(x0)
        self.v = np.zeros_like(x0)
    
    def step(self, grad_x):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad_x
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad_x ** 2
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        self.x = self.x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return self.x
    
    def get_a(self):
        return softmax_from_x(self.x)


def adam_optimize(a0, beta_schedule, total_iters, n=None):

def lbfgsb_optimize(a0, beta, maxiter=500):
    """L-BFGS-B optimization with softmax parameterization."""
    n = len(a0)

def coordinate_descent(a, n_iters=100):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_obj = compute_objective(a)
    
    for it in range(n_iters):
        improved = False
        # Shuffle order for stochastic CD
        order = np.random.permutation(n)
        for j in order:
            # Try different values for a[j]
            old_val = a[j]
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[j] = old_val * scale
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                obj = compute_objective(a)
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
            else:
                a[j] = old_val
        
        if not improved:
            break
    
    return a


def structured_perturbation(a, rng, mode='edges'):
    """Apply a structured perturbation to the sequence."""
    n = len(a)
    a = a.copy()
    
    if mode == 'edges':
        # Perturb the first and last 20% of elements
        edge = max(1, n // 5)
        mask = np.zeros(n, dtype=bool)
        mask[:edge] = True
        mask[-edge:] = True
        noise = rng.standard_normal(n) * 0.05
        a[mask] += noise[mask] * a[mask].max()
    
    elif mode == 'center':
        center = n // 4
        mask = np.zeros(n, dtype=bool)
        mask[center:n-center] = True
        noise = rng.standard_normal(n) * 0.03
        a[mask] += noise[mask] * a[mask].max()
    
    elif mode == 'block':
        # Perturb a random block
        block_size = max(1, n // 8)
        start = rng.integers(0, n - block_size)
        a[start:start+block_size] *= np.exp(rng.standard_normal(block_size) * 0.1)
    
    elif mode == 'global':
        a *= np.exp(rng.standard_normal(n) * 0.02)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam with softmax parameterization,
L-BFGS-B, multi-length exploration, and peak-targeting strategies.
Key innovations: adaptive peak-targeting penalties, structured perturbations,
and a two-level optimization (smooth max annealing + exact max refinement).
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def smooth_max_and_weights(c, beta):
    """Returns (smooth_max_value, softmax_weights)."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return c_max, w
    w = exp_s / total
    sm = c_max + np.log(total) / beta
    return sm, w


def conv_gradient(w, a):
    """
    Compute d/da of sum_k w[k]*conv[k] where conv = linear_conv(a,a).
    Result: grad[j] = 2 * sum_k w[k]*a[k-j] = 2*(w*a)_linear[j]
    """
    n = len(a)
    m = len(w)
    N = 1 << (n + m - 1 + 6).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    result = np.real(ifft(W * A, N))[:n]
    return 2.0 * result


def softmax_from_x(x):
    """Convert log-parameter to simplex."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    return a


def softmax_grad_transform(a, grad_a):
    """Transform gradient from a-space to x-space for softmax param."""
    return a * (grad_a - np.dot(a, grad_a))


def adam_step(x, m, v, grad_x, lr, beta1, beta2, eps, t):
    """Single Adam step, returns updated (x, m, v)."""
    m = beta1 * m + (1.0 - beta1) * grad_x
    v = beta2 * v + (1.0 - beta2) * grad_x ** 2
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x, m, v


def adam_optimize(a0, beta_schedule, total_iters, lr_init=0.003, peak_target=False):
    """
    Adam optimization on simplex with softmax parameterization.
    beta_schedule: list of (beta, n_iters) pairs.
    peak_target: if True, add extra weight to the current peak lag.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x = np.log(a)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    total_done = 0
    for beta, n_iter in beta_schedule:
        for it in range(n_iter):
            if total_done >= total_iters:
                break
            
            a = softmax_from_x(x)
            c = fast_convolve(a)
            sm, w = smooth_max_and_weights(c, beta)
            
            if peak_target and total_done > 100:
                # Boost the peak lag weight
                peak_idx = int(np.argmax(c))
                w_boosted = w.copy()
                w_boosted[peak_idx] *= 2.0
                w_boosted /= w_boosted.sum()
                w = w_boosted
            
            grad_a = conv_gradient(w, a)
            grad_x = softmax_grad_transform(a, grad_a)
            
            # Adaptive lr: decay over time
            lr = lr_init / (1.0 + 0.001 * total_done)
            x, m, v = adam_step(x, m, v, grad_x, lr, beta1, beta2, eps, t_count + 1)
            t_count += 1
            total_done += 1
            
            # Periodic objective check
            if (it + 1) % 100 == 0 or (it == n_iter - 1):
                obj = compute_objective(softmax_from_x(x))
                if obj < best_obj:
                    best_obj = obj
                    best_a = softmax_from_x(x).copy()
    
    return best_a, best_obj


def lbfgsb_optimize(a0, beta, maxiter=300, ftol=1e-14):
    """L-BFGS-B with softmax parameterization and smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x0 = np.log(a)
    
    def obj_fn(x):
        a_local = softmax_from_x(x)
        c = fast_convolve(a_local)
        sm, _ = smooth_max_and_weights(c, beta)
        return 2.0 * n * sm
    
    def jac_fn(x):
        a_local = softmax_from_x(x)
        c = fast_convolve(a_local)
        sm, w = smooth_max_and_weights(c, beta)
        grad_a = conv_gradient(w, a_local)
        return softmax_grad_transform(a_local, grad_a)
    
    result = minimize(obj_fn, x0, jac=jac_fn, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 40, 'ftol': ftol})
    
    a_opt = softmax_from_x(result.x)
    return a_opt


def nesterov_optimize(a0, n_iters=2000, lr_init=0.01, beta_target=50.0):
    """Nesterov accelerated gradient on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x = np.log(a)
    v = np.zeros_like(x)
    
    best_obj = compute_objective(a)
    best_a = a.copy()
    mu = 0.9
    
    for it in range(n_iters):
        a_y = softmax_from_x(x + mu * v)
        c = fast_convolve(a_y)
        sm, w = smooth_max_and_weights(c, beta_target)
        grad_a = conv_gradient(w, a_y)
        grad_x = softmax_grad_transform(a_y, grad_a)
        
        lr = lr_init / (1.0 + 0.002 * it)
        v_new = mu * v - lr * grad_x
        x = x + v_new
        v = v_new
        
        if (it + 1) % 200 == 0:
            obj = compute_objective(softmax_from_x(x))
            if obj < best_obj:
                best_obj = obj
                best_a = softmax_from_x(x).copy()
    
    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descend(a, n_iters=50, rng=None):
    """Coordinate descent with multiplicative updates."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_obj = compute_objective(a)
    
    for it in range(n_iters):
        improved = False
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        
        # Prioritize coordinates that contribute to the peak
        order = np.arange(n)
        # Coordinates that contribute to peak: i and peak_k - i
        contributing = set()
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contributing.add(i)
                contributing.add(j)
        contrib_list = sorted(contributing)
        non_contrib = [i for i in order if i not in contributing]
        search_order = contrib_list + non_contrib
        
        rng.shuffle(search_order)
        
        for j in search_order:
            old_val = a[j]
            best_local = best_obj
            best_scale = 1.0
            for scale in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
                a[j] = old_val * scale
                obj = compute_objective(a)
                if obj < best_local:
                    best_local = obj
                    best_scale = scale
            if best_scale != 1.0:
                a[j] = old_val * best_scale
                best_obj = best_local
                improved = True
            else:
                a[j] = old_val
        
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if not improved:
            break
    
    return a


def structured_perturb(a, mode, rng):
    """Apply structured perturbation."""
    n = len(a)
    a = a.copy()
    if mode == 'edges':
        edge = max(2, n // 5)
        mask = np.zeros(n, dtype=bool)
        mask[:edge] = True
        mask[-edge:] = True
        noise = rng.standard_normal(n) * 0.08
        a[mask] *= np.exp(noise[mask])
    elif mode == 'center':
        c = max(2, n // 4)
        mask = np.zeros(n, dtype=bool)
        mask[c:n-c] = True
        noise = rng.standard_normal(n) * 0.05
        a[mask] *= np.exp(noise[mask])
    elif mode == 'block':
        bs = max(2, n // 6)
        start = int(rng.integers(0, max(1, n - bs)))
        a[start:start+bs] *= np.exp(rng.standard_normal(bs) * 0.1)
    elif mode == 'ramp':
        # Asymmetric ramp perturbation
        x = np.arange(n) / n
        a *= np.exp(0.1 * rng.uniform(-1, 1) * (x - 0.5))
    else:
        a *= np.exp(rng.standard_normal(n) * 0.03)
    
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def interpolate_sequence(parent, n):
    """Interpolate parent sequence to length n."""
    pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
    pp = np.maximum(pp, 0.0)
    ps = pp.sum()
    if ps > 1e-10:
        return pp / ps
    return np.ones(n) / n


def flat_spectrum_init(n, rng):
    """Initialize with flat power spectrum (random phases)."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    
    for trial in range(10):
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        for k in range(1, fft_size // 2 + 1):
            A[k] = np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * phases[k])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a

            schedule = [(25.0, total_iters // 2), (10.0, total_iters // 2)]
            a_opt, _ = adam_optimize(a_pert, schedule, total_iters, lr_init=0.002)
            try_update(a_opt)
    
    print("Phase 4 done: best=%.6f" % best_score)
    
    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # Low-beta L-BFGS-B sequence
    if remaining > 30:
        for beta in [5.0, 2.5, 1.5]:
            if time.monotonic() > deadline - 15:
                break
            t_single = remaining / 4.0
            maxiter = int(min(300, max(30, t_single * 3)))
            a_opt = lbfgsb_optimize(best_sequence, beta, maxiter)
            try_update(a_opt)
            remaining = deadline - time.monotonic()
    
    # Coordinate descent
    if time.monotonic() < deadline - 25:
        a_cd = coordinate_descend(best_sequence, n_iters=40, rng=rng)
        cd_score = try_update(a_cd)
        if np.isfinite(cd_score) and cd_score < best_score:
            if time.monotonic() < deadline - 15:
                a_opt = lbfgsb_optimize(best_sequence, beta=8.0, maxiter=150)
                try_update(a_opt)
    
    # Final Adam with very low beta
    if time.monotonic() < deadline - 20:
        t_alloc = max(5.0, deadline - time.monotonic() - 15)
        n_iters = int(min(2000, t_alloc * 30))
        schedule = [(8.0, n_iters // 2), (4.0, n_iters // 2)]
        a_opt, _ = adam_optimize(best_sequence, schedule, n_iters, lr_init=0.001)
        try_update(a_opt)
    
    # Absolute final L-BFGS-B
    if time.monotonic() < deadline - 10:
        a_opt = lbfgsb_optimize(best_sequence, beta=2.0, maxiter=100)
        try_update(a_opt)
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
