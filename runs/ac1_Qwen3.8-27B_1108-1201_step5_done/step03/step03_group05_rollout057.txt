3. The minimum possible value of `max(conv(a, a))` with `sum(a) = 1` and `a >= 0` is related to the "flatness" of the convolution.

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using targeted LP with top-constraint
selection, exact subgradient descent, and adaptive coordinate descent.
Combines multiple optimization strategies with diverse multi-start.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a, top_k=5):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a, focusing on top_k peaks.
    Returns (gradient, max_value, peak_indices).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    
    # Find top peaks
    k = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k)[-k:]
    # Only keep peaks that are close to the max
    threshold = max_val * 0.999
    peak_indices = peak_indices[conv[peak_indices] >= threshold]
    
    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv)])
    
    # Subgradient: grad_i = 2 * sum_{peaks k} a[k-i] (for valid i)
    grad = np.zeros(n)
    for k_idx in peak_indices:
        k_idx = int(k_idx)
        i_start = max(0, k_idx - n + 1)
        i_end = min(n - 1, k_idx)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k_idx - i]
    
    return grad, max_val, peak_indices


def subgradient_descent(a0, time_limit, top_k=5, lr_init=0.01, lr_decay=0.999,
                        seed=0, max_iters=100000):
    """
    Subgradient descent on exact max(conv(a,a)) with simplex projection.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    avg_obj = best_obj
    
    for it in range(max_iters):
        if time.monotonic() - t0 >= time_limit:
            break
        
        grad, cur_obj, peaks = subgradient_max_conv(a, top_k)
        
        # Polyak-like step size
        lr = lr_init * (lr_decay ** it)
        lr = max(lr, 1e-6)
        
        a_new = project_simplex(a - lr * grad)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Periodic perturbation
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            no_improve = 0
            lr = lr_init * 0.5
        
    return best_a, best_obj


def lp_top_constraints(a, M_base, top_k=10, time_limit_lp=10):
    """
    Solve LP: max sum(b) s.t. b >= 0, sum(b)=1, (a*b)_k <= M_k for top_k peaks.
    M_k = M_base for top peaks, relaxed for others.
    """
    n = len(a)
    conv_a_a = fast_convolve(a)
    
    # Find top peaks
    k = min(top_k, 2 * n - 1)
    peak_indices = np.argpartition(conv_a_a, -k)[-k:]
    peak_indices = peak_indices[conv_a_a[peak_indices] >= M_base * 0.95]
    
    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv_a_a)])
    
    # Build constraint matrix for only the peak constraints
    # (a*b)_k = sum_i a[i] * b[k-i] for valid i
    # Constraint: sum_i a[i] * b[k-i] <= M_k
    
    m = len(peak_indices)
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    for idx, k in enumerate(peak_indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[idx, i] = a[k - i]
        b_ub[idx] = M_base  # Use same M for all (or could use individual)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    # Add equality constraint sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def lp_full(a, M, time_limit_lp=10):
    """
    Solve LP: max sum(b) s.t. b >= 0, sum(b)=1, (a*b)_k <= M for ALL k.
    """
    n = len(a)
    
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum

def coordinate_descent(a0, time_limit, seed=0):
    """

    if s < 1e-15:

    n = len(a)

            a = project_simplex(a + noise)

        j = rng.integers(0, n)
        if i == j:

        # Try different amounts

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Structured
    inits['bell'] = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits['bell'] /= inits['bell'].sum()
    
    inits['bimodal'] = np.zeros(n)
    edge = max(2, n // 5)
    inits['bimodal'][:edge] = 1.0
    inits['bimodal'][-edge:] = 1.0
    inits['bimodal'] /= inits['bimodal'].sum()
    
    inits['ramp'] = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] /= inits['ramp'].sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
                inits['parent'] = pp / ps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using exact subgradient descent with
adaptive Polyak steps, targeted LP with top-constraint selection, paired
coordinate descent with bisection, and diverse multi-start across lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_k=5):
    """
    Compute subgradient of max(conv(a,a)) focusing on top_k peaks.
    Returns (gradient, max_value, peak_indices).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()

    k = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k)[-k:]
    threshold = max_val * 0.9995
    peak_indices = peak_indices[conv[peak_indices] >= threshold]

    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv)])

    # Subgradient: grad_i = sum_{k in peaks} 2*a[k-i] (valid indices only)
    grad = np.zeros(n)
    for k_idx in peak_indices:
        k_idx = int(k_idx)
        i_start = max(0, k_idx - n + 1)
        i_end = min(n - 1, k_idx)
        if i_end >= i_start:
            grad[i_start:i_end + 1] += 2.0 * a[k_idx - i_end:k_idx - i_start + 1][::-1]

    return grad, max_val, peak_indices


def subgradient_optimize(a0, time_limit, top_k=5, lr_init=0.01, lr_decay=0.9995,
                         seed=0, perturb_interval=800, perturb_scale=0.003):
    """
    Projected subgradient descent on exact max(conv(a,a)) with simplex projection.
    Uses adaptive step size and periodic perturbations.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    # Running average for adaptive step
    obj_avg = best_obj
    obj_avg_alpha = 0.95

    while True:
        if time.monotonic() - t0 >= time_limit:
            break

        grad, cur_obj, peaks = compute_subgradient(a, top_k)

        # Adaptive step: scale by gap
        obj_avg = obj_avg_alpha * obj_avg + (1 - obj_avg_alpha) * cur_obj
        gap = max(obj_avg - best_obj, 1e-8)
        lr_eff = lr * max(0.5, min(2.0, gap / max(cur_obj, 1e-10)))

        a_new = project_simplex(a - lr_eff * grad)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1
        lr = lr_init * (lr_decay ** it)
        lr = max(lr, 1e-7)

        # Periodic perturbation to escape local minima
        if no_improve >= perturb_interval:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            # Also try scaling perturbation
            noise2 = rng.standard_normal(n) * perturb_scale * 0.5
            a_alt = project_simplex(a + noise2)
            conv_alt = fast_convolve(a_alt)
            if conv_alt.max() < fast_convolve(a).max():
                a = a_alt
            no_improve = 0
            perturb_interval = int(perturb_interval * 1.3)
            perturb_scale *= 0.85
            lr = lr_init * 0.4

    return best_a, best_obj


def lp_top_constraints(a, M_target, top_k=20, time_limit_lp=5):
    """
    Solve LP: max sum(b) s.t. b >= 0, sum(b)=1, (a*b)_k <= M_target for top_k peaks.
    Returns normalized b or None.
    """
    n = len(a)
    conv_ref = fast_convolve(a)

    # Find top peaks
    k = min(top_k, 2 * n - 1)
    peak_indices = np.argpartition(conv_ref, -k)[-k:]
    # Keep peaks close to max
    threshold = conv_ref.max() * 0.95
    peak_indices = peak_indices[conv_ref[peak_indices] >= threshold]

    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv_ref)])

    m = len(peak_indices)
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)

    for idx, ki in enumerate(peak_indices):
        ki = int(ki)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        for ii in range(i_start, i_end + 1):
            A_ub[idx, ii] = a[ki - ii]
        b_ub[idx] = M_target

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def lp_full_constraints(a, M_target, time_limit_lp=10):
    """
    Solve LP: max sum(b) s.t. b >= 0, sum(b)=1, (a*b)_k <= M_target for ALL k.
    """
    n = len(a)

    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    c_obj = -np.ones(n)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def lp_optimize(a0, time_limit, seed=0):
    """
    LP-based optimization: repeatedly solve LP and move towards solution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        if stuck > 20:
            break

        conv_cur = fast_convolve(a)
        M_cur = conv_cur.max()

        # Try with progressively tighter M
        b = None
        for factor in [1.0, 0.999, 0.998, 0.995, 0.99]:
            M_try = M_cur * factor
            b = lp_top_constraints(a, M_try, top_k=25, time_limit_lp=5)
            if b is not None:
                break

        if b is None:
            b = lp_full_constraints(a, M_cur * 1.001, time_limit_lp=8)

        if b is None:
            # Perturb and retry
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            stuck += 1
            continue

        stuck = 0

        # Try different step sizes
        improved = False
        steps = [0.3, 0.2, 0.15, 0.25, 0.1, 0.4, 0.5, 0.05]
        for t in steps:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
            elif obj_try < M_cur * 0.9995:
                a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            # Tiny step
            a_try = (1 - 0.03) * a + 0.03 * b
            a_try = np.maximum(a_try, 0)
            if a_try.sum() > 1e-10:
                a = a_try / a_try.sum()

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 0.3)

    # Bimodal (U-shaped)
    bi = np.zeros(n)
    edge = max(3, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Add some in middle
    mid = max(1, n // 8)
    bi[n // 2 - mid:n // 2 + mid] = 0.3
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Smooth bimodal
    xi = x / max(n - 1, 1)
    bi_smooth = (xi ** 2 + (1 - xi) ** 2 + 0.1)
    bi_smooth = np.maximum(bi_smooth, 1e-6)
    bi_smooth /= bi_smooth.sum()
    inits['smooth_bi'] = bi_smooth

    # Cosine window (raised)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Inverse cosine (peaks at edges)
    inv_cos = 1.0 / (cos_w + 0.1)
    inv_cos /= inv_cos.sum()
    inits['inv_cos'] = inv_cos

    # Triangle (peak in middle)
    tri = 1.0 - 2.0 * np.abs(xi - 0.5)
    tri = np.maximum(tri, 0.05)
    tri /= tri.sum()
    inits['triangle'] = tri

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
                inits['parent'] = pp / ps

        # Perturbed parent
        if 'parent' in inits:
            noise = rng.standard_normal(n) * 0.05
            pp = inits['parent'] + noise
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_pert'] = pp / ps

    return inits

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)

    # Select top candidates for deep optimization
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 4:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add diverse randoms at best length
    best_n = len(best_sequence) if best_sequence is not None else 256
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.4)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.5)))

    remaining = deadline - time.monotonic() - 150.0
    n_configs = len(top_configs)
    per_config = max(25.0, remaining / max(n_configs, 1))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 130:
            break
        t_alloc = min(per_config, 300.0)
        if t_alloc < 20:
            break

        print(f"  Config {i}: n={n}, t_alloc={t_alloc:.0f}s", flush=True)
        a_opt, obj_opt = combined_optimize(a0, t_alloc, seed=rng.integers(0, 100000))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"    Deep: {score:.6f}", flush=True)
            else:
                print(f"    Deep: {score:.6f} (no improvement)", flush=True)

    # Phase 3: Polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Coordinate descent polish
        t_cd = remaining * 0.35
        if t_cd > 10:
            a_cd, obj_cd = paired_coordinate_descent(best_sequence, t_cd, seed=111)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {best_score:.6f}", flush=True)

        # Fine subgradient
        remaining = deadline - time.monotonic() - 10.0
        t_sg = remaining * 0.30
        if t_sg > 5:
            a_sg, obj_sg = subgradient_optimize(
                best_sequence, t_sg, top_k=3, lr_init=0.001,
                lr_decay=0.99995, seed=222,
                perturb_interval=2000, perturb_scale=0.0005)
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                print(f"  FineSG: {best_score:.6f}", flush=True)

        # LP polish
        remaining = deadline - time.monotonic() - 10.0
        t_lp = remaining * 0.25
        if t_lp > 10:
            a_lp, obj_lp = lp_optimize(best_sequence, t_lp, seed=333)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LPPolish: {best_score:.6f}", flush=True)

        # Final coordinate descent
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_c, obj_c = paired_coordinate_descent(best_sequence, remaining, seed=444)
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"  FinalCoord: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
