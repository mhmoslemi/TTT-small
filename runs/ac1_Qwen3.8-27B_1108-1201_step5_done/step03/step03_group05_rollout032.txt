```python
    A2 = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(P2 * np.conj(A2), fft_size)[:n]
    
    # Gradient: grad_i = (2n/s^2) * h_i - (4n * log_sum_exp) / (beta * s^3)
    grad = (2.0 * n / (s * s)) * h - (4.0 * n * log_sum_exp) / (beta * s * s * s)
    
    # Ensure gradient is valid
    grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    
    return F, grad


def lbfqsb_optimize(a0, beta, time_limit, maxiter=500):
    """
    Run L-BFGS-B on the smoothed objective with non-negativity constraints.
    """
    n = len(a0)
    x0 = np.maximum(a0.copy(), 1e-15)  # strictly positive to avoid division by zero
    
    def fun(x):
        val, _ = smooth_obj_and_grad(x, beta)
        return val
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g
    
    t0 = time.monotonic()
    
    # Callback to check time
    def callback(xk):
        return time.monotonic() - t0 > time_limit
    
    try:
        result = minimize(fun, x0, jac=grad, method='L-BFGS-B',
                         bounds=[(0, None)] * n, options={'maxiter': maxiter,
                                                     'maxfun': 10000,
                                                     'ftol': 1e-14,
                                                     'gtol': 1e-8,
                                                     'maxls': 50})
        a_best = np.maximum(result.x, 0)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}")
        a_best = x0.copy()
    
    return a_best, time.monotonic() - t0


def peak_coordinate_descent(a, time_limit, seed=42):
    """
    Directly target the top convolution peaks by moving mass from 
    high-contributing elements to low-contributing ones.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    peak_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        
        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Random restart within the current basin
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            no_improve = 0
            continue
        
        # Find top peaks
        n_peaks = min(10, len(conv))
        top_peaks = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            k = int(k)
            if conv[k] < max_val * 0.9:
                continue
            
            # Find contributing pairs
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            if i_start > i_end:
                continue
            
            # Contributions: a[i] * a[k-i] for i in [i_start, i_end]
            indices = np.arange(i_start, i_end + 1)
            partners = k - indices
            contribs = a[indices] * a[partners]
            
            # Find the dominant contributor
            max_c_idx = np.argmax(contribs)
            i_dom = indices[max_c_idx]
            p_dom = partners[max_c_idx]
            
            # Try moving mass from the dominant element to a random position
            # that doesn't contribute to this peak
            for attempt in range(5):
                j = rng.integers(0, n)
                # Check that j doesn't contribute to peak k
                if (j >= i_start and j <= i_end) or (k - j >= i_start and k - j <= i_end):
                    continue
                
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = a[i_dom] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[i_dom] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < max_val - 1e-12:
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved and no_improve % 50 == 0:
            # Try random pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-8:
                for frac in [0.01, 0.02, 0.05]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < conv.max() - 1e-12:
                        a = a_try
                        improved = True
                        break
    
    return best_a, best_max


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n, all on the simplex."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian at center
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g
    
    # Bimodal
    edge = max(2, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Raised cosine
    if n > 1:
        cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        cos_w = np.maximum(cos_w, 1e-6)
        cos_w /= cos_w.sum()
        inits['raised_cos'] = cos_w
    
    # Sine-based
    sine = np.sin(np.pi * x / n)
    sine = np.maximum(sine, 1e-6)
    sine /= sine.sum()
    inits['sine'] = sine
    
    # Power ramp
    for pwr in [2, 4]:
        ramp = (x + 1) ** pwr
        ramp /= ramp.sum()
        inits[f'power_{pwr}'] = ramp
        ramp2 = ramp[::-1].copy()
        inits[f'power_{pwr}_rev'] = ramp2
    
    # Random Dirichlet
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        inits[f'dir_{alpha}'] = rng.dirichlet(np.ones(n) * alpha)
    
    # Chirp-like (non-uniform spacing)
    chirp = np.exp(1j * np.pi * x**2 / n).real
    chirp = np.abs(chirp)
    chirp = np.maximum(chirp, 1e-6)
    chirp /= chirp.sum()
    inits['chirp'] = chirp
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                # Interpolate to length n
                pp_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), parent)
                pp_interp = np.maximum(pp_interp, 0)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2
    
    return inits
```

```python
def softmax_obj_and_grad(x, beta):
    """
    Objective and gradient in log-space (softmax parameterization).
    a = softmax(x) = exp(x) / sum(exp(x))
    F = 2n * logsumexp(beta * conv(a,a)) / (beta * 1) since sum(a) = 1
    = 2n * logsumexp(beta * conv(a,a)) / beta
    """
    n = len(x)
    # Compute a = softmax(x)
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum()
    
    conv = fast_convolve(a)
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    log_sum_exp = max_bc + np.log(np.sum(np.exp(beta_conv - max_bc)))
    
    F = 2.0 * n * log_sum_exp / beta
    
    # Gradient: dF/dx_i = sum_j (dF/da_j) * (da_j/dx_i)
    # da_j/dx_i = a_j * (delta_{ij} - a_i) = a_j * delta_{ij} - a_i * a_j
    # dF/da_j = (2n/(beta)) * d(logsumexp)/da_j = (2n/(beta)) * beta * h_j = 2n * h_j
    #   where h_j = sum_k p_k * a[k-j] (as before, with p = softmax(beta*conv))
    
    # So dF/dx_i = sum_j 2n * h_j * a_j * (delta_{ij} - a_i)
    # = 2n * (a_i * h_i - a_i * sum_j h_j * a_j)
    # = 2n * a_i * (h_i - sum_j h_j * a_j)
    
    p = np.exp(beta_conv - max_bc)
    p_sum = p.sum()
    p = p / p_sum
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    h = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    h_dot_a = h.dot(a)
    grad_x = 2.0 * n * a * (h - h_dot_a)
    
    return F, grad_x
```

```python
"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized L-BFGS-B
with adaptive beta annealing, peak coordinate descent, and diverse
multi-scale initialization.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def compute_h(a, p):
    """
    Compute h[i] = sum_j a[j] * p[i+j] using FFT.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    return np.fft.irfft(P * np.conj(A), fft_size)[:n]


def softmax_objective(x, beta):
    """
    F(x) = 2n * logsumexp(beta * conv(softmax(x), softmax(x))) / beta
    Since sum(softmax(x)) = 1, the normalization factor is 1.
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    log_sum_exp = max_bc + np.log(np.sum(np.exp(beta_conv - max_bc)))
    return 2.0 * n * log_sum_exp / beta


def softmax_gradient(x, beta):
    """
    Gradient of softmax_objective w.r.t. x.
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    
    p = np.exp(beta_conv - max_bc)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    h = compute_h(a, p)
    h_dot_a = np.dot(h, a)
    grad = 2.0 * n * a * (h - h_dot_a)
    
    return grad


def obj_grad_combined(x, beta):
    """Combined objective and gradient for scipy.optimize."""
    n = len(x)
    a = softmax(x)
```

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_func(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def compute_h(a, p):
    """Compute h[i] = sum_j a[j] * p[i+j] using FFT. a has length n, p has length 2n-1."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    return np.fft.irfft(P * np.conj(A), fft_size)[:n]


def obj_and_grad(x, beta):
    """
    Objective and gradient in softmax parameterization.
    F(x) = 2n * logsumexp(beta * conv(softmax(x), softmax(x))) / beta
    """
    n = len(x)
    a = softmax_func(x)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    exp_shifted = np.exp(beta_conv - max_bc)
    p_sum = exp_shifted.sum()
    log_sum_exp = max_bc + np.log(p_sum)
    F = 2.0 * n * log_sum_exp / beta

    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_shifted / p_sum

    h = compute_h(a, p)
    h_dot_a = np.dot(h, a)
    grad = 2.0 * n * a * (h - h_dot_a)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    return F, grad
