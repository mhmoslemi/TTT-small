    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_dense(a):
    """Build dense matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def build_conv_matrix_sparse(a):
    """Build sparse matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        valid = a != 0
        if valid.any():
            idx = np.where(valid)[0]
            rows.extend(range(i, i + len(idx)))
            cols.extend([i] * len(idx))
            vals.extend(a[idx])
    return sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(a, M, use_sparse=False):
    """
    Solve LP: maximize sum(g) s.t. conv(a, g) <= M (componentwise), g >= 0.
    Returns (g, g_sum) or (None, 0) on failure.
    """
    n = len(a)
    if use_sparse:
        A_ub = build_conv_matrix_sparse(a)
    else:
        A_ub = build_conv_matrix_dense(a)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum
    except Exception:
        pass
    return None, 0.0


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = (u - (cssv - 1.0) / ar) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temp):
    """Compute weighted subgradient at top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    peak_vals = conv[peaks]
    scaled = peak_vals / max(temp, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]
    return sg


def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp=0.0002):
    """Nesterov-accelerated projected subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    a_prev = a.copy()
    t = 1.0
    
    _, conv = 2.0 * n * 0, fast_convolve(a)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    batch = 20
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov lookahead
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + (t - 1.0) / t_new * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            
            # Compute subgradient at y
            conv_y = fast_convolve(y)
            sg = compute_subgradient(y, conv_y, top_k, temp)
            
            # Update
            a_new = y - lr * sg
            a_new = project_simplex(a_new)
            
            # Evaluate
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a_prev = a
            a = a_new
            t = t_new
        
        lr *= 0.9995
        temp *= 1.003
        if no_improve >= 200:
            break
    
    return best_a, best_obj


def lp_optimize(a0, time_limit, use_sparse=False, verbose=False):
    """
    LP-based iterative optimization with adaptive M and line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 100
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        
        if remaining < 1.0:
            break
        
        # Adaptive M: slowly decrease target to force progress
        M_target = M * (1.0 - 0.002 * min(iteration / 1000.0, 1.0))
        M_lp = max(M * 1.0001, M_target)
        
        # Solve LP
        g, g_sum = solve_lp_direction(a, M_lp, use_sparse=use_sparse)
        
        if g is None or g_sum <= 1.0 + 1e-7:
            no_improve += 1
        else:
            d = g / g_sum - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Line search with backtracking
                best_t = 0.0
                best_obj_ls = best_obj
                
                # Try geometric sequence
                t_candidates = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]
                for t_cand in t_candidates:
                    a_try = a + t_cand * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    
                    if obj_try < best_obj_ls - 1e-14:
                        best_obj_ls = obj_try
                        best_t = t_cand
                
                # Refine around best_t
                if 0.01 < best_t < 0.99:
                    step = 0.05
                    for _ in range(3):
                        improved = False
                        for dt in [-step, step]:
                            t_try = best_t + dt
                            if t_try <= 0 or t_try >= 1:
                                continue
                            a_try = a + t_try * d
                            a_try = np.maximum(a_try, 0.0)
                            s_try = a_try.sum()
                            if s_try < 1e-15:
                                continue
                            a_try = a_try / s_try
                            conv_try = fast_convolve(a_try)
                            obj_try = 2.0 * n * conv_try.max()
                            if obj_try < best_obj_ls - 1e-14:
                                best_obj_ls = obj_try
                                best_t = t_try
                                improved = True
                        if not improved:
                            step *= 0.5
                
                if best_t > 0 and best_obj_ls < best_obj - 1e-13:
                    a_new = a + best_t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.003 + 0.005 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(500, 100 + 50 * perturb_count)
            
            if perturb_count > 50:
                break
    
    return best_a, best_obj


def spectral_polish(a0, time_limit, alpha=0.1):
    """
    Frequency-domain spectral shaping to reduce autocorrelation peaks.
    Iteratively modifies the magnitude spectrum to flatten the autocorrelation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute current spectrum
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        mag = np.abs(A)
        
        # Compute autocorrelation
        r = fast_convolve(a)
        r_max = r.max()
        
        # Create peak reduction mask in time domain
        threshold = 0.85 * r_max
        peak_mask = (r > threshold).astype(np.float64)
        # Weight by how much above threshold
        peak_weights = np.maximum(r - threshold, 0)
        if peak_weights.sum() > 1e-15:
            peak_weights = peak_weights / peak_weights.sum()
            # Zero-pad to fft_size
            pw_full = np.zeros(fft_size)
            pw_full[:len(peak_weights)] = peak_weights
            # Transform to frequency domain
            dp = np.fft.rfft(pw_full)
            # Update magnitude spectrum
            mag_sq = mag * mag
            mag_sq = mag_sq - alpha * np.abs(dp)
            mag_sq = np.maximum(mag_sq, 1e-20)
            mag_new = np.sqrt(mag_sq)
        else:
            mag_new = mag.copy()
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce alpha
            alpha *= 0.9
            if alpha < 1e-4:
                break
        
        iteration += 1
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine bump
    inits['sine'] = (np.sin(np.pi * (x + 0.5) / n) + 0.01)
    inits['sine'] /= inits['sine'].sum()
    
    # Double sine
    inits['dsine'] = (np.sin(np.pi * (x + 0.5) / n) + 0.3 * np.sin(2 * np.pi * (x + 0.5) / n) + 0.1)
    inits['dsine'] = np.maximum(inits['dsine'], 0)
    inits['dsine'] /= inits['dsine'].sum()
    
    # Exponential decay
    inits['exp'] = np.exp(-x / (n * 0.3))
    inits['exp'] /= inits['exp'].sum()
    
    # Bi-modal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                # Downsample
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                # Upsample
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits


def subgradient_optimize(a0, time_limit, lr_init=0.012, momentum=0.92, top_k=60, temp=0.00015):
    """
    Projected subgradient descent with momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    batch = 30
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            sg = compute_subgradient(a, conv, top_k, temp)
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.999
        temp *= 1.002
        momentum = min(0.97, momentum + 0.0003)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP, subgradient,
    spectral polishing, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    def update_best(a):
        nonlocal best_sequence, best_score
        if a is not None and len(a) > 0:
            try:
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a.copy()
                    return score
            except Exception:
                pass
        return float('inf')
    
    # Phase 1: Quick scan to find promising (n, init) combinations
    print("=== Phase 1: Quick Scan ===")
    t_start = time.monotonic()
    phase1_limit = min(300.0, (deadline - time.monotonic()) * 0.15)
    
    lengths = [256, 512, 1024]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > deadline - phase1_limit - 30:
            break
        
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        # Prioritize parent-based inits
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random', 'sine', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]
        
        t_per = max(10.0, phase1_limit / (len(lengths) * 4))
        
        for name in init_names:
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            # Quick subgradient for initial scan (faster than LP)
            a_opt, obj_opt = subgradient_optimize(a0, time_limit=t_alloc,
                                                   lr_init=0.015, momentum=0.9,
                                                   top_k=40, temp=0.0002)
            score = update_best(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                print(f"  n={n}, {name}: {score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # Phase 2: Deep LP optimization on top candidates
    print("=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()
    
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt, score))
            seen.add(key)
        if len(top_configs) >= 5:
            break
    
    # Always include the current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy(), best_score))
    
    remaining = deadline - time.monotonic() - 100.0
    n_candidates = max(1, len(top_configs))
    t_per_deep = max(30.0, remaining * 0.55 / n_candidates)
    
    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 50)
        if t_alloc < 15:
            break
        
        print(f"  Deep LP {i} (n={n}, init_score={sc:.6f}), time_alloc={t_alloc:.0f}s")
        
        # Use sparse for large n
        use_sparse = n > 512
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, use_sparse=use_sparse)
        score = update_best(a_opt)
        if np.isfinite(score):
            print(f"    -> {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # Phase 3: Nesterov subgradient + spectral polish
    print("=== Phase 3: Nesterov + Spectral Polish ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 30:
            n_best = len(best_sequence)
            
            # Nesterov with different hyperparameters
            nest_configs = [
                {'lr_init': 0.008, 'momentum': 0.93, 'top_k': 60, 'temp': 0.0001},
                {'lr_init': 0.015, 'momentum': 0.88, 'top_k': 80, 'temp': 0.0003},
                {'lr_init': 0.005, 'momentum': 0.95, 'top_k': 40, 'temp': 0.00008},
            ]
            
            t_per = remaining / (len(nest_configs) + 2)
            
            for ci, cfg in enumerate(nest_configs):

                if t_alloc < 10:

                score = update_best(a_opt)
                if np.isfinite(score):

            # Spectral polish
            if time.monotonic() < deadline - 20:
                t_alloc = min(30.0, deadline - time.monotonic() - 10)
                if t_alloc > 10:
                    a_opt, obj_opt = spectral_polish(best_sequence, time_limit=t_alloc, alpha=0.05)
                    score = update_best(a_opt)
                    if np.isfinite(score):
                        print(f"  Spectral: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # Phase 4: Final LP polish + perturbation restarts
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 20:
            n_best = len(best_sequence)
            use_sparse = n_best > 512
            
            # Final LP polish on best
            t_alloc = min(remaining * 0.4, deadline - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc, use_sparse=use_sparse)
                score = update_best(a_opt)
                if np.isfinite(score):
                    print(f"  Final LP: {score:.6f}")
            
            # Perturbation restarts
            remaining = deadline - time.monotonic() - 5.0
            if remaining > 15:
                n_perturb = max(1, int(remaining / 20))
                t_per_perturb = remaining / n_perturb
                
                for pi in range(n_perturb):
                    if time.monotonic() > deadline - 5:
                        break
                    if t_per_perturb < 5:
                        break
                    
                    # Try different perturbation scales
                    scale = 0.002 + 0.003 * (pi % 5)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a_pert = best_sequence + noise
                    a_pert = np.maximum(a_pert, 0.0)
                    s = a_pert.sum()
                    if s > 1e-15:
                        a_pert = a_pert / s
                    
                    t_alloc = min(t_per_perturb, deadline - time.monotonic() - 3)
                    if t_alloc < 5:

                    # Alternate between LP and subgradient
                    if pi % 2 == 0:
                        a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, use_sparse=use_sparse)
                    else:
                        a_opt, obj_opt = subgradient_optimize(a_pert, time_limit=t_alloc,
                                                               lr_init=0.01, momentum=0.9,
                                                               top_k=50, temp=0.00015)
                    score = update_best(a_opt)
                    if np.isfinite(score):
                        print(f"  Perturb {pi}: {score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    try:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    except Exception:
        pass
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
