def fast_convolve(a):

def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)

        w[np.argmax(x)] = 1.0
        return x_max, w

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Exponential (softmax) reparameterization for strict positivity
- Adaptive log-sum-exp smoothing of the max
- L-BFGS-B with exact gradient
- Multi-start over diverse initializations and lengths
- Active-set guided emphasis
- Final projected gradient polish

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_conv_gradient(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j]
    which equals sum_k w[k] * d(conv[k])/d(a[j])
    """
    n = len(a)
    # grad[j] = 2 * convolve(w, a[::-1])[n-1+j]
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A_rev = fft(a[::-1], N)
    conv_full = np.real(ifft(W * A_rev, N))
    grad = 2.0 * conv_full[n-1:n-1+n]
    return grad


def softmax_from_logx(logx):
    """Convert log-space to probability vector."""
    x = logx - logx.max()
    a = np.exp(x)
    a = a / a.sum()
    return a


def logx_from_softmax(a):
    """Convert probability vector to log-space."""
    a = np.maximum(a, 1e-30)
    return np.log(a)


def objective_from_logx(logx, n, beta):
    """Objective function in log-space parameterization."""
    a = softmax_from_logx(logx)
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def gradient_from_logx(logx, n, beta):
    """Gradient in log-space: J^T * grad_a where J is the Jacobian of softmax."""
    a = softmax_from_logx(logx)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    
    # grad_a[j] = 2*n * sum_k w[k] * 2*a[k-j] = 4*n * sum_k w[k]*a[k-j]
    # Wait, let me recompute.
    # f(a) = 2*n*smooth_max(conv(a))
    # df/da[j] = 2*n * sum_k w[k] * d(conv[k])/d(a[j]) = 2*n * 2*sum_k w[k]*a[k-j]
    # df/da[j] = 4*n * sum_k w[k]*a[k-j]
    
    conv_grad = compute_conv_gradient(a, w)  # This is 2*sum_k w[k]*a[k-j]
    grad_a = 2.0 * n * conv_grad  # = 4*n*sum_k w[k]*a[k-j]
    
    # Now, d(log-softmax)/d(logx) = a*(I - outer(a,a))
    # So grad_logx[j] = sum_i (da[i]/d(logx[j])) * grad_a[i]
    # = a[j] * grad_a[j] - a[j] * sum_i a[i]*grad_a[i]
    # = a[j] * (grad_a[j] - dot(a, grad_a))
    
    dot = np.dot(a, grad_a)
    grad_logx = a * (grad_a - dot)
    return grad_logx


def optimize_lbfgsb_logx(logx0, n, beta, maxiter=300, time_budget=30):
    """Optimize in log-space using L-BFGS-B."""
    t_start = time.monotonic()
    
    def obj(x):
        if time.monotonic() - t_start > time_budget:
            raise TimeoutError
        return objective_from_logx(x, n, beta)
    
    def jac(x):
        if time.monotonic() - t_start > time_budget:
            raise TimeoutError
        return gradient_from_logx(x, n, beta)
    
    try:
        result = minimize(obj, logx0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'maxls': 30})
        return result.x, result.fun
    except TimeoutError:
        return logx0, objective_from_logx(logx0, n, beta)
    except Exception as e:
        return logx0, objective_from_logx(logx0, n, beta)


def projected_gradient_step(a, w, n, lr):
    """One projected gradient step on the simplex."""
    conv_grad = compute_conv_gradient(a, w)
    grad = 2.0 * n * conv_grad
    a_new = a - lr * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    a_new = project_simplex(a_new)
    return a_new


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond))
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_pg(a0, beta, lr=0.05, n_iter=500, time_budget=30):
    """Projected gradient descent on simplex."""
    n = len(a0)
    a = a0.copy()
    t_start = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iter):
        if time.monotonic() - t_start > time_budget:
            break
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        a_new = projected_gradient_step(a, w, n, lr)
        lr *= 0.999
        a = a_new
    
    return best_a, best_obj


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = a / a.sum()
    return a


def sqrt_geometric_init(n):
    """Initialization based on sqrt of geometric series (C(2k,k)/4^k)."""
    a = np.zeros(n)
    c = 1.0
    for k in range(n):
        if k == 0:
            c = 1.0
        else:
            c = c * (2*k) * (2*k - 1) / (k * k)  # C(2k,k)/C(2k-2,k-1) = (2k)(2k-1)/(k^2)
            # Actually: C(2k,k) = C(2k-2,k-1) * (2k)(2k-1)/(k*k)
            # And we want C(2k,k)/4^k
            # C(2k,k)/4^k = [C(2k-2,k-1)/4^{k-1}] * (2k)(2k-1)/(k^2 * 4)
            # = prev * (2k-1)/(2k)
            # Wait: C(2k,k) = C(2k-2,k-1) * (2k)!/((k!)^2) / [(2k-2)!/((k-1)!)^2]
            # = C(2k-2,k-1) * (2k)(2k-1)/(k^2)
            # C(2k,k)/4^k = C(2k-2,k-1)/4^{k-1} * (2k)(2k-1)/(k^2*4)
            # = prev * (2k)(2k-1)/(4k^2)
            # = prev * (2k-1)/(2k)
            c = c * (2*k - 1) / (2*k)
        a[k] = c
    a = a / a.sum()
    return a


def bell_init(n, exponent=1.5):
    """Bell-shaped initialization: a[k] proportional to (1 - |2k/n - 1|^exponent)."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - np.abs(2*x - 1) ** exponent)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
