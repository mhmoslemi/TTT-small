"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based subgradient Adam
with adaptive temperature, LP refinement on active peaks, multi-length
exploration, and coordinate descent polishing.
"""

import numpy as np
import time
import sys
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(w, a):
    """
    Compute the subgradient of sum_k w[k]*C[k] with respect to a.
    where C = conv(a,a).
    Result: sg[i] = 2 * sum_k w[k] * a[k-i] (exact formula).
    Computed via FFT in O(n log n).
    """
    n = len(a)
    # B = reversed a
    B = a[::-1]
    # conv(w, B) has length (2n-1) + n - 1 = 3n-2
    # We need indices [n-1, 2n-2] of this convolution
    fft_size = 2 ** (3 * n).bit_length()
    W_hat = np.fft.rfft(w, fft_size)
    B_hat = np.fft.rfft(B, fft_size)
    conv_wB = np.fft.irfft(W_hat * B_hat, fft_size)[:3 * n - 1]
    # sg[i] = 2 * conv(w,B)[n-1+i] for i=0,...,n-1
    sg = 2.0 * conv_wB[n - 1:2 * n - 1]
    return sg


def adam_update(a, sg, m, v, t, lr, beta1, beta2, eps):
    """Adam optimizer step with simplex projection."""
    m = beta1 * m + (1.0 - beta1) * sg
    v = beta2 * v + (1.0 - beta2) * sg * sg
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a - step)
    return a_new, m, v


def optimize_adam(a_init, time_limit, lr_init=0.01, beta1=0.9, beta2=0.999,
                  temp_init=0.05, temp_min=0.001, temp_cool=0.9995,
                  lr_decay=0.9999, perturb_every=300, perturb_size=0.01):
    """
    Adam optimization on the simplex for minimizing max(conv(a,a)).
    Uses FFT-based subgradient computation and adaptive temperature.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    temp = temp_init
    
    conv = fast_convolve(a)

def solve_lp_topk(a, M, top_k=15):
    """
    Solve LP: max sum(g) s.t. g>=0, sum(g)=1, (f*g)_k <= M for top K peaks.
    Returns the optimal direction g.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top K peak indices
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]  # sort descending
    
    # Build constraint matrix for these K constraints
    # (a * g)[k] = sum_j a[j] * g[k-j] <= M
    A_ub = np.zeros((len(peak_idx), n))
    for idx, k in enumerate(peak_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = a[k - j]
    
    b_ub = np.full(len(peak_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': max(1.0, time_limit_global - (time.monotonic() - t0_global))})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
                return g
    except Exception:
        pass
    return None

time_limit_global = 0.0
t0_global = 0.0


def lp_refinement(a_init, time_limit, top_k=15, blend=0.3):
    """
    LP-based refinement: iteratively find the best direction g that
    maintains the peak constraint, then blend and project.
    """
    global time_limit_global, t0_global
    time_limit_global = time_limit
    t0_global = time.monotonic()
    
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    M = conv.max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_topk(a, M, top_k)
        if g is None:
            break
        
        # Check if g is too similar to a (no progress)
        if np.allclose(g, a, atol=1e-6):
            no_improve += 1
        else:
            a_new = (1.0 - blend) * a + blend * g
            a = project_simplex(a_new)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= 20:
            # Reduce blend and try less aggressive LP
            blend = max(0.05, blend * 0.5)
            no_improve = 0
            if blend < 0.05:
                break
        
        # Slightly increase M tolerance to allow progress
        # (the LP might be infeasible or give a trivial solution)
        # Actually, let's just break if no progress
        if no_improve >= 50:
            break
    
    return best_a, best_obj


def coordinate_descent_fast(a_in, max_rounds=30, time_limit=60.0):
    """
    Fast coordinate descent: try moving mass from high-gradient
    positions to low-gradient positions.
    """
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if conv.max() < 1e-15:
            break
        
        # Compute subgradient (uniform weight on max peak)
        pmax_idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[pmax_idx] = 1.0
        sg = compute_subgradient_fft(w, a)
        
        # Find the best coordinate to decrease (highest positive gradient)
        # and the best to increase (lowest gradient)
        # But we need to be in the simplex, so we do pairwise swaps.
        
        # Sort by gradient: decrease where sg is high, increase where sg is low
        # But also consider the current values (don't move too much from small entries)
        
        improved = False
        # Try top few swaps
        sgRanked = np.argsort(sg)
        # Best source: high sg, high a value
        # Best target: low sg
        candidates = []
        for i in range(min(10, n)):
            src = n - 1 - sgRanked[i]  # high sg index
            for j in range(min(5, n)):
                dst = sgRanked[j]  # low sg index
                if src != dst and a[src] > 1e-6:
                    candidates.append((src, dst))
        
        for src, dst in candidates[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            # Try different fractions
            for frac in [0.02, 0.05, 0.1]:
                delta = a[src] * frac
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[src] -= delta
                a_test[dst] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        
        if not improved:
            break
    
    return a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Wider Gaussian
    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss_narrow'] = g2 / g2.sum()
    
    # Edge (mass near boundaries)
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Power law (decay from center)
    d = np.abs(x - 0.5 * (n - 1))
    pl = np.exp(-d / (0.2 * n))
    inits['power'] = pl / pl.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                inits['parent_interp'] = interpolate_sequence(p / ps, n)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based subgradient Adam
with adaptive temperature, top-K peak targeting, LP refinement, multi-length
exploration, and coordinate descent polishing. Key innovation: exact O(n log n)
subgradient via FFT of the autocorrelation weights.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(w, a):
    """
    Compute exact subgradient of sum_k w[k]*C[k] w.r.t. a, where C=conv(a,a).
    Uses the identity: sg[i] = 2*sum_k w[k]*a[k-i], computed via FFT.
    """
    n = len(a)
    B = a[::-1]
    total_len = 3 * n - 2
    fft_size = 2 ** (total_len).bit_length()
    W_hat = np.fft.rfft(w, fft_size)
    B_hat = np.fft.rfft(B, fft_size)
    conv_wB = np.fft.irfft(W_hat * B_hat, fft_size)[:total_len + 1]
    sg = 2.0 * conv_wB[n - 1:2 * n - 1]
    return sg


def softmax_weights(conv, temperature):
    """Compute softmax weights over convolution peaks."""
    pmax = conv.max()
    if pmax < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    scaled = conv / (temperature * pmax + 1e-15)
    w = np.exp(scaled - scaled.max())
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w


def topk_weights(conv, k, temperature):
    """Compute weights concentrated on top-k peaks with softmax within them."""
    if k >= len(conv):
        return softmax_weights(conv, temperature)
    idx = np.argpartition(conv, -k)[-k:]
    w = np.zeros(len(conv))
    vals = conv[idx]
    t = temperature * vals.max() + 1e-15
    sw = np.exp(vals / t)
    w[idx] = sw / sw.sum()
    return w


def adam_step(a, sg, m, v, t, lr, beta1, beta2, eps):
    """Adam optimizer step with simplex projection."""
    m = beta1 * m + (1.0 - beta1) * sg
    v = beta2 * v + (1.0 - beta2) * sg * sg
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a - step)
    return a_new, m, v


def nesterov_step(a, a_prev, sg, lr, mu, t):
    """Nesterov accelerated gradient step with simplex projection."""
    alpha = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
    mu = 1.0 / alpha
    y = a + mu * (a - a_prev)
    y = project_simplex(y)
    # Recompute at y would be ideal but expensive; use sg from a as approximation
    a_new = project_simplex(a - lr * sg)
    return a_new, y


def optimize_adam(a_init, time_limit, lr_init=0.02, beta1=0.9, beta2=0.999,

def solve_lp_direction(a, M, top_k, sub_time_limit=5.0):
    """
    Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for top-k peak lags.
    Returns normalized direction g or None.
    """
    n = len(a)
    conv = fast_convolve(a)

    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

    A_ub = np.zeros((len(peak_idx), n))
    for idx, k in enumerate(peak_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = a[k_int - j]

    b_ub = np.full(len(peak_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': sub_time_limit})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_refinement(a_init, time_limit, top_k=12, blend_init=0.25):
    """
    LP-based directional refinement: iteratively find best feasible direction,
    blend with current, and project onto simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)

        if M < 1e-15:
            break

        t_lp = max(1.0, min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3))
        g = solve_lp_direction(a, M, top_k, sub_time_limit=t_lp)
        if g is None:
            break

        # Check if LP gives meaningful direction
        g_diff = np.linalg.norm(g - a)
        if g_diff < 1e-7:
            no_improve += 1
        else:
            a_new = (1.0 - blend) * a + blend * g
            a = project_simplex(a_new)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj

                no_improve = 0
            else:
                no_improve += 1

            blend = max(0.05, blend * 0.5)
            no_improve = 0
            if blend < 0.05:

        if no_improve >= 40:

def coordinate_descent_fast(a_in, max_rounds=40, time_limit=60.0):
    """
    Fast coordinate descent using gradient-guided pairwise mass transfer.
    """
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if conv.max() < 1e-15:
            break

        # Find the dominant peak
        pmax_idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[pmax_idx] = 1.0
        sg = compute_subgradient_fft(w, a)

        # Rank coordinates by gradient (positive sg means increasing a increases peak)
        # We want to decrease a where sg is high, increase where sg is low
        sg_rank = np.argsort(sg)
        low_sg = sg_rank[:min(15, n // 4)]
        high_sg = sg_rank[-min(15, n // 4):]

        improved = False
        for src in high_sg:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-7:
                continue
            for dst in low_sg:
                if src == dst:
                    continue
                for frac in [0.03, 0.07, 0.12]:
                    delta = a[src] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break

        if not improved:
            break

    return a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(a)
    if old_n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a)
    # Smooth slightly
    if new_n > 3:
        kernel = np.array([0.25, 0.5, 0.25])
        a_smooth = np.convolve(a_interp, kernel, mode='same')
        a_interp = a_smooth
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp
