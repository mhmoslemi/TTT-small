def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

def nesterov_subgrad(f0, time_limit, lr=0.002, mom=0.9, top_k=10):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Subgradient at y
        conv_y = fast_convolve(y)
        if top_k < len(conv_y):
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv_y))
        
        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]
        
        # Nesterov update
        v = mom * v - cur_lr * sg
        y_new = project_simplex(y + v)
        
        # Update y and f (Nesterov scheme)
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (alpha - 1.0) / alpha
        f_new = y_new + beta * (y_new - y)
        f_new = project_simplex(f_new)
        
        y = y_new
        f = f_new
        
        conv_f = fast_convolve(f)
        cur_max = conv_f.max()
        
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M, time_per_lp=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_per_lp})
    if res.success:
        return res.x
    return None

                    f_try[src] -= delta
                    f_try[j] += delta
                    st = f_try.sum()
                    f_try = f_try / st
                    conv_try = fast_convolve(f_try)
                    if conv_try.max() < cur_max - 1e-13:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try random small transfers
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            for _ in range(10):
                src = int(rng.choice(n, p=f / f.sum()))
                j = int(rng.choice(n))
                if src == j:
                    continue
                delta = 0.005 * f[src]
                if delta < 1e-12:
                    continue
                f_try = f.copy()
                f_try[src] -= delta
                f_try[j] += delta
                f_try = f_try / f_try.sum()
                if fast_convolve(f_try).max() < cur_max - 1e-13:
                    f = f_try
                    improved = True
                    break

            if not improved:
                # Escalate: random perturbation
                if stall > 50:
                    scale = 0.002 + 0.003 * min(stall // 100, 5)
                    rng = np.random.default_rng(stall * 7 + 1)
                    f = project_simplex(f + rng.standard_normal(n) * scale)
                    stall = 0

    return best_f, best_obj


def nesterov_sg(f0, time_limit, lr=0.002, mom=0.92, top_k=15, temp0=0.001):
    """Nesterov-accelerated projected subgradient with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    y = f.copy()
    prev_x = f.copy()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        if top_k < len(conv_y):
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv_y))

        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Nesterov update
        x_new = project_simplex(y - cur_lr * sg)
        # Momentum: y_{t+1} = x_{t+1} + beta * (x_{t+1} - x_t)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * (1.0 + np.sqrt(1.0 + 4.0 * t * t)))
        # Actually use standard heavy ball: beta = mom (simpler and works well)
        y_new = x_new + mom * (x_new - prev_x)
        y_new = project_simplex(y_new)

        prev_x = x_new
        y = y_new
        f = x_new

        conv_f = fast_convolve(f)
        cur_max = conv_f.max()

        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            prev_x = f.copy()
            cur_lr = max(lr * 0.5, 0.0005)
            cur_temp = temp0 * 1.5
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, pair_count=50):
    """
    Coordinate descent that tries pairwise transfers between coordinates.
    More targeted than pure subgradient: directly optimizes the transfer amount.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        # Identify the top peak
        k_top = int(np.argmax(conv))

        # For the top peak, find the best single-coordinate move
        js = max(0, k_top - n + 1)
        je = min(n - 1, k_top)
        idx = np.arange(js, je + 1)
        contributions = f[idx] * f[k_top - idx]

        # Try moving mass from the top few contributors
        top_contribs = np.argsort(contributions)[::-1][:5]

        for ci in top_contribs:
            src = int(idx[ci])
            if f[src] < 1e-10:
                continue

            # Binary search for optimal delta to move from src
            # We want to find delta that minimizes max(conv) after transfer
            # Try a few target positions
            # Target: position that minimizes the resulting max
            # Approximate: choose target j that minimizes max over peaks of
            #   new_contrib = (f[j]+delta)*f[peak-j] + f[j]*f[peak-j] (roughly)

            # Just try a few random targets + the "least dangerous" ones
            candidates = set()
            # Add the position farthest from the peak
            dists = np.abs(np.arange(n) - src)
            candidates.add(int(np.argmax(dists)))
            # Add some random ones
            for _ in range(10):
                candidates.add(int(rng.integers(n)))
            candidates.discard(src)

            for j in list(candidates)[:8]:
                # Try to find best delta via a few trials
                for delta_frac in [0.01, 0.005, 0.002, 0.001]:
                    delta = delta_frac * f[src]
                    if delta < 1e-12 or delta > f[src]:
                        continue
                    f_try = f.copy()
                    f_try[src] -= delta
                    f_try[j] += delta
                    st = f_try.sum()
                    f_try = f_try / st
                    cv = fast_convolve(f_try)
                    if cv.max() < cur_max - 1e-13:
                        f = f_try
                        stall = 0
                        break
                else:
                    continue
                break
            else:
                continue
            break
        else:
            if stall > 30:
                # Random perturbation
                scale = 0.001 + 0.002 * min(stall // 60, 4)
                f = project_simplex(f + rng.standard_normal(n) * scale)
                stall = 0

    return best_f, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.22 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.3
    inits.append(c / c.sum())

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits.append((p1 + p2) / (p1 + p2).sum())

    # Smooth random (low frequency)
    smooth = np.ones(n) + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n + rng.uniform(0, 2 * np.pi))
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Concentrated center (narrower)
    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-FW, peak shaving,
    Nesterov subgradient, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(p.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: LP-FW on parent (main workhorse) ===
    print("=== Phase 1: LP-FW Optimization ===")

    if parent is not None:
        n_p = len(parent)
        t1 = min(500, deadline - time.monotonic() - 400)
        if t1 > 60:
            print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, _ = fw_lp_solve(parent, time_limit=t1)
            try_seq(a_opt, "parent_fw")

    # Try other promising lengths with different inits
    lengths = [384, 256, 512]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    for n in lengths[1:]:  # Skip parent length (already tried)
        if time.monotonic() > deadline - 300:
            break
        inits = generate_inits(n, rng, parent)
        for i in range(min(2, len(inits))):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(50, (deadline - time.monotonic() - 200) / 4)
            if t_alloc < 20:
                t_alloc = 20
            a_opt, _ = fw_lp_solve(inits[i], time_limit=t_alloc)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  Peak shave (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, top_k=8)
            try_seq(a_opt, "peak_shave")

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Nesterov SG (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = nesterov_sg(best_seq, time_limit=t_alloc,
                                    lr=0.002, mom=0.92,
                                    top_k=max(10, n_best // 6), temp0=0.0008)
            try_seq(a_opt, "nesterov")

    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: Coordinate Descent ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Coord descent (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

    # === Phase 5: Final fine-tuning cycle ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Quick FW
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            a_opt, _ = fw_lp_solve(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "final_fw")

        # Peak shave again
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = peak_shave(best_seq, time_limit=min(rem, 25), top_k=5)
            try_seq(a_opt, "final_shave")

        # Ultra-fine Nesterov
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = nesterov_sg(best_seq, time_limit=min(rem, 15),
                                    lr=0.0008, mom=0.95,
                                    top_k=max(5, n_best // 4), temp0=0.0003)
            try_seq(a_opt, "final_nest")

        # Tiny perturbation + quick polish
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            for attempt in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0008
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = peak_shave(f_pert, time_limit=min(rem, 3), top_k=4)
                try_seq(a_opt, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
