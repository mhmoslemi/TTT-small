"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated projected
subgradient with adaptive temperature annealing and multi-phase search strategy.
Key innovations: (1) Fourier-based gradient computation for O(n log n) per step,
(2) Nesterov acceleration for faster convergence, (3) adaptive temperature
scheduling for the smooth-max approximation, (4) multi-restart exploration
across different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """Compute cross[j] = sum_k p[k]*a[k-j] for j=0..n-1."""
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(np.conj(A) * P, fft_size)[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the true objective 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_grad_smooth(a, t_temp):
    """Compute gradient of smooth-max objective (normalized to sum(a)=1)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t_temp * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    p = exp_scaled / sum_exp
    cross = fast_cross_corr(p, a)
    grad = 4.0 * n * cross / t_temp
    return grad, conv


def nesterov_subgrad(a0, time_limit, t_temp_init=30.0, t_temp_end=2000.0,
                      c_init=1.0, max_iters=100000, momentum_decay=True,
                      evaluate_every=500):
    """
    Nesterov-accelerated projected subgradient for smooth-max objective.
    Works in a-space (simplex).
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max_iters)
        t_temp = t_temp_init * (t_temp_end / t_temp_init) ** progress
        
        # Nesterov extrapolation
        beta = (it - 1 + np.sqrt(1 + 4 * (it - 1)**2)) / (2 * it + 2) if it > 1 else 0.0
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum
        
        # Compute gradient at y
        grad_y, _ = compute_grad_smooth(y, t_temp)
        
        # Step size
        lr = c_init / np.sqrt(it + 1)
        
        # Update
        a_new = project_simplex(y - lr * grad_y)
        a_prev = a
        a = a_new
        
        # Evaluate every K steps
        if (it + 1) % evaluate_every == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                # Reduce step size
                c_init *= 0.5
                no_improve = 0
                if c_init < 0.01:
                    break
    
    # Final evaluation
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj


def lp_refine(a0, time_limit, max_iters=200):
    """LP-based refinement: alternate between fixing a and optimizing b."""
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.3
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        
        # Build constraint matrix: A[b][k] = sum_j a[k-j]*b[j] = conv(a,b)[k]
        # We want: max sum(b) s.t. A@b <= M, b >= 0
        # This is equivalent to: the "direction of steepest ascent" for sum(b)
        # under the constraint that conv(a, b) doesn't exceed M.
        
        # For efficiency, use the fact that A is a Toeplitz-like matrix.
        # But for n=512, the matrix is 1023x512, which is manageable.
        num_rows = 2 * n - 1
        A_mat = np.zeros((num_rows, n))
        for k in range(num_rows):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A_mat[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
        
        c = -np.ones(n)
        res = linprog(c, A_ub=A_mat, b_ub=np.full(num_rows, M),
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        if np.linalg.norm(b - a) < 1e-10:
            break
        
        a_new = (1 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            step = min(0.8, step * 1.1)
        else:
            step = max(0.05, step * 0.9)
        
        a = a_new
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy combining:
(1) Nesterov-accelerated projected subgradient with adaptive temperature annealing,
(2) LP-based iterative refinement with line search for precise convergence,
(3) Multi-restart exploration across hyperparameters and sequence lengths.
Key efficiency: O(n log n) per subgradient step via FFT; LP solves use HiGHS.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """Compute cross[j] = sum_k p[k]*a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(np.conj(A) * P, fft_size)[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_true_obj(a):
    """Compute 2*n*max(conv)/(sum(a))^2 for the given sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def solve_lp_direction(a, M):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b (unnormalized)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def nesterov_subgrad(a0, time_limit, t_temp_init=30.0, t_temp_end=2000.0,
                     c_init=1.0, max_iters=100000, eval_interval=200,
                     seed=0):
    """
    Nesterov-accelerated projected subgradient with annealed smooth-max.
    Optimizes in the probability simplex.
    """
    a = normalize_nonneg(a0)
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    t0 = time.monotonic()
    t_nesterov = 1.0
    no_improve = 0
    c_cur = c_init
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max(max_iters, 1))
        t_temp = t_temp_init * (t_temp_end / t_temp_init) ** progress
        
        # Nesterov momentum coefficient
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
        beta = (t_nesterov - 1.0) / t_new
        t_nesterov = t_new
        
        # Extrapolation
        y = a + beta * (a - a_prev)
        y = normalize_nonneg(y)
        
        # Compute gradient of smooth-max at y
        conv_y = fast_convolve(y)
        scaled = t_temp * conv_y
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        p = exp_scaled / sum_exp
        cross = fast_cross_corr(p, y)
        grad = 4.0 * n * cross / t_temp
        
        # Step size (Nesterov optimal rate)
        lr = c_cur / np.sqrt(it + 1)
        
        # Projected gradient step
        a_new = project_simplex(y - lr * grad)
        a_prev = a
        a = a_new
        
        # Periodic evaluation
        if (it + 1) % eval_interval == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 10:
                c_cur *= 0.5
                no_improve = 0
                if c_cur < 0.005:
                    break
            elif no_improve == 1:
                # Small improvement in aggressiveness
                c_cur = min(c_init, c_cur * 1.1)
    
    # Final check
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj


def subgrad_momentum(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=20,
                     temp_init=0.001, perturb_every=3000, perturb_scale=0.003,
                     seed=42):

    a = normalize_nonneg(a0)
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        # Compute weighted subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum
        
        # Compute subgradient efficiently using FFT
        # sg[j] = sum_{k in peaks} 2*w[k]*a[k-j]
        # This is 2 * cross_correlation(w_ext, a) where w_ext is zero-padded
        w_ext = np.zeros(len(conv))
        for i in range(len(peaks)):
            if w[i] > 1e-14:
                w_ext[int(peaks[i])] = 2.0 * w[i]
        sg = fast_cross_corr(w_ext, a)
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.99985
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.15, lr * 0.5)
    
    return best_a, best_obj

            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_trial:
                best_trial = obj_try
                best_t = t
        
        a = (1.0 - best_t) * a + best_t * b
        a = normalize_nonneg(a)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            # Rebuild constraint matrix (only needed for next LP)
            A_mat = build_conv_matrix(a)
        elif cur_obj >= best_obj + 0.01:
            # Stalled significantly, break
            break
    
    return best_a, best_obj


def lp_refine_fixed(a0, time_limit, step=0.3, max_iters=300):
    """LP-based refinement with fixed step size (faster, no line search)."""
    a = normalize_nonneg(a0)
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        b = solve_lp_direction(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        if np.linalg.norm(b - a) < 1e-10:
            break
        
        a = (1.0 - step) * a + step * b
        a = normalize_nonneg(a)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj


def smooth_max_gradient_descent(a0, time_limit, lr_init=0.01, momentum=0.9,
                                t_init=30.0, t_end=500.0, max_iters=5000):
    """
    Gradient descent on smooth-max objective with momentum,
    using softmax parameterization to automatically maintain simplex.
    """
    a = normalize_nonneg(a0)
    n = len(a)
    
    # Initialize x such that softmax(x) = a
    x = np.log(a + 1e-30)
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max(max_iters, 1))
        t_temp = t_init * (t_end / t_init) ** progress
        
        # Compute softmax
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a_cur = exp_x / exp_x.sum()
        
        # Compute gradient w.r.t. x
        conv = fast_convolve(a_cur)
        scaled = t_temp * conv
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        p = exp_scaled / sum_exp
        cross = fast_cross_corr(p, a_cur)
        grad_a = 4.0 * n * cross / t_temp
        
        # Convert to gradient w.r.t. x via softmax Jacobian
        dot = np.dot(a_cur, grad_a)
        grad_x = a_cur * (grad_a - dot)
        
        # Momentum update
        v = momentum * v - lr * grad_x
        x = x + v
        
        # Evaluate
        a_new = exp_x / exp_x.sum()  # Recompute from updated x
        x_shifted2 = x - x.max()
        exp_x2 = np.exp(x_shifted2)
        a_new = exp_x2 / exp_x2.sum()
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        lr *= 0.9999
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Fast subgradient exploration (multiple configs)
    Phase 2: LP refinement with line search (precise)
    Phase 3: Subgradient fine-tuning
    Phase 4: LP refinement (smaller steps)
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: len={len(p)}, norm_sum={p.sum():.4f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  *** New best: {score:.6f} (n={len(a)}) ***")
        return score
    
    # ============ Phase 1: Nesterov Subgradient (fast exploration) ============
    print("\n=== Phase 1: Nesterov Subgradient ===")
    t1_start = time.monotonic()
    
    start_a = parent if parent is not None else np.ones(256) / 256
    p1_deadline = deadline - 800
    
    configs = [
        {'t_temp_init': 20.0, 't_temp_end': 500.0, 'c_init': 0.8, 'max_iters': 80000, 'eval_interval': 100},
        {'t_temp_init': 50.0, 't_temp_end': 2000.0, 'c_init': 1.2, 'max_iters': 80000, 'eval_interval': 150},
        {'t_temp_init': 100.0, 't_temp_end': 5000.0, 'c_init': 0.5, 'max_iters': 60000, 'eval_interval': 200},
    ]
    
    n_configs = len(configs) + 2  # extra for perturbed starts
    per_config = max(15.0, (p1_deadline - time.monotonic()) / n_configs)
    
    for cfg in configs:
        remaining = p1_deadline - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(per_config, remaining)
        print(f"  Running nesterov (t_init={cfg['t_temp_init']}, c={cfg['c_init']}), alloc={t_alloc:.0f}s")
        a_opt, _ = nesterov_subgrad(start_a, time_limit=t_alloc, **cfg)
        try_update(a_opt)
    
    # Perturbed starts
    for scale in [0.008, 0.015]:
        remaining = p1_deadline - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(40.0, remaining)
        a_pert = project_simplex(start_a + rng.standard_normal(len(start_a)) * scale)
        a_opt, _ = nesterov_subgrad(a_pert, time_limit=t_alloc,
                                     t_temp_init=50.0, t_temp_end=2000.0,
                                     c_init=0.8, max_iters=50000, eval_interval=150)
        try_update(a_opt)
    
    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")
    
    # ============ Phase 2: LP Refinement with Line Search ============
    print("\n=== Phase 2: LP Refinement (Line Search) ===")
    t2_start = time.monotonic()
    
    if best_sequence is not None:
        p2_deadline = deadline - 400
        base = best_sequence.copy()
        
        # Try line search version first (more precise)
        remaining = p2_deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(200.0, remaining)
            print(f"  LP line search, alloc={t_alloc:.0f}s")
            a_opt, _ = lp_refine_linesearch(base, time_limit=t_alloc, max_iters=200)
            try_update(a_opt)
            if best_sequence is not None:
                base = best_sequence.copy()
        
        # Fixed step LP (faster iterations)
        for step in [0.3, 0.5, 0.2]:
            remaining = p2_deadline - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(100.0, remaining)
            print(f"  LP fixed step={step}, alloc={t_alloc:.0f}s")
            a_opt, _ = lp_refine_fixed(base, time_limit=t_alloc, step=step, max_iters=300)
            try_update(a_opt)
            if best_sequence is not None:
                base = best_sequence.copy()
    
    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")
    
    # ============ Phase 3: Subgradient Fine-tuning ============
    print("\n=== Phase 3: Subgradient Fine-tuning ===")
    t3_start = time.monotonic()
    
    if best_sequence is not None:
        p3_deadline = deadline - 200
        n_best = len(best_sequence)
        
        sg_configs = [
            {'lr_init': 0.01, 'momentum': 0.92, 'top_k': 10, 'temp_init': 0.0005},
            {'lr_init': 0.005, 'momentum': 0.95, 'top_k': 5, 'temp_init': 0.0002},
            {'lr_init': 0.015, 'momentum': 0.90, 'top_k': 20, 'temp_init': 0.001},
        ]
        
        per_sg = max(20.0, (p3_deadline - time.monotonic()) / len(sg_configs))
        for cfg in sg_configs:
            remaining = p3_deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(per_sg, remaining)
            print(f"  Subgrad lr={cfg['lr_init']}, mom={cfg['momentum']}, topk={cfg['top_k']}, alloc={t_alloc:.0f}s")
            a_opt, _ = subgrad_momentum(best_sequence, time_limit=t_alloc,
                                         perturb_every=2000, perturb_scale=0.002,
                                         **cfg)
            try_update(a_opt)
    
    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")
    
    # ============ Phase 4: More LP Refinement ============
    print("\n=== Phase 4: LP Refinement Round 2 ===")
    t4_start = time.monotonic()
    
    if best_sequence is not None:
        p4_deadline = deadline - 100
        base = best_sequence.copy()
        
        for step in [0.2, 0.35, 0.45]:
            remaining = p4_deadline - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(80.0, remaining)
            print(f"  LP step={step}, alloc={t_alloc:.0f}s")
            a_opt, _ = lp_refine_fixed(base, time_limit=t_alloc, step=step, max_iters=250)
            try_update(a_opt)
            if best_sequence is not None:
                base = best_sequence.copy()
        
        # One more line search
        remaining = p4_deadline - time.monotonic()
        if remaining > 30 and best_sequence is not None:
            t_alloc = min(100.0, remaining)
            a_opt, _ = lp_refine_linesearch(best_sequence, time_limit=t_alloc, max_iters=150)
            try_update(a_opt)
    
    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")
    
    # ============ Phase 5: Final Polish ============
    print("\n=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = remaining
            print(f"  Final subgrad polish, alloc={t_alloc:.0f}s")
            a_opt, _ = subgrad_momentum(best_sequence, time_limit=t_alloc,
                                         lr_init=0.003, momentum=0.95, top_k=3,
                                         temp_init=0.0001, perturb_every=5000,
                                         perturb_scale=0.001, seed=999)
            try_update(a_opt)
        
        # One very small LP step
        remaining = deadline - time.monotonic()
        if remaining > 10 and best_sequence is not None:
            t_alloc = min(30.0, remaining)
            a_opt, _ = lp_refine_fixed(best_sequence, time_limit=t_alloc, step=0.15, max_iters=50)
            try_update(a_opt)
    
    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    print(f"{'='*50}")
    return [float(x) for x in best_sequence]
