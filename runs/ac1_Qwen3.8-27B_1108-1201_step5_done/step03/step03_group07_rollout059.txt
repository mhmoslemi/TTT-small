def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.max(rho[cond])
    theta = (cssv[rho_idx - 1] - 1.0) / rho_idx
    return np.maximum(v - theta, 0.0)

    n = len(a0)

    s = a.sum()
    if s < 1e-15:

    a = a / s

    best_max = fast_convolve(best).max()

    lr = lr_init
    eps = eps_init

        # Adaptive epsilon: start large, shrink over time

        threshold = max_b - eps * max_b
        S = np.where(b >= threshold)[0]

        # Subgradient

                        best_max = bn.max()
                        best = a_new.copy()
                        a = a_new
                        break

        # Also try: spread mass from peak contributors to flatten
        if a is not best:
            continue
        # Random perturbation + check
        perturb_idx = rng_local.integers(n)
        for target_idx in rng_local.integers(n, size=5):
            if target_idx == perturb_idx:
                continue
            for frac in [0.02, 0.05]:
                delta = a[perturb_idx] * frac
                a_new = a.copy()
                a_new[perturb_idx] -= delta
                a_new[target_idx] += delta
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn < 1e-15:
                    continue
                a_new = a_new / sn
                bn = fast_convolve(a_new)
                if bn.max() < best_max - 1e-12:
                    best_max = bn.max()
                    best = a_new.copy()
                    a = a_new
                    break

    return best


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum (random phases)."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_max = fast_convolve(best_a).max()

    for trial in range(30):
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A = np.zeros(fft_size, dtype=complex)
        A[0] = n  # DC component
        for k in range(1, fft_size // 2 + 1):
            A[k] = np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * phases[k])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        b = fast_convolve(a)
        if b.max() < best_max:
            best_max = b.max()
            best_a = a.copy()

    return best_a


def optimal_rising_init(n, rng):
    """Initialize with a rising-exponential shape (asymmetric to reduce center peak)."""
    best_a = np.ones(n) / n
    best_max = fast_convolve(best_a).max()

    for rate in np.concatenate([np.linspace(0.1, 3.0, 20), np.linspace(0.01, 0.1, 5)]):
        x = np.arange(n, dtype=np.float64)
        a = np.exp(rate * (x / n - 0.5))
        # Also try reversed
        for sign in [1, -1]:
            a_try = np.exp(sign * rate * (x / n - 0.5))
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            b = fast_convolve(a_try)
            if b.max() < best_max:
                best_max = b.max()
                best_a = a_try.copy()

    # Add random perturbation
    for _ in range(10):
        noise = rng.standard_normal(n) * 0.1
        a_try = best_a + noise
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        b = fast_convolve(a_try)
        if b.max() < best_max:
            best_max = b.max()
            best_a = a_try.copy()

    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Rising/exponential
    exp_a = np.exp(0.5 * x / n)
    inits['rising'] = exp_a / exp_a.sum()
    exp_b = np.exp(-0.5 * x / n)
    inits['falling'] = exp_b / exp_b.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng)

    # Optimal rising
    inits['opt_rising'] = optimal_rising_init(n, rng)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 0.5
    inits['bimodal'] = bi / bi.sum()

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits


def smooth_max_refine(a, beta, max_iter=200):
    """Final polish using smooth max (log-sum-exp) with L-BFGS on simplex."""
    from scipy.optimize import minimize
    n = len(a)
    a0 = np.maximum(a, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)

    def obj(x):
        a_t = np.exp(x - x.max())
        a_t = a_t / a_t.sum()
        b = fast_convolve(a_t)
        bm = b.max()
        shifted = beta * (b - bm)
        shifted = np.clip(shifted, -50, 50)
        sm = bm + np.log(np.exp(shifted).sum()) / beta
        return 2.0 * n * sm

    def jac(x):
        a_t = np.exp(x - x.max())
        a_t = a_t / a_t.sum()
        b = fast_convolve(a_t)
        bm = b.max()
        shifted = beta * (b - bm)
        shifted = np.clip(shifted, -50, 50)
        w = np.exp(shifted)
        w = w / w.sum()
        # grad of smooth max w.r.t. a_t
        grad_a = np.zeros(n)
        for k, wk in enumerate(w):
            if wk < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            grad_a[js] += wk * a_t[k - js]
        grad_a *= 2.0
        # Chain rule through softmax
        grad_x = a_t * (grad_a - np.dot(a_t, grad_a))
        return 2.0 * n * grad_x

    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxls': 30})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def propose_candidate():
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-start scan over lengths and initializations
    Phase 2: Deep subgradient optimization of best candidate
    Phase 3: Coordinate swap refinement + peak breaking
    Phase 4: Smooth max polish + exploration
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + 1770.0  # Leave 30s buffer

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d" % len(p))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return float('inf')
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" %
                  (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # === Phase 1: Multi-start scan ===
    print("=== Phase 1: Multi-start scan ===")
    lengths = [192, 256, 320, 384, 512]
    phase1_deadline = t_start + 300.0

    phase1_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        # Priority: parent, flat_spec, opt_rising, rising, random
        priority = ['parent', 'flat_spec', 'opt_rising', 'rising', 'random',
                    'gauss', 'bimodal', 'uniform', 'falling']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            t_alloc = min(55.0, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a0 = inits[name].copy()
            a_opt, obj_opt = subgradient_optimize(a0, max_time=t_alloc,
                                                  lr_init=0.04, eps_init=0.1,
                                                  eps_min=1e-8)
            score = score_and_update(a_opt)
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))

    phase1_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % best_score)

    # === Phase 2: Deep optimization ===
    print("=== Phase 2: Deep subgradient optimization ===")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        score_and_update(best_sequence)

    best_n = len(best_sequence)
    phase2_deadline = t_start + 900.0

    # Run extended optimization with multiple restarts
    a_cur = best_sequence / best_sequence.sum()
    remaining = phase2_deadline - time.monotonic()

    n_restarts = max(3, int(remaining / 150))
    restart_time = max(30.0, remaining / n_restarts)

    for rst in range(n_restarts):
        if time.monotonic() > phase2_deadline - 30:
            break

        # On restarts after the first, perturb the sequence
        if rst > 0:
            scale = 0.01 + 0.02 * rst / n_restarts
            noise = rng.standard_normal(best_n) * scale
            a_cur = project_simplex(a_cur + noise)

        t_alloc = min(restart_time, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        # Decreasing learning rate schedule across restarts
        lr_use = 0.04 * (0.8 ** rst)
        eps_use = 0.05 * (0.7 ** rst)

        a_opt, obj_opt = subgradient_optimize(a_cur, max_time=t_alloc,
                                              lr_init=lr_use, eps_init=max(0.005, eps_use),
                                              eps_min=1e-10)
        score = score_and_update(a_opt)
        a_cur = a_opt

    print("Phase 2 done. Best: %.6f" % best_score)

    # === Phase 3: Peak breaking + coordinate refinement ===
    print("=== Phase 3: Peak breaking & coordinate refinement ===")
    if best_sequence is not None:
        phase3_deadline = t_start + 1350.0
        a_cur = best_sequence / best_sequence.sum()
        n_cur = len(a_cur)

        # Peak breaking + re-optimize cycle
        cycle_deadline = phase3_deadline - 60
        for cycle in range(20):
            if time.monotonic() > cycle_deadline:
                break
            t_alloc = min(30.0, cycle_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a_broken = peak_breaking_step(a_cur, n_breaks=5)
            a_opt, _ = subgradient_optimize(a_broken, max_time=t_alloc,
                                            lr_init=0.02, eps_init=0.03,
                                            eps_min=1e-10)
            score = score_and_update(a_opt)
            if score < best_score:
                a_cur = a_opt

        # Coordinate swap refinement
        if time.monotonic() < phase3_deadline - 10:
            t_alloc = min(60.0, phase3_deadline - time.monotonic() - 5)
            a_final = coordinate_swap_refine(a_cur, n_rounds=int(t_alloc * 5))
            score = score_and_update(a_final)
            if score < best_score:
                a_cur = a_final

    print("Phase 3 done. Best: %.6f" % best_score)

    # === Phase 4: Smooth max polish + length exploration ===
    print("=== Phase 4: Final polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20

        # Smooth max polish (L-BFGS-B with very high beta for near-exact max)
        if remaining > 30:
            t_alloc = min(60.0, remaining / 3)
            a_opt = smooth_max_refine(best_sequence / best_sequence.sum(),
                                      beta=200.0, max_iter=int(t_alloc * 10))
            score = score_and_update(a_opt)
            if score < best_score:
                best_sequence = a_opt
            remaining = deadline - time.monotonic() - 20

        # Try another length
        if remaining > 60 and best_sequence is not None:
            alt_n = 384 if len(best_sequence) == 256 else 256
            inits_alt = generate_initializations(alt_n, best_sequence, rng)
            a0_alt = inits_alt.get('parent', inits_alt.get('flat_spec',
                                inits_alt['uniform']))
            t_alloc = min(100.0, remaining / 2)
            if t_alloc > 20:
                a_opt, _ = subgradient_optimize(a0_alt, max_time=t_alloc,
                                                lr_init=0.03, eps_init=0.05,
                                                eps_min=1e-9)
                score = score_and_update(a_opt)

        # Final very-fine subgradient on the best
        remaining = deadline - time.monotonic() - 10
        if remaining > 20 and best_sequence is not None:
            t_alloc = remaining
            a_opt, _ = subgradient_optimize(best_sequence / best_sequence.sum(),
                                            max_time=t_alloc,
                                            lr_init=0.005, eps_init=0.005,
                                            eps_min=1e-12)
            score = score_and_update(a_opt)
            if score < best_score:
                best_sequence = a_opt

        # One more smooth max pass
        remaining = deadline - time.monotonic() - 5
        if remaining > 15 and best_sequence is not None:
            a_opt = smooth_max_refine(best_sequence / best_sequence.sum(),
                                      beta=500.0, max_iter=int(remaining * 5))
            score = score_and_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        score_and_update(best_sequence)

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d, time=%.0fs)" %
          (best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
