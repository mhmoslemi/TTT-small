    """Compute soft objective and gradient."""
    a = np.maximum(x, 0)  # ensure non-negative (L-BFGS-B handles bounds, but safety)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    b = fast_convolve(a)  # length 2n-1
    scaled = b / T
    scaled -= scaled.max()  # for numerical stability
    log_sum_exp = scaled.max() + np.log(np.sum(np.exp(scaled)))  # = max(b/T) + log(sum(exp((b-max(b))/T)))
    # Actually, logsumexp(b/T) = max(b)/T + log(sum(exp((b - max(b))/T)))
    
    # Gradient of logsumexp
    w = np.exp(scaled)  # softmax weights (unnormalized by total sum, but that's fine)
    w_sum = w.sum()
    w = w / w_sum  # normalized softmax
    
    cross = np.correlate(w, a, mode='valid')[:n]  # length n
    
    # ∂L/∂a_i = cross[i] / T

    obj = 2.0 * n * L / (S * S)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Soft L-BFGS with temperature annealing for fast initial optimization
2. LP-based iterative direction finding with sparse constraint matrices
3. Nesterov-accelerated subgradient descent for fine-tuning
4. Coordinate descent polishing

import numpy as np
import time
from scipy.optimize import linprog, minimize

# Module-level state for soft objective
_soft_T = 0.01
_soft_n = 512


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def compute_objective(a):
    """Compute the objective value 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)


def _soft_obj_grad(x):
    """Compute soft objective and gradient."""
    n = _soft_n
    T = _soft_T
    a = x
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)

    w = exp_scaled / exp_sum
    cross = np.correlate(w, a, mode='valid')[:n]
    obj = 2.0 * n * L / (S * S)
    grad = 2.0 * n * (cross / T - 2.0 * L * a / S) / (S * S)
    return obj, grad


def _soft_obj_func(x):

def _soft_grad_func(x):

            max_evals -= 1

        obj_try = compute_objective(a_norm)
        max_evals -= 1

            best_obj = obj_try
            best_t = t

        for _ in range(4):

                t_try = best_t + dt
                if t_try <= 0 or t_try >= 1 or max_evals <= 0:
                    continue
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.93, top_k=30, temp=0.001):
    """Subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        
        # Find peak positions
        if top_k >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -top_k)[-top_k:]
        
        # Softmax weighting
        peak_vals = b[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]
        
        # Nesterov step
        v_new = momentum * v - lr * sg
        a_new = a - v  # lookahead
        a_new = a_new + v_new
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        v = v_new
        
        cur_b = fast_convolve(a)
        cur_obj = 2.0 * n * cur_b.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        temp *= 1.003
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_sweeps=100):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initializations + soft L-BFGS with temperature annealing
2. LP-based iterative direction finding with multi-M exploration
3. Nesterov-accelerated subgradient descent for fine-tuning
4. Final LP polish
Uses FFT-based convolution, sparse insights, and adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

# Module-level state for soft objective (avoid closures/lambdas)
_soft_T = 0.01
_soft_n = 512


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def _soft_obj_grad(x):
    """Compute soft (log-sum-exp) objective and gradient."""
    n = _soft_n
    T = _soft_T
    a = x
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = b_max / T + np.log(exp_sum)
    w = exp_scaled / exp_sum
    cross = np.correlate(w, a, mode='valid')[:n]
    obj = 2.0 * n * L / (S * S)
    grad = 2.0 * n * (cross / T - 2.0 * L * a / S) / (S * S)
    return obj, grad


def _soft_obj_func(x):
    return _soft_obj_grad(x)[0]


def _soft_grad_func(x):
    return _soft_obj_grad(x)[1]


def soft_lbfgs_optimize(a0, time_limit):
    """L-BFGS on soft objective with temperature annealing."""
    global _soft_T, _soft_n
    n = len(a0)
    _soft_n = n

    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n

    b = fast_convolve(a)
    M = b.max()

    T_init = 0.03 * M
    T_final = 0.00005 * M

    t0 = time.monotonic()
    T = T_init
    bounds = [(0.0, 1000.0)] * n

    while time.monotonic() - t0 < time_limit - 3:
        _soft_T = T

        result = minimize(
            _soft_obj_func,
            a,
            jac=_soft_grad_func,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-10}
        )

        if result.fun < 1e9:
            a = np.maximum(result.x, 0.0)

        T = max(T_final, T * 0.55)

        if T <= T_final * 1.05:
            _soft_T = T_final
            result = minimize(
                _soft_obj_func,
                a,
                jac=_soft_grad_func,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            if result.fun < 1e9:
                a = np.maximum(result.x, 0.0)
            break

    S = a.sum()
    if S > 1e-15:
        a = a / S
    return a


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x, result.x.sum()
    return None, 0.0


def line_search(a, d, current_obj, n, max_evals=12):
    """Geometric line search with refinement."""
    best_t = 0.0
    best_obj = current_obj

    candidates = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

    for t in candidates:
        if max_evals <= 0:
            break
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            max_evals -= 1
            continue
        a_norm = a_try / s
        obj_try = compute_objective(a_norm)
        max_evals -= 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    # Refine around best
    if 0.03 < best_t < 0.92 and max_evals > 2:
        step = 0.08
        for _ in range(4):
            improved = False
            for dt in [-step, step]:
                t_try = best_t + dt
                if t_try <= 0.001 or t_try >= 0.99 or max_evals <= 0:
                    continue
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_norm = a_try / s
                obj_try = compute_objective(a_norm)
                max_evals -= 1
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try
                    improved = True
            if not improved:
                step *= 0.4

    return best_t, best_obj


def lp_optimize(a0, time_limit):
    """LP-based iterative optimization with multi-M exploration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    M = b.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Try multiple M values to find best direction
        M_factors = [1.001, 1.0005, 0.9995]
        best_dir = None
        best_g_sum = 0.0

        for mf in M_factors:
            M_try = M * mf
            g, g_sum = solve_lp_direction(a, M_try)
            if g is not None and g_sum > 1.0 + 1e-7:
                if g_sum > best_g_sum:
                    best_g_sum = g_sum
                    best_dir = g

        if best_dir is None:
            no_improve += 1
        else:
            d = best_dir / best_g_sum - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)

                if obj_best < best_obj - 1e-13:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()

                    b = fast_convolve(a)
                    M = b.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        iteration += 1

        # Handle stagnation with adaptive perturbation
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + 7 + perturb_count * 31)
            scale = 0.002 + 0.004 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-15:
                a = best_a.copy()
            else:
                a = a_pert / s_pert
            b = fast_convolve(a)
            M = b.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 60 + 30 * perturb_count)

            if perturb_count > 40:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.008, momentum=0.94, top_k=40, temp=0.0005):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)

        if top_k >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -top_k)[-top_k:]

        peak_vals = b[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]

        # Nesterov momentum
        v_prev = v.copy()
        v = momentum * v - lr * sg
        a_new = a - v_prev  # extrapolation step
        a_new = a_new + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        cur_b = fast_convolve(a)
        cur_obj = 2.0 * n * cur_b.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.002

        if no_improve >= 800:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits['cosine'] = cos_w / cos_w.sum()

    # Sine window
    sin_w = np.sin(np.pi * (x + 1) / (n + 1)) + 0.01
    inits['sine'] = sin_w / sin_w.sum()

    # Raised cosine (Hann-like)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.01
    inits['hann'] = hann / hann.sum()

    # Bimodal (edges emphasized)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi += 0.05
    inits['bimodal'] = bi / bi.sum()

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Double cosine
    dc = 1.0 + 0.5 * np.cos(4 * np.pi * x / n)
    dc = np.maximum(dc, 0.01)
    inits['double_cos'] = dc / dc.sum()

    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

            print(f"  Deep LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Subgradient polishing
    print("=== Phase 3: Subgradient Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 20:
            sg_configs = [
                {'lr_init': 0.006, 'momentum': 0.94, 'top_k': 50, 'temp': 0.0003},
                {'lr_init': 0.01, 'momentum': 0.92, 'top_k': 30, 'temp': 0.0008},
                {'lr_init': 0.004, 'momentum': 0.96, 'top_k': 80, 'temp': 0.0002},
            ]

            t_per_sg = remaining / (len(sg_configs) + 2)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 50:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 30)
                if t_alloc < 8:
                    break

                a_opt, _ = subgradient_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final LP polish + perturbations
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        # Direct LP polish
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 15:
            t_alloc = min(remaining * 0.4, 120.0)
            if t_alloc > 10:
                a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f}")

        # Perturbation rounds
        remaining = deadline - time.monotonic() - 5.0
        n_perturb = max(1, int(remaining / 25))
        t_per_perturb = remaining / n_perturb

        for pi in range(n_perturb):
            if time.monotonic() > deadline - 5:
                break
            if t_per_perturb < 8:
                break

            scale = 0.002 * (pi + 1)
            noise = rng.standard_normal(len(best_sequence)) * scale
            a_pert = best_sequence + noise
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert / s_pert

            t_alloc = min(t_per_perturb, deadline - time.monotonic() - 3)
            if t_alloc < 5:
                break

            a_opt, _ = lp_optimize(a_pert, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
